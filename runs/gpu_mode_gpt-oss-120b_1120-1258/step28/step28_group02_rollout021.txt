@triton.jit
def apply_norm_gate_kernel(
    out_ptr, gate_ptr, mu_ptr, rstd_ptr, ln_w_ptr, ln_b_ptr, norm_out_ptr,
    total_rows, H, N,
    BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + j   # start offset of hidden dim 0

    # Load per row stats
    mu = tl.load(mu_ptr + offs_m, mask=mask_m, other=0.0)
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        ptr_out = out_ptr + base[:, None] + offs_h[None, :] * hidden_stride
        ptr_gate = gate_ptr + base[:, None] + offs_h[None, :] * hidden_stride

        out_vals = tl.load(ptr_out, mask=mask_m[:, None] & mask_h[None, :], other=0.0).to(tl.float32)  # (M, K)
        gate_vals = tl.load(ptr_gate, mask=mask_m[:, None] & mask_h[None, :], other=0.0).to(tl.float32)

        # LN weight and bias
        gamma = tl.load(ln_w_ptr + offs_h, mask=mask_h, other=0.0)
        beta = tl.load(ln_b_ptr + offs_h, mask=mask_h, other=0.0)

        # Compute normalized value
        diff = (out_vals - mu[:, None]) * rstd[:, None]      # (M,K)
        norm = diff * gamma[None, :] + beta[None, :]          # (M,K)
        norm_gate = norm * gate_vals                         # (M,K)

        # Store to norm_out
        dst_ptr = norm_out_ptr + base[:, None] + offs_h[None, :] * hidden_stride
        tl.store(dst_ptr, norm_gate.to(tl.float16), mask=mask_m[:, None] & mask_h[None, :])
```

def custom_kernel(data):
    x, mask, weights, cfg = data
    # config
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device
    B, N, _, _ = x.shape
    R = B * N * N
    H = hidden_dim
    # Step 1: LayerNorm
    x_flat = x.reshape(R, dim).contiguous()
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)
    BLOCK_M = 128; BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        weights["norm.weight"],
        weights["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    # Step 2: fused projection
    fused_w = torch.cat([
        weights["left_proj.weight"],
        weights["right_proj.weight"],
        weights["left_gate.weight"],
        weights["right_gate.weight"],
        weights["out_gate.weight"],
    ], dim=0).to(dtype=torch.float16, device=device)
    proj = torch.nn.functional.linear(x_norm, fused_w)  # shape (R,5*H) half
    # release some memory
    del x_norm, fused_w

    # Step 3: split and gate
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if not nomask and mask is not None:
        mask_flat = mask.to(dtype=torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # Step 4: batched GEMM
    left_mat = left.view(B * H, N, N)  # shape [B*H, N, N] half
    right_mat = right.view(B * H, N, N)  # half
    # compute left * right^T
    # right_mat transposed along last two dims: right_mat.transpose(1,2)
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))
    out = out.view(B, H, N, N)  # half

    # Free left/right maybe no longer needed
    del left, right

    # Step 5: rowwise stats
    mu = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 256
    BLOCK_K_STATS = 32  # may be <= H
    grid_stats = (triton.cdiv(R, BLOCK_M_STATS),)
    rowwise_stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=4,
    )

    # Step 6: apply second LN and gating
    normed = torch.empty_like(out)  # half

    BLOCK_M_NG = 128
    BLOCK_H_NG = 64
    grid_ng = (triton.cdiv(R, BLOCK_M_NG),)
    apply_norm_gate_kernel[grid_ng](
        out,
        out_gate,
        mu,
        rstd,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        normed,
        R,
        H,
        N,
        BLOCK_M=BLOCK_M_NG,
        BLOCK_H=BLOCK_H_NG,
        num_warps=8,
    )

    del out, out_gate

    # Step 7: final linear projection

    normed_perm = normed.permute(0, 2, 3, 1).contiguous()  # [B,N,N,H] half

    final = final_fp16.to(dtype=torch.float32).view(B, N, N, dim)  # float32

@triton.jit
def apply_norm_gate_kernel(
    out_ptr, gate_ptr,
    mu_ptr, rstd_ptr,
    ln_w_ptr, ln_b_ptr,
    norm_out_ptr,
    total_rows, H, N,
    BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + j  # start offset for hidden plane 0

    mu = tl.load(mu_ptr + offs_m, mask=mask_m, other=0.0)   # [BLOCK_M] fp32
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)  # [BLOCK_M] fp32

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # Load out and gate
        ptr_out = out_ptr + base[:, None] + offs_h[None, :] * hidden_stride
        ptr_gate = gate_ptr + base[:, None] + offs_h[None, :] * hidden_stride

        out_vals = tl.load(ptr_out,
                           mask=mask_m[:, None] & mask_h[None, :],
                           other=0.0).to(tl.float32)   # [M,K]
        gate_vals = tl.load(ptr_gate,
                            mask=mask_m[:, None] & mask_h[None, :],
                            other=0.0).to(tl.float32)   # [M,K]

        # Load LN params
        gamma = tl.load(ln_w_ptr + offs_h, mask=mask_h, other=0.0)   # [K] fp32
        beta = tl.load(ln_b_ptr + offs_h, mask=mask_h, other=0.0)    # [K] fp32

        # Compute normalized and gated value
        diff = (out_vals - mu[:, None]) * rstd[:, None]   # [M,K]
        norm = diff * gamma[None, :] + beta[None, :]       # [M,K]
        norm_gate = norm * gate_vals                       # [M,K]

        # Store
        dst_ptr = norm_out_ptr + base[:, None] + offs_h[None, :] * hidden_stride
        tl.store(dst_ptr,
                 norm_gate.to(tl.float16),
                 mask=mask_m[:, None] & mask_h[None, :])
```

"""
Outgoing TriMul (AlphaFold‑3) – forward‑only fused implementation.

Algorithm
---------
1️⃣ Row‑wise LayerNorm over the last channel (dim) of the flattened pair tensor.
2️⃣ Fuse the five linear maps (left, right, left_gate, right_gate, out_gate) into a
   single GEMM:  (R, dim) @ (5·H, dim).T  → (R, 5·H)    (R = B·N·N).
3️⃣ Split the fused projection, apply the optional pair mask, compute sigmoid gates,
   and write three tensors `left`, `right`, `out_gate` of shape [B, H, N, N] (FP16).
4️⃣ Batched GEMM (cuBLAS) computes the core einsum:
       out[b, h, i, j] = Σₖ left[b, h, i, k]·right[b, h, j, k]   → [B, H, N, N].
5️⃣ Row‑wise mean & rsqrt‑variance of `out` over the hidden dimension H  → µ, rstd.
6️⃣ Apply the second LayerNorm, multiply by `out_gate`, and finally project to the
   original channel dimension with a fused GEMM (cuBLAS).  The final tensor has
   shape [B, N, N, dim] (FP32).

All kernels are written for Triton 3.3.1 and run on an NVIDIA H100.
Only the forward pass is required – no gradients are computed.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,               # [R, dim] FP32 input
    y_ptr,               # [R, dim] FP16 output
    w_ptr,               # [dim]     γ (FP32)
    b_ptr,               # [dim]     β (FP32)
    rows,                # scalar R
    dim,                 # scalar dim
    eps,                 # scalar eps
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)               # [BLOCK_M]
    mask_m = offs_m < rows

    # ------- first pass: sum & sum‑of‑squares -------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------- second pass: write normalized values -------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, apply mask, sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,      # [R, 5·H] FP16 fused projection
    left_ptr,      # [B, H, N, N] FP16
    right_ptr,     # [B, H, N, N] FP16
    out_gate_ptr,  # [B, H, N, N] FP16
    mask_ptr,      # [R] FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)               # [BLOCK_R]
    mask_r = offs_r < R

    # Decode flat index: r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # Optional mask (broadcast over H)
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        m_val = m_val[:, None]                                       # [BLOCK_R, 1]
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + k                     # start of hidden plane 0

    # ------------------------------------------------------------------
    # iterate over hidden dimension
    # ------------------------------------------------------------------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)          # [BLOCK_H]
        mask_h = offs_h < H

        # Layout inside proj: [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(proj_ptr + proj_base,
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0)
        right_raw = tl.load(proj_ptr + proj_base + H,
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0)
        left_gate_raw = tl.load(proj_ptr + proj_base + 2 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_gate_raw = tl.load(proj_ptr + proj_base + 3 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        out_gate_raw = tl.load(proj_ptr + proj_base + 4 * H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)

        # sigmoid in FP32 for numerical stability
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * m_val * left_gate
        right = right_raw * m_val * right_gate
        outg  = out_gate

        dst = base[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr     + dst, left,  mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr    + dst, right, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst, outg,  mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 3️⃣ Row‑wise mean & rsqrt‑variance of the contracted tensor
# ----------------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr,   # [B, H, N, N] FP16
    mu_ptr,    # [R] FP32
    rstd_ptr,  # [R] FP32
    total_rows, H, N,
    eps,
    BLOCK_M: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)             # [BLOCK_M]
    mask_m = offs_m < total_rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + j             # start of hidden plane 0

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)          # [BLOCK_K]
        mask_k = offs_k < H

        ptr = out_ptr + base[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_m, mean,  mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)


# ----------------------------------------------------------------------
# 4️⃣ Apply second LayerNorm, the out_gate, and prepare for final projection
# ----------------------------------------------------------------------
@triton.jit
def apply_norm_gate_kernel(
    out_ptr,       # [B, H, N, N] FP16      (post‑einsum)
    gate_ptr,      # [B, H, N, N] FP16      (out_gate)
    mu_ptr,        # [R] FP32
    rstd_ptr,      # [R] FP32
    ln_w_ptr,      # [H] FP32  (to_out_norm.weight)
    ln_b_ptr,      # [H] FP32  (to_out_norm.bias)
    out_norm_ptr,  # [B, H, N, N] FP16  (LN‑gate result)
    total_rows, H, N,
    BLOCK_M: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)              # [BLOCK_M]
    mask_m = offs_m < total_rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + j              # start of hidden plane 0

    mu   = tl.load(mu_ptr   + offs_m, mask=mask_m, other=0.0)   # [BLOCK_M] FP32
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)   # [BLOCK_M] FP32

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)    # [BLOCK_H]
        mask_h = offs_h < H

        # Load raw tensors
        ptr_out  = out_ptr + base[:, None] + offs_h[None, :] * hidden_stride
        ptr_gate = gate_ptr + base[:, None] + offs_h[None, :] * hidden_stride

        out_vals  = tl.load(ptr_out,
                            mask=mask_m[:, None] & mask_h[None, :],
                            other=0.0).to(tl.float32)          # [M, K]
        gate_vals = tl.load(ptr_gate,
                            mask=mask_m[:, None] & mask_h[None, :],
                            other=0.0).to(tl.float32)          # [M, K]

        # LayerNorm parameters
        gamma = tl.load(ln_w_ptr + offs_h, mask=mask_h, other=0.0)   # [K] FP32
        beta  = tl.load(ln_b_ptr + offs_h, mask=mask_h, other=0.0)   # [K] FP32

        # (out - µ) * rstd → γ· + β → * out_gate
        diff = (out_vals - mu[:, None]) * rstd[:, None]              # [M, K]
        norm = diff * gamma[None, :] + beta[None, :]                 # [M, K]
        norm_gate = norm * gate_vals                                 # [M, K]

        dst_ptr = out_norm_ptr + base[:, None] + offs_h[None, :] * hidden_stride
        tl.store(dst_ptr,
                 norm_gate.to(tl.float16),
                 mask=mask_m[:, None] & mask_h[None, :])

def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul operator (no gradients).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
              (all float32)
            - config_dict must contain
                "dim" (int), "hidden_dim" (int), "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack inputs
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = bool(cfg.get("nomask", True))
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N               # number of pairwise tokens
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # --------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # --------------------------------------------------------------
    # 2️⃣ Fuse five linear maps (left, right, left_gate, right_gate, out_gate)
    # --------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16, device=device)    # [5·H, dim]
    proj = F.linear(x_norm, fused_w)       # (R, 5·H) FP16
    del fused_w, x_norm

    # --------------------------------------------------------------
    # 3️⃣ Split, mask, sigmoid gates → left/right/out_gate
    # --------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if not nomask and mask is not None:
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension (cuBLAS)
    # --------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)                  # FP16
    right_mat = right.view(B * H, N, N)                 # FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                         # [B, H, N, N] FP16
    del left, right

    # --------------------------------------------------------------
    # 5️⃣ Row‑wise mean & rsqrt‑variance of `out`
    # --------------------------------------------------------------
    mu   = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 256
    BLOCK_K_STATS = 32                     # H ≤ 128 ⇒ fits comfortably
    grid_stats = (triton.cdiv(R, BLOCK_M_STATS),)
    rowwise_stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm + out_gate
    # --------------------------------------------------------------
    normed = torch.empty_like(out)                 # FP16
    BLOCK_M_NG = 128
    BLOCK_H_NG = 64
    grid_ng = (triton.cdiv(R, BLOCK_M_NG),)
    apply_norm_gate_kernel[grid_ng](
        out,
        out_gate,
        mu,
        rstd,
        w["to_out_norm.weight"],   # γ  (FP32)
        w["to_out_norm.bias"],     # β  (FP32)
        normed,
        R,
        H,
        N,
        BLOCK_M=BLOCK_M_NG,
        BLOCK_H=BLOCK_H_NG,
        num_warps=8,
    )
    del out, out_gate, mu, rstd

    # --------------------------------------------------------------
    # 7️⃣ Final linear projection (to_out) – cuBLAS GEMM
    # --------------------------------------------------------------
    # reshape to [R, H] (channel‑last) and apply linear
    normed_perm = normed.permute(0, 2, 3, 1).contiguous()   # [B, N, N, H] FP16
    normed_flat = normed_perm.view(R, H)                     # [R, H] FP16

    to_out_w = w["to_out.weight"].to(torch.float16, device=device)   # [dim, H]
    final_fp16 = F.linear(normed_flat, to_out_w)                     # [R, dim] FP16
    final = final_fp16.to(torch.float32).view(B, N, N, dim)          # [B, N, N, dim] FP32

    return final
