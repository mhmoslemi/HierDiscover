    in_ptr, gate_ptr, gamma_ptr, beta_ptr,
    out_ptr, R, H, eps,
    BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr,

    pid = tl.program_id(0)

    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < R

    # First pass: compute sum & sumsq across H
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for h_off in range(0, H, BLOCK_H):
        offs_h = h_off + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        ptr = in_ptr + offs_m[:, None] * H + offs_h[None, :]
        vals = tl.load(ptr, mask=mask_m[:, None] & mask_h[None, :], other=0.0).to(tl.float32)  # [M, tile_H]

        sum_val += tl.sum(vals, axis=1)

"""
Outgoing TriMul (AlphaFold‑3) – forward‑only implementation.

Algorithm (high‑level)
----------------------
1️⃣ Row‑wise LayerNorm on the flattened pair tensor (FP32 → FP16) – Triton.
2️⃣ Fuse the five linear maps (left, right, left_gate, right_gate, out_gate)
   into a single GEMM (cuBLAS) → tensor (R, 5·H), R = B·N·N.
3️⃣ Split the fused projection, apply the optional pair‑mask, compute sigmoid
   gates and write the three tensors `left`, `right`, `out_gate` in layout
   [B, H, N, N] (FP16) – Triton.
4️⃣ Batched GEMM (cuBLAS) computes the core einsum:
        out[b, h, i, j] = Σₖ left[b, h, i, k]·right[b, h, j, k]
   → shape [B, H, N, N] (FP16).
5️⃣ Fuse the second LayerNorm, the `out_gate` multiplication and the final
   projection into a single Triton kernel that:
        * computes µ and σ⁻¹ per (b,i,j) over the hidden dimension H,
        * normalises, applies γ/β, multiplies by the gate,
        * writes a matrix M of shape (R, H) (FP16).
6️⃣ Final linear projection (cuBLAS) performs M @ W_outᵀ  →  (R, dim)
   (FP16 → cast to FP32) and reshapes to [B, N, N, dim].

Only the forward pass is required; no gradients are computed.
All kernels target Triton 3.3.1 on an NVIDIA H100 and use mixed precision
(FP32 for statistics, FP16 for the bulk of the work).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,   # [R, dim] FP32 input
    y_ptr,   # [R, dim] FP16 output
    w_ptr,   # [dim]   γ (FP32)
    b_ptr,   # [dim]   β (FP32)
    rows,    # scalar R
    dim,     # scalar dim
    eps,     # scalar eps
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)            # [BLOCK_M]
    mask_m = offs_m < rows

    # ----- first pass: accumulate sum & sum of squares -----
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0
        ).to(tl.float32)                                   # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                     # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)              # [BLOCK_M]

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ----- second pass: write normalised values -----
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0
        ).to(tl.float32)                                   # [BLOCK_M, BLOCK_D]

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * dim + offs_d[None, :],
            out.to(tl.float16),
            mask=mask_m[:, None] & mask_d[None, :],
        )


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, apply mask, sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,      # [R, 5·H]  FP16 fused projection
    left_ptr,      # [B, H, N, N] FP16
    right_ptr,     # [B, H, N, N] FP16
    out_gate_ptr,  # [B, H, N, N] FP16
    mask_ptr,      # [R]       FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)           # [BLOCK_R]
    mask_r = offs_r < R

    # Decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # Optional mask (broadcast on H)
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)  # [BLOCK_R]
        m_val = m_val[:, None]                                      # [BLOCK_R, 1]
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + k               # start of hidden plane 0

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)     # [BLOCK_H]
        mask_h = offs_h < H

        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]  # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # Sigmoid (computed in FP32 for accuracy)
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * m_val * left_gate
        right = right_raw * m_val * right_gate
        outg  = out_gate

        dst = base[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst, left,  mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst, right, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst, outg, mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 5️⃣ Fuse the second LayerNorm, out_gate & final projection (M = LN·gate)
# ----------------------------------------------------------------------
@triton.jit
def fused_ln_gate_kernel(
    in_ptr,        # [R, H]  FP16 – contract‑output tensor
    gate_ptr,      # [R, H]  FP16 – out_gate
    gamma_ptr,     # [H]    FP32 – to_out_norm.weight
    beta_ptr,      # [H]    FP32 – to_out_norm.bias
    out_ptr,       # [R, H]  FP16 – result M = LN(out) * out_gate
    R, H,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < R

    # ----- First pass: compute mean & variance over the hidden dimension -----
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for h_off in range(0, H, BLOCK_H):
        offs_h = h_off + tl.arange(0, BLOCK_H)          # [BLOCK_H]
        mask_h = offs_h < H

        ptr = in_ptr + offs_m[:, None] * H + offs_h[None, :]
        vals = tl.load(ptr, mask=mask_m[:, None] & mask_h[None, :], other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ----- Second pass: apply LN, gate and store -----
    for h_off in range(0, H, BLOCK_H):
        offs_h = h_off + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        ptr_in   = in_ptr   + offs_m[:, None] * H + offs_h[None, :]
        ptr_gate = gate_ptr + offs_m[:, None] * H + offs_h[None, :]
        vals = tl.load(ptr_in,   mask=mask_m[:, None] & mask_h[None, :], other=0.0).to(tl.float32)
        gates = tl.load(ptr_gate,mask=mask_m[:, None] & mask_h[None, :], other=0.0).to(tl.float32)

        # LayerNorm (FP32)
        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(gamma_ptr + offs_h, mask=mask_h, other=0.0)   # FP32
        beta  = tl.load(beta_ptr  + offs_h, mask=mask_h, other=0.0)   # FP32

        norm = norm * gamma[None, :] + beta[None, :]               # FP32
        out_vals = norm * gates                                    # FP32

        dst = out_ptr + offs_m[:, None] * H + offs_h[None, :]
        tl.store(dst, out_vals.to(tl.float16), mask=mask_m[:, None] & mask_h[None, :])

    # ------------------------------------------------------------------

    dim        = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask     = bool(cfg.get("nomask", True))

    R = B * N * N               # rows for pairwise tokens
    H = hidden_dim

    # ------------------------------------------------------------------

    BLOCK_D = 64

        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # ------------------------------------------------------------------
    # 2️⃣ Fuse five linear maps (left, right, left_gate, right_gate, out_gate)

        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)   # [5·H, dim]

    # ------------------------------------------------------------------

    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if not nomask and mask is not None:
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension (cuBLAS)
    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)                      # FP16
    right_mat = right.view(B * H, N, N)                     # FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                             # [B, H, N, N] FP16
    del left_mat, right_mat, left, right

    # ------------------------------------------------------------------
    # 5️⃣ Fuse second LayerNorm, out_gate, and prepare M = LN(out) * out_gate
    # ------------------------------------------------------------------
    # Pack tensors as [R, H] for easy access
    out_perm      = out.permute(0, 2, 3, 1).contiguous()          # [B, N, N, H]
    out_gate_perm = out_gate.permute(0, 2, 3, 1).contiguous()     # [B, N, N, H]

    out_flat      = out_perm.view(R, H)       # FP16
    out_gate_flat = out_gate_perm.view(R, H)  # FP16

    M = torch.empty(R, H, dtype=torch.float16, device=device)  # LN‑gated tensor

    BLOCK_M_OUT = 128   # rows per program
    BLOCK_H_OUT = 32    # hidden‑dim tile (fits in registers)
    grid_ln_gate = (triton.cdiv(R, BLOCK_M_OUT),)
    fused_ln_gate_kernel[grid_ln_gate](
        out_flat,
        out_gate_flat,
        w["to_out_norm.weight"],   # γ (FP32)
        w["to_out_norm.bias"],     # β (FP32)
        M,
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_H=BLOCK_H_OUT,
        num_warps=8,
    )
    del out, out_gate, out_perm, out_gate_perm, out_flat, out_gate_flat

    # ------------------------------------------------------------------
    # 6️⃣ Final linear projection (cuBLAS)
    # ------------------------------------------------------------------
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim]

    final_flat = torch.matmul(M, w_out_T)  # [R, dim] FP16
    final = final_flat.to(torch.float32).view(B, N, N, dim)   # BF16 → FP32

    # ------------------------------------------------------------------
    return final
```
