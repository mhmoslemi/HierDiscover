"""
TriMul (outgoing) – forward‑only Triton implementation.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the pairwise tensor (FP32 → FP16).
2️⃣ Fuse the five linear maps (left, right, left_gate, right_gate, out_gate) into one
   GEMM (cuBLAS) producing a matrix (R, 5·H) in FP16, where R = B·N·N.
3️⃣ Triton kernel splits the fused projection, applies the optional mask,
   computes the sigmoid gates and writes three tensors `left`, `right`,
   `out_gate` in layout [B, H, N, N] (FP16).
4️⃣ Batched GEMM (torch.bmm) evaluates the core einsum:
        out[b, h, i, j] = Σₖ left[b, h, i, k]·right[b, h, j, k]
   yielding [B, H, N, N] (FP16).
5️⃣ Row‑wise mean & rsqrt‑variance of `out` over the hidden dimension H
   (Triton kernel) → µ, rstd per (b,i,j).
6️⃣ Final Triton kernel fuses the second LayerNorm, the out‑gate multiplication
   and the output projection.  For each row (b,i,j) it computes
        out_fp32 = Σₕ ( ((out‑µ)·rstd·γ + β) * gate ) · W_out[h,d]
   and writes the result `[B, N, N, dim]` (FP32).

All kernels target Triton 3.3.1 (H100) and use mixed‑precision arithmetic.
No backward pass is required.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,            # [R, dim] input FP32
    y_ptr,            # [R, dim] output FP16
    w_ptr, b_ptr,    # [dim] LN weight & bias (FP32)
    rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [M]
    mask_m = offs_m < rows

    # ---------- pass 1: mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]   # [M, D]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- pass 2: write normalized output ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, apply mask & sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,        # [R, 5·H]  FP16  fused projection
    left_ptr,        # [B, H, N, N] FP16
    right_ptr,       # [B, H, N, N] FP16
    out_gate_ptr,    # [B, H, N, N] FP16
    mask_ptr,        # [R] FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [R]
    mask_r = offs_r < R

    # decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask (broadcast over H)
    if HAS_MASK:
        mval = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [R]
        mval = mval[:, None]                                        # [R,1]
    else:
        mval = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + k          # start of h=0 plane for this (b,i,k)

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # each projection occupies a contiguous chunk of size H
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [R, H]

        left_raw  = tl.load(proj_ptr + proj_base,
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0)
        right_raw = tl.load(proj_ptr + proj_base + H,
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0)
        left_g_raw  = tl.load(proj_ptr + proj_base + 2 * H,
                              mask=mask_r[:, None] & mask_h[None, :],
                              other=0.0)
        right_g_raw = tl.load(proj_ptr + proj_base + 3 * H,
                              mask=mask_r[:, None] & mask_h[None, :],
                              other=0.0)
        outg_raw    = tl.load(proj_ptr + proj_base + 4 * H,
                              mask=mask_r[:, None] & mask_h[None, :],
                              other=0.0)

        # sigmoid in FP32 → back to FP16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_g_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_g_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-outg_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mval * left_gate
        right = right_raw * mval * right_gate

        dst_off = base_offset[:, None] + offs_h[None, :] * hidden_stride   # [R, H]

        tl.store(left_ptr + dst_off,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_off,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_off,
                 out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 3️⃣ Row‑wise statistics (mean & rsqrt‑var) over hidden dim H
# ----------------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr,          # [B, H, N, N] FP16
    mu_ptr, rstd_ptr, # [R] FP32
    total_rows, H, N,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [M]
    mask_m = offs_m < total_rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + j          # start of h=0 plane

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_off in range(0, H, BLOCK_K):
        offs_k = k_off + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        ptr = out_ptr + base[:, None] + offs_k[None, :] * hidden_stride   # [M, K]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_m, mean,  mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)


# ----------------------------------------------------------------------
# 4️⃣ Final output: 2nd LayerNorm + out‑gate + output projection
# ----------------------------------------------------------------------
@triton.jit
def final_output_kernel(
    out_ptr,          # [B, H, N, N] FP16   (contracted tensor)
    gate_ptr,         # [B, H, N, N] FP16   (out_gate)
    mu_ptr, rstd_ptr, # [R] FP32
    ln_w_ptr, ln_b_ptr,                # [H] FP32  (to_out_norm weight/bias)
    w_out_T_ptr,                       # [H, dim] FP16  (transposed to_out weight)
    out_final_ptr,                     # [B, N, N, dim] FP32
    total_rows, H, N, dim,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr,
):
    pid_m = tl.program_id(0)    # rows (R) tile
    pid_n = tl.program_id(1)    # output‑dim (dim) tile

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)    # [M]
    offs_n = col_start + tl.arange(0, BLOCK_N)    # [N]
    mask_m = offs_m < total_rows
    mask_n = offs_n < dim

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + j    # start of plane h=0

    # load per‑row µ and rstd (FP32)
    mu   = tl.load(mu_ptr   + offs_m, mask=mask_m,  other=0.0)   # [M]
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m,  other=0.0)   # [M]

    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    for k_off in range(0, H, BLOCK_K):
        offs_k = k_off + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        # values from contracted tensor and gate (FP16 → FP32)
        out_tile  = tl.load(out_ptr  + base[:, None] + offs_k[None, :] * hidden_stride,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)          # [M, K]
        gate_tile = tl.load(gate_ptr + base[:, None] + offs_k[None, :] * hidden_stride,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)          # [M, K]

        # 2nd LayerNorm parameters (γ, β) are 1‑D over H
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [K]
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [K]

        # ((out‑µ)·rstd·γ + β)
        norm = ((out_tile - mu[:, None]) * rstd[:, None]) * gamma[None, :] + beta[None, :]

        # multiply by out_gate
        term = norm * gate_tile                                   # [M, K]
        term_fp16 = term.to(tl.float16)                           # cast back for tensor‑core dot

        # slice of W_out^T  (shape H×dim)
        w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]   # [K, N]
        w_tile = tl.load(w_ptr,
                         mask=mask_k[:, None] & mask_n[None, :],
                         other=0.0)                                         # [K, N]

        acc += tl.dot(term_fp16, w_tile).to(tl.float32)                # [M, N]

    # write final FP32 result
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Args:
        data (tuple):
            input_tensor : torch.Tensor [B, N, N, dim] (float32)
            mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            weights_dict : keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
                (all float32)
            config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim        = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask     = cfg.get("nomask", True)
    eps        = 1e-5
    device     = x.device

    B, N, _, _ = x.shape
    R = B * N * N               # total pairwise tokens
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                     # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear projection for the five heads (5·H)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                       # [5·H, dim]

    proj = F.linear(x_norm, fused_w)          # (R, 5·H) FP16
    del fused_w, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ Split, mask and sigmoid gates → left / right / out_gate
    # ------------------------------------------------------------------
    left     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right    = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK  = 1
        mask_ptr  = mask_flat
    else:
        HAS_MASK  = 0
        mask_ptr  = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM (left·rightᵀ) – core einsum
    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)                # [B·H, N, N] FP16
    right_mat = right.view(B * H, N, N)               # [B·H, N, N] FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                         # [B, H, N, N] FP16

    # ------------------------------------------------------------------
    # 5️⃣ Row‑wise mean & rsqrt‑var over hidden dim
    # ------------------------------------------------------------------
    mu   = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 256
    BLOCK_K_STATS = 128                     # H ≤ 128 in all supported configs
    grid_stats = (triton.cdiv(R, BLOCK_M_STATS),)
    rowwise_stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Final LayerNorm + out_gate + to_out projection
    # ------------------------------------------------------------------
    # to_out.weight has shape [dim, hidden_dim]; we need its transpose (H × dim)
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim]

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 128   # rows per program (tiles over R)
    BLOCK_K_OUT = 32    # hidden‑dim chunk processed per iteration (fits H=128)
    BLOCK_N_OUT = 128   # output‑dim chunk

    grid_out = (
        triton.cdiv(R, BLOCK_M_OUT),
        triton.cdiv(dim, BLOCK_N_OUT),
    )
    final_output_kernel[grid_out](
        out,
        out_gate,
        mu,
        rstd,
        w["to_out_norm.weight"],   # γ (FP32)
        w["to_out_norm.bias"],     # β (FP32)
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=8,
    )
    # cleanup (optional)
    del left, right, out_gate, out, mu, rstd, w_out_T

    return final
```

**Explanation of key steps**

* **LayerNorm kernel** – Computes mean/variance per row (R = B·N²) in FP32, then applies the learned γ/β and stores the normalized tensor in FP16.
* **Fused projection & gate kernel** – One GEMM (via `torch.nn.functional.linear`) produces the concatenated 5 × hidden projections. The Triton kernel slices this tensor, applies the optional mask, computes the sigmoid gates, and writes `left`, `right` and `out_gate` in the layout `[B, H, N, N]`.
* **Core pairwise multiplication** – Per‑hidden‑channel batched GEMM (`torch.bmm`) implements the `einsum('... i k d, ... j k d -> ... i j d')` efficiently.
* **Row‑wise statistics kernel** – Computes μ and rstd for the second LayerNorm across the hidden dimension H for every (b,i,j) position.
* **Final output kernel** – Fuses the second LayerNorm, multiplication by `out_gate`, and the projection back to the original `dim` (via the transposed `to_out` weight). The kernel processes tiles over rows (R) and output dimensions (dim) and writes the final FP32 result `[B, N, N, dim]`.

The implementation satisfies the required API, works for all listed configurations, and leverages Triton for the most expensive per‑element work while using cuBLAS (`torch.bmm`) for the large batched matrix multiplication.
