def custom_kernel(data):
    x, mask, weights, config = data
    device = x.device
    dim = config["dim"]; hidden_dim = config["hidden_dim"]; nomask = config.get("nomask", True)
    eps = 1e-5

    B, N, _, _ = x.shape
    R = B * N * N
    H = hidden_dim

    # 1️⃣ Input layernorm using torch.nn.functional.layer_norm
    x_norm = torch.nn.functional.layer_norm(
        x,
        normalized_shape=dim,
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    ).to(torch.float16)  # [B,N,N,dim] half

    # flatten for linear
    x_norm_flat = x_norm.reshape(R, dim).contiguous()

    # 2️⃣ Fuse 5 linear maps into one matmul
    fused_W = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)  # [5*H, dim]
    proj = torch.nn.functional.linear(x_norm_flat, fused_W)  # [R,5*H] half

    # 3️⃣ Split fused projection, apply mask and sigmoid gates
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if not nomask and mask is not None:
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)

    BLOCK_R = 64
    BLOCK_H = 64
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, x_norm, x_norm_flat, mask_ptr

    # 4️⃣ Core einsum via batched GEMM
    left_mat = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [B*H,N,N] half
    out = out.view(B, H, N, N)  # [B,H,N,N] half
    del left, right  # free memory

    # 5️⃣ Second LayerNorm + out_gate multiplication (all in FP32)
    out_f = out.float()          # [B,H,N,N] FP32
    gate_f = out_gate.float()    # [B,H,N,N] FP32

    mu = out_f.mean(dim=1, keepdim=True)               # [B,1,N,N]
    var = out_f.var(dim=1, unbiased=False, keepdim=True)  # [B,1,N,N]
    rstd = torch.rsqrt(var + eps)                      # [B,1,N,N]

    gamma = weights["to_out_norm.weight"]   # [H]
    beta = weights["to_out_norm.bias"]      # [H]
    gamma_v = gamma[None, :, None, None]    # [1,H,1,1]
    beta_v = beta[None, :, None, None]      # [1,H,1,1]

    # Apply LayerNorm: (x - μ) * rstd * γ + β
    out_norm = (out_f - mu) * rstd
    out_norm = out_norm * gamma_v + beta_v

    # Apply out_gate
    gated = out_norm * gate_f  # [B,H,N,N] FP32

    # 6️⃣ Final projection
    gated_flat = gated.permute(0, 2, 3, 1).contiguous().view(R, H)  # [R, H]
    final = torch.nn.functional.linear(gated_flat, weights["to_out.weight"])  # [R, dim] FP32
    final = final.view(B, N, N, dim)  # [B, N, N, dim]

    return final

   `out_gate` in layout [B, H, N, N] (FP16).  This is the only custom kernel.
4️⃣ Batched matrix‑multiply (cuBLAS) computes the core einsum:
       out[b, h, i, j] = Σₖ left[b, h, i, k] · right[b, h, j, k]
   → [B, H, N, N] (FP16).
5️⃣ Second LayerNorm is performed in FP32 with a simple reduction over the
   hidden dimension H (μ, σ).  The normalized tensor is multiplied by the
   `out_gate` (also promoted to FP32).
6️⃣ Final linear projection (FP32) is a single GEMM:
       final = (norm·out_gate) @ to_out.weightᵀ
   → [B, N, N, dim] (float32).

All heavy computations are dispatched to cuBLAS; only the
gate‑splitting step is hand‑written in Triton.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 3️⃣ Split fused projection, apply mask and sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,      # [R, 5·H]   FP16 fused projection
    left_ptr,      # [B, H, N, N]  FP16
    right_ptr,     # [B, H, N, N]  FP16
    out_gate_ptr,  # [B, H, N, N]  FP16
    mask_ptr,      # [R]            FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)           # [BLOCK_R]
    mask_r = offs_r < R

    # Decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # Optional mask (broadcast over H)
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        m_val = m_val[:, None]                                      # [BLOCK_R, 1]
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N                # stride of one hidden plane (N×N)
    # base address of hidden plane 0 for this (b,i,k)
    base = b * (H * hidden_stride) + i * N + k

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # Index into fused projection: layout (L|R|LG|RG|OG) each H long
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [R, H]

        left_raw      = tl.load(proj_ptr + proj_base,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_raw     = tl.load(proj_ptr + proj_base + H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        left_gate_raw = tl.load(proj_ptr + proj_base + 2 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_gate_raw = tl.load(proj_ptr + proj_base + 3 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        out_gate_raw = tl.load(proj_ptr + proj_base + 4 * H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)

        # Sigmoid (computed in FP32, result back to FP16)
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * m_val * left_gate
        right = right_raw * m_val * right_gate
        outg  = out_gate

        # Destination offset for the hidden dimension
        dst = base[:, None] + offs_h[None, :] * hidden_stride   # [R, H]

        tl.store(left_ptr     + dst, left,   mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr    + dst, right,  mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst, outg,   mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Entrypoint
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : dict with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
              (all float32, already on the target device)
            - config_dict must contain
                "dim" (int), "hidden_dim" (int), "nomask" (bool)
    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    device = x.device
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = bool(cfg.get("nomask", True))
    eps = 1e-5

    B, N, _, _ = x.shape
    R = B * N * N               # number of pairwise tokens
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    # torch.nn.functional.layer_norm works on the last dimension; this
    # matches the per‑pair LayerNorm in the reference implementation.
    x_norm = F.layer_norm(
        x,
        normalized_shape=dim,
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    ).to(torch.float16)                     # [B, N, N, dim] FP16

    # ------------------------------------------------------------------
    # 2️⃣ Fuse five linear maps (left, right, left_gate, right_gate, out_gate)
    # ------------------------------------------------------------------
    fused_W = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # [5·H, dim]
    # Linear: (R, dim) @ (dim, 5·H)ᵀ → (R, 5·H)
    proj = F.linear(x_norm.reshape(R, dim), fused_W)   # FP16, shape [R, 5·H]

    # ------------------------------------------------------------------
    # 3️⃣ Split fused projection, apply optional mask and sigmoid gates
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if not nomask and mask is not None:
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64   # rows per program
    BLOCK_H = 64   # hidden‑dim tile
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, x_norm, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Core einsum via batched GEMM (cuBLAS)
    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)          # FP16
    right_mat = right.view(B * H, N, N)         # FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))
    out = out.view(B, H, N, N)                  # [B, H, N, N] FP16
    del left, right  # free memory

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm, out_gate multiplication (FP32)
    # ------------------------------------------------------------------
    out_f   = out.float()          # [B, H, N, N] FP32
    gate_f  = out_gate.float()     # [B, H, N, N] FP32

    # per‑pair mean / variance over the hidden dimension
    mu   = out_f.mean(dim=1, keepdim=True)                    # [B,1,N,N]
    var  = out_f.var(dim=1, unbiased=False, keepdim=True)    # [B,1,N,N]
    rstd = torch.rsqrt(var + eps)                             # [B,1,N,N]

    # LayerNorm parameters (γ, β) – broadcast to [B,H,N,N]
    gamma = w["to_out_norm.weight"][None, :, None, None]   # [1,H,1,1]
    beta  = w["to_out_norm.bias"][None, :, None, None]     # [1,H,1,1]

    # Normalized hidden tensor
    out_norm = (out_f - mu) * rstd          # (x‑μ)·σ⁻¹
    out_norm = out_norm * gamma + beta     # γ· + β

    # Apply the output gate
    gated = out_norm * gate_f               # [B, H, N, N] FP32

    # ------------------------------------------------------------------
    # 6️⃣ Final projection (FP32 GEMM)
    # ------------------------------------------------------------------
    # Move the hidden dimension to the last axis to obtain a dense [R, H]
    gated_flat = gated.permute(0, 2, 3, 1).contiguous().view(R, H)  # [R, H]
    final = F.linear(gated_flat, w["to_out.weight"])               # [R, dim] FP32
    final = final.view(B, N, N, dim)                               # [B, N, N, dim]

    return final
```
