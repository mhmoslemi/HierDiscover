"""
TriMul (outgoing) – fast forward‑only Triton implementation.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the 4‑D pairwise tensor (FP32 → FP16) with a
    custom Triton kernel.

2️⃣ Fuse the five linear maps (left_proj, right_proj, left_gate,
    right_gate, out_gate) into a single GEMM (R × dim) → (R × 5·H)
    (FP16).  The result is split, masked and passed through sigmoid
    gates by a second Triton kernel, producing three tensors:
        left      : [B, H, N, N] (FP16)
        right     : [B, H, N, N] (FP16)
        out_gate  : [B, H, N, N] (FP16)

3️⃣ Batched GEMM (tensor‑core cuBLAS) evaluates the core einsum:
        out[b, h, i, j] = Σₖ left[b, h, i, k]·right[b, h, j, k]
    yielding a tensor `out` of shape [B, H, N, N] (FP16).

4️⃣ Per‑position mean / rsqrt‑variance over the hidden dimension H are
    obtained with built‑in PyTorch reductions (FP32).

5️⃣ The second LayerNorm, out‑gate multiplication and the final
    projection are performed with a few dense PyTorch ops:
        norm = ((out – μ)·rstd)·γ + β               (γ,β = to_out_norm)
        gated = norm * out_gate
        final = gated @ to_out.weightᵀ                (FP16 matmul)

   The result is reshaped to `[B, N, N, dim]` (float32).

The implementation keeps the heavy work (LayerNorm, projection split &
gates) in Triton kernels while delegating the large dense GEMMs and
reductions to highly‑optimized cuBLAS / Torch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,            # [R, dim] input FP32
    y_ptr,            # [R, dim] output FP16
    w_ptr, b_ptr,    # [dim] LN weight & bias (FP32)
    rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # ----- pass 1: mean & variance -----
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]          # [M, D]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ----- pass 2: write normalized output -----
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, apply mask & sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,        # [R, 5·H]  FP16 fused projection
    left_ptr,        # [B, H, N, N] FP16
    right_ptr,       # [B, H, N, N] FP16
    out_gate_ptr,    # [B, H, N, N] FP16
    mask_ptr,        # [R] FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [R]
    mask_r = offs_r < R

    # decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask (broadcast over H)
    if HAS_MASK:
        mval = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [R]
        mval = mval[:, None]                                       # [R,1]
    else:
        mval = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + k          # start of h=0 plane

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # each projection occupies a contiguous chunk of size H
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [R, H]

        left_raw  = tl.load(proj_ptr + proj_base,
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0)
        right_raw = tl.load(proj_ptr + proj_base + H,
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0)
        left_g_raw  = tl.load(proj_ptr + proj_base + 2 * H,
                              mask=mask_r[:, None] & mask_h[None, :],
                              other=0.0)
        right_g_raw = tl.load(proj_ptr + proj_base + 3 * H,
                              mask=mask_r[:, None] & mask_h[None, :],
                              other=0.0)
        outg_raw    = tl.load(proj_ptr + proj_base + 4 * H,
                              mask=mask_r[:, None] & mask_h[None, :],
                              other=0.0)

        # sigmoid (computed in FP32 then cast back)
        left_gate  = (1.0 / (1.0 + tl.exp(-left_g_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_g_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-outg_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mval * left_gate
        right = right_raw * mval * right_gate

        dst_off = base_offset[:, None] + offs_h[None, :] * hidden_stride   # [R, H]

        tl.store(left_ptr + dst_off, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_off, right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_off, out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor or None, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim        = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask     = cfg.get("nomask", True)          # True → ignore mask
    eps        = 1e-5
    device     = x.device

    B, N, _, _ = x.shape
    R = B * N * N                     # total pairwise tokens
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                     # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear projection for the five heads (5·H)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                       # [5·H, dim]

    proj = F.linear(x_norm, fused_w)          # (R, 5·H) FP16
    del fused_w, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ Split, apply optional mask, sigmoid gates → left/right/out_gate
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (mask is not None) and (not nomask):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Core einsum via batched GEMM: out[b, h, i, j] = Σₖ left·rightᵀ
    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)                # [B·H, N, N] FP16
    right_mat = right.view(B * H, N, N)               # [B·H, N, N] FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                         # [B, H, N, N] FP16

    # free memory that is no longer needed
    del left, right
    torch.cuda.empty_cache()

    # ------------------------------------------------------------------
    # 5️⃣ Row‑wise mean & rsqrt‑variance over the hidden dimension H
    # ------------------------------------------------------------------
    # torch.mean / var are heavily optimized; they compute in FP32.
    mu  = out.float().mean(dim=1)                                # [B, N, N] FP32
    var = out.float().var(dim=1, unbiased=False)                # [B, N, N] FP32
    rstd = (var + eps).rsqrt()                                   # [B, N, N] FP32

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm + out‑gate + output projection (large GEMM)
    # ------------------------------------------------------------------
    # bring tensors to shape [R, H] where R = B·N·N
    out_perm = out.permute(0, 2, 3, 1).contiguous()    # [B, N, N, H]
    out_flat = out_perm.view(R, H)                     # FP16

    gate_perm = out_gate.permute(0, 2, 3, 1).contiguous()
    gate_flat = gate_perm.view(R, H)                   # FP16

    mu_flat   = mu.view(R, 1)                          # FP32
    rstd_flat = rstd.view(R, 1)                        # FP32

    # cast scalar statistics to FP16 for the element‑wise norm
    mu_fp16   = mu_flat.to(torch.float16)
    rstd_fp16 = rstd_flat.to(torch.float16)

    # to_out_norm parameters
    gamma_fp16 = w["to_out_norm.weight"].to(torch.float16)   # [H]
    beta_fp16  = w["to_out_norm.bias"].to(torch.float16)    # [H]

    # (out - μ)·rstd·γ + β   – all in FP16
    norm_fp16 = ((out_flat - mu_fp16) * rstd_fp16) * gamma_fp16[None, :] + beta_fp16[None, :]

    # gating (out_gate) follows the second LayerNorm
    gated_fp16 = norm_fp16 * gate_flat

    # final projection:   gated_fp16 @ to_out.weightᵀ   (FP16 matmul)
    final_fp16 = torch.nn.functional.linear(
        gated_fp16,
        w["to_out.weight"].to(torch.float16)   # shape [dim, H]
    )   # -> [R, dim] FP16

    # reshape to the original 4‑D layout and cast back to FP32
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
