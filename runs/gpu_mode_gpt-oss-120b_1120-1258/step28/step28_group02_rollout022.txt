def split_fuse_gate_kernel_proj(
    proj_ptr,  # [R, 5*H] FP16
    left_ptr,  # [B, N, N, H] FP16
    right_ptr,
    out_gate_ptr,
    mask_ptr,   # [R] FP16 optional
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)  # [BLOCK_R]
    mask_r = offs_r < R

    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # compute base index into left/right/out_gate reshaped [B,N,N,H] flattened as 1D
    pair_idx = ((b * N + i) * N + k)   # shape [BLOCK_R]
    base = pair_idx * H                # starting offset for H dimension

    # Optional mask loading
    if HAS_MASK:
        m = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)[:, None]  # [BLOCK_R,1]
    else:
        m = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # load fused projection slices
        # proj layout: for each row r, the 5*H values are contiguous: left, right, left_gate, right_gate, out_gate
        # So offset = offs_r * (5*H) + offs_h
        base_proj = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(proj_ptr + base_proj, mask=mask_r[:, None] & mask_h[None, :], other=0.0)
        right_raw = tl.load(proj_ptr + base_proj + H, mask=mask_r[:, None] & mask_h[None, :], other=0.0)

        left = left_raw * m * left_gate
        right = right_raw * m * right_gate
        og = out_gate

        # destination pointers: base + offs_h + hidden stride zero? Actually hidden stride = 1, but we need to offset per hidden index.
        dst = base[:, None] + offs_h[None, :]   # [BLOCK_R, BLOCK_H] offsets

# -*- coding: utf-8 -*-
"""
Outgoing TriMul (AlphaFold‑3) – forward‑only implementation (high‑performance).

Key ideas
---------
1️⃣  Row‑wise LayerNorm on the flattened pair tensor is performed with the
    highly‑optimised PyTorch kernel (torch.nn.functional.layer_norm) – this
    removes a custom Triton launch and lets cuDNN handle the reduction
    in FP32.

2️⃣  Five linear maps (left, right, left_gate, right_gate, out_gate) are
    fused into a single GEMM (FP16) using torch.nn.functional.linear.

3️⃣  A single Triton kernel (`split_fuse_gate_kernel`) writes the three
    tensors *left*, *right* and *out_gate* directly into the layout
    ``[B, N, N, H]`` (H = hidden_dim) while:
    • applying the optional pair‑mask,
    • applying the sigmoid gates,
    • broadcasting the mask over the hidden dimension.

4️⃣  Contraction (the O(N³) “einsum”) is carried out with a batched GEMM:
    the tensors are permuted to ``[B, H, N, N]`` (contiguous across H) and
    a single ``torch.bmm`` is issued for each hidden slice.
    This yields the intermediate tensor ``out`` in the layout
    ``[B, N, N, H]`` (H is the fastest‑changing dimension).

5️⃣  The second LayerNorm, the element‑wise ``out_gate`` multiplication,
    and the final projection onto the model dimension are performed with
    stock PyTorch ops (layer_norm + linear).  Since the hidden dimension
    is now contiguous, the LayerNorm runs on a perfectly strided tensor.

All heavy arithmetic stays in FP16 (Tensor‑cores) and only the final
output is cast back to FP32.  The only Triton kernel needed is the fused
split / mask / gate kernel, which dramatically cuts launch overhead
compared with the reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
#  Kernel: split the fused projection, apply mask + sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,      # [R, 5*H]   FP16 fused projection
    left_ptr,      # [B, N, N, H] FP16
    right_ptr,     # [B, N, N, H] FP16
    out_gate_ptr,  # [B, N, N, H] FP16
    mask_ptr,      # [R]          FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,  # rows (pair‑tokens) per program
    BLOCK_H: tl.constexpr,  # hidden‑dim tiles per program
):
    pid = tl.program_id(0)

    # ------------------------------------------------------------------
    # 1) Row index range
    # ------------------------------------------------------------------
    offs_r = pid * BLOCK_R + tl.arange(0, BLOCK_R)   # [BLOCK_R]
    mask_r = offs_r < R

    # ------------------------------------------------------------------
    # 2) Optional mask (broadcast on the hidden dimension)
    # ------------------------------------------------------------------
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)[:, None]  # [BLOCK_R,1]
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # ------------------------------------------------------------------
    # 3) Loop over hidden‑dim tiles
    # ------------------------------------------------------------------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)          # [BLOCK_H]
        mask_h = offs_h < H

        # offset inside the fused projection matrix
        # each row r contains (5*H) contiguous values: [L|R|LG|RG|OG]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        # ----- load the five slices -----
        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # ----- sigmoid (computed in FP32 for accuracy) -----
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # ----- apply mask and gates -----
        left_val      = left_raw      * m_val * left_gate
        right_val     = right_raw     * m_val * right_gate
        out_gate_val  = out_gate

        # ----- write out (layout [B, N, N, H] -> flat index = r*H + h) -----
        dst = offs_r[:, None] * H + offs_h[None, :]                 # [BLOCK_R, BLOCK_H]
        tl.store(left_ptr     + dst, left_val,     mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr    + dst, right_val,    mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst, out_gate_val, mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
#  Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul operator (no gradients).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
              (all float32)
            - config_dict must contain "dim" (int), "hidden_dim" (int),
              "nomask" (bool, optional)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # 0️⃣ unpack
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim          = cfg["dim"]
    hidden_dim   = cfg["hidden_dim"]
    nomask       = bool(cfg.get("nomask", True))
    eps          = 1e-5
    device       = x.device

    B, N, _, _ = x.shape
    R = B * N * N                     # total number of pair‑tokens
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm + cast to FP16 (use the highly‑optimised
    #    PyTorch implementation – this is faster than a custom Triton kernel)
    # ------------------------------------------------------------------
    norm_w_fp16 = w["norm.weight"].to(torch.float16)
    norm_b_fp16 = w["norm.bias"].to(torch.float16)

    x_flat = x.view(R, dim)                           # [R, dim]  FP32
    # LayerNorm on the last dimension (normalized_shape = dim)
    x_norm = F.layer_norm(
        x_flat, (dim,), weight=norm_w_fp16, bias=norm_b_fp16, eps=eps
    ).to(torch.float16)                               # [R, dim]  FP16

    # ------------------------------------------------------------------
    # 2️⃣ Fuse the five linear maps (left, right, left_gate, right_gate,
    #    out_gate) into a single GEMM.
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                               # [5*H, dim]

    proj = F.linear(x_norm, fused_w)                   # [R, 5*H]  FP16

    # ------------------------------------------------------------------
    # 3️⃣ Split, mask and apply sigmoid gates → left / right / out_gate
    #    Layout: [B, N, N, H] (H is the fastest‑changing axis)
    # ------------------------------------------------------------------
    left      = torch.empty(B, N, N, H, dtype=torch.float16, device=device)
    right     = torch.empty_like(left)
    out_gate  = torch.empty_like(left)

    if not nomask and mask is not None:
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 128
    BLOCK_H = 32
    grid = (triton.cdiv(R, BLOCK_R),)

    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    # proj is no longer needed
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Core O(N³) contraction using a batched GEMM.
    #    Convert to [B, H, N, N] (contiguous across H) for cuBLAS.
    # ------------------------------------------------------------------
    #   left  : [B, N, N, H] -> permute -> [B, H, N, N]
    left_perm  = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    left_mat  = left_perm.view(B * H, N, N)                # FP16
    right_mat = right_perm.view(B * H, N, N)               # FP16

    # Contract over the “k” dimension (matrix multiply)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B*H, N, N] FP16

    # Reshape back to [B, N, N, H] (H fastest)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm (over H), multiply by out_gate, final Projection.
    # ------------------------------------------------------------------
    #   Cast LayerNorm parameters to FP16 for the FP16 tensor.
    out_norm_w_fp16 = w["to_out_norm.weight"].to(torch.float16)
    out_norm_b_fp16 = w["to_out_norm.bias"].to(torch.float16)

    out_norm = F.layer_norm(
        out, (H,), weight=out_norm_w_fp16, bias=out_norm_b_fp16, eps=eps
    )                                                     # FP16, shape [B,N,N,H]

    # Align out_gate to the same layout
    out_gate_perm = out_gate.permute(0, 2, 3, 1).contiguous()   # [B,N,N,H]
    gated = out_norm * out_gate_perm                               # element‑wise

    # Final linear map: H -> dim (use FP16 weight for Tensor‑Core utilisation)
    w_out_fp16 = w["to_out.weight"].to(torch.float16)              # [dim, H]
    final_fp16 = F.linear(gated, w_out_fp16)                      # [B,N,N,dim] FP16

    # Cast back to FP32 as required by the API
    final = final_fp16.to(torch.float32)

    return final
```
