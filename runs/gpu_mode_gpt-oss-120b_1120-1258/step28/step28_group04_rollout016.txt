"""
TriMul – outgoing direction (forward only)

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the last dimension of the 4‑D pair tensor.
   Input [B·N·N, dim] (FP32) → normalized output in FP16.

2️⃣ Fuse the five linear maps (left_proj, right_proj, left_gate,
   right_gate, out_gate) into a single GEMM (torch.nn.functional.linear).
   Result shape = [R, 5·H] where R = B·N·N and H = hidden_dim.

3️⃣ Split the fused projection, apply the optional pair mask, take
   sigmoid on the three gates and store `left`, `right`, `out_gate`
   in layout [B, H, N, N] (FP16).  This work is done by a Triton kernel.

4️⃣ Core contraction: out[b, h, i, j] = Σₖ left[b, h, i, k] · right[b, h, j, k].
   Implemented as a batched GEMM via `torch.bmm` (cuBLAS).

5️⃣ Compute per‑row statistics μ and rstd over the hidden dimension H.
   μ = mean(out, dim=1),  rstd = rsqrt(var + eps).  Performed with
   native PyTorch reductions (fast FP32 kernels).

6️⃣ Apply the second LayerNorm, multiply by the `out_gate` sigmoid,
   and perform the final linear projection (hidden → dim) with a
   fused `torch.nn.functional.linear`.  All remaining work is in
   FP32 and the final tensor is reshaped to [B, N, N, dim] (float32).

Only the forward pass is required; no gradient computation.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,   # [R, dim]  FP32 input
    y_ptr,   # [R, dim]  FP16 output
    w_ptr,   # [dim]    γ (FP32)
    b_ptr,   # [dim]    β (FP32)
    rows,    # scalar   R
    dim,     # scalar   dim
    eps,     # scalar   eps
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)       # [BLOCK_M]
    mask_m = offs_m < rows

    # ----- first pass: mean & variance ---------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)               # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                     # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)              # [BLOCK_M]

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ----- second pass: write normalized values ------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, apply mask & sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,      # [R, 5·H]        FP16 fused projection
    left_ptr,      # [B, H, N, N]    FP16
    right_ptr,     # [B, H, N, N]    FP16
    out_gate_ptr,  # [B, H, N, N]    FP16
    mask_ptr,      # [R]             FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)           # [BLOCK_R]
    mask_r = offs_r < R

    # Decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # Optional mask (broadcast on H)
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        m_val = m_val[:, None]                                     # [BLOCK_R, 1]
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N                       # stride of one H‑slice
    base = b * (H * hidden_stride) + i * N + k  # base offset for h = 0

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)  # [BLOCK_H]
        mask_h = offs_h < H

        # Index into the fused projection: [left|right|lg|rg|og]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]      # [BLOCK_R, BLOCK_H]

        left_raw       = tl.load(proj_ptr + proj_base,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        right_raw      = tl.load(proj_ptr + proj_base + H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        left_gate_raw  = tl.load(proj_ptr + proj_base + 2 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        right_gate_raw = tl.load(proj_ptr + proj_base + 3 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        out_gate_raw   = tl.load(proj_ptr + proj_base + 4 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)

        # Sigmoid (FP32 for stability)
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * m_val * left_gate
        right = right_raw * m_val * right_gate
        outg  = out_gate

        dst = base[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst,
                 outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Entry point – custom_kernel
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only outgoing TriMul.

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping with the following keys (all torch tensors):
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config_dict must contain:
                "dim" (int), "hidden_dim" (int), "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # 0️⃣ Unpack inputs
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = bool(cfg.get("nomask", True))
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                      # total pairwise tokens
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # --------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()                     # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat  # free memory early

    # --------------------------------------------------------------
    # 2️⃣ Fuse five linear maps (torch GEMM)
    # --------------------------------------------------------------
    fused_weight = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16).contiguous()          # [5·H, dim]

    proj = F.linear(x_norm, fused_weight)     # (R, 5·H) FP16
    del fused_weight, x_norm

    # --------------------------------------------------------------
    # 3️⃣ Split, apply mask & sigmoid gates
    # --------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if not nomask and mask is not None:
        # Flatten mask to [R] and cast to fp16
        mask_flat = mask.to(torch.float16).view(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        # dummy tensor – never dereferenced when HAS_MASK==0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣ Core contraction (einsum) via batched GEMM
    # --------------------------------------------------------------
    # reshape to [B·H, N, N] and perform left @ rightᵀ per (b,h)
    left_mat  = left.view(B * H, N, N)                     # FP16
    right_mat = right.view(B * H, N, N)                    # FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B·H, N, N) FP16
    out = out.view(B, H, N, N)                            # [B, H, N, N] FP16
    del left_mat, right_mat

    # --------------------------------------------------------------
    # 5️⃣ Row‑wise mean & rsqrt‑variance of `out` (FP32)
    # --------------------------------------------------------------
    # Convert to FP32 for stable statistics
    out_fp32 = out.float()
    mu   = out_fp32.mean(dim=1)               # [B, N, N]  FP32
    var  = out_fp32.pow(2).mean(dim=1) - mu.pow(2)  # unbiased=False
    rstd = torch.rsqrt(var + eps)             # [B, N, N]  FP32

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm, out_gate multiplication and final projection
    # --------------------------------------------------------------
    # Expand μ, rstd, LN parameters to broadcast over H‑axis
    mu   = mu.unsqueeze(1)        # [B,1,N,N]
    rstd = rstd.unsqueeze(1)      # [B,1,N,N]

    gamma = w["to_out_norm.weight"].view(1, H, 1, 1)   # [1,H,1,1] FP32
    beta  = w["to_out_norm.bias"].view(1, H, 1, 1)    # [1,H,1,1] FP32

    # LayerNorm over hidden dim H, multiply by out_gate, then linear proj.
    #   term = ((out - μ) * rstd) * γ + β
    #   term = term * out_gate
    term = ((out_fp32 - mu) * rstd) * gamma + beta      # [B,H,N,N] FP32
    term = term * out_gate.float()                      # elementwise gate

    # Flatten to [R, H] where rows follow (b,i,j) order
    term_flat = term.permute(0, 2, 3, 1).contiguous().view(R, H)   # [R, H]

    # Final linear projection (hidden → dim) – weight shape [dim, H]
    final = F.linear(term_flat, w["to_out.weight"])   # [R, dim] FP32

    # Reshape back to [B, N, N, dim]
    out_final = final.view(B, N, N, dim)

    # --------------------------------------------------------------
    return out_final
