"""
TriMul (outgoing) – forward‑only Triton implementation.

The operator works on a 4‑D pairwise tensor X ∈ ℝ^{B×N×N×D} and
produces Y ∈ ℝ^{B×N×N×D}.  The forward pass consists of

1️⃣ Row‑wise LayerNorm on the last dimension (D → fp16).
2️⃣ A single fused linear projection that computes the five
   hidden‑dim heads (left, right, left_gate, right_gate, out_gate)
   in one GEMM.
3️⃣ A Triton kernel that
      – splits the fused projection,
      – applies the optional mask,
      – evaluates the sigmoid gates,
      – writes the three tensors ``left``, ``right`` and ``out_gate``.
4️⃣ Batched matrix multiply (torch.bmm) implements the core
   Einsum   out[b,h,i,j] = Σ_k left[b,h,i,k]·right[b,h,j,k].
5️⃣ Row‑wise statistics (mean, rsqrt‑var) of the contracted tensor
   over the hidden dimension H (another Triton kernel).
6️⃣ A final Triton kernel that fuses the second LayerNorm,
   the out‑gate multiplication and the output linear projection.
   The result is written in FP32 as required by the model.

All kernels target Triton 3.3.1 on an H100 and use mixed‑precision
(FP16/Tensor‑cores for the heavy arithmetic, FP32 for the
normalisation scalars).  No backward pass is implemented.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,        # [R, D]  input FP32
    y_ptr,        # [R, D]  output FP16
    w_ptr, b_ptr, # [D]   LN weight & bias (FP32)
    rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [M]
    mask_m = offs_m < rows

    # ---------- pass 1: compute mean & var ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    num_d_iters = tl.cdiv(dim, BLOCK_D)
    for d_it in range(0, num_d_iters):
        offs_d = d_it * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]   # [M, D]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- pass 2: write normalised output ----------
    for d_it in range(0, num_d_iters):
        offs_d = d_it * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, apply mask & sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,       # [R, 5*H]  FP16 fused projection
    left_ptr,       # [B, H, N, N] FP16
    right_ptr,      # [B, H, N, N] FP16
    out_gate_ptr,   # [B, H, N, N] FP16
    mask_ptr,       # [R] optional FP16 mask (1.0 means keep)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [R]
    mask_r = offs_r < R

    # flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask (broadcast over H)
    if HAS_MASK:
        mval = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [R]
        mval = mval[:, None]                                       # [R,1]
    else:
        mval = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + k          # start of h=0 plane

    # iterate over hidden dimension in chunks
    for h_it in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_it * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # layout of fused projection: [R, 5*H] → stride H for each head
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [R, H]

        left_raw   = tl.load(proj_ptr + proj_base,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        right_raw  = tl.load(proj_ptr + proj_base + H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        lgate_raw  = tl.load(proj_ptr + proj_base + 2 * H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        rgate_raw  = tl.load(proj_ptr + proj_base + 3 * H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        outg_raw   = tl.load(proj_ptr + proj_base + 4 * H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)

        # sigmoid in FP32, cast back to FP16
        lgate = (1.0 / (1.0 + tl.exp(-lgate_raw.to(tl.float32)))).to(tl.float16)
        rgate = (1.0 / (1.0 + tl.exp(-rgate_raw.to(tl.float32)))).to(tl.float16)
        outg  = (1.0 / (1.0 + tl.exp(-outg_raw.to(tl.float32)))).to(tl.float16)

        # apply mask and gates
        left  = left_raw  * mval * lgate
        right = right_raw * mval * rgate

        dst_off = base_offset[:, None] + offs_h[None, :] * hidden_stride   # [R, H]

        tl.store(left_ptr + dst_off,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_off,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_off,
                 outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 3️⃣ Row‑wise mean & rsqrt‑var over hidden dim H (for 2nd LayerNorm)
# ----------------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr,               # [B, H, N, N] FP16
    mu_ptr, rstd_ptr,      # [R] FP32
    total_rows, H, N,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [M]
    mask_m = offs_m < total_rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + j          # start of h=0 plane

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_off in range(0, H, BLOCK_K):
        offs_k = k_off + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        ptr = out_ptr + base[:, None] + offs_k[None, :] * hidden_stride   # [M, K]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_m, mean,  mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)


# ----------------------------------------------------------------------
# 4️⃣ Final output kernel (2nd LN + out‑gate + projection)
# ----------------------------------------------------------------------
@triton.jit
def final_output_kernel(
    out_ptr,            # [B, H, N, N] FP16   contracted tensor
    gate_ptr,           # [B, H, N, N] FP16   out_gate
    mu_ptr, rstd_ptr,   # [R] FP32
    ln_w_ptr, ln_b_ptr, # [H] FP32  (to_out_norm γ, β)
    w_out_T_ptr,        # [H, dim] FP16 (transposed weight)
    out_final_ptr,      # [B, N, N, dim] FP32
    total_rows, H, N, dim,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr,
):
    pid_m = tl.program_id(0)        # rows (R) tile
    pid_n = tl.program_id(1)        # output‑dim (dim) tile

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)    # [M]
    offs_n = col_start + tl.arange(0, BLOCK_N)    # [D]
    mask_m = offs_m < total_rows
    mask_n = offs_n < dim

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + j    # start of h=0 plane

    # per‑row statistics
    mu   = tl.load(mu_ptr   + offs_m, mask=mask_m,  other=0.0)   # [M]
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m,  other=0.0)   # [M]

    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    for k_off in range(0, H, BLOCK_K):
        offs_k = k_off + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        # contracted tensor and gate (FP16 → FP32)
        out_tile  = tl.load(out_ptr + base[:, None] + offs_k[None, :] * hidden_stride,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)          # [M, K]
        gate_tile = tl.load(gate_ptr + base[:, None] + offs_k[None, :] * hidden_stride,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)          # [M, K]

        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [K]
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [K]

        # ((out‑µ)·rstd·γ + β)
        norm = ((out_tile - mu[:, None]) * rstd[:, None]) * gamma[None, :] + beta[None, :]

        # multiply by out_gate
        term = norm * gate_tile                                   # [M, K]
        term_fp16 = term.to(tl.float16)                           # cast for tensor‑core dot

        # slice of W_out^T  (H × dim)
        w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]   # [K, D]
        w_tile = tl.load(w_ptr,
                         mask=mask_k[:, None] & mask_n[None, :],
                         other=0.0)                                         # [K, D]

        acc += tl.dot(term_fp16, w_tile).to(tl.float32)                # [M, D]

    # write final result (FP32)
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Args:
        data (tuple):
            input_tensor : torch.Tensor [B, N, N, dim] (float32)
            mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            weights_dict : containing the following keys (all float32):
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            config_dict  : must contain "dim", "hidden_dim", "nomask"

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim        = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask     = cfg.get("nomask", True)
    eps        = 1e-5
    device     = x.device

    B, N, _, _ = x.shape
    R = B * N * N                     # total pairwise tokens
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    # flatten to [R, dim]
    x_flat = x.reshape(R, dim).contiguous()                     # FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M  = 128
    BLOCK_D  = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat  # free memory

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear projection for the five heads (5·H)
    # ------------------------------------------------------------------
    fused_weight = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # [5·H, dim]

    # (R, 5·H) FP16
    proj = F.linear(x_norm, fused_weight)
    del fused_weight, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ Split, mask and sigmoid gates → left / right / out_gate
    # ------------------------------------------------------------------
    left     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right    = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Core einsum via batched GEMM (left·rightᵀ)
    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)                # [B·H, N, N] FP16
    right_mat = right.view(B * H, N, N)               # [B·H, N, N] FP16
    # out_tmp = Σ_k left[b,h,i,k] * right[b,h,j,k]
    out_tmp = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out_tmp.view(B, H, N, N)                         # [B, H, N, N] FP16

    # ------------------------------------------------------------------
    # 5️⃣ Row‑wise statistics (mean & rsqrt‑var) over hidden dim H
    # ------------------------------------------------------------------
    mu   = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 256
    BLOCK_K_STATS = 128                     # H ≤ 128 for all supported configs
    grid_stats = (triton.cdiv(R, BLOCK_M_STATS),)
    rowwise_stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Final LayerNorm + out‑gate + output projection
    # ------------------------------------------------------------------
    # to_out.weight : [dim, hidden] in the original model → we need its transpose
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim]

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 128   # rows per program
    BLOCK_K_OUT = 32    # hidden‑dim chunk
    BLOCK_N_OUT = 128   # output‑dim chunk

    grid_out = (
        triton.cdiv(R, BLOCK_M_OUT),
        triton.cdiv(dim, BLOCK_N_OUT),
    )
    final_output_kernel[grid_out](
        out,
        out_gate,
        mu,
        rstd,
        w["to_out_norm.weight"],   # γ  (FP32)
        w["to_out_norm.bias"],     # β  (FP32)
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=8,
    )

    return final
