import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def split_fuse_gate_kernel(proj_ptr, left_ptr, right_ptr, out_gate_ptr, mask_ptr,
                           R, H, N,
                           HAS_MASK: tl.constexpr,
                           BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)  # [R]
    mask_r = offs_r < R

    # decode flat index to (b,i,k)
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask
    if HAS_MASK:
        mval = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)  # [R]
        mval = mval[:, None]  # broadcast over H
    else:
        mval = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + k

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # proj segment (R,5H) linear layout: [left, right, left_gate, right_gate, out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]

        left_raw = tl.load(proj_ptr + proj_base,
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0)
        right_raw = tl.load(proj_ptr + proj_base + H,
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0)
        left_gate_raw = tl.load(proj_ptr + proj_base + 2 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_gate_raw = tl.load(proj_ptr + proj_base + 3 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        out_gate_raw = tl.load(proj_ptr + proj_base + 4 * H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)

        # sigmoid in fp32
        left_gate = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left = left_raw * mval * left_gate
        right = right_raw * mval * right_gate

        # destination offset into dense tensor: [B, H, N, N]
        dst_off = base_offset[:, None] + offs_h[None, :] * hidden_stride # [R, H]

        tl.store(left_ptr + dst_off,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_off,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_off,
                 out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])

def custom_kernel(data):
    """
    Forward‑only outgoing TriMul operator implemented with a mix of
    PyTorch and Triton.  The main compute steps are:
    
    1. Row‑wise LayerNorm on the input tensor (FP32 → FP16).
    2. A single fused linear projection that produces the five
       intermediate tensors (left, right, left‑gate, right‑gate,
       out‑gate) in FP16.
    3. Triton kernel `split_fuse_gate_kernel` that
       *splits* the fused projection, applies the optional mask, &
       computes the sigmoid gates.
    4. Batched GEMM (left · rightᵀ) using torch.bmm to realise the
       core ``einsum``.
    5. Mean / variance over the hidden dimension (H) are computed with
       torch’s native reduction (fast on H100).
    6. Final LayerNorm over H, gating, and projection back to the
       original ``dim`` using torch.nn.functional.linear.
    
    All heavy arithmetic stays in FP16 where possible; reductions are
    performed in FP32 for numerical stability.  The implementation
    works for any supported batch size, sequence length and hidden
    dimension and respects optional masks.
    """
    import torch
    import torch.nn.functional as F
    import triton
    import triton.language as tl
    
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device
    
    B, N, _, _ = x.shape
    R = B * N * N
    H = hidden_dim
    
    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (torch implementation – fast, uses cuDNN)
    # ------------------------------------------------------------------
    # reshape to (R, dim)
    x_flat = x.reshape(R, dim)
    # torch.layer_norm works on the last dimension
    x_norm_fp32 = F.layer_norm(
        x_flat,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )
    # Cast to FP16 for the rest of the pipeline
    x_norm = x_norm_fp32.half()
    del x_flat, x_norm_fp32
    
    # ------------------------------------------------------------------
    # 2️⃣ Fused projection of five linear maps (5·H × dim)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)  # [5·H, dim]
    proj = F.linear(x_norm, fused_w)  # (R, 5·H)  FP16
    del x_norm, fused_w
    
    # ------------------------------------------------------------------
    # 3️⃣ Split → mask → sigmoid gates (Triton)
    # ------------------------------------------------------------------
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    
    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy
    
    BLOCK_R = 64   # rows per program
    BLOCK_H = 64   # hidden dimension chunk
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr
    
    # ------------------------------------------------------------------
    # 4️⃣ Core einsum via batched GEMM (left·rightᵀ)
    # ------------------------------------------------------------------
    left_mat = left.view(B * H, N, N)          # [B·H, N, N]   FP16
    right_mat = right.view(B * H, N, N)        # [B·H, N, N]   FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                # [B, H, N, N] FP16
    del left_mat, right_mat
    
    # ------------------------------------------------------------------
    # 5️⃣ Row‑wise mean & rsqrt‑var over hidden dim H (torch reduction)
    # ------------------------------------------------------------------
    out_fp32 = out.float()
    mu = out_fp32.mean(dim=1)                               # [B, N, N] FP32
    var = out_fp32.var(dim=1, unbiased=False)               # [B, N, N] FP32
    rstd = torch.rsqrt(var + eps)                           # [B, N, N] FP32
    
    # ------------------------------------------------------------------
    # 6️⃣ Apply LN (γ,β), out‑gate and final projection (torch)
    # ------------------------------------------------------------------
    # broadcast μ and rstd to hidden dimension
    mu_exp = mu.unsqueeze(1)        # [B,1,N,N]
    rstd_exp = rstd.unsqueeze(1)    # [B,1,N,N]
    
    # LayerNorm over hidden dimension H
    gamma = weights["to_out_norm.weight"]   # [H]
    beta = weights["to_out_norm.bias"]     # [H]
    # reshape to broadcast
    gamma_view = gamma[None, :, None, None]   # [1,H,1,1]
    beta_view = beta[None, :, None, None]     # [1,H,1,1]
    
    # Normalise, scale, shift
    normed = ((out_fp32 - mu_exp) * rstd_exp) * gamma_view + beta_view   # [B,H,N,N] FP32
    
    # Multiply by out_gate (cast to FP32)
    gated = normed * out_gate.float()                                    # [B,H,N,N] FP32
    
    # Final linear projection: hidden → dim (weight shape [dim, H])
    w_out = weights["to_out.weight"].float()   # [dim, H]
    # reshape to (B,N,N,H) for torch.nn.functional.linear
    gated_reshaped = gated.permute(0, 2, 3, 1).contiguous()    # [B,N,N,H]
    final = F.linear(gated_reshaped.view(R, H), w_out)        # [R, dim] FP32
    final = final.view(B, N, N, dim)                         # [B,N,N,dim]
    return final

"""
Outgoing TriMul (AlphaFold‑3) – forward only.

Algorithm (mixed PyTorch/Triton)
---------------------------------
1️⃣  Input LayerNorm (FP32 → FP16) – uses torch.layer_norm (cuDNN).

    * computes the sigmoid gates,
    * writes three tensors `left`, `right`, `out_gate` in layout
      [B, H, N, N] (FP16).
4️⃣  Core einsum is a batched GEMM performed with torch.bmm:
    out[b,h,i,j] = Σₖ left[b,h,i,k]·right[b,h,j,k]   → [B,H,N,N] FP16.
5️⃣  Row‑wise (over H) mean / variance are computed with torch’s native
    reductions (fast on H100).
6️⃣  Final LayerNorm over H, out‑gate multiplication and the projection
    back to the original `dim` are carried out with pure PyTorch
    (FP32 arithmetic, weight in FP16).  All heavy arithmetic stays in
    FP16 where possible; reductions are performed in FP32 for stability.

The implementation works for any supported batch‑size, sequence‑length,
hidden‑dim and respects an optional mask.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: split the fused projection, apply mask and compute sigmoids
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,        # [R, 5·H]  FP16 fused projection
    left_ptr,        # [B, H, N, N] FP16
    right_ptr,       # [B, H, N, N] FP16
    out_gate_ptr,    # [B, H, N, N] FP16
    mask_ptr,        # [R] FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [R]
    mask_r = offs_r < R

    # decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask (broadcast over H)
    if HAS_MASK:
        mval = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [R]
        mval = mval[:, None]                                       # [R,1]
    else:
        mval = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N
    # start of the h‑plane = b·(H·NN) + i·N + k  (h = 0)
    base_offset = b * (H * hidden_stride) + i * N + k

    # iterate over hidden dimension in chunks of BLOCK_H
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # each projection occupies a contiguous chunk of size H
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [R, H]

        left_raw   = tl.load(proj_ptr + proj_base,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        right_raw  = tl.load(proj_ptr + proj_base + H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        left_g_raw = tl.load(proj_ptr + proj_base + 2 * H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        right_g_raw = tl.load(proj_ptr + proj_base + 3 * H,
                              mask=mask_r[:, None] & mask_h[None, :],
                              other=0.0)
        outg_raw   = tl.load(proj_ptr + proj_base + 4 * H,
                              mask=mask_r[:, None] & mask_h[None, :],
                              other=0.0)

        # sigmoid in FP32 → back to FP16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_g_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_g_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-outg_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mval * left_gate
        right = right_raw * mval * right_gate

        # destination offset for the dense tensors (B, H, N, N)
        dst_off = base_offset[:, None] + offs_h[None, :] * hidden_stride   # [R, H]

        tl.store(left_ptr + dst_off,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_off,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_off,
                 out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Triton‑based TriMul forward (no backward)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Arguments
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
        - weights_dict : keys
            "norm.weight", "norm.bias",
            "left_proj.weight", "right_proj.weight",
            "left_gate.weight", "right_gate.weight", "out_gate.weight",
            "to_out_norm.weight", "to_out_norm.bias",
            "to_out.weight"
            (all float32 tensors)
        - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns
    -------
    torch.Tensor : [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N               # total pairwise tokens
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm – use cuDNN implementation (fast, FP32→FP16)
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim)                              # [R, dim]  (float32)
    x_norm_fp32 = F.layer_norm(
        x_flat,
        normalized_shape=(dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    )
    x_norm = x_norm_fp32.half()                             # [R, dim]  (float16)
    del x_flat, x_norm_fp32

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection of the five linear maps (5·H × dim)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                     # [5·H, dim]
    proj = F.linear(x_norm, fused_w)                         # (R, 5·H)  FP16
    del x_norm, fused_w

    # ------------------------------------------------------------------
    # 3️⃣ Split / mask / sigmoid gates – Triton kernel
    # ------------------------------------------------------------------
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy

    BLOCK_R = 64   # rows per program
    BLOCK_H = 64   # hidden‑dim chunk
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Core einsum via batched GEMM (left·rightᵀ)
    # ------------------------------------------------------------------
    left_mat = left.view(B * H, N, N)                    # [B·H, N, N]  FP16
    right_mat = right.view(B * H, N, N)                  # [B·H, N, N]  FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [B·H, N, N]  FP16
    out = out.view(B, H, N, N)                          # [B, H, N, N] FP16
    del left_mat, right_mat

    # ------------------------------------------------------------------
    # 5️⃣ Row‑wise mean & rsqrt‑var over the hidden dimension (torch)
    # ------------------------------------------------------------------
    out_fp32 = out.float()
    mu = out_fp32.mean(dim=1)                                 # [B, N, N]  FP32
    var = out_fp32.var(dim=1, unbiased=False)                # [B, N, N]  FP32
    rstd = torch.rsqrt(var + eps)                             # [B, N, N]  FP32

    # ------------------------------------------------------------------
    # 6️⃣ Final LayerNorm (γ,β), out‑gate, and projection back to `dim`
    # ------------------------------------------------------------------
    # broadcast μ and rstd to hidden dimension
    mu_exp   = mu.unsqueeze(1)        # [B,1,N,N]
    rstd_exp = rstd.unsqueeze(1)      # [B,1,N,N]

    gamma = w["to_out_norm.weight"]   # [H]
    beta  = w["to_out_norm.bias"]     # [H]
    gamma_view = gamma[None, :, None, None]   # [1,H,1,1]
    beta_view  = beta[None, :, None, None]    # [1,H,1,1]

    # LayerNorm over H
    normed = ((out_fp32 - mu_exp) * rstd_exp) * gamma_view + beta_view   # [B,H,N,N] FP32

    # Multiply by the out‑gate (cast to FP32)
    gated = normed * out_gate.float()                                     # [B,H,N,N] FP32

    # Final linear projection: hidden → dim (weight shape [dim, H])
    w_out = w["to_out.weight"].float()                                    # [dim, H]
    # reshape to (B·N·N, H) for a single GEMM
    gated_perm = gated.permute(0, 2, 3, 1).contiguous()                   # [B,N,N,H]
    final = F.linear(gated_perm.view(R, H), w_out)                        # [R, dim]  FP32
    final = final.view(B, N, N, dim)                                      # [B,N,N,dim]
    return final
```
