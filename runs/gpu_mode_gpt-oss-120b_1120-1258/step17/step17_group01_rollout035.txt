@triton.jit
def layernorm_kernel(
    x_ptr, y_ptr,
    weight_ptr, bias_ptr,
    rows, dim,
    eps,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # accumulate sum and sum squares
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        beta = tl.load(bias_ptr + offs_d, mask=mask_d, other=0.0)

        out = norm * gamma[None, :] + beta[None, :]
        out_fp16 = out.to(tl.float16)

        dst_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(dst_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])
```

@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,       # const *[R, 5*H] fp16
    left_ptr,       # mutable *[B, H, N, N] fp16
    right_ptr,
    out_gate_ptr,
    mask_ptr,
    R: tl.int32,
    H: tl.int32,
    N: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < R

    # decode b,i,k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask per position
    if HAS_MASK:
        # load mask as scalar per row
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)
        # broadcast to H dimension later
    else:
        mask_val = tl.full([BLOCK_R], 1.0, dtype=tl.float16)

    # compute base offset for (b,0,i,k) in B,H,N,N layout
    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + k   # [BLOCK_R]

    # iterate over hidden dimension in chunks
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # compute pointer offset into proj: each row starts at offs_r * (5*H)
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]  # [BLOCK_R, BLOCK_H]

        # load raw slices
        left_raw = tl.load(proj_ptr + proj_base,
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0)
        right_raw = tl.load(proj_ptr + proj_base + H,
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0)
        left_gate_raw = tl.load(proj_ptr + proj_base + 2 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_gate_raw = tl.load(proj_ptr + proj_base + 3 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        out_gate_raw = tl.load(proj_ptr + proj_base + 4 * H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)

        # sigmoid in fp32
        left_gate = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # broadcast mask if needed
        if HAS_MASK:
            mask_vec = mask_val[:, None]    # [BLOCK_R,1] broadcast across H
        else:
            mask_vec = tl.full([BLOCK_R,1], 1.0, dtype=tl.float16)

        left = left_raw * mask_vec * left_gate
        right = right_raw * mask_vec * right_gate
        outg = out_gate  # not masked

        # destination offsets
        dst_off = base[:, None] + offs_h[None, :] * hidden_stride

        tl.store(left_ptr + dst_off,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_off,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_off,
                 outg,
                 mask=mask_r[:, None] & mask_h[None, :])
```

@triton.jit
def rowwise_stats_kernel(
    out_ptr,   # const *[B, H, N, N] fp16
    mu_ptr,    # mutable *[R] float32
    rstd_ptr,  # mutable *[R] float32
    total_rows: tl.int32,
    H: tl.int32,
    N: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + j   # offset of hidden 0

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H
        ptr = out_ptr + base[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr + offs_m, mean, mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)

    H = hidden_dim

    # -----------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 -> float16)
    # -----------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 256
    BLOCK_D = 64
    grid = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],
        weights["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # -----------------------------------------------------------------
    # 2️⃣ Fused projection (left/right + three gates)
    # -----------------------------------------------------------------
    fused_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)  # (5*H, dim)

    proj = F.linear(x_norm, fused_w)  # (R, 5*H) fp16
    del fused_w, x_norm

    # -----------------------------------------------------------------
    # 3️⃣ Split, apply mask, compute sigmoid gates
    # -----------------------------------------------------------------
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if cfg.get("nomask", True) or mask is None:
        has_mask = 0
        dummy_mask = torch.empty(0, dtype=torch.float16, device=device)
        mask_ptr = dummy_mask
    else:
        # flatten mask and cast to fp16 (True -> 1.0, False -> 0.0)
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat

    BLOCK_R = 128
    BLOCK_H = 32
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # -----------------------------------------------------------------
    # 4️⃣ Batched GEMM (left @ right^T)  -> out (B, H, N, N)
    # -----------------------------------------------------------------
    left_mat = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)

    out = torch.bmm(left_mat, right_mat.transpose(1, 2))
    out = out.view(B, H, N, N)

    # -----------------------------------------------------------------
    # 5️⃣ Row‑wise mean / rsqrt‑var of out (per position over H)
    # -----------------------------------------------------------------
    mu = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 256
    BLOCK_K_STATS = 64
    grid_stats = (triton.cdiv(R, BLOCK_M_STATS),)
    rowwise_stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=8,
    )

    # -----------------------------------------------------------------
    # 6️⃣ Second LayerNorm, out_gate, final projection (float32)
    # -----------------------------------------------------------------
    # Convert tensors to fp32 for the final stage
    out_fp32 = out.to(torch.float32)            # (B, H, N, N)
    out_gate_fp32 = out_gate.to(torch.float32)  # (B, H, N, N)

    # reshape mu/rstd to [B,1,N,N] for broadcasting
    mu_ = mu.view(B, N, N).unsqueeze(1)    # (B,1,N,N)
    rstd_ = rstd.view(B, N, N).unsqueeze(1)

    # Apply first part of LayerNorm (normalize)
    normed = (out_fp32 - mu_) * rstd_  # (B, H, N, N)

    # Apply γ/β of to_out_norm
    gamma = weights["to_out_norm.weight"]   # (H,)
    beta = weights["to_out_norm.bias"]     # (H,)
    normed = normed * gamma.view(1, -1, 1, 1) + beta.view(1, -1, 1, 1)

    # Multiply by out_gate
    gated = normed * out_gate_fp32

    # Final linear projection to original dimension
    # bring to shape (B,N,N,H) for linear
    gated = gated.permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)
    gated_flat = gated.view(R, H)                    # (R, H)

    final = F.linear(gated_flat, weights["to_out.weight"], bias=None)  # (R, dim)
    final = final.view(B, N, N, dim)  # (B,N,N,dim)

    return final

"""
TriMul (outgoing) – Triton‑accelerated forward pass
===================================================

The implementation follows the reference PyTorch module *exactly* but
moves the heaviest element‑wise work to Triton:

1️⃣  Row‑wise LayerNorm (float32 → float16) – the mean/variance are
    accumulated in fp32, the normalised values are written back as fp16.

2️⃣  A single fused linear projection produces the five matrices
    (left, right, left_gate, right_gate, out_gate).  This is a standard
    fp16 GEMM (`torch.nn.functional.linear`).

3️⃣  A Triton kernel splits the fused matrix, applies the optional
    pair‑wise mask, evaluates the three sigmoid‑gates and stores
    ``left``, ``right`` and ``out_gate`` in the layout
    ``[B, H, N, N]`` required for a batched GEMM.

4️⃣  Batched GEMM (`torch.bmm`) contracts over the pair‑wise dimension:
    ``out[b, h] = left[b, h] @ right[b, h]ᵀ`` → shape ``(B·H, N, N)`` in fp16.

5️⃣  Another Triton kernel computes per‑position mean and rsqrt‑variance
    of ``out`` across the hidden dimension `H`.  These statistics are needed
    for the second LayerNorm.

6️⃣  The remaining steps (second LayerNorm, out‑gate multiplication and the
    final linear projection back to ``dim``) are performed in PyTorch.
    The final tensor has dtype ``float32`` and shape ``[B, N, N, dim]``.

All kernels run with FP16 arithmetic and FP32 accumulation and work for
any configuration in the test set (batch size 1‑2, sequence length up to
1024, hidden dimension 128, and optional mask).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (float32 → float16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const *[rows, dim] (float32)
    y_ptr,          # mutable *[rows, dim] (float16)
    weight_ptr,     # const *[dim] (float32) – γ
    bias_ptr,       # const *[dim] (float32) – β
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ---------- mean / variance (fp32) ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # [BLOCK_M, BLOCK_D]
        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- normalise, apply γ/β and store as fp16 ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        beta  = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β

        out = norm * gamma[None, :] + beta[None, :]                    # fp32
        out_fp16 = out.to(tl.float16)

        dst_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(dst_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, apply mask & sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,       # const *[R, 5*H] (fp16) – fused projection
    left_ptr,       # mutable *[B, H, N, N] (fp16)
    right_ptr,      # mutable *[B, H, N, N] (fp16)
    out_gate_ptr,   # mutable *[B, H, N, N] (fp16)
    mask_ptr,       # const *[R] (fp16) – optional mask
    R: tl.int32,
    H: tl.int32,
    N: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)               # [BLOCK_R]
    mask_r = offs_r < R

    # decode (b,i,k) from linear index
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional per‑pair mask (scalar, broadcast over H)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
    else:
        mask_val = tl.full([BLOCK_R], 1.0, dtype=tl.float16)          # [BLOCK_R]

    # layout stride for hidden dimension inside [B, H, N, N]
    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + k               # [BLOCK_R]

    # process hidden dimension in tiles
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)      # [BLOCK_H]
        mask_h = offs_h < H

        # start of the row (R) inside the fused matrix
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw      = tl.load(proj_ptr + proj_base,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        right_raw     = tl.load(proj_ptr + proj_base + H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        left_gate_raw = tl.load(proj_ptr + proj_base + 2 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        right_gate_raw= tl.load(proj_ptr + proj_base + 3 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        out_gate_raw  = tl.load(proj_ptr + proj_base + 4 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)

        # sigmoid (stable in fp32, result cast back to fp16)
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        mask_vec = mask_val[:, None]    # [BLOCK_R, 1]

        left  = left_raw  * mask_vec * left_gate
        right = right_raw * mask_vec * right_gate
        outg  = out_gate                         # not masked

        dst_off = base[:, None] + offs_h[None, :] * hidden_stride

        tl.store(left_ptr    + dst_off, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr   + dst_off, right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr+ dst_off, outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 3️⃣ Row‑wise mean & rsqrt‑variance of the contracted tensor
# ----------------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr,        # const *[B, H, N, N] (fp16)
    mu_ptr,         # mutable *[R] (fp32)
    rstd_ptr,       # mutable *[R] (fp32)
    total_rows: tl.int32,   # = B * N * N
    H: tl.int32,
    N: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + j          # offset of hidden‑plane 0

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H
        ptr = out_ptr + base[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)            # [M, K]
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_m, mean,  mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd,  mask=mask_m)


# ----------------------------------------------------------------------
# Custom forward for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : mapping with the nine weight tensors used by the reference module
        - config_dict  : must contain "dim", "hidden_dim" and "nomask" (bool)

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    x, mask, w, cfg = data
    dim        = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps        = 1e-5
    device     = x.device

    B, N, _, _ = x.shape
    R = B * N * N               # total number of (b,i,k) positions
    H = hidden_dim              # == hidden_dim in the reference module

    # -----------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 → float16)
    # -----------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()                     # [R, dim] float32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 256
    BLOCK_D = 64
    grid = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        w["norm.weight"],   # γ (float32)
        w["norm.bias"],     # β (float32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # -----------------------------------------------------------------
    # 2️⃣ Fused projection (left/right + three gates)
    # -----------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                      # (5*H, dim)

    proj = F.linear(x_norm, fused_w)                         # (R, 5*H) fp16
    del fused_w, x_norm

    # -----------------------------------------------------------------
    # 3️⃣ Split, apply mask, compute sigmoid gates
    # -----------------------------------------------------------------
    left     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right    = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if cfg.get("nomask", True) or mask is None:
        has_mask = 0
        dummy_mask = torch.empty(0, dtype=torch.float16, device=device)
        mask_ptr = dummy_mask
    else:
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat

    BLOCK_R = 128
    BLOCK_H = 32
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # -----------------------------------------------------------------
    # 4️⃣ Batched GEMM (left @ rightᵀ)  → out (B, H, N, N)
    # -----------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)

    out = torch.bmm(left_mat, right_mat.transpose(1, 2))      # fp16, shape (B·H, N, N)
    out = out.view(B, H, N, N)                               # back to (B, H, N, N)

    # -----------------------------------------------------------------
    # 5️⃣ Row‑wise mean / rsqrt‑var of `out` (across H)
    # -----------------------------------------------------------------
    mu   = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 256
    BLOCK_K_STATS = 64
    grid_stats = (triton.cdiv(R, BLOCK_M_STATS),)
    rowwise_stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=8,
    )

    # -----------------------------------------------------------------
    # 6️⃣ Second LayerNorm, out‑gate, final projection
    # -----------------------------------------------------------------
    # Convert to fp32 for the final stage
    out_fp32      = out.to(torch.float32)          # (B, H, N, N)
    out_gate_fp32 = out_gate.to(torch.float32)    # (B, H, N, N)

    mu_   = mu.view(B, N, N).unsqueeze(1)         # (B, 1, N, N)
    rstd_ = rstd.view(B, N, N).unsqueeze(1)       # (B, 1, N, N)

    # Normalise and apply γ / β of `to_out_norm`
    normed = (out_fp32 - mu_) * rstd_
    gamma = w["to_out_norm.weight"]               # (H,)
    beta  = w["to_out_norm.bias"]                 # (H,)
    normed = normed * gamma.view(1, -1, 1, 1) + beta.view(1, -1, 1, 1)

    # Multiply by out_gate
    gated = normed * out_gate_fp32                # (B, H, N, N)

    # Final linear projection back to `dim`
    gated = gated.permute(0, 2, 3, 1).contiguous()   # (B, N, N, H)
    gated_flat = gated.view(R, H)                     # (R, H)

    final = F.linear(gated_flat, w["to_out.weight"], bias=None)  # (R, dim)
    final = final.view(B, N, N, dim)                         # (B, N, N, dim)

    return final
```
