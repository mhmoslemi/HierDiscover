"""
Outgoing TriMul (AlphaFold‑3) forward pass – fast Triton‑accelerated implementation.

Algorithm
---------
1️⃣ Row‑wise LayerNorm over the input channel dimension `dim` (FP32 → FP16).
2️⃣ Fuse the five Linear maps (left, right, left_gate, right_gate, out_gate)
   into a single GEMM `proj = x_norm @ W_fused.T` (FP16).
3️⃣ A Triton kernel splits the fused projection, applies the optional pair‑wise
   mask, evaluates the sigmoid gates and writes three tensors
   `left`, `right` and `out_gate` in layout `[B, H, N, N]` (FP16).
4️⃣ Batched GEMM (cuBLAS) computes the core einsum:
        out[b, h, i, j] = Σₖ left[b, h, i, k] * right[b, h, j, k]
   producing `[B, H, N, N]` (FP16).
5️⃣ Row‑wise LayerNorm (mean & rsqrt‑variance) over the hidden dimension `H`,
   with learned weight & bias (`to_out_norm`).  This is a second launch of the
   same LayerNorm kernel from step 1.
6️⃣ Element‑wise multiplication with `out_gate`.
7️⃣ Final linear projection `to_out` is performed with a single cuBLAS call:
        out_final = (out_norm * out_gate) @ W_out.T   (FP16 → FP32).

Only the two reduction kernels (layer‑norm & split‑&‑gate) are written in Triton;
all dense matrix multiplications use cuBLAS which is optimal on H100.
The implementation works for any `B∈{1,2}`, `N∈{128,256,512,1024}`,
`dim∈{128,384}` and `hidden_dim≤768`.  The final tensor is returned as
`torch.float32` to match the reference PyTorch module.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm kernel (works for any `dim`)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,        # [rows, dim]   input (FP32 for the first LN, FP16 for the second)
    y_ptr,        # [rows, dim]   output (FP16)
    w_ptr,        # [dim]         γ (FP32)
    b_ptr,        # [dim]         β (FP32)
    rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)         # [BLOCK_M]
    mask_m = offs_m < rows

    # ------------------------------------------------------------------
    # First pass – compute mean & variance per row
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)            # [BLOCK_D]
        mask_d = offs_d < dim

        # Load a tile of the row (as FP32 regardless of the source dtype)
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------------------------------------------------------
    # Second pass – write normalized values with γ/β
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)  # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)  # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, apply optional mask & sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # [R, 5*H]  fused projection (FP16)
    left_ptr,          # [B, H, N, N]   left    (FP16)
    right_ptr,         # [B, H, N, N]   right   (FP16)
    out_gate_ptr,      # [B, H, N, N]   out_gate(FP16)
    mask_ptr,          # [R] optional mask (FP16) – scalar per (b,i,k)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)      # [BLOCK_R]
    mask_r = offs_r < R

    # Decode flat index r -> (b, i, k)
    nn = N * N
    b   = offs_r // nn
    rem = offs_r - b * nn
    i   = rem // N
    k   = rem - i * N

    # Load optional mask (shape [R]) – broadcast along hidden dim later
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                   # [BLOCK_R, 1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # Destination stride for a single hidden plane
    hidden_stride = N * N
    # Base offset for h = 0 in the destination tensors
    base_offset = b * (H * hidden_stride) + i * N + k   # [BLOCK_R]

    # Process hidden channels in tiles
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # Layout of proj: [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [R, H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # Sigmoid in FP32 → back to FP16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # Apply mask and gates
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate

        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride
        tl.store(left_ptr + dst_offset,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset,
                 outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Entry point – outgoing TriMul forward (no backward)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradient).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim]      (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : dict with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
              (all float32)
            - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                     # total pairwise positions
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm over the input channels (FP32 → FP16)
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()          # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 256
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear projection: 5 * hidden_dim outputs (FP16)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                 # [5*H, dim]

    proj = F.linear(x_norm, fused_w)    # (R, 5*H) FP16
    del fused_w, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ Split projection, apply optional mask & sigmoid gates
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy

    BLOCK_R = 128
    BLOCK_H = 32
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension (cuBLAS)
    # ------------------------------------------------------------------
    # left/right layout -> (B*H, N, N)
    left_mat  = left.view(B * H, N, N)                     # [B*H, N, N] FP16
    right_mat = right.view(B * H, N, N)                    # [B*H, N, N] FP16

    # out = left @ right^T for each (b,h) pair
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B*H, N, N] FP16
    out = out.view(B, H, N, N)                             # [B, H, N, N] FP16

    # ------------------------------------------------------------------
    # 5️⃣ Row‑wise LayerNorm on the hidden dimension (FP16 → FP16)
    # ------------------------------------------------------------------
    # Permute to (B, N, N, H) -> flatten rows = R, dim = H
    out_perm = out.permute(0, 2, 3, 1).contiguous()       # [B, N, N, H]
    out_flat = out_perm.view(R, H)                        # [R, H] FP16
    out_norm_flat = torch.empty_like(out_flat)            # [R, H] FP16

    # Re‑use the same layernorm kernel (weights are to_out_norm)
    BLOCK_M2 = 256
    BLOCK_D2 = 64
    grid_ln2 = (triton.cdiv(R, BLOCK_M2),)
    layernorm_kernel[grid_ln2](
        out_flat,
        out_norm_flat,
        w["to_out_norm.weight"],   # γ (FP32)
        w["to_out_norm.bias"],     # β (FP32)
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M2,
        BLOCK_D=BLOCK_D2,
        num_warps=8,
    )
    # Restore the (B, H, N, N) layout
    out_norm = out_norm_flat.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    del out_perm, out_flat, out_norm_flat

    # ------------------------------------------------------------------
    # 6️⃣ Multiply by the out‑gate (FP16)
    # ------------------------------------------------------------------
    gated = out_norm * out_gate    # [B, H, N, N] FP16
    del out_norm, out_gate

    # ------------------------------------------------------------------
    # 7️⃣ Final linear projection (FP16 → FP32)
    # ------------------------------------------------------------------
    # Flatten to (R, H) for the final matmul
    gated_perm = gated.permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]
    gated_flat = gated_perm.view(R, H)                     # [R, H] FP16

    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()  # [H, dim] FP16
    final_flat = F.linear(gated_flat, w_out_T)                       # [R, dim] FP16
    final = final_flat.view(B, N, N, dim).to(torch.float32)          # FP32

    return final
