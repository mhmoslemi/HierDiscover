def custom_kernel(data):
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device
    B, N, _, _ = x.shape
    R = B * N * N
    H = hidden_dim
    # 1. LayerNorm
    x_norm = torch.nn.functional.layer_norm(x, (dim,), w["norm.weight"], w["norm.bias"])
    x_norm = x_norm.to(torch.float16)
    # 2. fused linear projection
    fused_w = torch.cat([w["left_proj.weight"],
                         w["right_proj.weight"],
                         w["left_gate.weight"],
                         w["right_gate.weight"],
                         w["out_gate.weight"]], dim=0).to(torch.float16)
    proj = torch.nn.functional.linear(x_norm, fused_w)  # (R,5*H)
    # 3. split + gate (Triton)
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    HAS_MASK = 0 if nomask else 1
    if HAS_MASK:
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        mask_ptr = mask_flat
    else:
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)
    BLOCK_R = 64
    BLOCK_H = 64
    split_fuse_gate_kernel[(triton.cdiv(R, BLOCK_R),)](
        proj, left, right, out_gate, mask_ptr,
        R, H, N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R, BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    # free proj
    del proj
    # 4. batched GEMM
    left_mat = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H,N,N) fp16
    out = out.view(B, H, N, N)
    # 5. rowwise LN (hidden dim)
    mu = out.mean(dim=1, keepdim=True, dtype=torch.float32)  # (B,1,N,N)
    var = out.var(dim=1, unbiased=False, keepdim=True, dtype=torch.float32)
    rstd = torch.rsqrt(var + eps)
    # Normalize
    out_f = out.float()
    norm = (out_f - mu) * rstd
    # Apply second LN weight/bias
    ln_w = w["to_out_norm.weight"].view(1, H, 1, 1)
    ln_b = w["to_out_norm.bias"].view(1, H, 1, 1)
    norm = norm * ln_w + ln_b
    # Multiply gate (out_gate)
    gated = (norm.to(torch.float16) * out_gate)  # (B,H,N,N) half
    # Final projection

    final = final.to(torch.float32).view(B, N, N, dim)
    return final

"""
Outgoing TriMul (AlphaFold‑3) – fast forward‑only Triton implementation.

Algorithm
---------
1️⃣  LayerNorm (FP32 → FP16) on the pairwise input using the highly‑optimised
    PyTorch `torch.nn.functional.layer_norm` (no custom kernel needed).

2️⃣  Fuse the five linear maps (left, right, left_gate, right_gate, out_gate)
    into a single half‑precision GEMM.

3️⃣  Split the fused projection, apply the optional pair‑wise mask and the
    sigmoid gates **inside a Triton kernel** (the only Triton kernel we keep).

4️⃣  Batched GEMM (cuBLAS) computes the core einsum:
        out[b, h, i, j] = Σₖ left[b, h, i, k] · right[b, h, j, k]

5️⃣  Row‑wise statistics (mean / rsqrt‑variance) over the hidden dimension `H`
    are obtained with a single call to `torch.mean` / `torch.var` (FP32).

6️⃣  Apply the second LayerNorm, multiply by `out_gate` and a final
    projection back to `dim`.  All of these steps are done with
    PyTorch tensor ops – the only custom kernel is the split/gate kernel.

Result
-------
`custom_kernel` returns a tensor of shape `[B, N, N, dim]` (float32) and
produces the same numerical results as the reference PyTorch implementation
while being considerably faster on an NVIDIA H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 3️⃣ Split fused projection, apply mask & sigmoid gates (Triton)
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,        # [R, 5·H]   FP16 fused projection
    left_ptr,        # [B, H, N, N] FP16
    right_ptr,       # [B, H, N, N] FP16
    out_gate_ptr,    # [B, H, N, N] FP16
    mask_ptr,        # [R]        FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)      # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index r = b·N·N + i·N + k
    nn = N * N
    b   = offs_r // nn
    rem = offs_r - b * nn
    i   = rem // N
    k   = rem - i * N

    # optional per‑pair mask (broadcast over hidden dim)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # FP16
        mask_val = mask_val[:, None]                                   # [BLOCK_R, 1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # stride between consecutive hidden planes in the destination tensors
    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + k   # offset of h = 0

    # iterate over hidden dimension in tiles
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # view of the fused projection: [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]          # [R, H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # sigmoid (computed in FP32 → back to FP16)
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate

        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride
        tl.store(
            left_ptr + dst_offset,
            left,
            mask=mask_r[:, None] & mask_h[None, :],
        )
        tl.store(
            right_ptr + dst_offset,
            right,
            mask=mask_r[:, None] & mask_h[None, :],
        )
        tl.store(
            out_gate_ptr + dst_offset,
            outg,
            mask=mask_r[:, None] & mask_h[None, :],
        )


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only implementation of the outgoing TriMul operator.

    Args:
        data (tuple):
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)
    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim        = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask     = cfg.get("nomask", True)
    eps        = 1e-5
    device     = x.device

    B, N, _, _ = x.shape
    R = B * N * N                     # total number of pair positions
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣  LayerNorm (FP32 -> FP16) ------------------------------------
    # ------------------------------------------------------------------
    # torch.nn.functional.layer_norm is highly optimised on H100
    x_norm = F.layer_norm(
        x,
        (dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    ).to(torch.float16)             # [B, N, N, dim]  FP16

    # ------------------------------------------------------------------
    # 2️⃣  Fused linear projection (5·H) ---------------------------------
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)             # [5·H, dim]  FP16

    proj = F.linear(x_norm, fused_w)   # (R, 5·H)  FP16
    del fused_w, x_norm

    # ------------------------------------------------------------------
    # 3️⃣  Split, optional mask, sigmoid gates (Triton) -----------------
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    HAS_MASK = 0
    if not nomask and mask is not None:
        # mask is broadcast over the hidden dimension
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        # dummy pointer – never read when HAS_MASK == 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣  Batched GEMM (cuBLAS) ---------------------------------------
    # ------------------------------------------------------------------
    # shape: [B·H, N, N]  (fp16)
    left_mat  = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)

    # out[b, h, i, j] = Σₖ left[b, h, i, k] • right[b, h, j, k]
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))
    out = out.view(B, H, N, N)          # [B, H, N, N]  fp16

    # ------------------------------------------------------------------
    # 5️⃣  Row‑wise LayerNorm over the hidden dimension -----------------
    # ------------------------------------------------------------------
    # mean & variance are computed in float32 for numerical stability
    mu   = out.mean(dim=1, keepdim=True, dtype=torch.float32)            # [B,1,N,N]
    var  = out.var(dim=1, unbiased=False, keepdim=True, dtype=torch.float32)
    rstd = torch.rsqrt(var + eps)                                        # [B,1,N,N]

    out_fp32 = out.float()
    norm = (out_fp32 - mu) * rstd                                          # [B,H,N,N] fp32

    # apply the second LayerNorm parameters (γ, β)
    ln_w = w["to_out_norm.weight"].view(1, H, 1, 1)                        # fp32
    ln_b = w["to_out_norm.bias"].view(1, H, 1, 1)                          # fp32
    norm = norm * ln_w + ln_b                                             # fp32

    # ------------------------------------------------------------------
    # 6️⃣  Multiply by out_gate and final projection --------------------
    # ------------------------------------------------------------------
    gated = (norm.to(torch.float16) * out_gate)                           # fp16
    gated_flat = gated.view(R, H)                                          # [R, H] fp16

    # final linear projection (half‑precision weight for speed)
    w_out = w["to_out.weight"].to(torch.float16)                           # [dim, H]
    final = F.linear(gated_flat, w_out)                                    # [R, dim] fp16
    final = final.to(torch.float32).view(B, N, N, dim)                     # fp32

    return final
