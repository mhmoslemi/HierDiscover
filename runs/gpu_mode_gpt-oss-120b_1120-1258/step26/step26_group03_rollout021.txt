"""
Outgoing TriMul (AlphaFold3) – optimized forward‑only implementation.

Key ideas
---------
1️⃣  Fuse the five linear projections into a single FP16 GEMM.
2️⃣  Split the fused projection and apply the mask/gates in a single Triton kernel.
3️⃣  Use a single batched GEMM (cuBLAS) to contract the pairwise tensors.
4️⃣  Fuse the row‑wise statistics (mean/variance) and the second LayerNorm,
    gating and final projection into ONE Triton kernel.  This removes the
    previous 2‑pass scan over the large `[B,H,N,N]` tensor and cuts the
    memory traffic in half.

The kernel computes for every pair (b,i,j) the three accumulators

    A = Σₕ outₕ·γₕ·gₕ·Wₕ      (out after the first matmul)
    B = Σₕ γₕ·gₕ·Wₕ
    C = Σₕ βₕ·gₕ·Wₕ

while simultaneously accumulating the raw sums needed for the mean and
variance of `out`.  The final result is

    μ   = Σₕ outₕ / H
    σ² = Σₕ outₕ² / H – μ²
    ρ   = rsqrt(σ² + eps)

    out = ρ·(A – μ·B) + C                (shape [B,N,N,dim])

All heavy work is done in FP16/FP32 fused kernels; the final tensor is
returned as float32.  The implementation works for all benchmark shapes
listed in the problem statement.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr, y_ptr, w_ptr, b_ptr, rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # ---- first pass: mean & variance ----
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- second pass: write normalized values ----
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2️⃣ Split the fused projection, apply mask & sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr, left_ptr, right_ptr, out_gate_ptr, mask_ptr,
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < R

    # decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)[:, None]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N                # stride between hidden slices
    # destination offset for h == 0  (layout [B,H,N,N])
    base_dest = b * (H * hidden_stride) + i * N + k

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]

        left_raw = tl.load(proj_ptr + proj_base,
                           mask=mask_r[:, None] & mask_h[None, :], other=0.0)
        right_raw = tl.load(proj_ptr + proj_base + H,
                            mask=mask_r[:, None] & mask_h[None, :], other=0.0)
        left_gate_raw = tl.load(proj_ptr + proj_base + 2 * H,
                                mask=mask_r[:, None] & mask_h[None, :], other=0.0)
        right_gate_raw = tl.load(proj_ptr + proj_base + 3 * H,
                                 mask=mask_r[:, None] & mask_h[None, :], other=0.0)
        out_gate_raw = tl.load(proj_ptr + proj_base + 4 * H,
                               mask=mask_r[:, None] & mask_h[None, :], other=0.0)

        # sigmoid in FP32 → back to FP16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left   = left_raw  * mask_val * left_gate
        right  = right_raw * mask_val * right_gate
        outg   = out_gate

        dst_offset = base_dest[:, None] + offs_h[None, :] * hidden_stride   # [R,H]

        tl.store(left_ptr + dst_offset,
                 left, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset,
                 right, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset,
                 outg, mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 3️⃣ Fuse statistics, second LayerNorm, gating and final projection
# ----------------------------------------------------------------------
@triton.jit
def final_fused_kernel(
    out_ptr,                 # [B, H, N, N] FP16        (contracted tensor)
    gate_ptr,                # [B, H, N, N] FP16        (out_gate)
    ln_w_ptr,                # [H] FP32 (to_out_norm.weight)
    ln_b_ptr,                # [H] FP32 (to_out_norm.bias)
    w_out_T_ptr,             # [H, dim] FP16 (transposed to_out weight)
    out_final_ptr,           # [B, N, N, dim] FP32
    total_rows, H, N, dim,
    eps,
    BLOCK_M: tl.constexpr,  # rows per program (tiles over R)
    BLOCK_K: tl.constexpr,  # hidden dimension chunk (covers H)
    BLOCK_N: tl.constexpr,  # output‑dim chunk
):
    """
    Compute for each (b,i,j):
        μ   = Σₕ outₕ / H
        σ² = Σₕ outₕ² / H – μ²
        ρ   = rsqrt(σ² + eps)

        A = Σₕ outₕ·γₕ·gₕ·Wₕ   (dim‑vector)
        B = Σₕ γₕ·gₕ·Wₕ
        C = Σₕ βₕ·gₕ·Wₕ

        out = ρ·(A – μ·B) + C
    """
    pid_m = tl.program_id(0)      # rows
    pid_n = tl.program_id(1)      # dim tiles

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)
    offs_n = col_start + tl.arange(0, BLOCK_N)

    mask_m = offs_m < total_rows
    mask_n = offs_n < dim

    # decode (b,i,j) from linear row index
    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + j      # start of hidden‑plane 0

    # per‑row statistics
    sum_out = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    # three accumulators for the final projection
    acc_A = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)
    acc_B = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)
    acc_C = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        # ------------------- load out & gate -------------------
        out_tile = tl.load(
            out_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)                     # [M, K]

        gate_tile = tl.load(
            gate_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)                     # [M, K]

        # accumulate raw statistics
        sum_out += tl.sum(out_tile, axis=1)                     # Σₕ outₕ
        sum_sq  += tl.sum(out_tile * out_tile, axis=1)         # Σₕ outₕ²

        # ------------------- LayerNorm params -------------------
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [K]
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [K]

        # common factor gate*γ  and gate*β  (both FP32)
        gate_gamma = gate_tile * gamma[None, :]               # [M, K]
        gate_beta  = gate_tile * beta[None, :]                # [M, K]

        # ---- A term: out·gate·γ ----------
        a_input = out_tile * gate_gamma                       # [M, K]
        a_fp16  = a_input.to(tl.float16)

        # ---- B term: gate·γ ----------
        b_fp16 = gate_gamma.to(tl.float16)

        # ---- C term: gate·β ----------
        c_fp16 = gate_beta.to(tl.float16)

        # load slice of the final projection weight (H, dim) – already transposed
        w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]
        w_tile = tl.load(
            w_ptr,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )                                                   # [K, TILE_N]

        # accumulate three dot‑products
        acc_A += tl.dot(a_fp16, w_tile).to(tl.float32)
        acc_B += tl.dot(b_fp16, w_tile).to(tl.float32)
        acc_C += tl.dot(c_fp16, w_tile).to(tl.float32)

    # ----- compute mean / rsqrt‑variance (per row) -----
    mu   = sum_out / H
    var  = sum_sq / H - mu * mu
    rstd = tl.rsqrt(var + eps)                               # [M]

    # ----- final formula: ρ·(A – μ·B) + C -----
    final_tile = rstd[:, None] * (acc_A - mu[:, None] * acc_B) + acc_C

    # write result in layout [B, N, N, dim] (contiguous)
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, final_tile,
             mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradient).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"   (all float32)
            - config_dict must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                      # number of (b,i,j) rows
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm (FP32 → FP16)
    # --------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()               # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # --------------------------------------------------------------
    # 2️⃣ Fused projection (5·hidden_dim)
    # --------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                 # [5·H, dim]

    proj = F.linear(x_norm, fused_w)                    # (R, 5·H) FP16
    del fused_w, x_norm

    # --------------------------------------------------------------
    # 3️⃣ Split, mask & sigmoid gates → left/right/out_gate
    # --------------------------------------------------------------
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension (cuBLAS)
    # --------------------------------------------------------------
    # reinterpret as [B·H, N, N] for torch.bmm
    left_mat  = left.view(B * H, N, N)                      # [B·H, N, N] FP16
    right_mat = right.view(B * H, N, N)                     # [B·H, N, N] FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                              # [B, H, N, N] FP16

    # --------------------------------------------------------------
    # 5️⃣ Final fused kernel (stats + LN + gating + projection)
    # --------------------------------------------------------------
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim]

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    # tune block sizes – BLOCK_K covers the whole hidden dimension (≤128)
    BLOCK_M_OUT = 64          # rows per program (tune for register pressure)
    BLOCK_K_OUT = 128         # hidden‑dim chunk (covers full H)
    BLOCK_N_OUT = 128         # output‑dim chunk

    grid_out = (
        triton.cdiv(R, BLOCK_M_OUT),
        triton.cdiv(dim, BLOCK_N_OUT),
    )
    final_fused_kernel[grid_out](
        out,
        out_gate,
        w["to_out_norm.weight"],    # γ (FP32)
        w["to_out_norm.bias"],      # β (FP32)
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=8,
    )
    # ------------------------------------------------------------------
    # Return result
    # ------------------------------------------------------------------
    return final
```
