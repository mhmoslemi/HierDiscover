col_start = pid_n * BLOCK_N

mask_n = offs_n < dim

rem = offs_m - b * NN
i = rem // N
j = rem - i * N

# For each row we compute sum and sumsq across hidden dimension

sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

# Allocate shared memory for out and gate
sh_out = tl.alloc([BLOCK_M, H], dtype=tl.float16)
sh_gate = tl.alloc([BLOCK_M, H], dtype=tl.float16)

# First pass: load out and gate into shared memory and accumulate sums
for h_start in range(0, H, BLOCK_K):
    offs_h = h_start + tl.arange(0, BLOCK_K)
    mask_h = offs_h < H
    
    # Load out values from global memory
    out_vals = tl.load(
        out_ptr + base_offset[:, None] + offs_h[None, :] * hidden_stride,
        mask=mask_m[:, None] & mask_h[None, :],
        other=0.0,
    ) # shape [BLOCK_M, BLOCK_K]
    
    gate_vals = tl.load(
        out_gate_ptr + base_offset[:, None] + offs_h[None, :] * hidden_stride,
        mask=mask_m[:, None] & mask_h[None, :],
        other=0.0,
    )
    
    # Store to shared memory
    # compute shared pointer offsets
    smem_out_ptr = sh_out + offs_m[:, None] * H + offs_h[None, :]
    smem_gate_ptr = sh_gate + offs_m[:, None] * H + offs_h[None, :]
    tl.store(smem_out_ptr, out_vals, mask=mask_m[:, None] & mask_h[None, :])
    tl.store(smem_gate_ptr, gate_vals, mask=mask_m[:, None] & mask_h[None, :])
    
    # Accumulate sum and sumsq
    sum_val += tl.sum(out_vals, axis=1)
    sum_sq += tl.sum(out_vals * out_vals, axis=1)

var = sum_sq / H - mean * mean
rstd = tl.rsqrt(var + eps)

# Initialize accumulator for final output

# Second pass: compute LN, gate, dot with weight

    mask_h = offs_h < H

    smem_out_ptr = sh_out + offs_m[:, None] * H + offs_h[None, :]
    out_vals = tl.load(smem_out_ptr, mask=mask_m[:, None] & mask_h[None, :], other=0.0)

    smem_gate_ptr = sh_gate + offs_m[:, None] * H + offs_h[None, :]
    gate_vals = tl.load(smem_gate_ptr, mask=mask_m[:, None] & mask_h[None, :], other=0.0)

    gamma = tl.load(ln_w_ptr + offs_h, mask=mask_h, other=0.0) # fp32
    beta = tl.load(ln_b_ptr + offs_h, mask=mask_h, other=0.0)   # fp32

    # mean, rstd shape [BLOCK_M], broadcast to [BLOCK_M, BLOCK_K]
    norm = ((out_vals - mean[:, None]) * rstd[:, None]) * gamma[None, :] + beta[None, :]

    # Apply out gate

    # Dot product: (BLOCK_M, BLOCK_K) dot (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)

# Store final result to output

tl.store(dst_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])

"""
Outgoing TriMul (AlphaFold‑3) – forward‑only implementation.

Algorithm
---------
1️⃣  LayerNorm over the last (dim) axis of the flattened pair representation.
2️⃣  Fuse the five linear maps (left, right, left_gate, right_gate, out_gate)
    into a single GEMM (cuBLAS) → tensor (R, 5·H) in fp16, where
    R = B·N·N and H = hidden_dim.
3️⃣  Split the fused projection, apply the (optional) mask, compute sigmoid
    gates and store the three tensors *left*, *right* and *out_gate*
    in layout [B, H, N, N] (fp16) via a Triton kernel.
4️⃣  Batched GEMM (cuBLAS) computes the core einsum:
       out[b, h, i, j] = Σₖ left[b, h, i, k]·right[b, h, j, k]
    → tensor `[B, H, N, N]` (fp16).
5️⃣  Fuse the second LayerNorm, the out‑gate multiplication and the final
    projection into a single Triton kernel.  The kernel
    *stores* the input tensor `out` and `out_gate` into shared memory,
    computes per‑row mean / variance (over H), applies the hidden‑dim
    LayerNorm, multiplies by `out_gate`, and finally performs the
    hidden‑dim × dim matrix multiplication (`to_out` weight) to produce
    the final `[B, N, N, dim]` tensor (fp32).

All kernels target Triton 3.3.1 on an NVIDIA H100 and use mixed precision.
"""

# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16) over the last dimension (dim)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,    # [R, dim] float32
    y_ptr,    # [R, dim] float16 output
    w_ptr,    # [dim]   LN weight (γ) float32
    b_ptr,    # [dim]   LN bias   (β) float32
    rows, dim,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)            # [BLOCK_M]
    mask_m = offs_m < rows

    # ------------------------------------------------------------------
    # First pass – compute mean / variance
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                                 # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------------------------------------------------------
    # Second pass – write normalized values (γ·x̂ + β) as fp16
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                                 # [BLOCK_M, BLOCK_D]

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # fp32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # fp32

        out = norm * gamma[None, :] + beta[None, :]               # fp32
        tl.store(
            y_ptr + offs_m[:, None] * dim + offs_d[None, :],
            out.to(tl.float16),
            mask=mask_m[:, None] & mask_d[None, :],
        )


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, apply mask and sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,        # [R, 5·H] fp16 fused projection
    left_ptr,        # [B, H, N, N] fp16
    right_ptr,       # [B, H, N, N] fp16
    out_gate_ptr,    # [B, H, N, N] fp16
    mask_ptr,        # [R] fp16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index r = b·N·N + i·N + k
    NN = N * N
    b = offs_r // NN
    rem = offs_r - b * NN
    i = rem // N
    k = rem - i * N

    # optional mask (broadcast over hidden dim)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                   # [BLOCK_R, 1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N
    # base offset for first hidden slice (h = 0)
    base_offset = b * (H * hidden_stride) + i * N + k   # [BLOCK_R]

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # Gather the 5·H fused view
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # Sigmoid (computed in fp32 for stability)
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate

        # Destination offset in the flattened [B, H, N, N] view.
        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst_offset, left,  mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset, right, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset, outg, mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 5️⃣ Fuse hidden‑dim LayerNorm, out‑gate multiplication and the final
#     projection into a single kernel (memory‑efficient).
# ----------------------------------------------------------------------
@triton.jit
def fused_output_kernel(
    out_ptr,        # [B, H, N, N] fp16
    gate_ptr,       # [B, H, N, N] fp16   (out_gate)
    ln_w_ptr,       # [H] fp32 (to_out_norm.weight)
    ln_b_ptr,       # [H] fp32 (to_out_norm.bias)
    w_out_T_ptr,    # [H, dim] fp16 (transposed to_out weight)
    out_final_ptr,  # [B, N, N, dim] fp32
    total_rows,     # R = B·N·N   (int)
    N, dim,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr,
    H: tl.constexpr,
):
    """
    For each flattened pair index r = b·N·N + i·N + j:
    1. Load the hidden‑dim vector out[r, :] and out_gate[r, :] into shared memory.
    2. Compute mean / variance over the hidden axis (row‑wise).
    3. Apply hidden‑dim LayerNorm, multiply by out_gate,
       and accumulate the product with the final weight matrix (H × dim).
    """
    pid_m = tl.program_id(0)   # rows dimension
    pid_n = tl.program_id(1)   # output‑dim tiles

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)   # [BLOCK_M]
    offs_n = col_start + tl.arange(0, BLOCK_N)   # [BLOCK_N]

    mask_m = offs_m < total_rows
    mask_n = offs_n < dim

    NN = N * N
    hidden_stride = NN

    # Decode b,i,j from the flattened row index
    b = offs_m // NN
    rem = offs_m - b * NN
    i = rem // N
    j = rem - i * N

    # Base offset for hidden‑dim slice 0
    base_offset = b * (H * hidden_stride) + i * N + j   # [BLOCK_M]

    # ------------------------------------------------------------------
    # Allocate shared memory for the full hidden vector of each row
    # ------------------------------------------------------------------
    sh_out = tl.alloc([BLOCK_M, H], dtype=tl.float16)
    sh_gate = tl.alloc([BLOCK_M, H], dtype=tl.float16)

    # ------------------------------------------------------------------
    # First pass: load out & gate into shared memory and accumulate statistics
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for h_start in range(0, H, BLOCK_K):
        offs_h = h_start + tl.arange(0, BLOCK_K)          # [BLOCK_K]
        mask_h = offs_h < H

        # Load a tile of out (fp16) and gate (fp16) from global memory
        out_tile = tl.load(
            out_ptr + base_offset[:, None] + offs_h[None, :] * hidden_stride,
            mask=mask_m[:, None] & mask_h[None, :],
            other=0.0,
        )
        gate_tile = tl.load(
            gate_ptr + base_offset[:, None] + offs_h[None, :] * hidden_stride,
            mask=mask_m[:, None] & mask_h[None, :],
            other=0.0,
        )

        # Store into shared memory (will be reused in the second pass)
        smem_out_ptr  = sh_out  + offs_m[:, None] * H + offs_h[None, :]
        smem_gate_ptr = sh_gate + offs_m[:, None] * H + offs_h[None, :]
        tl.store(smem_out_ptr,  out_tile,  mask=mask_m[:, None] & mask_h[None, :])
        tl.store(smem_gate_ptr, gate_tile, mask=mask_m[:, None] & mask_h[None, :])

        # Update running sum / sum‑squared (cast to fp32 for accuracy)
        out_fp32 = out_tile.to(tl.float32)
        sum_val += tl.sum(out_fp32, axis=1)
        sum_sq  += tl.sum(out_fp32 * out_fp32, axis=1)

    # --------------------------------------------------------------
    # Compute mean and rsqrt‑variance for each row
    # --------------------------------------------------------------
    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)       # fp32

    # --------------------------------------------------------------
    # Second pass: LayerNorm + out_gate + projection
    # --------------------------------------------------------------
    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    for h_start in range(0, H, BLOCK_K):
        offs_h = h_start + tl.arange(0, BLOCK_K)          # [BLOCK_K]
        mask_h = offs_h < H

        # Load from shared memory (fast)
        smem_out_ptr  = sh_out  + offs_m[:, None] * H + offs_h[None, :]
        smem_gate_ptr = sh_gate + offs_m[:, None] * H + offs_h[None, :]
        out_tile  = tl.load(smem_out_ptr,  mask=mask_m[:, None] & mask_h[None, :], other=0.0)
        gate_tile = tl.load(smem_gate_ptr, mask=mask_m[:, None] & mask_h[None, :], other=0.0)

        # Load LayerNorm parameters for this hidden slice (γ, β)
        gamma = tl.load(ln_w_ptr + offs_h, mask=mask_h, other=0.0)   # fp32
        beta  = tl.load(ln_b_ptr + offs_h, mask=mask_h, other=0.0)   # fp32

        # ---- Hidden‑dim LayerNorm (in fp32 for stability) ----
        out_fp32   = out_tile.to(tl.float32)                         # [BLOCK_M, BLOCK_K]
        norm_fp32  = ((out_fp32 - mean[:, None]) * rstd[:, None]) * gamma[None, :] + beta[None, :]
        norm_fp16  = norm_fp32.to(tl.float16)                         # back to fp16

        # Apply out_gate (already fp16)
        gated = norm_fp16 * gate_tile                                 # [BLOCK_M, BLOCK_K] fp16

        # Load weight slice w_out_T[h, d]  (row‑major: hidden × dim)
        w_ptr = w_out_T_ptr + offs_h[:, None] * dim + offs_n[None, :]   # [BLOCK_K, BLOCK_N]
        w_tile = tl.load(
            w_ptr,
            mask=mask_h[:, None] & mask_n[None, :],
            other=0.0,
        )                                                             # fp16

        # Multiply‑accumulate (MFMA)
        acc += tl.dot(gated, w_tile).to(tl.float32)

    # --------------------------------------------------------------
    # Store final result (float32)
    # --------------------------------------------------------------
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]   # [BLOCK_M, BLOCK_N]
    tl.store(dst_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])

    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # --------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # --------------------------------------------------------------
    # 2️⃣ Fused linear projection (5·hidden_dim)
    # --------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # [5·H, dim]
    proj = F.linear(x_norm, fused_w)        # (R, 5·H) fp16
    del fused_w, x_norm

    # --------------------------------------------------------------
    # 3️⃣ Split, mask, sigmoid gates → left / right / out_gate
    # --------------------------------------------------------------
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣ Core einsum via batched GEMM (cuBLAS)
    # --------------------------------------------------------------
    # reshape to [B·H, N, N] for torch.bmm
    left_mat = left.view(B * H, N, N)                # fp16
    right_mat = right.view(B * H, N, N)              # fp16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # fp16, shape [B·H, N, N]
    out = out.view(B, H, N, N)                       # fp16

    # --------------------------------------------------------------
    # 5️⃣ Fuse hidden‑dim LayerNorm, out_gate and final projection
    # --------------------------------------------------------------
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim] fp16

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    # Choose block sizes that fit into shared memory (H = 128)
    BLOCK_M_OUT = 64      # rows per program (≤ 64 to keep shared memory < 64KB)
    BLOCK_K_OUT = 32      # hidden‑dim tile
    BLOCK_N_OUT = 128     # output‑dim tile (covers dim up to 384)

    grid_out = (
        triton.cdiv(R, BLOCK_M_OUT),
        triton.cdiv(dim, BLOCK_N_OUT),
    )
    fused_output_kernel[grid_out](
        out,
        out_gate,
        w["to_out_norm.weight"],
        w["to_out_norm.bias"],
        w_out_T,
        final,
        R,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        H=H,                          # compile‑time hidden dimension
        num_warps=8,
    )

    return final
```
