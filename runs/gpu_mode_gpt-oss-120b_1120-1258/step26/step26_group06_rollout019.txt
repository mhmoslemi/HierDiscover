"""
Outgoing TriMul (AlphaFold3) – fused Triton implementation.
================================================================

The forward pass consists of:
1️⃣ Row‑wise LayerNorm over the channel dimension (float32 → float16).
2️⃣ A single fused linear projection that yields the five tensors
   (left, right, left_gate, right_gate, out_gate).  The mask (if any)
   and the sigmoid gates are applied inside a Triton kernel, producing
   `left`, `right` and `out_gate` in half‑precision.
3️⃣ A batched half‑precision matrix‑multiply (cuBLAS) computes the
   einsum `out[b,i,j,k] = Σ_k left[b,i,k] * right[b,j,k]`.
4️⃣ A final Triton kernel fuses:
   * the second LayerNorm (its mean/rstd are computed on‑the‑fly),
   * multiplication by `out_gate`,
   * the final linear projection `to_out`.
   The kernel accumulates three dot‑products over the hidden dimension
   to avoid a second pass over `out`:
       acc1 = Σ (out·gate·γ)·W
       acc2 = Σ (gate·γ)·W          (used for the μ term)
       acc3 = Σ (gate·β)·W
   With μ and rstd known, the output is
       rstd·acc1 – rstd·μ·acc2 + acc3.
All kernels use mixed precision (FP16 arithmetic, FP32 accumulation)
and run completely on‑device, yielding a float32 tensor
`[B, N, N, dim]` identical (up to rounding) to the PyTorch reference.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # [R, C] FP32 input
    y_ptr,          # [R, C] FP16 output
    w_ptr,          # [C]    LayerNorm weight γ (FP32)
    b_ptr,          # [C]    LayerNorm bias   β (FP32)
    rows,           # R = B·N·N
    dim,            # C
    eps,            # epsilon
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ---- first pass: mean & variance ------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                                   # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                     # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)              # [BLOCK_M]

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- second pass: write normalized values ---------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                                   # [BLOCK_M, BLOCK_D]

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * dim + offs_d[None, :],
            out.to(tl.float16),
            mask=mask_m[:, None] & mask_d[None, :],
        )


# ----------------------------------------------------------------------
# 2️⃣ Split‑fused projection, optional mask, sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,       # [R, 5·H] fused projection (FP16)
    left_ptr,       # [B, H, N, N] (FP16)
    right_ptr,      # [B, H, N, N] (FP16)
    out_gate_ptr,   # [B, H, N, N] (FP16)
    mask_ptr,       # [R] (FP16)  – optional
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask (broadcast across H)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)
        mask_val = mask_val[:, None]                     # [BLOCK_R, 1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N
    # base offset for hidden‑plane 0 at (b,i,k)
    base_offset = b * (H * hidden_stride) + i * N + k

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # layout: [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # sigmoid in FP32 for stability, cast back to FP16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate                                 # no mask

        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst_offset,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset,
                 outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 3️⃣ Fuse second LayerNorm, out_gate multiplication and final projection
# ----------------------------------------------------------------------
@triton.jit
def final_output_kernel(
    out_ptr,            # [B, H, N, N]  (FP16) – result of the batched GEMM
    gate_ptr,           # [B, H, N, N]  (FP16) – out_gate
    ln_w_ptr,           # [H]           (FP32) – to_out_norm weight γ
    ln_b_ptr,           # [H]           (FP32) – to_out_norm bias   β
    w_out_T_ptr,        # [H, dim]      (FP16) – transposed to_out weight
    out_final_ptr,      # [B, N, N, dim] (FP32) – final result
    total_rows, H, N, dim,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr,
):
    pid_m = tl.program_id(0)     # rows
    pid_n = tl.program_id(1)     # output‑dim tiles

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)   # [M]
    offs_n = col_start + tl.arange(0, BLOCK_N)   # [N]

    mask_m = offs_m < total_rows
    mask_n = offs_n < dim

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + j   # start of hidden plane 0

    # accumulators
    sum_o   = tl.zeros([BLOCK_M], dtype=tl.float32)   # Σ out
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)   # Σ out²
    acc1    = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)   # Σ out·gate·γ·W
    acc2    = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)   # Σ gate·γ·W
    acc3    = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)   # Σ gate·β·W

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)          # [K]
        mask_k = offs_k < H

        # ------------------------------------------------------------------
        # Load tensors for this hidden‑dim slice
        # ------------------------------------------------------------------
        out_vals = tl.load(
            out_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)                                   # [M, K]

        gate_vals = tl.load(
            gate_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)                                   # [M, K]

        # update mean / variance statistics
        sum_o  += tl.sum(out_vals, axis=1)                 # [M]
        sum_sq += tl.sum(out_vals * out_vals, axis=1)      # [M]

        # LayerNorm parameters
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0).to(tl.float32)   # [K]
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0).to(tl.float32)   # [K]

        # ------------------------------------------------------------------
        # Prepare three terms for the fused dot‑product
        # ------------------------------------------------------------------
        term1 = (out_vals * gate_vals) * gamma[None, :]      # out·gate·γ
        term2 = gate_vals * gamma[None, :]                   # gate·γ
        term3 = gate_vals * beta[None, :]                    # gate·β

        term1_fp16 = term1.to(tl.float16)
        term2_fp16 = term2.to(tl.float16)
        term3_fp16 = term3.to(tl.float16)

        # Load slice of the final projection weight
        w_tile = tl.load(
            w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :],
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )                                                   # [K, N]

        # Accumulate three dot‑products
        acc1 += tl.dot(term1_fp16, w_tile).to(tl.float32)
        acc2 += tl.dot(term2_fp16, w_tile).to(tl.float32)
        acc3 += tl.dot(term3_fp16, w_tile).to(tl.float32)

    # ----------------------------------------------------------------------
    # Compute per‑row mean / rsqrt‑variance
    # ----------------------------------------------------------------------
    mean = sum_o / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)                                 # [M]

    # ----------------------------------------------------------------------
    # Assemble final output
    # ----------------------------------------------------------------------
    final = rstd[:, None] * acc1 - rstd[:, None] * mean[:, None] * acc2 + acc3

    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, final, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only implementation of the outgoing TriMul operator.

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, C] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
              All tensors are float32 and already on the target device.
            - config_dict must contain:
                "dim" (C), "hidden_dim" (H), "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, N2, _ = x.shape
    assert N == N2, "TriMul expects a square pairwise matrix"
    R = B * N * N                          # total (b,i,k) rows
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                # [R, C] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    layernorm_kernel[(triton.cdiv(R, BLOCK_M),)](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear projection (5·H) and split + gate handling
    # ------------------------------------------------------------------
    # Concatenate weights for a single matmul
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16).contiguous()          # [5·H, C]

    proj = F.linear(x_norm, fused_w)            # [R, 5·H] (FP16)
    del fused_w, x_norm

    # Allocate output tensors for the split step
    left     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right    = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    split_fuse_gate_kernel[(triton.cdiv(R, BLOCK_R),)](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 3️⃣ Batched GEMM (left • rightᵀ) – the core einsum
    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)                     # [B·H, N, N] (FP16)
    right_mat = right.view(B * H, N, N)                    # [B·H, N, N] (FP16)
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [B·H, N, N] (FP16)
    out = out.view(B, H, N, N)                            # [B, H, N, N] (FP16)

    # free memory that is no longer needed
    del left_mat, right_mat, left, right

    # ------------------------------------------------------------------
    # 4️⃣ Final fused LayerNorm + out_gate + projection
    # ------------------------------------------------------------------
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim] (FP16)

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 64          # rows per program
    BLOCK_K_OUT = 32          # hidden‑dim chunk
    BLOCK_N_OUT = 64          # output‑dim chunk

    grid = (
        triton.cdiv(R, BLOCK_M_OUT),
        triton.cdiv(dim, BLOCK_N_OUT),
    )
    final_output_kernel[grid](
        out,
        out_gate,
        w["to_out_norm.weight"],   # γ (FP32)
        w["to_out_norm.bias"],     # β (FP32)
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=8,
    )
    return final
