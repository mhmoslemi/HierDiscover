"""
Fast outgoing TriMul (AlphaFold‑3) forward pass.

Algorithm
---------
1️⃣  Row‑wise LayerNorm on the input tensor (custom Triton kernel).  
    The kernel works on fp32 data but writes fp16 – this avoids extra casts.

2️⃣  Fuse the five dense layers (left_proj, right_proj, left_gate,
    right_gate, out_gate) into a *single* half‑precision GEMM
    `[R, dim] @ [5*hidden, dim]ᵀ → [R, 5*hidden]`, where `R = B·N·N`.

3️⃣  Apply the optional mask (only on the two projected tensors) and the
    three sigmoid gates.

4️⃣  Triangular multiplicative update  
    `out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]`  
    is performed as a batched GEMM using `torch.bmm`:
    each hidden channel becomes an independent batch element,
    i.e. a set of `B·hidden` matrix‑multiplications of size `N×N`.

5️⃣  Second LayerNorm (same Triton kernel, operating on the hidden
    dimension).

6️⃣  Multiply by the output gate and project back to the original
    `dim` with a final half‑precision linear layer; the result is cast
    to float32 before returning.

Only the LayerNorms are written as explicit Triton kernels – the heavy
cubic computation is carried out by a highly‑optimised cuBLAS batched
GEMM (`torch.bmm`).  All heavy linear work is done in fp16 to exploit
Tensor‑Cores on an H100, while the final output is fp32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton row‑wise LayerNorm (output in fp16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const *  [total_rows, dim]   (float32 or fp16)
    y_ptr,          # mutable *[total_rows, dim]   (fp16 output)
    w_ptr,          # const *  [dim]               (float32, γ)
    b_ptr,          # const *  [dim]               (float32, β)
    total_rows,     # int (runtime)
    dim: tl.constexpr,      # compile‑time
    eps: tl.constexpr,      # compile‑time
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)                 # [BLOCK_M]
    mask_m = offs_m < total_rows

    # --------------------------------------------------------------
    # 1st pass: compute mean and variance for each row
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)           # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        vals = tl.load(ptrs, mask=mask_m[:, None] & mask_d[None, :], other=0.0)
        vals_f32 = vals.to(tl.float32)

        sum_val += tl.sum(vals_f32, axis=1)
        sum_sq  += tl.sum(vals_f32 * vals_f32, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                     # 1 / sqrt(var + eps)

    # --------------------------------------------------------------
    # 2nd pass: write normalized value * γ + β  (as fp16)
    # --------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        vals = tl.load(ptrs, mask=mask_m[:, None] & mask_d[None, :], other=0.0)
        vals_f32 = vals.to(tl.float32)

        norm = (vals_f32 - mean[:, None]) * rstd[:, None]

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]                # fp32

        out_fp16 = out.to(tl.float16)
        out_ptrs = y_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom forward kernel
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]      (float32/float16)
        - weights_dict : mapping with the model weights
        - config_dict  : contains ``dim`` and ``hidden_dim`` plus ``nomask``

    Returns
    -------
    torch.Tensor [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    device = input_tensor.device
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1) First LayerNorm (row‑wise) – emits fp16
    # --------------------------------------------------------------
    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N                     # R = B·N·N

    # flatten to 2‑D tensor [R, dim] (float32)
    x_flat = input_tensor.view(total_rows, dim).contiguous()

    # allocate fp16 buffer for the normalized tensor
    x_norm = torch.empty(total_rows, dim, dtype=torch.float16, device=device)

    BLOCK_M = 256
    BLOCK_D = 64                               # fits dim up to 384 with few loops
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,                     # input (fp32)
        x_norm,                     # output (fp16)
        weights["norm.weight"].to(torch.float32),
        weights["norm.bias"].to(torch.float32),
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # --------------------------------------------------------------
    # 2) Fuse the five dense layers into one half‑precision GEMM
    # --------------------------------------------------------------
    # concat: [left_proj, right_proj, left_gate, right_gate, out_gate]  → (5*H, dim)
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)

    # fused linear: [R, dim] @ [5*H, dim]ᵀ → [R, 5*H]
    fused_out = F.linear(x_norm, w_concat)          # fp16, shape [R, 5*hidden]

    H = hidden_dim
    left_proj   = fused_out[:, 0:H]
    right_proj  = fused_out[:, H:2 * H]
    left_gate_p = fused_out[:, 2 * H:3 * H]
    right_gate_p = fused_out[:, 3 * H:4 * H]
    out_gate_p   = fused_out[:, 4 * H:5 * H]

    # --------------------------------------------------------------
    # 3) Non‑linearities and optional mask
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate_p)          # fp16
    right_gate = torch.sigmoid(right_gate_p)        # fp16
    out_gate   = torch.sigmoid(out_gate_p)          # fp16

    if not nomask:
        # mask is broadcast over the hidden dimension
        mask_h = mask_tensor.to(torch.float16).view(total_rows, 1)   # [R,1]
        left_proj  = left_proj * mask_h
        right_proj = right_proj * mask_h

    left  = left_proj * left_gate
    right = right_proj * right_gate

    # reshape back to 4‑D tensors [B, N, N, H]
    left  = left.view(B, N, N, H)
    right = right.view(B, N, N, H)

    # --------------------------------------------------------------
    # 4) Triangular multiplicative update via batched GEMM
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # --------------------------------------------------------------
    # move hidden dim to batch dimension for torch.bmm
    left_bh  = left.permute(0, 3, 1, 2).contiguous()    # [B, H, N, N]
    right_bh = right.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]

    left_flat  = left_bh.view(B * H, N, N)               # [B*H, N, N]
    right_flat = right_bh.view(B * H, N, N)              # [B*H, N, N]

    # batched matrix multiplication (Tensor‑Core accelerated)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))  # [B*H, N, N]

    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1)   # [B, N, N, H] (fp16)

    # --------------------------------------------------------------
    # 5) Second LayerNorm over the hidden dimension (still fp16)
    # --------------------------------------------------------------
    out_flat2 = out.contiguous().view(total_rows, H)       # [R, H]
    out_norm_flat = torch.empty_like(out_flat2, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat2,
        out_norm_flat,
        weights["to_out_norm.weight"].to(torch.float32),
        weights["to_out_norm.bias"].to(torch.float32),
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    out_norm = out_norm_flat.view(B, N, N, H)              # fp16

    # --------------------------------------------------------------
    # 6) Apply output gate and final projection back to `dim`
    # --------------------------------------------------------------
    out_norm = out_norm * out_gate                         # element‑wise fp16

    out_norm_flat3 = out_norm.view(total_rows, H)          # [R, H]

    final_half = F.linear(
        out_norm_flat3,
        weights["to_out.weight"].to(torch.float16)        # [dim, H]
    )                                                      # [R, dim]

    final = final_half.view(B, N, N, dim).to(torch.float32)  # cast to float32

    return final
