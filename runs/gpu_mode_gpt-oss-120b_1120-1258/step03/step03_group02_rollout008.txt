"""
TriMul "outgoing" forward pass (AlphaFold‑3 style) – mixed‑precision Triton
==========================================================================

The operator works on a 4‑D tensor `x` of shape [B, N, N, C] and produces a
tensor of the same shape.  The core steps are:

1️⃣ Row‑wise LayerNorm on the input (float32 → float16) – a tiny Triton
   kernel computes mean/variance per row and writes the normalized result in
   FP16, saving a full‑precision copy.

2️⃣ Fuse projection and gate linear layers for the left and right paths:
   a single FP16 GEMM produces the projection and pre‑gate tensors (2 ×H
   columns) → 4 GEMMs total (left/right projection + gate).

3️⃣ Optional pairwise mask (broadcast over the hidden dimension) is applied
   in‑place.

4️⃣ Apply sigmoid‑gates and multiply gated projections.

5️⃣ Tri‑multiplicative update:
        out_h(i,j) = Σₖ left(i,k,h) · right(j,k,h)
   This is a batched GEMM with batch size B·H (each GEMM is N×N×N).  PyTorch's
   `torch.bmm` (FP16) is used – it maps to cuBLAS Tensor‑Core kernels.

6️⃣ **Fused** LayerNorm + output‑gate on the hidden tensor and immediate
   write of the result in FP16.  The kernel performs the usual `(x‑μ)/σ`,
   applies the learned γ/β, multiplies by the already‑computed output‑gate,
   and stores the final hidden representation ready for the final projection.

7️⃣ Final linear projection (hidden_dim → dim) – another FP16 GEMM, after which
   the result is cast back to float32.

All heavy per‑row reductions are done in Triton, while the large dense
matrix multiplications (steps 2, 5, 7) use PyTorch’s highly‑optimised cuBLAS
kernels.  The implementation covers all shapes from the prompt and works
with or without a mask.  No gradients are required.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (float32 → float16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,            # const float*   [total_rows, dim]
    y_ptr,            #       half*   [total_rows, dim] (output)
    w_ptr,            # const float*  [dim] (γ)
    b_ptr,            # const float*  [dim] (β)
    total_rows,       # int (runtime)
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,    # rows per program
    BLOCK_D: tl.constexpr,    # columns per inner loop
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ----- pass 1: compute mean & variance --------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)   # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)          # 1 / sqrt(var+eps)

    # ----- pass 2: write normalized values ---------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]   # (x-μ)/σ

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # β

        out = norm * w[None, :] + b[None, :]                # fp32
        out_fp16 = out.to(tl.float16)

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 6️⃣ Fused LayerNorm + output‑gate (FP16 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_gate_fp16_kernel(
    x_ptr,            # const half*   [total_rows, dim]   (input hidden)
    gate_ptr,         # const half*   [total_rows, dim]   (out_gate)
    y_ptr,            #       half*   [total_rows, dim]   (output = norm*gate)
    w_ptr,            # const float*  [dim]               (γ)
    b_ptr,            # const float*  [dim]               (β)
    total_rows,       # int (runtime)
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,    # rows per program
    BLOCK_D: tl.constexpr,    # columns per inner loop
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ----- pass 1: mean & variance -----------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ----- pass 2: normalize, apply γ/β and multiply by gate -----------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        # load hidden value
        x_vals = tl.load(
            x_ptr + offs_m[:, None] * dim + offs_d[None, :],
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)
        # load gate value (already sigmoided)
        g_vals = tl.load(
            gate_ptr + offs_m[:, None] * dim + offs_d[None, :],
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # β

        norm = (x_vals - mean[:, None]) * rstd[:, None]
        out  = (norm * w[None, :] + b[None, :]) * g_vals   # fp32
        out_fp16 = out.to(tl.float16)

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.

    Args:
        data: (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, C]    (float32)
            - mask_tensor: torch.Tensor [B, N, N]      (bool/float)
            - weights_dict: mapping of all weight tensors
            - config_dict : contains "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N                     # rows for flattened view

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32 → FP16) – Triton
    # --------------------------------------------------------------
    x_flat = input_tensor.reshape(total_rows, dim).contiguous()
    x_norm_h = torch.empty((total_rows, dim), dtype=torch.float16, device=device)

    BLOCK_M = 256          # rows per program (tuned)
    BLOCK_D = 128          # columns per inner loop (covers dim up to 384)
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        x_flat,
        x_norm_h,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    # No explicit sync – let downstream ops handle it.

    # --------------------------------------------------------------
    # 2️⃣ Fuse projection + gate (4 GEMMs total, all FP16)
    # --------------------------------------------------------------
    # Concatenate proj and gate weights for left/right paths
    left_concat_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16, non_blocking=True)
    right_concat_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16, non_blocking=True)

    # Linear passes – shape: [total_rows, 2*hidden_dim]
    left_concat = F.linear(x_norm_h, left_concat_w)          # FP16 GEMM
    right_concat = F.linear(x_norm_h, right_concat_w)

    left_proj_h   = left_concat[:, :hidden_dim]               # [rows, H]
    left_gate_pre = left_concat[:, hidden_dim:]               # [rows, H]
    right_proj_h  = right_concat[:, :hidden_dim]
    right_gate_pre = right_concat[:, hidden_dim:]

    # Out‑gate (needs same input, compute once)
    out_gate_h = torch.sigmoid(
        F.linear(x_norm_h, weights["out_gate.weight"].to(torch.float16))
    )   # [rows, hidden_dim] – already FP16

    # --------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over hidden dimension)
    # --------------------------------------------------------------
    if not nomask:
        mask_h = mask_tensor.reshape(total_rows, 1).to(torch.float16)
        left_proj_h = left_proj_h * mask_h
        right_proj_h = right_proj_h * mask_h

    # --------------------------------------------------------------
    # 4️⃣ Apply sigmoid‑gates to left/right paths
    # --------------------------------------------------------------
    left_gate_h  = torch.sigmoid(left_gate_pre)
    right_gate_h = torch.sigmoid(right_gate_pre)

    left_h  = left_proj_h * left_gate_h            # [rows, H]
    right_h = right_proj_h * right_gate_h

    # --------------------------------------------------------------
    # 5️⃣ Tri‑multiplicative update (batched GEMM)
    # --------------------------------------------------------------
    # Reshape to [B, N, N, H] then bring H to batch dimension
    left  = left_h.view(B, N, N, hidden_dim)
    right = right_h.view(B, N, N, hidden_dim)

    # Permute to [B, H, N, N] – each (B,H) slice is a matrix to multiply
    left_perm  = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    BH = B * hidden_dim
    left_mat  = left_perm.view(BH, N, N)            # [BH, N, N]
    right_mat = right_perm.view(BH, N, N)

    # out_mat = left_mat @ right_matᵀ  (sum over k)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [BH, N, N]

    # Back to [B, N, N, H] (FP16)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6️⃣ Fused LayerNorm + output‑gate on hidden tensor
    # --------------------------------------------------------------
    total_rows_h = B * N * N
    out_flat = out.view(total_rows_h, hidden_dim).contiguous()
    out_norm_gate_h = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows_h, BLOCK_M),)
    layernorm_gate_fp16_kernel[grid2](
        out_flat,
        out_gate_h,                       # already sigmoided
        out_norm_gate_h,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows_h,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    # No sync needed; the following linear forces completion.

    # --------------------------------------------------------------
    # 7️⃣ Final linear projection (hidden_dim → dim) – FP16 GEMM
    # --------------------------------------------------------------
    final_h = F.linear(
        out_norm_gate_h,
        weights["to_out.weight"].to(torch.float16)
    )   # [total_rows_h, dim]  (FP16)

    final = final_h.view(B, N, N, dim).to(torch.float32)   # back to FP32
    return final
