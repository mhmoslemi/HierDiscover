"""
Highly‑optimized forward kernel for the “outgoing” TriMul operation (AlphaFold‑3).

Algorithm
---------
1. **Row‑wise LayerNorm (FP16)** – a custom Triton kernel computes per‑row
   mean/variance in FP32, normalises the FP32 input and writes the result
   directly as half‑precision.  This replaces the PyTorch `nn.LayerNorm`.

2. **Fused linear projections + gates** – the five weight matrices
   (`left_proj`, `right_proj`, `left_gate`, `right_gate`, `out_gate`)
   are concatenated (shape `(5*H, C)`) and a single `torch.nn.functional.linear`
   produces a `(R, 5*H)` tensor in FP16.  A single `torch.split` recovers the
   five `(R, H)` tensors.

3. **Optional mask** – the binary mask is applied to the *left* and *right*
   projections (still in FP16) before gating.

4. **Sigmoid gates** – `torch.sigmoid` works on FP16, producing the three
   gate tensors.  The gated projections are obtained with in‑place
   multiplication to keep the memory footprint low.

5. **Triangular multiplicative update** – the expensive contraction

       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]

   is performed by a single `torch.einsum` which reads the tensors in their
   original layout (`[B,N,N,H]`) and avoids the expensive `permute(...).contiguous()`
   copies required for a batched `bmm`.  This yields `out` of shape
   `[B,N,N,H]` in FP16.

6. **Fused LayerNorm + output gate** – another Triton kernel combines the
   second LayerNorm (over the hidden dimension) with the pre‑computed
   `out_gate`.  The kernel mirrors step 1 but also multiplies by the gate
   before writing the result back as FP16.

7. **Final projection** – a standard FP16 GEMM (`F.linear`) maps the hidden
   dimension back to the original channel dimension `C`.  The result is
   cast to FP32 for the function return.

All heavy‑weight element‑wise work (both LayerNorms) runs on custom Triton
kernels, while the large dense GEMMs (`linear` and the triangular update)
use the highly‑optimised cuBLAS Tensor‑Core kernels.  The implementation
requires only the Triton kernels; everything else is native PyTorch
operations on the same CUDA device.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1) Row‑wise LayerNorm that writes FP16
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,                     # *const float*  [total_rows, dim] (FP32 input)
    y_ptr,                     # *mut   half*   [total_rows, dim] (FP16 output)
    weight_ptr,                # *const float*  [dim]          (γ)
    bias_ptr,                  # *const float*  [dim]          (β)
    total_rows,                # int
    dim: tl.constexpr,         # compile‑time static dimension
    eps: tl.constexpr,         # epsilon
    BLOCK_M: tl.constexpr,     # rows per program
    BLOCK_D: tl.constexpr,     # columns per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)              # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ---- first pass: row‑wise mean & variance (FP32 accumulation) ----
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)        # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                               # 1 / sqrt(var+eps)

    # ---- second pass: normalise, apply γ, β and store as FP16 ----
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # FP32

        norm = (vals - mean[:, None]) * rstd[:, None]       # (x-µ)/σ
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # β
        out = norm * w[None, :] + b[None, :]                # FP32

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2) Fused LayerNorm + output‑gate (both in FP16)
# ----------------------------------------------------------------------
@triton.jit
def fused_layernorm_gate_kernel(
    x_ptr,                     # *const half* [total_rows, dim] (pre‑norm)
    y_ptr,                     # *mut   half* [total_rows, dim] (post‑norm·gate)
    weight_ptr,                # *const float* [dim] (γ)
    bias_ptr,                  # *const float* [dim] (β)
    gate_ptr,                  # *const half*  [total_rows, dim] (out_gate)
    total_rows,
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # ---- mean / variance (FP32) ----
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # FP32
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- normalise, apply γ, β, multiply by gate, store FP16 ----
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # FP32

        norm = (vals - mean[:, None]) * rstd[:, None]           # (x-µ)/σ
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # β
        out = norm * w[None, :] + b[None, :]                    # FP32

        gate_vals = tl.load(gate_ptr + offs_m[:, None] * dim + offs_d[None, :],
                            mask=mask_m[:, None] & mask_d[None, :],
                            other=0.0).to(tl.float32)          # FP32
        out = out * gate_vals

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
            input_tensor : torch.Tensor [B, N, N, C]    (float32)
            mask_tensor  : torch.Tensor [B, N, N]     (float32 / bool)
            weights_dict : dict of model parameters (float32)
            config_dict  : dict containing:
                - dim (int)          : input channel dimension C
                - hidden_dim (int)   : projection dimension H
                - nomask (bool)      : True → no mask
                - eps (float)        : LayerNorm epsilon (optional)

    Returns:
        torch.Tensor [B, N, N, C] (float32) on the same CUDA device.
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    inp, mask, weights, cfg = data
    device = inp.device
    dim   = cfg["dim"]
    hidden = cfg["hidden_dim"]
    eps   = cfg.get("eps", 1e-5)
    nomask = cfg.get("nomask", True)

    B, N, _, _ = inp.shape
    total_rows = B * N * N                       # rows for the flattened tensors

    ##################################################################
    # 1) Input LayerNorm (produces FP16)
    ##################################################################
    inp_flat = inp.view(total_rows, dim).contiguous()   # [R, C] float32
    x_norm_fp16 = torch.empty(total_rows, dim, dtype=torch.half,
                              device=device)          # [R, C] FP16

    BLOCK_M = 256
    BLOCK_D = 32

    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        inp_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()

    ##################################################################
    # 2) Fused linear projections + gates (single GEMM, FP16)
    ##################################################################
    # Concatenate the five weight tensors (shape (5*hidden, dim))
    concat_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).half().contiguous()                       # (5*H, C)

    # One GEMM for all five tensors
    proj_all = F.linear(x_norm_fp16, concat_w)   # (R, 5*H)  FP16
    left_proj_fp16, right_proj_fp16, left_gate_fp16, right_gate_fp16, out_gate_fp16 = torch.split(
        proj_all, hidden, dim=1
    )                                            # each (R, H)

    ##################################################################
    # 2a) Optional mask (apply before gating)
    ##################################################################
    if not nomask:
        # mask has shape [B, N, N]; broadcast over hidden dim
        mask_flat = mask.view(total_rows, 1).to(inp.dtype).half()
        left_proj_fp16 = left_proj_fp16 * mask_flat
        right_proj_fp16 = right_proj_fp16 * mask_flat

    ##################################################################
    # 2b) Sigmoid gates (still FP16)
    ##################################################################
    left_gate = torch.sigmoid(left_gate_fp16)          # (R, H) FP16
    right_gate = torch.sigmoid(right_gate_fp16)       # (R, H) FP16
    out_gate = torch.sigmoid(out_gate_fp16)           # (R, H) FP16

    # In‑place gating to keep the memory footprint low
    left_fp16 = left_proj_fp16 * left_gate
    right_fp16 = right_proj_fp16 * right_gate

    ##################################################################
    # 3) Reshape to [B,N,N,H] for the triangular update
    ##################################################################
    left = left_fp16.view(B, N, N, hidden)      # (B, N, N, H)
    right = right_fp16.view(B, N, N, hidden)    # (B, N, N, H)

    ##################################################################
    # 4) Triangular multiplicative update (einsum avoids copies)
    ##################################################################
    # out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    out = torch.einsum('b i k h, b j k h -> b i j h', left, right)   # (B,N,N,H) FP16

    ##################################################################
    # 5) Second LayerNorm + output gate (fused Triton kernel)
    ##################################################################
    out_flat = out.view(total_rows, hidden).contiguous()   # (R, H) FP16
    out_norm_gate_fp16 = torch.empty_like(out_flat)        # (R, H) FP16

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    fused_layernorm_gate_kernel[grid2](
        out_flat,
        out_norm_gate_fp16,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        out_gate,                      # already sigmoid’ed, FP16
        total_rows,
        hidden,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()

    ##################################################################
    # 6) Final linear projection back to `dim`
    ##################################################################
    to_out_w_fp16 = weights["to_out.weight"].half()          # (dim, hidden)
    final_fp16 = F.linear(out_norm_gate_fp16, to_out_w_fp16) # (R, dim) FP16
    final = final_fp16.view(B, N, N, dim).to(torch.float32)  # cast to FP32

    return final
