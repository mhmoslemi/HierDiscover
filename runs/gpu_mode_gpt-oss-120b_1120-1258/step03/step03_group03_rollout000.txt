"""
TriMul (outgoing) – fast forward pass
====================================

The kernel implements the “outgoing” variant of the TriMul operator
described in AlphaFold‑3.  The implementation follows the reference
PyTorch code but moves the two LayerNorms to Triton while keeping the
heavy matrix‑multiplications (the N³ update) in cuBLAS/Tensor‑Core
(batched GEMM).  Mixed‑precision (FP16) is used for all linear
operations; the final result is cast back to FP32 to match the
original model.

Algorithm
---------
1. **LayerNorm over the input channel dimension (float32 → FP16)**  
   A custom Triton kernel computes per‑row mean/variance in FP32,
   applies the learned γ/β (float32) and writes the normalized tensor
   in FP16 (`x_norm`).

2. **Fused linear projection**  
   The five weight matrices (`left_proj`, `left_gate`, `right_proj`,
   `right_gate`, `out_gate`) are concatenated into a single matrix
   `W_concat` (shape `(5*H, D)`).  A single `F.linear` computes all
   projections and gates at once: `proj_all = x_norm @ W_concatᵀ`.

3. **Mask, sigmoid, gating**  
   The optional mask (if provided) is broadcast over the hidden
   dimension and applied to `left_proj`/`right_proj`.  The three gate
   slices are passed through a sigmoid and multiplied with their
   respective projections, yielding `left` and `right` (shape
   `[B·N·N, H]`).

4. **Tri‑multiplicative update**  
   `left` and `right` are reshaped to `[B, N, N, H]`, permuted so that
   `H` becomes a batch axis, and a batched matrix multiply
   `torch.bmm(left, rightᵀ)` computes the N³ sum for every hidden
   slice in a single call to cuBLAS.

5. **Second LayerNorm + output gate**  
   A second Triton kernel fuses a half‑precision LayerNorm over the
   hidden dimension with an element‑wise multiplication by the gated
   `out_gate`.

6. **Final linear projection**  
   The result is projected back to the original channel dimension with
   `to_out.weight` (FP16) and cast to FP32.

Only the two LayerNorms are custom Triton kernels; all dense
matrix‑multiplications are delegated to highly‑optimised cuBLAS
Tensor‑Core kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernels
# ----------------------------------------------------------------------


@triton.jit
def layernorm_kernel(
    x_ptr,          # const float*   [total_rows, dim]
    y_ptr,          #       float*   [total_rows, dim]  (output)
    weight_ptr,     # const float*   [dim]               (γ)
    bias_ptr,       # const float*   [dim]               (β)
    total_rows,     # i32
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """
    Row‑wise LayerNorm in FP32 (input & output are FP32).
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # 1st pass – row‑wise sum & sum‑of‑squares
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                               # [BLOCK_M, BLOCK_D]
        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                          # 1 / sqrt(var+eps)

    # 2nd pass – normalize, apply γ/β, store result
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)

        out = norm * w[None, :] + b[None, :]
        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out, mask=mask_m[:, None] & mask_d[None, :])


@triton.jit
def layernorm_gate_kernel(
    x_ptr,          # const half*    [total_rows, dim]   (input)
    y_ptr,          #       half*    [total_rows, dim]   (output)
    gate_ptr,       # const half*    [total_rows, dim]   (gate)
    weight_ptr,     # const half*   [dim]               (γ)
    bias_ptr,       # const half*   [dim]               (β)
    total_rows,     # i32
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """
    Half‑precision LayerNorm followed by element‑wise multiplication
    with a gate tensor.  Accumulators are FP32.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # 1st pass – mean / variance (FP32)
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # 2nd pass – normalize, γ/β, multiply by gate, store as FP16
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        # load x
        x_ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        x_val = tl.load(
            x_ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        # normalize
        norm = (x_val - mean[:, None]) * rstd[:, None]

        # γ, β (half → float32)
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0).to(tl.float32)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0).to(tl.float32)

        out = norm * w[None, :] + b[None, :]

        # gate (half → float32)
        g_ptrs = gate_ptr + offs_m[:, None] * dim + offs_d[None, :]
        gate = tl.load(
            g_ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        out = out * gate

        # store back as FP16
        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out.to(tl.float16), mask=mask_m[:, None] & mask_d[None, :])

# Main entry point

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor, [B, N, N, D]  (float32)
            - mask_tensor  : torch.Tensor, [B, N, N]   (bool/int) or None
            - weights_dict : dict of torch.Tensors matching the module
            - config_dict  : dict with keys ``dim``, ``hidden_dim``, ``nomask``

    Returns:
        torch.Tensor of shape [B, N, N, D] (float32)
    """
    # ---------------------------------------------------------
    # 1. Unpack arguments & constants
    # ---------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                 # D
    hidden_dim = config["hidden_dim"]   # H
    eps = 1e-5
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N             # R = B·N·N

    # ---------------------------------------------------------
    # 2. First LayerNorm (float32 → FP16)
    # ---------------------------------------------------------
    x = input_tensor.contiguous()                     # [B,N,N,D]  FP32
    x_flat = x.view(total_rows, dim)                  # [R,D]      FP32
    x_norm_flat = torch.empty_like(x_flat)            # FP32 output

    BLOCK_M = 256
    BLOCK_D = 32
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm_flat,
        weights["norm.weight"],      # γ (FP32, [D])
        weights["norm.bias"],        # β (FP32, [D])
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # cast to half for the rest of the pipeline
    x_norm_h = x_norm_flat.to(torch.float16)          # [R,D]  FP16

    # ---------------------------------------------------------
    # 3. Fused linear projection for all gates / projections
    # ---------------------------------------------------------
    # Concatenate weight matrices: shape (5*H, D)
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)      # (5*H, D)

    # Linear: [R,D] @ (5*H,D)^T → [R,5*H] (FP16)
    proj_all = F.linear(x_norm_h, w_concat)  # FP16

    H = hidden_dim
    left_proj   = proj_all[:, 0:H]          # [R, H]
    left_gate   = proj_all[:, H:2*H]        # [R, H]
    right_proj  = proj_all[:, 2*H:3*H]      # [R, H]
    right_gate  = proj_all[:, 3*H:4*H]      # [R, H]
    out_gate    = proj_all[:, 4*H:5*H]      # [R, H]

    # ---------------------------------------------------------
    # 4. Apply optional mask & sigmoids, then gating
    # ---------------------------------------------------------
    if not nomask:
        # mask shape: [B,N,N] → [R,1] (FP16)
        mask_h = mask_tensor.to(torch.float16).view(total_rows, 1)
        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h

    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    left = left_proj * left_gate   # [R, H] FP16
    right = right_proj * right_gate   # [R, H] FP16

    # ---------------------------------------------------------
    # 5. Tri‑multiplicative update via batched GEMM
    # ---------------------------------------------------------
    left = left.view(B, N, N, H)        # [B,N,N,H]
    right = right.view(B, N, N, H)      # [B,N,N,H]

    # Move hidden dimension to batch axis for bmm
    left_mat = left.permute(0, 3, 1, 2).contiguous()   # [B,H,N,N]
    right_mat = right.permute(0, 3, 2, 1).contiguous() # [B,H,N,N] (j,k) → (k,j)

    BH = B * H
    left_mat = left_mat.view(BH, N, N)      # [B*H, N, N]
    right_mat = right_mat.view(BH, N, N)    # [B*H, N, N]

    # Batched matrix multiply (Tensor‑Core GEMM, FP16)
    out_mat = torch.bmm(left_mat, right_mat)   # [B*H, N, N] FP16

    # Restore original layout: [B,N,N,H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # FP16

    # ---------------------------------------------------------
    # 6. Second LayerNorm over hidden dim + output gate
    # ---------------------------------------------------------
    out_flat = out.view(total_rows, H)               # [R, H] FP16
    out_gate_flat = out_gate.view(total_rows, H)     # [R, H] FP16
    out_norm_gate = torch.empty_like(out_flat)       # FP16 buffer

    BLOCK_M2 = 256
    BLOCK_D2 = 32
    grid2 = (triton.cdiv(total_rows, BLOCK_M2),)
    layernorm_gate_kernel[grid2](
        out_flat,
        out_norm_gate,
        out_gate_flat,
        weights["to_out_norm.weight"].to(torch.float16),   # γ (FP16, [H])
        weights["to_out_norm.bias"].to(torch.float16),     # β (FP16, [H])
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M2,
        BLOCK_D=BLOCK_D2,
    )

    # ---------------------------------------------------------
    # 7. Final linear projection back to original dimension
    # ---------------------------------------------------------
    w_to_out = weights["to_out.weight"].to(torch.float16)   # [D, H]
    final_half = F.linear(out_norm_gate, w_to_out)         # [R, D] FP16
    final = final_half.view(B, N, N, dim).to(torch.float32)  # FP32

    return final
