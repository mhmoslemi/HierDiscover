"""
Highly‑optimized forward pass of the “outgoing” TriMul operator
(AlphaFold‑3).  

Algorithm
---------

1. **Row‑wise LayerNorm** – a custom Triton kernel normalises the input
   tensor over the channel dimension (dim) and writes the result directly
   in fp16.

2. **Fused linear projections** – the five linear layers (left/right
   projections, three gates) are concatenated into a single GEMM
   (fp16, Tensor‑Core).  The result is split into the five tensors.

3. **Fused gating + mask** – a Triton kernel computes  
      `left = left_proj * sigmoid(left_gate)`,  
      `right = right_proj * sigmoid(right_gate)` and also applies the
   optional binary mask in‑place.  The output‑gate is sigmoid‑ed here
   as well, so no extra torch‑call is needed later.

4. **Triangular multiplicative update** – the core operation  
      `out[i,j,d] = Σ_k left[i,k,d] * right[j,k,d]`  
   is performed as a batched matrix‑multiply (`torch.bmm`) after a
   cheap permutation that yields two tensors of shape
   `(B*hidden, N, N)`.  This exploits cuBLAS/Tensor‑Core on the H100.

5. **Fused LayerNorm + output‑gate** – a second Triton kernel normalises
   the result of the triangular update, applies per‑channel weight &
   bias, multiplies by the pre‑sigmoid‑ed `out_gate` and stores fp16.

6. **Final linear projection** – a last fp16 GEMM projects back to the
   original channel dimension and the result is cast to fp32 for the
   output.

Only the two LayerNorms and the gating/mask steps are written in
Triton; the remaining dense linear algebra uses cuBLAS (Tensor‑Core) via
PyTorch which is already optimal on the H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1) Row‑wise LayerNorm (input → fp16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,                     # *const float*   [R, dim]   (FP32 input)
    y_ptr,                     # *mut   half*    [R, dim]   (FP16 output)
    weight_ptr,                # *const float*   [dim]      (γ)
    bias_ptr,                  # *const float*   [dim]      (β)
    total_rows,                # int
    dim: tl.constexpr,         # compile‑time static
    eps: tl.constexpr,         # epsilon
    BLOCK_M: tl.constexpr,     # rows per program
    BLOCK_D: tl.constexpr,     # columns per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)            # [BLOCK_M]
    mask_m = offs_m < total_rows

    # -----------------------------------------------------------------
    # First pass: mean / variance (float32 accumulator)
    # -----------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)        # 1/√(var+eps)

    # -----------------------------------------------------------------
    # Second pass: normalise, apply γ, β and store as fp16
    # -----------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]                      # FP32

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2) Fused gating + optional mask
# ----------------------------------------------------------------------
@triton.jit
def gating_and_mask_kernel(
    left_proj_ptr,   # *const half* [R, H]
    right_proj_ptr,  # *const half* [R, H]
    left_gate_ptr,   # *const half* [R, H]
    right_gate_ptr,  # *const half* [R, H]
    out_gate_ptr,    # *const half* [R, H]  (pre‑sigmoid)
    mask_ptr,        # *const half* [R]     (optional)
    left_out_ptr,    # *mut   half* [R, H]
    right_out_ptr,   # *mut   half* [R, H]
    out_gate_out_ptr,# *mut   half* [R, H] (post‑sigmoid)
    total_rows, hidden,
    HAS_MASK: tl.constexpr,      # compile‑time bool
    BLOCK_R: tl.constexpr,      # rows per program
    BLOCK_H: tl.constexpr,      # hidden per program
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)            # [BLOCK_R]
    mask_r = offs_r < total_rows

    offs_h = tl.arange(0, BLOCK_H)                        # [BLOCK_H]
    mask_h = offs_h < hidden
    active = mask_r[:, None] & mask_h[None, :]            # [BLOCK_R, BLOCK_H]

    # load all tensors in FP32 for arithmetic
    lp = tl.load(left_proj_ptr + offs_r[:, None] * hidden + offs_h[None, :],
                 mask=active, other=0.0).to(tl.float32)
    rp = tl.load(right_proj_ptr + offs_r[:, None] * hidden + offs_h[None, :],
                 mask=active, other=0.0).to(tl.float32)
    lg = tl.load(left_gate_ptr + offs_r[:, None] * hidden + offs_h[None, :],
                 mask=active, other=0.0).to(tl.float32)
    rg = tl.load(right_gate_ptr + offs_r[:, None] * hidden + offs_h[None, :],
                 mask=active, other=0.0).to(tl.float32)
    og = tl.load(out_gate_ptr + offs_r[:, None] * hidden + offs_h[None, :],
                 mask=active, other=0.0).to(tl.float32)

    # sigmoid gates
    lg_sig = tl.sigmoid(lg)
    rg_sig = tl.sigmoid(rg)
    og_sig = tl.sigmoid(og)

    # optional binary mask (broadcasted over hidden)
    if HAS_MASK:
        m = tl.load(mask_ptr + offs_r, mask=mask_r, other=0.0).to(tl.float32)
        m = m[:, None]                # [BLOCK_R, 1] → broadcast
        lp = lp * m
        rp = rp * m

    left_out  = lp * lg_sig
    right_out = rp * rg_sig

    # store results back as FP16
    tl.store(left_out_ptr  + offs_r[:, None] * hidden + offs_h[None, :],
             left_out .to(tl.float16), mask=active)
    tl.store(right_out_ptr + offs_r[:, None] * hidden + offs_h[None, :],
             right_out.to(tl.float16), mask=active)
    tl.store(out_gate_out_ptr + offs_r[:, None] * hidden + offs_h[None, :],
             og_sig   .to(tl.float16), mask=active)


# ----------------------------------------------------------------------
# 3) Fused LayerNorm + output‑gate (post‑tri‑mul)
# ----------------------------------------------------------------------
@triton.jit
def fused_layernorm_gate_kernel(
    x_ptr,                     # *const half*   [R, dim] (pre‑norm)
    y_ptr,                     # *mut   half*   [R, dim] (post‑norm⋅gate)
    weight_ptr,                # *const float*  [dim]   (γ)
    bias_ptr,                  # *const float*  [dim]   (β)
    gate_ptr,                  # *const half*   [R, dim] (post‑sigmoid out_gate)
    total_rows,
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # ---------------------------------------------------------------
    # Mean / var (float32)
    # ---------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------------------------------------------------------------
    # Normalise, apply γ, β and multiply by gate
    # ---------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]   # (x-µ)/σ

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # β
        out = norm * w[None, :] + b[None, :]                    # FP32

        # multiply by the already‑sigmoided out‑gate
        gate = tl.load(gate_ptr + offs_m[:, None] * dim + offs_d[None, :],
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        out = out * gate

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
            input_tensor : torch.Tensor [B, N, N, C]  (float32)
            mask_tensor  : torch.Tensor [B, N, N]   (float32 or bool)
            weights_dict : dict of model parameters (float32)
            config_dict  : dict containing:
                - dim (int)          : input channel dimension C
                - hidden_dim (int)   : projection dimension H
                - nomask (bool)      : True → mask is all‑ones
                - eps (float)        : LayerNorm epsilon (optional)

    Returns:
        torch.Tensor [B, N, N, C] (float32) on the same CUDA device.
    """
    # ------------------------------------------------------------------
    # 0) unpack
    # ------------------------------------------------------------------
    inp, mask, weights, cfg = data
    device = inp.device
    dim   = cfg["dim"]
    hidden = cfg["hidden_dim"]
    eps   = cfg.get("eps", 1e-5)
    nomask = cfg.get("nomask", True)

    B, N, _, _ = inp.shape
    total_rows = B * N * N                     # flattened rows for kernels

    # ------------------------------------------------------------------
    # 1) Input LayerNorm → fp16
    # ------------------------------------------------------------------
    inp_flat = inp.view(total_rows, dim).contiguous()          # [R, C] float32
    x_norm_fp16 = torch.empty(total_rows, dim, dtype=torch.half,
                              device=device)               # [R, C] fp16

    BLOCK_M = 128
    BLOCK_D = 32
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        inp_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()

    # ------------------------------------------------------------------
    # 2) Fused linear projections (5 * hidden) → fp16
    # ------------------------------------------------------------------
    # concatenate the five projection/gate weight matrices (shape (5*H, C))
    concat_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).half().contiguous()               # (5*H, C)

    proj_all = F.linear(x_norm_fp16, concat_w)   # (R, 5*H) fp16
    left_proj_fp16, right_proj_fp16, left_gate_fp16, \
        right_gate_fp16, out_gate_fp16 = torch.split(
            proj_all, hidden, dim=1
        )  # each (R, H) fp16

    # ------------------------------------------------------------------
    # 3) Fuse gating, mask and sigmoid(out_gate)  → fp16 tensors
    # ------------------------------------------------------------------
    left_fp16   = torch.empty_like(left_proj_fp16)
    right_fp16  = torch.empty_like(right_proj_fp16)
    out_gate_sig_fp16 = torch.empty_like(out_gate_fp16)   # already sigmoid‑ed

    if not nomask:
        mask_flat = mask.view(total_rows).to(torch.float16)   # (R,)
    else:
        # dummy tensor – never accessed when HAS_MASK=0
        mask_flat = torch.empty((0,), dtype=torch.half, device=device)

    BLOCK_R = 128
    BLOCK_H = 32
    grid_gate = (triton.cdiv(total_rows, BLOCK_R),)

    gating_and_mask_kernel[grid_gate](
        left_proj_fp16,
        right_proj_fp16,
        left_gate_fp16,
        right_gate_fp16,
        out_gate_fp16,
        mask_flat,
        left_fp16,
        right_fp16,
        out_gate_sig_fp16,
        total_rows,
        hidden,
        HAS_MASK=0 if nomask else 1,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
    )
    torch.cuda.synchronize()

    # ------------------------------------------------------------------
    # 4) Triangular multiplicative update (batched GEMM)
    # ------------------------------------------------------------------
    # reshape to (B, N, N, H)
    left  = left_fp16.view(B, N, N, hidden)   # (B, N, N, H)
    right = right_fp16.view(B, N, N, hidden)  # (B, N, N, H)

    # Permute for batched GEMM: left → (B, H, N, N) with (i,k) layout,
    # right → (B, H, N, N) with (k,j) layout.
    left_mat  = left.permute(0, 3, 1, 2).reshape(B * hidden, N, N)   # (B*H, N, N)
    right_mat = right.permute(0, 3, 2, 1).reshape(B * hidden, N, N)  # (B*H, N, N)

    out_mat = torch.bmm(left_mat, right_mat)                 # (B*H, N, N) fp16
    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 5) Second LayerNorm + output‑gate (fused Triton)
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, hidden).contiguous()       # (R, H) fp16
    out_norm_gate_fp16 = torch.empty_like(out_flat)            # (R, H) fp16

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    fused_layernorm_gate_kernel[grid2](
        out_flat,
        out_norm_gate_fp16,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        out_gate_sig_fp16,               # already sigmoid‑ed
        total_rows,
        hidden,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()

    # ------------------------------------------------------------------
    # 6) Final linear projection back to `dim`
    # ------------------------------------------------------------------
    to_out_w_fp16 = weights["to_out.weight"].half()            # (dim, H)
    final_fp16 = F.linear(out_norm_gate_fp16, to_out_w_fp16)   # (R, dim) fp16
    final = final_fp16.view(B, N, N, dim).to(torch.float32)    # cast to FP32

    return final
