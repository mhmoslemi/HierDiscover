Pseudo:

@triton.jit
def trmul_kernel(
    left_ptr, right_ptr, out_ptr,
    batch, M, N, K,
    stride_left_batch, stride_left_m, stride_left_k,
    stride_right_batch, stride_right_n, stride_right_k,
    stride_out_batch, stride_out_m, stride_out_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)  # row block
    pid_n = tl.program_id(1)  # col block
    pid_b = tl.program_id(2)  # batch

    # Compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Compute masks
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Load left tile (BLOCK_M, BLOCK_K)
    left_ptrs = left_ptr + pid_b * stride_left_batch + offs_m[:, None] * stride_left_m + offs_k[None, :] * stride_left_k
    left = tl.load(left_ptrs, mask=mask_m[:, None] & (offs_k[None,:] < K), other=0.0)

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # load right tile (BLOCK_K, BLOCK_N)
        right_ptrs = right_ptr + pid_b * stride_right_batch + offs_k[:, None] * stride_right_k + offs_n[None,:] * stride_right_n
        right = tl.load(right_ptrs, mask=(offs_k[:, None] < K) & mask_n[None,:], other=0.0)

        # dot product
        acc += tl.dot(left, right)  # left: [BLOCK_M,BLOCK_K], right->[BLOCK_K,BLOCK_N] -> [BLOCK_M,BLOCK_N] in accum dtype

        # advance pointers
        left_ptrs += BLOCK_K * stride_left_k
        right_ptrs += BLOCK_K * stride_right_k
        # Also update left representation (load next left block)
        left = tl.load(left_ptrs, mask=mask_m[:, None] & ((offs_k + (k+1)*BLOCK_K) < K), other=0.0)

    # Write out
    out_ptrs = out_ptr + pid_b * stride_out_batch + offs_m[:, None] * stride_out_m + offs_n[None,:] * stride_out_n
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None,:])

"""
Highly‑optimized Triton + PyTorch implementation of the *outgoing* TriMul
operator (AlphaFold‑3).  

Algorithm
---------
1️⃣ Row‑wise LayerNorm over the input channel dimension C.  The kernel
   computes mean/variance in fp32 and writes the normalized result as fp16.

2️⃣ Fuse the five Linear layers (left/right projections + three gates) into a
   single fp16 GEMM by concatenating their weight matrices.  The result is split
   back into the individual tensors.

3️⃣ If a pair‑wise mask is present, apply it to the left/right projections.

4️⃣ Apply sigmoid to the three gate tensors and element‑wise gate the left/right
   projections.

5️⃣ Triangular multiplicative update (the expensive O(N³·H) step):
        out_h[i,j] = Σ_k left_h[i,k] * right_h[j,k]
   This is a batched matrix multiply `left @ rightᵀ` for every hidden channel.
   It is implemented as a custom Triton kernel (batched GEMM) that works on
   fp16 inputs and accumulates in fp32.

6️⃣ Fuse the second LayerNorm (over the hidden dimension H) with the final
   output‑gate multiplication in a single Triton kernel.

7️⃣ Linear projection back to the original channel dimension C with a tiny
   fp16 GEMM (torch.nn.functional.linear) and cast the result to fp32.

All heavy reductions run in Triton; the dense linear algebra uses Tensor‑Core
GEMMs (cuBLAS) via `torch.nn.functional.linear` and the custom batched GEMM.
The implementation works for all test configurations listed in the prompt.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernels
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,                # *const float*   [total_rows, dim]   (fp32 or fp16 input)
    y_ptr,                # *mut   half*    [total_rows, dim]   (fp16 output)
    weight_ptr,           # *const float*   [dim]               (γ)
    bias_ptr,             # *const float*   [dim]               (β)
    total_rows,           # int (runtime)
    dim: tl.constexpr,    # int (compile‑time)
    eps: tl.constexpr,    # float (compile‑time)
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """Row‑wise LayerNorm that writes fp16 output."""
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)                 # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ---------- first pass : mean & variance (float32) ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)           # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)               # [BLOCK_M, BLOCK_D]
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                                 # 1 / sqrt(var + eps)

    # ---------- second pass : write normalized values ----------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]          # x̂

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # β

        out = norm * w[None, :] + b[None, :]                   # fp32
        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]

        tl.store(out_ptr,
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


@triton.jit
def layernorm_gate_fp16_kernel(
    x_ptr,                # *const half*   [total_rows, dim]
    gate_ptr,             # *const half*   [total_rows, dim]   (output‑gate)
    y_ptr,                # *mut   half*   [total_rows, dim]   (LN·γ·β·gate)
    weight_ptr,           # *const float*  [dim]               (γ of LN)
    bias_ptr,             # *const float*  [dim]               (β of LN)
    total_rows,           # int (runtime)
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """LayerNorm + output‑gate (γ·x̂+β)·gate, writes fp16."""
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # ---- mean & variance ------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- write normalized *gate -----------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        x = tl.load(x_ptr + offs_m[:, None] * dim + offs_d[None, :],
                    mask=mask_m[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)               # pre‑norm
        g = tl.load(gate_ptr + offs_m[:, None] * dim + offs_d[None, :],
                    mask=mask_m[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)               # gate

        norm = (x - mean[:, None]) * rstd[:, None]            # x̂

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β

        out = (norm * w[None, :] + b[None, :]) * g            # (γ·x̂+β)·gate
        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr,
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


@triton.jit
def trmul_kernel(
    left_ptr,            # *const half*   [batch, M, K]   (M = N, K = N)
    right_ptr,           # *const half*   [batch, K, N]   (right already transposed)
    out_ptr,             # *mut   half*   [batch, M, N]
    batch,               # int (runtime)
    M: tl.constexpr,     # rows of left (== N)
    N: tl.constexpr,     # cols of right (== N)
    K: tl.constexpr,     # reduction dim (== N)
    stride_left_batch,   # stride of batch dimension in left
    stride_left_m,       # stride of row dimension in left
    stride_left_k,       # stride of column (k) dimension in left
    stride_right_batch,  # stride of batch dimension in right
    stride_right_k,      # stride of k dimension in right
    stride_right_n,      # stride of column dimension in right
    stride_out_batch,    # stride of batch dimension in out
    stride_out_m,        # stride of row dimension in out
    stride_out_n,        # stride of col dimension in out
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM for the core TriMul update:
        out[b] = left[b] @ right[b]      # left: (M,K), right: (K,N)
    All tensors are fp16, accumulation in fp32.
    """
    pid_m = tl.program_id(0)      # block row
    pid_n = tl.program_id(1)      # block col
    pid_b = tl.program_id(2)      # batch index

    # offsets within M/N dimensions
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < K

        # left tile: [BLOCK_M, BLOCK_K]
        left_ptrs = left_ptr + pid_b * stride_left_batch \
                    + offs_m[:, None] * stride_left_m \
                    + k_offset[None, :] * stride_left_k
        left = tl.load(left_ptrs,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0)

        # right tile: [BLOCK_K, BLOCK_N]
        right_ptrs = right_ptr + pid_b * stride_right_batch \
                     + k_offset[:, None] * stride_right_k \
                     + offs_n[None, :] * stride_right_n
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_n[None, :],
                        other=0.0)

        acc += tl.dot(left, right)

    # write result back as fp16
    out_ptrs = out_ptr + pid_b * stride_out_batch \
               + offs_m[:, None] * stride_out_m \
               + offs_n[None, :] * stride_out_n
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, C] (float32, on the same CUDA device as the inputs)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                     # C
    hidden_dim = config["hidden_dim"]       # H
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", True)

    B, N, _, C = input_tensor.shape
    assert C == dim, "config['dim'] must match input last dimension"

    # ------------------------------------------------------------------
    # 1️⃣  LayerNorm over C → fp16
    # ------------------------------------------------------------------
    total_rows = B * N * N
    x_flat = input_tensor.contiguous().view(total_rows, dim)          # fp32
    x_norm_fp16 = torch.empty((total_rows, dim), dtype=torch.float16,
                              device=device)

    BLOCK_M_LN = 256   # rows per program (tune if needed)
    BLOCK_D_LN = 32    # columns per iteration
    grid_ln = (triton.cdiv(total_rows, BLOCK_M_LN),)

    layernorm_fp16_kernel[grid_ln](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M_LN,
        BLOCK_D=BLOCK_D_LN,
    )
    torch.cuda.synchronize()

    # ------------------------------------------------------------------
    # 2️⃣  Fused linear projections + gates (all fp16)
    # ------------------------------------------------------------------
    # concatenate the five weight matrices (shape: (5*H, C))
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).half().contiguous()                     # (5*H, C)

    proj_all = F.linear(x_norm_fp16, w_concat)      # fp16, shape [total_rows, 5*H]

    left_proj_fp16, right_proj_fp16, left_gate_fp16, right_gate_fp16, out_gate_fp16 = \
        torch.split(proj_all, hidden_dim, dim=1)

    # ------------------------------------------------------------------
    # 2a️⃣  Optional mask (fp16)
    # ------------------------------------------------------------------
    if not nomask:
        mask_flat = mask_tensor.contiguous().view(total_rows, 1).to(torch.float16)
        left_proj_fp16 = left_proj_fp16 * mask_flat
        right_proj_fp16 = right_proj_fp16 * mask_flat

    # ------------------------------------------------------------------
    # 2b️⃣  Gates (sigmoid) and element‑wise gating
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_fp16)      # [total_rows, H]
    right_gate = torch.sigmoid(right_gate_fp16)
    out_gate = torch.sigmoid(out_gate_fp16)

    left_fp16  = left_proj_fp16 * left_gate
    right_fp16 = right_proj_fp16 * right_gate

    # reshape to [B, N, N, H] for the next step
    left = left_fp16.view(B, N, N, hidden_dim)          # [B, N, N, H]
    right = right_fp16.view(B, N, N, hidden_dim)
    out_gate = out_gate.view(B, N, N, hidden_dim)       # used later

    # ------------------------------------------------------------------
    # 3️⃣  Batched GEMM: out_h = left_h @ right_hᵀ   (Triangular update)
    # ------------------------------------------------------------------
    # Prepare contiguous tensors with shape (B*H, N, N) where right is already transposed
    left_mat = left.permute(0, 3, 1, 2).contiguous()          # (B, H, N, N)
    right_mat = right.permute(0, 3, 2, 1).contiguous()        # (B, H, N, N) → (batch, K, N)

    batch = B * hidden_dim
    out_mat_fp16 = torch.empty_like(left_mat)                 # (B*H, N, N)

    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32
    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        batch,
    )

    trmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat_fp16,
        batch,
        N,                     # M
        N,                     # N
        N,                     # K
        left_mat.stride(0),    # batch stride
        left_mat.stride(1),    # row stride (M)
        left_mat.stride(2),    # col/k stride (K)
        right_mat.stride(0),   # batch stride
        right_mat.stride(1),   # k stride
        right_mat.stride(2),   # col stride (N)
        out_mat_fp16.stride(0),
        out_mat_fp16.stride(1),
        out_mat_fp16.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    torch.cuda.synchronize()

    # reshape back to [B, N, N, H]
    out = out_mat_fp16.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 4️⃣  Fuse second LayerNorm with the output‑gate
    # ------------------------------------------------------------------
    total_rows_h = B * N * N
    out_flat = out.view(total_rows_h, hidden_dim)                # [total_rows_h, H]
    out_gate_flat = out_gate.view(total_rows_h, hidden_dim)      # same shape

    fused_fp16 = torch.empty_like(out_flat)                      # fp16 buffer

    grid2 = (triton.cdiv(total_rows_h, BLOCK_M_LN),)
    layernorm_gate_fp16_kernel[grid2](
        out_flat,
        out_gate_flat,
        fused_fp16,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows_h,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M_LN,
        BLOCK_D=BLOCK_D_LN,
    )
    torch.cuda.synchronize()

    # ------------------------------------------------------------------
    # 5️⃣  Final linear projection back to C (fp16 GEMM) → cast to fp32
    # ------------------------------------------------------------------
    to_out_weight_fp16 = weights["to_out.weight"].half()          # (C, H)
    final_fp16 = F.linear(fused_fp16, to_out_weight_fp16)        # [total_rows_h, C] fp16
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
