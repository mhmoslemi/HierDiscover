    where `R = B·N·N`.  Slicing yields the projected tensors and gate
    vectors.

3️⃣  Apply the optional mask (broadcast over the hidden dimension) on the
    two projected tensors and then element‑wise sigmoids for each gate.

4️⃣  Triangular multiplicative update:
    `out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]`.
    This is a batched matrix multiplication of the hidden‑slices:
    for each hidden channel `d` we compute `left[...,d] @ right[...,d].T`
    via `torch.bmm`.  This uses cuBLAS Tensor‑Cores (FP16).

5️⃣  Second LayerNorm over the hidden dimension (`hidden_dim`).  Again a
    Triton kernel that writes FP16 results.

6️⃣  Multiply by the output gate, project back to the original dimension
    with a final half‑precision linear layer and cast the result to FP32.

All heavy arithmetic is performed in FP16, exploiting H100 Tensor‑Core
throughput, while the custom LayerNorm kernels eliminate the PyTorch
mean/var reductions.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton LayerNorm (row‑wise, works for FP32 or FP16 input, writes FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,            # const *  [total_rows, dim]   (float32 or float16)
    y_ptr,            # mutable *[total_rows, dim]   (float16 output)
    weight_ptr,       # const *  [dim]              (float32)
    bias_ptr,         # const *  [dim]              (float32)
    total_rows,       # int (runtime)
    dim: tl.constexpr,    # compile‑time constant
    eps: tl.constexpr,    # compile‑time constant
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # -----------------------------------------------------------------
    # 1st pass: compute per‑row sum and sum‑of‑squares
    # -----------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs, mask=mask_m[:, None] & mask_d[None, :], other=0.0)
        vals_f32 = vals.to(tl.float32)

        sum_val += tl.sum(vals_f32, axis=1)
        sum_sq  += tl.sum(vals_f32 * vals_f32, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                # 1 / sqrt(var + eps)

    # -----------------------------------------------------------------
    # 2nd pass: write normalized values * γ + β (FP16)
    # -----------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs, mask=mask_m[:, None] & mask_d[None, :], other=0.0)
        vals_f32 = vals.to(tl.float32)

        norm = (vals_f32 - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ  (float32)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β  (float32)
        out = norm * w[None, :] + b[None, :]                     # fp32

        out_fp16 = out.to(tl.float16)
        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16, mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom forward kernel
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [B, N, N, dim] (float32, on CUDA)
    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                 # input channel dimension
    hidden_dim = config["hidden_dim"]   # projection dimension (H)
    eps = 1e-5
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N  # R = B·N·N

    # --------------------------------------------------------------
    # 1) First LayerNorm (row‑wise) → FP16
    # --------------------------------------------------------------
    # Flatten to 2‑D [R, dim] and make contiguous
    x_flat = input_tensor.view(total_rows, dim).contiguous()
    # Allocate FP16 buffer for normalized tensor
    x_norm_fp16 = torch.empty(total_rows, dim, dtype=torch.float16, device=device)

    BLOCK_M = 256
    BLOCK_D = 32
    grid = (triton.cdiv(total_rows, BLOCK_M),)

    layernorm_kernel[grid](
        x_flat,                     # input (float32)
        x_norm_fp16,                # output (fp16)
        weights["norm.weight"],     # γ (float32)
        weights["norm.bias"],       # β (float32)
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # --------------------------------------------------------------
    # 2) Fuse five linear layers (proj + 3 gates) into a single GEMM
    # --------------------------------------------------------------
    # concatenate weights: [5*hidden, dim]
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(device, dtype=torch.float16)

    # Single half‑precision linear transformation
    # x_norm_fp16 : [R, dim]   →   out_concat : [R, 5*hidden]
    out_concat = F.linear(x_norm_fp16, w_concat)   # bias=None

    hs = hidden_dim
    left_proj   = out_concat[:, 0:hs]                 # [R, H]
    right_proj  = out_concat[:, hs:2*hs]              # [R, H]
    left_gate_p = out_concat[:, 2*hs:3*hs]            # [R, H]
    right_gate_p= out_concat[:, 3*hs:4*hs]            # [R, H]
    out_gate_p  = out_concat[:, 4*hs:5*hs]            # [R, H]

    # Sigmoid gates (still in fp16)
    left_gate   = torch.sigmoid(left_gate_p)
    right_gate  = torch.sigmoid(right_gate_p)
    out_gate    = torch.sigmoid(out_gate_p)

    # Optional mask (broadcast over hidden dim)
    if not nomask:
        # mask shape [B, N, N] → [R, 1]
        mask_h = mask_tensor.to(dtype=torch.float16, device=device).view(total_rows, 1)
        left_proj  = left_proj * mask_h
        right_proj = right_proj * mask_h

    # Apply gates
    left  = left_proj * left_gate          # [R, H]
    right = right_proj * right_gate        # [R, H]

    # Reshape back to 4‑D tensors [B, N, N, H]
    left  = left.view(B, N, N, hs)
    right = right.view(B, N, N, hs)
    out_gate = out_gate.view(B, N, N, hs)   # needed later

    # --------------------------------------------------------------
    # 3) Triangular multiplicative update (batched GEMM over hidden)
    # --------------------------------------------------------------
    # Rearrange to [B*H, N, N] to use torch.bmm efficiently
    # left:  [B, N, N, H] → permute → [B, H, N, N] → reshape → [B*H, N, N]
    left_flat = left.permute(0, 3, 1, 2).reshape(B * hs, N, N)   # fp16
    right_flat = right.permute(0, 3, 1, 2).reshape(B * hs, N, N) # fp16

    # Batched matrix multiplication: A @ B^T
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))   # [B*H, N, N] fp16

    # Restore shape to [B, N, N, H]
    out = out_flat.view(B, hs, N, N).permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]

    # --------------------------------------------------------------
    # 4) Second LayerNorm over hidden dimension (fp16)
    # --------------------------------------------------------------
    out_flat2 = out.view(total_rows, hs).contiguous()
    out_norm_fp16 = torch.empty_like(out_flat2, dtype=torch.float16)

    grid_ln2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid_ln2](
        out_flat2,
        out_norm_fp16,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        total_rows,
        hs,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    out_norm = out_norm_fp16.view(B, N, N, hs)   # [B, N, N, H] fp16

    # --------------------------------------------------------------
    # 5) Apply output gate and final projection back to `dim`
    # --------------------------------------------------------------
    out_norm = out_norm * out_gate                 # element‑wise fp16

    # Flatten for final linear layer
    out_norm_flat = out_norm.view(total_rows, hs)

    final_half = F.linear(
        out_norm_flat,
        weights["to_out.weight"].to(device, dtype=torch.float16)  # [dim, H]
    )  # fp16

    # Cast back to FP32 as required by the API
    final = final_half.view(B, N, N, dim).to(torch.float32)

    return final
```
