@triton.jit
def mask_gate_kernel(
    left_proj_ptr, right_proj_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    left_out_ptr, right_out_ptr,
    total_rows, hidden_dim,
    stride_lpr, stride_lpc,
    stride_rpr, stride_rpc,
    stride_lgr, stride_lgc,
    stride_rgr, stride_rgc,
    stride_mask_r,
    stride_lor, stride_loc,
    stride_ror, stride_roc,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_oob = offs_m >= total_rows

    for d in range(0, tl.cdiv(hidden_dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        col_oob = offs_d >= hidden_dim

        # load mask per row only once per iteration? It repeats for each column; that's fine.
        mask_val = tl.load(mask_ptr + offs_m * stride_mask_r, mask=~mask_oob, other=0.0)[:, None] # shape [BLOCK_M,1]

        left_val = tl.load(left_proj_ptr + offs_m[:, None] * stride_lpr + offs_d[None, :] * stride_lpc,
                            mask=~mask_oob[:, None] & ~col_oob[None, :], other=0.0)
        left_gate_val = tl.load(left_gate_ptr + offs_m[:, None] * stride_lgr + offs_d[None, :] * stride_lgc,
                            mask=~mask_oob[:, None] & ~col_oob[None, :], other=0.0)
        right_val = tl.load(right_proj_ptr + offs_m[:, None] * stride_rpr + offs_d[None, :] * stride_rpc,
                            mask=~mask_oob[:, None] & ~col_oob[None, :], other=0.0)
        right_gate_val = tl.load(right_gate_ptr + offs_m[:, None] * stride_rgr + offs_d[None, :] * stride_rgc,
                            mask=~mask_oob[:, None] & ~col_oob[None, :], other=0.0)

        # apply mask and gates
        left_out = left_val * left_gate_val * mask_val

"""
Highly‑optimized forward pass of the *outgoing* TriMul operator (AlphaFold‑3).

Algorithm
---------
1️⃣  Layer‑Norm the input tensor (shape [B,N,N,C]) with the native
    torch.nn.functional.layer_norm – this kernel computes mean/var in a
    single pass and is heavily tuned for NVIDIA GPUs (H100).

2️⃣  Fuse the five linear projections (left/right and three gates) into one
    half‑precision GEMM by concatenating their weight matrices.
    The result is split back into the individual tensors.

3️⃣  If a pair‑wise mask is present, apply it together with the two gate
    tensors in a single Triton kernel (mask + sigmoid(gate) + element‑wise
    multiplication).  When no mask is required we simply multiply the
    projections by their gates with a few torch ops.

4️⃣  Triangular multiplicative update:
        out_h[i,j] = Σ_k left_h[i,k] * right_h[j,k]
    Implemented as a batched half‑precision matrix‑multiply
    (torch.bmm) after reshaping to (B·H,N,N).

5️⃣  Layer‑Norm over the hidden dimension H using torch.nn.functional.layer_norm,
    then multiply by the output‑gate (sigmoid).

6️⃣  Final linear projection back to the original channel dimension C
    (half‑precision GEMM) and cast the result to float‑32.

Only the mask‑and‑gate fusion is written in Triton; all heavy
linear‑algebra and normalisation steps rely on the highly‑optimised
PyTorch/cuBLAS kernels.  This reduces kernel‑launch overhead
and memory traffic, achieving sub‑millisecond runtimes on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: mask + sigmoid(gate) + element‑wise multiplication
# ----------------------------------------------------------------------
@triton.jit
def apply_mask_gate_kernel(
    # input tensors
    left_proj_ptr, right_proj_ptr,          # [total_rows, hidden_dim]
    left_gate_ptr, right_gate_ptr,          # [total_rows, hidden_dim]
    mask_ptr,                               # [total_rows]  (scalar per row)
    # output tensors
    left_out_ptr, right_out_ptr,            # [total_rows, hidden_dim]
    # meta‑parameters
    total_rows, hidden_dim,
    # strides (row‑major contiguous)
    stride_lp_r, stride_lp_c,
    stride_rp_r, stride_rp_c,
    stride_lg_r, stride_lg_c,
    stride_rg_r, stride_rg_c,
    stride_mask_r,
    stride_lo_r, stride_lo_c,
    stride_ro_r, stride_ro_c,
    # compile‑time block sizes
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    """Fuse mask, sigmoid‑gate and element‑wise product.

    left_out = left_proj * sigmoid(left_gate) * mask
    right_out = right_proj * sigmoid(right_gate) * mask
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)               # [BLOCK_M]
    row_mask = offs_m < total_rows

    for d in range(0, tl.cdiv(hidden_dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)         # [BLOCK_D]
        col_mask = offs_d < hidden_dim

        # ---- mask (scalar per row, broadcasted) ----
        mask_val = tl.load(mask_ptr + offs_m * stride_mask_r,
                           mask=row_mask,
                           other=0.0)[:, None]            # [BLOCK_M,1]

        # ---- load tensors ----
        left = tl.load(
            left_proj_ptr + offs_m[:, None] * stride_lp_r + offs_d[None, :] * stride_lp_c,
            mask=row_mask[:, None] & col_mask[None, :],
            other=0.0,
        )
        right = tl.load(
            right_proj_ptr + offs_m[:, None] * stride_rp_r + offs_d[None, :] * stride_rp_c,
            mask=row_mask[:, None] & col_mask[None, :],
            other=0.0,
        )
        left_gate = tl.load(
            left_gate_ptr + offs_m[:, None] * stride_lg_r + offs_d[None, :] * stride_lg_c,
            mask=row_mask[:, None] & col_mask[None, :],
            other=0.0,
        )
        right_gate = tl.load(
            right_gate_ptr + offs_m[:, None] * stride_rg_r + offs_d[None, :] * stride_rg_c,
            mask=row_mask[:, None] & col_mask[None, :],
            other=0.0,
        )

        # ---- sigmoid (half‑precision) ----
        left_gate_sig = tl.sigmoid(left_gate)
        right_gate_sig = tl.sigmoid(right_gate)

        # ---- fused element‑wise multiplication ----
        left_out = left * left_gate_sig * mask_val
        right_out = right * right_gate_sig * mask_val

        # ---- store results ----
        tl.store(
            left_out_ptr + offs_m[:, None] * stride_lo_r + offs_d[None, :] * stride_lo_c,
            left_out,
            mask=row_mask[:, None] & col_mask[None, :],
        )
        tl.store(
            right_out_ptr + offs_m[:, None] * stride_ro_r + offs_d[None, :] * stride_ro_c,
            right_out,
            mask=row_mask[:, None] & col_mask[None, :],
        )

        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).half().contiguous()                       # (5*H, C)

    total_rows = B * N * N
    x_flat = x_norm.view(total_rows, dim)       # [total_rows, C]
    proj_all = F.linear(x_flat, concat_weight)  # [total_rows, 5*H] in fp16

    # Split back into the five tensors
    splits = torch.split(proj_all, hidden_dim, dim=1)
    left_proj, right_proj, left_gate, right_gate, out_gate_proj = splits

    # --------------------------------------------------------------
    # 3️⃣  Apply mask + sigmoid gates (fused Triton kernel)
    # --------------------------------------------------------------
    if not nomask:
        # mask -> [total_rows] (scalar per pair)
        mask_flat = mask_tensor.contiguous().view(total_rows).to(torch.float16)

        left_out = torch.empty_like(left_proj)
        right_out = torch.empty_like(right_proj)

        # Triton launch
        BLOCK_M = 128               # rows per program (tuned for H100)
        BLOCK_D = 128               # hidden_dim fits exactly → single iteration
        grid = (triton.cdiv(total_rows, BLOCK_M),)
        apply_mask_gate_kernel[grid](
            left_proj, right_proj,
            left_gate, right_gate,
            mask_flat,
            left_out, right_out,
            total_rows, hidden_dim,
            left_proj.stride(0), left_proj.stride(1),
            right_proj.stride(0), right_proj.stride(1),
            left_gate.stride(0), left_gate.stride(1),
            right_gate.stride(0), right_gate.stride(1),
            mask_flat.stride(0),
            left_out.stride(0), left_out.stride(1),
            right_out.stride(0), right_out.stride(1),
            BLOCK_M=BLOCK_M,
            BLOCK_D=BLOCK_D,
        )
        torch.cuda.synchronize()
    else:
        # No mask → simple element‑wise gating in PyTorch
        left_out = left_proj * torch.sigmoid(left_gate)
        right_out = right_proj * torch.sigmoid(right_gate)

    # Output‑gate (sigmoid) – always required
    out_gate = torch.sigmoid(out_gate_proj)      # [total_rows, H]

    # --------------------------------------------------------------
    # 4️⃣  Reshape tensors for the triangular multiplicative update
    # --------------------------------------------------------------
    left = left_out.view(B, N, N, hidden_dim)    # [B, N, N, H]
    right = right_out.view(B, N, N, hidden_dim)  # [B, N, N, H]
    out_gate = out_gate.view(B, N, N, hidden_dim)

    # Batched GEMM: out_h[i,j] = Σ_k left_h[i,k] * right_h[j,k]
    #   left_mat : (B*H, N, N)  (i as rows, k as cols)
    #   right_mat: (B*H, N, N)  (k as rows, j as cols) → permute to (k,j)
    left_mat = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)      # (B*H, N, N)
    right_mat = right.permute(0, 3, 2, 1).reshape(B * hidden_dim, N, N)    # (B*H, N, N) with (k,j)
    out_mat = torch.bmm(left_mat, right_mat)                               # (B*H, N, N) fp16

    # Back to [B, N, N, H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 5️⃣  Second LayerNorm + output‑gate (native torch)
    # --------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate

    # --------------------------------------------------------------
    # 6️⃣  Final linear projection back to C and cast to float32
    # --------------------------------------------------------------
    final_fp16 = F.linear(
        out.view(-1, hidden_dim),
        weights["to_out.weight"].half(),
    )                                          # [total_rows, C] fp16
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
```
