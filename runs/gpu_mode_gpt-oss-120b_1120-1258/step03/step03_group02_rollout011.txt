@triton.jit
def trimul_outgoing_kernel(
    left_ptr,    # FP16* shape [N, N, H] (i, k, h)
    right_ptr,   # FP16* shape [N, N, H] (j, k, h) B=?? we treat separate
    out_ptr,    # FP16* shape [N, N, H]
    N: tl.constexpr,
    H: tl.constexpr,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_h = tl.program_id(2)

    i_start = pid_i * BLOCK_I
    j_start = pid_j * BLOCK_J
    h_start = pid_h * BLOCK_H

    # Offsets for i, j, h dims
    offs_i = i_start + tl.arange(0, BLOCK_I)
    offs_j = j_start + tl.arange(0, BLOCK_J)
    offs_h = h_start + tl.arange(0, BLOCK_H)

    # Masks for out-of-bound
    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_h = offs_h < H

    # Initialize accumulator (float32)
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)

    # Loop over k dimension in blocks
    for k_start in range(0, N, BLOCK_K):
        # load left block: shape (BLOCK_I, BLOCK_K, BLOCK_H)
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Compute left offsets: i, k, h -> stride_i * i + stride_k * k + stride_h * h
        # Need stride info. We'll get stride_i, stride_k, stride_h from arguments? Instead we can compute based on pointer shape: left_ptr points to contiguous array with dimensions N, N, H. So layout: offset = i*stride_i + k*stride_k + h*stride_h. For row-major, stride_h = 1, stride_k = H, stride_i = N*H. Actually shape (i,k,h) meaning N (i) as first dimension, N (k) as second, H (h) as third. In memory contiguous, innermost dimension is h. So stride_h = 1, stride_k = H, stride_i = N*H.

        # So left addresses: left_ptr + offs_i[:, None, None] * (N*H) + offs_k[None, :, None] * H + offs_h[None, None, :]

        # Similarly for right shape (j,k,h) where j is first dimension, k second, h third. So stride_j = N*H, stride_k = H, stride_h = 1.

        # Build offsets accordingly.

        left_ptrs = left_ptr + offs_i[:, None, None] * (N * H) + offs_k[None, :, None] * H + offs_h[None, None, :]

        right_ptrs = right_ptr + offs_j[:, None, None] * (N * H) + offs_k[None, :, None] * H + offs_h[None, None, :]

        left_block = tl.load(left_ptrs, mask=(mask_i[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :]), other=0.0).to(tl.float32)

        right_block = tl.load(right_ptrs, mask=(mask_j[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :]), other=0.0).to(tl.float32)

        # For each k in BLOCK_K, accumulate outer product over i,j for each h
        # Vectorized: for each h, we need left_block[:, kk, h] (BLOCK_I), right_block[:, kk, h] (BLOCK_J)

        # Loop across k in compile time:
        for kk in range(0, BLOCK_K):
            # mask for k out-of-bounds may be needed; mask_k_kk = (k_start + kk) < N
            mask_kk = (k_start + kk) < N
            # Using broadcasted multiplication and accumulation
            l_slice = left_block[:, kk, :]    # shape (BLOCK_I, BLOCK_H)
            r_slice = right_block[:, kk, :]   # shape (BLOCK_J, BLOCK_H)

            # compute outer product per hidden dim: broadcast multiply

            # r_slice[None, :, :] shape (1, BLOCK_J, BLOCK_H)
            # product shape (BLOCK_I, BLOCK_J, BLOCK_H)

@triton.jit
def trimul_outgoing_kernel(
    left_ptr,
    right_ptr,
    out_ptr,
    stride_left_i,
    stride_left_k,
    stride_left_h,
    stride_right_i,
    stride_right_k,
    stride_right_h,
    stride_out_i,
    stride_out_k,
    stride_out_h,
    N: tl.constexpr,
    H: tl.constexpr,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_h = tl.program_id(2)

    i_start = pid_i * BLOCK_I
    j_start = pid_j * BLOCK_J
    h_start = pid_h * BLOCK_H

    offs_i = i_start + tl.arange(0, BLOCK_I)
    offs_j = j_start + tl.arange(0, BLOCK_J)
    offs_h = h_start + tl.arange(0, BLOCK_H)

    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_h = offs_h < H

    # accumulator
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)

    # loop over k dimension
    for k_start in range(0, N, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # compute pointer offsets for left and right
        left_ptrs = left_ptr + \
            offs_i[:, None, None] * stride_left_i + \
            offs_k[None, :, None] * stride_left_k + \
            offs_h[None, None, :] * stride_left_h

        right_ptrs = right_ptr + \
            offs_j[:, None, None] * stride_right_i + \
            offs_k[None, :, None] * stride_right_k + \
            offs_h[None, None, :] * stride_right_h

        left_block = tl.load(
            left_ptrs,
            mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
            other=0.0,
        ).to(tl.float32)

        right_block = tl.load(
            right_ptrs,
            mask=mask_j[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
            other=0.0,
        ).to(tl.float32)

        # inner loop over k within block
        for kk in range(0, BLOCK_K):
            # l_slice shape (BLOCK_I, BLOCK_H)
            l_slice = left_block[:, kk, :]   # (BLOCK_I, BLOCK_H)
            r_slice = right_block[:, kk, :]  # (BLOCK_J, BLOCK_H)

            # compute outer product per hidden dim and accumulate
            # broadcast multiply
            # shape (BLOCK_I, BLOCK_J, BLOCK_H)
            acc += l_slice[:, None, :] * r_slice[None, :, :]

    # write out
    out_ptrs = out_ptr + \
        offs_i[:, None, None] * stride_out_i + \
        offs_j[None, :, None] * stride_out_k + \
        offs_h[None, None, :] * stride_out_h

    tl.store(
        out_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
    )

"""
TriMul “outgoing” implementation (AlphaFold‑3 style) – Triton‑accelerated.

Algorithm
---------
1️⃣ Row‑wise LayerNorm (FP32 → FP16) – tiny Triton kernel that computes μ,
   σ² per row and writes the normalized values directly in half‑precision.
2️⃣ Fuse the two linear projections (proj + gate) per side into a single
   FP16 GEMM each (4 GEMMs total).  The resulting tensors are still in
   [B, N, N, H] format (H = hidden_dim).
3️⃣ Optional mask (broadcast over H) is applied in‑place.
4️⃣ Apply sigmoid‑gates and multiply the gated projections.
5️⃣ Core “tri‑multiplicative” update:
        out(b,i,j,h) = Σₖ left(b,i,k,h) * right(b,j,k,h)
   is performed by a custom Triton kernel that computes many
   (i,j)‑tiles in parallel while looping over the k‑dimension in blocks.
   This avoids the many small batched GEMMs used by a naïve torch.bmm
   implementation and keeps all intermediate data in registers.
6️⃣ Second LayerNorm (FP16 output) – same Triton kernel as step 1.
7️⃣ Multiply by the output‑gate and a final FP16 linear projection back to `dim`.
   The result is cast to FP32 for the module output.

The kernel works for all test configurations listed in the prompt,
supports the presence or absence of a mask, and runs on H100 GPUs
with Triton 3.3.1.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,          # const float*   [total_rows, dim]
    y_ptr,          #       half*   [total_rows, dim] (output)
    w_ptr,          # const float*  [dim]   (γ)
    b_ptr,          # const float*  [dim]   (β)
    total_rows,     # int (runtime)
    dim: tl.constexpr,          # static: number of columns
    eps: tl.constexpr,          # static epsilon
    BLOCK_M: tl.constexpr,      # rows per program
    BLOCK_D: tl.constexpr,      # columns per inner loop
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ----------- pass 1: compute mean & variance --------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                               # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)          # 1 / sqrt(var+eps)

    # ----------- pass 2: write normalized values --------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]   # (x-μ)/σ

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # β

        out = norm * w[None, :] + b[None, :]                # fp32
        out_fp16 = out.to(tl.float16)                       # cast to fp16

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 5️⃣ Tri‑multiplicative update kernel (outgoing version)
# ----------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    left_ptr,               # fp16*  [N, N, H]   (i, k, h)
    right_ptr,              # fp16*  [N, N, H]   (j, k, h)
    out_ptr,                # fp16*  [N, N, H]   (i, j, h)
    stride_left_i, stride_left_k, stride_left_h,
    stride_right_i, stride_right_k, stride_right_h,
    stride_out_i, stride_out_j, stride_out_h,
    N: tl.constexpr,                 # sequence length
    H: tl.constexpr,                 # hidden dimension
    BLOCK_I: tl.constexpr,           # tile size along i
    BLOCK_J: tl.constexpr,           # tile size along j
    BLOCK_K: tl.constexpr,           # tile size along k
    BLOCK_H: tl.constexpr,           # tile size along h
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_h = tl.program_id(2)

    i0 = pid_i * BLOCK_I
    j0 = pid_j * BLOCK_J
    h0 = pid_h * BLOCK_H

    offs_i = i0 + tl.arange(0, BLOCK_I)
    offs_j = j0 + tl.arange(0, BLOCK_J)
    offs_h = h0 + tl.arange(0, BLOCK_H)

    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_h = offs_h < H

    # accumulator in FP32 for numerical stability
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the reduction dimension k in blocks
    # ------------------------------------------------------------------
    for k0 in range(0, N, BLOCK_K):
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ------- load left block (i, k, h) -------
        left_ptrs = left_ptr + \
            offs_i[:, None, None] * stride_left_i + \
            offs_k[None, :, None] * stride_left_k + \
            offs_h[None, None, :] * stride_left_h

        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
            other=0.0,
        ).to(tl.float32)                                    # [BI, BK, BH]

        # ------- load right block (j, k, h) -------
        right_ptrs = right_ptr + \
            offs_j[:, None, None] * stride_right_i + \
            offs_k[None, :, None] * stride_right_k + \
            offs_h[None, None, :] * stride_right_h

        right = tl.load(
            right_ptrs,
            mask=mask_j[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
            other=0.0,
        ).to(tl.float32)                                    # [BJ, BK, BH]

        # ------- inner loop over k inside the block -------
        for kk in range(0, BLOCK_K):
            # left slice: (BI, BH)
            l_slice = left[:, kk, :]               # shape (BLOCK_I, BLOCK_H)
            # right slice: (BJ, BH)
            r_slice = right[:, kk, :]              # shape (BLOCK_J, BLOCK_H)

            # outer product per hidden channel and accumulate
            # shape broadcast -> (BI, BJ, BH)
            acc += l_slice[:, None, :] * r_slice[None, :, :]

    # ------------------------------------------------------------------
    # Store the result (FP16)
    # ------------------------------------------------------------------
    out_ptrs = out_ptr + \
        offs_i[:, None, None] * stride_out_i + \
        offs_j[None, :, None] * stride_out_j + \
        offs_h[None, None, :] * stride_out_h

    tl.store(
        out_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
    )


# ----------------------------------------------------------------------
# ️⃣ Entry point – custom_kernel
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32, on CUDA)
    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32 → FP16)
    # --------------------------------------------------------------
    x_flat = input_tensor.view(total_rows, dim).contiguous()
    x_norm_h = torch.empty((total_rows, dim), dtype=torch.float16, device=device)

    BLOCK_M = 256                     # rows per program
    BLOCK_D = 64                      # columns per inner loop (covers dim up to 384)
    grid_ln1 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid_ln1](
        x_flat,
        x_norm_h,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()
    x_norm_h = x_norm_h.view(B, N, N, dim)          # [B, N, N, dim] (FP16)

    # --------------------------------------------------------------
    # 2️⃣ Fused projection + gate (4 GEMMs total)
    # --------------------------------------------------------------
    # ----- left side -----
    left_concat_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16, non_blocking=True)

    left_concat = F.linear(
        x_norm_h.view(total_rows, dim), left_concat_w
    )  # [total_rows, 2*hidden_dim]

    left_proj_h = left_concat[:, :hidden_dim].view(B, N, N, hidden_dim)
    left_gate_pre_h = left_concat[:, hidden_dim:].view(B, N, N, hidden_dim)

    # ----- right side -----
    right_concat_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16, non_blocking=True)

    right_concat = F.linear(
        x_norm_h.view(total_rows, dim), right_concat_w
    )
    right_proj_h = right_concat[:, :hidden_dim].view(B, N, N, hidden_dim)
    right_gate_pre_h = right_concat[:, hidden_dim:].view(B, N, N, hidden_dim)

    # ----- output gate (once per element) -----
    out_gate_weight = weights["out_gate.weight"].to(torch.float16, non_blocking=True)
    out_gate_h = torch.sigmoid(
        F.linear(x_norm_h.view(total_rows, dim), out_gate_weight)
    ).view(B, N, N, hidden_dim)                # [B, N, N, hidden_dim] (FP16)

    # --------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over hidden dimension)
    # --------------------------------------------------------------
    if not nomask:
        mask_h = mask_tensor.unsqueeze(-1).to(torch.float16)   # [B, N, N, 1]
        left_proj_h = left_proj_h * mask_h
        right_proj_h = right_proj_h * mask_h

    # --------------------------------------------------------------
    # 4️⃣ Apply sigmoid‑gates
    # --------------------------------------------------------------
    left_gate_h = torch.sigmoid(left_gate_pre_h)
    right_gate_h = torch.sigmoid(right_gate_pre_h)

    left_h = left_proj_h * left_gate_h          # [B, N, N, hidden]
    right_h = right_proj_h * right_gate_h

    # --------------------------------------------------------------
    # 5️⃣ Tri‑multiplicative update (custom Triton kernel)
    # --------------------------------------------------------------
    out_h = torch.empty((B, N, N, hidden_dim), dtype=torch.float16, device=device)

    # Tiling parameters – tuned for H100 registers & shared memory
    BLOCK_I = 16
    BLOCK_J = 16
    BLOCK_H = 8
    BLOCK_K = 8

    # Strides for a contiguous tensor of shape (N, N, H) -> (i, k, h)
    stride_i = N * hidden_dim        # i‑stride
    stride_k = hidden_dim            # k‑stride
    stride_h = 1                     # h‑stride (innermost)

    # Launch one kernel per batch (B ≤ 2)
    for b in range(B):
        left_ptr  = left_h[b].contiguous()
        right_ptr = right_h[b].contiguous()
        out_ptr   = out_h[b].contiguous()

        grid = (
            triton.cdiv(N, BLOCK_I),
            triton.cdiv(N, BLOCK_J),
            triton.cdiv(hidden_dim, BLOCK_H),
        )
        trimul_outgoing_kernel[grid](
            left_ptr,
            right_ptr,
            out_ptr,
            stride_i,
            stride_k,
            stride_h,
            stride_i,
            stride_k,
            stride_h,
            stride_i,
            stride_k,
            stride_h,
            N,
            hidden_dim,
            BLOCK_I=BLOCK_I,
            BLOCK_J=BLOCK_J,
            BLOCK_K=BLOCK_K,
            BLOCK_H=BLOCK_H,
        )
    torch.cuda.synchronize()

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm (FP16)
    # --------------------------------------------------------------
    total_rows_h = B * N * N
    out_flat = out_h.view(total_rows_h, hidden_dim).contiguous()
    out_norm_h = torch.empty_like(out_flat, dtype=torch.float16)

    grid_ln2 = (triton.cdiv(total_rows_h, BLOCK_M),)
    layernorm_fp16_kernel[grid_ln2](
        out_flat,
        out_norm_h,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows_h,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()
    out_norm = out_norm_h.view(B, N, N, hidden_dim)   # FP16

    # --------------------------------------------------------------
    # 7️⃣ Apply output‑gate and final projection
    # --------------------------------------------------------------
    out_norm = out_norm * out_gate_h

    to_out_weight = weights["to_out.weight"].to(torch.float16, non_blocking=True)
    final_h = F.linear(
        out_norm.view(total_rows_h, hidden_dim), to_out_weight
    )   # [total_rows_h, dim] (FP16)

    final = final_h.view(B, N, N, dim).to(torch.float32)   # back to FP32

    return final
