"""
TriMul (outgoing) – fast Triton‑augmented forward pass

Algorithm
---------
1. **LayerNorm (input → hidden)**
   Use PyTorch's native `torch.nn.functional.layer_norm` (float32) which is
   heavily tuned on NVIDIA GPUs.  This replaces the first custom Triton LN
   and removes a separate write‑read of the normalized tensor.

2. **Fused projection + gating**
   Concatenate the five weight matrices (`left_proj, left_gate, right_proj,
   right_gate, out_gate`) into a single matrix of shape `(5*H, D)` and call a
   single `F.linear`.  This yields the five intermediate tensors in one
   GEMM (FP16).

3. **Mask & sigmoid**
   Optional mask is applied only to the two projection tensors.
   Gates are passed through a sigmoid and multiplied element‑wise.

4. **Tri‑multiplicative update**
   ```
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   ```
   Implemented as a batched GEMM:
   - reshape `left/right` to `[B*H, N, N]`
   - use `torch.bmm` (Tensor‑Core GEMM) → `[B*H, N, N]`
   - reshape back to `[B, N, N, H]`.

5. **LayerNorm + output gate (hidden → hidden)**
   A small Triton kernel (`layernorm_gate_kernel`) computes
   row‑wise LayerNorm over the hidden dimension, applies the learned
   scale/shift (`to_out_norm.{weight,bias}`) and multiplies by the output
   gate, all in FP16 (float32 accumulators).  The result stays in FP16.

6. **Final linear projection**
   `to_out` projects from hidden (`H`) back to the model dimension (`D`).
   A final `F.linear` (FP16) produces the output which is cast back to
   `float32`.

Only the hidden‑dim LayerNorm + gate is written in Triton – satisfying the
requirement that “part of the operations” be kernel‑based while keeping the
heavy GEMM paths in cuBLAS/Tensor‑Cores for maximal performance.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: hidden‑dim LayerNorm + output‑gate
# ----------------------------------------------------------------------
@triton.jit
def layernorm_gate_kernel(
    x_ptr,          # const half*   [total_rows, H]   (input from tri‑mul)
    y_ptr,          #       half*   [total_rows, H]   (LN‑output * out_gate)
    gate_ptr,       # const half*   [total_rows, H]   (out_gate)
    weight_ptr,     # const half*   [H]               (γ of to_out_norm)
    bias_ptr,       # const half*   [H]               (β of to_out_norm)
    total_rows: tl.constexpr,
    H: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_H: tl.constexpr,   # hidden dimension tile
):
    """
    Row‑wise LayerNorm (float32 accumulator) over the hidden dimension
    followed by an element‑wise multiplication with the output gate.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # -------------------------------------------------------------
    # 1st pass – mean / variance (float32 accumulator)
    # -------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for h in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h * BLOCK_H + tl.arange(0, BLOCK_H)    # [BLOCK_H]
        mask_h = offs_h < H
        ptrs = x_ptr + offs_m[:, None] * H + offs_h[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_h[None, :],
                       other=0.0).to(tl.float32)        # [BLOCK_M, BLOCK_H]

        sum_val += tl.sum(vals, axis=1)                 # row‑wise sum
        sum_sq  += tl.sum(vals * vals, axis=1)          # row‑wise sum of squares

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # -------------------------------------------------------------
    # 2nd pass – normalize, apply γ/β, multiply by gate
    # -------------------------------------------------------------
    for h in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h * BLOCK_H + tl.arange(0, BLOCK_H)    # [BLOCK_H]
        mask_h = offs_h < H

        # load hidden values
        x_vals = tl.load(x_ptr + offs_m[:, None] * H + offs_h[None, :],
                         mask=mask_m[:, None] & mask_h[None, :],
                         other=0.0).to(tl.float32)

        # normalize
        norm = (x_vals - mean[:, None]) * rstd[:, None]

        # γ / β (cast to float32 for the arithmetic)
        w = tl.load(weight_ptr + offs_h, mask=mask_h, other=0.0).to(tl.float32)
        b = tl.load(bias_ptr   + offs_h, mask=mask_h, other=0.0).to(tl.float32)

        out = norm * w[None, :] + b[None, :]

        # gate (half → float32)
        g = tl.load(gate_ptr + offs_m[:, None] * H + offs_h[None, :],
                    mask=mask_m[:, None] & mask_h[None, :],
                    other=0.0).to(tl.float32)

        out = out * g

        # store result as half
        tl.store(y_ptr + offs_m[:, None] * H + offs_h[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_h[None, :])

# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, D] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool / int, optional)
        - weights_dict : key → torch.Tensor (model parameters)
        - config_dict  : {"dim": D, "hidden_dim": H, "nomask": bool}

    Returns
    -------
    torch.Tensor of shape [B, N, N, D] (float32)
    """
    # ------------------------------------------------------------------
    # 0. unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                # D
    hidden_dim = config["hidden_dim"]  # H
    eps = 1e-5
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N  # R = B·N·N

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32, use native implementation)
    # ------------------------------------------------------------------
    # torch.nn.functional.layer_norm is highly optimized on NVIDIA GPUs.
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B,N,N,D] float32

    # ------------------------------------------------------------------
    # 2. Fuse all linear projections (FP16)
    # ------------------------------------------------------------------
    x_norm_h = x_norm.to(torch.float16)          # [R, D] half

    # concatenate the five weight matrices once (shape: (5*H, D))
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)  # (5*H, D)

    # single GEMM → [R, 5*H]
    proj_all = F.linear(x_norm_h, w_concat)      # half

    # ------------------------------------------------------------------
    # 3. Slice, mask and apply sigmoid gates
    # ------------------------------------------------------------------
    H = hidden_dim
    left_proj   = proj_all[:, 0:H]               # [R, H]
    left_gate   = proj_all[:, H:2*H]
    right_proj  = proj_all[:, 2*H:3*H]
    right_gate  = proj_all[:, 3*H:4*H]
    out_gate    = proj_all[:, 4*H:5*H]

    if not nomask:
        # mask shape [B,N,N] -> [R,1] and broadcast over H
        mask_h = mask_tensor.to(torch.float16).view(total_rows, 1)
        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h

    left_gate   = torch.sigmoid(left_gate)
    right_gate  = torch.sigmoid(right_gate)
    out_gate    = torch.sigmoid(out_gate)

    left  = left_proj * left_gate      # [R, H] half
    right = right_proj * right_gate    # [R, H] half

    # ------------------------------------------------------------------
    # 4. Tri‑multiplicative update (batched GEMM)
    # ------------------------------------------------------------------
    # reshape to 4‑D tensors
    left  = left.view(B, N, N, H)   # [B,N,N,H]
    right = right.view(B, N, N, H)  # [B,N,N,H]

    # bring hidden dimension into the batch axis for torch.bmm
    # (no extra copy – we only reshaped after permuting)
    left_mat  = left.permute(0, 3, 1, 2).contiguous()   # [B,H,N,N]
    right_mat = right.permute(0, 3, 2, 1).contiguous()  # [B,H,N,N] (transpose k ↔ j)

    BH = B * H
    left_mat  = left_mat.view(BH, N, N)    # [B*H, N, N]
    right_mat = right_mat.view(BH, N, N)   # [B*H, N, N]

    # batched matrix multiplication; Tensor‑Cores handle FP16 very efficiently
    out_mat = torch.bmm(left_mat, right_mat)   # [B*H, N, N] half

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # half

    # ------------------------------------------------------------------
    # 5. Hidden‑dim LayerNorm + output‑gate (Triton)
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, H)                 # [R, H] half
    out_gate_flat = out_gate.view(total_rows, H)       # [R, H] half

    out_norm_gate = torch.empty_like(out_flat)         # half, final pre‑projection

    # Triton launch configuration – a few hundred rows per program works well
    BLOCK_M = 256          # rows per program
    BLOCK_H = 32           # hidden‑dim tile (H ≤ 384)
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_gate_kernel[grid](
        out_flat,
        out_norm_gate,
        out_gate_flat,
        weights["to_out_norm.weight"].to(torch.float16),   # γ (half) : [H]
        weights["to_out_norm.bias"].to(torch.float16),     # β (half) : [H]
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_H=BLOCK_H,
    )

    # ------------------------------------------------------------------
    # 6. Final output projection back to model dimension
    # ------------------------------------------------------------------
    w_to_out = weights["to_out.weight"].to(torch.float16)   # [D, H]
    final_half = F.linear(out_norm_gate, w_to_out)        # [R, D] half
    final = final_half.view(B, N, N, dim).to(torch.float32)

    return final
