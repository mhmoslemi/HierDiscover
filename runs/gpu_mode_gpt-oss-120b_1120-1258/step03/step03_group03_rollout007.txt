```python
@triton.jit
def layernorm_fp16_kernel(
    x_ptr, y_ptr,
    weight_ptr, bias_ptr,
    total_rows,
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows
    
    # compute sum and sum_sq across dim per row
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)
    
    # loop over dimension blocks
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)   # [BLOCK_M, BLOCK_D]
        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)
    
    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)
    
    # second pass: write normalized values cast to fp16
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        norm = (vals - mean[:, None]) * rstd[:, None]
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0).to(tl.float32)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0).to(tl.float32)
        out = norm * w[None, :] + b[None, :]
        # cast to half and store
        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out.to(tl.float16), mask=mask_m[:, None] & mask_d[None, :])
```

"""
TriMul (outgoing) – fast forward pass for AlphaFold‑3.

Algorithm
---------
1️⃣ Row‑wise LayerNorm (float32 accumulator, FP16 output) on the
   input tensor X ∈ [B,N,N,D].  Implemented as a small Triton kernel
   that computes mean/variance per row and writes (X‑μ)/σ·γ+β in
   half‑precision (reduces memory traffic vs. FP32 → FP16 cast).

2️⃣ Fuse the five linear projections/gates (left_proj, left_gate,
   right_proj, right_gate, out_gate) into a single GEMM:
      P = X̂ @ Wᵀ      where W ∈ [5·H, D] (all weights are FP16).
   The result P ∈ [R,5·H] (R = B·N·N) is sliced into the five tensors.

3️⃣ Apply sigmoid to the three gate slices and element‑wise
   gating of the projection slices.  When a mask is present it is
   multiplied (broadcast) onto the two projection tensors only.

4️⃣ Tri‑multiplicative update:
   left  = left_proj * left_gate   → [R,H] → reshape (B,N,N,H)
   right = right_proj * right_gate → [R,H] → reshape (B,N,N,H)
   For each batch‑hidden pair we compute
        O = L @ Rᵀ   (batched GEMM of size N×N)
   using a single torch.bmm on tensors of shape [B·H, N, N].

5️⃣ LayerNorm over the hidden dimension plus output‑gate:
   A second Triton kernel normalises O (FP32 accumulator), applies
   γ/β (to_out_norm) and multiplies by out_gate – all in one pass,
   writing the result directly in FP16.

6️⃣ Final linear projection back to the original dimension D
   (FP16 GEMM) and cast to FP32 for the module output.

Only the two LayerNorm steps are custom Triton kernels; all large
matrix multiplications use cuBLAS/Tensor‑Core kernels (highly optimal on
H100).  Mixed precision (FP16) is used throughout the heavy work,
keeping the final result in FP32 as required.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernels
# ----------------------------------------------------------------------


@triton.jit
def layernorm_fp16_kernel(
    x_ptr,            # const float*   [total_rows, dim]   (input, FP32)
    y_ptr,            #       half*   [total_rows, dim]   (output, FP16)
    weight_ptr,       # const float*   [dim]               (γ, FP32)
    bias_ptr,         # const float*   [dim]               (β, FP32)
    total_rows,       # i32
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """
    Row‑wise LayerNorm (mean/var in FP32) that writes the normalized
    result in half‑precision.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)            # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ---------- 1st pass: compute mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        )                                                   # FP32
        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                             # 1/√(var+ε)

    # ---------- 2nd pass: write normalized output ----------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        )                                                   # FP32
        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0).to(tl.float32)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0).to(tl.float32)

        out = norm * w[None, :] + b[None, :]                # FP32
        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


@triton.jit
def layernorm_gate_fp16_kernel(
    x_ptr,            # const half*    [total_rows, dim]   (input, FP16)
    y_ptr,            #       half*   [total_rows, dim]   (output, FP16)
    gate_ptr,         # const half*    [total_rows, dim]   (out_gate, FP16)
    weight_ptr,       # const half*    [dim]               (γ, FP16)
    bias_ptr,         # const half*    [dim]               (β, FP16)
    total_rows,       # i32
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """
    LayerNorm over the hidden dimension (FP32 accumulator) followed by
    element‑wise multiplication with a gate tensor.  Result is written
    in half‑precision.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # ---------- mean / var ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                                   # FP16 → FP32
        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- normalize, γ/β, gate ----------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        # load input
        x_ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        x_val = tl.load(
            x_ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (x_val - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0).to(tl.float32)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0).to(tl.float32)

        out = norm * w[None, :] + b[None, :]

        # gate
        g_ptrs = gate_ptr + offs_m[:, None] * dim + offs_d[None, :]
        gate = tl.load(
            g_ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        out = out * gate

        # write back as half
        y_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(y_ptrs, out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, D] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/int) or None
            - weights_dict : mapping from layer names to torch.Tensor
            - config_dict  : {"dim": D, "hidden_dim": H, "nomask": bool}

    Returns:
        torch.Tensor [B, N, N, D] (float32)
    """
    # -------------------------------------------------
    # 1️⃣ Unpack & constants
    # -------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device

    dim = config["dim"]                     # D
    hidden_dim = config["hidden_dim"]       # H
    eps = 1e-5
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N                  # R = B·N·N

    # -------------------------------------------------
    # 2️⃣ First LayerNorm (float32 → half)
    # -------------------------------------------------
    x = input_tensor.contiguous()          # [B,N,N,D]  FP32
    x_flat = x.view(total_rows, dim)       # [R,D]      FP32
    x_norm_half = torch.empty(
        (total_rows, dim), dtype=torch.float16, device=device
    )                                       # [R,D]      FP16 output

    # block sizes tuned for the allowed dimensions (≤ 384)
    BLOCK_M = 1024
    BLOCK_D = 64 if dim > 32 else 32
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        x_flat,
        x_norm_half,
        weights["norm.weight"],            # γ (FP32)
        weights["norm.bias"],              # β (FP32)
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )  # result in x_norm_half (FP16)

    # -------------------------------------------------
    # 3️⃣ Fused projection + gates (single GEMM)
    # -------------------------------------------------
    # concatenate the five (H×D) weight matrices → (5H, D)
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # shape (5*H, D)

    # GEMM: [R, D] @ (5H, D)ᵀ → [R, 5H]  (all FP16)
    proj_all = F.linear(x_norm_half, w_concat)   # [R, 5*H]

    H = hidden_dim
    left_proj   = proj_all[:, 0:H]
    left_gate   = proj_all[:, H:2*H]
    right_proj  = proj_all[:, 2*H:3*H]
    right_gate  = proj_all[:, 3*H:4*H]
    out_gate    = proj_all[:, 4*H:5*H]

    # sigmoid non‑linearity on gates
    left_gate   = torch.sigmoid(left_gate)
    right_gate  = torch.sigmoid(right_gate)
    out_gate    = torch.sigmoid(out_gate)

    # element‑wise gating of the projections
    left  = left_proj * left_gate          # [R, H]  FP16
    right = right_proj * right_gate        # [R, H]  FP16

    # optional mask (broadcast over hidden dim)
    if not nomask and mask_tensor is not None:
        mask_h = mask_tensor.to(torch.float16).view(total_rows, 1)  # [R,1]
        left = left * mask_h
        right = right * mask_h

    # -------------------------------------------------
    # 4️⃣ Tri‑multiplicative update (batched GEMM)
    # -------------------------------------------------
    left = left.view(B, N, N, H)          # [B,N,N,H]
    right = right.view(B, N, N, H)        # [B,N,N,H]

    # bring hidden dim to the batch axis and transpose right for Rᵀ
    left_mat = left.permute(0, 3, 1, 2).contiguous()   # [B,H,N,N]
    right_mat = right.permute(0, 3, 2, 1).contiguous() # [B,H,N,N] (k ↔ j)

    BH = B * H
    left_mat = left_mat.view(BH, N, N)    # [B·H, N, N]
    right_mat = right_mat.view(BH, N, N)  # [B·H, N, N]

    # each batch does L @ Rᵀ   (Tensor‑Core GEMM)
    out_mat = torch.bmm(left_mat, right_mat)          # [B·H, N, N] FP16

    # restore original layout  → [B,N,N,H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # FP16

    # -------------------------------------------------
    # 5️⃣ Second LayerNorm + output‑gate (fused kernel)
    # -------------------------------------------------
    out_flat = out.view(total_rows, H)               # [R, H] FP16
    out_gate_flat = out_gate.view(total_rows, H)     # [R, H] FP16
    out_norm_gate = torch.empty_like(out_flat)       # FP16 buffer

    BLOCK_M2 = 1024
    BLOCK_D2 = 64 if H > 32 else 32
    grid2 = (triton.cdiv(total_rows, BLOCK_M2),)
    layernorm_gate_fp16_kernel[grid2](
        out_flat,
        out_norm_gate,
        out_gate_flat,
        weights["to_out_norm.weight"].to(torch.float16),   # γ (FP16)
        weights["to_out_norm.bias"].to(torch.float16),     # β (FP16)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M2,
        BLOCK_D=BLOCK_D2,
    )  # result in out_norm_gate (FP16)

    # -------------------------------------------------
    # 6️⃣ Final linear projection back to dim
    # -------------------------------------------------
    final_half = F.linear(out_norm_gate, weights["to_out.weight"].to(torch.float16))
    final = final_half.view(B, N, N, dim).to(torch.float32)  # cast to FP32

    return final
