@triton.jit
def fused_ln_gate_linear_kernel(
    out_ptr,   # half* [R, H]
    gate_ptr,  # half* [R, H] (out_gate)
    out_final_ptr, # half* [R, D] (or float32)
    weight_ptr,  # half [D, H]
    bias_ptr,    # half [D]
    R, H, D,
    eps,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < R

    # Compute per-row mean & var
    # Need to sum over H dimension --> loop over h blocks
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_R], dtype=tl.float32)
    for h in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H
        # load out and gate maybe later
        x_ptrs = out_ptr + offs_r[:, None] * H + offs_h[None, :]
        x = tl.load(x_ptrs, mask=mask_r[:, None] & mask_h[None, :], other=0.0).to(tl.float32)
        sum_val += tl.sum(x, axis=1)
        sum_sq += tl.sum(x * x, axis=1)
    mean = sum_val / H
    var = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # Now compute dot product with weight and output
    # We will accumulate for each D tile
    for d in range(0, tl.cdiv(D, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < D
        acc = tl.zeros([BLOCK_R, BLOCK_D], dtype=tl.float32)

        # Loop over H again to compute normalized*gate dot weight
        for h in range(0, tl.cdiv(H, BLOCK_H)):
            offs_h = h * BLOCK_H + tl.arange(0, BLOCK_H)
            mask_h = offs_h < H

            # Load out row slice
            x_ptrs = out_ptr + offs_r[:, None] * H + offs_h[None, :]
            x = tl.load(x_ptrs, mask=mask_r[:, None] & mask_h[None, :], other=0.0).to(tl.float32)

            # Normalize
            norm = (x - mean[:, None]) * rstd[:, None]

            # Load gate
            g_ptrs = gate_ptr + offs_r[:, None] * H + offs_h[None, :]
            g = tl.load(g_ptrs, mask=mask_r[:, None] & mask_h[None, :], other=0.0).to(tl.float32)

            y = norm * g

            # Load weight slice [D,H] at current d block and h block
            # weight_ptr is shape (D, H); we need weight[d, h] slice
            w_ptrs = weight_ptr + offs_d[:, None] * H + offs_h[None, :]  # shape (BLOCK_D, BLOCK_H)
            w = tl.load(w_ptrs, mask=mask_d[:, None] & mask_h[None, :], other=0.0).to(tl.float32)

            # Compute contribution: for each row r, for each output d, accumulate dot(y, w[d])
            # Use tl.dot on vectors: y shape (BLOCK_R, BLOCK_H) and w shape (BLOCK_D, BLOCK_H)
            # We can compute for each d tile:
            # for each row, compute y[ row, :] dot w[d,:] => scalar; accumulate into acc[row,d]
            # Use tl.dot if we reshape y and w accordingly.

@triton.jit
def fused_ln_gate_out_kernel(
    out_ptr, gate_ptr, out_final_ptr,
    weight_ptr, bias_ptr,
    total_rows, H, D,
    eps: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < total_rows

    # First pass: compute mean & var across H
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_R], dtype=tl.float32)
    for h in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H
        # load out (half)
        x_ptrs = out_ptr + offs_r[:, None] * H + offs_h[None, :]
        x = tl.load(x_ptrs, mask=mask_r[:, None] & mask_h[None, :], other=0.0)
        x = x.to(tl.float32)
        sum_val += tl.sum(x, axis=1)
        sum_sq += tl.sum(x * x, axis=1)
    mean = sum_val / H
    var = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # Loop over D blocks
    for d in range(0, tl.cdiv(D, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < D
        # bias slice
        bias = tl.load(bias_ptr + offs_d, mask=mask_d, other=0.0)

        acc = tl.zeros([BLOCK_R, BLOCK_D], dtype=tl.float32)

        for h in range(0, tl.cdiv(H, BLOCK_H)):
            offs_h = h * BLOCK_H + tl.arange(0, BLOCK_H)
            mask_h = offs_h < H

            # load out and gate
            x_ptrs = out_ptr + offs_r[:, None] * H + offs_h[None, :]
            x = tl.load(x_ptrs, mask=mask_r[:, None] & mask_h[None, :], other=0.0).to(tl.float32)

            g_ptrs = gate_ptr + offs_r[:, None] * H + offs_h[None, :]
            g = tl.load(g_ptrs, mask=mask_r[:, None] & mask_h[None, :], other=0.0).to(tl.float32)

            # normalize and gate
            y = (x - mean[:, None]) * rstd[:, None] * g   # shape [BLOCK_R, BLOCK_H]

            # load weight slice [BLOCK_D, BLOCK_H]
            w_ptrs = weight_ptr + offs_d[:, None] * H + offs_h[None, :]
            w = tl.load(w_ptrs, mask=mask_d[:, None] & mask_h[None, :], other=0.0).to(tl.float32)

            # accumulate dot product across H dimension
            # For each row r and output d, acc[r,d] += sum_{h} y[r,h] * w[d,h]
            # Use broadcast and sum over h dimension
            acc += tl.sum(y[:, None, :] * w[None, :, :], axis=2)   # [BLOCK_R, BLOCK_D]

        # Add bias
        out_ptrs = out_final_ptr + offs_r[:, None] * D + offs_d[None, :]
        tl.store(out_ptrs, (acc + bias[None, :]).to(tl.float16), mask=mask_r[:, None] & mask_d[None, :])

def fused_ln_gate_out_kernel(
    out_ptr, gate_ptr, out_final_ptr, weight_ptr,
    total_rows, H, D,
    eps: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
   pid = tl.program_id(0)
   row_start = pid * BLOCK_R
   offs_r = row_start + tl.arange(0, BLOCK_R)
   mask_r = offs_r < total_rows

   # compute mean,var
   sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
   sum_sq = tl.zeros([BLOCK_R], dtype=tl.float32)
   for h in range(0, tl.cdiv(H, BLOCK_H)):
       offs_h = h * BLOCK_H + tl.arange(0, BLOCK_H)
       mask_h = offs_h < H
       ptrs = out_ptr + offs_r[:, None] * H + offs_h[None, :]
       x = tl.load(ptrs, mask=mask_r[:, None] & mask_h[None, :], other=0.0)
       x_f = x.to(tl.float32)
       sum_val += tl.sum(x_f, axis=1)
       sum_sq += tl.sum(x_f * x_f, axis=1)

   mean = sum_val / H
   var = sum_sq / H - mean * mean
   rstd = tl.rsqrt(var + eps)

   # Loop over D blocks
   for d in range(0, tl.cdiv(D, BLOCK_D)):
       offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
       mask_d = offs_d < D

       acc = tl.zeros([BLOCK_R, BLOCK_D], dtype=tl.float32)

       for h in range(0, tl.cdiv(H, BLOCK_H)):
           offs_h = h * BLOCK_H + tl.arange(0, BLOCK_H)
           mask_h = offs_h < H

           # load out
           out_ptrs = out_ptr + offs_r[:, None] * H + offs_h[None, :]
           out_val = tl.load(out_ptrs, mask=mask_r[:, None] & mask_h[None, :], other=0.0).to(tl.float32)

           # load gate
           gate_ptrs = gate_ptr + offs_r[:, None] * H + offs_h[None, :]
           gate_val = tl.load(gate_ptrs, mask=mask_r[:, None] & mask_h[None, :], other=0.0).to(tl.float32)

           # normalize and apply gate
           y = (out_val - mean[:, None]) * rstd[:, None] * gate_val   # [BLOCK_R, BLOCK_H]

           # load weight slice
           w_ptrs = weight_ptr + offs_d[:, None] * H + offs_h[None, :]
           w = tl.load(w_ptrs, mask=mask_d[:, None] & mask_h[None, :], other=0.0).to(tl.float32)

           # accumulate dot product

           acc += tl.sum(y[:, None, :] * w[None, :, :], axis=2)

       # store output
       out_ptrs = out_final_ptr + offs_r[:, None] * D + offs_d[None, :]
       tl.store(out_ptrs, acc.to(tl.float16), mask=mask_r[:, None] & mask_d[None, :])

def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]; hidden_dim = config["hidden_dim"]; nomask = config.get("nomask", True)

    device = input_tensor.device

    # 1. First LayerNorm using torch's nn.functional.layer_norm
    # We have bias and weight
    # Use built-in: torch.nn.functional.layer_norm(input, normalized_shape=dim, weight=..., bias=..., eps=1e-5)
    # Input shape [B,N,N,dim]
    # Ensure normalized_shape = dim (last dim).
    # Use contiguous input.

    # Move input to float32; already.
    x_norm = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=1e-5)      # shape [B,N,N,dim]

    # 2. Flatten for linear
    total_rows = B*N*N
    R = total_rows
    x_norm_flat = x_norm.view(R, dim).contiguous()
    x_norm_h = x_norm_flat.to(torch.float16)

    # 3. Concatenated linear
    w_concat = torch.cat([...], dim=0).to(torch.float16)   # shape (5*H, D)
    proj_all = torch.nn.functional.linear(x_norm_h, w_concat)   # shape [R,5*H]

    # Slice
    H = hidden_dim
    left_proj = proj_all[:, 0:H]
    left_gate = proj_all[:, H:2*H]
    right_proj = proj_all[:, 2*H:3*H]
    right_gate = proj_all[:, 3*H:4*H]
    out_gate = proj_all[:, 4*H:5*H]

    if not nomask:
        mask_h = mask_tensor.view(R, 1).to(torch.float16)
        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h

    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    left = left_proj * left_gate
    right = right_proj * right_gate

    # 4. Tri-mul using batched GEMM
    left = left.view(B, N, N, H)
    right = right.view(B, N, N, H)

    # shape for batch matmul: left_mat [B,H,N,N] (i.e., left per hidden); right_mat [B,H,N,N] transposed.
    left_mat = left.permute(0,3,1,2)
    right_mat = right.permute(0,3,2,1)
    # Get contiguous to improve memory (maybe .contiguous()) - memory copy cost for left_mat & right_mat but relatively small.

    # Reshape to BH x N x N
    BH = B*H
    left_mat = left_mat.reshape(BH, N, N)
    right_mat = right_mat.reshape(BH, N, N)

    out_mat = torch.bmm(left_mat, right_mat)   # half

    # Reshape out to [R, H]
    out = out_mat.view(B, H, N, N).permute(0,2,3,1).contiguous()
    out_flat = out.view(R, H).to(torch.float16)

    # out_gate already shape [R, H] half (already computed). Good.

    # 5. Fuse LN+gate+output linear via Triton kernel
    # Allocate final buffer
    final_half = torch.empty((R, dim), dtype=torch.float16, device=device)

    # weight pointer: weights["to_out.weight"] shape [dim, hidden_dim] half.
    # Ensure half.
    weight_to_out = weights["to_out.weight"].to(torch.float16)

    # Launch kernel
    BLOCK_R = 128  # choose 128 rows per block
    BLOCK_H = 32   # hidden dim block size
    BLOCK_D = 64   # output dim block size

    grid = (triton.cdiv(R, BLOCK_R),)
    fused_ln_gate_out_kernel[grid](
        out_flat,
        out_gate,
        final_half,
        weight_to_out,
        R,
        H,
        dim,
        eps=1e-5,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        BLOCK_D=BLOCK_D,
    )
    # final result: reshape to [B,N,N,dim] and cast to float32
    final = final_half.view(B, N, N, dim).to(torch.float32)

    return final

"""
TriMul (outgoing) – fast forward pass
------------------------------------
The implementation follows the reference PyTorch module but moves the two
expensive LayerNorms onto Triton:

1. **Input LayerNorm** – performed with the highly‑optimised `torch.nn.functional.layer_norm`
   (FP32, float‑accumulators).  This avoids the custom Triton kernel and gives the
   best possible throughput on H100.

2. **Linear projection + gates** – a single fused `F.linear` computes the five
   required projections (`left_proj`, `right_proj`, `left_gate`,
   `right_gate`, `out_gate`) in one GEMM.

3. **Tri‑multiplicative update** – `left` and `right` are reshaped and a
   batched matrix‑multiply (`torch.bmm`) carries out the `einsum`
   `'... i k d, ... j k d -> ... i j d'`.  This runs on Tensor‑Core cuBLAS
   and dominates the FLOP count.

4. **Fused hidden‑dim LayerNorm + output‑gate + final linear** – a custom
   Triton kernel normalises the intermediate tensor, multiplies by the
   pre‑computed `out_gate`, and immediately applies the output linear
   projection (`to_out.weight`).  All three steps are fused, removing two
   extra memory passes.

The result is cast back to `float32` and returned.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fused LayerNorm (hidden dim) + gate + final linear
# ----------------------------------------------------------------------
@triton.jit
def fused_ln_gate_out_kernel(
    out_ptr,        # const half* [R, H]   – output of the Tri‑mul (R = B·N·N)
    gate_ptr,       # const half* [R, H]   – out_gate (after sigmoid)
    out_final_ptr,  #       half* [R, D]   – final tensor (before FP32 cast)
    weight_ptr,     # const half* [D, H]   – to_out.weight
    total_rows,     # i32
    H,              # i32   hidden dimension
    D,              # i32   model dimension (dim)
    eps: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """
    Row‑wise (over hidden dim H) LayerNorm in fp32, element‑wise multiply
    with `gate`, then a dense projection to dimension D.
    All operations are fused to reduce memory traffic.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < total_rows

    # -----------------------------------------------------------------
    # 1️⃣ Compute mean & variance for each row (float32 accumulator)
    # -----------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_R], dtype=tl.float32)

    for h in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H
        ptrs = out_ptr + offs_r[:, None] * H + offs_h[None, :]
        vals = tl.load(ptrs, mask=mask_r[:, None] & mask_h[None, :], other=0.0)
        vals_f = vals.to(tl.float32)
        sum_val += tl.sum(vals_f, axis=1)
        sum_sq  += tl.sum(vals_f * vals_f, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)      # 1 / sqrt(var + eps)

    # -----------------------------------------------------------------
    # 2️⃣ For each tile of the output dimension D, accumulate the
    #     projection `y = (out - mean) / sqrt(var+eps) * gate` dot
    #     weight[d, :] .
    # -----------------------------------------------------------------
    for d in range(0, tl.cdiv(D, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < D

        # accumulator for the current D‑tile (fp32)
        acc = tl.zeros([BLOCK_R, BLOCK_D], dtype=tl.float32)

        for h in range(0, tl.cdiv(H, BLOCK_H)):
            offs_h = h * BLOCK_H + tl.arange(0, BLOCK_H)
            mask_h = offs_h < H

            # ---- load and normalise the hidden vector -----------------
            out_vals = tl.load(out_ptr + offs_r[:, None] * H + offs_h[None, :],
                               mask=mask_r[:, None] & mask_h[None, :], other=0.0).to(tl.float32)
            gate_vals = tl.load(gate_ptr + offs_r[:, None] * H + offs_h[None, :],
                                mask=mask_r[:, None] & mask_h[None, :], other=0.0).to(tl.float32)

            y = (out_vals - mean[:, None]) * rstd[:, None] * gate_vals   # [BLOCK_R, BLOCK_H]

            # ---- weight slice -----------------------------------------
            w = tl.load(weight_ptr + offs_d[:, None] * H + offs_h[None, :],
                        mask=mask_d[:, None] & mask_h[None, :], other=0.0).to(tl.float32)
            # w shape: [BLOCK_D, BLOCK_H]

            # contribution to the output tile: sum_h y[h] * w[d,h]
            acc += tl.sum(y[:, None, :] * w[None, :, :], axis=2)          # [BLOCK_R, BLOCK_D]

        # ---- write the final tile (store as FP16) --------------------
        out_ptrs = out_final_ptr + offs_r[:, None] * D + offs_d[None, :]
        tl.store(out_ptrs, acc.to(tl.float16),
                 mask=mask_r[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : [B, N, N, D]  (float32)
        - mask_tensor  : [B, N, N]    (bool/int) – may be ignored if `nomask=True`
        - weights_dict : dict of model weights (float32)
        - config_dict  : {"dim": D, "hidden_dim": H, "nomask": bool}

    Returns
    -------
    torch.Tensor of shape [B, N, N, D] (float32)
    """
    # ---------------------------------------------------------
    # 0️⃣ unpack inputs & constants
    # ---------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                 # D
    hidden_dim = config["hidden_dim"]   # H
    eps = 1e-5
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape
    R = B * N * N                       # total rows = B·N·N

    # ---------------------------------------------------------
    # 1️⃣ Input LayerNorm (use the highly‑optimised torch kernel)
    # ---------------------------------------------------------
    # normalized input – still fp32
    x_norm = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )                                   # [B, N, N, D]  fp32

    # ---------------------------------------------------------
    # 2️⃣ Fused linear that produces projections and gates
    # ---------------------------------------------------------
    x_flat = x_norm.view(R, dim).contiguous()
    x_half = x_flat.to(torch.float16)                     # [R, D]  fp16

    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                   # (5*H, D)

    # fused linear → [R, 5*H]  fp16
    proj_all = F.linear(x_half, w_concat)

    H = hidden_dim
    left_proj   = proj_all[:, 0:H]
    left_gate   = proj_all[:, H:2 * H]
    right_proj  = proj_all[:, 2 * H:3 * H]
    right_gate  = proj_all[:, 3 * H:4 * H]
    out_gate    = proj_all[:, 4 * H:5 * H]

    if not nomask:
        # mask is broadcast over the hidden dimension
        mask_flat = mask_tensor.to(torch.float16).view(R, 1)   # [R, 1]
        left_proj  = left_proj * mask_flat
        right_proj = right_proj * mask_flat

    # sigmoid gates (still fp16)
    left_gate  = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate   = torch.sigmoid(out_gate)

    left  = left_proj * left_gate      # [R, H] fp16
    right = right_proj * right_gate    # [R, H] fp16

    # ---------------------------------------------------------
    # 3️⃣ Tri‑multiplicative update via batched GEMM
    # ---------------------------------------------------------
    left = left.view(B, N, N, H)        # [B, N, N, H]
    right = right.view(B, N, N, H)      # [B, N, N, H]

    # bring hidden dim to the batch axis
    left_mat  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_mat = right.permute(0, 3, 2, 1).contiguous()  # [B, H, N, N] (j↔k)

    BH = B * H
    left_mat  = left_mat.view(BH, N, N)   # [B·H, N, N]
    right_mat = right_mat.view(BH, N, N)  # [B·H, N, N]

    # batched matrix multiply – everything stays in fp16
    out_mat = torch.bmm(left_mat, right_mat)   # [B·H, N, N] fp16

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # fp16
    out_flat = out.view(R, H).contiguous()                          # [R, H] fp16

    # ---------------------------------------------------------
    # 4️⃣ Fused hidden‑dim LayerNorm + out_gate + final linear
    # ---------------------------------------------------------
    # to_out.weight : (D, H) fp16   (bias is disabled in the reference)
    w_out = weights["to_out.weight"].to(torch.float16)

    final_half = torch.empty((R, dim), dtype=torch.float16, device=device)

    # Triton launch configuration
    BLOCK_R = 128                      # rows per program
    BLOCK_H = 32                       # tile over hidden dimension
    BLOCK_D = 64                       # tile over output (model) dimension
    grid = (triton.cdiv(R, BLOCK_R),)

    fused_ln_gate_out_kernel[grid](
        out_flat,
        out_gate,
        final_half,
        w_out,
        R,
        H,
        dim,
        eps=eps,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        BLOCK_D=BLOCK_D,
    )

    # ---------------------------------------------------------
    # 5️⃣ Cast back to float32 and reshape to the original layout
    # ---------------------------------------------------------
    final = final_half.view(B, N, N, dim).to(torch.float32)

    return final
