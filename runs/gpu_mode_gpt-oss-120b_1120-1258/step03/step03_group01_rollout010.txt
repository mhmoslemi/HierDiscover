"""
TriMul (outgoing) forward pass – Triton‑accelerated implementation.

Algorithm
---------
1. Row‑wise LayerNorm on the input tensor (shape [B,N,N,dim]) – performed by a
   custom Triton kernel that writes the normalized result directly in fp16.

2. Fuse the five dense layers (left_proj, right_proj, left_gate, right_gate,
   out_gate) into a single half‑precision GEMM:
   `[R, dim] @ [5*hidden, dim].T → [R, 5*hidden]`, where `R = B·N·N`.

3. Slice the concatenated result, apply the optional mask, compute the three
   sigmoid gates and obtain the gated projected tensors `left` and `right`
   (shape [B,N,N,hidden]).

4. Triangular multiplicative update:
   `out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]`.
   This is a batched matrix multiplication `left @ rightᵀ` for every hidden
   channel, implemented with `torch.bmm` on the reshaped tensors
   (`[B*hidden, N, N]`).

5. Second LayerNorm over the hidden dimension (again a Triton kernel).

6. Apply the output gate, project back to the original dimension with a final
   half‑precision linear layer, and cast the result to float32.

All heavy arithmetic (LayerNorm, GEMMs, and the O(N³) update) runs in fp16,
exploiting Tensor‑Core throughput on an H100.  The only required custom
kernel is the row‑wise LayerNorm; the rest uses highly‑optimized PyTorch
operations.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def layernorm_kernel(
    x_ptr,          # const *  [total_rows, dim]  (fp32 or fp16 input)
    y_ptr,          # mutable *[total_rows, dim] (fp16 output)
    weight_ptr,     # const *  [dim]            (fp32)
    bias_ptr,       # const *  [dim]            (fp32)
    total_rows,     # int (runtime)
    dim,            # int (compile‑time)
    eps,            # float (compile‑time)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # cols per loop (tile over dim)
):
    """
    Row‑wise LayerNorm (no bias after the mean/var reduction).
    Computes (x - μ) / sqrt(var + eps) * weight + bias and writes fp16.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ----------------- 1st pass: mean & variance -----------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs, mask=mask_m[:, None] & mask_d[None, :], other=0.0)
        vals_f32 = vals.to(tl.float32)

        sum_val += tl.sum(vals_f32, axis=1)
        sum_sq  += tl.sum(vals_f32 * vals_f32, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)        # 1 / sqrt(var + eps)

    # ----------------- 2nd pass: write normalized result ------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs, mask=mask_m[:, None] & mask_d[None, :], other=0.0)
        vals_f32 = vals.to(tl.float32)

        norm = (vals_f32 - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]                      # fp32

        out_fp16 = out.to(tl.float16)
        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16, mask=mask_m[:, None] & mask_d[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float)
        - weights_dict : mapping str → torch.Tensor containing all model
                         parameters (float32)
        - config_dict  : {"dim": int,
                          "hidden_dim": int,
                          "nomask": bool (optional)}

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, on the same device).
    """
    # --------------------------------------------------------------
    # Unpack
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    device = input_tensor.device

    # --------------------------------------------------------------
    # 1) First LayerNorm (row‑wise) – output fp16
    # --------------------------------------------------------------
    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N                     # R = B·N·N

    # Flatten to [R, dim]
    x_flat = input_tensor.view(total_rows, dim).contiguous()
    x_norm_flat = torch.empty(total_rows, dim, dtype=torch.float16,
                              device=device)

    # Triton launch parameters (tuned for typical dim ≤ 384)
    BLOCK_M = 256
    BLOCK_D = 64

    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,                     # input pointer (fp32)
        x_norm_flat,                # output pointer (fp16)
        weights["norm.weight"],     # γ (fp32)
        weights["norm.bias"],       # β (fp32)
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # --------------------------------------------------------------
    # 2) Fuse five linear layers (proj + 3 gates) into one GEMM
    # --------------------------------------------------------------
    # Concatenate weight matrices: [5*hidden, dim]
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
            weights["out_gate.weight"],    # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)

    # Single half‑precision linear transformation
    out_concat = F.linear(x_norm_flat, w_concat)          # [R, 5*H] (fp16)

    H = hidden_dim
    left_proj   = out_concat[:, 0:H]                     # [R, H]
    right_proj  = out_concat[:, H:2*H]
    left_gate_p = out_concat[:, 2*H:3*H]
    right_gate_p= out_concat[:, 3*H:4*H]
    out_gate_p  = out_concat[:, 4*H:5*H]

    # Sigmoids (still fp16)
    left_gate   = torch.sigmoid(left_gate_p)
    right_gate  = torch.sigmoid(right_gate_p)
    out_gate    = torch.sigmoid(out_gate_p)

    # --------------------------------------------------------------
    # 3) Optional mask (broadcasted over hidden dim)
    # --------------------------------------------------------------
    if not config.get("nomask", True):
        # mask shape [B,N,N] → flatten to [R,1] and cast to half
        mask_h = mask_tensor.to(torch.float16).view(total_rows, 1)
        left_proj  = left_proj * mask_h
        right_proj = right_proj * mask_h

    # --------------------------------------------------------------
    # 4) Apply gates & reshape to [B,N,N,H]
    # --------------------------------------------------------------
    left  = left_proj * left_gate          # [R, H]
    right = right_proj * right_gate        # [R, H]

    left  = left.view(B, N, N, H)
    right = right.view(B, N, N, H)
    out_gate = out_gate.view(B, N, N, H)   # later used after 2nd LN

    # --------------------------------------------------------------
    # 5) Triangular multiplicative update via batched GEMM
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # --------------------------------------------------------------
    # Rearrange to [B, H, N, N] → (B*H, N, N)
    left_batch  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_batch = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # BMM: (B*H, N, N) @ (B*H, N, N)^T  → (B*H, N, N)
    out_batch = torch.bmm(left_batch, right_batch.transpose(1, 2))

    # Restore shape [B, N, N, H]
    out = out_batch.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6) Second LayerNorm over hidden dim (fp16)
    # --------------------------------------------------------------
    out_flat = out.view(total_rows, H).contiguous()
    out_norm_flat = torch.empty_like(out_flat, dtype=torch.float16)

    grid_ln2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid_ln2](
        out_flat,
        out_norm_flat,
        weights["to_out_norm.weight"],   # γ (fp32)
        weights["to_out_norm.bias"],     # β (fp32)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    out_norm = out_norm_flat.view(B, N, N, H)   # fp16

    # --------------------------------------------------------------
    # 7) Apply output gate and final projection back to `dim`
    # --------------------------------------------------------------
    out_norm = out_norm * out_gate                         # element‑wise fp16
    out_norm_flat2 = out_norm.view(total_rows, H)          # [R, H]

    final_half = F.linear(
        out_norm_flat2,
        weights["to_out.weight"].to(torch.float16)        # [dim, H]
    )                                                       # [R, dim] fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)

    return final
