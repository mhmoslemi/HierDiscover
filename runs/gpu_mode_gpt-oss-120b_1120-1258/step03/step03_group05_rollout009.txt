```
@triton.jit
def final_norm_proj_kernel(
    out_ptr, out_gate_ptr, w_norm_ptr, b_norm_ptr,
    w_out_ptr,  # shape [dim, hidden]
    out_final_ptr,  # shape [R, dim]
    R, H, dim,
    eps: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < R

    # Compute mean & var for each row
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_R], dtype=tl.float32)

    for h0 in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h0 * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H
        # load out values (fp16)
        ptr = out_ptr + offs_r[:, None] * H + offs_h[None, :]
        out_vals = tl.load(ptr,
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0).to(tl.float32)  # [BLOCK_R, BLOCK_H]
        sum_val += tl.sum(out_vals, axis=1)
        sum_sq += tl.sum(out_vals * out_vals, axis=1)

    mean = sum_val / H
    var = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # second pass: compute final output accumulation per dim
    # initialize output accumulators [BLOCK_R, BLOCK_D] as float32
    acc = tl.zeros([BLOCK_R, BLOCK_D], dtype=tl.float32)
    for h0 in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h0 * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H
        # load x, gate
        out_vals = tl.load(out_ptr + offs_r[:, None] * H + offs_h[None, :],
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0).to(tl.float32)
        gate_vals = tl.load(out_gate_ptr + offs_r[:, None] * H + offs_h[None, :],
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0).to(tl.float32)
        # normalize
        norm = (out_vals - mean[:, None]) * rstd[:, None]
        # apply LN affine (gamma,beta) weights (float32)
        gamma = tl.load(w_norm_ptr + offs_h, mask=mask_h, other=0.0) # shape [H]; broadcast across rows
        beta = tl.load(b_norm_ptr + offs_h, mask=mask_h, other=0.0)
        norm = norm * gamma[None, :] + beta[None, :]
        # apply gate
        norm = norm * gate_vals

        # multiply with to_out weight
        # load weight slice of shape [dim, BLOCK_H]
        w_out_slice = tl.load(w_out_ptr + offs_h[None, :] + tl.arange(0, BLOCK_D)[:, None] * H,
                              mask=tl.arange(0, BLOCK_D)[:, None] < dim and mask_h[None, :],
                              other=0.0).to(tl.float32)  # [BLOCK_D, BLOCK_H]
        # w_out_ptr indexing: offset = d*H + h
```

```
@triton.jit
def final_norm_proj_kernel(
    out_ptr,                                                                                                                                
    out_gate_ptr,
    w_norm_ptr,        # gamma (float32) shape [H]
    b_norm_ptr,        # beta (float32) shape [H]
    w_out_ptr,         # weight_T shape [H, dim] as half (H rows, dim columns)
    out_final_ptr,     # output [R, dim] float32
    R, H, dim,
    eps: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid_r = tl.program_id(0) # row block
    pid_d = tl.program_id(1) # dim block

    row_start = pid_r * BLOCK_R
    col_start = pid_d * BLOCK_D

    offs_r = row_start + tl.arange(0, BLOCK_R)      # [BLOCK_R]
    offs_d = col_start + tl.arange(0, BLOCK_D)      # [BLOCK_D]

    mask_r = offs_r < R
    mask_d = offs_d < dim

    # Compute mean & variance per row
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_R], dtype=tl.float32)

    for h0 in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h0 * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H
        # load out values
        # offset = r * H + h
        ptr = out_ptr + offs_r[:, None] * H + offs_h[None, :]
        val = tl.load(ptr,
                      mask=mask_r[:, None] & mask_h[None, :],
                      other=0.0).to(tl.float32)
        sum_val += tl.sum(val, axis=1)
        sum_sq += tl.sum(val * val, axis=1)

    mean = sum_val / H
    var = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # second pass accumulation across hidden dimension, also compute dot product
    acc = tl.zeros([BLOCK_R, BLOCK_D], dtype=tl.float32)

    for h0 in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h0 * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # load x
        x_ptr = out_ptr + offs_r[:, None] * H + offs_h[None, :]
        x_vals = tl.load(x_ptr,
                         mask=mask_r[:, None] & mask_h[None, :],
                         other=0.0).to(tl.float32) # [BLOCK_R, BLOCK_H]
        # load gate
        gate_ptr = out_gate_ptr + offs_r[:, None] * H + offs_h[None, :]
        gate_vals = tl.load(gate_ptr,
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0).to(tl.float32)

        # norm
        norm = (x_vals - mean[:, None]) * rstd[:, None]   # [BLOCK_R, BLOCK_H]
        # apply LN affine gamma, beta
        gamma = tl.load(w_norm_ptr + offs_h, mask=mask_h, other=0.0)  # [BLOCK_H]
        beta  = tl.load(b_norm_ptr + offs_h, mask=mask_h, other=0.0)
        norm = norm * gamma[None, :] + beta[None, :]
        # apply gate
        norm = norm * gate_vals

        # load weight slice: shape [BLOCK_H, BLOCK_D]
        # memory layout: weight_T is [H, dim] row-major: offset = h * dim + d
        w_ptr = w_out_ptr + offs_h[:, None] * dim + offs_d[None, :]  # [BLOCK_H, BLOCK_D]
        w_vals = tl.load(w_ptr,
                         mask=mask_h[:, None] & mask_d[None, :],
                         other=0.0).to(tl.float32)  # [BLOCK_H, BLOCK_D]

        # accumulate: (norm) [BLOCK_R, BLOCK_H] dot w_vals [BLOCK_H, BLOCK_D]
        # using tl.dot: shape [BLOCK_R, BLOCK_D]
        acc += tl.dot(norm, w_vals)  # registers

    # Write final output
    out_ptrs = out_final_ptr + offs_r[:, None] * dim + offs_d[None, :]
    tl.store(out_ptrs,
             acc,
             mask=mask_r[:, None] & mask_d[None, :])
```

        weights["left_gate.weight"],
        weights["right_proj.weight"],
        weights["right_gate.weight"],
        weights["out_gate.weight"],

        proj, hidden_dim, dim=1)

        right_proj_raw = right_proj_raw * mask_h

    # 4️⃣ Triangular multiplicative update (batched GEMM)
    # ------------------------------------------------------------------

    right = right.view(B, N, N, hidden_dim)      # [B,N,N,H]

    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B,H,N,N] (i‑k)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # [B,H,N,N] (j‑k)

    right_mat = right_perm.view(BH, N, N)        # [BH,N,N]

    # ------------------------------------------------------------------

    final = torch.empty((R, dim), dtype=torch.float32, device=input_tensor.device)

    BLOCK_H = 32          # hidden tiles
    BLOCK_D = 64          # output‑channel tiles

        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()

    final = final.view(B, N, N, dim)   # [B, N, N, dim] float32

    offs_c = col_start + tl.arange(0, BLOCK_D)

    mask_r = offs_r < R
    mask_c = offs_c < D

    # --------------------------------------------------------------
    # Pass 1 – compute mean & variance per row (FP32)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_R], dtype=tl.float32)

    for h0 in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h0 * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        x = tl.load(x_ptr + offs_r[:, None] * H + offs_h[None, :],
                    mask=mask_r[:, None] & mask_h[None, :],
                    other=0.0).to(tl.float32)              # [BLOCK_R, BLOCK_H]

        sum_val += tl.sum(x, axis=1)
        sum_sq  += tl.sum(x * x, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------------------------------------------------------------
    # Pass 2 – LN → affine → gate → dot(weight_T) accumulation
    # --------------------------------------------------------------
    acc = tl.zeros([BLOCK_R, BLOCK_D], dtype=tl.float32)

    for h0 in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h0 * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # load x and gate (half → float32)
        x = tl.load(x_ptr + offs_r[:, None] * H + offs_h[None, :],
                    mask=mask_r[:, None] & mask_h[None, :],
                    other=0.0).to(tl.float32)
        g = tl.load(gate_ptr + offs_r[:, None] * H + offs_h[None, :],
                    mask=mask_r[:, None] & mask_h[None, :],
                    other=0.0).to(tl.float32)

        # LayerNorm normalisation
        norm = (x - mean[:, None]) * rstd[:, None]               # [BLOCK_R, BLOCK_H]

        # γ / β (float32 vectors length H)
        gamma = tl.load(wnorm_ptr + offs_h, mask=mask_h, other=0.0)   # [BLOCK_H]
        beta  = tl.load(bnorm_ptr + offs_h, mask=mask_h, other=0.0)   # [BLOCK_H]

        norm = norm * gamma[None, :] + beta[None, :]            # affine
        norm = norm * g                                         # apply output‑gate

        # load slice of W_out^T : shape [BLOCK_H, BLOCK_D]
        # layout of wout_ptr is row‑major [H, D] → offset = h*D + d
        w = tl.load(wout_ptr + offs_h[:, None] * D + offs_c[None, :],
                    mask=mask_h[:, None] & mask_c[None, :],
                    other=0.0).to(tl.float32)                # [BLOCK_H, BLOCK_D]

        # accumulate matrix product: (BLOCK_R, BLOCK_H) @ (BLOCK_H, BLOCK_D)
        acc += tl.dot(norm, w)                                 # [BLOCK_R, BLOCK_D]

    # write final output
    out_ptr = y_ptr + offs_r[:, None] * D + offs_c[None, :]
    tl.store(out_ptr,
             acc,
             mask=mask_r[:, None] & mask_c[None, :])

# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul module.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32, on CUDA).
    """
    input_tensor, mask_tensor, weights, config = data
    dim          = config["dim"]          # input / output channel dimension
    hidden_dim   = config["hidden_dim"]   # inner hidden dimension
    eps          = 1e-5
    nomask       = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (custom Triton kernel)
    # ------------------------------------------------------------------
    B, N, _, _ = input_tensor.shape                     # [B, N, N, dim]
    total_rows = B * N * N                               # rows after flattening

    x_flat = input_tensor.contiguous().view(total_rows, dim)   # [R, dim]
    x_norm_flat = torch.empty_like(x_flat)

    BLOCK_M = 64
    BLOCK_D = 32
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm_flat,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()
    x_norm = x_norm_flat.view(B, N, N, dim)               # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2️⃣ Fuse all linear projections + gates into ONE GEMM (FP16)
    # ------------------------------------------------------------------
    x_norm_h = x_norm.view(total_rows, dim).to(torch.float16)

    # Concatenate the five weight matrices (shape [(5*H), dim])
    weight_concat = torch.cat([
        weights["left_proj.weight"],
        weights["left_gate.weight"],
        weights["right_proj.weight"],
        weights["right_gate.weight"],
        weights["out_gate.weight"],
    ], dim=0).to(torch.float16)

    proj = F.linear(x_norm_h, weight_concat)             # [R, 5*H]

    # Split into the logical branches
    left_proj_raw, left_gate_raw, right_proj_raw, right_gate_raw, out_gate_raw = \
        torch.split(proj, hidden_dim, dim=1)

    # ------------------------------------------------------------------
    # 3️⃣ Apply mask (if present) and sigmoid‑gates
    # ------------------------------------------------------------------
    if not nomask:
        mask_h = mask_tensor.view(total_rows, 1).to(torch.float16)   # [R,1]
        left_proj_raw  = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    left = left_proj_raw * torch.sigmoid(left_gate_raw)    # [R, H]
    right = right_proj_raw * torch.sigmoid(right_gate_raw) # [R, H]
    out_gate = torch.sigmoid(out_gate_raw)                # [R, H]

    # ------------------------------------------------------------------
    # 4️⃣ Triangular multiplicative update (batched GEMM)
    # ------------------------------------------------------------------
    left = left.view(B, N, N, hidden_dim)                 # [B,N,N,H]
    right = right.view(B, N, N, hidden_dim)               # [B,N,N,H]

    left_perm  = left.permute(0, 3, 1, 2).contiguous()    # [B,H,N,N]
    right_perm = right.permute(0, 3, 2, 1).contiguous()   # [B,H,N,N]

    BH = B * hidden_dim
    left_mat  = left_perm.view(BH, N, N)                  # [BH,N,N]
    right_mat = right_perm.view(BH, N, N)                 # [BH,N,N]

    out_mat = torch.bmm(left_mat, right_mat)              # [BH,N,N]

    # Back to [B,N,N,H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # FP16

    # ------------------------------------------------------------------
    # 5️⃣ Fused second LayerNorm + output‑gate + final linear projection
    # ------------------------------------------------------------------
    R = B * N * N
    out_flat      = out.view(R, hidden_dim)                # [R, H]  (FP16)
    out_gate_flat = out_gate.view(R, hidden_dim)          # [R, H]  (FP16)

    # transpose final weight once: shape [H, D] (row‑major)
    w_out_T = weights["to_out.weight"].t().contiguous().to(torch.float16)  # [H, D]

    final = torch.empty((R, dim), dtype=torch.float32, device=input_tensor.device)

    BLOCK_R = 64          # rows per program
    BLOCK_H = 16          # hidden‑dimension tile
    BLOCK_D = 32          # output‑channel tile
    grid_r = triton.cdiv(R, BLOCK_R)
    grid_c = triton.cdiv(dim, BLOCK_D)

    final_norm_proj_kernel[(grid_r, grid_c)](
        out_flat,
        out_gate_flat,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        w_out_T,
        final,
        R,
        hidden_dim,
        dim,
        eps,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()

    final = final.view(B, N, N, dim)   # [B, N, N, dim] (float32)
    return final
