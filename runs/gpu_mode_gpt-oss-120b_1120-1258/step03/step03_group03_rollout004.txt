"""
TriMul (outgoing) forward pass – highly‑optimized Triton + PyTorch.

Algorithm
---------
1. **LayerNorm (input)** – use PyTorch’s native `F.layer_norm` (FP32) which is
   already heavily tuned on GPUs.  This avoids the extra kernel launch for the
   first LN.

2. **Fused Linear + Gates** – concatenate the five weight matrices
   (left_proj, left_gate, right_proj, right_gate, out_gate) and perform a
   single half‑precision `F.linear`.  The result is sliced into the three
   projection tensors and three gate tensors.  Sigmoid is applied in‑place.

3. **Mask** – if a mask is supplied it is broadcasted and applied only to the
   two *projection* tensors (as in the reference implementation).

4. **Tri‑multiplicative update** – compute
   `out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]` directly with
   `torch.einsum`.  This removes the costly `permute → view → bmm`
   pipeline and eliminates two large (≈ GB) memory copies.

5. **LayerNorm + Output‑gate** – a custom Triton kernel (`layernorm_gate_kernel`)
   normalises the hidden‑dimension tensor (`out`), applies the learned
   `γ/β` and multiplies element‑wise with the `out_gate`.  The kernel works in
   FP16 with FP32 accumulators for numerical stability.

6. **Final projection** – a half‑precision linear layer maps back to the model
   dimension and the result is cast to FP32.

Only step 5 is implemented in Triton (as required); all other heavy work is
performed by highly‑optimised cuBLAS/Tensor‑Core kernels or PyTorch’s native
layer‑norm.  The implementation runs entirely on the GPU and respects all
provided test configurations.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: hidden‑dim LayerNorm + element‑wise gate
# ----------------------------------------------------------------------
@triton.jit
def layernorm_gate_kernel(
    x_ptr,          # const half*   [total_rows, dim]   (input)
    y_ptr,          #       half*   [total_rows, dim]   (output)
    gate_ptr,       # const half*   [total_rows, dim]   (gate)
    weight_ptr,     # const half*   [dim]               (γ)
    bias_ptr,       # const half*   [dim]               (β)
    total_rows,     # i32
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # -----------------------------------------------------------------
    # 1st pass – mean / variance (float32 accumulator)
    # -----------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                     # [BLOCK_M, BLOCK_D]
        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                # 1 / sqrt(var+eps)

    # -----------------------------------------------------------------
    # 2nd pass – normalize, apply γ/β, multiply by gate
    # -----------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        # load x
        x_ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        x = tl.load(
            x_ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        # layer‑norm
        norm = (x - mean[:, None]) * rstd[:, None]

        # γ, β (half → float32)
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0).to(tl.float32)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0).to(tl.float32)

        out = norm * w[None, :] + b[None, :]

        # gate (half → float32)
        g = tl.load(
            gate_ptr + offs_m[:, None] * dim + offs_d[None, :],
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        out = out * g

        # store back as half
        y_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(y_ptrs, out.to(tl.float16), mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, D]  (float32)
            - mask_tensor  : torch.Tensor [B, N, N]   (bool/int, optional)
            - weights_dict : model parameters (see code)
            - config_dict  : {"dim": D, "hidden_dim": H, "nomask": bool}

    Returns:
        torch.Tensor [B, N, N, D] (float32)
    """
    # -------------------------------------------------
    # 1. Unpack arguments
    # -------------------------------------------------
    x, mask, w, cfg = data
    device = x.device
    dim = cfg["dim"]                # D
    hidden_dim = cfg["hidden_dim"]  # H
    eps = 1e-5
    nomask = cfg.get("nomask", True)

    B, N, _, _ = x.shape
    total_rows = B * N * N          # R = B·N·N

    # -------------------------------------------------
    # 2. Input LayerNorm (use highly‑optimised torch kernel)
    # -------------------------------------------------
    x_norm = F.layer_norm(
        x,
        (dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    )                               # [B,N,N,D]  float32

    # -------------------------------------------------
    # 3. Linear projection & gates (single fused GEMM)
    # -------------------------------------------------
    x_norm_h = x_norm.to(torch.float16)                 # [R,D] half
    # Concatenate the five weight matrices (shape (5*H, D))
    w_concat = torch.cat(
        [
            w["left_proj.weight"],
            w["left_gate.weight"],
            w["right_proj.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                 # (5H, D)

    # fused linear: [R,D] @ (5H,D)ᵀ  →  [R,5H]
    proj = F.linear(x_norm_h, w_concat)                # half

    H = hidden_dim
    left_proj   = proj[:, 0:H]
    left_gate   = proj[:, H:2*H]
    right_proj  = proj[:, 2*H:3*H]
    right_gate  = proj[:, 3*H:4*H]
    out_gate    = proj[:, 4*H:5*H]

    # optional mask (broadcast over hidden dim)
    if not nomask:
        mask_h = mask.to(torch.float16).view(total_rows, 1)  # [R,1]
        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h

    # sigmoid gates (half → half)
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    # gated projections
    left = left_proj * left_gate      # [R,H] half
    right = right_proj * right_gate   # [R,H] half

    # -------------------------------------------------
    # 4. Tri‑multiplicative update (einsum, no extra copies)
    # -------------------------------------------------
    left = left.view(B, N, N, H)      # [B,N,N,H]
    right = right.view(B, N, N, H)    # [B,N,N,H]

    # out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
    out = torch.einsum("b i k d, b j k d -> b i j d", left, right)  # half

    # -------------------------------------------------
    # 5. Hidden‑dim LayerNorm + output‑gate (custom Triton kernel)
    # -------------------------------------------------
    out_flat = out.view(total_rows, H)          # [R, H] half
    out_gate_flat = out_gate.view(total_rows, H)  # [R, H] half

    out_norm_gate = torch.empty_like(out_flat)  # half, result buffer

    BLOCK_M = 256
    BLOCK_D = 32
    grid = (triton.cdiv(total_rows, BLOCK_M),)

    layernorm_gate_kernel[grid](
        out_flat,
        out_norm_gate,
        out_gate_flat,
        w["to_out_norm.weight"].to(torch.float16),   # γ  (half)
        w["to_out_norm.bias"].to(torch.float16),     # β  (half)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # -------------------------------------------------
    # 6. Final linear projection back to model dimension
    # -------------------------------------------------
    final_half = F.linear(out_norm_gate, w["to_out.weight"].to(torch.float16))  # [R, D] half
    final = final_half.view(B, N, N, dim).to(torch.float32)                    # cast back

    return final
