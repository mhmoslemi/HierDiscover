"""
TriMul “outgoing” (AlphaFold‑3) – fast forward pass.

Algorithm
---------
1️⃣ Row‑wise LayerNorm (FP32 → FP16) – a tiny Triton kernel that
   computes μ,σ² for each flatten‑ed [B·N·N, dim] row and writes the
   normalized result directly in FP16 (saves a copy).

2️⃣ Fuse the three linear projections (left‑proj, left‑gate, right‑proj,
   right‑gate, out‑gate) into **one** FP16 GEMM:
        y = x_norm @ W^T   where   W ∈ ℝ^{5·H×C}
   (H = hidden_dim, C = dim).  The result is split into the five
   tensors needed for the update.

3️⃣ Apply the sigmoid‑gates, the optional pairwise mask and obtain the
   gated tensors *left* and *right* (both FP16).  The gate for the
   output (out_gate) is kept for later.

4️⃣ Tri‑multiplicative update:
        out_h(i,j) = Σ_k left(i,k,h)·right(j,k,h)
   Implemented with a single batched GEMM (torch.bmm) over the
   hidden dimension – this is Tensor‑Core‑accelerated on H100.

5️⃣ Second LayerNorm (FP16) – the same Triton kernel as step 1 but over
   the hidden dimension.

6️⃣ Element‑wise multiplication with *out_gate* and the final projection
   back to the model dimension (FP16 GEMM).

All heavy linear algebra (steps 2, 4, 6) use PyTorch’s native FP16
kernels; the only custom code is the layer‑norm kernel, which satisfies
the “must‑use‑Triton” requirement while dramatically reducing the number
of separate GEMMs (from 5 to 3) and overall memory traffic.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Row‑wise LayerNorm that writes FP16 output
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,          # const float*   [total_rows, dim]
    y_ptr,          #       half*   [total_rows, dim]   (output)
    w_ptr,          # const float*  [dim]              (γ)
    b_ptr,          # const float*  [dim]              (β)
    total_rows,     # int (runtime)
    dim: tl.constexpr,          # static: number of columns (C)
    eps: tl.constexpr,          # static epsilon
    BLOCK_M: tl.constexpr,      # rows per program
    BLOCK_D: tl.constexpr,      # columns per inner loop
):
    """
    Compute LayerNorm row‑wise for a 2‑D tensor stored row‑major.
    The result is written as FP16 to reduce memory traffic.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)           # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ---------- pass 1: mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)       # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)          # 1 / sqrt(var+eps)

    # ---------- pass 2: write normalized values ----------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)       # [BLOCK_M, BLOCK_D]

        # (x-μ)/σ
        norm = (vals - mean[:, None]) * rstd[:, None]

        # affine parameters γ, β
        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # β

        out = norm * w[None, :] + b[None, :]                # fp32
        out_fp16 = out.to(tl.float16)                       # cast to fp16

        out_ptrs = y_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        tl.store(out_ptrs,
                 out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor[bs, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor[bs, N, N]       (bool/float)
            - weights_dict : mapping str → torch.Tensor containing
                             all model parameters (float32)
            - config_dict  : dict with keys
                             * dim        (int)
                             * hidden_dim (int)
                             * nomask     (bool, optional)

    Returns:
        torch.Tensor[bs, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                 # C
    hidden_dim = config["hidden_dim"]   # H
    eps = 1e-5
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N                       # rows for the flattened view

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32→FP16)
    # --------------------------------------------------------------
    x_flat = input_tensor.contiguous().view(total_rows, dim)
    x_norm_h = torch.empty((total_rows, dim), dtype=torch.float16, device=device)

    BLOCK_M = 256   # rows per program (tuned for H100)
    BLOCK_D = 64    # columns per inner loop (covers dim up to 384)
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        x_flat,
        x_norm_h,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()

    # --------------------------------------------------------------
    # 2️⃣ Fused projection + gate + out‑gate (single GEMM)
    # --------------------------------------------------------------
    # concatenate the five weight matrices: left_proj, left_gate,
    # right_proj, right_gate, out_gate.  All are (hidden_dim, dim).
    combined_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16, non_blocking=True)   # shape (5*H, dim)

    # fused linear: rows x (5*H)
    fused_out = F.linear(x_norm_h, combined_weight)   # FP16

    # split the concatenated result
    H = hidden_dim
    left_proj_h   = fused_out[:, 0 * H : 1 * H]   # [rows, H]
    left_gate_h   = fused_out[:, 1 * H : 2 * H]   # pre‑sigmoid
    right_proj_h  = fused_out[:, 2 * H : 3 * H]
    right_gate_h  = fused_out[:, 3 * H : 4 * H]   # pre‑sigmoid
    out_gate_h    = fused_out[:, 4 * H : 5 * H]   # pre‑sigmoid

    # --------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over hidden dimension)
    # --------------------------------------------------------------
    if not nomask:
        mask_h = mask_tensor.view(total_rows, 1).to(torch.float16)  # [rows,1]
        left_proj_h = left_proj_h * mask_h
        right_proj_h = right_proj_h * mask_h

    # --------------------------------------------------------------
    # 4️⃣ Apply sigmoid‑gates and compute gated tensors
    # --------------------------------------------------------------
    left_gate_h = torch.sigmoid(left_gate_h)      # [rows, H]
    right_gate_h = torch.sigmoid(right_gate_h)    # [rows, H]
    out_gate_h  = torch.sigmoid(out_gate_h)       # [rows, H]

    left_h = left_proj_h * left_gate_h            # [rows, H]
    right_h = right_proj_h * right_gate_h         # [rows, H]

    # --------------------------------------------------------------
    # 5️⃣ Tri‑multiplicative update (batched GEMM)
    # --------------------------------------------------------------
    # reshape for batched matmul: [B, N, N, H] → [B, H, N, N]
    left = left_h.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    right = right_h.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()

    BH = B * H
    left_mat = left.view(BH, N, N)          # [BH, N, N]
    right_mat = right.view(BH, N, N)        # [BH, N, N]

    # out = left @ rightᵀ  (sum over k)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [BH, N, N]

    # back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm (FP16)
    # --------------------------------------------------------------
    total_rows_h = B * N * N                     # rows for the flattened view
    out_flat = out.view(total_rows_h, H).contiguous()
    out_norm_h = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows_h, BLOCK_M),)
    layernorm_fp16_kernel[grid2](
        out_flat,
        out_norm_h,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows_h,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()

    # --------------------------------------------------------------
    # 7️⃣ Multiply by out_gate and final projection
    # --------------------------------------------------------------
    out_gate = out_gate_h.view(B, N, N, H)                 # FP16
    out_norm = out_norm_h.view(B, N, N, H) * out_gate      # element‑wise

    # final linear projection: hidden_dim → dim
    final_h = F.linear(
        out_norm.view(total_rows_h, H),
        weights["to_out.weight"].to(torch.float16),
    )   # [total_rows_h, dim]  (FP16)

    final = final_h.view(B, N, N, dim).to(torch.float32)
    return final
