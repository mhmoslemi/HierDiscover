def fused_norm_gate_linear_kernel(
    inp_ptr, gate_ptr,
    norm_w_ptr, norm_b_ptr,
    out_w_ptr, out_ptr,
    total_rows, hidden, dim,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_D: tl.constexpr
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # compute mean and var
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    # loop over hidden blocks
    for h in range(0, tl.cdiv(hidden, BLOCK_H)):
        offs_h = h * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < hidden
        # load input values
        inp = tl.load(inp_ptr + offs_m[:, None] * hidden + offs_h[None, :],
                      mask=mask_m[:, None] & mask_h[None, :],
                      other=0.0).to(tl.float32)
        sum_val += tl.sum(inp, axis=1)
        sum_sq += tl.sum(inp * inp, axis=1)

    mean = sum_val / hidden
    var = sum_sq / hidden - mean * mean
    rstd = tl.rsqrt(var + eps)

    # Outer loop over output dim blocks
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        # accumulator for this block
        acc = tl.zeros([BLOCK_M, BLOCK_D], dtype=tl.float32)

        # loop hidden
        for h in range(0, tl.cdiv(hidden, BLOCK_H)):
            offs_h = h * BLOCK_H + tl.arange(0, BLOCK_H)
            mask_h = offs_h < hidden

            # load raw input
            inp = tl.load(inp_ptr + offs_m[:, None] * hidden + offs_h[None, :],
                          mask=mask_m[:, None] & mask_h[None, :],

            # apply layernorm weight and bias per hidden feature

"""
Highly‑optimized forward implementation of the “outgoing” TriMul operation
(AlphaFold‑3).  

Algorithm
---------
1. Input LayerNorm – performed with torch.nn.functional.layer_norm (cuDNN).
2. Fused linear projection for left/right tensors and three gates
   (one GEMM on the concatenated weight matrix, executed in FP16).
3. Optional binary mask applied to the left/right projections.
4. Sigmoid gates and element‑wise gating (still FP16).
5. Triangular multiplicative update:
        out[i,j,d] = Σ_k left[i,k,d] * right[j,k,d]
   Implemented as a batched GEMM (torch.bmm) after a permute+reshape.
6. Fused LayerNorm + output gate (FP16) – a custom Triton kernel that
   computes mean/variance per row, normalises, scales+shifts, multiplies
   by the pre‑computed `out_gate`, and writes the result.
7. Final linear projection (`hidden → dim`) using torch.nn.functional.linear.
   The result is cast back to FP32 for the returned tensor.

Only the two Triton kernels (input norm is delegated to cuDNN) are required;
the rest relies on highly‑optimised cuBLAS/Tensor‑Core kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 6) Fused LayerNorm + output‑gate (both in FP16)
# ----------------------------------------------------------------------
@triton.jit
def fused_layernorm_gate_kernel(
    x_ptr,                     # *const half*  [total_rows, hidden]
    y_ptr,                     # *mut   half*  [total_rows, hidden] (post‑norm⋅gate)
    weight_ptr,                # *const float* [hidden] (γ)
    bias_ptr,                  # *const float* [hidden] (β)
    gate_ptr,                  # *const half*  [total_rows, hidden] (out_gate)
    total_rows,
    hidden: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # -----------------------------------------------------------------
    # Compute mean & variance (float32 accumulation)
    # -----------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for h in range(0, tl.cdiv(hidden, BLOCK_D)):
        offs_h = h * BLOCK_D + tl.arange(0, BLOCK_D)          # [BLOCK_D]
        mask_h = offs_h < hidden
        ptrs = x_ptr + offs_m[:, None] * hidden + offs_h[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_h[None, :],
                       other=0.0).to(tl.float32)          # [BLOCK_M, BLOCK_D]
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / hidden
    var  = sum_sq / hidden - mean * mean
    rstd = tl.rsqrt(var + eps)       # 1 / sqrt(var + eps)

    # -----------------------------------------------------------------
    # Second pass: normalise, apply γ, β and output‑gate, store as FP16
    # -----------------------------------------------------------------
    for h in range(0, tl.cdiv(hidden, BLOCK_D)):
        offs_h = h * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_h = offs_h < hidden

        # load raw hidden values
        ptrs = x_ptr + offs_m[:, None] * hidden + offs_h[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_h[None, :],
                       other=0.0).to(tl.float32)          # [BLOCK_M, BLOCK_D]

        # normalise
        norm = (vals - mean[:, None]) * rstd[:, None]          # (x-µ)/σ  (FP32)

        # apply LayerNorm weight (γ) and bias (β)
        w = tl.load(weight_ptr + offs_h, mask=mask_h, other=0.0)   # γ  (FP32)
        b = tl.load(bias_ptr   + offs_h, mask=mask_h, other=0.0)   # β  (FP32)
        norm = norm * w[None, :] + b[None, :]                    # (FP32)

        # multiply by pre‑computed out_gate (half → FP32)
        gate_vals = tl.load(gate_ptr + offs_m[:, None] * hidden + offs_h[None, :],
                            mask=mask_m[:, None] & mask_h[None, :],
                            other=0.0).to(tl.float32)          # (FP32)

        norm = norm * gate_vals

        # store result
        tl.store(y_ptr + offs_m[:, None] * hidden + offs_h[None, :],
                 norm.to(tl.float16),
                 mask=mask_m[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Entry‑point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
            input_tensor : torch.Tensor [B, N, N, C] (float32)
            mask_tensor  : torch.Tensor [B, N, N] (float32 or bool)
            weights_dict : dict of model parameters (float32)
            config_dict  : dict containing:
                - dim (int)          : input channel dimension C
                - hidden_dim (int)   : projection dimension H
                - nomask (bool)      : True → no mask
                - eps (float)        : LayerNorm epsilon (optional)

    Returns:
        torch.Tensor [B, N, N, C] (float32) on the same CUDA device.
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    inp, mask, weights, cfg = data
    device = inp.device
    dim = cfg["dim"]               # C
    hidden = cfg["hidden_dim"]     # H
    eps = cfg.get("eps", 1e-5)
    nomask = cfg.get("nomask", True)

    B, N, _, _ = inp.shape
    total_rows = B * N * N                         # rows after flattening

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (fast cuDNN implementation, output FP16)
    # ------------------------------------------------------------------
    # Convert weights to FP16 for the LayerNorm call (cuDNN expects same dtype)
    norm_w_fp16 = weights["norm.weight"].to(torch.half)
    norm_b_fp16 = weights["norm.bias"].to(torch.half)

    inp_half = inp.to(torch.half)                               # [B,N,N,C] FP16
    x_norm = F.layer_norm(inp_half,
                          normalized_shape=(dim,),
                          weight=norm_w_fp16,
                          bias=norm_b_fp16,
                          eps=eps)                         # FP16
    x_norm_flat = x_norm.reshape(total_rows, dim).contiguous()   # [R,C]

    # ------------------------------------------------------------------
    # 2) Fused linear projections + gates (single GEMM, FP16)
    # ------------------------------------------------------------------
    concat_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.half).contiguous()                              # (5*H, C)

    proj_all = F.linear(x_norm_flat, concat_w)                  # (R, 5*H) FP16
    left_proj, right_proj, left_gate, right_gate, out_gate = torch.split(
        proj_all, hidden, dim=1
    )  # each (R, H) FP16

    # ------------------------------------------------------------------
    # 2a) Optional binary mask
    # ------------------------------------------------------------------
    if not nomask:
        # mask is [B,N,N] → flatten to (R,1) and broadcast
        mask_flat = mask.view(total_rows, 1).to(inp.dtype).to(torch.half)  # (R,1) FP16
        left_proj = left_proj * mask_flat
        right_proj = right_proj * mask_flat

    # ------------------------------------------------------------------
    # 2b) Sigmoid gates (still FP16)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)          # (R, H) FP16

    # ------------------------------------------------------------------
    # 3) Apply gates to projections
    # ------------------------------------------------------------------
    left = left_proj * left_gate                     # (R, H) FP16
    right = right_proj * right_gate                  # (R, H) FP16

    # ------------------------------------------------------------------
    # 4) Reshape for the triangular update
    # ------------------------------------------------------------------
    left = left.view(B, N, N, hidden)                # (B,N,N,H)
    right = right.view(B, N, N, hidden)              # (B,N,N,H)

    # Batched GEMM: out[i,j] = Σ_k left[i,k] * right[j,k]
    # → left as (B*H,N,N), right transposed as (B*H,N,N)
    left_mat = left.permute(0, 3, 1, 2).reshape(B * hidden, N, N)   # (B*H,N,N)
    right_mat = right.permute(0, 3, 2, 1).reshape(B * hidden, N, N) # (B*H,N,N)
    out_mat = torch.bmm(left_mat, right_mat)                         # (B*H,N,N) FP16
    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 5) Second LayerNorm + output‑gate (fused Triton kernel)
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, hidden).contiguous()               # (R, H) FP16
    out_norm_gate = torch.empty_like(out_flat)                         # (R, H) FP16

    BLOCK_M = 128          # rows per program
    BLOCK_D = 32           # hidden‑dim chunk size (must divide hidden≈128)
    grid = (triton.cdiv(total_rows, BLOCK_M),)

    fused_layernorm_gate_kernel[grid](
        out_flat,
        out_norm_gate,
        weights["to_out_norm.weight"],   # γ  (float32)
        weights["to_out_norm.bias"],     # β  (float32)
        out_gate,                        # pre‑sigmoid‑ed gate (FP16)
        total_rows,
        hidden,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()

    # ------------------------------------------------------------------
    # 6) Final linear projection back to `dim`
    # ------------------------------------------------------------------
    to_out_w_fp16 = weights["to_out.weight"].to(torch.half)   # (dim, hidden)
    final_fp16 = F.linear(out_norm_gate, to_out_w_fp16)       # (R, dim) FP16
    final = final_fp16.view(B, N, N, dim).to(torch.float32)   # cast back to FP32

    return final
