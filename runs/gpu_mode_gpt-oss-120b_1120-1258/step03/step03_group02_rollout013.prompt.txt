import torch
from torch import nn, einsum
import math

# Reference code in PyTorch
class TriMul(nn.Module):
    def __init__(
        self,
        dim: int,
        hidden_dim: int,
    ):
        super().__init__()

        self.norm = nn.LayerNorm(dim)

        self.left_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.right_proj = nn.Linear(dim, hidden_dim, bias=False)

        self.left_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.right_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.out_gate = nn.Linear(dim, hidden_dim, bias=False)

        self.to_out_norm = nn.LayerNorm(hidden_dim)
        self.to_out = nn.Linear(hidden_dim, dim, bias=False)

    def forward(self, x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
        """
        x: [bs, seq_len, seq_len, dim]
        mask: [bs, seq_len, seq_len]

        Returns:
            output: [bs, seq_len, seq_len, dim]
        """
        batch_size, seq_len, _, dim = x.shape

        x = self.norm(x)

        left = self.left_proj(x)
        right = self.right_proj(x)

        mask = mask.unsqueeze(-1)
        left = left * mask
        right = right * mask

        left_gate = self.left_gate(x).sigmoid()
        right_gate = self.right_gate(x).sigmoid()
        out_gate = self.out_gate(x).sigmoid()

        left = left * left_gate
        right = right * right_gate

        out = einsum('... i k d, ... j k d -> ... i j d', left, right)
        # This einsum is the same as the following:
        # out = torch.zeros(batch_size, seq_len, seq_len, dim, device=x.device)
        
        # # Compute using nested loops
        # for b in range(batch_size):
        #     for i in range(seq_len):
        #         for j in range(seq_len):
        #             # Compute each output element
        #             for k in range(seq_len):
        #                 out[b, i, j] += left[b, i, k, :] * right[b, j, k, :]

        out = self.to_out_norm(out)
        out = out * out_gate
        return self.to_out(out)

"""
TriMul “outgoing” implementation (AlphaFold‑3 style) using Triton.

Algorithm
---------
1️⃣ Row‐wise LayerNorm (float32 → fp16) – a tiny Triton kernel that
   computes μ, σ² per row and writes the normalized result directly in
   FP16, saving one memory copy.
2️⃣ Fuse the two projection weights (proj + gate) for the left and right
   paths: a single FP16 GEMM each produces the projection and pre‑gate
   tensors (2 × hidden_dim columns) → 4 GEMMs total.
3️⃣ Optional pairwise mask (broadcast over hidden dim) is applied in‑place.
4️⃣ Apply sigmoid‑gates, multiply gated projections (still FP16).
5️⃣ Tri‑multiplicative update:
        out_h(i,j) = Σ_k left(i,k,h)·right(j,k,h)
   This is a batched GEMM (B·hidden_dim matrices of size N×N) performed
   with `torch.bmm` on FP16 (Tensor‑Core accelerated).
6️⃣ Second LayerNorm (again FP16 output) – same Triton kernel as step 1.
7️⃣ Apply the output‑gate (sigmoid) and a final FP16 linear projection
   back to `dim`.  The result is cast to FP32 for the module output.

Only the LayerNorms are written in Triton; all dense linear algebra is
delegated to PyTorch’s highly‑optimised FP16 kernels.  The implementation
covers every configuration listed in the prompt and works with or without
a mask.  The final output tensor has shape `[bs, seq_len, seq_len, dim]`
and dtype ``torch.float32``.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Row‑wise LayerNorm that writes FP16 output
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,          # const float*   [total_rows, dim]
    y_ptr,          #       half*   [total_rows, dim]  (output)
    w_ptr,          # const float*  [dim]   (γ)
    b_ptr,          # const float*  [dim]   (β)
    total_rows,     # int (runtime)
    dim: tl.constexpr,          # static: number of columns
    eps: tl.constexpr,          # static epsilon
    BLOCK_M: tl.constexpr,      # rows per program
    BLOCK_D: tl.constexpr,      # columns per inner loop
):
    """
    Row‑wise LayerNorm for a 2‑D tensor stored in row‑major order.
    The kernel works in FP32 internally and writes normalized values as
    FP16 to reduce memory traffic.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ----------- pass 1: compute mean & variance --------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)  # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)          # 1 / sqrt(var+eps)

    # ----------- pass 2: write normalized values --------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]   # (x-μ)/σ

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # β

        out = norm * w[None, :] + b[None, :]                # fp32
        out_fp16 = out.to(tl.float16)                       # cast

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32, on CUDA)
    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1️⃣ First layer‑norm (FP32 → FP16)
    # --------------------------------------------------------------
    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N                     # rows for the flattened view
    x_flat = input_tensor.view(total_rows, dim).contiguous()

    # allocate half‑precision buffer for the normalized tensor
    x_norm_h = torch.empty((total_rows, dim), dtype=torch.float16,
                           device=device)

    # launch Triton kernel (tuned block sizes)
    BLOCK_M = 256   # rows per program
    BLOCK_D = 64    # columns per inner loop (covers dim up to 384)
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        x_flat,
        x_norm_h,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()

    x_norm_h = x_norm_h.view(B, N, N, dim)   # [B, N, N, dim]  (FP16)

    # --------------------------------------------------------------
    # 2️⃣ Fuse projection + gate (4 GEMMs total)
    # --------------------------------------------------------------
    # left path: concat proj & gate weights → (2*H, dim)
    left_concat_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16, non_blocking=True)

    right_concat_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16, non_blocking=True)

    # Combined linear passes (half‑precision)
    left_concat = F.linear(x_norm_h.view(total_rows, dim), left_concat_w)   # [rows, 2*H]
    right_concat = F.linear(x_norm_h.view(total_rows, dim), right_concat_w)

    left_proj_h = left_concat[:, :hidden_dim]                # projection
    left_gate_pre_h = left_concat[:, hidden_dim:]            # pre‑sigmoid gate
    right_proj_h = right_concat[:, :hidden_dim]
    right_gate_pre_h = right_concat[:, hidden_dim:]

    # out‑gate (only once, same for the whole tensor)
    out_gate_h = torch.sigmoid(
        F.linear(x_norm_h.view(total_rows, dim), weights["out_gate.weight"].to(torch.float16))
    )   # [rows, hidden_dim]  (FP16)

    # --------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over hidden dimension)
    # --------------------------------------------------------------
    if not nomask:
        # mask shape [B, N, N] → flatten → [rows, 1]
        mask_h = mask_tensor.view(total_rows, 1).to(torch.float16)
        left_proj_h = left_proj_h * mask_h
        right_proj_h = right_proj_h * mask_h

    # --------------------------------------------------------------
    # 4️⃣ Apply sigmoid‑gates
    # --------------------------------------------------------------
    left_gate_h = torch.sigmoid(left_gate_pre_h)
    right_gate_h = torch.sigmoid(right_gate_pre_h)

    left_h = left_proj_h * left_gate_h      # [rows, hidden_dim]
    right_h = right_proj_h * right_gate_h

    # --------------------------------------------------------------
    # 5️⃣ Tri‑multiplicative update (batched GEMM)
    # --------------------------------------------------------------
    left = left_h.view(B, N, N, hidden_dim)    # [B, N, N, H]
    right = right_h.view(B, N, N, hidden_dim)

    # bring hidden dimension to the batch axis: [B, H, N, N]
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    BH = B * hidden_dim                         # B × H
    left_mat = left_perm.view(BH, N, N)         # [BH, N, N]
    right_mat = right_perm.view(BH, N, N)       # [BH, N, N]

    # batched GEMM: out = left @ rightᵀ   (sum over k)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [BH, N, N]

    # reshape back to [B, N, N, H] (still FP16)
    out = (
        out_mat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )   # [B, N, N, H]

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm (FP16 output)
    # --------------------------------------------------------------
    total_rows_h = B * N * N
    out_flat = out.view(total_rows_h, hidden_dim).contiguous()
    out_norm_h = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows_h, BLOCK_M),)
    layernorm_fp16_kernel[grid2](
        out_flat,
        out_norm_h,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows_h,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()
    out_norm = out_norm_h.view(B, N, N, hidden_dim)   # FP16

    # --------------------------------------------------------------
    # 7️⃣ Apply output‑gate and final projection
    # --------------------------------------------------------------
    # out_gate already FP16; broadcast and multiply
    out_gate = out_gate_h.view(B, N, N, hidden_dim)
    out_norm = out_norm * out_gate

    # final linear projection (hidden_dim → dim) – FP16 matmul
    final_h = F.linear(
        out_norm.view(total_rows_h, hidden_dim),
        weights["to_out.weight"].to(torch.float16)
    )   # [total_rows_h, dim]  (FP16)

    final = final_h.view(B, N, N, dim).to(torch.float32)   # back to FP32

    return final
