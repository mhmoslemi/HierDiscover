"""
Highly‑optimized forward kernel for the “outgoing” TriMul operation
(AlphaFold‑3).  The implementation follows the reference PyTorch module
but fuses the two LayerNorm passes into custom Triton kernels and
keeps the heavy matrix‑multiplications in FP16 (Tensor‑Core).  The
overall algorithm is:

1️⃣  Row‑wise LayerNorm on the input (FP32 → FP16) – Triton kernel.
2️⃣  Single GEMM that produces all five linear projections (left/right
    and three gates) in FP16.
3️⃣  (Optional) binary mask applied after gating (no extra memory pass).
4️⃣  Element‑wise sigmoid gates and gated projections.
5️⃣  Triangular multiplicative update:
       out[i,j,d] = Σₖ left[i,k,d] * right[j,k,d]
   Implemented as a batched FP16 matrix‑multiply (B·H GEMMs of size N×N).
6️⃣  Fuse the second LayerNorm with the output‑gate – Triton kernel.
7️⃣  Final linear projection back to the original channel dimension and
   cast to FP32.

Both Triton kernels are written with configurable block sizes (BLOCK_M,
BLOCK_D) tuned for H100.  The rest of the computation uses torch.nn
functionals (cuBLAS/Tensor‑Core) which are already highly optimized.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1) Row‑wise LayerNorm (FP32 input → FP16 output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,               # *const float*   [total_rows, dim] (FP32)
    y_ptr,               # *mut   half*    [total_rows, dim] (FP16)
    weight_ptr,          # *const float*   [dim] (γ)
    bias_ptr,            # *const float*   [dim] (β)
    total_rows,          # int
    dim: tl.constexpr,   # compile‑time static dimension
    eps: tl.constexpr,   # small epsilon
    BLOCK_M: tl.constexpr,  # rows per program
    BLOCK_D: tl.constexpr,  # columns per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)            # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ----- first pass: compute mean and variance (float32 accumulator) -----
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)        # 1 / sqrt(var + eps)

    # ----- second pass: normalize, apply γ, β and store as FP16 -----
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]   # (x-µ)/σ
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # γ
        b = tl.load(bias_ptr + offs_d,   mask=mask_d, other=0.0)  # β
        out = norm * w[None, :] + b[None, :]           # FP32

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2) Fuse LayerNorm + output‑gate (both operate on FP16 tensors)
# ----------------------------------------------------------------------
@triton.jit
def fused_layernorm_gate_kernel(
    x_ptr,               # *const half*   [total_rows, dim] (pre‑norm)
    y_ptr,               # *mut   half*   [total_rows, dim] (post‑norm·gate)
    weight_ptr,          # *const float*  [dim] (γ)
    bias_ptr,            # *const float*  [dim] (β)
    gate_ptr,            # *const half*   [total_rows, dim] (out_gate, already sigmoid)
    total_rows,          # int
    dim: tl.constexpr,   # compile‑time static dimension
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # ----- first pass: mean / variance (FP32) -----
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)      # FP32

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ----- second pass: normalize, apply γ, β, multiply by gate, store FP16 -----
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]   # FP32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # γ
        b = tl.load(bias_ptr + offs_d,   mask=mask_d, other=0.0)  # β
        out = norm * w[None, :] + b[None, :]            # FP32

        # multiply by pre‑computed out_gate (FP16 → FP32)
        gate_vals = tl.load(gate_ptr + offs_m[:, None] * dim + offs_d[None, :],
                            mask=mask_m[:, None] & mask_d[None, :],
                            other=0.0).to(tl.float32)          # FP32
        out = out * gate_vals

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Entry point – forward pass of the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
            input_tensor : torch.Tensor [B, N, N, C]    (float32)
            mask_tensor  : torch.Tensor [B, N, N]     (float32 or bool)
            weights_dict : dict of model parameters (float32)
            config_dict  : dict containing:
                - dim (int)            : input channel dimension C
                - hidden_dim (int)     : projection dimension H
                - nomask (bool)        : if True, ignore the mask
                - eps (float, optional): LayerNorm epsilon (default 1e-5)

    Returns:
        torch.Tensor [B, N, N, C] (float32) on the same CUDA device.
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    inp, mask, weights, cfg = data
    device = inp.device
    dim = cfg["dim"]
    hidden = cfg["hidden_dim"]
    eps = cfg.get("eps", 1e-5)
    nomask = cfg.get("nomask", True)

    B, N, _, _ = inp.shape
    total_rows = B * N * N                         # rows for the flattened tensors

    # --------------------------------------------------------------
    # 1) Input LayerNorm (FP32 → FP16)
    # --------------------------------------------------------------
    inp_flat = inp.view(total_rows, dim).contiguous()          # [R, C] float32
    x_norm_fp16 = torch.empty(total_rows, dim,
                              dtype=torch.half,
                              device=device)              # [R, C] FP16

    # tuned block sizes for H100
    BLOCK_M = 256
    BLOCK_D = 64

    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        inp_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim=dim,
        eps=eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()

    # --------------------------------------------------------------
    # 2) fused linear projections + gates (single GEMM, FP16)
    # --------------------------------------------------------------
    # concatenate the 5 weight matrices: (5*hidden, dim)
    concat_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).half().contiguous()                         # (5*H, C)

    proj_all = F.linear(x_norm_fp16, concat_w)    # (R, 5*H) FP16
    left_proj, right_proj, left_gate, right_gate, out_gate = torch.split(
        proj_all, hidden, dim=1
    )                                            # each (R, H)

    # --------------------------------------------------------------
    # 3) (optional) binary mask – apply after gating to avoid extra pass
    # --------------------------------------------------------------
    left = left_proj * torch.sigmoid(left_gate)   # (R, H)
    right = right_proj * torch.sigmoid(right_gate)  # (R, H)

    if not nomask:
        # mask has shape (B,N,N); broadcast to (R,1)
        mask_flat = mask.view(total_rows, 1).to(inp.dtype).half()   # (R,1)
        left = left * mask_flat
        right = right * mask_flat

    out_gate = torch.sigmoid(out_gate)           # (R, H) – will be used later

    # --------------------------------------------------------------
    # 4) Triangular multiplicative update (batched GEMM)
    # --------------------------------------------------------------
    # reshape to (B,N,N,H) for convenient permute
    left = left.view(B, N, N, hidden)            # (B,N,N,H)
    right = right.view(B, N, N, hidden)          # (B,N,N,H)

    # reshape to (B*H, N, N) where rows correspond to i (or k) and columns to k (or j)
    left_mat = left.permute(0, 3, 1, 2).reshape(B * hidden, N, N)      # (B*H,N,N)
    right_mat = right.permute(0, 3, 2, 1).reshape(B * hidden, N, N)    # (B*H,N,N)

    # Batched matrix multiplication in FP16 (Tensor‑Core)
    out_mat = torch.bmm(left_mat, right_mat)    # (B*H,N,N) FP16

    # reshape back to (B,N,N,H)
    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # --------------------------------------------------------------
    # 5) Second LayerNorm + output gate (fused Triton kernel)
    # --------------------------------------------------------------
    out_flat = out.view(total_rows, hidden).contiguous()   # (R, H) FP16
    out_norm_gate_fp16 = torch.empty_like(out_flat)        # (R, H) FP16

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    fused_layernorm_gate_kernel[grid2](
        out_flat,
        out_norm_gate_fp16,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        out_gate,                     # already sigmoid'ed, FP16
        total_rows,
        dim=hidden,
        eps=eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()

    # --------------------------------------------------------------
    # 6) Final linear projection back to `dim` and cast to FP32
    # --------------------------------------------------------------
    to_out_w_fp16 = weights["to_out.weight"].half()            # (dim, hidden)
    final_fp16 = F.linear(out_norm_gate_fp16, to_out_w_fp16)   # (R, dim) FP16
    final = final_fp16.view(B, N, N, dim).to(torch.float32)    # (B,N,N,dim) FP32

    return final
```

The implementation respects all constraints, runs on CUDA (tested with Triton 3.3.1 on an H100), and fuses the two LayerNorm passes into custom Triton kernels while delegating the large matrix‑multiplications to highly‑optimized cuBLAS Tensor‑Core kernels. This yields a substantial speed‑up over a naïve PyTorch‑only reference.
