"""
Optimized “outgoing” TriMul forward pass (AlphaFold‑3) using Triton.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the input tensor (shape [B,N,N,C]).
   A custom Triton kernel computes mean/variance in FP32 and writes the
   normalized result directly in FP16 (γ and β stay in FP32).

2️⃣ Fuse the five dense layers (left_proj, right_proj, left_gate,
   right_gate, out_gate) into a single half‑precision GEMM:
   `[R, C] @ [5*H, C]^T → [R, 5*H]` where `R = B·N·N`.
   Slicing the result yields the two projected tensors and the three
   gate vectors.

3️⃣ Optional binary mask (broadcast over the hidden dimension) is applied
   to the two projected tensors, then each projection is multiplied by its
   sigmoid gate.

4️⃣ Triangular multiplicative update:
   `out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]`.
   This is exactly a batched matrix multiplication
   `left[b,:, :, d] @ right[b,:, :, d].T` for every hidden channel `d`.
   We achieve it with a single `torch.matmul` on tensors of shape
   `[B, H, N, N]`, which lets cuBLAS/Tensor‑Cores handle the N³ work
   efficiently.

5️⃣ Second LayerNorm over the hidden dimension `H` – same Triton kernel
   as in step 1.

6️⃣ Multiply by the output gate and project back to the original channel
   dimension with a final FP16 linear layer.  The result is cast to
   `float32` before returning.

All heavy linear work runs in FP16, enabling full Tensor‑Core utilisation
on an H100.  Only the two LayerNorms are implemented in Triton; the rest
leverages PyTorch’s highly‑optimised GEMM kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Row‑wise LayerNorm kernel (FP32 accumulation, FP16 output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,            # const *  [total_rows, dim]   (fp32 or fp16)
    y_ptr,            # mutable *[total_rows, dim]   (fp16 output)
    weight_ptr,       # const *  [dim]               (fp32)
    bias_ptr,         # const *  [dim]               (fp32)
    total_rows,       # int (runtime)
    dim: tl.constexpr,      # compile‑time size of the channel dimension
    eps: tl.constexpr,      # compile‑time epsilon
    BLOCK_M: tl.constexpr,  # rows per program
    BLOCK_D: tl.constexpr,  # columns per program
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)        # [BLOCK_M]
    mask_m = offs_m < total_rows

    # -----------------------------------------------------------------
    # First pass: compute mean & variance (float32)
    # -----------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_block in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_block * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        # Load a BLOCK_M x BLOCK_D tile
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0)                         # respects FP16/FP32
        vals_f32 = vals.to(tl.float32)

        sum_val += tl.sum(vals_f32, axis=1)        # sum over d
        sum_sq  += tl.sum(vals_f32 * vals_f32, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                     # 1/√(var+eps)

    # -----------------------------------------------------------------
    # Second pass: write normalized output (γ,β are FP32)
    # -----------------------------------------------------------------
    for d_block in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_block * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0)
        vals_f32 = vals.to(tl.float32)

        norm = (vals_f32 - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # gamma
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # beta

        out = norm * w[None, :] + b[None, :]   # still fp32
        out_fp16 = out.to(tl.float16)

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom forward kernel (outgoing TriMul)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor, shape [B, N, N, C] (float32)
        - mask_tensor  : torch.Tensor, shape [B, N, N] (float32/bool) or None
        - weights_dict : mapping of all weight tensors (float32)
        - config_dict  : {"dim": C, "hidden_dim": H, "nomask": bool, ...}

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32, on CUDA)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, cfg = data
    dim = cfg["dim"]                     # input channel dimension C
    hidden_dim = cfg["hidden_dim"]       # hidden dimension H
    eps = 1e-5
    device = input_tensor.device
    nomask = cfg.get("nomask", True)

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N               # R = B·N·N

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → fp16
    # ------------------------------------------------------------------
    # flatten [R, C]
    x_flat = input_tensor.view(total_rows, dim).contiguous()
    x_norm_fp16 = torch.empty(total_rows, dim,
                              dtype=torch.float16,
                              device=device)

    BLOCK_M = 256
    BLOCK_D = 32
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fuse linear projections + gates into one GEMM
    # ------------------------------------------------------------------
    # concatenate weight matrices: shape [5*H, C]
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],   # [H, C]
            weights["right_proj.weight"],  # [H, C]
            weights["left_gate.weight"],   # [H, C]
            weights["right_gate.weight"],  # [H, C]
            weights["out_gate.weight"],    # [H, C]
        ],
        dim=0,
    ).to(torch.float16)

    # fused linear: [R, C] @ [5*H, C]^T → [R, 5*H]
    proj_gate = F.linear(x_norm_fp16, w_concat)      # fp16, shape [R, 5*H]

    H = hidden_dim
    left_proj   = proj_gate[:, 0:H]                # [R, H]
    right_proj  = proj_gate[:, H:2*H]
    left_gate_p = proj_gate[:, 2*H:3*H]
    right_gate_p= proj_gate[:, 3*H:4*H]
    out_gate_p  = proj_gate[:, 4*H:5*H]

    # sigmoid gates (fp16)
    left_gate   = torch.sigmoid(left_gate_p)
    right_gate  = torch.sigmoid(right_gate_p)
    out_gate    = torch.sigmoid(out_gate_p)

    # optional mask (broadcast over hidden dim)
    if not nomask:
        # mask shape [B, N, N] → [R, 1]
        mask_h = mask_tensor.to(torch.float16).view(total_rows, 1)
        left_proj  = left_proj * mask_h
        right_proj = right_proj * mask_h

    # apply gates
    left  = left_proj * left_gate   # [R, H]
    right = right_proj * right_gate # [R, H]

    # reshape back to 4‑D tensors
    left  = left.view(B, N, N, H)    # [B, N, N, H]
    right = right.view(B, N, N, H)   # [B, N, N, H]
    out_gate = out_gate.view(B, N, N, H)  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 3️⃣ Triangular multiplicative update (batched matmul)
    # ------------------------------------------------------------------
    # permute to (B, H, N, N) for a single torch.matmul call
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    # out_tmp = left_perm @ right_perm.transpose(-1, -2)
    out_tmp = torch.matmul(left_perm,
                           right_perm.transpose(-1, -2))   # (B, H, N, N)

    # bring back to original layout [B, N, N, H]
    out = out_tmp.permute(0, 2, 3, 1).contiguous()      # (B, N, N, H) fp16

    # ------------------------------------------------------------------
    # 4️⃣ Second LayerNorm over hidden dimension H
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, H).contiguous()
    out_norm_fp16 = torch.empty_like(out_flat, dtype=torch.float16)

    grid_ln2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid_ln2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    out_norm = out_norm_fp16.view(B, N, N, H)   # (B, N, N, H) fp16

    # ------------------------------------------------------------------
    # 5️⃣ Apply output gate and final projection back to C
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate               # element‑wise fp16
    out_norm_flat = out_norm.view(total_rows, H)

    # final linear: [R, H] @ [C, H]^T → [R, C]
    to_out_weight = weights["to_out.weight"].to(torch.float16)   # [C, H]
    final_fp16 = F.linear(out_norm_flat, to_out_weight)        # fp16, shape [R, C]

    final = final_fp16.view(B, N, N, dim).to(torch.float32)   # cast to FP32
    return final
```
