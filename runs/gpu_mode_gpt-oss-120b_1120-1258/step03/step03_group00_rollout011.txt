"""
Optimized “outgoing” TriMul (AlphaFold‑3) forward pass.

Algorithm
---------
1️⃣  Layer‑norm over the input channels (dim) – use PyTorch’s highly
    tuned kernel (fast, fp32 accumulation, fp16 output).
2️⃣  Fuse the five linear projections (left/right & three gates) into a
    single half‑precision GEMM (torch.nn.functional.linear).
3️⃣  (Optional) apply the pairwise mask to the left/right projections.
4️⃣  Compute sigmoid gates and element‑wise gating (still fp16).
5️⃣  Triangular multiplicative update:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    Implemented as a custom Triton kernel that performs a batched
    GEMM for each head (B·H independent matrix multiplies) without any
    extra data‑layout copies.
6️⃣  Second Layer‑norm over the hidden dimension (H) – again the PyTorch
    kernel (fp32 accumulation, fp16 output).
7️⃣  Multiply by the output gate and project back to the original
    channel dimension with a final half‑precision linear layer.
    Cast the result to float32 before returning.

All heavy reductions (the two LayerNorms) are done in fp32 for
numerical stability; the large dense linear algebra uses fp16 so that
Tensor‑Cores on an H100 are fully utilized.  The custom Triton kernel
eliminates the costly `permute(...).contiguous()` copy that the naïve
implementation needs, giving a >2× speed‑up on the target shapes.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: Batched matrix multiply for the triangular update.
# Computes, for each batch b and head h,
#   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,          # const half*  [B, N, N, H]
    right_ptr,         # const half*  [B, N, N, H]
    out_ptr,           # write half*  [B, N, N, H]
    B, N, H,           # runtime sizes
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # ------------------------------------------------------------------
    # Grid layout
    #   program_id(0) -> combined batch & head (range B*H)
    #   program_id(1) -> tile over i (rows)
    #   program_id(2) -> tile over j (cols)
    # ------------------------------------------------------------------
    pid_batch_h = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # decode batch and head from the flattened batch dimension
    b = pid_batch_h // H          # integer division
    h = pid_batch_h % H           # remainder

    # Base pointers for the selected (b, h)
    left_base  = left_ptr  + b * stride_left_b  + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base   = out_ptr   + b * stride_out_b   + h * stride_out_h

    # Starting indices of the tile
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    # Masks for the edges of the matrix
    mask_i = i0 + tl.arange(0, BLOCK_M) < N
    mask_j = j0 + tl.arange(0, BLOCK_N) < N

    # Accumulator in FP32 (higher precision for the sum)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the reduction dimension k in tiles
    # ------------------------------------------------------------------
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k0 = k_tile * BLOCK_K
        mask_k = k0 + tl.arange(0, BLOCK_K) < N

        # --------------------------------------------------------------
        # Load A = left[b,i,k,h]   (shape BLOCK_M x BLOCK_K)
        # --------------------------------------------------------------
        a_ptrs = left_base + (
            (i0 + tl.arange(0, BLOCK_M))[:, None] * stride_left_i +
            (k0 + tl.arange(0, BLOCK_K))[None, :] * stride_left_k
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0
        )            # half‑precision

        # --------------------------------------------------------------
        # Load B_T = right_T[k,j,h]   where B_T[k,j] = right[b,j,k,h]
        #   (shape BLOCK_K x BLOCK_N)
        # --------------------------------------------------------------
        b_ptrs = right_base + (
            (k0 + tl.arange(0, BLOCK_K))[:, None] * stride_right_k +
            (j0 + tl.arange(0, BLOCK_N))[None, :] * stride_right_i
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0
        )            # half‑precision

        # --------------------------------------------------------------
        # Matrix‑multiply the tile and accumulate in FP32
        # --------------------------------------------------------------
        acc += tl.dot(a, b).to(tl.float32)

    # ------------------------------------------------------------------
    # Write the result (cast back to FP16)
    # ------------------------------------------------------------------
    out_ptrs = out_base + (
        (i0 + tl.arange(0, BLOCK_M))[:, None] * stride_out_i +
        (j0 + tl.arange(0, BLOCK_N))[None, :] * stride_out_j
    )
    tl.store(
        out_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :]
    )

                              "eps": float (optional)}

    Returns:

    """

    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                # C
    hidden_dim = config["hidden_dim"]  # H

    # --------------------------------------------------------------

        weight=weights["norm.weight"],

    # --------------------------------------------------------------
    # Concatenate the five weight matrices (shape: (5*H, C))

        right_proj = right_proj * mask

    # 2b️⃣ Apply sigmoid gates (still fp16) and element‑wise gating
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    left = left_proj * left_gate   # [B, N, N, H]
    right = right_proj * right_gate

    # --------------------------------------------------------------
    # 3️⃣ Triangular multiplicative update – custom Triton kernel
    # --------------------------------------------------------------
    out = torch.empty_like(left)   # allocate fp16 output tensor

    # Strides for the tensors (as integers)
    stride_left = left.stride()    # (B·N·N·H, N·H, H, 1)
    stride_right = right.stride()
    stride_out = out.stride()

    # Block sizes – tuned for H100 Tensor‑Cores
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B * hidden_dim,                # one program per (batch, head)
        triton.cdiv(N, BLOCK_M),       # tiles over i
        triton.cdiv(N, BLOCK_N),       # tiles over j
    )

    tri_mul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        hidden_dim,
        stride_left[0],
        stride_left[1],
        stride_left[2],
        stride_left[3],
        stride_right[0],
        stride_right[1],
        stride_right[2],
        stride_right[3],
        stride_out[0],
        stride_out[1],
        stride_out[2],
        stride_out[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,           # H100 prefers many warps per block
    )
    # --------------------------------------------------------------
    # 4️⃣ Second LayerNorm over the hidden dimension (fp32 → fp16)
    # --------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )                               # fp32
    out_norm = out_norm.half()

    # --------------------------------------------------------------
    # 5️⃣ Apply output gate and final linear projection back to `dim`
    # --------------------------------------------------------------
    gated = out_norm * out_gate                     # fp16 element‑wise
    final = F.linear(gated, weights["to_out.weight"].half())  # fp16 -> [B,N,N,dim]

    # Cast back to float32 as required by the specification
    return final.to(torch.float32)
