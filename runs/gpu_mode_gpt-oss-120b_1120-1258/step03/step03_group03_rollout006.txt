"""
TriMul (outgoing) – fast forward pass

Algorithm (high‑level)
----------------------
1️⃣  Input ‑‑> Row‑wise LayerNorm (FP32) using the native torch kernel.
2️⃣  Cast the normalized tensor to FP16.
3️⃣  Fuse the five linear projections/gates (left, right, left_gate,
    right_gate, out_gate) into a single half‑precision GEMM.
4️⃣  Apply the optional pairwise mask to the left/right projections.
5️⃣  Apply element‑wise sigmoid to the three gate tensors and fuse the
    gates with the corresponding projections.
6️⃣  Reshape the gated tensors to (B, H, N, N) and perform the
    triangle‑multiplicative update as a batched GEMM
    (B·H independent matrix‑multiplications of size N×N).
7️⃣  Fuse the second LayerNorm (over the hidden dimension), the learned
    γ/β of the output‑norm and the element‑wise multiplication with
    `out_gate` into a single Triton kernel (`layernorm_gate_kernel`).
8️⃣  Final half‑precision linear projection back to the original channel
    dimension.
9️⃣  Cast the result back to FP32.

Only the second LayerNorm + output‑gate step is custom‑implemented in
Triton; the rest exploits highly‑optimized cuBLAS/Tensor‑Core kernels.
Mixed precision (FP16) is used for the bulk of the work, which yields
sub‑millisecond runtimes on an H100 for the target shapes.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel: hidden‑dim LayerNorm + element‑wise multiplication
# with `out_gate`.  Output is stored directly as FP16, avoiding an
# intermediate buffer.
# ----------------------------------------------------------------------
@triton.jit
def layernorm_gate_kernel(
    x_ptr,          # const half*   [total_rows, dim]   (input to LN)
    y_ptr,          #       half*   [total_rows, dim]   (LN output * gate)
    gate_ptr,       # const half*   [total_rows, dim]   (out_gate)
    weight_ptr,     # const half*   [dim]                (γ)
    bias_ptr,       # const half*   [dim]                (β)
    total_rows,     # i32
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """
    Row‑wise LayerNorm (FP32 accumulation) on a half‑precision matrix,
    followed by an element‑wise multiplication with a gate tensor.
    The result is written back as half‑precision.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)           # [BLOCK_M]
    mask_m = offs_m < total_rows

    # -----------------------------------------------------------------
    # 1st pass – compute mean / variance (FP32 accumulator)
    # -----------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0
        ).to(tl.float32)                               # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                           # 1 / sqrt(var+eps)

    # -----------------------------------------------------------------
    # 2nd pass – normalize, apply γ/β and multiply by gate
    # -----------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim

        # load raw x
        x_ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        x = tl.load(
            x_ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0
        ).to(tl.float32)

        # normalize
        norm = (x - mean[:, None]) * rstd[:, None]

        # γ, β (half → float32)
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0).to(tl.float32)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0).to(tl.float32)

        out = norm * w[None, :] + b[None, :]

        # gate (half → float32)
        g_ptrs = gate_ptr + offs_m[:, None] * dim + offs_d[None, :]
        gate = tl.load(
            g_ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0
        ).to(tl.float32)

        out = out * gate

        # store back as half
        y_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(y_ptrs, out.to(tl.float16), mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, D]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]    (bool/int/float, optional)
        - weights_dict : mapping from layer name to torch.Tensor
        - config_dict  : contains "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, D] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    dim        = config["dim"]            # D
    hidden_dim = config["hidden_dim"]    # H
    nomask     = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N               # R = B·N·N

    # ------------------------------------------------------------------
    # 1️⃣  First LayerNorm (use native cuDNN for speed)
    # ------------------------------------------------------------------
    # torch.nn.functional.layer_norm works on the last dimension.
    x_norm = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )                       # [B,N,N,D]  float32

    # Cast to half for the bulk of the computation
    x_norm_h = x_norm.to(torch.float16)          # [R, D] after flatten
    x_norm_h = x_norm_h.view(total_rows, dim)    # [R, D]

    # ------------------------------------------------------------------
    # 2️⃣  Fused linear projection for left / right and all gates
    # ------------------------------------------------------------------
    # Concatenate the five weight matrices (shape: (5*H, D))
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],   # (H, D)
            weights["left_gate.weight"],   # (H, D)
            weights["right_proj.weight"],  # (H, D)
            weights["right_gate.weight"],  # (H, D)
            weights["out_gate.weight"],    # (H, D)
        ],
        dim=0,
    ).to(torch.float16)  # (5*H, D)

    # fused GEMM: [R, D] @ (5*H, D)^T → [R, 5*H]
    proj = F.linear(x_norm_h, w_concat)          # half, shape (R,5*H)

    H = hidden_dim
    left_proj = proj[:, 0:H]
    left_gate = proj[:, H:2*H]
    right_proj = proj[:, 2*H:3*H]
    right_gate = proj[:, 3*H:4*H]
    out_gate   = proj[:, 4*H:5*H]

    # ------------------------------------------------------------------
    # 3️⃣  Optional mask (only on the two projection streams)
    # ------------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        # mask shape: [B, N, N] -> flatten -> [R, 1]
        mask_h = mask_tensor.to(torch.float16).view(total_rows, 1)
        left_proj  = left_proj * mask_h
        right_proj = right_proj * mask_h

    # ------------------------------------------------------------------
    # 4️⃣  Sigmoid gates (performed in FP32 for numerical stability)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate.float()).to(torch.float16)
    right_gate = torch.sigmoid(right_gate.float()).to(torch.float16)
    out_gate   = torch.sigmoid(out_gate.float()).to(torch.float16)

    # Apply the gates to the projections
    left  = left_proj * left_gate          # [R, H] half
    right = right_proj * right_gate        # [R, H] half

    # ------------------------------------------------------------------
    # 5️⃣  Triangle‑multiplicative update (batched GEMM)
    # ------------------------------------------------------------------
    # Reshape to 4‑D tensors and move the hidden dimension to the batch axis
    left  = left.view(B, N, N, H)           # [B, N, N, H]
    right = right.view(B, N, N, H)          # [B, N, N, H]

    # Convert to (B, H, N, N) for the batched matrix multiply
    left_mat  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_mat = right.permute(0, 3, 2, 1).contiguous()  # [B, H, N, N] (transpose k↔j)

    BH = B * H
    left_mat  = left_mat.view(BH, N, N)    # [B·H, N, N]
    right_mat = right_mat.view(BH, N, N)   # [B·H, N, N]

    # Batched GEMM: left_mat @ right_mat  ->  out_mat [B·H, N, N]
    out_mat = torch.bmm(left_mat, right_mat)   # half precision

    # Restore layout [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # half

    # ------------------------------------------------------------------
    # 6️⃣  Second LayerNorm + output gate (single Triton kernel)
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, H)                     # [R, H] half
    out_gate_flat = out_gate.view(total_rows, H)           # [R, H] half

    out_norm_gate = torch.empty_like(out_flat)             # half buffer

    # Parameters for the kernel
    BLOCK_M = 1024                # rows per program (tuned for large R)
    BLOCK_D = 32                  # hidden‑dim block size (H ≤ 128)
    grid = (triton.cdiv(total_rows, BLOCK_M),)

    layernorm_gate_kernel[grid](
        out_flat,
        out_norm_gate,
        out_gate_flat,
        weights["to_out_norm.weight"].to(torch.float16),   # γ (half)
        weights["to_out_norm.bias"].to(torch.float16),     # β (half)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 7️⃣  Final linear projection back to the original channel dimension
    # ------------------------------------------------------------------
    w_to_out = weights["to_out.weight"].to(torch.float16)   # (D, H)
    final_half = F.linear(out_norm_gate, w_to_out)          # [R, D] half
    final = final_half.view(B, N, N, dim).to(torch.float32) # cast back to FP32

    return final
