"""
TriMul “outgoing” implementation (AlphaFold‑3 style) using Triton.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the input tensor (FP32 → FP16) – a lightweight
   Triton kernel that computes mean / variance per row and writes the
   normalized values directly in FP16, avoiding an extra FP32 copy.

2️⃣ Fuse the five linear projections (left_proj, left_gate, right_proj,
   right_gate, out_gate) into a single GEMM:
        Y = X_norm @ W_all.T
   where `W_all` concatenates the five weight matrices along the output‑
   feature axis. This reduces three passes over the large input matrix
   (`X_norm`) to a single pass, cutting memory traffic by ~2×.

3️⃣ Split `Y` into the five components, apply the optional pairwise mask,
   compute sigmoid‑gates and obtain the gated tensors `left` and `right`.

4️⃣ Batched GEMM for the multiplicative update:
        out_h(i,j) = Σ_k left(i,k,h) · right(j,k,h)
   Implemented as a single `torch.bmm` on a view of shape
   (B·H, N, N).

5️⃣ Second LayerNorm over the hidden dimension – the same Triton kernel as
   in step 1, operating on FP16 and writing FP16 results.

6️⃣ Multiply by the output‑gate (already sigmoid‑activated) and apply the
   final linear projection back to `dim`.

Only the LayerNorms are written in Triton; all dense linear algebra (the
GEMMs, sigmoid, masking, and the final projection) uses PyTorch’s highly‑
optimised FP16 kernels.  The result is a `float32` tensor of shape
`[bs, seq_len, seq_len, dim]`.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Row‑wise LayerNorm writing FP16 output (used for both normalizations)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,          # const float*   [total_rows, dim]  (FP32 input)
    y_ptr,          #       half*   [total_rows, dim]  (FP16 output)
    w_ptr,          # const float*  [dim] (γ)
    b_ptr,          # const float*  [dim] (β)
    total_rows,     # int (runtime)
    dim: tl.constexpr,          # static: number of columns
    eps: tl.constexpr,          # static epsilon
    BLOCK_M: tl.constexpr,      # number of rows processed per program
    BLOCK_D: tl.constexpr,      # column tile size
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------- 1st pass: compute mean & variance per row -----------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)   # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------- 2nd pass: write normalized + affine transformed values ----
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]   # (x-μ)/σ

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # β

        out = norm * w[None, :] + b[None, :]                # fp32
        out_fp16 = out.to(tl.float16)

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Triton‑accelerated forward pass of the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

            input_tensor : torch.Tensor [bs, seq_len, seq_len, dim]   (float32)
            mask_tensor  : torch.Tensor [bs, seq_len, seq_len]       (bool/float)
            weights_dict : mapping of parameter names → torch.Tensor
            config_dict  : {"dim": int, "hidden_dim": int, "nomask": bool, ...}

    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ First layer‑norm (row‑wise, FP32→FP16)
    # ------------------------------------------------------------------
    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N                     # rows after flattening (M)
    x_flat = input_tensor.view(total_rows, dim).contiguous()  # FP32

    x_norm_h = torch.empty((total_rows, dim), dtype=torch.float16, device=device)

    # launch Triton LayerNorm (tuned block sizes)
    BLOCK_M = 256
    BLOCK_D = 64                     # works for dim up to 384
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        x_flat,
        x_norm_h,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()

    # reshape back to 4‑D tensor (still FP16)
    x_norm_h = x_norm_h.view(B, N, N, dim)   # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2️⃣ Fuse all five linear projections into one GEMM
    # ------------------------------------------------------------------
    # concatenate weight matrices: [left_proj, left_gate, right_proj, right_gate, out_gate]
    all_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,  # concatenate along output‑feature axis
    ).to(torch.float16, non_blocking=True)   # shape (5*H, dim)

    # perform the single GEMM (rows = total_rows)
    y = F.linear(
        x_norm_h.view(total_rows, dim),   # (M, dim)
        all_weight,                       # (5*H, dim)
    )  # result: (M, 5*hidden_dim), dtype=half

    # ------------------------------------------------------------------
    # 3️⃣ Split the fused result and apply mask / sigmoid gates
    # ------------------------------------------------------------------
    H = hidden_dim
    left_proj_h      = y[:, 0 * H : 1 * H]      # (M, H)
    left_gate_pre_h  = y[:, 1 * H : 2 * H]
    right_proj_h     = y[:, 2 * H : 3 * H]
    right_gate_pre_h = y[:, 3 * H : 4 * H]
    out_gate_pre_h   = y[:, 4 * H : 5 * H]

    if not nomask:
        mask_h = mask_tensor.view(total_rows, 1).to(torch.float16)
        left_proj_h = left_proj_h * mask_h
        right_proj_h = right_proj_h * mask_h

    left_gate_h  = torch.sigmoid(left_gate_pre_h)
    right_gate_h = torch.sigmoid(right_gate_pre_h)
    out_gate_h   = torch.sigmoid(out_gate_pre_h)

    left_h  = left_proj_h * left_gate_h      # (M, H)
    right_h = right_proj_h * right_gate_h    # (M, H)

    # ------------------------------------------------------------------
    # 4️⃣ Reshape for the tri‑multiplicative update
    # ------------------------------------------------------------------
    left_h  = left_h.view(B, N, N, H)          # [B, N, N, H]
    right_h = right_h.view(B, N, N, H)

    # bring hidden dimension to the batch axis: (B, H, N, N)
    left_perm  = left_h.permute(0, 3, 1, 2).contiguous()
    right_perm = right_h.permute(0, 3, 1, 2).contiguous()

    BH = B * H
    left_mat  = left_perm.view(BH, N, N)      # (BH, N, N)
    right_mat = right_perm.view(BH, N, N)     # (BH, N, N)

    # batched GEMM: out = left @ rightᵀ   (sum over k)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (BH, N, N)

    # reshape back to [B, N, N, H] (still FP16)
    out = (
        out_mat.view(B, H, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    total_rows_h = B * N * N
    out_flat = out.view(total_rows_h, H).contiguous()
    out_norm_h = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows_h, BLOCK_M),)
    layernorm_fp16_kernel[grid2](
        out_flat,
        out_norm_h,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows_h,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()
    out_norm = out_norm_h.view(B, N, N, H)    # FP16

    # ------------------------------------------------------------------
    # 6️⃣ Apply output‑gate and final projection back to `dim`
    # ------------------------------------------------------------------
    out_gate = out_gate_h.view(B, N, N, H)
    out_norm = out_norm * out_gate            # element‑wise (FP16)

    final_h = F.linear(
        out_norm.view(total_rows_h, H),
        weights["to_out.weight"].to(torch.float16, non_blocking=True),
    )  # (total_rows_h, dim)  FP16

    final = final_h.view(B, N, N, dim).to(torch.float32)   # back to FP32
    return final
