"""
Highly‑optimized “outgoing” TriMul (AlphaFold‑3) forward pass.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the input channels (C) – custom Triton kernel.
2️⃣ Fuse the five linear projections (left, right and three gates) into one
   fp16 GEMM (torch.nn.functional.linear).
3️⃣ Optional mask applied to left/right projections.
4️⃣ Sigmoid gates → element‑wise gated tensors L, R (shape [B,N,N,H]).
5️⃣ Triangular multiplicative update:
        out[i,j] = Σ_k L[i,k] · R[j,k]
   realized as a batched fp16 GEMM `L @ Rᵀ` (torch.bmm).
6️⃣ Fuse the second LayerNorm (over H) with the output‑gate scaling in a
   single Triton kernel, producing the tensor ready for the final projection.
7️⃣ Final fp16 linear projection back to C, cast to fp32.

Only the two LayerNorm passes are written in Triton (the second one also
applies the per‑position output gate, saving a separate element‑wise
multiplication and one memory write).  All heavy dense algebra uses
Tensor‑Core‑accelerated fp16 GEMMs, giving sub‑millisecond runtimes on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton LayerNorm (Row‑wise, fp32 reduction, fp16/fp32 output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # *const   float*   [total_rows, dim]
    y_ptr,          # *mut     float*   [total_rows, dim]
    weight_ptr,     # *const   float*   [dim]
    bias_ptr,       # *const   float*   [dim]
    total_rows,     # int (runtime)
    dim: tl.constexpr,          # int (compile‑time)
    eps: tl.constexpr,          # float (compile‑time)
    BLOCK_M: tl.constexpr,      # rows per program
    BLOCK_D: tl.constexpr,      # cols per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # --------------------------------------------------------------
    # 1️⃣ mean & variance (fp32)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim

        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                               # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                 # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                          # 1 / sqrt(var+eps)

    # --------------------------------------------------------------
    # 2️⃣ write normalized output (γ,β) – fp32 → fp16 or fp32 depending
    # --------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)

        out = norm * w[None, :] + b[None, :]

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out, mask=mask_m[:, None] & mask_d[None, :])

    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]

    # --------------------------------------------------------------
    # 1️⃣ mean & variance (fp32)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim

        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]

            ptrs,

        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------------------------------------------------------------
    # 2️⃣ normalized output * (γ,β) * per‑element scale
    # --------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out.to(tl.float16), mask=mask_m[:, None] & mask_d[None, :])

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor of shape [B, N, N, C] (float32)
            - mask_tensor  : torch.Tensor of shape [B, N, N] (bool/float)
            - weights_dict : mapping from layer names to weight tensors
            - config_dict  : {"dim": C, "hidden_dim": H, "nomask": bool (opt.)}

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]               # input channel dimension C
    hidden_dim = config["hidden_dim"] # hidden dimension H
    eps = 1e-5
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Prepare basic shapes
    # ------------------------------------------------------------------
    x = input_tensor.contiguous()
    B, N, _, _ = x.shape
    total_rows = B * N * N                     # rows for the flattened view

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm over the input channel dimension C
    # ------------------------------------------------------------------
    x_flat = x.view(total_rows, dim)           # [total_rows, C] (float32)
    x_norm_flat = torch.empty_like(x_flat)     # fp32 buffer for normalized values

    BLOCK_M = 256
    BLOCK_D = 64
    grid_x = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid_x](
        x_flat,
        x_norm_flat,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fuse the five linear projections (left/right + 3 gates)
    # ------------------------------------------------------------------
    x_norm_fp16 = x_norm_flat.half()             # [total_rows, C]
    # Concatenate weight matrices (each [H, C])
    w_left   = weights["left_proj.weight"]
    w_right  = weights["right_proj.weight"]
    w_lgate  = weights["left_gate.weight"]
    w_rgate  = weights["right_gate.weight"]
    w_outgate= weights["out_gate.weight"]

    proj_weight = torch.cat(
        [w_left, w_right, w_lgate, w_rgate, w_outgate],
        dim=0,
    ).half().contiguous()                        # shape (5*H, C)

    # Fused linear: [total_rows, C] @ [C, 5*H]^T → [total_rows, 5*H]
    proj_all = F.linear(x_norm_fp16, proj_weight)  # fp16

    left_proj_fp16, right_proj_fp16, left_gate_fp16, right_gate_fp16, out_gate_fp16 = torch.split(
        proj_all, hidden_dim, dim=1
    )

    # ------------------------------------------------------------------
    # 3️⃣ Optional masking (applied before gating)
    # ------------------------------------------------------------------
    if not nomask:
        mask_flat = mask_tensor.contiguous().view(total_rows, 1).to(torch.float32)
        mask_fp16 = mask_flat.half()
        left_proj_fp16  = left_proj_fp16 * mask_fp16
        right_proj_fp16 = right_proj_fp16 * mask_fp16

    # ------------------------------------------------------------------
    # 4️⃣ Apply sigmoid gates and element‑wise gating
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate_fp16)   # [total_rows, H]
    right_gate = torch.sigmoid(right_gate_fp16)
    out_gate   = torch.sigmoid(out_gate_fp16)    # [total_rows, H]

    left_fp16  = left_proj_fp16 * left_gate      # [total_rows, H]
    right_fp16 = right_proj_fp16 * right_gate    # [total_rows, H]

    # ------------------------------------------------------------------
    # 5️⃣ Triangular multiplicative update (batched GEMM)
    #    out_h[i, j] = Σ_k left_h[i, k] * right_h[j, k]
    # ------------------------------------------------------------------
    left = left_fp16.view(B, N, N, hidden_dim)          # [B, N, N, H]
    right = right_fp16.view(B, N, N, hidden_dim)        # [B, N, N, H]
    out_gate = out_gate.view(B, N, N, hidden_dim)      # [B, N, N, H]

    # Rearrange to (B, H, N, N) → (B*H, N, N)
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # transpose on k‑axis

    BH = B * hidden_dim
    left_mat  = left_perm.view(BH, N, N)   # [B*H, N, N]
    right_mat = right_perm.view(BH, N, N)  # [B*H, N, N]

    out_mat = torch.bmm(left_mat, right_mat)               # fp16, [B*H, N, N]

    # Flatten the result for the second LayerNorm
    total_rows_h = B * N * N
    out_flat = out_mat.view(total_rows_h, hidden_dim)      # [total_rows_h, H]
    out_gate_flat = out_gate.view(total_rows_h, hidden_dim)  # same shape, fp16

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm (over hidden dim) + output‑gate scaling
    # ------------------------------------------------------------------
    out_scaled_flat = torch.empty_like(out_flat)           # fp16 buffer

    grid_y = (triton.cdiv(total_rows_h, BLOCK_M),)
    layernorm_scale_kernel[grid_y](
        out_flat,
        out_scaled_flat,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        out_gate_flat,
        total_rows_h,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 7️⃣ Final linear projection back to input dimension C
    # ------------------------------------------------------------------
    to_out_w_fp16 = weights["to_out.weight"].half()
    final_fp16 = F.linear(out_scaled_flat, to_out_w_fp16)   # [total_rows_h, C]

    # ------------------------------------------------------------------
    # 8️⃣ Reshape & cast to fp32
    # ------------------------------------------------------------------
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
