"""
Optimised “outgoing” TriMul (AlphaFold‑3) forward pass.

Algorithm
---------
1️⃣  Layer‑norm over the input channel *dim* (custom Triton kernel, fp16 output,
    fp32 accumulation for numeric stability).
2️⃣  Fuse the five linear projections (left/right and three gates) into a single
    GEMM:  [B·N·N, dim] × [dim, 5·hidden_dim]ᵀ → [B·N·N, 5·hidden_dim].
3️⃣  Optional pair‑wise mask (applied in‑place before gating).
4️⃣  Apply sigmoid gates and element‑wise gating (all fp16).
5️⃣  Triangular multiplicative update:
       out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
    realised as a batched matrix‑multiply:
       left_mat @ right_matᵀ   with batch = B·hidden_dim.
6️⃣  Second Layer‑norm over the hidden dimension (same Triton kernel).
7️⃣  In‑place multiplication by the output gate and a final linear projection
    back to *dim* (fp16 GEMM).  Cast result to fp32 before returning.

Only the LayerNorms are custom Triton kernels; the heavy dense linear algebra
uses cuBLAS/Tensor‑Core GEMMs (torch.nn.functional.linear, torch.bmm),
which gives the best performance on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton LayerNorm – fp16 output, fp32 accumulation
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,          # *const float*   [total_rows, dim]   (input, fp32)
    y_ptr,          # *mut   half*    [total_rows, dim]   (output, fp16)
    weight_ptr,     # *const float*   [dim]              (γ)
    bias_ptr,       # *const float*   [dim]              (β)
    total_rows,     # int (runtime)
    dim: tl.constexpr,          # int (compile‑time)
    eps: tl.constexpr,          # float (compile‑time)
    BLOCK_M: tl.constexpr,      # rows per program
    BLOCK_D: tl.constexpr,      # columns per tile
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # -----------------------------------------------------------------
    # Pass 1: compute row‑wise mean & variance (fp32)
    # -----------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)       # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # -----------------------------------------------------------------
    # Pass 2: write normalized values (fp16)
    # -----------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)       # fp32 raw data

        norm = (vals - mean[:, None]) * rstd[:, None]   # fp32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)

        out_fp32 = norm * w + b          # fp32
        out_fp16 = out_fp32.to(tl.float16)

        out_ptrs = y_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim], float32
            - mask_tensor : torch.Tensor [B, N, N] (float or bool)
            - weights_dict: dict of model parameters (float32)
            - config_dict : dict with keys "dim", "hidden_dim", "nomask" (optional)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                 # C
    hidden_dim = config["hidden_dim"]   # H
    eps = 1e-5
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Shapes
    # ------------------------------------------------------------------
    B, N, _, _ = input_tensor.shape                # [B, N, N, C]
    total_rows = B * N * N                         # rows for flattened view

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (custom Triton)
    # ------------------------------------------------------------------
    x_flat = input_tensor.view(total_rows, dim)    # no copy, just a view
    x_norm_fp16 = torch.empty((total_rows, dim), dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 32
    grid = (triton.cdiv(total_rows, BLOCK_M),)

    layernorm_fp16_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim=dim,
        eps=eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear projections & gates
    # ------------------------------------------------------------------
    # Concatenate the five weight matrices (shape: (5*H, C)) and cast once
    concat_weight = torch.cat([
        weights["left_proj.weight"],
        weights["right_proj.weight"],
        weights["left_gate.weight"],
        weights["right_gate.weight"],
        weights["out_gate.weight"],
    ], dim=0).half().contiguous()                  # (5*H, C)

    # Single GEMM: [total_rows, C] @ [C, 5*H]ᵀ → [total_rows, 5*H]
    proj_all = F.linear(x_norm_fp16, concat_weight)   # fp16

    # Split into the five tensors
    left_proj, right_proj, left_gate, right_gate, out_gate = torch.split(
        proj_all, hidden_dim, dim=1
    )  # each: [total_rows, H] (view, no copy)

    # ------------------------------------------------------------------
    # 2a️⃣ Optional mask (apply in‑place)
    # ------------------------------------------------------------------
    if not nomask:
        mask_flat = mask_tensor.view(total_rows, 1).to(dtype=torch.float16)
        left_proj.mul_(mask_flat)
        right_proj.mul_(mask_flat)

    # ------------------------------------------------------------------
    # 2b️⃣ Sigmoid gates (still fp16)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    # Element‑wise gating (in‑place to save a write)
    left_proj.mul_(left_gate)
    right_proj.mul_(right_gate)

    # ------------------------------------------------------------------
    # Reshape to 4‑D tensors for the triangular update
    # ------------------------------------------------------------------
    left = left_proj.view(B, N, N, hidden_dim)          # [B, N, N, H]
    right = right_proj.view(B, N, N, hidden_dim)        # [B, N, N, H]
    out_gate = out_gate.view(B, N, N, hidden_dim)      # [B, N, N, H]

    # ------------------------------------------------------------------
    # 3️⃣ Triangular multiplicative update (batched GEMM)
    # ------------------------------------------------------------------
    # Bring the hidden dimension to the batch axis and make matrices contiguous
    left_perm = left.permute(0, 3, 1, 2).contiguous()    # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    BH = B * hidden_dim
    left_mat = left_perm.view(BH, N, N)                  # [B*H, N, N]
    right_mat = right_perm.view(BH, N, N)                # [B*H, N, N]

    # Batched matmul: left @ rightᵀ  →  out[b,i,j,d]
    out_mat = torch.bmm(left_mat, right_mat.transpose(-2, -1))   # fp16

    # Reshape back to [B, N, N, H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 4️⃣ Second LayerNorm (custom Triton)
    # ------------------------------------------------------------------
    total_rows_h = B * N * N
    out_flat = out.view(total_rows_h, hidden_dim)       # [B·N·N, H]
    out_norm_fp16 = torch.empty_like(out_flat)          # fp16 result

    grid2 = (triton.cdiv(total_rows_h, BLOCK_M),)
    layernorm_fp16_kernel[grid2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows_h,
        dim=hidden_dim,
        eps=eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 5️⃣ Multiply by output gate (in‑place) and final projection
    # ------------------------------------------------------------------
    out_gate_flat = out_gate.view(total_rows_h, hidden_dim)
    out_norm_fp16.mul_(out_gate_flat)   # in‑place gating

    # Final linear projection H → C
    final_fp16 = F.linear(out_norm_fp16, weights["to_out.weight"].half())  # fp16

    # Cast back to fp32 as required by the specification
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
