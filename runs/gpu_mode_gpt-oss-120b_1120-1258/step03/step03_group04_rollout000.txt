"""
Highly‑optimized outgoing TriMul (AlphaFold‑3) forward pass.

Algorithm
---------
1️⃣ Row‑wise LayerNorm over the input channel (C) – custom Triton kernel
   (FP32 accumulation, FP16 output).

2️⃣ Fuse the 5 linear projections (left, right, 3 gates) into a single
   FP16 GEMM by concatenating their weight matrices.

3️⃣ Optional mask applied to the left/right projections before gating.

4️⃣ Apply sigmoid gates and element‑wise gating (still FP16).

5️⃣ Tri‑multiplicative update:
      out_h[i, j] = Σ_k left_h[i, k] * right_h[j, k]
   realized as a batched GEMM `left @ rightᵀ` on tensors of shape
   (B·H, N, N) in FP16.

6️⃣ Second row‑wise LayerNorm over the hidden dimension (H) – same Triton
   kernel (FP32 accumulation, FP16 output).

7️⃣ Multiply by the output gate and a final FP16 linear projection back
   to the original channel dimension C.

8️⃣ Cast to FP32 for the final result.

Only the two LayerNorms are written in Triton; all heavy dense linear
algebra uses cuBLAS/Tensor‑Core FP16 GEMMs.  This design keeps memory
traffic low and matches the performance characteristics of the
reference implementation while improving runtime by eliminating
unnecessary FP32↔FP16 copies.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton row‑wise LayerNorm (FP32 reduction, FP16 output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # *const   float*   [total_rows, dim]       (input FP32)
    y_ptr,          # *mut     half*    [total_rows, dim]       (output FP16)
    w_ptr,          # *const   float*   [dim]                   (γ, FP32)
    b_ptr,          # *const   float*   [dim]                   (β, FP32)
    total_rows,     # i32 (runtime)
    dim: tl.constexpr,          # i32 compile‑time
    eps: tl.constexpr,          # fp32 compile‑time
    BLOCK_M: tl.constexpr,      # rows per program
    BLOCK_D: tl.constexpr,      # cols per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # -------------------------------------------------
    # 1️⃣ Compute mean & variance (FP32)
    # -------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim

        # Load a tile [BLOCK_M, BLOCK_D] as FP32
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                # 1 / sqrt(var+eps)

    # -------------------------------------------------
    # 2️⃣ Write normalized output as FP16
    # -------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        # (x - mean) / sqrt(var+eps)
        norm = (vals - mean[:, None]) * rstd[:, None]

        # γ (weight) and β (bias) – still FP32
        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)

        out_f32 = norm * w[None, :] + b[None, :]
        out_f16 = out_f32.to(tl.float16)

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_f16, mask=mask_m[:, None] & mask_d[None, :])


def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, C], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float), may be omitted when ``nomask=True``
        - weights_dict : dict with the following keys (all float32 tensors):
              "norm.weight", "norm.bias",
              "left_proj.weight", "right_proj.weight",
              "left_gate.weight", "right_gate.weight", "out_gate.weight",
              "to_out_norm.weight", "to_out_norm.bias",
              "to_out.weight"
        - config_dict  : dict with keys "dim", "hidden_dim", "nomask" (optional)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, C], dtype float32, on the same CUDA device.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]               # C
    hidden_dim = config["hidden_dim"] # H
    eps = 1e-5
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Flatten input for LayerNorm
    # ------------------------------------------------------------------
    x = input_tensor.contiguous()
    B, N, _, _ = x.shape
    total_rows = B * N * N                     # rows for flattened view

    x_flat = x.view(total_rows, dim)           # [total_rows, C] (FP32)

    # Decide block sizes (tuned for H100, N up to 1024)
    BLOCK_M = 1024 if total_rows >= 1024 * 512 else 256
    BLOCK_D = 64  # works for dim ≤ 384

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP16 output)
    # ------------------------------------------------------------------
    x_norm_fp16 = torch.empty((total_rows, dim), dtype=torch.float16, device=device)
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fuse five linear projections (left, right, three gates)
    # ------------------------------------------------------------------
    # Convert to half for the GEMM
    # (x_norm_fp16 already half)
    # Concatenate weight matrices (each [hidden_dim, dim])
    w_left   = weights["left_proj.weight"]
    w_right  = weights["right_proj.weight"]
    w_lgate  = weights["left_gate.weight"]
    w_rgate  = weights["right_gate.weight"]
    w_outgate= weights["out_gate.weight"]

    concat_w = torch.cat(
        [w_left, w_right, w_lgate, w_rgate, w_outgate],
        dim=0,
    ).half().contiguous()                      # shape (5*H, C)

    # Fused projection: [total_rows, C] @ [C, 5*H]^T -> [total_rows, 5*H]
    proj_all = F.linear(x_norm_fp16, concat_w)  # FP16 result

    # Split back into components
    left_proj_fp16, right_proj_fp16, left_gate_fp16, right_gate_fp16, out_gate_fp16 = torch.split(
        proj_all, hidden_dim, dim=1
    )

    # ------------------------------------------------------------------
    # 3️⃣ Optional masking (applied before gating)
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] -> [total_rows, 1] (FP16)
        mask_fp16 = mask_tensor.contiguous().view(total_rows, 1).to(torch.float16)
        left_proj_fp16 = left_proj_fp16 * mask_fp16
        right_proj_fp16 = right_proj_fp16 * mask_fp16

    # ------------------------------------------------------------------
    # 4️⃣ Apply sigmoid gates and element‑wise gating (still FP16)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate_fp16)   # [total_rows, H]
    right_gate = torch.sigmoid(right_gate_fp16)
    out_gate   = torch.sigmoid(out_gate_fp16)

    left_fp16  = left_proj_fp16 * left_gate      # [total_rows, H]
    right_fp16 = right_proj_fp16 * right_gate    # [total_rows, H]

    # Reshape to 4‑D tensors
    left  = left_fp16.view(B, N, N, hidden_dim)    # [B, N, N, H]
    right = right_fp16.view(B, N, N, hidden_dim)   # [B, N, N, H]
    out_gate = out_gate.view(B, N, N, hidden_dim) # [B, N, N, H]

    # ------------------------------------------------------------------
    # 5️⃣ Triangular multiplicative update (batched GEMM)
    # ------------------------------------------------------------------
    # Permute to (B, H, N, N) and collapse batch+heads
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    # right needs transpose on the k‑axis (the last two dimensions)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # [B, H, N, N]

    BH = B * hidden_dim
    left_mat  = left_perm.view(BH, N, N)   # [B*H, N, N] (FP16)
    right_mat = right_perm.view(BH, N, N)  # [B*H, N, N] (FP16)

    # Batched matrix multiplication (Tensor‑Core accelerated)
    out_mat = torch.bmm(left_mat, right_mat)  # [B*H, N, N] (FP16)

    # Reshape back to original layout
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm over hidden dimension H
    # ------------------------------------------------------------------
    total_rows_h = B * N * N
    out_flat = out.view(total_rows_h, hidden_dim)       # [total_rows_h, H] (FP16)
    out_norm_fp16 = torch.empty_like(out_flat)          # FP16 buffer

    grid2 = (triton.cdiv(total_rows_h, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows_h,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    out_norm = out_norm_fp16.view(B, N, N, hidden_dim)  # [B, N, N, H] (FP16)

    # ------------------------------------------------------------------
    # 7️⃣ Final gating + linear projection back to C
    # ------------------------------------------------------------------
    gated = out_norm * out_gate                     # element‑wise, FP16

    # Linear projection hidden_dim -> dim
    to_out_w_fp16 = weights["to_out.weight"].half()
    gated_flat = gated.view(total_rows_h, hidden_dim)    # [total_rows_h, H]
    final_fp16 = F.linear(gated_flat, to_out_w_fp16)    # [total_rows_h, C]

    # ------------------------------------------------------------------
    # 8️⃣ Cast back to FP32 and return
    # ------------------------------------------------------------------
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
