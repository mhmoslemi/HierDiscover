"""
Highly‑optimized “outgoing” TriMul (AlphaFold‑3) forward pass.

Algorithm
---------
1️⃣ First LayerNorm – use the highly‑tuned torch.nn.functional.layer_norm (FP32).
2️⃣ Fuse the five linear layers (left_proj, left_gate, right_proj,
   right_gate, out_gate) into a single matmul:
        Y = X_norm @ W_fused.T   (FP16)
   Split Y into the five parts and apply the optional mask.
3️⃣ Apply the sigmoid gates (still FP16) and obtain
        left  = proj_left  * sigmoid(gate_left)
        right = proj_right * sigmoid(gate_right)
        out_gate = sigmoid(gate_out)
4️⃣ Triangular multiplicative update:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched GEMM (torch.bmm) over the
   `(batch * hidden_dim)` dimension.
5️⃣ Fuse the second LayerNorm and the output‑gate into one Triton kernel
   (layernorm_gate_kernel).
6️⃣ Final FP16 linear projection back to `dim` and cast to FP32.

Only the two LayerNorm‑style kernels are written in Triton; all dense
linear algebra uses cuBLAS/Tensor‑Core kernels which are already optimal
on an H100.  The implementation works for all configurations described
in the problem statement.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fused LayerNorm + element‑wise gate (used for the second norm)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_gate_kernel(
    x_ptr,          # const half*   [total_rows, dim]   (input)
    gate_ptr,       # const half*   [total_rows, dim]   (sigmoid‑gate)
    y_ptr,          #       half*   [total_rows, dim]   (output)
    w_ptr,          # const float*  [dim]               (γ)
    b_ptr,          # const float*  [dim]               (β)
    total_rows,     # int
    dim: tl.constexpr,          # static (hidden_dim)
    eps: tl.constexpr,          # static
    BLOCK_M: tl.constexpr,      # rows per program
    BLOCK_D: tl.constexpr,      # cols per inner loop
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # Compute mean & variance (float32) for each row
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------------------------------------------------------
    # Normalize, apply affine transform, multiply gate and store (FP16)
    # ------------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        # load x (FP16 → FP32)
        x_ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        x_vals = tl.load(x_ptrs,
                         mask=mask_m[:, None] & mask_d[None, :],
                         other=0.0).to(tl.float32)

        # load already‑sigmoid‑ed gate (FP16 → FP32)
        g_ptrs = gate_ptr + offs_m[:, None] * dim + offs_d[None, :]
        g_vals = tl.load(g_ptrs,
                         mask=mask_m[:, None] & mask_d[None, :],
                         other=0.0).to(tl.float32)

        norm = (x_vals - mean[:, None]) * rstd[:, None]

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)

        out = norm * w[None, :] + b[None, :]
        out = out * g_vals                       # apply the output‑gate

        y_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(y_ptrs, out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        torch.Tensor of shape [bs, seq_len, seq_len, dim] (float32, on CUDA)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]                 # input / output channel dimension
    hidden_dim = config["hidden_dim"]   # hidden dimension inside TriMul
    eps = 1e-5
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape   # input: [B, N, N, dim]
    total_rows = B * N * N

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm – use torch's native implementation (FP32)
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )   # [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------
    # 2️⃣ Fuse all five linear layers (projections + gates) into one matmul
    # ------------------------------------------------------------------
    # flatten and cast to FP16 for the GEMM
    x_norm_flat = x_norm.view(-1, dim).to(torch.float16)          # [R, dim]

    # concatenate the five weight matrices (out_features, in_features)
    weight_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                          # [(5*H), dim]

    # single fused projection (no bias)
    proj = F.linear(x_norm_flat, weight_concat)                  # [R, 5*H]  FP16

    # split into logical branches
    left_proj_raw, left_gate_raw, right_proj_raw, right_gate_raw, out_gate_raw = torch.split(
        proj, hidden_dim, dim=1
    )

    # ------------------------------------------------------------------
    # 3️⃣ Mask (if present) – only on the *projections*
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape: [B, N, N] -> [R, 1]
        mask_h = mask_tensor.view(-1, 1).to(torch.float16)
        left_proj_raw = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    # ------------------------------------------------------------------
    # 4️⃣ Apply sigmoid‑gates (still FP16)
    # ------------------------------------------------------------------
    left = left_proj_raw * torch.sigmoid(left_gate_raw)   # [R, H]
    right = right_proj_raw * torch.sigmoid(right_gate_raw)  # [R, H]
    out_gate = torch.sigmoid(out_gate_raw)               # [R, H]

    # ------------------------------------------------------------------
    # 5️⃣ Triangular multiplicative update (batched GEMM)
    #     out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    # reshape to [B, N, N, H]
    left = left.view(B, N, N, hidden_dim)
    right = right.view(B, N, N, hidden_dim)

    # bring hidden dimension to batch axis and transpose right's (i,k) → (k,i)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 2, 1).contiguous() # [B, H, N, N] (j,k → k,j)

    BH = B * hidden_dim
    left_mat = left_perm.view(BH, N, N)    # [BH, N, N]
    right_mat = right_perm.view(BH, N, N)  # [BH, N, N]

    # batched matrix multiplication (FP16, Tensor‑Core on H100)
    out_mat = torch.bmm(left_mat, right_mat)               # [BH, N, N]

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H] FP16

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm + output‑gate (fused Triton kernel)
    # ------------------------------------------------------------------
    out_flat = out.view(-1, hidden_dim)                    # [R, H]   FP16
    out_gate_flat = out_gate.view(-1, hidden_dim)          # [R, H]   FP16
    out_norm_gated = torch.empty_like(out_flat)            # allocate FP16 output

    # Triton launch parameters – chosen to give good occupancy on H100
    BLOCK_M = 128
    BLOCK_D = 32 if hidden_dim <= 128 else 64
    grid = (triton.cdiv(out_flat.shape[0], BLOCK_M),)

    layernorm_gate_kernel[grid](
        out_flat,
        out_gate_flat,
        out_norm_gated,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        out_flat.shape[0],
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()

    # ------------------------------------------------------------------
    # 7️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    final_fp16 = F.linear(
        out_norm_gated,
        weights["to_out.weight"].to(torch.float16),  # shape [dim, hidden]
    )  # [R, dim]  FP16

    final = final_fp16.view(B, N, N, dim).to(torch.float32)  # cast back to FP32
    return final
