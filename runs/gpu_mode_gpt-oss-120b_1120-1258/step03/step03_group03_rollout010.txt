Define a kernel `bmm_trimul_kernel`:

```
@triton.jit
def bmm_trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_k, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    # Compute batch and hidden index
    bh = pid
    b = bh // H
    h = bh % H

    # Compute offsets for row (i) and column (j)
    offs_i = tl.arange(0, BLOCK_M) + tl.program_id(1) * BLOCK_M
    offs_j = tl.arange(0, BLOCK_N) + tl.program_id(2) * BLOCK_N

    # Mask for boundaries
    i_valid = offs_i < N
    j_valid = offs_j < N

    # Compute base pointers for this batch/hidden
    left_base = left_ptr + b * stride_left_b + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base = out_ptr + b * stride_out_b + h * stride_out_h

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = tl.arange(0, BLOCK_K) + k * BLOCK_K
        k_valid = offs_k < N

        # Load left tile: shape [BLOCK_M, BLOCK_K]
        left_ptrs = left_base + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        left_tile = tl.load(left_ptrs, mask=i_valid[:, None] & k_valid[None, :], other=0.0).to(tl.float32)

        # Load right tile transposed: shape [BLOCK_K, BLOCK_N]
        right_ptrs = right_base + offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_i  # note stride_right_i for col j
        right_tile = tl.load(right_ptrs, mask=k_valid[:, None] & j_valid[None, :], other=0.0).to(tl.float32)

        # dot
        acc += tl.dot(left_tile, right_tile)

        # no pointer increment needed; offsets computed each iteration

    # Write out
    out_ptrs = out_base + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_k
    tl.store(out_ptrs, acc.to(tl.float16), mask=i_valid[:, None] & j_valid[None, :])
```

"""
TriMul (outgoing) – fully‑fused forward pass
------------------------------------------------
* Row‑wise LayerNorm on the input (float32) – Triton kernel.
* One fused Linear computes the three projections and three gates at once.
* Optional pairwise mask is applied before the two value projections.
* Gates are sigmoid‑ed and combined with the projected values.
* **Core contraction** (einsum ‘… i k d, … j k d → … i j d’) is implemented as a
  custom Triton batched GEMM that works directly on the original
  `[B, N, N, H]` layout – no `permute(...).contiguous()` copies are required.
  For each hidden slice `h` we compute `out[b,i,j,h] = Σ_k left[b,i,k,h] *
  right[b,j,k,h]` by a tiled `tl.dot` over `k`.  This eliminates two
  ~64 MiB data shuffles and reduces kernel‑launch overhead.
* Second LayerNorm + output‑gate + final Linear are fused in a second Triton
  kernel (`layernorm_gate_kernel`), then a cheap half‑precision Linear
  produces the final FP32 tensor.

The implementation uses mixed‑precision (FP16) for the heavy tensor work,
float32 for the statistics in LayerNorm, and runs efficiently on H100
(Tensor‑Core friendly tile sizes).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (float32) – used on the input tensor
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const float*   [total_rows, dim]
    y_ptr,          #       float*   [total_rows, dim]
    weight_ptr,     # const float*   [dim]   (γ)
    bias_ptr,       # const float*   [dim]   (β)
    total_rows,     # i32
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ----- pass 1: mean / variance (float32 accumulator) -----
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ----- pass 2: normalize, apply γ/β and write back -----
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)

        out = norm * w[None, :] + b[None, :]
        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out, mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2️⃣ LayerNorm + output‑gate + final Linear (half‑precision) – fused
# ----------------------------------------------------------------------
@triton.jit
def layernorm_gate_kernel(
    x_ptr,          # const half*    [total_rows, dim]  (input)
    y_ptr,          #       half*    [total_rows, dim]  (LN output * gate)
    gate_ptr,       # const half*    [total_rows, dim]  (out_gate)
    weight_ptr,     # const half*    [dim]              (γ)
    bias_ptr,       # const half*    [dim]              (β)
    total_rows,     # i32
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # ----- pass 1: mean / variance (float32) -----
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ----- pass 2: normalize, γ/β, multiply by gate, store -----
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        x_ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        x = tl.load(
            x_ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (x - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0).to(tl.float32)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0).to(tl.float32)
        out = norm * w[None, :] + b[None, :]

        g_ptrs = gate_ptr + offs_m[:, None] * dim + offs_d[None, :]
        gate = tl.load(
            g_ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        out = out * gate
        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out.to(tl.float16), mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 3️⃣ Core triangle‑multiplicative update – Triton batched GEMM
# ----------------------------------------------------------------------
@triton.jit
def tril_mul_bmm_kernel(
    left_ptr,       # const half*   [B, N, N, H]
    right_ptr,      # const half*   [B, N, N, H]
    out_ptr,        #       half*   [B, N, N, H] (already allocated)
    B: tl.constexpr,
    N: tl.constexpr,
    H: tl.constexpr,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,   # int64 strides for left
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,   # int64 strides for right
    stride_o_b, stride_o_i, stride_o_k, stride_o_h,   # int64 strides for out
    BLOCK_M: tl.constexpr,   # tile rows (i)
    BLOCK_N: tl.constexpr,   # tile cols (j)
    BLOCK_K: tl.constexpr,   # reduction tile (k)
):
    """
    batched GEMM for TriMul:
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    The kernel iterates over the hidden dimension as part of the batch
    (BH = B*H program IDs).  Strides make the transposition of right
    (k ↔ j) implicit – no data copy is performed.
    """
    pid = tl.program_id(0)          # combined batch‑hidden index
    pid_i = tl.program_id(1)        # tile over i
    pid_j = tl.program_id(2)        # tile over j

    bh = pid
    b = bh // H
    h = bh % H

    # start offsets for this tile
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    i_valid = offs_i < N
    j_valid = offs_j < N

    # base pointers that already encode batch & hidden offsets
    left_base  = left_ptr  + b * stride_l_b  + h * stride_l_h
    right_base = right_ptr + b * stride_r_b  + h * stride_r_h
    out_base   = out_ptr   + b * stride_o_b  + h * stride_o_h

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension (the shared “k” axis)
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        k_valid = offs_k < N

        # ---- left tile : (i,k) ----
        left_ptrs = left_base \
            + offs_i[:, None] * stride_l_i \
            + offs_k[None, :] * stride_l_k
        left_tile = tl.load(
            left_ptrs,
            mask=i_valid[:, None] & k_valid[None, :],
            other=0.0,
        ).to(tl.float32)

        # ---- right tile (transposed) : (k,j) ----
        # note: stride_r_k is the stride for moving along k (row)
        #       stride_r_i is the stride for moving along j (col)
        right_ptrs = right_base \
            + offs_k[:, None] * stride_r_k \
            + offs_j[None, :] * stride_r_i
        right_tile = tl.load(
            right_ptrs,
            mask=k_valid[:, None] & j_valid[None, :],
            other=0.0,
        ).to(tl.float32)

        acc += tl.dot(left_tile, right_tile)   # BLOCK_M × BLOCK_N accumulation

    # write back
    out_ptrs = out_base \
        + offs_i[:, None] * stride_o_i \
        + offs_j[None, :] * stride_o_k
    tl.store(out_ptrs, acc.to(tl.float16), mask=i_valid[:, None] & j_valid[None, :])

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (AlphaFold‑3).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, D], float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/int) or None
        - weights_dict : dict of torch.Tensors (model parameters)
        - config_dict  : { "dim": D, "hidden_dim": H, "nomask": bool }

    Returns
    -------
    torch.Tensor of shape [B, N, N, D], float32
    """
    # -------------------------------------------------
    # 0️⃣ unpack & constants
    # -------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                 # D
    hidden_dim = config["hidden_dim"]   # H
    eps = 1e-5
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N                # R = B·N·N

    # -------------------------------------------------
    # 1️⃣ Input LayerNorm (float32)
    # -------------------------------------------------
    x_flat = input_tensor.contiguous().view(total_rows, dim)          # [R, D]
    x_norm_flat = torch.empty_like(x_flat)                            # [R, D]

    BLOCK_M = 256
    BLOCK_D = 32
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm_flat,
        weights["norm.weight"],      # γ   (float32, [D])
        weights["norm.bias"],        # β   (float32, [D])
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # -------------------------------------------------
    # 2️⃣ Project + gates (single fused linear)
    # -------------------------------------------------
    x_norm_h = x_norm_flat.to(torch.float16)               # [R, D] FP16

    # concatenate the five weight matrices (shape (5*H, D))
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)   # (5*H, D)

    proj_all = F.linear(x_norm_h, w_concat)                 # [R, 5*H] (FP16)

    H = hidden_dim
    left_proj   = proj_all[:, 0:H]
    left_gate   = proj_all[:, H:2*H]
    right_proj  = proj_all[:, 2*H:3*H]
    right_gate  = proj_all[:, 3*H:4*H]
    out_gate    = proj_all[:, 4*H:5*H]

    if not nomask:
        # mask expanded to (R,1) for broadcasting
        mask_h = mask_tensor.to(torch.float16).view(total_rows, 1)
        left_proj  = left_proj * mask_h
        right_proj = right_proj * mask_h

    # sigmoid gates
    left_gate  = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate   = torch.sigmoid(out_gate)

    # apply value‑gates
    left  = left_proj * left_gate          # [R, H] FP16
    right = right_proj * right_gate        # [R, H] FP16

    # -------------------------------------------------
    # 3️⃣ Reshape to 4‑D tensors (no copy – view only)
    # -------------------------------------------------
    left  = left.view(B, N, N, H)   # [B, N, N, H]
    right = right.view(B, N, N, H)  # [B, N, N, H]

    # -------------------------------------------------
    # 4️⃣ Tri‑multiplicative update – custom batched GEMM
    # -------------------------------------------------
    out = torch.empty_like(left)     # allocate output tensor [B,N,N,H] (FP16)

    # Strides (in elements) for the four‑dim tensors
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_k, stride_o_h = out.stride()

    # Kernel launch configuration
    BLOCK_M = 128   # rows of output tile (i)
    BLOCK_N = 128   # cols of output tile (j)
    BLOCK_K = 64    # reduction tile (k)

    grid = (
        B * H,                                   # batch‑hidden dimension
        triton.cdiv(N, BLOCK_M),                 # i‑tiles
        triton.cdiv(N, BLOCK_N),                 # j‑tiles
    )
    tril_mul_bmm_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_i, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_k, stride_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -------------------------------------------------
    # 5️⃣ Second LayerNorm + output‑gate fused, then final linear
    # -------------------------------------------------
    out_flat = out.view(total_rows, H)                     # [R, H] FP16
    out_gate_flat = out_gate.view(total_rows, H)           # [R, H] FP16

    out_norm_gate = torch.empty_like(out_flat)             # FP16 buffer

    BLOCK_M2 = 256
    BLOCK_D2 = 32
    grid2 = (triton.cdiv(total_rows, BLOCK_M2),)
    layernorm_gate_kernel[grid2](
        out_flat,
        out_norm_gate,
        out_gate_flat,
        weights["to_out_norm.weight"].to(torch.float16),   # γ   (FP16, [H])
        weights["to_out_norm.bias"].to(torch.float16),     # β   (FP16, [H])
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M2,
        BLOCK_D=BLOCK_D2,
    )

    # final projection back to original dim
    w_to_out = weights["to_out.weight"].to(torch.float16)   # [D, H]
    final_half = F.linear(out_norm_gate, w_to_out)          # [R, D] FP16
    final = final_half.view(B, N, N, dim).to(torch.float32) # cast to FP32

    return final
