@triton.jit
def fused_ln_linear_kernel(
    x_ptr, y_ptr, weight_ptr, bias_ptr,
    rows, dim, out_dim,
    eps,
    stride_xm, stride_xd,
    stride_wm, stride_wd,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
    BLOCK_O: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # Compute mean/var
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    # Iterating over dim blocks
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        x_ptrs = x_ptr + offs_m[:, None] * stride_xm + offs_d[None, :] * stride_xd
        x = tl.load(x_ptrs, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)
        sum_val += tl.sum(x, axis=1)
        sum_sq += tl.sum(x * x, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # Accumulator for output
    acc = tl.zeros([BLOCK_M, BLOCK_O], dtype=tl.float32)

    # second pass: compute dot product with weight_scaled
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        # load x block
        x_ptrs = x_ptr + offs_m[:, None] * stride_xm + offs_d[None, :] * stride_xd
        x_block_f32 = tl.load(x_ptrs, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)
        # normalize
        x_norm_f32 = (x_block_f32 - mean[:, None]) * rstd[:, None]
        # cast to half for dot
        x_norm = x_norm_f32.to(tl.float16)

        # load weight block
        offs_o = tl.arange(0, BLOCK_O)
        mask_o = offs_o < out_dim
        w_ptrs = weight_ptr + offs_o[:, None] * stride_wm + offs_d[None, :] * stride_wd
        w = tl.load(w_ptrs, mask=mask_o[:, None] & mask_d[None, :], other=0.0).to(tl.float16)

        # dot
        # tl.dot supports half arguments, accum dtype defined by out_dtype
        acc += tl.dot(x_norm, w, out_dtype=tl.float32)

    # add bias
    offs_o = tl.arange(0, BLOCK_O)
    mask_o = offs_o < out_dim
    b = tl.load(bias_ptr + offs_o, mask=mask_o, other=0.0).to(tl.float32)
    acc = acc + b[None, :]

    # store result as half
    y_ptrs = y_ptr + offs_m[:, None] * out_dim + offs_o[None, :]
    tl.store(y_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_o[None, :])

                           mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)

        sum_sq += tl.sum(out_vals * out_vals, axis=1)

   var = sum_sq / hidden_dim - mean * mean
   rstd = tl.rsqrt(var + eps)

   # allocate accumulator for final output
   acc = tl.zeros([BLOCK_M, BLOCK_O], dtype=tl.float32)

   # second pass over hidden dim
   for d in range(0, tl.cdiv(hidden_dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < hidden_dim

        # load out
        out_vals_f32 = tl.load(out_ptr + offs_m[:, None] * stride_out_m + offs_d[None, :] * stride_out_h,
                               mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)
        # load gate values
        gate_vals_f32 = tl.load(gate_ptr + offs_m[:, None] * stride_gate_m + offs_d[None, :] * stride_gate_h,
                                mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)

        # normalize
        norm_vals = (out_vals_f32 - mean[:, None]) * rstd[:, None]   # fp32

        # load LN weight and bias
        gamma = tl.load(norm_weight_ptr + offs_d, mask=mask_d, other=0.0).to(tl.float32)
        beta = tl.load(norm_bias_ptr + offs_d, mask=mask_d, other=0.0).to(tl.float32)

        # apply LN weight & bias
        norm_vals = norm_vals * gamma[None, :] + beta[None, :]

        # apply gate
        gated = norm_vals * gate_vals_f32

        # cast to half for dot
        gated_f16 = gated.to(tl.float16)

        # load weight slice from w_out (shape [out_dim, hidden_dim])
        offs_o = tl.arange(0, BLOCK_O)
        mask_o = offs_o < out_dim
        w_ptrs = w_out_ptr + offs_o[:, None] * stride_wout_m + offs_d[None, :] * stride_wout_h
        w = tl.load(w_ptrs, mask=mask_o[:, None] & mask_d[None, :], other=0.0).to(tl.float16)

        acc += tl.dot(gated_f16, w, out_dtype=tl.float32)

   # store final output
   # y_ptr = out_ptr_out (i.e., output buffer)
   offs_o = tl.arange(0, BLOCK_O)
   mask_o = offs_o < out_dim
   y_ptrs = out_ptr_out + offs_m[:, None] * out_dim + offs_o[None, :]
   tl.store(y_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_o[None, :])

    weight_scaled = (weight_concat * norm_weight.unsqueeze(0)).to(torch.float16)   # broadcasting (5*H, dim)
    bias_concat = torch.matmul(norm_bias.unsqueeze(0), weight_concat).squeeze(0).to(torch.float16)  # shape (5*H)

    # Output buffer for projection
    O = 5 * hidden_dim
    proj_all = torch.empty((R, O), dtype=torch.float16, device=device)

    # kernel launch
    BLOCK_M = 64    # Try smaller to reduce register pressure
    BLOCK_D = 32
    BLOCK_O = 64

    grid = (triton.cdiv(R, BLOCK_M), triton.cdiv(O, BLOCK_O))

    fused_ln_linear_kernel[grid](
        x_ptr = input_tensor,
        y_ptr = proj_all,
        weight_ptr = weight_scaled,
        bias_ptr = bias_concat,
        rows = R,
        dim = dim,
        out_dim = O,
        eps = eps,
        stride_xm = dim,
        stride_xd = 1,
        stride_wm = dim,
        stride_wd = 1,
        BLOCK_M = BLOCK_M,
        BLOCK_D = BLOCK_D,
        BLOCK_O = BLOCK_O,
    )
    # now use proj_all
    left_proj_raw = proj_all[:, 0:hidden_dim]
    left_gate_raw = proj_all[:, hidden_dim:2*hidden_dim]
    right_proj_raw = proj_all[:, 2*hidden_dim:3*hidden_dim]
    right_gate_raw = proj_all[:, 3*hidden_dim:4*hidden_dim]
    out_gate_raw = proj_all[:, 4*hidden_dim:5*hidden_dim]

    # Apply mask
    if not nomask:
        mask_flat = mask_tensor.to(input_tensor.dtype).view(R, 1)  # float32 mask
        left_proj_raw = left_proj_raw * mask_flat
        right_proj_raw = right_proj_raw * mask_flat

    # Sigmoid gates
    left_gate = torch.sigmoid(left_gate_raw).to(torch.float16)
    right_gate = torch.sigmoid(right_gate_raw).to(torch.float16)
    out_gate = torch.sigmoid(out_gate_raw).to(torch.float16)

    left = left_proj_raw * left_gate
    right = right_proj_raw * right_gate

    # Tri mul update
    left = left.view(B, N, N, hidden_dim)
    right = right.view(B, N, N, hidden_dim)

    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 2, 1).contiguous()

    left_mat = left_perm.view(B*hidden_dim, N, N)
    right_mat = right_perm.view(B*hidden_dim, N, N)

    out_mat = torch.bmm(left_mat, right_mat)  # half
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H) half

    # Output stage fused kernel
    # out shape (R, hidden_dim) flatten
    out_flat = out.view(R, hidden_dim)

    # Prepare weight for final projection and norm params
    to_out_norm_weight = weights["to_out_norm.weight"].to(torch.float16)   # shape [hidden_dim]
    to_out_norm_bias = weights["to_out_norm.bias"].to(torch.float16)     # shape [hidden_dim]
    to_out_weight = weights["to_out.weight"].to(torch.float16)   # shape [dim, hidden_dim]

    # Output buffer
    final_half = torch.empty((R, dim), dtype=torch.float16, device=device)

    # Grid for output kernel
    BLOCK_O2 = 64  # maybe same as BLOCK_O but now output dim's columns
    grid2 = (triton.cdiv(R, BLOCK_M), triton.cdiv(dim, BLOCK_O2))

    out_ln_gate_linear_kernel[grid2](
        out_ptr = out_flat,
        gate_ptr = out_gate,
        norm_weight_ptr = to_out_norm_weight,
        norm_bias_ptr = to_out_norm_bias,
        w_out_ptr = to_out_weight,
        out_ptr_out = final_half,
        rows = R,
        hidden_dim = hidden_dim,
        out_dim = dim,
        eps = eps,
        stride_out_m = hidden_dim,
        stride_out_h = 1,
        stride_gate_m = hidden_dim,
        stride_gate_h = 1,
        stride_norm = 1,
        stride_wout_m = hidden_dim,
        stride_wout_h = 1,
        BLOCK_M = BLOCK_M,
        BLOCK_D = 32,
        BLOCK_O = BLOCK_O2,
    )

    # Reshape final output
    final = final_half.view(B, N, N, dim).to(torch.float32)

    return final

@triton.jit
def fused_ln_linear_kernel(
    x_ptr, y_ptr, weight_ptr, bias_ptr,
    rows, dim, out_dim,
    eps,
    stride_xm, stride_xd,
    stride_wm, stride_wd,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
    BLOCK_O: tl.constexpr,
):
    pid0 = tl.program_id(0) # row block
    pid1 = tl.program_id(1) # output column block
    row_start = pid0 * BLOCK_M
    col_start = pid1 * BLOCK_O

    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    offs_o = col_start + tl.arange(0, BLOCK_O)
    mask_o = offs_o < out_dim

    # Compute mean & var across dim
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        x_ptrs = x_ptr + offs_m[:, None] * stride_xm + offs_d[None, :] * stride_xd
        x = tl.load(x_ptrs, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)
        sum_val += tl.sum(x, axis=1)
        sum_sq += tl.sum(x * x, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # accumulator
    acc = tl.zeros([BLOCK_M, BLOCK_O], dtype=tl.float32)

    # second pass: compute normalized and dot with weight
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        # load input block
        x_ptrs = x_ptr + offs_m[:, None] * stride_xm + offs_d[None, :] * stride_xd
        x_block_f32 = tl.load(x_ptrs, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)

        # normalize: (x - mean) * rstd
        x_norm_f32 = (x_block_f32 - mean[:, None]) * rstd[:, None]
        # cast to half for dot with weight (which is already scaled by gamma)
        x_norm = x_norm_f32.to(tl.float16)

        # load weight block for current output columns
        w_ptrs = weight_ptr + offs_o[:, None] * stride_wm + offs_d[None, :] * stride_wd
        w = tl.load(w_ptrs, mask=mask_o[:, None] & mask_d[None, :], other=0.0).to(tl.float16)

        # dot product: sum over D block
        acc += tl.dot(x_norm, w, out_dtype=tl.float32)
    # add bias
    b = tl.load(bias_ptr + offs_o, mask=mask_o, other=0.0).to(tl.float32)
    acc = acc + b[None, :]

    # store result
    y_ptrs = y_ptr + offs_m[:, None] * out_dim + offs_o[None, :]
    tl.store(y_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_o[None, :])

@triton.jit
def out_ln_gate_linear_kernel(
    out_ptr, gate_ptr,
    norm_weight_ptr, norm_bias_ptr,
    w_out_ptr, out_ptr_out,
    rows, hidden_dim, out_dim,
    eps,
    stride_out_m, stride_out_h,
    stride_gate_m, stride_gate_h,

    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
    BLOCK_O: tl.constexpr,
):
    pid0 = tl.program_id(0)
    pid1 = tl.program_id(1)
    row_start = pid0 * BLOCK_M
    col_start = pid1 * BLOCK_O

    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    offs_o = col_start + tl.arange(0, BLOCK_O)
    mask_o = offs_o < out_dim

    # compute mean/var for each row (over hidden_dim)
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(hidden_dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < hidden_dim
        out_vals = tl.load(out_ptr + offs_m[:, None] * stride_out_m + offs_d[None, :] * stride_out_h,
                           mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)
        sum_val += tl.sum(out_vals, axis=1)
        sum_sq += tl.sum(out_vals * out_vals, axis=1)

    mean = sum_val / hidden_dim
    var = sum_sq / hidden_dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # accumulator for final output columns
    acc = tl.zeros([BLOCK_M, BLOCK_O], dtype=tl.float32)

    # second pass
    for d in range(0, tl.cdiv(hidden_dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < hidden_dim

        # load out values and gate
        out_vals = tl.load(out_ptr + offs_m[:, None] * stride_out_m + offs_d[None, :] * stride_out_h,
                           mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)
        gate_vals = tl.load(gate_ptr + offs_m[:, None] * stride_gate_m + offs_d[None, :] * stride_gate_h,
                            mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)

        # normalize
        norm_vals = (out_vals - mean[:, None]) * rstd[:, None]   # fp32

        # apply LN gamma and beta
        gamma = tl.load(norm_weight_ptr + offs_d, mask=mask_d, other=0.0).to(tl.float32)
        beta = tl.load(norm_bias_ptr + offs_d, mask=mask_d, other=0.0).to(tl.float32)
        norm_vals = norm_vals * gamma[None, :] + beta[None, :]

        # apply gate
        gated = norm_vals * gate_vals

        # cast to half
        gated_f16 = gated.to(tl.float16)

        # load weight slice for output columns
        w_ptrs = w_out_ptr + offs_o[:, None] * stride_wout_m + offs_d[None, :] * stride_wout_h
        w = tl.load(w_ptrs, mask=mask_o[:, None] & mask_d[None, :], other=0.0).to(tl.float16)

        acc += tl.dot(gated_f16, w, out_dtype=tl.float32)

    # store final output
    out_ptrs = out_ptr_out + offs_m[:, None] * out_dim + offs_o[None, :]
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_o[None, :])

"""TriMul (outgoing) – fused LayerNorm + Linear kernels (Trident)

The forward consists of two major stages:

1️⃣  Row‑wise LayerNorm (γ,β) followed by a single fused Linear that
    produces the five projected tensors  
    `[left_proj, left_gate, right_proj, right_gate, out_gate]`.  
    By scaling the weight matrix with the LayerNorm scale (γ) and
    pre‑computing the bias `β @ W` we can fuse reduction of the mean/
    variance, the normalization, and the matrix‑multiply into one
    Triton kernel.  This saves one full read of the massive
    `[B·N·N, dim]` tensor.

2️⃣  After the multiplicative update (batched GEMM) we again apply a
    LayerNorm (γ_out,β_out) to the hidden tensor, multiply element‑wise
    by the `out_gate` (sigmoid) and finally project back to the original
    channel dimension `dim`.  The three operations are fused into a
    second Triton kernel that computes the per‑row statistics,
    normalizes, applies the gate, and performs the final matrix‑multiply
    with `to_out.weight`.

Both kernels use FP16 for the heavy arithmetic, while the reduction
( mean / variance ) is accumulated in FP32 for numerical stability.
The bmm that implements the “tri‑multiplicative” update remains a
highly‑optimised cuBLAS call (Tensor‑Cores).  The implementation works
for all shapes required by the test suite (dim ∈ {128,384},
hidden_dim = 128, B ∈ {1,2}, N ∈ {128,256,512,1024}) and respects the
optional mask."""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Kernel 1 – fused LayerNorm + Linear for the five projections
# ----------------------------------------------------------------------
@triton.jit
def fused_ln_linear_kernel(
    x_ptr,                # const float*   [total_rows, dim]
    y_ptr,                #       half*   [total_rows, out_dim]
    w_ptr,                # const half*   [out_dim, dim]   (already scaled by γ)
    b_ptr,                # const half*   [out_dim]        (β @ W)
    total_rows, dim, out_dim,
    eps,
    stride_xm, stride_xd,   # strides for x (row, col)
    stride_wm, stride_wd,   # strides for weight (row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
    BLOCK_O: tl.constexpr,
):
    # 2‑D grid: rows × output‑column blocks
    pid_row = tl.program_id(0)
    pid_col = tl.program_id(1)

    row_start = pid_row * BLOCK_M
    col_start = pid_col * BLOCK_O

    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    offs_o = col_start + tl.arange(0, BLOCK_O)
    mask_o = offs_o < out_dim

    # -----------------------------------------------------------------
    # 1️⃣  Compute per‑row mean / variance (float32 accumulator)
    # -----------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_d[None, :] * stride_xd,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                         # [BLOCK_M, BLOCK_D]
        sum_val += tl.sum(x, axis=1)
        sum_sq  += tl.sum(x * x, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                     # 1/sqrt(var+eps)

    # -----------------------------------------------------------------
    # 2️⃣  Second pass – normalized row × weight → output block
    # -----------------------------------------------------------------
    acc = tl.zeros([BLOCK_M, BLOCK_O], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        # load a tile of the input
        x = tl.load(
            x_ptr + offs_m[:, None] * stride_xm + offs_d[None, :] * stride_xd,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        # (x - mean) / sqrt(var+eps)
        x_norm = ((x - mean[:, None]) * rstd[:, None]).to(tl.float16)

        # load the corresponding weight slice
        w = tl.load(
            w_ptr + offs_o[:, None] * stride_wm + offs_d[None, :] * stride_wd,
            mask=mask_o[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float16)

        # accumulate dot product (FP32 accumulator)
        acc += tl.dot(x_norm, w, out_dtype=tl.float32)

    # add the pre‑computed bias
    b = tl.load(b_ptr + offs_o, mask=mask_o, other=0.0).to(tl.float32)
    acc = acc + b[None, :]

    # write result as half
    y_ptrs = y_ptr + offs_m[:, None] * out_dim + offs_o[None, :]
    tl.store(y_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_o[None, :])


# ----------------------------------------------------------------------
# Kernel 2 – fused LayerNorm (γ_out,β_out) + gate + final Linear
# ----------------------------------------------------------------------
@triton.jit
def out_ln_gate_linear_kernel(
    inp_ptr,               # const half*   [total_rows, hidden_dim]
    gate_ptr,              # const half*   [total_rows, hidden_dim]
    ln_w_ptr,              # const half*   [hidden_dim]   γ_out
    ln_b_ptr,              # const half*   [hidden_dim]   β_out
    w_out_ptr,             # const half*   [out_dim, hidden_dim]   (to_out.weight)
    out_ptr,               #       half*   [total_rows, out_dim]   final result
    total_rows, hidden_dim, out_dim,
    eps,
    stride_in_m, stride_in_h,
    stride_gate_m, stride_gate_h,
    stride_ln,             # =1 (1‑D tensors)
    stride_wout_m, stride_wout_h,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
    BLOCK_O: tl.constexpr,
):
    pid_row = tl.program_id(0)
    pid_col = tl.program_id(1)

    row_start = pid_row * BLOCK_M
    col_start = pid_col * BLOCK_O

    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    offs_o = col_start + tl.arange(0, BLOCK_O)
    mask_o = offs_o < out_dim

    # -----------------------------------------------------------------
    # 1️⃣  Row‑wise mean / variance over the hidden dimension
    # -----------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(hidden_dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < hidden_dim
        x = tl.load(
            inp_ptr + offs_m[:, None] * stride_in_m + offs_d[None, :] * stride_in_h,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)
        sum_val += tl.sum(x, axis=1)
        sum_sq  += tl.sum(x * x, axis=1)

    mean = sum_val / hidden_dim
    var  = sum_sq / hidden_dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # -----------------------------------------------------------------
    # 2️⃣  Normalise → LN (γ_out,β_out) → gate → final Linear
    # -----------------------------------------------------------------
    acc = tl.zeros([BLOCK_M, BLOCK_O], dtype=tl.float32)

    for d in range(0, tl.cdiv(hidden_dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < hidden_dim

        # load hidden values and corresponding gate
        x = tl.load(
            inp_ptr + offs_m[:, None] * stride_in_m + offs_d[None, :] * stride_in_h,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)
        g = tl.load(
            gate_ptr + offs_m[:, None] * stride_gate_m + offs_d[None, :] * stride_gate_h,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        # (x - μ) / √σ²+ε
        norm = (x - mean[:, None]) * rstd[:, None]           # FP32

        # LayerNorm γ_out, β_out
        gamma = tl.load(ln_w_ptr + offs_d, mask=mask_d, other=0.0).to(tl.float32)
        beta  = tl.load(ln_b_ptr + offs_d, mask=mask_d, other=0.0).to(tl.float32)
        norm = norm * gamma[None, :] + beta[None, :]

        # apply the output gate
        gated = norm * g

        # cast to half for the matrix‑multiply
        gated_f16 = gated.to(tl.float16)

        # load weight block for this output‑column tile
        w = tl.load(
            w_out_ptr + offs_o[:, None] * stride_wout_m + offs_d[None, :] * stride_wout_h,
            mask=mask_o[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float16)

        # Accumulate: gated (M×Dₖ) ⋅ w (N×Dₖ)ᵀ  → (M×N)
        acc += tl.dot(gated_f16, w, out_dtype=tl.float32)

    # write final half‑precision result
    out_p = out_ptr + offs_m[:, None] * out_dim + offs_o[None, :]
    tl.store(out_p, acc.to(tl.float16), mask=mask_m[:, None] & mask_o[None, :])


# ----------------------------------------------------------------------
# Main entry point – outgoing TriMul forward pass
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (AlphaFold‑3 style).

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor: torch.Tensor [B, N, N, D] (float32)
        - mask_tensor : torch.Tensor [B, N, N] (bool/int) – may be omitted
        - weights_dict : mapping of all linear / norm parameters (float32)
        - config_dict  : {"dim": D, "hidden_dim": H, "nomask": bool}
    Returns:
        torch.Tensor of shape [B, N, N, D] (float32)
    """
    # ------------------------------------------------------------------
    # 0️⃣ Unpack inputs and constants
    # ------------------------------------------------------------------
    inp, mask, weights, cfg = data
    device = inp.device
    dim = cfg["dim"]                # D
    hidden = cfg["hidden_dim"]      # H (always 128 in tests)
    eps = 1e-5
    nomask = cfg.get("nomask", True)

    B, N, _, _ = inp.shape
    total_rows = B * N * N          # R = B·N·N
    out_dim = dim

    # ------------------------------------------------------------------
    # 1️⃣ Fuse first LayerNorm + all five linear projections
    # ------------------------------------------------------------------
    # concatenate the five projection matrices (shape: 5H × D)
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],   # (H, D)
            weights["left_gate.weight"],   # (H, D)
            weights["right_proj.weight"],  # (H, D)
            weights["right_gate.weight"],  # (H, D)
            weights["out_gate.weight"],    # (H, D)
        ],
        dim=0,
    ).to(device)                       # (5H, D)  float32

    # scale weight by the first LayerNorm γ (broadcast over columns)
    gamma = weights["norm.weight"].to(device)               # (D,)
    beta  = weights["norm.bias"].to(device)                # (D,)

    w_scaled = (w_concat * gamma[None, :]).to(torch.float16)   # (5H, D)
    b_scaled = (beta @ w_concat).to(torch.float16)            # (5H,)

    # output buffer for the fused op
    proj_all = torch.empty((total_rows, 5 * hidden), dtype=torch.float16, device=device)

    # tuning parameters – small enough to fit registers but big enough for occupancy
    BLOCK_M = 64    # rows per program
    BLOCK_D = 32    # inner‑dim tile
    BLOCK_O = 64    # output‑column tile (5*H ≤ 640 → 10 tiles)

    grid = (triton.cdiv(total_rows, BLOCK_M), triton.cdiv(5 * hidden, BLOCK_O))
    fused_ln_linear_kernel[grid](
        x_ptr=inp,
        y_ptr=proj_all,
        w_ptr=w_scaled,
        b_ptr=b_scaled,
        total_rows=total_rows,
        dim=dim,
        out_dim=5 * hidden,
        eps=eps,
        stride_xm=dim,
        stride_xd=1,
        stride_wm=dim,
        stride_wd=1,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        BLOCK_O=BLOCK_O,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Slice, apply mask (if any) and sigmoids
    # ------------------------------------------------------------------
    left_proj   = proj_all[:, 0 * hidden:1 * hidden]
    left_gate_r = proj_all[:, 1 * hidden:2 * hidden]
    right_proj  = proj_all[:, 2 * hidden:3 * hidden]
    right_gate_r = proj_all[:, 3 * hidden:4 * hidden]
    out_gate_r  = proj_all[:, 4 * hidden:5 * hidden]

    if not nomask:
        mask_f = mask.to(torch.float16).view(total_rows, 1)   # (R,1)
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    left_gate  = torch.sigmoid(left_gate_r).to(torch.float16)
    right_gate = torch.sigmoid(right_gate_r).to(torch.float16)
    out_gate   = torch.sigmoid(out_gate_r).to(torch.float16)

    left  = left_proj * left_gate      # (R, H)   half
    right = right_proj * right_gate    # (R, H)   half

    # ------------------------------------------------------------------
    # 3️⃣ Tri‑multiplicative update (batched GEMM)
    # ------------------------------------------------------------------
    left  = left.view(B, N, N, hidden)          # (B,N,N,H)
    right = right.view(B, N, N, hidden)         # (B,N,N,H)

    # bring hidden dim to batch axis for a single bmm per head
    left_mat  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_mat = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N) with (k ↔ j)

    BH = B * hidden
    left_mat = left_mat.view(BH, N, N)          # (B·H, N, N)
    right_mat = right_mat.view(BH, N, N)        # (B·H, N, N)

    out_mat = torch.bmm(left_mat, right_mat)   # (B·H, N, N)  half
    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H) half

    # ------------------------------------------------------------------
    # 4️⃣ Fuse second LayerNorm + out‑gate + final Linear
    # ------------------------------------------------------------------
    out_flat   = out.view(total_rows, hidden)       # (R, H) half
    out_gate_f = out_gate.view(total_rows, hidden) # (R, H) half

    # to_out_norm (LayerNorm over hidden dim) + final projection
    ln_out_w = weights["to_out_norm.weight"].to(torch.float16)   # (H,)
    ln_out_b = weights["to_out_norm.bias"].to(torch.float16)    # (H,)
    w_out    = weights["to_out.weight"].to(torch.float16)       # (D, H)

    final_half = torch.empty((total_rows, out_dim), dtype=torch.float16, device=device)

    # reuse BLOCK_M and BLOCK_D, choose a modest BLOCK_O for the final dim
    BLOCK_O2 = 64
    grid2 = (triton.cdiv(total_rows, BLOCK_M), triton.cdiv(out_dim, BLOCK_O2))
    out_ln_gate_linear_kernel[grid2](
        inp_ptr=out_flat,
        gate_ptr=out_gate_f,
        ln_w_ptr=ln_out_w,
        ln_b_ptr=ln_out_b,
        w_out_ptr=w_out,
        out_ptr=final_half,
        total_rows=total_rows,
        hidden_dim=hidden,
        out_dim=out_dim,
        eps=eps,
        stride_in_m=hidden,
        stride_in_h=1,
        stride_gate_m=hidden,
        stride_gate_h=1,
        stride_ln=1,
        stride_wout_m=hidden,
        stride_wout_h=1,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        BLOCK_O=BLOCK_O2,
    )

    # reshape back to [B, N, N, D] and cast to FP32
    out_fp32 = final_half.view(B, N, N, out_dim).to(torch.float32)

    return out_fp32
