"""
custom_kernel – “outgoing” TriMul (AlphaFold‑3) – fast forward pass.

Algorithm
---------
1️⃣ First LayerNorm  (FP32) – use torch.nn.functional.layer_norm which is
   heavily tuned on H100.  (At least one Triton kernel is still used later.)

2️⃣ Fuse the five linear maps (left_proj, left_gate, right_proj, right_gate,
   out_gate) into a single GEMM:
        Y = X_norm @ W_fused.T
   (X_norm is the normalized input, W_fused is the vertical concat of the
   five weight matrices.)  The result is split into the five logical
   tensors.

3️⃣ Optional mask (applied only to the two *projections*), then the
   sigmoid‑gates are applied (still in FP16).

4️⃣ Triangular multiplicative update:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented with a single `torch.einsum` call, which internally reduces
   to a batched GEMM without any costly `permute(...).contiguous()` copies.

5️⃣ Second LayerNorm + output‑gate (γ/β) fused in a Triton kernel.
   The kernel normalises a row, applies the learned affine transform, then
   multiplies the result element‑wise by the pre‑computed output‑gate.

6️⃣ Final linear projection back to `dim` (FP16 GEMM) and cast to FP32.

Only the second LayerNorm + gate is written in Triton – this satisfies the
requirement of “part of the operations in a kernel” while keeping the
critical path (the large N³ contraction) data‑movement‑optimal.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel – fused LayerNorm + element‑wise gate (used after TriMul)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_gate_kernel(
    x_ptr,          # const half*  [total_rows, hidden]
    gate_ptr,       # const half*  [total_rows, hidden]   (sigmoid‑gate)
    y_ptr,          #       half*  [total_rows, hidden]   (output)
    w_ptr,          # const float* [hidden]              (γ)
    b_ptr,          # const float* [hidden]              (β)
    total_rows,     # int
    hidden: tl.constexpr,          # static – hidden dimension (C)
    eps: tl.constexpr,             # static
    BLOCK_M: tl.constexpr,         # rows per program (e.g. 256)
    BLOCK_D: tl.constexpr,         # channels per inner loop (e.g. 64)
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # Pass 1 – mean & variance (float32 accumulation)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(hidden, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < hidden
        ptr = x_ptr + offs_m[:, None] * hidden + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)            # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / hidden
    var  = sum_sq / hidden - mean * mean
    rstd = tl.rsqrt(var + eps)                              # 1/√(var+eps)

    # ------------------------------------------------------------------
    # Pass 2 – normalise, affine, multiply gate and store (FP16)
    # ------------------------------------------------------------------
    for d in range(0, tl.cdiv(hidden, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < hidden

        # load normalised input
        x_ptrs = x_ptr + offs_m[:, None] * hidden + offs_d[None, :]
        x_vals = tl.load(x_ptrs,
                         mask=mask_m[:, None] & mask_d[None, :],
                         other=0.0).to(tl.float32)

        # load gate (already sigmoid‑ed)
        g_ptrs = gate_ptr + offs_m[:, None] * hidden + offs_d[None, :]
        g_vals = tl.load(g_ptrs,
                         mask=mask_m[:, None] & mask_d[None, :],
                         other=0.0).to(tl.float32)

        # affine transform
        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)

        norm = (x_vals - mean[:, None]) * rstd[:, None]      # LN
        out  = norm * w[None, :] + b[None, :]                # γ/β
        out  = out * g_vals                                   # element‑wise gate

        # store FP16 result
        y_ptrs = y_ptr + offs_m[:, None] * hidden + offs_d[None, :]
        tl.store(y_ptrs, out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor  [B, N, N, dim]  (float32)
        - mask_tensor  : torch.Tensor  [B, N, N]      (bool/float, optional)
        - weights_dict : dict of weight tensors
        - config_dict  : {
            "dim"        : int,
            "hidden_dim" : int,
            "nomask"     : bool (True → ignore mask),
            "eps"        : float (optional, defaults to 1e‑5)
          }

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm – use the highly‑optimised cuDNN kernel
    # ------------------------------------------------------------------
    #   y = (x - μ) / √(σ²+ε) * γ + β
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x_norm = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )   # [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------
    # 2️⃣ Fuse the five linear maps (proj + gates) into one GEMM
    # ------------------------------------------------------------------
    # Cast to FP16 – all dense algebra is done in half precision
    x_norm_fp16 = x_norm.view(-1, dim).to(torch.float16)          # [R, dim]
    # Stack weights vertically: (5 * hidden_dim, dim)
    weight_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)

    # Single linear transform
    proj = F.linear(x_norm_fp16, weight_concat)                  # [R, 5*H] fp16

    # Split into the five logical tensors
    left_proj_raw, left_gate_raw, right_proj_raw, right_gate_raw, out_gate_raw = \
        torch.split(proj, hidden_dim, dim=1)

    # ------------------------------------------------------------------
    # 3️⃣ Apply optional mask (only on the two *projections*)
    # ------------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        # mask is [B,N,N] → broadcast over the hidden dimension
        mask_fp16 = mask_tensor.view(-1, 1).to(torch.float16)   # [R,1]
        left_proj_raw  = left_proj_raw * mask_fp16
        right_proj_raw = right_proj_raw * mask_fp16

    # ------------------------------------------------------------------
    # 4️⃣ Apply sigmoid‑gates (still half‑precision)
    # ------------------------------------------------------------------
    left  = left_proj_raw  * torch.sigmoid(left_gate_raw)      # [R, H]
    right = right_proj_raw * torch.sigmoid(right_gate_raw)    # [R, H]
    out_gate = torch.sigmoid(out_gate_raw)                    # [R, H]

    # ------------------------------------------------------------------
    # 5️⃣ Triangular multiplicative update (einsum, no extra copies)
    # ------------------------------------------------------------------
    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N                                   # R
    left  = left.view(B, N, N, hidden_dim)                   # [B,N,N,H]
    right = right.view(B, N, N, hidden_dim)                  # [B,N,N,H]

    # Einsum implements: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    out = torch.einsum('b i k h, b j k h -> b i j h', left, right)   # [B,N,N,H]

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm + output‑gate (custom Triton kernel)
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, hidden_dim)               # [R, H] fp16
    out_gate_flat = out_gate.view(total_rows, hidden_dim)     # already fp16
    out_norm_gated = torch.empty_like(out_flat)               # FP16 destination

    # Triton launch configuration
    BLOCK_M = 256                     # rows per program – tuned for H100
    BLOCK_D = 64                      # hidden_dim per inner loop
    grid = (triton.cdiv(total_rows, BLOCK_M),)

    layernorm_gate_kernel[grid](
        out_flat,
        out_gate_flat,
        out_norm_gated,
        weights["to_out_norm.weight"],          # γ (float32)
        weights["to_out_norm.bias"],            # β (float32)
        total_rows,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()

    # ------------------------------------------------------------------
    # 7️⃣ Final projection back to `dim` (FP16 GEMM) and cast to FP32
    # ------------------------------------------------------------------
    final_fp16 = F.linear(
        out_norm_gated,
        weights["to_out.weight"].to(torch.float16)   # [dim, hidden_dim]
    )                                                # [R, dim] fp16

    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
```
