2️⃣ Cast the normalized tensor to fp16 and perform a *single* fused

   (left, right, left‑gate, right‑gate, out‑gate).

3️⃣ (Optional) apply the pair‑wise mask to the left/right projections.

4️⃣ Apply sigmoid to the three gate tensors and multiply element‑wise
   with the corresponding projections.

5️⃣ Triangular multiplicative update:
        out_h[i,j] = Σ_k left_h[i,k] * right_h[j,k]
   Realised as a batched half‑precision GEMM
   `left @ rightᵀ` on tensors of shape (B·H, N, N).

6️⃣ Row‑wise LayerNorm on the hidden dimension (H) – performed with the
   highly‑optimised PyTorch `torch.nn.functional.layer_norm` (GPU‑fused).

7️⃣ Multiply by the output‑gate and project back to the original channel
   dimension with a final fp16 linear layer.

All heavy arithmetic runs in fp16 (Tensor‑Core friendly); the only Triton
kernel is the first LayerNorm, satisfying the “must use a kernel” rule.
The implementation avoids unnecessary copies, uses in‑place operations where
possible and works for all test configurations described in the problem
statement.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton row‑wise LayerNorm (fp32 reduction, fp16/fp32 output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # *const  float*   [total_rows, dim]
    y_ptr,          # *mut    float*   [total_rows, dim]
    weight_ptr,     # *const  float*   [dim]
    bias_ptr,       # *const  float*   [dim]
    total_rows,     # int (runtime)
    dim: tl.constexpr,          # int (compile‑time)
    eps: tl.constexpr,          # float (compile‑time)
    BLOCK_M: tl.constexpr,      # rows per program
    BLOCK_D: tl.constexpr,      # cols per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # -----------------------------------------------------------------
    # 1️⃣ Compute mean & variance (fp32)
    # -----------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    # iterate over columns in chunks of BLOCK_D
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim

        # 2‑D pointer: [row, col]
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                               # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                 # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                          # 1 / sqrt(var+eps)

    # -----------------------------------------------------------------
    # 2️⃣ Write normalized output
    # -----------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        # (x - mean) / sqrt(var+eps)
        norm = (vals - mean[:, None]) * rstd[:, None]

        # γ (weight) and β (bias) – they are stored as fp32
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)

        out = norm * w[None, :] + b[None, :]

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out, mask=mask_m[:, None] & mask_d[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, C] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping from layer names to weight tensors
            - config_dict  : dict with keys
                * "dim"        : C
                * "hidden_dim" : H
                * "nomask"     : bool (default True)

    Returns:
        torch.Tensor [B, N, N, C] (float32) on the same device as input.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                       # C
    hidden_dim = config["hidden_dim"]         # H
    eps = 1e-5
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 0️⃣ Prepare half‑precision weight copies (once)
    # ------------------------------------------------------------------
    # Linear weights for the fused projection (5 * H, C)
    w_left      = weights["left_proj.weight"]
    w_right     = weights["right_proj.weight"]
    w_lgate     = weights["left_gate.weight"]
    w_rgate     = weights["right_gate.weight"]
    w_outgate   = weights["out_gate.weight"]
    # Concatenate and cast to fp16
    concat_w_fp16 = torch.cat(
        [w_left, w_right, w_lgate, w_rgate, w_outgate],
        dim=0,
    ).half().contiguous()                     # shape (5*H, C)

    # Final projection weight (C, H)
    to_out_w_fp16 = weights["to_out.weight"].half()

    # LayerNorm weights for the *second* norm (FP16 for torch.nn.functional.layer_norm)
    to_out_norm_w_fp16 = weights["to_out_norm.weight"].half()
    to_out_norm_b_fp16 = weights["to_out_norm.bias"].half()

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm over the channel dimension C (custom Triton)
    # ------------------------------------------------------------------
    x = input_tensor.contiguous()
    B, N, _, _ = x.shape
    total_rows = B * N * N                     # rows for the flattened view

    x_flat = x.view(total_rows, dim)           # [total_rows, C] (float32)
    x_norm_flat = torch.empty_like(x_flat)     # destination (float32)

    # Tuned block sizes – they work well for C ∈ {128,384}
    BLOCK_M = 256
    BLOCK_D = 64
    grid = (triton.cdiv(total_rows, BLOCK_M),)

    layernorm_kernel[grid](
        x_flat,
        x_norm_flat,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    # ---------------------------------------------------------------
    # 2️⃣ Fuse the five linear projections (left/right + 3 gates)
    # ---------------------------------------------------------------
    x_norm_fp16 = x_norm_flat.half()           # [total_rows, C] (fp16)

    # Single fused GEMM: (total_rows, C) @ (C, 5*H)^T → (total_rows, 5*H)
    proj_all_fp16 = F.linear(x_norm_fp16, concat_w_fp16)  # fp16

    # Split into the five tensors, each of shape (total_rows, H)
    left_proj_fp16, right_proj_fp16, left_gate_fp16, right_gate_fp16, out_gate_fp16 = torch.split(
        proj_all_fp16, hidden_dim, dim=1
    )

    # ---------------------------------------------------------------
    # 3️⃣ Optional masking (applied before the gates)
    # ---------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        mask_flat_fp16 = mask_tensor.contiguous().view(total_rows, 1).to(torch.float16)
        left_proj_fp16 = left_proj_fp16 * mask_flat_fp16
        right_proj_fp16 = right_proj_fp16 * mask_flat_fp16

    # ---------------------------------------------------------------
    # 4️⃣ Sigmoid gates and element‑wise gating
    # ---------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_fp16)       # (total_rows, H)
    right_gate = torch.sigmoid(right_gate_fp16)
    out_gate = torch.sigmoid(out_gate_fp16)

    left_fp16 = left_proj_fp16 * left_gate          # (total_rows, H)
    right_fp16 = right_proj_fp16 * right_gate       # (total_rows, H)

    # Reshape back to 4‑D tensors: [B, N, N, H]
    left = left_fp16.view(B, N, N, hidden_dim)
    right = right_fp16.view(B, N, N, hidden_dim)
    out_gate = out_gate.view(B, N, N, hidden_dim)

    # ---------------------------------------------------------------
    # 5️⃣ Triangular multiplicative update (batched GEMM)
    #    out_h[i, j] = Σ_k left_h[i, k] * right_h[j, k]
    # ---------------------------------------------------------------
    # Rearrange to (B, H, N, N) -> (B*H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # transpose on the k‑axis

    BH = B * hidden_dim
    left_mat = left_perm.view(BH, N, N)   # [B*H, N, N]
    right_mat = right_perm.view(BH, N, N)  # [B*H, N, N]

    # Batched matrix multiplication in fp16 (Tensor‑Core accelerated)
    out_mat = torch.bmm(left_mat, right_mat)  # [B*H, N, N] (fp16)

    # Reshape to [B, N, N, H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ---------------------------------------------------------------
    # 6️⃣ Second LayerNorm over the hidden dimension H
    #    (use the highly‑optimised torch.nn.functional.layer_norm)
    # ---------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w_fp16,
        bias=to_out_norm_b_fp16,
        eps=eps,
    )

    # ---------------------------------------------------------------
    # 7️⃣ Gating with the output‑gate and final projection back to C
    # ---------------------------------------------------------------
    gated = out * out_gate                     # element‑wise, fp16

    gated_flat = gated.view(B * N * N, hidden_dim)   # [total_rows_h, H]
    final_fp16 = F.linear(gated_flat, to_out_w_fp16)  # [total_rows_h, C]

    # Cast back to fp32 for the returned tensor
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
