"""
Highly‑optimized Triton + PyTorch implementation of the *outgoing* TriMul
operator (AlphaFold‑3).

Algorithm
---------
1️⃣ Row‑wise LayerNorm over the input channel dimension C.  The kernel
   `layernorm_fp16_kernel` reads FP32 values, computes mean/variance in FP32,
   then writes the normalized (γ·x̂+β) result as FP16.

2️⃣ Fuse the five linear projections (left/right and three gates) into a
   single half‑precision GEMM by concatenating their weight matrices.
   The result is split back into the individual tensors.

3️⃣ If a pair‑wise mask is present, apply it to the left/right projections
   (still FP16).

4️⃣ Apply sigmoid to the three gate tensors and perform element‑wise gating
   on the left/right projections.

5️⃣ Triangular multiplicative update:
        out_h[i,j] = Σ_k left_h[i,k] * right_h[j,k]   (batched GEMM)
   Implemented with a single `torch.bmm` over the batch dimension
   (B·H matrices of size N×N).  This uses Tensor‑Core FP16 GEMM.

6️⃣ Fuse the second LayerNorm (over the hidden dimension H) with the final
   output‑gate multiplication in `layernorm_gate_fp16_kernel`, writing the
   result directly as FP16.

7️⃣ Final linear projection back to the original channel dimension C
   (FP16 GEMM) and cast the result to FP32.

All heavy reductions (the two LayerNorms) run in custom Triton kernels,
while dense linear algebra uses cuBLAS‑accelerated GEMMs via
`torch.nn.functional.linear` and `torch.bmm`.  The implementation works
for all configurations in the test suite (mask/no‑mask, various
sequence lengths, and channel dimensions 128/384).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernels
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,            # *const float*   [total_rows, dim]
    y_ptr,            # *mut   half*    [total_rows, dim]
    weight_ptr,       # *const float*   [dim]   (γ)
    bias_ptr,         # *const float*   [dim]   (β)
    total_rows,       # int (runtime)
    dim: tl.constexpr,      # int (compile‑time)
    eps: tl.constexpr,      # float (compile‑time)
    BLOCK_M: tl.constexpr,  # rows per program
    BLOCK_D: tl.constexpr,  # cols per iteration
):
    """Row‑wise LayerNorm that writes FP16 output."""
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)                     # [BLOCK_M]
    mask_m = offs_m < total_rows

    # -------------------------------------------------------------
    # First pass – compute mean & variance (FP32 accumulators)
    # -------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    num_blocks_d = tl.cdiv(dim, BLOCK_D)
    for d in range(0, num_blocks_d):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)               # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                                          # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                                      # 1 / sqrt(var+eps)

    # -------------------------------------------------------------
    # Second pass – write normalized (γ·x̂+β) as FP16
    # -------------------------------------------------------------
    for d in range(0, num_blocks_d):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]   # (x - μ)/σ

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β

        out = norm * w[None, :] + b[None, :]           # FP32
        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out.to(tl.float16), mask=mask_m[:, None] & mask_d[None, :])

    weight_ptr,       # *const float*  [dim]               (γ of LN)

    # -------------------------------------------------------------

        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).half().contiguous()                     # (5*H, C)

    # Linear projection: [total_rows, C] @ [C, 5*H].T -> [total_rows, 5*H] (FP16)
    proj_all = F.linear(x_norm_fp16, concat_weight)

    # Split back into individual tensors
    left_proj_fp16, right_proj_fp16, left_gate_fp16, right_gate_fp16, out_gate_fp16 = torch.split(
        proj_all, hidden_dim, dim=1
    )                                          # each: (total_rows, H)

    # --------------------------------------------------------------
    # 2a. Optional mask (applied to left/right projections)
    # --------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] -> [total_rows, 1] (float16)
        mask_fp16 = mask_tensor.to(dtype=torch.float16).contiguous().view(total_rows, 1)
        left_proj_fp16 = left_proj_fp16 * mask_fp16
        right_proj_fp16 = right_proj_fp16 * mask_fp16

    # --------------------------------------------------------------
    # 2b. Apply sigmoid gates and element‑wise gating
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_fp16)      # (total_rows, H)
    right_gate = torch.sigmoid(right_gate_fp16)
    out_gate = torch.sigmoid(out_gate_fp16)

    left_fp16 = left_proj_fp16 * left_gate          # (total_rows, H)
    right_fp16 = right_proj_fp16 * right_gate

    # Reshape back to [B, N, N, H]
    left = left_fp16.view(B, N, N, hidden_dim)          # (B, N, N, H)
    right = right_fp16.view(B, N, N, hidden_dim)
    out_gate = out_gate.view(B, N, N, hidden_dim)      # used later

    # --------------------------------------------------------------
    # 3. Triangular multiplicative update (batched GEMM)
    #    out_h[i,j] = Σ_k left_h[i,k] * right_h[j,k]
    # --------------------------------------------------------------
    # Prepare (B*H, N, N) matrices: L @ Rᵀ  (FP16 GEMM)
    left_mat = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)     # (B*H, N, N)
    right_mat = right.permute(0, 3, 2, 1).reshape(B * hidden_dim, N, N)   # (B*H, N, N) with k as first axis

    out_mat = torch.bmm(left_mat, right_mat)      # (B*H, N, N), FP16

    # Restore shape [B, N, N, H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # --------------------------------------------------------------
    # 4. Fuse second LayerNorm (over H) with the final output gate
    # --------------------------------------------------------------
    total_rows_h = B * N * N
    out_flat = out.view(total_rows_h, hidden_dim)                # (total_rows_h, H)
    out_gate_flat = out_gate.view(total_rows_h, hidden_dim)      # (total_rows_h, H)

    fused_fp16 = torch.empty_like(out_flat)                      # buffer for LN·gate output (FP16)

    grid2 = (triton.cdiv(total_rows_h, BLOCK_M),)
    layernorm_gate_fp16_kernel[grid2](
        out_flat,
        out_gate_flat,
        fused_fp16,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows_h,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    # No sync needed – next op depends on `fused_fp16`.

    # --------------------------------------------------------------
    # 5. Final linear projection back to C (FP16 GEMM) and cast to FP32
    # --------------------------------------------------------------
    to_out_weight_fp16 = weights["to_out.weight"].half()
    final_fp16 = F.linear(fused_fp16, to_out_weight_fp16)      # (total_rows_h, C) FP16

    final = final_fp16.view(B, N, N, dim).to(dtype=torch.float32)

    return final
