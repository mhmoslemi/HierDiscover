"""
TriMul (outgoing) – Triton‑accelerated forward pass
===================================================

The module processes a 4‑D tensor `x` of shape [B, N, N, C] (C = dim) and
computes the “outgoing” TriMul operation described in the AlphaFold3 paper.

The forward pass consists of six fused stages:

1️⃣  Row‑wise LayerNorm (FP32 → FP16).  Statistics are accumulated in FP32,
    normalised values are stored in FP16.

2️⃣  A fused linear projection computes the five weight matrices
    (left_proj, right_proj, left_gate, right_gate, out_gate) in a single
    `torch.nn.functional.linear` call.

3️⃣  A Triton kernel splits the fused projection, applies the optional
    pair‑wise mask, evaluates the three sigmoid gates and writes the three
    tensors `left`, `right` and `out_gate` in layout [B, H, N, N] (H = hidden_dim).

4️⃣  Batched GEMM (`torch.bmm`) contracts the tensors:
    `out[b, h] = left[b, h] @ right[b, h].T`   →   shape (B·H, N, N) in FP16.

5️⃣  Row‑wise mean / rsqrt‑variance of `out` are computed with a Triton kernel
    (required for the second LayerNorm).

6️⃣  A final Triton kernel fuses the second LayerNorm, the out‑gate,
    and the final linear projection (`to_out`).  It normalises each hidden
    vector, applies the γ/β of `to_out_norm`, multiplies by the out‑gate
    and finally projects to the original `dim` using the transposed weight
    `to_out.weightᵀ`.  The result is written as a contiguous float‑32 tensor
    of shape [B, N, N, dim].

All kernels are written for the H100 (CUDA compute capability ≥ 9.0) and use
FP16 arithmetic with FP32 accumulation for numerical stability.  The
implementation works with both masked and un‑masked inputs and with any
allowed combination of batch size, sequence length and feature sizes.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (float32 → float16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,    # const   *[rows, dim] (float32)
    y_ptr,    # mutable *[rows, dim] (float16)
    w_ptr,    # const   *[dim]      (float32)   gamma
    b_ptr,    # const   *[dim]      (float32)   beta
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ------------------------------------------------------------------
    # Compute mean & variance (fp32)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # [BLOCK_M, BLOCK_D]
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------------------------------------------------------
    # Normalise, apply gamma / beta and store as FP16
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # [BLOCK_M, BLOCK_D]

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # gamma
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # beta

        out = norm * gamma[None, :] + beta[None, :]              # fp32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

# 2️⃣ Split fused projection, apply mask & sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,       # const   *[R, 5*H]   (fp16)   fused projection
    left_ptr,       # mutable *[B, H, N, N] (fp16)
    right_ptr,      # mutable *[B, H, N, N] (fp16)
    out_gate_ptr,   # mutable *[B, H, N, N] (fp16)
    mask_ptr,       # const   *[R]        (fp16)   (ignored if HAS_MASK=0)
    R: tl.int32,
    H: tl.int32,
    N: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)               # [BLOCK_R]
    mask_r = offs_r < R

    # Decode (b,i,k) from flat position
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                   # broadcast over H
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # Base offset of (b, 0, i, k) inside the [B, H, N, N] buffer
    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + k           # [BLOCK_R]

    # Process hidden dimension in chunks
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)          # [BLOCK_H]
        mask_h = offs_h < H

        # Layout in proj: [ left | right | left_gate | right_gate | out_gate ]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw      = tl.load(proj_ptr + proj_base,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_raw     = tl.load(proj_ptr + proj_base + H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        left_gate_raw = tl.load(proj_ptr + proj_base + 2 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_gate_raw= tl.load(proj_ptr + proj_base + 3 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        out_gate_raw  = tl.load(proj_ptr + proj_base + 4 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)

        # sigmoid in FP32 for stability → cast back to FP16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # Apply mask and gates
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        # out_gate is NOT masked – matches reference implementation
        outg  = out_gate

        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride

        tl.store(left_ptr    + dst_offset, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr   + dst_offset, right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr+ dst_offset, outg,
                 mask=mask_r[:, None] & mask_h[None, :])

# 3️⃣ Row‑wise mean & rsqrt‑variance of the contracted tensor
# ----------------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr,    # const *[B, H, N, N] (fp16)
    mu_ptr,     # mutable *[R]         (float32)
    rstd_ptr,   # mutable *[R]         (float32)
    total_rows: tl.int32,   # = B * N * N
    H: tl.int32,
    N: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + j          # offset of hidden‑plane 0

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H
        ptr = out_ptr + base[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)            # [M, K]
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_m, mean,  mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
            - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
            - weights_dict : mapping with Linear / LayerNorm parameters
                             (all float32, Linear layers have no bias)
            - config_dict  : must contain "dim", "hidden_dim", optionally "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim        = cfg["dim"]               # input feature dimension (C)
    hidden_dim = cfg["hidden_dim"]        # hidden dimension H
    eps        = 1e-5
    device     = x.device

    B, N, _, _ = x.shape
    R = B * N * N            # total number of (b,i,k) positions

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 → float16)
    # ------------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()                     # [R, dim] float32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 256
    BLOCK_D = 64
    grid = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],   # gamma (float32)
        weights["norm.bias"],     # beta  (float32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection (left/right + three gates)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                      # [5*H, dim]

    proj = F.linear(x_norm, fused_w)                          # (R, 5*H) fp16
    del fused_w, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ Split, apply mask, evaluate sigmoid gates
    # ------------------------------------------------------------------
    left      = torch.empty(B, hidden_dim, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, hidden_dim, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, hidden_dim, N, N, dtype=torch.float16, device=device)

    if cfg.get("nomask", True) or mask is None:
        has_mask = 0
        dummy_mask = torch.empty(0, dtype=torch.float16, device=device)
        mask_ptr = dummy_mask
    else:
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat

    BLOCK_R = 128
    BLOCK_H = 32
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        hidden_dim,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM (Tensor‑Core)
    # ------------------------------------------------------------------
    left_mat  = left.view(B * hidden_dim, N, N)
    right_mat = right.view(B * hidden_dim, N, N)

    out = torch.bmm(left_mat, right_mat.transpose(1, 2))    # [B·H, N, N] fp16
    out = out.view(B, hidden_dim, N, N)                     # back to (B, H, N, N)

    # ------------------------------------------------------------------
    # 5️⃣ Row‑wise statistics (mean & rsqrt‑var) of `out`
    # ------------------------------------------------------------------
    mu   = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 256
    BLOCK_K_STATS = 64
    grid_stats = (triton.cdiv(R, BLOCK_M_STATS),)
    rowwise_stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R,
        hidden_dim,
        N,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Fuse second LayerNorm, out_gate and final linear projection
    # ------------------------------------------------------------------
    w_out_T = weights["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim]

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 128          # rows per program (positions)
    BLOCK_K_OUT = 32           # hidden‑dim chunk
    BLOCK_N_OUT = 64           # output‑dim chunk
    grid_out = (
        triton.cdiv(R, BLOCK_M_OUT),
        triton.cdiv(dim, BLOCK_N_OUT),
    )
    final_output_kernel[grid_out](
        out,
        out_gate,
        mu,
        rstd,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        w_out_T,
        final,
        R,
        hidden_dim,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # Return result (float32)
    # ------------------------------------------------------------------
    return final
```
