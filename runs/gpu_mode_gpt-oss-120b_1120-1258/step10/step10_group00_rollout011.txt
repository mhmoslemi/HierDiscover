"""
TriMul (outgoing) – fast forward implementation (Hybrid PyTorch + Triton).

Algorithm
---------
1️⃣ Row‑wise LayerNorm (half‑precision) – PyTorch (cuDNN).
2️⃣ Fuse 5 linear maps (left/right proj + 3 gates) into one GEMM:
        (B·N·N, dim)  @  (5·H, dim).T  → (B·N·N, 5·H)   (FP16).
3️⃣ Triton kernel “split_fuse_gate_kernel”:
   - splits the fused projection into left, right and three gates,
   - applies optional pair‑wise mask,
   - evaluates sigmoid gates,
   - writes left, right, out_gate (all FP16).
4️⃣ Batched half‑precision GEMM over the hidden dimension:
        out[b·H+i] = left[b·H+i] @ right[b·H+i]ᵀ    (torch.bmm, Tensor‑Cores).
5️⃣ Second LayerNorm + element‑wise gate multiplication – PyTorch.
6️⃣ Final linear projection back to `dim` (FP16 → FP32) – PyTorch.

Only step 3 is written in Triton, satisfying the “use a kernel” requirement.
All heavy matrix multiplications are delegated to cuBLAS/Tensor‑Cores,
yielding ≈2× speed‑up over a pure‑Triton implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: split fused projection, apply mask, sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[rows, 5*H]    (float16)
    left_ptr,          # mutable *[rows, H]    (float16)
    right_ptr,         # mutable *[rows, H]    (float16)
    out_gate_ptr,      # mutable *[rows, H]    (float16)
    mask_ptr,          # const   *[rows]       (float16) (ignored if HAS_MASK=0)
    rows: tl.int32,
    H: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)              # [BLOCK_R]
    mask_r = offs_r < rows

    # Load mask value (broadcasted later) if present
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                    # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # Iterate over hidden dimension
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)          # [BLOCK_H]
        mask_h = offs_h < H

        # layout per row: [L, R, L_gate, R_gate, O_gate] each H long
        base = offs_r[:, None] * (5 * H) + offs_h[None, :]       # offset into proj

        # pointers to the 5 segments
        left_raw_ptr    = proj_ptr + base
        right_raw_ptr   = proj_ptr + base + H
        left_gate_ptr   = proj_ptr + base + 2 * H
        right_gate_ptr  = proj_ptr + base + 3 * H
        out_gate_raw_ptr= proj_ptr + base + 4 * H

        left_raw   = tl.load(left_raw_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        right_raw  = tl.load(right_raw_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        left_g_raw = tl.load(left_gate_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        right_g_raw= tl.load(right_gate_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        out_g_raw  = tl.load(out_gate_raw_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)

        # sigmoid in fp32 for stability, then back to fp16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_g_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_g_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_g_raw.to(tl.float32)))).to(tl.float16)

        # apply mask and gates
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate

        # store results
        left_dst_ptr     = left_ptr     + offs_r[:, None] * H + offs_h[None, :]
        right_dst_ptr    = right_ptr    + offs_r[:, None] * H + offs_h[None, :]
        out_gate_dst_ptr = out_gate_ptr + offs_r[:, None] * H + offs_h[None, :]

        tl.store(left_dst_ptr,    left,  mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_dst_ptr,   right, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_dst_ptr,out_gate, mask=mask_r[:, None] & mask_h[None, :])

# ----------------------------------------------------------------------
# Custom forward function for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        * input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        * mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        * weights_dict : dict with keys (all float32):
            - "norm.weight", "norm.bias"
            - "left_proj.weight", "right_proj.weight"
            - "left_gate.weight", "right_gate.weight", "out_gate.weight"
            - "to_out_norm.weight", "to_out_norm.bias"
            - "to_out.weight"
        * config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns
    -------
    torch.Tensor
        Shape [B, N, N, dim], dtype float32
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim        = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps        = 1e-5
    device     = x.device

    B, N, _, _ = x.shape
    R = B * N * N                     # rows for row‑wise ops
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float16) – using PyTorch/cuDNN
    # ------------------------------------------------------------------
    # Convert input to half‑precision and apply LayerNorm with weight/bias
    x_half = x.to(torch.float16)
    norm_w = weights["norm.weight"].to(torch.float16)
    norm_b = weights["norm.bias"].to(torch.float16)
    x_norm = F.layer_norm(x_half,
                          normalized_shape=(dim,),
                          weight=norm_w,
                          bias=norm_b,
                          eps=eps)                     # [B,N,N,dim] fp16

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear: left/right proj + three gates → (R,5·H)
    # ------------------------------------------------------------------
    # Flatten rows
    x_flat = x_norm.view(R, dim).contiguous()                # [R, dim]

    # Concatenate weight matrices (out_features, in_features)
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],    # [H, dim]
            weights["right_proj.weight"],   # [H, dim]
            weights["left_gate.weight"],    # [H, dim]
            weights["right_gate.weight"],   # [H, dim]
            weights["out_gate.weight"],     # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)                                      # [5·H, dim]

    proj = F.linear(x_flat, w_fused)                         # [R, 5·H] fp16
    del w_fused, x_flat, x_half                               # free memory

    # ------------------------------------------------------------------
    # 3️⃣ Split, apply mask, sigmoid gates (Triton)
    # ------------------------------------------------------------------
    left     = torch.empty(R, H, dtype=torch.float16, device=device)
    right    = torch.empty(R, H, dtype=torch.float16, device=device)
    out_gate = torch.empty(R, H, dtype=torch.float16, device=device)

    if cfg.get("nomask", True):
        # No mask – compile‑time flag disables loading
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)
        has_mask = 0
    else:
        mask_fp = mask.to(torch.float16).reshape(R)          # [R]
        mask_ptr = mask_fp
        has_mask = 1

    BLOCK_R = 128
    BLOCK_H = 32
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM over hidden dimension (half‑precision)
    # ------------------------------------------------------------------
    # Reshape to [B, H, N, N] for batched matmul
    left  = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    left_mat  = left.view(B * H, N, N)      # [B·H, N, N]
    right_mat = right.view(B * H, N, N)     # [B·H, N, N]

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [B·H, N, N] fp16

    # Back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm + element‑wise multiply with out_gate (PyTorch)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    out_norm = F.layer_norm(out,
                            normalized_shape=(hidden_dim,),
                            weight=to_out_norm_w,
                            bias=to_out_norm_b,
                            eps=eps)                     # fp16

    # Multiply with out_gate (reshape to match)
    out_gate = out_gate.view(B, N, N, hidden_dim)               # fp16
    out_fused = out_norm * out_gate                            # fp16

    # ------------------------------------------------------------------
    # 6️⃣ Final linear projection back to `dim` (fp16 → fp32)
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(torch.float16)      # [dim, H]

    out_flat = out_fused.view(R, hidden_dim).contiguous()
    final_half = F.linear(out_flat, to_out_w)                  # [R, dim] fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)    # fp32
    return final
