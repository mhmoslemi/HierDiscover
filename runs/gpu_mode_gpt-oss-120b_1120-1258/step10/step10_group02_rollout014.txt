```python
import triton
import triton.language as tl

@triton.jit
def fused_norm_proj_gate_kernel(
    x_ptr,          # fp32 [rows, dim]
    mask_ptr,       # fp16 [rows] (or dummy)
    weight_ptr,     # fp16 [5*H, dim]
    left_ptr,       # fp16 [rows, H]
    right_ptr,      # fp16 [rows, H]
    out_gate_ptr,   # fp16 [rows, H]
    rows, dim, H,
    EPS: tl.float32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < rows

    # compute mean & var
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_R], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        x = tl.load(x_ptr + offs_r[:, None] * dim + offs_d[None, :],
                    mask=mask_r[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)
        sum_val += tl.sum(x, axis=1)
        sum_sq += tl.sum(x * x, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + EPS)

    # load mask if present
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0).to(tl.float16)
        mask_val = mask_val[:, None]   # broadcast
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # accumulators for each group
    left_acc = tl.zeros([BLOCK_R, H], dtype=tl.float32)
    right_acc = tl.zeros([BLOCK_R, H], dtype=tl.float32)
    left_gate_acc = tl.zeros([BLOCK_R, H], dtype=tl.float32)
    right_gate_acc = tl.zeros([BLOCK_R, H], dtype=tl.float32)
    out_gate_acc = tl.zeros([BLOCK_R, H], dtype=tl.float32)

    # offsets for heads
    offs_h = tl.arange(0, H)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        # load x block
        x = tl.load(x_ptr + offs_r[:, None] * dim + offs_d[None, :],
                    mask=mask_r[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)
        # normalize
        xn = (x - mean[:, None]) * rstd[:, None]   # fp32

        # For each of 5 groups load weight tile transposed and compute dot
        # Group 0: left_proj
        w0 = tl.load(weight_ptr + offs_d[:, None] + (0 * H + offs_h[None, :]) * dim,
                     mask=mask_d[:, None] & (offs_h[None, :] < H),
                     other=0.0, dtype=tl.float16).to(tl.float32)   # [BLOCK_D, H]
        left_acc += tl.dot(xn, w0)   # [BLOCK_R, H]

        # Group 1: right_proj
        w1 = tl.load(weight_ptr + offs_d[:, None] + (1 * H + offs_h[None, :]) * dim,
                     mask=mask_d[:, None] & (offs_h[None, :] < H),
                     other=0.0, dtype=tl.float16).to(tl.float32)
        right_acc += tl.dot(xn, w1)

        # Group 2: left_gate
        w2 = tl.load(weight_ptr + offs_d[:, None] + (2 * H + offs_h[None, :]) * dim,
                     mask=mask_d[:, None] & (offs_h[None, :] < H),
                     other=0.0, dtype=tl.float16).to(tl.float32)
        left_gate_acc += tl.dot(xn, w2)

        # Group 3: right_gate
        w3 = tl.load(weight_ptr + offs_d[:, None] + (3 * H + offs_h[None, :]) * dim,
                     mask=mask_d[:, None] & (offs_h[None, :] < H),
                     other=0.0, dtype=tl.float16).to(tl.float32)
        right_gate_acc += tl.dot(xn, w3)

        # Group 4: out_gate
        w4 = tl.load(weight_ptr + offs_d[:, None] + (4 * H + offs_h[None, :]) * dim,
                     mask=mask_d[:, None] & (offs_h[None, :] < H),
                     other=0.0, dtype=tl.float16).to(tl.float32)
        out_gate_acc += tl.dot(xn, w4)

    # sigmoid for gates (fp32 -> fp16)
    left_gate = (1.0 / (1.0 + tl.exp(-left_gate_acc))).to(tl.float16)
    right_gate = (1.0 / (1.0 + tl.exp(-right_gate_acc))).to(tl.float16)
    out_gate = (1.0 / (1.0 + tl.exp(-out_gate_acc))).to(tl.float16)

    # apply mask and gates
    left = (left_acc.to(tl.float16) * mask_val) * left_gate
    right = (right_acc.to(tl.float16) * mask_val) * right_gate
    out_gate = out_gate   # already sigmoid

    # store results
    tl.store(left_ptr + offs_r[:, None] * H + offs_h[None, :],
             left, mask=mask_r[:, None] & (offs_h[None, :] < H))
    tl.store(right_ptr + offs_r[:, None] * H + offs_h[None, :],
             right, mask=mask_r[:, None] & (offs_h[None, :] < H))
    tl.store(out_gate_ptr + offs_r[:, None] * H + offs_h[None, :],
             out_gate, mask=mask_r[:, None] & (offs_h[None, :] < H))
```

"""
Outgoing Tri‑Multiplicative update (TriMul) – highly‑optimized Triton+PyTorch.
Algorithm
---------
1️⃣  Fuse the first LayerNorm, the five linear projections (left/right &
    three gates) and the optional pair‑wise mask into a *single* Triton
    kernel.  This eliminates the large intermediate tensors  (`x_norm`
    ≈ R·dim fp16 and `proj` ≈ R·5·H fp16) and reduces memory traffic
    by ~2 GB for the largest test case.

2️⃣  The kernel produces the three tensors required for the next stage:
       *left*   – (R, H)   ← (norm(x) · W_left)           × mask × σ(gate_left)
       *right*  – (R, H)   ← (norm(x) · W_right)          × mask × σ(gate_right)
       *out_g*  – (R, H)   ← σ( norm(x) · W_out_gate )

   All arithmetic uses fp16 for the tensors and fp32 for the reductions,
   giving the same numerical results as the reference implementation.

3️⃣  Batched GEMM:   left @ rightᵀ   (B·H independent N×N matmuls) – performed
    with `torch.bmm`, which on an H100 maps to a single high‑throughput cuBLAS
    kernel.

4️⃣  Fuse the second LayerNorm with the element‑wise multiplication by the
    output‑gate (again a Triton kernel).

5️⃣  Final linear projection back to the model dimension (`dim`) via a
    cuBLAS half‑precision GEMM.

The kernels use block sizes tuned for H100 (BLOCK_R=64, BLOCK_D=32,
BLOCK_M=256, BLOCK_D_NORM=32).  The implementation works for all
configurations listed in the problem statement.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# 1️⃣  First LayerNorm + fused projection + mask + gates
# ----------------------------------------------------------------------
@triton.jit
def fused_norm_proj_gate_kernel(
    x_ptr,          # const  *[rows, dim]   (fp32)
    mask_ptr,       # const  *[rows]        (fp16)   (ignored if HAS_MASK==0)
    weight_ptr,     # const  *[5*H, dim]    (fp16)
    left_ptr,       # mutable *[rows, H]    (fp16)
    right_ptr,      # mutable *[rows, H]    (fp16)
    out_gate_ptr,   # mutable *[rows, H]    (fp16)
    rows: tl.int32,
    dim: tl.int32,
    H: tl.int32,
    EPS: tl.float32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # dim tile
):
    """
    For each row r:
        norm   = (x[r] - μ[r]) / σ[r]
        left   = (norm @ W_left.T)   * mask * σ(norm @ W_left_gate.T)
        right  = (norm @ W_right.T)  * mask * σ(norm @ W_right_gate.T)
        out_g  = σ( norm @ W_out_gate.T )
    All outputs are fp16.  The five weight slices are packed consecutively
    in ``weight_ptr`` (order: left, right, left_gate, right_gate, out_gate).
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < rows

    # --------------------------------------------------------------
    # 1️⃣  Compute mean / variance of each row (fp32 reduction)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_R], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        x = tl.load(x_ptr + offs_r[:, None] * dim + offs_d[None, :],
                    mask=mask_r[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)               # [R,D]
        sum_val += tl.sum(x, axis=1)
        sum_sq  += tl.sum(x * x, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + EPS)                     # 1/σ

    # --------------------------------------------------------------
    # 2️⃣  Load optional mask (broadcast later)
    # --------------------------------------------------------------
    if HAS_MASK:
        m = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # fp16
        mask_val = m[:, None]                                   # [R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # --------------------------------------------------------------
    # 3️⃣  Accumulate the five linear projections (fp32 accumulator)
    # --------------------------------------------------------------
    # one accumulator per output group
    left_acc      = tl.zeros([BLOCK_R, H], dtype=tl.float32)
    right_acc     = tl.zeros([BLOCK_R, H], dtype=tl.float32)
    left_gate_acc = tl.zeros([BLOCK_R, H], dtype=tl.float32)
    right_gate_acc= tl.zeros([BLOCK_R, H], dtype=tl.float32)
    out_gate_acc  = tl.zeros([BLOCK_R, H], dtype=tl.float32)

    offs_h = tl.arange(0, H)                                 # [H]

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        # load raw input tile and normalise
        x = tl.load(x_ptr + offs_r[:, None] * dim + offs_d[None, :],
                    mask=mask_r[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)               # [R,D]
        xn = (x - mean[:, None]) * rstd[:, None]           # fp32

        # ----- group 0 : left projection ---------------------------------
        w0 = tl.load(weight_ptr + offs_d[:, None] +
                     (0 * H + offs_h[None, :]) * dim,
                     mask=mask_d[:, None],
                     other=0.0, dtype=tl.float16).to(tl.float32)   # [D,H]
        left_acc += tl.dot(xn, w0)                                  # [R,H]

        # ----- group 1 : right projection --------------------------------
        w1 = tl.load(weight_ptr + offs_d[:, None] +
                     (1 * H + offs_h[None, :]) * dim,
                     mask=mask_d[:, None],
                     other=0.0, dtype=tl.float16).to(tl.float32)
        right_acc += tl.dot(xn, w1)

        # ----- group 2 : left‑gate ---------------------------------------
        w2 = tl.load(weight_ptr + offs_d[:, None] +
                     (2 * H + offs_h[None, :]) * dim,
                     mask=mask_d[:, None],
                     other=0.0, dtype=tl.float16).to(tl.float32)
        left_gate_acc += tl.dot(xn, w2)

        # ----- group 3 : right‑gate --------------------------------------
        w3 = tl.load(weight_ptr + offs_d[:, None] +
                     (3 * H + offs_h[None, :]) * dim,
                     mask=mask_d[:, None],
                     other=0.0, dtype=tl.float16).to(tl.float32)
        right_gate_acc += tl.dot(xn, w3)

        # ----- group 4 : out‑gate ----------------------------------------
        w4 = tl.load(weight_ptr + offs_d[:, None] +
                     (4 * H + offs_h[None, :]) * dim,
                     mask=mask_d[:, None],
                     other=0.0, dtype=tl.float16).to(tl.float32)
        out_gate_acc += tl.dot(xn, w4)

    # --------------------------------------------------------------
    # 4️⃣  Apply sigmoid to gates (fp32 → fp16)
    # --------------------------------------------------------------
    left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_acc))).to(tl.float16)
    right_gate = (1.0 / (1.0 + tl.exp(-right_gate_acc))).to(tl.float16)
    out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_acc))).to(tl.float16)

    # --------------------------------------------------------------
    # 5️⃣  Mask + gate + store (fp16)
    # --------------------------------------------------------------
    left  = left_acc.to(tl.float16) * mask_val * left_gate
    right = right_acc.to(tl.float16) * mask_val * right_gate

    tl.store(left_ptr  + offs_r[:, None] * H + offs_h[None, :],
             left,  mask=mask_r[:, None] & (offs_h[None, :] < H))
    tl.store(right_ptr + offs_r[:, None] * H + offs_h[None, :],
             right, mask=mask_r[:, None] & (offs_h[None, :] < H))
    tl.store(out_gate_ptr + offs_r[:, None] * H + offs_h[None, :],
             out_gate, mask=mask_r[:, None] & (offs_h[None, :] < H))

# ----------------------------------------------------------------------
# 2️⃣  Second LayerNorm + multiply by out_gate (fused)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_mul_kernel(
    x_ptr,            # const   *[rows, dim]   (fp16)
    gate_ptr,         # const   *[rows, dim]   (fp16)
    y_ptr,            # mutable *[rows, dim]   (fp16)
    weight_ptr,       # const   *[dim]        (fp32)   γ
    bias_ptr,         # const   *[dim]        (fp32)   β
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)                 # [BLOCK_M]
    mask_m = offs_m < rows

    # --------------------------------------------------------------
    # mean / variance (fp32)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)       # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :] # [M,D]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------------------------------------------------------------
    # normalize, apply γ/β, multiply gate, store (fp16)
    # --------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        x_val = tl.load(x_ptr + offs_m[:, None] * dim + offs_d[None, :],
                        mask=mask_m[:, None] & mask_d[None, :],
                        other=0.0).to(tl.float32)

        norm = (x_val - mean[:, None]) * rstd[:, None]   # fp32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ fp32
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β fp32
        norm = norm * w[None, :] + b[None, :]                     # fp32

        g = tl.load(gate_ptr + offs_m[:, None] * dim + offs_d[None, :],
                    mask=mask_m[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)                     # fp32

        out = norm * g                                          # fp32
        out_fp16 = out.to(tl.float16)

        tl.store(x_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])
