"""
Outgoing Tri‑Multiplicative update (TriMul) – fast Triton implementation.

The program follows the exact computation of the reference PyTorch module
but rearranges the work so that the heavy per‑row operations are done in
custom Triton kernels while the expensive N×N×N bilinear product uses
batched Tensor‑Core GEMMs from cuBLAS.  The forward pass consists of:

1️⃣ Row‑wise LayerNorm (fp32 → fp16) on the input tensor of shape
   [B·N·N, dim].

2️⃣ A *single* fused linear projection that computes all five weight
   matrices (left_proj, right_proj, left_gate, right_gate, out_gate)
   together:  X_norm @ W_fusedᵀ → (R, 5·H)  where R = B·N·N,
   H = hidden_dim.  The concatenated weight matrix lives in fp16 to
   trigger Tensor‑Core GEMM.

3️⃣ Triton kernel `split_fuse_gate_kernel` splits the fused result,
   applies the optional mask, evaluates sigmoid gates (in fp32 for
   stability) and writes three tensors: `left`, `right` and `out_gate`
   (all fp16).

4️⃣ Batched matrix‑multiply over the hidden dimension:
        out[b·H+i] = left[b·H+i] @ right[b·H+i]ᵀ
   Implemented with `torch.bmm`, which internally calls cuBLAS and
   delivers near‑peak performance on H100.

5️⃣ Triton kernel `layernorm_mul_kernel` fuses the second LayerNorm with
   the element‑wise multiplication by `out_gate`.

6️⃣ Final linear projection back to `dim` using a fp16 GEMM.

The implementation works for any configuration appearing in the test
suite (different batch sizes, sequence lengths, hidden dimensions and
optional masks).  All kernels are written for Triton 3.3.1 and tuned
with block sizes that give good occupancy on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (fp32 → fp16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   *[rows, dim]   (fp32)
    y_ptr,          # mutable *[rows, dim]   (fp16)
    weight_ptr,     # const   *[dim]        (fp32)   γ
    bias_ptr,       # const   *[dim]        (fp32)   β
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)                 # [BLOCK_M]
    mask_m = offs_m < rows

    # ---------- mean / variance (fp32) ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)       # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :] # [M,D]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- write normalized values ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]          # fp32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ (fp32)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β (fp32)

        out = norm * w[None, :] + b[None, :]                    # fp32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, apply mask, sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[rows, 5*H]    (fp16)
    left_ptr,          # mutable *[rows, H]    (fp16)
    right_ptr,         # mutable *[rows, H]    (fp16)
    out_gate_ptr,      # mutable *[rows, H]    (fp16)
    mask_ptr,          # const   *[rows]       (fp16)   (ignored if HAS_MASK==0)
    rows: tl.int32,
    H: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)               # [BLOCK_R]
    mask_r = offs_r < rows

    # ------------------------------------------------------------------
    # Load mask value (scalar per row) if present
    # ------------------------------------------------------------------
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                    # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # ------------------------------------------------------------------
    # Loop over hidden dim blocks
    # ------------------------------------------------------------------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)        # [BLOCK_H]
        mask_h = offs_h < H

        # base offset for the current row in the concatenated tensor
        row_off = offs_r[:, None] * (5 * H)                     # [R,1]

        # ----------- left -------------
        left_raw = tl.load(
            proj_ptr + row_off + offs_h[None, :],
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0).to(tl.float16)

        # ----------- right ------------
        right_raw = tl.load(
            proj_ptr + row_off + H + offs_h[None, :],
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0).to(tl.float16)

        # ----------- left gate ---------
        left_g_raw = tl.load(
            proj_ptr + row_off + 2 * H + offs_h[None, :],
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0).to(tl.float16)

        # ----------- right gate --------
        right_g_raw = tl.load(
            proj_ptr + row_off + 3 * H + offs_h[None, :],
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0).to(tl.float16)

        # ----------- out gate ----------
        out_g_raw = tl.load(
            proj_ptr + row_off + 4 * H + offs_h[None, :],
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0).to(tl.float16)

        # sigmoid in fp32 (more stable)
        left_gate  = (1.0 / (1.0 + tl.exp(-left_g_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_g_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_g_raw.to(tl.float32)))).to(tl.float16)

        # apply mask and gates
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate

        # store
        tl.store(left_ptr    + offs_r[:, None] * H + offs_h[None, :],
                 left,  mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr   + offs_r[:, None] * H + offs_h[None, :],
                 right, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr+ offs_r[:, None] * H + offs_h[None, :],
                 out_gate, mask=mask_r[:, None] & mask_h[None, :])

# ----------------------------------------------------------------------
# 3️⃣ Second LayerNorm + multiply by out_gate (fused)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_mul_kernel(
    x_ptr,            # const   *[rows, dim]   (fp16)
    gate_ptr,         # const   *[rows, dim]   (fp16)   (out_gate)
    y_ptr,            # mutable *[rows, dim]   (fp16)   (result)
    weight_ptr,       # const   *[dim]        (fp32)   γ
    bias_ptr,         # const   *[dim]        (fp32)   β
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)                 # [BLOCK_M]
    mask_m = offs_m < rows

    # ---------- mean / variance (fp32) ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)       # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :] # [M,D]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- normalize, apply γ/β, multiply by gate ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        # load x (fp16 → fp32)
        x_val = tl.load(x_ptr + offs_m[:, None] * dim + offs_d[None, :],
                        mask=mask_m[:, None] & mask_d[None, :],
                        other=0.0).to(tl.float32)

        # (x‑μ)·rstd
        norm = (x_val - mean[:, None]) * rstd[:, None]

        # γ, β
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # fp32
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # fp32
        norm = norm * w[None, :] + b[None, :]

        # gate (fp16 → fp32)
        g = tl.load(gate_ptr + offs_m[:, None] * dim + offs_d[None, :],
                    mask=mask_m[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)

        out = norm * g
        out_fp16 = out.to(tl.float16)

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor, [B, N, N, dim], dtype float32
            - mask_tensor  : torch.Tensor, [B, N, N] (bool/float) or None
            - weights_dict : dict of model parameters (all float32, Linear layers
                             have bias=False)
            - config_dict  : contains "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (dtype float32)
    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                # total rows for row‑wise ops
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (fp32 → fp16)
    # --------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()                     # [R, dim] fp32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 512          # more rows per program to improve occupancy
    BLOCK_D = 64           # more columns per iteration for better coalescing
    grid_ln1 = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln1](
        x_flat,
        x_norm,
        weights["norm.weight"],   # γ (fp32)
        weights["norm.bias"],     # β (fp32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    del x_flat

    # --------------------------------------------------------------
    # 2️⃣ Fused projection: left/right + 3 gates
    # --------------------------------------------------------------
    # Concatenate the five weight matrices (shape: [5*H, dim])
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)   # fp16 → Tensor‑Core friendly

    # Linear projection (R, dim) @ (5*H, dim).T → (R, 5*H)
    proj = F.linear(x_norm, w_concat)   # bias=None
    del w_concat, x_norm

    # --------------------------------------------------------------
    # 3️⃣ Split, mask, sigmoid gates (single Triton kernel)
    # --------------------------------------------------------------
    left     = torch.empty(R, H, dtype=torch.float16, device=device)
    right    = torch.empty(R, H, dtype=torch.float16, device=device)
    out_gate = torch.empty(R, H, dtype=torch.float16, device=device)

    if cfg.get("nomask", True) or mask is None:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)
    else:
        HAS_MASK = 1
        mask_half = mask.to(torch.float16).reshape(R)      # [R]
        mask_ptr = mask_half

    BLOCK_R = 256
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣ Batched GEMM over hidden dim (left @ rightᵀ)
    # --------------------------------------------------------------
    # Reshape to [B, N, N, H] → permute to [B, H, N, N] for bmm
    left = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    left_mat = left.view(B * H, N, N)      # [B*H, N, N]
    right_mat = right.view(B * H, N, N)    # [B*H, N, N]

    # out_mat[b·H+i] = left_mat @ right_matᵀ   (fp16)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))
    del left, right, left_mat, right_mat

    # Back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm + multiply by out_gate (fused)
    # --------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()          # [R, H] fp16
    out_fused = torch.empty_like(out_flat, dtype=torch.float16, device=device)

    BLOCK_M2 = 512
    BLOCK_D2 = 64
    grid_ln2 = (triton.cdiv(R, BLOCK_M2),)
    layernorm_mul_kernel[grid_ln2](
        out_flat,
        out_gate.view(R, H),
        out_fused,
        weights["to_out_norm.weight"],   # γ (fp32)
        weights["to_out_norm.bias"],     # β (fp32)
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M2,
        BLOCK_D=BLOCK_D2,
        num_warps=4,
    )
    del out_flat, out_gate, out

    # --------------------------------------------------------------
    # 6️⃣ Final linear projection back to `dim`
    # --------------------------------------------------------------
    final_half = F.linear(
        out_fused,
        weights["to_out.weight"].to(torch.float16),   # [dim, H]
    )   # fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
