@triton.jit
def norm_gate_proj_out_kernel(
    x_ptr,                # const *[rows, H] fp16 (out_mat)
    gate_ptr,             # const *[rows, H] fp16 (out_gate)
    norm_weight_ptr,      # const *[H] fp32
    norm_bias_ptr,        # const *[H] fp32
    final_weight_ptr,     # const *[H, dim] fp16 (transposed weight)
    out_ptr,              # mutable *[rows, dim] fp16
    rows: tl.int32,
    hidden: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < rows

    # First compute mean & var across hidden
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_R], dtype=tl.float32)

    for h_off in range(0, tl.cdiv(hidden, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < hidden

        ptr = x_ptr + offs_r[:, None] * hidden + offs_h[None, :]
        vals = tl.load(ptr,
                       mask=mask_r[:, None] & mask_h[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    mean = sum_val / hidden
    var = sum_sq / hidden - mean * mean
    rstd = tl.rsqrt(var + eps)

    # Second pass: accumulate final output over hidden blocks
    # accumulator for dim
    acc = tl.zeros([BLOCK_R, BLOCK_D], dtype=tl.float32)

    for h_off in range(0, tl.cdiv(hidden, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < hidden

        # load x slice
        x_ptr_off = x_ptr + offs_r[:, None] * hidden + offs_h[None, :]
        x_vals = tl.load(x_ptr_off,
                         mask=mask_r[:, None] & mask_h[None, :],
                         other=0.0).to(tl.float32)

        # normalize: (x - mean) * rstd * norm_weight + norm_bias
        w = tl.load(norm_weight_ptr + offs_h, mask=mask_h, other=0.0)
        b = tl.load(norm_bias_ptr + offs_h, mask=mask_h, other=0.0)
        norm = (x_vals - mean[:, None]) * rstd[:, None] * w[None, :] + b[None, :]  # fp32

        # load gate
        g_ptr = gate_ptr + offs_r[:, None] * hidden + offs_h[None, :]
        g = tl.load(g_ptr,
                    mask=mask_r[:, None] & mask_h[None, :],
                    other=0.0).to(tl.float32)

        norm_gate = norm * g  # fp32

        # load final_weight slice (hidden block x dim block)
        for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
            offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
            mask_d = offs_d < dim

            # weight pointer: final_weight_ptr + offs_h[:, None] * dim + offs_d[None, :]
            w_ptr = final_weight_ptr + offs_h[:, None] * dim + offs_d[None, :]
            w_block = tl.load(w_ptr,
                              mask=mask_h[:, None] & mask_d[None, :],
                              other=0.0).to(tl.float32)  # shape [BLOCK_H, BLOCK_D]

            # accumulate dot product: (BLOCK_R, BLOCK_H) dot (BLOCK_H, BLOCK_D) -> (BLOCK_R, BLOCK_D)
            # using matmul: sum over hidden block dimension
            # compute matmul manually via tl.dot? We can use tl.dot as:
            # partial = tl.dot(norm_gate, w_block)  # (BLOCK_R, BLOCK_D)
            # However tl.dot expects (M,K) x (K,N) with M and N compile-time known.
            # Here norm_gate shape (BLOCK_R, BLOCK_H) and w_block shape (BLOCK_H, BLOCK_D)
            # So we can compute partial = tl.dot(norm_gate, w_block) and accumulate.
            partial = tl.dot(norm_gate, w_block)  # result fp32 after conversion?
            acc += partial
        # end dim loop
    # end h loop

    # store result
    out_ptr_base = out_ptr + offs_r[:, None] * dim + offs_d[None, :]
    out_fp16 = acc.to(tl.float16)
    tl.store(out_ptr_base, out_fp16,
             mask=mask_r[:, None] & mask_d[None, :])
```

"""
Highly‑optimized Triton implementation of the *outgoing* TriMul operator.

Algorithm
---------
1️⃣  Fuse the first LayerNorm, the five linear projections (left/right
    projections + 3 gates) and the optional pair‑wise mask into a
    single Triton kernel:
        * Row‑wise mean / variance over the input dimension `dim`.
        * Normalise the row, apply the LayerNorm γ/β.
        * For every hidden block compute the five ϵ‑dot‑products
          (left, right, left_gate, right_gate, out_gate) using the
          normalised values.
        * Apply sigmoid to the three gate tensors, multiply the mask
          (if present) and write three tensors `left`, `right`,
          `out_gate` of shape `[B·N·N, hidden_dim]`.

2️⃣  The core multiplicative update:
        out[b,i,j] = Σₖ left[b,i,k] * right[b,j,k]
   implemented with a single batched GEMM (`torch.bmm`).

3️⃣  Fuse the second LayerNorm, the element‑wise multiplication with
    `out_gate` and the final linear projection back to `dim` into a
    second Triton kernel:
        * Row‑wise mean / variance over the hidden dimension.
        * Normalise, apply γ/β, multiply by `out_gate`.
        * For every output‑column block compute the mat‑vec product
          with the transposed `to_out.weight`.
   The kernel writes the final result directly in FP16; the caller
   casts it to FP32.

All heavy arithmetic is performed in FP16 with FP32 accumulation,
the two LayerNorm passes are fully fused, and the only extra
PyTorch operation is the batched GEMM of the `left`/`right` tensors.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Kernel 1: Row‑wise LayerNorm + fused projections + gates + mask
# ----------------------------------------------------------------------
@triton.jit
def norm_proj_gate_kernel(
    x_ptr,                     # const  *[rows, dim]   (float32)
    left_ptr,                  # mutable *[rows, hidden] (float16)
    right_ptr,                 # mutable *[rows, hidden] (float16)
    out_gate_ptr,              # mutable *[rows, hidden] (float16)
    mask_ptr,                  # const  *[rows]       (float16) – ignored if HAS_MASK=0
    norm_w_ptr,                # const  *[dim]        (float32) – γ
    norm_b_ptr,                # const  *[dim]        (float32) – β
    proj_w_ptr,                # const  *[dim, 5*hidden] (float16) – concatenated weight^T
    rows: tl.int32,
    dim: tl.int32,
    hidden: tl.int32,
    HAS_MASK: tl.constexpr,
    eps: tl.float32,
    BLOCK_R: tl.constexpr,
    BLOCK_D: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < rows

    # ------------------------------------------------------------------
    # 1️⃣  Compute mean / variance for each row (fp32)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_R], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < dim
        x = tl.load(x_ptr + offs_r[:, None] * dim + offs_d[None, :],
                    mask=mask_r[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)                # [R,D]
        sum_val += tl.sum(x, axis=1)
        sum_sq  += tl.sum(x * x, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------------------------------------------------------
    # 2️⃣  Compute the five dot‑products (left, right, left_gate,
    #      right_gate, out_gate) while iterating over the input dim.
    # ------------------------------------------------------------------
    # Allocate accumulators (fp32) for the five raw results.
    left_acc      = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
    right_acc     = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
    left_gate_acc = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
    right_gate_acc= tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
    out_gate_acc  = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)

    # Offsets of the five weight groups inside `proj_w_ptr`
    off_left      = 0
    off_right     = hidden
    off_left_gate = 2 * hidden
    off_right_gate= 3 * hidden
    off_out_gate  = 4 * hidden
    col_stride    = 5 * hidden                       # total number of columns

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < dim

        # -------- normalize the input slice (including LayerNorm γ/β) ----
        x = tl.load(x_ptr + offs_r[:, None] * dim + offs_d[None, :],
                    mask=mask_r[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)                # [R,D]

        w = tl.load(norm_w_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(norm_b_ptr + offs_d, mask=mask_d, other=0.0)   # β

        # (x - μ)·rstd·γ + β  →  fp32, then cast to fp16 for dot products
        norm = ((x - mean[:, None]) * rstd[:, None]) * w[None, :] + b[None, :]
        norm_fp16 = norm.to(tl.float16)                           # [R,D]

        # -------- load the five weight slices for this hidden block ----
        offs_h = tl.arange(0, BLOCK_H)
        mask_h = (offs_h + 0) < hidden                              # same mask for all groups

        # left weight slice
        col_l = off_left + offs_h
        w_left = tl.load(proj_w_ptr + offs_d[:, None] * col_stride + col_l[None, :],
                         mask=mask_d[:, None] & mask_h[None, :],
                         other=0.0).to(tl.float32)                # [D,Hblk]

        # right weight slice
        col_r = off_right + offs_h
        w_right = tl.load(proj_w_ptr + offs_d[:, None] * col_stride + col_r[None, :],
                          mask=mask_d[:, None] & mask_h[None, :],
                          other=0.0).to(tl.float32)

        # left_gate weight slice
        col_lg = off_left_gate + offs_h
        w_lgate = tl.load(proj_w_ptr + offs_d[:, None] * col_stride + col_lg[None, :],
                          mask=mask_d[:, None] & mask_h[None, :],
                          other=0.0).to(tl.float32)

        # right_gate weight slice
        col_rg = off_right_gate + offs_h
        w_rgate = tl.load(proj_w_ptr + offs_d[:, None] * col_stride + col_rg[None, :],
                          mask=mask_d[:, None] & mask_h[None, :],
                          other=0.0).to(tl.float32)

        # out_gate weight slice
        col_og = off_out_gate + offs_h
        w_og = tl.load(proj_w_ptr + offs_d[:, None] * col_stride + col_og[None, :],
                       mask=mask_d[:, None] & mask_h[None, :],
                       other=0.0).to(tl.float32)

        # -------- accumulate dot‑products (fp32) -----------------------
        left_acc      += tl.dot(norm_fp16, w_left)      # [R,Hblk] fp16 → fp32
        right_acc     += tl.dot(norm_fp16, w_right)
        left_gate_acc += tl.dot(norm_fp16, w_lgate)
        right_gate_acc+= tl.dot(norm_fp16, w_rgate)
        out_gate_acc  += tl.dot(norm_fp16, w_og)

    # ------------------------------------------------------------------
    # 3️⃣  Apply mask (if any) and sigmoid gates, then write results.
    # ------------------------------------------------------------------
    # sigmoid in fp32 for stability
    left_gate   = (1.0 / (1.0 + tl.exp(-left_gate_acc))) .to(tl.float16)
    right_gate  = (1.0 / (1.0 + tl.exp(-right_gate_acc))).to(tl.float16)
    out_gate    = (1.0 / (1.0 + tl.exp(-out_gate_acc)))  .to(tl.float16)

    # optional pair‑wise mask
    if HAS_MASK:
        m = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)          # [R]
        m = m[:, None]                                                # broadcast
    else:
        m = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    left  = left_acc .to(tl.float16) * m * left_gate
    right = right_acc.to(tl.float16) * m * right_gate

    # store
    left_ptrs  = left_ptr  + offs_r[:, None] * hidden + offs_h[None, :]
    right_ptrs = right_ptr + offs_r[:, None] * hidden + offs_h[None, :]
    og_ptrs    = out_gate_ptr + offs_r[:, None] * hidden + offs_h[None, :]

    tl.store(left_ptrs,  left,  mask=mask_r[:, None] & mask_h[None, :])
    tl.store(right_ptrs, right, mask=mask_r[:, None] & mask_h[None, :])
    tl.store(og_ptrs,   out_gate, mask=mask_r[:, None] & mask_h[None, :])

# ----------------------------------------------------------------------
# Kernel 2: second LayerNorm + out_gate multiplication + final projection
# ----------------------------------------------------------------------
@triton.jit
def norm_gate_proj_out_kernel(
    x_ptr,               # const  *[rows, hidden] (float16)  – result of left·right^T
    gate_ptr,            # const  *[rows, hidden] (float16)  – out_gate
    norm_w_ptr,          # const  *[hidden]       (float32) – γ (to_out_norm)
    norm_b_ptr,          # const  *[hidden]       (float32) – β (to_out_norm)
    proj_w_ptr,          # const  *[hidden, dim]  (float16) – (to_out.weight)^T
    out_ptr,             # mutable* [rows, dim]   (float16) – final result
    rows: tl.int32,
    hidden: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < rows

    # --------------------------------------------------------------
    # 1️⃣ mean / variance over the hidden dimension (fp32)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_R], dtype=tl.float32)

    for h_off in range(0, tl.cdiv(hidden, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < hidden

        x = tl.load(x_ptr + offs_r[:, None] * hidden + offs_h[None, :],
                    mask=mask_r[:, None] & mask_h[None, :],
                    other=0.0).to(tl.float32)                     # [R,Hblk]

        sum_val += tl.sum(x, axis=1)
        sum_sq  += tl.sum(x * x, axis=1)

    mean = sum_val / hidden
    var  = sum_sq / hidden - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------------------------------------------------------------
    # 2️⃣ for each output‑column block compute (norm·gate)·W
    # --------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        acc = tl.zeros([BLOCK_R, BLOCK_D], dtype=tl.float32)    # accumulator for this col‑block

        for h_off in range(0, tl.cdiv(hidden, BLOCK_H)):
            offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
            mask_h = offs_h < hidden

            # ----- load, normalize and apply to_out_norm γ/β -----
            x = tl.load(x_ptr + offs_r[:, None] * hidden + offs_h[None, :],
                        mask=mask_r[:, None] & mask_h[None, :],
                        other=0.0).to(tl.float32)

            w = tl.load(norm_w_ptr + offs_h, mask=mask_h, other=0.0)   # γ
            b = tl.load(norm_b_ptr + offs_h, mask=mask_h, other=0.0)   # β

            norm = ((x - mean[:, None]) * rstd[:, None]) * w[None, :] + b[None, :]   # fp32

            # ----- multiply by out_gate -----
            g = tl.load(gate_ptr + offs_r[:, None] * hidden + offs_h[None, :],
                        mask=mask_r[:, None] & mask_h[None, :],
                        other=0.0).to(tl.float32)

            norm_gate = norm * g   # fp32, shape [R, Hblk]

            # ----- load the final weight slice (Hblk × Dblk) -----
            w_final = tl.load(proj_w_ptr + offs_h[:, None] * dim + offs_d[None, :],
                              mask=mask_h[:, None] & mask_d[None, :],
                              other=0.0).to(tl.float32)                 # [Hblk,Dblk]

            # ----- accumulate dot product -----
            acc += tl.dot(norm_gate, w_final)   # (R,Hblk)·(Hblk,Dblk) → (R,Dblk)

        # ----- write the finished column block -----
        out_ptrs = out_ptr + offs_r[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, acc.to(tl.float16),
                 mask=mask_r[:, None] & mask_d[None, :])

def _choose_blocks(rows: int, dim: int, hidden: int):

# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim       = cfg["dim"]
    hidden    = cfg["hidden_dim"]
    eps       = 1e-5
    nomask    = cfg.get("nomask", True)
    device    = x.device

    B, N, _, _ = x.shape
    rows = B * N * N                     # flattened pair‑wise dimension

    # ------------------------------------------------------------------
    # 1️⃣  Fused LayerNorm + projections + gates
    # ------------------------------------------------------------------
    # flatten input
    x_flat = x.view(rows, dim).contiguous()                # [R, dim] float32
    left      = torch.empty(rows, hidden, dtype=torch.float16, device=device)
    right     = torch.empty(rows, hidden, dtype=torch.float16, device=device)
    out_gate  = torch.empty(rows, hidden, dtype=torch.float16, device=device)

        weights["left_proj.weight"].to(torch.float16).t(),
        weights["right_proj.weight"].to(torch.float16).t(),
        weights["left_gate.weight"].to(torch.float16).t(),
        weights["right_gate.weight"].to(torch.float16).t(),
        weights["out_gate.weight"].to(torch.float16).t(),
    ], dim=1).contiguous()                                 # [dim, 5*hidden]

    if nomask:
        dummy_mask = torch.empty(0, dtype=torch.float16, device=device)
        HAS_MASK   = 0
        mask_ptr   = dummy_mask
    else:
        mask_fp16 = mask.to(torch.float16).reshape(rows).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_fp16

    BLOCK_R, BLOCK_D, BLOCK_H = _choose_blocks(rows, dim, hidden)

    grid = (triton.cdiv(rows, BLOCK_R),)
    norm_proj_gate_kernel[grid](
        x_flat,
        left,
        right,
        out_gate,
        mask_ptr,
        weights["norm.weight"],      # γ  (float32)
        weights["norm.bias"],        # β  (float32)
        proj_w,
        rows,
        dim,
        hidden,
        HAS_MASK=HAS_MASK,
        eps=eps,
        BLOCK_R=BLOCK_R,
        BLOCK_D=BLOCK_D,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del x_flat, proj_w

    # ------------------------------------------------------------------
    # 2️⃣  Batched GEMM: left @ rightᵀ   (shape [B*hidden, N, N])
    # ------------------------------------------------------------------
    # reshape to [B, N, N, hidden] → permute → batch
    left_mat  = left.view(B, N, N, hidden).permute(0, 3, 1, 2).contiguous()
    right_mat = right.view(B, N, N, hidden).permute(0, 3, 1, 2).contiguous()

    left_mat  = left_mat.view(B * hidden, N, N)      # [B·H, N, N]
    right_mat = right_mat.view(B * hidden, N, N)     # [B·H, N, N]

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N]

    # reshape back to flattened [rows, hidden]
    out_flat = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()
    out_flat = out_flat.view(rows, hidden)          # [R, hidden]

    del left_mat, right_mat, left, right, out_mat

    # ------------------------------------------------------------------
    # 3️⃣  Second LayerNorm + out_gate * final linear projection
    # ------------------------------------------------------------------
    final = torch.empty(rows, dim, dtype=torch.float16, device=device)

    # to_out.norm parameters (float32) and weight (transposed fp16)
    to_out_norm_w = weights["to_out_norm.weight"]   # (hidden,)
    to_out_norm_b = weights["to_out_norm.bias"]     # (hidden,)
    to_out_w_T    = weights["to_out.weight"].to(torch.float16).t().contiguous()  # [hidden, dim]

    BLOCK_R2, BLOCK_H2, BLOCK_D2 = _choose_blocks(rows, hidden, dim)  # note swapped dims
    grid2 = (triton.cdiv(rows, BLOCK_R2),)

    norm_gate_proj_out_kernel[grid2](
        out_flat,
        out_gate,
        to_out_norm_w,
        to_out_norm_b,
        to_out_w_T,
        final,
        rows,
        hidden,
        dim,
        eps=eps,
        BLOCK_R=BLOCK_R2,
        BLOCK_H=BLOCK_H2,
        BLOCK_D=BLOCK_D2,
        num_warps=8,
    )
    del out_flat, out_gate, to_out_w_T

    # ------------------------------------------------------------------
    # 4️⃣  Cast back to float32 and reshape to [B, N, N, dim]
    # ------------------------------------------------------------------
    out = final.view(B, N, N, dim).to(torch.float32)
    return out
```
