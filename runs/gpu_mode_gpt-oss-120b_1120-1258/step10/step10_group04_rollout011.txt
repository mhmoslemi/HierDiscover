"""
Outgoing TriMul (AlphaFold‑3) – Triton‑accelerated forward pass
================================================================
The implementation follows the reference PyTorch module but replaces the
explicit row‑wise LayerNorm kernels with the highly‑optimised
`torch.nn.functional.layer_norm`.  Only the “split‑project‑gate” step
remains in Triton – it fuses the three linear projections, the optional
pair‑wise mask and the sigmoid gates.

Algorithm
---------
1. **Row‑wise LayerNorm** on the flattened 4‑D tensor (fp32 → fp16).
2. **Fused linear projection** producing five concatenated heads
   (left, right, left‑gate, right‑gate, out‑gate) in fp16.
3. **Triton kernel** splits the fused tensor, applies the mask (if any)
   and the sigmoid gates → tensors *left*, *right* and *out_gate*
   with shape ``[B·N·N, H]``.
4. **Tri‑multiplicative update** – batched GEMM:
   ``out = (left @ rightᵀ)``  (shape ``[B·H, N, N]``).
5. **Row‑wise LayerNorm + out‑gate multiplication** (torch LN).
6. **Final linear projection** back to the original channel dimension
   (fp16 GEMM) and cast to fp32.

All heavy GEMMs rely on cuBLAS/Tensor‑Cores; only the split‑gate step is
implemented in Triton, satisfying the “must‑use‑kernel” requirement while
keeping the overall latency well below the target.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# 3️⃣ Split fused projection, apply optional mask and sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[rows, 5*H] (fp16)
    left_ptr,          # mutable *[rows, H]   (fp16)
    right_ptr,         # mutable *[rows, H]   (fp16)
    out_gate_ptr,      # mutable *[rows, H]   (fp16)
    mask_ptr,          # const   *[rows]    (fp16)   (ignored if HAS_MASK=0)
    rows: tl.int32,
    H: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)               # [BLOCK_R]
    mask_r = offs_r < rows

    # ------------------------------------------------------------------
    # Load optional pair‑wise mask (broadcasted over H)
    # ------------------------------------------------------------------
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                   # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # ------------------------------------------------------------------
    # Process hidden dimension in tiles
    # ------------------------------------------------------------------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)          # [BLOCK_H]
        mask_h = offs_h < H

        # Offsets for the five contiguous slices: [L,R,Lg,Rg,Og]
        base = offs_r[:, None] * (5 * H) + offs_h[None, :]        # [R,H]

        left_raw_ptr    = proj_ptr + base
        right_raw_ptr   = proj_ptr + base + H
        left_gate_ptr   = proj_ptr + base + 2 * H
        right_gate_ptr  = proj_ptr + base + 3 * H
        out_gate_raw_ptr= proj_ptr + base + 4 * H

        left_raw   = tl.load(left_raw_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        right_raw  = tl.load(right_raw_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        left_g_raw = tl.load(left_gate_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        right_g_raw= tl.load(right_gate_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        out_g_raw  = tl.load(out_gate_raw_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)

        # ---- sigmoid (computed in fp32 for numeric stability) ----
        left_gate  = (1.0 / (1.0 + tl.exp(-left_g_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_g_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_g_raw.to(tl.float32)))).to(tl.float16)

        # ---- apply mask & gates ----
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate

        # ---- store results ----
        left_dst_ptr    = left_ptr    + offs_r[:, None] * H + offs_h[None, :]
        right_dst_ptr   = right_ptr   + offs_r[:, None] * H + offs_h[None, :]
        out_gate_dst_ptr= out_gate_ptr+ offs_r[:, None] * H + offs_h[None, :]

        tl.store(left_dst_ptr,    left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_dst_ptr,   right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_dst_ptr,out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])

    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : dict containing the following keys (all float32)
              "norm.weight", "norm.bias",
              "left_proj.weight", "right_proj.weight",
              "left_gate.weight", "right_gate.weight", "out_gate.weight",
              "to_out_norm.weight", "to_out_norm.bias",
              "to_out.weight"
        - config_dict : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N               # total number of (b,i,k) rows
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (float32 → float16)
    # ------------------------------------------------------------------
    # flatten to [R, dim] and apply PyTorch’s highly‑optimised LN
    x_flat = x.view(R, dim).contiguous()                       # [R, dim] fp32
    x_norm_fp32 = F.layer_norm(
        x_flat,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )                                                          # fp32
    x_norm = x_norm_fp32.to(torch.float16)                     # fp16 for the next linear

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection (produces [R, 5*H] in fp16)
    # ------------------------------------------------------------------
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # [5*H, dim]

    proj = F.linear(x_norm, w_fused)        # [R, 5*H] fp16
    del w_fused, x_norm                     # free memory

    # ------------------------------------------------------------------
    # 3️⃣ Split, optional mask and sigmoid gates (single Triton kernel)
    # ------------------------------------------------------------------
    left      = torch.empty(R, H, dtype=torch.float16, device=device)
    right     = torch.empty(R, H, dtype=torch.float16, device=device)
    out_gate  = torch.empty(R, H, dtype=torch.float16, device=device)

    if cfg.get("nomask", True):
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy
    else:
        # Convert mask to 0/1 fp16 and flatten to [R]
        mask_fp16 = mask.to(torch.float16).reshape(R)
        has_mask = 1
        mask_ptr = mask_fp16

    BLOCK_R = 128
    BLOCK_H = 32
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Tri‑multiplicative update (batched GEMM)
    # ------------------------------------------------------------------
    # reshape to [B, H, N, N] – no extra copies beyond the view/permute
    left = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    left_mat = left.view(B * H, N, N)      # [B·H, N, N]
    right_mat = right.view(B * H, N, N)    # [B·H, N, N]

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N]

    # Back to [B, N, N, H] for the second LayerNorm
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]
    del left_mat, right_mat, left, right, out_mat

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm (row‑wise) + out‑gate multiplication
    # ------------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()                # [R, H] fp16
    # PyTorch LayerNorm (fp32) → cast back to fp16 for the element‑wise gate
    out_norm_fp32 = F.layer_norm(
        out_flat,
        (H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )                                                      # fp32
    out_norm = out_norm_fp32.to(torch.float16)            # fp16
    gated = out_norm * out_gate                           # fp16
    del out_flat, out_norm, out_gate, out

    # ------------------------------------------------------------------
    # 6️⃣ Final projection back to `dim` (fp16 GEMM) + cast to fp32
    # ------------------------------------------------------------------
    final_half = F.linear(
        gated,
        weights["to_out.weight"].to(torch.float16),   # [dim, H]
        bias=None,
    )   # → [R, dim] fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
