"""

Algorithm
---------
1️⃣  Row‑wise LayerNorm on the 4‑D input (B·N·N × dim) – fp32 → fp16.
2️⃣  Fuse the five linear maps (left/right projections + three gates) into a
    single GEMM:
        X_norm @ W_fusedᵀ  →  (R, 5·H)   with R = B·N·N.
3️⃣  Triton kernel splits the fused result, applies the (optional) pair‑wise mask,
    evaluates the sigmoid gates and writes three tensors:
        left   = (proj_left  * mask) * sigmoid(proj_left_gate)
        right  = (proj_right * mask) * sigmoid(proj_right_gate)
        out_gate = sigmoid(proj_out_gate)
   All arithmetic is in fp16 (sigmoid evaluated in fp32 for stability).
4️⃣  Batched half‑precision GEMM over the hidden dimension:
        out[b·H+i] = left[b·H+i] @ right[b·H+i]ᵀ
   (implemented with `torch.bmm`, which uses Tensor‑Core cuBLAS kernels).
5️⃣  Second LayerNorm + element‑wise multiplication with `out_gate`
   (single Triton kernel).
6️⃣  Final linear projection back to the original `dim`
   (fp16 → fp32 output).

Only the three heavy kernels are written in Triton; all matrix‑multiplications
are delegated to cuBLAS for peak performance on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Kernel 1: row‑wise LayerNorm (float32 → float16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   *[rows, dim]   (float32)
    y_ptr,          # mutable *[rows, dim]   (float16)
    w_ptr,          # const   *[dim]        (float32)   gamma
    b_ptr,          # const   *[dim]        (float32)   beta
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ------------------------------------------------------------------
    # Compute mean & variance (fp32)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------------------------------------------------------
    # Write normalized result (γ·(x‑μ)·rstd + β) as fp16
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]       # fp32
        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)  # γ (fp32)
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)  # β (fp32)

        out = norm * w[None, :] + b[None, :]                # fp32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

# Kernel 2: split fused projection, apply mask & sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[rows, 5*H]    (float16)
    left_ptr,          # mutable *[rows, H]    (float16)
    right_ptr,         # mutable *[rows, H]    (float16)
    out_gate_ptr,      # mutable *[rows, H]    (float16)
    mask_ptr,          # const   *[rows]       (float16)  (ignored if HAS_MASK=0)
    rows: tl.int32,
    H: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    For each row i (i indexes B·N·N):
        left   = (proj_left  * mask[i]) * sigmoid(proj_left_gate)
        right  = (proj_right * mask[i]) * sigmoid(proj_right_gate)
        out_gate = sigmoid(proj_out_gate)

    All arithmetic in fp16 except sigmoid (computed in fp32 for stability).
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)               # [BLOCK_R]
    mask_r = offs_r < rows

    # Load scalar mask per row if present
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                    # [BLOCK_R, 1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)          # [BLOCK_H]
        mask_h = offs_h < H

        # Layout: [left, right, left_gate, right_gate, out_gate]
        base = offs_r[:, None] * (5 * H) + offs_h[None, :]        # offset base

        left_raw_ptr     = proj_ptr + base
        right_raw_ptr    = proj_ptr + base + H
        left_gate_ptr    = proj_ptr + base + 2 * H
        right_gate_ptr   = proj_ptr + base + 3 * H
        out_gate_raw_ptr = proj_ptr + base + 4 * H

        left_raw      = tl.load(left_raw_ptr,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_raw     = tl.load(right_raw_ptr,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        left_gate_raw = tl.load(left_gate_ptr,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_gate_raw = tl.load(right_gate_ptr,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        out_gate_raw   = tl.load(out_gate_raw_ptr,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)

        # sigmoid in fp32 (stable)
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate

        left_dst_ptr     = left_ptr     + offs_r[:, None] * H + offs_h[None, :]
        right_dst_ptr    = right_ptr    + offs_r[:, None] * H + offs_h[None, :]
        out_gate_dst_ptr = out_gate_ptr + offs_r[:, None] * H + offs_h[None, :]

        tl.store(left_dst_ptr,     left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_dst_ptr,    right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_dst_ptr, out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])

# Kernel 3: second LayerNorm + element‑wise gate multiplication
# ----------------------------------------------------------------------
@triton.jit
def layernorm_mul_kernel(
    x_ptr,          # const   *[rows, dim]   (float16)
    gate_ptr,       # const   *[rows, dim]   (float16)   (out_gate)
    y_ptr,          # mutable *[rows, dim]   (float16)
    w_ptr,          # const   *[dim]        (float32)   γ
    b_ptr,          # const   *[dim]        (float32)   β
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """
    Row‑wise LayerNorm on `x` (fp16) followed by element‑wise multiplication
    with `gate`. Result stored as fp16.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)                 # [BLOCK_M]
    mask_m = offs_m < rows

    # ------------------------------------------------------------------
    # Compute mean & variance (fp32)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)       # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------------------------------------------------------
    # Normalize, apply γ/β, multiply with gate, write out
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        x_val = tl.load(x_ptr + offs_m[:, None] * dim + offs_d[None, :],
                        mask=mask_m[:, None] & mask_d[None, :],

        # (x‑μ)·rstd
        norm = (x_val - mean[:, None]) * rstd[:, None]           # fp32

        # γ, β (fp32)
        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)

        norm = norm * w[None, :] + b[None, :]                    # fp32

        # gate (fp16 → fp32)
        gate_val = tl.load(gate_ptr + offs_m[:, None] * dim + offs_d[None, :],
                           mask=mask_m[:, None] & mask_d[None, :],
                           other=0.0).to(tl.float32)

        out = norm * gate_val                                    # fp32
        out_fp16 = out.to(tl.float16)

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

        x_norm,
        w["norm.weight"],          # γ (float32)
        w["norm.bias"],            # β (float32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,

        num_stages=2,
    )

            w["left_gate.weight"],

        ],
        dim=0,

    BLOCK_H = 32       # hidden dim chunk
    grid_split = (triton.cdiv(R, BLOCK_R),)

        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,

        BLOCK_H=BLOCK_H,
        num_warps=4,
        num_stages=1,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣ Tri‑multiplicative update (batched GEMM over hidden dim)
    # --------------------------------------------------------------
    # reshape to [B, H, N, N] for efficient batched GEMM
    left  = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    left_mat  = left.view(B * H, N, N)    # [B·H, N, N]
    right_mat = right.view(B * H, N, N)   # [B·H, N, N]

    # Batched GEMM: left @ rightᵀ  (half‑precision, Tensor‑Core)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N]

    # back‑to [B, N, N, H] (fp16) – no extra copy beyond view/permute
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm + multiply with out_gate (fused kernel)
    # --------------------------------------------------------------
    out_flat   = out.view(R, H).contiguous()
    out_fused  = torch.empty_like(out_flat, dtype=torch.float16, device=device)

    grid2 = (triton.cdiv(R, BLOCK_M),)
    layernorm_mul_kernel[grid2](
        out_flat,
        out_gate.view(R, H),               # gate pointer
        out_fused,
        w["to_out_norm.weight"],           # γ (float32)
        w["to_out_norm.bias"],             # β (float32)
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
        num_stages=2,
    )
    del out_flat, out_gate, out

    # --------------------------------------------------------------
    # 6️⃣ Final projection back to `dim`
    # --------------------------------------------------------------
    final_half = F.linear(
        out_fused,
        w["to_out.weight"].to(torch.float16),   # [dim, H]
    )   # fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
