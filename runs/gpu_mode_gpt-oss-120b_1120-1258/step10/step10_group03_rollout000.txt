"""
TriMul (outgoing) – fast forward pass

Algorithm (all tensor dimensions are given in the docstring of `custom_kernel`)
---------------------------------------------------------------------------
1️⃣ Row‑wise LayerNorm on the 4‑D input (B·N·N × dim).  Implemented with
   cuDNN via `torch.nn.functional.layer_norm` (fp16).

2️⃣ Fuse the five linear maps (left/right projections + three gates) into one
   GEMM:
        X_norm @ W_fusedᵀ → (R, 5·H)   where  R = B·N·N
   `W_fused` = [left_proj; right_proj; left_gate; right_gate; out_gate] (fp16).

3️⃣ One Triton kernel splits the fused result, applies the optional pair‑wise
   mask, evaluates the three sigmoid gates and writes three tensors:
        left  = (proj_left  * mask) * sigmoid(left_gate)
        right = (proj_right * mask) * sigmoid(right_gate)
        out_gate = sigmoid(out_gate)

4️⃣ Batched half‑precision GEMM over the hidden dimension:
        out[b·H+i] = left_mat[b·H+i] @ right_mat[b·H+i]ᵀ
   performed with `torch.bmm` (cuBLAS Tensor‑cores).

5️⃣ Second LayerNorm (cuDNN) on the result and element‑wise multiplication by
   `out_gate`.

6️⃣ Final linear projection back to `dim` (fp16 → fp32).

Only step 3 is written in Triton; all heavy matrix‑multiplications use
cuBLAS/ cuDNN which gives near‑peak performance on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: split fused projection, apply mask and three sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[rows, 5*H]    (fp16)
    left_ptr,          # mutable *[rows, H]    (fp16)
    right_ptr,         # mutable *[rows, H]    (fp16)
    out_gate_ptr,      # mutable *[rows, H]    (fp16)
    mask_ptr,          # const   *[rows]       (fp16) – ignored if HAS_MASK=0
    rows: tl.int32,
    H: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    For each row:
        left  = (proj_left  * mask) * sigmoid(proj_left_gate)
        right = (proj_right * mask) * sigmoid(proj_right_gate)
        out_gate = sigmoid(proj_out_gate)
    All arithmetic is in fp16, sigmoid evaluated in fp32 for numeric stability.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < rows

    # Load mask if present (broadcast later)
    if HAS_MASK:
        m = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        m = m[:, None]                                           # [BLOCK_R,1]
    else:
        m = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # Iterate over hidden dimension in chunks
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # Offsets for the five consecutive slices inside the fused projection
        # Layout: [left, right, left_gate, right_gate, out_gate] each of size H
        base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw_ptr    = proj_ptr + base
        right_raw_ptr   = proj_ptr + base + H
        left_gate_ptr   = proj_ptr + base + 2 * H
        right_gate_ptr  = proj_ptr + base + 3 * H
        out_gate_raw_ptr= proj_ptr + base + 4 * H

        left_raw   = tl.load(left_raw_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        right_raw  = tl.load(right_raw_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        left_g_raw = tl.load(left_gate_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        right_g_raw= tl.load(right_gate_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        out_g_raw  = tl.load(out_gate_raw_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)

        # Sigmoid in fp32 → back to fp16
        left_gate   = (1.0 / (1.0 + tl.exp(-left_g_raw.to(tl.float32)))).to(tl.float16)
        right_gate  = (1.0 / (1.0 + tl.exp(-right_g_raw.to(tl.float32)))).to(tl.float16)
        out_gate    = (1.0 / (1.0 + tl.exp(-out_g_raw.to(tl.float32)))).to(tl.float16)

        # Apply mask and gates
        left  = left_raw  * m * left_gate
        right = right_raw * m * right_gate

        # Store results
        left_dst_ptr     = left_ptr    + offs_r[:, None] * H + offs_h[None, :]
        right_dst_ptr    = right_ptr   + offs_r[:, None] * H + offs_h[None, :]
        out_gate_dst_ptr = out_gate_ptr+ offs_r[:, None] * H + offs_h[None, :]

        tl.store(left_dst_ptr,     left,  mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_dst_ptr,    right, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_dst_ptr,out_gate, mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Helper: choose block sizes for the split kernel
# ----------------------------------------------------------------------
def _choose_split_block(rows: int, hidden: int):
    """Pick block sizes that give good occupancy on H100."""
    BLOCK_R = 128                     # rows per program – fits nicely in 8 warps
    BLOCK_H = 32 if hidden >= 32 else hidden
    return BLOCK_R, BLOCK_H


# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : dict with the following keys (all float32):
            "norm.weight", "norm.bias",
            "left_proj.weight", "right_proj.weight",
            "left_gate.weight", "right_gate.weight", "out_gate.weight",
            "to_out_norm.weight", "to_out_norm.bias",
            "to_out.weight"
        - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # 0️⃣ Unpack arguments
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    device = x.device
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    nomask = cfg.get("nomask", True)

    B, N, _, _ = x.shape
    R = B * N * N            # total number of rows for the row‑wise ops
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (cuDNN) – fp16
    # ------------------------------------------------------------------
    # Cast weights to fp16 for the LN
    norm_w = weights["norm.weight"].to(torch.float16, non_blocking=True)
    norm_b = weights["norm.bias"].to(torch.float16, non_blocking=True)

    x_fp16 = x.view(R, dim).to(torch.float16, non_blocking=True)   # [R, dim]
    x_norm = F.layer_norm(
        x_fp16,
        (dim,),
        weight=norm_w,
        bias=norm_b,
        eps=eps,
    )   # fp16, shape [R, dim]

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear: left/right projections + all gates (5·H)
    # ------------------------------------------------------------------
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16, non_blocking=True)   # [5*H, dim]

    # (R, dim) @ (5*H, dim).T → (R, 5*H)   (half‑precision GEMM)
    proj = F.linear(x_norm, w_fused)  # fp16
    # No longer need x_norm or w_fused
    del x_norm, w_fused

    # ------------------------------------------------------------------
    # 3️⃣ Split, apply mask and sigmoid gates (single Triton kernel)
    # ------------------------------------------------------------------
    left   = torch.empty(R, H, dtype=torch.float16, device=device)
    right  = torch.empty(R, H, dtype=torch.float16, device=device)
    out_gate = torch.empty(R, H, dtype=torch.float16, device=device)

    if nomask:
        dummy_mask = torch.empty(0, dtype=torch.float16, device=device)
        HAS_MASK = 0
        mask_ptr = dummy_mask
    else:
        # mask shape [B, N, N] → flatten and cast to fp16 (0/1)
        mask_fp16 = mask.to(torch.float16, non_blocking=True).reshape(R)
        HAS_MASK = 1
        mask_ptr = mask_fp16

    BLOCK_R, BLOCK_H = _choose_split_block(R, H)
    grid = (triton.cdiv(R, BLOCK_R),)

    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Tri‑multiplicative update (batched GEMM over hidden dimension)
    # ------------------------------------------------------------------
    # Reshape to [B, H, N, N] for a single batch GEMM per head
    left = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    left_mat = left.view(B * H, N, N)   # [B*H, N, N]
    right_mat = right.view(B * H, N, N) # [B*H, N, N]

    # (B*H, N, N) @ (B*H, N, N)ᵀ → (B*H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Back‑to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm (cuDNN) + multiplication by out_gate
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16, non_blocking=True)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16, non_blocking=True)

    out_flat = out.view(R, H).contiguous()                # [R, H] fp16
    out_gate_flat = out_gate.view(R, H).contiguous()      # [R, H] fp16

    # cuDNN LayerNorm on the hidden dimension
    out_norm = F.layer_norm(
        out_flat,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )   # fp16

    # Element‑wise gating
    out_norm = out_norm * out_gate_flat   # fp16

    # ------------------------------------------------------------------
    # 6️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(torch.float16, non_blocking=True)  # [dim, H]

    final_half = F.linear(out_norm, to_out_w)   # (R, dim) fp16
    final = final_half.view(B, N, N, dim).to(torch.float32)  # cast back to fp32

    return final
