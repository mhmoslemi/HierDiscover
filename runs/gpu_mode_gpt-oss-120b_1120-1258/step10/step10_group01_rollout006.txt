"""
TriMul (outgoing) – fast Triton+PyTorch implementation
=====================================================

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the 4‑D input (B·N·N × dim).  We reuse the
   cuDNN LayerNorm kernel via ``torch.nn.functional.layer_norm`` (fp16
   input, fp32 statistics, fp16 output).

2️⃣ Fuse the five linear maps (left/right projections + three gates) into a
   single GEMM:
        X_norm @ W_fused.T   →   (R, 5·H)   with R = B·N·N.
   The fused weight concatenates the five weight matrices (fp16).

3️⃣ One Triton kernel splits the fused result, applies the optional pair‑wise
   mask and the sigmoid gates, and writes three tensors:
        left      = (proj_left      * mask) * sigmoid(proj_left_gate)
        right     = (proj_right     * mask) * sigmoid(proj_right_gate)
        out_gate  = sigmoid(proj_out_gate)
   All arithmetic is fp16, the sigmoid is evaluated in fp32 for stability.

4️⃣ Batched GEMM over the hidden dimension:
        out[b·H+i] = left[b·H+i] @ right[b·H+i].T
   Implemented with ``torch.bmm`` (Tensor‑Core cuBLAS) on tensors of shape
   (B·H, N, N).

5️⃣ Second LayerNorm + element‑wise multiplication with ``out_gate``.
   Again we use cuDNN LayerNorm (fp16) and a simple ``*``.

6️⃣ Final projection back to the original model dimension (fp16 GEMM) and
   cast to float32.

Only the split‑&‑gate kernel is written in Triton; all heavy matrix‑multiplications
are delegated to cuBLAS/cuDNN, which gives near‑peak H100 performance.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel 1: split fused projection, apply mask & sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[rows, 5*H]  (fp16)
    left_ptr,          # mutable *[rows, H]   (fp16)
    right_ptr,         # mutable *[rows, H]   (fp16)
    out_gate_ptr,      # mutable *[rows, H]   (fp16)
    mask_ptr,          # const   *[rows]    (fp16)  (ignored if HAS_MASK=0)
    rows: tl.int32,
    H: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,   # rows per program
    BLOCK_H: tl.constexpr,   # hidden dim chunk
):
    """
    For each row i (i indexes B·N·N):
        left  = (proj_left  * mask[i]) * sigmoid(proj_left_gate)
        right = (proj_right * mask[i]) * sigmoid(proj_right_gate)
        out_gate = sigmoid(proj_out_gate)

    Layout of proj (contiguous, row‑major):
        [ left_proj | right_proj | left_gate | right_gate | out_gate ]

    All arithmetic in fp16, sigmoid computed in fp32.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < rows

    # Load mask scalar per row if present
    if HAS_MASK:
        # mask is stored as fp16 scalar per row (0 or 1)
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)
        mask_val = mask_val[:, None]                    # broadcast over H
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # Iterate over hidden dimension in chunks
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)    # [BLOCK_H]
        mask_h = offs_h < H

        # Base pointer to the beginning of the row in `proj`
        base = offs_r[:, None] * (5 * H) + offs_h[None, :]  # [BLOCK_R, BLOCK_H]

        # Offsets for the five segments
        left_raw_ptr    = proj_ptr + base
        right_raw_ptr   = proj_ptr + base + H
        left_gate_ptr   = proj_ptr + base + 2 * H
        right_gate_ptr  = proj_ptr + base + 3 * H
        out_gate_raw_ptr= proj_ptr + base + 4 * H

        # Load raw values
        left_raw    = tl.load(left_raw_ptr,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0).to(tl.float16)
        right_raw   = tl.load(right_raw_ptr,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0).to(tl.float16)
        left_gate_raw = tl.load(left_gate_ptr,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0).to(tl.float16)
        right_gate_raw = tl.load(right_gate_ptr,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0).to(tl.float16)
        out_gate_raw = tl.load(out_gate_raw_ptr,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0).to(tl.float16)

        # Sigmoid (stable, compute in fp32)
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # Apply mask and gates
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate

        # Store results
        left_dst_ptr     = left_ptr     + offs_r[:, None] * H + offs_h[None, :]
        right_dst_ptr    = right_ptr    + offs_r[:, None] * H + offs_h[None, :]
        out_gate_dst_ptr = out_gate_ptr + offs_r[:, None] * H + offs_h[None, :]

        tl.store(left_dst_ptr,     left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_dst_ptr,    right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_dst_ptr, out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])

# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
            - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
            - weights_dict : mapping with all Linear/LayerNorm parameters
                             (float32 tensors, bias=False for Linear)
            - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim        = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]          # H
    eps        = 1e-5
    device     = x.device

    B, N, _, _ = x.shape
    R = B * N * N                         # total rows for row‑wise ops
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (fp32 → fp16) using cuDNN
    # ------------------------------------------------------------------
    # Flatten to (R, dim) and cast to fp16
    x_flat_fp16 = x.view(R, dim).contiguous().to(torch.float16)
    # cuDNN LayerNorm: returns fp16 (statistics in fp32 internally)
    x_norm = F.layer_norm(
        x_flat_fp16,
        (dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    )   # shape [R, dim], dtype fp16

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear: left/right proj + three gates → (R, 5·H)
    # ------------------------------------------------------------------
    fused_weight = torch.cat(
        [
            w["left_proj.weight"],   # (H, dim)
            w["right_proj.weight"],  # (H, dim)
            w["left_gate.weight"],   # (H, dim)
            w["right_gate.weight"],  # (H, dim)
            w["out_gate.weight"],    # (H, dim)
        ],
        dim=0,
    ).to(torch.float16)               # (5·H, dim)

    # (R, dim) @ (5·H, dim).T  →  (R, 5·H)   – half‑precision Tensor‑Core GEMM
    proj = F.linear(x_norm, fused_weight)   # fp16
    del x_norm, fused_weight                # free early

    # ------------------------------------------------------------------
    # 3️⃣ Split → mask & sigmoid gates (single Triton kernel)
    # ------------------------------------------------------------------
    left      = torch.empty(R, H, dtype=torch.float16, device=device)
    right     = torch.empty(R, H, dtype=torch.float16, device=device)
    out_gate  = torch.empty(R, H, dtype=torch.float16, device=device)

    if cfg.get("nomask", True) or mask is None:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy
    else:
        # mask values are {0,1} (float).  Convert to fp16 and flatten.
        mask_fp16 = mask.to(torch.float16).reshape(R)
        has_mask = 1
        mask_ptr = mask_fp16

    BLOCK_R = 128   # rows per program
    BLOCK_H = 32    # hidden dim chunk (fits 128‑dim hidden nicely)
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Tri‑multiplicative update: batched GEMM over hidden dimension
    # ------------------------------------------------------------------
    # Reshape to [B, H, N, N] for efficient batched GEMM
    left  = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    # Merge batch and hidden dims → (B·H, N, N)
    left_mat  = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)

    # Batched half‑precision matrix multiply (Tensor‑Core cuBLAS)
    # out_mat[i] = left_mat[i] @ right_mat[i].T
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Restore original layout: [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    del left_mat, right_mat, left, right

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm + element‑wise multiplication with out_gate
    # ------------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()       # (R, H)
    # cuDNN LayerNorm on hidden dimension
    out_norm = F.layer_norm(
        out_flat,
        (H,),
        weight=w["to_out_norm.weight"],
        bias=w["to_out_norm.bias"],
        eps=eps,
    )                                            # fp16
    # Element‑wise multiply with the gate (already fp16)
    out_norm = out_norm * out_gate
    del out, out_gate

    # ------------------------------------------------------------------
    # 6️⃣ Final linear projection back to `dim` and cast to fp32
    # ------------------------------------------------------------------
    final = F.linear(
        out_norm,
        w["to_out.weight"].to(torch.float16),   # (dim, H)
    )   # fp16

    final_fp32 = final.view(B, N, N, dim).to(torch.float32)
    return final_fp32
