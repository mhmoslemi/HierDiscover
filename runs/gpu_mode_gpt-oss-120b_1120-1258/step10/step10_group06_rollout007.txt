1️⃣ Row‑wise LayerNorm on the input tensor (fp32 → fp16) using the
   highly‑optimised PyTorch `layer_norm` implementation.
2️⃣ Fuse the five Linear layers (left/right projections + three gates) into a
   single GEMM:  X_norm @ W_fusedᵀ  →  (R, 5·H) where R = B·N·N and H = hidden_dim.
3️⃣ Triton kernel `split_fuse_gate_kernel`:
   * Splits the fused result into the three needed tensors.
   * Applies the optional pair‑wise mask.
   * Evaluates sigmoid gates (in fp32 for stability) and writes:
        left  = (proj_left  * mask) * sigmoid(proj_left_gate)
        right = (proj_right * mask) * sigmoid(proj_right_gate)
        out_gate = sigmoid(proj_out_gate)
4️⃣ Batched half‑precision GEMM (`torch.bmm`) over the hidden dimension:
        out[b·H+i] = left[b·H+i] @ right[b·H+i]ᵀ
5️⃣ Second LayerNorm on the GEMM result (fp16) using PyTorch, then element‑wise
   multiplication with `out_gate`.
6️⃣ Final linear projection back to `dim` (fp16 → fp32).

All heavy linear algebra is done with cuBLAS (Tensor‑Cores).  Only the
splitting / gating step is custom Triton code – this satisfies the
“must have a kernel” requirement while keeping launch overhead minimal.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: split fused projection, apply mask, evaluate sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[rows, 5*H]    (fp16)
    left_ptr,          # mutable *[rows, H]     (fp16)
    right_ptr,         # mutable *[rows, H]     (fp16)
    out_gate_ptr,      # mutable *[rows, H]     (fp16)
    mask_ptr,          # const   *[rows]       (fp16) – ignored if HAS_MASK=0
    rows: tl.int32,
    H: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,   # rows per program
    BLOCK_H: tl.constexpr,   # hidden dim per iteration
):
    """
    For each row   r ∈ [0, rows):
        left  = (proj_left  * mask) * sigmoid(proj_left_gate)
        right = (proj_right * mask) * sigmoid(proj_right_gate)
        out_gate = sigmoid(proj_out_gate)
    All computation is in fp16 except the sigmoid (computed in fp32 for
    numerical stability).
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)        # [BLOCK_R]
    mask_r = offs_r < rows

    # ---- optional per‑row mask (0/1) ---------------------------------
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)  # [BLOCK_R]
        mask_val = mask_val[:, None]   # broadcast over hidden dim
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # ---- iterate over hidden dimension --------------------------------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # layout inside proj: [left, right, left_gate, right_gate, out_gate]
        base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        p_left      = proj_ptr + base
        p_right     = proj_ptr + base + H
        p_left_gate = proj_ptr + base + 2 * H
        p_right_gate= proj_ptr + base + 3 * H
        p_out_gate  = proj_ptr + base + 4 * H

        left_raw      = tl.load(p_left,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0).to(tl.float16)
        right_raw     = tl.load(p_right,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0).to(tl.float16)
        left_gate_raw = tl.load(p_left_gate,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0).to(tl.float16)
        right_gate_raw= tl.load(p_right_gate,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0).to(tl.float16)
        out_gate_raw  = tl.load(p_out_gate,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0).to(tl.float16)

        # sigmoid in fp32
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # apply mask and gates
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate

        # store
        dst_left    = left_ptr    + offs_r[:, None] * H + offs_h[None, :]
        dst_right   = right_ptr   + offs_r[:, None] * H + offs_h[None, :]
        dst_outgate = out_gate_ptr+ offs_r[:, None] * H + offs_h[None, :]

        tl.store(dst_left,    left,    mask=mask_r[:, None] & mask_h[None, :])
        tl.store(dst_right,   right,   mask=mask_r[:, None] & mask_h[None, :])
        tl.store(dst_outgate, out_gate,mask=mask_r[:, None] & mask_h[None, :])

# ----------------------------------------------------------------------
# Custom forward function for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : dict with Linear/LayerNorm parameters (float32)
        - config_dict  : dict containing at least ``dim``, ``hidden_dim`` and
                         ``nomask`` (bool)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim], dtype float32.
    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    x, mask, weights, cfg = data
    dim        = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps        = 1e-5
    device     = x.device

    B, N, _, _ = x.shape
    R = B * N * N               # total pairwise rows
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (fp32 → fp16) using torch.nn.functional.layer_norm
    # --------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()                      # [R, dim] (fp32)
    x_norm_fp32 = torch.nn.functional.layer_norm(
        x_flat,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )                                                          # fp32
    x_norm = x_norm_fp32.to(torch.float16)                     # fp16
    del x_flat, x_norm_fp32

    # --------------------------------------------------------------
    # 2️⃣ Fused projection: 5 * Linear -> (R, 5*H)
    # --------------------------------------------------------------
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                         # (5*H, dim)
    proj = F.linear(x_norm, w_fused)                           # (R, 5*H), fp16
    del x_norm, w_fused

    # --------------------------------------------------------------
    # 3️⃣ Split fused result, apply mask, evaluate sigmoid gates
    # --------------------------------------------------------------
    left      = torch.empty(R, H, dtype=torch.float16, device=device)
    right     = torch.empty(R, H, dtype=torch.float16, device=device)
    out_gate  = torch.empty(R, H, dtype=torch.float16, device=device)

    if cfg.get("nomask", True) or mask is None:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)
    else:
        # mask shape [B, N, N] → [R]; convert to 0/1 fp16
        mask_fp16 = mask.to(torch.float16).reshape(R)
        has_mask = 1
        mask_ptr = mask_fp16

    BLOCK_R = 128
    BLOCK_H = 64
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣ Tri‑multiplicative update (batched GEMM over hidden dimension)
    # --------------------------------------------------------------
    # reshape to [B, N, N, H] and bring H into the batch axis
    left = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    left_mat  = left.reshape(B * H, N, N)    # [B·H, N, N]
    right_mat = right.reshape(B * H, N, N)   # [B·H, N, N]

    # half‑precision batched matrix multiplication
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [B·H, N, N]

    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]
    del left_mat, right_mat, left, right, out_mat

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm + element‑wise multiplication with out_gate
    # --------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()          # [R, H]  (fp16)

    out_norm = torch.nn.functional.layer_norm(
        out_flat,
        (H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )                                                # fp16

    # multiply with the previously computed gate
    out_fused = out_norm * out_gate                  # fp16
    del out_flat, out_norm, out_gate

    # --------------------------------------------------------------
    # 6️⃣ Final linear projection back to `dim` (fp16 → fp32)
    # --------------------------------------------------------------
    final_half = F.linear(
        out_fused,
        weights["to_out.weight"].to(torch.float16),  # (dim, H)
    )                                                # fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
