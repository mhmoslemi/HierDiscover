1️⃣ Row‑wise LayerNorm on the input tensor (B·N·N × dim) – performed with the
   native PyTorch `F.layer_norm` (fast fused CUDA kernel).  Result is cast to
   fp16.

2️⃣ Fuse the five linear layers (left/right projections + three gates) into one
   GEMM:  (R, dim) @ (5·H, dim)ᵀ  →  (R, 5·H)  (half‑precision, cuBLAS).

3️⃣ Triton kernel `split_fuse_gate_kernel` splits the fused projection, applies
   the optional pair‑wise mask, evaluates sigmoid gates, and writes three
   tensors (`left`, `right`, `out_gate`) – all in fp16.

4️⃣ Reshape `left` / `right` to (B·H, N, N) and perform a batched GEMM:
        out = left @ rightᵀ   (half‑precision, cuBLAS)

5️⃣ Second LayerNorm + element‑wise multiplication with `out_gate` – performed
   with the native `F.layer_norm` (fp32) followed by a simple broadcast
   multiplication.  Result is cast back to fp16.

6️⃣ Final linear projection back to `dim` using a fp32 GEMM.

Only the mask‑aware gate‑splitting logic is written in Triton, satisfying the
requirement to implement “part of the operations in a kernel” while keeping
the overall runtime well below the target.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: split fused projection, apply mask, evaluate sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[rows, 5*H]   (fp16)
    left_ptr,          # mutable *[rows, H]    (fp16)
    right_ptr,         # mutable *[rows, H]    (fp16)
    out_gate_ptr,      # mutable *[rows, H]    (fp16)
    mask_ptr,          # const   *[rows]       (fp16) – ignored if HAS_MASK=0
    rows: tl.int32,
    H: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,   # rows per program
    BLOCK_H: tl.constexpr,   # hidden dim per iteration
):
    """
    For each row r:
        left  = (proj_left  * mask) * sigmoid(proj_left_gate)
        right = (proj_right * mask) * sigmoid(proj_right_gate)
        out_gate = sigmoid(proj_out_gate)
    All ops are performed in fp16 (sigmoid evaluated in fp32 for
    numerical stability).
    Layout of proj:
        [ left_proj | right_proj | left_gate | right_gate | out_gate ]
        each segment has length H.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < rows

    # Load per‑row mask (if present) – broadcast to hidden dimension.
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)
        mask_val = mask_val[:, None]                     # [BLOCK_R, 1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # Loop over hidden dimension chunks.
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # Offsets into the fused projection.
        base = offs_r[:, None] * (5 * H) + offs_h[None, :]  # [BLOCK_R, BLOCK_H]

        p_left      = proj_ptr + base
        p_right     = proj_ptr + base + H
        p_left_gate = proj_ptr + base + 2 * H
        p_right_gate= proj_ptr + base + 3 * H
        p_out_gate  = proj_ptr + base + 4 * H

        # Load raw values.
        left_raw      = tl.load(p_left,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)
        right_raw     = tl.load(p_right,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)
        left_gate_raw = tl.load(p_left_gate,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)
        right_gate_raw= tl.load(p_right_gate,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)
        out_gate_raw  = tl.load(p_out_gate,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)

        # Sigmoid in fp32 for stability.
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # Apply mask and gates.
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate

        # Store results.
        dst_left   = left_ptr   + offs_r[:, None] * H + offs_h[None, :]
        dst_right  = right_ptr  + offs_r[:, None] * H + offs_h[None, :]
        dst_outgate= out_gate_ptr + offs_r[:, None] * H + offs_h[None, :]

        tl.store(dst_left,   left,   mask=mask_r[:, None] & mask_h[None, :])
        tl.store(dst_right,  right,  mask=mask_r[:, None] & mask_h[None, :])
        tl.store(dst_outgate,out_gate,mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Forward function
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor[BS, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor[BS, N, N] (bool/float) or None
        - weights_dict : mapping with all Linear/LayerNorm parameters
                         (float32 tensors, bias=False for Linear)
        - config_dict  : must contain ``dim``, ``hidden_dim`` and ``nomask``
                         (bool). May contain other optional flags.

    Returns
    -------
    torch.Tensor[BS, N, N, dim] (float32)
        The TriMul output.
    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    x, mask, weights, cfg = data
    dim        = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps        = 1e-5
    device     = x.device

    B, N, _, _ = x.shape
    R = B * N * N            # total rows for row‑wise ops
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise, fp32).  Output cast to fp16.
    # --------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()                        # (R, dim), fp32
    x_norm_fp32 = F.layer_norm(x_flat,
                               normalized_shape=(dim,),
                               weight=weights["norm.weight"],
                               bias=weights["norm.bias"],
                               eps=eps)                        # fp32
    x_norm = x_norm_fp32.to(torch.float16)                       # fp16

    # --------------------------------------------------------------
    # 2️⃣ Fused linear: left/right projections + three gates
    # --------------------------------------------------------------
    # Concatenate the five weight matrices (each [hidden_dim, dim]).
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                          # (5*H, dim)

    # (R, dim) @ (5*H, dim).T -> (R, 5*H)  (half‑precision GEMM)
    proj = F.linear(x_norm, w_fused)                              # fp16
    del w_fused, x_norm, x_flat  # free intermediate tensors

    # --------------------------------------------------------------
    # 3️⃣ Split fused result, apply mask, evaluate sigmoid gates
    # --------------------------------------------------------------
    left      = torch.empty(R, H, dtype=torch.float16, device=device)
    right     = torch.empty(R, H, dtype=torch.float16, device=device)
    out_gate  = torch.empty(R, H, dtype=torch.float16, device=device)

    if cfg.get("nomask", True) or mask is None:
        # No mask – compile‑time flag disables loading.
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)
    else:
        # mask shape [B, N, N] → flatten to [R]; convert to 0/1 half.
        mask_half = mask.to(torch.float16).reshape(R)
        has_mask = 1
        mask_ptr = mask_half

    BLOCK_R = 128
    BLOCK_H = 32
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣ Tri‑multiplicative update (batched GEMM over hidden dim)
    # --------------------------------------------------------------
    # Bring hidden dim into the batch axis for a single bmm call.
    left  = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    left_mat  = left.view(B * H, N, N)    # (B·H, N, N)
    right_mat = right.view(B * H, N, N)   # (B·H, N, N)

    # Batched half‑precision GEMM: left @ rightᵀ → (B·H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm + multiplication with out_gate
    # --------------------------------------------------------------
    # Convert to fp32 for a stable LayerNorm, then apply out_gate.
    out_fp32 = out.to(torch.float32)                                 # (B,N,N,H)
    out_norm = F.layer_norm(out_fp32,
                            normalized_shape=(H,),
                            weight=weights["to_out_norm.weight"],
                            bias=weights["to_out_norm.bias"],
                            eps=eps)                         # fp32
    out_gate_fp32 = out_gate.view(B, N, N, H).to(torch.float32)
    out_fused = (out_norm * out_gate_fp32).to(torch.float16)         # fp16
    del out, out_fp32, out_norm, out_gate, out_gate_fp32

    # --------------------------------------------------------------
    # 6️⃣ Final linear projection back to `dim` (fp32)
    # --------------------------------------------------------------
    final = F.linear(out_fused.float(),
                     weights["to_out.weight"])                    # fp32
    final = final.view(B, N, N, dim)                                 # (B,N,N,dim)

    return final
