def split_fuse_gate_kernel(proj_ptr, left_ptr, right_ptr, out_gate_ptr, mask_ptr, rows, H, HAS_MASK, BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < rows

    # load mask values (scalar per row)

        mask_val = mask_val[:, None] # broadcast

    for off_h in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = off_h * BLOCK_H + tl.arange(0, BLOCK_H)

        right_raw = tl.load(proj_ptr + base + H,

        left_gate_raw = tl.load(proj_ptr + base + 2 * H,

"""
TriMul (outgoing) – fast forward pass.
The implementation fuses the five linear projections + three sigmoid gates
into a single half‑precision GEMM and uses a custom Triton kernel only for the
mask / gate handling.  Both LayerNorm passes are delegated to
`torch.nn.functional.layer_norm`, which is highly optimized on H100.
The heavy N³ update is performed by a batched fp16 GEMM (`torch.bmm`).

Input:
    data = (x, mask, weights, cfg)
    - x      : torch.Tensor, shape [B, N, N, dim], dtype float32
    - mask   : torch.Tensor, shape [B, N, N] (bool/float) or None
    - weights: dict of tensors (all float32, Linear bias=False)
    - cfg    : dict with keys "dim", "hidden_dim", "nomask" (bool)

Output:
    torch.Tensor, shape [B, N, N, dim], dtype float32
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: split fused projection, apply optional mask and sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[rows, 5*H]    (fp16)
    left_ptr,          # mutable *[rows, H]    (fp16)
    right_ptr,         # mutable *[rows, H]    (fp16)
    out_gate_ptr,      # mutable *[rows, H]    (fp16)
    mask_ptr,          # const   *[rows]       (fp16) – ignored if HAS_MASK==0
    rows: tl.int32,
    H: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,   # rows per program
    BLOCK_H: tl.constexpr,   # hidden dim chunk
):
    """
    For each row:
        left  = (proj_left  * mask) * sigmoid(proj_left_gate)
        right = (proj_right * mask) * sigmoid(proj_right_gate)
        out_gate = sigmoid(proj_out_gate)
    All ops are fp16; sigmoid evaluated in fp32 for stability.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)                # [BLOCK_R]
    mask_r = offs_r < rows

    # -------------------- load mask (scalar per row) --------------------
    if HAS_MASK:
        m = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        m = m[:, None]                                          # broadcast to [BLOCK_R,1]
    else:
        m = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # -------------------- iterate over hidden dimension chunks ------------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)        # [BLOCK_H]
        mask_h = offs_h < H

        # base offset for the five concatenated slices
        base = offs_r[:, None] * (5 * H) + offs_h[None, :]      # [BLOCK_R, BLOCK_H]

        # ----- load raw projection results --------------------------------
        left_raw      = tl.load(proj_ptr + base,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_raw     = tl.load(proj_ptr + base + H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        left_gate_raw = tl.load(proj_ptr + base + 2 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_gate_raw= tl.load(proj_ptr + base + 3 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        out_gate_raw  = tl.load(proj_ptr + base + 4 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)

        # ----- sigmoid (computed in fp32) ---------------------------------
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # ----- apply mask and gates ---------------------------------------
        left  = left_raw  * m * left_gate
        right = right_raw * m * right_gate

        # ----- store results ------------------------------------------------
        tl.store(left_ptr     + offs_r[:, None] * H + offs_h[None, :],
                 left,  mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr    + offs_r[:, None] * H + offs_h[None, :],
                 right, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + offs_r[:, None] * H + offs_h[None, :],
                 out_gate, mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Helper to pick reasonable block sizes for the split kernel
# ----------------------------------------------------------------------
def _split_block_cfg(rows: int, H: int):
    """Return (BLOCK_R, BLOCK_H) tuned to problem size."""
    # More rows → more programs, but keep at least 64 rows per program.
    BLOCK_R = 128 if rows >= 128 else rows
    # Hidden dim is modest (≤384). Use a chunk that fits in registers.
    BLOCK_H = 64 if H > 64 else H
    return BLOCK_R, BLOCK_H


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (x, mask, weights, cfg)
            x      : torch.Tensor, [B, N, N, dim] (float32)
            mask   : torch.Tensor, [B, N, N] (bool/float) or None
            weights: dict of model parameters (float32)
            cfg    : dict with "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor, shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    nomask = cfg.get("nomask", True)

    device = x.device
    B, N, _, _ = x.shape
    R = B * N * N                     # total rows for row‑wise ops
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) – use torch’s highly‑optimized kernel
    # ------------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()          # [R, dim] float32
    x_norm_fp32 = F.layer_norm(
        x_flat,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )
    x_norm = x_norm_fp32.to(torch.float16)        # [R, dim] fp16

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear projection (left/right + 3 gates)
    # ------------------------------------------------------------------
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],   # (H, dim)
            weights["right_proj.weight"],  # (H, dim)
            weights["left_gate.weight"],   # (H, dim)
            weights["right_gate.weight"],  # (H, dim)
            weights["out_gate.weight"],    # (H, dim)
        ],
        dim=0,
    ).to(torch.float16)                # [5*H, dim] fp16

    # (R, dim) @ (5*H, dim)ᵀ → (R, 5*H)   (half‑precision GEMM)
    proj = F.linear(x_norm, w_fused)   # fp16
    del w_fused, x_norm                # free memory

    # ------------------------------------------------------------------
    # 3️⃣ Split, apply mask and sigmoid gates (single Triton kernel)
    # ------------------------------------------------------------------
    left      = torch.empty(R, H, dtype=torch.float16, device=device)
    right     = torch.empty(R, H, dtype=torch.float16, device=device)
    out_gate  = torch.empty(R, H, dtype=torch.float16, device=device)

    if nomask:
        dummy_mask = torch.empty(0, dtype=torch.float16, device=device)
        HAS_MASK = 0
        mask_ptr = dummy_mask
    else:
        mask_fp16 = mask.to(torch.float16).reshape(R)   # broadcast mask per row
        HAS_MASK = 1
        mask_ptr = mask_fp16

    BLOCK_R, BLOCK_H = _split_block_cfg(R, H)
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Tri‑multiplicative update (batched GEMM over hidden dim)
    # ------------------------------------------------------------------
    # reshape to [B, N, N, H] → permute to [B, H, N, N] → flatten batch/H
    left = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    left_mat  = left.view(B * H, N, N)     # [B·H, N, N]
    right_mat = right.view(B * H, N, N)    # [B·H, N, N]

    # left @ rightᵀ  (fp16 batched GEMM)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N]

    # reshape back to [R, H] (fp16)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]
    out_flat = out.view(R, H).contiguous()
    del left_mat, right_mat, left, right, out_mat, out

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm + element‑wise multiply by out_gate
    # ------------------------------------------------------------------
    out_norm_fp32 = F.layer_norm(
        out_flat,
        (H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    # gate is fp16 → promote to fp32 for the multiplication
    out_gate_fp32 = out_gate.view(R, H).to(torch.float32)
    out_fused_fp32 = out_norm_fp32 * out_gate_fp32
    out_fused = out_fused_fp32.to(torch.float16)   # [R, H] fp16
    del out_flat, out_gate, out_norm_fp32, out_gate_fp32, out_fused_fp32

    # ------------------------------------------------------------------
    # 6️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    final_half = F.linear(
        out_fused,
        weights["to_out.weight"].to(torch.float16),   # [dim, H]
    )   # (R, dim) fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
