        acc_right = tl.zeros((BLOCK_R, BLOCK_H), dtype=tl.float32)

        acc_right_gate = tl.zeros((BLOCK_R, BLOCK_H), dtype=tl.float32)

            mask_d = offs_d < dim

            # Load X tile (R, D) as float32
            x = tl.load(x_ptr + offs_r[:, None] * dim + offs_d[None, :],
                        mask=mask_r[:, None] & mask_d[None, :],
                        other=0.0).to(tl.float32)  # [BLOCK_R, BLOCK_D]

            # Load weight tiles for each group (D, H_tile)
            lp = tl.load(left_proj_ptr + offs_d[:, None] * H + offs_h[None, :],

                         mask=mask_d[:, None] & mask_h[None, :],
                         other=0.0).to(tl.float32)
            og = tl.load(out_gate_w_ptr + offs_d[:, None] * H + offs_h[None, :],
                         mask=mask_d[:, None] & mask_h[None, :],
                         other=0.0).to(tl.float32)

            # Accumulate dot products: (R,D) @ (D,H) -> (R,H)
            acc_left += tl.dot(x, lp)
            acc_right += tl.dot(x, rp)
            acc_left_gate += tl.dot(x, lg)
            acc_right_gate += tl.dot(x, rg)
            acc_out_gate += tl.dot(x, og)

        right = acc_right.to(tl.float16) * mask_expand * right_gate_sig
        out_gate = out_gate_sig

        # Store results
        tl.store(left_ptr + offs_r[:, None] * H + offs_h[None, :],

        tl.store(right_ptr + offs_r[:, None] * H + offs_h[None, :],

                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + offs_r[:, None] * H + offs_h[None, :],

any combination of the test configurations described in the problem
statement.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 2️⃣ Fused projection + gating kernel
# ----------------------------------------------------------------------
@triton.jit
def proj_gate_fused_kernel(
    # inputs
    x_ptr,                      # const   *[R, dim]      (fp16)
    # outputs
    left_ptr,                   # mutable *[R, H]       (fp16)
    right_ptr,                  # mutable *[R, H]       (fp16)
    out_gate_ptr,               # mutable *[R, H]       (fp16)
    # optional mask
    mask_ptr,                   # const   *[R]           (fp16)   (ignored if HAS_MASK==0)
    # transposed weight matrices (dim, H)
    left_proj_ptr,              # const   *[dim, H]      (fp16)
    right_proj_ptr,             # const   *[dim, H]      (fp16)
    left_gate_ptr,              # const   *[dim, H]      (fp16)
    right_gate_ptr,             # const   *[dim, H]      (fp16)
    out_gate_weight_ptr,        # const   *[dim, H]      (fp16)
    # shape / compile‑time parameters
    R, dim, H,                  # int32
    HAS_MASK: tl.constexpr,     # int32  (0 or 1)
    BLOCK_R: tl.constexpr,      # rows processed per program
    BLOCK_H: tl.constexpr,      # hidden‑dim tile size
    BLOCK_D: tl.constexpr,      # input‑dim tile size
):
    """\
    For each row r (flattened batch × i × k) compute:
        left  = (x·W_left   * mask) * sigmoid(x·W_left_gate)
        right = (x·W_right  * mask) * sigmoid(x·W_right_gate)
        out_gate = sigmoid(x·W_out_gate)
    All work is done in fp32 accumulators, the final tensors are stored
    as fp16.  The kernel processes `BLOCK_R` rows and a `BLOCK_H` slice
    of the hidden dimension at a time.
    """
    pid = tl.program_id(0)                     # one program per row‑tile
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)  # [BLOCK_R]
    mask_r = offs_r < R

    # ------------------------------------------------------------------
    # Load mask once (if present) – broadcast later
    # ------------------------------------------------------------------
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r,
                           mask=mask_r,
                           other=1.0).to(tl.float16)          # [BLOCK_R]
        mask_val = mask_val[:, None]                         # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # ------------------------------------------------------------------
    # Loop over hidden‑dim tiles
    # ------------------------------------------------------------------
    num_h_tiles = tl.cdiv(H, BLOCK_H)
    num_d_tiles = tl.cdiv(dim, BLOCK_D)

    for h_tile in range(0, num_h_tiles):
        offs_h = h_tile * BLOCK_H + tl.arange(0, BLOCK_H)        # [BLOCK_H]
        mask_h = offs_h < H

        # fp32 accumulators for the five dot‑products
        acc_left      = tl.zeros((BLOCK_R, BLOCK_H), dtype=tl.float32)
        acc_right     = tl.zeros((BLOCK_R, BLOCK_H), dtype=tl.float32)
        acc_left_gate = tl.zeros((BLOCK_R, BLOCK_H), dtype=tl.float32)
        acc_right_gate= tl.zeros((BLOCK_R, BLOCK_H), dtype=tl.float32)
        acc_out_gate  = tl.zeros((BLOCK_R, BLOCK_H), dtype=tl.float32)

        # ------------------------------------------------------------------
        # Loop over input‑dim tiles (inner dimension of the GEMM)
        # ------------------------------------------------------------------
        for d_tile in range(0, num_d_tiles):
            offs_d = d_tile * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
            mask_d = offs_d < dim

            # ---- load X (R × D) -------------------------------------------------
            x = tl.load(x_ptr + offs_r[:, None] * dim + offs_d[None, :],
                        mask=mask_r[:, None] & mask_d[None, :],
                        other=0.0).to(tl.float32)                # [BLOCK_R, BLOCK_D]

            # ---- load weight tiles (D × H_tile) ----------------------------------
            # weights are stored transposed as (dim, H)
            w_left = tl.load(left_proj_ptr + offs_d[:, None] * H + offs_h[None, :],
                             mask=mask_d[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float32)              # [BLOCK_D, BLOCK_H]
            w_right= tl.load(right_proj_ptr+ offs_d[:, None] * H + offs_h[None, :],
                             mask=mask_d[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float32)
            w_lgate= tl.load(left_gate_ptr + offs_d[:, None] * H + offs_h[None, :],
                             mask=mask_d[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float32)
            w_rgate= tl.load(right_gate_ptr+ offs_d[:, None] * H + offs_h[None, :],
                             mask=mask_d[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float32)
            w_out  = tl.load(out_gate_weight_ptr + offs_d[:, None] * H + offs_h[None, :],
                             mask=mask_d[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float32)

            # ---- accumulate the five matrix‑vector products -----------------------
            acc_left      += tl.dot(x, w_left)       # (R,H_tile)
            acc_right     += tl.dot(x, w_right)
            acc_left_gate += tl.dot(x, w_lgate)
            acc_right_gate+= tl.dot(x, w_rgate)
            acc_out_gate  += tl.dot(x, w_out)

        # --------------------------------------------------------------------
        # Apply mask, sigmoid gates and store the results
        # --------------------------------------------------------------------
        # sigmoid in fp32 for stability, then cast back to fp16
        left_gate   = (1.0 / (1.0 + tl.exp(-acc_left_gate))).to(tl.float16)
        right_gate  = (1.0 / (1.0 + tl.exp(-acc_right_gate))).to(tl.float16)
        out_gate    = (1.0 / (1.0 + tl.exp(-acc_out_gate))).to(tl.float16)

        left  = acc_left .to(tl.float16) * mask_val * left_gate
        right = acc_right.to(tl.float16) * mask_val * right_gate

        # store – respect masks for rows / hidden‑dim tails
        tl.store(left_ptr     + offs_r[:, None] * H + offs_h[None, :],
                 left,  mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr    + offs_r[:, None] * H + offs_h[None, :],
                 right, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + offs_r[:, None] * H + offs_h[None, :],
                 out_gate, mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 4️⃣ Second LayerNorm + out‑gate multiplication (fused)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_mul_kernel(
    x_ptr,            # const   *[rows, dim]   (fp16)
    gate_ptr,         # const   *[rows, dim]   (fp16)   (out_gate)
    y_ptr,            # mutable *[rows, dim]   (fp16)   (result)
    weight_ptr,       # const   *[dim]        (fp32)   (γ)
    bias_ptr,         # const   *[dim]        (fp32)   (β)
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)                 # [BLOCK_M]
    mask_m = offs_m < rows

    # --------------------------------------------------------------
    # Compute mean / variance (fp32)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)       # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :] # [M,D]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------------------------------------------------------------
    # Normalize, apply γ/β, multiply with gate, store (fp16)
    # --------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        # load x
        x_val = tl.load(x_ptr + offs_m[:, None] * dim + offs_d[None, :],
                        mask=mask_m[:, None] & mask_d[None, :],
                        other=0.0).to(tl.float32)

        # (x‑μ)·rstd
        norm = (x_val - mean[:, None]) * rstd[:, None]           # fp32

        # γ, β (float32)
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β

        norm = norm * w[None, :] + b[None, :]                     # fp32

        # gate
        gate = tl.load(gate_ptr + offs_m[:, None] * dim + offs_d[None, :],
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)                # fp32

        out = norm * gate                                        # fp32
        out_fp16 = out.to(tl.float16)

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor, shape [B, N, N, dim], float32
            - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
            - weights_dict : dict with LayerNorm/Linear parameters
                               (all tensors float32, Linear layers bias=False)
            - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim         = cfg["dim"]
    hidden_dim  = cfg["hidden_dim"]
    eps         = 1e-5
    device      = x.device

    B, N, _, _ = x.shape
    R = B * N * N                         # total number of (batch,i,k) rows
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 → fp16) – use the highly‑optimized
    #    PyTorch implementation.
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        x.float(),
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    ).to(torch.float16)                # [B, N, N, dim] → fp16

    x_flat = x_norm.view(R, dim).contiguous()   # [R, dim]

    # ------------------------------------------------------------------
    # 2️⃣ Transpose weight matrices once (dim, hidden) to enable a
    #    GEMM‑style dot product without extra transposes inside the kernel.
    # ------------------------------------------------------------------
    left_proj_T   = weights["left_proj.weight"].t().contiguous().to(torch.float16)
    right_proj_T  = weights["right_proj.weight"].t().contiguous().to(torch.float16)
    left_gate_T   = weights["left_gate.weight"].t().contiguous().to(torch.float16)
    right_gate_T  = weights["right_gate.weight"].t().contiguous().to(torch.float16)
    out_gate_T    = weights["out_gate.weight"].t().contiguous().to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Fused projection + gating kernel
    # ------------------------------------------------------------------
    left      = torch.empty(R, H, dtype=torch.float16, device=device)
    right     = torch.empty(R, H, dtype=torch.float16, device=device)
    out_gate  = torch.empty(R, H, dtype=torch.float16, device=device)

    if cfg.get("nomask", True) or mask is None:
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)
    else:
        has_mask = 1
        mask_ptr = mask.to(torch.float16).reshape(R)

    # Tunable block sizes – chosen to keep register pressure modest while
    # delivering high occupancy on an H100.
    BLOCK_R = 128   # rows per program
    BLOCK_H = 32    # hidden‑dim tile
    BLOCK_D = 32    # input‑dim tile

    grid = (triton.cdiv(R, BLOCK_R),)
    proj_gate_fused_kernel[grid](
        x_flat,
        left,
        right,
        out_gate,
        mask_ptr,
        left_proj_T,
        right_proj_T,
        left_gate_T,
        right_gate_T,
        out_gate_T,
        R,
        dim,
        H,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM over hidden dimension (left @ rightᵀ)
    # ------------------------------------------------------------------
    # reshape to [B, N, N, H] → permute to [B, H, N, N] for the batched
    # matmul that fuses the (i,k) and (j,k) axes.
    left  = left .view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]

    left_mat  = left .view(B * H, N, N)   # [B*H, N, N]
    right_mat = right.view(B * H, N, N)   # [B*H, N, N]

    # out_mat[b·H+i] = left_mat[b·H+i] @ right_mat[b·H+i]ᵀ
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))          # fp16

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm + multiplication by out_gate (fused)
    # ------------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()          # [R, H] fp16
    out_fused = torch.empty_like(out_flat, dtype=torch.float16, device=device)

    grid_ln2 = (triton.cdiv(R, BLOCK_R),)
    layernorm_mul_kernel[grid_ln2](
        out_flat,
        out_gate.view(R, H),
        out_fused,
        weights["to_out_norm.weight"],   # γ  (float32)
        weights["to_out_norm.bias"],     # β  (float32)
        R,
        H,
        eps,
        BLOCK_M=BLOCK_R,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Final projection back to `dim`
    # ------------------------------------------------------------------
    final_half = F.linear(
        out_fused,
        weights["to_out.weight"].t().contiguous().to(torch.float16),  # [dim, H]
    )   # fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
