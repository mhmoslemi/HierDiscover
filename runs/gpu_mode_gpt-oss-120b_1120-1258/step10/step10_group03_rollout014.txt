    out_gate_ptr,     # mutable *[rows, H] fp16
    mask_ptr,         # const *[rows] fp16 (if present)
    w_left_ptr, w_right_ptr, w_left_gate_ptr, w_right_gate_ptr, w_out_gate_ptr,
    rows: tl.int32,
    dim: tl.int32,
    H: tl.int32,
    eps: tl.float32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)  # [BLOCK_R]
    mask_r = offs_r < rows

    # First compute mean and variance across dim (row-wise) to perform LayerNorm
    # We need to compute these for each row.
    # We'll have accumulators sum_val, sum_sq of shape [BLOCK_R] in fp32.
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_R], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        x_ptrs = x_ptr + offs_r[:, None] * dim + offs_d[None, :]
        x = tl.load(x_ptrs,
                    mask=mask_r[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)

        sum_val += tl.sum(x, axis=1)
        sum_sq += tl.sum(x * x, axis=1)

    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

def custom_kernel(data):
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    nomask = cfg.get("nomask", True)
    device = x.device
    B,N,_,_ = x.shape
    R = B*N*N
    H = hidden_dim

    # 1. first LayerNorm using torch (fast)
    x_flat = x.view(R, dim)
    x_norm = torch.nn.functional.layer_norm(x_flat, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps)
    x_norm = x_norm.to(torch.float16).contiguous()

    # 2. fused projection + gates (custom kernel)
    left = torch.empty(R, H, dtype=torch.float16, device=device)
    right = torch.empty(R, H, dtype=torch.float16, device=device)
    out_gate = torch.empty(R, H, dtype=torch.float16, device=device)

    # pack weight tensors
    w_left = weights["left_proj.weight"].to(torch.float16).contiguous()
    w_right = weights["right_proj.weight"].to(torch.float16).contiguous()
    w_left_gate = weights["left_gate.weight"].to(torch.float16).contiguous()
    w_right_gate = weights["right_gate.weight"].to(torch.float16).contiguous()
    w_out_gate = weights["out_gate.weight"].to(torch.float16).contiguous()

    # mask handling
    if nomask:
        HAS_MASK = 0
        dummy_mask = torch.empty(0, dtype=torch.float16, device=device)
        mask_ptr = dummy_mask
    else:
        HAS_MASK = 1
        mask_fp16 = mask.to(torch.float16).reshape(R)  # flatten
        mask_ptr = mask_fp16

    BLOCK_R = 128  # rows per program
    BLOCK_H = 16   # hidden per iteration
    BLOCK_D = 32   # dim tile
    grid = (triton.cdiv(R, BLOCK_R),)

    proj_gate_fused_kernel[grid](
        x_norm,
        left,
        right,
        out_gate,
        mask_ptr,
        w_left,
        w_right,
        w_left_gate,
        w_right_gate,
        w_out_gate,
        R,
        dim,
        H,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    # free x_norm
    del x_norm

    # 3. batched GEMM
    left = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # B, H, N, N
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    left_mat = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # B, N, N, H

    # 4. second LayerNorm + out_gate multiplication (custom kernel)
    out_flat = out.view(R, H).contiguous()
    out_gate_flat = out_gate.view(R, H).contiguous()
    out_fused = torch.empty_like(out_flat, dtype=torch.float16, device=device)
    BLOCK_M2, BLOCK_D2 = _choose_block_sizes(R, H)

    layernorm_mul_kernel[(triton.cdiv(R, BLOCK_M2),)](
        out_flat,
        out_gate_flat,
        out_fused,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M2,
        BLOCK_D=BLOCK_D2,
        num_warps=4,
    )
    # clean up
    del out, out_gate, out_flat, out_gate_flat

    # 5. final linear projection
    final_half = F.linear(out_fused, weights["to_out.weight"].to(torch.float16))
    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final

"""
Tri‑Mul (outgoing) – fast forward implementation.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the input tensor (B·N·N × dim) – done with
   torch.nn.functional.layer_norm (high‑performance cuDNN kernel).

2️⃣ Fuse the five linear maps (left/right projection + three gates) into a
   single Triton kernel:
        * reads a row of the LN‑output (fp16)
        * computes the five matrix‑vector products with the corresponding
          weight matrices (fp16 → fp32 accumulation)
        * applies the optional pair‑wise mask, sigmoid gates and writes the
          three results: left, right and out_gate (all fp16).

   This eliminates the huge intermediate “proj” tensor (R×5·H) and halves
   memory traffic.

3️⃣ Batched half‑precision GEMM (torch.bmm) over the hidden dimension:
        out[b·H+i] = left_mat[b·H+i] @ right_mat[b·H+i].T
   (B·H independent GEMMs of size N×N → we reuse cuBLAS tensor‑core kernels).

4️⃣ Second LayerNorm + multiplication by out_gate – fused in a tiny Triton
   kernel (row‑wise LN on the hidden dimension, then element‑wise gate).

5️⃣ Final Linear projection back to “dim”.

Only the projection/gate and the second LayerNorm are custom Triton kernels;
all heavy GEMMs use cuBLAS for peak throughput on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: second LayerNorm + multiplication with out_gate
# ----------------------------------------------------------------------
@triton.jit
def layernorm_mul_kernel(
    x_ptr,            # const   *[rows, dim]   (fp16)
    gate_ptr,         # const   *[rows, dim]   (fp16)   (out_gate)
    y_ptr,            # mutable *[rows, dim]   (fp16)   (result)
    weight_ptr,       # const   *[dim]        (fp32)   γ
    bias_ptr,         # const   *[dim]        (fp32)   β
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # ----- mean / variance (fp32) ------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ----- normalize, apply γ/β, multiply by gate --------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        # load X
        x = tl.load(x_ptr + offs_m[:, None] * dim + offs_d[None, :],
                    mask=mask_m[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)

        # (x‑μ)·rstd
        norm = (x - mean[:, None]) * rstd[:, None]

        # γ, β
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)
        norm = norm * w[None, :] + b[None, :]

        # gate
        g = tl.load(gate_ptr + offs_m[:, None] * dim + offs_d[None, :],
                    mask=mask_m[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)

        out = norm * g
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
# Triton kernel: fused projection + three gates (no intermediate buffer)
# ----------------------------------------------------------------------
@triton.jit
def proj_gate_fused_kernel(
    x_ptr,          # const   *[rows, dim]   (fp16)   – already layer‑normed
    left_ptr,       # mutable *[rows, H]    (fp16)
    right_ptr,      # mutable *[rows, H]    (fp16)
    out_gate_ptr,   # mutable *[rows, H]    (fp16)
    mask_ptr,       # const   *[rows]       (fp16)   (ignored if HAS_MASK=0)
    w_left_ptr,     # const   *[H, dim]    (fp16)
    w_right_ptr,    # const   *[H, dim]    (fp16)
    w_lgate_ptr,    # const   *[H, dim]    (fp16)
    w_rgate_ptr,    # const   *[H, dim]    (fp16)
    w_out_ptr,      # const   *[H, dim]    (fp16)
    rows: tl.int32,
    dim: tl.int32,
    H: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """
    For each row r:
        left_raw  = Σ_d x[r,d] * W_left[d]          (fp32)
        right_raw = Σ_d x[r,d] * W_right[d]
        left_gate_raw  = Σ_d x[r,d] * W_lgate[d]
        right_gate_raw = Σ_d x[r,d] * W_rgate[d]
        out_gate_raw   = Σ_d x[r,d] * W_out[d]

        left  = (left_raw  * mask) * sigmoid(left_gate_raw)
        right = (right_raw * mask) * sigmoid(right_gate_raw)
        out_gate = sigmoid(out_gate_raw)
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)                # [BLOCK_R]
    mask_r = offs_r < rows

    # load mask (if present)
    if HAS_MASK:
        m = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)[:, None]   # [BLOCK_R,1] fp16
    else:
        m = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)                # broadcast 1

    m_f32 = m.to(tl.float32)

    # ------------------------------------------------------------------
    # hidden‑dimension loop
    # ------------------------------------------------------------------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)         # [BLOCK_H]
        mask_h = offs_h < H

        # accumulators for the five products (fp32)
        acc_left      = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        acc_right     = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        acc_lgate     = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        acc_rgate     = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        acc_out_gate  = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)

        # ----------------------------------------------------------------
        # dot‑product over the input dimension
        # ----------------------------------------------------------------
        for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
            offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
            mask_d = offs_d < dim

            # load a chunk of the input row (fp32)
            x = tl.load(x_ptr + offs_r[:, None] * dim + offs_d[None, :],
                        mask=mask_r[:, None] & mask_d[None, :],
                        other=0.0).to(tl.float32)                # [R,D]

            # weight slices (fp32)
            w_l = tl.load(w_left_ptr   + offs_h[:, None] * dim + offs_d[None, :],
                          mask=mask_h[:, None] & mask_d[None, :],
                          other=0.0).to(tl.float32)                # [H,D]
            w_r = tl.load(w_right_ptr  + offs_h[:, None] * dim + offs_d[None, :],
                          mask=mask_h[:, None] & mask_d[None, :],
                          other=0.0).to(tl.float32)
            w_lg = tl.load(w_lgate_ptr + offs_h[:, None] * dim + offs_d[None, :],
                           mask=mask_h[:, None] & mask_d[None, :],
                           other=0.0).to(tl.float32)
            w_rg = tl.load(w_rgate_ptr + offs_h[:, None] * dim + offs_d[None, :],
                           mask=mask_h[:, None] & mask_d[None, :],
                           other=0.0).to(tl.float32)
            w_o  = tl.load(w_out_ptr   + offs_h[:, None] * dim + offs_d[None, :],
                           mask=mask_h[:, None] & mask_d[None, :],
                           other=0.0).to(tl.float32)

            # broadcast multiply‑accumulate: (R,1,D) * (1,H,D) → (R,H,D) → sum over D
            acc_left     += tl.sum(x[:, None, :] * w_l[None, :, :], axis=2)
            acc_right    += tl.sum(x[:, None, :] * w_r[None, :, :], axis=2)
            acc_lgate    += tl.sum(x[:, None, :] * w_lg[None, :, :], axis=2)
            acc_rgate    += tl.sum(x[:, None, :] * w_rg[None, :, :], axis=2)
            acc_out_gate += tl.sum(x[:, None, :] * w_o[None, :, :], axis=2)

        # ----------------------------------------------------------------
        # apply mask and sigmoid gates (still fp32)
        # ----------------------------------------------------------------
        left_raw  = acc_left  * m_f32                 # [R,H]
        right_raw = acc_right * m_f32

        left_gate  = 1.0 / (1.0 + tl.exp(-acc_lgate))  # fp32
        right_gate = 1.0 / (1.0 + tl.exp(-acc_rgate))
        out_gate   = 1.0 / (1.0 + tl.exp(-acc_out_gate))

        # fused results (fp16)
        left  = (left_raw  * left_gate).to(tl.float16)
        right = (right_raw * right_gate).to(tl.float16)
        outg  = out_gate.to(tl.float16)

        # store
        dst_left  = left_ptr  + offs_r[:, None] * H + offs_h[None, :]
        dst_right = right_ptr + offs_r[:, None] * H + offs_h[None, :]
        dst_gate  = out_gate_ptr + offs_r[:, None] * H + offs_h[None, :]

        tl.store(dst_left,  left,  mask=mask_r[:, None] & mask_h[None, :])
        tl.store(dst_right, right, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(dst_gate,  outg,  mask=mask_r[:, None] & mask_h[None, :])

# ----------------------------------------------------------------------
# Helper: choose sensible block sizes
# ----------------------------------------------------------------------
def _choose_block_sizes(rows: int, dim: int):
    BLOCK_M = 128 if rows >= 128 else max(1, rows)
    BLOCK_D = 32 if dim >= 32 else dim
    return BLOCK_M, BLOCK_D

# ----------------------------------------------------------------------
# Entrypoint
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor, [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, [B, N, N] (bool / float) or None
        - weights_dict : mapping of all Linear/LayerNorm parameters (float32)
        - config_dict  : must contain "dim", "hidden_dim", "nomask"

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim         = cfg["dim"]
    hidden_dim  = cfg["hidden_dim"]
    eps         = 1e-5
    nomask      = cfg.get("nomask", True)
    device      = x.device

    B, N, _, _ = x.shape
    R = B * N * N               # rows after flattening
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (use cuDNN for speed)
    # ------------------------------------------------------------------
    x_flat = x.view(R, dim)
    x_norm = torch.nn.functional.layer_norm(
        x_flat,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    ).to(torch.float16).contiguous()            # [R, dim] fp16

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection + gates (custom Triton kernel)
    # ------------------------------------------------------------------
    left      = torch.empty(R, H, dtype=torch.float16, device=device)
    right     = torch.empty(R, H, dtype=torch.float16, device=device)
    out_gate  = torch.empty(R, H, dtype=torch.float16, device=device)

    # weight tensors (fp16, contiguous)
    w_left   = weights["left_proj.weight"].to(torch.float16).contiguous()
    w_right  = weights["right_proj.weight"].to(torch.float16).contiguous()
    w_lgate  = weights["left_gate.weight"].to(torch.float16).contiguous()
    w_rgate  = weights["right_gate.weight"].to(torch.float16).contiguous()
    w_outg   = weights["out_gate.weight"].to(torch.float16).contiguous()

    if nomask:
        HAS_MASK = 0
        dummy_mask = torch.empty(0, dtype=torch.float16, device=device)
        mask_ptr = dummy_mask
    else:
        HAS_MASK = 1
        mask_ptr = mask.to(torch.float16).reshape(R)   # flatten

    BLOCK_R = 128          # rows per program (≤1024 threads per block)
    BLOCK_H = 16           # hidden dim chunk
    BLOCK_D = 32           # input‑dim chunk
    grid = (triton.cdiv(R, BLOCK_R),)

    proj_gate_fused_kernel[grid](
        x_norm,
        left,
        right,
        out_gate,
        mask_ptr,
        w_left,
        w_right,
        w_lgate,
        w_rgate,
        w_outg,
        R,
        dim,
        H,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    del x_norm, w_left, w_right, w_lgate, w_rgate, w_outg

    # ------------------------------------------------------------------
    # 3️⃣ Batched GEMM over the hidden dimension (cuBLAS Tensor‑cores)
    # ------------------------------------------------------------------
    left = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    left_mat  = left.view(B * H, N, N)      # [B·H, N, N]
    right_mat = right.view(B * H, N, N)     # [B·H, N, N]

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N]

    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]
    del left_mat, right_mat, left, right, out_mat

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()

    )   # (R, dim) fp16
