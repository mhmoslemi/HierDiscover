@triton.jit
def split_fuse_gate_kernel(
    proj_ptr, left_ptr, right_ptr, out_gate_ptr, mask_ptr,
    rows: tl.int32, N: tl.int32, H: tl.int32, N2: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < rows

    if HAS_MASK:
        m = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)
        m = m[:, None] # broadcast later

    # decode coordinates
    b = offs_r // N2
    rem = offs_r - b * N2
    i = rem // N
    k = rem - i * N  # remainder

    # broadcast coordinates for hidden dim
    # will be used as off_h loops
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # compute offsets: left/right offset
        # off_lr = ((b * H + h) * N2) + (i * N + k) = ((b * H + h) * N2) + rem
        # Since b, h and rem are vectors of shape BLOCK_R and BLOCK_H
        # We'll compute using broadcasting: ((b[:,None] * H + offs_h[None,:]) * N2) + rem[:,None]
        off_lr = ((b[:, None] * H + offs_h[None, :]) * N2) + rem[:, None]  # shape [BLOCK_R, BLOCK_H]

        # out_gate offset: offs_r * H + h
        off_og = offs_r[:, None] * H + offs_h[None, :]

        # compute projection offsets (same as before)
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]  # [R, H]

        # slices
        left_raw = tl.load(proj_ptr + proj_base,
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0)
        right_raw = tl.load(proj_ptr + proj_base + H,
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0)
        left_gate_raw = tl.load(proj_ptr + proj_base + 2*H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_gate_raw = tl.load(proj_ptr + proj_base + 3*H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        out_gate_raw = tl.load(proj_ptr + proj_base + 4*H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)

        # sigmoid in fp32
        left_gate = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate_val = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # apply mask
        if HAS_MASK:
            mask_val = m
        else:
            mask_val = tl.full([BLOCK_R,1], 1.0, dtype=tl.float16)

        left = left_raw * mask_val * left_gate
        right = right_raw * mask_val * right_gate

        # store left/right/out_gate
        tl.store(left_ptr + off_lr, left, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + off_lr, right, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + off_og, out_gate_val, mask=mask_r[:, None] & mask_h[None, :])

            out_gate     : (B, N, N, H)   (row‑major, H last)
4️⃣ Batched half‑precision GEMM over the hidden dimension using
   torch.bmm (no extra copies):
       out[b,i,j,h] = Σ_k left[b,h,i,k] * right[b,h,j,k]
5️⃣ torch.nn.functional.layer_norm on the result (normalises over H),
   multiplies by the already‑computed out_gate, and finally a half‑precision
   linear projection back to `dim`.
All heavy GEMM work is delegated to cuBLAS; only the two custom kernels
are needed, dramatically reducing memory traffic and launch overhead.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (fp32 → fp16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   *[rows, dim]   (float32)
    y_ptr,          # mutable *[rows, dim]   (float16)
    w_ptr,          # const   *[dim]        (float32)   γ
    b_ptr,          # const   *[dim]        (float32)   β
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ---- mean / variance (fp32) ------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- write normalized values (fp16) -----------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]       # fp32

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # γ (fp32)
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # β (fp32)

        out = norm * w[None, :] + b[None, :]                 # fp32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2️⃣ Split‑fuse‑gate kernel – writes directly in the layout needed
#      for the batched GEMM.
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,           # const   *[rows, 5*H]   (float16)
    left_ptr,           # mutable *[B, H, N, N] (float16)
    right_ptr,          # mutable *[B, H, N, N] (float16)
    out_gate_ptr,       # mutable *[B, N, N, H] (float16)
    mask_ptr,           # const   *[rows]      (float16) (ignored if HAS_MASK=0)
    rows: tl.int32,
    N: tl.int32,
    H: tl.int32,
    N2: tl.int32,        # N*N  (pre‑computed)
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    For each (b,i,k) row (flattened as `rows`) we compute:
        left  = (proj_left  * mask) * sigmoid(proj_left_gate)
        right = (proj_right * mask) * sigmoid(proj_right_gate)
        out_gate = sigmoid(proj_out_gate)

    Results are stored in the target layout:
        left/right : (B, H, N, N)
        out_gate  : (B, N, N, H)
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)                # [BLOCK_R]
    mask_r = offs_r < rows

    # ---- optional mask (broadcast later) ----
    if HAS_MASK:
        m = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)    # [BLOCK_R]
        m = m[:, None]                                            # [BLOCK_R,1]
    else:
        m = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # ---- decode the flattened row index into (b,i,k) ----
    b = offs_r // N2                         # [BLOCK_R]
    rem = offs_r - b * N2                    # i*N + k
    i = rem // N                             # [BLOCK_R]
    k = rem - i * N                          # [BLOCK_R]   (k = rem % N)

    # ---- iterate over hidden dimension in chunks ----
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)       # [BLOCK_H]
        mask_h = offs_h < H

        # ---- offsets into the fused projection (5*H) ----
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]    # [R, H]

        left_raw   = tl.load(proj_ptr + proj_base,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        right_raw  = tl.load(proj_ptr + proj_base + H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        left_g_raw = tl.load(proj_ptr + proj_base + 2 * H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        right_g_raw= tl.load(proj_ptr + proj_base + 3 * H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        out_g_raw  = tl.load(proj_ptr + proj_base + 4 * H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)

        # ---- sigmoid in fp32 (numerical stability) ----
        left_gate  = (1.0 / (1.0 + tl.exp(-left_g_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_g_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_g_raw.to(tl.float32)))).to(tl.float16)

        # ---- apply mask and gates ----
        left  = left_raw  * m * left_gate
        right = right_raw * m * right_gate

        # ---- compute destination offsets -----------------
        # left/right layout: (B, H, N, N) → offset = ((b*H + h)*N2) + (i*N + k)
        # out_gate layout: (B, N, N, H) → offset = (offs_r * H) + h
        lr_off = ((b[:, None] * H + offs_h[None, :]) * N2) + rem[:, None]   # [R, H]
        og_off = offs_r[:, None] * H + offs_h[None, :]                       # [R, H]

        # ---- store results (respecting masks) ----------
        tl.store(left_ptr  + lr_off, left,  mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + lr_off, right, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + og_off, out_gate, mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Helper: choose reasonable block sizes
# ----------------------------------------------------------------------
def _choose_block_sizes(rows: int, dim: int):
    # a bit larger block over rows reduces kernel launch overhead
    BLOCK_M = 256 if rows >= 256 else max(1, rows)
    BLOCK_D = 32 if dim >= 32 else dim
    return BLOCK_M, BLOCK_D


# ----------------------------------------------------------------------
# Custom forward implementation of the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Args
    ----
    data = (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : mapping with all linear / layer‑norm parameters
                         (all float32, Linear layers have bias=False)
        - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # 0️⃣ unpack
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    nomask = cfg.get("nomask", True)
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                     # total rows for row‑wise ops
    H = hidden_dim
    N2 = N * N

    # ------------------------------------------------------------------
    # 1️⃣ first LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()          # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M, BLOCK_D = _choose_block_sizes(R, dim)
    grid = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],    # γ  (float32)
        weights["norm.bias"],      # β  (float32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # ------------------------------------------------------------------
    # 2️⃣ fused projection: left/right + three gates → (R, 5*H)
    # ------------------------------------------------------------------
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # [5*H, dim]

    proj = F.linear(x_norm, w_fused)         # (R, 5*H), fp16
    del w_fused, x_norm                     # free memory

    # ------------------------------------------------------------------
    # 3️⃣ split, mask, sigmoid – write directly in GEMM layout
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)   # (B,H,N,N)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)   # (B,H,N,N)
    out_gate  = torch.empty(B, N, N, H, dtype=torch.float16, device=device)   # (B,N,N,H)

    if nomask or mask is None:
        HAS_MASK = 0
        dummy_mask = torch.empty(0, dtype=torch.float16, device=device)
        mask_ptr = dummy_mask
    else:
        HAS_MASK = 1
        mask_half = mask.to(torch.float16).view(R)   # flatten to [R]
        mask_ptr = mask_half

    BLOCK_R = 128
    BLOCK_H = 32
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        N,
        H,
        N2,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ batched GEMM over the hidden dimension (no extra copies)
    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)       # (B*H, N, N) – contiguous view
    right_mat = right.view(B * H, N, N)      # (B*H, N, N)

    # L @ Rᵀ  → (B*H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape to (B, N, N, H) without materialising a copy
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1)  # (B, N, N, H)
    del left_mat, right_mat, left, right, out_mat

    # ------------------------------------------------------------------
    # 5️⃣ second LayerNorm + out_gate multiplication (torch.nn.functional)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate                     # element‑wise
    del out_gate

    # ------------------------------------------------------------------
    # 6️⃣ final projection back to `dim`
    # ------------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()   # (R, H) – required for linear
    final_half = F.linear(
        out_flat,
        weights["to_out.weight"].to(torch.float16),   # (dim, H)
    )   # (R, dim) fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
