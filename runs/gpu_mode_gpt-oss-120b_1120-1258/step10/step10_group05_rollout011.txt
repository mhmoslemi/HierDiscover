"""
Highly‑optimized forward TriMul (outgoing) operator.

Algorithm
---------
1️⃣  Row‑wise LayerNorm on the input (float32 → float16) using
    `torch.nn.functional.layer_norm` (fast native kernel).

2️⃣  Fused linear projection that produces the five tensors needed for the
    update (left, right, left_gate, right_gate, out_gate).  The projection is
    performed with a single FP16 GEMM (cuBLAS).

3️⃣  Triton kernel `split_fuse_gate_kernel` splits the fused projection,
    applies the optional pair‑wise mask and the three sigmoid gates, producing
    three FP16 tensors: `left`, `right` and `out_gate`.

4️⃣  Multiplicative update:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    Implemented with a single `torch.einsum` (no layout copies).  This yields
    a Tensor of shape [B, N, N, H] in FP16.

5️⃣  Second LayerNorm (over the hidden dimension) using the native kernel,
    then element‑wise multiplication with `out_gate`.

6️⃣  Final linear projection back to `dim` (FP16 GEMM) and cast to float32.

Only the split‑gate operation is implemented in Triton, satisfying the
requirement that “part of the operations” be a custom kernel.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 3️⃣ Split fused projection, apply mask, apply sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[rows, 5*H]   (fp16)
    left_ptr,          # mutable *[rows, H]    (fp16)
    right_ptr,         # mutable *[rows, H]    (fp16)
    out_gate_ptr,      # mutable *[rows, H]    (fp16)
    mask_ptr,          # const   *[rows]       (fp16) – ignored if HAS_MASK=0
    rows: tl.int32,
    H: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)         # [BLOCK_R]
    mask_r = offs_r < rows

    # ------------------------------------------------------------------
    # optional binary mask (broadcasted over the hidden dim)
    # ------------------------------------------------------------------
    if HAS_MASK:
        m = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        m = m[:, None]                                          # broadcast -> [BLOCK_R,1]
    else:
        m = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # ------------------------------------------------------------------
    # loop over hidden dimension
    # ------------------------------------------------------------------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # layout in the fused projection:
        #   [left, right, left_gate, right_gate, out_gate] each of size H
        base = offs_r[:, None] * (5 * H) + offs_h[None, :]  # [R, H]

        left_raw_ptr    = proj_ptr + base
        right_raw_ptr   = proj_ptr + base + H
        left_gate_ptr   = proj_ptr + base + 2 * H
        right_gate_ptr  = proj_ptr + base + 3 * H
        out_gate_raw_ptr= proj_ptr + base + 4 * H

        left_raw   = tl.load(left_raw_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        right_raw  = tl.load(right_raw_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        left_g_raw = tl.load(left_gate_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        right_g_raw= tl.load(right_gate_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        out_g_raw  = tl.load(out_gate_raw_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)

        # sigmoid (computed in fp32 for stability)
        lg = (1.0 / (1.0 + tl.exp(-left_g_raw.to(tl.float32)))).to(tl.float16)
        rg = (1.0 / (1.0 + tl.exp(-right_g_raw.to(tl.float32)))).to(tl.float16)
        og = (1.0 / (1.0 + tl.exp(-out_g_raw.to(tl.float32)))).to(tl.float16)

        # apply mask and gates
        left  = left_raw  * m * lg
        right = right_raw * m * rg

        # store
        left_dst_ptr     = left_ptr     + offs_r[:, None] * H + offs_h[None, :]
        right_dst_ptr    = right_ptr    + offs_r[:, None] * H + offs_h[None, :]
        out_gate_dst_ptr = out_gate_ptr + offs_r[:, None] * H + offs_h[None, :]

        tl.store(left_dst_ptr,     left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_dst_ptr,    right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_dst_ptr, og,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : dict with the following keys (all float32):
            * "norm.weight", "norm.bias"
            * "left_proj.weight", "right_proj.weight"
            * "left_gate.weight", "right_gate.weight", "out_gate.weight"
            * "to_out_norm.weight", "to_out_norm.bias"
            * "to_out.weight"
        - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim], dtype float32
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    device = x.device
    B, N, _, _ = x.shape
    R = B * N * N                     # total rows for row‑wise ops
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 → fp16) using native kernel
    # ------------------------------------------------------------------
    # torch.nn.functional.layer_norm works on the last dimension; we flatten
    # the first three dimensions to treat each (b,i,j) as a separate row.
    x_norm = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    ).to(torch.float16)                # [B, N, N, dim]  fp16

    # flatten for the fused projection
    x_norm_flat = x_norm.view(R, dim)   # [R, dim] contiguous

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection to obtain 5*H channels (no bias)
    # ------------------------------------------------------------------
    # concatenate the five weight matrices (each [H, dim]) → [5*H, dim]
    fused_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                # [5*H, dim]

    # linear projection (FP16 GEMM)
    proj = F.linear(x_norm_flat, fused_w)   # [R, 5*H]  fp16
    del x_norm_flat, fused_w               # free memory early

    # ------------------------------------------------------------------
    # 3️⃣ Split into left/right/out_gate, apply mask + sigmoid gates
    # ------------------------------------------------------------------
    left      = torch.empty(R, H, dtype=torch.float16, device=device)
    right     = torch.empty(R, H, dtype=torch.float16, device=device)
    out_gate  = torch.empty(R, H, dtype=torch.float16, device=device)

    if cfg.get("nomask", True) or mask is None:
        HAS_MASK = 0
        dummy_mask = torch.empty(0, dtype=torch.float16, device=device)
        mask_ptr = dummy_mask
    else:
        HAS_MASK = 1
        # mask is broadcasted over the hidden dimension; 1 → keep, 0 → discard
        mask_ptr = mask.to(torch.float16).reshape(R)

    BLOCK_R = 128                     # rows per program
    BLOCK_H = 32                      # hidden‑dim tiles
    grid = (triton.cdiv(R, BLOCK_R),)

    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, mask_ptr                      # free

    # ------------------------------------------------------------------
    # 4️⃣ Tri‑multiplicative update: einsum over the sequence dimension k
    # ------------------------------------------------------------------
    # reshape to 4‑D tensors without any transposes (no extra copies)
    left_4d  = left.view(B, N, N, H)    # [B, N, N, H]
    right_4d = right.view(B, N, N, H)   # [B, N, N, H]

    # out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    out_4d = torch.einsum('b i k d, b j k d -> b i j d', left_4d, right_4d)  # fp16
    del left, right                         # free

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm (over hidden dimension) + out‑gate multiplication
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(
        out_4d,
        (H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )                                       # fp16, same shape [B,N,N,H]

    out_gate_4d = out_gate.view(B, N, N, H)
    out_gated = out_norm * out_gate_4d    # element‑wise fp16
    del out_4d, out_norm, out_gate, out_gate_4d  # free

    # ------------------------------------------------------------------
    # 6️⃣ Final projection back to `dim` and cast to float32
    # ------------------------------------------------------------------
    out_flat = out_gated.view(R, H)                # [R, H]
    final_half = F.linear(
        out_flat,
        weights["to_out.weight"].to(torch.float16),   # [dim, H]
        bias=None,
    )                                                # [R, dim] fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
