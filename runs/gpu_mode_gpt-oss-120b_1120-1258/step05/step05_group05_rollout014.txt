def custom_kernel(data):
    input, mask, weights, config = data
    device = input.device
    B,N,_,dim = input.shape
    total_rows = B*N*N
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # 1. Compute per-row mean and rstd of input (x)
    mean = torch.empty(total_rows, dtype=torch.float32, device=device)
    rstd = torch.empty(total_rows, dtype=torch.float32, device=device)

    BLOCK_M = 256
    BLOCK_D = 64
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    row_stats_kernel[grid](
        input.view(total_rows, dim), 
        mean,
        rstd,
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # 2. Prepare scaled weights and constants
    # Get norm weight gamma and bias beta
    gamma = weights["norm.weight"]  # shape [dim]
    beta = weights["norm.bias"]     # shape [dim]

    # Concatenate weights for projections
    left_w = weights["left_proj.weight"] # shape [hidden_dim, dim]  (Wy as? Actually torch Linear expects weight shape [out_features, in_features]
    right_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    concat_w = torch.cat([left_w, right_w, left_gate_w, right_gate_w, out_gate_w], dim=0)  # shape [5*hidden_dim, dim]
    # Scale by gamma
    concat_w_scaled = (concat_w * gamma[None, :]).to(torch.float16)  # half

    # Precompute weight_sum_scaled and bias_const (beta @ weight)
    # weight_sum_scaled = concat_w_scaled.sum(dim=1).float()
    weight_sum_scaled = concat_w_scaled.to(torch.float32).sum(dim=1)  # shape [5*H]
    # bias_const = (concat_w * beta[None,:]).sum(dim=1).float()
    bias_const = (concat_w * beta[None, :]).to(torch.float32).sum(dim=1)  # measure

    # 3. GEMM
    x_half = input.to(torch.float16).view(total_rows, dim)   # we can keep original as half, maybe input is float32; cast to half.
    proj_all = torch.nn.functional.linear(x_half, concat_w_scaled) # shape [total_rows, 5*H] half
    proj_all_f32 = proj_all.to(torch.float32)

    # 4. Apply LN scaling and bias (normalize)
    # out = (proj_all_f32 - mean[:,None]*weight_sum_scaled[None,:]) * rstd[:,None] + bias_const[None,:]
    # broadcasting requires same dtype; we compute as float.
    proj_all_norm = (proj_all_f32 - mean.unsqueeze(1) * weight_sum_scaled) * rstd.unsqueeze(1) + bias_const

    # split
    left_proj, right_proj, left_gate_pre, right_gate_pre, out_gate_pre = torch.split(proj_all_norm, hidden_dim, dim=1)

    # convert to half for further
    left_proj_fp16 = left_proj.to(torch.float16)
    right_proj_fp16 = right_proj.to(torch.float16)
    left_gate_fp16 = left_gate_pre.to(torch.float16)
    right_gate_fp16 = right_gate_pre.to(torch.float16)
    out_gate_fp16 = out_gate_pre.to(torch.float16)

    # 5. Apply mask if needed
    if not config.get("nomask", True):
        mask_fp16 = mask.view(total_rows, 1).to(torch.float16)
        left_proj_fp16 = left_proj_fp16 * mask_fp16
        right_proj_fp16 = right_proj_fp16 * mask_fp16

    # 6. Gating: sigmoid
    left_gate = torch.sigmoid(left_gate_fp16)
    right_gate = torch.sigmoid(right_gate_fp16)
    out_gate = torch.sigmoid(out_gate_fp16)

    left = left_proj_fp16 * left_gate
    right = right_proj_fp16 * right_gate

    # Reshape to 4D then permute
    left = left.view(B, N, N, hidden_dim).permute(0, 3, 1, 2).contiguous()
    right = right.view(B, N, N, hidden_dim).permute(0, 3, 1, 2).contiguous()
    out_gate = out_gate.view(B, N, N, hidden_dim)

    # 7. Triangular multiplication
    BH = B * hidden_dim
    left_mat = left.view(BH, N, N)
    right_mat = right.view(BH, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1,2))
    out = out_mat.view(B, hidden_dim, N, N).permute(0,2,3,1).contiguous()

    # 8. Second LN + gating fused kernel
    # We need to compute mean/rstd per row of out (float16) and apply LN + multiply out_gate
    # Let's write kernel layernorm_gate_kernel
    out_flat = out.view(total_rows, hidden_dim)  # half
    # allocate buffer gated_norm_fp16
    gated_norm_fp16 = torch.empty_like(out_flat, dtype=torch.float16)

    # precompute gamma2 (to_out_norm.weight) and beta2 (to_out_norm.bias)
    gamma2 = weights["to_out_norm.weight"]
    beta2 = weights["to_out_norm.bias"]
    # Launch kernel
    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_gate_kernel[grid2](
        out_flat,       # input half
        out_gate.view(total_rows, hidden_dim),   # out_gate half
        gated_norm_fp16, # output half
        gamma2,
        beta2,
        total_rows,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # 9. Final linear projection
    to_out_w_fp16 = weights["to_out.weight"].half()
    final_fp16 = torch.nn.functional.linear(gated_norm_fp16.view(total_rows, hidden_dim), to_out_w_fp16)

    # 10. Cast to float32 shape [B,N,N,dim]
    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final

@triton.jit
def row_stats_kernel(x_ptr, mean_ptr, rstd_ptr, total_rows, dim: tl.constexpr, eps: tl.constexpr, BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)  # [BLOCK_M]
    mask_m = offs_m < total_rows

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # store
    tl.store(mean_ptr + offs_m, mean, mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)

@triton.jit
def layernorm_gate_kernel(
    x_ptr, gate_ptr, out_ptr,
    gamma_ptr, beta_ptr,
    total_rows, dim: tl.constexpr, eps: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # compute sum & sum_sq for x
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    # we will also need to load gate values later; but we can compute them after mean/rstd available.
    # We'll compute per block per element later in second loop
    # First pass: compute sums
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        x_ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(x_ptrs, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)  # [BLOCK_M]

    # second loop: normalize, apply gamma, beta, gate
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        # load x
        x_ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        x_vals = tl.load(x_ptrs, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)

        # load gate
        gate_ptrs = gate_ptr + offs_m[:, None] * dim + offs_d[None, :]
        gate_vals = tl.load(gate_ptrs, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)

        # load gamma and beta
        gamma = tl.load(gamma_ptr + offs_d, mask=mask_d, other=0.0)  # float32
        beta = tl.load(beta_ptr + offs_d, mask=mask_d, other=0.0)    # float32

        # Normalize: (x - mean) * rstd
        norm = (x_vals - mean[:, None]) * rstd[:, None]  # shape [BLOCK_M, BLOCK_D]

        # Apply LN scaling & bias: norm * gamma + beta
        ln_out = norm * gamma[None, :] + beta[None, :]   # shape [BLOCK_M, BLOCK_D]

        # Multiply by gate
        out_vals = ln_out * gate_vals

        # Store to out
        out_ptrs = out_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_vals.to(tl.float16), mask=mask_m[:, None] & mask_d[None, :])

"""
Fast outgoing TriMul (AlphaFold‑3) forward pass.

Algorithm
---------
1. Compute row‑wise mean and 1/√var (rstd) of the input tensor `x`
   with a tiny Triton kernel (no output tensor is written).
2. Fuse the LayerNorm scaling into the linear projections:
   * Scale each weight matrix by `norm.weight` (γ) → `Ŵ`.
   * Pre‑compute for every output dimension
        sum_w = Σc Ŵc      (used for the mean term)
        bias_c = Σc βc·Wc   (β = `norm.bias`, *no* γ scaling)
3. Perform a single half‑precision GEMM
        p = x @ Ŵᵀ                     (shape [R,5·H])
   and transform it to the Layer‑Normed projections:
        p = (p - mean·sum_w)·rstd + bias_c
   (R = B·N², H = hidden_dim).
4. Split the five tensors, apply the (optional) mask,
   apply sigmoid gates and element‑wise gating.
5. Triangular multiplicative update:
        out[i,j] = Σk left[i,k]·right[j,k]
   implemented as a batched GEMM `left @ rightᵀ`.
6. Fuse the second LayerNorm (`to_out_norm`) with the gating
   `out_gate` in a second Triton kernel.
7. Final half‑precision linear projection back to `dim`.
8. Cast the result to FP32.

Both custom kernels run once per forward pass; the heavy
matrix‑multiplications rely on cuBLAS Tensor‑Core kernels,
which is optimal on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Kernel 1: row‑wise mean and 1/√var (rstd) – no output tensor
# ----------------------------------------------------------------------
@triton.jit
def row_stats_kernel(
    x_ptr,                # *const   half*   [total_rows, dim]
    mean_ptr,             # *mut      float* [total_rows]
    rstd_ptr,             # *mut      float* [total_rows]
    total_rows,
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # dimension chunk size
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)        # [BLOCK_M]
    mask_m = offs_m < total_rows

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    # ----- reduction over the feature dimension -----
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)  # [BLOCK_D]
        mask_d = offs_d < dim

        # load a BLOCK_D slice of the row
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)       # fp32

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                         # 1/√(var+eps)

    # ----- write results -------------------------------------------------
    tl.store(mean_ptr + offs_m, mean, mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)


# ----------------------------------------------------------------------
# Kernel 2: LayerNorm + gating (used after the triangular update)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_gate_kernel(
    x_ptr,               # *const half*   [R, H]
    gate_ptr,            # *const half*   [R, H]
    out_ptr,             # *mut   half*   [R, H]
    gamma_ptr,           # *const float*  [H]   (to_out_norm.weight)
    beta_ptr,            # *const float*  [H]   (to_out_norm.bias)
    total_rows,
    dim: tl.constexpr,   # H
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)        # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ---- first pass: compute mean & rstd -------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        x_ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(x_ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                         # [BLOCK_M]

    # ---- second pass: normalize, apply γ/β and the gate -----------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        # load x, gate, γ, β
        x_ptrs   = x_ptr   + offs_m[:, None] * dim + offs_d[None, :]
        g_ptrs   = gate_ptr + offs_m[:, None] * dim + offs_d[None, :]
        x_vals   = tl.load(x_ptrs,
                           mask=mask_m[:, None] & mask_d[None, :],
                           other=0.0).to(tl.float32)   # fp32
        gate_vals = tl.load(g_ptrs,
                           mask=mask_m[:, None] & mask_d[None, :],
                           other=0.0).to(tl.float32)   # fp32
        gamma = tl.load(gamma_ptr + offs_d, mask=mask_d, other=0.0)   # fp32
        beta  = tl.load(beta_ptr  + offs_d, mask=mask_d, other=0.0)   # fp32

        # (x‑mean)·rstd
        norm = (x_vals - mean[:, None]) * rstd[:, None]         # fp32
        # LayerNorm:  norm·γ + β
        ln_out = norm * gamma[None, :] + beta[None, :]          # fp32
        # multiply by the element‑wise gate
        out_vals = ln_out * gate_vals

        # store result as half
        out_ptrs = out_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs,
                 out_vals.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Main entry‑point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul operator.

    Args:
        data : tuple (input, mask, weights, config)
            input   – Tensor[bs, N, N, C]  (float32)
            mask    – Tensor[bs, N, N]    (bool/float)  or None
            weights – dict of model weights (float32)
            config  – {"dim": C, "hidden_dim": H, "nomask": bool}
    Returns:
        Tensor[bs, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, N, _, C = input_tensor.shape
    H = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 0) prepare half‑precision input and flatten
    # ------------------------------------------------------------------
    x_fp16 = input_tensor.to(torch.float16).contiguous()
    total_rows = B * N * N
    x_flat = x_fp16.view(total_rows, C)                # [R, C]

    # ------------------------------------------------------------------
    # 1) row‑wise mean / rstd of the original tensor (no write‑back)
    # ------------------------------------------------------------------
    mean = torch.empty(total_rows, dtype=torch.float32, device=device)
    rstd = torch.empty(total_rows, dtype=torch.float32, device=device)

    BLOCK_M = 512
    BLOCK_D = 64
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    row_stats_kernel[grid](
        x_flat,
        mean,
        rstd,
        total_rows,
        C,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 2) Fuse LayerNorm into the five linear projections
    # ------------------------------------------------------------------
    # 2.1 γ, β from the first LayerNorm
    gamma = weights["norm.weight"]          # [C]  fp32
    beta  = weights["norm.bias"]           # [C]  fp32

    # 2.2 concatenate raw weights (bias=False in all Linear modules)
    w_left    = weights["left_proj.weight"]   # [H, C]
    w_right   = weights["right_proj.weight"]
    w_lgate   = weights["left_gate.weight"]
    w_rgate   = weights["right_gate.weight"]
    w_outgate = weights["out_gate.weight"]

    concat_w = torch.cat([w_left, w_right, w_lgate, w_rgate, w_outgate], dim=0)   # [(5*H), C]

    # 2.3 scale by γ (LayerNorm scale) – this is the part that multiplies (x‑mean)
    w_scaled = (concat_w * gamma[None, :]).to(torch.float16)   # [(5*H), C]  fp16

    # 2.4 pre‑compute per‑output constants
    #     sum_w = Σc (γc·Wc)      – needed for the mean term
    #     bias  = Σc (βc·Wc)      – constant after LayerNorm
    sum_w_fp32   = w_scaled.to(torch.float32).sum(dim=1)          # [(5*H)]  fp32
    bias_fp32    = (concat_w * beta[None, :]).to(torch.float32).sum(dim=1)  # [(5*H)]

    # 2.5 GEMM: X @ Ŵᵀ  (half‑precision, fully‑fused)
    proj_all = F.linear(x_flat, w_scaled)            # [R, 5*H]  fp16

    # 2.6 Apply the (mean, rstd) correction and add bias
    #      proj = (proj - mean·sum_w)·rstd + bias
    mean_fp16 = mean.unsqueeze(1).to(torch.float16)   # [R,1]
    rstd_fp16 = rstd.unsqueeze(1).to(torch.float16)   # [R,1]
    sum_w_fp16 = sum_w_fp32.unsqueeze(0).to(torch.float16)   # [1,5*H]
    bias_fp16  = bias_fp32.unsqueeze(0).to(torch.float16)    # [1,5*H]

    proj_all = (proj_all - mean_fp16 * sum_w_fp16) * rstd_fp16 + bias_fp16   # still fp16

    # ------------------------------------------------------------------
    # 3) Split the five tensors
    # ------------------------------------------------------------------
    left_proj, right_proj, left_gate_pre, right_gate_pre, out_gate_pre = torch.split(
        proj_all, H, dim=1
    )                                    # each [R, H] fp16

    # ------------------------------------------------------------------
    # 4) Optional mask (applied before gating)
    # ------------------------------------------------------------------
    if (not nomask) and (mask_tensor is not None):
        mask_fp16 = mask_tensor.reshape(total_rows, 1).to(torch.float16)
        left_proj  = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # ------------------------------------------------------------------
    # 5) Sigmoid gates and element‑wise gating
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate_pre.float()).half()
    right_gate = torch.sigmoid(right_gate_pre.float()).half()
    out_gate   = torch.sigmoid(out_gate_pre.float()).half()

    left  = left_proj * left_gate                     # [R, H]
    right = right_proj * right_gate                   # [R, H]

    # ------------------------------------------------------------------
    # 6) Reshape to 4‑D tensors and prepare for batched GEMM
    # ------------------------------------------------------------------
    left  = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    BH = B * H
    left_mat  = left.view(BH, N, N)               # [BH, N, N]
    right_mat = right.view(BH, N, N)              # [BH, N, N]

    # ------------------------------------------------------------------
    # 7) Triangular multiplicative update (outgoing version)
    # ------------------------------------------------------------------
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [BH, N, N]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 8) Fuse second LayerNorm (to_out_norm) with the gating
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, H)                # [R, H] fp16
    out_gate_flat = out_gate.view(total_rows, H)      # [R, H] fp16

    gated_norm = torch.empty_like(out_flat, dtype=torch.float16)

    gamma2 = weights["to_out_norm.weight"]   # [H] fp32
    beta2  = weights["to_out_norm.bias"]    # [H] fp32

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_gate_kernel[grid2](
        out_flat,
        out_gate_flat,
        gated_norm,
        gamma2,
        beta2,
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 9) Final linear projection back to the original channel dimension
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(torch.float16)   # [C, H]
    final_fp16 = F.linear(gated_norm.view(total_rows, H), to_out_w)   # [R, C] fp16

    # ------------------------------------------------------------------
    # 10) Cast back to FP32 and restore [B, N, N, C] layout
    # ------------------------------------------------------------------
    final = final_fp16.view(B, N, N, C).to(torch.float32)
    return final
