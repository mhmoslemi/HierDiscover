def final_ln_gate_linear_kernel(
    src_ptr,         # fp16 [B,N,N,H]
    gate_ptr,        # fp16 [R,H]
    dst_ptr,         # fp32 [R, dim]
    ln_weight_ptr,   # fp32 [H]
    ln_bias_ptr,     # fp32 [H]
    w_out_ptr,       # fp32 [H, dim] (transposed weight)
    total_rows,      # int
    hidden_dim,      # int (compile-time)
    seq_len,         # int
    dim,             # int (compile-time)
    eps,             # float
    stride_b, stride_i, stride_j, stride_h,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # decode (b,i,j)
    seq_len_sq = seq_len * seq_len
    b_idx = offs_m // seq_len_sq
    rem = offs_m % seq_len_sq
    i_idx = rem // seq_len
    j_idx = rem % seq_len
    base_off = b_idx * stride_b + i_idx * stride_i + j_idx * stride_j

    # pass 1: compute mean/var over hidden dim H
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)
    for d in range(0, tl.cdiv(hidden_dim, BLOCK_D)):
        offs_h = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_h = offs_h < hidden_dim
        # load src tile
        ptr = src_ptr + base_off[:, None] + offs_h[None, :] * stride_h
        val = tl.load(ptr, mask=mask_m[:, None] & mask_h[None, :], other=0.0).to(tl.float32)
        sum_val += tl.sum(val, axis=1)
        sum_sq += tl.sum(val * val, axis=1)

    mean = sum_val / hidden_dim
    var = sum_sq / hidden_dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # pass 2: LN, gate, and linear combine
    # accumulator for output (BLOCK_M, dim)
    out_acc = tl.zeros([BLOCK_M, dim], dtype=tl.float32)

    for d in range(0, tl.cdiv(hidden_dim, BLOCK_D)):
        offs_h = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_h = offs_h < hidden_dim

        # Load raw hidden
        ptr = src_ptr + base_off[:, None] + offs_h[None, :] * stride_h
        raw = tl.load(ptr, mask=mask_m[:, None] & mask_h[None, :], other=0.0).to(tl.float32)

        # Normalization
        norm = (raw - mean[:, None]) * rstd[:, None]  # [BLOCK_M, tile_h]

        # Load LN weights/bias
        w_ln = tl.load(ln_weight_ptr + offs_h, mask=mask_h, other=0.0)  # [tile_h]
        b_ln = tl.load(ln_bias_ptr + offs_h, mask=mask_h, other=0.0)    # [tile_h]

        hidden = norm * w_ln[None, :] + b_ln[None, :]   # fp32

        # Apply out_gate
        gate = tl.load(gate_ptr + offs_m[:, None] * hidden_dim + offs_h[None, :],
                       mask=mask_m[:, None] & mask_h[None, :],
                       other=0.0).to(tl.float32)
        hidden = hidden * gate

        # Load weight tile (tile_h, dim)
        w_ptr = w_out_ptr + offs_h[:, None] * dim + tl.arange(0, dim)[None, :]
        w_tile = tl.load(w_ptr,
                         mask=mask_h[:, None] & (tl.arange(0, dim)[None, :] < dim),
                         other=0.0).to(tl.float32)

        # Accumulate dot product
        out_acc += tl.dot(hidden, w_tile)   # (BLOCK_M, dim)

    # Store final output
    dst_ptrs = dst_ptr + offs_m[:, None] * dim + tl.arange(0, dim)[None, :]
    tl.store(dst_ptrs, out_acc, mask=mask_m[:, None] & (tl.arange(0, dim)[None, :] < dim))

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    device = input_tensor.device

    total_rows = B * N * N

    # Layernorm first
    x_flat = input_tensor.view(total_rows, dim).contiguous()
    x_norm = torch.empty(total_rows, dim, dtype=torch.float16, device=device)

    BLOCK_M = 256
    BLOCK_D = 64

        x_flat,
        x_norm,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,

        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    right_gate = torch.sigmoid(right_gate)

Outgoing TriMul (AlphaFold‑3) – Triton‑accelerated forward pass
----------------------------------------------------------------
The operator works on a 4‑D tensor [B, N, N, C] and consists of:
1️⃣  Row‑wise LayerNorm on the input (FP32 → FP16) – Triton.
2️⃣  A single fused linear projection that produces the three

3️⃣  Optional pair‑wise mask applied to left/right projections.
4️⃣  Apply sigmoid‑gates → gated tensors *left* and *right*.
5️⃣  Pairwise multiplication across the sequence dimension:
        out[b,i,j,h] = Σₖ left[b,i,k,h]·right[b,j,k,h]

6️⃣  Second LayerNorm + out‑gate (both per (b,i,j) row) fused into a

7️⃣  Final linear projection back to the model dimension (FP16 GEMM)

All heavy work (LayerNorms, the pairwise multiplication and the final

def layernorm_kernel(
    x_ptr,            # const   *  [total_rows, dim]   (fp32)
    y_ptr,            # mutable *  [total_rows, dim]   (fp16)
    weight_ptr,       # const   *  [dim]               (fp32)
    bias_ptr,         # const   *  [dim]               (fp32)
    total_rows,       # int     (runtime)
    dim,              # int     (runtime)
    eps,              # float   (compile‑time)
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]

        sum_sq  += tl.sum(val * val, axis=1)

        out = norm * w[None, :] + b[None, :]                # fp32

        # out‑gate (fp16 → fp32)
        gate = tl.load(gate_ptr + offs_m[:, None] * hidden_dim + offs_h[None, :],
                       mask=mask_m[:, None] & mask_h[None, :],
                       other=0.0).to(tl.float32)

        out = out * gate

        # store as fp16
        out_fp16 = out.to(tl.float16)
        dst_ptrs = dst_ptr + offs_m[:, None] * hidden_dim + offs_h[None, :]
        tl.store(dst_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_h[None, :])

# ---------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ---------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [B, N, N, dim] (float32, on the same CUDA device)
    """
    # --------------------------------------------------------------
    # 0) unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim        = config["dim"]           # model dimension (e.g., 128 or 384)
    hidden_dim = config["hidden_dim"]   # hidden dimension (always 128 in tests)
    eps        = 1e-5
    device     = input_tensor.device
    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N

    # --------------------------------------------------------------
    # 1) First LayerNorm – output in fp16
    # --------------------------------------------------------------
    x_flat = input_tensor.view(total_rows, dim).contiguous()
    x_norm = torch.empty(total_rows, dim, dtype=torch.float16, device=device)

    BLOCK_M = 256
    BLOCK_D = 64
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # --------------------------------------------------------------
    # 2) Fused projection producing left/right + three gates
    # --------------------------------------------------------------
    # concatenate the five weight matrices (dim → 5·hidden_dim)
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # shape (5*H, dim)

    proj_out = F.linear(x_norm, w_concat)    # [R, 5·H]  (fp16)

    H = hidden_dim
    left_proj  = proj_out[:, 0:H]
    right_proj = proj_out[:, H:2*H]
    left_gate  = proj_out[:, 2*H:3*H]
    right_gate = proj_out[:, 3*H:4*H]
    out_gate   = proj_out[:, 4*H:5*H]

    # sigmoid gates (still fp16)
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    # --------------------------------------------------------------
    # 3) Optional pair‑wise mask (applied only to the two projections)
    # --------------------------------------------------------------
    if not config.get("nomask", True):
        mask_h = mask_tensor.to(torch.float16).view(total_rows, 1)
        left_proj  = left_proj * mask_h
        right_proj = right_proj * mask_h

    # --------------------------------------------------------------
    # 4) Apply gates → gated tensors `left` and `right`
    # --------------------------------------------------------------
    left  = left_proj * left_gate      # [R, H]
    right = right_proj * right_gate    # [R, H]

    left  = left.view(B, N, N, H)      # (B, N, N, H)
    right = right.view(B, N, N, H)     # (B, N, N, H)

    # --------------------------------------------------------------
    # 5) Pairwise multiplication (einsum) – result (B, N, N, H)
    # --------------------------------------------------------------
    out = torch.einsum('b i k h, b j k h -> b i j h', left, right).contiguous()
    # out is fp16, layout (B, N, N, H) with H as the innermost dimension

    # --------------------------------------------------------------
    # 6) Second LayerNorm + out‑gate (fused Triton kernel)
    #    Output is a contiguous (R, H) fp16 tensor `hidden_out`.
    # --------------------------------------------------------------
    # strides of `out` (contiguous layout)
    stride_b = out.stride(0)   # B‑stride
    stride_i = out.stride(1)   # i‑stride
    stride_j = out.stride(2)   # j‑stride
    # hidden stride is 1 (last dimension)

    hidden_out = torch.empty(total_rows, H, dtype=torch.float16, device=device)

    # launch kernel
    BLOCK_M2 = 128   # rows per program (kept moderate)
    BLOCK_D2 = 64    # hidden‑dim tile
    grid2 = (triton.cdiv(total_rows, BLOCK_M2),)
    fused_ln_gate_contig_kernel[grid2](
        out,                                 # src (B,N,N,H) fp16
        out_gate,                            # gate (R,H) fp16
        hidden_out,                          # dst (R,H) fp16
        weights["to_out_norm.weight"],       # LN γ   (fp32)
        weights["to_out_norm.bias"],         # LN β   (fp32)
        total_rows,
        H,
        N,
        eps,
        stride_b,
        stride_i,
        stride_j,
        BLOCK_M=BLOCK_M2,
        BLOCK_D=BLOCK_D2,
    )

    # --------------------------------------------------------------
    # 7) Final projection back to `dim` (FP16 GEMM) → cast to FP32
    # --------------------------------------------------------------
    w_out = weights["to_out.weight"].to(torch.float16)   # shape (dim, H)
    final_half = F.linear(hidden_out, w_out)            # [R, dim] fp16
    final = final_half.view(B, N, N, dim).to(torch.float32)

    return final
