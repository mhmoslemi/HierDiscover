"""
Highly‑optimized Triton + PyTorch forward for the **outgoing** TriMul operator
(AlphaFold‑3).

Algorithm
---------
1️⃣  Row‑wise LayerNorm on the input (FP32 reduction → FP16 output) – Triton.
2️⃣  Fuse the five linear projections (left/right + three gates) into a single
    FP16 GEMM (torch.nn.functional.linear).  This is the dominant FLOP count.
3️⃣  (Optional) apply the binary mask to the left/right projections.
4️⃣  Apply sigmoid on the three gate tensors and fuse the element‑wise gating.
5️⃣  Triangular multiplicative update:
    `out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]`
    performed with a batched half‑precision matrix‑multiply (`torch.bmm`).
6️⃣  Fused per‑row LayerNorm + out‑gate + final linear projection (all FP32) –
    Triton kernel.  This eliminates two extra memory passes.

All tensors stay on‑device; the final result is returned as FP32
with shape `[B, N, N, dim]`.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1) Row‑wise LayerNorm (FP32 reduction → FP16 output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,               # *const float*  [total_rows, dim]   input (FP32)
    y_ptr,               # *mut   half*   [total_rows, dim]   output (FP16)
    weight_ptr,          # *const float*  [dim]               γ
    bias_ptr,            # *const float*  [dim]               β
    total_rows: tl.int32,
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,      # rows per program
    BLOCK_D: tl.constexpr,      # columns per block
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)         # [BLOCK_M]
    mask_m = offs_m < total_rows

    # -----------------------------------------------------------------
    # Pass‑1: compute mean & variance (FP32 accumulation)
    # -----------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    num_d_blocks = tl.cdiv(dim, BLOCK_D)
    for db in range(num_d_blocks):
        offs_d = db * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)       # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                          # 1/σ

    # -----------------------------------------------------------------
    # Pass‑2: normalise, apply γ,β and store as FP16
    # -----------------------------------------------------------------
    for db in range(num_d_blocks):
        offs_d = db * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]   # (x-μ)/σ

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]                     # FP32

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 6) Fused LayerNorm + out‑gate + final linear projection (FP32 output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_gate_linear_kernel(
    x_ptr,               # *const half*  [total_rows, hidden]  pre‑norm (FP16)
    gate_ptr,            # *const half*  [total_rows, hidden]  out_gate (sigmoid, FP16)
    norm_w_ptr,          # *const float* [hidden]                γ of to_out_norm
    norm_b_ptr,          # *const float* [hidden]                β of to_out_norm
    out_w_ptr,           # *const half*  [dim, hidden]           final linear weight
    out_ptr,             # *mut   float* [total_rows, dim]       result (FP32)
    total_rows: tl.int32,
    dim: tl.constexpr,
    hidden: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # output‑channel block size
):
    pid = tl.program_id(0)          # row block index
    pid_d = tl.program_id(1)        # output‑channel block index
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)               # [BLOCK_M]
    mask_m = offs_m < total_rows

    # -----------------------------------------------------------------
    # Load full hidden vector for each selected row
    # -----------------------------------------------------------------
    offs_h = tl.arange(0, hidden)                           # [hidden]
    mask_h = offs_h < hidden

    # x and gate are stored as half → cast to FP32 for arithmetic
    x = tl.load(x_ptr + offs_m[:, None] * hidden + offs_h[None, :],
                mask=mask_m[:, None] & mask_h[None, :],
                other=0.0).to(tl.float32)                  # (BLOCK_M, hidden)
    g = tl.load(gate_ptr + offs_m[:, None] * hidden + offs_h[None, :],
                mask=mask_m[:, None] & mask_h[None, :],
                other=0.0).to(tl.float32)                  # (BLOCK_M, hidden)

    # -----------------------------------------------------------------
    # Per‑row mean & rstd (FP32)
    # -----------------------------------------------------------------
    sum_val = tl.sum(x, axis=1)                           # (BLOCK_M)
    sum_sq  = tl.sum(x * x, axis=1)                       # (BLOCK_M)
    mean = sum_val / hidden
    var  = sum_sq / hidden - mean * mean
    rstd = tl.rsqrt(var + eps)

    # -----------------------------------------------------------------
    # Apply LayerNorm weight/bias and out‑gate
    # -----------------------------------------------------------------
    w_norm = tl.load(norm_w_ptr + offs_h, mask=mask_h, other=0.0)   # γ
    b_norm = tl.load(norm_b_ptr + offs_h, mask=mask_h, other=0.0)   # β

    norm = (x - mean[:, None]) * rstd[:, None]                     # (BLOCK_M, hidden)
    norm = norm * w_norm[None, :] + b_norm[None, :]                # (BLOCK_M, hidden)
    gated = norm * g                                               # (BLOCK_M, hidden)

    # -----------------------------------------------------------------
    # Final linear projection (hidden → dim)
    # -----------------------------------------------------------------
    d_start = pid_d * BLOCK_D
    offs_d = d_start + tl.arange(0, BLOCK_D)                       # [BLOCK_D]
    mask_d = offs_d < dim

    # Load weight slice: original weight shape [dim, hidden];
    # we need its transpose (hidden, BLOCK_D)
    out_w_T = tl.load(out_w_ptr + offs_h[:, None] + offs_d[None, :] * hidden,
                      mask=mask_h[:, None] & mask_d[None, :],
                      other=0.0).to(tl.float32)                  # (hidden, BLOCK_D)

    # (BLOCK_M, hidden) @ (hidden, BLOCK_D) → (BLOCK_M, BLOCK_D)
    out_chunk = tl.dot(gated, out_w_T)                            # (BLOCK_M, BLOCK_D)

    tl.store(out_ptr + offs_m[:, None] * dim + offs_d[None, :],
             out_chunk,
             mask=mask_m[:, None] & mask_d[None, :])

    ).half().contiguous()                                 # (5*H, C)

    # Linear: (R, C) → (R, 5*H) in FP16
    proj_all = F.linear(x_norm_fp16, concat_w)             # (R, 5*H) FP16

    # Split back to the five tensors (each (R, H))
    left_proj_fp16, right_proj_fp16, left_gate_fp16, \
    right_gate_fp16, out_gate_fp16 = torch.split(proj_all, hidden, dim=1)

    # ------------------------------------------------------------------
    # 2a) Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask_flat = mask.view(total_rows, 1).to(torch.float16)
        left_proj_fp16 = left_proj_fp16 * mask_flat
        right_proj_fp16 = right_proj_fp16 * mask_flat

    # ------------------------------------------------------------------
    # 2b) Sigmoid gates + element‑wise gating
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_fp16)              # (R, H)
    right_gate = torch.sigmoid(right_gate_fp16)            # (R, H)
    out_gate = torch.sigmoid(out_gate_fp16)                # (R, H) – later used

    left = left_proj_fp16 * left_gate                      # (R, H)
    right = right_proj_fp16 * right_gate                   # (R, H)

    # ------------------------------------------------------------------
    # 3️⃣ Triangular multiplicative update (batched matmul)
    # ------------------------------------------------------------------
    # Reshape to (B, N, N, H)
    left = left.view(B, N, N, hidden)
    right = right.view(B, N, N, hidden)

    # Prepare batched matrices: (B*H, N, N)
    left_mat = left.permute(0, 3, 1, 2).reshape(B * hidden, N, N)   # rows=i, cols=k
    right_mat = right.permute(0, 3, 2, 1).reshape(B * hidden, N, N) # rows=k, cols=j

    out_mat = torch.bmm(left_mat, right_mat)            # (B*H, N, N) FP16

    # Back‑to (B, N, N, H)
    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 4️⃣ Fused LayerNorm + out‑gate + final projection
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, hidden).contiguous()    # (R, H) FP16
    final_fp32 = torch.empty(total_rows, dim,
                             dtype=torch.float32,
                             device=device)

    grid_out = (
        triton.cdiv(total_rows, BLOCK_M),
        triton.cdiv(dim, BLOCK_D),
    )
    layernorm_gate_linear_kernel[grid_out](
        out_flat,                                          # x_ptr
        out_gate,                                          # gate_ptr (sigmoid‑ed)
        weights["to_out_norm.weight"],                     # γ
        weights["to_out_norm.bias"],                       # β
        weights["to_out.weight"].half().contiguous(),     # (dim, hidden)
        final_fp32,                                        # output buffer (FP32)
        total_rows,
        dim=dim,
        hidden=hidden,
        eps=eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 5️⃣ Reshape to final tensor shape
    # ------------------------------------------------------------------
    final = final_fp32.view(B, N, N, dim)                 # [B, N, N, dim] FP32
    return final
