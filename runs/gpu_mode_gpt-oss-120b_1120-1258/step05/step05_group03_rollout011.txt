import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# LayerNorm (FP16 output) kernel
@triton.jit
def layernorm_fp16_kernel(
    x_ptr, y_ptr,
    weight_ptr, bias_ptr,
    total_rows: tl.int32,
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M) # [BLOCK_M]
    mask_m = offs_m < total_rows

    # pass 1: compute mean and variance
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32) # [BLOCK_M, BLOCK_D]
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # pass 2: normalize, apply LN weight & bias, store FP16
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        norm = (vals - mean[:, None]) * rstd[:, None]  # (x-µ)/σ
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # β
        out = norm * w[None, :] + b[None, :]  # FP32
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])

def fused_layernorm_gate_kernel(
    x_ptr, y_ptr,
    weight_ptr, bias_ptr,
    gate_ptr,

def custom_kernel(data):
    """
    Forward pass for the *outgoing* TriMul operator (AlphaFold‑3 style).
    The implementation fuses the input LayerNorm, the five linear
    projections (left_proj, left_gate, right_proj, right_gate, out_gate),
    and the second LayerNorm+out_gate into custom Triton kernels, while
    the heavy pairwise update and the final projection reuse highly‑
    optimized Torch GEMM (FP16 Tensor‑Core) calls.
    """
    inp, mask, weights, cfg = data
    device = inp.device
    dim = cfg["dim"]
    hidden = cfg["hidden_dim"]
    eps = cfg.get("eps", 1e-5)
    nomask = cfg.get("nomask", True)

    B, N, _, _ = inp.shape
    total_rows = B * N * N

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP16)
    # ------------------------------------------------------------------
    inp_flat = inp.view(total_rows, dim).contiguous()
    x_norm_fp16 = torch.empty(total_rows, dim, dtype=torch.half, device=device)

    BLOCK_M = 128
    BLOCK_D = 128 if dim > 256 else 64
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        inp_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear projection + gating (single GEMM)
    # ------------------------------------------------------------------
    # Cast weights to half once.
    left_proj_w   = weights["left_proj.weight"].half().contiguous()
    left_gate_w   = weights["left_gate.weight"].half().contiguous()
    right_proj_w  = weights["right_proj.weight"].half().contiguous()
    right_gate_w  = weights["right_gate.weight"].half().contiguous()
    out_gate_w    = weights["out_gate.weight"].half().contiguous()

    # concat into a single weight of shape (5*hidden, dim)
    combined_w = torch.cat([
        left_proj_w,
        left_gate_w,
        right_proj_w,
        right_gate_w,
        out_gate_w,
    ], dim=0)  # (5*hidden, dim)

    # Perform one fused GEMM
    combined = F.linear(x_norm_fp16, combined_w)  # (R, 5*hidden) half

    # Split
    splits = torch.split(combined, hidden, dim=1)
    left_proj_raw, left_gate_raw, right_proj_raw, right_gate_raw, out_gate_raw = splits

    # Apply sigmoid gates
    left_gate = torch.sigmoid(left_gate_raw)          # (R, hidden) half
    right_gate = torch.sigmoid(right_gate_raw)        # (R, hidden) half
    out_gate = torch.sigmoid(out_gate_raw)            # (R, hidden) half

    left = left_proj_raw * left_gate                  # (R, hidden) half
    right = right_proj_raw * right_gate               # (R, hidden) half

    # ------------------------------------------------------------------
    # 2a) Apply optional binary mask (broadcast to hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask_flat = mask.view(total_rows, 1).to(torch.float16)
        left = left * mask_flat
        right = right * mask_flat

    # ------------------------------------------------------------------
    # 3️⃣ Triangular multiplicative update (batched GEMM)
    # ------------------------------------------------------------------
    left = left.view(B, N, N, hidden)    # (B, N, N, H)
    right = right.view(B, N, N, hidden)  # (B, N, N, H)

    left_mat = left.permute(0, 3, 1, 2).reshape(B * hidden, N, N)   # (B*H, N, N)
    right_mat = right.permute(0, 3, 2, 1).reshape(B * hidden, N, N) # (B*H, N, N)

    # FP16 batched matmul
    out_mat = torch.bmm(left_mat, right_mat)  # (B*H, N, N)

    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 4️⃣ Second LayerNorm + output gate (fused)
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, hidden).contiguous()
    out_norm_gate = torch.empty_like(out_flat)  # same dtype (half)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    fused_layernorm_gate_kernel[grid2](
        out_flat,
        out_norm_gate,

"""
Highly‑optimized forward for the *outgoing* TriMul operator (AlphaFold‑3).

Algorithm
---------
1️⃣  Row‑wise LayerNorm (FP16) – Triton kernel that computes mean/variance
    in FP32, applies γ/β and writes a half‑precision normalised tensor.

2️⃣  **Single** fused linear projection (+ sigmoid gate) – all five
    learned projections (left_proj, left_gate, right_proj, right_gate,
    out_gate) are concatenated into one weight matrix and evaluated by one
    half‑precision GEMM (`torch.nn.functional.linear`).  This removes two
    extra reads of the normalised tensor compared to the naïve
    implementation.

3️⃣  Apply the optional binary mask (broadcasted over the hidden dim).

4️⃣  Triangular multiplicative update – realised as a batched
    half‑precision matrix‑multiply:
        out_h = left_h @ right_hᵀ   (h = hidden index)
    implemented via `torch.bmm` on tensors of shape (B·H, N, N).

5️⃣  Second LayerNorm + output‑gate – fused Triton kernel that normalises
    `out`, applies the learned γ/β, multiplies element‑wise with the
    previously computed `out_gate` and stores the result as FP16.

6️⃣  Final linear projection back to `dim` – regular half‑precision GEMM.

All heavy compute is performed in FP16 (Tensor‑Core) while the LayerNorm
reductions stay in FP32 for numerical stability.  The design removes one
full‑tensor GEMM and merges two LayerNorm passes, giving a considerable
speed‑up over the reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1) Row‑wise LayerNorm (FP16 output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,               # *const float*  [total_rows, dim] (FP32)
    y_ptr,               # *mut   half*   [total_rows, dim] (FP16)
    weight_ptr,          # *const float*  [dim]          (γ)
    bias_ptr,            # *const float*  [dim]          (β)
    total_rows: tl.int32,
    dim: tl.constexpr,   # static (≤ 384)
    eps: tl.constexpr,   # epsilon
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ---------------------------------------------------------------
    # First pass: mean & variance (FP32 accumulation)
    # ---------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)      # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                         # 1/√(var+eps)

    # ---------------------------------------------------------------
    # Second pass: normalise, apply γ, β and store as FP16
    # ---------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]            # (x-µ)/σ

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]                      # FP32

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 6) Fused LayerNorm + out‑gate (both FP16)
# ----------------------------------------------------------------------
@triton.jit
def fused_layernorm_gate_kernel(
    x_ptr,               # *const half*  [total_rows, hidden] (pre‑norm)
    y_ptr,               # *mut   half*  [total_rows, hidden] (post‑norm⋅gate)
    weight_ptr,          # *const float* [hidden] (γ)
    bias_ptr,            # *const float* [hidden] (β)
    gate_ptr,            # *const half*  [total_rows, hidden] (out_gate, already sigmoid’ed)
    total_rows: tl.int32,
    hidden: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ---------------------------------------------------------------
    # First pass: mean / variance (FP32)
    # ---------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(hidden, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < hidden
        ptrs = x_ptr + offs_m[:, None] * hidden + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # FP32

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / hidden
    var  = sum_sq / hidden - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------------------------------------------------------------
    # Second pass: normalise, apply γ, β, multiply by out_gate, store FP16
    # ---------------------------------------------------------------
    for d in range(0, tl.cdiv(hidden, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < hidden
        ptrs = x_ptr + offs_m[:, None] * hidden + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # FP32

        norm = (vals - mean[:, None]) * rstd[:, None]   # (x‑µ)/σ

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # β
        out = norm * w[None, :] + b[None, :]            # FP32

        gate_vals = tl.load(gate_ptr + offs_m[:, None] * hidden + offs_d[None, :],
                            mask=mask_m[:, None] & mask_d[None, :],
                            other=0.0).to(tl.float32)          # FP32
        out = out * gate_vals

        tl.store(y_ptr + offs_m[:, None] * hidden + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the **outgoing** TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, C] (float32)
        - mask_tensor  : torch.Tensor [B, N, N]   (float32/bool) – ignored if config["nomask"]=True
        - weights_dict : model parameters (float32)
        - config_dict  : {"dim":C,
                          "hidden_dim":H,
                          "nomask":bool,
                          "eps":float (optional)}

    Returns
    -------
    torch.Tensor [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    inp, mask, weights, cfg = data
    device = inp.device

    dim   = cfg["dim"]            # C
    hidden = cfg["hidden_dim"]    # H
    eps   = cfg.get("eps", 1e-5)
    nomask = cfg.get("nomask", True)

    B, N, _, _ = inp.shape
    total_rows = B * N * N                      # rows for flattened tensors

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP16 output)
    # ------------------------------------------------------------------
    inp_flat = inp.view(total_rows, dim).contiguous()          # [R, C] float32
    x_norm_fp16 = torch.empty(total_rows, dim, dtype=torch.half,
                              device=device)                # [R, C] half

    BLOCK_M = 128
    BLOCK_D = 128 if dim > 256 else 64

    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        inp_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Single fused GEMM for all five linear projections (proj + gates)
    # ------------------------------------------------------------------
    # Cast weight matrices to FP16 once
    left_proj_w   = weights["left_proj.weight"].half().contiguous()
    left_gate_w   = weights["left_gate.weight"].half().contiguous()
    right_proj_w  = weights["right_proj.weight"].half().contiguous()
    right_gate_w  = weights["right_gate.weight"].half().contiguous()
    out_gate_w    = weights["out_gate.weight"].half().contiguous()

    # Concatenate to a single (5*H, C) matrix
    combined_w = torch.cat([
        left_proj_w,
        left_gate_w,
        right_proj_w,
        right_gate_w,
        out_gate_w,
    ], dim=0)                                   # (5*H, C)

    # One fused linear: (R, C) @ (C, 5*H)ᵀ -> (R, 5*H)
    combined = F.linear(x_norm_fp16, combined_w)  # (R, 5*H) half

    # Split into the five parts
    left_proj_raw, left_gate_raw, right_proj_raw, right_gate_raw, out_gate_raw = \
        torch.split(combined, hidden, dim=1)

    # Sigmoid gates
    left_gate = torch.sigmoid(left_gate_raw)      # (R, H) half
    right_gate = torch.sigmoid(right_gate_raw)    # (R, H) half
    out_gate = torch.sigmoid(out_gate_raw)        # (R, H) half

    # Apply gates to the projections
    left = left_proj_raw * left_gate              # (R, H) half
    right = right_proj_raw * right_gate           # (R, H) half

    # ------------------------------------------------------------------
    # 2a) Apply binary mask (if present)
    # ------------------------------------------------------------------
    if not nomask:
        mask_flat = mask.view(total_rows, 1).to(torch.float16)
        left = left * mask_flat
        right = right * mask_flat

    # ------------------------------------------------------------------
    # 3️⃣ Triangular multiplicative update (batched GEMM)
    # ------------------------------------------------------------------
    left = left.view(B, N, N, hidden)    # (B, N, N, H)
    right = right.view(B, N, N, hidden)  # (B, N, N, H)

    # (B, H, N, N) -> (B*H, N, N)
    left_mat  = left.permute(0, 3, 1, 2).reshape(B * hidden, N, N)   # (B*H, N, N)
    right_mat = right.permute(0, 3, 2, 1).reshape(B * hidden, N, N)  # (B*H, N, N)

    # Batched matrix‑multiply: out_h = left_h @ right_hᵀ
    out_mat = torch.bmm(left_mat, right_mat)   # (B*H, N, N)

    # Reshape back to (B, N, N, H)
    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 4️⃣ Second LayerNorm + output‑gate (fused Triton kernel)
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, hidden).contiguous()   # (R, H) half
    out_norm_gate = torch.empty_like(out_flat)            # (R, H) half

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    fused_layernorm_gate_kernel[grid2](
        out_flat,
        out_norm_gate,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        out_gate,                         # already sigmoid’ed, half
        total_rows,
        hidden,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 5️⃣ Final linear projection back to `dim` (FP16 GEMM)
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].half()   # (C, H)
    final_fp16 = F.linear(out_norm_gate, to_out_w)  # (R, C) half
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
