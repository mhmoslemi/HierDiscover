def row_stats_kernel(x_ptr, mean_ptr, rstd_ptr, total_rows, dim, eps, BLOCK_M, BLOCK_D):
   pid = tl.program_id(0)
   row_start = pid * BLOCK_M
   offs_m = row_start + tl.arange(0, BLOCK_M)
   mask_m = offs_m < total_rows

   sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
   sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)
   for d in range(0, tl.cdiv(dim, BLOCK_D)):
       offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
       mask_d = offs_d < dim
       ptrs = x_ptr + (offs_m[:, None] * dim + offs_d[None, :]) * 1  # x is half pointer
       vals = tl.load(ptrs, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)
       sum_val += tl.sum(vals, axis=1)
       sum_sq += tl.sum(vals * vals, axis=1)

   mean = sum_val / dim
   var = sum_sq / dim - mean * mean
   rstd = tl.rsqrt(var + eps)

   # store
   tl.store(mean_ptr + offs_m, mean, mask=mask_m)
   tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)

def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    B, N, _, C = input_tensor.shape
    total_rows = B * N * N

    # Cast input to half
    x_fp16 = input_tensor.to(torch.float16)

    # Compute per-row mean and rstd via Triton kernel
    mean = torch.empty(total_rows, dtype=torch.float32, device=device)
    rstd = torch.empty(total_rows, dtype=torch.float32, device=device)

    BLOCK_M = 1024  # rows per program
    BLOCK_D = 128  # dimension tile
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    row_stats_kernel[grid](x_fp16, mean, rstd,
                           total_rows, dim, eps,
                           BLOCK_M=BLOCK_M,
                           BLOCK_D=BLOCK_D,
                           num_warps=4)  # optional

    # Convert to half for further operations
    mean_fp16 = mean.to(torch.float16)
    rstd_fp16 = rstd.to(torch.float16)

    # Load weights & cast to half
    norm_weight_fp16 = weights["norm.weight"].to(torch.float16)
    norm_bias_fp16 = weights["norm.bias"].to(torch.float16)

    # Concatenate linear weights
    w_left = weights["left_proj.weight"]
    w_right = weights["right_proj.weight"]
    w_lgate = weights["left_gate.weight"]
    w_rgate = weights["right_gate.weight"]
    w_outgate = weights["out_gate.weight"]
    linear_weights_fp16 = torch.cat([w_left, w_right, w_lgate, w_rgate, w_outgate],
                                   dim=0).to(torch.float16).contiguous()  # shape (5*H, C)

    # weight_prime = linear_weights * norm_weight (broadcast over dim)
    weight_prime_fp16 = linear_weights_fp16 * norm_weight_fp16[None, :]  # shape (5*H, C)

    # Precompute sum_w_prime and bias_proj (fp32)
    sum_w_prime_fp32 = weight_prime_fp16.float().sum(dim=1)  # (5*H)
    bias_proj_fp32 = (norm_bias_fp16.float() @ linear_weights_fp16.float().t())  # (5*H)
    sum_w_prime_fp16 = sum_w_prime_fp32.to(torch.float16)
    bias_proj_fp16 = bias_proj_fp32.to(torch.float16)

    # Scale input by norm_weight before linear
    x_scaled = x_fp16 * norm_weight_fp16  # broadcasting

    # Compute dot product: (total_rows, C) @ (5*H, C).T => (total_rows, 5*H)
    proj_all = torch.nn.functional.linear(x_scaled, weight_prime_fp16)  # half

    # Apply LN scaling: rstd * (proj - mean * sum_w_prime) + bias
    proj_all = rstd_fp16[:, None] * (proj_all - mean_fp16[:, None] * sum_w_prime_fp16[None, :]) + bias_proj_fp16[None, :]

    # Split into tensors
    left_proj_fp16, right_proj_fp16, left_gate_fp16, right_gate_fp16, out_gate_fp16 = torch.split(proj_all,
                                                                                                   hidden_dim,
                                                                                                   dim=1)

    # Apply mask if needed
    if not nomask and mask_tensor is not None:
        mask_fp16 = mask_tensor.reshape(total_rows, 1).to(torch.float16)
        left_proj_fp16 = left_proj_fp16 * mask_fp16
        right_proj_fp16 = right_proj_fp16 * mask_fp16

    # Gate sigmoids
    left_gate = torch.sigmoid(left_gate_fp16)
    right_gate = torch.sigmoid(right_gate_fp16)
    out_gate = torch.sigmoid(out_gate_fp16)

    left = left_proj_fp16 * left_gate
    right = right_proj_fp16 * right_gate

    # Reshape to 4-D
    left = left.view(B, N, N, hidden_dim)
    right = right.view(B, N, N, hidden_dim)
    out_gate = out_gate.view(B, N, N, hidden_dim)

    # Triangular multiplicative update (outgoing)
    # Permute to (B, H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    BH = B * hidden_dim
    left_mat = left_perm.view(BH, N, N)
    right_mat = right_perm.view(BH, N, N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # half

    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # shape (B,N,N,H) half

    # Second LayerNorm (builtin for speed)
    to_out_norm_weight_fp16 = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_bias_fp16 = weights["to_out_norm.bias"].to(torch.float16)

    out_norm = torch.nn.functional.layer_norm(out,
                                              (hidden_dim,),
                                              weight=to_out_norm_weight_fp16,
                                              bias=to_out_norm_bias_fp16,
                                              eps=eps)

    # Apply output gate
    out_norm = out_norm * out_gate

    # Final linear projection
    to_out_weight_fp16 = weights["to_out.weight"].to(torch.float16)
    final_fp16 = torch.nn.functional.linear(out_norm.view(total_rows, hidden_dim),
                                            to_out_weight_fp16)

    # Cast to fp32 and reshape
    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final
```

```python
import triton
import triton.language as tl

@triton.jit
def row_stats_kernel(x_ptr, mean_ptr, rstd_ptr,
                     total_rows, dim: tl.constexpr, eps: tl.constexpr,
                     BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        vals = tl.load(ptrs, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mean_ptr + offs_m, mean, mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)

"""
custom_kernel
==============

Fast forward pass for the *outgoing* TriMul operator (AlphaFold‑3).

Algorithm (half‑precision, fused where possible):
1️⃣  Compute per‑row mean and 1/√var (rstd) of the input tensor
    (shape [B·N·N, C]) with a tiny Triton reduction kernel.
2️⃣  Fuse LayerNorm and the five linear projections (left, right,
    three gates) into a single GEMM:
        • Pre‑scale the weight matrix with the LayerNorm γ.
        • Compute   dot = X @ W′ᵀ   where  W′ = W·γ   (half‑precision).
        • Recover the LayerNorm output analytically:
              proj = rstd·(dot – mean·ΣW′) + Σ(bias·W)
        (no intermediate normalized tensor is materialised).
3️⃣  Split the fused projection into left/right/value tensors
    and optionally apply the pair‑wise mask.
4️⃣  Apply sigmoid gates and element‑wise gating (half).
5️⃣  Triangular multiplicative update (outgoing version):
        out_h[i, j] = Σₖ left_h[i, k] · right_h[j, k]
    performed as a batched GEMM `left @ rightᵀ` with cuBLAS.
6️⃣  Row‑wise LayerNorm over the hidden dimension (built‑in
    `torch.nn.functional.layer_norm` – highly optimised on H100).
7️⃣  Multiply by the output gate and a final linear projection
    back to the original channel dimension.
8️⃣  Cast the result to FP32 and return.

Only the per‑row statistics (mean & rstd) are computed by a custom
Triton kernel; all dense linear algebra uses cuBLAS/Torch kernels,
yielding sub‑millisecond runtimes on the target hardware.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: per‑row mean and inverse‑std (rstd) for half‑inputs
# ----------------------------------------------------------------------
@triton.jit
def row_stats_kernel(
    x_ptr,          # *const half*  [total_rows, dim]
    mean_ptr,       # *mut   float* [total_rows]
    rstd_ptr,       # *mut   float* [total_rows]
    total_rows,     # i32
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,   # rows processed per program
    BLOCK_D: tl.constexpr,   # cols processed per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)           # [BLOCK_M]
    mask_m = offs_m < total_rows

    # accumulate sum(x) and sum(x²) in FP32
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    # loop over the feature dimension
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)     # [BLOCK_D]
        mask_d = offs_d < dim

        # flat pointer: row * dim + col
        ptrs = x_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        # load half → up‑cast to FP32
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)       # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                 # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)          # [BLOCK_M]

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mean_ptr + offs_m, mean, mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)


# ----------------------------------------------------------------------
# Main Triton‑accelerated TriMul entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, C] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict  : mapping name → torch.Tensor (float32)
            - config_dict   : {"dim": C, "hidden_dim": H, "nomask": bool}
    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                # input channel dimension C
    hidden_dim = config["hidden_dim"]  # hidden dimension H (always 128)
    eps = 1e-5
    nomask = config.get("nomask", True)

    B, N, _, C = input_tensor.shape
    total_rows = B * N * N

    # ------------------------------------------------------------------
    # 1️⃣  Cast input to FP16 and compute per‑row statistics (mean, rstd)
    # ------------------------------------------------------------------
    x_flat = input_tensor.reshape(total_rows, dim).contiguous()
    x_fp16 = x_flat.to(torch.float16)

    mean = torch.empty(total_rows, dtype=torch.float32, device=device)
    rstd = torch.empty(total_rows, dtype=torch.float32, device=device)

    # launch Triton kernel ------------------------------------------------
    BLOCK_M = 1024            # rows per program (tunable)
    BLOCK_D = 128             # tile over feature dim (covers max dim)
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    row_stats_kernel[grid](
        x_fp16,               # half input
        mean,
        rstd,
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    # Cast statistics back to FP16 for subsequent arithmetic
    mean_fp16 = mean.to(torch.float16)
    rstd_fp16 = rstd.to(torch.float16)

    # ------------------------------------------------------------------
    # 2️⃣  Fuse LayerNorm and the five projections into one GEMM
    # ------------------------------------------------------------------
    # Load weight tensors (FP16)
    norm_weight_fp16 = weights["norm.weight"].to(torch.float16)
    norm_bias_fp16   = weights["norm.bias"].to(torch.float16)

    # Concatenate the five projection matrices (shape: [5*H, C])
    linear_weights_fp16 = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16).contiguous()               # (5*H, C)

    # Incorporate LayerNorm γ into the projection weight:
    #   W′ = W ⊙ γ   (broadcast over the C dimension)
    weight_prime_fp16 = linear_weights_fp16 * norm_weight_fp16[None, :]   # (5*H, C)

    # Pre‑compute per‑output constants:
    #   ΣW′  – sum of the scaled weights per output (5*H,)
    sum_w_prime_fp16 = weight_prime_fp16.float().sum(dim=1).to(torch.float16)

    #   bias_proj = bias ⋅ W   (5*H,)
    bias_proj_fp16 = (norm_bias_fp16.float() @ linear_weights_fp16.float().t()).to(torch.float16)

    # GEMM: dot = X @ W′ᵀ   (half precision)
    dot = F.linear(x_fp16, weight_prime_fp16)      # (total_rows, 5*H)

    # Recover the LayerNorm output analytically
    # proj = rstd·(dot – mean·ΣW′) + bias_proj
    proj_all = (
        rstd_fp16[:, None] *
        (dot - mean_fp16[:, None] * sum_w_prime_fp16[None, :])
        + bias_proj_fp16[None, :]
    )                                              # (total_rows, 5*H)

    # ------------------------------------------------------------------
    # 3️⃣  Split fused projection & apply optional mask
    # ------------------------------------------------------------------
    left_proj, right_proj, left_gate, right_gate, out_gate = torch.split(
        proj_all, hidden_dim, dim=1
    )                                              # each (total_rows, H)

    if not nomask and mask_tensor is not None:
        mask_fp16 = mask_tensor.reshape(total_rows, 1).to(torch.float16)
        left_proj  = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # ------------------------------------------------------------------
    # 4️⃣  Apply sigmoid gates and element‑wise gating
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate)    # (total_rows, H)
    right_gate = torch.sigmoid(right_gate)
    out_gate   = torch.sigmoid(out_gate)

    left  = left_proj  * left_gate          # (total_rows, H)
    right = right_proj * right_gate

    # Reshape back to 4‑D tensors: [B, N, N, H]
    left  = left.view(B, N, N, hidden_dim)
    right = right.view(B, N, N, hidden_dim)
    out_gate = out_gate.view(B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 5️⃣  Triangular multiplicative update (outgoing)
    # ------------------------------------------------------------------
    # Permute to (B, H, N, N) → view as (B·H, N, N)
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    BH = B * hidden_dim
    left_mat  = left_perm.view(BH, N, N)                # (BH, N, N)
    right_mat = right_perm.view(BH, N, N)               # (BH, N, N)

    # Batched GEMM: left @ rightᵀ  → (BH, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Restore original layout: [B, N, N, H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # half

    # ------------------------------------------------------------------
    # 6️⃣  Second (hidden‑dim) LayerNorm – use Torch's fused kernel
    # ------------------------------------------------------------------
    to_out_norm_weight_fp16 = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_bias_fp16   = weights["to_out_norm.bias"].to(torch.float16)

    out_norm = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight_fp16,
        bias=to_out_norm_bias_fp16,
        eps=eps,
    )   # (B, N, N, H) half

    # ------------------------------------------------------------------
    # 7️⃣  Apply output gate and final linear projection
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate               # element‑wise, half

    to_out_weight_fp16 = weights["to_out.weight"].to(torch.float16)   # (C, H)
    final_fp16 = F.linear(out_norm.view(total_rows, hidden_dim), to_out_weight_fp16)

    # ------------------------------------------------------------------
    # 8️⃣  Cast back to FP32 and reshape to [B, N, N, C]
    # ------------------------------------------------------------------
    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final
