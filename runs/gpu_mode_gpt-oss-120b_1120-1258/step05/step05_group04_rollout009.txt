def custom_kernel(data):
    inp, mask, weights, cfg = data
    device = inp.device
    dim = cfg['dim']
    hidden = cfg['hidden_dim']
    eps = cfg.get('eps', 1e-5)
    nomask = cfg.get('nomask', True)

    # 1. Input LayerNorm using torch.nn.functional
    # Norm handle weight and bias
    x_norm = torch.nn.functional.layer_norm(
        inp,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    ) # shape B,N,N,dim, float32
    x_norm_fp16 = x_norm.half()
    total_rows = inp.shape[0] * inp.shape[1] * inp.shape[2] # B*N*N

    # 2. Linear projection and gates
    # Concatenate weights
    concat_weight = torch.cat([
        weights["left_proj.weight"],
        weights["right_proj.weight"],
        weights["left_gate.weight"],
        weights["right_gate.weight"],
        weights["out_gate.weight"],
    ], dim=0).half().contiguous() # shape (5*hidden, dim)

    # No bias
    proj_all = torch.nn.functional.linear(x_norm_fp16, concat_weight)  # shape (B,N,N,5*hidden)
    # split
    proj_all = proj_all.view(total_rows, 5*hidden)
    left_proj_fp16, right_proj_fp16, left_gate_fp16, right_gate_fp16, out_gate_fp16 = torch.split(proj_all, hidden, dim=1)

    # 2a mask
    if not nomask:
        mask_flat = mask.view(total_rows, 1).to(torch.float16)
        left_proj_fp16 = left_proj_fp16 * mask_flat
        right_proj_fp16 = right_proj_fp16 * mask_flat

    # 2b gates and elementwise apply
    left_gate = torch.sigmoid(left_gate_fp16.float()).half()
    right_gate = torch.sigmoid(right_gate_fp16.float()).half()
    out_gate = torch.sigmoid(out_gate_fp16.float()).half()

    left = left_proj_fp16 * left_gate
    right = right_proj_fp16 * right_gate

    # 3. Triangular multiplicative update
    B, N, _, _ = inp.shape  # get B and N
    left = left.view(B, N, N, hidden)   # (B,N,N,H)
    right = right.view(B, N, N, hidden)

    # Use torch.bmm batched multiplication
    left_mat = left.permute(0,3,1,2).reshape(B*hidden, N, N)
    right_mat = right.permute(0,3,2,1).reshape(B*hidden, N, N)
    out_mat = torch.bmm(left_mat, right_mat) # (B*hidden, N, N) float16

    out = out_mat.view(B, hidden, N, N).permute(0,2,3,1).contiguous()  # (B,N,N,H) FP16

    # 4. Final fused LN + out_gate + linear (Triton kernel)
    out_flat = out.view(total_rows, hidden).contiguous()
    final_fp32 = torch.empty(total_rows, dim, dtype=torch.float32, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid = (triton.cdiv(total_rows, BLOCK_M), triton.cdiv(dim, BLOCK_D))
    layernorm_gate_linear_kernel[grid](
        out_flat,
        out_gate,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        weights["to_out.weight"].half().contiguous(),
        final_fp32,
        total_rows,
        dim=dim,
        hidden=hidden,
        eps=eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    final = final_fp32.view(B, N, N, dim)
    return final

"""
Highly‑optimized forward pass for the **outgoing** TriMul operator.

Algorithm
---------
1️⃣  Input LayerNorm – we use the native `torch.nn.functional.layer_norm`,
    which is heavily tuned in cuDNN (much faster than a naïve Triton
    implementation).  Result is cast to FP16 for the downstream GEMMs.

2️⃣  Fuse the 5 linear projections (left/right + 3 gates) into a single
    half‑precision GEMM (`torch.nn.functional.linear`).  The large weight
    matrix is assembled on‑the‑fly, saving kernel launches.

3️⃣  Optional binary mask is applied only to the two “value” projections
    (left‑proj / right‑proj) – this is a cheap element‑wise multiply.

4️⃣  Apply sigmoid to the three gate tensors and fuse the gating with the
    projections.

5️⃣  Triangular multiplicative update – a batched matrix‑multiply across
    the sequence dimension:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    Implemented with `torch.bmm` on the reshaped tensors, which maps to a
    single cuBLAS call (Tensor‑Core, FP16).  No explicit Python loops.

6️⃣  **Fused** LayerNorm + out‑gate + final linear projection.
    A custom Triton kernel (`layernorm_gate_linear_kernel`) reads the
    half‑precision `out` tensor, computes per‑row mean / variance (FP32),
    applies the `to_out_norm` affine transformation, multiplies by the
    pre‑computed out‑gate, and finally projects to the model dimension
    (`to_out.weight`).  This eliminates two extra memory passes (the
    separate LayerNorm and linear layers) and speeds up the final stage
    dramatically.

Only the final fused stage is written in Triton; everything else relies
on highly‑optimised PyTorch primitives.  The implementation works for all
test configurations listed in the statement and returns a `float32`
tensor of shape `[B, N, N, dim]`.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# 6) Fused LayerNorm + out‑gate + final linear projection (all FP32)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_gate_linear_kernel(
    x_ptr,               # *const half*  [total_rows, hidden]   (pre‑norm)
    gate_ptr,            # *const half*  [total_rows, hidden]   (out_gate, sigmoid’ed)
    norm_w_ptr,          # *const float* [hidden]               (γ of to_out_norm)
    norm_b_ptr,          # *const float* [hidden]               (β of to_out_norm)
    out_w_ptr,           # *const half*  [dim, hidden]          (to_out weight)
    out_ptr,             # *mut   float* [total_rows, dim]      (final FP32 output)
    total_rows: tl.int32,
    dim: tl.constexpr,
    hidden: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,   # rows per program   (e.g. 128)
    BLOCK_D: tl.constexpr,   # output‑channel chunk (e.g. 64)
):
    pid = tl.program_id(0)          # row block index
    pid_d = tl.program_id(1)        # output‑channel block index
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)                 # [BLOCK_M]
    mask_m = offs_m < total_rows

    # -----------------------------------------------------------------
    # Load the whole hidden vector for each selected row (BLOCK_M × hidden)
    # -----------------------------------------------------------------
    offs_h = tl.arange(0, hidden)                               # [hidden]
    mask_h = offs_h < hidden

    # x and gate are stored as half → cast to FP32 for math
    x = tl.load(x_ptr + offs_m[:, None] * hidden + offs_h[None, :],
                mask=mask_m[:, None] & mask_h[None, :],
                other=0.0).to(tl.float32)                       # (BLOCK_M, hidden)
    g = tl.load(gate_ptr + offs_m[:, None] * hidden + offs_h[None, :],
                mask=mask_m[:, None] & mask_h[None, :],
                other=0.0).to(tl.float32)                       # (BLOCK_M, hidden)

    # -----------------------------------------------------------------
    # Compute mean / rstd for each row (FP32)
    # -----------------------------------------------------------------
    sum_val = tl.sum(x, axis=1)                                 # (BLOCK_M)
    sum_sq  = tl.sum(x * x, axis=1)                             # (BLOCK_M)
    mean = sum_val / hidden
    var  = sum_sq / hidden - mean * mean
    rstd = tl.rsqrt(var + eps)

    # -----------------------------------------------------------------
    # Apply LayerNorm weight/bias and the out‑gate
    # -----------------------------------------------------------------
    w_norm = tl.load(norm_w_ptr + offs_h, mask=mask_h, other=0.0)   # (hidden)
    b_norm = tl.load(norm_b_ptr + offs_h, mask=mask_h, other=0.0)   # (hidden)

    norm = (x - mean[:, None]) * rstd[:, None]                     # (BLOCK_M, hidden)
    norm = norm * w_norm[None, :] + b_norm[None, :]                # apply γ,β
    gated = norm * g                                                # (BLOCK_M, hidden)

    # -----------------------------------------------------------------
    # Output‑channel block handling
    # -----------------------------------------------------------------
    d_start = pid_d * BLOCK_D
    offs_d = d_start + tl.arange(0, BLOCK_D)                       # [BLOCK_D]
    mask_d = offs_d < dim

    # Load the slice of the final weight matrix (hidden × BLOCK_D)
    # out_w has shape [dim, hidden]; we need its transpose (hidden, BLOCK_D)
    out_w_T = tl.load(out_w_ptr + offs_h[:, None] + offs_d[None, :] * hidden,
                      mask=mask_h[:, None] & mask_d[None, :],
                      other=0.0).to(tl.float32)                  # (hidden, BLOCK_D)

    # Compute the mat‑mul: (BLOCK_M, hidden) × (hidden, BLOCK_D) → (BLOCK_M, BLOCK_D)
    out_chunk = tl.dot(gated, out_w_T)                            # (BLOCK_M, BLOCK_D)

    # Store the result (FP32)
    tl.store(out_ptr + offs_m[:, None] * dim + offs_d[None, :],
             out_chunk,
             mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the **outgoing** TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, C] (float32)
        - mask_tensor  : torch.Tensor [B, N, N]   (bool/float) – ignored if config["nomask"]=True
        - weights_dict : model parameters (float32)
        - config_dict  : {"dim": C, "hidden_dim": H, "nomask": bool, "eps": float (opt.)}

    Returns
    -------
    torch.Tensor [B, N, N, C] (float32) on the same CUDA device.
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    inp, mask, weights, cfg = data
    device = inp.device
    dim = cfg["dim"]                # input channel dimension C
    hidden = cfg["hidden_dim"]      # hidden dimension H
    eps = cfg.get("eps", 1e-5)
    nomask = cfg.get("nomask", True)

    B, N, _, _ = inp.shape
    total_rows = B * N * N                 # flattened rows = B·N·N

    # --------------------------------------------------------------
    # 1️⃣  Input LayerNorm (use native cuDNN implementation)
    # --------------------------------------------------------------
    # LayerNorm normalises over the last dimension (dim)
    x_norm = F.layer_norm(
        inp,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )                                        # [B, N, N, dim] float32
    x_norm_fp16 = x_norm.half()               # cast to FP16 for the GEMM

    # --------------------------------------------------------------
    # 2️⃣  Fused projection + gates (single half‑precision GEMM)
    # --------------------------------------------------------------
    # Concatenate the five weight matrices (each [hidden, dim])
    concat_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).half().contiguous()                     # shape (5*hidden, dim)

    # Linear projection of the normalised input
    proj_all = F.linear(x_norm_fp16, concat_w)   # (B, N, N, 5*hidden)  FP16
    proj_all = proj_all.view(total_rows, 5 * hidden)   # flatten spatial dims

    # Split into the five tensors
    left_proj_fp16, right_proj_fp16, left_gate_fp16, \
    right_gate_fp16, out_gate_fp16 = torch.split(
        proj_all, hidden, dim=1
    )                                           # each (total_rows, hidden)

    # --------------------------------------------------------------
    # 2a) Optional mask (broadcasted on hidden dim)
    # --------------------------------------------------------------
    if not nomask:
        mask_flat = mask.view(total_rows, 1).to(torch.float16)
        left_proj_fp16  = left_proj_fp16  * mask_flat
        right_proj_fp16 = right_proj_fp16 * mask_flat

    # --------------------------------------------------------------
    # 2b) Sigmoid gates and element‑wise gating
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_fp16.float()).half()
    right_gate = torch.sigmoid(right_gate_fp16.float()).half()
    out_gate = torch.sigmoid(out_gate_fp16.float()).half()   # (total_rows, hidden)

    left = left_proj_fp16 * left_gate          # (total_rows, hidden)
    right = right_proj_fp16 * right_gate       # (total_rows, hidden)

    # --------------------------------------------------------------
    # 3️⃣  Triangular multiplicative update (batched matmul)
    # --------------------------------------------------------------
    left = left.view(B, N, N, hidden)          # (B, N, N, H)
    right = right.view(B, N, N, hidden)        # (B, N, N, H)

    # reshape for a single batched GEMM: (B*H, N, N)
    left_mat = left.permute(0, 3, 1, 2).reshape(B * hidden, N, N)   # (B*H, N, N)
    right_mat = right.permute(0, 3, 2, 1).reshape(B * hidden, N, N) # (B*H, N, N)

    # FP16 batched matrix multiply – leverages Tensor Cores
    out_mat = torch.bmm(left_mat, right_mat)   # (B*H, N, N)  FP16

    # reshape back to (B, N, N, hidden)
    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H) FP16

    # --------------------------------------------------------------
    # 4️⃣  Fused LayerNorm + out‑gate + final projection (Triton)
    # --------------------------------------------------------------
    out_flat = out.view(total_rows, hidden).contiguous()   # (R, H) FP16
    final_fp32 = torch.empty(total_rows, dim,
                             dtype=torch.float32,
                             device=device)

    # Tune block sizes for large row counts
    BLOCK_M = 128          # rows per program (chosen to halve kernel launches)
    BLOCK_D = 64           # output‑channel chunk
    grid = (
        triton.cdiv(total_rows, BLOCK_M),
        triton.cdiv(dim, BLOCK_D),
    )
    layernorm_gate_linear_kernel[grid](
        out_flat,                               # x_ptr
        out_gate,                               # gate_ptr
        weights["to_out_norm.weight"],          # γ
        weights["to_out_norm.bias"],            # β
        weights["to_out.weight"].half().contiguous(),  # final linear weight
        final_fp32,                             # out_ptr
        total_rows,
        dim=dim,
        hidden=hidden,
        eps=eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # --------------------------------------------------------------
    # 5️⃣  Reshape to final tensor shape (float32) and return
    # --------------------------------------------------------------
    final = final_fp32.view(B, N, N, dim)       # [B, N, N, dim] float32
    return final
