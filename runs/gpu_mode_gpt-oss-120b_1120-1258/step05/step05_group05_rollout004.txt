def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N

    # 1. LayerNorm on input (dim)
    x_norm_fp16 = torch.empty(total_rows, dim, dtype=torch.float16, device=device)
    BLOCK_M = 1024
    BLOCK_D = 64
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        input_tensor.contiguous().view(total_rows, dim),
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M, BLOCK_D=BLOCK_D
    )
    # 2. fused linear projection
    # concat weight matrices of shape (5*hidden_dim, dim)
    w_concat = torch.cat([weights["left_proj.weight"],
                          weights["right_proj.weight"],
                          weights["left_gate.weight"],
                          weights["right_gate.weight"],
                          weights["out_gate.weight"]], dim=0).half()
    proj = F.linear(x_norm_fp16, w_concat)  # shape [total_rows, 5*hidden_dim]
    # split
    left_proj, right_proj, left_gate, right_gate, out_gate = torch.split(proj, hidden_dim, dim=1)
    # 3. mask
    if not nomask and mask_tensor is not None:
        mask_fp16 = mask_tensor.view(total_rows, 1).to(torch.float16)
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16
    # 4. apply gates
    left = left_proj * torch.sigmoid(left_gate)   # each half
    right = right_proj * torch.sigmoid(right_gate)
    out_gate_sig = torch.sigmoid(out_gate)

    # reshape to [B, N, N, hidden_dim]
    left = left.view(B, N, N, hidden_dim)
    right = right.view(B, N, N, hidden_dim)
    out_gate_sig = out_gate_sig.view(B, N, N, hidden_dim)

    # 5. triangular update
    # using einsum
    out = torch.einsum('b i k h, b j k h -> b i j h', left, right)

    # 6. second layernorm over hidden dim
    out_flat = out.view(total_rows, hidden_dim)
    out_norm_fp16 = torch.empty_like(out_flat, dtype=torch.float16)
    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid2](
        out_flat,
        out_norm_fp16,

        weights["to_out_norm.bias"],
        total_rows,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D
    )
    out_norm = out_norm_fp16.view(B, N, N, hidden_dim)

    gated = out_norm * out_gate_sig

"""
Fast outgoing TriMul (AlphaFold‑3) – pure–FP16 forward pass.

Algorithm
---------
1️⃣ Row‑wise LayerNorm over the input channel dimension **C** (FP32
    reduction → FP16 output). Implemented with a single Triton kernel.

2️⃣ Fuse the five linear projections (left/right projections, three
    gating‑projections) into one GEMM:
       [B·N·N, C] @ [C, 5·H]ᵀ → [B·N·N, 5·H]
   (H = hidden_dim). The weight matrix is built once per call and
   kept in half‑precision.

3️⃣ Optional pair‑wise mask is applied **after** the projection (the mask
   is broadcast along the hidden dimension).

4️⃣ Apply sigmoid gates and element‑wise gating (still FP16).

5️⃣ Triangular multiplicative update (outgoing version) via a single
   `torch.einsum`:
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   This eliminates the explicit `permute` + `contiguous` that caused an
   extra 2 × (N²·H) copy.

6️⃣ Row‑wise LayerNorm over the hidden dimension **H** (same Triton kernel).

7️⃣ Multiply by the output gate, then a final linear projection back to **C**.

All heavy linear algebra (step 2, 5, 7) is performed by cuBLAS/Tensor‑Core
kernels on FP16.  Only the two LayerNorms are executed in Triton, keeping
the implementation compliant with the “must use Triton” requirement while
reducing memory traffic and kernel launch overhead.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton row‑wise LayerNorm (fp16 output, fp32 reduction)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,          # *const void*   [total_rows, dim]   (float16/float32 input)
    y_ptr,          # *mut   float16* [total_rows, dim]   (fp16 output)
    weight_ptr,     # *const float*  [dim]               (γ)
    bias_ptr,       # *const float*  [dim]               (β)
    total_rows,     # i32
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,   # rows processed per program
    BLOCK_D: tl.constexpr,   # columns processed per loop iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)            # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # 1. Mean & variance (fp32 accumulator)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < dim

        # 2‑D pointer: [row, col]
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                                   # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                     # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)              # [BLOCK_M]

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                               # 1 / √(var+eps)

    # ------------------------------------------------------------------
    # 2. Write normalized output (fp16)
    # ------------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]      # fp32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)

        out_fp16 = (norm * w[None, :] + b[None, :]).to(tl.float16)

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, C] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : {"dim": C, "hidden_dim": H, "nomask": bool}

    Returns
    -------
    torch.Tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                # C
    hidden_dim = config["hidden_dim"]  # H
    eps = 1e-5
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N                     # rows for the flattened view

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP16 output)
    # ------------------------------------------------------------------
    # flatten to [total_rows, dim] – keep original layout, cast on the fly
    x_flat = input_tensor.contiguous().view(total_rows, dim)

    x_norm = torch.empty(total_rows, dim,
                         dtype=torch.float16,
                         device=device)

    # choose a relatively large BLOCK_M to cut launch overhead
    BLOCK_M = min(1024, total_rows)   # cap at 1024 rows per program
    BLOCK_D = 64
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection (left, right, three gates)
    # ------------------------------------------------------------------
    w_left    = weights["left_proj.weight"]
    w_right   = weights["right_proj.weight"]
    w_lgate   = weights["left_gate.weight"]
    w_rgate   = weights["right_gate.weight"]
    w_outgate = weights["out_gate.weight"]

    # concat → [(5*H), C] and convert once to half‑precision
    W_fused = torch.cat([w_left, w_right, w_lgate, w_rgate, w_outgate],
                        dim=0).half().contiguous()

    # (total_rows, C) @ (C, 5*H)ᵀ  → (total_rows, 5*H)
    proj = F.linear(x_norm, W_fused)               # FP16

    # split back into the five tensors, each (total_rows, H)
    left_proj, right_proj, left_gate, right_gate, out_gate = torch.split(
        proj, hidden_dim, dim=1
    )

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        mask_fp16 = mask_tensor.view(total_rows, 1).to(torch.float16)
        left_proj  = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # ------------------------------------------------------------------
    # 4️⃣ Apply sigmoid gates and element‑wise gating (still FP16)
    # ------------------------------------------------------------------
    left  = left_proj  * torch.sigmoid(left_gate)
    right = right_proj * torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)   # will be used later

    # reshape to 4‑D layout for the contraction
    left  = left.view(B, N, N, hidden_dim)      # [B, N, N, H]
    right = right.view(B, N, N, hidden_dim)    # [B, N, N, H]
    out_gate = out_gate.view(B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 5️⃣ Triangular multiplicative update (outgoing)
    #     out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    # torch.einsum internally performs a batched GEMM without the
    # explicit permute‑contiguous copy used previously.
    out = torch.einsum("b i k h, b j k h -> b i j h", left, right)

    # ------------------------------------------------------------------
    # 6️⃣ Hidden‑dim LayerNorm (FP16 output)
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, hidden_dim)
    out_norm = torch.empty_like(out_flat,
                                dtype=torch.float16,
                                device=device)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid2](
        out_flat,
        out_norm,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    out_norm = out_norm.view(B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 7️⃣ Final gating + projection back to C
    # ------------------------------------------------------------------
    gated = out_norm * out_gate                     # element‑wise FP16
    final = F.linear(gated.view(total_rows, hidden_dim),
                     weights["to_out.weight"].half())   # (total_rows, C) FP16

    # ------------------------------------------------------------------
    # 8️⃣ Cast to FP32 as required by the API
    # ------------------------------------------------------------------
    return final.view(B, N, N, dim).to(torch.float32)
