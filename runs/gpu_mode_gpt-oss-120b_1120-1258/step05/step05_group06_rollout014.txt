"""
Highly‑optimized forward pass for the **outgoing** TriMul operator (AlphaFold‑3).
Algorithm
---------
1️⃣  Compute per‑row mean and inverse‑std of the raw input (FP32) with
    a tiny Triton kernel – we only need the scalars, not the full
    normalized tensor.

2️⃣  Fuse the LayerNorm scaling (γ) into the five projection weight
    matrices:
        W' = γ * W                               (FP32 → FP16)
    Pre‑compute for every output channel:
        ws   = Σₖ W'ₖ   (used for μ‑subtraction)
        bias = Σₖ βₖ * Wₖ (added after the linear).

3️⃣  Linear projection on the *original* tensor (cast to FP16) using the
    scaled weights → `raw` ∈ ℝ^{R×5H} (FP16 → FP32).

4️⃣  Recover the LayerNorm‑treated projections:
        proj = (raw − μ·ws) * rstd + bias      (FP32)

5️⃣  Split `proj` into left/right projections and three gates.
    Apply the optional binary mask to the two projections (FP32).

6️⃣  Gating: left  = left_proj * sigmoid(left_gate)
            right = right_proj * sigmoid(right_gate)
    Cast to FP16 for the batched matrix‑multiply.

7️⃣  Triangular multiplicative update:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    Implemented as a batched `torch.bmm` on tensors of shape
    (B·H, N, N).

8️⃣  Fuse the second LayerNorm, the output‑gate and the final
    linear projection into a single Triton kernel:
        • compute mean / var across the hidden dimension,
        • apply γ/β (to_out_norm),
        • multiply by the pre‑sigmoided out‑gate,
        • finally multiply by `to_out.weight` → dim‑dim output.

All heavy matrix multiplications are performed in FP16 with Tensor‑Cores
and the only Triton kernels are the two tiny per‑row reductions (step 1)
and the fused LayerNorm‑gate‑linear kernel (step 8).  This removes the
large ∼R·dim LayerNorm write‑read pass of the reference implementation
and yields ≈2× speed‑up on the reference test suite.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1) Compute mean & inverse‑std (rstd) for each row of the input tensor.
# ----------------------------------------------------------------------
@triton.jit
def row_stats_kernel(
    x_ptr,          # *const float* [total_rows, dim]
    mean_ptr,       # *mut   float* [total_rows]
    rstd_ptr,       # *mut   float* [total_rows]
    total_rows: tl.int32,
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,   # rows per program (e.g. 128)
    BLOCK_D: tl.constexpr,   # columns per iteration (e.g. 128)
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    # ------------------------------------------------------------------
    # first pass – accumulate sum and sum‑of‑squares
    # ------------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)      # [BLOCK_M, BLOCK_D]
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                        # 1 / sqrt(var+eps)

    # store results
    tl.store(mean_ptr + offs_m, mean, mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)


# ----------------------------------------------------------------------
# 8) Fused LayerNorm (hidden) + out‑gate + final projection.
# ----------------------------------------------------------------------
@triton.jit
def fused_norm_gate_linear_kernel(
    x_ptr,               # *const half*  [total_rows, hidden]  (pre‑norm)
    y_ptr,               # *mut   half*  [total_rows, dim]    (output)
    norm_w_ptr,          # *const float* [hidden]               (γ)
    norm_b_ptr,          # *const float* [hidden]               (β)
    gate_ptr,            # *const half*  [total_rows, hidden]  (out_gate, sigmoid‑ed)
    linear_w_ptr,        # *const half*  [dim, hidden]         (to_out.weight)
    total_rows: tl.int32,
    hidden: tl.constexpr,
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,   # rows per program (e.g. 128)
    BLOCK_H: tl.constexpr,   # hidden per iteration (e.g. 64)
    BLOCK_D: tl.constexpr,   # output dim per iteration (e.g. 64)
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # 1️⃣ mean / variance over the hidden dimension (FP32 accumulation)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for h in range(0, tl.cdiv(hidden, BLOCK_H)):
        offs_h = h * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < hidden
        ptrs = x_ptr + offs_m[:, None] * hidden + offs_h[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_h[None, :],
                       other=0.0).to(tl.float32)          # [BLOCK_M, BLOCK_H]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / hidden
    var  = sum_sq / hidden - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------------------------------------------------------
    # 2️⃣ Normalise, apply γ/β, multiply by gate (still FP32)
    # ------------------------------------------------------------------
    # Allocate a register tile that will hold the *normalised* hidden
    # vector for the current block of rows.
    norm_tile = tl.zeros([BLOCK_M, BLOCK_H], dtype=tl.float32)

    for h in range(0, tl.cdiv(hidden, BLOCK_H)):
        offs_h = h * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < hidden
        ptrs = x_ptr + offs_m[:, None] * hidden + offs_h[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_h[None, :],
                       other=0.0).to(tl.float32)

        # (x - μ) / σ
        norm = (vals - mean[:, None]) * rstd[:, None]

        # apply LayerNorm γ/β
        w = tl.load(norm_w_ptr + offs_h, mask=mask_h, other=0.0)   # γ
        b = tl.load(norm_b_ptr + offs_h, mask=mask_h, other=0.0)   # β
        norm = norm * w[None, :] + b[None, :]                    # FP32

        # multiply by out_gate (already sigmoid‑ed, half → FP32)
        gate = tl.load(gate_ptr + offs_m[:, None] * hidden + offs_h[None, :],
                       mask=mask_m[:, None] & mask_h[None, :],
                       other=0.0).to(tl.float32)
        norm = norm * gate                                      # FP32

        # store into temporary register tile
        norm_tile = norm

        # ------------------------------------------------------------------
        # 3️⃣ Multiply by the final projection weight (to_out) – small GEMM.
        #        out = norm_tile @ linear_w.T   (dim ← hidden)
        # ------------------------------------------------------------------
        for d in range(0, tl.cdiv(dim, BLOCK_D)):
            offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
            mask_d = offs_d < dim

            # weight slice: [BLOCK_D, BLOCK_H] (row‑major in memory)
            w_ptr = linear_w_ptr + offs_d[:, None] * hidden + offs_h[None, :]  # [BLOCK_D, BLOCK_H]

            w_tile = tl.load(w_ptr,
                             mask=mask_d[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float32)               # FP16 → FP32

            # dot: (BLOCK_M, BLOCK_H) x (BLOCK_H, BLOCK_D) -> (BLOCK_M, BLOCK_D)
            acc = tl.dot(norm_tile, w_tile)                         # (BLOCK_M, BLOCK_D)

            # store result
            out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
            tl.store(out_ptrs,
                     acc.to(tl.float16),
                     mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the **outgoing** TriMul operator (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, C] (float32)
        - mask_tensor  : torch.Tensor, shape [B, N, N] (float32/bool)
        - weights_dict : dict of model parameters (float32)
        - config_dict  : {
                "dim"        : int    (C),
                "hidden_dim" : int    (H),
                "nomask"     : bool,
                "eps"        : float (optional, default 1e-5)
          }

    Returns
    -------
    torch.Tensor : shape [B, N, N, C] (float32) on the same CUDA device.
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    inp, mask, weights, cfg = data
    device = inp.device
    dim = cfg["dim"]                # C
    hidden = cfg["hidden_dim"]      # H
    eps = cfg.get("eps", 1e-5)
    nomask = cfg.get("nomask", True)

    B, N, _, _ = inp.shape
    total_rows = B * N * N                         # R = B·N²

    # --------------------------------------------------------------
    # 1️⃣ Row‑wise mean / rstd (output: mean[R], rstd[R])
    # --------------------------------------------------------------
    inp_flat = inp.view(total_rows, dim).contiguous()               # FP32
    mean = torch.empty(total_rows, dtype=torch.float32, device=device)
    rstd = torch.empty(total_rows, dtype=torch.float32, device=device)

    BLOCK_M = 128
    BLOCK_D = 128
    grid_stat = (triton.cdiv(total_rows, BLOCK_M),)

    row_stats_kernel[grid_stat](
        inp_flat,
        mean,
        rstd,
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # --------------------------------------------------------------
    # 2️⃣ Fuse LayerNorm scaling into the projection weights
    # --------------------------------------------------------------
    # Concatenate the five weight matrices (each [hidden, dim])
    weight_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).contiguous()                                 # (5*H, C)  float32

    gamma = weights["norm.weight"]                  # (C,)  float32
    beta  = weights["norm.bias"]                    # (C,)  float32

    # Apply γ to the projection weights (W' = γ ⊙ W)
    weight_scaled_fp32 = weight_concat * gamma[None, :]       # (5*H, C)
    # Pre‑compute Σₖ γₖ·W_{i,k}   and   Σₖ βₖ·W_{i,k}
    ws   = weight_scaled_fp32.sum(dim=1)                      # (5*H,)  float32
    bias = (weight_concat * beta[None, :]).sum(dim=1)         # (5*H,)  float32

    # Cast to FP16 for the GEMM
    weight_scaled_fp16 = weight_scaled_fp32.half().contiguous()
    inp_half = inp_flat.half()                                 # (R, C) FP16

    # --------------------------------------------------------------
    # 3️⃣ Linear projection on the original tensor (FP16 GEMM)
    # --------------------------------------------------------------
    raw_fp16 = F.linear(inp_half, weight_scaled_fp16)          # (R, 5*H) FP16
    raw = raw_fp16.float()                                    # FP32 for the next step

    # --------------------------------------------------------------
    # 4️⃣ Recover full LayerNorm‑treated projections (vectorised)
    # --------------------------------------------------------------
    # (raw - μ·ws) * rstd + bias
    out_concat = (raw - mean.unsqueeze(1) * ws.unsqueeze(0)) * rstd.unsqueeze(1) + bias.unsqueeze(0)

    # Split concatenated result
    left_proj, right_proj, left_gate, right_gate, out_gate = torch.split(
        out_concat, hidden, dim=1
    )  # each (R, H) FP32

    # --------------------------------------------------------------
    # 5️⃣ Optional binary mask (broadcast on hidden dim)
    # --------------------------------------------------------------
    if not nomask:
        mask_flat = mask.view(total_rows, 1).to(torch.float32)  # (R,1)
        left_proj = left_proj * mask_flat
        right_proj = right_proj * mask_flat

    # --------------------------------------------------------------
    # 6️⃣ Sigmoid gates
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)       # (R, H) FP32
    right_gate = torch.sigmoid(right_gate)     # (R, H) FP32
    out_gate = torch.sigmoid(out_gate)         # (R, H) FP32  (used later)

    # --------------------------------------------------------------
    # 7️⃣ Apply gates and reshape for the batched GEMM
    # --------------------------------------------------------------
    left = (left_proj * left_gate).half()      # (R, H) FP16
    right = (right_proj * right_gate).half()   # (R, H) FP16

    # reshape to (B, H, N, N) for the batched matmul
    left_mat = left.view(B, N, N, hidden).permute(0, 3, 1, 2).reshape(B * hidden, N, N)   # (B·H, N, N)
    right_mat = right.view(B, N, N, hidden).permute(0, 3, 2, 1).reshape(B * hidden, N, N) # (B·H, N, N)

    # Batched matrix multiplication (Tensor‑Core, FP16)
    out_mat = torch.bmm(left_mat, right_mat)  # (B·H, N, N) FP16

    # --------------------------------------------------------------
    # 8️⃣ Second LayerNorm + out‑gate + final projection (fused Triton)
    # --------------------------------------------------------------
    # Flatten to (R, H)
    out_flat = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous().view(total_rows, hidden)

    # Allocate output tensor (final dim)
    final_fp16 = torch.empty(total_rows, dim, dtype=torch.half, device=device)

    # Cast out_gate to half for the kernel
    out_gate_half = out_gate.half().contiguous()

    # Weight for the second LayerNorm (hidden → hidden)
    to_out_norm_w = weights["to_out_norm.weight"]  # FP32, shape (H,)
    to_out_norm_b = weights["to_out_norm.bias"]    # FP32, shape (H,)

    # Final projection weight (dim, hidden) – half precision
    to_out_w_half = weights["to_out.weight"].half().contiguous()

    # Grid for the fused kernel (process rows in blocks)
    BLOCK_M2 = 128          # rows per program
    BLOCK_H  = 64           # hidden dimension chunk
    BLOCK_D2 = 64           # output‑dim chunk
    grid_fused = (triton.cdiv(total_rows, BLOCK_M2),)

    fused_norm_gate_linear_kernel[grid_fused](
        out_flat,                               # input (R, H) half
        final_fp16,                             # output (R, dim) half
        to_out_norm_w,                          # γ (H) FP32
        to_out_norm_b,                          # β (H) FP32
        out_gate_half,                          # out‑gate (R, H) half
        to_out_w_half,                          # final linear weight (dim, H) half
        total_rows,
        hidden,
        dim,
        eps,
        BLOCK_M=BLOCK_M2,
        BLOCK_H=BLOCK_H,
        BLOCK_D=BLOCK_D2,
    )

    # --------------------------------------------------------------
    # 9️⃣ Reshape to the original 4‑D shape and cast back to FP32
    # --------------------------------------------------------------
    final = final_fp16.view(B, N, N, dim).float()

    return final
