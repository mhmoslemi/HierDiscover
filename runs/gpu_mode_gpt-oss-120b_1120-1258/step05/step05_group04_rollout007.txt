import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# Triton kernels

@triton.jit
def layernorm_gate_linear_kernel(
    x_ptr,               # *const half*  [total_rows, hidden]
    gate_ptr,            # *const half*  [total_rows, hidden] (sigmoid out_gate)
    norm_w_ptr,          # *const float* [hidden] (γ)
    norm_b_ptr,          # *const float* [hidden] (β)
    out_w_ptr,           # *const half*  [dim, hidden] (to_out weight)
    out_ptr,             # *mut   float* [total_rows, dim]
    total_rows: tl.int32,
    dim: tl.constexpr,
    hidden: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,   # rows per block
    BLOCK_D: tl.constexpr,   # output feature chunk
):
    pid = tl.program_id(0)
    pid_d = tl.program_id(1)

    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # Hidden dimension load
    offs_h = tl.arange(0, hidden)
    mask_h = offs_h < hidden

    # Load x and gate
    x = tl.load(x_ptr + offs_m[:, None] * hidden + offs_h[None, :],
                mask=mask_m[:, None] & mask_h[None, :],
                other=0.0).to(tl.float32)   # (BLOCK_M, hidden)

    g = tl.load(gate_ptr + offs_m[:, None] * hidden + offs_h[None, :],
                mask=mask_m[:, None] & mask_h[None, :],
                other=0.0).to(tl.float32)   # (BLOCK_M, hidden)

    # Mean / variance per row
    sum_val = tl.sum(x, axis=1)
    sum_sq = tl.sum(x * x, axis=1)
    mean = sum_val / hidden
    var = sum_sq / hidden - mean * mean
    rstd = tl.rsqrt(var + eps)

    # Apply LN weight/bias
    w_norm = tl.load(norm_w_ptr + offs_h, mask=mask_h, other=0.0)
    b_norm = tl.load(norm_b_ptr + offs_h, mask=mask_h, other=0.0)

    norm = (x - mean[:, None]) * rstd[:, None]   # (BLOCK_M, hidden)
    norm = norm * w_norm[None, :] + b_norm[None, :]
    gated = norm * g                               # (BLOCK_M, hidden)

    # Output-channel block
    d_start = pid_d * BLOCK_D
    offs_d = d_start + tl.arange(0, BLOCK_D)
    mask_d = offs_d < dim

    # Load output weight slice (hidden, BLOCK_D) – note weight stored as [dim, hidden]
    w = tl.load(out_w_ptr + offs_h[:, None] + offs_d[None, :] * hidden,
                mask=mask_h[:, None] & mask_d[None, :],
                other=0.0).to(tl.float32)      # (hidden, BLOCK_D)

    out_chunk = tl.dot(gated, w)                 # (BLOCK_M, BLOCK_D)

    tl.store(out_ptr + offs_m[:, None] * dim + offs_d[None, :],
             out_chunk,
             mask=mask_m[:, None] & mask_d[None, :])

---------

    out_gate   = torch.sigmoid(out_gate_fp16)        # (R, H) FP16 (used later)

    left  = left_proj_fp16 * left_gate               # (R, H) FP16
    right = right_proj_fp16 * right_gate             # (R, H) FP16

    # ------------------------------------------------------------------
    # 3️⃣ Triangular multiplicative update (batched half‑precision GEMM)
    # ------------------------------------------------------------------
    # Reshape to (B, N, N, H) → (B, H, N, N) → (B·H, N, N)
    left = left.view(B, N, N, hidden).permute(0, 3, 1, 2).reshape(B * hidden, N, N)   # (B·H, N, N)
    right = right.view(B, N, N, hidden).permute(0, 3, 2, 1).reshape(B * hidden, N, N) # (B·H, N, N)

    # Half‑precision batched matrix multiply.
    out_mat = torch.bmm(left, right)                 # (B·H, N, N) FP16

    # Reshape back to (B, N, N, H)
    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H) FP16

    # ------------------------------------------------------------------
    # 4️⃣ Fused LayerNorm + out‑gate + final projection (Triton)
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, hidden).contiguous()   # (R, H) FP16
    final_fp32 = torch.empty(total_rows, dim, dtype=torch.float32, device=device)

    BLOCK_M = 128          # rows per program (tuned for H100)
    BLOCK_D = 128          # output‑channel tile size

    grid = (
        triton.cdiv(total_rows, BLOCK_M),
        triton.cdiv(dim, BLOCK_D),
    )
    layernorm_gate_linear_kernel[grid](
        out_flat,                               # x_ptr
        out_gate,                               # gate_ptr (already sigmoid)
        weights["to_out_norm.weight"],          # γ
        weights["to_out_norm.bias"],            # β
        weights["to_out.weight"].half().contiguous(),  # to_out weight (dim, hidden)
        final_fp32,                             # out_ptr
        total_rows,
        dim=dim,
        hidden=hidden,
        eps=eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 5️⃣ Reshape to final tensor shape (B, N, N, C) and return
    # ------------------------------------------------------------------
    final = final_fp32.view(B, N, N, dim)        # float32
    return final
```
