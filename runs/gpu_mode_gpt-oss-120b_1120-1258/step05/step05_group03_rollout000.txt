"""
Highly‑optimized forward for the **outgoing** TriMul operator (AlphaFold‑3).

Algorithm
---------
1️⃣  Input LayerNorm (FP16) – we reuse PyTorch’s fused LayerNorm kernel
    (still meets the requirement of using a Triton kernel later).

2️⃣  **One** fused linear projection:
        W = [left_proj; left_gate; right_proj; right_gate; out_gate]  # (5*H, C)
    This replaces three separate GEMMs with a single half‑precision matmul.
    The result is split into the 5 required tensors and the sigmoid gates
    are applied.

3️⃣  Optional binary mask – broadcasted on the hidden dimension.

4️⃣  Triangular multiplicative update:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    Implemented as a batched half‑precision GEMM via `torch.bmm`
    (Tensor‑Core on H100).

5️⃣  Fused LayerNorm + output‑gate – a custom Triton kernel that
    computes the mean/variance in FP32, applies the learned γ/β
    (to_out_norm) and multiplies by the pre‑computed `out_gate`.

6️⃣  Final linear projection back to `dim` – half‑precision GEMM.

All heavy arithmetic stays in FP16/Tensor‑Core; only the reductions in
LayerNorm are carried out in FP32 for numerical stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 5️⃣  Fused LayerNorm + output‑gate (half → half)
# ----------------------------------------------------------------------
@triton.jit
def fused_layernorm_gate_kernel(
    x_ptr,           # const half *,   [R, H]
    y_ptr,           # mutable half *,[R, H]   (norm·gate)
    weight_ptr,      # const float *, [H]   (γ)
    bias_ptr,        # const float *, [H]   (β)
    gate_ptr,        # const half *,  [R, H] (out_gate already sigmoid’ed)
    total_rows: tl.int32,
    hidden: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)            # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ---------- first pass: mean / var ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(hidden, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < hidden
        ptrs = x_ptr + offs_m[:, None] * hidden + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)        # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / hidden
    var  = sum_sq / hidden - mean * mean
    rstd = tl.rsqrt(var + eps)                           # 1/√(var+ε)

    # ---------- second pass: norm, γ/β, out‑gate ----------
    for d in range(0, tl.cdiv(hidden, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < hidden
        # load normalized values
        ptrs = x_ptr + offs_m[:, None] * hidden + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]    # (x-μ)/σ

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]                     # FP32

        # multiply by the pre‑computed out_gate
        gate_vals = tl.load(gate_ptr + offs_m[:, None] * hidden + offs_d[None, :],
                            mask=mask_m[:, None] & mask_d[None, :],
                            other=0.0).to(tl.float32)            # FP32
        out = out * gate_vals

        tl.store(y_ptr + offs_m[:, None] * hidden + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, C] (float32)
        - mask_tensor  : torch.Tensor [B, N, N]   (float32/bool). Ignored when config["nomask"]=True.
        - weights_dict : model parameters (float32)
        - config_dict  : {"dim":C, "hidden_dim":H, "nomask":bool, "eps":float (optional)}

    Returns
    -------
    torch.Tensor [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # 0️⃣ unpack
    # ------------------------------------------------------------------
    inp, mask, weights, cfg = data
    device = inp.device
    dim = cfg["dim"]                       # C
    hidden = cfg["hidden_dim"]             # H
    eps = cfg.get("eps", 1e-5)
    nomask = cfg.get("nomask", True)

    B, N, _, _ = inp.shape
    total_rows = B * N * N                # R = B·N·N

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP16) – use PyTorch’s fused implementation
    # ------------------------------------------------------------------
    # Convert to half and apply LayerNorm across the last dimension.
    inp_fp16 = inp.half()
    weight_norm = weights["norm.weight"].half()
    bias_norm   = weights["norm.bias"].half()
    x_norm = F.layer_norm(inp_fp16,
                          (dim,),
                          weight=weight_norm,
                          bias=bias_norm,
                          eps=eps)                 # (B, N, N, C) half

    # Flatten for the fused linear projection
    x_flat = x_norm.view(total_rows, dim).contiguous()   # (R, C)

    # ------------------------------------------------------------------
    # 2️⃣ One fused linear projection for left/right/out gates
    # ------------------------------------------------------------------
    # Gather all five weight matrices (each H×C)
    left_proj_w = weights["left_proj.weight"].half()
    left_gate_w = weights["left_gate.weight"].half()
    right_proj_w = weights["right_proj.weight"].half()
    right_gate_w = weights["right_gate.weight"].half()
    out_gate_w = weights["out_gate.weight"].half()

    # Combined weight: (5*H, C)
    combined_weight = torch.cat([
        left_proj_w,
        left_gate_w,
        right_proj_w,
        right_gate_w,
        out_gate_w,
    ], dim=0).contiguous()                               # (5*H, C)

    # Single half‑precision GEMM
    combined = F.linear(x_flat, combined_weight)        # (R, 5*H)

    # Split into the five required tensors
    left_proj_raw, left_gate_raw, right_proj_raw, right_gate_raw, out_gate_raw = torch.split(
        combined, hidden, dim=1
    )                                                    # each (R, H)

    # Apply sigmoid gates
    left_gate = torch.sigmoid(left_gate_raw)            # (R, H)
    right_gate = torch.sigmoid(right_gate_raw)          # (R, H)
    out_gate = torch.sigmoid(out_gate_raw)              # (R, H) – will be used later

    # Gated projections
    left = left_proj_raw * left_gate                    # (R, H)
    right = right_proj_raw * right_gate                 # (R, H)

    # ------------------------------------------------------------------
    # 2a️⃣ Optional binary mask (broadcasted on hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask_flat = mask.view(total_rows, 1).to(torch.float16)
        left = left * mask_flat
        right = right * mask_flat

    # ------------------------------------------------------------------
    # 3️⃣ Triangular multiplicative update (batched GEMM)
    # ------------------------------------------------------------------
    # Reshape to (B, N, N, H)
    left = left.view(B, N, N, hidden)          # (B, N, N, H) half
    right = right.view(B, N, N, hidden)        # (B, N, N, H) half

    # Prepare for batched matmul: (B*H, N, N)
    left_mat = left.permute(0, 3, 1, 2).reshape(B * hidden, N, N)
    right_mat = right.permute(0, 3, 2, 1).reshape(B * hidden, N, N)

    # Compute out_mat = left_mat @ right_mat  (half, Tensor‑Core)
    out_mat = torch.bmm(left_mat, right_mat)   # (B*H, N, N)

    # Back to (B, N, N, H)
    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()  # half

    # ------------------------------------------------------------------
    # 4️⃣ Fused LayerNorm + out‑gate (Triton kernel)
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, hidden).contiguous()   # (R, H) half
    out_norm_gate = torch.empty_like(out_flat)             # (R, H) half

    # Choose block sizes that cover the hidden dimension efficiently
    BLOCK_M = 128
    BLOCK_D = 64 if hidden <= 256 else 128

    grid = (triton.cdiv(total_rows, BLOCK_M),)
    fused_layernorm_gate_kernel[grid](
        out_flat,
        out_norm_gate,
        weights["to_out_norm.weight"],          # float32 γ
        weights["to_out_norm.bias"],            # float32 β
        out_gate,                               # half, already sigmoid’ed
        total_rows,
        hidden,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 5️⃣ Final linear projection back to dim (half GEMM)
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].half()           # (dim, H)
    final_fp16 = F.linear(out_norm_gate, to_out_w)      # (R, dim) half

    # Reshape to original 4‑D layout and cast to float32 as required
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
