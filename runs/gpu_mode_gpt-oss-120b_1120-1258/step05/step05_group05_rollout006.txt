"""
Fast outgoing TriMul forward (AlphaFold‑3) – Triton‑accelerated layer‑norms.

Algorithm
---------
1️⃣  Row‑wise LayerNorm over the input channel dimension `C`.
    - Input cast to FP16 → mean/var accumulated in FP32.
    - Normalised result written as FP16.

2️⃣  Fuse the five linear projections (left/right + three gates) into a single
    half‑precision GEMM by concatenating their weight matrices.
    - Optional pair‑wise mask applied after the projection.
    - Sigmoid gates are computed in FP16.

3️⃣  Reshape the projected tensors to `[B, H, N, N]` and perform the
    triangular multiplicative update (outgoing version) as a batched GEMM:
        out_h[i, j] = Σ_k left_h[i, k] * right_h[j, k]
    – realised with `torch.bmm(left, right.transpose(1, 2))`.

4️⃣  Row‑wise LayerNorm on the hidden dimension `H`.

5️⃣  Multiply by the output gate, then a final half‑precision linear projection
    back to the original dimension `C`.

All heavy reductions (both LayerNorms) are performed by a custom Triton kernel,
while the dense linear algebra (projections, batched GEMM, final projection) is
delegated to cuBLAS/Tensor‑Core kernels.  The implementation works for the
full test matrix of shapes, supports optional masking and runs completely on
the GPU.  The final tensor is returned in FP32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton row‑wise LayerNorm (FP16 output, FP32 reduction)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,          # *const void*   [total_rows, dim]  (FP16 or FP32)
    y_ptr,          # *mut   fp16*   [total_rows, dim]
    weight_ptr,     # *const float* [dim]   (γ)
    bias_ptr,       # *const float* [dim]   (β)
    total_rows,     # i32
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per loop iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # -------------------------------------------------
    # 1. Compute mean & variance (FP32 reduction)
    # -------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    # iterate over dimension in BLOCK_D sized chunks
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim

        # address: [row, col]
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        # Load as half (or float) and up‑cast to FP32
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                                 # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                  # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)           # [BLOCK_M]

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                           # 1 / sqrt(var+eps)

    # -------------------------------------------------
    # 2. Write normalised output (FP16)
    # -------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                                 # fp32

        norm = (vals - mean[:, None]) * rstd[:, None]    # fp32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # fp32
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # fp32

        out_fp16 = (norm * w[None, :] + b[None, :]).to(tl.float16)

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
# Custom kernel entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor of shape [B, N, N, C] (float32)
            - mask_tensor  : torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights_dict : mapping name → torch.Tensor (float32)
            - config_dict  : {"dim": C, "hidden_dim": H, "nomask": bool}
    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # --------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                # input channel dimension C
    hidden_dim = config["hidden_dim"]  # hidden dimension H
    eps = 1e-5
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N

    # --------------------------------------------------
    # 1️⃣ First LayerNorm (FP16 output)
    # --------------------------------------------------
    # Cast input to FP16 once, then normalise with Triton
    input_half = input_tensor.to(torch.float16).contiguous()
    x_flat = input_half.view(total_rows, dim)                # [R, C]

    x_norm_fp16 = torch.empty(total_rows, dim,
                              dtype=torch.float16,
                              device=device)

    # launch parameters (tuned for H100)
    BLOCK_M = 512          # rows per program
    BLOCK_D = 128          # columns per iteration (covers up to dim=384 in ≤3 loops)
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # --------------------------------------------------
    # 2️⃣ Fuse five linear projections (left/right + three gates)
    # --------------------------------------------------
    # Concatenate weight matrices (each: [H, C])
    w_left    = weights["left_proj.weight"]
    w_right   = weights["right_proj.weight"]
    w_lgate   = weights["left_gate.weight"]
    w_rgate   = weights["right_gate.weight"]
    w_outgate = weights["out_gate.weight"]

    concat_w = torch.cat([w_left, w_right, w_lgate, w_rgate, w_outgate],
                         dim=0).to(torch.float16).contiguous()   # [(5*H), C]

    # Single GEMM: (R, C) @ (C, 5*H)^T → (R, 5*H)
    proj_all = F.linear(x_norm_fp16, concat_w)                # half

    # Split back to the five tensors
    left_proj_fp16, right_proj_fp16, left_gate_fp16, \
        right_gate_fp16, out_gate_fp16 = torch.split(
            proj_all, hidden_dim, dim=1
        )                                                    # each (R, H)

    # --------------------------------------------------
    # 3️⃣ Apply optional mask (before gating)
    # --------------------------------------------------
    if (not nomask) and mask_tensor is not None:
        mask_fp16 = mask_tensor.reshape(total_rows, 1).to(torch.float16)
        left_proj_fp16  = left_proj_fp16 * mask_fp16
        right_proj_fp16 = right_proj_fp16 * mask_fp16

    # --------------------------------------------------
    # 4️⃣ Sigmoid gates and element‑wise gating (still FP16)
    # --------------------------------------------------
    left_gate   = torch.sigmoid(left_gate_fp16)      # (R, H)
    right_gate  = torch.sigmoid(right_gate_fp16)
    out_gate    = torch.sigmoid(out_gate_fp16)

    left_fp16   = left_proj_fp16 * left_gate
    right_fp16  = right_proj_fp16 * right_gate

    # Reshape back to 4‑D layout: [B, N, N, H]
    left  = left_fp16.view(B, N, N, hidden_dim)
    right = right_fp16.view(B, N, N, hidden_dim)
    out_gate = out_gate.view(B, N, N, hidden_dim)

    # --------------------------------------------------
    # 5️⃣ Triangular multiplicative update (batched GEMM)
    #    out_h[i, j] = Σ_k left_h[i, k] * right_h[j, k]
    # --------------------------------------------------
    # Move hidden dimension to the second axis for batched GEMM
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    BH = B * hidden_dim
    left_mat  = left_perm.view(BH, N, N)                 # [BH, N, N]  (FP16)
    right_mat = right_perm.view(BH, N, N)                # [BH, N, N]  (FP16)

    # Batched matmul: left @ rightᵀ → [BH, N, N]
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Restore layout [B, N, N, H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------
    # 6️⃣ Second LayerNorm over the hidden dimension H
    # --------------------------------------------------
    out_flat = out.view(total_rows, hidden_dim)          # [R, H]
    out_norm_fp16 = torch.empty_like(out_flat,
                                     dtype=torch.float16,
                                     device=device)

    # Grid for second LN (same BLOCK_M, BLOCK_D=128 works for H≤128)
    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid2](
        out_flat,                     # FP16 input (cast inside)
        out_norm_fp16,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    out_norm = out_norm_fp16.view(B, N, N, hidden_dim)   # [B, N, N, H]

    # --------------------------------------------------
    # 7️⃣ Final gating + output projection
    # --------------------------------------------------
    gated = out_norm * out_gate                            # FP16

    to_out_w_fp16 = weights["to_out.weight"].to(torch.float16).contiguous()
    gated_flat = gated.view(total_rows, hidden_dim)        # [R, H]
    final_fp16 = F.linear(gated_flat, to_out_w_fp16)       # (R, C) half

    # --------------------------------------------------
    # 8️⃣ Cast back to FP32 and return
    # --------------------------------------------------
    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final
