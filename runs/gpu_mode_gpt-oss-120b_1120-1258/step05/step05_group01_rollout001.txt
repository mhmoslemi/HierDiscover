"""
Outgoing TriMul (AlphaFold‑3) – Triton‑accelerated forward pass (no gradients)

Algorithm
---------
1️⃣  Row‑wise LayerNorm (FP32 input → FP16 output).
2️⃣  Fuse the five linear projections (left_proj, right_proj,
    left_gate, right_gate, out_gate) into a single FP16 GEMM.
3️⃣  Optional pair‑wise mask applied only to the two projection tensors.
4️⃣  Apply sigmoid gates → gated tensors *left* and *right*.
5️⃣  Batched matrix multiplication:
        out_perm[b, h, i, j] = Σₖ left[b, i, k, h] · right[b, j, k, h]
    using torch.matmul on non‑contiguous views (no extra copies).
6️⃣  Fuse the second LayerNorm, the output‑gate and write to a contiguous
    (R, H) buffer (R = B·N·N) with a custom Triton kernel.
7️⃣  Final linear projection back to the model dimension and cast to FP32.

All heavy work (layer‑norms, final LN+gate) is performed in Triton
kernels, while the two GEMM’s (projection & final linear) rely on
cuBLAS/Tensor‑Core FP16 matrix‑multiplication for maximum throughput.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# 1️⃣  Row‑wise LayerNorm kernel (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   * [total_rows, dim]   (fp32)
    y_ptr,          # mutable * [total_rows, dim]   (fp16)
    w_ptr,          # const   * [dim]               (fp32)   gamma
    b_ptr,          # const   * [dim]               (fp32)   beta
    total_rows,
    dim,
    eps,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # -------- pass 1: compute mean & variance per row ------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim

        # load a tile (BLOCK_M, BLOCK_D)
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        val = tl.load(ptr,
                      mask=mask_m[:, None] & mask_d[None, :],
                      other=0.0).to(tl.float32)

        sum_val += tl.sum(val, axis=1)
        sum_sq  += tl.sum(val * val, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # -------- pass 2: write normalized values (γ/β) as fp16 ----------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        val = tl.load(ptr,
                      mask=mask_m[:, None] & mask_d[None, :],
                      other=0.0).to(tl.float32)

        norm = (val - mean[:, None]) * rstd[:, None]                # fp32

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)         # γ  (fp32)
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)         # β  (fp32)

        out = norm * w[None, :] + b[None, :]                        # fp32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16, mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
# 6️⃣  Fused second LayerNorm + out‑gate kernel (writes contiguous (R,H))
# ----------------------------------------------------------------------
@triton.jit
def fused_ln_gate_kernel(
    src_ptr,                # const   * [B, H, N, N] (fp16) – non‑contiguous
    gate_ptr,               # const   * [R, H]      (fp16) – already fused sigmoid(out_gate)
    dst_ptr,                # mutable * [R, H]      (fp16) – final hidden after LN & gate
    ln_w_ptr,               # const   * [H]         (fp32) – γ of second LN
    ln_b_ptr,               # const   * [H]         (fp32) – β of second LN
    total_rows,             # int      = B·N·N
    hidden_dim,             # int      (compile‑time, = H)
    seq_len,                # int      = N
    stride_b,               # int
    stride_h,               # int
    stride_i,               # int
    stride_j,               # int
    eps,
    BLOCK_M: tl.constexpr,  # rows per program
    BLOCK_D: tl.constexpr,  # tile over hidden dimension
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ----- decode linear row index → (b,i,j) ---------------------------
    seq_len_sq = seq_len * seq_len
    b_idx = offs_m // seq_len_sq
    rem   = offs_m % seq_len_sq
    i_idx = rem // seq_len
    j_idx = rem % seq_len

    # base offset for the first hidden element of each row
    base_off = (b_idx * stride_b) + (i_idx * stride_i) + (j_idx * stride_j)

    # -------- pass 1: compute mean & variance over hidden dim ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(hidden_dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < hidden_dim

        # load a hidden tile (stride over hidden dim = stride_h)
        ptr = src_ptr + base_off[:, None] + offs_d[None, :] * stride_h
        val = tl.load(ptr,
                      mask=mask_m[:, None] & mask_d[None, :],
                      other=0.0).to(tl.float32)

        sum_val += tl.sum(val, axis=1)
        sum_sq  += tl.sum(val * val, axis=1)

    mean = sum_val / hidden_dim
    var  = sum_sq / hidden_dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # -------- pass 2: normalize, apply γ/β, multiply with out_gate -----
    for d in range(0, tl.cdiv(hidden_dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < hidden_dim

        # raw hidden values
        ptr = src_ptr + base_off[:, None] + offs_d[None, :] * stride_h
        raw = tl.load(ptr,
                      mask=mask_m[:, None] & mask_d[None, :],
                      other=0.0).to(tl.float32)

        # LayerNorm
        norm = (raw - mean[:, None]) * rstd[:, None]      # fp32

        # γ/β (fp32)
        w = tl.load(ln_w_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(ln_b_ptr + offs_d, mask=mask_d, other=0.0)
        out = norm * w[None, :] + b[None, :]               # fp32

        # out_gate (fp16 → fp32)
        gate = tl.load(gate_ptr + offs_m[:, None] * hidden_dim + offs_d[None, :],
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        out = out * gate                                   # fp32

        # store as fp16
        out_fp16 = out.to(tl.float16)
        dst_ptrs = dst_ptr + offs_m[:, None] * hidden_dim + offs_d[None, :]
        tl.store(dst_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

# Custom forward function for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [B, N, N, dim] (float32, on CUDA)
    """
    # --------------------------------------------------------------
    # 0) unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim          = config["dim"]           # e.g. 128 or 384
    hidden_dim   = config["hidden_dim"]   # always 128 in the provided tests
    nomask       = config.get("nomask", True)
    eps          = 1e-5
    device       = input_tensor.device
    B, N, _, _   = input_tensor.shape
    total_rows   = B * N * N               # R = B·N·N

    # --------------------------------------------------------------
    # 1) First LayerNorm (row‑wise) – output in fp16
    # --------------------------------------------------------------
    # Flatten to (R, dim)
    x_flat = input_tensor.view(total_rows, dim).contiguous()
    x_norm = torch.empty(total_rows, dim, dtype=torch.float16, device=device)

    # Tile sizes – chosen to fit both dim=128 and dim=384 efficiently
    BLOCK_M = 256
    BLOCK_D_IN = 64 if dim > 128 else 128

    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D_IN,
    )

    # --------------------------------------------------------------
    # 2) Fused linear projection (5 * hidden_dim outputs)
    # --------------------------------------------------------------
    # Concatenate weight matrices once and cast to fp16
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)               # shape (5*H, dim)

    # Linear: [R, dim] @ [dim, 5*H]^T  →  [R, 5*H]  (FP16 GEMM)
    proj_out = F.linear(x_norm, w_concat)   # bias is None

    H = hidden_dim
    left_proj   = proj_out[:, 0:H]
    right_proj  = proj_out[:, H:2*H]
    left_gate   = proj_out[:, 2*H:3*H]
    right_gate  = proj_out[:, 3*H:4*H]
    out_gate    = proj_out[:, 4*H:5*H]

    # sigmoid gates (still fp16)
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    # --------------------------------------------------------------
    # 3) Optional mask (applied only to the two projections)
    # --------------------------------------------------------------
    if not nomask:
        # mask shape [B, N, N] → [R, 1]
        mask_flat = mask_tensor.to(torch.float16).view(total_rows, 1)
        left_proj  = left_proj * mask_flat
        right_proj = right_proj * mask_flat

    # --------------------------------------------------------------
    # 4) Apply gates → gated tensors `left` and `right`
    # --------------------------------------------------------------
    left  = left_proj * left_gate    # (R, H)
    right = right_proj * right_gate  # (R, H)

    # reshape back to (B, N, N, H) – still contiguous because H is last dim
    left  = left.view(B, N, N, H)
    right = right.view(B, N, N, H)
    # out_gate stays flat (R, H) for the fused LN+gate kernel

    # --------------------------------------------------------------
    # 5) Batched matrix multiplication: out_perm = left @ rightᵀ
    #    Shape (B, H, N, N)
    # --------------------------------------------------------------
    left_perm  = left.permute(0, 3, 1, 2)          # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2)         # (B, H, N, N)
    out_perm = torch.matmul(left_perm, right_perm.transpose(-1, -2))  # (B, H, N, N)

    # --------------------------------------------------------------
    # 6) Second LayerNorm + out‑gate (fused Triton kernel)
    #    Output -> contiguous (R, H) fp16 tensor `hidden_out`
    # --------------------------------------------------------------
    # Strides of the source tensor (non‑contiguous view)
    stride_b = out_perm.stride(0)   # stride over B
    stride_h = out_perm.stride(1)   # stride over H (normally N*N)
    stride_i = out_perm.stride(2)   # stride over i (N)
    stride_j = out_perm.stride(3)   # stride over j (1)

    hidden_out = torch.empty(total_rows, H, dtype=torch.float16, device=device)

    BLOCK_D_H = 128   # hidden_dim is always 128 → one tile fits exactly
    grid_ln2 = (triton.cdiv(total_rows, BLOCK_M),)
    fused_ln_gate_kernel[grid_ln2](
        out_perm,                       # src (B, H, N, N) fp16
        out_gate,                       # gate (R, H) fp16
        hidden_out,                     # dst (R, H) fp16
        weights["to_out_norm.weight"],  # γ (fp32)
        weights["to_out_norm.bias"],    # β (fp32)
        total_rows,
        H,
        N,
        stride_b,
        stride_h,
        stride_i,
        stride_j,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D_H,
    )

    # --------------------------------------------------------------
    # 7) Final projection back to model dimension and cast to FP32
    # --------------------------------------------------------------
    w_out = weights["to_out.weight"].to(torch.float16)   # shape (dim, H)
    final_half = F.linear(hidden_out, w_out)             # (R, dim) fp16
    final = final_half.view(B, N, N, dim).to(torch.float32)

    return final
