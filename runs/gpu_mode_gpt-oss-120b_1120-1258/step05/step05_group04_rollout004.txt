"""
Highly‑optimized Triton + PyTorch forward for the **outgoing** TriMul
operator (AlphaFold‑3).  

Algorithm
---------
1️⃣ Input LayerNorm – use the native PyTorch kernel (fast, fused
   reduction) and cast to FP16 for the remaining pipeline.
2️⃣ Fuse the five linear projections (left/right + 3 gates) into one
   half‑precision GEMM (`torch.nn.functional.linear`).  This yields the
   tensors `left_proj, right_proj, left_gate, right_gate, out_gate`.
3️⃣ Optional binary mask is applied only to the left/right projections.
4️⃣ Apply the sigmoid to the three gate tensors and fuse the gating
   (`proj * gate`).
5️⃣ **Triangular multiplicative update** – compute  
   `out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]`  
   as a batched matrix‑multiply with `torch.matmul`.  By keeping the
   tensors in the layout `(B, H, N, N)` we avoid any extra copies that
   `reshape`+`bmm` would incur.
6️⃣ Fuse the final LayerNorm, out‑gate, and output projection into a
   single Triton kernel (`layernorm_gate_linear_kernel`).  This removes
   two separate memory‑bound passes.
7️⃣ Reshape to `[B, N, N, dim]` (FP32) and return.

All heavy arithmetic stays in FP16/Tensor‑Core, while the two LayerNorm
steps run in FP32 for numerical stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# 1) LayerNorm → FP16 (used for the *output* tensor, not the input)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_gate_linear_kernel(
    x_ptr,               # const half [total_rows, hidden]   (pre‑norm)
    gate_ptr,            # const half [total_rows, hidden]   (out_gate sigmoid)
    norm_w_ptr,          # const float[hidden]                (γ of to_out_norm)
    norm_b_ptr,          # const float[hidden]                (β of to_out_norm)
    out_w_ptr,           # const half [dim, hidden]           (to_out weight)
    out_ptr,             # mut float [total_rows, dim]        (final FP32)
    total_rows: tl.int32,
    dim: tl.constexpr,
    hidden: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,   # rows per program (e.g. 64)
    BLOCK_D: tl.constexpr,   # output‑channel chunk (e.g. 64)
):
    pid = tl.program_id(0)          # row block
    pid_d = tl.program_id(1)        # output‑channel block
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # -----------------------------------------------------------------
    # Load the entire hidden vector for the selected rows
    # -----------------------------------------------------------------
    offs_h = tl.arange(0, hidden)
    mask_h = offs_h < hidden

    x = tl.load(x_ptr + offs_m[:, None] * hidden + offs_h[None, :],
                mask=mask_m[:, None] & mask_h[None, :],
                other=0.0).to(tl.float32)           # (BLOCK_M, hidden)

    g = tl.load(gate_ptr + offs_m[:, None] * hidden + offs_h[None, :],
                mask=mask_m[:, None] & mask_h[None, :],
                other=0.0).to(tl.float32)           # (BLOCK_M, hidden)

    # -----------------------------------------------------------------
    # Row‑wise mean / variance (FP32)
    # -----------------------------------------------------------------
    sum_val = tl.sum(x, axis=1)
    sum_sq  = tl.sum(x * x, axis=1)
    mean = sum_val / hidden
    var  = sum_sq / hidden - mean * mean
    rstd = tl.rsqrt(var + eps)

    # -----------------------------------------------------------------
    # Apply LayerNorm weight/bias and the out‑gate
    # -----------------------------------------------------------------
    w_norm = tl.load(norm_w_ptr + offs_h, mask=mask_h, other=0.0)   # γ
    b_norm = tl.load(norm_b_ptr + offs_h, mask=mask_h, other=0.0)   # β

    norm = (x - mean[:, None]) * rstd[:, None]                     # (BLOCK_M, hidden)
    norm = norm * w_norm[None, :] + b_norm[None, :]                # γ·x+β
    gated = norm * g                                                # apply out_gate

    # -----------------------------------------------------------------
    # Compute final linear projection (hidden → dim) on the fly
    # -----------------------------------------------------------------
    d_start = pid_d * BLOCK_D
    offs_d = d_start + tl.arange(0, BLOCK_D)
    mask_d = offs_d < dim

    # out_w is stored as (dim, hidden); we need its transpose (hidden, dim)
    # Access it as: out_w_T[hidden, dim] = out_w[dim, hidden]
    out_w_T = tl.load(out_w_ptr + offs_h[:, None] + offs_d[None, :] * hidden,
                      mask=mask_h[:, None] & mask_d[None, :],
                      other=0.0).to(tl.float32)                # (hidden, BLOCK_D)

    # (BLOCK_M, hidden) × (hidden, BLOCK_D) → (BLOCK_M, BLOCK_D)
    out_chunk = tl.dot(gated, out_w_T)

    tl.store(out_ptr + offs_m[:, None] * dim + offs_d[None, :],
             out_chunk,
             mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Triton entry point for the TriMul forward pass
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass for the **outgoing** TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, C] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float) – ignored if config["nomask"]=True
        - weights_dict : model parameters (float32, on same device)
        - config_dict  : {"dim": C, "hidden_dim": H, "nomask": bool, "eps": float (opt.)}

    Returns
    -------
    torch.Tensor [B, N, N, C] (float32) on the same CUDA device.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    inp, mask, weights, cfg = data
    device = inp.device
    dim = cfg["dim"]
    hidden = cfg["hidden_dim"]
    eps = cfg.get("eps", 1e-5)
    nomask = cfg.get("nomask", True)

    B, N, _, _ = inp.shape
    total_rows = B * N * N                     # rows for flattened tensors

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (use native kernel – fast, fused reduction)
    # ------------------------------------------------------------------
    # We normalize over the last dimension only.
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    # Ensure weight/bias are on the correct device and dtype
    norm_weight = norm_weight.to(device, dtype=torch.float32)
    norm_bias   = norm_bias.to(device, dtype=torch.float32)

    # Native layer‑norm over dim; returns float32 → cast to FP16 for later ops
    x_norm = F.layer_norm(
        inp,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    ).half()                                   # [B, N, N, dim] half

    # ------------------------------------------------------------------
    # 2️⃣ Fuse the five linear projections (left/right + 3 gates)
    # ------------------------------------------------------------------
    # Concatenate weight matrices (shape: (5*hidden, dim))
    concat_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(device, dtype=torch.float16).contiguous()   # (5*H, C)

    proj_all = F.linear(x_norm, concat_w)               # (B,N,N,5*H) FP16

    # Split projections & gates
    left_proj_fp16, right_proj_fp16, left_gate_fp16, \
    right_gate_fp16, out_gate_fp16 = torch.split(
        proj_all, hidden, dim=-1
    )  # each (B,N,N,H) FP16

    # ------------------------------------------------------------------
    # 2a) Optional binary mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape: (B,N,N) → (B,N,N,1) → flatten to (total_rows,1)
        mask_flat = mask.view(total_rows, 1).to(device, dtype=torch.float16)
        left_proj_fp16  = left_proj_fp16.reshape(total_rows, hidden) * mask_flat
        right_proj_fp16 = right_proj_fp16.reshape(total_rows, hidden) * mask_flat
        # Restore original shape for later steps
        left_proj_fp16  = left_proj_fp16.view(B, N, N, hidden)
        right_proj_fp16 = right_proj_fp16.view(B, N, N, hidden)

    # ------------------------------------------------------------------
    # 2b) Sigmoid gates and element‑wise gating
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_fp16)     # (B,N,N,H)
    right_gate = torch.sigmoid(right_gate_fp16)   # (B,N,N,H)
    out_gate = torch.sigmoid(out_gate_fp16)       # (B,N,N,H) – used later

    left = left_proj_fp16 * left_gate              # (B,N,N,H)
    right = right_proj_fp16 * right_gate           # (B,N,N,H)

    # ------------------------------------------------------------------
    # 3️⃣ Triangular multiplicative update (batched matrix‑multiply)
    # ------------------------------------------------------------------
    # Keep tensors in (B, H, N, N) layout → torch.matmul avoids the copies
    # required by reshaping to (B*H, N, N) + bmm.
    left_perm = left.permute(0, 3, 1, 2)            # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1)          # (B, H, N, N) (k‑axis transposed)

    # (B, H, N, N) @ (B, H, N, N) → (B, H, N, N)
    out = torch.matmul(left_perm, right_perm)      # FP16

    # Bring back to the original layout expected by the final kernel
    out = out.permute(0, 2, 3, 1).contiguous()     # (B, N, N, H)

    # ------------------------------------------------------------------
    # 4️⃣ Fused LayerNorm + out‑gate + final projection (Triton)
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, hidden).contiguous()    # (R, H) FP16
    out_gate_flat = out_gate.view(total_rows, hidden).contiguous()  # (R, H) FP16

    final_fp32 = torch.empty(total_rows, dim,
                             dtype=torch.float32,
                             device=device)

    # Kernel launch parameters (tuned for H100)
    BLOCK_M = 64                 # rows per program
    BLOCK_D = 64                 # output‑channel chunk
    grid = (
        triton.cdiv(total_rows, BLOCK_M),
        triton.cdiv(dim, BLOCK_D),
    )
    layernorm_gate_linear_kernel[grid](
        out_flat,
        out_gate_flat,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        weights["to_out.weight"].half().contiguous(),
        final_fp32,
        total_rows,
        dim=dim,
        hidden=hidden,
        eps=eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 5️⃣ Reshape to final tensor shape (FP32)
    # ------------------------------------------------------------------
    final = final_fp32.view(B, N, N, dim)          # (B, N, N, dim) float32
    return final
