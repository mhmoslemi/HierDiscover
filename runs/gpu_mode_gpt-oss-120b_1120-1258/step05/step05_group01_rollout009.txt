    src_ptr, gate_ptr, weight_ptr,
    ln_weight_ptr, ln_bias_ptr,
    dst_ptr,
    total_rows, hidden_dim, dim_out,
    seq_len,
    stride_b, stride_h, stride_i, stride_j,
    eps,
    BLOCK_M: tl.constexpr,
    BLOCK_D_H: tl.constexpr,
    BLOCK_D_OUT: tl.constexpr,
):

    pid_col = tl.program_id(1)
    row_start = pid_row * BLOCK_M
    col_start = pid_col * BLOCK_D_OUT

    # row indices
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    j_idx = rem % seq_len

    # First pass: mean/var

"""
Outgoing TriMul (AlphaFold‑3) – Triton‑fused LayerNorm + gate + final projection

Algorithm
---------
1️⃣  Layer‑Norm the 4‑D input tensor (FP32 → FP16) with the built‑in
    torch.nn.functional.layer_norm (highly tuned in cuDNN).

2️⃣  Fuse the five linear projections (left_proj, right_proj,
    left_gate, right_gate, out_gate) into a single half‑precision GEMM
    (`torch.nn.functional.linear`).  No biases are used in the model.

3️⃣  Apply the optional binary mask only to the two value‑projections.

4️⃣  Apply sigmoid gates and obtain the gated tensors `left` and `right`.

5️⃣  Batched matrix multiplication (Tensor‑Core GEMM):
        out_perm[b, h, i, j] = Σₖ left[b, i, k, h] · right[b, j, k, h]
    `out_perm` has shape (B, H, N, N) and stays in fp16.

6️⃣  **Fused Triton kernel** – for every row (b,i,j) we
      * compute the per‑row mean/variance of `out_perm`,
      * apply LayerNorm (γ/β),
      * multiply by the element‑wise `out_gate`,
      * and finally perform the projection to the model dimension
        (`out = (norm * out_gate) @ w_out`).

   The kernel works on a 2‑D grid (rows × output‑dim tiles) and
   completely removes the intermediate hidden‑state buffer and a
   second GEMM.  All arithmetic is performed in float32 and the
   result is stored as fp16.

7️⃣  Reshape the contiguous (R, dim) output back to (B, N, N, dim) and
    cast to float32.

The heavy work (steps 5 and 6) runs on the GPU using fp16 Tensor Cores,
while only a single Triton kernel is launched.  This implementation
conforms to the required signature `custom_kernel(data)` and works for
all test configurations listed in the problem statement.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 6️⃣  Fused LayerNorm + out‑gate + final linear projection
# ----------------------------------------------------------------------
@triton.jit
def ln_gate_out_linear_kernel(
    src_ptr,          # const   *  [B, H, N, N]   (fp16)
    gate_ptr,         # const   *  [R, H]        (fp16)
    weight_ptr,       # const   *  [dim, H]      (fp16)   – final linear weight
    ln_weight_ptr,    # const   *  [H]           (fp32)   – LN γ
    ln_bias_ptr,      # const   *  [H]           (fp32)   – LN β
    dst_ptr,          # mutable *  [R, dim]      (fp16)
    total_rows,       # int                     = B·N·N
    hidden_dim,       # int                     = H
    dim_out,          # int                     = dim
    seq_len,          # int                     = N
    stride_b,         # int                     stride on dim 0 of src
    stride_h,         # int                     stride on dim 1 of src
    stride_i,         # int                     stride on dim 2 of src
    stride_j,         # int                     stride on dim 3 of src
    eps,              # float32                 LayerNorm epsilon
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D_H: tl.constexpr, # hidden‑dim tile
    BLOCK_D_OUT: tl.constexpr, # output‑dim tile
):
    # ------------------------------------------------------------------
    # 2‑D grid: (row‑tile, col‑tile)
    # ------------------------------------------------------------------
    pid_row = tl.program_id(0)
    pid_col = tl.program_id(1)

    row_start = pid_row * BLOCK_M
    col_start = pid_col * BLOCK_D_OUT

    offs_m = row_start + tl.arange(0, BLOCK_M)               # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # Decode (b,i,j) from the linear row index.
    # ------------------------------------------------------------------
    seq_len_sq = seq_len * seq_len
    b_idx = offs_m // seq_len_sq
    rem   = offs_m % seq_len_sq
    i_idx = rem // seq_len
    j_idx = rem % seq_len

    # Base offset of hidden‑dim element h = 0 for each row.
    base_off = b_idx * stride_b + i_idx * stride_i + j_idx * stride_j   # [BLOCK_M]

    # ------------------------------------------------------------------
    # Pass‑1: compute per‑row mean & variance over the hidden dimension.
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for h_block in range(0, tl.cdiv(hidden_dim, BLOCK_D_H)):
        offs_h = h_block * BLOCK_D_H + tl.arange(0, BLOCK_D_H)   # [BLOCK_D_H]
        mask_h = offs_h < hidden_dim

        # src layout: (B, H, N, N) with stride_h separating the hidden dim.
        src_ptrs = src_ptr + base_off[:, None] + offs_h[None, :] * stride_h
        vals = tl.load(src_ptrs,
                       mask=mask_m[:, None] & mask_h[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / hidden_dim
    var  = sum_sq / hidden_dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------------------------------------------------------
    # Pass‑2: LayerNorm → gate → linear projection.
    # ------------------------------------------------------------------
    out_acc = tl.zeros([BLOCK_M, BLOCK_D_OUT], dtype=tl.float32)

    for h_block in range(0, tl.cdiv(hidden_dim, BLOCK_D_H)):
        offs_h = h_block * BLOCK_D_H + tl.arange(0, BLOCK_D_H)   # [BLOCK_D_H]
        mask_h = offs_h < hidden_dim

        # ----- load raw src values -----
        src_ptrs = src_ptr + base_off[:, None] + offs_h[None, :] * stride_h
        src_vals = tl.load(src_ptrs,
                           mask=mask_m[:, None] & mask_h[None, :],
                           other=0.0).to(tl.float32)

        # ----- LayerNorm (γ/β) -----
        norm = (src_vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(ln_weight_ptr + offs_h, mask=mask_h, other=0.0)   # fp32
        beta  = tl.load(ln_bias_ptr   + offs_h, mask=mask_h, other=0.0)   # fp32
        norm = norm * gamma[None, :] + beta[None, :]

        # ----- apply out‑gate -----
        gate_ptrs = gate_ptr + (offs_m[:, None] * hidden_dim + offs_h[None, :])
        gate_vals = tl.load(gate_ptrs,
                            mask=mask_m[:, None] & mask_h[None, :],
                            other=0.0).to(tl.float32)

        norm = norm * gate_vals

        # ----- load weight slice (transposed view): shape (BLOCK_D_H, BLOCK_D_OUT) -----
        offs_out = col_start + tl.arange(0, BLOCK_D_OUT)            # [BLOCK_D_OUT]
        mask_out = offs_out < dim_out

        # weight layout: (dim_out, hidden_dim). We need (hidden_dim, dim_out) → transpose on‑the‑fly.
        w_ptrs = weight_ptr + offs_h[:, None] + offs_out[None, :] * hidden_dim
        w_tile = tl.load(w_ptrs,
                         mask=mask_h[:, None] & mask_out[None, :],
                         other=0.0).to(tl.float32)                # [BLOCK_D_H, BLOCK_D_OUT]

        # ----- accumulate: (BLOCK_M, BLOCK_D_H) @ (BLOCK_D_H, BLOCK_D_OUT) -----
        out_acc += tl.dot(norm, w_tile)   # result [BLOCK_M, BLOCK_D_OUT]

    # ------------------------------------------------------------------
    # Store the final output (fp16) – layout [R, dim_out].
    # ------------------------------------------------------------------
    col_offs = col_start + tl.arange(0, BLOCK_D_OUT)                # [BLOCK_D_OUT]
    mask_col = col_offs < dim_out

    dst_ptrs = dst_ptr + offs_m[:, None] * dim_out + col_offs[None, :]
    tl.store(dst_ptrs,
             out_acc.to(tl.float16),
             mask=mask_m[:, None] & mask_col[None, :])

    """

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    dim        = config["dim"]            # model dimension (e.g., 128 or 384)
    hidden_dim = config["hidden_dim"]    # hidden dimension (always 128 in tests)
    eps        = 1e-5
    device     = input_tensor.device
    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N                       # R = B·N·N

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm – FP32 → FP16
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    ).to(torch.float16)                           # shape [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2️⃣ One fused linear projection (5 × hidden_dim outputs)
    # ------------------------------------------------------------------
    # concatenate weight matrices once and cast to fp16
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                           # (dim, 5*H)

    proj = F.linear(x_norm.view(total_rows, dim), w_concat)   # (R, 5*H)

    H = hidden_dim
    left_proj  = proj[:, 0:H]
    right_proj = proj[:, H:2*H]
    left_gate  = proj[:, 2*H:3*H]
    right_gate = proj[:, 3*H:4*H]
    out_gate   = proj[:, 4*H:5*H]

    # ------------------------------------------------------------------
    # 3️⃣ Sigmoid gates (still fp16)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate   = torch.sigmoid(out_gate)

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask (applied only to the two value‑projections)
    # ------------------------------------------------------------------
    if not config.get("nomask", True):
        mask_h = mask_tensor.to(torch.float16).view(total_rows, 1)
        left_proj  = left_proj * mask_h
        right_proj = right_proj * mask_h

    # ------------------------------------------------------------------
    # 5️⃣ Apply gates → gated tensors `left` and `right`
    # ------------------------------------------------------------------
    left  = left_proj * left_gate       # (R, H)
    right = right_proj * right_gate     # (R, H)

    # reshape back to (B, N, N, H) – still contiguous (H is the last dim)
    left  = left.view(B, N, N, H)
    right = right.view(B, N, N, H)

    # ------------------------------------------------------------------
    # 6️⃣ Batched matrix multiplication (B, H, N, N) = left @ rightᵀ
    # ------------------------------------------------------------------
    left_perm  = left.permute(0, 3, 1, 2)                 # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2)                # (B, H, N, N)
    out_perm = torch.matmul(left_perm, right_perm.transpose(-1, -2))  # (B, H, N, N), fp16

    # ------------------------------------------------------------------
    # 7️⃣ Fused LayerNorm + out‑gate + final linear projection
    # ------------------------------------------------------------------
    stride_b = out_perm.stride(0)   # B‑stride
    stride_h = out_perm.stride(1)   # H‑stride (= N*N)
    stride_i = out_perm.stride(2)   # i‑stride (= N)
    stride_j = out_perm.stride(3)   # j‑stride (= 1)

    # output buffer (R, dim) – fp16
    output_fp16 = torch.empty(total_rows, dim, dtype=torch.float16, device=device)

    # final projection weight (dim, hidden_dim) – fp16
    w_out = weights["to_out.weight"].to(torch.float16)

    BLOCK_D_H    = 32          # hidden‑dim tile
    BLOCK_D_OUT  = 64          # output‑dim tile

    grid = (

        triton.cdiv(dim,       BLOCK_D_OUT),    # column dimension

        out_perm,                               # src (B, H, N, N)

        w_out,                                  # final weight (dim, H)
        weights["to_out_norm.weight"],          # LN γ (fp32)
        weights["to_out_norm.bias"],            # LN β (fp32)
        output_fp16,                            # dst (R, dim)
        total_rows,
        hidden_dim,
        dim,
        N,
        stride_b,
        stride_h,
        stride_i,
        stride_j,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D_H=BLOCK_D_H,
        BLOCK_D_OUT=BLOCK_D_OUT,
    )

    # 8️⃣ Reshape to (B, N, N, dim) and cast to float32
    # ------------------------------------------------------------------
    output = output_fp16.view(B, N, N, dim).to(torch.float32)
    return output
