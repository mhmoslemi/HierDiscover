                               total_rows: tl.int32,
                               hidden: tl.constexpr,
                               BLOCK_M: tl.constexpr,
                               BLOCK_H: tl.constexpr):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # Load mask values for each row
    mask_vals = tl.load(mask_ptr + offs_m, mask=mask_m, other=0.0).to(tl.float32)   # [BLOCK_M]

    # For each hidden chunk
    for h0 in range(0, hidden, BLOCK_H):
        offs_h = h0 + tl.arange(0, BLOCK_H)
        mask_h = offs_h < hidden

        # offset base in proj for each row: stride = hidden * 5
        stride_proj = hidden * 5
        base = offs_m[:, None] * stride_proj   # shape [BLOCK_M,1]

        # load slices
        # left_proj = proj_ptr + base + offs_h
        left_proj = tl.load(proj_ptr + base + offs_h,
                            mask=mask_m[:, None] & mask_h[None, :],
                            other=0.0).to(tl.float32)
        # similarly other slices
        right_proj = tl.load(proj_ptr + base + hidden + offs_h,
                             mask=mask_m[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float32)
        left_gate = tl.load(proj_ptr + base + 2*hidden + offs_h,
                             mask=mask_m[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float32)

                             mask=mask_m[:, None] & mask_h[None, :],

                             mask=mask_m[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float32)

        # store
        tl.store(left_ptr + offs_m[:, None] * hidden + offs_h,

All heavy reductions (LayerNorm) and the element‑wise gating are
implemented in Triton, while the two large dense GEMMs (the projection
and the N³ update) are delegated to cuBLAS/Tensor‑Core which already
provides optimal performance on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# -------------------------------------------------------------
# 1️⃣ FP16 LayerNorm (input)
# -------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,            # const float*  [total_rows, dim]
    y_ptr,            #      half*   [total_rows, dim] (output)
    w_ptr,            # const float* [dim]       (γ)
    b_ptr,            # const float* [dim]       (β)
    total_rows: tl.int32,
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # cols per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # ---- first pass : mean / variance (FP32) -----------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                      # [BM,BD]
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                 # 1/σ

    # ---- second pass : normalise, γ & β, store FP16 ---------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]    # (x-µ)/σ
        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)  # γ
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)  # β
        out = norm * w[None, :] + b[None, :]                 # FP32

        tl.store(
            y_ptr + offs_m[:, None] * dim + offs_d[None, :],
            out.to(tl.float16),
            mask=mask_m[:, None] & mask_d[None, :],
        )


# -------------------------------------------------------------
# 3️⃣ Fuse projection, sigmoid‑gate, and optional mask
# -------------------------------------------------------------
@triton.jit
def proj_gate_mask_fuse_kernel(
    proj_ptr,          # const half*  [R, 5*H]  (output of concat Linear)
    mask_ptr,          # const half*  [R]       (1.0 or real mask)
    left_ptr,          #      half*  [R, H]     (result)
    right_ptr,         #      half*  [R, H]     (result)
    out_gate_ptr,      #      half*  [R, H]     (sigmoid(out_gate))
    total_rows: tl.int32,
    hidden: tl.constexpr,
    BLOCK_M: tl.constexpr,          # rows per program
    BLOCK_H: tl.constexpr,          # hidden chunk per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # load mask (scalar per row) once
    mask_val = tl.load(mask_ptr + offs_m, mask=mask_m, other=0.0).to(tl.float32)  # [BM]

    stride = hidden * 5                     # distance between consecutive rows in proj_ptr

    for h0 in range(0, hidden, BLOCK_H):
        offs_h = h0 + tl.arange(0, BLOCK_H)
        mask_h = offs_h < hidden

        # -----------------------------------------------------------------
        # Load the five slices of the concatenated projection tensor
        # -----------------------------------------------------------------
        base = offs_m[:, None] * stride

        left_proj = tl.load(
            proj_ptr + base + offs_h,
            mask=mask_m[:, None] & mask_h[None, :],
            other=0.0,
        ).to(tl.float32)
        right_proj = tl.load(
            proj_ptr + base + hidden + offs_h,
            mask=mask_m[:, None] & mask_h[None, :],
            other=0.0,
        ).to(tl.float32)

        left_gate = tl.load(
            proj_ptr + base + 2 * hidden + offs_h,
            mask=mask_m[:, None] & mask_h[None, :],
            other=0.0,
        ).to(tl.float32)
        right_gate = tl.load(
            proj_ptr + base + 3 * hidden + offs_h,
            mask=mask_m[:, None] & mask_h[None, :],
            other=0.0,
        ).to(tl.float32)
        out_gate = tl.load(
            proj_ptr + base + 4 * hidden + offs_h,
            mask=mask_m[:, None] & mask_h[None, :],
            other=0.0,
        ).to(tl.float32)

        # -------------------------------------------------------------
        # sigmoid (stable enough for fp16 range)
        # -------------------------------------------------------------
        sig = lambda x: 1.0 / (1.0 + tl.exp(-x))

        left_gate_sig = sig(left_gate)
        right_gate_sig = sig(right_gate)
        out_gate_sig  = sig(out_gate)

        # -------------------------------------------------------------
        # Apply mask (if any) and gates
        # -------------------------------------------------------------
        # mask_val[:,None] broadcasts over hidden
        mask_vec = mask_val[:, None]
        left = left_proj * left_gate_sig * mask_vec
        right = right_proj * right_gate_sig * mask_vec

        # -------------------------------------------------------------
        # Store results
        # -------------------------------------------------------------
        tl.store(
            left_ptr + offs_m[:, None] * hidden + offs_h,
            left.to(tl.float16),
            mask=mask_m[:, None] & mask_h[None, :],
        )
        tl.store(
            right_ptr + offs_m[:, None] * hidden + offs_h,
            right.to(tl.float16),
            mask=mask_m[:, None] & mask_h[None, :],
        )
        tl.store(
            out_gate_ptr + offs_m[:, None] * hidden + offs_h,
            out_gate_sig.to(tl.float16),
            mask=mask_m[:, None] & mask_h[None, :],
        )


# -------------------------------------------------------------
# 5️⃣ Fused LayerNorm + out‑gate + final Linear
# -------------------------------------------------------------
@triton.jit
def layernorm_gate_linear_kernel(
    x_ptr,               # const half*   [R, H]   (pre‑norm)
    gate_ptr,            # const half*   [R, H]   (sigmoid(out_gate))
    norm_w_ptr,          # const float*  [H]      (γ of to_out_norm)
    norm_b_ptr,          # const float*  [H]      (β of to_out_norm)
    out_w_ptr,           # const half*   [dim, H] (to_out weight)
    out_ptr,             #      float*   [R, dim] (final FP32)
    total_rows: tl.int32,
    dim: tl.constexpr,
    hidden: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,      # rows per program
    BLOCK_D: tl.constexpr,      # output‑channel chunk
):
    pid = tl.program_id(0)            # row block
    pid_d = tl.program_id(1)          # out‑channel block
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # ---- load a whole hidden vector for the selected rows -------
    offs_h = tl.arange(0, hidden)
    mask_h = offs_h < hidden

    x = tl.load(
        x_ptr + offs_m[:, None] * hidden + offs_h[None, :],
        mask=mask_m[:, None] & mask_h[None, :],
        other=0.0,
    ).to(tl.float32)                     # (BM, H)
    g = tl.load(
        gate_ptr + offs_m[:, None] * hidden + offs_h[None, :],
        mask=mask_m[:, None] & mask_h[None, :],
        other=0.0,
    ).to(tl.float32)                     # (BM, H)

    # ---- per‑row mean / var (FP32) ------------------------------
    sum_val = tl.sum(x, axis=1)
    sum_sq  = tl.sum(x * x, axis=1)
    mean = sum_val / hidden
    var  = sum_sq / hidden - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- apply LN weight / bias and out‑gate --------------------
    w_norm = tl.load(norm_w_ptr + offs_h, mask=mask_h, other=0.0)   # (H)
    b_norm = tl.load(norm_b_ptr + offs_h, mask=mask_h, other=0.0)   # (H)

    norm = (x - mean[:, None]) * rstd[:, None]        # (BM, H)
    norm = norm * w_norm[None, :] + b_norm[None, :]   # γ,β
    gated = norm * g                                   # (BM, H)

    # ---- output‑channel block -----------------------------------
    d_start = pid_d * BLOCK_D
    offs_d = d_start + tl.arange(0, BLOCK_D)
    mask_d = offs_d < dim

    # load slice of final weight matrix (dim, hidden) → transpose to (hidden, BLOCK_D)
    out_w_T = tl.load(
        out_w_ptr + offs_h[:, None] + offs_d[None, :] * hidden,
        mask=mask_h[:, None] & mask_d[None, :],
        other=0.0,
    ).to(tl.float32)                                 # (H, BD)

    # (BM, H) @ (H, BD) → (BM, BD)
    out_chunk = tl.dot(gated, out_w_T)

    tl.store(
        out_ptr + offs_m[:, None] * dim + offs_d[None, :],
        out_chunk,
        mask=mask_m[:, None] & mask_d[None, :],
    )


# -------------------------------------------------------------
# Entry point
# -------------------------------------------------------------
def custom_kernel(data):
    """
    Forward of the outgoing TriMul operator.

    Arguments
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor   [B, N, N, C]  (float32)
        - mask_tensor  : torch.Tensor   [B, N, N]    (bool / float, ignored if nomask=True)
        - weights_dict : model parameters (float32 tensors)
        - config_dict  : {"dim": C, "hidden_dim": H, "nomask": bool, "eps": float?}

    Returns
    -------
    torch.Tensor [B, N, N, C] (float32) on the same CUDA device.
    """
    # ---------------------------------------------------------
    # unpack
    # ---------------------------------------------------------
    inp, mask, weights, cfg = data
    device = inp.device
    dim = cfg["dim"]
    hidden = cfg["hidden_dim"]
    eps = cfg.get("eps", 1e-5)
    nomask = cfg.get("nomask", True)

    B, N, _, _ = inp.shape
    total_rows = B * N * N                       # R = B·N²

    # ---------------------------------------------------------
    # 1️⃣ Input LayerNorm → FP16
    # ---------------------------------------------------------
    inp_flat = inp.view(total_rows, dim).contiguous()          # [R, C] float32
    x_norm_fp16 = torch.empty(total_rows, dim,
                              dtype=torch.half,
                              device=device)

    BLOCK_M_LN = 128          # rows per block
    BLOCK_D_LN = 64           # cols per iteration
    grid_ln = (triton.cdiv(total_rows, BLOCK_M_LN),)

    layernorm_fp16_kernel[grid_ln](
        inp_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M_LN,
        BLOCK_D=BLOCK_D_LN,
    )

    # ---------------------------------------------------------
    # 2️⃣ 5‑way fused projection (Linear)
    # ---------------------------------------------------------
    # concatenate the five weight matrices (each [hidden, dim])
    concat_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).half().contiguous()                               # (5·H, C)

    proj_all = F.linear(x_norm_fp16, concat_w)          # (R, 5·H)  FP16

    # ---------------------------------------------------------
    # 3️⃣ Fuse sigmoid‑gates, mask, and produce left/right/out_gate
    # ---------------------------------------------------------
    if nomask:
        mask_fp16 = torch.ones(total_rows,
                               dtype=torch.half,
                               device=device)
    else:
        mask_fp16 = mask.view(total_rows).to(torch.float16)

    left  = torch.empty(total_rows, hidden, dtype=torch.half, device=device)
    right = torch.empty(total_rows, hidden, dtype=torch.half, device=device)
    out_gate = torch.empty(total_rows, hidden, dtype=torch.half, device=device)

    BLOCK_M_FUSE = 128
    BLOCK_H_FUSE = 64
    grid_fuse = (triton.cdiv(total_rows, BLOCK_M_FUSE),)
    proj_gate_mask_fuse_kernel[grid_fuse](
        proj_all,
        mask_fp16,
        left,
        right,
        out_gate,
        total_rows,
        hidden,
        BLOCK_M=BLOCK_M_FUSE,
        BLOCK_H=BLOCK_H_FUSE,
    )
    # -----------------------------------------------------------------
    # Memory clean‑up of the large intermediate (proj_all) – freed now
    # -----------------------------------------------------------------
    del proj_all

    # ---------------------------------------------------------
    # 4️⃣ Triangular multiplicative update (batched GEMM)
    # ---------------------------------------------------------
    # reshape to (B, N, N, H)
    left  = left.view(B, N, N, hidden)
    right = right.view(B, N, N, hidden)

    # (B, H, N, N) → (B·H, N, N)
    left_mat  = left.permute(0, 3, 1, 2).reshape(B * hidden, N, N)
    right_mat = right.permute(0, 3, 2, 1).reshape(B * hidden, N, N)

    # batched half‑precision matrix multiply (Tensor‑Core)
    out_mat = torch.bmm(left_mat, right_mat)               # (B·H, N, N) FP16

    # back to (B, N, N, H)
    out = (
        out_mat.view(B, hidden, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )                                                       # (B, N, N, H)

    # ---------------------------------------------------------
    # 5️⃣ Final fused LayerNorm + out‑gate + Linear → FP32
    # ---------------------------------------------------------
    out_flat = out.view(total_rows, hidden).contiguous()   # (R, H) half
    final_fp32 = torch.empty(total_rows, dim,
                             dtype=torch.float32,
                             device=device)

    BLOCK_M_OUT = 128
    BLOCK_D_OUT = 64
    grid_out = (
        triton.cdiv(total_rows, BLOCK_M_OUT),
        triton.cdiv(dim, BLOCK_D_OUT),
    )
    layernorm_gate_linear_kernel[grid_out](
        out_flat,
        out_gate,                                 # already sigmoid‑ed
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        weights["to_out.weight"].half().contiguous(),
        final_fp32,
        total_rows,
        dim=dim,
        hidden=hidden,
        eps=eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_D=BLOCK_D_OUT,
    )

    # ---------------------------------------------------------
    # reshape to final answer
    # ---------------------------------------------------------
    final = final_fp32.view(B, N, N, dim)               # [B, N, N, C] float32
    return final
