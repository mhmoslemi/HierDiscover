"""
Optimized forward pass for the *outgoing* TriMul operator (AlphaFold‑3).

Algorithm (FP16/FP32 mixed):
1. Row‑wise LayerNorm over the input channel dim (C) – custom Triton kernel.
2. Fuse the 5 linear projections (left/right + three gates) into a single GEMM.
3. (Optional) apply the pair‑wise mask to the left/right projections.
4. Apply sigmoid gates and element‑wise gating (all FP16).
5. Triangular multiplicative update (outgoing):
        out_h[i, j] = Σ_k left_h[i, k] * right_h[j, k]
   Implemented as a batched GEMM `left @ rightᵀ` on tensors of shape (B·H, N, N)
   using `torch.bmm` (Tensor‑Core accelerated).
6. Row‑wise LayerNorm over the hidden dim (H) – same Triton kernel.
7. Multiply by the output gate, then a final linear projection back to C.
8. Cast to FP32 and return.
All heavy reductions are performed in Triton; dense linear algebra uses
cuBLAS (the fastest path on an H100). The implementation supports both
masked and un‑masked inputs and works for all shapes required by the test
suite.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (FP16 output, FP32 accumulation)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,          # *const void*  [total_rows, dim] (float16 or float32 input)
    y_ptr,          # *mut   float16* [total_rows, dim] (fp16 output)
    weight_ptr,     # *const float* [dim]  (γ)
    bias_ptr,       # *const float* [dim]  (β)
    total_rows,     # i32
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per inner loop
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # ---------- mean & variance (FP32) ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                     # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)       # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                # 1 / sqrt(var+eps)

    # ---------- write normalized result (FP16) ----------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                     # fp32

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)

        out_fp16 = (norm * w[None, :] + b[None, :]).to(tl.float16)

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
# Entry point used by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor  [B, N, N, C]  (float32)
            - mask_tensor  : torch.Tensor  [B, N, N]    (bool/float) or None
            - weights_dict : mapping from weight name to torch.Tensor (float32)
            - config_dict  : {"dim": C, "hidden_dim": H, "nomask": bool}
    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, cfg = data
    device = input_tensor.device
    dim = cfg["dim"]                     # input channel dim C
    hidden_dim = cfg["hidden_dim"]       # hidden dim H
    eps = 1e-5
    nomask = cfg.get("nomask", True)

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N                     # rows for flattened view

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (fp16 output)
    # --------------------------------------------------------------
    x_flat = input_tensor.contiguous().view(total_rows, dim)   # [R, C] fp32
    x_norm_fp16 = torch.empty(total_rows, dim,
                              dtype=torch.float16,
                              device=device)

    # Tuned block sizes – a good trade‑off for the supported dimensions.
    BLOCK_M = 1024
    BLOCK_D = 64
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # --------------------------------------------------------------
    # 2️⃣ Fuse the five linear projections (left/right + three gates)
    # --------------------------------------------------------------
    w_left    = weights["left_proj.weight"]      # (H, C)
    w_right   = weights["right_proj.weight"]    # (H, C)
    w_lgate   = weights["left_gate.weight"]     # (H, C)
    w_rgate   = weights["right_gate.weight"]    # (H, C)
    w_outgate = weights["out_gate.weight"]      # (H, C)

    # Concatenate on the output dimension → (5*H, C)
    concat_w = torch.cat([w_left, w_right, w_lgate, w_rgate, w_outgate],
                         dim=0).half().contiguous()

    # Fused GEMM (FP16) → (R, 5*H)
    proj_all = F.linear(x_norm_fp16, concat_w)   # bias is zero in all 5 layers

    # Split back to the five tensors   → each (R, H)
    (left_proj_fp16,
     right_proj_fp16,
     left_gate_fp16,
     right_gate_fp16,
     out_gate_fp16) = torch.split(proj_all, hidden_dim, dim=1)

    # --------------------------------------------------------------
    # 3️⃣ Mask (applied before gating)
    # --------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        mask_fp16 = mask_tensor.reshape(total_rows, 1).to(torch.float16)
        left_proj_fp16  = left_proj_fp16 * mask_fp16
        right_proj_fp16 = right_proj_fp16 * mask_fp16

    # --------------------------------------------------------------
    # 4️⃣ Sigmoid gates & element‑wise gating (still FP16)
    # --------------------------------------------------------------
    left_gate   = torch.sigmoid(left_gate_fp16)       # (R, H)
    right_gate  = torch.sigmoid(right_gate_fp16)
    out_gate    = torch.sigmoid(out_gate_fp16)

    left_fp16   = left_proj_fp16 * left_gate
    right_fp16  = right_proj_fp16 * right_gate

    # Reshape back to 4‑D tensors  → (B, N, N, H)
    left  = left_fp16.view(B, N, N, hidden_dim)
    right = right_fp16.view(B, N, N, hidden_dim)
    out_gate = out_gate.view(B, N, N, hidden_dim)

    # --------------------------------------------------------------
    # 5️⃣ Triangular multiplicative update (batched GEMM)
    #    out_h[i, j] = Σ_k left_h[i, k] * right_h[j, k]
    # --------------------------------------------------------------
    # Prepare tensors as (B·H, N, N) for torch.bmm
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    BH = B * hidden_dim
    left_mat  = left_perm.view(BH, N, N)                 # (BH, N, N)
    right_mat = right_perm.view(BH, N, N)                # (BH, N, N)

    # Batched matmul : left @ rightᵀ  → (BH, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Restore shape (B, N, N, H)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm over hidden dim H (fp16 output)
    # --------------------------------------------------------------
    out_flat = out.view(total_rows, hidden_dim)          # (R, H)
    out_norm_fp16 = torch.empty_like(out_flat,
                                     dtype=torch.float16,
                                     device=device)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid2](
        out_flat,                # fp16 input (will be up‑cast internally)
        out_norm_fp16,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    out_norm = out_norm_fp16.view(B, N, N, hidden_dim)   # (B, N, N, H)

    # --------------------------------------------------------------
    # 7️⃣ Final gating + linear projection back to C
    # --------------------------------------------------------------
    gated = out_norm * out_gate      # element‑wise FP16

    # Final linear: (R, H) @ (C, H)^T → (R, C)
    to_out_w_fp16 = weights["to_out.weight"].half()
    gated_flat = gated.view(total_rows, hidden_dim)
    final_fp16 = F.linear(gated_flat, to_out_w_fp16)     # bias is zero

    # --------------------------------------------------------------
    # 8️⃣ Cast back to FP32 and return
    # --------------------------------------------------------------
    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final
