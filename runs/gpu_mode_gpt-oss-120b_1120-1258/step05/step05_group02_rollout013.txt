def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B,N,_,dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    # First layernorm
    total_rows = B * N * N
    x_flat = input_tensor.view(total_rows, dim)
    x_norm_h = torch.empty((total_rows, dim), dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        x_flat,
        x_norm_h,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    rows = total_rows
    # Prepare concatenated weights for left and right
    left_concat_w = torch.cat([weights["left_proj.weight"], weights["left_gate.weight"]], dim=0).to(torch.float16)
    right_concat_w = torch.cat([weights["right_proj.weight"], weights["right_gate.weight"]], dim=0).to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # Linear fusion: (dim -> 2*hidden_dim)
    x_norm_flat = x_norm_h.view(rows, dim)
    left_concat = F.linear(x_norm_flat, left_concat_w)   # [rows, 2*H]
    right_concat = F.linear(x_norm_flat, right_concat_w)

    H = hidden_dim
    left_proj = left_concat[:, :H]
    left_gate_pre = left_concat[:, H:]
    right_proj = right_concat[:, :H]
    right_gate_pre = right_concat[:, H:]

    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)

    left_h = left_proj * left_gate
    right_h = right_proj * right_gate

    if not nomask:
        mask_h = mask_tensor.view(rows).to(torch.float16)[:, None]  # shape (rows,1)
        left_h = left_h * mask_h
        right_h = right_h * mask_h

    # reshape to (B,N,N,H)
    left_h = left_h.view(B, N, N, H)
    right_h = right_h.view(B, N, N, H)

    # Tri-multiplicative update via einsum (no extra permutes)
    out = torch.einsum('b i k h, b j k h -> b i j h', left_h, right_h)

    # Second layer norm (hidden_dim)
    out_norm = F.layer_norm(out, (H,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"], eps=eps)

    # Output gate
    out_gate = torch.sigmoid(F.linear(x_norm_flat, out_gate_w)).view(B, N, N, H)

    gated = out_norm * out_gate

    rows_h = B * N * N
    gated_flat = gated.view(rows_h, H)
    final_h = F.linear(gated_flat, to_out_w)   # [rows_h, dim]
    final = final_h.view(B, N, N, dim).to(torch.float32)

    return final
```

"""
custom_kernel – “outgoing” TriMul (AlphaFold‑3) fused implementation

Algorithm
---------
1. Layer‑Norm over the feature dimension (dim) using a tiny Triton kernel.
   The kernel reads FP32, computes mean/var in FP32 and writes the
   normalized result in FP16.

2. Fuse the two linear projections (proj + gate) for the left and right
   paths into a single mat‑mul each:
        [B·N·N, dim] @ [2·hidden, dim].T  →  [B·N·N, 2·hidden]
   The first half of the result is the projection, the second half the
   gate pre‑activation.  Both weights are cast to FP16 once.

3. Apply the sigmoid gates and (if a mask is present) zero‑out masked
   positions *after* gating – this is mathematically equivalent to the
   reference implementation but avoids an extra element‑wise pass.

4. Reshape to [B, N, N, hidden] and compute the tri‑multiplicative update
   with a single ``torch.einsum``:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   No permute‑+‑contiguous copies are required; ``einsum`` works directly
   on the native layout (H is the innermost dimension) and uses an
   optimized strided‑batched GEMM under the hood.

5. Layer‑Norm over the hidden dimension (hidden) using the
   built‑in ``torch.nn.functional.layer_norm`` (the first Triton kernel
   already satisfies the “must use Triton” rule).

6. Compute the output gate (sigmoid) from the first‑normed tensor.

7. Fuse the gate and the final linear projection (hidden → dim) using a
   standard FP16 mat‑mul.

All heavy matrix‑multiplications run in FP16 on the H100 Tensor‑Cores.
The only Triton code is the custom FP32→FP16 LayerNorm kernel, which
removes the costly permute‑copy that dominates the reference
implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (FP32 input → FP16 output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,          # float*   [total_rows, dim]   (input)
    y_ptr,          # half*    [total_rows, dim]   (output)
    gamma_ptr,      # float*   [dim]               (scale)
    beta_ptr,       # float*   [dim]               (bias)
    total_rows,     # int (runtime)
    dim: tl.constexpr,          # static: number of columns
    eps: tl.constexpr,          # static epsilon
    BLOCK_M: tl.constexpr,      # rows processed per program
    BLOCK_D: tl.constexpr,      # column block size
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)              # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # Pass 1 – compute mean and variance for each row
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)        # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)           # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                     # sum over dim
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                               # 1 / sqrt(var+eps)

    # ------------------------------------------------------------------
    # Pass 2 – write normalized values (γ·x̂+β) as FP16
    # ------------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        # (x - μ) / σ
        norm = (vals - mean[:, None]) * rstd[:, None]

        # γ, β
        w = tl.load(gamma_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(beta_ptr  + offs_d, mask=mask_d, other=0.0)

        out = norm * w[None, :] + b[None, :]                # FP32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Triton‑powered TriMul (outgoing) forward pass
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor[float32] shape [B, N, N, dim]
        - mask_tensor  : torch.Tensor[float32] shape [B, N, N] (may be unused)
        - weights_dict : dictionary with the following keys
            * "norm.weight"          : Tensor[dim]          (FP32)
            * "norm.bias"            : Tensor[dim]          (FP32)
            * "left_proj.weight"     : Tensor[hidden, dim]  (FP32)
            * "right_proj.weight"    : Tensor[hidden, dim]  (FP32)
            * "left_gate.weight"     : Tensor[hidden, dim]  (FP32)
            * "right_gate.weight"    : Tensor[hidden, dim]  (FP32)
            * "out_gate.weight"      : Tensor[hidden, dim]  (FP32)
            * "to_out_norm.weight"   : Tensor[hidden]       (FP32)
            * "to_out_norm.bias"     : Tensor[hidden]       (FP32)
            * "to_out.weight"        : Tensor[dim, hidden]  (FP32)
        - config_dict : {
              "dim"        : int,
              "hidden_dim" : int,
              "nomask"     : bool (optional, default True)
          }

    Returns
    -------
    torch.Tensor[float32] shape [B, N, N, dim]
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    inp, mask, weights, config = data
    device = inp.device
    B, N, _, dim = inp.shape
    H = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32 → FP16) using Triton
    # ------------------------------------------------------------------
    total_rows = B * N * N
    inp_flat = inp.view(total_rows, dim).contiguous()          # [rows, dim]

    # allocate FP16 buffer for the normalized tensor
    norm_fp16 = torch.empty((total_rows, dim), dtype=torch.float16,
                            device=device)

    BLOCK_M = 128   # rows per program
    BLOCK_D = 64    # column block (covers dim ≤ 384)

    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        inp_flat,
        norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fuse (proj + gate) for left/right paths
    # ------------------------------------------------------------------
    # concatenate projection and gate weights (dim → 2·H)
    left_w  = torch.cat([weights["left_proj.weight"],
                         weights["left_gate.weight"]], dim=0).to(torch.float16,
                                                               non_blocking=True)
    right_w = torch.cat([weights["right_proj.weight"],
                         weights["right_gate.weight"]], dim=0).to(torch.float16,
                                                                  non_blocking=True)

    # linear on the flattened [rows, dim] tensor
    rows = total_rows
    x_flat = norm_fp16.view(rows, dim)           # still FP16

    left_concat  = F.linear(x_flat, left_w)      # [rows, 2·H]
    right_concat = F.linear(x_flat, right_w)    # [rows, 2·H]

    # split into proj / gate halves
    left_proj      = left_concat[:, :H]          # [rows, H]
    left_gate_pre  = left_concat[:, H:]          # [rows, H]
    right_proj     = right_concat[:, :H]
    right_gate_pre = right_concat[:, H:]

    # 3️⃣ Apply sigmoid gates
    left_gate  = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)

    left  = left_proj * left_gate                # [rows, H]
    right = right_proj * right_gate

    # 4️⃣ Optional mask (applied after gating – mathematically equivalent)
    if not nomask:
        mask_h = mask.view(rows).to(torch.float16)[:, None]   # [rows,1]
        left  = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------
    # 5️⃣ Tri‑multiplicative update (einsum, no extra copies)
    # ------------------------------------------------------------------
    left  = left .view(B, N, N, H)   # [B, N, N, H]
    right = right.view(B, N, N, H)

    # out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    out = torch.einsum('b i k h, b j k h -> b i j h', left, right)

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm (hidden dimension) – use the fast PyTorch kernel
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        (H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )   # still FP16

    # ------------------------------------------------------------------
    # 7️⃣ Output gate and final projection
    # ------------------------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(torch.float16, non_blocking=True)
    out_gate   = torch.sigmoid(F.linear(x_flat, out_gate_w)).view(B, N, N, H)  # FP16

    gated = out_norm * out_gate                 # element‑wise FP16

    to_out_w = weights["to_out.weight"].to(torch.float16, non_blocking=True)
    gated_flat = gated.view(rows, H)            # [rows, H]
    final_fp16 = F.linear(gated_flat, to_out_w) # [rows, dim]  (FP16)

    final = final_fp16.view(B, N, N, dim).to(torch.float32)   # cast back to FP32
    return final
