"""
Optimized Triton‑based implementation of the *outgoing* TriMul operator
(AlphaFold‑3).

Algorithm
---------
1️⃣  Row‑wise LayerNorm on the flattened input (B·N·N×dim) – a custom
    Triton kernel that computes mean/variance in fp32 and writes the
    normalised result in fp16.

2️⃣  Fuse the four linear projections (left/right) and the two gate
    linear layers into a single GEMM:
        [R, dim] @ [4·H, dim].T  →  [R, 4·H]      (R = B·N·N)
    The result is split into the projected tensors and the corresponding
    gate pre‑activations.

3️⃣  Compute the output‑gate (a fifth linear layer) – also a GEMM.

4️⃣  Apply the optional mask, in‑place sigmoid on the gates and
    element‑wise gating:
        left  = (proj_left)  * sigmoid(gate_left)
        right = (proj_right) * sigmoid(gate_right)

5️⃣  Tri‑multiplicative update.
    After reshaping, the problem reduces to a batched matrix‑multiply:
        (B·H, N, N) @ (B·H, N, N)ᵀ   →   (B·H, N, N)
    Implemented with the highly‑optimised cuBLAS routine ``torch.bmm``.

6️⃣  Second LayerNorm over the hidden dimension (again via the Triton
    kernel).  The normalised tensor stays in fp16.

7️⃣  Multiply by the output‑gate and apply the final linear projection
    (hidden_dim → dim).  The result is cast back to fp32, matching the
    reference implementation.

All heavy linear layers are executed with fp16 tensor‑core GEMMs.
Only the two LayerNorms are written in Triton, which reduces the
extra memory traffic of the reference implementation while keeping the
overall arithmetic bound to the matrix‑multiplications.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (fp32 → fp16)
# ----------------------------------------------------------------------
@triton.jit
def row_layernorm_kernel(
    x_ptr,          # const   * [total_rows, dim]   (float32)
    y_ptr,          # mutable * [total_rows, dim]   (float16)
    weight_ptr,     # const   * [dim]               (float32)   – γ
    bias_ptr,       # const   * [dim]               (float32)   – β
    total_rows,     # scalar  (int32)
    dim,            # scalar  (int32)
    eps,            # const   (float32)           – ε
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """
    Row‑wise LayerNorm:
        y = ((x - μ) / sqrt(σ² + ε)) * γ + β
    The computation uses fp32 accumulation and writes fp16 results.
    """
    pid = tl.program_id(0)                     # 1‑D grid
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M) # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # 1. Compute mean and variance of each row
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    # iterate over the channel dimension in blocks
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        # (row, col) linearised pointer
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)      # fp32 load

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)          # 1 / sqrt(var + eps)

    # ------------------------------------------------------------------
    # 2. Write normalised values
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]      # (x-μ)/σ

        gamma = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        beta  = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β

        out = norm * gamma[None, :] + beta[None, :]                 # fp32

        # store fp16
        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
# Custom forward pass
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul (AlphaFold‑3) operator.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        * input_tensor – torch.float32, shape [B, N, N, dim]
        * mask_tensor  – torch.float32/torch.bool, shape [B, N, N] (may be None)
        * weights_dict – mapping with the following keys:
            - "norm.weight", "norm.bias"
            - "left_proj.weight", "right_proj.weight"
            - "left_gate.weight", "right_gate.weight"
            - "out_gate.weight"
            - "to_out_norm.weight", "to_out_norm.bias"
            - "to_out.weight"
        * config_dict  – must contain "dim", "hidden_dim" and boolean "nomask"

    Returns
    -------
    torch.Tensor – shape [B, N, N, dim], dtype torch.float32
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    inp, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    device = inp.device

    B, N, _, _ = inp.shape
    R = B * N * N                     # total number of (b,i,j) rows

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm  (fp32 → fp16)
    # ------------------------------------------------------------------
    # flatten to [R, dim]
    inp_flat = inp.view(R, dim).contiguous()
    norm_fp16 = torch.empty(R, dim, dtype=torch.float16, device=device)

    # launch Triton kernel
    BLOCK_M = 256                      # rows per program
    BLOCK_D = 32                       # columns per program
    grid = (triton.cdiv(R, BLOCK_M),)
    row_layernorm_kernel[grid](
        inp_flat,
        norm_fp16,
        w["norm.weight"],              # γ  – FP32
        w["norm.bias"],                # β  – FP32
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fuse left/right projections + their gates (4·H outputs)
    # ------------------------------------------------------------------
    H = hidden_dim
    # concatenate the four projection/gate weight matrices [4·H, dim]
    w_proj_gate = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                # fp16 for GEMM

    # GEMM: [R, dim] @ [4·H, dim].T → [R, 4·H]
    proj_gate = F.linear(norm_fp16, w_proj_gate)   # fp16

    # split the concatenated tensor (no extra copy, just views)
    left_raw   = proj_gate[:, 0:H]         # [R, H]
    right_raw  = proj_gate[:, H:2*H]       # [R, H]
    left_gate_raw  = proj_gate[:, 2*H:3*H] # [R, H]
    right_gate_raw = proj_gate[:, 3*H:4*H] # [R, H]

    # 2️⃣‑b Output gate (separate linear)
    out_gate_raw = F.linear(
        norm_fp16,
        w["out_gate.weight"].to(torch.float16),  # [H, dim]
    )                                              # fp16, shape [R, H]

    # ------------------------------------------------------------------
    # 3️⃣ Mask (if present) + sigmoid gates + element‑wise gating
    # ------------------------------------------------------------------
    if not cfg.get("nomask", True):
        # mask is [B,N,N] → flatten to [R,1] and broadcast
        mask_h = mask.to(torch.float16).reshape(R, 1)
        left_raw  = left_raw * mask_h
        right_raw = right_raw * mask_h

    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    # In‑place gating to reuse memory (destroy raw tensors)
    left  = left_raw.mul_(left_gate)      # [R, H] fp16
    right = right_raw.mul_(right_gate)    # [R, H] fp16

    # ------------------------------------------------------------------
    # 4️⃣ Reshape for the O(N³) multiplicative update
    # ------------------------------------------------------------------
    left  = left.view(B, N, N, H)          # [B, N, N, H]
    right = right.view(B, N, N, H)         # [B, N, N, H]
    out_gate = out_gate.view(B, N, N, H)   # [B, N, N, H]

    # bring head dimension next to batch for the batched GEMM
    left  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    # collapse batch+head -> (B·H, N, N)
    left_mat  = left.view(B * H, N, N)    # [B·H, N, N]
    right_mat = right.view(B * H, N, N)   # [B·H, N, N]

    # batched GEMM: (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N)
    # (Tensor Cores: fp16×fp16→fp32 accumulate → fp16)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm over the hidden dimension (fp16)
    # ------------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()
    out_norm_fp16 = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(R, BLOCK_M),)
    row_layernorm_kernel[grid2](
        out_flat,
        out_norm_fp16,
        w["to_out_norm.weight"],       # γ – FP32
        w["to_out_norm.bias"],         # β – FP32
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Apply output‑gate and final linear projection
    # ------------------------------------------------------------------
    out_norm = out_norm_fp16.view(B, N, N, H) * out_gate   # fp16, in‑place mult

    # final linear: hidden_dim → dim
    final_fp16 = F.linear(
        out_norm.view(R, H),
        w["to_out.weight"].to(torch.float16),      # [dim, H]
    )                                               # fp16, shape [R, dim]

    # cast back to fp32 to match the reference implementation
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
