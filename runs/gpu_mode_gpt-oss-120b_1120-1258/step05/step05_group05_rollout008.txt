"""
custom_kernel
==============

Fast forward pass for the *outgoing* TriMul operator (AlphaFold‑3).

Algorithm (mixed FP16/FP32):
1️⃣ First LayerNorm on the input channel dimension (C).  Implemented in a
   Triton kernel that streams each row twice (once for mean/var, once for the
   normalized output).  The kernel works on FP32 input and writes FP16.

2️⃣ Fuse the five linear projections (left/right + three gates) into a single
   GEMM by concatenating their weight matrices.  The fused GEMM is performed
   with `torch.nn.functional.linear` on the FP16‑normalized tensor.

3️⃣ (Optional) apply the pair‑wise mask to the left/right projections.

4️⃣ Apply sigmoid gates and element‑wise gating (still in FP16).

5️⃣ Triangular multiplicative update (outgoing version):
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This contraction is executed with `torch.einsum`, which avoids the
   explicit transposition and copy that a batched GEMM would require.

6️⃣ Second LayerNorm over the hidden dimension (H).  Here we use the
   highly‑optimized `torch.nn.functional.layer_norm` (FP32 accumulation,
   FP16 output).

7️⃣ Combine the output gate (sigmoid) with the normalized tensor and apply
   the final linear projection back to the original channel dimension.

All heavy reductions are performed in FP32; the bulk of the computation
uses FP16 Tensor‑Core kernels.  The implementation works for the full
range of test configurations (variable batch size, sequence length,
input dimension and hidden dimension) and supports the presence/absence
of a mask.

Author: Triton engineer
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton row‑wise LayerNorm (fp16 output, fp32 reduction)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,          # const   *void*   [total_rows, dim]   (float16/float32)
    y_ptr,          # mutable *float16* [total_rows, dim]   (fp16 output)
    weight_ptr,     # const   *float*  [dim]               (γ)
    bias_ptr,       # const   *float*  [dim]               (β)
    total_rows,     # i32
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)            # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ----------- 1. mean & variance (fp32) -------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < dim

        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                                   # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                     # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)              # [BLOCK_M]

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                               # 1 / sqrt(var+eps)

    # ----------- 2. write normalized output (fp16) -----
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                                   # fp32

        norm = (vals - mean[:, None]) * rstd[:, None]      # fp32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)

        out_fp16 = (norm * w[None, :] + b[None, :]).to(tl.float16)

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, C] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (bool/float) or None
        - weights_dict : dict of model weights (float32 tensors)
        - config_dict  : {"dim": C, "hidden_dim": H, "nomask": bool}

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                # input channel dimension C
    hidden_dim = config["hidden_dim"]  # hidden dimension H
    eps = 1e-5
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Shape preparation
    # ------------------------------------------------------------------
    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N                      # rows for the flattened view

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (fp32 reduction → fp16 output)
    # ------------------------------------------------------------------
    # Flatten to [total_rows, dim] (float32 layout)
    x_flat = input_tensor.contiguous().view(total_rows, dim)

    # Output buffer for the normalized tensor (FP16)
    x_norm_fp16 = torch.empty(total_rows, dim, dtype=torch.float16, device=device)

    # Triton launch parameters (tuned for dim ≤ 384)
    BLOCK_M = 256
    BLOCK_D = 128   # 1‑2 loads per row instead of many small ones
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fuse the five linear projections into a single GEMM
    # ------------------------------------------------------------------
    # Gather the weight matrices (all of shape [H, C])
    w_left    = weights["left_proj.weight"]
    w_right   = weights["right_proj.weight"]
    w_lgate   = weights["left_gate.weight"]
    w_rgate   = weights["right_gate.weight"]
    w_outgate = weights["out_gate.weight"]

    # Concatenate → shape [(5*H), C]  and cast to FP16 once
    concat_w = torch.cat([w_left, w_right, w_lgate, w_rgate, w_outgate],
                         dim=0).half().contiguous()

    # Fused GEMM:  [total_rows, C] @ [C, 5*H]^T → [total_rows, 5*H]  (FP16)
    proj_all = F.linear(x_norm_fp16, concat_w)   # (half)

    # Split into five tensors of shape [total_rows, H]
    left_proj_fp16, right_proj_fp16, left_gate_fp16, \
        right_gate_fp16, out_gate_fp16 = torch.split(
        proj_all, hidden_dim, dim=1
    )

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (applied before gating)
    # ------------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        # mask has shape [B, N, N] → flatten → [total_rows, 1]
        mask_fp16 = mask_tensor.reshape(total_rows, 1).to(torch.float16)
        left_proj_fp16  = left_proj_fp16 * mask_fp16
        right_proj_fp16 = right_proj_fp16 * mask_fp16

    # ------------------------------------------------------------------
    # 4️⃣ Apply sigmoid gates and element‑wise gating (still FP16)
    # ------------------------------------------------------------------
    left_gate   = torch.sigmoid(left_gate_fp16)       # [total_rows, H]
    right_gate  = torch.sigmoid(right_gate_fp16)
    out_gate    = torch.sigmoid(out_gate_fp16)

    left_fp16   = left_proj_fp16 * left_gate          # [total_rows, H]
    right_fp16  = right_proj_fp16 * right_gate        # [total_rows, H]

    # Reshape back to 4‑D layout for the contraction
    left  = left_fp16.view(B, N, N, hidden_dim)   # [B, N, N, H] (FP16)
    right = right_fp16.view(B, N, N, hidden_dim)   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 5️⃣ Triangular multiplicative update (outgoing)
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    Implemented with torch.einsum to avoid the explicit
    #    permute/contiguous copy required by a batched GEMM.
    # ------------------------------------------------------------------
    out = torch.einsum('b i k d, b j k d -> b i j d', left, right)  # (FP16)

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm over the hidden dimension H (FP32 reduction)
    #     Using the highly‑optimized torch.nn.functional.layer_norm.
    # ------------------------------------------------------------------
    out_norm = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].half(),
        bias=weights["to_out_norm.bias"].half(),
        eps=eps,
    )  # (FP16)

    # ------------------------------------------------------------------
    # 7️⃣ Final gating + linear projection back to C
    # ------------------------------------------------------------------
    gated = out_norm * out_gate  # element‑wise FP16

    # Linear projection to original dimension C
    to_out_w_fp16 = weights["to_out.weight"].half()
    gated_flat = gated.view(total_rows, hidden_dim)   # [total_rows, H]
    final_fp16 = F.linear(gated_flat, to_out_w_fp16)  # [total_rows, C] (FP16)

    # ------------------------------------------------------------------
    # 8️⃣ Cast back to FP32 and return
    # ------------------------------------------------------------------
    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final
