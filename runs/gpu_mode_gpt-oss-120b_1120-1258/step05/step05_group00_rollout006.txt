@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    N, total_batch,
    stride_left_bn, stride_left_n, stride_left_n2,
    stride_right_bn, stride_right_n, stride_right_n2,
    stride_out_bn, stride_out_n, stride_out_n2,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # Batch dimension index
    batch_idx = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # start offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # Base pointer for this batch
    left_batch_ptr = left_ptr + batch_idx * stride_left_bn
    right_batch_ptr = right_ptr + batch_idx * stride_right_bn
    out_batch_ptr = out_ptr + batch_idx * stride_out_bn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k_idx in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k_idx * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile [M, K]
        left_ptrs = left_batch_ptr + offs_m[:, None] * stride_left_n + offs_k[None, :] * stride_left_n2
        left_tile = tl.load(left_ptrs,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)

        # right tile [K, N]
        right_ptrs = right_batch_ptr + offs_k[:, None] * stride_right_n + offs_n[None, :] * stride_right_n2
        right_tile = tl.load(right_ptrs,
                             mask=mask_k[:, None] & mask_n[None, :],
                             other=0.0).to(tl.float32)

        # accumulate
        acc += tl.dot(left_tile, right_tile)

    # Write out
    out_ptrs = out_batch_ptr + offs_m[:, None] * stride_out_n + offs_n[None, :] * stride_out_n2
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])

"""
Optimized outgoing TriMul forward pass (AlphaFold‑3).

Algorithm
---------
1️⃣ Row‑wise LayerNorm (FP32 → FP16) – custom Triton kernel.
2️⃣ Fuse the four linear layers (left/right projection + left/right gate) into a
   single FP16 GEMM:  [R, dim] @ [4·H, dim].T → [R, 4·H].  The result is sliced
   into the four tensors needed later.
3️⃣ Compute the output gate with a second FP16 GEMM.
4️⃣ Apply the optional mask, sigmoid‑gate (in‑place) and element‑wise
   multiplication to obtain `left` and `right` (shape [B,N,N,H] FP16).
5️⃣ Batched triangular multiplicative update:
   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h].
   Implemented as a single Triton kernel that performs a batched GEMM
   (B·H, N, N) @ (B·H, N, N)ᵀ using FP16 inputs / FP32 accumulation.
6️⃣ Second LayerNorm on the hidden dimension – same Triton kernel as step‑1.
7️⃣ Multiply by the output gate, final FP16 linear projection and cast to
   FP32.

All heavy per‑row normalisations are done with Triton, while the large
matrix multiplications leverage the GPU’s Tensor‑Cores via custom Triton
code.  No autograd is required – only the forward pass is implemented.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (FP32 input → FP16 output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   * [total_rows, dim]   (float32)
    y_ptr,          # mutable * [total_rows, dim]   (float16)
    weight_ptr,     # const   * [dim]               (float32) – γ
    bias_ptr,       # const   * [dim]               (float32) – β
    total_rows,     # int – runtime
    dim,            # int – runtime
    eps,            # float – compile‑time const
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # ----- mean & variance ------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ----- write normalized output ----------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]                       # FP32

        out_fp16 = out.to(tl.float16)
        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Triton kernel: batched triangular multiplicative update
# Computes   out = left @ rightᵀ   for many (batch,head) matrices.
# Input/outputs are FP16, accumulation is FP32.
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,          # const * [B*H, N, N] (fp16)
    right_ptr,         # const * [B*H, N, N] (fp16)
    out_ptr,           # mutable * [B*H, N, N] (fp16)
    N,                 # int – sequence length
    stride_l_batch,    # int – stride between batch elements (left)
    stride_l_i,        # int – stride between rows   (left)
    stride_l_j,        # int – stride between cols   (left)
    stride_r_batch,    # int – stride between batch elements (right)
    stride_r_i,        # int – stride between rows   (right)
    stride_r_j,        # int – stride between cols   (right)
    stride_o_batch,    # int – stride between batch elements (out)
    stride_o_i,        # int – stride between rows   (out)
    stride_o_j,        # int – stride between cols   (out)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    batch_idx = tl.program_id(0)          # which (B*H) matrix
    pid_m = tl.program_id(1)              # tile row
    pid_n = tl.program_id(2)              # tile col

    # offsets for the output tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # base pointers for this batch element
    left_base  = left_ptr  + batch_idx * stride_l_batch
    right_base = right_ptr + batch_idx * stride_r_batch
    out_base   = out_ptr   + batch_idx * stride_o_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension k
    for k_off in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k_off * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # A: left[batch, m, k]
        a_ptrs = left_base + offs_m[:, None] * stride_l_i + offs_k[None, :] * stride_l_j
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # B: right[batch, k, n]   (note: we use the non‑transposed layout,
        #       so this directly yields the needed K‑dim)
        b_ptrs = right_base + offs_k[:, None] * stride_r_i + offs_n[None, :] * stride_r_j
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)

        acc += tl.dot(a, b)   # (M,K)·(K,N) → (M,N)

    # store result (FP16)
    out_ptrs = out_base + offs_m[:, None] * stride_o_i + offs_n[None, :] * stride_o_j
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data = (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    device = input_tensor.device

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N      # R = B·N·N

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → FP16
    # ------------------------------------------------------------------
    x_flat = input_tensor.view(total_rows, dim).contiguous()          # [R, dim] float32
    x_norm_half = torch.empty(total_rows, dim,
                              dtype=torch.float16,
                              device=device)

    BLOCK_M = 256
    BLOCK_D = 32
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm_half,
        weights["norm.weight"],    # γ  (float32)
        weights["norm.bias"],      # β  (float32)
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fuse left/right projections + left/right gates (4·H outputs)
    # ------------------------------------------------------------------
    H = hidden_dim
    # Concatenate the four projection/gate weights: shape (4·H, dim)
    w_proj_gate = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                 # (4·H, dim)

    # One GEMM:  [R, dim] @ [4·H, dim].T → [R, 4·H]
    proj_gate = F.linear(x_norm_half, w_proj_gate)      # FP16

    # Split into the four components
    left_proj_raw  = proj_gate[:, 0:H]            # [R, H]
    right_proj_raw = proj_gate[:, H:2*H]          # [R, H]
    left_gate_raw  = proj_gate[:, 2*H:3*H]        # [R, H]
    right_gate_raw = proj_gate[:, 3*H:4*H]        # [R, H]

    # Output gate (separate linear)
    out_gate_raw = F.linear(
        x_norm_half,
        weights["out_gate.weight"].to(torch.float16)  # (H, dim)
    )                                                # [R, H]

    # ------------------------------------------------------------------
    # 3️⃣ Mask, sigmoid gates, element‑wise gating
    # ------------------------------------------------------------------
    if not config.get("nomask", True):
        # mask: [B,N,N] → [R,1] (float16) and broadcast
        mask_h = mask_tensor.to(torch.float16).reshape(total_rows, 1)
        left_proj_raw  = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    # In‑place sigmoid to avoid extra allocations
    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    left  = left_proj_raw * left_gate      # [R, H]  fp16
    right = right_proj_raw * right_gate    # [R, H]  fp16

    # ------------------------------------------------------------------
    # 4️⃣ Reshape for triangular multiplicative update
    # ------------------------------------------------------------------
    # From [R, H] → [B, N, N, H]
    left  = left.view(B, N, N, H)
    right = right.view(B, N, N, H)
    out_gate = out_gate.view(B, N, N, H)

    # Move head dimension into the batch axis: [B, H, N, N]
    left_perm  = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    # Flatten batch*head → one big batch for the kernel
    left_flat  = left_perm.view(B * H, N, N).contiguous()
    right_flat = right_perm.view(B * H, N, N).contiguous()
    out_flat   = torch.empty_like(left_flat, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 5️⃣ Batched triangular multiplicative update (Triton GEMM)
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (
        B * H,                                 # batch dimension
        triton.cdiv(N, BLOCK_M),               # row tiles
        triton.cdiv(N, BLOCK_N),               # col tiles
    )
    tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        right_flat.stride(0), right_flat.stride(1), right_flat.stride(2),
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # Reshape back to [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm over hidden dimension → FP16
    # ------------------------------------------------------------------
    out_flat2 = out.view(total_rows, H).contiguous()
    out_norm_half = torch.empty_like(out_flat2, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat2,
        out_norm_half,
        weights["to_out_norm.weight"],   # γ  (float32)
        weights["to_out_norm.bias"],     # β  (float32)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    out_norm = out_norm_half.view(B, N, N, H)   # FP16

    # ------------------------------------------------------------------
    # 7️⃣ Apply output gate, final projection and cast to FP32
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate               # [B,N,N,H] FP16

    final_half = F.linear(
        out_norm.view(total_rows, H),
        weights["to_out.weight"].to(torch.float16)   # (dim, H)
    )                                                # [R, dim] FP16

    final = final_half.view(B, N, N, dim).to(torch.float32)

    return final
