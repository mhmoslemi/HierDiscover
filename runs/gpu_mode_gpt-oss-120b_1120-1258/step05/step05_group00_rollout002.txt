"""
Fast outgoing TriMul (AlphaFold‑3) forward pass.

Key ideas
---------
1️⃣  Row‑wise LayerNorm (float32 accumulation → fp16 output) implemented in Triton.
2️⃣  Fuse the four projection/gate linear layers **and** the output‑gate linear
    into a *single* half‑precision GEMM by concatenating the 5 weight matrices.
3️⃣  Apply the (optional) mask only to the two projected tensors.
4️⃣  Compute the N³ multiplicative update with a batched half‑precision GEMM
    `torch.bmm(left, right.transpose(1, 2))` (leverages Tensor Cores on H100).
5️⃣  Second LayerNorm (Triton) over the hidden dimension.
6️⃣  Apply the output gate and a final half‑precision linear projection.
   Cast the result back to float32 to match the reference implementation.

All heavy per‑row normalisations are performed by the custom Triton kernel,
while the large matrix multiplications use cuBLAS/Tensor‑Core GEMM for peak
throughput.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton row‑wise LayerNorm (FP32 accum → FP16 output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,      # const   * [total_rows, dim]   (float32)
    y_ptr,      # mutable * [total_rows, dim]   (float16)
    w_ptr,      # const   * [dim]               (float32) – γ
    b_ptr,      # const   * [dim]               (float32) – β
    total_rows, # int (runtime)
    dim,        # int (runtime)
    eps,        # float (compile‑time constexpr)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # cols per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows                         # row mask

    # ------------------------------------------------------------------
    # Compute mean & variance for each row
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        # load a tile [BLOCK_M, BLOCK_D]
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # fp32
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)            # 1 / sqrt(var + eps)

    # ------------------------------------------------------------------
    # Write normalized output: γ·(x−μ)·rstd + β   (fp16)
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # β

        out = norm * w[None, :] + b[None, :]                # fp32
        out_fp16 = out.to(tl.float16)

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) – may be
                         ignored if config["nomask"] is True
        - weights_dict : mapping with all Linear/LayerNorm parameters
        - config_dict  : must contain "dim", "hidden_dim" and boolean "nomask"

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim], dtype float32
    """
    # ------------------------------------------------------------------
    # Unpack ------------------------------------------------------------
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim         = config["dim"]
    hidden_dim  = config["hidden_dim"]
    eps         = 1e-5
    device      = input_tensor.device

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N                # R = B·N·N

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → fp16
    # ------------------------------------------------------------------
    x_flat = input_tensor.view(total_rows, dim).contiguous()   # [R, dim] fp32
    x_norm = torch.empty(total_rows, dim,
                         dtype=torch.float16,
                         device=device)                     # [R, dim] fp16

    BLOCK_M = 1024    # rows per program – larger blocks reduce launch overhead
    BLOCK_D = 64      # cols per iteration – aligns well with typical dim (≤384)
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],          # γ (float32)
        weights["norm.bias"],            # β (float32)
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fuse all five linear transforms (4*H + H) into ONE GEMM
    # ------------------------------------------------------------------
    # Concatenate the five weight matrices: left_proj, right_proj,
    # left_gate, right_gate, out_gate   → shape [5*H, dim]
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                 # [5*H, dim]

    # Single half‑precision linear:  [R, dim] @ [5*H, dim].T → [R, 5*H]
    proj = F.linear(x_norm, w_concat)                   # fp16

    H = hidden_dim
    left_proj_raw   = proj[:, 0:H]               # [R, H]
    right_proj_raw  = proj[:, H:2*H]
    left_gate_raw   = proj[:, 2*H:3*H]
    right_gate_raw  = proj[:, 3*H:4*H]
    out_gate_raw    = proj[:, 4*H:5*H]           # [R, H]

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask, sigmoid gates, element‑wise gating
    # ------------------------------------------------------------------
    if not config.get("nomask", True):
        # mask: [B,N,N] → [R,1] (fp16)
        mask_h = mask_tensor.to(torch.float16).reshape(total_rows, 1)
        left_proj_raw  = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    # In‑place sigmoid to save memory
    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    # Apply gates
    left  = left_proj_raw * left_gate          # [R, H] fp16
    right = right_proj_raw * right_gate        # [R, H] fp16

    # ------------------------------------------------------------------
    # 4️⃣ N³ multiplicative update  (batched GEMM)
    # ------------------------------------------------------------------
    # Reshape to [B, N, N, H]
    left  = left.view(B, N, N, H)
    right = right.view(B, N, N, H)
    out_gate = out_gate.view(B, N, N, H)

    # Bring head dimension into the batch axis for torch.bmm
    # → shape (B*H, N, N) for both operands
    left  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    left_mat  = left.view(B * H, N, N)      # [B*H, N, N]  fp16
    right_mat = right.view(B * H, N, N)     # [B*H, N, N]  fp16

    # Batched GEMM: (B*H, N, N) @ (B*H, N, N)ᵀ → (B*H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm over hidden dimension → fp16
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, H).contiguous()
    out_norm = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    out_norm = out_norm.view(B, N, N, H)   # fp16

    # ------------------------------------------------------------------
    # 6️⃣ Apply output gate and final projection back to dim
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate          # [B,N,N,H] fp16

    final_half = F.linear(
        out_norm.view(total_rows, H),
        weights["to_out.weight"].to(torch.float16)   # [dim, H] → fp16
    )                                            # fp16, shape [R, dim]

    final = final_half.view(B, N, N, dim).to(torch.float32)

    return final
