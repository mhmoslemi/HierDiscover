def custom_kernel(data):
    inp, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden = cfg["hidden_dim"]
    eps = cfg.get("eps", 1e-5)
    nomask = cfg.get("nomask", True)
    device = inp.device
    B,N,_,_ = inp.shape
    total_rows = B * N * N

    # First LayerNorm using torch.nn.functional
    # Input: inp_flat: (R, dim) FP32
    inp_flat = inp.view(total_rows, dim)
    # compute LN
    x_norm_fp32 = torch.nn.functional.layer_norm(inp_flat, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps)
    # cast to half
    x_norm_fp16 = x_norm_fp32.half()

    # Concatenated projection weight 5*hidden x dim (dim is columns)
    concat_w = torch.cat([...], dim=0).half().contiguous()
    proj_all = torch.nn.functional.linear(x_norm_fp16, concat_w) # (R,5*hidden)

    left_proj_fp16, right_proj_fp16, left_gate_fp16, right_gate_fp16, out_gate_fp16 = torch.split(proj_all, hidden, dim=1)

    if not nomask:
        mask_flat = mask.reshape(total_rows, 1).to(torch.float16)
        left_proj_fp16 = left_proj_fp16 * mask_flat
        right_proj_fp16 = right_proj_fp16 * mask_flat

    left_gate = torch.sigmoid(left_gate_fp16)
    right_gate = torch.sigmoid(right_gate_fp16)
    out_gate = torch.sigmoid(out_gate_fp16)

    left_fp16 = left_proj_fp16 * left_gate
    right_fp16 = right_proj_fp16 * right_gate

    # Reshape for bmm
    left = left_fp16.view(B, N, N, hidden)
    right = right_fp16.view(B, N, N, hidden)

    # matmul: left[b,i,k,h] * right[b,j,k,h] across k
    # need left_mat shape (B*hidden, N, N) row i, col k
    # we can use torch.einsum as alternate but will be slower.
    left_mat = left.permute(0, 3, 1, 2).reshape(B * hidden, N, N)
    right_mat = right.permute(0, 3, 2, 1).reshape(B * hidden, N, N)

    out_mat = torch.bmm(left_mat, right_mat)  # (B*hidden,N,N)
    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # second layernorm + gate using kernel
    out_flat = out.view(total_rows, hidden).contiguous()
    out_norm_gate_fp16 = torch.empty_like(out_flat, dtype=torch.half, device=device)

    BLOCK_M = 256

        hidden,
        eps,

        BLOCK_D=BLOCK_D,

"""
Highly‑optimized forward implementation of the **outgoing** TriMul operator
(AlphaFold‑3).  

Algorithm
---------
1️⃣  Row‑wise LayerNorm on the input tensor (FP32 → FP16) – we use the
    fused PyTorch `torch.nn.functional.layer_norm` which is already
    heavily vectorized on H100.  

2️⃣  Fuse the 5 linear maps (left/right projections + 3 gates) into a single
    half‑precision GEMM (`torch.nn.functional.linear`).  

3️⃣  Optional binary mask is applied to the left/right projected tensors.  

4️⃣  Sigmoid gates (still FP16) and element‑wise gating.  

5️⃣  Triangular multiplicative update:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    Implemented as a batched half‑precision GEMM (`torch.bmm`) on shape
    (B·H, N, N).  This is the dominant O(H·N³) work and leverages
    Tensor‑Cores automatically.  

6️⃣  Fused LayerNorm + output‑gate (both half‑precision) – a custom Triton
    kernel (`fused_layernorm_gate_kernel`).  It computes per‑row mean /
   variance in FP32, normalises, applies the learned γ/β and finally
   multiplies by the already‑sigmoided `out_gate`.  

7️⃣  Final linear projection back to the model dimension (`dim`) – another
    half‑precision GEMM.  

All heavy linear algebra uses cuBLAS/Tensor‑Core which is optimal on an H100.
At least one operation (the second LayerNorm + gate) is written in Triton
as required.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 6) Fused LayerNorm + out‑gate (FP16 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def fused_layernorm_gate_kernel(
    x_ptr,          # const   half *[total_rows, dim]  (input)
    y_ptr,          # mutable half *[total_rows, dim]  (output)
    w_ptr,          # const   float* [dim]           (γ)
    b_ptr,          # const   float* [dim]           (β)
    gate_ptr,       # const   half *[total_rows, dim] (out_gate, already sigmoid‑ed)
    total_rows: tl.int32,
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,   # rows processed per program (e.g. 256)
    BLOCK_D: tl.constexpr,   # columns processed per inner loop (e.g. 128)
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ---------------------------------------------------------------
    # First pass: per‑row mean & variance (FP32 accumulation)
    # ---------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)      # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                         # 1 / sqrt(var+eps)

    # ---------------------------------------------------------------
    # Second pass: normalise, apply γ, β and multiply by out_gate
    # ---------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # FP32

        # (x - μ) / σ
        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]                 # FP32

        # multiply by out_gate (half → FP32)
        gate = tl.load(gate_ptr + offs_m[:, None] * dim + offs_d[None, :],
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        out = out * gate

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])

        - input_tensor : torch.Tensor, shape [B, N, N, C] (float32)
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) – ignored if config["nomask"]=True
        - weights_dict : dict of model parameters (float32)
        - config_dict  : {
                "dim"        : int (C),
                "hidden_dim" : int (H),
                "nomask"     : bool,
                "eps"        : float, optional (default 1e-5)
          }

    Returns
    -------
    torch.Tensor
        Shape [B, N, N, C] (float32) on the same CUDA device.
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    inp, mask, weights, cfg = data
    device = inp.device
    dim = cfg["dim"]               # input channel dimension C
    hidden = cfg["hidden_dim"]     # hidden dimension H
    eps = cfg.get("eps", 1e-5)
    nomask = cfg.get("nomask", True)

    B, N, _, _ = inp.shape
    total_rows = B * N * N                     # rows for flattened tensors

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32 → FP16) – use PyTorch fused implementation
    # ------------------------------------------------------------------
    inp_flat = inp.view(total_rows, dim)                       # (R, C) FP32
    x_norm_fp32 = F.layer_norm(
        inp_flat,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )                                                          # (R, C) FP32
    x_norm_fp16 = x_norm_fp32.half()                          # (R, C) FP16

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear projection for left/right + three gates
    # ------------------------------------------------------------------
    #   Weight layout: [5*H, C] (no bias) – cast once to FP16
    concat_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).half().contiguous()                                     # (5*H, C)

    proj_all = F.linear(x_norm_fp16, concat_w)                # (R, 5*H)

    # split into the five tensors
    left_proj, right_proj, left_gate, right_gate, out_gate = torch.split(
        proj_all, hidden, dim=1
    )  # each (R, H), all FP16

    # ------------------------------------------------------------------
    # 2a) Apply mask (if present) – broadcast on hidden dimension
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask.reshape(total_rows, 1).to(torch.float16)
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # ------------------------------------------------------------------
    # 2b) Sigmoid gates and element‑wise gating
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)          # (R, H) FP16
    right_gate = torch.sigmoid(right_gate)        # (R, H) FP16
    out_gate  = torch.sigmoid(out_gate)           # (R, H) FP16  – used later

    left = left_proj * left_gate   # (R, H) FP16
    right = right_proj * right_gate  # (R, H) FP16

    # ------------------------------------------------------------------
    # 3️⃣ Triangular multiplicative update (batched GEMM)
    # ------------------------------------------------------------------
    # reshape to (B, N, N, H) each
    left = left.view(B, N, N, hidden)
    right = right.view(B, N, N, hidden)

    # view as (B*H, N, N) for torch.bmm
    left_mat = left.permute(0, 3, 1, 2).reshape(B * hidden, N, N)   # (B·H, N, N)
    right_mat = right.permute(0, 3, 2, 1).reshape(B * hidden, N, N) # (B·H, N, N)

    out_mat = torch.bmm(left_mat, right_mat)                         # (B·H, N, N)

    # back to (B, N, N, H)
    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 4️⃣ Second LayerNorm + out‑gate (fused Triton kernel)
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, hidden).contiguous()            # (R, H) half
    out_norm_gate = torch.empty_like(out_flat, dtype=torch.half, device=device)

    # Block sizes – tuned for H ≤ 128
    BLOCK_M = 256                      # rows per program
    BLOCK_D = 128                      # columns per inner loop (covers hidden in one pass)
    grid = (triton.cdiv(total_rows, BLOCK_M),)

    fused_layernorm_gate_kernel[grid](
        out_flat,
        out_norm_gate,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        out_gate,                     # already sigmoid‑ed, half
        total_rows,
        hidden,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 5️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].half()                # (dim, H)
    final_fp16 = F.linear(out_norm_gate, to_out_w)           # (R, dim) half

    # ------------------------------------------------------------------
    # Cast to FP32 as expected by the original module and reshape
    # ------------------------------------------------------------------
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
