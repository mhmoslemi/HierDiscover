"""
TriMul “outgoing” forward pass (AlphaFold‑3) – Triton + PyTorch.

Key ideas
---------
* The two LayerNorms are written in Triton (row‑wise, FP32 statistics, FP16
  output) – they replace the PyTorch `nn.LayerNorm` and avoid an extra
  synchronisation.
* Linear projections and gates are fused:
        [proj , gate] = Linear(x)   →   split → sigmoid(gate)
  This saves one memory read/write for each of the four linear layers.
* The heavy O(N³·H) contraction is performed with a single ``torch.einsum``
  which lets PyTorch pick the best cuTensor/cuBLAS path (no explicit
  ``permute``/``contiguous`` copies).
* The second LayerNorm is again a Triton kernel; the final linear
  projection uses the usual fast `F.linear`.
* All tensors (except the original input) live in FP16 after the first
  LayerNorm, matching Tensor‑Core precision on H100.  The output is cast
  back to FP32 to stay compatible with the reference module.

The implementation meets the required signature ``custom_kernel(data)`` and
contains a Triton kernel for LayerNorm.  It works for all configurations
listed in the problem statement (batch ≤ 2, N ≤ 1024, dim ∈ {128,384},
hidden = 128).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (FP32 statistics → FP16 output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,          # float*   [total_rows, dim]   (input, fp32)
    y_ptr,          # half*    [total_rows, dim]   (output, fp16)
    gamma_ptr,      # float*   [dim]               (scale)
    beta_ptr,       # float*   [dim]               (bias)
    total_rows,     # int (runtime)
    dim: tl.constexpr,          # static: number of columns
    eps: tl.constexpr,          # static epsilon
    BLOCK_M: tl.constexpr,      # rows per program
    BLOCK_D: tl.constexpr,      # column block size
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)              # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ---------- Pass 1 – mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)        # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)           # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                               # 1/√(var+ε)

    # ---------- Pass 2 – write normalized values ----------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(gamma_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(beta_ptr  + offs_d, mask=mask_d, other=0.0)

        out = norm * w[None, :] + b[None, :]                # FP32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the "outgoing" TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        * input_tensor : torch.Tensor[float32]  [B, N, N, dim]
        * mask_tensor  : torch.Tensor[float32] [B, N, N]  (may be all‑ones)
        * weights_dict : dictionary with the nine weight tensors required
        * config_dict  : {"dim": int, "hidden_dim": int, "nomask": bool (opt.)}

    Returns
    -------
    torch.Tensor[float32]  [B, N, N, dim]
    """
    # --------------------------------------------------------------
    # 0. unpack
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32 → FP16)
    # --------------------------------------------------------------
    total_rows = B * N * N
    x_flat = input_tensor.view(total_rows, dim).contiguous()          # [Rows, Dim]

    x_norm_h = torch.empty((total_rows, dim), dtype=torch.float16,
                           device=device)                            # FP16 output

    # reasonable block sizes for H100
    BLOCK_M = 256
    BLOCK_D = 128
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        x_flat,
        x_norm_h,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    # back to the 4‑D view, still FP16
    x_norm_h = x_norm_h.view(B, N, N, dim)

    # --------------------------------------------------------------
    # 2️⃣ Fuse projection + gate (4 linear layers → 2 fused matmuls)
    # --------------------------------------------------------------
    # cast all weight matrices once to FP16
    w_left_proj = weights["left_proj.weight"].to(torch.float16, non_blocking=True)
    w_right_proj = weights["right_proj.weight"].to(torch.float16, non_blocking=True)
    w_left_gate = weights["left_gate.weight"].to(torch.float16, non_blocking=True)
    w_right_gate = weights["right_gate.weight"].to(torch.float16, non_blocking=True)
    w_out_gate = weights["out_gate.weight"].to(torch.float16, non_blocking=True)

    rows = total_rows
    x_norm_h_flat = x_norm_h.view(rows, dim)          # (Rows, Dim)

    # left projection + left gate packed into one weight matrix
    left_concat_w = torch.cat([w_left_proj, w_left_gate], dim=0)    # (2·H, Dim)
    right_concat_w = torch.cat([w_right_proj, w_right_gate], dim=0) # (2·H, Dim)

    left_concat = F.linear(x_norm_h_flat, left_concat_w)   # (Rows, 2·H)
    right_concat = F.linear(x_norm_h_flat, right_concat_w)

    # split into proj / pre‑gate tensors
    left_proj_h      = left_concat[:, :hidden]
    left_gate_pre_h  = left_concat[:, hidden:]
    right_proj_h     = right_concat[:, :hidden]
    right_gate_pre_h = right_concat[:, hidden:]

    out_gate_h = torch.sigmoid(F.linear(x_norm_h_flat, w_out_gate))

    # reshape back to 4‑D tensors (still FP16)
    left_proj_h      = left_proj_h.view(B, N, N, hidden)
    left_gate_pre_h  = left_gate_pre_h.view(B, N, N, hidden)
    right_proj_h     = right_proj_h.view(B, N, N, hidden)
    right_gate_pre_h = right_gate_pre_h.view(B, N, N, hidden)
    out_gate_h       = out_gate_h.view(B, N, N, hidden)

    # --------------------------------------------------------------
    # 3️⃣ optional mask (broadcast over the hidden dim)
    # --------------------------------------------------------------
    if not nomask:
        mask_h = mask_tensor.unsqueeze(-1).to(torch.float16, non_blocking=True)
        left_proj_h = left_proj_h * mask_h
        right_proj_h = right_proj_h * mask_h

    # --------------------------------------------------------------
    # 4️⃣ apply sigmoid gates
    # --------------------------------------------------------------
    left_gate_h  = torch.sigmoid(left_gate_pre_h)
    right_gate_h = torch.sigmoid(right_gate_pre_h)

    left_h  = left_proj_h * left_gate_h          # (B, N, N, H)
    right_h = right_proj_h * right_gate_h

    # --------------------------------------------------------------
    # 5️⃣ Tri‑multiplicative update (batched GEMM via einsum)
    # --------------------------------------------------------------
    # einsum contracts over the shared 'k' dimension, leaving the hidden
    # dimension free on both sides → shape (B, N, N, H) without any
    # permute / copy.
    out_h = torch.einsum('b i k h, b j k h -> b i j h', left_h, right_h)

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm (over hidden) → FP16
    # --------------------------------------------------------------
    rows_h = B * N * N
    out_flat = out_h.view(rows_h, hidden).contiguous()
    out_norm_h = torch.empty_like(out_flat, dtype=torch.float16, device=device)

    grid2 = (triton.cdiv(rows_h, BLOCK_M),)
    layernorm_fp16_kernel[grid2](
        out_flat,
        out_norm_h,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        rows_h,
        hidden,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    out_norm = out_norm_h.view(B, N, N, hidden)         # (B,N,N,H) FP16

    # --------------------------------------------------------------
    # 7️⃣ Output gate → final linear projection
    # --------------------------------------------------------------
    gated = out_norm * out_gate_h                         # element‑wise FP16
    gated_flat = gated.view(rows_h, hidden)

    final_h = F.linear(gated_flat,
                       weights["to_out.weight"].to(torch.float16, non_blocking=True))

    # cast back to FP32 (the original module’s dtype)
    final = final_h.view(B, N, N, dim).to(torch.float32)

    return final
