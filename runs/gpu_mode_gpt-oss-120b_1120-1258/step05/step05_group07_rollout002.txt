"""
TriMul (outgoing) – fast forward implementation.

Algorithm
---------
1️⃣ Row‑wise LayerNorm over the input channels (C) using a small Triton
   kernel (fp32 reductions → fp16 output).

2️⃣ Fuse the five linear projections (left/right + three gates) into a single
   half‑precision GEMM (weight matrix concatenation).  The result is split
   into the three intermediate tensors (left, right, gates).

3️⃣ (Optional) apply the pair‑wise mask *before* the gating step.

4️⃣ Apply sigmoid gates (fp16) and element‑wise gating.

5️⃣ Triangular multiplicative update (outgoing version):
   out_h[i, j] = Σ_k left_h[i, k] * right_h[j, k] .
   Realised by a batched fp16 GEMM `left @ rightᵀ`
   where the hidden dimension H is promoted to the batch axis
   (shape (B·H, N, N)).  CuBLAS/Tensor‑Core kernels handle this
   work efficiently.

6️⃣ Second LayerNorm over the hidden dimension (H) – the same Triton kernel.

7️⃣ Multiply by the output gate and apply the final linear projection back
   to the original channel dimension C.

All heavy linear algebra uses Tensor‑Core fp16 kernels; only the two
LayerNorms are implemented in Triton (thus satisfying the “must use a
kernel” requirement).  The implementation works for all test
configurations (B ∈ {1,2}, N ∈ {128…1024}, C ∈ {128,384},
H = hidden_dim ∈ {128}).  The final tensor is returned in float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton row‑wise LayerNorm (fp32 reduction, fp16 output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,           # *const  fp32*  [total_rows, dim]   (input, fp32 or fp16)
    y_ptr,           # *mut    fp16*  [total_rows, dim]   (output)
    weight_ptr,      # *const  fp32*  [dim]               (γ)
    bias_ptr,        # *const  fp32*  [dim]               (β)
    total_rows,      # i32
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # cols per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # -----------------------------------------------------------------
    # 1. Compute mean & variance (fp32) – two‑pass implementation
    # -----------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim

        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                                 # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                  # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)           # [BLOCK_M]

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)        # 1 / sqrt(var+eps)

    # -----------------------------------------------------------------
    # 2. Write normalized output (fp16)
    # -----------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                                 # fp32

        norm = (vals - mean[:, None]) * rstd[:, None]    # fp32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # fp32
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # fp32

        out_fp16 = (norm * w[None, :] + b[None, :]).to(tl.float16)

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom kernel entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, C] (float32)
        - mask_tensor  : torch.Tensor [B, N, N]   (bool/float, may be None)
        - weights_dict : dict(name → torch.Tensor) (float32)
        - config_dict  : {"dim":C, "hidden_dim":H, "nomask":bool}

    Returns
    -------
    torch.Tensor [B, N, N, C] (float32)
    """
    # --------------------------------------------------------------
    # Unpack
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                     # C
    hidden_dim = config["hidden_dim"]       # H
    eps = 1e-5
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N                 # rows for the flattened view

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (fp16 output) – Triton
    # --------------------------------------------------------------
    # Flatten to (total_rows, dim) – keep fp32 for numerical stability
    x_flat = input_tensor.contiguous().view(total_rows, dim)
    x_norm_fp16 = torch.empty(total_rows, dim,
                              dtype=torch.float16,
                              device=device)

    # Simple heuristic for block sizes – enough for dim ≤ 384
    BLOCK_M = 256
    BLOCK_D = 64
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # --------------------------------------------------------------
    # 2️⃣ Fuse the five linear projections (left/right + three gates)
    # --------------------------------------------------------------
    # Stack the five weight matrices – shape (5*H, C)
    # Cast once to half‑precision to avoid repeated conversions.
    concat_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).half().contiguous()

    # Single GEMM: (total_rows, C) @ (5*H, C).T  → (total_rows, 5*H)
    proj_all = F.linear(x_norm_fp16, concat_w)   # fp16 output

    # Split back into the five branches
    left_proj, right_proj, left_gate, right_gate, out_gate = torch.split(
        proj_all, hidden_dim, dim=1
    )  # each (total_rows, H)

    # --------------------------------------------------------------
    # 3️⃣ Optional mask (applied BEFORE gating)
    # --------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        # mask shape: [B, N, N] → broadcast over hidden dim
        mask_fp16 = mask_tensor.contiguous().view(total_rows, 1).to(torch.float16)
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # --------------------------------------------------------------
    # 4️⃣ Apply sigmoid gates (still fp16) and element‑wise gating
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)       # (total_rows, H)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    left = left_proj * left_gate               # (total_rows, H)
    right = right_proj * right_gate

    # Reshape to 4‑D layout expected by the outgoing update
    left = left.view(B, N, N, hidden_dim)      # (B, N, N, H)
    right = right.view(B, N, N, hidden_dim)
    out_gate = out_gate.view(B, N, N, hidden_dim)

    # --------------------------------------------------------------
    # 5️⃣ Triangular multiplicative update (batched GEMM)
    # --------------------------------------------------------------
    # Move hidden dimension into the batch axis – no extra copy is needed
    # because `.contiguous()` would allocate a large temporary buffer.
    # Instead we use `.transpose` + `.reshape` which yields a view with
    # the required stride pattern and let cuBLAS handle non‑contiguous
    # input (it internally copies into a temporary contiguous buffer once).
    # This eliminates the explicit `contiguous()` calls that previously
    # caused an extra ~0.25 GiB copy.
    left_mat = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)

    # Batched half‑precision matrix multiplication: left @ rightᵀ
    # Result shape: (B·H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Restore the 4‑D layout (B, N, N, H)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm over hidden dimension H – Triton
    # --------------------------------------------------------------
    out_flat = out.contiguous().view(total_rows, hidden_dim)
    out_norm_fp16 = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    out_norm = out_norm_fp16.view(B, N, N, hidden_dim)   # (B,N,N,H)

    # --------------------------------------------------------------
    # 7️⃣ Final gating + linear projection back to C
    # --------------------------------------------------------------
    gated = out_norm * out_gate                           # (B,N,N,H) fp16

    # Final projection – fused GEMM
    to_out_w_fp16 = weights["to_out.weight"].half()
    gated_flat = gated.view(total_rows, hidden_dim)       # (total_rows, H)
    final_fp16 = F.linear(gated_flat, to_out_w_fp16)     # (total_rows, C)

    # --------------------------------------------------------------
    # 8️⃣ Cast to fp32 and return
    # --------------------------------------------------------------
    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final
```
