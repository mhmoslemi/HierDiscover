"""
custom_kernel
==============

Fast forward‑pass for the **outgoing** TriMul operator (AlphaFold‑3).

Algorithm (mixed FP16/FP32):
1️⃣ Row‑wise LayerNorm over the input channel dimension C.
   – reduction in FP32 (mean/var) → output FP16.
2️⃣ Fuse the **five** linear projections (left, right and three gates)
   into a single GEMM by concatenating their weight matrices.
3️⃣ (Optional) apply the pair‑wise mask to the left/right projections.
4️⃣ Apply sigmoid gates and element‑wise gating (still FP16).
5️⃣ Triangular multiplicative update (outgoing version):
        out_h[i, j] = Σ_k left_h[i, k] * right_h[j, k]
   – realized as a batched GEMM `left @ rightᵀ` on tensors of shape
     (B·H, N, N) using `torch.bmm` (cuBLAS Tensor‑Cores).
6️⃣ Row‑wise LayerNorm over the hidden dimension H.
7️⃣ Multiply by the output gate, then a final linear projection back to C.
8️⃣ Cast the result to FP32 and return.

All heavy reductions (LayerNorm) are executed in Triton; dense linear algebra
uses cuBLAS which gives the best performance on an H100.  The implementation
supports the presence/absence of a mask and works for any
`dim ∈ {128,384}` and `hidden_dim = 128` as required by the test suite.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton Row‑wise LayerNorm (FP16 output, FP32 accumulation)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,          # const * (total_rows, dim)   input  (fp16 or fp32)
    y_ptr,          #      * (total_rows, dim)   output (fp16)
    weight_ptr,     # const * (dim)               γ
    bias_ptr,       # const * (dim)               β
    total_rows,     # i32
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,   # rows processed per program
    BLOCK_D: tl.constexpr,   # columns processed per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # -------------------------------------------------
    # 1. Compute mean and variance (FP32 reduction)
    # -------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        # pointer for a 2‑D slice [row, col]
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        # load as whatever dtype the pointer holds, then up‑cast
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                     # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)       # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                # 1 / sqrt(var + eps)

    # -------------------------------------------------
    # 2. Write normalized output (FP16)
    # -------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        # (x - μ) / √(σ²+ε)
        norm = (vals - mean[:, None]) * rstd[:, None]

        # γ and β are FP32
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)

        out_fp16 = (norm * w[None, :] + b[None, :]).to(tl.float16)

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor of shape [B, N, N, C] (float32)
            - mask_tensor  : torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights_dict : mapping from str → torch.Tensor (float32)
            - config_dict  : {"dim": C, "hidden_dim": H, "nomask": bool}
    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # -------------------------------------------------
    # Unpack inputs
    # -------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                 # input channel dim C
    hidden_dim = config["hidden_dim"]   # hidden size H
    eps = 1e-5
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N

    # -------------------------------------------------
    # 1️⃣ First LayerNorm (FP16 output)
    # -------------------------------------------------
    # Flatten to [total_rows, dim] (float32)
    x_flat = input_tensor.contiguous().view(total_rows, dim)

    # Allocate fp16 buffer for normalized tensor
    x_norm_fp16 = torch.empty(total_rows, dim,
                              dtype=torch.float16,
                              device=device)

    BLOCK_M = 256          # rows per program (tuned for H100)
    BLOCK_D = 64           # columns per iteration
    grid = (triton.cdiv(total_rows, BLOCK_M),)

    layernorm_fp16_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # -------------------------------------------------
    # 2️⃣ Fuse the five linear projections
    #    (left, right, left_gate, right_gate, out_gate)
    # -------------------------------------------------
    # Concatenate weight matrices: each has shape [hidden_dim, dim]
    w_left    = weights["left_proj.weight"]
    w_right   = weights["right_proj.weight"]
    w_lgate   = weights["left_gate.weight"]
    w_rgate   = weights["right_gate.weight"]
    w_outgate = weights["out_gate.weight"]

    # (5*H, C) matrix, half precision
    concat_w = torch.cat([w_left, w_right, w_lgate, w_rgate, w_outgate],
                         dim=0).half().contiguous()

    # Fused GEMM: [total_rows, C] @ [C, 5*H]^T -> [total_rows, 5*H]
    proj_all = F.linear(x_norm_fp16, concat_w)   # FP16, shape (total_rows, 5*H)

    # Split into the five heads
    left_proj_fp16, right_proj_fp16, left_gate_fp16, \
        right_gate_fp16, out_gate_fp16 = torch.split(
        proj_all, hidden_dim, dim=1
    )                                            # each (total_rows, H)

    # -------------------------------------------------
    # 3️⃣ Optional mask (applied before gating)
    # -------------------------------------------------
    if (not nomask) and (mask_tensor is not None):
        # mask shape [B, N, N] → flatten → [total_rows, 1]
        mask_fp16 = mask_tensor.reshape(total_rows, 1).to(torch.float16)
        left_proj_fp16  = left_proj_fp16 * mask_fp16
        right_proj_fp16 = right_proj_fp16 * mask_fp16

    # -------------------------------------------------
    # 4️⃣ Apply sigmoid gates and element‑wise gating
    # -------------------------------------------------
    left_gate   = torch.sigmoid(left_gate_fp16)
    right_gate  = torch.sigmoid(right_gate_fp16)
    out_gate    = torch.sigmoid(out_gate_fp16)

    left_fp16   = left_proj_fp16 * left_gate
    right_fp16  = right_proj_fp16 * right_gate

    # Reshape back to 4‑D tensors: [B, N, N, H]
    left  = left_fp16.view(B, N, N, hidden_dim)
    right = right_fp16.view(B, N, N, hidden_dim)
    out_gate = out_gate.view(B, N, N, hidden_dim)

    # -------------------------------------------------
    # 5️⃣ Triangular multiplicative update (batched GEMM)
    #     out_h[i, j] = Σ_k left_h[i, k] * right_h[j, k]
    # -------------------------------------------------
    # Move hidden dim to the second axis and treat each head as an
    # independent batch: (B, H, N, N) → (B·H, N, N)
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    BH = B * hidden_dim
    left_mat  = left_perm.view(BH, N, N)                 # [BH, N, N] FP16
    right_mat = right_perm.view(BH, N, N)                # [BH, N, N] FP16

    # Batched matrix multiplication: left @ rightᵀ  → [BH, N, N]
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Restore original layout: [B, N, N, H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # -------------------------------------------------
    # 6️⃣ Second LayerNorm over hidden dimension H
    # -------------------------------------------------
    out_flat = out.view(total_rows, hidden_dim)   # [total_rows, H]
    out_norm_fp16 = torch.empty_like(out_flat,
                                     dtype=torch.float16,
                                     device=device)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    out_norm = out_norm_fp16.view(B, N, N, hidden_dim)   # [B,N,N,H]

    # -------------------------------------------------
    # 7️⃣ Final gating + linear projection back to C
    # -------------------------------------------------
    gated = out_norm * out_gate                # FP16 element‑wise

    # Final linear: (total_rows, H) @ (C, H)^T → (total_rows, C)
    to_out_w_fp16 = weights["to_out.weight"].half()
    gated_flat = gated.view(total_rows, hidden_dim)
    final_fp16 = F.linear(gated_flat, to_out_w_fp16)   # FP16

    # -------------------------------------------------
    # 8️⃣ Cast back to FP32 and return
    # -------------------------------------------------
    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final
