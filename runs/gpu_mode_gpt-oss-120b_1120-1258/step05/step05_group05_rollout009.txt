"""
Fast outgoing TriMul (AlphaFold‑3) implementation.

Algorithm (FP16/FP32 mixed‑precision):
1️⃣  Layer‑norm over the input channel dimension C  (torch.nn.functional.layer_norm,
    FP32 accumulation, FP16 output).
2️⃣  Fuse the five linear projections (left/right + three gates) into a single
    GEMM by concatenating their weight matrices (half‑precision).
3️⃣  Optional pair‑wise mask – applied with a tiny Triton kernel that multiplies
    each row of the left/right projections by the scalar mask value.
4️⃣  Apply sigmoid gates and element‑wise gating (still FP16).
5️⃣  Triangular multiplicative update (outgoing version):
       out_h[i, j] = Σ_k left_h[i, k] * right_h[j, k]
    Implemented as a batched GEMM `left @ rightᵀ` on tensors of shape
    (B·H, N, N) using cuBLAS (torch.bmm, Tensor‑Core kernels).
6️⃣  Second Layer‑norm over the hidden dimension H (torch.nn.functional.layer_norm,
    FP16 output).
7️⃣  Multiply by the output gate and a final linear projection back to C
    (half‑precision GEMM).
8️⃣  Cast the result to FP32 and return.

A Triton kernel is used only for the mask‑multiplication step (mandatory
kernel usage). All heavy linear‑algebra is delegated to cuBLAS, which gives the
best possible throughput on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: element‑wise multiplication by a per‑row mask.
# Each element `proj[i, h]` is multiplied by `mask[i]`.
# ----------------------------------------------------------------------
@triton.jit
def mask_mul_kernel(
    proj_ptr,          # *mut   half*  [total_elements]   (left or right proj)
    mask_ptr,          # *const half* [total_rows]      (scalar mask per row)
    total_elements,   # i32
    hidden_dim: tl.constexpr,   # i32 – size of the inner (gate) dimension
    BLOCK_SIZE: tl.constexpr,   # threads per program
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    active = offs < total_elements

    # row = offs // hidden_dim  (integer division)
    row = offs // hidden_dim

    mask_val = tl.load(mask_ptr + row, mask=active)
    proj_val = tl.load(proj_ptr + offs, mask=active)
    tl.store(proj_ptr + offs, proj_val * mask_val, mask=active)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul operator.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, C] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict  : mapping of weight names to tensors (float32)
            - config_dict   : {"dim": C, "hidden_dim": H,
                              "nomask": bool, "eps": optional float}
    Returns:
        torch.Tensor [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                # input channel dimension C
    hidden_dim = config["hidden_dim"]  # hidden dimension H
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N                          # rows = B·N·N
    total_elements = total_rows * hidden_dim       # elements in a projection

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP16 output)
    # ------------------------------------------------------------------
    # torch.nn.functional.layer_norm is highly optimized (CUDA/CuDNN)
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )                               # [B, N, N, C]  (float32)
    x_norm_fp16 = x_norm.half()    # cast once for the forthcoming GEMM

    # ------------------------------------------------------------------
    # 2️⃣ Fuse five linear projections (left/right + three gates)
    # ------------------------------------------------------------------
    w_left    = weights["left_proj.weight"]
    w_right   = weights["right_proj.weight"]
    w_lgate   = weights["left_gate.weight"]
    w_rgate   = weights["right_gate.weight"]
    w_outgate = weights["out_gate.weight"]
    # Concatenate along the output‑feature dimension → [(5·H), C]
    concat_w = torch.cat(
        [w_left, w_right, w_lgate, w_rgate, w_outgate], dim=0
    ).half().contiguous()

    # Fused GEMM: [total_rows, C] @ [C, 5·H]ᵀ → [total_rows, 5·H]   (FP16)
    proj_all = F.linear(x_norm_fp16, concat_w)   # shape [total_rows, 5·hidden_dim]

    # ------------------------------------------------------------------
    # 3️⃣ Split projections back to the five tensors
    # ------------------------------------------------------------------
    left_proj_fp16, right_proj_fp16, left_gate_fp16, \
        right_gate_fp16, out_gate_fp16 = torch.split(
            proj_all, hidden_dim, dim=1
        )  # each [total_rows, H]

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask (applied before gating)
    # ------------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        # mask → [total_rows] (0/1) in half‑precision
        mask_fp16 = mask_tensor.reshape(total_rows).to(torch.float16)

        # Triton kernel: left_proj_fp16 *= mask, right_proj_fp16 *= mask
        BLOCK_SIZE = 1024
        grid = (triton.cdiv(total_elements, BLOCK_SIZE),)

        mask_mul_kernel[grid](
            left_proj_fp16, mask_fp16, total_elements,
            hidden_dim, BLOCK_SIZE=BLOCK_SIZE
        )
        mask_mul_kernel[grid](
            right_proj_fp16, mask_fp16, total_elements,
            hidden_dim, BLOCK_SIZE=BLOCK_SIZE
        )

    # ------------------------------------------------------------------
    # 5️⃣ Apply sigmoid gates and element‑wise gating (still FP16)
    # ------------------------------------------------------------------
    left_gate   = torch.sigmoid(left_gate_fp16)       # [total_rows, H]
    right_gate  = torch.sigmoid(right_gate_fp16)
    out_gate    = torch.sigmoid(out_gate_fp16)       # will be used later

    left  = left_proj_fp16 * left_gate               # [total_rows, H]
    right = right_proj_fp16 * right_gate

    # ------------------------------------------------------------------
    # 6️⃣ Reshape to (B, H, N, N) for batched GEMM
    # ------------------------------------------------------------------
    left  = left.view(B, N, N, hidden_dim).permute(0, 3, 1, 2).contiguous()
    right = right.view(B, N, N, hidden_dim).permute(0, 3, 1, 2).contiguous()
    # Shapes → [B, H, N, N]

    BH = B * hidden_dim
    left_mat  = left.view(BH, N, N)          # [BH, N, N] (FP16)
    right_mat = right.view(BH, N, N)         # [BH, N, N] (FP16)

    # ------------------------------------------------------------------
    # 7️⃣ Triangular multiplicative update → batched GEMM
    # ------------------------------------------------------------------
    # out_mat[b] = left_mat[b] @ right_mat[b]ᵀ
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Restore layout → [B, N, N, H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8️⃣ Second LayerNorm over hidden dimension H
    # ------------------------------------------------------------------
    out_norm = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].half(),
        bias=weights["to_out_norm.bias"].half(),
        eps=eps,
    )   # FP16

    # ------------------------------------------------------------------
    # 9️⃣ Final gating (output gate) and linear projection back to C
    # ------------------------------------------------------------------
    out_gate = out_gate.view(B, N, N, hidden_dim)       # [B, N, N, H]
    gated = out_norm * out_gate                         # FP16

    to_out_w = weights["to_out.weight"].half()
    final_fp16 = F.linear(gated.view(total_rows, hidden_dim), to_out_w)  # [total_rows, C]

    # ------------------------------------------------------------------
    # 10️⃣ Cast back to FP32 and return
    # ------------------------------------------------------------------
    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final
