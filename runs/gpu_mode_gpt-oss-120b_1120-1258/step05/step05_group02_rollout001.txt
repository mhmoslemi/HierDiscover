    device = input_tensor.device
    B,N,_,dim = input_tensor.shape

    eps = 1e-5
    nomask = config.get("nomask", True)

    # Cast to half (more efficient)
    # first layernorm
    x_norm = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps)

    # combine proj+gate weights for left and right
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)

    # compute left_proj and left_gate in fused linear: need weight shape (2*hidden, dim)
    left_w = torch.cat([left_proj_w, left_gate_w], dim=0)    # [2*H, dim]
    right_w = torch.cat([right_proj_w, right_gate_w], dim=0)  # [2*H, dim]

    rows = B*N*N
    x_flat = x_norm.view(rows, dim)  # fp16
    left_concat = F.linear(x_flat, left_w)   # [rows, 2*H]
    right_concat = F.linear(x_flat, right_w)

    left_proj = left_concat[:, :hidden]
    left_gate_pre = left_concat[:, hidden:]
    right_proj = right_concat[:, :hidden]
    right_gate_pre = right_concat[:, hidden:]

    # out_gate
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x_flat, out_gate_w))

    if not nomask:
        mask_flat = mask_tensor.view(rows, 1).to(torch.float16)  # broadcast over hidden
        left_proj = left_proj * mask_flat
        right_proj = right_proj * mask_flat

    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)

    left = left_proj * left_gate   # rows x H
    right = right_proj * right_gate

    # reshape

Algorithm
~~~~~~~~~
1️⃣  Layer‑norm on the input tensor `[B, N, N, C]` (FP32 → FP16).  
    The kernel `layernorm_fp16` processes a whole row at a time, computes the
    mean/variance in FP32 and writes the normalized result in FP16.

2️⃣  Two linear projections (proj + gate) are fused by concatenating the weights
    (`proj.weight` and `gate.weight`) → a single `F.linear` per side.

3️⃣  Optional pair‑wise mask (broadcast over the hidden dimension) is applied
    *before* gating.

4️⃣  Sigmoid gates are evaluated (`torch.sigmoid`).

5️⃣  Tri‑multiplicative update    
    `out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]`    
    is a batched GEMM over the hidden dimension:
    * reshape to `[B·H, N, N]`
    * `torch.bmm(left, right.transpose(1,2))`
    – the heavy compute is performed by cuBLAS Tensor‑Core kernels.

6️⃣  Second Layer‑norm on the hidden dimension (again with `layernorm_fp16`).

7️⃣  Apply the output gate (`sigmoid`) and a final linear projection
    (`hidden → dim`).  Result is cast back to `float32`.

The implementation keeps **one Triton kernel** (the custom layer‑norm) and
uses highly‑optimized PyTorch kernels for all linear / GEMM operations,
which is the fastest combination on an H100 for the tested shapes.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (FP32 accumulator → FP16 output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16(
    x_ptr,            # *float32   [total_rows, dim]
    y_ptr,            # *float16   [total_rows, dim]
    gamma_ptr,        # *float32   [dim]
    beta_ptr,         # *float32   [dim]
    total_rows,       # i32 (runtime)
    dim: tl.constexpr,          # compile‑time: number of columns
    eps: tl.constexpr,          # compile‑time: epsilon
    BLOCK_M: tl.constexpr,      # rows per program
    BLOCK_D: tl.constexpr,      # column tile size
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)            # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # Pass 1 – compute mean & variance per row (FP32)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)         # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                  # sum over dim
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                           # 1 / sqrt(var+eps)

    # ------------------------------------------------------------------
    # Pass 2 – write normalized result (γ·x̂+β) as FP16
    # ------------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        # (x - μ) * rstd
        norm = (vals - mean[:, None]) * rstd[:, None]

        # γ, β (both FP32)
        w = tl.load(gamma_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(beta_ptr  + offs_d, mask=mask_d, other=0.0)

        out = norm * w[None, :] + b[None, :]            # FP32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        * input_tensor : torch.Tensor[float32]  [B, N, N, C]
        * mask_tensor  : torch.Tensor[float32] [B, N, N] (may be unused)
        * weights_dict : {
            "norm.weight":          Tensor[C],
            "norm.bias":            Tensor[C],
            "left_proj.weight":     Tensor[hidden, C],
            "right_proj.weight":    Tensor[hidden, C],
            "left_gate.weight":     Tensor[hidden, C],
            "right_gate.weight":    Tensor[hidden, C],
            "out_gate.weight":      Tensor[hidden, C],
            "to_out_norm.weight":   Tensor[hidden],
            "to_out_norm.bias":     Tensor[hidden],
            "to_out.weight":        Tensor[C, hidden],
        }
        * config_dict : {
            "dim": int,
            "hidden_dim": int,
            "nomask": bool (optional, default True)
        }

    Returns
    -------
    torch.Tensor[float32]  [B, N, N, C]
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, N, _, C = input_tensor.shape
    H = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣  First LayerNorm (FP32 → FP16) – Triton kernel
    # ------------------------------------------------------------------
    total_rows = B * N * N
    x_flat = input_tensor.view(total_rows, C).contiguous()      # [rows, C]

    # allocate FP16 buffer for the normalized tensor
    x_norm_h = torch.empty_like(x_flat, dtype=torch.float16)

    # launch parameters – tuned for C up to 384
    BLOCK_M = 256          # rows per program (good occupancy)
    BLOCK_D = 128          # column tile (covers C≤384 in ≤3 passes)
    grid = (triton.cdiv(total_rows, BLOCK_M),)

    layernorm_fp16[grid](
        x_flat,
        x_norm_h,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        C,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    # reshape back to [B, N, N, C] – keep as FP16 for the rest of the pipeline
    x_norm_h = x_norm_h.view(B, N, N, C)

    # ------------------------------------------------------------------
    # 2️⃣  Fused projection + gate (left & right)
    # ------------------------------------------------------------------
    # concatenate (proj, gate) weights for a single matmul per side
    left_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16, non_blocking=True)          # [2·H, C]

    right_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16, non_blocking=True)          # [2·H, C]

    rows = total_rows
    x_flat_h = x_norm_h.view(rows, C)                # FP16

    # Linear: (rows, C) @ (2·H, C).T → (rows, 2·H)
    left_concat = F.linear(x_flat_h, left_w)         # [rows, 2·H]
    right_concat = F.linear(x_flat_h, right_w)       # [rows, 2·H]

    # split into projection and pre‑gate tensors
    left_proj_h      = left_concat[:, :H]            # [rows, H]
    left_gate_pre_h  = left_concat[:, H:]            # [rows, H]
    right_proj_h     = right_concat[:, :H]
    right_gate_pre_h = right_concat[:, H:]

    # out‑gate (shared across the pair‑wise dimension)
    out_gate_h = torch.sigmoid(
        F.linear(
            x_flat_h,
            weights["out_gate.weight"].to(torch.float16, non_blocking=True)
        )
    )                                                # [rows, H]

    # ------------------------------------------------------------------
    # 3️⃣  Optional mask (broadcast over hidden_dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask_h = mask_tensor.view(rows, 1).to(torch.float16, non_blocking=True)
        left_proj_h = left_proj_h * mask_h
        right_proj_h = right_proj_h * mask_h

    # ------------------------------------------------------------------
    # 4️⃣  Apply the sigmoid gates
    # ------------------------------------------------------------------
    left_gate_h  = torch.sigmoid(left_gate_pre_h)
    right_gate_h = torch.sigmoid(right_gate_pre_h)

    left_h  = left_proj_h * left_gate_h        # [rows, H]
    right_h = right_proj_h * right_gate_h      # [rows, H]

    # ------------------------------------------------------------------
    # 5️⃣  Tri‑multiplicative update (batched GEMM over hidden dimension)
    # ------------------------------------------------------------------
    # reshape to [B, N, N, H] and permute so that H becomes part of batch
    left  = left_h.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right_h.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()

    BH = B * H
    left_mat  = left.view(BH, N, N)          # [BH, N, N]
    right_mat = right.view(BH, N, N)         # [BH, N, N]

    # GEMM: (N×N) · (N×N)^T → sum over k
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [BH, N, N]

    # reshape back -> [B, N, N, H]   (still FP16)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6️⃣  Second LayerNorm (over hidden dimension) – Triton kernel again
    # ------------------------------------------------------------------
    rows_h = B * N * N
    out_flat = out.view(rows_h, H).contiguous()
    out_norm_h = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(rows_h, BLOCK_M),)
    layernorm_fp16[grid2](
        out_flat,
        out_norm_h,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        rows_h,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    out_norm = out_norm_h.view(B, N, N, H)     # FP16

    # ------------------------------------------------------------------
    # 7️⃣  Apply output gate and final linear projection
    # ------------------------------------------------------------------
    gated = out_norm * out_gate_h.view(B, N, N, H)   # element‑wise FP16

    final_h = F.linear(
        gated.view(rows_h, H),
        weights["to_out.weight"].to(torch.float16, non_blocking=True)
    )                                                  # [rows_h, C] (FP16)

    final = final_h.view(B, N, N, C).to(torch.float32)   # back to FP32

    return final
```
