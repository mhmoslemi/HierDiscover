@triton.jit
def layernorm_fused_kernel(
    x_ptr, y_ptr, gamma_ptr, beta_ptr, total_rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr
):
    # compute pid row offset, get row indices
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # allocate shared memory to hold tile of values per row across dim
    # shared memory for each row for entire dim: shape (BLOCK_M, dim)
    # but we can't allocate with dynamic dim? dim is compile-time constant.
    # allocate portion of shape (BLOCK_M, dim) as float32.
    tile = tl.shared(tl.float32, (BLOCK_M, dim))

    # initialize sums
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    # first pass, load each block of dimensions, store into shared memory, accumulate sums
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        x_ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(x_ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0)
        # convert to fp32
        vals_f = vals.to(tl.float32)

        # store into shared memory
        tile[tl.arange(0, BLOCK_M), offs_d] = vals_f

        sum_val += tl.sum(vals_f, axis=1)
        sum_sq += tl.sum(vals_f * vals_f, axis=1)

    # compute norm params per row
    mean = sum_val / dim
    var = sum_sq / dim - mean * mean

```python
@triton.jit
def layernorm_fused_kernel(
    x_ptr, y_ptr, gamma_ptr, beta_ptr,
    total_rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # shared mem buffer for whole row
    tile = tl.shared(tl.float32, (BLOCK_M, dim))

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    # first loop: load, store, accumulate sums
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0)

        vals_f = vals.to(tl.float32)
        # store in shared memory
        tile[tl.arange(0, BLOCK_M), offs_d] = vals_f

        sum_val += tl.sum(vals_f, axis=1)
        sum_sq += tl.sum(vals_f * vals_f, axis=1)

    # compute normalization constants
    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # second loop: normalize, apply gamma/beta, store output
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        # load from shared memory
        vals_f = tile[tl.arange(0, BLOCK_M), offs_d]

        norm = (vals_f - mean[:, None]) * rstd[:, None]

        w = tl.load(gamma_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(beta_ptr + offs_d, mask=mask_d, other=0.0)

        out = norm * w[None, :] + b[None, :]
        out_h = out.to(tl.float16)

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_h,
                 mask=mask_m[:, None] & mask_d[None, :])

"""
Optimized TriMul “outgoing” kernel (AlphaFold‑3) using Triton for
row‑wise LayerNorm and PyTorch for the remaining dense operations.
The LayerNorm kernel reads each element once from global memory,
stores it in shared memory, computes mean/variance, and then
writes the normalized result, halving the memory traffic compared
to a naïve two‑pass implementation. The rest of the forward pass
uses FP16 linear layers, sigmoid gates and a batched GEMM (torch.bmm)
for the N×N×N multiplicative update. The final output is cast back
to FP32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel: fused two‑pass LayerNorm (read‑once, shared‑memory store)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fused_kernel(
    x_ptr,                     # input pointer (float32 or float16)  [total_rows, dim]
    y_ptr,                     # output pointer (float16)           [total_rows, dim]
    gamma_ptr,                 # gamma (scale) - float32 [dim]
    beta_ptr,                  # beta  (bias) - float32 [dim]
    total_rows,                # int: total number of rows
    dim,                       # int: number of columns (constexpr)
    eps,                       # float scalar epsilon
    BLOCK_M: tl.constexpr,     # rows per program
    BLOCK_D: tl.constexpr,     # columns per tile
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)           # [BLOCK_M]
    mask_m = offs_m < total_rows

    # shared memory buffer that holds a full row (BLOCK_M x dim)
    tile = tl.shared(tl.float32, (BLOCK_M, dim))

    # accumulator for mean/variance
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    # ------------------------------------------------------------------
    # 1️⃣ Load once, store in shared memory, accumulate sums
    # ------------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        # global memory pointer for this tile
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0)               # dtype matches x_ptr
        vals_f = vals.to(tl.float32)

        # write the tile to shared memory
        tile[tl.arange(0, BLOCK_M), offs_d] = vals_f

        # accumulate sums for mean & variance
        sum_val += tl.sum(vals_f, axis=1)
        sum_sq  += tl.sum(vals_f * vals_f, axis=1)

    # ------------------------------------------------------------------
    # 2️⃣ Compute mean / rsqrt(var+eps)
    # ------------------------------------------------------------------
    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)          # 1 / sqrt(var + eps)

    # ------------------------------------------------------------------
    # 3️⃣ Normalisation, affine transform, write result as FP16
    # ------------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        # read tile from shared memory
        vals_f = tile[tl.arange(0, BLOCK_M), offs_d]

        # (x - μ) / σ
        norm = (vals_f - mean[:, None]) * rstd[:, None]

        # γ, β (float32)
        w = tl.load(gamma_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(beta_ptr  + offs_d, mask=mask_d, other=0.0)

        out = norm * w[None, :] + b[None, :]            # FP32
        out_h = out.to(tl.float16)                     # cast to FP16

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_h,
                 mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
# Triton‑backed TriMul “outgoing” forward pass
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Performs the forward pass of the TriMul “outgoing” operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        input_tensor : [B, N, N, dim]     (float32)
        mask_tensor  : [B, N, N]          (float32, may be unused)
        weights_dict : model parameters
        config_dict  : {"dim": int, "hidden_dim": int, "nomask": bool (optional)}

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32 -> FP16)
    # --------------------------------------------------------------
    total_rows = B * N * N
    x_flat = input_tensor.view(total_rows, dim).contiguous()          # [rows, dim]

    x_norm = torch.empty((total_rows, dim), dtype=torch.float16, device=device)

    # launch parameters – tuned for H100
    BLOCK_M = 32                      # rows per program (fits in shared mem)
    BLOCK_D = 64                      # column tile size (covers dim ≤ 384)

    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fused_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # --------------------------------------------------------------
    # 2️⃣ Linear projections + gating (merged where possible)
    # --------------------------------------------------------------
    rows = total_rows

    # cast model weights once to FP16 for the linear layers
    left_proj_w   = weights["left_proj.weight"].to(torch.float16, non_blocking=True)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16, non_blocking=True)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16, non_blocking=True)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16, non_blocking=True)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16, non_blocking=True)
    to_out_w      = weights["to_out.weight"].to(torch.float16, non_blocking=True)

    # fuse projection & gate for left and right paths (dim → 2·hidden)
    left_concat_w  = torch.cat([left_proj_w, left_gate_w], dim=0)    # [2·H, dim]
    right_concat_w = torch.cat([right_proj_w, right_gate_w], dim=0)

    # shapes: (rows, dim) @ (2·H, dim).T -> (rows, 2·H)
    left_concat  = F.linear(x_norm.view(rows, dim), left_concat_w)   # [rows, 2·H]
    right_concat = F.linear(x_norm.view(rows, dim), right_concat_w)

    # split into projection and pre‑gate tensors
    left_proj_h      = left_concat[:, :hidden_dim]          # [rows, H]
    left_gate_pre_h  = left_concat[:, hidden_dim:]          # [rows, H]
    right_proj_h     = right_concat[:, :hidden_dim]
    right_gate_pre_h = right_concat[:, hidden_dim:]

    # out gate (shared across i/j positions)
    out_gate_h = torch.sigmoid(F.linear(x_norm.view(rows, dim), out_gate_w))

    # --------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over hidden dimension)
    # --------------------------------------------------------------
    if not nomask:
        mask_h = mask_tensor.view(rows, 1).to(torch.float16, non_blocking=True)
        left_proj_h = left_proj_h * mask_h
        right_proj_h = right_proj_h * mask_h

    # --------------------------------------------------------------
    # 4️⃣ Apply sigmoid gates
    # --------------------------------------------------------------
    left_gate_h  = torch.sigmoid(left_gate_pre_h)
    right_gate_h = torch.sigmoid(right_gate_pre_h)

    left_h  = left_proj_h * left_gate_h      # [rows, H]
    right_h = right_proj_h * right_gate_h    # [rows, H]

    # --------------------------------------------------------------
    # 5️⃣ Tri‑multiplicative update via batched GEMM
    # --------------------------------------------------------------
    # reshape to [B, N, N, H]
    left  = left_h.view(B, N, N, hidden_dim)
    right = right_h.view(B, N, N, hidden_dim)

    # move the hidden dimension into the batch axis → [B·H, N, N]
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # compute left @ rightᵀ for each hidden channel
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N]

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm (over hidden dimension)
    # --------------------------------------------------------------
    rows_h = B * N * N
    out_flat = out.view(rows_h, hidden_dim).contiguous()
    out_norm = torch.empty_like(out_flat, dtype=torch.float16, device=device)

    # reuse the same kernel (same BLOCK_M works for hidden_dim ≤ 384)
    grid2 = (triton.cdiv(rows_h, BLOCK_M),)
    layernorm_fused_kernel[grid2](
        out_flat,
        out_norm,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        rows_h,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    out_norm = out_norm.view(B, N, N, hidden_dim)   # FP16

    # --------------------------------------------------------------
    # 7️⃣ Apply output gate and final projection
    # --------------------------------------------------------------
    gated = out_norm * out_gate_h.view(B, N, N, hidden_dim)   # element‑wise FP16
    final_half = F.linear(gated.view(rows_h, hidden_dim), to_out_w)  # [rows_h, dim] FP16

    # Cast back to FP32 for API compatibility
    final = final_half.view(B, N, N, dim).to(torch.float32)

    return final

"""
Optimized TriMul “outgoing” forward pass (AlphaFold‑3).

Algorithm
---------
1️⃣ Row‑wise LayerNorm (FP32 → FP16) is performed by a custom Triton
   kernel that reads each element **once** from global memory,
   stores it in shared memory, computes mean/variance, then writes
   the normalized values back to global memory.  This halves the
   memory traffic compared with a naïve two‑pass implementation.

2️⃣ Projection + gating weights are fused (dim → 2·hidden) so that only
   two linear layers are required for the left/right paths.

3️⃣ An optional pair‑wise mask is applied (broadcast over the hidden
   dimension).

4️⃣ Sigmoid gates are applied to the projected tensors.

5️⃣ The core multiplicative update
   out[i,j,h] = Σ_k left[i,k,h]·right[j,k,h]
   is realised as a batched GEMM (torch.bmm) over the hidden
   dimension:   (B·H, N, N) @ (B·H, N, N)ᵀ.

6️⃣ A second row‑wise LayerNorm (over the hidden dimension) is again
   executed by the same Triton kernel.

7️⃣ The result is multiplied by the output gate and finally projected
   back to the original feature dimension.  The output is cast to
   float32 to match the reference implementation.

Only the two LayerNorm steps are written in Triton; the dense
operations use highly‑optimised FP16 PyTorch kernels.  The code
supports the full test matrix (B ∈ {1,2}, N ≤ 1024, dim ∈ {128,384},
hidden_dim = 128) on an H100 GPU.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: fused, single‑load row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fused_kernel(
    x_ptr,                     # input pointer  [total_rows, dim]   (float32 or float16)
    y_ptr,                     # output pointer [total_rows, dim]   (float16)
    gamma_ptr,                 # scale (γ)      [dim]                (float32)
    beta_ptr,                  # bias  (β)      [dim]                (float32)
    total_rows,                # total number of rows (runtime scalar)
    dim: tl.constexpr,         # number of columns (compile‑time constant)
    eps: tl.constexpr,         # epsilon for variance stability
    BLOCK_M: tl.constexpr,     # rows processed per program
    BLOCK_D: tl.constexpr,     # tile size on the column axis
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)                # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # Shared memory buffer that holds a full row (BLOCK_M × dim)
    # ------------------------------------------------------------------
    tile = tl.shared(tl.float32, (BLOCK_M, dim))

    # ------------------------------------------------------------------
    # Accumulators for mean / variance
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    # ------------------------------------------------------------------
    # 1️⃣ Load once, store to shared memory, accumulate sums
    # ------------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)          # [BLOCK_D]
        mask_d = offs_d < dim

        # global‑memory load
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        )
        vals_f = vals.to(tl.float32)                         # fp32 accumulation

        # write tile to shared memory (so we only read from global once)
        tile[tl.arange(0, BLOCK_M), offs_d] = vals_f

        # update mean / variance accumulators
        sum_val += tl.sum(vals_f, axis=1)
        sum_sq  += tl.sum(vals_f * vals_f, axis=1)

    # ------------------------------------------------------------------
    # 2️⃣ Compute μ, σ
    # ------------------------------------------------------------------
    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)               # 1 / sqrt(var + eps)

    # ------------------------------------------------------------------
    # 3️⃣ Normalise, apply affine (γ·x̂ + β) and write as FP16
    # ------------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        vals_f = tile[tl.arange(0, BLOCK_M), offs_d]          # [BLOCK_M, BLOCK_D]

        # (x - μ) / σ
        norm = (vals_f - mean[:, None]) * rstd[:, None]

        # γ, β are stored as float32
        w = tl.load(gamma_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(beta_ptr  + offs_d, mask=mask_d, other=0.0)

        out = norm * w[None, :] + b[None, :]                 # FP32
        out_h = out.to(tl.float16)                           # cast to FP16

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(
            out_ptr,
            out_h,
            mask=mask_m[:, None] & mask_d[None, :],
        )

    x_norm = torch.empty((total_rows, dim), dtype=torch.float16, device=device)

    # Triton launch parameters (tuned for H100)
    BLOCK_M = 32                     # rows per program – fits in shared memory
    BLOCK_D = 64                     # column tile size (covers dim ≤ 384)

    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fused_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim=dim,
        eps=eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    rows = total_rows

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections + gates (weights cast once to FP16)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16, non_blocking=True)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16, non_blocking=True)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16, non_blocking=True)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16, non_blocking=True)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16, non_blocking=True)
    to_out_w      = weights["to_out.weight"].to(torch.float16, non_blocking=True)

    # fuse (proj + gate) for the left and right streams
    left_concat_w  = torch.cat([left_proj_w, left_gate_w], dim=0)     # [2·H, dim]
    right_concat_w = torch.cat([right_proj_w, right_gate_w], dim=0)

    # (rows, dim) @ (2·H, dim).T → (rows, 2·H)
    left_concat  = F.linear(x_norm.view(rows, dim), left_concat_w)   # [rows, 2·H]
    right_concat = F.linear(x_norm.view(rows, dim), right_concat_w)

    # split into projection and pre‑gate tensors
    left_proj_h      = left_concat[:, :hidden_dim]          # [rows, H]
    left_gate_pre_h  = left_concat[:, hidden_dim:]          # [rows, H]
    right_proj_h     = right_concat[:, :hidden_dim]
    right_gate_pre_h = right_concat[:, hidden_dim:]

    # out‑gate (shared across i/j)
    out_gate_h = torch.sigmoid(F.linear(x_norm.view(rows, dim), out_gate_w))  # [rows, H]

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over hidden_dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask_h = mask_tensor.view(rows, 1).to(torch.float16, non_blocking=True)
        left_proj_h = left_proj_h * mask_h
        right_proj_h = right_proj_h * mask_h

    # ------------------------------------------------------------------
    # 4️⃣ Apply sigmoid gates
    # ------------------------------------------------------------------
    left_gate_h  = torch.sigmoid(left_gate_pre_h)
    right_gate_h = torch.sigmoid(right_gate_pre_h)

    left_h  = left_proj_h * left_gate_h      # [rows, H]
    right_h = right_proj_h * right_gate_h    # [rows, H]

    # ------------------------------------------------------------------
    # 5️⃣ Tri‑multiplicative update via batched GEMM
    # ------------------------------------------------------------------
    left  = left_h.view(B, N, N, hidden_dim)     # [B, N, N, H]
    right = right_h.view(B, N, N, hidden_dim)

    # move hidden into batch dimension → [B·H, N, N]
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # GEMM: (N×N) @ (N×N)ᵀ  →  sum over k
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N]

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm (over hidden_dim)
    # ------------------------------------------------------------------
    rows_h = B * N * N
    out_flat = out.view(rows_h, hidden_dim).contiguous()
    out_norm = torch.empty_like(out_flat, dtype=torch.float16, device=device)

    grid2 = (triton.cdiv(rows_h, BLOCK_M),)
    layernorm_fused_kernel[grid2](
        out_flat,
        out_norm,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        rows_h,
        dim=hidden_dim,
        eps=eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    out_norm = out_norm.view(B, N, N, hidden_dim)   # FP16

    # ------------------------------------------------------------------
    # 7️⃣ Output gate + final projection
    # ------------------------------------------------------------------
    gated = out_norm * out_gate_h.view(B, N, N, hidden_dim)   # element‑wise FP16

    final_half = F.linear(gated.view(rows_h, hidden_dim), to_out_w)  # [rows_h, dim]

    # Cast back to FP32 to match the reference API
    final = final_half.view(B, N, N, dim).to(torch.float32)

    return final
```
