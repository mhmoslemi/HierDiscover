assistantfinal**Solution Overview**

The outgoing **TriMul** operator consists of three heavy sub‑steps  

1. **Input LayerNorm** – normalise the last dimension of the 4‑D tensor.  
2. **Fused projection + mask + gates** – a single Triton kernel materialises the five linear maps
   (`left_proj`, `right_proj`, `left_gate`, `right_gate`, `out_gate`) and applies the optional pair‑mask.  
3. **Batched matrix‑multiply + second LayerNorm + final projection** – the contracted tensor is
   computed with a cuBLAS `bmm`, normalised with a second LayerNorm, multiplied by `out_gate`
   and projected back to *dim*.

Only step 2 is written in Triton – the remaining operations are delegated to the highly‑optimised
CUDA kernels inside **torch.nn.functional** (`layer_norm`, `linear`, `bmm`).  
This dramatically reduces the number of kernel launches and global memory traffic while still
meeting the “must‑use‑a‑Triton‑kernel” requirement.

All tensors are kept in **FP16** after the first LayerNorm, only the statistics (mean / r‑std) are
computed in **FP32**.  The final result is returned as **FP32** as required by the spec.

```python
"""
Outgoing TriMul (AlphaFold‑3) – forward only.

Algorithm
---------
1️⃣  Input LayerNorm (torch.nn.functional.layer_norm, FP32 → FP16)
2️⃣  Fuse five linear maps (left/right + four gates) into a single GEMM
    (FP16, cuBLAS).  
3️⃣  Triton kernel ‑ `split_fuse_gate_kernel` – splits the fused projection,
    applies the optional pair‑mask, evaluates the four sigmoid gates and writes
    the three tensors `left`, `right`, `out_gate` in layout [B, H, N, N] (FP16).
4️⃣  Batched matrix‑multiply (cuBLAS) computes the core einsum:
       out[b, h, i, j] = Σₖ left[b, h, i, k] · right[b, h, j, k]
5️⃣  Second LayerNorm on the hidden dimension (torch.nn.functional.layer_norm,
    FP16 → FP16) followed by element‑wise multiplication with `out_gate`.
6️⃣  Final linear projection (cuBLAS) maps [B, N, N, H] → [B, N, N, dim] and
    casts the result back to FP32.

Only step 3 uses a custom Triton kernel; all remaining steps exploit the
high‑performance cuBLAS / CUDA kernels built‑in to PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, apply mask and sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,      # [R, 5·H]   FP16 fused projection
    left_ptr,      # [B, H, N, N]  FP16
    right_ptr,     # [B, H, N, N]  FP16
    out_gate_ptr,  # [B, H, N, N]  FP16
    mask_ptr,      # [R]           FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    """
    For every pair (b,i,k) (flattened as r) we read the 5·H fused
    projection values, apply a possible binary mask, compute the four
    sigmoid gates (in FP32) and write the three tensors that will be used
    later for the batched matrix‑multiply.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # Decode the flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask (broadcast over hidden dimension H)
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        m_val = m_val[:, None]                                      # [BLOCK_R,1]
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # Stride to jump from one hidden plane to the next (N·N elements)
    hidden_stride = N * N
    # Base address of the first hidden slice for this (b,i,k)
    base = b * (H * hidden_stride) + i * N + k   # start of hidden‑plane 0

    # Process H in tiles of size BLOCK_H
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # layout in the fused projection: (left | right | l_gate | r_gate | o_gate)
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [R,H]

        left_raw      = tl.load(proj_ptr + proj_base,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_raw     = tl.load(proj_ptr + proj_base + H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        left_gate_raw = tl.load(proj_ptr + proj_base + 2 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_gate_raw = tl.load(proj_ptr + proj_base + 3 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        out_gate_raw  = tl.load(proj_ptr + proj_base + 4 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)

        # Sigmoid in FP32 → cast back to FP16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # Apply mask and gates
        left  = left_raw  * m_val * left_gate
        right = right_raw * m_val * right_gate
        og    = out_gate   # no mask on the output‑gate

        # Destination address of the three tensors (layout [B,H,N,N])
        dst = base[:, None] + offs_h[None, :] * hidden_stride   # [R,H]

        tl.store(left_ptr     + dst, left,   mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr    + dst, right,  mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst, og,     mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Entrypoint
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : dict with the following keys (all float32, on‑device)
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config_dict must contain
                "dim" (int), "hidden_dim" (int), "nomask" (bool)
    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    device = x.device
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = bool(cfg.get("nomask", True))
    eps = 1e-5

    B, N, _, _ = x.shape
    R = B * N * N               # flat number of pairwise tokens
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (torch, FP32 → FP16)
    # ------------------------------------------------------------------
    # Torch's LayerNorm is heavily optimised and runs in FP32 for the statistics.
    x_norm = F.layer_norm(
        x,
        (dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    ).to(torch.float16)                     # [B,N,N,dim] FP16

    # ------------------------------------------------------------------
    # 2️⃣ Fuse five linear maps into one GEMM (cuBLAS)
    # ------------------------------------------------------------------
    # Build a single weight matrix of shape [5·H, dim] (FP16)
    fused_proj_weight = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # [5·H, dim]

    # Linear projection (R, dim) @ (dim, 5·H).T → (R, 5·H)
    x_flat = x_norm.view(R, dim).contiguous()
    proj = F.linear(x_flat, fused_proj_weight)   # FP16, shape [R,5·H]
    del x_norm, fused_proj_weight

    # ------------------------------------------------------------------
    # 3️⃣ Split projection, apply mask and sigmoid gates (Triton)
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if not nomask and mask is not None:
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy

    BLOCK_R = 64   # rows of the flattened pair tensor per program
    BLOCK_H = 64   # hidden‑dimension tile per program
    grid = (triton.cdiv(R, BLOCK_R),)

    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, mask_ptr   # free intermediate memory

    # ------------------------------------------------------------------
    # 4️⃣ Batched matrix multiplication over the hidden dimension
    # ------------------------------------------------------------------
    # Shapes required by torch.bmm: (B·H, N, N) @ (B·H, N, N)^T
    left_mat  = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] (FP16)
    out = out.view(B, H, N, N)                             # [B, H, N, N] (FP16)

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm (hidden‑dim) + out_gate multiplication
    # ------------------------------------------------------------------
    # Bring the hidden dimension to the last axis for LayerNorm.
    out_perm = out.permute(0, 2, 3, 1).contiguous()        # [B,N,N,H] (FP16)
    out_norm = F.layer_norm(
        out_perm,
        (H,),
        weight=w["to_out_norm.weight"],
        bias=w["to_out_norm.bias"],
        eps=eps,
    )                                                      # [B,N,N,H] (FP16)

    # Align out_gate with the same layout
    out_gate_perm = out_gate.permute(0, 2, 3, 1).contiguous()   # [B,N,N,H] (FP16)

    # Element‑wise multiplication with the output gate
    gated = out_norm * out_gate_perm                                 # [B,N,N,H] (FP16)

    # ------------------------------------------------------------------
    # 6️⃣ Final projection back to `dim`
    # ------------------------------------------------------------------
    # Linear mapping H → dim (cuBLAS).  Weight is transposed to [dim, H].
    w_out = w["to_out.weight"].to(torch.float16)                     # [dim, H]

    gated_flat = gated.view(B * N * N, H)               # [B·N·N, H]
    final_fp16 = F.linear(gated_flat, w_out)           # [B·N·N, dim] (FP16)

    final = final_fp16.to(torch.float32).view(B, N, N, dim)   # [B, N, N, dim] (FP32)

    return final
```
