import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
@triton.jit
def layernorm_kernel(
    x_ptr, y_ptr, w_ptr, b_ptr, rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    # first pass: mean & variance
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # second pass: write normalized values
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])

# 2️⃣ Split fused projection, apply optional mask and sigmoid gates
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,  # [R,5*H]  FP16
    left_ptr,  # [B,H,N,N] FP16
    right_ptr,
    out_gate_ptr,
    mask_ptr,  # [R] FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < R

    # decode (b,i,k) from flat index r
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask broadcast over hidden dim
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)[:, None]  # [BLOCK_R,1]
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N
    # base offset for hidden index=0 in the B,H,N,N layout
    base = b * (H * hidden_stride) + i * N + k   # shape [BLOCK_R]

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # layout of proj: (R, 5*H) row‑major
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]               # [R,H]

        left_raw      = tl.load(proj_ptr + proj_base,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_raw     = tl.load(proj_ptr + proj_base + H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        left_gate_raw = tl.load(proj_ptr + proj_base + 2 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_gate_raw = tl.load(proj_ptr + proj_base + 3 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        out_gate_raw = tl.load(proj_ptr + proj_base + 4 * H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)

        # sigmoid in FP32, cast back
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * m_val * left_gate
        right = right_raw * m_val * right_gate
        og    = out_gate

        # destination pointer: each hidden plane strides by hidden_stride
        dst = base[:, None] + offs_h[None, :] * hidden_stride   # [R,H]

        tl.store(left_ptr + dst, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst, right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst, og,
                 mask=mask_r[:, None] & mask_h[None, :])

# 3️⃣ Row‑wise stats (mean & rsqrt‑var) over hidden dim
@triton.jit
def rowwise_stats_kernel(
    out_ptr,   # [B,H,N,N] FP16
    mu_ptr,    # [R] FP32
    rstd_ptr,  # [R] FP32
    total_rows, H, N,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + j   # start of hidden plane 0

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        ptr = out_ptr + base[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr + offs_m, mean, mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)

# 4️⃣ Apply second LayerNorm, out‑gate and store flattened (R,H) tensor
@triton.jit
def apply_norm_gate_kernel(
    out_ptr,         # [B,H,N,N] FP16
    gate_ptr,        # [B,H,N,N] FP16 (out_gate)
    mu_ptr,          # [R] FP32
    rstd_ptr,        # [R] FP32
    ln_w_ptr,        # [H] FP32
    ln_b_ptr,        # [H] FP32
    dst_ptr,         # [R,H] FP16
    total_rows, H, N,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # decode (b,i,j) -> row index
    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + j   # start of hidden index 0

    # load per‑row statistics once
    mu_val   = tl.load(mu_ptr   + offs_m, mask=mask_m, other=0.0)
    rstd_val = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)
    mu_tile   = mu_val[:, None]   # [M,1]
    rstd_tile = rstd_val[:, None] # [M,1]

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        # load tensors for this hidden tile
        ptr_out  = out_ptr  + base[:, None] + offs_k[None, :] * hidden_stride
        ptr_gate = gate_ptr + base[:, None] + offs_k[None, :] * hidden_stride

        vals_out  = tl.load(ptr_out,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)   # [M,K]
        vals_gate = tl.load(ptr_gate,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)   # [M,K]

        # LayerNorm parameters for this hidden tile
        ln_w = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)  # [K] FP32
        ln_b = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)  # [K]

        # ((out - μ) * rstd) * γ + β
        norm = (vals_out - mu_tile) * rstd_tile
        norm = norm * ln_w[None, :] + ln_b[None, :]

        # multiply by out‑gate
        fused = norm * vals_gate

        # write flattened (R,H) FP16
        dst = offs_m[:, None] * H + offs_k[None, :]   # [M,K]
        tl.store(dst_ptr + dst,
                 fused.to(tl.float16),
                 mask=mask_m[:, None] & mask_k[None, :])

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : dict with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
              (all float32, already on the target device)
            - config_dict must contain
                "dim" (int), "hidden_dim" (int), "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    device = x.device
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = bool(cfg.get("nomask", True))
    eps = 1e-5

    B, N, _, _ = x.shape
    R = B * N * N
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm over the pair dimension (row‑wise)
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # ------------------------------------------------------------------
    # 2️⃣ Fuse five linear maps (left, right, left_gate, right_gate, out_gate)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)  # [5*H, dim]

    proj = F.linear(x_norm, fused_w)   # (R,5*H) FP16
    del fused_w, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ Split projection, apply mask and sigmoid gates
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if not nomask and mask is not None:
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM over hidden dimension (cuBLAS)
    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)               # [B*H, N, N] FP16
    right_mat = right.view(B * H, N, N)              # [B*H, N, N] FP16

    out = torch.bmm(left_mat, right_mat.transpose(1, 2))
    out = out.view(B, H, N, N)                      # [B, H, N, N] FP16

    # ------------------------------------------------------------------
    # 5️⃣ Row‑wise mean & rsqrt‑variance of the contracted tensor
    # ------------------------------------------------------------------
    mu   = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 256
    BLOCK_K_STATS = 128      # H ≤ 128 → fits in one tile
    grid_stats = (triton.cdiv(R, BLOCK_M_STATS),)
    rowwise_stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Fuse second LayerNorm + out_gate → (R,H) tensor
    # ------------------------------------------------------------------
    out_norm_gate = torch.empty(R, H, dtype=torch.float16, device=device)

    BLOCK_M_OUT = 128
    BLOCK_K_OUT = 32   # hidden tiles
    grid_outnorm = (triton.cdiv(R, BLOCK_M_OUT),)
    apply_norm_gate_kernel[grid_outnorm](
        out,
        out_gate,
        mu,
        rstd,
        w["to_out_norm.weight"],
        w["to_out_norm.bias"],
        out_norm_gate,
        R,
        H,
        N,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        num_warps=8,
    )
    # clean up intermediate tensors we no longer need
    del out, out_gate, mu, rstd

    # ------------------------------------------------------------------
    # 7️⃣ Final linear projection (cuBLAS). Output must be FP32.
    # ------------------------------------------------------------------
    w_out = w["to_out.weight"].to(torch.float16)   # [dim, H]
    final_fp16 = torch.nn.functional.linear(out_norm_gate, w_out)  # (R, dim) FP16
    final = final_fp16.to(torch.float32).view(B, N, N, dim)

    return final

"""
Outgoing TriMul (AlphaFold‑3) – forward only, mixed‑precision, Triton‑fused.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the flattened pair tensor (FP32 → FP16).
2️⃣ Fuse the five linear maps (left, right, left_gate, right_gate, out_gate)
   into a single GEMM (cuBLAS) → tensor (R, 5·H) where R = B·N·N.
3️⃣ Triton kernel splits the fused projection, applies an optional pair‑mask,
   evaluates the sigmoid gates and writes three tensors `left`, `right`,
   `out_gate` in layout [B, H, N, N] (FP16).
4️⃣ Batched GEMM (cuBLAS) computes the core einsum:
       out[b, h, i, j] = Σₖ left[b, h, i, k]·right[b, h, j, k]
   → shape [B, H, N, N] (FP16).
5️⃣ Row‑wise mean & rsqrt‑variance of `out` (H‑dim) → μ and rstd per (b,i,j).
6️⃣ Triton kernel fuses the second LayerNorm, the `out_gate` multiplication
   and stores a flattened (R, H) tensor in FP16.
7️⃣ Final projection uses a single cuBLAS GEMM:
       (R, H) @ (H, dim) → (R, dim)  (FP16 → FP32)

All heavy work is performed in FP16; statistics are carried out in FP32.
The implementation requires at least one Triton kernel (several are used)
and runs on H100 (Triton 3.3.1).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,  # [R, dim] FP32 input
    y_ptr,  # [R, dim] FP16 output
    w_ptr,  # [dim]    γ (FP32)
    b_ptr,  # [dim]    β (FP32)
    rows,   # scalar   R
    dim,    # scalar   dim
    eps,    # scalar   epsilon
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ---- first pass: mean & variance ------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]   # [M,D]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- second pass: write normalized values ---------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, apply mask and sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,      # [R,5*H]   FP16 fused projection
    left_ptr,      # [B,H,N,N] FP16
    right_ptr,     # [B,H,N,N] FP16
    out_gate_ptr,  # [B,H,N,N] FP16
    mask_ptr,      # [R]       FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)           # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask (broadcast over H)
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)[:, None]   # [BLOCK_R,1]
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N                # stride of one hidden slice
    base = b * (H * hidden_stride) + i * N + k   # start of hidden‑plane 0

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [R,H]

        left_raw      = tl.load(proj_ptr + proj_base,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_raw     = tl.load(proj_ptr + proj_base + H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        left_gate_raw = tl.load(proj_ptr + proj_base + 2 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_gate_raw = tl.load(proj_ptr + proj_base + 3 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        out_gate_raw = tl.load(proj_ptr + proj_base + 4 * H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)

        # sigmoid in FP32 → back to FP16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * m_val * left_gate
        right = right_raw * m_val * right_gate
        og    = out_gate

        dst = base[:, None] + offs_h[None, :] * hidden_stride   # [R,H]

        tl.store(left_ptr     + dst, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr    + dst, right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst, og,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 3️⃣ Row‑wise mean & rsqrt‑variance of the contracted tensor
# ----------------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr,   # [B,H,N,N]  FP16
    mu_ptr,    # [R]        FP32
    rstd_ptr,  # [R]        FP32
    total_rows, H, N,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + j   # start of hidden plane 0

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        ptr = out_ptr + base[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_m, mean,  mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)


# ----------------------------------------------------------------------
# 4️⃣ Fuse second LayerNorm and out‑gate → flattened (R,H) tensor
# ----------------------------------------------------------------------
@triton.jit
def apply_norm_gate_kernel(
    out_ptr,        # [B,H,N,N] FP16
    gate_ptr,       # [B,H,N,N] FP16 (out_gate)
    mu_ptr,         # [R]       FP32
    rstd_ptr,       # [R]       FP32
    ln_w_ptr,       # [H]       FP32 (to_out_norm.weight)
    ln_b_ptr,       # [H]       FP32 (to_out_norm.bias)
    dst_ptr,        # [R,H]     FP16
    total_rows, H, N,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # decode (b,i,j) → flat row index
    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + j   # start of hidden‑plane 0

    # per‑row statistics (load once)
    mu   = tl.load(mu_ptr   + offs_m, mask=mask_m, other=0.0)   # [M]
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)   # [M]
    mu_tile   = mu[:,   None]      # [M,1]
    rstd_tile = rstd[:, None]      # [M,1]

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        # load tensors for this hidden tile
        ptr_out  = out_ptr  + base[:, None] + offs_k[None, :] * hidden_stride
        ptr_gate = gate_ptr + base[:, None] + offs_k[None, :] * hidden_stride

        vals_out  = tl.load(ptr_out,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)   # [M,K]
        vals_gate = tl.load(ptr_gate,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)   # [M,K]

        # LayerNorm γ,β for this hidden tile
        ln_w = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [K] FP32
        ln_b = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [K] FP32

        # ((out - μ) * rstd) * γ + β
        norm = (vals_out - mu_tile) * rstd_tile
        norm = norm * ln_w[None, :] + ln_b[None, :]

        # multiply by out‑gate
        fused = norm * vals_gate

        # write flattened (R,H) FP16
        dst_off = offs_m[:, None] * H + offs_k[None, :]               # [M,K]
        tl.store(dst_ptr + dst_off,
                 fused.to(tl.float16),
                 mask=mask_m[:, None] & mask_k[None, :])

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : dict with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
              (all float32, already on the target device)
            - config_dict must contain
                "dim" (int), "hidden_dim" (int), "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    device = x.device
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = bool(cfg.get("nomask", True))
    eps = 1e-5

    B, N, _, _ = x.shape
    R = B * N * N               # number of pairwise tokens
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                 # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # ------------------------------------------------------------------
    # 2️⃣ Fuse five linear maps (left, right, left_gate, right_gate, out_gate)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                 # [5*H, dim]

    proj = F.linear(x_norm, fused_w)                     # (R,5*H) FP16
    del fused_w, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ Split projection, apply mask and sigmoid gates
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if not nomask and mask is not None:
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension (cuBLAS)
    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)               # [B*H, N, N] FP16
    right_mat = right.view(B * H, N, N)              # [B*H, N, N] FP16

    out = torch.bmm(left_mat, right_mat.transpose(1, 2))
    out = out.view(B, H, N, N)                       # [B, H, N, N] FP16

    # ------------------------------------------------------------------
    # 5️⃣ Row‑wise mean & rsqrt‑variance of `out`
    # ------------------------------------------------------------------
    mu   = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 256
    BLOCK_K_STATS = 128                # H ≤ 128 fits in one tile
    grid_stats = (triton.cdiv(R, BLOCK_M_STATS),)
    rowwise_stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Fuse second LayerNorm + out_gate → (R,H) tensor (FP16)
    # ------------------------------------------------------------------
    out_norm_gate = torch.empty(R, H, dtype=torch.float16, device=device)

    BLOCK_M_OUT = 128
    BLOCK_K_OUT = 32      # hidden‑dim tiles
    grid_norm = (triton.cdiv(R, BLOCK_M_OUT),)
    apply_norm_gate_kernel[grid_norm](
        out,
        out_gate,
        mu,
        rstd,
        w["to_out_norm.weight"],   # γ (FP32)
        w["to_out_norm.bias"],     # β (FP32)
        out_norm_gate,
        R,
        H,
        N,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        num_warps=8,
    )
    # release intermediate buffers
    del out, out_gate, mu, rstd

    # ------------------------------------------------------------------
    # 7️⃣ Final projection (cuBLAS) and cast back to FP32
    # ------------------------------------------------------------------
    w_out = w["to_out.weight"].to(torch.float16)   # [dim, H]
    final_fp16 = torch.nn.functional.linear(out_norm_gate, w_out)   # (R, dim) FP16
    final = final_fp16.to(torch.float32).view(B, N, N, dim)

    return final
