"""
TriMul (outgoing) – fast forward‑only implementation.

Algorithm
---------
1️⃣  Row‑wise LayerNorm on the 4‑D input (FP32→FP16) – pure Triton.
2️⃣  Fuse the five linear projections (left, right, left‑gate, right‑gate,
    out‑gate) into one matmul (`torch.nn.functional.linear`) on the
    normalized tensor.
3️⃣  Decode the fused projection, optionally mask it, apply the
    sigmoid gates and write three tensors `left`, `right` and `out_gate`
    (layout [B, H, N, N], FP16).  All of this is done inside a single
    Triton kernel (no Python loops).
4️⃣  Batched GEMM (`torch.bmm`) over the sequence dimension N:
    out[b, h, i, j] = Σₖ left[b, h, i, k] * right[b, h, j, k]   (FP16)
5️⃣  Fuse the second LayerNorm, the element‑wise multiplication by the
    `out_gate` tensor, and the final linear projection back to the model
    dimension `dim` in a single Triton kernel.  The kernel computes the
    per‑row mean / rsqrt‑variance on‑the‑fly, normalises, gates, and
    performs the H×dim matrix multiplication in a tiled fashion.
    The result is a FP32 tensor of shape [B, N, N, dim].

The implementation works for both masked / un‑masked inputs and for the
full set of test configurations listed in the problem statement.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,   # [R, C]   input FP32
    y_ptr,   # [R, C]   output FP16
    w_ptr,   # [C]      gamma (FP32)
    b_ptr,   # [C]      beta  (FP32)
    rows,    # R = B·N·N
    dim,     # C
    eps,     # epsilon
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ------------------------------------------------------------------
    # first pass: compute sum and sum‑of‑squares
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                                   # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                     # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)              # [BLOCK_M]
    # ------------------------------------------------------------------
    # compute mean / rstd
    # ------------------------------------------------------------------
    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------------------------------------------------------
    # second pass: write normalized values
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]              # FP32
        tl.store(
            y_ptr + offs_m[:, None] * dim + offs_d[None, :],
            out.to(tl.float16),
            mask=mask_m[:, None] & mask_d[None, :],
        )


# ----------------------------------------------------------------------
# 2️⃣ Decode fused projection, apply mask and sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,       # [R, 5·H]  FP16 fused projection
    left_ptr,       # [B, H, N, N] FP16
    right_ptr,      # [B, H, N, N] FP16
    out_gate_ptr,   # [B, H, N, N] FP16
    mask_ptr,       # [R] FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index r → (b, i, k)
    NN = N * N
    b = offs_r // NN
    rem = offs_r - b * NN
    i = rem // N
    k = rem - i * N

    # optional mask broadcast over hidden dimension
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = m_val[:, None]                                   # [BLOCK_R, 1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N               # distance between two hidden planes
    # start offset of (b, *, i, k) inside the 4‑D tensors
    base_offset = b * (H * hidden_stride) + i * N + k

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # layout inside proj: [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # sigmoid in fp32 → back to fp16
        left_gate = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate   # never masked

        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst_offset,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset,
                 outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 5️⃣ Fuse second LayerNorm, out_gate multiplication & final projection
# ----------------------------------------------------------------------
@triton.jit
def final_output_kernel(
    out_ptr,        # [B, H, N, N] FP16 (post‑GEMM)
    gate_ptr,       # [B, H, N, N] FP16 (out_gate)
    ln_w_ptr,       # [H] FP32 (to_out_norm.weight)
    ln_b_ptr,       # [H] FP32 (to_out_norm.bias)
    w_out_T_ptr,    # [H, dim] FP16 (to_out weight transposed)
    out_final_ptr,  # [B, N, N, dim] FP32
    total_rows, H, N, dim,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr,
):
    """
    Tile‑wise kernel that:
        * computes per‑row mean / rstd over the hidden dimension H,
        * normalises, multiplies by the out‑gate,
        * applies the final linear projection (H → dim) and writes FP32 result.
    """
    pid_m = tl.program_id(0)   # rows (R = B·N·N)
    pid_n = tl.program_id(1)   # output‑dim

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)   # [BLOCK_M]
    offs_n = col_start + tl.arange(0, BLOCK_N)   # [BLOCK_N]
    mask_m = offs_m < total_rows
    mask_n = offs_n < dim

    NN = N * N
    b = offs_m // NN
    rem = offs_m - b * NN
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + j   # start of hidden slice 0

    # load per‑row statistics
    # (these are computed on‑the‑fly inside the kernel)
    # ------------------------------------------------------------------
    # first pass: compute sum & sum‑sq over H (same pattern as layernorm)
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)   # [BLOCK_K]
        mask_k = offs_k < H

        ptr = out_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)                             # [M, K]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mu   = sum_val / H
    var  = sum_sq / H - mu * mu
    rstd = tl.rsqrt(var + eps)

    # ------------------------------------------------------------------
    # second pass: normalise, gate and project
    # ------------------------------------------------------------------
    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)               # [K]
        mask_k = offs_k < H

        # load hidden activation and its gate
        ptr_out  = out_ptr  + base_offset[:, None] + offs_k[None, :] * hidden_stride
        ptr_gate = gate_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride

        val_out = tl.load(
            ptr_out,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)   # [M, K]

        val_gate = tl.load(
            ptr_gate,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)   # [M, K]

        # LayerNorm parameters (γ, β) for this hidden slice
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [K]
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [K]

        # (out - μ) * rstd  →  *γ + β   →  *gate
        norm   = ((val_out - mu[:, None]) * rstd[:, None]) * gamma[None, :] + beta[None, :]
        gated  = norm * val_gate                                   # [M, K]

        # convert to fp16 for the dot with the final weight matrix
        gated_fp16 = gated.to(tl.float16)

        # load a tile of the final projection weight (H × dim)
        w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]   # [K, TILE_N]
        w_tile = tl.load(
            w_ptr,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )                                                             # [K, TILE_N]

        # accumulate
        acc += tl.dot(gated_fp16, w_tile).to(tl.float32)   # (M,K)·(K,N) → (M,N)

    # write final output (row‑major layout [B, N, N, dim])
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only Triton implementation of the *outgoing* TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, C] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
        - weights_dict : mapping with keys
              "norm.weight", "norm.bias",
              "left_proj.weight", "right_proj.weight",
              "left_gate.weight", "right_gate.weight", "out_gate.weight",
              "to_out_norm.weight", "to_out_norm.bias",
              "to_out.weight"
        - config_dict  : {"dim": C, "hidden_dim": H, "nomask": bool, ...}

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]                # model dimension C
    hidden_dim = cfg["hidden_dim"]  # hidden dimension H
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, N2, _ = x.shape
    assert N == N2, "TriMul expects a square pairwise matrix"
    R = B * N * N            # total number of (b,i,k) rows
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ layernorm (row‑wise) → FP16
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                     # [R, C] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 256
    BLOCK_D = 64
    layernorm_kernel[(triton.cdiv(R, BLOCK_M),)](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # ------------------------------------------------------------------
    # 2️⃣ fused projection (5·H outputs)
    # ------------------------------------------------------------------
    # concatenate the five weight matrices once (FP16)
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                         # [5·H, C]
    proj = F.linear(x_norm, fused_w)                             # [R, 5·H] FP16
    del fused_w, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ split, mask and sigmoid gates → left, right, out_gate
    # ------------------------------------------------------------------
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 128
    BLOCK_H = 64
    split_fuse_gate_kernel[(triton.cdiv(R, BLOCK_R),)](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM over the sequence dimension (FP16)
    # ------------------------------------------------------------------
    # left & right are [B, H, N, N]; we treat each slice (b,h) as a
    # matrix product left[b,h] @ right[b,h].T
    left_mat = left.view(B * H, N, N)            # [B·H, N, N] FP16
    right_mat = right.view(B * H, N, N)          # [B·H, N, N] FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                  # [B, H, N, N] FP16

    # ------------------------------------------------------------------
    # 5️⃣ Combine 2nd LayerNorm, out_gate and final projection
    # ------------------------------------------------------------------
    # transposed final projection weight: [H, dim] (FP16)
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    # block sizes for the fused kernel – tuned for the H100
    BLOCK_M_OUT = 128   # rows (R) per program
    BLOCK_K_OUT = 32    # hidden‑dim chunk
    BLOCK_N_OUT = 128   # output‑dim chunk

    grid = (
        triton.cdiv(R, BLOCK_M_OUT),
        triton.cdiv(dim, BLOCK_N_OUT),
    )
    final_output_kernel[grid](
        out,
        out_gate,
        w["to_out_norm.weight"],   # γ (FP32)
        w["to_out_norm.bias"],     # β (FP32)
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=8,
    )
    return final
