@triton.jit
def norm_proj_gate_kernel(
    x_ptr, # [R, dim], fp32
    left_ptr, right_ptr, out_gate_ptr,
    mask_ptr,
    ln_w_ptr, ln_b_ptr,
    left_w_ptr, right_w_ptr,
    left_gate_w_ptr, right_gate_w_ptr, out_gate_w_ptr,
    R, H, N, dim,
    HAS_MASK: tl.constexpr,
    EPS: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_D: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < R

    # compute mean, var
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_R], dtype=tl.float32)
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_r[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr, mask=mask_r[:, None] & mask_d[None, :], other=0.0)
        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + EPS)

    # optional mask load (per row)
    if HAS_MASK:

```
@triton.jit
def fused_norm_proj_gate_kernel(
    x_ptr, left_ptr, right_ptr, out_gate_ptr,
    mask_ptr,
    ln_w_ptr, ln_b_ptr,
    left_w_ptr, right_w_ptr, left_gate_w_ptr, right_gate_w_ptr, out_gate_w_ptr,
    R, H, N, dim,
    HAS_MASK: tl.constexpr,
    EPS: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_D: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < R
    
    # compute mean and variance
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_R], dtype=tl.float32)
    
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        
        x_vals = tl.load(x_ptr + offs_r[:, None] * dim + offs_d[None, :],
                         mask=mask_r[:, None] & mask_d[None, :],
                         other=0.0)
        sum_val += tl.sum(x_vals, axis=1)
        sum_sq += tl.sum(x_vals * x_vals, axis=1)
    
    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + EPS)
    
    # optional mask
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r,
                           mask=mask_r,
                           other=1.0).to(tl.float32)  # [BLOCK_R]
    else:
        mask_val = tl.full([BLOCK_R], 1.0, dtype=tl.float32)
    mask_val = mask_val[:, None]   # broadcast

    # decode b,i,k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N
    
    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + k  # offset for hidden plane 0
    
    # iterate over hidden dimension chunks
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H
        
        # accumulators for this hidden tile
        left_acc = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        right_acc = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        left_gate_acc = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        right_gate_acc = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        out_gate_acc = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        
        for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
            offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
            mask_d = offs_d < dim
            
            # load x
            x_vals = tl.load(x_ptr + offs_r[:, None] * dim + offs_d[None, :],
                             mask=mask_r[:, None] & mask_d[None, :],
                             other=0.0)   # fp32
            
            # normalize
            norm = (x_vals - mean[:, None]) * rstd[:, None]   # fp32
            
            # apply LN scale and bias
            ln_w = tl.load(ln_w_ptr + offs_d, mask=mask_d, other=0.0)   # fp32
            ln_b = tl.load(ln_b_ptr + offs_d, mask=mask_d, other=0.0)   # fp32
            norm = norm * ln_w[None, :] + ln_b[None, :]  # fp32
            
            # load weight tiles (transposed)
            # shape [BLOCK_D, BLOCK_H]
            left_w = tl.load(left_w_ptr + offs_d[:, None] * H + offs_h[None, :],
                             mask=mask_d[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float32)
            right_w = tl.load(right_w_ptr + offs_d[:, None] * H + offs_h[None, :],
                              mask=mask_d[:, None] & mask_h[None, :],
                              other=0.0).to(tl.float32)
            left_gate_w = tl.load(left_gate_w_ptr + offs_d[:, None] * H + offs_h[None, :],
                                  mask=mask_d[:, None] & mask_h[None, :],
                                  other=0.0).to(tl.float32)
            right_gate_w = tl.load(right_gate_w_ptr + offs_d[:, None] * H + offs_h[None, :],
                                   mask=mask_d[:, None] & mask_h[None, :],
                                   other=0.0).to(tl.float32)
            out_gate_w = tl.load(out_gate_w_ptr + offs_d[:, None] * H + offs_h[None, :],
                                 mask=mask_d[:, None] & mask_h[None, :],
                                 other=0.0).to(tl.float32)
            
            # dot products
            left_acc += tl.dot(norm, left_w)           # (BLOCK_R,BLOCK_H)
            right_acc += tl.dot(norm, right_w)
            left_gate_acc += tl.dot(norm, left_gate_w)
            right_gate_acc += tl.dot(norm, right_gate_w)
            out_gate_acc += tl.dot(norm, out_gate_w)
        
        # apply sigmoid gates
        left_gate = 1.0 / (1.0 + tl.exp(-left_gate_acc))
        right_gate = 1.0 / (1.0 + tl.exp(-right_gate_acc))
        out_gate_val = 1.0 / (1.0 + tl.exp(-out_gate_acc))
        
        # apply mask to left and right
        left_val = left_acc * left_gate * mask_val
        right_val = right_acc * right_gate * mask_val
        # out_gate_val no mask
        
        # store
        dst = base[:, None] + offs_h[None, :] * hidden_stride
        tl.store(left_ptr + dst,
                 left_val.to(tl.float16),
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst,
                 right_val.to(tl.float16),
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst,
                 out_gate_val.to(tl.float16),
                 mask=mask_r[:, None] & mask_h[None, :])
```

"""
Outgoing TriMul (AlphaFold‑3) – fused forward‑only implementation.

Algorithm
---------
1️⃣  Row‑wise LayerNorm (γ,β) and simultaneous projection of the
    normalized pairwise token into five hidden tensors:
        left_proj , right_proj , left_gate , right_gate , out_gate.
    All five linear maps share the same input, the LN statistics are
    computed on‑the‑fly and the mask (if any) is applied only to
    left/right tensors (the out‑gate is NOT masked).
    The five results are written in FP16 layout [B, H, N, N].

2️⃣  Batched GEMM (cuBLAS) computes the core einsum
        out[b,h,i,j] = Σ_k left[b,h,i,k]·right[b,h,j,k]
    using a single torch.bmm over the hidden dimension.

3️⃣  Row‑wise mean & rsqrt‑variance of the contracted tensor
    (second LayerNorm statistics) – a tiny Triton kernel.

4️⃣  Fuse the second LayerNorm, the out‑gate multiplication and the
    final projection (H → dim) into one Triton kernel, writing the
    final tensor `[B, N, N, dim]` in FP32.

All kernels use mixed precision (FP32 for statistics, FP16 for the
bulk work) and are targeted at Triton 3.3.1 on an NVIDIA H100.
Only the forward pass is required.
"""

import torch
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣  Fused LayerNorm + 5×Linear + gating + optional mask
# ----------------------------------------------------------------------
@triton.jit
def fused_norm_proj_gate_kernel(
    # pointers
    x_ptr,                 # [R, dim]                FP32 input
    left_ptr,              # [B, H, N, N]            FP16 output
    right_ptr,             # [B, H, N, N]            FP16 output
    out_gate_ptr,          # [B, H, N, N]            FP16 output
    mask_ptr,              # [R]                     FP16 mask (optional)

    # LayerNorm parameters
    ln_w_ptr,              # [dim]                   FP32 γ
    ln_b_ptr,              # [dim]                   FP32 β

    # Linear weights (already transposed → [dim, H])
    left_w_ptr,            # [dim, H]                FP16
    right_w_ptr,           # [dim, H]                FP16
    left_gate_w_ptr,       # [dim, H]                FP16
    right_gate_w_ptr,      # [dim, H]                FP16
    out_gate_w_ptr,        # [dim, H]                FP16

    # sizes & compile‑time constants
    R, H, N, dim,
    HAS_MASK: tl.constexpr,   # 1 → mask present, 0 → no mask
    EPS: tl.constexpr,        # LayerNorm epsilon
    BLOCK_R: tl.constexpr,    # rows per program
    BLOCK_D: tl.constexpr,    # dim chunk per iteration
    BLOCK_H: tl.constexpr,    # hidden‑dim chunk per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)            # [BLOCK_R]
    mask_r = offs_r < R

    # ------------------------------------------------------------------
    # 1️⃣  Compute per‑row mean & variance of the raw input x
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_R], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        x = tl.load(x_ptr + offs_r[:, None] * dim + offs_d[None, :],
                    mask=mask_r[:, None] & mask_d[None, :],
                    other=0.0)                               # FP32
        sum_val += tl.sum(x, axis=1)
        sum_sq  += tl.sum(x * x, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + EPS)                             # 1/√(var+ε)

    # ------------------------------------------------------------------
    # 2️⃣  Broadcast optional mask (mask is 1.0 for unmasked positions)
    # ------------------------------------------------------------------
    if HAS_MASK:
        m = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # FP16 → cast
        m = m.to(tl.float32)                                    # FP32
    else:
        m = tl.full([BLOCK_R], 1.0, dtype=tl.float32)           # FP32

    m = m[:, None]   # [BLOCK_R,1] – broadcast over hidden dim

    # ------------------------------------------------------------------
    # 3️⃣  Decode (b,i,k) from flat row index; compute base offset for
    #     hidden planes in the output tensors.
    # ------------------------------------------------------------------
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N
    hidden_stride = N * N                     # distance between two hidden slices
    base = b * (H * hidden_stride) + i * N + k   # offset for hidden‑plane 0

    # ------------------------------------------------------------------
    # 4️⃣  Iterate over hidden‑dim tiles and accumulate the five linear
    #     projections (raw values).  The final sigmoid gates and the
    #     optional mask are applied after the accumulation.
    # ------------------------------------------------------------------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # accumulators for this tile   (FP32)
        left_acc      = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        right_acc     = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        left_gate_acc = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        right_gate_acc= tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        out_gate_acc  = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)

        # ------------------------------------------------------------------
        #   Loop over the feature dimension to compute dot‑products
        # ------------------------------------------------------------------
        for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
            offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
            mask_d = offs_d < dim

            # raw input slice
            x = tl.load(x_ptr + offs_r[:, None] * dim + offs_d[None, :],
                        mask=mask_r[:, None] & mask_d[None, :],
                        other=0.0)                               # FP32

            # ----- LayerNorm scaling -----
            # (x - μ) * rstd
            x_norm = (x - mean[:, None]) * rstd[:, None]       # FP32

            # γ,β of the first LayerNorm
            ln_w = tl.load(ln_w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
            ln_b = tl.load(ln_b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
            x_norm = x_norm * ln_w[None, :] + ln_b[None, :]              # FP32

            # ----- Load transposed weight tiles (dim × H) -----
            # shape [BLOCK_D, BLOCK_H] for each matrix
            left_w       = tl.load(left_w_ptr       + offs_d[:, None] * H + offs_h[None, :],
                                   mask=mask_d[:, None] & mask_h[None, :],
                                   other=0.0).to(tl.float32)
            right_w      = tl.load(right_w_ptr      + offs_d[:, None] * H + offs_h[None, :],
                                   mask=mask_d[:, None] & mask_h[None, :],
                                   other=0.0).to(tl.float32)
            left_gate_w  = tl.load(left_gate_w_ptr  + offs_d[:, None] * H + offs_h[None, :],
                                   mask=mask_d[:, None] & mask_h[None, :],
                                   other=0.0).to(tl.float32)
            right_gate_w = tl.load(right_gate_w_ptr + offs_d[:, None] * H + offs_h[None, :],
                                   mask=mask_d[:, None] & mask_h[None, :],
                                   other=0.0).to(tl.float32)
            out_gate_w   = tl.load(out_gate_w_ptr   + offs_d[:, None] * H + offs_h[None, :],
                                   mask=mask_d[:, None] & mask_h[None, :],
                                   other=0.0).to(tl.float32)

            # ----- Dot‑products (BLOCK_R × BLOCK_D) • (BLOCK_D × BLOCK_H) -----
            left_acc      += tl.dot(x_norm, left_w)          # [BLOCK_R, BLOCK_H]
            right_acc     += tl.dot(x_norm, right_w)
            left_gate_acc += tl.dot(x_norm, left_gate_w)
            right_gate_acc+= tl.dot(x_norm, right_gate_w)
            out_gate_acc  += tl.dot(x_norm, out_gate_w)

        # ----- Sigmoid gates (still in FP32) -----
        left_gate  = 1.0 / (1.0 + tl.exp(-left_gate_acc))
        right_gate = 1.0 / (1.0 + tl.exp(-right_gate_acc))
        out_gate   = 1.0 / (1.0 + tl.exp(-out_gate_acc))

        # ----- Apply mask to left/right, cast to FP16, store -----
        left_val  = left_acc  * left_gate  * m
        right_val = right_acc * right_gate * m
        # out_gate is NOT masked
        dst = base[:, None] + offs_h[None, :] * hidden_stride    # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst,
                 left_val.to(tl.float16),
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst,
                 right_val.to(tl.float16),
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst,
                 out_gate.to(tl.float16),
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 3️⃣ Row‑wise mean & rsqrt‑variance of the contracted tensor
# ----------------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr,    # [B, H, N, N]   FP16
    mu_ptr,     # [R]            FP32
    rstd_ptr,   # [R]            FP32
    R, H, N,
    EPS: tl.constexpr,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_K: tl.constexpr,   # hidden‑dim chunk
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < R

    # decode (b,i,j)
    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + j      # start of plane h=0

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        ptr = out_ptr + base[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)          # [BLOCK_M, BLOCK_K]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mu   = sum_val / H
    var  = sum_sq / H - mu * mu
    rstd = tl.rsqrt(var + EPS)

    tl.store(mu_ptr   + offs_m, mu,   mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)


# ----------------------------------------------------------------------
# 4️⃣ Fuse second LayerNorm, out_gate and final projection
# ----------------------------------------------------------------------
@triton.jit
def final_output_kernel(
    out_ptr,        # [B, H, N, N] FP16   (contracted tensor)
    gate_ptr,       # [B, H, N, N] FP16   (out_gate)
    mu_ptr,         # [R] FP32
    rstd_ptr,       # [R] FP32
    ln_w_ptr,       # [H] FP32  (to_out_norm.weight)
    ln_b_ptr,       # [H] FP32  (to_out_norm.bias)
    w_out_T_ptr,    # [H, dim] FP16 (to_out weight transposed)
    out_final_ptr,  # [B, N, N, dim] FP32
    R, H, N, dim,
    EPS: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr,
):
    pid_m = tl.program_id(0)   # rows (R)
    pid_n = tl.program_id(1)   # output‑dim tiles

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)   # [BLOCK_M]
    offs_n = col_start + tl.arange(0, BLOCK_N)   # [BLOCK_N]

    mask_m = offs_m < R
    mask_n = offs_n < dim

    # decode (b,i,j)
    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + j   # start of plane h=0

    # per‑row stats
    mu   = tl.load(mu_ptr   + offs_m, mask=mask_m, other=0.0)   # [BLOCK_M]
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)   # [BLOCK_M]

    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        # load contracted tensor and gate
        ptr_out  = out_ptr  + base[:, None] + offs_k[None, :] * hidden_stride
        ptr_gate = gate_ptr + base[:, None] + offs_k[None, :] * hidden_stride

        vals_out = tl.load(ptr_out,
                           mask=mask_m[:, None] & mask_k[None, :],
                           other=0.0).to(tl.float32)          # [M,K]
        vals_gate= tl.load(ptr_gate,
                           mask=mask_m[:, None] & mask_k[None, :],
                           other=0.0).to(tl.float32)          # [M,K]

        # LayerNorm (γ,β) for this hidden slice
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [K]
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [K]

        # (out - μ) * rstd → γ· + β → * out_gate
        diff = (vals_out - mu[:, None]) * rstd[:, None]              # [M,K]
        norm = diff * gamma[None, :] + beta[None, :]                 # [M,K]
        term = norm * vals_gate                                      # [M,K]
        term_fp16 = term.to(tl.float16)                              # fp16 for dot

        # final linear projection (H → dim)
        w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]   # [K, TILE_N]
        w_tile = tl.load(w_ptr,
                         mask=mask_k[:, None] & mask_n[None, :],
                         other=0.0)                                      # [K, TILE_N]

        acc += tl.dot(term_fp16, w_tile).to(tl.float32)               # [M,N]

    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])

# Entrypoint
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only implementation of the *outgoing* TriMul operator.

    Args:
        data (tuple):
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping containing the model weights:
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config_dict  : must contain "dim", "hidden_dim",
                             optionally "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim        = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask     = bool(cfg.get("nomask", True))
    eps        = 1e-5
    device     = x.device

    B, N, _, _ = x.shape
    R = B * N * N                     # number of pairwise tokens
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣  Fused LayerNorm + 5×Linear + gating (+ optional mask)
    # ------------------------------------------------------------------
    # transpose all weight matrices so they are stored as [dim, H]
    left_w_T      = w["left_proj.weight"].to(torch.float16).t().contiguous()
    right_w_T     = w["right_proj.weight"].to(torch.float16).t().contiguous()
    left_gate_w_T = w["left_gate.weight"].to(torch.float16).t().contiguous()
    right_gate_w_T= w["right_gate.weight"].to(torch.float16).t().contiguous()
    out_gate_w_T  = w["out_gate.weight"].to(torch.float16).t().contiguous()

    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty_like(left)
    out_gate  = torch.empty_like(left)

    if not nomask and mask is not None:
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy

    BLOCK_R = 32          # rows per program (tuned for register pressure)
    BLOCK_D = 32          # feature‑dim chunk
    BLOCK_H = 64          # hidden‑dim chunk

    grid = (triton.cdiv(R, BLOCK_R),)
    fused_norm_proj_gate_kernel[grid](
        x,
        left,
        right,
        out_gate,
        mask_ptr,
        w["norm.weight"],         # γ  (FP32)
        w["norm.bias"],           # β  (FP32)
        left_w_T,
        right_w_T,
        left_gate_w_T,
        right_gate_w_T,
        out_gate_w_T,
        R, H, N, dim,
        HAS_MASK=HAS_MASK,
        EPS=eps,
        BLOCK_R=BLOCK_R,
        BLOCK_D=BLOCK_D,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    # ------------------------------------------------------------------
    # 2️⃣  Core Einsum via batched GEMM (cuBLAS)
    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)                     # FP16
    right_mat = right.view(B * H, N, N)                    # FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                            # [B, H, N, N] FP16

    # ------------------------------------------------------------------
    # 3️⃣  Row‑wise mean & rsqrt‑variance of the contracted tensor
    # ------------------------------------------------------------------
    mu   = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 256
    BLOCK_K_STATS = 128   # H ≤ 128 in all test configs
    grid_stats = (triton.cdiv(R, BLOCK_M_STATS),)
    rowwise_stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        EPS=eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 4️⃣  Second LayerNorm + out_gate + final projection (H→dim)
    # ------------------------------------------------------------------
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim]

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 128
    BLOCK_K_OUT = 32
    BLOCK_N_OUT = 128
    grid_out = (
        triton.cdiv(R, BLOCK_M_OUT),
        triton.cdiv(dim, BLOCK_N_OUT),
    )
    final_output_kernel[grid_out](
        out,
        out_gate,
        mu,
        rstd,
        w["to_out_norm.weight"],   # γ  (FP32)
        w["to_out_norm.bias"],     # β  (FP32)
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        EPS=eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=8,
    )
    # ------------------------------------------------------------------
    return final
