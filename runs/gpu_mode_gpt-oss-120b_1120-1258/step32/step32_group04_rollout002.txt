"""
Outgoing TriMul – forward‑only Triton implementation.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the flattened [B·N·N, dim] tensor (FP32 → FP16).
2️⃣ Fuse the five linear maps (left, right, left_gate, right_gate, out_gate)
   into a single FP16 GEMM.  A Triton kernel
   * applies the optional pairwise mask,
   * computes sigmoid gates,
   * writes left, right and out_gate tensors in [B, H, N, N] layout.
3️⃣ Batched GEMM (cuBLAS) computes the core “einsum”
   out[b, h, i, j] = Σₖ left[b, h, i, k]·right[b, h, j, k]   →  FP16.
4️⃣ A *single* Triton kernel fuses:
      – row‑wise mean & rsqrt‑variance of the contracted tensor,
      – the second LayerNorm (to_out_norm),
      – gating by out_gate,
      – the final projection (to_out) .
   The kernel works tile‑wise over the output‑dimension (dim) and the
   hidden dimension (H).  All intermediate statistics stay in registers,
   eliminating a separate statistics kernel and reducing memory traffic.
5️⃣ The result is written as FP32 [B, N, N, dim].

All kernels target Triton 3.3.1 on an NVIDIA H100 and use mixed precision
(FP32 for statistics, FP16 for the bulk of the work) to maximize throughput.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# --------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,   # [R, dim]   FP32 input
    y_ptr,   # [R, dim]   FP16 output
    w_ptr,   # [dim]      γ (FP32)
    b_ptr,   # [dim]      β (FP32)
    rows,    # scalar R
    dim,     # scalar dim
    eps,     # scalar eps
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # ------- first pass: compute mean & variance -------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------- second pass: write normalized values -------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])

# --------------------------------------------------------------
# 2️⃣ Split + gate + optional mask
# --------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,      # [R, 5·H]   FP16 fused projection
    left_ptr,      # [B, H, N, N] FP16
    right_ptr,     # [B, H, N, N] FP16
    out_gate_ptr,  # [B, H, N, N] FP16
    mask_ptr,      # [R]       FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)           # [BLOCK_R]
    mask_r = offs_r < R

    # Decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # Optional mask (broadcast across hidden dim)
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)          # [BLOCK_R]
        m_val = m_val[:, None]                                            # [BLOCK_R, 1]
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N
    # base address for hidden‑dim = 0
    base = b * (H * hidden_stride) + i * N + k   # [BLOCK_R]

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # Index into fused projection: [left | right | lgate | rgate | ogate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]    # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # Sigmoid (FP32 → FP16)
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * m_val * left_gate
        right = right_raw * m_val * right_gate
        outg  = out_gate

        # Destination offsets for hidden = h_off … h_off+BLOCK_H‑1
        dst = base[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst, left,  mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst, right, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst, outg, mask=mask_r[:, None] & mask_h[None, :])

# --------------------------------------------------------------
# 3️⃣ Fused final kernel  (row‑wise stats + LN + gating + proj)
# --------------------------------------------------------------
@triton.jit
def final_fused_kernel(
    out_ptr,        # [B, H, N, N] FP16  (contracted left·right)
    gate_ptr,       # [B, H, N, N] FP16  (out_gate)
    ln_w_ptr,       # [H]          FP32 (to_out_norm.weight)
    ln_b_ptr,       # [H]          FP32 (to_out_norm.bias)
    w_out_T_ptr,    # [H, dim]     FP16 (to_out weight transposed)
    out_final_ptr,  # [B, N, N, dim] FP32
    B, H, N, dim,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr,
):
    pid_m = tl.program_id(0)   # rows (R = B·N·N)
    pid_n = tl.program_id(1)   # output‑dim tiles

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)   # [BLOCK_M]
    offs_n = col_start + tl.arange(0, BLOCK_N)   # [BLOCK_N]

    mask_m = offs_m < (B * N * N)
    mask_n = offs_n < dim

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + j   # start offset for hidden=0

    # ------------------------------------------------------------------
    # First pass: row‑wise mean & variance over hidden dimension
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H
        ptr = out_ptr + base[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------------------------------------------------------
    # Second pass: LayerNorm + out_gate + projection → final output
    # ------------------------------------------------------------------
    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        # Load hidden slice of out and out_gate
        ptr_out  = out_ptr + base[:, None] + offs_k[None, :] * hidden_stride
        ptr_gate = gate_ptr + base[:, None] + offs_k[None, :] * hidden_stride

        out_val = tl.load(ptr_out,
                          mask=mask_m[:, None] & mask_k[None, :],
                          other=0.0).to(tl.float32)
        gate_val = tl.load(ptr_gate,
                           mask=mask_m[:, None] & mask_k[None, :],
                           other=0.0).to(tl.float32)

        # Normalize with the statistics computed above
        diff = (out_val - mean[:, None]) * rstd[:, None]

        # Apply the second LayerNorm (γ,β) on the hidden dimension
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [K]
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [K]

        norm = diff * gamma[None, :] + beta[None, :]   # [M,K]

        # Multiply by out_gate
        term = norm * gate_val                         # [M,K]
        term_fp16 = term.to(tl.float16)               # fp16 for dot

        # Load slice of final projection weight  (H, dim)
        w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]   # [K, BLOCK_N]
        w_tile = tl.load(w_ptr,
                         mask=mask_k[:, None] & mask_n[None, :],
                         other=0.0)                                    # [K, BLOCK_N]

        # Accumulate (M,K)·(K,N) → (M,N)   (fp16·fp16 → fp32)
        acc += tl.dot(term_fp16, w_tile).to(tl.float32)

    # Store final result: flattened rows (B·N·N) × dim
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])

# --------------------------------------------------------------
# Entry point
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul operator (no gradients).

    Args:
        data (tuple):
            - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
            - weights_dict : dict containing the model weights
            - config_dict  : must contain ``dim``, ``hidden_dim`` and ``nomask``

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = bool(cfg.get("nomask", True))
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                 # total pairwise tokens
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()            # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # ------------------------------------------------------------------
    # 2️⃣ Fuse five linear maps (left, right, left_gate, right_gate, out_gate)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                 # shape [5·H, dim]

    proj = F.linear(x_norm, fused_w)    # (R, 5·H) FP16
    del fused_w, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ Split, mask, sigmoid gates → left, right, out_gate
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if not nomask and mask is not None:
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        # dummy tensor – never read when HAS_MASK == 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)

    BLOCK_R = 128
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)

    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension (cuBLAS)
    # ------------------------------------------------------------------
    # left/right: [B, H, N, N] → view as [B·H, N, N]
    left_mat  = left.view(B * H, N, N)                 # FP16
    right_mat = right.view(B * H, N, N)                # FP16
    # out = Σₖ left[...,k]·right[...,k]   (i‑j contraction)
    contracted = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = contracted.view(B, H, N, N)                               # [B, H, N, N] FP16

    # ------------------------------------------------------------------
    # 5️⃣ Final fused kernel (statistics + LN + gate + projection)
    # ------------------------------------------------------------------
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim]

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    # Choose tile sizes respecting typical H100 capacity
    BLOCK_M_OUT = 128                               # rows per program
    BLOCK_K_OUT = 32                                # hidden‑dim chunk
    BLOCK_N_OUT = 64 if dim <= 192 else 128        # output‑dim tile

    grid_out = (
        triton.cdiv(R, BLOCK_M_OUT),
        triton.cdiv(dim, BLOCK_N_OUT),
    )
    final_output_kernel = final_fused_kernel

    final_output_kernel[grid_out](
        out,
        out_gate,
        w["to_out_norm.weight"],   # γ  FP32
        w["to_out_norm.bias"],     # β  FP32
        w_out_T,
        final,
        B,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=8,
    )
    # ------------------------------------------------------------------
    return final
