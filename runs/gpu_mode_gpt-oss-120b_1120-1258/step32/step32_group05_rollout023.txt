@triton.jit
def proj_gate_fused_kernel(
    x_ptr,               # [R, C] FP32
    left_ptr,            # [B, H, N, N] FP16
    right_ptr,           # same
    out_gate_ptr,        # same
    mask_ptr,            # [R] FP16 optional
    weight_T_ptr,        # [C, 5*H] FP16
    R, H, N, C,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
    BLOCK_C: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < R

    # compute batch and position indices
    NN = N * N
    b = offs_r // NN
    rem = offs_r - b * NN
    i = rem // N
    k = rem - i * N

    # mask for optional mask
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)    # FP16
        mask_val = m_val[:, None]    # shape [BLOCK_R, 1] broadcast to hidden
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # Blockwise compute sum & sumsq for layernorm
    # accumulators for sum and sumsq
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_R], dtype=tl.float32)

    # Compute mean, var
    for c_off in range(0, C, BLOCK_C):
        cur_c = c_off + tl.arange(0, BLOCK_C)
        mask_c = cur_c < C

        # Load x values
        x_vals = tl.load(
            x_ptr + offs_r[:, None] * C + cur_c[None, :],
            mask=mask_r[:, None] & mask_c[None, :],
            other=0.0,
        ).to(tl.float32)  # [BLOCK_R, BLOCK_C]

        sum_val += tl.sum(x_vals, axis=1)   # sum across C
        sum_sq += tl.sum(x_vals * x_vals, axis=1)

    mean = sum_val / C
    var = sum_sq / C - mean * mean
    rstd = tl.rsqrt(var + 1e-5)

    # Now second pass: compute projection & gates
    # Use accumulators for each projection group as float32: shape [BLOCK_R, BLOCK_H]
    acc_left = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
    acc_right = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
    acc_lgate = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
    acc_rgate = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
    acc_outgate = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)

    for c_off in range(0, C, BLOCK_C):
        cur_c = c_off + tl.arange(0, BLOCK_C)
        mask_c = cur_c < C

        # Load x values and normalize them
        x_vals = tl.load(
            x_ptr + offs_r[:, None] * C + cur_c[None, :],
            mask=mask_r[:, None] & mask_c[None, :],
            other=0.0,
        ).to(tl.float32)  # [BLOCK_R, BLOCK_C]

        norm_vals = (x_vals - mean[:, None]) * rstd[:, None]   # [BLOCK_R, BLOCK_C] (float32)
        # convert to FP16 for multiplication with weight_T (FP16) but we can keep as fp32 and cast weight
        # weight_T slice per group
        for g in range(5):
            # Compute offset for hidden h range
            # weight_T shape [C, 5*H]; we need slice for rows cur_c and columns (g*H + cur_h)
            # cur_h as vector of hidden offsets in block
            cur_h = tl.arange(0, BLOCK_H)  # offsets within block
            mask_h = cur_h < H
            # base column index for group g
            col_offset = g * H + cur_h  # shape [BLOCK_H]

            # pointer for weight_T slice: weight_T_ptr + cur_c[:, None] * (5*H) + col_offset[None, :]
            w_ptr = weight_T_ptr + cur_c[:, None] * (5 * H) + col_offset[None, :]   # shape [BLOCK_C, BLOCK_H]

            w_vals = tl.load(
                w_ptr,
                mask=mask_c[:, None] & mask_h[None, :],
                other=0.0,
            ).to(tl.float32)   # [BLOCK_C, BLOCK_H]

            # dot product: norm_vals @ w_vals  (block_R, block_C) * (block_C, block_H) => (block_R, block_H)
            # Use tl.dot
            prod = tl.dot(norm_vals, w_vals)   # float32

            if g == 0:
                acc_left += prod
            elif g == 1:
                acc_right += prod
            elif g == 2:
                acc_lgate += prod
            elif g == 3:
                acc_rgate += prod
            else:
                acc_outgate += prod

    # After loops, we have accumulators for each projection (float32).
    # Apply sigmoid to gate accumulators
    # Use approximated stable sigmoid: 1/(1+exp(-x))
    lgate = (1.0 / (1.0 + tl.exp(-acc_lgate))).to(tl.float16)
    rgate = (1.0 / (1.0 + tl.exp(-acc_rgate))).to(tl.float16)
    outgate = (1.0 / (1.0 + tl.exp(-acc_outgate))).to(tl.float16)

    # left and right raw values: convert to fp16
    left_raw = acc_left.to(tl.float16)
    right_raw = acc_right.to(tl.float16)

    # Apply mask and gates to left/right
    left = left_raw * mask_val * lgate   # broadcast mask over hidden
    right = right_raw * mask_val * rgate

    outg = outgate  # no mask

    # compute storage offsets for hidden dimension
    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + k

    dst_off = base_offset[:, None] + cur_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H] offset

    tl.store(left_ptr + dst_off, left, mask=mask_r[:, None] & (cur_h < H)[None, :])
    tl.store(right_ptr + dst_off, right, mask=mask_r[:, None] & (cur_h < H)[None, :])
    tl.store(out_gate_ptr + dst_off, outg, mask=mask_r[:, None] & (cur_h < H)[None, :])

@triton.jit
def proj_gate_fused_kernel(
    x_ptr,                 # [R, C] FP32
    left_ptr,              # [B, H, N, N] FP16
    right_ptr,             # [B, H, N, N] FP16
    out_gate_ptr,          # [B, H, N, N] FP16
    mask_ptr,              # [R] FP16 (optional)
    norm_w_ptr,            # [C] FP32
    norm_b_ptr,            # [C] FP32
    left_w_ptr,            # [C, H] FP16
    right_w_ptr,           # [C, H] FP16
    left_gate_w_ptr,       # [C, H] FP16
    right_gate_w_ptr,      # [C, H] FP16
    out_gate_w_ptr,        # [C, H] FP16
    R, H, N, C,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr, BLOCK_C: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < R

    # decode b,i,k
    NN = N * N
    b = offs_r // NN
    rem = offs_r - b * NN
    i = rem // N
    k = rem - i * N

    # mask for optional mask
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)
        mask_val = m_val[:, None]  # shape [BLOCK_R,1] FP16
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # Compute mean and variance for LayerNorm
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_R], dtype=tl.float32)

    for c_off in range(0, C, BLOCK_C):
        cur_c = c_off + tl.arange(0, BLOCK_C)
        mask_c = cur_c < C
        x_vals = tl.load(
            x_ptr + offs_r[:, None] * C + cur_c[None, :],
            mask=mask_r[:, None] & mask_c[None, :],
            other=0.0,
        ).to(tl.float32)   # [BLOCK_R, BLOCK_C]

        sum_val += tl.sum(x_vals, axis=1)
        sum_sq += tl.sum(x_vals * x_vals, axis=1)

    mean = sum_val / C
    var = sum_sq / C - mean * mean
    rstd = tl.rsqrt(var + 1e-5)

    # Prepare hidden dimension vector
    cur_h = tl.arange(0, BLOCK_H)

    # Accumulators for projections and gates
    acc_left = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
    acc_right = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
    acc_lgate = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
    acc_rgate = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
    acc_outgate = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)

    # Second pass: compute normalized values and dot with projection weights
    for c_off in range(0, C, BLOCK_C):
        cur_c = c_off + tl.arange(0, BLOCK_C)
        mask_c = cur_c < C

        # Load x block
        x_vals = tl.load(
            x_ptr + offs_r[:, None] * C + cur_c[None, :],
            mask=mask_r[:, None] & mask_c[None, :],
            other=0.0,
        ).to(tl.float32)   # [BLOCK_R, BLOCK_C]

        # Normalize
        norm_vals = (x_vals - mean[:, None]) * rstd[:, None]   # [BLOCK_R, BLOCK_C]

        # Apply LayerNorm weight and bias (gamma, beta)
        gamma = tl.load(norm_w_ptr + cur_c, mask=mask_c, other=0.0).to(tl.float32)   # [BLOCK_C]
        beta = tl.load(norm_b_ptr + cur_c, mask=mask_c, other=0.0).to(tl.float32)    # [BLOCK_C]

        # broadcast

        beta_row = beta[None, :]

        normed = norm_vals * gamma_row + beta_row   # [BLOCK_R, BLOCK_C]

        # Now dot with each projection weight (transposed)

        # left
        left_w = tl.load(

            mask=mask_c[:, None] & mask_h[None, :],

        acc_left += tl.dot(normed, left_w)

        # right
        right_w = tl.load(
            right_w_ptr + cur_c[:, None] * H + cur_h[None, :],
            mask=mask_c[:, None] & mask_h[None, :],
            other=0.0,
        ).to(tl.float32)

        acc_right += tl.dot(normed, right_w)

        # left_gate
        lgate_w = tl.load(
            left_gate_w_ptr + cur_c[:, None] * H + cur_h[None, :],
            mask=mask_c[:, None] & mask_h[None, :],
            other=0.0,
        ).to(tl.float32)

        acc_lgate += tl.dot(normed, lgate_w)

        # right_gate
        rgate_w = tl.load(
            right_gate_w_ptr + cur_c[:, None] * H + cur_h[None, :],
            mask=mask_c[:, None] & mask_h[None, :],
            other=0.0,

        # out_gate
        outgate_w = tl.load(

            mask=mask_c[:, None] & mask_h[None, :],
            other=0.0,

    left_raw = acc_left.to(tl.float16)

    left = left_raw * mask_val * lgate
    right = right_raw * mask_val * rgate

    # Store to output tensors

             mask=mask_r[:, None] & mask_h[None, :])
    tl.store(right_ptr + dst_off,
             right,
             mask=mask_r[:, None] & mask_h[None, :])
    tl.store(out_gate_ptr + dst_off,
             outg,
             mask=mask_r[:, None] & mask_h[None, :])

"""
TriMul (outgoing) – Triton fused implementation
------------------------------------------------
We fuse the three heavy stages of the outgoing TriMul operator:

1️⃣ Row‑wise LayerNorm (γ/β) on the input tensor (FP32 → FP16) and
   five linear projections (left, right, left‑gate, right‑gate, out‑gate).
   This is done in a single kernel that:
   * computes mean / variance per row,
   * normalises the row and applies the LayerNorm γ/β,
   * performs the five matrix‑vector products (C→H) for each hidden block.

2️⃣ Batched GEMM (`torch.bmm`) over the sequence dimension N:
   out[b, h, i, j] = Σₖ left[b, h, i, k] * right[b, h, j, k].

3️⃣ Row‑wise mean / rsqrt‑variance of the contracted tensor (needed for the
   second LayerNorm).

4️⃣ Fuse the second LayerNorm, element‑wise multiplication by the
   out‑gate, and the final linear projection back to the model dimension C
   into a single Triton kernel.

All kernels are written for H ≤ 128, C ≤ 384 and run on H100 GPUs.
The final tensor is returned as float32 with shape [B, N, N, C].
"""

import torch
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣  Fused LayerNorm + 5‑projection kernel
# ----------------------------------------------------------------------
@triton.jit
def proj_gate_fused_kernel(
    x_ptr,                 # [R, C]  FP32  (input)
    left_ptr,              # [B, H, N, N] FP16 (output left)
    right_ptr,             # [B, H, N, N] FP16 (output right)
    out_gate_ptr,          # [B, H, N, N] FP16 (output out‑gate)
    mask_ptr,              # [R] FP16   (optional mask)
    ln_w_ptr,              # [C] FP32   (LayerNorm weight γ)
    ln_b_ptr,              # [C] FP32   (LayerNorm bias β)
    left_w_ptr,            # [C, H] FP16 (left projection‑T)
    right_w_ptr,           # [C, H] FP16 (right projection‑T)
    left_gate_w_ptr,       # [C, H] FP16 (left_gate‑T)
    right_gate_w_ptr,      # [C, H] FP16 (right_gate‑T)
    out_gate_w_ptr,        # [C, H] FP16 (out_gate‑T)
    R, H, N, C,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr, BLOCK_C: tl.constexpr,
    EPS: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # -------- decode (b,i,k) from flat row index -----------------------
    NN = N * N
    b  = offs_r // NN
    rem = offs_r - b * NN
    i   = rem // N
    k   = rem - i * N

    # -------- optional per‑row mask (broadcast over hidden dim) -------
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # FP16
        mask_val = m_val[:, None]                                   # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # ----------- 1st pass: compute mean & variance --------------------
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_R], dtype=tl.float32)

    for c_off in range(0, C, BLOCK_C):
        cur_c = c_off + tl.arange(0, BLOCK_C)                 # [BLOCK_C]
        mask_c = cur_c < C

        x_vals = tl.load(
            x_ptr + offs_r[:, None] * C + cur_c[None, :],
            mask=mask_r[:, None] & mask_c[None, :],
            other=0.0,
        ).to(tl.float32)                                      # [BLOCK_R, BLOCK_C]

        sum_val += tl.sum(x_vals, axis=1)                      # [BLOCK_R]
        sum_sq  += tl.sum(x_vals * x_vals, axis=1)            # [BLOCK_R]

    mean = sum_val / C
    var  = sum_sq / C - mean * mean
    rstd = tl.rsqrt(var + EPS)

    # ----------- 2nd pass: proj/gate for each hidden‑block -----------
    hidden_stride = N * N

    for h_off in range(0, H, BLOCK_H):                # runtime loop over hidden blocks
        cur_h = h_off + tl.arange(0, BLOCK_H)         # [BLOCK_H]
        mask_h = cur_h < H

        # per‑block accumulators (float32)
        acc_left   = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        acc_right  = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        acc_lgate  = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        acc_rgate  = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        acc_outgate= tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)

        # iterate over the channel dimension C
        for c_off in range(0, C, BLOCK_C):
            cur_c = c_off + tl.arange(0, BLOCK_C)                 # [BLOCK_C]
            mask_c = cur_c < C

            # load raw input and normalise
            x_vals = tl.load(
                x_ptr + offs_r[:, None] * C + cur_c[None, :],
                mask=mask_r[:, None] & mask_c[None, :],
                other=0.0,
            ).to(tl.float32)                                      # [BLOCK_R, BLOCK_C]

            normed = (x_vals - mean[:, None]) * rstd[:, None]     # [BLOCK_R, BLOCK_C]

            # apply LayerNorm γ/β (γ/β are FP32)
            gamma = tl.load(ln_w_ptr + cur_c, mask=mask_c, other=0.0).to(tl.float32)   # [BLOCK_C]
            beta  = tl.load(ln_b_ptr + cur_c, mask=mask_c, other=0.0).to(tl.float32)   # [BLOCK_C]

            normed = normed * gamma[None, :] + beta[None, :]      # [BLOCK_R, BLOCK_C]

            # ---------- dot with each projection weight (C×H) ----------
            # all weight matrices are stored transposed as [C, H] (FP16)
            left_w = tl.load(
                left_w_ptr + cur_c[:, None] * H + cur_h[None, :],
                mask=mask_c[:, None] & mask_h[None, :],
                other=0.0,
            ).to(tl.float32)                                      # [BLOCK_C, BLOCK_H]
            acc_left  += tl.dot(normed, left_w)

            right_w = tl.load(
                right_w_ptr + cur_c[:, None] * H + cur_h[None, :],
                mask=mask_c[:, None] & mask_h[None, :],
                other=0.0,
            ).to(tl.float32)
            acc_right += tl.dot(normed, right_w)

            lgate_w = tl.load(
                left_gate_w_ptr + cur_c[:, None] * H + cur_h[None, :],
                mask=mask_c[:, None] & mask_h[None, :],
                other=0.0,
            ).to(tl.float32)
            acc_lgate += tl.dot(normed, lgate_w)

            rgate_w = tl.load(
                right_gate_w_ptr + cur_c[:, None] * H + cur_h[None, :],
                mask=mask_c[:, None] & mask_h[None, :],
                other=0.0,
            ).to(tl.float32)
            acc_rgate += tl.dot(normed, rgate_w)

            outgate_w = tl.load(
                out_gate_w_ptr + cur_c[:, None] * H + cur_h[None, :],
                mask=mask_c[:, None] & mask_h[None, :],
                other=0.0,
            ).to(tl.float32)
            acc_outgate += tl.dot(normed, outgate_w)

        # ---------- sigmoid on gate tensors ----------
        lgate   = (1.0 / (1.0 + tl.exp(-acc_lgate))).to(tl.float16)
        rgate   = (1.0 / (1.0 + tl.exp(-acc_rgate))).to(tl.float16)
        outgate = (1.0 / (1.0 + tl.exp(-acc_outgate))).to(tl.float16)

        # ---------- final left/right tensors (FP16) ----------
        left_raw  = acc_left.to(tl.float16)
        right_raw = acc_right.to(tl.float16)

        left  = left_raw  * mask_val * lgate
        right = right_raw * mask_val * rgate
        outg  = outgate

        # ---------- store results ----------
        base_offset = b * (H * hidden_stride) + i * N + k           # start of hidden‑dim 0
        dst_off = base_offset[:, None] + cur_h[None, :] * hidden_stride  # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst_off,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_off,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_off,
                 outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 5️⃣ Row‑wise mean & rsqrt‑var of the contracted tensor (mean & rstd)
# ----------------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr,    # [B, H, N, N] FP16
    mu_ptr,     # [R] FP32
    rstd_ptr,   # [R] FP32
    total_rows, H, N,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    NN = N * N
    b = offs_m // NN
    rem = offs_m - b * NN
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + j   # start of hidden‑dim 0

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)        # [BLOCK_K]
        mask_k = offs_k < H

        ptr = out_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)                                 # [BLOCK_M, BLOCK_K]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_m, mean,  mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)


# ----------------------------------------------------------------------
# 6️⃣ Fuse 2nd LayerNorm, out_gate multiplication & final projection
# ----------------------------------------------------------------------
@triton.jit
def final_output_kernel(
    out_ptr,        # [B, H, N, N] FP16 (post‑GEMM)
    gate_ptr,       # [B, H, N, N] FP16 (out_gate)
    mu_ptr,         # [R] FP32
    rstd_ptr,       # [R] FP32
    ln_w_ptr,       # [H] FP32   (to_out_norm.weight)
    ln_b_ptr,       # [H] FP32   (to_out_norm.bias)
    w_out_T_ptr,    # [H, dim] FP16 (to_out weight transposed)
    out_final_ptr,  # [B, N, N, dim] FP32
    total_rows, H, N, dim,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr,
):
    pid_m = tl.program_id(0)   # rows
    pid_n = tl.program_id(1)   # output‑dim tiles

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)   # [BLOCK_M]
    offs_n = col_start + tl.arange(0, BLOCK_N)   # [BLOCK_N]
    mask_m = offs_m < total_rows
    mask_n = offs_n < dim

    NN = N * N
    b = offs_m // NN
    rem = offs_m - b * NN
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + j   # start of hidden‑dim 0

    # load per‑row statistics
    mu   = tl.load(mu_ptr   + offs_m, mask=mask_m, other=0.0)   # [M]
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)   # [M]

    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)               # [K]
        mask_k = offs_k < H

        # hidden activation and gate (FP16 → FP32)
        ptr_out  = out_ptr  + base_offset[:, None] + offs_k[None, :] * hidden_stride
        ptr_gate = gate_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride

        val_out  = tl.load(
            ptr_out,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)   # [M, K]

        val_gate = tl.load(
            ptr_gate,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)   # [M, K]

        # 2nd LayerNorm parameters for this hidden slice
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [K]
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [K]

        # ((out - mu) * rstd) * γ + β   →   * out_gate
        norm  = ((val_out - mu[:, None]) * rstd[:, None]) * gamma[None, :] + beta[None, :]
        gated = norm * val_gate                                   # [M, K]

        gated_fp16 = gated.to(tl.float16)

        # final linear projection weight tile (H × dim)
        w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]   # [K, TILE_N]
        w_tile = tl.load(
            w_ptr,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )                                                             # [K, TILE_N]

        acc += tl.dot(gated_fp16, w_tile).to(tl.float32)   # (M,K)·(K,N) → (M,N)

    # write final output  [B, N, N, dim] (row‑major)
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (gradient‑free).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, C] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
        - weights_dict : dict with keys
              "norm.weight", "norm.bias",
              "left_proj.weight", "right_proj.weight",
              "left_gate.weight", "right_gate.weight", "out_gate.weight",
              "to_out_norm.weight", "to_out_norm.bias",
              "to_out.weight"
        - config_dict  : {"dim": C, "hidden_dim": H, "nomask": bool, ...}

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, C] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]                # model dimension C
    hidden_dim = cfg["hidden_dim"]  # hidden dimension H
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, N2, _ = x.shape
    assert N == N2, "TriMul expects a square pairwise matrix"

    R = B * N * N      # total number of (b,i,k) rows
    H = hidden_dim
    C = dim

    # ------------------------------------------------------------------
    # 1️⃣  Fuse LayerNorm + 5 Linear projections (produces left/right/out_gate)
    # ------------------------------------------------------------------
    # LayerNorm parameters (γ/β)
    norm_weight = w["norm.weight"]
    norm_bias   = w["norm.bias"]

    # Transpose projection weights once (C × H, FP16)
    left_w_T     = w["left_proj.weight"].to(torch.float16).t().contiguous()
    right_w_T    = w["right_proj.weight"].to(torch.float16).t().contiguous()
    left_gate_w_T= w["left_gate.weight"].to(torch.float16).t().contiguous()
    right_gate_w_T= w["right_gate.weight"].to(torch.float16).t().contiguous()
    out_gate_w_T = w["out_gate.weight"].to(torch.float16).t().contiguous()

    # Allocate output tensors
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    # Optional mask handling
    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    # Kernel launch parameters
    BLOCK_R = 64
    BLOCK_H = 32
    BLOCK_C = 32

    proj_gate_fused_kernel[(triton.cdiv(R, BLOCK_R),)](
        x.reshape(R, C).contiguous(),  # x_ptr
        left,
        right,
        out_gate,
        mask_ptr,
        norm_weight,
        norm_bias,
        left_w_T,
        right_w_T,
        left_gate_w_T,
        right_gate_w_T,
        out_gate_w_T,
        R,
        H,
        N,
        C,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        BLOCK_C=BLOCK_C,
        EPS=eps,
        num_warps=8,
    )
    # ------------------------------------------------------------------
    # 2️⃣ Batched GEMM over hidden dimension (FP16, TensorCores)
    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)          # [B·H, N, N] FP16
    right_mat = right.view(B * H, N, N)         # [B·H, N, N] FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                # [B, H, N, N] FP16

    # ------------------------------------------------------------------
    # 3️⃣ Row‑wise mean & rsqrt‑variance of `out`
    # ------------------------------------------------------------------
    mu   = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 256
    BLOCK_K_STATS = 128                     # H ≤ 128 for all test configs
    rowwise_stats_kernel[(triton.cdiv(R, BLOCK_M_STATS),)](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 4️⃣ Fuse 2nd LayerNorm, out_gate multiplication & final projection
    # ------------------------------------------------------------------
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim] FP16

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 128    # rows per program
    BLOCK_K_OUT = 32     # hidden‑dim chunk
    BLOCK_N_OUT = 128    # output‑dim chunk

    grid_out = (
        triton.cdiv(R, BLOCK_M_OUT),
        triton.cdiv(dim, BLOCK_N_OUT),
    )
    final_output_kernel[grid_out](
        out,
        out_gate,
        mu,
        rstd,
        w["to_out_norm.weight"],   # γ (FP32)
        w["to_out_norm.bias"],     # β (FP32)
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=8,
    )
    return final
