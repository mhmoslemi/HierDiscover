def final_norm_gate_proj_kernel(
    out_ptr, gate_ptr,
    ln_w_ptr, ln_b_ptr,
    w_out_T_ptr,
    out_final_ptr,
    total_rows, H, dim,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr,
):
    pid_m = tl.program_id(0)  # rows
    pid_n = tl.program_id(1)  # output dim

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)
    offs_n = col_start + tl.arange(0, BLOCK_N)

    mask_m = offs_m < total_rows
    mask_n = offs_n < dim

    # Compute mean & variance across hidden dimension
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        # Load out and gate
        out_vals = tl.load(out_ptr + offs_m[:, None] * H + offs_k[None, :], mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
        sum_val += tl.sum(out_vals, axis=1)
        sum_sq += tl.sum(out_vals * out_vals, axis=1)

    mean = sum_val / H
    var = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # Now compute contribution to final output using dot product.
    # We'll accumulate over hidden dimension again.
    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        out_vals = tl.load(out_ptr + offs_m[:, None] * H + offs_k[None, :], mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
        gate_vals = tl.load(gate_ptr + offs_m[:, None] * H + offs_k[None, :], mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Normalize out_vals
        diff = (out_vals - mean[:, None]) * rstd[:, None]

        # Apply layernorm parameters
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)  # [K]
        beta = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)

        norm = diff * gamma[None, :] + beta[None, :]  # [M,K]
        term = norm * gate_vals  # [M,K]  (multiply gating)

        term_fp16 = term.to(tl.float16)  # cast

        w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]  # [K,N_tile]
        w_tile = tl.load(w_ptr, mask=mask_k[:, None] & mask_n[None, :], other=0.0)  # [K,N_tile]

        acc += tl.dot(term_fp16, w_tile).to(tl.float32)

    # store result
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])

    Outgoing TriMul operator forward pass using
    - Torch fused linear for the five projections
    - Triton for mask gating, split, and the final layernorm+projection.
    The pairwise tensor is processed in FP16 with an FP32
    layer‑norm. No gradients are computed.
    """
    import torch
    import torch.nn.functional as F
    import triton
    import triton.language as tl

    # unpack input...
    # 1. Input layernorm
    eps=1e-5
    R = bs * N * N
    # Convert to FP16 after layernorm
    x_flat = x.reshape(R, dim)
    # Use torch's layernorm (fast)
    x_norm = F.layer_norm(x_flat, (dim,), weight=w["norm.weight"], bias=w["norm.bias"], eps=eps).to(torch.float16)

    # 2. Fused linear for 5 projections
    fused_w = torch.cat([...], dim=0).to(torch.float16)
    proj = F.linear(x_norm, fused_w)  # (R, 5*H) FP16
    del fused_w, x_norm

    # 3. Split and apply mask & sigmoid gates using Triton
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if not nomask and mask is not None:
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)
    # launch split kernel
    BLOCK_R = 64
    BLOCK_H = 64
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj

    # 4. Batched GEMM
    left_mat = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [B*H,N,N] FP16
    out = out.view(B, H, N, N)

    out = out.permute(0, 2, 3, 1).contiguous()      # [B,N,N,H]
    out_gate = out_gate.permute(0, 2, 3, 1).contiguous()   # [B,N,N,H]

    # flatten
    out_flat = out.reshape(R, H)
    out_gate_flat = out_gate.reshape(R, H)
    del out, out_gate

    # 5. Final layernorm + gate + projection (single Triton kernel)
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()  # [H, dim]
    final = torch.empty(R, dim, dtype=torch.float32, device=device)

        BLOCK_N=BLOCK_N_OUT,
        num_warps=8,
    )
    # reshape back to [B,N,N,dim]
    final = final.view(B, N, N, dim)

"""
Outgoing TriMul (AlphaFold3) – forward only.

Algorithm
---------
1️⃣  Row‑wise LayerNorm on the input pairwise tensor (FP32 → FP16) using
    `torch.nn.functional.layer_norm` (highly‑optimized CUDA kernel).

2️⃣  Fuse the five linear maps (left, right, left_gate, right_gate, out_gate)
    into a single GEMM (`torch.nn.functional.linear`).  The result has shape
    (R, 5·H) where R = B·N·N and H = hidden_dim.

3️⃣  Split the fused projection, apply the optional mask, and compute the
    sigmoid gates in a Triton kernel
    `split_fuse_gate_kernel`.  The kernel writes *left*, *right* and *out_gate*
    tensors in layout [B, H, N, N] (FP16).

4️⃣  Batched GEMM (`torch.bmm`) computes the core einsum
    out[b, h, i, j] = Σₖ left[b, h, i, k]·right[b, h, j, k]  →  [B, H, N, N]
    (FP16, tensor‑cores).

5️⃣  Permute the contracted tensor so that the hidden dimension becomes the
    innermost one (layout [B, N, N, H]).  This enables a single Triton kernel
    `final_norm_gate_proj_kernel` to (a) compute the row‑wise mean/variance,
    (b) apply the second LayerNorm, (c) multiply by the out‑gate, and
    (d) project to the final `dim` using the transposed `to_out` weight.
    The kernel fuses all these steps, avoiding an extra global‑memory pass.

All kernels run on Triton 3.3.1, use mixed precision (FP32 for statistics,
FP16 for bulk work) and produce a float‑32 output tensor of shape
[B, N, N, dim].

No gradients are computed.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 2️⃣  Split fused projection, apply mask, compute sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # [R, 5·H]   FP16 fused projection
    left_ptr,          # [B, H, N, N] FP16
    right_ptr,         # [B, H, N, N] FP16
    out_gate_ptr,      # [B, H, N, N] FP16
    mask_ptr,          # [R]       FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)           # [BLOCK_R]
    mask_r = offs_r < R

    # Decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # Optional pair‑mask (broadcast over hidden dim)
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)          # [BLOCK_R]
        m_val = m_val[:, None]                                             # [BLOCK_R,1]
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + k   # start offset for hidden‑plane 0

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # Linear layout: [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]    # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # Sigmoid (computed in FP32, result stored as FP16)
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * m_val * left_gate
        right = right_raw * m_val * right_gate

        dst = base[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst, left,  mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst, right, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst, out_gate, mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 5️⃣  Fuse second LayerNorm, out‑gate and final projection
# ----------------------------------------------------------------------
@triton.jit
def final_norm_gate_proj_kernel(
    out_ptr,          # [R, H]   FP16 (contracted tensor)
    gate_ptr,         # [R, H]   FP16 (out_gate)
    ln_w_ptr,         # [H]      FP32 (to_out_norm.weight)
    ln_b_ptr,         # [H]      FP32 (to_out_norm.bias)
    w_out_T_ptr,      # [H, dim] FP16 (to_out weight transposed)
    out_final_ptr,    # [R, dim] FP32
    total_rows, H, dim,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr,
):
    pid_m = tl.program_id(0)   # rows of the flattened tensor
    pid_n = tl.program_id(1)   # output‑dim tiles

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)   # [BLOCK_M]
    offs_n = col_start + tl.arange(0, BLOCK_N)   # [BLOCK_N]

    mask_m = offs_m < total_rows
    mask_n = offs_n < dim

    # ------------------------------------------------------------------
    # 1️⃣  Compute row‑wise mean & variance (FP32)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)               # [BLOCK_K]
        mask_k = offs_k < H

        # out is stored contiguously as [row, hidden]
        vals = tl.load(
            out_ptr + offs_m[:, None] * H + offs_k[None, :],
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------------------------------------------------------
    # 2️⃣  Apply LayerNorm, out‑gate and project to `dim`
    # ------------------------------------------------------------------
    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)               # [BLOCK_K]
        mask_k = offs_k < H

        # Load contracted values and gate (both FP16 → FP32)
        out_vals = tl.load(
            out_ptr + offs_m[:, None] * H + offs_k[None, :],
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)

        gate_vals = tl.load(
            gate_ptr + offs_m[:, None] * H + offs_k[None, :],
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)

        # Normalise (out - µ)·rstd  →  γ· + β
        diff = (out_vals - mean[:, None]) * rstd[:, None]   # [M,K]

        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [K]
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [K]

        norm = diff * gamma[None, :] + beta[None, :]                # [M,K]

        # Apply out‑gate
        term = norm * gate_vals                                    # [M,K]
        term_fp16 = term.to(tl.float16)                            # fp16 for the dot

        # Load a slice of the final projection weight (H, dim)
        w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]   # [K, TILE_N]
        w_tile = tl.load(
            w_ptr,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )                                                             # [K, TILE_N]

        # Accumulate (M,K)·(K,N) → (M,N)  (fp16·fp16 → fp32)
        acc += tl.dot(term_fp16, w_tile).to(tl.float32)

    # Write final result (flattened first three dims → row “R”)
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul operator (no gradients).

    Args:
        data (tuple):
            - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
            - weights_dict : dict containing the model weights
            - config_dict  : must contain ``dim``, ``hidden_dim`` and ``nomask``

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # 0️⃣  Unpack inputs
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = bool(cfg.get("nomask", True))
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N            # total number of pairwise tokens
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣  Row‑wise LayerNorm (FP32 → FP16) – use the highly‑optimised
    #      torch implementation (faster than a custom Triton kernel)
    # --------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                # [R, dim] FP32
    x_norm = F.layer_norm(
        x_flat,
        (dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    ).to(torch.float16)                                     # [R, dim] FP16
    del x_flat

    # --------------------------------------------------------------
    # 2️⃣  Fuse five linear maps (left, right, left_gate, right_gate, out_gate)
    # --------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                      # [5·H, dim]  (FP16)

    proj = F.linear(x_norm, fused_w)          # (R, 5·H) FP16
    del fused_w, x_norm

    # --------------------------------------------------------------
    # 3️⃣  Split, apply optional mask, and compute sigmoid gates
    # --------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if not nomask and mask is not None:
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)

    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣  Batched GEMM over the hidden dimension (cuBLAS)
    # --------------------------------------------------------------
    # left/right are [B, H, N, N] – view as batch dimension B·H
    left_mat  = left.view(B * H, N, N)                      # FP16
    right_mat = right.view(B * H, N, N)                     # FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                             # [B, H, N, N] FP16

    # --------------------------------------------------------------
    # 5️⃣  Prepare tensors for the fused final kernel:
    #       - make hidden dimension innermost → [B, N, N, H]
    #       - flatten to [R, H]
    # --------------------------------------------------------------
    out = out.permute(0, 2, 3, 1).contiguous()            # [B, N, N, H]
    out_gate = out_gate.permute(0, 2, 3, 1).contiguous() # [B, N, N, H]

    out_flat      = out.view(R, H)        # FP16
    out_gate_flat = out_gate.view(R, H)   # FP16
    del out, out_gate

    # --------------------------------------------------------------
    # 6️⃣  Second LayerNorm + out_gate + final projection (single Triton kernel)
    # --------------------------------------------------------------
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim]

    final = torch.empty(R, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 128   # rows per program
    BLOCK_K_OUT = 32    # hidden‑dim chunk (128 = 4·32)
    BLOCK_N_OUT = 128   # output‑dim tile

    grid_out = (
        triton.cdiv(R, BLOCK_M_OUT),
        triton.cdiv(dim, BLOCK_N_OUT),
    )
    final_norm_gate_proj_kernel[grid_out](
        out_flat,
        out_gate_flat,
        w["to_out_norm.weight"],   # γ  FP32
        w["to_out_norm.bias"],     # β  FP32
        w_out_T,
        final,
        R,
        H,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=8,
    )
    # reshape back to [B, N, N, dim]
    final = final.view(B, N, N, dim)

    return final
