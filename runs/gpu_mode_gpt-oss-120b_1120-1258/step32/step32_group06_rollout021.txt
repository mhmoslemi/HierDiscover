byte‑wise work into Triton kernels:

1️⃣  Row‑wise LayerNorm on the flattened pairwise tensor (type FP32 → FP16).
2️⃣  Five linear maps (left, right, left_gate, right_gate, out_gate) are
    performed with a single torch.nn.functional.linear call.
3️⃣  A Triton kernel (`split_fuse_gate_kernel`) splits the concatenated
    projection, applies the optional pair mask, and evaluates the three
    sigmoid gates.
4️⃣  Per‑hidden‑plane matrix‑multiply:
       out[b, h, i, j] = Σₖ left[b, h, i, k]·right[b, h, j, k]
    realised with a batched cuBLAS GEMM (`torch.bmm`).
5️⃣  The final LayerNorm, multiplication by the output‑gate and the
    projection back to the model dimension are fused in a single Triton
    kernel (`final_output_fused_kernel`).  This kernel computes the mean /
    variance over the hidden dimension, normalises, applies the
    `to_out_norm` parameters, multiplies by `out_gate` and finally
    performs the hidden‑→‑dim linear projection.

All kernels use mixed precision: statistics are accumulated in FP32,
bulk work in FP16.  The function `custom_kernel` is the required entry
point.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣  Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,   # [R, dim]      input  (FP32)
    y_ptr,   # [R, dim]      output (FP16)
    w_ptr,   # [dim]         γ (FP32)
    b_ptr,   # [dim]         β (FP32)
    rows,    # scalar R
    dim,     # scalar dim
    eps,     # scalar eps
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # dim chunk per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ----- first pass (mean / var) -----
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]  # [M,D]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)          # [M]
        sum_sq  += tl.sum(vals * vals, axis=1)   # [M]

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ----- second pass (write normalized values) -----
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]  # [M,D]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)          # [M,D]

        norm = (vals - mean[:, None]) * rstd[:, None]      # [M,D]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)  # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)  # FP32

        out = norm * gamma[None, :] + beta[None, :]            # FP32
        tl.store(
            y_ptr + offs_m[:, None] * dim + offs_d[None, :],
            out.to(tl.float16),
            mask=mask_m[:, None] & mask_d[None, :],
        )


# ----------------------------------------------------------------------
# 2️⃣  Split, mask, and compute three sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,      # [R, 5·H]   (FP16 fused projection)
    left_ptr,      # [B, H, N, N] (FP16)
    right_ptr,     # [B, H, N, N] (FP16)
    out_gate_ptr,  # [B, H, N, N] (FP16)
    mask_ptr,      # [R]        (FP16, optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index -> (b, i, k)
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask (broadcast over hidden dimension)
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        m_val = m_val[:, None]                                      # [BLOCK_R,1]
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # stride inside the left/right/out_gate tensors
    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + k    # start address for h = 0

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # layout of proj: [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [R,H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # sigmoid in FP32 for stability, cast back to FP16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * m_val * left_gate
        right = right_raw * m_val * right_gate
        outg  = out_gate

        dst = base[:, None] + offs_h[None, :] * hidden_stride   # [R,H]

        tl.store(left_ptr + dst,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst,
                 outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 5️⃣+6️⃣  Final LayerNorm + out‑gate + projection (fused)
# ----------------------------------------------------------------------
@triton.jit
def final_output_fused_kernel(
    out_ptr,        # [B, H, N, N]      (FP16) – result of left·rightᵀ
    gate_ptr,       # [B, H, N, N]      (FP16) – out_gate
    ln_w_ptr,       # [H]               (FP32) – to_out_norm.weight
    ln_b_ptr,       # [H]               (FP32) – to_out_norm.bias
    w_out_T_ptr,    # [H, dim]          (FP16) – to_out weight transposed
    out_final_ptr,  # [B, N, N, dim]    (FP32) – final tensor
    total_rows, H, N, dim,
    eps,            # scalar (float32)
    BLOCK_M: tl.constexpr,    # rows per program (tiles over B·N·N)
    BLOCK_K: tl.constexpr,    # hidden‑dim chunk
    BLOCK_N: tl.constexpr,    # output‑dim chunk
):
    pid_m = tl.program_id(0)   # row tile
    pid_n = tl.program_id(1)   # output‑dim tile

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)   # [BLOCK_M]
    offs_n = col_start + tl.arange(0, BLOCK_N)   # [BLOCK_N]

    mask_m = offs_m < total_rows
    mask_n = offs_n < dim

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + j   # start address for h = 0

    # --------------------------------------------------------------
    # 1️⃣ Compute mean & rsqrt‑variance over the hidden dimension
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)    # [BLOCK_K]
        mask_k = offs_k < H

        ptr = out_ptr + base[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)                              # [M,K]

        sum_val += tl.sum(vals, axis=1)               # [M]
        sum_sq  += tl.sum(vals * vals, axis=1)        # [M]

    mu   = sum_val / H
    var  = sum_sq / H - mu * mu
    rstd = tl.rsqrt(var + eps)

    # --------------------------------------------------------------
    # 2️⃣ Normalise, apply to_out_norm, multiply by out_gate,
    #    then project to the model dimension.
    # --------------------------------------------------------------
    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)    # [BLOCK_K]
        mask_k = offs_k < H

        # load hidden‑plane values and the corresponding out‑gate
        ptr_out  = out_ptr  + base[:, None] + offs_k[None, :] * hidden_stride
        ptr_gate = gate_ptr + base[:, None] + offs_k[None, :] * hidden_stride

        vals_out  = tl.load(
            ptr_out,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)                               # [M,K]

        vals_gate = tl.load(
            ptr_gate,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)                               # [M,K]

        # LayerNorm parameters for the current hidden slice
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)  # [K] FP32
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)  # [K] FP32

        # (out - μ)·rstd → γ· + β → * out_gate
        diff = (vals_out - mu[:, None]) * rstd[:, None]          # [M,K]
        norm = diff * gamma[None, :] + beta[None, :]             # [M,K]
        term = norm * vals_gate                                   # [M,K]

        term_fp16 = term.to(tl.float16)                          # cast back for dot

        # load a tile of the final projection matrix: shape [K, TILE_N]
        w_tile = tl.load(
            w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :],
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )                                                        # [K, TILE_N]

        # accumulate (M,K)·(K,N) → (M,N)
        acc += tl.dot(term_fp16, w_tile).to(tl.float32)

    # write final output, layout [B, N, N, dim]
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entrypoint
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only outgoing TriMul.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor[B, N, N, dim]  (float32)
        - mask_tensor  : torch.Tensor[B, N, N]      (bool/float) or None
        - weights_dict : mapping with the following keys
            "norm.weight", "norm.bias",
            "left_proj.weight", "right_proj.weight",
            "left_gate.weight", "right_gate.weight", "out_gate.weight",
            "to_out_norm.weight", "to_out_norm.bias",
            "to_out.weight"
        - config_dict must contain
            "dim" (model dimension), "hidden_dim" (hidden dimension),
            "nomask" (bool, optional)

    Returns
    -------
    torch.Tensor[B, N, N, dim] (float32) – the TriMul output.
    """
    # --------------------------------------------------------------
    # unpack arguments
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = bool(cfg.get("nomask", True))
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                 # number of pairwise tokens
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣  Row‑wise LayerNorm (FP32 → FP16)
    # --------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()               # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # --------------------------------------------------------------
    # 2️⃣  Fuse the five linear maps (left, right, gates)
    # --------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                   # [5·H, dim]

    proj = F.linear(x_norm, fused_w)                       # (R, 5·H) FP16
    del fused_w, x_norm

    # --------------------------------------------------------------
    # 3️⃣  Split, apply mask and sigmoid gates
    # --------------------------------------------------------------
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if not nomask and mask is not None:
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣  Core bilinear reduction (left · rightᵀ) – batched GEMM
    # --------------------------------------------------------------
    left_mat = left.view(B * H, N, N)                      # [B·H, N, N] FP16
    right_mat = right.view(B * H, N, N)                    # [B·H, N, N] FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                             # [B, H, N, N] FP16

    # --------------------------------------------------------------
    # 5️⃣+6️⃣  Final LayerNorm + out_gate multiplication + projection
    # --------------------------------------------------------------
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()  # [H, dim]

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 128          # rows per program (tiles over B·N·N)
    BLOCK_K_OUT = 64           # hidden‑dim chunk
    BLOCK_N_OUT = 128          # output‑dim chunk

    grid_out = (
        triton.cdiv(R, BLOCK_M_OUT),
        triton.cdiv(dim, BLOCK_N_OUT),
    )
    final_output_fused_kernel[grid_out](
        out,
        out_gate,
        w["to_out_norm.weight"],      # γ  (FP32)
        w["to_out_norm.bias"],        # β  (FP32)
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=8,
    )

    return final
