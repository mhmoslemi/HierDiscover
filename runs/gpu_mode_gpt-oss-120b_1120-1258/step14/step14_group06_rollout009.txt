"""
custom_kernel – Fast “outgoing” TriMul (AlphaFold3) forward pass

The implementation follows the reference PyTorch module but fuses the
expensive work into a few Triton kernels:

1️⃣  Row‑wise LayerNorm on the flattened input (fp32 → fp16)  
2️⃣  A single GEMM performs the five linear maps (left/right projections
    + three gates) and stores the result as a contiguous fp16 tensor.  
3️⃣  A Triton kernel splits that fused tensor, applies the pair‑wise mask,
    evaluates the sigmoid gates and writes the three tensors
    (left, right, out_gate) in the layout (B, H, N, N).  
4️⃣  Batched half‑precision GEMM (cuBLAS) contracts the left and right
    tensors over the sequence dimension *k*:  
        out[b·H+i] = left[b·H+i] @ right[b·H+i]ᵀ    → (B·H, N, N).  
5️⃣  Row‑wise mean and rsqrt‑variance of *out* are computed by a second
    Triton kernel (statistics needed for the second LayerNorm).  
6️⃣  The last Triton kernel fuses the second LayerNorm, the element‑wise
    multiplication with ``out_gate`` and the final linear projection
    (hidden → dim).  For each position (b,i,j) it loads the hidden slice,
    normalises it with the statistics from step 5, applies the γ/β
    parameters of ``to_out_norm``, multiplies by the gate, and performs a
    Tensor‑Core dot with the transposed output weight.  The result is
    written as a dense float‑32 tensor of shape [B, N, N, dim].

All kernels are written for Triton 3.3.1 and tuned for the NVIDIA H100
(128‑thread blocks, 4 warps).  The forward pass returns a float‑32
tensor and does **not** compute any gradients.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (float32 → float16)
# --------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,    # const   *[rows, dim] (float32)
    y_ptr,    # mutable *[rows, dim] (float16)
    w_ptr,    # const   *[dim]      (float32) γ
    b_ptr,    # const   *[dim]      (float32) β
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # ---- accumulate sum and sum‑of‑squares (fp32) -----------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- write normalised values ---------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]          # fp32
        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)    # γ (fp32)
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)    # β (fp32)

        out = norm * w[None, :] + b[None, :]                   # fp32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

# --------------------------------------------------------------
# 2️⃣ Split fused projection, optional mask, and sigmoid gates
# --------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[R, 5*H]    (fp16)
    left_ptr,          # mutable *[B, H, N, N] (fp16)
    right_ptr,         # mutable *[B, H, N, N] (fp16)
    out_gate_ptr,      # mutable *[B, H, N, N] (fp16)
    mask_ptr,          # const   *[R]        (fp16) – ignored if HAS_MASK=0
    R: tl.int32,
    H: tl.int32,
    N: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # decode (b,i,k) from flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask (broadcast across hidden dimension)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                   # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N                               # stride for hidden dim in (B,H,N,N)
    base_offset = b * (H * N * N) + i * N + k           # offset for h=0

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # [R, H] view of the fused projection (5 groups of H)
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [R, H]

        left_raw      = tl.load(proj_ptr + proj_base,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_raw     = tl.load(proj_ptr + proj_base + H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        left_gate_raw = tl.load(proj_ptr + proj_base + 2 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_gate_raw= tl.load(proj_ptr + proj_base + 3 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        out_gate_raw  = tl.load(proj_ptr + proj_base + 4 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)

        # sigmoid (fp32 for stability, back to fp16)
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate

        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride   # [R, H]

        tl.store(left_ptr     + dst_offset,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr    + dst_offset,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset,
                 out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])

# --------------------------------------------------------------
# 3️⃣ Row‑wise mean & rsqrt‑variance of contracted tensor
# --------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr,      # const   *[B, H, N, N] (fp16)
    mu_ptr,       # mutable *[R]          (float32)
    rstd_ptr,     # mutable *[R]          (float32)
    total_rows: tl.int32,   # = B * N * N
    H: tl.int32,
    N: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * N * N) + i * N + j          # offset of h=0

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H
        ptr = out_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_m, mean,  mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)

# --------------------------------------------------------------
# 4️⃣ Fuse second LayerNorm, out_gate and final projection
# --------------------------------------------------------------
@triton.jit
def final_output_kernel(
    out_ptr,          # const *[B, H, N, N] (fp16)
    gate_ptr,         # const *[B, H, N, N] (fp16)   (out_gate)
    mu_ptr,           # const *[R]            (float32)
    rstd_ptr,         # const *[R]            (float32)
    ln_w_ptr,         # const *[H]            (float32) γ of to_out_norm
    ln_b_ptr,         # const *[H]            (float32) β of to_out_norm
    w_out_ptr,        # const *[H, dim]       (fp16)   (H×dim, transposed weight)
    out_final_ptr,    # mutable *[B, N, N, dim] (float32)
    total_rows: tl.int32,
    H: tl.int32,
    N: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    pid_m = tl.program_id(0)   # rows tile
    pid_n = tl.program_id(1)   # dim tile

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    offs_n = col_start + tl.arange(0, BLOCK_N)          # [BLOCK_N]

    mask_m = offs_m < total_rows
    mask_n = offs_n < dim

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * N * N) + i * N + j          # offset of h=0

    # per‑position mean / rsqrt‑var
    mu   = tl.load(mu_ptr   + offs_m, mask=mask_m, other=0.0)   # [M]
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)   # [M]

    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        # load contracted tensor and gate
        ptr_out  = out_ptr  + base_offset[:, None] + offs_k[None, :] * hidden_stride
        ptr_gate = gate_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride

        out_vals  = tl.load(ptr_out,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)   # [M, K]
        gate_vals = tl.load(ptr_gate,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)   # [M, K]

        # LayerNorm parameters for the hidden slice k
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [K] fp32
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [K] fp32

        # (out - μ) * rstd → normalise, then γ/β, then multiply by gate
        diff = (out_vals - mu[:, None]) * rstd[:, None]              # [M, K]
        norm = diff * gamma[None, :] + beta[None, :]                 # [M, K]
        term = norm * gate_vals                                      # [M, K] fp32
        term_fp16 = term.to(tl.float16)                              # cast for dot

        # load weight tile (H×dim) – contiguous over hidden dimension
        w_ptr = w_out_ptr + offs_k[:, None] * dim + offs_n[None, :]
        w_tile = tl.load(w_ptr,
                         mask=mask_k[:, None] & mask_n[None, :],
                         other=0.0)                                   # [K, N]

        # accumulate using Tensor‑Core dot (fp16×fp16 → fp32)
        acc += tl.dot(term_fp16, w_tile).to(tl.float32)

    # write final result: (B, N, N, dim) → flattened as (R, dim)
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, acc,
             mask=mask_m[:, None] & mask_n[None, :])

# --------------------------------------------------------------
# Custom forward entry point
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
            - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
            - weights_dict : mapping with all Linear/LayerNorm parameters
                             (float32 tensors, bias=False for Linear)
            - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # 0️⃣ Unpack arguments
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N           # total (b,i,j) positions
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (fp32 → fp16)
    # --------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()               # [R, dim] fp32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],   # γ (float32)
        w["norm.bias"],     # β (float32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    del x_flat

    # --------------------------------------------------------------
    # 2️⃣ Fused projection (left/right + three gates) – single GEMM
    # --------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)           # shape [5·H, dim]

    # Linear without bias: (R, dim) @ (dim, 5·H).T → (R, 5·H)
    proj = F.linear(x_norm, fused_w)   # fp16
    del fused_w, x_norm

    # --------------------------------------------------------------
    # 3️⃣ Split, apply optional mask, sigmoid gates
    # --------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if cfg.get("nomask", True) or mask is None:
        has_mask = 0
        dummy_mask = torch.empty(0, dtype=torch.float16, device=device)
        mask_ptr = dummy_mask
    else:
        has_mask = 1
        mask_flat = mask.to(torch.float16).reshape(R)   # values 0/1
        mask_ptr = mask_flat

    BLOCK_R = 128
    BLOCK_H = 32
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣ Batched GEMM (cuBLAS) over the hidden dimension
    # --------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)          # (B·H, N, N)
    right_mat = right.view(B * H, N, N)         # (B·H, N, N)

    # out[b·H+i] = left[b·H+i] @ right[b·H+i]ᵀ
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B·H, N, N)
    out = out.view(B, H, N, N)                             # (B, H, N, N)

    # --------------------------------------------------------------
    # 5️⃣ Row‑wise mean & rsqrt‑variance of `out`
    # --------------------------------------------------------------
    mu   = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 128
    BLOCK_K_STATS = 32
    grid_stats = (triton.cdiv(R, BLOCK_M_STATS),)
    rowwise_stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 6️⃣ Fuse second LayerNorm, out_gate and final linear projection
    # --------------------------------------------------------------
    # Weight for the final Linear (hidden → dim) is stored transposed
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim]

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 64          # rows per program (positions)
    BLOCK_K_OUT = 32          # hidden‑dim chunk
    BLOCK_N_OUT = 64          # output‑dim chunk
    grid_out = (
        triton.cdiv(R, BLOCK_M_OUT),
        triton.cdiv(dim, BLOCK_N_OUT),
    )
    final_output_kernel[grid_out](
        out,
        out_gate,
        mu,
        rstd,
        w["to_out_norm.weight"],   # γ (float32)
        w["to_out_norm.bias"],     # β (float32)
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=4,
    )
    # ------------------------------------------------------------------
    # Return result in the expected shape
    # ------------------------------------------------------------------
    return final
