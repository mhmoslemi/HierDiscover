"""
TriMul – outgoing version (AlphaFold‑3) forward pass

The implementation follows the reference PyTorch module but removes all
unnecessary memory traffic:

*   The first LayerNorm is performed by a custom Triton kernel and writes the
    result directly in fp16.
*   The four linear projections (left/right and their gates) are merged into a
    single fp16 GEMM (`torch.nn.functional.linear`) by concatenating the weight
    matrices.
*   The cubic multiplicative update is expressed as a batched matrix‑multiply
    (`torch.bmm`) with the hidden dimension folded into the batch dimension.
*   The second LayerNorm, the output‑gate and the final linear projection are
    fused into a second Triton kernel.  This kernel computes per‑row
    mean/variance, applies the learned scale/bias, multiplies by the output‑gate
    and writes the final `dim`‑dimensional result directly – no intermediate
    tensor is materialised.

All heavy arithmetic is carried out in fp16 (the kernels promote to fp32 only
where higher precision is required).  The code works for the configurations
used in the tests (B∈{1,2}, N∈{128,…,1024}, dim ∈ {128,384},
hidden_dim = 128) and supports optional binary masks.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (float32 → float16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,            # const  * [R, dim]   (float32)
    y_ptr,            # mut    * [R, dim]   (float16)
    weight_ptr,       # const  * [dim]      (float32)
    bias_ptr,         # const  * [dim]      (float32)
    R,                # total rows (int)
    dim,              # feature dim (int)
    eps,              # epsilon (float)
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """
    Row‑wise LayerNorm.
    y = ((x - μ) / √(σ²+eps)) * γ + β
    Output is stored as float16.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)                # [BLOCK_M]
    mask_m = offs_m < R

    # ------------------------------------------------------------------
    # 1️⃣  Compute per‑row mean / variance in fp32
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    # iterate over the feature dimension
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < dim

        x_ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(x_ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)           # fp32 load

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                     # 1 / √(var+eps)

    # ------------------------------------------------------------------
    # 2️⃣  Write normalised values (fp16)
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        x_ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(x_ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # β
        out = norm * w[None, :] + b[None, :]                      # fp32
        out_fp16 = out.to(tl.float16)

        y_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(y_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2️⃣ LayerNorm + output‑gate + final linear (fused)
# ----------------------------------------------------------------------
@triton.jit
def ln_gate_linear_kernel(
    inp_ptr,          # const  * [R, H]        (fp16)  ← output of cubic mul
    gate_ptr,         # const  * [R, H]        (fp16)  ← out_gate
    wnorm_ptr,        # const  * [H]           (float32)  ← to_out_norm.weight
    bnorm_ptr,        # const  * [H]           (float32)  ← to_out_norm.bias
    wlin_ptr,         # const  * [H, D]        (fp16)  ← to_out.weight.T
    out_ptr,          # mut    * [R, D]        (fp16)
    R,                # total rows (int)
    H,                # hidden dimension (int)
    D,                # output dimension (int)
    eps,              # epsilon (float)
    BLOCK_M: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """
    Row‑wise LayerNorm + element‑wise gate + linear projection.
    Computes for each row r:

    norm = (x - μ) / √(σ²+eps)           (μ,σ² from the row)
    norm = norm * γ + β                 (γ,β = wnorm,bnorm)
    gated = norm * gate
    out   = gated @ W                     (W = to_out.weight.T)

    All arithmetic is fp32 inside the kernel; the result is stored as fp16.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)                # [BLOCK_M]
    mask_m = offs_m < R

    # ------------------------------------------------------------------
    # 1️⃣  Compute mean / variance across H (fp32)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)      # [BLOCK_H]
        mask_h = offs_h < H

        inp_ptrs = inp_ptr + offs_m[:, None] * H + offs_h[None, :]
        vals = tl.load(inp_ptrs,
                       mask=mask_m[:, None] & mask_h[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------------------------------------------------------
    # 2️⃣  Produce final output in blocks of D columns
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(D, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < D

        # accumulator for this column block
        acc = tl.zeros([BLOCK_M, BLOCK_D], dtype=tl.float32)

        # loop over hidden dimension blocks
        for h_off in range(0, tl.cdiv(H, BLOCK_H)):
            offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)  # [BLOCK_H]
            mask_h = offs_h < H

            # ------------------- load & normalize --------------------
            inp_vals = tl.load(inp_ptr + offs_m[:, None] * H + offs_h[None, :],
                               mask=mask_m[:, None] & mask_h[None, :],
                               other=0.0).to(tl.float32)

            norm = (inp_vals - mean[:, None]) * rstd[:, None]   # (BM, BH)

            # ------------------- load LN parameters --------------------
            gamma = tl.load(wnorm_ptr + offs_h, mask=mask_h, other=0.0)   # (BH)
            beta  = tl.load(bnorm_ptr + offs_h, mask=mask_h, other=0.0)   # (BH)

            # ------------------- load gate ----------------------------
            gate_vals = tl.load(gate_ptr + offs_m[:, None] * H + offs_h[None, :],
                                mask=mask_m[:, None] & mask_h[None, :],
                                other=0.0).to(tl.float32)

            # combined = ((norm * γ) + β) * gate
            combined = (norm * gamma[None, :] + beta[None, :]) * gate_vals   # (BM, BH)

            # ------------------- load linear weight --------------------
            w_ptrs = wlin_ptr + offs_h[:, None] * D + offs_d[None, :]  # (BH, BD)
            w = tl.load(w_ptrs,
                        mask=mask_h[:, None] & mask_d[None, :],
                        other=0.0).to(tl.float32)                     # fp32

            # ------------------- accumulate ----------------------------
            acc += tl.dot(combined, w)                               # (BM, BD)

        # store result
        out_ptrs = out_ptr + offs_m[:, None] * D + offs_d[None, :]
        tl.store(out_ptrs, acc.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom forward for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]  (float32)
        - mask_tensor  : torch.Tensor [B, N, N]      (bool/float, optional)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain "dim", "hidden_dim", "nomask"

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    inp, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    device = inp.device

    B, N, _, _ = inp.shape
    R = B * N * N                                 # total rows for 2‑D view
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → fp16
    # ------------------------------------------------------------------
    inp_flat = inp.reshape(R, dim).contiguous()   # [R, dim] float32
    norm_out = torch.empty(R, dim,
                           dtype=torch.float16,
                           device=device)

    BLOCK_M = 256              # rows per program (tune if needed)
    BLOCK_D = 128              # feature‑dim block; covers dim up to 384
    grid = (triton.cdiv(R, BLOCK_M),)

    layernorm_kernel[grid](
        inp_flat,
        norm_out,
        weights["norm.weight"],          # γ  (float32)
        weights["norm.bias"],            # β  (float32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 2️⃣ 4× Linear projections (left/right & their gates) – fused GEMM
    # ------------------------------------------------------------------
    # concatenate the four weight matrices (each [H, dim])
    w_4h = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                         # [4*H, dim]

    proj_gate = F.linear(norm_out, w_4h)        # [R, 4*H] fp16

    left_proj_raw  = proj_gate[:, 0:H]
    right_proj_raw = proj_gate[:, H:2*H]
    left_gate_raw  = proj_gate[:, 2*H:3*H]
    right_gate_raw = proj_gate[:, 3*H:4*H]

    # out_gate is a separate linear projection
    out_gate_raw = F.linear(
        norm_out,
        weights["out_gate.weight"].to(torch.float16)   # [H, dim]
    )                                                   # [R, H] fp16

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask, sigmoid gates and element‑wise gating
    # ------------------------------------------------------------------
    # apply sigmoid in‑place
    left_gate = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate = out_gate_raw.sigmoid_()            # [R, H] fp16

    if not cfg.get("nomask", True):
        # mask shape [B,N,N] → flatten to [R,1] and broadcast
        mask_flat = mask.reshape(R, 1).to(torch.float16)
        left = left_proj_raw * left_gate * mask_flat
        right = right_proj_raw * right_gate * mask_flat
    else:
        left = left_proj_raw * left_gate
        right = right_proj_raw * right_gate

    # ------------------------------------------------------------------
    # 4️⃣ Reshape tensors to 4‑D layout
    # ------------------------------------------------------------------
    left = left.view(B, N, N, H)      # [B, N, N, H] fp16
    right = right.view(B, N, N, H)    # [B, N, N, H] fp16
    out_gate = out_gate.view(B, N, N, H)  # [B, N, N, H] fp16

    # ------------------------------------------------------------------
    # 5️⃣ Cubic multiplicative update (batched GEMM)
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    # bring 'H' into the batch dimension
    left_mat = left.permute(0, 3, 1, 2).reshape(B * H, N, N)      # [B*H, N, N]
    right_mat = right.permute(0, 3, 1, 2).reshape(B * H, N, N)    # [B*H, N, N]

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))      # [B*H, N, N] fp16

    # restore 4‑D layout (still fp16)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1)           # [B, N, N, H] fp16

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm + output‑gate + final linear (fused)
    # ------------------------------------------------------------------
    out_flat = out.reshape(R, H).contiguous()          # [R, H] fp16
    out_gate_flat = out_gate.reshape(R, H).contiguous()

    # pre‑transpose the final linear weight so that we can treat it as [H, D]
    w_out_T = weights["to_out.weight"].t().contiguous().to(torch.float16)  # [H, dim]

    final_fp16 = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_H = 64                     # hidden‑dim block size (tune if needed)
    BLOCK_D = 128                    # output‑dim block size
    grid2 = (triton.cdiv(R, BLOCK_M),)

    ln_gate_linear_kernel[grid2](
        out_flat,
        out_gate_flat,
        weights["to_out_norm.weight"],   # γ  (float32)
        weights["to_out_norm.bias"],     # β  (float32)
        w_out_T,
        final_fp16,
        R,
        H,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_H=BLOCK_H,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 7️⃣ Cast back to float32 and reshape to original 4‑D shape
    # ------------------------------------------------------------------
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
