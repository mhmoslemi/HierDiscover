import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def elementwise_mul_kernel(a_ptr, b_ptr, out_ptr, size, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    start = pid * BLOCK
    offs = start + tl.arange(0, BLOCK)
    mask = offs < size
    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    tl.store(out_ptr + offs, a * b, mask=mask)

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator, optimized with
    PyTorch fused linear layers and a tiny Triton kernel for the final
    element‑wise gating. The implementation follows the reference
    ``TriMul`` module: LayerNorm → four linear projections (left/right
    and gates) → optional mask → sigmoid gates → batched matrix
    multiplication (N×N) → LayerNorm on hidden dim → multiply by
    output‑gate → final linear projection. All heavy matrix ops use
    FP16 and cuBLAS, while the only Triton kernel performs the
    out_norm * out_gate multiplication.
    """
    # Unpack inputs
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)
    device = input_tensor.device

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N  # R

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP16) – use PyTorch built‑in for speed
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(torch.float16, device=device)
    norm_b = weights["norm.bias"].to(torch.float16, device=device)

    # Cast input to half and apply LN over the last dimension
    x_norm = F.layer_norm(
        input_tensor.half(),
        normalized_shape=(dim,),
        weight=norm_w,
        bias=norm_b,
        eps=eps,
    )  # [B, N, N, dim]  (FP16)

    # ------------------------------------------------------------------
    # 2️⃣ Fuse left/right projections and their gates in one GEMM
    # ------------------------------------------------------------------
    w_left = weights["left_proj.weight"].to(torch.float16, device=device)   # [H, dim]
    w_right = weights["right_proj.weight"].to(torch.float16, device=device) # [H, dim]
    w_lgate = weights["left_gate.weight"].to(torch.float16, device=device) # [H, dim]
    w_rgate = weights["right_gate.weight"].to(torch.float16, device=device) # [H, dim]

    # Concatenate for a single matmul: (R, dim) @ (4*H, dim).T → (R, 4*H)
    w_concat = torch.cat([w_left, w_right, w_lgate, w_rgate], dim=0)  # [4H, dim]
    proj_gate = F.linear(x_norm.view(total_rows, dim), w_concat)       # [R, 4H] FP16

    H = hidden_dim
    left_proj_raw = proj_gate[:, 0:H]               # [R, H]
    right_proj_raw = proj_gate[:, H:2 * H]          # [R, H]
    left_gate_raw = proj_gate[:, 2 * H:3 * H]       # [R, H]
    right_gate_raw = proj_gate[:, 3 * H:4 * H]      # [R, H]

    # ------------------------------------------------------------------
    # 3️⃣ Output‑gate projection (single GEMM)
    # ------------------------------------------------------------------
    w_out_gate = weights["out_gate.weight"].to(torch.float16, device=device)  # [H, dim]
    out_gate_raw = F.linear(x_norm.view(total_rows, dim), w_out_gate)         # [R, H] FP16

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask + sigmoid gates
    # ------------------------------------------------------------------
    if not nomask:
        mask_h = mask_tensor.to(torch.float16, device=device).reshape(total_rows, 1)
        left_proj_raw = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    left_gate = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate = out_gate_raw.sigmoid_()

    left = left_proj_raw * left_gate     # [R, H] FP16
    right = right_proj_raw * right_gate  # [R, H] FP16

    # ------------------------------------------------------------------
    # 5️⃣ Batched matrix multiplication (the “tri‑mul”)
    # ------------------------------------------------------------------
    # Reshape to (B, N, N, H) → (B, H, N, N) → (B·H, N, N)
    left = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    left_mat = left.view(B * H, N, N)   # [B·H, N, N]
    right_mat = right.view(B * H, N, N) # [B·H, N, N]

    # (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Back to original layout – now [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 6️⃣ Hidden‑dim LayerNorm + multiplication by out_gate
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16, device=device)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16, device=device)

    out_norm = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps
    )  # [B, N, N, H] FP16

    # Fuse the final gating with a tiny Triton kernel (elementwise mul)
    out_norm_flat = out_norm.view(-1)   # total_rows * H elements
    out_gate_flat = out_gate.view(-1)
    gated_flat = torch.empty_like(out_norm_flat)

    total_elems = out_norm_flat.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total_elems, BLOCK),)
    elementwise_mul_kernel[grid](
        out_norm_flat,
        out_gate_flat,
        gated_flat,
        total_elems,
        BLOCK=BLOCK,
        num_warps=4,
    )
    out_norm_gated = gated_flat.view(B, N, N, H)   # [B, N, N, H] FP16

    # ------------------------------------------------------------------
    # 7️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    w_to_out = weights["to_out.weight"].to(torch.float16, device=device)  # [dim, H]
    final_half = F.linear(out_norm_gated.view(total_rows, H), w_to_out)   # [R, dim] FP16

    # Cast to FP32 as required by the reference implementation
    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final

    # --------------------------------------------------------------
    w_out_gate = weights["out_gate.weight"].to(torch.float16, device=device)   # [H, dim]
    out_gate_raw = F.linear(x_norm.view(total_rows, dim), w_out_gate)          # [R, H] (FP16)

    # --------------------------------------------------------------
    # 4️⃣ Mask (if any) + sigmoid gates
    # --------------------------------------------------------------
    if not nomask:
        mask_h = mask_tensor.to(torch.float16, device=device).reshape(total_rows, 1)
        left_proj_raw  = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    left  = left_proj_raw * left_gate          # [R, H] (FP16)
    right = right_proj_raw * right_gate        # [R, H] (FP16)

    # --------------------------------------------------------------
    # 5️⃣ Tri‑multiplicative update (batched GEMM)
    # --------------------------------------------------------------
    # Reshape to (B, H, N, N) → (B·H, N, N) for a simple bmm
    left  = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    left_mat  = left.view(B * H, N, N)    # [B·H, N, N]
    right_mat = right.view(B * H, N, N)   # [B·H, N, N]

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [B·H, N, N]

    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    # --------------------------------------------------------------
    # 6️⃣ Hidden‑dim LayerNorm + element‑wise gating (Triton kernel)
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16, device=device)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16, device=device)

    out_norm = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )                                          # [B, N, N, H] (FP16)

    # Triton kernel: out_norm * out_gate  (both [R, H] FP16)
    out_norm_flat = out_norm.view(-1)
    out_gate_flat = out_gate.view(-1)
    gated_flat = torch.empty_like(out_norm_flat)

    total_elems = out_norm_flat.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total_elems, BLOCK),)
    elementwise_mul_kernel[grid](
        out_norm_flat,
        out_gate_flat,
        gated_flat,
        total_elems,
        BLOCK=BLOCK,
        num_warps=4,
    )
    out_norm_gated = gated_flat.view(B, N, N, H)   # [B, N, N, H] (FP16)

    # --------------------------------------------------------------
    # 7️⃣ Final projection back to `dim`
    # --------------------------------------------------------------
    w_to_out = weights["to_out.weight"].to(torch.float16, device=device)   # [dim, H]
    final_half = F.linear(out_norm_gated.view(total_rows, H), w_to_out)    # [R, dim] (FP16)

    # Cast to FP32 – this matches the reference implementation’s output dtype
    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
