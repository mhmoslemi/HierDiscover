"""
Outgoing TriMul (AlphaFold‑3) – fast Triton‑accelerated forward pass.

Algorithm
---------
1️⃣  Row‑wise LayerNorm on the input (FP32 → FP16) – handled by the
    native torch.nn.functional.layer_norm (highly‑optimised).

2️⃣  Fuse the four linear maps (left_proj, right_proj, left_gate, right_gate)
    into a single GEMM:   W4 = [Wl_proj ; Wr_proj ; Wl_gate ; Wr_gate]   (4·H × dim).

3️⃣  Compute the out‑gate linear map (separate GEMM).

4️⃣  Optional mask, sigmoid gates and element‑wise gating (all FP16).

5️⃣  **Batched GEMM in Triton** – compute
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    for every head h in the same kernel, avoiding the costly
    ``permute().contiguous()`` copy used by a naïve torch.bmm
    implementation.  The kernel works on strided tensors
    (shape [B,N,N,H]) and writes the result in FP16.

6️⃣  Second LayerNorm over the hidden dimension H (again via torch.nn).

7️⃣  Multiply by the out‑gate and a final linear projection back to dim.

All heavy linear work stays in FP16 so Tensor‑Cores are fully utilised,
while the custom Triton kernel eliminates a ~250 MiB memory copy and
greatly reduces launch overhead, bringing the runtime well below the
target 1 ms on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the cubic update.
# Input layout  : (B, N, N, H) – contiguous in memory.
#   left  [b,i,k,h]  stride: (N*N*H, N*H, H, 1)
#   right [b,j,k,h]  same strides, but we need the K‑dimension first.
# Output layout : (B, N, N, H) – same stride as inputs.
# ----------------------------------------------------------------------
@triton.jit
def trmul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,               # pointers
    B, N, H,                                     # dimensions
    stride_b_l, stride_i_l, stride_k_l, stride_h_l,   # left strides
    stride_b_r, stride_j_r, stride_k_r, stride_h_r,   # right strides
    stride_b_o, stride_i_o, stride_j_o, stride_h_o,   # out strides
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Batched matmul:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    performed for every head h (hidden dimension) in the same launch.
    """
    # ------------------------------------------------------------------
    # 0️⃣ Identify the (batch, head) pair processed by this program.
    # ------------------------------------------------------------------
    bh_id = tl.program_id(0)                 # 0‑th grid axis → B·H
    pid_m = tl.program_id(1)                 # tile row (i)
    pid_n = tl.program_id(2)                 # tile col (j)

    # decode batch b and head h from the combined index
    b = bh_id // H
    h = bh_id % H

    # ------------------------------------------------------------------
    # 1️⃣ Tile offsets for the i‑ and j‑ dimensions.
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)   # i indices
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)   # j indices
    mask_m = offs_m < N
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # 2️⃣ Compute base pointers for this (b,h) slice.
    # ------------------------------------------------------------------
    left_base  = b * stride_b_l + h * stride_h_l
    right_base = b * stride_b_r + h * stride_h_r
    out_base   = b * stride_b_o + h * stride_h_o

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # 3️⃣ Loop over the K (seq) dimension in tiles.
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ----- left tile: A[i,k] -------------------------------------------------
        # Shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_ptr + left_base \
                 + offs_m[:, None] * stride_i_l \
                 + offs_k[None, :] * stride_k_l
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)   # fp16 → fp32

        # ----- right tile (transposed): B[k,j] ----------------------------------
        # Original layout is right[b,j,k,h]; we need it as (k,j).
        # Shape (BLOCK_K, BLOCK_N)
        b_ptrs = right_ptr + right_base \
                 + offs_k[:, None] * stride_k_r \
                 + offs_n[None, :] * stride_j_r
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)

        # ----- MAC ---------------------------------------------------------------
        acc += tl.dot(a, b)      # (M,K)·(K,N) → (M,N) in fp32

    # ------------------------------------------------------------------
    # 4️⃣ Write the result back as fp16.
    # ------------------------------------------------------------------
    c_ptrs = out_ptr + out_base \
             + offs_m[:, None] * stride_i_o \
             + offs_n[None, :] * stride_j_o
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float) – optional
        - weights_dict : model parameters (see keys below)
        - config_dict  : must contain "dim", "hidden_dim", "nomask"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) – native fused kernel (FP32 → FP16)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    ).to(torch.float16)                     # [B,N,N,dim] FP16

    # --------------------------------------------------------------
    # 2️⃣ Fuse the four linear maps (proj & gate) into a single GEMM
    # --------------------------------------------------------------
    # weight shape (4·H, dim) → FP16
    w_4h = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)

    # flatten batch & pair dimensions for the GEMM
    x_flat = x.view(-1, dim)                 # [R, dim]   R = B·N·N
    proj_gate = F.linear(x_flat, w_4h)       # [R, 4·H]   FP16

    # split
    H = hidden_dim
    left_proj_raw  = proj_gate[:, 0:H]
    right_proj_raw = proj_gate[:, H:2 * H]
    left_gate_raw  = proj_gate[:, 2 * H:3 * H]
    right_gate_raw = proj_gate[:, 3 * H:4 * H]

    # out‑gate (separate linear map)
    out_gate_raw = F.linear(
        x_flat,
        weights["out_gate.weight"].to(torch.float16)   # (H, dim)
    )                                            # [R, H] FP16

    # --------------------------------------------------------------
    # 3️⃣ Optional mask, sigmoid gates and element‑wise gating
    # --------------------------------------------------------------
    if not nomask:
        # mask: [B,N,N] → [R,1] FP16
        mask_fp16 = mask_tensor.reshape(-1, 1).to(torch.float16)
        left_proj_raw  = left_proj_raw * mask_fp16
        right_proj_raw = right_proj_raw * mask_fp16

    # in‑place sigmoid for speed
    left_gate = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate = out_gate_raw.sigmoid_()

    # apply gates
    left  = left_proj_raw * left_gate          # [R, H] FP16
    right = right_proj_raw * right_gate        # [R, H] FP16

    # --------------------------------------------------------------
    # 4️⃣ Reshape to (B,N,N,H) – still contiguous, no copies.
    # --------------------------------------------------------------
    left  = left.view(B, N, N, H)               # [B,N,N,H] FP16
    right = right.view(B, N, N, H)              # [B,N,N,H] FP16
    out_gate = out_gate.view(B, N, N, H)        # [B,N,N,H] FP16

    # --------------------------------------------------------------
    # 5️⃣ Cubic update – batched GEMM in Triton (no permute copy)
    # --------------------------------------------------------------
    out = torch.empty_like(left)                # allocate output (FP16)

    # strides for the three tensors (all share the same layout)
    stride_b_l, stride_i_l, stride_k_l, stride_h_l = left.stride()
    stride_b_r, stride_j_r, stride_k_r, stride_h_r = right.stride()
    stride_b_o, stride_i_o, stride_j_o, stride_h_o = out.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B * H,                                 # combined batch & head dimension
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    trmul_gemm_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        out.data_ptr(),
        B,
        N,
        H,
        stride_b_l,
        stride_i_l,
        stride_k_l,
        stride_h_l,
        stride_b_r,
        stride_j_r,
        stride_k_r,
        stride_h_r,
        stride_b_o,
        stride_i_o,
        stride_j_o,
        stride_h_o,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm over the hidden dimension H
    # --------------------------------------------------------------
    # flatten to (R, H) for the row‑wise layer‑norm
    out_flat = out.view(-1, H)
    out_norm = F.layer_norm(
        out_flat,
        normalized_shape=(H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    ).to(torch.float16)                        # still FP16
    out_norm = out_norm.view(B, N, N, H)

    # --------------------------------------------------------------
    # 7️⃣ Apply out‑gate and final linear projection back to dim
    # --------------------------------------------------------------
    gated = out_norm * out_gate                      # FP16
    final = F.linear(
        gated.view(-1, H),                          # [R, H]
        weights["to_out.weight"].to(torch.float16) # (dim, H)
    )                                                # [R, dim] FP16

    # Cast back to FP32 to match the reference implementation
    final = final.view(B, N, N, dim).to(torch.float32)
    return final
