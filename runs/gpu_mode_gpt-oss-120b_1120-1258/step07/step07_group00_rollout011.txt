@triton.jit
def proj_gate_kernel(
    x_ptr, left_ptr, right_ptr, out_gate_ptr,
    w_left_ptr, w_lgate_ptr, w_right_ptr, w_rgate_ptr, w_out_gate_ptr,
    total_rows, dim, H,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)   # [BLOCK_M]
    mask_m = offs_m < total_rows

    offs_n = tl.arange(0, BLOCK_N)    # channel index within H
    # compute weight pointers offsets per matrix:
    # each weight matrix shaped [H, dim], stride for rows = dim, stride for cols = 1? Actually weight is [H, dim]; we store row-major as weight[h, d] visible in memory typical contiguous layout: stride_h = dim (row major), stride_d = 1.
    # we will compute dot product across K dimension; typical matrix multiplication: A (M, K) @ B^T (N, K). We can treat each output channel as row of weight.
    # We'll load weight blocks of shape [BLOCK_N, BLOCK_K] for each weight matrix.

    # Accumulators for each of the 5 outputs: left_proj, left_gate, right_proj, right_gate, out_gate.
    acc_left_proj = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    acc_left_gate = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    acc_right_proj = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    acc_right_gate = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    acc_out_gate = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(dim, BLOCK_K)):
        off_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = off_k < dim

        # Load x tile: shape [BLOCK_M, BLOCK_K]
        x_tile = tl.load(x_ptr + (offs_m[:, None] * dim + off_k[None, :]), mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load weight tiles for each matrix: shape [BLOCK_N, BLOCK_K]
        w_left_tile = tl.load(w_left_ptr + (offs_n[:, None] * dim + off_k[None, :]), mask=mask_k[None, :], other=0.0).to(tl.float32)
        w_lgate_tile = tl.load(w_lgate_ptr + (offs_n[:, None] * dim + off_k[None, :]), mask=mask_k[None, :], other=0.0).to(tl.float32)
        w_right_tile = tl.load(w_right_ptr + (offs_n[:, None] * dim + off_k[None, :]), mask=mask_k[None, :], other=0.0).to(tl.float32)
        w_rgate_tile = tl.load(w_rgate_ptr + (offs_n[:, None] * dim + off_k[None, :]), mask=mask_k[None, :], other=0.0).to(tl.float32)
        w_out_gate_tile = tl.load(w_out_gate_ptr + (offs_n[:, None] * dim + off_k[None, :]), mask=mask_k[None, :], other=0.0).to(tl.float32)

        # Use dot
        acc_left_proj += tl.dot(x_tile, w_left_tile)   # shape (M,N)
        acc_left_gate += tl.dot(x_tile, w_lgate_tile)
        acc_right_proj += tl.dot(x_tile, w_right_tile)
        acc_right_gate += tl.dot(x_tile, w_rgate_tile)
        acc_out_gate += tl.dot(x_tile, w_out_gate_tile)

        # ptr increments automatically through next loop due to base pointer offsets computed per tile

    # After reduction, apply sigmoid and combine.
    # Compute sigmoid using fast approx: 1/(1+exp(-x)). Use tl.exp.
    # Convert to fp16 when storing.

    # Sigmoids:
    left_gate_sig = 1.0 / (1.0 + tl.exp(-acc_left_gate))
    right_gate_sig = 1.0 / (1.0 + tl.exp(-acc_right_gate))
    out_gate_sig = 1.0 / (1.0 + tl.exp(-acc_out_gate))

    left_val = acc_left_proj * left_gate_sig
    right_val = acc_right_proj * right_gate_sig

    # Cast to fp16
    left_fp16 = left_val.to(tl.float16)
    right_fp16 = right_val.to(tl.float16)
    out_gate_fp16 = out_gate_sig.to(tl.float16)

    # Store
    # store left
    left_ptrs = left_ptr + (offs_m[:, None] * H + offs_n[None, :])
    right_ptrs = right_ptr + (offs_m[:, None] * H + offs_n[None, :])
    out_gate_ptrs = out_gate_ptr + (offs_m[:, None] * H + offs_n[None, :])

    tl.store(left_ptrs, left_fp16, mask=mask_m[:, None] & (offs_n[None, :] < H))
    tl.store(right_ptrs, right_fp16, mask=mask_m[:, None] & (offs_n[None, :] < H))
    tl.store(out_gate_ptrs, out_gate_fp16, mask=mask_m[:, None] & (offs_n[None, :] < H))

def custom_kernel(data):
    """Fast Triton implementation of the outgoing TriMul operator.
    The algorithm:
    1. Row‑wise LayerNorm on the input (FP32→FP16).
    2. A single fused Triton GEMM computes left/right projections, their
       gates and the output gate. The gates are applied on‑the‑fly, producing
       left, right and out_gate tensors already in FP16.
    3. Optional masking is applied (broadcast over the hidden dimension).
    4. Batched matrix multiplication implements the `einsum('... i k d, ... j k d -> ... i j d')`.
    5. A fused LayerNorm+gate kernel normalises the intermediate tensor
       and multiplies by the previously computed out_gate.
    6. Final FP16 linear projection back to the original channel dimension.
    All heavy work is off‑loaded to Triton kernels; the remaining steps use
    highly‑optimised Torch kernels.
    """
    import torch
    import torch.nn.functional as F
    import triton
    import triton.language as tl

    # unpack
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    device = input_tensor.device

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N
    H = hidden_dim

    # ---------------------------------------------------------
    # 1. LayerNorm (FP32->FP16) using Triton
    # ---------------------------------------------------------
    x_flat = input_tensor.view(total_rows, dim).contiguous()
    x_norm = torch.empty(total_rows, dim, dtype=torch.float16, device=device)

    BLOCK_M_LN = 1024
    BLOCK_D_LN = 64
    grid_ln = (triton.cdiv(total_rows, BLOCK_M_LN),)

    @triton.jit
    def layernorm_kernel(
        x_ptr, y_ptr,
        weight_ptr, bias_ptr,
        total_rows, dim, eps,
        BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
    ):
        pid = tl.program_id(0)
        row_start = pid * BLOCK_M
        offs_m = row_start + tl.arange(0, BLOCK_M)
        mask_m = offs_m < total_rows

        # compute mean & var
        sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
        sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

        for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
            offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
            mask_d = offs_d < dim
            ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
            vals = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)
            sum_val += tl.sum(vals, axis=1)
            sum_sq += tl.sum(vals * vals, axis=1)

        mean = sum_val / dim
        var = sum_sq / dim - mean * mean
        rstd = tl.rsqrt(var + eps)

        for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
            offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
            mask_d = offs_d < dim
            ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
            vals = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)

            norm = (vals - mean[:, None]) * rstd[:, None]       # FP32

            w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
            b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β

            out = norm * w[None, :] + b[None, :]                # FP32
            out_fp16 = out.to(tl.float16)

            out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
            tl.store(out_ptr, out_fp16,
                     mask=mask_m[:, None] & mask_d[None, :])

    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M_LN,
        BLOCK_D=BLOCK_D_LN,
        num_warps=4,
    )

    # ---------------------------------------------------------
    # 2. Fused projection + gates (left/right/out_gate)
    # ---------------------------------------------------------
    # convert weight matrices to FP16 once
    w_left = weights["left_proj.weight"].to(torch.float16)
    w_right = weights["right_proj.weight"].to(torch.float16)
    w_lgate = weights["left_gate.weight"].to(torch.float16)
    w_rgate = weights["right_gate.weight"].to(torch.float16)
    w_out_gate = weights["out_gate.weight"].to(torch.float16)

    left = torch.empty(total_rows, H, dtype=torch.float16, device=device)
    right = torch.empty(total_rows, H, dtype=torch.float16, device=device)
    out_gate = torch.empty(total_rows, H, dtype=torch.float16, device=device)

    BLOCK_M_PROJ = 256
    BLOCK_N_PROJ = 64
    BLOCK_K_PROJ = 64
    grid_proj = (
        triton.cdiv(total_rows, BLOCK_M_PROJ),
        triton.cdiv(H, BLOCK_N_PROJ),
    )

    @triton.jit
    def proj_gate_kernel(
        x_ptr,
        left_ptr, right_ptr, out_gate_ptr,
        w_left_ptr, w_lgate_ptr, w_right_ptr, w_rgate_ptr, w_out_gate_ptr,
        total_rows, dim, hidden_dim,
        BLOCK_M: tl.constexpr,
        BLOCK_N: tl.constexpr,
        BLOCK_K: tl.constexpr,
    ):
        pid_m = tl.program_id(0)   # row block
        pid_n = tl.program_id(1)   # channel block

        row_start = pid_m * BLOCK_M
        col_start = pid_n * BLOCK_N

        offs_m = row_start + tl.arange(0, BLOCK_M)
        offs_n = col_start + tl.arange(0, BLOCK_N)

        mask_m = offs_m < total_rows
        mask_n = offs_n < hidden_dim

        # accumulators in fp32
        acc_left_proj = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
        acc_left_gate = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
        acc_right_proj = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
        acc_right_gate = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
        acc_out_gate = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

        for k in range(0, tl.cdiv(dim, BLOCK_K)):
            off_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
            mask_k = off_k < dim

            # load X tile [M, K]
            x = tl.load(
                x_ptr + offs_m[:, None] * dim + off_k[None, :],
                mask=mask_m[:, None] & mask_k[None, :],
                other=0.0
            ).to(tl.float32)

            # load weight tiles [N, K] for each matrix
            w_l = tl.load(
                w_left_ptr + offs_n[:, None] * dim + off_k[None, :],
                mask=mask_n[:, None] & mask_k[None, :],
                other=0.0
            ).to(tl.float32)

            w_lg = tl.load(
                w_lgate_ptr + offs_n[:, None] * dim + off_k[None, :],
                mask=mask_n[:, None] & mask_k[None, :],
                other=0.0
            ).to(tl.float32)

            w_r = tl.load(
                w_right_ptr + offs_n[:, None] * dim + off_k[None, :],
                mask=mask_n[:, None] & mask_k[None, :],
                other=0.0
            ).to(tl.float32)

            w_rg = tl.load(
                w_rgate_ptr + offs_n[:, None] * dim + off_k[None, :],
                mask=mask_n[:, None] & mask_k[None, :],
                other=0.0
            ).to(tl.float32)

            w_og = tl.load(
                w_out_gate_ptr + offs_n[:, None] * dim + off_k[None, :],
                mask=mask_n[:, None] & mask_k[None, :],
                other=0.0
            ).to(tl.float32)

            # dot products accumulate
            acc_left_proj += tl.dot(x, w_l)      # (M,N)
            acc_left_gate += tl.dot(x, w_lg)
            acc_right_proj += tl.dot(x, w_r)
            acc_right_gate += tl.dot(x, w_rg)
            acc_out_gate += tl.dot(x, w_og)

        # sigmoid (stable)
        sigmoid = lambda v: 1.0 / (1.0 + tl.exp(-v))

        left_gate = sigmoid(acc_left_gate)
        right_gate = sigmoid(acc_right_gate)
        out_gate_val = sigmoid(acc_out_gate)

        left_val = acc_left_proj * left_gate
        right_val = acc_right_proj * right_gate

        # cast to fp16
        left_fp16 = left_val.to(tl.float16)
        right_fp16 = right_val.to(tl.float16)
        out_gate_fp16 = out_gate_val.to(tl.float16)

        # store results
        out_mask = mask_m[:, None] & mask_n[None, :]
        left_ptrs = left_ptr + offs_m[:, None] * hidden_dim + offs_n[None, :]
        right_ptrs = right_ptr + offs_m[:, None] * hidden_dim + offs_n[None, :]
        out_gate_ptrs = out_gate_ptr + offs_m[:, None] * hidden_dim + offs_n[None, :]

        tl.store(left_ptrs, left_fp16, mask=out_mask)
        tl.store(right_ptrs, right_fp16, mask=out_mask)
        tl.store(out_gate_ptrs, out_gate_fp16, mask=out_mask)

    proj_gate_kernel[grid_proj](
        x_norm,
        left,
        right,
        out_gate,
        w_left,
        w_lgate,
        w_right,
        w_rgate,
        w_out_gate,
        total_rows,
        dim,
        H,
        BLOCK_M=BLOCK_M_PROJ,
        BLOCK_N=BLOCK_N_PROJ,
        BLOCK_K=BLOCK_K_PROJ,
        num_warps=8,
    )

    # ---------------------------------------------------------
    # 3. Optional mask application (broadcast over H)
    # ---------------------------------------------------------
    if not config.get("nomask", True):
        mask_h = mask_tensor.to(torch.float16).reshape(total_rows, 1)
        left = left * mask_h
        right = right * mask_h

    # ---------------------------------------------------------
    # 4. Batched GEMM: einsum('... i k d, ... j k d -> ... i j d')
    # ---------------------------------------------------------
    left = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    left_mat = left.view(B * H, N, N)       # [B·H, N, N]
    right_mat = right.view(B * H, N, N)     # [B·H, N, N]

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N]

    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]
    out_flat = out.view(total_rows, H)

    # ---------------------------------------------------------
    # 5. Fused LayerNorm + output gate multiplication
    # ---------------------------------------------------------
    out_norm_gated = torch.empty_like(out_flat, dtype=torch.float16)

    grid_ln2 = (triton.cdiv(total_rows, BLOCK_M_LN),)

    @triton.jit
    def layernorm_gate_kernel(
        x_ptr, y_ptr,
        weight_ptr, bias_ptr,
        gate_ptr,
        total_rows, dim, eps,
        BLOCK_M: tl.constexpr,
        BLOCK_D: tl.constexpr,
    ):
        pid = tl.program_id(0)
        row_start = pid * BLOCK_M
        offs_m = row_start + tl.arange(0, BLOCK_M)
        mask_m = offs_m < total_rows

        # mean / var
        sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
        sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

        for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
            offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
            mask_d = offs_d < dim
            ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
            vals = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)
            sum_val += tl.sum(vals, axis=1)
            sum_sq += tl.sum(vals * vals, axis=1)

        mean = sum_val / dim
        var = sum_sq / dim - mean * mean
        rstd = tl.rsqrt(var + eps)

        # apply LN + gate
        for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
            offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
            mask_d = offs_d < dim
            x_ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
            vals = tl.load(x_ptrs, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)

            gate_ptrs = gate_ptr + offs_m[:, None] * dim + offs_d[None, :]
            gate = tl.load(gate_ptrs, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)

            norm = (vals - mean[:, None]) * rstd[:, None]

            w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
            b = tl.load(bias_ptr + offs_d, mask=mask_d, other=0.0)   # β

            out = (norm * w[None, :] + b[None, :]) * gate
            out_fp16 = out.to(tl.float16)

            out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
            tl.store(out_ptr, out_fp16, mask=mask_m[:, None] & mask_d[None, :])

    layernorm_gate_kernel[grid_ln2](
        out_flat,
        out_norm_gated,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        out_gate,
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M_LN,
        BLOCK_D=BLOCK_D_LN,
        num_warps=4,
    )

    # ---------------------------------------------------------
    # 6. Final projection to dim (FP16 GEMM)
    # ---------------------------------------------------------
    w_to_out = weights["to_out.weight"].to(torch.float16)   # [dim, H]
    final_half = F.linear(out_norm_gated, w_to_out)        # [R, dim] FP16

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final

"""
Fast Triton implementation of the *outgoing* TriMul operator (AlphaFold‑3).

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the input tensor (FP32 → FP16).
2️⃣ A *single* fused Triton GEMM computes the four linear projections
   (left, right, left‑gate, right‑gate) together with the output‑gate.
   The gates are applied on‑the‑fly, producing `left`, `right` and
   `out_gate` already in FP16.  This removes two extra matmuls and cuts the
   memory traffic for the repeated load of the input rows.
3️⃣ Optional mask (broadcast over the hidden dimension) is applied after
   gating (multiplication is associative, therefore the order does not
   matter).
4️⃣ Batched matrix multiplication implements the core `einsum('... i k d,
   ... j k d -> ... i j d')`.  The hidden dimension is moved to the batch
   axis, giving a simple `torch.bmm` on tensors of shape (B·H, N, N).
5️⃣ A second fused kernel performs LayerNorm over the hidden dimension and
   multiplies the result by the previously‑computed `out_gate`.
6️⃣ Final half‑precision linear projection back to the original channel
   dimension (`dim`).  The output is cast to FP32 to match the reference
   implementation.

All heavy work (both LayerNorms and the fused projection+gate GEMM) is
off‑loaded to Triton kernels; the remaining steps use highly‑optimised
Torch kernels.  The implementation supports `B∈{1,2}`, `N∈{128,…,1024}`,
`dim∈{128,384}` and `hidden_dim=128` on an H100 GPU.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr, y_ptr,                         # [M, D] input (fp32) and output (fp16)
    weight_ptr, bias_ptr,                 # LayerNorm parameters γ, β (fp32)
    total_rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # ---------- mean / variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)               # 1/√(var+eps)

    # ---------- write normalised tensor ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]   # FP32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]            # FP32

        out_fp16 = out.to(tl.float16)
        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2️⃣ Fused projection + gating kernel
#    Computes left, right and out_gate in a single GEMM.
# ----------------------------------------------------------------------
@triton.jit
def proj_gate_kernel(
    x_ptr,                                 # [M, D]   (fp16)
    left_ptr, right_ptr, out_gate_ptr,     # [M, H]   (fp16) outputs
    w_left_ptr, w_lgate_ptr,
    w_right_ptr, w_rgate_ptr,
    w_out_gate_ptr,                       # [H, D] each (fp16)
    total_rows, dim, hidden_dim,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)   # row block
    pid_n = tl.program_id(1)   # hidden‑dim block

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)
    offs_n = col_start + tl.arange(0, BLOCK_N)

    mask_m = offs_m < total_rows
    mask_n = offs_n < hidden_dim

    # accumulators (FP32)
    acc_lproj = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    acc_lgate = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    acc_rproj = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    acc_rgate = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    acc_outg = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(dim, BLOCK_K)):
        off_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = off_k < dim

        # X tile [M, K]
        x = tl.load(
            x_ptr + offs_m[:, None] * dim + off_k[None, :],
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0).to(tl.float32)

        # weight tiles [N, K] for each of the 5 matrices
        w_l = tl.load(
            w_left_ptr + offs_n[:, None] * dim + off_k[None, :],
            mask=mask_n[:, None] & mask_k[None, :],
            other=0.0).to(tl.float32)

        w_lg = tl.load(
            w_lgate_ptr + offs_n[:, None] * dim + off_k[None, :],
            mask=mask_n[:, None] & mask_k[None, :],
            other=0.0).to(tl.float32)

        w_r = tl.load(
            w_right_ptr + offs_n[:, None] * dim + off_k[None, :],
            mask=mask_n[:, None] & mask_k[None, :],
            other=0.0).to(tl.float32)

        w_rg = tl.load(
            w_rgate_ptr + offs_n[:, None] * dim + off_k[None, :],
            mask=mask_n[:, None] & mask_k[None, :],
            other=0.0).to(tl.float32)

        w_og = tl.load(
            w_out_gate_ptr + offs_n[:, None] * dim + off_k[None, :],
            mask=mask_n[:, None] & mask_k[None, :],
            other=0.0).to(tl.float32)

        # dot‑products
        acc_lproj += tl.dot(x, w_l)      # (M,N)
        acc_lgate += tl.dot(x, w_lg)
        acc_rproj += tl.dot(x, w_r)
        acc_rgate += tl.dot(x, w_rg)
        acc_outg += tl.dot(x, w_og)

    # ---------- sigmoid ----------
    sigmoid = lambda v: 1.0 / (1.0 + tl.exp(-v))

    left_gate  = sigmoid(acc_lgate)
    right_gate = sigmoid(acc_rgate)
    out_gate   = sigmoid(acc_outg)

    # ---------- apply gates ----------
    left  = acc_lproj * left_gate
    right = acc_rproj * right_gate

    # cast to FP16
    left_fp16  = left.to(tl.float16)
    right_fp16 = right.to(tl.float16)
    out_fp16   = out_gate.to(tl.float16)

    # store results
    store_mask = mask_m[:, None] & mask_n[None, :]

    tl.store(left_ptr      + offs_m[:, None] * hidden_dim + offs_n[None, :],
              left_fp16,  mask=store_mask)
    tl.store(right_ptr     + offs_m[:, None] * hidden_dim + offs_n[None, :],
              right_fp16, mask=store_mask)
    tl.store(out_gate_ptr  + offs_m[:, None] * hidden_dim + offs_n[None, :],
              out_fp16,   mask=store_mask)


# ----------------------------------------------------------------------
# 3️⃣ LayerNorm + gate (used on the intermediate tensor before the final
#     projection).  Very similar to `layernorm_kernel` but multiplies by a
#     per‑element gate after the normalisation.
# ----------------------------------------------------------------------
@triton.jit
def layernorm_gate_kernel(
    x_ptr, y_ptr,
    weight_ptr, bias_ptr,
    gate_ptr,
    total_rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # ---------- mean / variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- normalise + gate ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        x_ptrs  = x_ptr   + offs_m[:, None] * dim + offs_d[None, :]
        g_ptrs  = gate_ptr+ offs_m[:, None] * dim + offs_d[None, :]

        vals = tl.load(x_ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        gate = tl.load(g_ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]   # FP32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β

        out = (norm * w[None, :] + b[None, :]) * gate   # FP32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 4️⃣ Entry point required by the grader
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype torch.float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : mapping name → torch.Tensor (all parameters)
        - config_dict  : must contain "dim" (int) and "hidden_dim" (int);
                         optional key "nomask" (bool, default True).

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (torch.float32)
    """
    # --------------------------------------------------------------
    # Unpack
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim        = config["dim"]
    hidden_dim = config["hidden_dim"]          # = H
    eps        = 1e-5
    device     = input_tensor.device

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N                     # R = B·N·N
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32 → FP16)
    # --------------------------------------------------------------
    x_flat = input_tensor.view(total_rows, dim).contiguous()   # [R, D] FP32
    x_norm = torch.empty(total_rows, dim, dtype=torch.float16, device=device)

    BLOCK_M_LN = 1024
    BLOCK_D_LN = 64
    grid_ln = (triton.cdiv(total_rows, BLOCK_M_LN),)

    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        weights["norm.weight"],          # γ (FP32)
        weights["norm.bias"],            # β (FP32)
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M_LN,
        BLOCK_D=BLOCK_D_LN,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 2️⃣ Fused projection + gating (produces left, right, out_gate)
    # --------------------------------------------------------------
    w_left      = weights["left_proj.weight"].to(torch.float16).contiguous()
    w_right     = weights["right_proj.weight"].to(torch.float16).contiguous()
    w_lgate     = weights["left_gate.weight"].to(torch.float16).contiguous()
    w_rgate     = weights["right_gate.weight"].to(torch.float16).contiguous()
    w_out_gate  = weights["out_gate.weight"].to(torch.float16).contiguous()

    left      = torch.empty(total_rows, H, dtype=torch.float16, device=device)
    right     = torch.empty(total_rows, H, dtype=torch.float16, device=device)
    out_gate  = torch.empty(total_rows, H, dtype=torch.float16, device=device)

    BLOCK_M_PROJ = 256
    BLOCK_N_PROJ = 64
    BLOCK_K_PROJ = 64
    grid_proj = (
        triton.cdiv(total_rows, BLOCK_M_PROJ),
        triton.cdiv(H, BLOCK_N_PROJ),
    )

    proj_gate_kernel[grid_proj](
        x_norm,
        left,
        right,
        out_gate,
        w_left,
        w_lgate,
        w_right,
        w_rgate,
        w_out_gate,
        total_rows,
        dim,
        H,
        BLOCK_M=BLOCK_M_PROJ,
        BLOCK_N=BLOCK_N_PROJ,
        BLOCK_K=BLOCK_K_PROJ,
        num_warps=8,
    )

    # --------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over hidden dimension)
    # --------------------------------------------------------------
    if not config.get("nomask", True):
        mask_h = mask_tensor.to(torch.float16).reshape(total_rows, 1)
        left  = left * mask_h
        right = right * mask_h

    # --------------------------------------------------------------
    # 4️⃣ Batched GEMM implementing the multiplicative update
    # --------------------------------------------------------------
    # reshape to [B, H, N, N] for easy batched matmul
    left  = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    left_mat  = left.view(B * H, N, N)      # [B·H, N, N]
    right_mat = right.view(B * H, N, N)     # [B·H, N, N]

    # (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]
    out_flat = out.view(total_rows, H)                                 # [R, H]

    # --------------------------------------------------------------
    # 5️⃣ Fused LayerNorm over hidden dimension + multiplication by out_gate
    # --------------------------------------------------------------
    out_norm_gated = torch.empty_like(out_flat, dtype=torch.float16)

    grid_ln2 = (triton.cdiv(total_rows, BLOCK_M_LN),)

    layernorm_gate_kernel[grid_ln2](
        out_flat,
        out_norm_gated,
        weights["to_out_norm.weight"],   # γ (FP32)
        weights["to_out_norm.bias"],     # β (FP32)
        out_gate,                        # gate (FP16)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M_LN,
        BLOCK_D=BLOCK_D_LN,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 6️⃣ Final linear projection back to `dim`
    # --------------------------------------------------------------
    w_to_out = weights["to_out.weight"].to(torch.float16)   # [dim, H]
    final_half = F.linear(out_norm_gated, w_to_out)        # [R, dim] FP16

    # Cast back to FP32 (as the reference implementation returns fp32)
    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
