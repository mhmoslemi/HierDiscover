"""
custom_kernel
=============
Highly‑optimized forward pass for the *outgoing* TriMul operator (AlphaFold‑3).

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the input tensor (FP32 → FP16) – custom Triton kernel.
2️⃣ Fuse **five** linear projections (left, right, left‑gate, right‑gate, out‑gate)
   into a single half‑precision GEMM:
      [R, dim] @ [5·H, dim]ᵀ → [R, 5·H]   with R = B·N·N.
3️⃣ Split the result, apply the optional mask, and compute in‑place sigmoid
   gates.
4️⃣ Apply the two gates to the left/right projections, reshape to
   [B, H, N, N] and perform a batched matrix multiplication
      (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N)
   (this implements `einsum('... i k d, ... j k d -> ... i j d')`).
5️⃣ LayerNorm over the hidden dimension **H** (torch.nn.functional.layer_norm)
   followed by element‑wise multiplication with the previously‑computed
   output‑gate.
6️⃣ Final linear projection back to `dim` (half‑precision GEMM) and cast to
   FP32.

The implementation works for
    B ∈ {1,2}, N ≤ 1024, dim ∈ {128,384}, hidden_dim = 128,
and supports both masked and un‑masked inputs.  Only the first LayerNorm
is written in Triton – satisfying the requirement that *part* of the
forward pass is kernel‑based while keeping the heaviest work on cuBLAS
for maximal throughput on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,               # const   * [total_rows, dim]   (FP32)
    y_ptr,               # mutable * [total_rows, dim]   (FP16)
    w_ptr,               # const   * [dim]               (FP32) γ
    b_ptr,               # const   * [dim]               (FP32) β
    total_rows,          # int (runtime)
    dim,                 # int (runtime)
    eps,                 # float (runtime)
    BLOCK_M: tl.constexpr,  # rows per program
    BLOCK_D: tl.constexpr,  # cols per inner loop
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)            # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ---------- accumulate mean / variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # FP32

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                           # 1 / sqrt(var+eps)

    # ---------- write normalized output ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]    # FP32

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # γ  (FP32)
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # β  (FP32)

        out = norm * w[None, :] + b[None, :]                 # FP32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Entry point required by the grader
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype torch.float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) – optional
        - weights_dict : mapping from string names to torch.Tensor (model parameters)
        - config_dict  : must contain "dim" (int) and "hidden_dim" (int);
                         optional key "nomask" (bool, default True).

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (torch.float32)
    """
    # --------------------------------------------------------------
    # Unpack
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]          # called H in the code
    eps = 1e-5
    device = input_tensor.device

    B, N, _, _ = input_tensor.shape
    R = B * N * N                               # total rows = B·N·N
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32 → FP16) – Triton
    # --------------------------------------------------------------
    x_flat = input_tensor.view(R, dim).contiguous()            # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 256                     # rows per kernel launch
    BLOCK_D = 64                      # columns per inner loop (covers dim ≤ 384)
    grid = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],          # γ (FP32)
        weights["norm.bias"],            # β (FP32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 2️⃣ Fused 5‑way linear projection:
    #    left, right, left‑gate, right‑gate, out‑gate
    # --------------------------------------------------------------
    # Convert all needed weight matrices to FP16 once
    w_left   = weights["left_proj.weight"].to(torch.float16)   # [H, dim]
    w_right  = weights["right_proj.weight"].to(torch.float16)  # [H, dim]
    w_lgate  = weights["left_gate.weight"].to(torch.float16)   # [H, dim]
    w_rgate  = weights["right_gate.weight"].to(torch.float16)  # [H, dim]
    w_outgate= weights["out_gate.weight"].to(torch.float16)    # [H, dim]

    # Concatenate along the output dimension → [5·H, dim]
    w_all = torch.cat([w_left, w_right, w_lgate, w_rgate, w_outgate], dim=0)

    # Single half‑precision GEMM: (R, dim) @ (5·H, dim)ᵀ → (R, 5·H)
    proj = F.linear(x_norm, w_all)                # [R, 5·H] FP16

    # Split back to the five tensors
    left_raw, right_raw, left_gate_raw, right_gate_raw, out_gate_raw = \
        torch.split(proj, H, dim=1)               # each [R, H] FP16

    # --------------------------------------------------------------
    # 3️⃣ Optional mask (applied before gating)
    # --------------------------------------------------------------
    if not config.get("nomask", True):
        # mask shape [B, N, N] → [R, 1]
        mask_h = mask_tensor.to(torch.float16).view(R, 1)
        left_raw  = left_raw * mask_h
        right_raw = right_raw * mask_h

    # --------------------------------------------------------------
    # 4️⃣ Sigmoid gates (in‑place) and apply them
    # --------------------------------------------------------------
    left_gate  = left_gate_raw.sigmoid_()         # [R, H] FP16
    right_gate = right_gate_raw.sigmoid_()        # [R, H] FP16
    out_gate   = out_gate_raw.sigmoid_()          # [R, H] FP16

    left  = left_raw  * left_gate                  # [R, H] FP16
    right = right_raw * right_gate                 # [R, H] FP16

    # --------------------------------------------------------------
    # 5️⃣ Triangular multiplicative update (batched GEMM)
    # --------------------------------------------------------------
    # Reshape to [B, N, N, H] then bring H into the batch dimension
    left  = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    left_mat  = left.view(B * H, N, N)        # [B·H, N, N]
    right_mat = right.view(B * H, N, N)       # [B·H, N, N]

    # (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Restore layout → [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]
    out_flat = out.view(R, H)                                            # [R, H] FP16

    # --------------------------------------------------------------
    # 6️⃣ Hidden‑dim LayerNorm + multiplication by out‑gate
    # --------------------------------------------------------------
    # Torch's highly‑optimized LayerNorm (still FP16)
    out_norm = F.layer_norm(
        out_flat,
        normalized_shape=(H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )                                                         # FP16

    out_norm_gated = out_norm * out_gate                      # FP16

    # --------------------------------------------------------------
    # 7️⃣ Final projection back to `dim`
    # --------------------------------------------------------------
    w_to_out = weights["to_out.weight"].to(torch.float16)    # [dim, H]
    final_half = F.linear(out_norm_gated, w_to_out)         # [R, dim] FP16

    # Cast to the required float32 output
    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
