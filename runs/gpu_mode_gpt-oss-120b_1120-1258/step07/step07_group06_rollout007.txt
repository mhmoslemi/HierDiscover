"""
custom_kernel
==============

Fast forward implementation of the *outgoing* TriMul operator (AlphaFold‑3).

Algorithm
---------
1. **Row‑wise LayerNorm** – a Triton kernel normalises the input
   `[B·N·N, dim]` and writes the result in FP16.
2. **Fused projection + gates** – the four linear maps
   (left/right projection and their gates) are combined into a single
   half‑precision GEMM (cuBLAS) by concatenating their weight matrices.
3. **Mask & sigmoid** – optional pair‑wise mask is applied to the
   projection tensors, sigmoid is performed in‑place on the gate tensors.
4. **Tri‑multiplicative update** – using `torch.einsum` we compute
   `out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]` directly on the
   `[B,N,N,H]` layout, avoiding costly permute‑and‑contiguous copies.
5. **Second LayerNorm** – another Triton kernel normalises the hidden
   tensor `[B·N·N, H]`.
6. **Output gating & linear projection** – multiply by the output gate
   and project back to `dim`.  The final result is cast to `float32`.

The implementation keeps almost all heavy arithmetic in FP16, only the
final output is returned as FP32.  Two custom Triton kernels (input and
output LayerNorm) are required; the remaining operations rely on
high‑performance cuBLAS kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (float32 → float16)
# --------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   *[total_rows, dim]   (float32)
    y_ptr,          # mutable *[total_rows, dim]   (float16)
    weight_ptr,     # const   *[dim]               (float32) γ
    bias_ptr,       # const   *[dim]               (float32) β
    total_rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,  # rows per program
    BLOCK_D: tl.constexpr,  # cols per iteration
):
    """
    Row‑wise LayerNorm:
        y = ((x – μ) / sqrt(σ²+eps)) * γ + β
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # 1️⃣ Compute mean & variance (fp32 accumulation)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------------------------------------------------------
    # 2️⃣ Write normalised values (γ·(x‑μ)·rstd + β) in fp16
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]                      # fp32

        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# --------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool or float) or None
        - weights_dict : mapping with all Linear/LayerNorm parameters
                         (float32 tensors, bias=False for Linear)
        - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim], dtype float32
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                     # total rows for the row‑wise ops

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()                     # [R, dim] float32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 256          # rows per program
    BLOCK_D = 32           # columns per iteration
    grid = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],    # γ   (float32)
        weights["norm.bias"],      # β   (float32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fuse left/right projections + gates into a single GEMM
    # ------------------------------------------------------------------
    H = hidden_dim
    # concatenate the four weight matrices → [4*H, dim] (half‑precision)
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)

    # [R, dim] @ [4*H, dim].T  →  [R, 4*H]   (FP16)
    proj_gate = F.linear(x_norm, w_concat)

    left_proj_raw  = proj_gate[:, 0:H]
    right_proj_raw = proj_gate[:, H:2 * H]
    left_gate_raw  = proj_gate[:, 2 * H:3 * H]
    right_gate_raw = proj_gate[:, 3 * H:4 * H]

    # Output gate (separate linear)
    out_gate_raw = F.linear(
        x_norm,
        weights["out_gate.weight"].to(torch.float16),   # [H, dim]
    )   # FP16

    # ------------------------------------------------------------------
    # 3️⃣ Apply optional mask, sigmoid gates, and gating
    # ------------------------------------------------------------------
    if not cfg.get("nomask", True):
        # mask shape [B,N,N] → [R,1] broadcastable
        mask_flat = mask.reshape(R).to(torch.float16).unsqueeze(-1)  # [R,1]
        left_proj_raw = left_proj_raw * mask_flat
        right_proj_raw = right_proj_raw * mask_flat

    # inplace sigmoid for speed
    left_gate = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate = out_gate_raw.sigmoid_()

    # gated tensors
    left = left_proj_raw * left_gate   # [R, H] FP16
    right = right_proj_raw * right_gate   # [R, H] FP16

    # ------------------------------------------------------------------
    # 4️⃣ Tri‑multiplicative update (batched GEMM) – no extra copies
    # ------------------------------------------------------------------
    # reshape to [B, N, N, H] (still FP16, view‑only)
    left = left.view(B, N, N, H)
    right = right.view(B, N, N, H)

    # out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    # torch.einsum will delegate to a batched GEMM with batch = B*H,
    # avoiding the costly permute+contiguous steps of the naïve implementation.
    out = torch.einsum("b i k h, b j k h -> b i j h", left, right)   # FP16

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm over hidden dimension (FP16)
    # ------------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()
    out_norm = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Apply output gate and final projection back to `dim`
    # ------------------------------------------------------------------
    # out_norm and out_gate both have shape [R, H] (FP16)
    out_norm = out_norm * out_gate

    final_half = F.linear(
        out_norm,
        weights["to_out.weight"].to(torch.float16),   # [dim, H]
    )   # FP16

    final = final_half.view(B, N, N, dim).to(torch.float32)

    return final
