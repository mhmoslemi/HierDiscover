import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def layernorm_kernel(
    x_ptr, y_ptr, weight_ptr, bias_ptr,
    total_rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # compute mean & var
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # write normalized result
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr + offs_d, mask=mask_d, other=0.0)
        out = norm * w[None, :] + b[None, :]   # fp32

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])

def custom_kernel(data):
    """
    Outgoing TriMul forward pass (no gradients).

    The implementation follows the AlphaFold3 outgoing TriMul:
        1. Row‑wise LayerNorm over the input channels (fp32→fp16).
        2. Fuse left/right projections and their gates into a single FP16 GEMM.
        3. Compute the output‑gate with a second FP16 GEMM.
        4. Apply optional binary mask and sigmoid gates.
        5. Batched N×N matrix multiplication per head (FP16, Tensor‑Core).
        6. Second LayerNorm over the hidden dimension.
        7. Apply output‑gate and final linear projection, cast back to fp32.
    """
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    total_rows = B * N * N
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → fp16
    # --------------------------------------------------------------
    x_flat = x.view(total_rows, dim).contiguous()           # fp32
    x_norm_fp16 = torch.empty(total_rows, dim,
                              dtype=torch.float16,
                              device=device)

    BLOCK_M = 256
    BLOCK_D = 32
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],   # γ, fp32
        weights["norm.bias"],     # β, fp32
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 2️⃣ Fuse left/right projections + gates (4·H outputs)
    # --------------------------------------------------------------
    # Concatenate weight matrices (all are [H, dim])
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                   # [4H, dim]

    # [R, dim] @ [4H, dim]^T → [R, 4H] (FP16)
    fused = F.linear(x_norm_fp16, w_fused)                # fp16

    # split
    left_raw = fused[:, 0:H]
    right_raw = fused[:, H:2*H]
    left_gate_raw = fused[:, 2*H:3*H]
    right_gate_raw = fused[:, 3*H:4*H]

    # output gate
    out_gate_raw = F.linear(
        x_norm_fp16,
        weights["out_gate.weight"].to(torch.float16)    # [H, dim]
    )                                                     # [R, H]

    # --------------------------------------------------------------
    # 3️⃣ Mask, sigmoid gates and combine
    # --------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] → [R, 1]
        mask_fp16 = mask.to(torch.float16).reshape(total_rows, 1)
        left_raw = left_raw * mask_fp16
        right_raw = right_raw * mask_fp16

    left_gate = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate = out_gate_raw.sigmoid_()

    left = left_raw * left_gate          # [R, H] fp16
    right = right_raw * right_gate       # [R, H] fp16

    # --------------------------------------------------------------
    # 4️⃣ N³ multiplicative update (batched GEMM)
    # --------------------------------------------------------------
    left = left.view(B, N, N, H)
    right = right.view(B, N, N, H)
    out_gate = out_gate.view(B, N, N, H)

    # bring hidden dim to batch axis → [B, H, N, N]
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    left_mat = left_perm.view(B * H, N, N)               # [BH, N, N]
    right_mat = right_perm.view(B * H, N, N)             # [BH, N, N]

    # batched matmul: (BH,N,N) @ (BH,N,N)^T → (BH,N,N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # fp16

    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # [B,N,N,H]

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm over hidden dim → fp16
    # --------------------------------------------------------------
    out_flat = out.view(total_rows, H).contiguous()
    out_norm_fp16 = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],   # γ, fp32
        weights["to_out_norm.bias"],     # β, fp32
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    out_norm = out_norm_fp16.view(B, N, N, H)       # fp16

    # --------------------------------------------------------------
    # 6️⃣ Apply output gate and final projection
    # --------------------------------------------------------------
    out_norm = out_norm * out_gate                  # fp16

    final_fp16 = F.linear(
        out_norm.view(total_rows, H),
        weights["to_out.weight"].to(torch.float16)   # [dim, H]
    )                                                # [R, dim] fp16

    # back to fp32
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final

    BLOCK_D: tl.constexpr,   # cols per tile
):

    y = ((x - μ) / sqrt(σ² + eps)) * γ + β
