"""
TriMul (outgoing) – fast forward pass

Algorithm
---------
1. Row‑wise LayerNorm on the input tensor (shape B·N·N×dim).  The kernel
   normalises each row independently and writes FP16 results.
2. Fuse the four linear maps (left/right projections + their gates) into a
   single half‑precision GEMM:   X_norm @ W_concatᵀ → [R, 4·H].
3. Split the concatenated result, apply the pair‑wise mask (if any) and
   sigmoid‑activate the gates.  Gated tensors `left` and `right` are kept in
   FP16.
4. Tri‑multiplicative update: for every pair (i,j) compute
        out[i,j] = Σₖ left[i,k] * right[j,k] .
   This is exactly a batched matrix multiplication:
        (B·H, N, N) @ (B·H, N, N)ᵀ .
5. Second row‑wise LayerNorm over the hidden dimension H (FP16).
6. Multiply by the output‑gate, project back to `dim` and cast to FP32.

Only the two LayerNorms are written in Triton; all GEMMs use cuBLAS (via
torch.nn.functional.linear / torch.bmm), which is optimal on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (fp32 → fp16)
# ----------------------------------------------------------------------
@triton.jit
def row_layernorm_kernel(
    x_ptr,      # const   *[total_rows, dim]   (float32)
    y_ptr,      # mutable *[total_rows, dim]   (float16)
    w_ptr,      # const   *[dim]               (float32)  γ
    b_ptr,      # const   *[dim]               (float32)  β
    total_rows: tl.int32,
    dim:        tl.int32,
    eps:        tl.float32,
    BLOCK_M:    tl.constexpr,
    BLOCK_D:    tl.constexpr,
):
    """
    Compute a row‑wise LayerNorm:
        y = ((x - μ) / sqrt(σ²+eps)) * γ + β
    Input  : fp32
    Output : fp16 (accumulation in fp32)
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ---------- mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]  # [M,D]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- normalised & scaled ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]                  # fp32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : dict with all Linear/LayerNorm parameters
                         (float32 tensors, bias=False for Linear)
        - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    total_rows = B * N * N                     # rows for row‑wise ops

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (fp32 → fp16) – Triton
    # ------------------------------------------------------------------
    x_flat = x.view(total_rows, dim).contiguous()          # [R, dim]  fp32
    x_norm = torch.empty(total_rows, dim, dtype=torch.float16, device=device)

    # launch configuration – tuned for H100
    BLOCK_M = 256                # rows per program
    BLOCK_D = 32                 # columns per iteration
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    row_layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],   # γ  (float32)
        weights["norm.bias"],     # β  (float32)
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fuse left/right projections + gates into a single GEMM
    # ------------------------------------------------------------------
    H = hidden_dim
    # concat weights → shape [4·H, dim]
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                 # fp16, contiguous

    # [R, dim] @ [4·H, dim]ᵀ  →  [R, 4·H]   (fp16)
    proj_gate = F.linear(x_norm, w_concat)

    left_proj_raw  = proj_gate[:, 0:H]
    right_proj_raw = proj_gate[:, H:2 * H]
    left_gate_raw  = proj_gate[:, 2 * H:3 * H]
    right_gate_raw = proj_gate[:, 3 * H:4 * H]

    # output‑gate (separate linear)
    out_gate_raw = F.linear(
        x_norm,
        weights["out_gate.weight"].to(torch.float16),   # [H, dim]
    )   # fp16

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask, sigmoid gates and gating
    # ------------------------------------------------------------------
    if not cfg.get("nomask", True):
        # mask shape [B,N,N] → broadcast to [R,1]
        mask_fp16 = mask.to(torch.float16).reshape(total_rows, 1)
        left_proj_raw = left_proj_raw * mask_fp16
        right_proj_raw = right_proj_raw * mask_fp16

    # inplace sigmoid for speed
    left_gate = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate = out_gate_raw.sigmoid_()

    left = left_proj_raw * left_gate        # [R, H]  fp16
    right = right_proj_raw * right_gate     # [R, H]  fp16

    # ------------------------------------------------------------------
    # 4️⃣ Tri‑multiplicative update (batched GEMM)
    # ------------------------------------------------------------------
    # reshape to [B, N, N, H] → bring H to batch dim
    left  = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    left_mat  = left.view(B * H, N, N)    # [B·H, N, N]
    right_mat = right.view(B * H, N, N)   # [B·H, N, N]

    # (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # back to [B, N, N, H] (still fp16)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm over the hidden dimension (fp16)
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, H).contiguous()
    out_norm = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    row_layernorm_kernel[grid2](
        out_flat,
        out_norm,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    out_norm = out_norm.view(B, N, N, H)   # fp16

    # ------------------------------------------------------------------
    # 6️⃣ Apply output gate + final projection back to `dim`
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate.view(B, N, N, H)   # [B,N,N,H] fp16

    final_half = F.linear(
        out_norm.view(total_rows, H),
        weights["to_out.weight"].to(torch.float16),   # [dim, H]
    )   # fp16

    # cast to float32 as required by the model
    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
