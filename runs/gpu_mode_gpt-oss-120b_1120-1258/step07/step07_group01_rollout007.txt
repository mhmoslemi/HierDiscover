def _batched_matmul_triton(left: torch.Tensor, right: torch.Tensor, block_m=64, block_n=64, block_k=32):
    # left and right shape [batch, N, N], contiguous with row-major layout.
    # right will be used transposed.
    batch, N, _ = left.shape
    out = torch.empty_like(left) # same dtype and shape

    # compute strides in terms of elements: left.stride() returns (N*N, N, 1)
    left_stride_batch, left_stride_m, left_stride_k = left.stride()
    right_stride_batch, right_stride_row, right_stride_col = right.stride() # same as left
    
    # For right transposed, we set stride_bk = right_stride_col (1)
    # stride_bn = right_stride_row (N)

    # Output strides
    out_stride_batch, out_stride_m, out_stride_n = out.stride()

    grid = (triton.cdiv(N, block_m), triton.cdiv(N, block_n), batch)
    _tri_mul_kernel[grid](
        left,
        right,
        out,
        batch,
        N,
        N,
        N,
        left_stride_batch,
        left_stride_m,
        left_stride_k,
        right_stride_batch,
        right_stride_col,  # stride_bk
        right_stride_row, # stride_bn
        out_stride_batch,
        out_stride_m,
        out_stride_n,
        BLOCK_M=block_m,
        BLOCK_N=block_n,
        BLOCK_K=block_k,
        num_warps=4,
    )
    return out

"""
Fast outgoing TriMul implementation (AlphaFold‑3).

Algorithm
---------
1️⃣  Row‑wise LayerNorm (float32 → float16) on the input tensor.
2️⃣  Fuse the four linear projections/gates (left/right proj & gate) into a
    single half‑precision GEMM.
3️⃣  Apply optional mask, sigmoid gates and element‑wise multiplication.
4️⃣  Reshape to (B, H, N, N) and perform the core N³ update:
        out[b, h, i, j] = Σ_k left[b, h, i, k] * right[b, h, j, k]
    This is a batched half‑precision GEMM (B·H independent ∈‑matrices) performed
    by a custom Triton kernel – one launch handles the whole batch,
    eliminating per‑batch kernel‑launch overhead.
5️⃣  Second LayerNorm over the hidden dimension.
6️⃣  Apply the output gate and a final half‑precision projection back to `dim`
    (float32 output).

All heavy linear work is done in fp16, while the two LayerNorms are
implemented in Triton for low‑overhead reduction across the (very large)
row dimension. The final result matches the reference PyTorch module.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (fp32 → fp16)
# ----------------------------------------------------------------------
@triton.jit
def _layernorm_kernel(
    x_ptr,          # const   * [rows, dim] (float32 or fp16)
    y_ptr,          # mutable * [rows, dim] (float16)
    weight_ptr,     # const   * [dim]      (float32) – gamma
    bias_ptr,       # const   * [dim]      (float32) – beta
    rows,           # int32   (runtime)
    dim,            # int32   (runtime)
    eps,            # float32 (compile‑time)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per program
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)            # [BLOCK_M]
    mask_m = offs_m < rows

    # ------------------------------------------------------------------
    # 1️⃣ compute per‑row mean & variance
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)               # fp32 load (or fp16 → fp32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                            # 1 / sqrt(var+eps)

    # ------------------------------------------------------------------
    # 2️⃣ write normalized output in fp16
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)
        norm = (vals - mean[:, None]) * rstd[:, None]      # (x-μ)/σ

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # gamma
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # beta
        out = norm * w[None, :] + b[None, :]               # fp32

        tl.store(
            y_ptr + offs_m[:, None] * dim + offs_d[None, :],
            out.to(tl.float16),
            mask=mask_m[:, None] & mask_d[None, :],
        )


def _layernorm_triton(
    x: torch.Tensor,
    weight: torch.Tensor,
    bias: torch.Tensor,
    eps: float = 1e-5,
    block_m: int = 256,
    block_d: int = 32,
) -> torch.Tensor:
    """
    Row‑wise LayerNorm on a 2‑D tensor ``x`` of shape [rows, dim] (float32 or fp16)
    → fp16 tensor of the same shape.
    """
    rows, dim = x.shape
    y = torch.empty(rows, dim, dtype=torch.float16, device=x.device)

    grid = (triton.cdiv(rows, block_m),)
    _layernorm_kernel[grid](
        x,
        y,
        weight,
        bias,
        rows,
        dim,
        eps,
        BLOCK_M=block_m,
        BLOCK_D=block_d,
        num_warps=4,
    )
    return y


# ----------------------------------------------------------------------
# Triton kernel: batched half‑precision GEMM for the N³ update
# Computes   out[b, i, j] = Σ_k left[b, i, k] * right[b, j, k]
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    left_ptr,   # const   * [batch, M, K] (fp16)
    right_ptr,  # const   * [batch, K, N] (fp16) – we read it transposed
    out_ptr,    # mutable * [batch, M, N] (fp16)

    batch,      # int32   (runtime)
    M,          # int32   (runtime) – rows of left ( = N )
    N,          # int32   (runtime) – cols of right ( = N )
    K,          # int32   (runtime) – shared dimension ( = N )

    stride_al, stride_am, stride_ak,   # left strides
    stride_br, stride_bk, stride_bn,   # right strides (bk = col‑stride, bn = row‑stride for transposed view)
    stride_cl, stride_cm, stride_cn,   # out strides

    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)   # tile row
    pid_n = tl.program_id(1)   # tile col
    pid_b = tl.program_id(2)   # batch index

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in fp32 for precision
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Base offsets for this batch
    a_batch_off = pid_b * stride_al
    b_batch_off = pid_b * stride_br
    c_batch_off = pid_b * stride_cl

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile   [BLOCK_M, BLOCK_K]
        a_ptrs = left_ptr + a_batch_off + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)

        # Load B tile (transposed)   [BLOCK_K, BLOCK_N]
        #   B_T[k, n] = B[n, k] in memory → row stride = stride_bn, col stride = stride_bk
        b_ptrs = right_ptr + b_batch_off + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        ).to(tl.float32)

        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_N)

    # Write result
    c_ptrs = out_ptr + c_batch_off + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )


def _batched_matmul_triton(
    left: torch.Tensor,
    right: torch.Tensor,
    block_m: int = 64,
    block_n: int = 64,
    block_k: int = 32,
) -> torch.Tensor:
    """
    Batched half‑precision GEMM: out = left @ rightᵀ
    left, right : [B, N, N] (fp16, row‑major)
    Returns out  : [B, N, N] (fp16)
    """
    assert left.dtype == torch.float16 and right.dtype == torch.float16
    batch, M, K = left.shape
    _, K2, N = right.shape
    assert K == K2 and M == N, "Both matrices must be square and share the inner dimension."

    out = torch.empty_like(left)

    # strides in elements
    stride_al, stride_am, stride_ak = left.stride()
    stride_br, stride_br_row, stride_br_col = right.stride()  # row‑major: (N*N, N, 1)

    # for the transposed view we use:
    stride_bk = stride_br_col                     # column stride (1)
    stride_bn = stride_br_row                     # row stride    (N)

    stride_cl, stride_cm, stride_cn = out.stride()

    grid = (
        triton.cdiv(M, block_m),
        triton.cdiv(N, block_n),
        batch,
    )
    _tri_mul_kernel[grid](
        left,
        right,
        out,
        batch,
        M,
        N,
        K,
        stride_al,
        stride_am,
        stride_ak,
        stride_br,
        stride_bk,
        stride_bn,
        stride_cl,
        stride_cm,
        stride_cn,
        BLOCK_M=block_m,
        BLOCK_N=block_n,
        BLOCK_K=block_k,
        num_warps=8,
    )
    return out

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) – ignored if config["nomask"] is True
        - weights_dict : mapping of all Linear / LayerNorm parameters (float32)
        - config_dict  : must contain "dim", "hidden_dim" and optional "nomask"

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # 0️⃣ unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N  # R = B·N·N

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → fp16
    # --------------------------------------------------------------
    x_flat = input_tensor.view(total_rows, dim).contiguous()  # [R, dim] fp32
    x_norm = _layernorm_triton(
        x_flat,
        weights["norm.weight"],
        weights["norm.bias"],
        eps=eps,
    )  # fp16, shape [R, dim]

    # --------------------------------------------------------------
    # 2️⃣ Fused projection + gate linear (produces 4·H outputs)
    # --------------------------------------------------------------
    H = hidden_dim
    # Concatenate the four projection/gate weight matrices (each [H, dim])
    w_proj_gate = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)                      # [4·H, dim]

    # X_norm (R, dim) @ Wᵀ (dim, 4·H) → (R, 4·H)  (all fp16)
    proj_gate = F.linear(x_norm, w_proj_gate)

    # split
    left_proj_raw  = proj_gate[:, 0:H]                     # [R, H]
    right_proj_raw = proj_gate[:, H:2 * H]                 # [R, H]
    left_gate_raw  = proj_gate[:, 2 * H:3 * H]             # [R, H]
    right_gate_raw = proj_gate[:, 3 * H:4 * H]             # [R, H]

    # output gate (separate linear)
    out_gate_raw = F.linear(
        x_norm,
        weights["out_gate.weight"].to(torch.float16),     # [H, dim]
    )                                                     # [R, H]

    # --------------------------------------------------------------
    # 3️⃣ Optional mask, sigmoid gating
    # --------------------------------------------------------------
    if not nomask:
        # mask → [R, 1] (fp16) and broadcast over hidden dim
        mask_h = mask_tensor.to(torch.float16).reshape(total_rows, 1)
        left_proj_raw = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    # in‑place sigmoid (stay in fp16)
    left_gate = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate = out_gate_raw.sigmoid_()

    # Apply element‑wise gates
    left = left_proj_raw * left_gate      # [R, H] fp16
    right = right_proj_raw * right_gate   # [R, H] fp16

    # --------------------------------------------------------------
    # 4️⃣ Reshape for the N³ multiplicative update
    # --------------------------------------------------------------
    left = left.view(B, N, N, H)          # [B, N, N, H]
    right = right.view(B, N, N, H)        # [B, N, N, H]
    out_gate = out_gate.view(B, N, N, H)  # [B, N, N, H]

    # Bring hidden dimension to axis‑1 → (B, H, N, N) and make contiguous
    left_bh = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_bh = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    # Collapse batch & head into a single GEMM batch dimension
    left_flat = left_bh.view(-1, N, N)   # [B·H, N, N] fp16
    right_flat = right_bh.view(-1, N, N) # [B·H, N, N] fp16

    # Core N³ update via custom Triton batched GEMM
    out_flat = _batched_matmul_triton(left_flat, right_flat)

    # Restore (B, H, N, N) layout
    out_bh = out_flat.view(B, H, N, N)                # [B, H, N, N]
    out = out_bh.permute(0, 2, 3, 1).contiguous()    # [B, N, N, H]

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm over hidden dimension → fp16
    # --------------------------------------------------------------
    out_flat2 = out.view(total_rows, H).contiguous()   # [R, H] fp16
    out_norm = _layernorm_triton(
        out_flat2,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        eps=eps,
    )                                                   # fp16, [R, H]
    out_norm = out_norm.view(B, N, N, H)                # [B, N, N, H]

    # --------------------------------------------------------------
    # 6️⃣ Apply output‑gate and final projection back to `dim`
    # --------------------------------------------------------------
    out_norm = out_norm * out_gate                     # [B, N, N, H] fp16

    final_half = F.linear(
        out_norm.view(total_rows, H),
        weights["to_out.weight"].to(torch.float16),   # [dim, H]
    )                                                   # fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
```
