"""
custom_kernel
=============
Highly‑optimized forward pass for the *outgoing* TriMul operator (AlphaFold‑3).

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the input tensor (FP32 → FP16) – custom Triton kernel.
2️⃣ Fuse the four linear projections (left, right, left‑gate, right‑gate) into a
   single half‑precision GEMM (R×dim)·(4·H×dim)ᵀ → (R,4·H).
3️⃣ Compute the output‑gate projection (half‑precision GEMM).
4️⃣ Apply optional mask, then in‑place sigmoid on the gate tensors and element‑wise
   gating of the left/right projections.
5️⃣ Batched matrix multiplication (B·H, N, N) × (B·H, N, N)ᵀ → (B·H, N, N)
   by using ``torch.matmul`` on a non‑contiguous view – no extra copies.
6️⃣ Fuse hidden‑dim LayerNorm and multiplication by the output‑gate
   (custom Triton kernel).
7️⃣ Final linear projection back to ``dim`` (half‑precision GEMM) and cast to FP32.

Only the two LayerNorm passes are written in Triton; the heavy GEMM /
batched‑matmul work is delegated to cuBLAS (via ``torch.nn.functional.linear``
and ``torch.matmul``), which on an H100 runs at Tensor‑Core speed.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   * [total_rows, dim]   (FP32)
    y_ptr,          # mutable * [total_rows, dim]   (FP16)
    gamma_ptr,      # const   * [dim]               (FP32)
    beta_ptr,       # const   * [dim]               (FP32)
    total_rows,     # int (runtime)
    dim,            # int (runtime)
    eps,            # float (runtime)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per inner loop
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ---- mean / variance (FP32) ----
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    cols = tl.cdiv(dim, BLOCK_D)
    for d_off in range(cols):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)               # 1 / sqrt(var+eps)

    # ---- write normalized tensor (γ·(x‑μ)·rstd + β) ----
    for d_off in range(cols):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(gamma_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(beta_ptr   + offs_d, mask=mask_d, other=0.0)   # β

        out = norm * w[None, :] + b[None, :]                     # FP32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2️⃣ LayerNorm + gate fusion (hidden dim)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_gate_kernel(
    x_ptr,          # const   * [total_rows, dim]   (FP16)
    y_ptr,          # mutable * [total_rows, dim]   (FP16)
    gamma_ptr,      # const   * [dim]               (FP32)
    beta_ptr,       # const   * [dim]               (FP32)
    gate_ptr,       # const   * [total_rows, dim]   (FP16)
    total_rows,     # int (runtime)
    dim,            # int (runtime)
    eps,            # float (runtime)
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # ---- mean / variance (FP32) ----
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    cols = tl.cdiv(dim, BLOCK_D)
    for d_off in range(cols):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- normalized + gate *****
    for d_off in range(cols):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        # load normalized input
        x_ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(x_ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        # load gate (FP16 → FP32)
        g_ptrs = gate_ptr + offs_m[:, None] * dim + offs_d[None, :]
        gate = tl.load(g_ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(gamma_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(beta_ptr   + offs_d, mask=mask_d, other=0.0)   # β

        out = (norm * w[None, :] + b[None, :]) * gate          # FP32
        out_fp16 = out.to(tl.float16)

        y_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(y_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 3️⃣ Forward entry point (required by the grader)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype torch.float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool / float) or None
        - weights_dict : dict of parameter tensors (see code)
        - config_dict  : must contain ``dim`` and ``hidden_dim``; optional key
                         ``nomask`` (bool, default True)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (torch.float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]          # H
    eps = 1e-5
    device = input_tensor.device

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N                     # R = B·N·N
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    x_flat = input_tensor.view(total_rows, dim).contiguous()   # [R, dim] FP32
    x_norm = torch.empty(total_rows, dim, dtype=torch.float16, device=device)

    BLOCK_M = 256
    BLOCK_D = 64
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],      # γ (FP32)
        weights["norm.bias"],        # β (FP32)
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection for left/right and their gates
    # ------------------------------------------------------------------
    # convert weight tensors once to FP16
    w_left   = weights["left_proj.weight"].to(torch.float16)    # [H, dim]
    w_right  = weights["right_proj.weight"].to(torch.float16)   # [H, dim]
    w_lgate  = weights["left_gate.weight"].to(torch.float16)   # [H, dim]
    w_rgate  = weights["right_gate.weight"].to(torch.float16)  # [H, dim]

    # concatenate along output dimension → [4·H, dim]
    w_concat = torch.cat([w_left, w_right, w_lgate, w_rgate], dim=0)

    # (R, dim) @ (4·H, dim).T  → (R, 4·H)   (FP16)
    proj_gate = F.linear(x_norm, w_concat)                # [R, 4·H] FP16

    left_proj_raw  = proj_gate[:, 0:H]                # [R, H]
    right_proj_raw = proj_gate[:, H:2 * H]            # [R, H]
    left_gate_raw  = proj_gate[:, 2 * H:3 * H]        # [R, H]
    right_gate_raw = proj_gate[:, 3 * H:4 * H]        # [R, H]

    # ------------------------------------------------------------------
    # 3️⃣ Output‑gate projection (HP → dim)
    # ------------------------------------------------------------------
    w_out_gate = weights["out_gate.weight"].to(torch.float16)    # [H, dim]
    out_gate_raw = F.linear(x_norm, w_out_gate)                 # [R, H] FP16

    # ------------------------------------------------------------------
    # 4️⃣ Apply mask (if present) & sigmoid gates
    # ------------------------------------------------------------------
    if not config.get("nomask", True):
        # mask shape [B, N, N] → [R, 1] (FP16) for broadcasting
        mask_h = mask_tensor.to(torch.float16).reshape(total_rows, 1)
        left_proj_raw  = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    left_gate = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate = out_gate_raw.sigmoid_()

    # element‑wise gating
    left = left_proj_raw * left_gate          # [R, H] FP16
    right = right_proj_raw * right_gate       # [R, H] FP16

    # ------------------------------------------------------------------
    # 5️⃣ Batched matrix multiplication (B·H, N, N) @ (B·H, N, N)ᵀ
    #    No extra copies – work on non‑contiguous views.
    # ------------------------------------------------------------------
    # reshape to [B, N, N, H]
    left = left.view(B, N, N, H)
    right = right.view(B, N, N, H)

    # (B, H, N, N) where the inner‑product dimension (k) is the *last* axis
    left = left.permute(0, 3, 1, 2)            # (B, H, N, N)
    right = right.permute(0, 3, 2, 1)          # (B, H, N, N)  – k‑axis moved to dim‑2

    # (B, H, N, N) @ (B, H, N, N)ᵀ  → (B, H, N, N)
    out_mat = torch.matmul(left, right)       # uses cuBLAS, stays in FP16

    # back to [B, N, N, H]
    out = out_mat.permute(0, 2, 3, 1).contiguous()   # (B, N, N, H)
    out_flat = out.view(total_rows, H)                # (R, H)

    # ------------------------------------------------------------------
    # 6️⃣ Fused LayerNorm over hidden dim + multiplication by out‑gate
    # ------------------------------------------------------------------
    out_norm_gated = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_gate_kernel[grid2](
        out_flat,
        out_norm_gated,
        weights["to_out_norm.weight"],   # γ (FP32)
        weights["to_out_norm.bias"],     # β (FP32)
        out_gate.view(total_rows, H),    # gate (FP16)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 7️⃣ Final linear projection back to ``dim``
    # ------------------------------------------------------------------
    w_to_out = weights["to_out.weight"].to(torch.float16)   # [dim, H]
    final_half = F.linear(out_norm_gated, w_to_out)         # (R, dim) FP16

    # cast back to FP32 as required by the reference implementation
    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
