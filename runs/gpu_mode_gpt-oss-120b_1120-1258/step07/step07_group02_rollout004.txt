"""
Outgoing TriMul (AlphaFold‑3) – Triton‑accelerated forward pass.

Algorithm
---------
1️⃣  Row‑wise LayerNorm (FP32→FP16) on the input tensor.
2️⃣  Fuse the four linear maps (left_proj, right_proj, left_gate, right_gate)
    into a single GEMM:  W₄ₕ = [Wₗₚ; Wᵣₚ; Wₗg; Wᵣg]  (shape 4·H × dim).
3️⃣  Compute the separate out‑gate linear map.
4️⃣  (Optional) apply the pairwise mask, apply sigmoids to the three gates,
    and multiply the projections by their gates.
5️⃣  Reshape to (B,N,N,H) and perform the cubic update:
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h].
    This is a batched GEMM over the combined batch‑hidden dimension
    (B·H, N, N) → (B·H, N, N).
6️⃣  Row‑wise LayerNorm over the hidden dimension H.
7️⃣  Multiply by the out‑gate, then a final linear projection back to the
    original feature dimension dim.
All heavy linear work stays in FP16 to fully exploit Tensor‑Cores.
The final result is cast back to float32 to match the reference
implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton row‑wise LayerNorm (input FP32, output FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const  * [total_rows, dim] (float32)
    y_ptr,          # mutable* [total_rows, dim] (float16)
    w_ptr,          # const  * [dim]            (float32) – gamma
    b_ptr,          # const  * [dim]            (float32) – beta
    total_rows,     # int32
    dim,            # int32
    eps,            # float32
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per inner loop
):
    """
    Per‑row layer‑norm:
      μ = mean(x, dim=-1)
      σ² = var(x, dim=-1)
      y = ((x - μ) / sqrt(σ² + eps)) * γ + β
    The computation is performed in FP32, the result is stored in FP16.
    """
    pid = tl.program_id(0)

    # ------------------------------------------------------------------
    # 1️⃣  Compute row sums & sum‑of‑squares
    # ------------------------------------------------------------------
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    # iterate over feature dimension in tiles
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        # linear address: row * dim + col
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        sum_val += tl.sum(vals, 1)          # sum over columns → [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, 1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)   # 1 / sqrt(var + eps)

    # ------------------------------------------------------------------
    # 2️⃣  Write normalized values (store as FP16)
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)

        out = norm * gamma[None, :] + beta[None, :]   # FP32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16, mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float) – optional
        - weights_dict : model parameters (see keys below)
        - config_dict  : must contain "dim", "hidden_dim", "nomask"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim          = config["dim"]
    hidden_dim   = config["hidden_dim"]
    eps          = 1e-5
    device       = input_tensor.device
    B, N, _, _   = input_tensor.shape
    total_rows   = B * N * N          # R = B·N·N
    H            = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → FP16
    # --------------------------------------------------------------
    x_flat = input_tensor.reshape(total_rows, dim).contiguous()  # [R, dim] FP32
    x_norm = torch.empty(total_rows, dim, dtype=torch.float16, device=device)

    BLOCK_M = 256                     # rows per program (tunable)
    BLOCK_D = 128                     # features per inner loop (covers dim ≤ 384)
    grid = (triton.cdiv(total_rows, BLOCK_M),)

    layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],    # gamma (FP32)
        weights["norm.bias"],      # beta  (FP32)
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    # --------------------------------------------------------------
    # 2️⃣ Fuse the four projection/gate linear maps (4·H × dim)
    # --------------------------------------------------------------
    w_4h = torch.cat(
        [
            weights["left_proj.weight"],   # (H, dim)
            weights["right_proj.weight"],  # (H, dim)
            weights["left_gate.weight"],   # (H, dim)
            weights["right_gate.weight"],  # (H, dim)
        ],
        dim=0,
    ).to(torch.float16)                     # (4·H, dim)

    proj_gate = F.linear(x_norm, w_4h)       # (R, 4·H) FP16
    left_proj_raw  = proj_gate[:, 0:H]
    right_proj_raw = proj_gate[:, H:2 * H]
    left_gate_raw  = proj_gate[:, 2 * H:3 * H]
    right_gate_raw = proj_gate[:, 3 * H:4 * H]

    # out‑gate (separate linear map)
    out_gate_raw = F.linear(
        x_norm,
        weights["out_gate.weight"].to(torch.float16)   # (H, dim)
    )   # (R, H) FP16

    # --------------------------------------------------------------
    # 3️⃣ Optional mask, sigmoid on gates and element‑wise gating
    # --------------------------------------------------------------
    if not config.get("nomask", True):
        # mask: [B,N,N] → [R,1] (float16)
        mask_fp16 = mask_tensor.reshape(total_rows, 1).to(torch.float16)
        left_proj_raw  = left_proj_raw * mask_fp16
        right_proj_raw = right_proj_raw * mask_fp16

    # in‑place sigmoid for speed
    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    # apply gates
    left  = left_proj_raw * left_gate        # (R, H) FP16
    right = right_proj_raw * right_gate      # (R, H) FP16

    # --------------------------------------------------------------
    # 4️⃣ Reshape to (B, N, N, H)
    # --------------------------------------------------------------
    left = left.view(B, N, N, H)
    right = right.view(B, N, N, H)
    out_gate = out_gate.view(B, N, N, H)

    # --------------------------------------------------------------
    # 5️⃣ Cubic update via batched GEMM:
    #    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    # --------------------------------------------------------------
    # bring H into the batch dimension → (B·H, N, N)
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Batched matrix multiply A @ Bᵀ  (A = left_mat, B = right_mat)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B·H, N, N)

    # Reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm over hidden dimension H (row‑wise)
    # --------------------------------------------------------------
    out_flat = out.reshape(total_rows, H).contiguous()          # (R, H)
    out_norm = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm,
        weights["to_out_norm.weight"],   # gamma (FP32)
        weights["to_out_norm.bias"],     # beta  (FP32)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    out_norm = out_norm.view(B, N, N, H)   # (B, N, N, H) FP16

    # --------------------------------------------------------------
    # 7️⃣ Apply output gate and final linear projection back to dim
    # --------------------------------------------------------------
    gated = out_norm * out_gate                       # (B,N,N,H) FP16

    final = F.linear(
        gated.view(total_rows, H),
        weights["to_out.weight"].to(torch.float16)   # (dim, H)
    )   # (R, dim) FP16

    final = final.view(B, N, N, dim).to(torch.float32)

    return final
