@triton.jit
def mask_gate_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    out_left_ptr, out_right_ptr,
    total_elems,
    BLOCK_SIZE: tl.constexpr,
    USE_MASK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems
    left = tl.load(left_ptr + offs, mask=mask, other=0.0).to(tl.float32)

    # convert weight/bias to fp16 for speed
    x_norm = torch.nn.functional.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"].to(device=device, dtype=torch.float32),
        bias=weights["norm.bias"].to(device=device, dtype=torch.float32),
        eps=eps,
    ).to(torch.float16)

    # 2. Fused projection + gate linear (4H)
    w_proj = torch.cat([
        weights["left_proj.weight"],
        weights["right_proj.weight"],
        weights["left_gate.weight"],
        weights["right_gate.weight"],
    ], dim=0).to(device=device, dtype=torch.float16)  # [4H, dim]

    # flatten normalized tensor to [R, dim]
    x_flat = x_norm.view(total_rows, dim)

    fused = torch.nn.functional.linear(x_flat, w_proj)  # [R, 4H] fp16

    # split
    left_raw = fused[:, 0:H]
    right_raw = fused[:, H:2*H]
    left_gate_raw = fused[:, 2*H:3*H]
    right_gate_raw = fused[:, 3*H:4*H]

    # out gate
    out_gate_raw = torch.nn.functional.linear(
        x_flat,
        weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    )  # [R, H] fp16

    # Apply mask and gates in fused kernel
    if not nomask and mask is not None:
        mask_flat = mask.to(device=device, dtype=torch.float16).reshape(total_rows, 1)
    else:
        mask_flat = None

    # allocate output tensors
    left = torch.empty_like(left_raw)
    right = torch.empty_like(right_raw)

    total_elem = left_raw.numel() # R*H
    BLOCK_SIZE = 1024
    grid = (triton.cdiv(total_elem, BLOCK_SIZE),)
    mask_gate_kernel[grid](
        left_raw,
        right_raw,
        left_gate_raw,
        right_gate_raw,
        mask_flat if mask_flat is not None else torch.empty(0, device=device, dtype=torch.float16),
        left,
        right,
        total_elem,
        H,
        USE_MASK=0 if nomask else 1,
        BLOCK_SIZE=BLOCK_SIZE,
        num_warps=4,
    )

    out_gate = out_gate_raw.sigmoid_()  # [R, H] fp16

    # reshape for batched matmul
    left = left.view(B, N, N, H)
    right = right.view(B, N, N, H)
    out_gate = out_gate.view(B, N, N, H)

    # permute to (B, H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    left_mat = left_perm.view(B * H, N, N)
    right_mat = right_perm.view(B * H, N, N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # restore shape
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # second layernorm (fp16 output)

    out_norm = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(H,),

        bias=weights["to_out_norm.bias"].to(device=device, dtype=torch.float32),
        eps=eps,
    ).to(torch.float16)

    out_norm = out_norm * out_gate

    # final linear

        out_norm.view(total_rows, H),
        weights["to_out.weight"].to(device=device, dtype=torch.float16)
    )
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

"""
Highly‑optimized forward pass for the **outgoing** TriMul operator
(AlphaFold‑3 style).

Algorithm
---------
1. Cast the input to FP16 and apply a row‑wise LayerNorm (torch
   implementation – fast and fully fused).  The LayerNorm parameters
   are kept in FP16, the reduction is performed in FP32 internally.

2. Fuse the four linear projections (left, right, left‑gate, right‑gate)
   into a single GEMM (`[R,dim] @ [4·H,dim]^T`) producing a tensor of
   shape `[R,4·H]`.  Split the result into the four components.

3. Compute the output‑gate (`out_gate`) with a second GEMM
   (`[R,dim] @ [H,dim]^T`).

4. A Triton kernel fuses three element‑wise operations for *left* and
   *right*:
        * optional mask multiplication,
        * sigmoid activation of the gate tensors,
        * final element‑wise scaling (`left = left_raw*mask*sigmoid(gate)`).
   This saves two passes over the large tensors and eliminates
   temporary gate tensors.

5. Reshape the gated tensors to `[B,H,N,N]` and perform the
   N³‑multiplicative update via a batched matrix multiplication
   (`torch.bmm`).  This leverages cuBLAS Tensor‑Cores on the H100.

6. Apply a second LayerNorm (again the PyTorch implementation) over
   the hidden dimension.

7. Multiply by the output‑gate, apply the final linear projection
   (`to_out`) and cast back to FP32.

Only the mask‑gate fusion is written in Triton; all large dense GEMMs
use the highly‑optimized cuBLAS kernels.  The implementation works for
all configurations described in the prompt.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fuse mask, sigmoid(gate) and element‑wise scaling.
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_kernel(
    left_ptr,               # fp16,   [total_elems]
    right_ptr,              # fp16,   [total_elems]
    left_gate_ptr,          # fp16,   [total_elems]
    right_gate_ptr,         # fp16,   [total_elems]
    mask_ptr,               # fp16,   [total_rows]   (used only if USE_MASK=1)
    out_left_ptr,           # fp16,   [total_elems] (output)
    out_right_ptr,          # fp16,   [total_elems] (output)
    total_elems,            # int,    total number of elements (=R*H)
    hidden_dim,             # int,    H
    total_rows,             # int,    R = B*N*N
    USE_MASK: tl.constexpr,   # compile‑time bool
    BLOCK_SIZE: tl.constexpr, # compile‑time block size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)          # [BLOCK_SIZE]
    valid = offs < total_elems

    # ------------------------------------------------------------------
    # Load the four raw tensors
    # ------------------------------------------------------------------
    left      = tl.load(left_ptr      + offs, mask=valid, other=0.0).to(tl.float32)
    right     = tl.load(right_ptr     + offs, mask=valid, other=0.0).to(tl.float32)
    left_g    = tl.load(left_gate_ptr + offs, mask=valid, other=0.0).to(tl.float32)
    right_g   = tl.load(right_gate_ptr+ offs, mask=valid, other=0.0).to(tl.float32)

    # ------------------------------------------------------------------
    # Sigmoid of the gate tensors (in‑place, fp32)
    # ------------------------------------------------------------------
    left_g  = 1.0 / (1.0 + tl.exp(-left_g))
    right_g = 1.0 / (1.0 + tl.exp(-right_g))

    # ------------------------------------------------------------------
    # Optional mask (broadcasted over the hidden dimension)
    # ------------------------------------------------------------------
    if USE_MASK:
        # each element belongs to row = floor(idx / H)
        row_idx   = offs // hidden_dim
        row_valid = row_idx < total_rows
        m_val = tl.load(mask_ptr + row_idx, mask=row_valid, other=1.0).to(tl.float32)
        left  = left  * m_val
        right = right * m_val

    # ------------------------------------------------------------------
    # Apply the gates
    # ------------------------------------------------------------------
    left_out  = (left  * left_g ).to(tl.float16)
    right_out = (right * right_g).to(tl.float16)

    # ------------------------------------------------------------------
    # Store results
    # ------------------------------------------------------------------
    tl.store(out_left_ptr  + offs, left_out,  mask=valid)
    tl.store(out_right_ptr + offs, right_out, mask=valid)

    # 1️⃣ First LayerNorm (fp16 output) – use the highly‑optimized torch kernel
    # ------------------------------------------------------------------
    # Cast to FP16 first to cut memory traffic; torch’s LayerNorm internally
    # computes mean/var in FP32.
    x_fp16 = x.to(torch.float16)
    norm_weight = weights["norm.weight"].to(device=device, dtype=torch.float16)
    norm_bias   = weights["norm.bias"].to(device=device, dtype=torch.float16)
    x_norm = F.layer_norm(x_fp16,
                          normalized_shape=(dim,),
                          weight=norm_weight,
                          bias=norm_bias,
                          eps=eps)                     # [B,N,N,dim] FP16

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection + gate linear (4·H outputs)
    # ------------------------------------------------------------------
    # Concatenate the four projection matrices once (converted to FP16)
    w_proj = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
        ],
        dim=0,
    ).to(device=device, dtype=torch.float16)               # [4·H, dim]

    # flatten to [R, dim] for the linear call
    x_flat = x_norm.view(R, dim)                            # FP16

    fused = F.linear(x_flat, w_proj)                        # [R, 4·H] FP16

    # Split the fused result
    left_raw      = fused[:, 0:H]               # [R, H]
    right_raw     = fused[:, H:2*H]              # [R, H]
    left_gate_raw = fused[:, 2*H:3*H]            # [R, H]
    right_gate_raw= fused[:, 3*H:4*H]            # [R, H]

    # ------------------------------------------------------------------
    # 3️⃣ Separate output‑gate linear
    # ------------------------------------------------------------------
    out_gate_weight = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    out_gate_raw = F.linear(x_flat, out_gate_weight)       # [R, H] FP16

    # ------------------------------------------------------------------
    # 4️⃣ Fuse mask, sigmoid‑gates and element‑wise weighting with Triton
    # ------------------------------------------------------------------
    # Allocate destination tensors for the gated left/right tensors
    left  = torch.empty_like(left_raw)    # FP16, shape [R, H]
    right = torch.empty_like(right_raw)   # FP16, shape [R, H]

    # If a mask is present, keep it FP16; otherwise we will launch the kernel
    # with USE_MASK=0 so the mask pointer can be a dummy tensor.
    if not nomask and mask is not None:
        mask_fp16 = mask.to(device=device, dtype=torch.float16).reshape(R, 1)  # [R,1]
        mask_use = mask_fp16
    else:
        mask_use = torch.empty(0, device=device, dtype=torch.float16)  # dummy

    total_elems = left_raw.numel()          # R * H
    BLOCK_SIZE = 1024
    grid = (triton.cdiv(total_elems, BLOCK_SIZE),)

    mask_gate_kernel[grid](
        left_raw,
        right_raw,
        left_gate_raw,
        right_gate_raw,
        mask_use,
        left,
        right,
        total_elems,
        H,                # hidden_dim
        R,                # total_rows
        USE_MASK=0 if nomask else 1,
        BLOCK_SIZE=BLOCK_SIZE,
        num_warps=4,
    )

    # Output‑gate (sigmoid) – no mask involved
    out_gate = out_gate_raw.sigmoid_()                     # [R, H] FP16

    # ------------------------------------------------------------------
    # 5️⃣ Reshape for the N³ multiplicative update
    # ------------------------------------------------------------------
    left  = left.view(B, N, N, H)        # [B,N,N,H]
    right = right.view(B, N, N, H)       # [B,N,N,H]
    out_gate = out_gate.view(B, N, N, H) # [B,N,N,H]

    # Move the head dimension into the batch axis: [B, H, N, N]
    left_perm  = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    # Collapse (B·H) as the batch dimension for torch.bmm
    left_mat  = left_perm.view(B * H, N, N)           # [B·H, N, N]
    right_mat = right_perm.view(B * H, N, N)          # [B·H, N, N]

    # Batched matrix multiplication → (B·H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Restore shape [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm over the hidden dimension (H)
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_bias   = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)

    out_norm = F.layer_norm(out,
                            normalized_shape=(H,),
                            weight=to_out_norm_weight,
                            bias=to_out_norm_bias,
                            eps=eps)                     # FP16

    # Apply the output‑gate
    out_norm = out_norm * out_gate                               # FP16

    # ------------------------------------------------------------------
    # 7️⃣ Final linear projection & cast back to FP32
    # ------------------------------------------------------------------
    to_out_weight = weights["to_out.weight"].to(device=device, dtype=torch.float16)  # [dim, H]

    final_fp16 = F.linear(out_norm.view(R, H), to_out_weight)    # [R, dim] FP16
    final = final_fp16.view(B, N, N, dim).to(torch.float32)       # FP32 result

    return final
```
