"""
Highly‑optimized forward pass for the **outgoing** TriMul operator
(AlphaFold‑3 style).  The implementation follows the reference PyTorch
module but fuses the most expensive work:

1️⃣ Row‑wise LayerNorm (FP32 → FP16) – custom Triton kernel.
2️⃣ One fused FP16 GEMM that computes left/right projections together
   with their two gates (4·H outputs) in a single `torch.nn.functional.linear`.
3️⃣ Separate GEMM for the output‑gate.
4️⃣ Optional mask application and in‑place sigmoid for the three gates.
5️⃣ Batched matrix multiplication `left @ rightᵀ` (B·H, N, N) performed
   with cuBLAS Tensor‑Cores (fp16).  This implements the Einstein‑summation
   `einsum('... i k d, ... j k d -> ... i j d', left, right)`.
6️⃣ Second LayerNorm over the hidden dimension (still in FP16, Triton‑based).
7️⃣ Element‑wise output‑gate, final linear projection and cast back to FP32.

All heavy element‑wise work (both layer‑norms and optional masking) is
performed in Triton while the large dense matrix products use cuBLAS for
maximum throughput.  The function `custom_kernel` receives a tuple
`(input, mask, weights, config)` and returns the processed tensor
`[B, N, N, dim]` (float32).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (float32 → float16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   * [total_rows, dim]   (float32)
    y_ptr,          # mutable * [total_rows, dim]   (float16)
    weight_ptr,     # const   * [dim]               (float32)
    bias_ptr,       # const   * [dim]               (float32)
    total_rows,     # int (runtime)
    dim,            # int (runtime)
    eps,            # float (compile‑time constexpr)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # cols per iteration
):
    """
    Row‑wise LayerNorm:
        y = ((x - μ) / sqrt(σ² + eps)) * γ + β
    """
    pid = tl.program_id(0)                    # one program = BLOCK_M rows
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)   # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # 1️⃣ Compute per‑row mean and variance
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    # iterate over the dim dimension in chunks of BLOCK_D
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        # load a BLOCK_M × BLOCK_D tile
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                 # fp32 load

        sum_val += tl.sum(vals, axis=1)   # sum over columns
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)           # 1 / sqrt(var + eps)

    # ------------------------------------------------------------------
    # 2️⃣ Write normalized values γ·(x−μ)·rstd + β   (fp16)
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]   # (x-μ)/sqrt(var+eps)

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # β
        out = norm * w[None, :] + b[None, :]                # fp32

        out_fp16 = out.to(tl.float16)
        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

            - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
            - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
            - weights_dict : dict with all Linear/LayerNorm parameters
            - config_dict  : dict with keys "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # 0️⃣ Unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    device = x.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)   # True → ignore mask argument
    eps = 1e-5

    B, N, _, _ = x.shape
    total_rows = B * N * N          # R = B·N·N
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (fp32 → fp16)
    # ------------------------------------------------------------------
    # flatten to [R, dim] for the row‑wise kernel
    x_flat = x.view(total_rows, dim).contiguous()
    ln1_out = torch.empty(total_rows, dim,
                          dtype=torch.float16,
                          device=device)

    # Tune block sizes: 512 rows × 128 columns = 65536 threads per program,
    # comfortably below hardware limits and gives good occupancy.
    BLOCK_M = 512
    BLOCK_D = 128
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        ln1_out,
        weights["norm.weight"],   # γ (float32)
        weights["norm.bias"],     # β (float32)
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fuse left/right projections + their two gates (4·H outputs)
    # ------------------------------------------------------------------
    # concatenate the four projection weight matrices into one
    w_proj = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
        ],
        dim=0,
    ).to(device=device, dtype=torch.float16)   # [4·H, dim]

    # fused GEMM:   [R, dim] @ [4·H, dim]ᵀ  →  [R, 4·H]
    fused = F.linear(ln1_out, w_proj)          # bias omitted, fp16

    # split the fused result
    left_raw       = fused[:, 0:H]
    right_raw      = fused[:, H:2*H]
    left_gate_raw  = fused[:, 2*H:3*H]
    right_gate_raw = fused[:, 3*H:4*H]

    # output‑gate (separate GEMM)
    out_gate_raw = F.linear(
        ln1_out,
        weights["out_gate.weight"].to(device=device, dtype=torch.float16)  # [H, dim]
    )                                            # [R, H]

    # ------------------------------------------------------------------
    # 3️⃣ Mask (if present), sigmoid gates and element‑wise weighting
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask shape: [B, N, N] → [R, 1] (fp16)
        mask_flat = mask.to(device=device, dtype=torch.float16).reshape(total_rows, 1)
        left_raw  = left_raw * mask_flat
        right_raw = right_raw * mask_flat

    # in‑place sigmoid saves temporaries
    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    left  = left_raw  * left_gate   # [R, H]
    right = right_raw * right_gate  # [R, H]

    # ------------------------------------------------------------------
    # 4️⃣ Reshape for the N³ multiplicative update
    # ------------------------------------------------------------------
    left  = left.view(B, N, N, H)
    right = right.view(B, N, N, H)
    out_gate = out_gate.view(B, N, N, H)

    # bring head dimension into the batch axis → [B, H, N, N]
    left_perm  = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    # collapse (B·H) → batch for torch.bmm
    left_mat  = left_perm.view(B * H, N, N)          # [B·H, N, N]
    right_mat = right_perm.view(B * H, N, N)         # [B·H, N, N]

    # batched matrix multiplication using Tensor‑Cores (fp16)
    # result: (B·H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # restore original layout: [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm over the hidden dimension (H)
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, H).contiguous()
    ln2_out = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        ln2_out,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Apply output gate, final linear projection and cast to float32
    # ------------------------------------------------------------------
    out_norm = ln2_out.view(B, N, N, H)          # fp16
    out_norm = out_norm * out_gate               # element‑wise fp16

    # final fp16 linear projection  [R, H] @ [dim, H]ᵀ → [R, dim]
    final_fp16 = F.linear(
        out_norm.view(total_rows, H),
        weights["to_out.weight"].to(device=device, dtype=torch.float16)   # [dim, H]
    )   # fp16, no bias

    # cast back to float32 as required by the API
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
```
