"""
Outgoing TriMul (AlphaFold‑3) forward pass – Triton + PyTorch.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the input tensor (B·N·N × dim).  
   Implemented in Triton (fp32 accumulation → fp16 output).

2️⃣ Fuse the four linear layers (left_proj, right_proj, left_gate,
   right_gate) into a single GEMM:
        [R, dim] @ [4·H, dim]ᵀ → [R, 4·H]   (R = B·N·N).

3️⃣ Compute the output‑gate linear projection separately.

4️⃣ If a mask is provided, apply it (broadcast over the hidden dim)
   to the *projected* left/right tensors, then sigmoid‑gate all
   three gate tensors (in‑place, half‑precision).

5️⃣ Reshape to [B, H, N, N] and perform the N³ multiplicative update
   with a batched half‑precision GEMM:
        out = left @ rightᵀ       →   [B·H, N, N].

6️⃣ Second LayerNorm over the hidden dimension (Triton).

7️⃣ Apply the output‑gate, final linear projection back to `dim`,
   and cast to fp32.

Only the two LayerNorms are custom Triton kernels; all dense matrix
multiplications use cuBLAS (via `torch.nn.functional.linear` /
`torch.bmm`), which is optimal on H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton: row‑wise LayerNorm (fp32 → fp16)
# ----------------------------------------------------------------------
@triton.jit
def rowwise_layernorm_kernel(
    x_ptr,            # const   * [rows, dim]  (float32)
    y_ptr,            # mutable * [rows, dim]  (float16)
    gamma_ptr,        # const   * [dim]       (float32)   scale
    beta_ptr,         # const   * [dim]       (float32)   bias
    rows,             # int runtime
    dim,              # int runtime
    eps,              # compile‑time float
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # cols per tile
):
    """
    Performs LayerNorm for each row independently.
    The reduction (mean / var) uses fp32 for numerical stability.
    Output is stored as fp16.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ---------- 1️⃣  Compute mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]   # [M, D]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)        # 1/sqrt(var+eps)

    # ---------- 2️⃣  Write normalized output ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]          # (x-μ)/σ

        gamma = tl.load(gamma_ptr + offs_d, mask=mask_d, other=0.0)   # scale
        beta  = tl.load(beta_ptr  + offs_d, mask=mask_d, other=0.0)   # bias

        out = norm * gamma[None, :] + beta[None, :]            # fp32
        out_fp16 = out.to(tl.float16)

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : dict of torch.Tensors containing:
            * "norm.weight", "norm.bias"
            * "left_proj.weight", "right_proj.weight",
              "left_gate.weight", "right_gate.weight"
            * "out_gate.weight"
            * "to_out_norm.weight", "to_out_norm.bias"
            * "to_out.weight"
        - config_dict  : must contain keys "dim", "hidden_dim"
                         optional key "nomask" (True → no mask)

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    device = input_tensor.device

    B, N, _, _ = input_tensor.shape
    R = B * N * N                     # rows for the first LN

    # --------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm on the input (fp32 → fp16)
    # --------------------------------------------------------------
    x_flat = input_tensor.view(R, dim).contiguous()          # [R, dim] float32
    x_norm_fp16 = torch.empty(R, dim, dtype=torch.float16,
                              device=device)

    BLOCK_M = 128   # rows per program
    BLOCK_D = 32    # cols per tile
    grid_rows = (R + BLOCK_M - 1) // BLOCK_M

    rowwise_layernorm_kernel[grid_rows](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],    # γ  (float32)
        weights["norm.bias"],      # β  (float32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    # --------------------------------------------------------------
    # 2️⃣ Fuse left/right projections + left/right gates (single GEMM)
    # --------------------------------------------------------------
    #   [R, dim] @ [4·H, dim]ᵀ  →  [R, 4·H]
    proj_gate_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                         # [4·H, dim] in fp16

    proj_gate = F.linear(x_norm_fp16, proj_gate_weight)   # [R, 4·H] fp16

    H = hidden_dim
    left_proj_raw   = proj_gate[:, 0:H]          # [R, H]
    right_proj_raw  = proj_gate[:, H:2*H]        # [R, H]
    left_gate_raw   = proj_gate[:, 2*H:3*H]      # [R, H]
    right_gate_raw  = proj_gate[:, 3*H:4*H]      # [R, H]

    # Output‑gate (separate linear, still on the normalized input)
    out_gate_raw = F.linear(
        x_norm_fp16,
        weights["out_gate.weight"].to(torch.float16)   # [H, dim]
    )   # → [R, H] fp16

    # --------------------------------------------------------------
    # 3️⃣ Optional mask + sigmoid gates + gating
    # --------------------------------------------------------------
    if not config.get("nomask", True):
        # mask: [B, N, N] → [R, 1] (float16)
        mask_flat = mask_tensor.to(torch.float16).view(R, 1)
        left_proj_raw = left_proj_raw * mask_flat
        right_proj_raw = right_proj_raw * mask_flat

    # In‑place sigmoid (half‑precision)
    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    # Apply gates
    left  = left_proj_raw * left_gate      # [R, H] fp16
    right = right_proj_raw * right_gate    # [R, H] fp16

    # --------------------------------------------------------------
    # 4️⃣ N³ multiplicative update  (batched GEMM)
    # --------------------------------------------------------------
    # reshape to [B, N, N, H] → permute → [B·H, N, N] for bmm
    left  = left.view(B, N, N, H)
    right = right.view(B, N, N, H)

    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    left_mat  = left_perm.view(B * H, N, N)      # [B·H, N, N]
    right_mat = right_perm.view(B * H, N, N)     # [B·H, N, N]

    # (B·H, N, N) @ (B·H, N, N)ᵀ  → (B·H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm over hidden dim (fp16)
    # --------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()
    out_norm_fp16 = torch.empty_like(out_flat, dtype=torch.float16)

    grid_out = (R + BLOCK_M - 1) // BLOCK_M
    rowwise_layernorm_kernel[grid_out](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    out_norm = out_norm_fp16.view(B, N, N, H)          # fp16

    # --------------------------------------------------------------
    # 6️⃣ Apply output‑gate, final linear projection, cast to fp32
    # --------------------------------------------------------------
    out_norm = out_norm * out_gate.view(B, N, N, H)    # gating (fp16)

    final_fp16 = F.linear(
        out_norm.view(R, H),
        weights["to_out.weight"].to(torch.float16)    # [dim, H]
    )   # → [R, dim] fp16

    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
