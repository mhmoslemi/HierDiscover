```python
def custom_kernel(data):
    x, mask, weights, config = data
    B,N,_,dim = x.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5
    device = x.device
    # 1. LayerNorm
    x_norm = torch.nn.functional.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps
    )
    # Convert to fp16 for compute
    x_norm_fp16 = x_norm.to(torch.float16)
    # 2. Fuse linear projection and gates
    # combine weight matrices
    w_fused = torch.cat([...], dim=0).to(torch.float16)
    fused = torch.nn.functional.linear(x_norm_fp16, w_fused) # [R, 4H]
    # split
    H = hidden_dim
    left_raw = fused[:, :H]
    right_raw = fused[:, H:2*H]
    left_gate_raw = fused[:, 2*H:3*H]
    right_gate_raw = fused[:, 3*H:4*H]
    # out gate
    out_gate_raw = torch.nn.functional.linear(x_norm_fp16, weights["out_gate.weight"].to(torch.float16))
    # 3. mask + sigmoid
    if not nomask:
        mask_fp16 = mask.to(torch.float16).reshape(-1,1)
        left_raw = left_raw * mask_fp16
        right_raw = right_raw * mask_fp16
    left_gate = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate = out_gate_raw.sigmoid_()
    left = left_raw * left_gate
    right = right_raw * right_gate
    # 4. reshape for N^3 multiplication
    left = left.view(B,N,N,H)
    right = right.view(B,N,N,H)
    out_gate = out_gate.view(B,N,N,H)
    # Bring hidden dimension into batch
    left_mat = left.permute(0,3,1,2).reshape(B*H, N, N)
    right_mat = right.permute(0,3,1,2).reshape(B*H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1,2))
    out = out_mat.view(B, H, N, N).permute(0,2,3,1).contiguous()
    # 5. second layernorm
    out_norm = torch.nn.functional.layer_norm(
        out,
        (H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps
    ).to(torch.float16) # convert to fp16
    # 6. final gated projection using fused Triton kernel
    # combine out_norm and out_gate elementwise and then linear.
    # We'll write a kernel that takes out_norm (R, H), out_gate (R, H), weight (dim, H) and outputs R, dim.
```

"""
TriMul (outgoing) – fast forward pass
=====================================

Algorithm (FP16‑heavy, 2 LayerNorms, 1 fused projection kernel)
----------------------------------------------------------------
1️⃣ Row‑wise LayerNorm over the last channel (torch.nn.functional.layer_norm,
   fp32 → fp16 conversion right after).

2️⃣ Fuse the four linear maps (left/right projections + their gates) into a
   single GEMM:  X_norm @ [left_proj; right_proj; left_gate; right_gate]ᵀ.
   The result is split into *left_raw*, *right_raw*, *left_gate_raw*,
   *right_gate_raw* (all fp16).

3️⃣ Compute the output‑gate with a second GEMM.

4️⃣ Optional binary mask, in‑place sigmoid on the three gate tensors and
   element‑wise multiplication:
        left  = left_raw  * sigmoid(left_gate_raw)
        right = right_raw * sigmoid(right_gate_raw)

5️⃣ Reshape to [B, H, N, N] and perform the N³ update as a batched matrix
   multiplication (Tensor‑Core BMM):
        out[b,h,i,j] = Σₖ left[b,h,i,k] · right[b,h,j,k]ᵀ

6️⃣ Second LayerNorm over the hidden dimension (torch.nn.functional.layer_norm).

7️⃣ Fuse the element‑wise output‑gate with the final linear projection
   (hidden → dim) into a custom Triton kernel:
        out_final = (out_norm * out_gate) @ Wᵀ
   – the kernel performs the per‑row scaling on‑the‑fly and accumulates in
     FP32, writing the result back in FP16.

The final tensor is reshaped to [B, N, N, dim] and cast to float32.
All heavy linear algebra runs in FP16 to exploit H100 Tensor‑Cores.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fuse output‑gate scaling and final linear projection
# ----------------------------------------------------------------------
@triton.jit
def final_proj_kernel(
    out_norm_ptr,   # fp16 * [R, H]   (scaled hidden states)
    out_gate_ptr,   # fp16 * [R, H]   (output gate)
    weight_T_ptr,   # fp16 * [H, D]   (Wᵀ, where D = dim)
    out_ptr,        # fp16 * [R, D]   (final result)
    total_rows,     # int
    hidden_dim,     # int
    out_dim,        # int
    BLOCK_M: tl.constexpr,  # rows per program
    BLOCK_N: tl.constexpr,  # output‑channel tile size
    BLOCK_K: tl.constexpr,  # reduction tile size
):
    pid_m = tl.program_id(0)                # row tile
    pid_n = tl.program_id(1)                # column tile

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)   # [BLOCK_M]
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)   # [BLOCK_N]

    mask_m = offs_m < total_rows
    mask_n = offs_n < out_dim

    # accumulator in FP32 for numeric stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(hidden_dim, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < hidden_dim

        # ------------------------------------------------------------------
        # Load and combine out_norm & out_gate  →  combined = out_norm * out_gate
        # ------------------------------------------------------------------
        out_norm = tl.load(
            out_norm_ptr + offs_m[:, None] * hidden_dim + offs_k[None, :],
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        out_gate = tl.load(
            out_gate_ptr + offs_m[:, None] * hidden_dim + offs_k[None, :],
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        combined = out_norm * out_gate                # fp16

        # ------------------------------------------------------------------
        # Load a tile of Wᵀ : shape (BLOCK_K, BLOCK_N)
        # ------------------------------------------------------------------
        w = tl.load(
            weight_T_ptr + offs_k[:, None] * out_dim + offs_n[None, :],
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )                                              # fp16

        # ------------------------------------------------------------------
        # Accumulate dot product over the reduction dimension K
        # ------------------------------------------------------------------
        acc += tl.dot(combined.to(tl.float32), w.to(tl.float32))

    # write back FP16 result
    out = acc.to(tl.float16)
    out_ptrs = out_ptr + offs_m[:, None] * out_dim + offs_n[None, :]
    tl.store(out_ptrs, out, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Triton‑backed TriMul forward pass (outgoing)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : dictionary with all Linear / LayerNorm parameters
        - config_dict  : dict with keys "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack
    # --------------------------------------------------------------
    x, mask, weights, cfg = data
    dim          = cfg["dim"]
    hidden_dim   = cfg["hidden_dim"]
    nomask       = cfg.get("nomask", True)
    eps          = 1e-5
    device       = x.device

    B, N, _, _ = x.shape
    R = B * N * N                     # total rows = B·N·N
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (fp32) → fp16
    # --------------------------------------------------------------
    x_norm = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )                                 # fp32
    x_norm_fp16 = x_norm.to(torch.float16)   # fp16, contiguous by default

    # --------------------------------------------------------------
    # 2️⃣ Fused linear for left/right projections + gates
    # --------------------------------------------------------------
    # concatenate the four weight matrices (no bias)
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)                 # [4·H, dim]

    fused = F.linear(x_norm_fp16, w_fused)   # [R, 4·H]  (FP16 GEMM)

    left_raw        = fused[:, 0:H]
    right_raw       = fused[:, H:2*H]
    left_gate_raw   = fused[:, 2*H:3*H]
    right_gate_raw  = fused[:, 3*H:4*H]

    # output‑gate (separate GEMM)
    out_gate_raw = F.linear(
        x_norm_fp16,
        weights["out_gate.weight"].to(torch.float16)   # [H, dim]
    )                                           # [R, H]

    # --------------------------------------------------------------
    # 3️⃣ Mask, sigmoid gates and combine
    # --------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] → flatten to [R, 1] (fp16)
        mask_fp16 = mask.to(torch.float16).reshape(R, 1)
        left_raw  = left_raw * mask_fp16
        right_raw = right_raw * mask_fp16

    # in‑place sigmoid (fp16) – cheap
    left_gate   = left_gate_raw.sigmoid_()
    right_gate  = right_gate_raw.sigmoid_()
    out_gate    = out_gate_raw.sigmoid_()

    left  = left_raw * left_gate      # [R, H]  fp16
    right = right_raw * right_gate    # [R, H]  fp16

    # --------------------------------------------------------------
    # 4️⃣ N³ multiplicative update (batched GEMM)
    # --------------------------------------------------------------
    left  = left.view(B, N, N, H)          # [B, N, N, H]
    right = right.view(B, N, N, H)         # [B, N, N, H]
    out_gate = out_gate.view(B, N, N, H)   # keep for later

    # bring hidden dim into batch axis → [B·H, N, N]
    left_mat  = left.permute(0, 3, 1, 2).reshape(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).reshape(B * H, N, N)

    # batched matrix multiplication (Tensor‑Core accelerated)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N]  fp16

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm (fp32 → fp16)
    # --------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        normalized_shape=(H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    ).to(torch.float16)               # fp16

    # --------------------------------------------------------------
    # 6️⃣ Fuse output‑gate scaling + final linear projection (Triton)
    # --------------------------------------------------------------
    # flatten tensors to [R, H] and [R, D]
    out_norm_flat = out_norm.view(R, H).contiguous()
    out_gate_flat = out_gate.view(R, H).contiguous()
    weight_T = weights["to_out.weight"].to(torch.float16).t().contiguous()  # [H, D]

    final_fp16 = torch.empty(R, dim, dtype=torch.float16, device=device)

    # tile sizes – tuned for H100 Tensor‑Core GEMMs
    BLOCK_M = 128    # rows per program
    BLOCK_N = 128    # output‑channel tiles
    BLOCK_K = 32     # reduction tiles (hidden_dim ≤ 128 → fits nicely)

    grid = (
        triton.cdiv(R, BLOCK_M),
        triton.cdiv(dim, BLOCK_N),
    )
    final_proj_kernel[grid](
        out_norm_flat,
        out_gate_flat,
        weight_T,
        final_fp16,
        R,
        H,
        dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # reshape to model output shape and cast back to float32
    out_final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return out_final
```
