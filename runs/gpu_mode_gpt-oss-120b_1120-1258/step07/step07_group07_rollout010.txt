@triton.jit
def proj_gate_mask_kernel(
    proj_ptr, mask_ptr, left_ptr, right_ptr, out_gate_ptr,
    total_rows, hidden_dim,
    BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr, MASKED: tl.constexpr
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)  # [BLOCK_M]
    mask_m = offs_m < total_rows

    # iterate over hidden dim blocks
    for h_off in range(0, tl.cdiv(hidden_dim, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)  # [BLOCK_H]
        mask_h = offs_h < hidden_dim

        # compute pointers
        # left_proj raw idx start 0
        left_proj_ptrs = proj_ptr + offs_m[:, None] * (5*hidden_dim) + offs_h[None, :]  # careful stride: ROW_STRIDE = 5*H

        # Using pointer arithmetic: each row has 5*H entries.

        # However need to compute offsets for each slice: left_proj_offset = 0, right_proj_offset=H, left_gate_offset=2*H, right_gate_offset=3*H, out_gate_offset=4*H.

        # We'll use local base offset.

        base = offs_m[:, None] * (5*hidden_dim)

        # left_proj_raw
        left_proj = tl.load(proj_ptr + base + offs_h[None, :], mask=mask_m[:, None] & mask_h[None, :], other=0.0)

        # right_proj_raw
        right_proj = tl.load(proj_ptr + base + hidden_dim + offs_h[None, :], mask=mask_m[:, None] & mask_h[None, :], other=0.0)

        # left_gate_raw
        left_gate_raw = tl.load(proj_ptr + base + 2*hidden_dim + offs_h[None, :], mask=mask_m[:, None] & mask_h[None, :], other=0.0)

        # right_gate_raw
        right_gate_raw = tl.load(proj_ptr + base + 3*hidden_dim + offs_h[None, :], mask=mask_m[:, None] & mask_h[None, :], other=0.0)

        # out_gate_raw
        out_gate_raw = tl.load(proj_ptr + base + 4*hidden_dim + offs_h[None, :], mask=mask_m[:, None] & mask_h[None, :], other=0.0)

        # sigmoid
        left_gate = 1.0 / (1.0 + tl.exp(-left_gate_raw))
        right_gate = 1.0 / (1.0 + tl.exp(-right_gate_raw))
        out_gate = 1.0 / (1.0 + tl.exp(-out_gate_raw))

        if MASKED:
            mask_val = tl.load(mask_ptr + offs_m, mask=mask_m, other=0.0)[:, None]  # [BLOCK_M,1]
            left_proj = left_proj * mask_val
            right_proj = right_proj * mask_val

        # Apply gate
        left = left_proj * left_gate
        right = right_proj * right_gate

        # store
        tl.store(left_ptr + base + offs_h[None, :], left, mask=mask_m[:, None] & mask_h[None, :])
        tl.store(right_ptr + base + offs_h[None, :], right, mask=mask_m[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + base + offs_h[None, :], out_gate, mask=mask_m[:, None] & mask_h[None, :])

"""
TriMul – outgoing version (AlphaFold‑3) forward pass, highly‑fused Triton/PyTorch.

Key ideas
---------
* Row‑wise LayerNorm (float32 → float16) is performed by a custom Triton kernel.
* The four linear projections (left/right + left/right‑gate) **and** the out‑gate
  are packed into a *single* GEMM (shape (R,5·H)) – this removes one kernel launch.
* A second Triton kernel fuses:
        – optional pair‑wise mask application,
        – sigmoid‑gating,
        – element‑wise multiplication producing the three tensors
          `left`, `right` and `out_gate`.
  All three results stay in FP16, avoiding extra memory traffic.
* The cubic multiplicative update
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
  is exactly a batched matrix multiplication `left @ rightᵀ`,
  executed with cuBLAS/Tensor‑Cores (`torch.bmm`).
* The second LayerNorm is fused with the output‑gate (as in the reference
  implementation) and the final linear projection `to_out` is kept as a
  regular fast GEMM.

The implementation works for all test configurations
(B∈{1,2}, N∈{128,…,1024}, dim∈{128,384},
hidden_dim=128) and supports the presence or absence of a mask.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (float32 → float16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,            # const   * [total_rows, dim]   (float32)
    y_ptr,            # mutable * [total_rows, dim]   (float16)
    weight_ptr,       # const   * [dim]               (float32)
    bias_ptr,         # const   * [dim]               (float32)
    total_rows,       # int (runtime)
    dim,              # int (runtime)
    eps,              # float (runtime)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # feature‑dim block (must ≥ dim)
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # 1️⃣  Compute per‑row mean / variance in fp32
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # fp32 load
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                     # 1 / √(var+eps)

    # ------------------------------------------------------------------
    # 2️⃣  Write normalised values (fp16)
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]                     # fp32
        out_fp16 = out.to(tl.float16)

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2️⃣ Project → mask → sigmoid‑gate (all in one kernel)
# ----------------------------------------------------------------------
@triton.jit
def proj_gate_mask_kernel(
    proj_ptr,         # const   * [total_rows, 5*H]   (fp16)
    mask_ptr,         # const   * [total_rows]       (fp16) – ignored when MASKED==0
    left_ptr,         # mutable * [total_rows, H]   (fp16)
    right_ptr,        # mutable * [total_rows, H]   (fp16)
    out_gate_ptr,     # mutable * [total_rows, H]   (fp16)
    total_rows: tl.constexpr,
    hidden_dim: tl.constexpr,
    MASKED: tl.constexpr,          # 1 ⇒ use mask, 0 ⇒ no mask
    BLOCK_M: tl.constexpr,         # rows per program
    BLOCK_H: tl.constexpr,         # hidden‑dim block (≤ hidden_dim)
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # stride inside the “proj” tensor (5 * H entries per row)
    proj_row_stride = hidden_dim * 5

    # ------------------------------------------------------------------
    # Loop over the hidden dimension in blocks
    # ------------------------------------------------------------------
    for h_off in range(0, tl.cdiv(hidden_dim, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < hidden_dim

        # ---------- load raw tensors ----------
        # base pointer for the current row block (projected tensor)
        base = offs_m[:, None] * proj_row_stride + offs_h[None, :]

        left_proj_raw  = tl.load(proj_ptr + base,
                                 mask=mask_m[:, None] & mask_h[None, :])
        right_proj_raw = tl.load(proj_ptr + base + hidden_dim,
                                 mask=mask_m[:, None] & mask_h[None, :])
        left_gate_raw  = tl.load(proj_ptr + base + 2 * hidden_dim,
                                 mask=mask_m[:, None] & mask_h[None, :])
        right_gate_raw = tl.load(proj_ptr + base + 3 * hidden_dim,
                                 mask=mask_m[:, None] & mask_h[None, :])
        out_gate_raw   = tl.load(proj_ptr + base + 4 * hidden_dim,
                                 mask=mask_m[:, None] & mask_h[None, :])

        # ---------- sigmoid ----------
        left_gate  = 1.0 / (1.0 + tl.exp(-left_gate_raw))
        right_gate = 1.0 / (1.0 + tl.exp(-right_gate_raw))
        out_gate   = 1.0 / (1.0 + tl.exp(-out_gate_raw))

        # ---------- optional mask ----------
        if MASKED:
            # broadcast mask scalar per hidden block
            mask_val = tl.load(mask_ptr + offs_m,
                               mask=mask_m,
                               other=1.0)[:, None]          # [BLOCK_M,1]
            left_proj  = left_proj_raw * mask_val
            right_proj = right_proj_raw * mask_val
        else:
            left_proj  = left_proj_raw
            right_proj = right_proj_raw

        # ---------- apply gates ----------
        left  = left_proj  * left_gate
        right = right_proj * right_gate

        # ---------- store results ----------
        left_ptrs      = left_ptr  + offs_m[:, None] * hidden_dim + offs_h[None, :]
        right_ptrs     = right_ptr + offs_m[:, None] * hidden_dim + offs_h[None, :]
        out_gate_ptrs  = out_gate_ptr + offs_m[:, None] * hidden_dim + offs_h[None, :]

        tl.store(left_ptrs, left,
                 mask=mask_m[:, None] & mask_h[None, :])
        tl.store(right_ptrs, right,
                 mask=mask_m[:, None] & mask_h[None, :])
        tl.store(out_gate_ptrs, out_gate,
                 mask=mask_m[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 3️⃣ Second LayerNorm + element‑wise out‑gate (fused)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_gate_kernel(
    x_ptr,            # const   * [total_rows, dim]   (float32)
    y_ptr,            # mutable * [total_rows, dim]   (float16)
    weight_ptr,       # const   * [dim]               (float32)
    bias_ptr,         # const   * [dim]               (float32)
    gate_ptr,         # const   * [total_rows, dim]   (float16)
    total_rows,       # int (runtime)
    dim,              # int (runtime)
    eps,              # float (runtime)
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ---------- accumulate mean / var ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- normalisation + gating ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        # original x
        x_ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        x_vals = tl.load(x_ptrs,
                         mask=mask_m[:, None] & mask_d[None, :],
                         other=0.0).to(tl.float32)

        # normalise
        norm = (x_vals - mean[:, None]) * rstd[:, None]

        # γ / β
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        norm = norm * w[None, :] + b[None, :]                     # fp32

        # gate (fp16 → fp32)
        g = tl.load(gate_ptr + offs_m[:, None] * dim + offs_d[None, :],
                    mask=mask_m[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)

        out = norm * g                                            # fp32
        out_fp16 = out.to(tl.float16)

        y_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(y_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 4️⃣ Full forward pass
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float, optional)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain "dim", "hidden_dim", "nomask"

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # 0️⃣  Global settings
    # --------------------------------------------------------------
    torch.backends.cuda.matmul.allow_tf32 = True
    torch.backends.cuda.matmul.allow_fp16_reduced_precision_reduction = True

    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    device = input_tensor.device

    B, N, _, _ = input_tensor.shape                # [B, N, N, dim]
    total_rows = B * N * N                          # R = B·N·N
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣  First LayerNorm (row‑wise) → fp16
    # --------------------------------------------------------------
    x_flat = input_tensor.reshape(total_rows, dim).contiguous()   # [R, dim]  fp32
    x_norm = torch.empty(total_rows, dim,
                         dtype=torch.float16, device=device)

    BLOCK_M = 256
    BLOCK_D = 128                     # covers both dim==128 and dim==384
    grid_norm = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid_norm](
        x_flat,
        x_norm,
        weights["norm.weight"],          # γ (float32)
        weights["norm.bias"],            # β (float32)
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    # --------------------------------------------------------------
    # 2️⃣  Single GEMM producing: left_proj, right_proj,
    #     left_gate, right_gate, out_gate   (all in fp16)
    # --------------------------------------------------------------
    # concatenate the five weight matrices (shape: [5·H, dim])
    w_all = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # [5·H, dim]

    # GEMM:  (R, dim)  ×  (dim, 5·H)ᵀ  →  (R, 5·H)
    proj_all = F.linear(x_norm, w_all)      # fp16, shape [R,5H]

    # --------------------------------------------------------------
    # 3️⃣  Fuse mask, sigmoid‑gates and write final tensors
    # --------------------------------------------------------------
    left  = torch.empty_like(x_norm, shape=(total_rows, H), dtype=torch.float16)
    right = torch.empty_like(x_norm, shape=(total_rows, H), dtype=torch.float16)
    out_gate = torch.empty_like(x_norm, shape=(total_rows, H), dtype=torch.float16)

    if config.get("nomask", True):
        mask_fp16 = torch.empty(0, device=device)   # dummy buffer
        MASKED = 0
    else:
        mask_fp16 = mask_tensor.reshape(total_rows).to(torch.float16)
        MASKED = 1

    BLOCK_H = 32                         # hidden‑dim block size
    grid_pg = (triton.cdiv(total_rows, BLOCK_M),)
    proj_gate_mask_kernel[grid_pg](
        proj_all,
        mask_fp16,
        left,
        right,
        out_gate,
        total_rows,
        H,
        MASKED,
        BLOCK_M=BLOCK_M,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )

    # --------------------------------------------------------------
    # 4️⃣  Reshape tensors back to 4‑D layout
    # --------------------------------------------------------------
    left = left.view(B, N, N, H)          # [B, N, N, H]  fp16
    right = right.view(B, N, N, H)        # [B, N, N, H]  fp16
    out_gate = out_gate.view(B, N, N, H)  # [B, N, N, H]  fp16

    # --------------------------------------------------------------
    # 5️⃣  Cubic multiplicative update:  left @ rightᵀ  (batched GEMM)
    # --------------------------------------------------------------
    # reshape to (B·H, N, N) → treat each head as an independent batch
    left_mat  = left.permute(0, 3, 1, 2).reshape(B * H, N, N)   # [B·H, N, N]
    right_mat = right.permute(0, 3, 1, 2).reshape(B * H, N, N)  # [B·H, N, N]

    # batched matrix multiply, Tensor‑Core accelerated
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))    # [B·H, N, N] fp16

    # restore original layout
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1)          # [B, N, N, H] fp16

    # --------------------------------------------------------------
    # 6️⃣  Second LayerNorm + output‑gate (fused)
    # --------------------------------------------------------------
    out_flat = out.reshape(total_rows, H).contiguous()
    out_norm = torch.empty_like(out_flat, dtype=torch.float16)

    grid_lg = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_gate_kernel[grid_lg](
        out_flat,
        out_norm,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        out_gate.reshape(total_rows, H), # fp16 gate
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    # --------------------------------------------------------------
    # 7️⃣  Final linear projection back to `dim`
    # --------------------------------------------------------------
    final = F.linear(
        out_norm,
        weights["to_out.weight"].to(torch.float16)   # [dim, H]
    )                                   # fp16, shape [R, dim]

    # Cast to float32 as required by the reference implementation
    final = final.view(B, N, N, dim).to(torch.float32)

    return final
