def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    batch, M, N, K,
    stride_ab, stride_ak, stride_a_batch,
    stride_bk, stride_bn, stride_b_batch,
    stride_cm, stride_cn, stride_c_batch,
    total_batches: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # Determine indices
    batch_idx = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers base for this batch
    a_batch_ptr = a_ptr + batch_idx * stride_a_batch
    b_batch_ptr = b_ptr + batch_idx * stride_b_batch
    c_batch_ptr = c_ptr + batch_idx * stride_c_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch_ptr + offs_m[:, None] * stride_ab + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load B tile (transposed) shape (BLOCK_K, BLOCK_N)
        b_ptrs = b_batch_ptr + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        acc += tl.dot(a, b)

    # Write back
    c_ptrs = c_batch_ptr + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn

"""
Outgoing TriMul (AlphaFold‑3) – Triton‑accelerated forward pass.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the input (FP32 → FP16) using a custom Triton kernel.
2️⃣ Fuse the four linear maps (left_proj, right_proj, left_gate, right_gate) into a
   single GEMM:  W₄ₕ = [Wₗₚ; Wᵣₚ; Wₗg; Wᵣg]  (shape 4·H × dim).
3️⃣ Compute the out‑gate linear map (dim → H).
4️⃣ (Optional) apply the pair‑wise mask, then sigmoid the three gate tensors and
   multiply the projections by their gates.
5️⃣ Reshape to (B,N,N,H) and perform the cubic update:
   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h].
   This is a batched GEMM over the combined batch‑hidden dimension
   (B·H, N, N) × (B·H, N, N)ᵀ → (B·H, N, N) and is executed by a custom
   Triton kernel that uses Tensor‑Cores (fp16 dot, fp32 accumulation).
6️⃣ Row‑wise LayerNorm over the hidden dimension H (again via Triton).
7️⃣ Multiply by the out‑gate, then a final linear projection back to the original
   feature dimension dim.  The result is cast back to float32 to match the
   reference implementation.

All heavy linear work stays in fp16 so Tensor‑Cores are fully utilised.
A Triton kernel is used for both layer‑norms and for the batched GEMM;
the remaining ops (linear, sigmoid, view/permute) are done with
high‑performance PyTorch primitives.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const  * [total_rows, dim]      (float32)
    y_ptr,          # mutable* [total_rows, dim]      (float16)
    w_ptr,          # const  * [dim]                   (float32) – gamma
    b_ptr,          # const  * [dim]                   (float32) – beta
    total_rows,     # int32
    dim,            # int32
    eps,            # float32
    BLOCK_M: tl.constexpr,   # rows processed per program
    BLOCK_D: tl.constexpr,   # columns (feature) tile size
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # --------------------------------------------------------------
    # 1️⃣ Compute per‑row mean / variance (FP32)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, 1)
        sum_sq  += tl.sum(vals * vals, 1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)           # 1 / sqrt(var + eps)

    # --------------------------------------------------------------
    # 2️⃣ Write normalised values (FP16)
    # --------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)

        out = norm * gamma[None, :] + beta[None, :]
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
# 5️⃣ Batched GEMM (A @ Bᵀ) for the cubic update
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_transpose_kernel(
    a_ptr,          # const  * [B, M, K]   (fp16)
    b_ptr,          # const  * [B, N, K]   (fp16)   – we will read it as (B, K, N) via transposed load
    c_ptr,          # mutable* [B, M, N]   (fp16)
    M, N, K,
    stride_am, stride_ak, stride_ab,      # A strides (row, col, batch)
    stride_bk, stride_bn, stride_bb,      # B strides (k, n, batch)  – original B is [B, N, K]
    stride_cm, stride_cn, stride_cb,      # C strides (row, col, batch)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    batch_idx = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch
    a_batch_ptr = a_ptr + batch_idx * stride_ab
    b_batch_ptr = b_ptr + batch_idx * stride_bb
    c_batch_ptr = c_ptr + batch_idx * stride_cb

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile: [BLOCK_M, BLOCK_K]
        a_ptrs = a_batch_ptr + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # Load B tile transposed: treat B as (K, N)
        # Original B layout is (batch, N, K) → stride_bn across N, stride_bk across K
        b_ptrs = b_batch_ptr + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)

        acc += tl.dot(a, b)      # (BLOCK_M, BLOCK_K) • (BLOCK_K, BLOCK_N) → (BLOCK_M, BLOCK_N)

    # Write back result in fp16
    c_ptrs = c_batch_ptr + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float) – optional
        - weights_dict : model parameters (see required keys below)
        - config_dict  : must contain "dim", "hidden_dim", "nomask"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    assert input_tensor.is_cuda, "TriMul kernel requires CUDA tensors"

    dim         = config["dim"]
    hidden_dim  = config["hidden_dim"]
    eps         = 1e-5
    device      = input_tensor.device
    B, N, _, _ = input_tensor.shape
    total_rows  = B * N * N                # R = B·N·N
    H           = hidden_dim               # hidden dimension

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → fp16
    # ------------------------------------------------------------------
    x_flat = input_tensor.reshape(total_rows, dim).contiguous()           # [R, dim] FP32
    x_norm = torch.empty(total_rows, dim, dtype=torch.float16, device=device)

    BLOCK_M = 256          # rows per kernel launch (tuned for H100)
    BLOCK_D = 128          # feature tile (covers dim up to 384)
    grid = (triton.cdiv(total_rows, BLOCK_M),)

    layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],   # gamma  (FP32)
        weights["norm.bias"],     # beta   (FP32)
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection + gate linear maps (4·H × dim)
    # ------------------------------------------------------------------
    # Concatenate projection and gate weights (all without bias)
    w_4h = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16, device=device)               # [4·H, dim]

    # Linear on the normalized input
    proj_gate = F.linear(x_norm, w_4h)                 # [R, 4·H]  (FP16)

    left_proj_raw  = proj_gate[:, 0:H]
    right_proj_raw = proj_gate[:, H:2 * H]
    left_gate_raw  = proj_gate[:, 2 * H:3 * H]
    right_gate_raw = proj_gate[:, 3 * H:4 * H]

    # Out‑gate does not share the fused weight matrix
    out_gate_raw = F.linear(
        x_norm,
        weights["out_gate.weight"].to(torch.float16, device=device)   # (H, dim)
    )   # [R, H] FP16

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask, sigmoid gates and element‑wise gating
    # ------------------------------------------------------------------
    if not config.get("nomask", True):
        # mask shape: [B, N, N] → broadcast to [R, 1]
        mask_fp16 = mask_tensor.reshape(total_rows, 1).to(torch.float16, device=device)
        left_proj_raw  = left_proj_raw * mask_fp16
        right_proj_raw = right_proj_raw * mask_fp16

    # In‑place sigmoid for speed
    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    # Apply gates
    left  = left_proj_raw * left_gate      # [R, H] FP16
    right = right_proj_raw * right_gate    # [R, H] FP16

    # ------------------------------------------------------------------
    # 4️⃣ Reshape to (B, N, N, H)
    # ------------------------------------------------------------------
    left = left.view(B, N, N, H)
    right = right.view(B, N, N, H)
    out_gate = out_gate.view(B, N, N, H)

    # ------------------------------------------------------------------
    # 5️⃣ Cubic update via batched GEMM: out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    #    We treat (B·H) as the batch dimension and compute
    #    out_mat = left_mat @ right_matᵀ   where each matrix is (N×N).
    # ------------------------------------------------------------------
    # Bring H into the batch dimension
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    # Merge batch and hidden dimensions
    left_mat = left_perm.view(B * H, N, N)   # [B·H, N, N]  fp16
    right_mat = right_perm.view(B * H, N, N) # [B·H, N, N]  fp16

    # Allocate output for the GEMM
    out_mat = torch.empty_like(left_mat, dtype=torch.float16, device=device)  # [B·H, N, N]

    # Triton block sizes – these values work well on H100 for N up to 1024
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        B * H,                         # batch dimension
        triton.cdiv(N, BLOCK_M),      # rows
        triton.cdiv(N, BLOCK_N),      # cols
    )

    # Strides for A (left_mat) – shape (B·H, N, N)
    stride_am = left_mat.stride(1)   # stride between rows (M dimension)   = N
    stride_ak = left_mat.stride(2)   # stride between cols (K dimension)   = 1
    stride_ab = left_mat.stride(0)   # stride between batches

    # Strides for B (right_mat) – original layout (batch, N, K)
    stride_bk = right_mat.stride(2)  # stride across K (inner dim) = 1
    stride_bn = right_mat.stride(1)  # stride across N                = N
    stride_bb = right_mat.stride(0)  # batch stride

    # Strides for C (out_mat)
    stride_cm = out_mat.stride(1)
    stride_cn = out_mat.stride(2)
    stride_cb = out_mat.stride(0)

    batched_gemm_transpose_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N, N, N,                     # M, N, K (all equal to sequence length)
        stride_am, stride_ak, stride_ab,
        stride_bk, stride_bn, stride_bb,
        stride_cm, stride_cn, stride_cb,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # Restore original layout: (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # fp16

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm over hidden dimension H (row‑wise)
    # ------------------------------------------------------------------
    out_flat = out.reshape(total_rows, H).contiguous()
    out_norm = torch.empty_like(out_flat, dtype=torch.float16, device=device)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)

    layernorm_kernel[grid2](
        out_flat,
        out_norm,
        weights["to_out_norm.weight"],   # gamma (FP32)
        weights["to_out_norm.bias"],     # beta  (FP32)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    out_norm = out_norm.view(B, N, N, H)    # fp16

    # ------------------------------------------------------------------
    # 7️⃣ Apply output gate and final linear projection back to dim
    # ------------------------------------------------------------------
    gated = out_norm * out_gate          # [B, N, N, H]  fp16

    final = F.linear(
        gated.view(total_rows, H),                     # (R, H)
        weights["to_out.weight"].to(torch.float16, device=device)   # (dim, H)
    )   # [R, dim] fp16

    final = final.view(B, N, N, dim).to(torch.float32)   # cast to float32 as reference

    return final
