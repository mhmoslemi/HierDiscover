    out_gate_ptr,           # [R, H] half

    bias_ln_ptr,            # [H] float32

    out_final_ptr,          # [R, dim] float32
    R, H, dim,
    eps: tl.float32,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_D: tl.constexpr,
):

    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < R

    sum_sq  = tl.zeros([BLOCK_R], dtype=tl.float32)

"""
Tri‑Multiplicative (outgoing) update – AlphaFold‑3 style.

Algorithm (all tensors stay on the GPU):
1️⃣  Row‑wise LayerNorm on the 4‑D input (B·N·N × dim) → fp16.
2️⃣  Fuse the four Linear maps (left/right & their gates) into a single
    half‑precision GEMM:   X_norm @ W_concatᵀ → [R, 4·H]   (R = B·N·N).
3️⃣  Split the concatenated result, apply the sigmoid gates and (optional) mask.
4️⃣  Tri‑multiplicative update
        out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
    implemented with `torch.einsum`, which internally uses a highly‑optimised
    batched GEMM and avoids the extra copy required by a manual `permute+view`.
5️⃣  Second row‑wise LayerNorm over the hidden dimension (R × H) → fp16.
6️⃣  Multiply by the output‑gate and a final linear projection back to `dim`.
    The result is cast back to float32.

Only the two LayerNorms are written in Triton; all large matrix‑multiplications
use cuBLAS (via `torch.nn.functional.linear` / `torch.einsum`) which gives
near‑optimal performance on an H100.  The kernel is heavily tiled (BLOCK_M,
BLOCK_D) and uses fp32 accumulation for numerical stability.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (float32 → float16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   *[total_rows, dim]   (float32)
    y_ptr,          # mutable *[total_rows, dim]   (float16)
    weight_ptr,     # const   *[dim]               (float32)   γ
    bias_ptr,       # const   *[dim]               (float32)   β
    total_rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """Row‑wise LayerNorm:
       y = ((x – μ) / sqrt(σ²+eps)) * γ + β
       Output in fp16, accumulation in fp32.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    row_mask = offs_m < total_rows

    # ------------------------------------------------------------------
    # 1️⃣  Compute mean & variance (fp32)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    d_blocks = tl.cdiv(dim, BLOCK_D)
    for db in range(d_blocks):
        offs_d = db * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        col_mask = offs_d < dim

        # load a tile of the input
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]   # [M,D]
        vals = tl.load(ptr,
                       mask=row_mask[:, None] & col_mask[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------------------------------------------------------
    # 2️⃣  Write normalized values  γ·(x‑μ)·rstd + β   (fp16)
    # ------------------------------------------------------------------
    for db in range(d_blocks):
        offs_d = db * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        col_mask = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]   # [M,D]
        vals = tl.load(ptr,
                       mask=row_mask[:, None] & col_mask[None, :],
                       other=0.0).to(tl.float32)

        # (x - μ) * rstd
        norm = (vals - mean[:, None]) * rstd[:, None]

        # γ and β
        w = tl.load(weight_ptr + offs_d, mask=col_mask, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=col_mask, other=0.0)   # β

        out = norm * w[None, :] + b[None, :]          # fp32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=row_mask[:, None] & col_mask[None, :])

    H = hidden_dim
    # concatenate the four projection matrices → [4·H, dim]
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)          # fp16 weight matrix

    # [R, dim] @ [4·H, dim]ᵀ  →  [R, 4·H]   (half‑precision GEMM)
    proj_gate = F.linear(x_norm, w_concat)                     # fp16

    left_proj_raw  = proj_gate[:, 0:H]
    right_proj_raw = proj_gate[:, H:2 * H]
    left_gate_raw  = proj_gate[:, 2 * H:3 * H]
    right_gate_raw = proj_gate[:, 3 * H:4 * H]

    # output gate (its own linear)
    out_gate_raw = F.linear(
        x_norm,
        weights["out_gate.weight"].to(torch.float16),          # [H, dim]
    )   # fp16

    # --------------------------------------------------------------
    # 3️⃣  Optional mask, sigmoid gates, and gating
    # --------------------------------------------------------------
    if not cfg.get("nomask", True):
        # mask shape [B,N,N] → [R,1] for broadcasting
        mask_h = mask.to(torch.float16).reshape(R, 1)
        left_proj_raw  = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    left  = left_proj_raw * left_gate        # [R, H]   fp16
    right = right_proj_raw * right_gate      # [R, H]   fp16

    # --------------------------------------------------------------
    # 4️⃣  Tri‑multiplicative update (batched GEMM via einsum)
    # --------------------------------------------------------------
    left  = left.view(B, N, N, H)            # [B, N, N, H]
    right = right.view(B, N, N, H)           # [B, N, N, H]

    # einsum internally maps to a batch‑gemm kernel and avoids the extra
    # copy caused by a manual permute+view → faster for large N.
    out = torch.einsum('b i k d, b j k d -> b i j d', left, right)   # fp16

    # --------------------------------------------------------------
    # 5️⃣  Second LayerNorm over the hidden dimension (fp16)
    # --------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()
    out_norm = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm,
        weights["to_out_norm.weight"],   # γ  (float32)
        weights["to_out_norm.bias"],     # β  (float32)
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    out_norm = out_norm.view(B, N, N, H)    # fp16

    # --------------------------------------------------------------
    # 6️⃣  Apply output gate and final projection back to `dim`
    # --------------------------------------------------------------
    out_norm = out_norm * out_gate          # element‑wise gate (fp16)

    final_half = F.linear(
        out_norm.view(R, H),
        weights["to_out.weight"].to(torch.float16),   # [dim, H]
    )   # fp16

    # Cast back to float32 as required by the public API
    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
```
