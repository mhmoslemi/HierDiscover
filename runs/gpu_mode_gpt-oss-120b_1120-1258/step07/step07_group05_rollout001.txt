"""
custom_kernel
==============

Highly‑optimized forward pass of the **outgoing** TriMul operator (AlphaFold‑3
style).  The implementation fuses as many element‑wise operations as possible
into a small set of Triton kernels while delegating the large dense matrix
products to cuBLAS (via torch.bmm / torch.nn.functional.linear).  The main
steps are:

1️⃣  Row‑wise LayerNorm (fp32 → fp16) – one Triton kernel.
2️⃣  Fused linear projection that computes the left/right projections **and**
    their two gates in a single GEMM (4·H outputs).  No extra temporaries are
    allocated.
3️⃣  Separate fused linear for the output‑gate.
4️⃣  Optional mask application (broadcast) + in‑place sigmoid on the three
    gates.
5️⃣  Multiplicative update :  out[i,j,d] = Σₖ left[i,k,d]·right[j,k,d].
    Implemented as a batched mat‑mul
    ``torch.bmm(left, right.transpose(1,2))`` where the head dimension H is
    folded into the batch axis (B·H, N, N).  This lets cuBLAS exploit Tensor‑Cores
    on H100.
6️⃣  Second LayerNorm (over the hidden dimension) – same Triton kernel.
7️⃣  Apply the output‑gate, a final linear projection back to *dim* and cast to
    float32.

All heavy element‑wise work (LayerNorm, optional mask) is performed on the
GPU with Triton, keeping memory traffic low.  Mixed‑precision (fp16) is used
throughout the bulk of the computation; the final result is returned in fp32.

The function signature matches the specification:

    custom_kernel(data) → torch.Tensor of shape [B, N, N, dim] (float32)

where ``data`` is a tuple ``(input_tensor, mask_tensor, weights_dict, config_dict)``.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (fp32 → fp16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   * [total_rows, dim]   (float32)
    y_ptr,          # mutable * [total_rows, dim]   (float16)
    weight_ptr,     # const   * [dim]               (float32)
    bias_ptr,       # const   * [dim]               (float32)
    total_rows,     # int (runtime)
    dim,            # int (runtime)
    eps,            # float (compile‑time constexpr)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # cols per iteration
):
    """
    Row‑wise LayerNorm:
        y = ((x - μ) / sqrt(σ² + eps)) * γ + β
    The kernel works on a 2‑D view of the tensor: [total_rows, dim].
    """
    pid = tl.program_id(0)                     # one program per BLOCK_M rows
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # -------------------- 1. per‑row mean & variance --------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    # iterate over the dim axis in BLOCK_D sized chunks
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < dim
        # pointer to the current tile
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                     # 1 / sqrt(var + eps)

    # -------------------- 2. write normalized + affine transformed --------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        # (x - μ) * rstd -> normalized
        norm = (vals - mean[:, None]) * rstd[:, None]

        # γ (weight) and β (bias) are float32; we keep the math in fp32
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]                       # fp32

        out_fp16 = out.to(tl.float16)                             # store fp16
        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : dict containing the following keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight",
                "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
          All weight tensors are stored as float32 (the same as in the PyTorch
          reference implementation).
        - config_dict  : dict with keys
                "dim" (int), "hidden_dim" (int), "nomask" (bool, optional)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] and dtype float32.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    device = x.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)          # True → mask is ignored
    eps = 1e-5

    B, N, _, _ = x.shape
    total_rows = B * N * N                       # R = B·N·N
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (fp32 → fp16)
    # ------------------------------------------------------------------
    # Flatten [B,N,N,dim] -> [R,dim] (row‑wise layout)
    x_flat = x.view(total_rows, dim).contiguous()
    ln1_out = torch.empty(total_rows, dim,
                          dtype=torch.float16,
                          device=device)

    # Heuristic block sizes – they work well for dim ∈ {128,384,768}
    BLOCK_M = 256
    BLOCK_D = 64
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        ln1_out,
        weights["norm.weight"],   # γ (float32)
        weights["norm.bias"],     # β (float32)
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear that produces left, right and the two gates
    #    (4·H outputs).  Weight matrix is stacked vertically.
    # ------------------------------------------------------------------
    # Stack weight matrices: shape [4·H, dim]
    W_fused = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
        ],
        dim=0,
    ).to(device=device, dtype=torch.float16)   # fp16 for the GEMM

    # Linear: [R, dim] @ [4·H, dim].T → [R, 4·H]   (no bias)
    fused = F.linear(ln1_out, W_fused)          # fp16

    # Split the fused result
    left_raw      = fused[:, 0:H]                # [R, H]
    right_raw     = fused[:, H:2*H]               # [R, H]
    left_gate_raw = fused[:, 2*H:3*H]             # [R, H]
    right_gate_raw = fused[:, 3*H:4*H]            # [R, H]

    # Separate linear for the output‑gate (still fp16)
    out_gate_raw = F.linear(
        ln1_out,
        weights["out_gate.weight"].to(device=device, dtype=torch.float16)  # [H, dim]
    )                                            # [R, H]

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask + sigmoid gates (in‑place to save memory)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask shape [B,N,N] → [R,1]; cast to fp16 for the element‑wise mul
        mask_flat = mask.to(device=device, dtype=torch.float16).reshape(total_rows, 1)
        left_raw  = left_raw * mask_flat
        right_raw = right_raw * mask_flat

    # In‑place sigmoid – avoids an extra temporary tensor
    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    # Apply gates
    left  = left_raw  * left_gate   # [R, H] fp16
    right = right_raw * right_gate  # [R, H] fp16

    # ------------------------------------------------------------------
    # 4️⃣ Reshape for the N³ multiplicative update
    # ------------------------------------------------------------------
    # Shape back to [B, N, N, H]
    left  = left.view(B, N, N, H)
    right = right.view(B, N, N, H)
    out_gate = out_gate.view(B, N, N, H)

    # Move head dimension into the batch axis: [B, H, N, N]
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    # Collapse batch & head → [B·H, N, N]
    left_mat  = left_perm.view(B * H, N, N)
    right_mat = right_perm.view(B * H, N, N)

    # Batched matrix multiplication: (B·H, N, N) @ (B·H, N, N)ᵀ
    # This is the core O(N³·H) operation.
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # fp16

    # Restore layout: [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm over the hidden dimension H
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, H).contiguous()
    ln2_out = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        ln2_out,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Apply output gate, final linear projection, cast to float32
    # ------------------------------------------------------------------
    out_norm = ln2_out.view(B, N, N, H)        # fp16
    out_norm = out_norm * out_gate             # element‑wise fp16

    # Final linear: [R, H] @ [dim, H].T → [R, dim] (fp16)
    final_fp16 = F.linear(
        out_norm.view(total_rows, H),
        weights["to_out.weight"].to(device=device, dtype=torch.float16)   # [dim, H]
    )   # fp16

    # Cast back to fp32 as required by the public API
    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final
