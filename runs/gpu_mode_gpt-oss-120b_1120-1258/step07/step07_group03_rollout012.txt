@triton.jit
def batch_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_bk, stride_bn,
    stride_cm, stride_cn,
    batch_stride_a, batch_stride_b, batch_stride_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)  # row block
    pid_n = tl.program_id(1)  # col block
    pid_b = tl.program_id(2)  # batch group id

    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # boundaries
    mask_m = offs_m < M
    mask_n = offs_n < N

    # offsets for batch
    batch_offset_a = pid_b * batch_stride_a
    batch_offset_b = pid_b * batch_stride_b
    batch_offset_c = pid_b * batch_stride_c

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # compute a and b tile pointers
        a_ptrs = a_ptr + batch_offset_a + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        b_ptrs = b_ptr + batch_offset_b + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

        # load tiles
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float16)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float16)

        # use dot (tensor core)
        acc += tl.dot(a, b)  # accumulates in fp32

    # write result
    c_ptrs = c_ptr + batch_offset_c + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

"""
TriMul (outgoing) forward pass – Triton‑accelerated.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the input tensor (fp32 → fp16) using a
   custom Triton kernel `rowwise_layernorm_kernel`.

2️⃣ Fuse the four linear projections (`left_proj`, `right_proj`,
   `left_gate`, `right_gate`) into a single GEMM:
   (B·N·N, dim) @ (4·H, dim).T → (B·N·N, 4·H) (fp16).

3️⃣ Compute the output‑gate projection separately.

4️⃣ Apply optional mask, sigmoids and gating:
   left  = (x·W_left)   * sigmoid(x·W_lgate)
   right = (x·W_right)  * sigmoid(x·W_rgate)

5️⃣ Batched multiplicative update:
   For each (batch, head) we need
      out_{i,j,h} = Σ_k left_{i,k,h} * right_{j,k,h}
   This is a batched matrix multiplication
      (B·H, N, N) @ (B·H, N, N)^T → (B·H, N, N).
   Implemented with a dedicated Triton kernel `batch_matmul_kernel`
   that exploits Tensor‑Cores (fp16×fp16→fp32 accumulation).

6️⃣ Second row‑wise LayerNorm over the hidden dimension (fp16).

7️⃣ Multiply by the output‑gate, project back to `dim` with a final
   linear layer, and cast the result to float32.

The heavy N³ part (step 5) is the only operation that required a
custom kernel; the LayerNorms are also Triton‑based to avoid extra
memory traffic. All other dense linear layers use PyTorch’s
cuBLAS‑backed `torch.nn.functional.linear`, which is already optimal.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (fp32 → fp16)
# ----------------------------------------------------------------------
@triton.jit
def rowwise_layernorm_kernel(
    x_ptr,    # const   * [total_rows, dim]   (float32)
    y_ptr,    # mutable * [total_rows, dim]   (float16)
    w_ptr,    # const   * [dim]               (float32)  γ
    b_ptr,    # const   * [dim]               (float32)  β
    total_rows,
    dim,
    eps,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """
    Row‑wise LayerNorm:
        y = ((x - μ) / sqrt(σ² + eps)) * γ + β
    The reduction (mean / var) is performed in fp32,
    while the output is stored in fp16.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)                # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------- 1️⃣ compute mean & var per row -------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    # loop over feature dimension
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < dim

        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)               # 1 / sqrt(var + eps)

    # ------------------- 2️⃣ write normalized output -------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)

        out = norm * gamma[None, :] + beta[None, :]      # fp32
        out_fp16 = out.to(tl.float16)

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication (fp16 × fp16 → fp16)
# Computes:
#   C[b] = A[b] @ B[b].T    for b = 0 .. batch‑1
# where A, B, C are (batch, M, K) , (batch, N, K) , (batch, M, N)
# ----------------------------------------------------------------------
@triton.jit
def batch_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,            # A: (batch, M, K)
    stride_bk, stride_bn,            # B (treated as K×N): (batch, K, N)
    stride_cm, stride_cn,            # C: (batch, M, N)
    batch_stride_a, batch_stride_b, batch_stride_c,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM using Tensor‑Core tiles.
    Each program instance computes a BLOCK_M×BLOCK_N tile of C
    for a specific batch (pid_z) and tile indices (pid_x, pid_y).
    """
    pid_m = tl.program_id(0)   # row block index
    pid_n = tl.program_id(1)   # col block index
    pid_b = tl.program_id(2)   # batch index

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # batch offsets (in elements)
    a_batch_off = pid_b * batch_stride_a
    b_batch_off = pid_b * batch_stride_b
    c_batch_off = pid_b * batch_stride_c

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile:   [BLOCK_M, BLOCK_K]
        a_ptrs = a_ptr + a_batch_off \
                 + offs_m[:, None] * stride_am \
                 + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float16)

        # B tile (transposed): [BLOCK_K, BLOCK_N]
        b_ptrs = b_ptr + b_batch_off \
                 + offs_k[:, None] * stride_bk \
                 + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float16)

        # Tensor‑Core MMA
        acc += tl.dot(a, b)      # fp16×fp16 → fp32 accumulation

    # Store the result tile
    c_ptrs = c_ptr + c_batch_off \
             + offs_m[:, None] * stride_cm \
             + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (AlphaFold‑3 variant).

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]  (float32)
        - mask_tensor  : torch.Tensor [B, N, N]      (bool/float, optional)
        - weights_dict : dict of torch.Tensors containing all linear/
                         layer‑norm parameters.
        - config_dict  : must contain "dim", "hidden_dim".  May also
                         contain "nomask" (bool).

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    device = input_tensor.device

    B, N, _, _ = input_tensor.shape
    R = B * N * N                     # total rows for the first LN

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → fp16
    # ------------------------------------------------------------------
    x_flat = input_tensor.reshape(R, dim).contiguous()           # [R, dim] fp32
    x_norm_half = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 256
    BLOCK_D = 32
    grid = (triton.cdiv(R, BLOCK_M),)
    rowwise_layernorm_kernel[grid](
        x_flat,
        x_norm_half,
        weights["norm.weight"],    # γ, fp32
        weights["norm.bias"],      # β, fp32
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fuse left/right projections + gate projections (single GEMM)
    # ------------------------------------------------------------------
    H = hidden_dim
    # Concatenate the four projection weights -> [4·H, dim]
    w_proj_gate = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)   # fp16 for Tensor‑Core GEMM

    # [R, dim] @ [4·H, dim].T → [R, 4·H]  (fp16)
    proj_gate = F.linear(x_norm_half, w_proj_gate)   # bias = None

    left_proj_raw  = proj_gate[:, 0:H]               # [R, H]
    right_proj_raw = proj_gate[:, H:2 * H]           # [R, H]
    left_gate_raw  = proj_gate[:, 2 * H:3 * H]       # [R, H]
    right_gate_raw = proj_gate[:, 3 * H:4 * H]       # [R, H]

    # Output‑gate linear (separate weight)
    out_gate_raw = F.linear(
        x_norm_half,
        weights["out_gate.weight"].to(torch.float16)   # [H, dim]
    )   # → [R, H]

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask, sigmoid gates and gating
    # ------------------------------------------------------------------
    if not config.get("nomask", True):
        # mask: [B, N, N] → broadcast over hidden dimension
        mask_h = mask_tensor.to(torch.float16).reshape(R, 1)   # [R,1]
        left_proj_raw  = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    # Sigmoids (in‑place, half‑precision)
    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    # Apply gates
    left  = left_proj_raw * left_gate       # [R, H] fp16
    right = right_proj_raw * right_gate     # [R, H] fp16

    # ------------------------------------------------------------------
    # 4️⃣ Reshape for the N³ multiplicative update
    # ------------------------------------------------------------------
    left  = left.view(B, N, N, H)          # [B, N, N, H]
    right = right.view(B, N, N, H)         # [B, N, N, H]
    out_gate = out_gate.view(B, N, N, H)   # [B, N, N, H]

    # Bring the hidden dimension into the batch axis
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    left_mat  = left_perm.view(B * H, N, N)      # [B·H, N, N]
    right_mat = right_perm.view(B * H, N, N)     # [B·H, N, N]

    # ------------------------------------------------------------------
    # 5️⃣ Batched half‑precision GEMM using Triton (N³ update)
    # ------------------------------------------------------------------
    out_mat = torch.empty_like(left_mat)        # fp16, shape [B·H, N, N]

    # Strides in elements (not bytes)
    stride_am = left_mat.stride(1)   # = N
    stride_ak = left_mat.stride(2)   # = 1
    stride_bk = right_mat.stride(1)  # = N   (row stride of the transposed view)
    stride_bn = right_mat.stride(2)  # = 1
    stride_cm = out_mat.stride(1)    # = N
    stride_cn = out_mat.stride(2)    # = 1

    batch_stride_a = left_mat.stride(0)   # = N * N
    batch_stride_b = right_mat.stride(0)  # = N * N
    batch_stride_c = out_mat.stride(0)    # = N * N

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    grid = (
        triton.cdiv(N, BLOCK_M),   # pid_m
        triton.cdiv(N, BLOCK_N),   # pid_n
        B * H,                     # pid_b
    )
    batch_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N, N, N,
        stride_am, stride_ak,
        stride_bk, stride_bn,
        stride_cm, stride_cn,
        batch_stride_a, batch_stride_b, batch_stride_c,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # Reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # fp16

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm over hidden dimension → fp16
    # ------------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()
    out_norm_half = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(R, BLOCK_M),)
    rowwise_layernorm_kernel[grid2](
        out_flat,
        out_norm_half,
        weights["to_out_norm.weight"],   # γ, fp32
        weights["to_out_norm.bias"],     # β, fp32
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    out_norm = out_norm_half.view(B, N, N, H)   # fp16

    # ------------------------------------------------------------------
    # 7️⃣ Apply output gate and final linear projection
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate                # gate (fp16)

    final_half = F.linear(
        out_norm.view(R, H),
        weights["to_out.weight"].to(torch.float16)   # [dim, H]
    )   # → [R, dim] fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)

    return final
