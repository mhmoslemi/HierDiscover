def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    dim, hidden_dim = config["dim"], config["hidden_dim"]
    eps = 1e-5
    B,N,_,_ = input_tensor.shape
    total_rows = B * N * N
    H = hidden_dim

    # convert weights to fp16 (already FP32 input)
    norm_weight = weights["norm.weight"].contiguous()
    norm_bias = weights["norm.bias"].contiguous()
    # fused projection+gate weights
    w_proj_gate = torch.cat([
        weights["left_proj.weight"],
        weights["right_proj.weight"],
        weights["left_gate.weight"],
        weights["right_gate.weight"],
        weights["out_gate.weight"]
    ], dim=0).contiguous().to(torch.float16)
    # to_out_norm
    to_out_norm_weight = weights["to_out_norm.weight"].contiguous()
    to_out_norm_bias = weights["to_out_norm.bias"].contiguous()
    # to_out
    to_out_weight = weights["to_out.weight"].contiguous().to(torch.float16)

    # 1. layernorm
    x_flat = input_tensor.reshape(total_rows, dim).contiguous()
    x_norm = torch.empty(total_rows, dim, dtype=torch.float16, device=input_tensor.device)

    BLOCK_M = 256
    BLOCK_D = 128
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        norm_weight,
        norm_bias,
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    # 2. fused linear maps
    proj_gate = F.linear(x_norm, w_proj_gate)   # shape [R, 5*H]
    # split
    left_raw = proj_gate[:, 0:H]
    right_raw = proj_gate[:, H:2*H]
    left_gate_raw = proj_gate[:, 2*H:3*H]
    right_gate_raw = proj_gate[:, 3*H:4*H]
    out_gate_raw = proj_gate[:, 4*H:5*H]

    # mask
    if not config.get("nomask", True):
        mask_flat = mask_tensor.reshape(total_rows, 1).to(torch.float16)
        left_raw = left_raw * mask_flat
        right_raw = right_raw * mask_flat

    # sigmoids inplace
    left_gate = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate = out_gate_raw.sigmoid_()

---------
1️⃣ Row‑wise LayerNorm on the input (FP32 → FP16) using a custom Triton kernel.
2️⃣ Fuse the four projection + two gate linear maps (left_proj, right_proj,
   left_gate, right_gate, out_gate) into a single GEMM:
   W5 = [Wₗₚ; Wᵣₚ; Wₗg; Wᵣg; Wₒg]   (shape 5·H × dim).
   This yields tensors  left, right, left_gate, right_gate, out_gate
   in one pass.
3️⃣ Apply optional pairwise mask, sigmoid the three gate tensors and
   multiply the projections by their gates.
4️⃣ Perform the cubic update
   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]   (batched GEMM on (B·H,N,N)).
5️⃣ Second row‑wise LayerNorm over the hidden dimension H (another Triton kernel).
6️⃣ Multiply by the output‑gate, then a final linear projection back to the
   original feature dimension dim.
All expensive linear work stays in FP16 so Tensor‑Cores on an H100 are fully
utilised.  The only Triton kernels are the two LayerNorms; the rest relies on
high‑throughput cuBLAS GEMM calls.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton row‑wise LayerNorm (FP32 input → FP16 output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const  * [total_rows, dim] (float32)
    y_ptr,          # mutable* [total_rows, dim] (float16)
    w_ptr,          # const  * [dim]            (float32) – gamma
    b_ptr,          # const  * [dim]            (float32) – beta
    total_rows,     # int32
    dim,            # int32
    eps,            # float32
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # cols per inner loop
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # 1️⃣ Compute row‑wise mean / variance (FP32 accumulation)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, 1)
        sum_sq  += tl.sum(vals * vals, 1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)            # 1 / sqrt(var + eps)

    # ------------------------------------------------------------------
    # 2️⃣ Write normalised values (store as FP16)
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)

        out = norm * gamma[None, :] + beta[None, :]
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Outgoing TriMul forward pass.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float) – optional
        - weights_dict : model parameters (see keys in docstring)
        - config_dict  : must contain "dim", "hidden_dim", "nomask"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    B, N, _, _ = input_tensor.shape
    R = B * N * N                # total rows for row‑wise ops
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → FP16
    # ------------------------------------------------------------------
    x_flat = input_tensor.reshape(R, dim).contiguous()                # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)   # [R, dim] FP16

    BLOCK_M = 256     # rows per program (tuned for H100)
    BLOCK_D = 128     # feature tile (covers dim up to 384)
    grid = (triton.cdiv(R, BLOCK_M),)

    layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],     # gamma (FP32)
        weights["norm.bias"],       # beta  (FP32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fuse the four projection + two gate linear maps
    #    into a single GEMM of shape (5·H, dim)
    # ------------------------------------------------------------------
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # [5·H, dim]

    proj_gate = F.linear(x_norm, w_fused)    # [R, 5·H] FP16
    # split the fused tensor
    left_raw      = proj_gate[:, 0:H]               # [R, H]
    right_raw     = proj_gate[:, H:2*H]              # [R, H]
    left_gate_raw = proj_gate[:, 2*H:3*H]            # [R, H]
    right_gate_raw= proj_gate[:, 3*H:4*H]            # [R, H]
    out_gate_raw  = proj_gate[:, 4*H:5*H]            # [R, H]

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask & gate sigmoid + element‑wise gating
    # ------------------------------------------------------------------
    if not config.get("nomask", True):
        # mask: [B,N,N] → [R,1] (float16)
        mask_fp16 = mask_tensor.reshape(R, 1).to(torch.float16)
        left_raw  = left_raw  * mask_fp16
        right_raw = right_raw * mask_fp16

    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    left  = left_raw  * left_gate   # [R, H] FP16
    right = right_raw * right_gate  # [R, H] FP16

    # reshape to (B, N, N, H)
    left = left.view(B, N, N, H)
    right = right.view(B, N, N, H)
    out_gate = out_gate.view(B, N, N, H)

    # ------------------------------------------------------------------
    # 4️⃣ Cubic update via batched GEMM (B·H independent matrix multiplications)
    #    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    # bring H into the batch dimension
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    left_mat  = left_perm.view(B * H, N, N)              # [B·H, N, N]
    right_mat = right_perm.view(B * H, N, N)             # [B·H, N, N]

    # batch‑wise matrix multiplication (left @ rightᵀ)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N]

    # reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N)               # [B, H, N, N]
    out = out.permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm over hidden dimension H (row‑wise)
    # ------------------------------------------------------------------
    out_flat = out.reshape(R, H).contiguous()
    out_norm = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm,
        weights["to_out_norm.weight"],   # gamma (FP32)
        weights["to_out_norm.bias"],     # beta  (FP32)
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    out_norm = out_norm.view(B, N, N, H)

    # ------------------------------------------------------------------
    # 6️⃣ Apply output‑gate, then final linear projection back to dim
    # ------------------------------------------------------------------
    gated = out_norm * out_gate                 # [B, N, N, H] FP16

    final = F.linear(
        gated.view(R, H),                     # [R, H]
        weights["to_out.weight"].to(torch.float16)  # (dim, H)
    )                                          # [R, dim] FP16

    final = final.view(B, N, N, dim).to(torch.float32)

    return final
