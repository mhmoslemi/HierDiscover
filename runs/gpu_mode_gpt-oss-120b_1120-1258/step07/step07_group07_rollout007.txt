"""
TriMul (outgoing) – highly‑optimised forward pass.

Algorithm
---------
1️⃣ Row‑wise LayerNorm (float32 → float16) – custom Triton kernel.
2️⃣ Fuse the four linear projections (left/right + left/right gates) into a
   single half‑precision GEMM.  The result is split into the four tensors.
3️⃣ Apply the optional pair mask, the sigmoidal gates and element‑wise
   gating (left *= sigmoid(left_gate), right *= sigmoid(right_gate)).
4️⃣ Reshape to 4‑D tensors (B,N,N,hidden) and compute the cubic update by a
   *batched* matrix multiplication:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is performed as a single `torch.bmm` after permuting the tensors so
   that the transpose is no longer required.
5️⃣ Second LayerNorm + output‑gate – fused Triton kernel.
6️⃣ Final half‑precision linear projection back to `dim` (torch.nn.functional.linear).

All heavy arithmetic stays in FP16; the final tensor is cast back to FP32 to
match the reference implementation.  The only Triton kernels are the two
row‑wise LayerNorms, which have been tuned (larger row block size) to cut
kernel‑launch overhead.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (fp32 → fp16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,            # const   * [total_rows, dim] (float32)
    y_ptr,            # mutable * [total_rows, dim] (float16)
    weight_ptr,       # const   * [dim]          (float32)
    bias_ptr,         # const   * [dim]          (float32)
    total_rows,       # int (runtime)
    dim,              # int (runtime)
    eps,              # float (runtime)
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)           # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # 1️⃣  Compute per‑row mean / variance (fp32)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # fp32 load
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                     # 1 / √(var+eps)

    # ------------------------------------------------------------------
    # 2️⃣  Normalise, scale+bias and write back as fp16
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]                     # fp32
        out_fp16 = out.to(tl.float16)

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2️⃣ Row‑wise LayerNorm + element‑wise gate (fp32 → fp16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_gate_kernel(
    x_ptr,            # const   * [total_rows, dim] (float32)
    y_ptr,            # mutable * [total_rows, dim] (float16)
    weight_ptr,       # const   * [dim]          (float32)
    bias_ptr,         # const   * [dim]          (float32)
    gate_ptr,         # const   * [total_rows, dim] (float16)
    total_rows,       # int (runtime)
    dim,              # int (runtime)
    eps,              # float (runtime)
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)           # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ---------- mean / variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- normalisation + gate ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        # load original x
        x_ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        x_vals = tl.load(x_ptrs,
                         mask=mask_m[:, None] & mask_d[None, :],
                         other=0.0).to(tl.float32)
        # normalise
        norm = (x_vals - mean[:, None]) * rstd[:, None]

        # γ / β
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)
        norm = norm * w[None, :] + b[None, :]          # fp32

        # gate (fp16 → fp32)
        g_ptrs = gate_ptr + offs_m[:, None] * dim + offs_d[None, :]
        gate = tl.load(g_ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        out = norm * gate                               # fp32
        out_fp16 = out.to(tl.float16)

        y_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(y_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 3️⃣  Custom forward for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Outgoing TriMul forward pass.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim          = config["dim"]
    hidden_dim   = config["hidden_dim"]
    eps          = 1e-5
    device       = input_tensor.device

    B, N, _, _ = input_tensor.shape               # [B, N, N, dim]
    total_rows = B * N * N                         # R = B·N·N
    H = hidden_dim

    # --------------------------------------------------------------
    # choose a row block size that keeps kernel launches small
    # --------------------------------------------------------------
    BLOCK_M = 1024 if total_rows >= 4096 else 256   # 1024 rows per program when feasible
    BLOCK_D = 128                                   # works for dim∈{128,384}

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → FP16
    # --------------------------------------------------------------
    x_flat = input_tensor.reshape(total_rows, dim).contiguous()   # [R, dim] FP32
    x_norm = torch.empty(total_rows, dim, dtype=torch.float16, device=device)

    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],      # γ   FP32
        weights["norm.bias"],        # β   FP32
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    # --------------------------------------------------------------
    # 2️⃣ Fused linear for left/right projections & their gates
    # --------------------------------------------------------------
    # concatenate the four weight matrices (shape: [4*H, dim])
    w_4h = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                 # [4*H, dim]

    # single GEMM: [R, dim] @ [dim, 4*H]ᵀ  ->  [R, 4*H]
    proj_gate = F.linear(x_norm, w_4h)                  # FP16

    left_proj_raw   = proj_gate[:, 0:H]
    right_proj_raw  = proj_gate[:, H:2 * H]
    left_gate_raw   = proj_gate[:, 2 * H:3 * H]
    right_gate_raw  = proj_gate[:, 3 * H:4 * H]

    # out_gate is a separate projection (shape: [H, dim])
    out_gate_raw = F.linear(
        x_norm,
        weights["out_gate.weight"].to(torch.float16)   # [H, dim]
    )                                                   # [R, H] FP16

    # --------------------------------------------------------------
    # 3️⃣ Mask (if present)  +  sigmoid gates
    # --------------------------------------------------------------
    if not config.get("nomask", True):
        # mask shape [B, N, N] → flatten → broadcast over hidden dim
        mask_h = mask_tensor.reshape(total_rows, 1).to(torch.float16)
        left_proj_raw  = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()               # [R, H] FP16

    left  = left_proj_raw * left_gate                  # [R, H] FP16
    right = right_proj_raw * right_gate                # [R, H] FP16

    # --------------------------------------------------------------
    # 4️⃣ Reshape to 4‑D tensors
    # --------------------------------------------------------------
    left  = left.view(B, N, N, H)          # [B, N, N, H]
    right = right.view(B, N, N, H)         # [B, N, N, H]
    out_gate = out_gate.view(B, N, N, H)   # [B, N, N, H]

    # --------------------------------------------------------------
    # 5️⃣ Cubic multiplicative update via batched GEMM
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    – avoid the explicit transpose by permuting right beforehand.
    # --------------------------------------------------------------
    # (B,H,N,N) layout: left -> (i,k), right -> (k,j)
    left_mat  = left.permute(0, 3, 1, 2).reshape(B * H, N, N)   # [B*H, N, N]
    right_mat = right.permute(0, 3, 2, 1).reshape(B * H, N, N)  # [B*H, N, N] (k as dim‑0)

    # batched matrix multiplication (Tensor‑Core accelerated)
    out_mat = torch.bmm(left_mat, right_mat)                    # [B*H, N, N] FP16

    # back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1)          # FP16

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm + output‑gate (fused)
    # --------------------------------------------------------------
    out_flat = out.reshape(total_rows, H).contiguous()
    out_norm = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_gate_kernel[grid2](
        out_flat,
        out_norm,
        weights["to_out_norm.weight"],   # γ   FP32
        weights["to_out_norm.bias"],     # β   FP32
        out_gate.reshape(total_rows, H), # gate (FP16)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    out_norm = out_norm.view(B, N, N, H)                     # [B, N, N, H] FP16

    # --------------------------------------------------------------
    # 7️⃣ Final linear projection back to `dim`
    # --------------------------------------------------------------
    final = F.linear(
        out_norm.reshape(total_rows, H),
        weights["to_out.weight"].to(torch.float16)   # [dim, H]
    )                                                # [R, dim] FP16

    # Cast to float32 – matches the reference implementation
    final = final.view(B, N, N, dim).to(torch.float32)

    return final
