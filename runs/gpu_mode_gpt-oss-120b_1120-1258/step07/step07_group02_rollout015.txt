def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    total_rows = B * N * N
    H = hidden_dim

    # flatten input: (R, dim)
    input_flat = input_tensor.reshape(total_rows, dim)

    # prepare weight matrix for first stage: combine left_proj, right_proj, left_gate, right_gate, out_gate
    # All have shape (hidden_dim, dim)
    left_proj_w = weights["left_proj.weight"]  # shape (hidden_dim, dim)
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    # Concatenate
    W_concat = torch.cat([left_proj_w, right_proj_w, left_gate_w, right_gate_w, out_gate_w], dim=0)  # shape (5*H, dim)
    # transpose to shape (dim, 5*H) for better memory access
    W_concat_T = W_concat.t().contiguous().to(torch.float16)   # shape (dim, 5*H)
    # allocate buffer for raw outputs
    out_raw = torch.empty(total_rows, 5 * H, dtype=torch.float16, device=device)

    # Launch kernel
    BLOCK_M = 64
    BLOCK_D = 64   # should divide dim up to 1024
    BLOCK_O = 64   # out_features per block
    grid = (triton.cdiv(total_rows, BLOCK_M), triton.cdiv(5*H, BLOCK_O))
    norm_linear_kernel[grid](
        input_flat,      # x_ptr float32
        W_concat_T,      # weight_T ptr float16 (dim, out_features)
        out_raw,         # output ptr float16 (R, out_features)
        total_rows,
        dim,
        5 * H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        BLOCK_O=BLOCK_O,
        num_warps=8,
    )

    # split outputs
    left_proj_raw = out_raw[:, 0:H]
    right_proj_raw = out_raw[:, H:2*H]
    left_gate_raw = out_raw[:, 2*H:3*H]
    right_gate_raw = out_raw[:, 3*H:4*H]
    out_gate_raw = out_raw[:, 4*H:5*H]

    # sigmoid gates
    left_gate = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate = out_gate_raw.sigmoid_()

    # apply gating
    left = left_proj_raw * left_gate
    right = right_proj_raw * right_gate

    # mask if needed
    if not config.get("nomask", True):
        mask_fp16 = mask_tensor.reshape(total_rows, 1).to(torch.float16)
        left = left * mask_fp16
        right = right * mask_fp16

    # reshape to (B,N,N,H)
    left = left.view(B, N, N, H)
    right = right.view(B, N, N, H)
    out_gate = out_gate.view(B, N, N, H)

    # cubic update via batched gemm
    left_perm = left.permute(0,3,1,2).contiguous()    # (B,H,N,N)
    right_perm = right.permute(0,3,1,2).contiguous()
    left_mat = left_perm.view(B*H, N, N)
    right_mat = right_perm.view(B*H, N, N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1,2))  # (B*H,N,N)
    out = out_mat.view(B, H, N, N).permute(0,2,3,1).contiguous()  # (B,N,N,H)

    # final fused LayerNorm+gate+linear
    # flatten out to (R, H)
    out_flat = out.view(total_rows, H).contiguous()
    out_gate_flat = out_gate.view(total_rows, H).contiguous()
    # allocate buffer for final output (R, dim) fp16
    final_fp16 = torch.empty(total_rows, dim, dtype=torch.float16, device=device)

    # prepare to_out_norm weight and bias (fp32) and to_out weight (dim, H) -> transposed to (H, dim) fp16
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w_T = weights["to_out.weight"].t().contiguous().to(torch.float16)  # shape (H, dim)

    # launch second kernel
    BLOCK_M2 = 64
    BLOCK_D2 = 64  # for hidden dimension H up to 128
    BLOCK_O2 = 64  # for output dim
    grid2 = (triton.cdiv(total_rows, BLOCK_M2), triton.cdiv(dim, BLOCK_O2))

    final_norm_linear_kernel[grid2](
        out_flat,
        out_gate_flat,
        to_out_norm_w,
        to_out_norm_b,
        to_out_w_T,
        final_fp16,
        total_rows,
        H,
        dim,
        eps,
        BLOCK_M=BLOCK_M2,
        BLOCK_D=BLOCK_D2,
        BLOCK_O=BLOCK_O2,
        num_warps=8,
    )

    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final

@triton.jit
def norm_linear_kernel(
    x_ptr,          # float32* [R, dim]
    w_ptr,          # float16* [dim, out_features] (transposed)
    out_ptr,        # float16* [R, out_features]
    norm_w_ptr,     # float32* [dim] gamma
    norm_b_ptr,     # float32* [dim] beta
    total_rows,     # int32
    dim,            # int32
    out_features,   # int32
    eps,            # float32
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
    BLOCK_O: tl.constexpr,
):
    pid_row = tl.program_id(0)
    pid_col = tl.program_id(1)

    row_start = pid_row * BLOCK_M
    col_start = pid_col * BLOCK_O

    rows = row_start + tl.arange(0, BLOCK_M)
    cols = col_start + tl.arange(0, BLOCK_O)

    row_mask = rows < total_rows
    col_mask = cols < out_features

    # compute mean & variance per row
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_start in range(0, tl.cdiv(dim, BLOCK_D)):
        d_off = d_start * BLOCK_D + tl.arange(0, BLOCK_D)
        d_mask = d_off < dim

        x = tl.load(
            x_ptr + rows[:, None] * dim + d_off[None, :],
            mask=row_mask[:, None] & d_mask[None, :],
            other=0.0
        ).to(tl.float32)

        sum_val += tl.sum(x, axis=1)
        sum_sq += tl.sum(x * x, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # accumulator for dot product
    acc = tl.zeros([BLOCK_M, BLOCK_O], dtype=tl.float32)

    for d_start in range(0, tl.cdiv(dim, BLOCK_D)):
        d_off = d_start * BLOCK_D + tl.arange(0, BLOCK_D)
        d_mask = d_off < dim

        # load raw values
        x = tl.load(
            x_ptr + rows[:, None] * dim + d_off[None, :],
            mask=row_mask[:, None] & d_mask[None, :],
            other=0.0
        ).to(tl.float32)

        # normalize
        x_norm = (x - mean[:, None]) * rstd[:, None]

        # apply layernorm gamma/beta per dim
        gamma = tl.load(norm_w_ptr + d_off, mask=d_mask, other=0.0)
        beta = tl.load(norm_b_ptr + d_off, mask=d_mask, other=0.0)

        x_norm = x_norm * gamma[None, :] + beta[None, :]

        # cast to fp16 for dot
        x_norm_fp16 = x_norm.to(tl.float16)

        # load weight block (dim, out_features) -> (BLOCK_D, BLOCK_O)
        w = tl.load(
            w_ptr + d_off[:, None] * out_features + cols[None, :],
            mask=d_mask[:, None] & col_mask[None, :],
            other=0.0
        )  # fp16

        # dot product accumulate
        acc += tl.dot(x_norm_fp16, w).to(tl.float32)

    # store result
    out = acc.to(tl.float16)
    out_ptrs = out_ptr + rows[:, None] * out_features + cols[None, :]
    tl.store(out_ptrs, out, mask=row_mask[:, None] & col_mask[None, :])

@triton.jit
def final_norm_linear_kernel(
    inp_ptr,         # float16* [R, H]  (out from cubic)
    gate_ptr,        # float16* [R, H]  (out_gate)
    norm_w_ptr,      # float32* [H]   (to_out_norm.weight)
    norm_b_ptr,      # float32* [H]   (to_out_norm.bias)
    w_ptr,           # float16* [H, dim] (to_out weight transposed)
    out_ptr,         # float16* [R, dim] (final output)
    total_rows,     # int32
    H,              # int32 hidden dim
    dim_out,        # int32 output dim
    eps,            # float32
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
    BLOCK_O: tl.constexpr,
):
    pid_row = tl.program_id(0)
    pid_col = tl.program_id(1)

    row_start = pid_row * BLOCK_M
    col_start = pid_col * BLOCK_O

    rows = row_start + tl.arange(0, BLOCK_M)
    cols = col_start + tl.arange(0, BLOCK_O)

    row_mask = rows < total_rows
    col_mask = cols < dim_out

    # compute mean and var across hidden dim H per row
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_start in range(0, tl.cdiv(H, BLOCK_D)):
        d_off = d_start * BLOCK_D + tl.arange(0, BLOCK_D)
        d_mask = d_off < H

        x = tl.load(
            inp_ptr + rows[:, None] * H + d_off[None, :],
            mask=row_mask[:, None] & d_mask[None, :],
            other=0.0
        ).to(tl.float32)

        sum_val += tl.sum(x, axis=1)
        sum_sq += tl.sum(x * x, axis=1)

    mean = sum_val / H
    var = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # accumulator for final output
    acc = tl.zeros([BLOCK_M, BLOCK_O], dtype=tl.float32)

    for d_start in range(0, tl.cdiv(H, BLOCK_D)):
        d_off = d_start * BLOCK_D + tl.arange(0, BLOCK_D)
        d_mask = d_off < H

        # load out and gate
        out_val = tl.load(
            inp_ptr + rows[:, None] * H + d_off[None, :],
            mask=row_mask[:, None] & d_mask[None, :],
            other=0.0
        ).to(tl.float32)

        gate_val = tl.load(
            gate_ptr + rows[:, None] * H + d_off[None, :],
            mask=row_mask[:, None] & d_mask[None, :],
            other=0.0
        ).to(tl.float32)

        # normalize out
        out_norm = (out_val - mean[:, None]) * rstd[:, None]

        # apply to_out_norm gamma/beta
        gamma = tl.load(norm_w_ptr + d_off, mask=d_mask, other=0.0)
        beta = tl.load(norm_b_ptr + d_off, mask=d_mask, other=0.0)

        out_norm = out_norm * gamma[None, :] + beta[None, :]

        # gating
        gated = out_norm * gate_val   # fp32

        # cast to fp16 for dot
        gated_fp16 = gated.to(tl.float16)

        # load final linear weight block (H, dim_out)
        w = tl.load(
            w_ptr + d_off[:, None] * dim_out + cols[None, :],
            mask=d_mask[:, None] & col_mask[None, :],
            other=0.0
        )  # fp16

        acc += tl.dot(gated_fp16, w).to(tl.float32)

    # store final output
    out_fp16 = acc.to(tl.float16)
    out_ptrs = out_ptr + rows[:, None] * dim_out + cols[None, :]
    tl.store(out_ptrs, out_fp16, mask=row_mask[:, None] & col_mask[None, :])

"""
Outgoing TriMul (AlphaFold‑3) – fused Triton kernels.

The forward pass consists of three stages:

1️⃣  Row‑wise LayerNorm + four linear maps (left_proj, right_proj,
    left_gate, right_gate) + out_gate are fused into a single
    Triton kernel (`norm_linear_kernel`).  The kernel first computes the
    per‑row mean/variance, applies the LayerNorm gamma/beta, and then
    multiplies the normalized row with a packed weight matrix
    (dim × 5·hidden) to produce all five raw tensors at once.
    All heavy work is performed in FP16 (Tensor‑Cores) while the
    statistics are accumulated in FP32.

2️⃣  The cubic update
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    is a batched GEMM and is executed with the highly‑optimised
    `torch.bmm` (B·H batches of N×N matrices).

3️⃣  The second LayerNorm, the output‑gate multiplication and the final
    linear projection back to the original feature dimension are fused
    into `final_norm_linear_kernel`.  This kernel again computes the
    per‑row statistics over the hidden dimension, applies the
    `to_out_norm` gamma/beta, multiplies by the pre‑computed
    `out_gate`, and finally performs a matrix multiplication with the
    transposed `to_out` weight (H × dim).

All intermediate tensors are kept in FP16; the final result is cast
back to FP32 to match the reference implementation.
"""

import torch
import triton
import triton.language as tl


# --------------------------------------------------------------
# 1️⃣  LayerNorm + 5·H linear maps (packed)
# --------------------------------------------------------------
@triton.jit
def norm_linear_kernel(
    x_ptr,                # float32*  [R, dim]   – input rows
    w_ptr,                # float16*  [dim, out_feat] – transposed weight (dim×5·H)
    out_ptr,              # float16*  [R, out_feat]    – raw outputs
    norm_w_ptr,           # float32*  [dim]            – LayerNorm γ
    norm_b_ptr,           # float32*  [dim]            – LayerNorm β
    R,                    # int32 total rows  = B·N·N
    dim,                  # int32 feature dimension
    out_feat,             # int32 = 5·hidden_dim
    eps,                  # float32 ε for LayerNorm
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # feature tile size
    BLOCK_O: tl.constexpr,   # output‑feature tile size
):
    pid_row = tl.program_id(0)
    pid_col = tl.program_id(1)

    row_start = pid_row * BLOCK_M
    col_start = pid_col * BLOCK_O

    rows = row_start + tl.arange(0, BLOCK_M)
    cols = col_start + tl.arange(0, BLOCK_O)

    row_mask = rows < R
    col_mask = cols < out_feat

    # ----------------------------------------------------------
    #   a) compute row‑wise mean and variance (FP32)
    # ----------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_start in range(0, tl.cdiv(dim, BLOCK_D)):
        d_off = d_start * BLOCK_D + tl.arange(0, BLOCK_D)
        d_mask = d_off < dim

        x = tl.load(
            x_ptr + rows[:, None] * dim + d_off[None, :],
            mask=row_mask[:, None] & d_mask[None, :],
            other=0.0
        ).to(tl.float32)

        sum_val += tl.sum(x, axis=1)
        sum_sq  += tl.sum(x * x, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                     # 1/√(var+ε)

    # ----------------------------------------------------------
    #   b) normalized row × packed weight → (R,5·H)
    # ----------------------------------------------------------
    acc = tl.zeros([BLOCK_M, BLOCK_O], dtype=tl.float32)

    for d_start in range(0, tl.cdiv(dim, BLOCK_D)):
        d_off = d_start * BLOCK_D + tl.arange(0, BLOCK_D)
        d_mask = d_off < dim

        # raw input values
        x = tl.load(
            x_ptr + rows[:, None] * dim + d_off[None, :],
            mask=row_mask[:, None] & d_mask[None, :],
            other=0.0
        ).to(tl.float32)

        # LayerNorm (γ,β) and normalisation
        gamma = tl.load(norm_w_ptr + d_off, mask=d_mask, other=0.0)
        beta  = tl.load(norm_b_ptr + d_off, mask=d_mask, other=0.0)

        x_norm = (x - mean[:, None]) * rstd[:, None]          # FP32
        x_norm = x_norm * gamma[None, :] + beta[None, :]       # γ·, +β
        x_norm = x_norm.to(tl.float16)                         # FP16 for dot

        # weight tile   (dim, out_feat) → (BLOCK_D, BLOCK_O)
        w = tl.load(
            w_ptr + d_off[:, None] * out_feat + cols[None, :],
            mask=d_mask[:, None] & col_mask[None, :],
            other=0.0
        )  # FP16

        # accumulate
        acc += tl.dot(x_norm, w).to(tl.float32)

    # write result
    out = acc.to(tl.float16)
    out_ptrs = out_ptr + rows[:, None] * out_feat + cols[None, :]
    tl.store(out_ptrs, out, mask=row_mask[:, None] & col_mask[None, :])


# --------------------------------------------------------------
# 2️⃣  Second LayerNorm + out‑gate + final linear (packed)
# --------------------------------------------------------------
@triton.jit
def final_norm_linear_kernel(
    inp_ptr,          # float16* [R, H]   – tensor after cubic update
    gate_ptr,         # float16* [R, H]   – out_gate (already sigmoid’ed)
    norm_w_ptr,       # float32* [H]      – to_out_norm γ
    norm_b_ptr,       # float32* [H]      – to_out_norm β
    w_ptr,            # float16* [H, dim] – transposed to_out weight (H×dim)
    out_ptr,          # float16* [R, dim] – final result
    R,                # int32 total rows = B·N·N
    H,                # int32 hidden dimension
    dim,              # int32 output feature dimension
    eps,              # float32 ε for LayerNorm
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
    BLOCK_O: tl.constexpr,
):
    pid_row = tl.program_id(0)
    pid_col = tl.program_id(1)

    row_start = pid_row * BLOCK_M
    col_start = pid_col * BLOCK_O

    rows = row_start + tl.arange(0, BLOCK_M)
    cols = col_start + tl.arange(0, BLOCK_O)

    row_mask = rows < R
    col_mask = cols < dim

    # ----------------------------------------------------------
    #   a) statistics over hidden dim H (Row‑wise)
    # ----------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_start in range(0, tl.cdiv(H, BLOCK_D)):
        d_off = d_start * BLOCK_D + tl.arange(0, BLOCK_D)
        d_mask = d_off < H

        x = tl.load(
            inp_ptr + rows[:, None] * H + d_off[None, :],
            mask=row_mask[:, None] & d_mask[None, :],
            other=0.0
        ).to(tl.float32)

        sum_val += tl.sum(x, axis=1)
        sum_sq  += tl.sum(x * x, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ----------------------------------------------------------
    #   b) (norm → gate) × final linear weight
    # ----------------------------------------------------------
    acc = tl.zeros([BLOCK_M, BLOCK_O], dtype=tl.float32)

    for d_start in range(0, tl.cdiv(H, BLOCK_D)):
        d_off = d_start * BLOCK_D + tl.arange(0, BLOCK_D)
        d_mask = d_off < H

        out_val = tl.load(
            inp_ptr + rows[:, None] * H + d_off[None, :],
            mask=row_mask[:, None] & d_mask[None, :],
            other=0.0
        ).to(tl.float32)

        gate_val = tl.load(
            gate_ptr + rows[:, None] * H + d_off[None, :],
            mask=row_mask[:, None] & d_mask[None, :],
            other=0.0
        ).to(tl.float32)

        # LayerNorm on the cubic output
        gamma = tl.load(norm_w_ptr + d_off, mask=d_mask, other=0.0)
        beta  = tl.load(norm_b_ptr + d_off, mask=d_mask, other=0.0)

        out_norm = (out_val - mean[:, None]) * rstd[:, None]
        out_norm = out_norm * gamma[None, :] + beta[None, :]

        # apply out‑gate
        gated = out_norm * gate_val                     # FP32

        # cast to fp16 for the matmul with the final weight
        gated_fp16 = gated.to(tl.float16)

        # weight tile (H, dim) → (BLOCK_D, BLOCK_O)
        w = tl.load(
            w_ptr + d_off[:, None] * dim + cols[None, :],
            mask=d_mask[:, None] & col_mask[None, :],
            other=0.0
        )  # FP16

        acc += tl.dot(gated_fp16, w).to(tl.float32)

    # write final tensor
    out = acc.to(tl.float16)
    out_ptrs = out_ptr + rows[:, None] * dim + cols[None, :]
    tl.store(out_ptrs, out, mask=row_mask[:, None] & col_mask[None, :])


# --------------------------------------------------------------
# Triton‑accelerated outgoing TriMul
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (AlphaFold‑3).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float) – optional
        - weights_dict : model weights (see key list below)
        - config_dict  : {"dim": int, "hidden_dim": int, "nomask": bool}

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    inp, mask, w, cfg = data
    dim         = cfg["dim"]
    hidden_dim  = cfg["hidden_dim"]
    eps         = 1e-5
    device      = inp.device
    B, N, _, _ = inp.shape
    R = B * N * N                       # total number of rows
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣  Fuse first LayerNorm + all linear maps (5·H outputs)
    # ------------------------------------------------------------------
    # flatten to (R, dim)
    inp_flat = inp.reshape(R, dim).contiguous()          # FP32

    # pack the five weight matrices and transpose for GEMM
    W5 = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                 # (5·H, dim)
    W5_T = W5.t().contiguous()          # (dim, 5·H)

    # allocate fused output buffer
    out5 = torch.empty(R, 5 * H, dtype=torch.float16, device=device)

    BLOCK_M = 64
    BLOCK_D = 64
    BLOCK_O = 64
    grid = (triton.cdiv(R, BLOCK_M), triton.cdiv(5 * H, BLOCK_O))

    norm_linear_kernel[grid](
        inp_flat,
        W5_T,
        out5,
        w["norm.weight"],       # γ  (dim)
        w["norm.bias"],         # β  (dim)
        R,
        dim,
        5 * H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        BLOCK_O=BLOCK_O,
        num_warps=8,
    )

    # split the fused result
    left_raw   = out5[:, 0:H]
    right_raw  = out5[:, H:2 * H]
    left_gate_raw  = out5[:, 2 * H:3 * H]
    right_gate_raw = out5[:, 3 * H:4 * H]
    out_gate_raw   = out5[:, 4 * H:5 * H]

    # --------------------------------------------------------------
    # 2️⃣  Apply sigmoids & optional pairwise mask
    # --------------------------------------------------------------
    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    left  = left_raw * left_gate       # (R, H) FP16
    right = right_raw * right_gate

    if not cfg.get("nomask", True):
        # mask broadcasted over the hidden dimension
        mask_fp16 = mask.reshape(R, 1).to(torch.float16)
        left  = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # reshape for the cubic update
    # ------------------------------------------------------------------
    left  = left.view(B, N, N, H)
    right = right.view(B, N, N, H)
    out_gate = out_gate.view(B, N, N, H)

    # ------------------------------------------------------------------
    # 3️⃣  Cubic update: out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    #     (batched GEMM over the hidden dimension)
    # ------------------------------------------------------------------
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    left_mat  = left_perm.view(B * H, N, N)               # (B·H, N, N)
    right_mat = right_perm.view(B * H, N, N)

    # batched matrix multiplication (Tensor‑Core accelerated)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B·H, N, N)

    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 4️⃣  Fuse second LayerNorm, out‑gate and final linear projection
    # ------------------------------------------------------------------
    out_flat = out.reshape(R, H).contiguous()                # (R, H) FP16
    out_gate_flat = out_gate.reshape(R, H).contiguous()      # (R, H) FP16
    final_fp16 = torch.empty(R, dim, dtype=torch.float16, device=device)

    # prepare final weight (H × dim) in FP16
    to_out_w_T = w["to_out.weight"].t().contiguous().to(torch.float16)

    grid2 = (triton.cdiv(R, BLOCK_M), triton.cdiv(dim, BLOCK_O))
    final_norm_linear_kernel[grid2](
        out_flat,
        out_gate_flat,
        w["to_out_norm.weight"],   # γ (H)
        w["to_out_norm.bias"],     # β (H)
        to_out_w_T,
        final_fp16,
        R,
        H,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        BLOCK_O=BLOCK_O,
        num_warps=8,
    )

    # cast back to float32 to match the reference implementation
    return final_fp16.view(B, N, N, dim).to(torch.float32)
