"""
custom_kernel – outgoing TriMul (AlphaFold‑3) forward pass
========================================================

Algorithm (all tensors reside on the same CUDA device)

1️⃣ Row‑wise LayerNorm over the last dimension (`dim`) – implemented in Triton.
   The kernel computes μ, σ² per row with fp32 accumulation and writes the
   normalized result in fp16 (or bf16).

2️⃣ Fuse the four linear layers (`left_proj`, `right_proj`, `left_gate`,
   `right_gate`) into a single GEMM:
      [R, dim] @ [4·H, dim]ᵀ → [R, 4·H]      (R = B·N·N, H = hidden_dim)
   The weight matrix is concatenated once and the GEMM is performed in
   half‑precision (fp16) via `torch.nn.functional.linear`.

3️⃣ Apply the optional mask (broadcast over the hidden dimension) and the
   three sigmoid gates (`left_gate`, `right_gate`, `out_gate`).

4️⃣ Reshape `left` and `right` to `[B, N, N, H]`, permute to `[B·H, N, N]`,
   and execute a batched matrix multiplication:
      out = left @ rightᵀ   (fp16, Tensor‑Core accelerated)
   This is the expensive N³ update (`B·H` independent GEMMs of size N×N).

5️⃣ Second row‑wise LayerNorm over the hidden dimension `H` (again a Triton
   kernel).  The output is multiplied by the output‑gate.

6️⃣ Final linear projection back to `dim` (fp16 GEMM) and cast to fp32.

Only the two LayerNorm steps are written in Triton; all dense matrix
multiplications use cuBLAS (via `torch.nn.functional.linear` / `torch.bmm`),
which is optimal on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (float32 → float16)
# ----------------------------------------------------------------------
@triton.jit
def rowwise_layernorm_kernel(
    x_ptr,          # const   * [total_rows, dim]   (float32)
    y_ptr,          # mutable * [total_rows, dim]   (float16)
    w_ptr,          # const   * [dim]               (float32)  γ
    b_ptr,          # const   * [dim]               (float32)  β
    total_rows,     # runtime int
    dim,            # runtime int
    eps,            # compile‑time float
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per tile
):
    """
    Row‑wise LayerNorm:
        y = ((x - μ) / sqrt(σ² + eps)) * γ + β
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # --------------------------------------------------------------
    # 1️⃣ Compute mean and variance for each row
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        # Load tile [BLOCK_M, BLOCK_D] (float32 → fp32)
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                 # 1 / sqrt(var + eps)

    # --------------------------------------------------------------
    # 2️⃣ Write normalized output (fp16)
    # --------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # β

        out = norm * gamma[None, :] + beta[None, :]               # fp32
        out_fp16 = out.to(tl.float16)

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (bool/float) – may be
                         ignored when ``config["nomask"]`` is True
        - weights_dict : dict containing the model parameters (all float32)
        - config_dict  : must contain "dim", "hidden_dim" and optional "nomask"

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    device = input_tensor.device

    B, N, _, _ = input_tensor.shape
    R = B * N * N                           # total rows for the first LN

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → fp16
    # --------------------------------------------------------------
    x_flat = input_tensor.view(R, dim).contiguous()          # [R, dim] fp32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 256               # rows per program
    BLOCK_D = 64                # columns per tile (covers dim≤384 with ≤6 loops)
    grid = (triton.cdiv(R, BLOCK_M),)

    rowwise_layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],          # γ  (float32)
        weights["norm.bias"],            # β  (float32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    # --------------------------------------------------------------
    # 2️⃣ Fuse left/right projections and their gates (single GEMM)
    # --------------------------------------------------------------
    H = hidden_dim
    # Concatenate the four weight matrices -> [4·H, dim]
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)   # use half‑precision for Tensor‑Core GEMM

    # (R, dim) @ (4·H, dim)ᵀ → (R, 4·H)
    proj_gate = F.linear(x_norm, w_fused)   # bias omitted (False in model)

    # Split the fused result
    left_proj_raw  = proj_gate[:, 0:H]           # [R, H]
    right_proj_raw = proj_gate[:, H:2*H]         # [R, H]
    left_gate_raw  = proj_gate[:, 2*H:3*H]       # [R, H]
    right_gate_raw = proj_gate[:, 3*H:4*H]       # [R, H]

    # Output‑gate linear (separate weight)
    out_gate_raw = F.linear(
        x_norm,
        weights["out_gate.weight"].to(torch.float16)   # [H, dim]
    )   # → [R, H]

    # --------------------------------------------------------------
    # 3️⃣ Mask (if any), sigmoid gates and gating
    # --------------------------------------------------------------
    if not config.get("nomask", True):
        # mask is broadcast over the hidden dimension
        mask_h = mask_tensor.to(torch.float16).reshape(R, 1)   # [R, 1]
        left_proj_raw  = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    # In‑place sigmoid (half‑precision)
    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    # Gating
    left  = left_proj_raw * left_gate      # [R, H] fp16
    right = right_proj_raw * right_gate    # [R, H] fp16

    # --------------------------------------------------------------
    # 4️⃣ N³ multiplicative update (batched GEMM)
    # --------------------------------------------------------------
    # Reshape to 4‑D tensors
    left  = left.view(B, N, N, H)          # [B, N, N, H]
    right = right.view(B, N, N, H)         # [B, N, N, H]
    out_gate = out_gate.view(B, N, N, H)   # [B, N, N, H]

    # Bring hidden dimension into the batch axis for a batched GEMM
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # [B·H, N, N]
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)  # [B·H, N, N]

    # (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Restore original layout: [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm over the hidden dimension → fp16
    # --------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()
    out_norm = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(R, BLOCK_M),)
    rowwise_layernorm_kernel[grid2](
        out_flat,
        out_norm,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    out_norm = out_norm.view(B, N, N, H)   # [B, N, N, H] fp16

    # --------------------------------------------------------------
    # 6️⃣ Apply output gate and final linear projection
    # --------------------------------------------------------------
    out_norm = out_norm * out_gate          # element‑wise gate, fp16

    # Final projection back to `dim`
    final_half = F.linear(
        out_norm.view(R, H),                     # [R, H]
        weights["to_out.weight"].to(torch.float16)   # [dim, H]
    )                                            # → [R, dim] (fp16)

    final = final_half.view(B, N, N, dim).to(torch.float32)

    return final
