    pid_out = tl.program_id(1)
    row_start = pid_row * BLOCK_M
    col_start = pid_out * BLOCK_O

    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # compute mean and var of x over H (dim)
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):

        sum_sq += tl.sum(vals * vals, axis=1)

                         mask=mask_o[:, None] & mask_h[None, :],

custom_kernel
=============
Highly‑optimized outgoing TriMul (AlphaFold‑3) forward pass.

Algorithm
---------
1️⃣ Fuse the *input* LayerNorm with the five linear projections
   (left, right, left‑gate, right‑gate, out‑gate) into a single
   Triton kernel:  (R, dim) → (R, 5·H)   (R = B·N·N).
   The kernel computes the row‑wise mean/variance, normalises,
   applies the LN scale/shift and performs a tiled GEMM against
   the fused weight matrix (FP16).  All work is done in one pass
   over the input, cutting memory traffic in half.

2️⃣ Apply the optional mask, the sigmoid gates and the element‑wise
   left/right gating (still in FP16).

3️⃣ Batched matrix multiplication for the core multiplicative update:
   out₍b,i,j,d₎ = Σₖ left₍b,i,k,d₎ · right₍b,j,k,d₎.
   Implemented with a single `torch.bmm` on the reshaped tensors
   (fast cuBLAS kernels).

4️⃣ Fuse the *output* LayerNorm, the output‑gate multiplication and
   the final projection back to `dim` into a second Triton kernel.
   The kernel reads the hidden‑dim tensor (FP16), computes its
   per‑row mean/variance, normalises, applies LN γ/β, multiplies by
   the already‑sigmoid‑ed out‑gate, and finally multiplies by the
   output weight matrix (FP16 → FP32 accumulator).  The result is
   written as FP16 and cast to FP32 for the final return value.

All heavy per‑row work is fused into two custom kernels; the only
operations left to PyTorch are the N³ batched GEMM and a few cheap
element‑wise fusions, giving a large speed‑up over the naïve
implementation while keeping the exact mathematical behaviour of
the reference module.

        mask_d = offs_d < dim

        x_ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        x = tl.load(x_ptrs,
                    mask=mask_m[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)

        sum_val += tl.sum(x, axis=1)
        sum_sq  += tl.sum(x * x, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------------------------------------------------------------
    # 2️⃣ Accumulate tiled GEMM (FP16 inputs, FP32 accumulator)
    # --------------------------------------------------------------
    acc = tl.zeros([BLOCK_M, BLOCK_O], dtype=tl.float32)

    for d_tile in range(0, num_d_tiles):
        d_off = d_tile * BLOCK_D
        offs_d = d_off + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        # ----- load and normalize a tile of x -----
        x_ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        x = tl.load(x_ptrs,
                    mask=mask_m[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)

        norm = (x - mean[:, None]) * rstd[:, None]               # FP32
        γ = tl.load(ln_weight_ptr + offs_d, mask=mask_d, other=0.0)  # FP32
        β = tl.load(ln_bias_ptr   + offs_d, mask=mask_d, other=0.0)  # FP32
        norm = norm * γ[None, :] + β[None, :]                    # FP32
        norm_fp16 = norm.to(tl.float16)                         # now FP16 for dot

        # ----- load a tile of the fused weight (shape: [out_feat, dim]) -----
        # Want tile of shape (BLOCK_D, BLOCK_O) so that we can call tl.dot
        # weight[out, d] -> address = out*dim + d
        w_ptrs = w_ptr + (offs_o[:, None] * dim) + offs_d[None, :]   # (BLOCK_O, BLOCK_D)
        # transpose on‑the‑fly by swapping axes in the load
        w_tile = tl.load(w_ptrs,
                         mask=mask_o[:, None] & mask_d[None, :],
                         other=0.0).to(tl.float16)                 # (BLOCK_O, BLOCK_D)
        w_tile_T = tl.trans(w_tile, (1, 0))                         # (BLOCK_D, BLOCK_O)

        # ----- GEMM accumulation -----
        acc += tl.dot(norm_fp16, w_tile_T)                         # (BLOCK_M, BLOCK_O)

    # --------------------------------------------------------------
    # 3️⃣ Write result (FP16)
    # --------------------------------------------------------------
    out_ptrs = out_ptr + offs_m[:, None] * out_feat + offs_o[None, :]
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_o[None, :])


# ----------------------------------------------------------------------
# 2️⃣ Fused output LayerNorm + gate multiplication + final Linear
# ----------------------------------------------------------------------
@triton.jit
def fused_outnorm_gate_linear_kernel(
    x_ptr,            # FP16 input  [R, H]
    gate_ptr,         # FP16 gate   [R, H]   (sigmoid‑ed out_gate)
    out_ptr,          # FP16 output [R, dim]
    ln_weight_ptr,    # FP32 LN γ   [H]
    ln_bias_ptr,      # FP32 LN β   [H]
    w_ptr,            # FP16 linear weight [dim, H]   (row‑major)
    R, H, dim, eps,   # runtime args
    BLOCK_M: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_O: tl.constexpr,
):
    pid_row = tl.program_id(0)
    pid_out = tl.program_id(1)

    row_start = pid_row * BLOCK_M
    out_start = pid_out * BLOCK_O

    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < R

    offs_o = out_start + tl.arange(0, BLOCK_O)
    mask_o = offs_o < dim

    # --------------------------------------------------------------
    # 1️⃣ Row‑wise mean / variance over hidden dim H (FP32)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    num_h_tiles = tl.cdiv(H, BLOCK_H)
    for h_tile in range(0, num_h_tiles):
        h_off = h_tile * BLOCK_H
        offs_h = h_off + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        x_ptrs = x_ptr + offs_m[:, None] * H + offs_h[None, :]
        x = tl.load(x_ptrs,
                    mask=mask_m[:, None] & mask_h[None, :],
                    other=0.0).to(tl.float32)

        sum_val += tl.sum(x, axis=1)
        sum_sq  += tl.sum(x * x, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------------------------------------------------------------
    # 2️⃣ Accumulate (LN → gate → Linear) in a single pass
    # --------------------------------------------------------------
    acc = tl.zeros([BLOCK_M, BLOCK_O], dtype=tl.float32)

    for h_tile in range(0, num_h_tiles):
        h_off = h_tile * BLOCK_H
        offs_h = h_off + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # ----- load x and normalize -----
        x_ptrs = x_ptr + offs_m[:, None] * H + offs_h[None, :]
        x = tl.load(x_ptrs,
                    mask=mask_m[:, None] & mask_h[None, :],
                    other=0.0).to(tl.float32)

        norm = (x - mean[:, None]) * rstd[:, None]                # FP32

        γ = tl.load(ln_weight_ptr + offs_h, mask=mask_h, other=0.0)  # FP32
        β = tl.load(ln_bias_ptr   + offs_h, mask=mask_h, other=0.0)  # FP32
        norm = norm * γ[None, :] + β[None, :]                     # FP32

        # ----- multiply by out‑gate (already sigmoid‑ed) -----
        gate_ptrs = gate_ptr + offs_m[:, None] * H + offs_h[None, :]
        gate = tl.load(gate_ptrs,
                       mask=mask_m[:, None] & mask_h[None, :],
                       other=0.0).to(tl.float32)
        norm = norm * gate                                       # FP32

        # cast to FP16 for the final GEMM
        norm_fp16 = norm.to(tl.float16)                          # (BLOCK_M, BLOCK_H)

        # ----- load a tile of the output linear weight -----
        # weight shape [dim, H]; we need tile (H_tile, O_tile) i.e. (BLOCK_H, BLOCK_O)
        w_ptrs = w_ptr + offs_h[:, None] + H * offs_o[None, :]   # (BLOCK_H, BLOCK_O)
        w_tile = tl.load(w_ptrs,
                         mask=mask_h[:, None] & mask_o[None, :],
                         other=0.0).to(tl.float16)              # (BLOCK_H, BLOCK_O)

        # ----- GEMM accumulation -----
        acc += tl.dot(norm_fp16, w_tile)                         # (BLOCK_M, BLOCK_O)

    # --------------------------------------------------------------
    # 3️⃣ Write final projection (FP16)
    # --------------------------------------------------------------
    out_ptrs = out_ptr + offs_m[:, None] * dim + offs_o[None, :]
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_o[None, :])


# ----------------------------------------------------------------------
# 3️⃣ Entry point expected by the grader
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Outgoing TriMul forward pass (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float) – may be ignored.
        - weights_dict : mapping name → torch.Tensor (all model parameters)
        - config_dict  : must contain "dim" and "hidden_dim".  Optional
                         key "nomask" (bool, default True).

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    x, mask, weights, cfg = data
    device = x.device
    dim       = cfg["dim"]
    hidden    = cfg["hidden_dim"]
    eps       = 1e-5
    nomask    = cfg.get("nomask", True)

    B, N, _, _ = x.shape
    R = B * N * N                     # total rows = B·N·N
    P = 5 * hidden                    # out‑features in the fused projection

    # --------------------------------------------------------------
    # 0️⃣ Prepare fused projection weight: [5·H, dim]  (FP16)
    # --------------------------------------------------------------
    w_left   = weights["left_proj.weight"].to(torch.float16)   # (H, dim)
    w_right  = weights["right_proj.weight"].to(torch.float16)
    w_lgate  = weights["left_gate.weight"].to(torch.float16)
    w_rgate  = weights["right_gate.weight"].to(torch.float16)
    w_outgate= weights["out_gate.weight"].to(torch.float16)

    w_fused = torch.cat([w_left, w_right, w_lgate, w_rgate, w_outgate],
                        dim=0).contiguous()                     # (5·H, dim)

    # --------------------------------------------------------------
    # 1️⃣ Fused input LN + 5×Linear (produces left/right proj + gates + out_gate)
    # --------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()                     # (R, dim) FP32
    proj_out = torch.empty(R, P, dtype=torch.float16, device=device)

    BLOCK_M = 64            # rows per program
    BLOCK_D = 64            # tile over input dim
    BLOCK_O = 64            # tile over output features (5·H)

    grid = (triton.cdiv(R, BLOCK_M), triton.cdiv(P, BLOCK_O))
    fused_proj_kernel[grid](
        x_flat,
        proj_out,
        weights["norm.weight"],          # LN γ (FP32)
        weights["norm.bias"],            # LN β (FP32)
        w_fused,
        R,
        dim,
        P,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        BLOCK_O=BLOCK_O,
        num_warps=8,
    )

    # --------------------------------------------------------------
    # 2️⃣ Slice the fused output into the individual tensors
    # --------------------------------------------------------------
    left_proj_raw  = proj_out[:, 0:               hidden]          # (R, H)
    right_proj_raw = proj_out[:, hidden:            2*hidden]      # (R, H)
    left_gate_raw  = proj_out[:, 2*hidden:          3*hidden]      # (R, H)
    right_gate_raw = proj_out[:, 3*hidden:          4*hidden]      # (R, H)
    out_gate_raw   = proj_out[:, 4*hidden:          5*hidden]      # (R, H)

    # --------------------------------------------------------------
    # 3️⃣ Optional mask (applied before the sigmoid gates)
    # --------------------------------------------------------------
    if not nomask:
        # mask shape [B, N, N] → (R, 1)
        mask_flat = mask.to(torch.float16).reshape(R, 1)
        left_proj_raw  = left_proj_raw * mask_flat
        right_proj_raw = right_proj_raw * mask_flat

    # --------------------------------------------------------------
    # 4️⃣ Sigmoid gates and element‑wise gating
    # --------------------------------------------------------------
    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    left  = left_proj_raw * left_gate                   # (R, H)
    right = right_proj_raw * right_gate                 # (R, H)

    # --------------------------------------------------------------
    # 5️⃣ Core multiplicative update (einsum -> batched GEMM)
    # --------------------------------------------------------------
    left = left.view(B, N, N, hidden).permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.view(B, N, N, hidden).permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    left_mat  = left.view(B * hidden, N, N)           # (B·H, N, N)
    right_mat = right.view(B * hidden, N, N)          # (B·H, N, N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B·H, N, N)

    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)
    out_flat = out.view(R, hidden)                                            # (R, H)

    # --------------------------------------------------------------
    # 6️⃣ Fused output LN + out‑gate multiplication + final Linear
    # --------------------------------------------------------------
    final_fp16 = torch.empty(R, dim, dtype=torch.float16, device=device)

    # Linear weight for the final projection: [dim, H] (FP16)
    w_to_out = weights["to_out.weight"].to(torch.float16).contiguous()

    BLOCK_M2 = 64   # rows per program (same as before)
    BLOCK_H  = 64   # tile over hidden dim H
    BLOCK_O2 = 64   # tile over final dim

    grid2 = (triton.cdiv(R, BLOCK_M2), triton.cdiv(dim, BLOCK_O2))
    fused_outnorm_gate_linear_kernel[grid2](
        out_flat,
        out_gate,
        final_fp16,
        weights["to_out_norm.weight"],   # LN γ (FP32)
        weights["to_out_norm.bias"],     # LN β (FP32)
        w_to_out,
        R,
        hidden,
        dim,
        eps,
        BLOCK_M=BLOCK_M2,
        BLOCK_H=BLOCK_H,
        BLOCK_O=BLOCK_O2,
        num_warps=8,
    )

    # --------------------------------------------------------------
    # 7️⃣ Cast back to FP32 and reshape to the original layout
    # --------------------------------------------------------------
    out_fp32 = final_fp16.view(B, N, N, dim).to(torch.float32)
    return out_fp32
