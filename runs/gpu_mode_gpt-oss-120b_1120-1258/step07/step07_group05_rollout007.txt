@triton.jit
def layernorm_gate_kernel(
    x_ptr, gate_ptr, y_ptr,
    weight_ptr, bias_ptr,
    total_rows, dim,
    eps,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # compute mean and var
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim

"""
Highly‑optimized forward pass for the **outgoing** TriMul operator
(AlphaFold‑3 style).  The implementation follows the reference PyTorch
module but fuses the most expensive element‑wise work in Triton kernels:

1️⃣ Row‑wise LayerNorm (float32 → float16) for the first LN.
2️⃣ A single fused half‑precision GEMM that computes the left/right
   projections together with their gates (4·H outputs).
3️⃣ Optional mask application and in‑place sigmoid for the three gates.
4️⃣ Batched matrix multiplication (B·H, N, N) @ (B·H, N, N)ᵀ using
   `torch.bmm` – leverages cuBLAS Tensor‑Cores.
5️⃣ A fused Triton kernel that performs the second LayerNorm **and**
   multiplies by the output‑gate in one pass (removes an extra kernel
   launch and an extra memory write).
6️⃣ Final linear projection (H → dim) using a regular FP16 GEMM.
All heavy element‑wise work (LayerNorms, gating) is performed in Triton,
while the large dense matrix products use cuBLAS for maximum throughput.
The function `custom_kernel` receives a tuple `(input, mask, weights,
config)` and returns a float‑32 tensor of shape `[B, N, N, dim]`.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (float32 → float16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   * [rows, dim]   (float32)
    y_ptr,          # mutable * [rows, dim]   (float16)
    weight_ptr,     # const   * [dim]       (float32)
    bias_ptr,       # const   * [dim]       (float32)
    rows,           # int (runtime)
    dim,            # int (runtime)
    eps,            # float (compile‑time)
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # ------------------------------------------------------------------
    # 1️⃣ Compute per‑row mean & variance
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------------------------------------------------------
    # 2️⃣ Write normalized values γ·(x−μ)·rstd + β   (fp16)
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # β
        out = norm * w[None, :] + b[None, :]                     # fp32

        out_fp16 = out.to(tl.float16)
        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16, mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
# Triton kernel: second LayerNorm + element‑wise output‑gate multiplication
# ----------------------------------------------------------------------
@triton.jit
def layernorm_gate_kernel(
    x_ptr,          # const   * [rows, dim]   (float16)
    gate_ptr,       # const   * [rows, dim]   (float16)
    y_ptr,          # mutable * [rows, dim]   (float16)
    weight_ptr,     # const   * [dim]       (float32)
    bias_ptr,       # const   * [dim]       (float32)
    rows,           # int (runtime)
    dim,            # int (runtime)
    eps,            # float (compile‑time)
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # ------------------------------------------------------------------
    # Compute mean / variance (accumulate in fp32)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------------------------------------------------------
    # Normalise, apply γ/β and multiply by output‑gate
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        x_ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        x_vals = tl.load(x_ptrs,
                         mask=mask_m[:, None] & mask_d[None, :],
                         other=0.0).to(tl.float32)

        g_ptrs = gate_ptr + offs_m[:, None] * dim + offs_d[None, :]
        g_vals = tl.load(g_ptrs,
                         mask=mask_m[:, None] & mask_d[None, :],
                         other=0.0).to(tl.float32)

        norm = (x_vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β

        out = (norm * w[None, :] + b[None, :]) * g_vals
        out_fp16 = out.to(tl.float16)

        y_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(y_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : mapping with all Linear/LayerNorm parameters
        - config_dict  : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    device = x.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)      # True → ignore mask
    eps = 1e-5

    B, N, _, _ = x.shape
    total_rows = B * N * N          # R = B·N·N
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (fp32 → fp16) using Triton
    # ------------------------------------------------------------------
    x_flat = x.view(total_rows, dim).contiguous()  # [R, dim] fp32
    ln1_out = torch.empty(total_rows, dim,
                          dtype=torch.float16,
                          device=device)

    BLOCK_M = 256
    BLOCK_D = 64
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        ln1_out,
        weights["norm.weight"],   # γ (float32)
        weights["norm.bias"],     # β (float32)
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection (left, right, left_gate, right_gate)
    # ------------------------------------------------------------------
    # Concatenate the four projection weight matrices and cast to fp16
    w_proj = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
        ],
        dim=0,
    ).to(device=device, dtype=torch.float16)  # [4·H, dim]

    # fused GEMM: [R, dim] @ [4·H, dim].T → [R, 4·H]
    fused = F.linear(ln1_out, w_proj)          # fp16, no bias

    # split fused result
    left_raw      = fused[:, 0:H]               # [R, H]
    right_raw     = fused[:, H:2*H]              # [R, H]
    left_gate_raw = fused[:, 2*H:3*H]            # [R, H]
    right_gate_raw = fused[:, 3*H:4*H]           # [R, H]

    # output‑gate (separate GEMM)
    out_gate_raw = F.linear(
        ln1_out,
        weights["out_gate.weight"].to(device=device, dtype=torch.float16)  # [H, dim]
    )                                          # [R, H] fp16

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask, sigmoid gates and element‑wise weighting
    # ------------------------------------------------------------------
    if (not nomask) and mask is not None:
        # mask: [B, N, N] → [R, 1] (fp16)
        mask_flat = mask.to(device=device, dtype=torch.float16).reshape(total_rows, 1)
        left_raw  = left_raw * mask_flat
        right_raw = right_raw * mask_flat

    # in‑place sigmoid to minimise temporaries
    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    # Apply gate scalings
    left  = left_raw  * left_gate   # [R, H] fp16
    right = right_raw * right_gate  # [R, H] fp16

    # ------------------------------------------------------------------
    # 4️⃣ Reshape for the N³ multiplicative update
    # ------------------------------------------------------------------
    left  = left.view(B, N, N, H)    # [B, N, N, H]
    right = right.view(B, N, N, H)   # [B, N, N, H]
    out_gate = out_gate.view(B, N, N, H)   # [B, N, N, H]

    # Bring head dimension before the N dimensions to match batched bmm layout
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    # Collapse batch & heads → [B·H, N, N]
    left_mat  = left_perm.view(B * H, N, N)
    right_mat = right_perm.view(B * H, N, N)

    # ------------------------------------------------------------------
    # 5️⃣ Batched matrix multiplication (N³ update)
    # ------------------------------------------------------------------
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B·H, N, N)

    # Restore original layout → [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm + output‑gate multiplication (fused Triton)
    # ------------------------------------------------------------------
    out_flat   = out.view(total_rows, H).contiguous()          # [R, H] fp16
    out_gate_flat = out_gate.view(total_rows, H).contiguous() # [R, H] fp16
    ln2_out = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_gate_kernel[grid2](
        out_flat,
        out_gate_flat,
        ln2_out,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 7️⃣ Final linear projection and cast back to float32
    # ------------------------------------------------------------------
    final_fp16 = F.linear(
        ln2_out,
        weights["to_out.weight"].to(device=device, dtype=torch.float16)  # [dim, H]
    )   # [R, dim] fp16

    final = final_fp16.view(B, N, N, dim).to(torch.float32)  # float32 output
    return final
