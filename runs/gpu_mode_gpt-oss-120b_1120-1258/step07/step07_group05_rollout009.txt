"""
Highly‑optimized forward pass for the **outgoing** TriMul operator
(AlphaFold‑3 style).

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the input (float32 → float16) – a
   custom Triton kernel computes mean/variance per row and
   writes the normalized result directly in fp16.

2️⃣ One fused half‑precision GEMM computes the four linear
   projections (left/right and their two gates) in a single
   call to `F.linear`.  All weights are cast to fp16 beforehand.

3️⃣ A second GEMM produces the output‑gate.

4️⃣ Optional mask is applied to the two projected tensors,
   then the sigmoid‐gated values are multiplied element‑wise.

5️⃣ The expensive N³ “multiplicative update’’ is performed with a
   single `torch.einsum` over the k‑dimension:
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   This avoids the costly `permute(...).contiguous()` that would
   otherwise copy the (B·N·N·H) tensor.

6️⃣ A second row‑wise LayerNorm (over the hidden dimension H) is
   fused in the same Triton kernel used in step 1.

7️⃣ The normalized tensor is multiplied by the output‑gate,
   then a final half‑precision linear projection maps H → dim.
   The result is cast back to float32 for the module output.

All heavy element‑wise work (both LayerNorms, optional mask,
sigmoids) is performed on the GPU with Triton, while the large
dense matrix products use cuBLAS/Tensor‑Cores.  The implementation
fits the required signature `custom_kernel(data)` and supports all
test configurations listed in the problem statement.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (float32 → float16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   * [total_rows, dim]   (float32)
    y_ptr,          # mutable * [total_rows, dim]   (float16)
    weight_ptr,     # const   * [dim]               (float32)
    bias_ptr,       # const   * [dim]               (float32)
    total_rows,     # int (runtime)
    dim,            # int (runtime)
    eps,            # float (compile‑time constexpr)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per iteration
):
    """
    Row‑wise LayerNorm:
        y = ((x - μ) / sqrt(σ² + eps)) * γ + β
    """
    pid = tl.program_id(0)                     # one program per BLOCK_M rows
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M) # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # 1️⃣ Compute per‑row mean and variance
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    num_dblocks = tl.cdiv(dim, BLOCK_D)
    for d_off in range(num_dblocks):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < dim
        # Load a BLOCK_M × BLOCK_D tile from x (fp32)
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                     # 1 / sqrt(var+eps)

    # ------------------------------------------------------------------
    # 2️⃣ Write normalized values γ·(x−μ)·rstd + β   (fp16)
    # ------------------------------------------------------------------
    for d_off in range(num_dblocks):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        # (x - μ) * rstd   – broadcasting across the dim axis
        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]                       # fp32

        out_fp16 = out.to(tl.float16)
        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : mapping with all Linear/LayerNorm parameters
        - config_dict  : dict with keys "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    device = x.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)          # True → ignore mask
    eps = 1e-5

    B, N, _, _ = x.shape
    total_rows = B * N * N                       # R = B·N·N
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 → fp16)
    # ------------------------------------------------------------------
    x_flat = x.view(total_rows, dim).contiguous()          # [R, dim] fp32
    ln1_out = torch.empty(total_rows, dim,
                          dtype=torch.float16,
                          device=device)

    BLOCK_M = 256
    BLOCK_D = 64
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        ln1_out,
        weights["norm.weight"],   # γ (float32)
        weights["norm.bias"],     # β (float32)
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fuse left/right projections + their two gates (4·H outputs)
    # ------------------------------------------------------------------
    # Concatenate the four weight matrices that all share the same
    # input dimension `dim`.  All weights are stored in float32 – cast
    # once to fp16 for the GEMM.
    w_proj = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
        ],
        dim=0,
    ).to(device=device, dtype=torch.float16)   # [4·H, dim]

    # Fused GEMM: [R, dim] @ [4·H, dim].T → [R, 4·H] (fp16, no bias)
    fused = F.linear(ln1_out, w_proj)

    # Split the fused result into the four tensors we need.
    left_raw      = fused[:, 0:H]                # [R, H]
    right_raw     = fused[:, H:2*H]               # [R, H]
    left_gate_raw = fused[:, 2*H:3*H]             # [R, H]
    right_gate_raw= fused[:, 3*H:4*H]             # [R, H]

    # Output‑gate (separate GEMM)
    out_gate_raw = F.linear(
        ln1_out,
        weights["out_gate.weight"].to(device=device, dtype=torch.float16)  # [H, dim]
    )   # [R, H] fp16

    # ------------------------------------------------------------------
    # 3️⃣ Mask, sigmoid gates and element‑wise weighting
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask: [B, N, N] → [R, 1] (float16)
        mask_flat = mask.to(device=device, dtype=torch.float16).reshape(total_rows, 1)
        left_raw  = left_raw * mask_flat
        right_raw = right_raw * mask_flat

    # In‑place sigmoid to avoid temporaries
    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    # Apply gates
    left  = left_raw  * left_gate   # [R, H] fp16
    right = right_raw * right_gate  # [R, H] fp16

    # ------------------------------------------------------------------
    # 4️⃣ N³ multiplicative update (no large memory copy)
    # ------------------------------------------------------------------
    # Reshape back to the original 4‑D layout; no permute/contiguous
    # because the einsum can consume the strided view directly.
    left  = left.view(B, N, N, H)   # [B, N, N, H]
    right = right.view(B, N, N, H)  # [B, N, N, H]
    out_gate = out_gate.view(B, N, N, H)   # [B, N, N, H]

    # Einstein‑summation over the k‑dimension.
    # This avoids the costly permutation / view that would otherwise
    # create an extra copy of size (B·H·N·N).
    out = torch.einsum('b i k h, b j k h -> b i j h', left, right)   # fp16

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm over the hidden dimension (H)
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, H).contiguous()
    ln2_out = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        ln2_out,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Apply output gate, final projection, cast to float32
    # ------------------------------------------------------------------
    out_norm = ln2_out.view(B, N, N, H)           # fp16
    out_norm = out_norm * out_gate                # element‑wise fp16

    # Final linear mapping H → dim (half‑precision GEMM)
    final_fp16 = F.linear(
        out_norm.view(total_rows, H),
        weights["to_out.weight"].to(device=device, dtype=torch.float16)   # [dim, H]
    )   # [R, dim] fp16

    # Cast back to FP32 for the module output
    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final
