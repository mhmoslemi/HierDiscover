import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def layernorm_kernel(
    x_ptr, y_ptr, weight_ptr, bias_ptr,
    total_rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # accumulate sums for mean & var
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    # iterate over channel blocks
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # write normalized output (fp16)
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)
        out = norm * w[None, :] + b[None, :]
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

def custom_kernel(data):
    """
    Outgoing TriMul forward pass (AlphaFold‑3 style).

    Steps:
    1️⃣ Row‑wise LayerNorm on the input (FP32 → FP16).
    2️⃣ Fuse left/right projections and the four gating linear layers into a single
       half‑precision GEMM (by concatenating the weight matrices).
    3️⃣ Compute the output‑gate with a second fused GEMM.
    4️⃣ Optionally mask the left/right tensors and apply sigmoid gates.
    5️⃣ Tri‑multiplicative update:
           out[b,i,j,:] = Σₖ left[b,i,k,:] * right[b,j,k,:]
       performed as a batched FP16 GEMM (tensor‑core friendly).
    6️⃣ Second LayerNorm over the hidden dimension.
    7️⃣ Multiply by the output‑gate and linearly project back to `dim`.
    Returns a float32 tensor of shape [B, N, N, dim].
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    total_rows = B * N * N          # rows for row‑wise ops

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (fp32 → fp16)
    # ------------------------------------------------------------------
    x_flat = x.view(total_rows, dim).contiguous()                # [R, dim] fp32
    x_norm = torch.empty(total_rows, dim, dtype=torch.float16, device=device)

    # launch triton kernel
    BLOCK_M = 256
    BLOCK_D = 32
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,                     # x_ptr
        x_norm,                     # y_ptr
        weights["norm.weight"],    # weight_ptr (fp32)
        weights["norm.bias"],      # bias_ptr   (fp32)
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fuse left/right projections + gates into one GEMM
    # ------------------------------------------------------------------
    H = hidden_dim
    # concatenate weight matrices (each [H, dim]) → [4*H, dim]
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16, non_blocking=True)   # fp16

    # fused linear: (R, dim) @ (4*H, dim)^T → (R, 4*H)
    proj_gate = F.linear(x_norm, w_concat)   # fp16
    left_raw   = proj_gate[:, :H]                     # [R, H]
    right_raw  = proj_gate[:, H:2*H]                  # [R, H]
    left_gate_raw  = proj_gate[:, 2*H:3*H]             # [R, H]
    right_gate_raw = proj_gate[:, 3*H:4*H]            # [R, H]

    # output gate – separate linear (still fp16)
    out_gate_raw = F.linear(
        x_norm,
        weights["out_gate.weight"].to(torch.float16, non_blocking=True),
    )   # fp16, shape [R, H]

    # ------------------------------------------------------------------
    # 3️⃣ Apply mask (if any) and sigmoid gates
    # ------------------------------------------------------------------
    if not cfg.get("nomask", True):
        # mask shape [B,N,N] → [R,1] broadcastable
        mask_h = mask.to(torch.float16, non_blocking=True).reshape(total_rows, 1)
        left_raw = left_raw * mask_h
        right_raw = right_raw * mask_h

    left_gate = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate = out_gate_raw.sigmoid_()

    left = left_raw * left_gate      # [R, H] fp16
    right = right_raw * right_gate   # [R, H] fp16

    # ------------------------------------------------------------------
    # 4️⃣ Tri‑multiplicative update (batched GEMM)
    # ------------------------------------------------------------------
    left = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous() # [B, H, N, N]
    out_gate = out_gate.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    # merge batch and head dimensions for bmm
    left_mat = left.view(B * H, N, N)     # [B*H, N, N]
    right_mat = right.view(B * H, N, N)   # [B*H, N, N]

    # (B·H, N, N) @ (B·H, N, N)^T -> (B·H, N, N); fp16 tensor‑core path
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # fp16

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm over hidden dim H (fp16)
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, H).contiguous()
    out_norm = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm,
        weights["to_out_norm.weight"],   # gamma (fp32)
        weights["to_out_norm.bias"],     # beta  (fp32)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    out_norm = out_norm.view(B, N, N, H)   # fp16

    # ------------------------------------------------------------------
    # 6️⃣ Apply output gate and final linear projection back to dim
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate          # [B,N,N,H] fp16

    final_half = F.linear(
        out_norm.view(total_rows, H),
        weights["to_out.weight"].to(torch.float16, non_blocking=True),   # [dim, H]
    )   # fp16

    # cast back to float32 as required by downstream modules
    final = final_half.view(B, N, N, dim).to(torch.float32)

    return final

"""
Outgoing Tri‑multiplicative update (AlphaFold‑3 style).

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the 4‑D input (B·N·N × dim).  The kernel runs in
   FP32 for the statistics and writes FP16 output.
2️⃣ Fuse the four linear maps (left/right projections + their gates) into a
   single GEMM by concatenating the four weight matrices.  This yields the
   raw projections and raw gates in one half‑precision matmul.
3️⃣ Compute the output‑gate with a second GEMM (still FP16).
4️⃣ (Optional) apply the pairwise mask, take sigmoids of the gates and
   obtain gated tensors `left` and `right`.
5️⃣ Tri‑multiplicative update:
       out[b,i,j,:] = Σₖ left[b,i,k,:] * right[b,j,k,:]
   performed as a batched FP16 GEMM (tensor‑core friendly) on the view
   (B·H, N, N).
6️⃣ Second LayerNorm over the hidden dimension H (FP16 → FP16).
7️⃣ Multiply by the output‑gate and linearly project back to `dim`.  The
   final tensor is cast to float32.

Only the two row‑wise LayerNorms are hand‑written in Triton; all large
matrix multiplications use cuBLAS (via `torch.nn.functional.linear` and
`torch.bmm`) which gives near‑optimal performance on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def layernorm_kernel(
    x_ptr,               # const   *[total_rows, dim]   (float32)
    y_ptr,               # mutable *[total_rows, dim]   (float16)
    weight_ptr,          # const   *[dim]               (float32) – γ
    bias_ptr,            # const   *[dim]               (float32) – β
    total_rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """
    Row‑wise LayerNorm:
        y = ((x – μ) / sqrt(σ²+eps)) * γ + β
    Output is fp16; accumulation uses fp32.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)                # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # 1️⃣ Mean & variance (fp32)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]  # [M,D]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------------------------------------------------------
    # 2️⃣ Write normalized values  γ·(x‑μ)·rstd + β   (fp16)
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]                     # fp32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weight_dict, config_dict)

        * input_tensor – torch.Tensor [B, N, N, dim] (float32)
        * mask_tensor  – torch.Tensor [B, N, N] (bool/float) or None
        * weight_dict  – mapping of all Linear/LayerNorm parameters
                         (float32, bias=False for Linear)
        * config_dict  – must contain ``dim``, ``hidden_dim`` and ``nomask``

    Returns
    -------
    torch.Tensor [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # 0️⃣ Unpack
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim        = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps        = 1e-5
    device     = x.device

    B, N, _, _ = x.shape
    R = B * N * N                     # total rows for row‑wise ops

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (fp32 → fp16)
    # ------------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()                 # [R, dim] fp32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 256
    BLOCK_D = 32
    grid = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],   # γ  (float32)
        weights["norm.bias"],     # β  (float32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fuse left/right projections & their gates into a single GEMM
    # ------------------------------------------------------------------
    H = hidden_dim
    # concatenate the four projection/gate weight matrices → [4H, dim]
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16, non_blocking=True)            # fp16

    # (R, dim) @ (4H, dim)^T → (R, 4H)
    proj_gate = F.linear(x_norm, w_concat)            # fp16

    left_raw      = proj_gate[:, 0:H]                 # [R, H]
    right_raw     = proj_gate[:, H:2*H]               # [R, H]
    left_gate_raw = proj_gate[:, 2*H:3*H]             # [R, H]
    right_gate_raw= proj_gate[:, 3*H:4*H]             # [R, H]

    # output‑gate – separate linear (still fp16)
    out_gate_raw = F.linear(
        x_norm,
        weights["out_gate.weight"].to(torch.float16, non_blocking=True),   # [H, dim]
    )   # fp16, shape [R, H]

    # ------------------------------------------------------------------
    if not cfg.get("nomask", True):

        left_raw  = left_raw  * mask_h
        right_raw = right_raw * mask_h

    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    left  = left_raw  * left_gate   # [R, H] fp16
    right = right_raw * right_gate  # [R, H] fp16

    # ------------------------------------------------------------------
    # 4️⃣ Tri‑multiplicative update (batched GEMM)
    # ------------------------------------------------------------------
    # reshape to [B, H, N, N] for easy batched matmul
    left  = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]
    out_gate = out_gate.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    # merge batch & head dims → [B·H, N, N]
    left_mat  = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)

    # (B·H, N, N) @ (B·H, N, N)^T → (B·H, N, N)   FP16 tensor‑core path
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to [B, N, N, H]  (still fp16)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm over hidden dim H (fp16)
    # ------------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()
    out_norm = torch.empty(R, H, dtype=torch.float16, device=device)

    layernorm_kernel[grid2](
        out_flat,
        out_norm,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,

    final = final_half.view(B, N, N, dim).to(torch.float32)
