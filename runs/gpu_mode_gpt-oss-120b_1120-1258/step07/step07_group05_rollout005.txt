"""
Highly‑optimized forward pass for the **outgoing** TriMul operator
(AlphaFold‑3 style).  The implementation mirrors the reference PyTorch
module but fuses the expensive parts:

1️⃣ Row‑wise LayerNorm (float32 → float16) via a custom Triton kernel.
2️⃣ A *single* fused half‑precision GEMM that computes the left/right
   projections together with their two gates **and** the output gate
   (5·H outputs).  This removes one extra GEMM pass.
3️⃣ Optional mask application and in‑place sigmoid for the three gates.
4️⃣ Batched matrix multiplication (B·H, N, N) @ (B·H, N, N).T using
   torch.bmm – lets cuBLAS Tensor‑Cores do the N³ work.
5️⃣ Second LayerNorm (over the hidden dimension) with the same Triton
   kernel.
6️⃣ Apply the output‑gate, a final half‑precision linear projection,
   and cast back to float32.

All heavy element‑wise work (LayerNorm, masking, sigmoid) is performed
in Triton while all large dense matrix products use cuBLAS for maximum
throughput on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (float32 → float16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   * [total_rows, dim]   (float32)
    y_ptr,          # mutable * [total_rows, dim]   (float16)
    weight_ptr,     # const   * [dim]               (float32)
    bias_ptr,       # const   * [dim]               (float32)
    total_rows,     # int (runtime)
    dim,            # int (runtime)
    eps,            # float (compile‑time constexpr)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns processed per inner loop
):
    """
    Row‑wise LayerNorm:
        y = ((x - μ) / sqrt(σ² + eps)) * γ + β
    Operates on a flattened 2‑D view of the input (R rows, D cols).
    """
    pid = tl.program_id(0)                 # one program per BLOCK_M rows
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)   # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # 1️⃣ Compute per‑row mean and variance
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    # iterate over the feature dimension in chunks of BLOCK_D
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < dim
        # load a BLOCK_M × BLOCK_D tile
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)          # sum over columns
        sum_sq  += tl.sum(vals * vals, axis=1)   # sum of squares

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                  # 1 / sqrt(var + eps)

    # ------------------------------------------------------------------
    # 2️⃣ Write normalized values γ·(x−μ)·rstd + β  (output fp16)
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        # (x - μ) * rstd
        norm = (vals - mean[:, None]) * rstd[:, None]

        # γ (weight) and β (bias)
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)

        out_fp32 = norm * w[None, :] + b[None, :]   # still fp32
        out_fp16 = out_fp32.to(tl.float16)

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

    # flatten to [R, dim]
    x_flat = x.view(R, dim).contiguous()          # fp32
    ln1_out = torch.empty(R, dim,
                          dtype=torch.float16,
                          device=device)      # fp16 output

    # launch kernel
    BLOCK_M = 256
    BLOCK_D = 64
    grid = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        ln1_out,
        weights["norm.weight"],   # γ (float32)
        weights["norm.bias"],     # β (float32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fuse *all* linear projections (left, right, gates, out_gate)
    # ------------------------------------------------------------------
    # Concatenate the five weight matrices -> shape [5*H, dim]
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
            weights["out_gate.weight"],    # [H, dim]
        ],
        dim=0,
    ).to(device=device, dtype=torch.float16)       # fp16

    # fused GEMM: [R, dim] @ [5*H, dim].T → [R, 5*H]
    fused = F.linear(ln1_out, w_fused)               # fp16, bias=None

    # split the fused result
    left_raw      = fused[:, 0:H]                # [R, H]
    right_raw     = fused[:, H:2*H]               # [R, H]
    left_gate_raw = fused[:, 2*H:3*H]             # [R, H]
    right_gate_raw= fused[:, 3*H:4*H]             # [R, H]
    out_gate_raw  = fused[:, 4*H:5*H]             # [R, H]

    # ------------------------------------------------------------------
    # 3️⃣ Mask, sigmoid gates and element‑wise multiplication
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask: [B, N, N] → [R, 1]  (float16)
        mask_flat = mask.to(device=device, dtype=torch.float16).reshape(R, 1)
        left_raw  = left_raw * mask_flat
        right_raw = right_raw * mask_flat

    # in‑place sigmoid to avoid temporaries
    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    # apply gates
    left  = left_raw  * left_gate   # [R, H] fp16
    right = right_raw * right_gate  # [R, H] fp16

    # ------------------------------------------------------------------
    # 4️⃣ N³ multiplicative update (batched matrix‑multiply)
    # ------------------------------------------------------------------
    # reshape to [B, N, N, H]
    left  = left.view(B, N, N, H)
    right = right.view(B, N, N, H)
    out_gate = out_gate.view(B, N, N, H)

    # bring head dim into batch axis: [B, H, N, N]
    left_perm  = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    # collapse batch & heads → [B*H, N, N]
    left_mat  = left_perm.view(B * H, N, N)
    right_mat = right_perm.view(B * H, N, N)

    # batched matmul: (B*H, N, N) @ (B*H, N, N)^T  -> (B*H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # restore layout: [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm over the hidden dimension (H)
    # ------------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()
    ln2_out = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        ln2_out,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Apply output gate, final projection and cast back to float32
    # ------------------------------------------------------------------
    out_norm = ln2_out.view(B, N, N, H)          # fp16
    out_norm = out_norm * out_gate                # element‑wise scaling (fp16)

    # final linear: [R, H] @ [dim, H].T → [R, dim]
    final_fp16 = F.linear(
        out_norm.view(R, H),
        weights["to_out.weight"].to(device=device, dtype=torch.float16)
    )   # fp16, no bias

    final = final_fp16.view(B, N, N, dim).to(torch.float32)   # float32 output
    return final
```
