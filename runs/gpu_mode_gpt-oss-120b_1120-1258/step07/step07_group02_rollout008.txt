"""
Outgoing TriMul (AlphaFold‑3) – Triton‑accelerated forward pass
================================================================
Algorithm (all heavy work in FP16):
1️⃣ Row‑wise LayerNorm on the input tensor (B·N·N rows × dim) → fp16.
2️⃣ Fuse the four projection / gate linear maps *and* the out‑gate linear map
   into a single GEMM:
      W5 = [W_left_proj;
            W_right_proj;
            W_left_gate;
            W_right_gate;
            W_out_gate]      # shape (5·H, dim)
   proj = x_norm @ W5.T                # → (R, 5·H)
3️⃣ Slice the result into left/right projections, left/right gates and out‑gate.
4️⃣ (Optional) apply pairwise mask on the two projection tensors.
5️⃣ Apply sigmoid to the three gate tensors and multiply:
      left  = left_proj  * sigmoid(left_gate)
      right = right_proj * sigmoid(right_gate)
6️⃣ Reshape left/right/out_gate to (B, N, N, H).
7️⃣ Cubic update without any data copy:
      out = torch.einsum('bikc,bjkc->bijc', left, right)   # (B,N,N,H)
8️⃣ Row‑wise LayerNorm over the hidden dimension H.
9️⃣ Multiply by the out‑gate (element‑wise) and a final linear projection
     back to dim:
        out = (LayerNorm(out) * sigmoid(out_gate)) @ W_to_out.T
10️⃣ Cast result to float32.

Key speed tricks
----------------
* The projection / gate linear maps are fused → one GEMM instead of four.
* The N³ update is performed with `torch.einsum`, which works on the
  original (B,N,N,H) layout and therefore avoids the expensive
  (B·H,N,N) copy required by the naïve permute→contiguous→bmm approach.
* All matrix multiplications use FP16 to exploit Tensor‑Cores on H100.
* Two custom Triton kernels implement per‑row LayerNorm (mean/var in FP32,
  output in FP16), keeping the kernel work lightweight.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton row‑wise LayerNorm kernel (input FP32 → output FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const  * [total_rows, dim] (float32)
    y_ptr,          # mutable* [total_rows, dim] (float16)
    w_ptr,          # const  * [dim]            (float32) – gamma
    b_ptr,          # const  * [dim]            (float32) – beta
    total_rows,     # int32
    dim,            # int32
    eps,            # float32
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # 1️⃣ Compute per‑row mean & variance (FP32)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------------------------------------------------------
    # 2️⃣ Write normalized values (FP16)
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)

        out = norm * gamma[None, :] + beta[None, :]
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float) – optional
        - weights_dict : model parameters (see required keys below)
        - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim        = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps        = 1e-5
    device     = input_tensor.device
    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N          # R = B·N·N
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → FP16
    # --------------------------------------------------------------
    x_flat = input_tensor.reshape(total_rows, dim).contiguous()   # [R, dim] FP32
    x_norm = torch.empty(total_rows, dim, dtype=torch.float16,
                         device=device)

    BLOCK_M = 256   # rows per program (tune if needed)
    BLOCK_D = 128   # columns per iteration (covers dim up to 384)
    grid = (triton.cdiv(total_rows, BLOCK_M),)

    layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],   # gamma (FP32)
        weights["norm.bias"],     # beta  (FP32)
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    # --------------------------------------------------------------
    # 2️⃣ Fused projection + gate (+ out‑gate) GEMM
    # --------------------------------------------------------------
    # Concatenate five linear weight matrices: left_proj, right_proj,
    # left_gate, right_gate, out_gate   → shape (5·H, dim)
    w_5h = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                # [5·H, dim]

    proj = F.linear(x_norm, w_5h)      # [R, 5·H]  FP16

    # slice the fused result
    left_proj_raw  = proj[:, 0:H]
    right_proj_raw = proj[:, H:2 * H]
    left_gate_raw  = proj[:, 2 * H:3 * H]
    right_gate_raw = proj[:, 3 * H:4 * H]
    out_gate_raw   = proj[:, 4 * H:5 * H]

    # --------------------------------------------------------------
    # 3️⃣ Optional mask, sigmoid gates and element‑wise gating
    # --------------------------------------------------------------
    if not config.get("nomask", True):
        # mask: [B,N,N] → [R,1] float16
        mask_fp16 = mask_tensor.reshape(total_rows, 1).to(torch.float16)
        left_proj_raw  = left_proj_raw * mask_fp16
        right_proj_raw = right_proj_raw * mask_fp16

    # in‑place sigmoid for speed
    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    # apply gates
    left  = left_proj_raw * left_gate          # [R, H] FP16
    right = right_proj_raw * right_gate        # [R, H] FP16

    # --------------------------------------------------------------
    # 4️⃣ reshape to (B, N, N, H)
    # --------------------------------------------------------------
    left      = left.view(B, N, N, H)
    right     = right.view(B, N, N, H)
    out_gate  = out_gate.view(B, N, N, H)

    # --------------------------------------------------------------
    # 5️⃣ cubic update via einsum, no extra copy
    # --------------------------------------------------------------
    # out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    out = torch.einsum('bikc,bjkc->bijc', left, right)   # [B,N,N,H] FP16

    # --------------------------------------------------------------
    # 6️⃣ second LayerNorm over hidden dimension H
    # --------------------------------------------------------------
    out_flat = out.reshape(total_rows, H).contiguous()
    out_norm = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm,
        weights["to_out_norm.weight"],   # gamma (FP32)
        weights["to_out_norm.bias"],     # beta  (FP32)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    out_norm = out_norm.view(B, N, N, H)

    # --------------------------------------------------------------
    # 7️⃣ apply out‑gate and final linear projection back to dim
    # --------------------------------------------------------------
    gated = out_norm * out_gate               # [B,N,N,H] FP16

    final = F.linear(
        gated.view(total_rows, H),            # [R, H]
        weights["to_out.weight"].to(torch.float16)   # (dim, H)
    )   # [R, dim] FP16

    return final.view(B, N, N, dim).to(torch.float32)
