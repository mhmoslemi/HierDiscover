"""Optimized forward pass for outgoing TriMul (AlphaFold3) using Triton.

Algorithm:
  1. Row‑wise LayerNorm on the input [B,N,N,dim] (float32→float16) via a
     Triton kernel.
  2. Fuse the four linear projections (left, right, left_gate, right_gate)
     into a single GEMM (fp16) using torch.nn.functional.linear.
  3. Separate linear for out_gate.
  4. Optional mask is applied to left/right before gating.
  5. Compute the N³ multiplicative update with a batched bmm:
        out[b,h,i,j] = Σ_k left[b,i,k,h] * right[b,j,k,h]
  6. Second row‑wise LayerNorm on the hidden dimension (fp16).
  7. Multiply by out_gate, final linear projection, cast back to float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def layernorm_kernel(
    x_ptr,   # const float32*   [R, D]
    y_ptr,   # mutable float16* [R, D]
    w_ptr,   # const float32*   [D]   (γ)
    b_ptr,   # const float32*   [D]   (β)
    total_rows,   # int
    dim,          # int
    eps,          # float (compile‑time)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per iteration
):
    """Row‑wise LayerNorm:
    y = ((x - μ) / sqrt(σ²+eps)) * γ + β
    All arithmetic is performed in fp32; the final result is stored as fp16.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)  # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ---- mean / variance accumulation ----
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        # Load tile [BLOCK_M, BLOCK_D]
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- write normalized values ----
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        # γ and β are fp32
        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)

        out = norm * w[None, :] + b[None, :]   # still fp32
        out_fp16 = out.to(tl.float16)

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data (tuple): (x, mask, weights, config)
            - x: torch.Tensor, shape [B, N, N, dim], dtype float32
            - mask: torch.Tensor, shape [B, N, N] (bool/float) or None
            - weights: dict with the following keys:
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight",
                "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config: dict with keys "dim", "hidden_dim", optionally "nomask"
    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    x, mask, weights, config = data
    device = x.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    B, N, _, _ = x.shape
    total_rows = B * N * N
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (fp32 → fp16), row‑wise over dim
    # --------------------------------------------------------------
    x_flat = x.view(total_rows, dim).contiguous()
    ln1_out = torch.empty(total_rows, dim,
                          dtype=torch.float16,
                          device=device)

    BLOCK_M = 256
    BLOCK_D = 64
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        ln1_out,
        weights["norm.weight"],         # γ (float32)
        weights["norm.bias"],           # β (float32)
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 2️⃣ Fuse left/right projections + their gates (4·H outputs)
    # --------------------------------------------------------------
    # Concatenate weight matrices: [4·H, dim]
    w_proj = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
        ],
        dim=0,
    ).to(device=device, dtype=torch.float16)   # fp16

    # fused GEMM: (R, dim) @ (4·H, dim).T → (R, 4·H)
    fused = F.linear(ln1_out, w_proj)          # fp16, bias=None

    left_raw      = fused[:, 0:H]            # (R, H)
    right_raw     = fused[:, H:2*H]          # (R, H)
    left_gate_raw = fused[:, 2*H:3*H]        # (R, H)
    right_gate_raw = fused[:, 3*H:4*H]       # (R, H)

    # out‑gate (separate GEMM)
    out_gate_raw = F.linear(
        ln1_out,
        weights["out_gate.weight"].to(device=device, dtype=torch.float16)   # [H, dim]
    )   # (R, H)

    # --------------------------------------------------------------
    # 3️⃣ Mask, sigmoid gates, and element‑wise weighting
    # --------------------------------------------------------------
    if not nomask and mask is not None:
        mask_flat = mask.reshape(total_rows, 1).to(device=device,
                                                   dtype=torch.float16)
        left_raw = left_raw * mask_flat
        right_raw = right_raw * mask_flat

    # In‑place sigmoid to minimise temporaries
    left_gate = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate = out_gate_raw.sigmoid_()

    left  = left_raw * left_gate      # (R, H)
    right = right_raw * right_gate    # (R, H)

    # --------------------------------------------------------------
    # 4️⃣ N³ multiplicative update via batched GEMM
    # --------------------------------------------------------------
    left  = left.view(B, N, N, H)
    right = right.view(B, N, N, H)
    out_gate = out_gate.view(B, N, N, H)

    # Move head dim into batch
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    left_mat  = left_perm.view(B * H, N, N)             # (B·H, N, N)
    right_mat = right_perm.view(B * H, N, N)            # (B·H, N, N)

    # B·H batched matmul: (i,k)·(j,k)ᵀ → sum_k
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B·H, N, N)

    # Reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm over hidden dimension
    # --------------------------------------------------------------
    out_flat = out.view(total_rows, H).contiguous()
    ln2_out = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        ln2_out,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 6️⃣ Apply output gate, final projection, and cast back to fp32
    # --------------------------------------------------------------
    out_norm = ln2_out.view(B, N, N, H)                # fp16
    out_norm = out_norm * out_gate                     # element‑wise fp16

    final_fp16 = F.linear(
        out_norm.view(total_rows, H),
        weights["to_out.weight"].to(device=device, dtype=torch.float16)   # [dim, H]
    )   # (R, dim), fp16

    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final

Algorithm
---------

2️⃣ Fuse the four linear projections (left, right, left_gate, right_gate) into a
   single fp16 GEMM.
3️⃣ Compute the output‑gate with a second fp16 GEMM.
4️⃣ Optional mask is applied to left/right before gating.
5️⃣ Perform the N³ multiplicative update with a batched matrix multiply:
      out[b,h,i,j] = Σ_k left[b,i,k,h] * right[b,j,k,h]
6️⃣ Second row‑wise LayerNorm over the hidden dimension (fp16) via Triton.
7️⃣ Multiply by the output gate, apply the final linear projection and cast

"""

    b_ptr,   # const float32*   [dim]   (β)
    total_rows,  # int (runtime)
    dim,         # int (runtime)
    eps,         # float (compile‑time constant)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per iteration
):
    """Row‑wise LayerNorm:
    y = ((x - μ) / sqrt(σ² + eps)) * γ + β
    Computation is performed in fp32, result stored in fp16.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)            # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # 1️⃣ Compute per‑row mean and variance
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)          # 1 / sqrt(var + eps)

    # ------------------------------------------------------------------
    # 2️⃣ Write normalized result γ·(x−μ)·rstd + β  (fp16)
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # γ (fp32)
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # β (fp32)

        out = norm * w[None, :] + b[None, :]                 # fp32
        out_fp16 = out.to(tl.float16)

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (x, mask, weights, config)
            - x: torch.Tensor, shape [B, N, N, dim], dtype float32
            - mask: torch.Tensor, shape [B, N, N] (bool/float) or None
            - weights: dict with the following keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight",
                "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config: dict with keys "dim", "hidden_dim", optionally "nomask"
    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    x, mask, weights, config = data
    device = x.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)      # True → ignore mask
    eps = 1e-5

    B, N, _, _ = x.shape
    total_rows = B * N * N                 # R = B·N·N
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (fp32 → fp16), row‑wise over dim
    # --------------------------------------------------------------
    x_flat = x.view(total_rows, dim).contiguous()  # [R, dim] fp32
    ln1_out = torch.empty(total_rows, dim,
                          dtype=torch.float16,
                          device=device)

    BLOCK_M = 256
    BLOCK_D = 64
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        ln1_out,
        weights["norm.weight"],   # γ (float32)
        weights["norm.bias"],     # β (float32)
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 2️⃣ Fuse left/right projections + their gates (4·H outputs)
    # --------------------------------------------------------------
    # Concatenate projection matrices: shape [4·H, dim]
    w_proj = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
        ],
        dim=0,
    ).to(device=device, dtype=torch.float16)   # fp16

    # Fused GEMM: (R, dim) @ (4·H, dim).T → (R, 4·H)
    fused = F.linear(ln1_out, w_proj)          # fp16, bias=None

    left_raw      = fused[:, 0:H]               # (R, H)
    right_raw     = fused[:, H:2*H]             # (R, H)
    left_gate_raw = fused[:, 2*H:3*H]           # (R, H)
    right_gate_raw = fused[:, 3*H:4*H]          # (R, H)

    # Output‑gate (separate GEMM)
    out_gate_raw = F.linear(
        ln1_out,
        weights["out_gate.weight"].to(device=device, dtype=torch.float16)   # [H, dim]
    )   # (R, H) fp16

    # --------------------------------------------------------------
    # 3️⃣ Apply mask (if any) and sigmoid gates
    # --------------------------------------------------------------
    if not nomask and mask is not None:
        mask_flat = mask.reshape(total_rows, 1).to(device=device,
                                                   dtype=torch.float16)
        left_raw  = left_raw * mask_flat
        right_raw = right_raw * mask_flat

    # In‑place sigmoid to reduce temporaries
    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    left  = left_raw * left_gate      # (R, H) fp16
    right = right_raw * right_gate    # (R, H) fp16

    # --------------------------------------------------------------
    # 4️⃣ N³ multiplicative update via batched GEMM
    # --------------------------------------------------------------
    left  = left.view(B, N, N, H)
    right = right.view(B, N, N, H)
    out_gate = out_gate.view(B, N, N, H)

    # Move head dimension into batch axis: (B, H, N, N)
    left_perm  = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    # Collapse batch & heads → (B·H, N, N)
    left_mat  = left_perm.view(B * H, N, N)
    right_mat = right_perm.view(B * H, N, N)

    # Batched matrix multiplication: sum over the k‑dimension
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B·H, N, N)

    # Restore original layout: (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm over the hidden dimension (fp16)
    # --------------------------------------------------------------
    out_flat = out.view(total_rows, H).contiguous()
    ln2_out = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        ln2_out,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 6️⃣ Apply output gate, final projection, and cast back to fp32
    # --------------------------------------------------------------
    out_norm = ln2_out.view(B, N, N, H)          # fp16
    out_norm = out_norm * out_gate               # element‑wise fp16

    final_fp16 = F.linear(
        out_norm.view(total_rows, H),
        weights["to_out.weight"].to(device=device, dtype=torch.float16)   # [dim, H]
    )   # (R, dim) fp16

    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final
