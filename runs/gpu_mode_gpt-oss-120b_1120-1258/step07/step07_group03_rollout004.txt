"""
TriMul (outgoing) forward pass – highly‑fused implementation.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the input tensor (B·N·N × dim) – custom Triton
   kernel that reduces in FP32 and writes FP16.

2️⃣ Fuse the four linear layers (left_proj, right_proj, left_gate,
   right_gate) into one GEMM:
       (R, dim) @ (4·H, dim)ᵀ → (R, 4·H)      where R = B·N·N.

3️⃣ Apply the optional mask *before* the sigmoid gates, compute the
   three sigmoid gates (in‑place) and obtain the gated tensors
   `left` and `right`.

4️⃣ Reshape to (B, N, N, H) and bring the hidden dimension into the
   batch axis → (B·H, N, N).  Perform the expensive N³ update with a
   single batched half‑precision GEMM:
       out = left @ rightᵀ   → (B·H, N, N).

5️⃣ Second LayerNorm over the hidden dimension (again Triton, FP32→FP16).

6️⃣ Multiply by the output‑gate, project back to `dim` with a final
   half‑precision linear layer and cast the result to FP32.

Only the two LayerNorms are written in Triton; all dense matmuls use
cuBLAS/F.linear which is optimal on H100 (Tensor‑Cores).  The kernel
works for both masked / unmasked inputs and for any batch size ≤ 2,
sequence length up to 1024 and hidden dimension 128.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (input FP32 → output FP16)
# ----------------------------------------------------------------------
@triton.jit
def rowwise_layernorm_kernel(
    x_ptr,        # const   * [total_rows, dim] (float32)
    y_ptr,        # mutable * [total_rows, dim] (float16)
    w_ptr,        # const   * [dim]            (float32)   γ
    b_ptr,        # const   * [dim]            (float32)   β
    total_rows,   # runtime int
    dim,          # runtime int
    eps,          # runtime float
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """Row‑wise LayerNorm:
        y = ((x - μ) / sqrt(σ² + eps)) * γ + β
    Reduction is done in FP32, output is FP16.
    """
    pid = tl.program_id(0)                     # one program per row tile
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)         # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # 1️⃣ Compute mean and variance per row (FP32 accumulator)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    num_d_tiles = tl.cdiv(dim, BLOCK_D)
    for d_tile in range(num_d_tiles):
        offs_d = d_tile * BLOCK_D + tl.arange(0, BLOCK_D)  # [BLOCK_D]
        mask_d = offs_d < dim

        # load tile [BLOCK_M, BLOCK_D] as FP32
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)          # sum over D dimension
        sum_sq  += tl.sum(vals * vals, axis=1)   # sum of squares

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                  # 1 / sqrt(var + eps)

    # ------------------------------------------------------------------
    # 2️⃣ Write normalized values (FP16)
    # ------------------------------------------------------------------
    for d_tile in range(num_d_tiles):
        offs_d = d_tile * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]   # (x-μ)/σ

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # β

        out = norm * gamma[None, :] + beta[None, :]               # FP32
        out_fp16 = out.to(tl.float16)

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16, mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor shape [B, N, N] (bool/float) – may be None
        - weights_dict : mapping of weight names to torch.Tensors
        - config_dict  : must contain "dim", "hidden_dim" and optional "nomask"

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    device = input_tensor.device

    B, N, _, _ = input_tensor.shape
    R = B * N * N                         # total rows for the first LN

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → FP16
    # --------------------------------------------------------------
    x_flat = input_tensor.view(R, dim).contiguous()   # [R, dim] FP32
    x_norm_half = torch.empty(R, dim,
                               dtype=torch.float16,
                               device=device)

    BLOCK_M = 256
    BLOCK_D = 32
    grid = (triton.cdiv(R, BLOCK_M),)

    rowwise_layernorm_kernel[grid](
        x_flat,
        x_norm_half,
        weights["norm.weight"],   # γ  (float32)
        weights["norm.bias"],     # β  (float32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    # --------------------------------------------------------------
    # 2️⃣ Fuse left/right projections + gate projections (single GEMM)
    # --------------------------------------------------------------
    H = hidden_dim
    # concatenate the four projection/gate weights: shape [4·H, dim]
    w_proj_gate = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                 # (4H, dim)

    # fused linear: (R, dim) @ (4H, dim)ᵀ → (R, 4H)   (no bias)
    proj_gate = F.linear(x_norm_half, w_proj_gate)     # FP16

    left_proj_raw   = proj_gate[:, 0:H]                # (R, H)
    right_proj_raw  = proj_gate[:, H:2 * H]            # (R, H)
    left_gate_raw   = proj_gate[:, 2 * H:3 * H]        # (R, H)
    right_gate_raw  = proj_gate[:, 3 * H:4 * H]        # (R, H)

    # output‑gate linear (separate weight) → (R, H)
    out_gate_raw = F.linear(
        x_norm_half,
        weights["out_gate.weight"].to(torch.float16)   # (H, dim)
    )   # no bias

    # --------------------------------------------------------------
    # 3️⃣ Optional mask, sigmoid gates and gating
    # --------------------------------------------------------------
    if not config.get("nomask", True):
        # mask shape [B,N,N] → [R,1] broadcast over hidden dim
        mask_h = mask_tensor.to(torch.float16).reshape(R, 1)   # (R,1)
        left_proj_raw  = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    # in‑place sigmoid (half‑precision)
    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    # gating
    left  = left_proj_raw * left_gate      # (R, H) fp16
    right = right_proj_raw * right_gate    # (R, H) fp16

    # --------------------------------------------------------------
    # 4️⃣ Reshape for the N³ multiplicative update
    # --------------------------------------------------------------
    left  = left.view(B, N, N, H)          # (B, N, N, H)
    right = right.view(B, N, N, H)         # (B, N, N, H)
    out_gate = out_gate.view(B, N, N, H)   # (B, N, N, H)

    # move hidden dimension to batch axis for a single batched GEMM
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    left_mat  = left_perm.view(B * H, N, N)      # (B·H, N, N)
    right_mat = right_perm.view(B * H, N, N)     # (B·H, N, N)

    # Batched half‑precision GEMM: (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # (B, N, N, H)

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm over hidden dimension → FP16
    # --------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()
    out_norm_half = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(R, BLOCK_M),)
    rowwise_layernorm_kernel[grid2](
        out_flat,
        out_norm_half,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    out_norm = out_norm_half.view(B, N, N, H)   # FP16

    # --------------------------------------------------------------
    # 6️⃣ Apply output gate and final projection back to `dim`
    # --------------------------------------------------------------
    out_norm = out_norm * out_gate               # (B,N,N,H) FP16

    final_half = F.linear(
        out_norm.view(R, H),                     # (R, H)
        weights["to_out.weight"].to(torch.float16)   # (dim, H)
    )   # → (R, dim) FP16

    final = final_half.view(B, N, N, dim).to(torch.float32)

    return final
