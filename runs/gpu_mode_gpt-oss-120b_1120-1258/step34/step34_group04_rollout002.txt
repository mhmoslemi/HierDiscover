"""
Outgoing TriMul (AlphaFold‑3) – forward‑only Triton implementation.

The operator works on a 4‑D pairwise tensor x ∈ ℝ^(B×N×N×C) and computes

    out[b,i,j] = Σₖ ( (x[b,i,k]·Wₗ)·σ(x[b,i,k]·Wₗg)·M[b,i,k] )
                 * ( (x[b,j,k]·Wᵣ)·σ(x[b,j,k]·W_rg)·M[b,j,k] )

followed by a hidden‑dimension LayerNorm, an output‑gate σ(x·W_og) and a final
linear projection back to the original channel dimension C.

Key ideas
---------
* The first LayerNorm (over the input channel dim) is implemented in a custom
  Triton kernel that reads FP32, computes mean/variance in FP32, and writes the
  normalized tensor in FP16.
* The five linear maps (left/right projection + three gates) are fused into a
  single GEMM (torch.nn.functional.linear) with a weight matrix of shape
  (5·H, C).  The result is then split, reshaped and the sigmoid gates are
  applied with standard torch ops.
* The heavy N³ contraction is performed with a batched GEMM (torch.bmm) on
  FP16 tensors – this is far faster than a naïve Python loop.
* The second LayerNorm (over the hidden dimension H) and the final projection
  are carried out with built‑in torch kernels; the final output is returned as
  FP32.

The implementation works for all configurations listed in the problem
statement (B∈{1,2}, N∈{128,…,1024}, C∈{128,384,768}, H=hidden_dim∈{128}).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣  LayerNorm over the flattened pairwise tokens (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp32_to_fp16_kernel(
    x_ptr,   # [R, D]   FP32 input (flattened pair tokens)
    y_ptr,   # [R, D]   FP16 output (normalized)
    w_ptr,   # [D]      γ (FP32)
    b_ptr,   # [D]      β (FP32)
    rows,    # scalar   R = B·N·N
    dim,     # scalar   D = C
    eps,     # scalar   ε
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # dim chunk per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)        # [BLOCK_M]
    mask_m = offs_m < rows

    # ---------- first pass: compute mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        # Load a tile of the input (FP32)
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- second pass: write normalized values ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * dim + offs_d[None, :],
            out.to(tl.float16),
            mask=mask_m[:, None] & mask_d[None, :],
        )


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only outgoing TriMul.

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor of shape [B, N, N, C] (float32)
            - mask_tensor  : torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights_dict : mapping with keys
                  "norm.weight", "norm.bias",
                  "left_proj.weight", "right_proj.weight",
                  "left_gate.weight", "right_gate.weight", "out_gate.weight",
                  "to_out_norm.weight", "to_out_norm.bias",
                  "to_out.weight"
              (all float32, already on the same device as inputs)
            - config_dict must contain:
                  "dim" (int), "hidden_dim" (int), "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = bool(cfg.get("nomask", True))
    eps = 1e-5
    device = x.device

    B, N, N2, C = x.shape
    assert N == N2 and C == dim, "input shape mismatch with config"

    R = B * N * N                     # total pairwise tokens
    H = hidden_dim                    # hidden dimension (same for all projections)

    # --------------------------------------------------------------
    # 1️⃣  LayerNorm over the input channel dimension (FP32 → FP16)
    # --------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                # [R, C] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128          # rows per kernel launch
    BLOCK_D = 64           # dim tiles
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_fp32_to_fp16_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat  # free memory

    # --------------------------------------------------------------
    # 2️⃣  Fuse all five linear maps into a single GEMM (FP16)
    # --------------------------------------------------------------
    fused_weight = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16).contiguous()      # shape (5·H, C)

    # (R, 5·H) = (R, C) @ (5·H, C).T   →  FP16 output
    proj = F.linear(x_norm, fused_weight, bias=None)   # [R, 5·H]
    del x_norm, fused_weight

    # --------------------------------------------------------------
    # 3️⃣  Split, apply mask, compute sigmoid gates
    # --------------------------------------------------------------
    left, right, left_gate_raw, right_gate_raw, out_gate_raw = torch.split(
        proj, H, dim=1
    )   # each [R, H]

    # reshape to [B, N, N, H]
    left = left.view(B, N, N, H)
    right = right.view(B, N, N, H)
    left_gate_raw = left_gate_raw.view(B, N, N, H)
    right_gate_raw = right_gate_raw.view(B, N, N, H)
    out_gate_raw = out_gate_raw.view(B, N, N, H)

    # optional mask (broadcasted across hidden dimension)
    if not nomask and mask is not None:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask_fp16
        right = right * mask_fp16
    # compute sigmoid gates (in FP16 – sufficient for inference)
    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate = torch.sigmoid(out_gate_raw)

    # apply gates
    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 4️⃣  Core N³ contraction via batched GEMM (left·rightᵀ)
    # --------------------------------------------------------------
    # reshape to [B, H, N, N] for the batched matmul
    left = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    left_mat = left.view(B * H, N, N)          # [B·H, N, N]  FP16
    right_mat = right.view(B * H, N, N)        # [B·H, N, N]  FP16

    # out_tmp[b·H, i, j] = Σₖ left[b·H, i, k] * right[b·H, j, k]
    out_tmp = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out_tmp.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]
    del left, right, left_mat, right_mat, out_tmp

    # --------------------------------------------------------------
    # 5️⃣  Second LayerNorm (over hidden dim H) + out‑gate multiplication
    # --------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        (H,),
        weight=w["to_out_norm.weight"],
        bias=w["to_out_norm.bias"],
        eps=eps,
    )   # still FP16

    gated = out_norm * out_gate   # [B, N, N, H]  FP16
    del out_norm, out_gate

    # --------------------------------------------------------------
    # 6️⃣  Final linear projection back to C (float32 output)
    # --------------------------------------------------------------
    w_out_fp16 = w["to_out.weight"].to(torch.float16).contiguous()  # [C, H]
    final_fp16 = F.linear(gated, w_out_fp16, bias=None)            # [B, N, N, C] FP16
    final = final_fp16.to(torch.float32)                           # cast to FP32

    return final
