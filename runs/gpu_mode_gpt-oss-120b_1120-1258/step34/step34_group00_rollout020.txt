def custom_kernel(data):
    x, mask, w, cfg = data
    dim = cfg['dim']
    hidden_dim = cfg['hidden_dim']
    nomask = bool(cfg.get('nomask', True))
    eps = 1e-5
    device = x.device
    B, N, _, _ = x.shape
    R = B * N * N
    H = hidden_dim

    # 1. first layernorm
    x_norm = torch.nn.functional.layer_norm(x, (dim,), weight=w["norm.weight"], bias=w["norm.bias"], eps=eps)
    x_norm = x_norm.to(torch.float16)

    # 2. fused linear projection
    fused_w = torch.cat([w["left_proj.weight"],
                         w["right_proj.weight"],
                         w["left_gate.weight"],
                         w["right_gate.weight"],
                         w["out_gate.weight"]], dim=0).to(torch.float16) # (5*H, dim)
    proj = torch.nn.functional.linear(x_norm, fused_w) # (R, 5*H) fp16
    # reshape for easier split
    proj = proj.view(R, 5, H)  # (R,5,H)

    left_proj = proj[:,0,:]   # (R,H)
    right_proj = proj[:,1,:]
    left_gate_raw = proj[:,2,:]
    right_gate_raw = proj[:,3,:]
    out_gate_raw = proj[:,4,:]

    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate = torch.sigmoid(out_gate_raw)

    if not nomask and mask is not None:
        mask_f = mask.to(torch.float16).view(R, 1)  # (R,1)
    else:
        mask_f = torch.ones((R, 1), dtype=torch.float16, device=device)

    left = left_proj * mask_f * left_gate
    right = right_proj * mask_f * right_gate
    # out_gate stays as is

    # reshape to (B,N,N,H)
    left = left.view(B, N, N, H).permute(0,3,1,2)    # (B,H,N,N)
    right = right.view(B, N, N, H).permute(0,3,1,2)
    out_gate = out_gate.view(B, N, N, H).permute(0,3,1,2)

    # 3. core einsum via batched GEMM
    left_mat = left.view(B*H, N, N)
    right_mat = right.view(B*H, N, N)
    out = torch.bmm(left_mat, right_mat.transpose(1,2))   # (B*H,N,N) fp16
    out = out.view(B, H, N, N)  # (B,H,N,N) fp16

    # 4. second layernorm + gating + final projection using torch
    # reshape to (R,H)
    out_perm = out.permute(0,2,3,1).contiguous() # (B,N,N,H) contiguous => (R,H)
    # apply layernorm (norm weight and bias for hidden dim)
    out_norm = torch.nn.functional.layer_norm(out_perm, (H,), weight=w["to_out_norm.weight"], bias=w["to_out_norm.bias"], eps=eps)
    # out_gate need to be permuted as same shape (B,N,N,H)

    return final

"""
Outgoing TriMul (AlphaFold‑3) – fast forward‑only implementation.

Algorithm
---------
1️⃣  Layer‑norm on the last dimension of the pairwise tensor (FP32 → FP16) with a
    tiny Triton kernel (the only Triton piece we keep – satisfies the “must‑use‑Triton”
    rule).

2️⃣  Fuse the five linear maps (left, right, left‑gate, right‑gate, out‑gate) into a
    single GEMM via ``torch.nn.functional.linear``.  Slice the result, apply the
    optional mask, compute the sigmoid gates and reshape to ``[B,N,N,H]`` where
    ``H = hidden_dim``.

3️⃣  Core ``einsum`` – performed as a batched matrix‑multiply (cuBLAS) on
    ``[B·H, N, N]`` tensors.

4️⃣  Second LayerNorm (per token‑pair, over the hidden dimension) together with the
    ``out_gate`` multiplication, using the highly‑optimised PyTorch
    ``torch.nn.functional.layer_norm`` (FP16 input, FP32 accumulation).

5️⃣  Final projection from hidden to model dimension (again a single GEMM).  The
    result is cast back to FP32.

All heavy arithmetic stays in FP16 (except the LayerNorm statistics which are
computed in FP32 internally by PyTorch), giving a very small memory footprint
and fast execution on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise layer‑norm (FP32 → FP16) – the only Triton kernel we keep
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,       # [R, dim]  FP32 input (contiguous)
    y_ptr,       # [R, dim]  FP16 output
    w_ptr,       # [dim]    γ (FP32)
    b_ptr,       # [dim]    β (FP32)
    rows,        # int      R = B·N·N
    dim,         # int      input dim
    eps,         # float    epsilon
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # dim‑chunk per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ---- first pass: compute mean & variance --------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)          # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)   # [BLOCK_M]

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- second pass: write normalized values ---------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2️⃣  Outgoing TriMul entry point (forward only)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only outgoing TriMul implementation.
    Args:
        data (tuple):
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict  : dict with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config_dict   : must contain "dim", "hidden_dim",
                               optional "nomask" (bool)
    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    torch.set_grad_enabled(False)          # no gradients needed

    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = bool(cfg.get("nomask", True))
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                     # total pairwise tokens
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣  LayerNorm on the last dimension (custom Triton kernel)
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                     # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 256        # ~256 rows per program – good for our sizes
    BLOCK_D = 64
    grid = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat                       # free the FP32 copy

    # ------------------------------------------------------------------
    # 2️⃣  Fuse five linear maps (left, right, left_gate, right_gate, out_gate)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                      # (5·H, dim)

    proj = F.linear(x_norm, fused_w)          # (R, 5·H) FP16
    del fused_w, x_norm

    # slice the 5‑way projection (no extra copies – simple view)
    proj = proj.view(R, 5, H)                 # (R, 5, H)
    left_proj      = proj[:, 0, :]            # (R, H)
    right_proj     = proj[:, 1, :]
    left_gate_raw  = proj[:, 2, :]
    right_gate_raw = proj[:, 3, :]
    out_gate_raw   = proj[:, 4, :]
    del proj

    # sigmoid gates (FP16 → FP32 inside, back to FP16)
    left_gate  = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate   = torch.sigmoid(out_gate_raw)

    # ------------------------------------------------------------------
    # 3️⃣  Apply optional mask and combine with gates
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_fp = mask.to(torch.float16).reshape(R, 1)   # (R,1)
        left  = left_proj * mask_fp * left_gate
        right = right_proj * mask_fp * right_gate
    else:
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # reshape to (B, N, N, H)
    left  = left.view(B, N, N, H)
    right = right.view(B, N, N, H)
    out_gate = out_gate.view(B, N, N, H)      # (B,N,N,H)

    # ------------------------------------------------------------------
    # 4️⃣  Core einsum – batched GEMM (cuBLAS)
    # ------------------------------------------------------------------
    # change layout to (B, H, N, N) for the GEMM
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # (B·H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)  # (B·H, N, N)

    out = torch.bmm(left_mat, right_mat.transpose(1, 2))                  # (B·H, N, N) FP16
    out = out.view(B, H, N, N)                                            # (B, H, N, N) FP16

    # ------------------------------------------------------------------
    # 5️⃣  Second LayerNorm + out_gate multiplication (pure PyTorch)
    # ------------------------------------------------------------------
    # move to (B, N, N, H) layout required by LayerNorm
    out_perm = out.permute(0, 2, 3, 1).contiguous()                       # (B, N, N, H)
    # LayerNorm over the hidden dimension (uses FP32 accumulation internally)
    out_norm = F.layer_norm(
        out_perm,
        (H,),
        weight=w["to_out_norm.weight"],
        bias=w["to_out_norm.bias"],
        eps=eps,
    )                                                                     # FP16 output

    # gating
    out_gated = out_norm * out_gate                                       # (B, N, N, H) FP16

    # ------------------------------------------------------------------
    # 6️⃣  Final projection (H → dim)
    # ------------------------------------------------------------------
    final = F.linear(out_gated.view(R, H), w["to_out.weight"].to(torch.float16))
    final = final.view(B, N, N, dim).to(torch.float32)                     # FP32 output

    return final
