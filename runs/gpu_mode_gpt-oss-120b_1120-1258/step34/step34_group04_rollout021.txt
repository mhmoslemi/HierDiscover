```python
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# split_fuse_gate kernel (unchanged but we can tweak block sizes)
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,      # [R, 5*H]
    left_ptr,      # [B, H, N, N]
    right_ptr,     # [B, H, N, N]
    out_gate_ptr,  # [B, H, N, N]
    mask_ptr,      # [R] (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)           # [BLOCK_R]
    mask_r = offs_r < R

    # decode r = b*N*N + i*N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=tl.float16(1.0))
        m_val = m_val[:, None]  # broadcast over hidden dim
    else:
        m_val = tl.full([BLOCK_R, 1], tl.float16(1.0), dtype=tl.float16)

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + k   # start offset for hidden dim 0

    # process hidden dim in blocks
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # column offset inside proj: each hidden dim occupies a column
        # proj layout: [R, 5*H]; stride in hidden dim is 1
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(proj_ptr + proj_base,
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=tl.float16(0.0))
        right_raw = tl.load(proj_ptr + proj_base + H,
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=tl.float16(0.0))
        left_gate_raw = tl.load(proj_ptr + proj_base + 2 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=tl.float16(0.0))
        right_gate_raw = tl.load(proj_ptr + proj_base + 3 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=tl.float16(0.0))
        out_gate_raw = tl.load(proj_ptr + proj_base + 4 * H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=tl.float16(0.0))

        # sigmoid in fp32 for numeric stability
        left_gate = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left = left_raw * m_val * left_gate
        right = right_raw * m_val * right_gate

        dst = base[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]
        tl.store(left_ptr + dst, left, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst, right, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst, out_gate, mask=mask_r[:, None] & mask_h[None, :])

def custom_kernel(data):
    """
    Outgoing TriMul forward‑only implementation.

    1️⃣ Layer‑norm over the input channel dimension (FP32 → FP16).
    2️⃣ Fuse the five linear maps (left, right, left‑gate, right‑gate, out‑gate) into a single GEMM.
    3️⃣ Triton kernel splits the fused projection, applies the optional mask and Sigmoid gates,
       writing three tensors ``left``, ``right`` and ``out_gate`` in layout ``[B, H, N, N]``.
    4️⃣ Core contraction is a batched GEMM (torch.bmm) corresponding to the einsum
       ``... i k d, ... j k d → ... i j d``.
    5️⃣ Second LayerNorm, multiplication with ``out_gate`` and the final projection are performed
       with native PyTorch kernels (torch.nn.functional.layer_norm + linear) which are highly
       optimized on H100.
    """
    # ------------------- unpack -------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = bool(cfg.get("nomask", True))
    eps = 1e-5
    device = x.device
    B, N, _, _ = x.shape
    R = B * N * N
    H = hidden_dim

    # ------------------- 1️⃣ Input LayerNorm (FP32→FP16) -------------------
    # torch.nn.functional.layer_norm is already a fast CUDA kernel.
    x_norm_fp32 = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim] float32
    x_norm = x_norm_fp32.to(torch.float16)   # cast for the fused GEMM

    # ------------------- 2️⃣ fused projection (5·H) -------------------
    fused_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)          # (5·H, dim)

    x_flat = x_norm.view(R, dim)  # [R, dim] contiguous FP16
    proj = F.linear(x_flat, fused_w)  # [R, 5·H] FP16
    del fused_w, x_norm, x_flat

    # ------------------- 3️⃣ split, mask, sigmoid → left/right/out_gate -------------------
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_fp16 = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_fp16
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------- 4️⃣ core contraction (einsum) -------------------
    left_mat = left.view(B * H, N, N)      # [B·H, N, N]
    right_mat = right.view(B * H, N, N)    # [B·H, N, N]
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [B·H, N, N] FP16
    out = out.view(B, H, N, N)           # [B, H, N, N] FP16

    # ------------------- 5️⃣ second LayerNorm + out_gate -------------------
    # Perform LayerNorm over the hidden dimension (H) per (b,i,j) token.
    out_fp32 = out.float()
    out_flat = out_fp32.view(R, H)        # [R, H]
    out_ln = F.layer_norm(
        out_flat,
        normalized_shape=(H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )                                     # [R, H] float32
    out_ln = out_ln.view(B, H, N, N)      # [B, H, N, N] float32

    # Multiply by output gate (broadcast over i/j)
    gated = out_ln * out_gate.float()     # [B, H, N, N] float32

    # ------------------- 6️⃣ final linear projection -------------------
    gated_flat = gated.view(R, H)         # [R, H] float32
    final = F.linear(gated_flat, weights["to_out.weight"])  # [R, dim] float32
    final = final.view(B, N, N, dim)     # [B, N, N, dim]

    return final

"""
Outgoing TriMul (AlphaFold‑3) – forward‑only implementation.

Algorithm
---------
1️⃣  Layer‑norm over the input channel dimension (float32 → float16).
2️⃣  Fuse the five linear maps (left, right, left‑gate, right‑gate, out‑gate)
    into a single GEMM (cuBLAS).  The resulting tensor has shape
    [R, 5·H] where R = B·N·N and H = hidden_dim.
3️⃣  Triton kernel `split_fuse_gate_kernel` splits the fused projection,
    applies the optional pair‑mask, computes the sigmoid gates and writes
    three tensors `left`, `right` and `out_gate` in layout `[B, H, N, N]`
    (float16).
4️⃣  Core contraction is a batched matrix‑multiply (torch.bmm) which
    implements the einsum
        out[b,i,j] = Σₖ left[b,i,k]·right[b,j,k] .
5️⃣  Second LayerNorm, multiplication with `out_gate` and the final linear
    projection are performed with highly‑optimised PyTorch kernels
    (`torch.nn.functional.layer_norm` + `torch.nn.functional.linear`).
The only custom Triton code required is the split/fuse/gate kernel,
which gives the biggest performance win on H100 GPUs.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Kernel 3️⃣ – split fused projection, apply mask, compute sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,      # [R, 5*H]   FP16 fused projection
    left_ptr,      # [B, H, N, N]  FP16
    right_ptr,     # [B, H, N, N]  FP16
    out_gate_ptr,  # [B, H, N, N]  FP16
    mask_ptr,      # [R]            FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)           # [BLOCK_R]
    mask_r = offs_r < R

    # ------------ decode flat index r = b·N·N + i·N + k ------------
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # ------------ optional per‑pair mask (broadcast across hidden dim) ------------
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=tl.float16(1.0))
        m_val = m_val[:, None]               # [BLOCK_R, 1]
    else:
        m_val = tl.full([BLOCK_R, 1], tl.float16(1.0), dtype=tl.float16)

    # ------------ layout helpers for [B, H, N, N] ----------------
    hidden_stride = N * N                     # stride between hidden planes
    base = b * (H * hidden_stride) + i * N + k   # start offset for hidden plane 0

    # ------------ iterate over hidden dimension in blocks ------------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # columns inside the fused projection (5·H columns, each hidden dim is contiguous)
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=tl.float16(0.0),
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=tl.float16(0.0),
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=tl.float16(0.0),
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=tl.float16(0.0),
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=tl.float16(0.0),
        )

        # Sigmoid in FP32 → back to FP16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # Apply mask and gates
        left  = left_raw  * m_val * left_gate
        right = right_raw * m_val * right_gate

        # Offsets into the destination tensors
        dst = base[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst, right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst, out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Entry point – forward pass of the outgoing TriMul
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only outgoing TriMul.

    Args:
        data (tuple):
            input_tensor : torch.Tensor [B, N, N, dim]   (float32)
            mask_tensor  : torch.Tensor [B, N, N]       (bool/float) or None
            weights_dict : model weight mapping containing
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"               (all float32)
            config_dict  : {"dim": int, "hidden_dim": int, "nomask": bool}
    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    # --------------------- unpack ---------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = bool(cfg.get("nomask", True))
    eps = 1e-5
    device = x.device
    B, N, _, _ = x.shape
    R = B * N * N
    H = hidden_dim

    # --------------------- 1️⃣ Input LayerNorm (FP32 → FP16) ---------------------
    # Fast native LayerNorm kernel
    x_norm_fp32 = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    )                                  # [B, N, N, dim]  float32
    x_norm = x_norm_fp32.to(torch.float16)  # cast to FP16 for the big GEMM

    # --------------------- 2️⃣ Fused projection (5·H) ---------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                 # (5·H, dim)

    x_flat = x_norm.view(R, dim)        # [R, dim] contiguous FP16
    proj = F.linear(x_flat, fused_w)    # [R, 5·H] FP16
    del fused_w, x_norm, x_flat

    # --------------------- 3️⃣ split, mask, sigmoid ---------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_fp16 = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_fp16
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy

    BLOCK_R = 64   # rows processed per program
    BLOCK_H = 128  # whole hidden dimension fits in a single block for H=128
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # --------------------- 4️⃣ core contraction (einsum) ---------------------
    left_mat  = left.view(B * H, N, N)      # [B·H, N, N] FP16
    right_mat = right.view(B * H, N, N)     # [B·H, N, N] FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)               # [B, H, N, N] FP16

    # --------------------- 5️⃣ second LayerNorm + out_gate ---------------------
    out_fp32 = out.float()
    out_flat = out_fp32.view(R, H)            # [R, H]

    out_ln = F.layer_norm(
        out_flat,
        normalized_shape=(H,),
        weight=w["to_out_norm.weight"],
        bias=w["to_out_norm.bias"],
        eps=eps,
    )                                         # [R, H]  float32
    out_ln = out_ln.view(B, H, N, N)          # [B, H, N, N] float32

    gated = out_ln * out_gate.float()         # element‑wise gate, still float32

    # --------------------- 6️⃣ final linear projection ---------------------
    gated_flat = gated.view(R, H)             # [R, H] float32
    final = F.linear(gated_flat, w["to_out.weight"])  # [R, dim] float32
    final = final.view(B, N, N, dim)          # [B, N, N, dim]

    return final
