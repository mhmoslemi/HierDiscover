"""
TriMul (outgoing) – fast forward‑only implementation using Triton.

The operator works on a 4‑D tensor x ∈ ℝ^{B×N×N×C} and returns a tensor
of the same shape.  The algorithm follows the AlphaFold‑3 description:

1️⃣  Row‑wise LayerNorm on the channel dimension C (FP32 → FP16).
2️⃣  Fuse the five linear projections (left, right, left_gate, right_gate,
    out_gate) into one matmul:   x_norm @ W_fusedᵀ.
3️⃣  Split the fused result, apply the optional mask, compute the sigmoid
    gates and store the three tensors left, right, out_gate (FP16).
4️⃣  Batched GEMM over the sequence dimension N:
        out[b,i,j,:] = Σ_k left[b,i,k,:] * right[b,j,k,:]      (FP16)
5️⃣  Compute per‑row mean and rsqrt‑variance of `out` across the hidden
    dimension H (Triton kernel).
6️⃣  Fuse the second LayerNorm, the element‑wise multiplication by the
    out‑gate and the final linear projection back to C in a single Triton
    kernel.

All kernels use mixed‑precision (FP16 arithmetic, FP32 accumulation) and
run entirely on‑device, yielding a float‑32 output that matches the
reference PyTorch implementation up to rounding error.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,        # [R, C]  FP32 input
    y_ptr,        # [R, C]  FP16 output
    w_ptr,        # [C]    γ (FP32)
    b_ptr,        # [C]    β (FP32)
    rows,         # R = B·N·N
    dim,          # C
    eps,          # epsilon
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ---- first pass: accumulate sum and sum of squares -------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        # load a tile of (BLOCK_M, BLOCK_D) values
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                                 # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                   # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)            # [BLOCK_M]

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- second pass: write normalized values ----------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                                 # [BLOCK_M, BLOCK_D]

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]               # FP32
        tl.store(
            y_ptr + offs_m[:, None] * dim + offs_d[None, :],
            out.to(tl.float16),
            mask=mask_m[:, None] & mask_d[None, :],
        )

# 2️⃣ Fuse 5 projections, apply mask and sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,      # [R, 5·H] FP16 fused projection (output of F.linear)
    left_ptr,      # [B, H, N, N] FP16
    right_ptr,     # [B, H, N, N] FP16
    out_gate_ptr,  # [B, H, N, N] FP16
    mask_ptr,      # [R] FP16 (optional, broadcast over H)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index r = b·N·N + i·N + k
    NN = N * N
    b = offs_r // NN
    rem = offs_r - b * NN
    i = rem // N
    k = rem - i * N

    # optional element‑wise mask (broadcast over hidden dimension)
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # FP16
        mask_val = m_val[:, None]                                   # broadcast
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N                     # stride of one hidden plane
    base_offset = b * (H * hidden_stride) + i * N + k   # start offset for hidden=0

    # Loop over hidden dimension in blocks
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # layout: [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]    # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # sigmoid (FP32 for stability) → cast back to FP16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate

        # destination offsets for [B, H, N, N] layout
        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst_offset,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset,
                 outg,
                 mask=mask_r[:, None] & mask_h[None, :])

# 5️⃣ Row‑wise stats of the contracted tensor (mean & rsqrt‑var)
# ----------------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr,    # [B, H, N, N] FP16
    mu_ptr,     # [R] FP32
    rstd_ptr,   # [R] FP32
    total_rows, H, N,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    NN = N * N
    b = offs_m // NN
    rem = offs_m - b * NN
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + j      # start of hidden plane 0

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)   # [BLOCK_K]
        mask_k = offs_k < H

        ptr = out_ptr + base[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)                         # [BLOCK_M, BLOCK_K]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_m, mean, mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)


# ----------------------------------------------------------------------
# 6️⃣ Fuse 2nd LayerNorm, out_gate multiplication and final projection
# ----------------------------------------------------------------------
@triton.jit
def final_output_kernel(
    out_ptr,        # [B, H, N, N] FP16 (post‑GEMM)
    gate_ptr,       # [B, H, N, N] FP16 (out_gate)
    mu_ptr,         # [R] FP32  (row means)
    rstd_ptr,       # [R] FP32  (row rsqrt‑var)
    ln_w_ptr,       # [H] FP32   (to_out_norm.weight)
    ln_b_ptr,       # [H] FP32   (to_out_norm.bias)
    w_out_T_ptr,    # [H, dim] FP16 (to_out weight transposed)
    out_final_ptr,  # [B, N, N, dim] FP32
    total_rows, H, N, dim,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr,
):
    pid_m = tl.program_id(0)   # tiles over rows (R)
    pid_n = tl.program_id(1)   # tiles over output dimensions

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)   # [M]
    offs_n = col_start + tl.arange(0, BLOCK_N)   # [N]

    mask_m = offs_m < total_rows
    mask_n = offs_n < dim

    NN = N * N
    b = offs_m // NN
    rem = offs_m - b * NN
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + j   # start offset of hidden‑plane 0

    # Load per‑row statistics (μ, rstd) – FP32
    mu   = tl.load(mu_ptr   + offs_m, mask=mask_m, other=0.0)   # [M]
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)   # [M]

    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    # Loop over hidden dimension in tiles of BLOCK_K
    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)               # [K]
        mask_k = offs_k < H

        # Load hidden activations and out‑gate values (FP16 → FP32)
        ptr_out  = out_ptr  + base[:, None] + offs_k[None, :] * hidden_stride
        ptr_gate = gate_ptr + base[:, None] + offs_k[None, :] * hidden_stride

        out_val  = tl.load(
            ptr_out,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)                     # [M, K]
        gate_val = tl.load(
            ptr_gate,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)                     # [M, K]

        # 2nd LayerNorm parameters for this hidden slice (FP32)
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [K]
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [K]

        # LayerNorm → * out_gate → cast back to FP16 for the dot
        norm = ((out_val - mu[:, None]) * rstd[:, None]) * gamma[None, :] + beta[None, :]
        term = norm * gate_val                                 # [M, K]
        term_fp16 = term.to(tl.float16)

        # Load a tile of the final projection weight (H x dim) – FP16
        w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]   # [K, TILE_N]
        w_tile = tl.load(
            w_ptr,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )                                                            # [K, TILE_N]

        # Accumulate (M,K)·(K,N) → (M,N)   (FP16·FP16 → FP32)
        acc += tl.dot(term_fp16, w_tile).to(tl.float32)

    # Write final output (row‑major layout [B, N, N, dim])
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(
        dst_ptr,
        acc,
        mask=mask_m[:, None] & mask_n[None, :],
    )

    -------

    """

    # Unpack inputs
    # ------------------------------------------------------------------

    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    assert N == N2, "TriMul expects a square pairwise matrix"
    R = B * N * N                # number of (b,i,k) rows
    H = hidden_dim

    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                      # [R, C] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_D = 64
    layernorm_kernel[(triton.cdiv(R, BLOCK_M),)](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # 2️⃣ Fuse the five linear projections (output 5·H)
    # ------------------------------------------------------------------

            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)       # shape [5·H, C]

    # ------------------------------------------------------------------

    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

        HAS_MASK = 1
        mask_ptr = mask_flat

        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_H = 64
    split_fuse_gate_kernel[(triton.cdiv(R, BLOCK_R),)](

        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,

    # 4️⃣ Batched GEMM over the hidden dimension (cuBLAS)
    # ------------------------------------------------------------------
    # torch.bmm expects shape [batch, M, K]; we reshape to [B·H, N, N]
    left_mat = left.view(B * H, N, N)          # FP16
    right_mat = right.view(B * H, N, N)        # FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                # [B, H, N, N] FP16

    # ------------------------------------------------------------------
    # 5️⃣ Row‑wise mean & rsqrt‑variance of `out`
    # ------------------------------------------------------------------
    mu = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 256
    BLOCK_K_STATS = 128                 # H ≤ 128 in all test configs
    rowwise_stats_kernel[(triton.cdiv(R, BLOCK_M_STATS),)](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Final LayerNorm, out_gate multiplication and projection
    # ------------------------------------------------------------------
    # to_out.weight : [C, H] → we need its transpose [H, C]
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim] FP16

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 128      # rows per program (tiles over R)
    BLOCK_K_OUT = 32       # hidden‑dim tile (tuned for H=128)
    BLOCK_N_OUT = 128      # output‑dim tile

    grid_out = (
        triton.cdiv(R, BLOCK_M_OUT),
        triton.cdiv(dim, BLOCK_N_OUT),
    )
    final_output_kernel[grid_out](
        out,
        out_gate,
        mu,
        rstd,
        w["to_out_norm.weight"],    # γ (FP32)
        w["to_out_norm.bias"],      # β (FP32)
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=8,
    )
    return final
```
