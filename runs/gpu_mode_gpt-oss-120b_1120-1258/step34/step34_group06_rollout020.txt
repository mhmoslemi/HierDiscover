    return final

"""
Outgoing TriMul (TriMul‑out) – Triton‑accelerated forward pass (inference only)

The operator works on a 4‑D tensor `x` of shape [B, N, N, D] and produces a
tensor of the same shape.  The computation consists of:

1️⃣ Row‑wise LayerNorm over the last dimension (float32 → float16) – done in
   a custom Triton kernel.

2️⃣ A fused linear projection that returns the concatenation of five
   hidden‑dim linear maps (left, right, left‑gate, right‑gate, out‑gate).

3️⃣ Split the fused projection, apply the optional mask, apply sigmoid gates
   and store `left`, `right` and `out_gate` in FP16.

4️⃣ Batched matrix multiplication (torch.bmm) evaluates the core
   einsum:
        out[b,i,j] = Σₖ left[b,i,k] * right[b,j,k]
   producing a tensor of shape [B, H, N, N] (H = hidden_dim, FP16).

5️⃣ Row‑wise LayerNorm over the hidden dimension, multiplication by the
   `out_gate`, and a final linear projection back to the original dimension D.
   All of these steps are performed with PyTorch (float32) for simplicity,
   while the critical LayerNorm is still executed in Triton.

All operations are performed without gradients (inference only) and use mixed‑precision
arithmetic for maximal throughput on an H100 GPU.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# --------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # [R, D]   FP32 input   (flattened tensor)
    y_ptr,          # [R, D]   FP16 output  (normalized tensor)
    w_ptr,          # [D]      LN weight γ  (FP32)
    b_ptr,          # [D]      LN bias β   (FP32)
    rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ---------- first pass: compute mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)           # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                   # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)            # [BLOCK_M]

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- second pass: write normalized values ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])

# --------------------------------------------------------------
# Triton entry point
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor of shape [B, N, N, D] (float32)
            - mask_tensor  : torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights_dict : mapping containing all model weights (float32)
            - config_dict  : must contain keys "dim", "hidden_dim", "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, D] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim         = cfg["dim"]
    hidden_dim  = cfg["hidden_dim"]
    nomask      = cfg.get("nomask", True)
    eps         = 1e-5
    device      = x.device

    B, N, _, _ = x.shape               # x: [B, N, N, D]
    R = B * N * N                       # total pairwise positions
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (float32 → float16)
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                       # [R, D] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear projection (5·hidden_dim)
    # ------------------------------------------------------------------
    fused_weight = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # [5·H, D]
    proj = F.linear(x_norm, fused_weight)   # (R, 5·H)  FP16
    del x_norm, fused_weight

    # ------------------------------------------------------------------
    # 3️⃣ Split, apply mask, sigmoid gates (pure PyTorch for speed)
    # ------------------------------------------------------------------
    # proj -> (R, 5, H)
    proj = proj.view(R, 5, H)

    left_raw      = proj[:, 0, :]   # (R, H)
    right_raw     = proj[:, 1, :]   # (R, H)
    left_gate_raw = proj[:, 2, :]
    right_gate_raw= proj[:, 3, :]
    out_gate_raw  = proj[:, 4, :]   # (R, H)

    # sigmoid gates (computed in fp32 for numerical stability)
    left_gate  = torch.sigmoid(left_gate_raw.to(torch.float32)).to(torch.float16)
    right_gate = torch.sigmoid(right_gate_raw.to(torch.float32)).to(torch.float16)
    out_gate   = torch.sigmoid(out_gate_raw.to(torch.float32)).to(torch.float16)

    if not nomask and mask is not None:
        mask_flat = mask.reshape(R).to(torch.float16)          # (R,)
        mask_exp  = mask_flat[:, None]                        # (R,1)
        left  = left_raw  * mask_exp * left_gate
        right = right_raw * mask_exp * right_gate
    else:
        left  = left_raw  * left_gate
        right = right_raw * right_gate

    # reshape to [B, H, N, N] (contiguous layout)
    left  = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    out_gate = out_gate.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()

    del proj, left_raw, right_raw, left_gate_raw, right_gate_raw, out_gate_raw
    del left_gate, right_gate, mask_flat if not nomask else None

    # ------------------------------------------------------------------
    # 4️⃣ Batched matrix multiplication (the core einsum)
    # ------------------------------------------------------------------
    # reshape to (B·H, N, N) for cuBLAS batched GEMM
    left_mat  = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)

    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B·H, N, N) FP16
    out = out.view(B, H, N, N)                             # (B, H, N, N) FP16

    del left, right

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm over hidden dim + out_gate + final projection
    # ------------------------------------------------------------------
    # bring hidden dimension to last axis -> shape (R, H)
    out_flat = out.permute(0, 2, 3, 1).contiguous().view(R, H)          # FP16
    out_gate_flat = out_gate.permute(0, 2, 3, 1).contiguous().view(R, H)  # FP16

    # compute per‑row mean / variance in FP32
    out_fp32 = out_flat.to(torch.float32)
    mu   = out_fp32.mean(dim=1, keepdim=True)                # (R,1)
    var  = out_fp32.var(dim=1, unbiased=False, keepdim=True) # (R,1)
    rstd = torch.rsqrt(var + eps)                            # (R,1)

    # LayerNorm over hidden dimension (γ,β from to_out_norm)
    gamma = w["to_out_norm.weight"].view(1, H)   # (1, H) FP32
    beta  = w["to_out_norm.bias"].view(1, H)     # (1, H) FP32

    norm = ((out_fp32 - mu) * rstd) * gamma + beta    # (R, H) FP32

    # apply out‑gate (cast gate to FP32)
    norm = norm * out_gate_flat.to(torch.float32)      # (R, H) FP32

    # final linear projection back to original dimension D
    # use the transposed weight (H, D) in FP16 – torch will up‑cast it automatically
    w_out = w["to_out.weight"].to(torch.float16)      # (D, H)
    final = F.linear(norm, w_out)                     # (R, D) FP32

    final = final.view(B, N, N, dim)                  # (B, N, N, D)

    return final
```
