"""
TriMul (outgoing) – Triton‑accelerated forward pass
==================================================
The implementation follows the PyTorch reference but moves the
memory‑bound parts to Triton:

1️⃣  Row‑wise LayerNorm on the flattened [B·N·N, C] input (FP32 → FP16)
2️⃣  A *single* linear projection that concatenates the five weight matrices
    (left, right, left_gate, right_gate, out_gate) → (R, 5·H).
3️⃣  Triton kernel `split_fuse_gate_kernel`:
    * splits the fused projection,
    * applies the optional pair‑wise mask,
    * applies sigmoid gates,
    * writes `left`, `right` and `out_gate` as FP16 tensors
      `[B, H, N, N]`.
4️⃣  Batched GEMM (`torch.bmm`) contracts the two projected tensors:
    `out[b, h, i, j] = Σₖ left[b, h, i, k]·right[b, h, j, k]`.
5️⃣  **Fused kernel** `final_fused_norm_proj_kernel`:
    * per‑row mean / variance of `out`,
    * the second LayerNorm (`to_out_norm`),
    * multiplication by `out_gate`,
    * final linear projection (`to_out`).

   The kernel computes for each row `r=(b,i,j)` the three
   intermediate reductions  
   `A = Σₖ outₖ·γₖ·gateₖ·Wₖd`,
   `B = Σₖ γₖ·gateₖ·Wₖd`,
   `C = Σₖ βₖ·gateₖ·Wₖd`,
   together with `μ = mean(outₖ)` and `σ⁻¹ = rsqrt(var+eps)`.
   The final output is  
   `Y = σ⁻¹·(A – μ·B) + C`   (all operations in FP32, stores FP32).

All kernels are written for Triton 3.3.1 (targeting an H100) and use
mixed‑precision (FP16 arithmetic, FP32 accumulation).  The forward
pass is gradient‑free.

"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr, y_ptr, w_ptr, b_ptr,
    rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # -------- first pass: sum & sum‑of‑squares ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # -------- second pass: write normalized values ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, apply optional mask, sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,      # [R, 5·H]  fused projection (FP16)
    left_ptr,      # [B, H, N, N]  FP16
    right_ptr,     # [B, H, N, N]  FP16
    out_gate_ptr,  # [B, H, N, N]  FP16
    mask_ptr,      # [R] (optional) FP16
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # Decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask broadcasted over hidden dimension
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                   # [BLOCK_R, 1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # base offset inside the 4‑D tensors [B, H, N, N]
    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + k               # start of plane h=0

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # Offsets inside the fused projection:
        # 5 groups of size H each
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]        # [BLOCK_R, BLOCK_H]

        left_raw      = tl.load(proj_ptr + proj_base,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_raw     = tl.load(proj_ptr + proj_base + H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        left_gate_raw = tl.load(proj_ptr + proj_base + 2 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_gate_raw= tl.load(proj_ptr + proj_base + 3 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        out_gate_raw  = tl.load(proj_ptr + proj_base + 4 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)

        # sigmoid (computed in FP32 for stability)
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate

        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst_offset,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset,
                 outg,
                 mask=mask_r[:, None] & mask_h[None, :])

# ----------------------------------------------------------------------
# 5️⃣ Fused second LayerNorm, out_gate, and final projection
# ----------------------------------------------------------------------
@triton.jit
def final_fused_norm_proj_kernel(
    out_ptr,        # [B, H, N, N]                FP16
    gate_ptr,       # [B, H, N, N]                FP16
    ln_w_ptr,       # [H]                         FP32 (γ)
    ln_b_ptr,       # [H]                         FP32 (β)
    w_out_T_ptr,    # [H, dim]                    FP16 (transposed to_out weight)
    out_final_ptr,  # [B, N, N, dim]              FP32
    total_rows, H, N, dim,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr,
):
    pid_m = tl.program_id(0)   # rows
    pid_n = tl.program_id(1)   # output‑dim chunks

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)   # [BLOCK_M]
    offs_n = col_start + tl.arange(0, BLOCK_N)   # [BLOCK_N]

    mask_m = offs_m < total_rows
    mask_n = offs_n < dim

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + j   # start of plane h=0

    # accumulators
    acc_A = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)   # Σ out·γ·gate·W
    acc_B = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)   # Σ γ·gate·W
    acc_C = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)   # Σ β·gate·W

    sum_out = tl.zeros([BLOCK_M], dtype=tl.float32)          # for μ
    sum_sq_out = tl.zeros([BLOCK_M], dtype=tl.float32)       # for σ²

    for k_start in range(0, H, BLOCK_K):
        k_size = tl.minimum(BLOCK_K, H - k_start)
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        # ----- load out & gate (shape [M, K]) -----
        ptr_out  = out_ptr  + base_offset[:, None] + offs_k[None, :] * hidden_stride
        ptr_gate = gate_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride

        out_tile  = tl.load(ptr_out,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0)           # fp16
        gate_tile = tl.load(ptr_gate,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0)           # fp16

        # ----- statistics over out (μ, σ²) -----
        out_tile_f32 = out_tile.to(tl.float32)
        sum_out    += tl.sum(out_tile_f32, axis=1)
        sum_sq_out += tl.sum(out_tile_f32 * out_tile_f32, axis=1)

        # ----- load LayerNorm parameters (γ, β) -----
        gamma_tile = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0).to(tl.float16)   # fp16
        beta_tile  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0).to(tl.float16)   # fp16

        # ----- prepare tensors for the three reductions -----
        # a = out * gate * γ
        a_tile = out_tile * gate_tile * gamma_tile[None, :]
        # b = gate * γ
        b_tile = gate_tile * gamma_tile[None, :]
        # c = gate * β
        c_tile = gate_tile * beta_tile[None, :]

        # ----- load weight tile (shape [K, N]) -----
        ptr_w = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]          # [K, N]
        w_tile = tl.load(ptr_w,
                         mask=mask_k[:, None] & mask_n[None, :],
                         other=0.0)                                           # fp16

        # ----- three dot‑products (accumulate in FP32) -----
        acc_A += tl.dot(a_tile, w_tile).to(tl.float32)
        acc_B += tl.dot(b_tile, w_tile).to(tl.float32)
        acc_C += tl.dot(c_tile, w_tile).to(tl.float32)

    # ----- compute μ, rsqrt‑variance -----
    mu   = sum_out / H
    var  = sum_sq_out / H - mu * mu
    rstd = tl.rsqrt(var + eps)

    # ----- final composition:
    #   Y = rstd * (A - μ * B) + C
    final = (acc_A - mu[:, None] * acc_B) * rstd[:, None] + acc_C

    # ----- store result (shape [B, N, N, dim]) -----
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, final, mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
# Entry point – outgoing TriMul forward (no gradients)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle‑Multiplicative update (TriMul).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, C] (FP32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
        - weights_dict : mapping containing all model parameters (FP32)
        - config_dict  : must contain ``dim``, ``hidden_dim`` and ``nomask``

    Returns
    -------
    torch.Tensor of shape [B, N, N, C] (FP32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                         # total (b,i,j) positions
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # --------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                     # [R, C]  FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # --------------------------------------------------------------
    # 2️⃣ Fused linear projection (5 × hidden_dim)
    # --------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                         # [5·H, C]
    proj = F.linear(x_norm, fused_w)            # (R, 5·H) FP16
    del fused_w, x_norm

    # --------------------------------------------------------------
    # 3️⃣ Split, mask, sigmoid gates → left / right / out_gate
    # --------------------------------------------------------------
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()   # [R]
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension (cuBLAS)
    # --------------------------------------------------------------
    # shape needed for torch.bmm: (B·H, N, N)
    left_mat  = left.view(B * H, N, N)               # FP16
    right_mat = right.view(B * H, N, N)              # FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                       # [B, H, N, N] FP16
    # No longer need left/right tensors
    del left, right

    # --------------------------------------------------------------
    # 5️⃣ Fused LayerNorm + gating + final projection
    # --------------------------------------------------------------
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim]

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 64   # rows per program (over R)
    BLOCK_K_OUT = 32   # hidden‑dim tile
    BLOCK_N_OUT = 64   # output‑dim tile

    grid_out = (
        triton.cdiv(R, BLOCK_M_OUT),
        triton.cdiv(dim, BLOCK_N_OUT),
    )
    final_fused_norm_proj_kernel[grid_out](
        out,
        out_gate,
        w["to_out_norm.weight"],   # γ  (FP32)
        w["to_out_norm.bias"],     # β  (FP32)
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=8,
    )
    # out, out_gate are no longer needed
    del out, out_gate, w_out_T

    return final
