import torch
import torch.nn.functional as F
import triton
import triton.language as tl

def custom_kernel(data):
    """
    Outgoing TriMul forward pass (no gradient).

    Algorithm:
        1. Per‑row LayerNorm on the input channel dimension (FP32 → FP16)
        2. Fuse the five linear layers (left, right, left_gate, right_gate,
           out_gate) into a single matmul.
        3. Decode the fused projection, apply optional mask, compute sigmoids
           and store left, right and out_gate tensors (FP16).
        4. Batched GEMM over the sequence dimension N:
               out[b, h] = left[b, h] @ right[b, h].T   (FP16)
        5. Row‑wise mean / rsqrt‑variance of ``out`` across the hidden dimension.
        6. Fuse the second LayerNorm, the element‑wise out_gate multiplication
           and the final linear projection into one Triton kernel.
        7. Return a float‑32 tensor of shape [B, N, N, C].
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, N2, _ = x.shape
    assert N == N2, "TriMul expects square pairwise matrix"
    R = B * N * N  # total rows for (b,i,k)
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()            # [R, C] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)  # [R, C] FP16

    BLOCK_M = 128
    BLOCK_D = 64
    @triton.jit
    def layernorm_kernel(
        x_ptr, y_ptr, w_ptr, b_ptr,
        rows, dim, eps,
        BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
    ):
        pid = tl.program_id(0)
        row_start = pid * BLOCK_M
        offs_m = row_start + tl.arange(0, BLOCK_M)
        mask_m = offs_m < rows

        # first pass: compute mean & variance
        sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
        sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

        for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
            offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
            mask_d = offs_d < dim

            ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
            vals = tl.load(ptr,
                           mask=mask_m[:, None] & mask_d[None, :],
                           other=0.0).to(tl.float32)

            sum_val += tl.sum(vals, axis=1)
            sum_sq  += tl.sum(vals * vals, axis=1)

        mean = sum_val / dim
        var  = sum_sq / dim - mean * mean
        rstd = tl.rsqrt(var + eps)

        # second pass: write normalized values
        for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
            offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
            mask_d = offs_d < dim

            ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
            vals = tl.load(ptr,
                           mask=mask_m[:, None] & mask_d[None, :],
                           other=0.0).to(tl.float32)

            norm = (vals - mean[:, None]) * rstd[:, None]

            gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
            beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

            out = norm * gamma[None, :] + beta[None, :]
            tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                     out.to(tl.float16),
                     mask=mask_m[:, None] & mask_d[None, :])

    layernorm_kernel[(triton.cdiv(R, BLOCK_M),)](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # ------------------------------------------------------------------
    # 2️⃣ Fuse the five linear projections (5·H outputs)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                    # [5·H, C]
    proj = F.linear(x_norm, fused_w)       # [R, 5·H] (FP16)
    del fused_w, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ Split fused tensor, apply mask, sigmoid gates, store left/right/out_gate
    # ------------------------------------------------------------------
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)

    BLOCK_R = 64
    BLOCK_H = 64

    @triton.jit
    def split_fuse_gate_kernel(
        proj_ptr, left_ptr, right_ptr, out_gate_ptr,
        mask_ptr,
        R, H, N,
        HAS_MASK: tl.constexpr,
        BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
    ):
        pid = tl.program_id(0)
        row_start = pid * BLOCK_R
        offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
        mask_r = offs_r < R

        # Decode (b,i,k) from flat index r = b·N·N + i·N + k
        NN = N * N
        b = offs_r // NN
        rem = offs_r - b * NN
        i = rem // N
        k = rem - i * N

        # Optional mask broadcast over H
        if HAS_MASK:
            m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
            mask_val = m_val[:, None]                                   # [BLOCK_R, 1]
        else:
            mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

        hidden_stride = N * N
        base_offset = b * (H * hidden_stride) + i * N + k   # start of plane 0

        for h_off in range(0, tl.cdiv(H, BLOCK_H)):
            offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
            mask_h = offs_h < H

            # [R, 5·H] layout: left | right | left_gate | right_gate | out_gate
            proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]          # [BLOCK_R, BLOCK_H]

            left_raw = tl.load(proj_ptr + proj_base,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)
            right_raw = tl.load(proj_ptr + proj_base + H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
            left_gate_raw = tl.load(proj_ptr + proj_base + 2 * H,
                                    mask=mask_r[:, None] & mask_h[None, :],
                                    other=0.0)
            right_gate_raw = tl.load(proj_ptr + proj_base + 3 * H,
                                     mask=mask_r[:, None] & mask_h[None, :],
                                     other=0.0)
            out_gate_raw = tl.load(proj_ptr + proj_base + 4 * H,
                                   mask=mask_r[:, None] & mask_h[None, :],
                                   other=0.0)

            # sigmoid (in FP32)
            left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
            right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
            out_gate_val = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

            left  = left_raw * mask_val * left_gate
            right = right_raw * mask_val * right_gate
            outg  = out_gate_val

            dst_off = base_offset[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

            tl.store(left_ptr + dst_off, left,
                     mask=mask_r[:, None] & mask_h[None, :])
            tl.store(right_ptr + dst_off, right,
                     mask=mask_r[:, None] & mask_h[None, :])
            tl.store(out_gate_ptr + dst_off, outg,
                     mask=mask_r[:, None] & mask_h[None, :])

        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------

    left_mat = left.view(B * H, N, N)            # FP16
    right_mat = right.view(B * H, N, N)          # FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B*H, N, N] FP16

    mu = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_K_STATS = 128

    def rowwise_stats_kernel(

        eps,
        BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr,

        row_start = pid * BLOCK_M
        offs_m = row_start + tl.arange(0, BLOCK_M)
        mask_m = offs_m < total_rows

        NN = N * N
        b = offs_m // NN
        rem = offs_m - b * NN
        i = rem // N
        j = rem - i * N

        hidden_stride = N * N
        base_offset = b * (H * hidden_stride) + i * N + j   # start of hidden plane 0

        sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
        sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

        for k_start in range(0, H, BLOCK_K):
            offs_k = k_start + tl.arange(0, BLOCK_K)
            mask_k = offs_k < H

            ptr = out_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride
            vals = tl.load(ptr,
                           mask=mask_m[:, None] & mask_k[None, :],
                           other=0.0).to(tl.float32)

            sum_val += tl.sum(vals, axis=1)
            sum_sq  += tl.sum(vals * vals, axis=1)

        mean = sum_val / H
        var  = sum_sq / H - mean * mean
        rstd = tl.rsqrt(var + eps)

        tl.store(mu_ptr + offs_m, mean, mask=mask_m)
        tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)

    rowwise_stats_kernel[(triton.cdiv(R, BLOCK_M_STATS),)](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=4,
    )

    # 6️⃣ Second LayerNorm + out_gate + final projection (fused)
    # ------------------------------------------------------------------
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim]

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 128
    BLOCK_K_OUT = 32
    BLOCK_N_OUT = 128

    @triton.jit
    def final_output_kernel(
        out_ptr, gate_ptr,
        mu_ptr, rstd_ptr,
        ln_w_ptr, ln_b_ptr,
        w_out_T_ptr, out_final_ptr,

        eps,
        BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr,

        pid_n = tl.program_id(1)   # output channel tiles

        row_start = pid_m * BLOCK_M
        col_start = pid_n * BLOCK_N

        offs_m = row_start + tl.arange(0, BLOCK_M)   # [M]
        offs_n = col_start + tl.arange(0, BLOCK_N)   # [N]

        mask_m = offs_m < total_rows
        mask_n = offs_n < dim

        NN = N * N
        b = offs_m // NN
        rem = offs_m - b * NN
        i = rem // N
        j = rem - i * N

        hidden_stride = N * N
        base_offset = b * (H * hidden_stride) + i * N + j   # start of hidden plane 0

        mu   = tl.load(mu_ptr + offs_m, mask=mask_m, other=0.0)   # [M]
        rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)   # [M]

        acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

        for k_start in range(0, H, BLOCK_K):
            offs_k = k_start + tl.arange(0, BLOCK_K)
            mask_k = offs_k < H

            # load hidden values (FP16 → FP32)
            ptr_out = out_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride
            ptr_gate = gate_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride

            val_out = tl.load(ptr_out,
                              mask=mask_m[:, None] & mask_k[None, :],
                              other=0.0).to(tl.float32)   # [M, K]
            val_gate = tl.load(ptr_gate,
                               mask=mask_m[:, None] & mask_k[None, :],
                               other=0.0).to(tl.float32)   # [M, K]

            gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [K]
            beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [K]

            # LayerNorm on hidden dim + out_gate multiplication
            norm = ((val_out - mu[:, None]) * rstd[:, None]) * gamma[None, :] + beta[None, :]
            term = norm * val_gate                     # [M, K]

            term_fp16 = term.to(tl.float16)

            # final linear projection slice (H, dim)
            w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]   # [K, TILE_N]
            w_tile = tl.load(w_ptr,
                             mask=mask_k[:, None] & mask_n[None, :],
                             other=0.0)                                   # [K, TILE_N]

            acc += tl.dot(term_fp16, w_tile).to(tl.float32)

        # write final output [B, N, N, dim] row‑major
        dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
        tl.store(dst_ptr,
                 acc,
                 mask=mask_m[:, None] & mask_n[None, :])

    grid_out = (triton.cdiv(R, BLOCK_M_OUT), triton.cdiv(dim, BLOCK_N_OUT))
    final_output_kernel[grid_out](
        out,
        out_gate,
        mu,
        rstd,
        w["to_out_norm.weight"],
        w["to_out_norm.bias"],
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=8,
    )

    return final

def custom_kernel(data):
    """
    Outgoing TriMul forward pass (no gradients).

    Algorithm
    ----------
    1️⃣ Per‑row LayerNorm on the channel dimension (FP32 → FP16).
    2️⃣ Fuse the five linear layers (left, right, left_gate, right_gate,
       out_gate) into a single matmul.
    3️⃣ Decode the fused tensor, apply the optional mask, compute sigmoids
       and store ``left``, ``right`` and ``out_gate`` (FP16).
    4️⃣ Batched GEMM over the sequence dimension N:
           out[b, h] = left[b, h] @ right[b, h].T          (FP16)
    5️⃣ Row‑wise mean / rsqrt‑variance of ``out`` across the hidden dim.
    6️⃣ Fuse the second LayerNorm, the element‑wise ``out_gate`` multiply
       and the final linear projection into a single Triton kernel.
    7️⃣ Return a float‑32 tensor of shape ``[B, N, N, C]``.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, N2, _ = x.shape
    assert N == N2, "TriMul expects a square pairwise matrix"
    R = B * N * N                     # total rows for (b,i,k)
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                 # [R, C] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64

    @triton.jit
    def layernorm_kernel(
        x_ptr, y_ptr, w_ptr, b_ptr,
        rows, dim, eps,
        BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
    ):
        pid = tl.program_id(0)
        row_start = pid * BLOCK_M
        offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
        mask_m = offs_m < rows

        # ---------- first pass: mean & variance ----------
        sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
        sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

        for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
            offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
            mask_d = offs_d < dim

            ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
            vals = tl.load(ptr,
                           mask=mask_m[:, None] & mask_d[None, :],
                           other=0.0).to(tl.float32)          # [BLOCK_M, BLOCK_D]

            sum_val += tl.sum(vals, axis=1)                     # [BLOCK_M]
            sum_sq  += tl.sum(vals * vals, axis=1)              # [BLOCK_M]

        mean = sum_val / dim
        var  = sum_sq / dim - mean * mean
        rstd = tl.rsqrt(var + eps)

        # ---------- second pass: write normalized values ----------
        for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
            offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
            mask_d = offs_d < dim

            ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
            vals = tl.load(ptr,
                           mask=mask_m[:, None] & mask_d[None, :],
                           other=0.0).to(tl.float32)          # [BLOCK_M, BLOCK_D]

            norm = (vals - mean[:, None]) * rstd[:, None]

            gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
            beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

            out = norm * gamma[None, :] + beta[None, :]
            tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                     out.to(tl.float16),
                     mask=mask_m[:, None] & mask_d[None, :])

        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

            w["right_proj.weight"],

        else:

        for h_off in range(0, tl.cdiv(H, BLOCK_H)):
            offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]

            proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

            left_raw = tl.load(proj_ptr + proj_base,
                               mask=mask_r[:, None] & mask_h[None, :],

            right_raw = tl.load(proj_ptr + proj_base + H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
            left_gate_raw = tl.load(proj_ptr + proj_base + 2 * H,
                                    mask=mask_r[:, None] & mask_h[None, :],
                                    other=0.0)
            right_gate_raw = tl.load(proj_ptr + proj_base + 3 * H,
                                     mask=mask_r[:, None] & mask_h[None, :],

            out_gate_raw = tl.load(proj_ptr + proj_base + 4 * H,

                                   other=0.0)

            # sigmoid (FP32 → FP16)
            left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
            right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
            out_gate_val = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

            tl.store(left_ptr + dst_off, left,
                     mask=mask_r[:, None] & mask_h[None, :])
            tl.store(right_ptr + dst_off, right,
                     mask=mask_r[:, None] & mask_h[None, :])
            tl.store(out_gate_ptr + dst_off, outg,
                     mask=mask_r[:, None] & mask_h[None, :])

    split_fuse_gate_kernel[(triton.cdiv(R, BLOCK_R),)](

        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )

    # 4️⃣ Batched GEMM over the hidden dimension (left @ rightᵀ)
    # ------------------------------------------------------------------
    left_mat = left.view(B * H, N, N)                   # FP16
    right_mat = right.view(B * H, N, N)                 # FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                         # [B, H, N, N] FP16

    # ------------------------------------------------------------------
    # 5️⃣ Row‑wise mean & rsqrt‑variance of `out` (over H)
    # ------------------------------------------------------------------
    mu = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 256
    BLOCK_K_STATS = 128

    @triton.jit
    def rowwise_stats_kernel(
        out_ptr, mu_ptr, rstd_ptr,
        total_rows, H, N,
        eps,
        BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr,

        pid = tl.program_id(0)
        row_start = pid * BLOCK_M
        offs_m = row_start + tl.arange(0, BLOCK_M)
        mask_m = offs_m < total_rows

        NN = N * N
        b = offs_m // NN
        rem = offs_m - b * NN

        j = rem - i * N

        base_offset = b * (H * hidden_stride) + i * N + j   # start of hidden plane 0

            mask_k = offs_k < H

            ptr = out_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride
            vals = tl.load(ptr,
                           mask=mask_m[:, None] & mask_k[None, :],
                           other=0.0).to(tl.float32)

            sum_val += tl.sum(vals, axis=1)
            sum_sq  += tl.sum(vals * vals, axis=1)

        mean = sum_val / H
        var  = sum_sq / H - mean * mean
        rstd = tl.rsqrt(var + eps)

        tl.store(mu_ptr + offs_m, mean, mask=mask_m)
        tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)

    rowwise_stats_kernel[(triton.cdiv(R, BLOCK_M_STATS),)](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm + out_gate + final projection (fused)
    # ------------------------------------------------------------------
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim]

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 128
    BLOCK_K_OUT = 32
    BLOCK_N_OUT = 128

    @triton.jit
    def final_output_kernel(
        out_ptr, gate_ptr,
        mu_ptr, rstd_ptr,
        ln_w_ptr, ln_b_ptr,
        w_out_T_ptr, out_final_ptr,
        total_rows, H, N, dim,
        eps,
        BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr,
    ):
        pid_m = tl.program_id(0)   # rows
        pid_n = tl.program_id(1)   # output‑dim tiles

        row_start = pid_m * BLOCK_M
        col_start = pid_n * BLOCK_N

        offs_m = row_start + tl.arange(0, BLOCK_M)   # [M]
        offs_n = col_start + tl.arange(0, BLOCK_N)   # [N]

        mask_m = offs_m < total_rows
        mask_n = offs_n < dim

        NN = N * N
        b = offs_m // NN
        rem = offs_m - b * NN
        i = rem // N
        j = rem - i * N

        hidden_stride = N * N
        base_offset = b * (H * hidden_stride) + i * N + j   # start of hidden plane 0

        mu   = tl.load(mu_ptr + offs_m, mask=mask_m, other=0.0)         # [M]
        rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)       # [M]

        acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

        for k_start in range(0, H, BLOCK_K):
            offs_k = k_start + tl.arange(0, BLOCK_K)
            mask_k = offs_k < H

            # load hidden values (FP16 → FP32)
            ptr_out = out_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride
            ptr_gate = gate_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride

            val_out = tl.load(ptr_out,
                              mask=mask_m[:, None] & mask_k[None, :],
                              other=0.0).to(tl.float32)   # [M, K]
            val_gate = tl.load(ptr_gate,
                               mask=mask_m[:, None] & mask_k[None, :],
                               other=0.0).to(tl.float32)   # [M, K]

            gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [K]
            beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [K]

            # LayerNorm over hidden dim + out_gate multiplication
            norm = ((val_out - mu[:, None]) * rstd[:, None]) * gamma[None, :] + beta[None, :]
            term = norm * val_gate                                      # [M, K]
            term_fp16 = term.to(tl.float16)

            # final linear projection slice (H, dim)
            w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]   # [K, TILE_N]
            w_tile = tl.load(w_ptr,
                             mask=mask_k[:, None] & mask_n[None, :],
                             other=0.0)                                    # [K, TILE_N]

            acc += tl.dot(term_fp16, w_tile).to(tl.float32)

        # write final output [B, N, N, dim] (row‑major)
        dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
        tl.store(dst_ptr,
                 acc,
                 mask=mask_m[:, None] & mask_n[None, :])

    grid_out = (triton.cdiv(R, BLOCK_M_OUT), triton.cdiv(dim, BLOCK_N_OUT))
    final_output_kernel[grid_out](
        out,
        out_gate,
        mu,
        rstd,
        w["to_out_norm.weight"],   # γ (FP32)
        w["to_out_norm.bias"],     # β (FP32)
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=8,
    )

    return final
```

This implementation follows the outlined algorithm, fuses the five linear projections, performs the pairwise multiplication via a batched GEMM, computes the required statistics with Triton kernels, and finally fuses the second LayerNorm, out‑gate multiplication and the output projection into a single Triton kernel. The result is a float‑32 tensor of shape **[B, N, N, C]** matching the PyTorch reference.
