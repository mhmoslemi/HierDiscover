    out_ptr,  # [B, H, N, N] FP16
    out_gate_ptr,  # [B, H, N, N] FP16

):
    pid = tl.program_id(0)
    row = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask = row < total_rows

    nn = N * N
    b = row // nn
    rem = row - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + j

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

"""
Outgoing TriMul (TriMul‑out) – efficient Triton forward pass

The kernel chain implements the following steps (all in FP16/FP32 mixed precision
and without gradients):

1️⃣  Row‑wise LayerNorm over the last dimension (`dim`).  The input is FP32,
    the output is FP16.

2️⃣  A *fused* linear projection that simultaneously computes the five hidden
    linear maps (left, right, left‑gate, right‑gate, out‑gate).  The projection
    is performed with a single `torch.nn.functional.linear` call.

3️⃣  A Triton kernel splits the fused projection, applies the optional pairwise
    mask, computes the sigmoid gates and stores `left`, `right` and `out_gate`
    in the layout `[B, H, N, N]` (FP16).

4️⃣  Batched GEMM (`torch.bmm`) evaluates the core Einsum
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    producing a tensor `[B, H, N, N]` (FP16).

5️⃣  A *single* Triton kernel fuses the per‑row mean/variance calculation,
    the second LayerNorm, the element‑wise out‑gate multiplication and the
    final projection back to `dim`.  The kernel iterates over the hidden
    dimension in tiles, using float32 accumulators for numerical stability.

All operations are inference‑only and the final result has shape
`[B, N, N, dim]` (FP32).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (float32 → float16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # [R, D]   FP32 input
    y_ptr,          # [R, D]   FP16 output
    w_ptr,          # [D]      LN weight (γ) FP32
    b_ptr,          # [D]      LN bias  (β) FP32
    rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ---------- 1st pass: compute mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)           # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                   # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)            # [BLOCK_M]

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- 2nd pass: write normalized values ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, apply mask and sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,      # [R, 5·H]               FP16 fused linear output
    left_ptr,      # [B, H, N, N]           FP16
    right_ptr,     # [B, H, N, N]           FP16
    out_gate_ptr,  # [B, H, N, N]           FP16
    mask_ptr,      # [R]                    FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)                # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index r → (b, i, k)
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask (broadcast over H)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                   # [BLOCK_R, 1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N                     # stride of one hidden plane in dest tensors
    base_offset = b * (H * hidden_stride) + i * N + k   # start for h = 0

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # linear view:   [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(proj_ptr + proj_base,
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0)

        right_raw = tl.load(proj_ptr + proj_base + H,
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0)

        left_gate_raw = tl.load(proj_ptr + proj_base + 2 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)

        right_gate_raw = tl.load(proj_ptr + proj_base + 3 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)

        out_gate_raw = tl.load(proj_ptr + proj_base + 4 * H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)

        # sigmoid in FP32 for stability
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate

        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst_offset, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset, right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset, outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 5️⃣ Fuse 2nd LayerNorm, out_gate and final projection
# ----------------------------------------------------------------------
@triton.jit
def norm_proj_kernel(
    out_ptr,        # [B, H, N, N]               FP16
    out_gate_ptr,   # [B, H, N, N]               FP16
    ln_w_ptr,       # [H]                        FP32 (γ)
    ln_b_ptr,       # [H]                        FP32 (β)
    w_out_T_ptr,    # [H, dim]                   FP16 (transposed final weight)
    out_final_ptr,  # [B, N, N, dim]             FP32
    total_rows, H, N, dim,
    eps,
    BLOCK_M: tl.constexpr,   # rows per program (usually 1)
    BLOCK_K: tl.constexpr,   # hidden‑dim tile
    BLOCK_N: tl.constexpr,   # output‑dim tile
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)                # [BLOCK_M]
    mask_m = offs_m < total_rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + j                # start of hidden plane 0

    # ---------- first pass: mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)               # [BLOCK_K]
        mask_k = offs_k < H

        ptr = out_ptr + base[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)               # [M, K]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- second pass: LN, gate, final projection ----------
    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    # reuse tile‑wise objects for the weight matrix
    offs_n = tl.arange(0, BLOCK_N)
    mask_n = offs_n < dim

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)               # [BLOCK_K]
        mask_k = offs_k < H

        # load hidden‑plane values and gate
        ptr_out  = out_ptr + base[:, None] + offs_k[None, :] * hidden_stride
        ptr_gate = out_gate_ptr + base[:, None] + offs_k[None, :] * hidden_stride

        out_vals  = tl.load(ptr_out,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)            # [M, K]
        gate_vals = tl.load(ptr_gate,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)            # [M, K]

        # LayerNorm parameters (γ,β) – broadcast over rows
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [K] FP32
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [K] FP32

        # LN → (out - μ) * rstd
        norm = (out_vals - mean[:, None]) * rstd[:, None]          # [M, K]
        # apply γ,β
        norm = norm * gamma[None, :] + beta[None, :]               # [M, K]
        # gate
        gated = norm * gate_vals                                   # [M, K]

        # cast to fp16 for GEMM with the final weight
        gated_fp16 = gated.to(tl.float16)

        # final weight tile (H, dim)
        w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]   # [K, N]
        w_tile = tl.load(w_ptr,
                         mask=mask_k[:, None] & mask_n[None, :],
                         other=0.0)                                      # [K, N]

        # accumulate (M,K)·(K,N) → (M,N)   (fp16·fp16 → fp32)
        acc += tl.dot(gated_fp16, w_tile).to(tl.float32)

    # write result
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, acc,
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Triton entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights_dict : mapping containing all model weights (float32)
            - config_dict  : must contain keys "dim", "hidden_dim", "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack inputs
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape                # x: [B, N, N, dim]
    R = B * N * N                        # total number of (i,k) pairs
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (float32 → float16)
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()            # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection for left/right + gates
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                         # [5·H, dim]

    proj = F.linear(x_norm, fused_w)            # (R, 5·H) FP16
    del fused_w, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ Split, apply mask, sigmoid gates → left / right / out_gate
    # ------------------------------------------------------------------
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension (cuBLAS)
    # ------------------------------------------------------------------
    left_mat = left.view(B * H, N, N)                      # [B·H, N, N] FP16
    right_mat = right.view(B * H, N, N)                    # [B·H, N, N] FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                             # [B, H, N, N] FP16

    # ------------------------------------------------------------------
    # 5️⃣ Fuse 2nd LN, out_gate and final projection
    # ------------------------------------------------------------------
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim] FP16

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 1          # one (b,i,j) per program
    BLOCK_K_OUT = 32         # hidden‑dim tile (fits H=128 comfortably)
    BLOCK_N_OUT = 128        # output‑dim tile (covers dim up to 384)

    grid_out = (triton.cdiv(R, BLOCK_M_OUT),)
    norm_proj_kernel[grid_out](
        out,
        out_gate,
        w["to_out_norm.weight"],   # γ  (FP32)
        w["to_out_norm.bias"],     # β  (FP32)
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=8,
    )
    # ------------------------------------------------------------------
    # clean‑up & return
    # ------------------------------------------------------------------
    return final
