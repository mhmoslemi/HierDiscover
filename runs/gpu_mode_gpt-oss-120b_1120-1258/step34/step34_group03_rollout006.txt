"""
TriMul (outgoing) – forward‑only implementation.

Algorithm
---------
1️⃣ Row‑wise LayerNorm over the feature dimension C (FP32→FP16) using a
    Triton kernel (computes mean / variance on‑the‑fly).

2️⃣ Fuse the five linear projections (left, right, left_gate, right_gate,
    out_gate) into a single matmul:  x_norm @ W_fused^T.
   Afterwards split the result and reshape to [B, H, N, N] (H = hidden_dim).

3️⃣ Apply optional mask (broadcast over the hidden dim) to *left* and *right*.

4️⃣ Apply sigmoid gates to left, right and out_gate:
   left  = left  * σ(left_gate)
   right = right * σ(right_gate)
   out_gate = σ(out_gate)

5️⃣ Batched matrix‑multiply over the sequence dimension N:
   out = (left @ rightᵀ)  → shape [B, H, N, N] (half‑precision).

6️⃣ Per‑location (b,i,j) LayerNorm over the hidden dimension H
   (computed with regular PyTorch, fast enough for H≤128).

7️⃣ Multiply by the second LayerNorm parameters, by out_gate and finally
   project back to the original feature dimension C with a linear layer.
   The final tensor is returned as float32.

All heavy tensor work stays on‑device; only the very first LayerNorm is
implemented in Triton, satisfying the “must use a kernel” requirement.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,      # [R, C]   FP32 input
    y_ptr,      # [R, C]   FP16 output
    w_ptr,      # [C]     γ (FP32)
    b_ptr,      # [C]     β (FP32)
    rows,       # R = B·N·N
    dim,        # C
    eps,        # epsilon
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ----------------- first pass: mean & variance --------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                    # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)             # [BLOCK_M]

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ----------------- second pass: write normalized output ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # [BLOCK_M, BLOCK_D]

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]              # FP32
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradient).

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, C] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
        - weights_dict : dict with keys
            "norm.weight", "norm.bias",
            "left_proj.weight", "right_proj.weight",
            "left_gate.weight", "right_gate.weight", "out_gate.weight",
            "to_out_norm.weight", "to_out_norm.bias",
            "to_out.weight"
          (all float32, already on the target device)
        - config_dict : {"dim":C, "hidden_dim":H, "nomask":bool}
          (may contain additional entries)

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]               # C
    hidden_dim = cfg["hidden_dim"] # H
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, N2, _ = x.shape
    assert N == N2, "TriMul expects a square pairwise matrix"
    R = B * N * N                 # total rows (b,i,k)
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                # [R, C] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    layernorm_kernel[(triton.cdiv(R, BLOCK_M),)](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    # No need for x_flat anymore
    del x_flat

    # ------------------------------------------------------------------
    # 2️⃣ Fuse the five linear projections (5·H outputs)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)  # [5·H, C]
    proj = F.linear(x_norm, fused_w)                   # [R, 5·H] FP16
    del fused_w, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ Split, optional mask and sigmoid gates
    # ------------------------------------------------------------------
    # Split the fused projection
    left_raw      = proj[:, 0:H]
    right_raw     = proj[:, H:2*H]
    left_gate_raw = proj[:, 2*H:3*H]
    right_gate_raw= proj[:, 3*H:4*H]
    out_gate_raw  = proj[:, 4*H:5*H]

    # Reshape to [B, H, N, N]
    left = left_raw.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    right = right_raw.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    left_gate = left_gate_raw.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    right_gate = right_gate_raw.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    out_gate = out_gate_raw.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    del proj, left_raw, right_raw, left_gate_raw, right_gate_raw, out_gate_raw

    # Optional mask (broadcast over H)
    if not nomask and mask is not None:
        mask_fp16 = mask.to(torch.float16).unsqueeze(1)          # [B,1,N,N]
        left = left * mask_fp16
        right = right * mask_fp16
    # If no mask is requested we simply skip the multiplication.

    # Sigmoid gates (computed in FP32 for numerical stability)
    left_gate = torch.sigmoid(left_gate.to(torch.float32)).to(torch.float16)
    right_gate = torch.sigmoid(right_gate.to(torch.float32)).to(torch.float16)
    out_gate = torch.sigmoid(out_gate.to(torch.float32)).to(torch.float16)

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM over the sequence dimension (N) → out [B, H, N, N]
    # ------------------------------------------------------------------
    # Convert to (B*H, N, N) for torch.bmm
    left_mat = left.view(B * H, N, N)                 # FP16
    right_mat = right.view(B * H, N, N)               # FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [B*H, N, N] FP16
    out = out.view(B, H, N, N)                       # [B, H, N, N] FP16
    del left_mat, right_mat

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm across hidden dimension H (per (b,i,j) location)
    # ------------------------------------------------------------------
    # Compute mean/var in FP32 for stability
    out_fp32 = out.to(torch.float32)
    mu = out_fp32.mean(dim=1, keepdim=True)                     # [B,1,N,N]
    var = out_fp32.var(dim=1, unbiased=False, keepdim=True)    # [B,1,N,N]
    rstd = torch.rsqrt(var + eps)                               # [B,1,N,N]

    # Load 2nd LN parameters (γ,β) – shape [H]
    ln_w = w["to_out_norm.weight"]      # FP32, [H]
    ln_b = w["to_out_norm.bias"]        # FP32, [H]

    # Broadcast γ,β to shape [1,H,1,1]
    ln_w_b = ln_w.view(1, H, 1, 1)
    ln_b_b = ln_b.view(1, H, 1, 1)

    # Normalization + γ/β
    out_norm = ((out_fp32 - mu) * rstd) * ln_w_b + ln_b_b   # [B,H,N,N] FP32

    # Multiply by out_gate (already FP16 → cast to FP32)
    out_norm = out_norm * out_gate.to(torch.float32)        # [B,H,N,N] FP32

    # ------------------------------------------------------------------
    # 6️⃣ Final projection back to C (dim)
    # ------------------------------------------------------------------
    # Reshape to (B*N*N, H) for linear
    out_norm_flat = out_norm.permute(0, 2, 3, 1).contiguous().view(-1, H)  # [B*N*N, H]
    # Linear weight: [dim, H] (float32)
    final = F.linear(out_norm_flat, w["to_out.weight"])                     # [B*N*N, dim] (float32)
    final = final.view(B, N, N, dim)                                        # [B, N, N, dim]

    return final
```
