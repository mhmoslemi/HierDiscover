import triton.language as tl

@triton.jit
def layernorm_kernel(x_ptr, y_ptr, w_ptr, b_ptr, rows, dim, eps, BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # compute mean and variance
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)

        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

"""
TriMul (outgoing) – fast Triton + torch implementation

Algorithm
---------
1️⃣  Row‑wise LayerNorm on the flattened [B·N·N, C] input (FP32→FP16) with a custom
    Triton kernel (the only required Triton component).

2️⃣  Fuse the five linear projections (left, right, left_gate, right_gate, out_gate)
    into a single GEMM (FP16).  The result is reshaped to (R,5,H) where
    R = B·N·N and H = hidden_dim.

3️⃣  Split the fused projection, apply the optional mask, compute the three
    sigmoids and obtain tensors *left*, *right* and *out_gate*.
    The tensors are reshaped to (B, H, N, N) – layout required by the
    batched GEMM that follows.

4️⃣  Batched GEMM (torch.bmm) implements the core einsum:
       out[b,h,i,j] = Σₖ left[b,h,i,k] · right[b,h,j,k]   →  (B,H,N,N) FP16.

5️⃣  Compute per‑position mean/variance of *out* over the hidden dimension H
    with native torch reductions (much faster than a custom kernel).

6️⃣  Apply the second LayerNorm (γ/β on the hidden dimension), multiply by
    *out_gate* and finally a linear projection to the original feature dimension
    using a fused torch linear (FP16 weight, FP16 activation → FP32 output).

The whole pipeline runs entirely on the GPU and returns a float‑32 tensor
with shape [B, N, N, C].
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣  Row‑wise LayerNorm (FP32 → FP16) – Triton kernel
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # [R, dim]   FP32 input
    y_ptr,          # [R, dim]   FP16 output
    w_ptr,          # [dim]      LayerNorm weight  (γ, FP32)
    b_ptr,          # [dim]      LayerNorm bias    (β, FP32)
    rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ------------------- first pass: mean & variance -------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)           # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                   # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)            # [BLOCK_M]

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------- second pass: write normalized values ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)           # [BLOCK_M, BLOCK_D]

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Entry point – outgoing TriMul forward (no gradient)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradient).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, C] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : contains the model parameters (all float32 tensors)
                * "norm.weight", "norm.bias"
                * "left_proj.weight", "right_proj.weight"
                * "left_gate.weight", "right_gate.weight", "out_gate.weight"
                * "to_out_norm.weight", "to_out_norm.bias"
                * "to_out.weight"
            - config_dict must contain "dim", "hidden_dim", "nomask" (bool)
    Returns:
        torch.Tensor: [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim          = cfg["dim"]            # input feature dimension C
    hidden_dim   = cfg["hidden_dim"]    # H
    nomask       = cfg.get("nomask", True)
    eps          = 1e-5
    device       = x.device

    B, N, _, _ = x.shape
    R = B * N * N                      # total (b,i,k) positions
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm on the flattened input (FP32 → FP16)
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                 # [R, C] float32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],          # γ  (float32)
        w["norm.bias"],            # β  (float32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # ------------------------------------------------------------------
    # 2️⃣ Fuse the five linear projections (5·H × C) with a single GEMM
    # ------------------------------------------------------------------
    fused_weight = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                         # [5·H, C]

    proj = F.linear(x_norm, fused_weight)       # (R, 5·H) FP16
    del fused_weight, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ Split, apply optional mask, compute sigmoids → left/right/out_gate
    # ------------------------------------------------------------------
    proj = proj.view(R, 5, H)                   # (R, 5, H)

    left_raw      = proj[:, 0, :]               # (R, H)
    right_raw     = proj[:, 1, :]               # (R, H)
    left_gate_raw = proj[:, 2, :]               # (R, H)
    right_gate_raw= proj[:, 3, :]               # (R, H)
    out_gate_raw  = proj[:, 4, :]               # (R, H)

    if (not nomask) and (mask is not None):
        # mask is broadcast over the hidden dimension
        mask_flat = mask.to(torch.float16).reshape(R)          # (R,)
        mask_flat = mask_flat.unsqueeze(-1)                    # (R,1)
        left_raw  = left_raw  * mask_flat
        right_raw = right_raw * mask_flat

    # sigmoid gates (stable in float32, cast back)
    left_gate  = torch.sigmoid(left_gate_raw).to(torch.float16)
    right_gate = torch.sigmoid(right_gate_raw).to(torch.float16)
    out_gate   = torch.sigmoid(out_gate_raw).to(torch.float16)

    left  = (left_raw  * left_gate).contiguous()
    right = (right_raw * right_gate).contiguous()
    out_gate = out_gate.contiguous()

    # reshape to [B, H, N, N] (the layout expected by the batched GEMM)
    left  = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    out_gate = out_gate.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    del proj, left_raw, right_raw, left_gate_raw, right_gate_raw, out_gate_raw

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension (core einsum)
    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)                     # [B·H, N, N] FP16
    right_mat = right.view(B * H, N, N)                    # [B·H, N, N] FP16

    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                             # [B, H, N, N] FP16
    del left_mat, right_mat, left, right

    # ------------------------------------------------------------------
    # 5️⃣ Per‑position LayerNorm over the hidden dimension (mean / var)
    # ------------------------------------------------------------------
    out_fp32 = out.float()                                 # cast once for reduction
    mu   = out_fp32.mean(dim=1, keepdim=True)              # [B,1,N,N] float32
    var  = out_fp32.var(dim=1, unbiased=False, keepdim=True)
    rstd = (var + eps).rsqrt()                             # [B,1,N,N] float32

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm, out_gate, final linear projection
    # ------------------------------------------------------------------
    gamma = w["to_out_norm.weight"].view(1, H, 1, 1)   # (1,H,1,1) float32
    beta  = w["to_out_norm.bias"  ].view(1, H, 1, 1)   # (1,H,1,1) float32

    # (out - μ) * rstd   →   normalized hidden representation
    normed = ((out_fp32 - mu) * rstd) * gamma + beta   # float32, shape (B,H,N,N)

    # multiply by out_gate (cast gate to fp32)
    normed = normed * out_gate.float()                 # float32

    # collapse the (B,N,N) positions → R rows and run the final linear
    normed_flat = normed.permute(0, 2, 3, 1).reshape(R, H)   # (R, H)
    w_out = w["to_out.weight"].to(torch.float16)             # (dim, H)

    final_fp16 = F.linear(normed_flat.to(torch.float16), w_out)   # (R, dim) FP16
    final = final_fp16.to(torch.float32).reshape(B, N, N, dim)    # (B,N,N,dim)

    return final
