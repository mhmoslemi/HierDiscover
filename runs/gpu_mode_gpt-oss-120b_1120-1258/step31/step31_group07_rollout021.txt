"""
TriMul (outgoing) – highly‑optimized forward implementation.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the input tensor (FP32 → FP16) – separate Triton kernel.
2️⃣ Fuse the five linear projections (left, right, left‑gate, right‑gate, out‑gate)
   into a single matmul, then split + apply the optional pairwise mask and sigmoid
   gating in a Triton kernel (produces left, right, out_gate tensors).
3️⃣ Batched GEMM `torch.bmm` computes the N³ contraction:
   out[b,h,i,j] = Σₖ left[b,h,i,k] * right[b,h,j,k]   (FP16).
4️⃣ Fuse the second LayerNorm, the out_gate multiplication and the final linear
   projection into a single Triton kernel.  The kernel computes per‑row mean &
   variance of the contracted tensor, applies the LN (γ,β), multiplies by the
   gate, and directly projects to the model dimension (dim).  No intermediate
   μ / rstd buffers are materialised, reducing global‑memory traffic.
5️⃣ Return the FP32 output of shape [B, N, N, dim].

All kernels are written for Triton 3.3.1 and target an NVIDIA H100.

# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# --------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,      # [R, C] FP32 input
    y_ptr,      # [R, C] FP16 output
    w_ptr,      # [C]   γ (FP32)
    b_ptr,      # [C]   β (FP32)
    rows,       # R = B·N·N
    dim,        # C
    eps,        # epsilon
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    #----- first pass: sum & sum of squares ---------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                                   # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                     # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)              # [BLOCK_M]

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    #----- second pass: write normalized values ------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                                   # [BLOCK_M, BLOCK_D]

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

    BLOCK_H: tl.constexpr,

    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # Decode flat index r → (b,i,k)
    NN = N * N
    b = offs_r // NN
    rem = offs_r - b * NN
    i = rem // N
    k = rem - i * N

    # Optional mask broadcast over the hidden dimension
    if HAS_MASK:

    hidden_stride = NN                     # stride between successive hidden planes

            other=0.0,

        right_raw = tl.load(

            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,

                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset,

                 mask=mask_r[:, None] & mask_h[None, :])

    base_offset_hidden = b * (H * hidden_stride) + i * N + j   # start of hidden plane h=0

    # Accumulators
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)   # Σ z_k
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)   # Σ z_k²

    acc_A = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)   # Σ (z·g·γ·w)
    acc_S = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)   # Σ (g·γ·w)
    acc_B = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)   # Σ (g·β·w)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        # ----- load contracted tensor ------------------------------------------------
        ptr_out = out_ptr + base_offset_hidden[:, None] + offs_k[None, :] * hidden_stride
        out_vals = tl.load(
            ptr_out,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)                               # [M, K]

        # ----- load gate values -------------------------------------------------------
        ptr_gate = gate_ptr + base_offset_hidden[:, None] + offs_k[None, :] * hidden_stride
        gate_vals = tl.load(
            ptr_gate,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)                               # [M, K]

        # accumulate mean/variance terms
        sum_val += tl.sum(out_vals, axis=1)
        sum_sq  += tl.sum(out_vals * out_vals, axis=1)

        # ----- LayerNorm parameters ---------------------------------------------------
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [K]   FP32
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [K]   FP32

        # ----- weight matrix tile ----------------------------------------------------
        w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]   # [K, TILE_N]
        w_tile = tl.load(
            w_ptr,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )                                                               # [K, TILE_N] FP16
        w_tile_f = w_tile.to(tl.float32)                               # cast to FP32

        # ----- compute contributions --------------------------------------------------
        # term_A : z * g * γ * w
        term_A = (out_vals * gate_vals) * gamma[None, :]                # [M, K]
        acc_A += tl.dot(term_A, w_tile_f)                              # [M, N]

        # term_S : g * γ * w
        term_S = gate_vals * gamma[None, :]                             # [M, K]
        acc_S += tl.dot(term_S, w_tile_f)

        # term_B : g * β * w
        term_B = gate_vals * beta[None, :]                              # [M, K]
        acc_B += tl.dot(term_B, w_tile_f)

    # ----- finalize LayerNorm ------------------------------------------------------
    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # output = rstd * (A - mean * S) + B
    out = rstd[:, None] * (acc_A - mean[:, None] * acc_S) + acc_B   # [M, N]

    # ----- write back ---------------------------------------------------------------
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]   # [M, N]
    tl.store(
        dst_ptr,
        out,
        mask=mask_m[:, None] & mask_n[None, :],
    )


# --------------------------------------------------------------
# Entry point
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only implementation of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, C] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
        - weights_dict : dict with the keys
              "norm.weight", "norm.bias",
              "left_proj.weight", "right_proj.weight",
              "left_gate.weight", "right_gate.weight", "out_gate.weight",
              "to_out_norm.weight", "to_out_norm.bias",
              "to_out.weight"
        - config_dict  : {"dim": C, "hidden_dim": H, "nomask": bool}
    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, C] (float32)
    """
    # --------------------------------------------------------------
    # unpack
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, N2, _ = x.shape
    assert N == N2, "TriMul expects a square pairwise matrix"
    R = B * N * N                      # total rows = B·N·N
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm (FP32 → FP16)
    # --------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()               # [R, C] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    layernorm_kernel[(triton.cdiv(R, BLOCK_M),)](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # --------------------------------------------------------------
    # 2️⃣ Fuse the five projections into one matmul
    # --------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                      # [5·H, C]

    proj = F.linear(x_norm, fused_w)                         # [R, 5·H] FP16
    del fused_w, x_norm

    # --------------------------------------------------------------
    # 3️⃣ Split + mask + sigmoid gates
    # --------------------------------------------------------------
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    split_fuse_gate_kernel[(triton.cdiv(R, BLOCK_R),)](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣ Batched GEMM (N³ contraction) – cuBLAS
    # --------------------------------------------------------------
    left_mat = left.view(B * H, N, N)          # FP16
    right_mat = right.view(B * H, N, N)        # FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                # [B, H, N, N] FP16
    del left_mat, right_mat                    # free memory

    # --------------------------------------------------------------
    # 5️⃣ Fuse 2nd LayerNorm, out_gate, and final projection
    # --------------------------------------------------------------
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim] FP16

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 128       # rows per program (over R)
    BLOCK_K_OUT = 32        # hidden‑dim tile
    BLOCK_N_OUT = 128       # output‑dim tile

    grid_out = (
        triton.cdiv(R, BLOCK_M_OUT),
        triton.cdiv(dim, BLOCK_N_OUT),
    )
    fused_output_kernel[grid_out](
        out,
        out_gate,
        w["to_out_norm.weight"],   # γ (FP32)
        w["to_out_norm.bias"],     # β (FP32)
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=8,
    )
    return final
```

This implementation removes the separate row‑wise statistics kernel, fusing the second LayerNorm, the out‑gate multiplication and the final linear projection into a single Triton kernel.  By cutting one global‑memory pass and one kernel launch, memory traffic and launch overhead are reduced, yielding a noticeable speed‑up (target ≈ 1000 μs on the reference configurations).
