@triton.jit
def sigmoid_kernel(
    src_ptr, dst_ptr,
    n_elements,
    BLOCK_SIZE: tl.constexpr
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < n_elements

    x = tl.load(src_ptr + offs, mask=mask, other=0.0)
    # compute sigmoid in fp32 for accuracy
    x_fp32 = x.to(tl.float32)
    y_fp32 = 1.0 / (1.0 + tl.exp(-x_fp32))
    y = y_fp32.to(tl.float16)
    tl.store(dst_ptr + offs, y, mask=mask)
```

"""
TriMul (outgoing) forward pass – Triton‑augmented implementation.

Algorithm
---------
1️⃣  Row‑wise LayerNorm on the input tensor (float32).
2️⃣  Five independent linear projections (left, right, left_gate, right_gate,
    out_gate) are computed with the provided FP32 weights.
3️⃣  An element‑wise Triton kernel evaluates the sigmoid for the three gate
    tensors (still in FP32 – the kernel uses the exact formula 1/(1+exp(−x))).
4️⃣  The optional pairwise mask is applied to the *left* and *right* tensors
    (broadcast on the hidden dimension).
5️⃣  Point‑wise multiplication with the corresponding gates yields the final
    *left* and *right* tensors.
6️⃣  Contraction  out[i,j] = Σₖ left[i,k]·right[j,k]  is performed as a batched
    matrix multiplication for each hidden slice (shape B·H, N, N).
7️⃣  A second LayerNorm normalises the contracted tensor over the hidden
    dimension, after which the out‑gate is applied.
8️⃣  A final linear projection maps the hidden dimension back to the model
    dimension → output shape [B, N, N, dim] (float32).

Only the sigmoid evaluation is implemented as a Triton kernel; all other
operations use highly‑optimised PyTorch kernels (including cuBLAS for the
batched GEMM).  The implementation works for any batch size (1‑2), sequence
length up to 1024, and model dimensions 128/384 as required by the test
suite.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: element‑wise sigmoid (FP32)
# ----------------------------------------------------------------------
@triton.jit
def sigmoid_fp32_kernel(
    src_ptr,          # input pointer (float32)
    dst_ptr,          # output pointer (float32)
    n_elements,       # total number of elements (int)
    BLOCK_SIZE: tl.constexpr,
):
    """Compute sigmoid(x) = 1 / (1 + exp(-x)) element‑wise."""
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)                 # [BLOCK_SIZE]
    mask = offs < n_elements

    # Load, compute sigmoid in FP32, store.
    x = tl.load(src_ptr + offs, mask=mask, other=0.0)       # FP32
    y = 1.0 / (1.0 + tl.exp(-x))
    tl.store(dst_ptr + offs, y, mask=mask)


def _sigmoid_triton(tensor: torch.Tensor) -> torch.Tensor:
    """Apply the FP32 sigmoid kernel to a contiguous tensor."""
    assert tensor.is_cuda and tensor.dtype == torch.float32
    flat_in = tensor.view(-1).contiguous()
    out = torch.empty_like(flat_in)
    BLOCK = 1024
    grid = (triton.cdiv(flat_in.numel(), BLOCK),)
    sigmoid_fp32_kernel[grid](
        flat_in,
        out,
        flat_in.numel(),
        BLOCK_SIZE=BLOCK,
        num_warps=4,
    )
    return out.view_as(tensor)


# ----------------------------------------------------------------------
# Triton‑augmented TriMul forward (no gradients)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (gradient‑free).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
        - weights_dict : mapping with keys
            "norm.weight", "norm.bias",
            "left_proj.weight", "right_proj.weight",
            "left_gate.weight", "right_gate.weight", "out_gate.weight",
            "to_out_norm.weight", "to_out_norm.bias",
            "to_out.weight"      (all float32)
        - config_dict must contain "dim" (int), "hidden_dim" (int), "nomask" (bool)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = bool(cfg.get("nomask", True))
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm over the last dimension (float32)
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim] float32

    # ------------------------------------------------------------------
    # 2️⃣ Five independent linear projections (no bias)
    # ------------------------------------------------------------------
    # Weight tensors are stored as (out_features, in_features)
    left_proj_w   = w["left_proj.weight"]
    right_proj_w  = w["right_proj.weight"]
    left_gate_w   = w["left_gate.weight"]
    right_gate_w  = w["right_gate.weight"]
    out_gate_w    = w["out_gate.weight"]

    # FP32 linear maps
    left_raw      = F.linear(x_norm, left_proj_w)       # [B,N,N,H]
    right_raw     = F.linear(x_norm, right_proj_w)      # [B,N,N,H]
    left_gate_raw = F.linear(x_norm, left_gate_w)      # [B,N,N,H]
    right_gate_raw= F.linear(x_norm, right_gate_w)     # [B,N,N,H]
    out_gate_raw  = F.linear(x_norm, out_gate_w)       # [B,N,N,H]

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask is [B,N,N]; broadcast to hidden dim
        mask_fp = mask.to(torch.float32).unsqueeze(-1)  # [B,N,N,1]
        left_raw  = left_raw * mask_fp
        right_raw = right_raw * mask_fp
        # note: gates are *not* masked (same as reference)
    # ------------------------------------------------------------------
    # 4️⃣ Sigmoid gates via Triton kernel (still FP32)
    # ------------------------------------------------------------------
    left_gate  = _sigmoid_triton(left_gate_raw)
    right_gate = _sigmoid_triton(right_gate_raw)
    out_gate   = _sigmoid_triton(out_gate_raw)

    # free raw gate tensors early
    del left_gate_raw, right_gate_raw, out_gate_raw

    # ------------------------------------------------------------------
    # 5️⃣ Apply gates to left/right tensors
    # ------------------------------------------------------------------
    left  = left_raw  * left_gate   # [B,N,N,H]
    right = right_raw * right_gate  # [B,N,N,H]
    del left_raw, right_raw, left_gate, right_gate

    # ------------------------------------------------------------------
    # 6️⃣ Core contraction: out[i,j] = Σₖ left[i,k]·right[j,k]
    #    → implemented as batched GEMM (B·H, N, N)
    # ------------------------------------------------------------------
    # transpose to [B, H, N, N] for batched matmul
    left  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    # left @ rightᵀ  (batch = B·H)
    out = torch.matmul(left, right.transpose(-2, -1))   # [B, H, N, N]

    # bring back to [B, N, N, H] for the second LayerNorm
    out = out.permute(0, 2, 3, 1).contiguous()        # [B, N, N, H]

    # ------------------------------------------------------------------
    # 7️⃣ Second LayerNorm over hidden dimension + out_gate multiplication
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=w["to_out_norm.weight"],
        bias=w["to_out_norm.bias"],
        eps=eps,
    )  # still float32

    # Apply out‑gate (element‑wise)
    out_norm = out_norm * out_gate
    del out, out_gate

    # ------------------------------------------------------------------
    # 8️⃣ Final linear projection back to model dimension
    # ------------------------------------------------------------------
    # to_out.weight has shape (dim, hidden_dim)
    final = F.linear(out_norm, w["to_out.weight"])   # [B, N, N, dim], float32

    return final
