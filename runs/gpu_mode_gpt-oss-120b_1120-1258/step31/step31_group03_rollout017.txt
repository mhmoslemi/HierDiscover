"""
Outgoing Triangular Multiplicative Update (TriMul) – forward only.

Algorithm (GPU‑only, mixed precision)
-------------------------------------
1️⃣  Row‑wise LayerNorm of the input tensor `x` (shape [B,N,N,dim]).
    Implemented with a small Triton kernel that works on FP32
    statistics and writes FP16 normalized results.

2️⃣  Fuse the five linear maps (left, right, left_gate, right_gate, out_gate)
    into a single GEMM:  proj = Linear(x_norm, W_fused)  →  shape (R,5·H) where
    R = B·N·N and H = hidden_dim.

3️⃣  Split the fused projection, apply the optional pair mask, compute the
    sigmoid gates and multiply left/right by the corresponding gates.
    All of this is done with plain torch ops (vectorised, fast).

4️⃣  Batched matrix‑multiply over the hidden dimension:
        out[b,h,i,j] = Σ_k left[b,h,i,k]·right[b,h,j,k]
    performed as `torch.bmm` on tensors (B·H, N, N).

5️⃣  Layer‑Norm over the hidden dimension H of `out`, followed by the
    out‑gate multiplication.  This uses the highly‑optimised
    `torch.nn.functional.layer_norm` implementation.

6️⃣  Final linear projection (to_out) merges the hidden dimension back to
    `dim`.  Implemented with a single `F.linear` on the flattened tensor.

All kernels run on the same CUDA device as the inputs.  The function
returns a float32 tensor of shape [B, N, N, dim] and does **not** compute
gradients.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,   # [R, dim]  FP32 input
    y_ptr,   # [R, dim]  FP16 output
    w_ptr,   # [dim]     γ (FP32)
    b_ptr,   # [dim]     β (FP32)
    rows,    # scalar R
    dim,     # scalar dim
    eps,     # scalar eps
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)            # [BLOCK_M]
    mask_m = offs_m < rows

    # ---- first pass: compute mean & variance ----
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- second pass: write normalized values ----
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Entrypoint – outgoing TriMul forward pass
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
              (all float32)
            - config_dict must contain
                "dim" (int), "hidden_dim" (int), "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = bool(cfg.get("nomask", True))
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                 # flattened pairwise token count
    H = hidden_dim                # hidden dimension

    # --------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # --------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # --------------------------------------------------------------
    # 2️⃣ Fuse five linear maps into one GEMM
    # --------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # [5·H, dim]

    proj = F.linear(x_norm, fused_w)         # (R, 5·H) FP16
    del fused_w, x_norm

    # --------------------------------------------------------------
    # 3️⃣ Split, apply optional mask, compute sigmoid gates,
    #    and multiply left/right by their gates
    # --------------------------------------------------------------
    # split the fused projection
    left_raw, right_raw, left_gate_raw, right_gate_raw, out_gate_raw = torch.split(
        proj, H, dim=1
    )
    # gates (sigmoid, stays FP16)
    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate = torch.sigmoid(out_gate_raw)

    if not nomask and mask is not None:
        # mask broadcast over the hidden dimension
        mask_flat = mask.to(torch.float16).reshape(R, 1)          # (R,1)
        left = left_raw * mask_flat * left_gate
        right = right_raw * mask_flat * right_gate
    else:
        left = left_raw * left_gate
        right = right_raw * right_gate

    # --------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension (cuBLAS)
    # --------------------------------------------------------------
    # reshape to (B, H, N, N) → (B·H, N, N) for torch.bmm
    left = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()

    left_mat = left.view(B * H, N, N)    # (B·H, N, N)
    right_mat = right.view(B * H, N, N)

    out = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B·H, N, N)
    out = out.view(B, H, N, N)                            # (B, H, N, N)

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm over hidden dimension + out_gate
    # --------------------------------------------------------------
    # flatten the first three dims → (R, H)
    out_flat = out.permute(0, 2, 3, 1).contiguous().view(R, H)

    out_norm = F.layer_norm(
        out_flat,
        (H,),
        weight=w["to_out_norm.weight"],
        bias=w["to_out_norm.bias"],
        eps=eps,
    )  # (R, H) FP32 → we keep FP32 for the next matmul

    # reshape back and apply out_gate (still FP16 → cast to FP32)
    out_norm = out_norm.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)
    out_gate = out_gate.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    out_scaled = out_norm.to(torch.float16) * out_gate                     # (B, H, N, N) FP16

    # --------------------------------------------------------------
    # 6️⃣ Final projection (to_out)
    # --------------------------------------------------------------
    out_scaled_flat = out_scaled.permute(0, 2, 3, 1).contiguous().view(R, H)  # (R, H)
    final = F.linear(out_scaled_flat.float(), w["to_out.weight"].float())      # (R, dim) FP32
    final = final.view(B, N, N, dim)                                            # (B, N, N, dim)

    return final
