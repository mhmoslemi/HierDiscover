```python
def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = bool(config.get("nomask", True))
    eps = 1e-5
    device = input_tensor.device
    B, N, _, _ = input_tensor.shape
    R = B * N * N
    H = hidden_dim

    # 1. LayerNorm (use torch.nn.functional.layer_norm for speed)
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    ).to(torch.float16)

    # 2. Fused projection (left,right, gates)
    fused_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)  # (5*H, dim)

    proj = torch.nn.functional.linear(x_norm.view(R, dim), fused_w)  # (R,5H) fp16
    # free x_norm
    del x_norm

    # 3. Split, mask and sigmoid gates
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if not nomask and mask is not None:
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)

    BLOCK_R = 64
    BLOCK_H = 64
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # 4. Batched GEMM for the core einsum
    left_mat = left.reshape(B * H, N, N)
    right_mat = right.reshape(B * H, N, N)
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))
    out = out.view(B, H, N, N)

    # 5. Compute per‑pair mean & rsqrt‑variance (torch reduction)
    out_flat = out.permute(0, 2, 3, 1).contiguous().view(R, H)  # (R, H)
    out_fp32 = out_flat.float()
    mu = out_fp32.mean(dim=1)                     # (R,)
    var = out_fp32.var(dim=1, unbiased=False)    # (R,)
    rstd = torch.rsqrt(var + eps)                # (R,)

    # 6. Final LayerNorm + out_gate + projection (all‑GPU GEMMs)
    gate_flat = out_gate.permute(0, 2, 3, 1).contiguous().view(R, H)  # (R, H)

    # pre‑multiply to_out weight with LN parameters
    w_out_T = weights["to_out.weight"].to(torch.float16).t().contiguous()   # (H, dim)
    weight_g = (weights["to_out_norm.weight"][:, None].to(torch.float16) * w_out_T)   # (H, dim)
    weight_b = (weights["to_out_norm.bias"][:, None].to(torch.float16) * w_out_T)     # (H, dim)

    # term 1 : (out * gate) @ weight_g
    term1 = torch.matmul((out_flat * gate_flat).to(torch.float32), weight_g.to(torch.float32))  # (R, dim)

    # term 2 : gate @ weight_g
    term2 = torch.matmul(gate_flat.to(torch.float32), weight_g.to(torch.float32))               # (R, dim)

    # combine LN scaling
    final = rstd.unsqueeze(-1) * (term1 - mu.unsqueeze(-1) * term2)   # (R, dim)

    final = final + term3

    return final
```

assistantfinal**Solution Overview**  
The outgoing TriMul operator consists of the following steps  

1. **Row‑wise LayerNorm** – normalises the input pair representation.  
2. **Fused linear projection** – a single matrix‑multiply produces the five
   projections (`left`, `right`, three gates).  
3. **Split + mask + sigmoid** – a Triton kernel separates the five
   projections, applies the optional mask and the sigmoid gates.  
4. **Core einsum** – the pair‑wise contraction `out[b,h,i,j] = Σₖ left[b,h,i,k]·right[b,h,j,k]`
   is performed with a batched GEMM (`torch.bmm`).  
5. **Row‑wise statistics** – mean and variance of `out` over the hidden
   dimension are obtained by a fast `torch.mean/var` reduction.  
6. **Final LayerNorm, gating and projection** – the remaining three
   matrix‑multiplications are expressed as standard GEMMs
   (`torch.matmul`).  This eliminates the second Triton kernel and
   avoids a second read of `out`.  

All heavy compute is performed in FP16, accumulations are in FP32, and the
final tensor is returned in FP32.

```python
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 3️⃣ Split, mask, apply sigmoids and store left/right/out_gate tensors
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,      # [R, 5·H]  FP16 fused projection
    left_ptr,      # [B, H, N, N] FP16
    right_ptr,     # [B, H, N, N] FP16
    out_gate_ptr,  # [B, H, N, N] FP16
    mask_ptr,      # [R] FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)           # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional pair‑mask (broadcast over hidden dimension)
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        m_val = m_val[:, None]                                      # [BLOCK_R,1]
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N
    # base address of the first hidden slice for this (b,i,k)
    base = b * (H * hidden_stride) + i * N + k

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # layout of proj: [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # sigmoid (computed in FP32 then cast back)
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * m_val * left_gate
        right = right_raw * m_val * right_gate
        outg  = out_gate

        dst = base[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst, left,  mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst, right, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst, outg, mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
#  Custom Triton‑accelerated forward pass
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only implementation of the *outgoing* TriMul operator (AlphaFold‑3).

    Steps
    -----
    1️⃣  Row‑wise LayerNorm on the input pair tensor (FP32 → FP16).
    2️⃣  Single fused linear projection producing left/right and three gates.
    3️⃣  Triton kernel ‑ split the projection, apply optional pair‑mask
        and the sigmoid gates → `left`, `right`, `out_gate`  (FP16).
    4️⃣  Batched GEMM (`torch.bmm`) implements the core einsum.
    5️⃣  Compute mean / variance of the contracted tensor with a fast
        reduction (`torch.mean/var`) – no extra kernel.
    6️⃣  Fuse the second LayerNorm, the gate multiplication and the final
        projection using three dense GEMMs (cuBLAS).  This avoids a second
        read of the huge tensor `out` and is fully GPU‑accelerated.
    Returns a float‑32 tensor of shape `[B, N, N, dim]`.
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = bool(cfg.get("nomask", True))
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N               # number of pairwise tokens (rows)
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (use the highly‑optimised torch kernel)
    # ------------------------------------------------------------------
    # Normalise over the last dimension `dim` and cast to FP16.
    x_norm = torch.nn.functional.layer_norm(
        x,
        (dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    ).to(torch.float16)                       # [B, N, N, dim] (FP16)

    # ------------------------------------------------------------------
    # 2️⃣ Fuse the five linear maps into one GEMM
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # (5·H, dim)

    proj = torch.nn.functional.linear(x_norm.view(R, dim), fused_w)  # (R, 5·H) FP16
    del x_norm

    # ------------------------------------------------------------------
    # 3️⃣ Split, mask, sigmoid → left / right / out_gate (FP16)
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if not nomask and mask is not None:
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Core contraction – batched GEMM
    # ------------------------------------------------------------------
    left_mat  = left.reshape(B * H, N, N)                     # FP16
    right_mat = right.reshape(B * H, N, N)                    # FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))      # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                               # [B, H, N, N] FP16

    # ------------------------------------------------------------------
    # 5️⃣ Row‑wise mean & rsqrt‑variance (torch reduction – fast)
    # ------------------------------------------------------------------
    # bring the hidden dimension to the last axis: (B, N, N, H) → (R, H)
    out_flat = out.permute(0, 2, 3, 1).contiguous().view(R, H)   # (R, H) FP16
    out_fp32 = out_flat.float()
    mu   = out_fp32.mean(dim=1)                     # (R,)
    var  = out_fp32.var(dim=1, unbiased=False)      # (R,)
    rstd = torch.rsqrt(var + eps)                   # (R,)

    # ------------------------------------------------------------------
    # 6️⃣ Fuse second LayerNorm, out_gate and final projection
    # ------------------------------------------------------------------
    gate_flat = out_gate.permute(0, 2, 3, 1).contiguous().view(R, H)   # (R, H) FP16

    # combine the final linear weight with the LayerNorm parameters
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # (H, dim)
    weight_g = (w["to_out_norm.weight"][:, None].to(torch.float16) * w_out_T)   # γ·W   (H, dim)
    weight_b = (w["to_out_norm.bias"][:, None].to(torch.float16)   * w_out_T)   # β·W   (H, dim)

    # term 1 : (out * gate) · (γ·W)
    term1 = torch.matmul((out_flat * gate_flat).to(torch.float32),
                         weight_g.to(torch.float32))                    # (R, dim)

    # term 2 : gate · (γ·W)
    term2 = torch.matmul(gate_flat.to(torch.float32),
                         weight_g.to(torch.float32))                    # (R, dim)

    # Apply the second LayerNorm and the out‑gate
    final = rstd.unsqueeze(-1) * (term1 - mu.unsqueeze(-1) * term2)       # (R, dim)

    # term 3 : gate · (β·W)
    term3 = torch.matmul(gate_flat.to(torch.float32),
                         weight_b.to(torch.float32))                    # (R, dim)

    final = final + term3

    # reshape back to `[B, N, N, dim]`
    final = final.view(B, N, N, dim)     # float32

    return final
```
