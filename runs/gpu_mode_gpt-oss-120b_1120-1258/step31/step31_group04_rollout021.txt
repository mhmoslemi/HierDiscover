Algorithm
---------
1️⃣  LayerNorm on the last dimension of the input tensor (FP32 → FP16) – done with

2️⃣  Fuse the five linear maps (left_proj, right_proj, left_gate, right_gate,

4️⃣  Batched GEMM (cuBLAS) implements the core einsum:

    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    # H ≤ 128 ⇒ one iteration is enough
    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)        # [BLOCK_K]
        mask_k = offs_k < H

        ptr = out_ptr + base[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)      # [BLOCK_M, BLOCK_K]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_m, mean,  mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)


# ---------------------------------------------------------------
# 6️⃣ Fuse second LayerNorm, out_gate & final projection
# ---------------------------------------------------------------
@triton.jit
def final_output_kernel(
    out_ptr,        # [B, H, N, N] FP16
    gate_ptr,       # [B, H, N, N] FP16 (out_gate)
    mu_ptr,         # [R] FP32
    rstd_ptr,       # [R] FP32
    ln_w_ptr,       # [H] FP32 (to_out_norm.weight)
    ln_b_ptr,       # [H] FP32 (to_out_norm.bias)
    w_out_T_ptr,    # [H, dim] FP16 (to_out weight transposed)
    out_final_ptr,  # [B, N, N, dim] FP32
    R, H, N, dim,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr,
):
    pid_m = tl.program_id(0)   # rows (R)
    pid_n = tl.program_id(1)   # output‑dim tiles

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)   # [BLOCK_M]
    offs_n = col_start + tl.arange(0, BLOCK_N)   # [BLOCK_N]

    mask_m = offs_m < R
    mask_n = offs_n < dim

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + j   # start of hidden plane 0

    # per‑row statistics (pre‑computed)
    mu   = tl.load(mu_ptr   + offs_m, mask=mask_m, other=0.0)   # [BLOCK_M]
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)   # [BLOCK_M]

    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    # H is ≤ 128 ⇒ a single iteration over the hidden dimension
    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)               # [BLOCK_K]
        mask_k = offs_k < H

        # load contracted tensor and gating values
        ptr_out  = out_ptr  + base[:, None] + offs_k[None, :] * hidden_stride
        ptr_gate = gate_ptr + base[:, None] + offs_k[None, :] * hidden_stride

        vals_out = tl.load(ptr_out,
                           mask=mask_m[:, None] & mask_k[None, :],
                           other=0.0).to(tl.float32)   # [M,K]
        vals_gate = tl.load(ptr_gate,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)   # [M,K]

        # LayerNorm (second LN)
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [K]
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [K]

        norm = ((vals_out - mu[:, None]) * rstd[:, None]) * gamma[None, :] + beta[None, :]
        term = norm * vals_gate                                   # [M,K] (FP32)

        # final linear projection: (M,K) × (K,dim_tile) → (M,dim_tile)
        w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]   # [K, TILE_N]
        w_tile = tl.load(w_ptr,
                         mask=mask_k[:, None] & mask_n[None, :],
                         other=0.0)                                 # [K,N]

        acc += tl.dot(term.to(tl.float16), w_tile).to(tl.float32)

    # store final tensor (flattened first 3 dims → row “R”)
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, acc,
             mask=mask_m[:, None] & mask_n[None, :])


# ---------------------------------------------------------------
# Entry point
# ---------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul operator (no gradients).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"            (all float32)
            - config_dict must contain
                "dim" (int), "hidden_dim" (int), "nomask" (bool, optional)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = bool(cfg.get("nomask", True))
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                    # number of pairwise tokens
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16) – use torch implementation
    # --------------------------------------------------------------
    x_norm_fp32 = torch.nn.functional.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim] FP32
    x_norm = x_norm_fp32.to(torch.float16)               # FP16 for the rest
    del x_norm_fp32

    # --------------------------------------------------------------
    # 2️⃣ Fuse five linear maps (left, right, left_gate, right_gate, out_gate)
    # --------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # [5·H, dim]

    x_flat = x_norm.view(R, dim)            # [R, dim] (contiguous)
    proj = F.linear(x_flat, fused_w)        # [R, 5·H] FP16
    del fused_w, x_flat

    # --------------------------------------------------------------
    # 3️⃣ Split, mask, apply sigmoid gates → left/right/out_gate (FP16)
    # --------------------------------------------------------------
    proj = proj.view(R, 5, H)               # [R, 5, H]

    if not nomask and mask is not None:
        mask_flat = mask.to(torch.float16).reshape(R, 1)  # [R,1]
    else:
        mask_flat = torch.ones(R, 1, dtype=torch.float16, device=device)

    # raw projections
    left_raw      = proj[:, 0, :]            # [R, H]
    right_raw     = proj[:, 1, :]
    left_gate_raw = proj[:, 2, :]
    right_gate_raw = proj[:, 3, :]
    out_gate_raw  = proj[:, 4, :]

    # sigmoid gates (done in FP32 for better accuracy)
    left_gate  = torch.sigmoid(left_gate_raw.float()).to(torch.float16)
    right_gate = torch.sigmoid(right_gate_raw.float()).to(torch.float16)
    out_gate   = torch.sigmoid(out_gate_raw.float()).to(torch.float16)

    # apply mask and gates
    left  = (left_raw * mask_flat) * left_gate      # [R, H]
    right = (right_raw * mask_flat) * right_gate    # [R, H]

    # reshape to 4‑D tensors [B, H, N, N]
    left = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    out_gate = out_gate.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    del proj, left_raw, right_raw, left_gate_raw, right_gate_raw, out_gate_raw

    # --------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension (cuBLAS)
    # --------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)                     # (B·H, N, N)
    right_mat = right.view(B * H, N, N)                    # (B·H, N, N)
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B·H, N, N) FP16
    out = out.view(B, H, N, N)                            # [B, H, N, N] FP16

    # --------------------------------------------------------------
    # 5️⃣ Row‑wise mean & rsqrt‑variance of `out`
    # --------------------------------------------------------------
    mu   = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 256          # rows per program
    BLOCK_K_STATS = 128          # H ≤ 128 ⇒ whole hidden dim
    grid_stats = (triton.cdiv(R, BLOCK_M_STATS),)
    rowwise_stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R, H, N,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm + out_gate + final projection (fused)
    # --------------------------------------------------------------
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim]

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 128   # rows processed per program
    BLOCK_K_OUT = 128   # full hidden dimension (H ≤ 128)
    BLOCK_N_OUT = 128   # output‑dim tile

    grid_out = (
        triton.cdiv(R, BLOCK_M_OUT),
        triton.cdiv(dim, BLOCK_N_OUT),
    )
    final_output_kernel[grid_out](
        out,
        out_gate,
        w["to_out_norm.weight"],   # γ  FP32
        w["to_out_norm.bias"],     # β  FP32
        w_out_T,
        final,
        R, H, N, dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=8,
    )

    return final
