"""
Outgoing TriMul (AlphaFold‑3) – forward‑only Triton implementation.

Algorithm
---------
1️⃣ LayerNorm over the channel dimension (dim) using the built‑in torch
   LayerNorm (fast cudnn kernel) → result cast to float16.
2️⃣ Fuse the five linear projections (left, right, left_gate,
   right_gate, out_gate) into a single GEMM:  (R, dim) × (5·H, dim)ᵀ.
3️⃣ Triton kernel `split_fuse_gate_kernel`:
   - decode the flat pair index (b,i,k) from the row index r,
   - apply the optional pair‑mask to *left* and *right*,
   - apply sigmoid to the three gate tensors,
   - write the three tensors `left`, `right` and `out_gate`
     in layout [B, H, N, N] (FP16).
4️⃣ Batched GEMM (cuBLAS) for the core einsum:
       out[b, h, i, j] = Σₖ left[b, h, i, k]·right[b, h, j, k]
   → executed as torch.bmm on (B·H, N, N) tensors (FP16).
5️⃣ Compute per‑pair (b,i,j) mean and variance of `out` over the
   hidden dimension H with a pure‑torch reduction (FP32 → rstd).
6️⃣ Triton kernel `final_output_kernel` fuses:
   - the second LayerNorm (γ/β),
   - multiplication by `out_gate`,
   - the final linear projection (H → dim).
   The kernel works on the flattened pair index R and tiles the
   output‑feature dimension.  All inner‑loop arithmetic uses Tensor
   Cores (FP16·FP16 → FP32) and the result is stored as FP32.

The implementation keeps the heavy memory‑bound steps (masking &
splitting, final LayerNorm + projection) in Triton kernels while
leveraging highly‑optimized PyTorch primitives for LayerNorm,
linear projection and the hidden‑dimension GEMM.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 3️⃣ Split, mask & apply sigmoid gates (Triton)
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # [R, 5·H]  FP16 fused projection
    left_ptr,          # [B, H, N, N] FP16
    right_ptr,         # [B, H, N, N] FP16
    out_gate_ptr,      # [B, H, N, N] FP16
    mask_ptr,          # [R]      FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)           # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional pair‑mask (broadcast over H)
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)  # [BLOCK_R]
        m_val = m_val[:, None]                                      # [BLOCK_R,1]
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # Strides inside the destination tensors
    hidden_stride = N * N                       # elements per hidden plane
    dst_base = b * (H * hidden_stride) + i * N + k   # start of H=0 plane

    # Loop over hidden dimension tiles
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # layout in proj: [left|right|lg|rg|og] each of size H
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(proj_ptr + proj_base,
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0)
        right_raw = tl.load(proj_ptr + proj_base + H,
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0)
        left_gate_raw = tl.load(proj_ptr + proj_base + 2 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_gate_raw = tl.load(proj_ptr + proj_base + 3 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        out_gate_raw = tl.load(proj_ptr + proj_base + 4 * H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)

        # sigmoid (more accurate in FP32)
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * m_val * left_gate
        right = right_raw * m_val * right_gate
        outg  = out_gate

        # destination offsets (flattened)
        dst = dst_base[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst, right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst, outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 6️⃣ Final LayerNorm + out_gate + projection (Triton)
# ----------------------------------------------------------------------
@triton.jit
def final_output_kernel(
    out_ptr,        # [B, H, N, N] FP16
    gate_ptr,       # [B, H, N, N] FP16   (out_gate)
    mu_ptr,         # [R] FP32   mean over H
    rstd_ptr,       # [R] FP32   rsqrt(var+eps)
    ln_w_ptr,       # [H] FP32   to_out_norm.weight
    ln_b_ptr,       # [H] FP32   to_out_norm.bias
    w_out_T_ptr,    # [H, dim] FP16  (to_out weight transposed)
    out_final_ptr,  # [B, N, N, dim] FP32
    R, H, N, dim,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr,
):
    pid_m = tl.program_id(0)   # rows (flattened pair index)
    pid_n = tl.program_id(1)   # output‑dim tiles

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)   # [BLOCK_M]
    offs_n = col_start + tl.arange(0, BLOCK_N)   # [BLOCK_N]

    mask_m = offs_m < R
    mask_n = offs_n < dim

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + j   # start of H=0 plane

    # per‑row statistics (already computed)
    mu   = tl.load(mu_ptr   + offs_m, mask=mask_m, other=0.0)   # [BLOCK_M]
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)   # [BLOCK_M]

    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)               # [BLOCK_K]
        mask_k = offs_k < H

        # load hidden‑plane values and gate
        ptr_out  = out_ptr  + base[:, None] + offs_k[None, :] * hidden_stride
        ptr_gate = gate_ptr + base[:, None] + offs_k[None, :] * hidden_stride

        vals_out = tl.load(ptr_out,
                           mask=mask_m[:, None] & mask_k[None, :],
                           other=0.0).to(tl.float32)   # [M,K]
        vals_gate = tl.load(ptr_gate,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)   # [M,K]

        # LayerNorm on hidden dimension (γ/β are per‑hidden)
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [K]
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [K]

        norm = ((vals_out - mu[:, None]) * rstd[:, None]) * gamma[None, :] + beta[None, :]
        term = norm * vals_gate                         # [M,K] (FP32)

        # final projection weight slice  (H, dim) -> (K, TILE_N)
        w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]   # [K,N]
        w_tile = tl.load(w_ptr,
                         mask=mask_k[:, None] & mask_n[None, :],
                         other=0.0)                                 # [K,N]

        # accumulate (M,K)·(K,N) → (M,N) using Tensor‑Core dot
        acc += tl.dot(term.to(tl.float16), w_tile).to(tl.float32)

    # write final tensor (flattened 3‑D -> row “R”)
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, acc,
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul operator (no gradients).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"            (all float32)
            - config_dict must contain
                "dim" (int), "hidden_dim" (int), "nomask" (bool)
    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = bool(cfg.get("nomask", True))
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                      # number of pairwise tokens
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm over channel dimension (fast torch kernel)
    # --------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()          # [R, dim] float32
    x_norm_fp32 = F.layer_norm(
        x_flat,
        normalized_shape=(dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    )
    x_norm = x_norm_fp32.to(torch.float16)           # [R, dim] FP16
    del x_flat, x_norm_fp32

    # --------------------------------------------------------------
    # 2️⃣ Fuse five linear projections (left, right, left_gate,
    #    right_gate, out_gate) into a single GEMM.
    # --------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                               # [5·H, dim]

    proj = F.linear(x_norm, fused_w)                  # (R, 5·H) FP16
    del fused_w, x_norm

    # --------------------------------------------------------------
    # 3️⃣ Split, apply mask, sigmoid gates → left/right/out_gate
    # --------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if not nomask and mask is not None:
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy

    BLOCK_R = 64   # rows per program (tune for occupancy)
    BLOCK_H = 64   # hidden tiles per program
    grid_split = (triton.cdiv(R, BLOCK_R),)

    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R, H, N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣ Core contraction (batched GEMM over hidden dim)
    # --------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)               # (B·H, N, N) FP16
    right_mat = right.view(B * H, N, N)              # (B·H, N, N) FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B·H, N, N) FP16
    out = out.view(B, H, N, N)                       # [B, H, N, N] FP16

    # --------------------------------------------------------------
    # 5️⃣ Compute per‑pair mean / rsqrt‑variance over hidden dim H
    #    (pure‑torch reduction, very fast on H100)
    # --------------------------------------------------------------
    out_fp32 = out.to(torch.float32)
    mu = out_fp32.mean(dim=1)                         # [B, N, N] FP32
    var = out_fp32.square().mean(dim=1) - mu * mu     # unbiased=False
    rstd = torch.rsqrt(var + eps)                     # [B, N, N] FP32

    mu_flat = mu.reshape(R).contiguous()
    rstd_flat = rstd.reshape(R).contiguous()
    del out_fp32, mu, var

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm + out_gate + final projection (Triton)
    # --------------------------------------------------------------
    # Transpose final weight for better memory coalescing
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim]

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 128   # rows per program (flattened pair index)
    BLOCK_K_OUT = 32    # hidden‑dim chunk
    BLOCK_N_OUT = 128   # output‑dim chunk

    grid_out = (
        triton.cdiv(R, BLOCK_M_OUT),
        triton.cdiv(dim, BLOCK_N_OUT),
    )
    final_output_kernel[grid_out](
        out,
        out_gate,
        mu_flat,
        rstd_flat,
        w["to_out_norm.weight"],   # γ  FP32
        w["to_out_norm.bias"],     # β  FP32
        w_out_T,
        final,
        R, H, N, dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=8,
    )
    # Clean up temporaries
    del out, out_gate, mu_flat, rstd_flat, w_out_T

    return final
