"""
Outgoing TriMul (AlphaFold‑3) – forward‑only implementation.

Algorithm
---------
1️⃣  Layer‑Norm over the last dimension of the input tensor (float32 → float16)
    using the built‑in torch.nn.functional.layer_norm (exactly matches the reference).

2️⃣  Fuse the five linear maps (left, right, left_gate, right_gate, out_gate)
    into a single GEMM:   X_norm @ W_fused.T → (R,5·H)   where
    R = B·N·N  (pairwise tokens)  and  H = hidden_dim.

3️⃣  Split the fused projection, apply the optional pair mask,
    compute the sigmoid gates and write the three tensors
    `left`, `right` and `out_gate` (shape [B, H, N, N], fp16)
    – all of this is done by a small Triton kernel.

4️⃣  Batched matrix‑multiply over the sequence dimension:
        out[b,h,i,j] = Σₖ left[b,h,i,k] · right[b,h,j,k]
    This is a regular torch.bmm on view‑shaped tensors (cuBLAS).

5️⃣  Layer‑Norm over the hidden dimension H of `out`
    (again using torch.nn.functional.layer_norm on fp16).

6️⃣  Element‑wise gate (out_gate) → (out_norm * out_gate).

7️⃣  Final linear projection to the model dimension `dim`:
        result = (out_norm * out_gate) @ W_out.T
    performed with torch.nn.functional.linear (Tensor‑core fast).

All heavy compute (step 3) is fused in Triton; the rest relies on
highly‑optimized cuBLAS / PyTorch kernels.  No gradients are required.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 3️⃣ Split the fused projection, mask & apply sigmoid gates (Triton)
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,      # [R, 5·H]  FP16 fused projection
    left_ptr,      # [B, H, N, N] FP16
    right_ptr,     # [B, H, N, N] FP16
    out_gate_ptr,  # [B, H, N, N] FP16
    mask_ptr,      # [R] FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)           # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional pair‑mask (broadcast on H)
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)  # [BLOCK_R]
        m_val = m_val[:, None]                                      # [BLOCK_R,1]
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N                     # stride of one hidden plane
    dst_base = b * (H * hidden_stride) + i * N + k   # base offset for h = 0

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # layout in proj: [left|right|lg|rg|og] each of size H
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw      = tl.load(proj_ptr + proj_base,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_raw     = tl.load(proj_ptr + proj_base + H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        left_gate_raw = tl.load(proj_ptr + proj_base + 2 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_gate_raw= tl.load(proj_ptr + proj_base + 3 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        out_gate_raw  = tl.load(proj_ptr + proj_base + 4 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)

        # sigmoid in FP32 → back to FP16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * m_val * left_gate
        right = right_raw * m_val * right_gate
        og    = out_gate

        # write to the destination tensors (flattened layout)
        dst = dst_base[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst,
                 og,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul operator (no gradients).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"            (all float32)
            - config_dict must contain
                "dim" (int), "hidden_dim" (int), "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = bool(cfg.get("nomask", True))
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                     # number of pairwise tokens
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm over the input (float32 → float16)
    # --------------------------------------------------------------
    # exact replica of nn.LayerNorm(dim) in the reference model
    x_norm = F.layer_norm(
        x,
        (dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps
    ).to(torch.float16)               # [B, N, N, dim] → fp16 for the linear maps

    # --------------------------------------------------------------
    # 2️⃣ Fuse the five linear maps into a single GEMM
    # --------------------------------------------------------------
    # Concatenate the five weight matrices (each: hidden_dim × dim)
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)               # [5·H, dim]

    # Flatten the tensor so that we can call F.linear once
    x_flat = x_norm.view(R, dim)       # [R, dim]
    proj = F.linear(x_flat, fused_w)   # [R, 5·H]
    del x_norm, x_flat, fused_w

    # --------------------------------------------------------------
    # 3️⃣ Split, mask and apply sigmoid gates → left, right, out_gate
    # --------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if not nomask and mask is not None:
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy tensor

    BLOCK_R = 64   # rows per program
    BLOCK_H = 64   # hidden‑dim tiles per program
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R, H, N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣ Batched GEMM over the sequence dimension (left @ rightᵀ)
    # --------------------------------------------------------------
    # left/right : [B, H, N, N] → (B·H, N, N)
    left_mat  = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] (fp16)
    out = out.view(B, H, N, N)                             # [B, H, N, N]

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm (over hidden dimension H)
    # --------------------------------------------------------------
    # reshape to (R, H) where the hidden dimension is the innermost axis
    out_flat = out.permute(0, 2, 3, 1).contiguous().view(R, H)          # fp16
    out_gate_flat = out_gate.permute(0, 2, 3, 1).contiguous().view(R, H)  # fp16

    # LayerNorm on the hidden dimension, matching the reference implementation
    out_norm = F.layer_norm(
        out_flat,
        (H,),
        weight=w["to_out_norm.weight"].to(torch.float16),
        bias=w["to_out_norm.bias"].to(torch.float16),
        eps=eps,
    )   # fp16, same dtype as input

    # --------------------------------------------------------------
    # 6️⃣ Apply out_gate and final linear projection
    # --------------------------------------------------------------
    gated = out_norm * out_gate_flat          # element‑wise product, fp16

    # final projection weight (dim × hidden_dim) – use fp16 for Tensor‑core GEMM
    w_out = w["to_out.weight"].to(torch.float16)   # [dim, H]

    # torch.nn.functional.linear does: out = gated @ w_out.T
    # (gated: [R, H], w_out.T: [H, dim]) → [R, dim]
    final = F.linear(gated, w_out)               # fp16

    # reshape back to [B, N, N, dim] and cast to fp32 (as required by the spec)
    final = final.view(B, N, N, -1).to(torch.float32)

    return final
