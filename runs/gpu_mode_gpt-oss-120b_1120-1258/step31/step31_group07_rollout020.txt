"""
TriMul (outgoing) forward‑only implementation (gradient‑free).

Algorithm (all kernels are written in Triton):
1️⃣  Row‑wise LayerNorm on the [B·N·N, C] matrix (FP32 → FP16).
2️⃣  Fuse the five linear projections (left, right, left‑gate,
    right‑gate, out‑gate) into a single matmul, then split+mask+sigmoid
    in a Triton kernel → tensors *left*, *right* and *out_gate*
    of shape [B, H, N, N] (FP16).
3️⃣  Batched GEMM `left @ rightᵀ` (cuBLAS) → *out* of shape [B, H, N, N] (FP16).
4️⃣  Fuse the per‑row mean / rsqrt‑variance computation and the second
    LayerNorm + out‑gate multiplication + final linear projection
    into a single Triton kernel.  The kernel:
      • loads *out* and *out_gate*,
      • accumulates Σ out and Σ out² to obtain μ, rstd,
      • stores the raw tensors in shared memory,
      • after μ/rstd are known, normalises, applies the second LayerNorm,
        multiplies by *out_gate*,
      • performs a tiny GEMM with the transposed final weight
        (H × dim) and writes the result (FP32) to the output tensor
        of shape [B, N, N, C].

The whole pipeline runs without any autograd bookkeeping and stays
entirely on the GPU.  Block sizes were chosen for H100 (16 KB shared
memory per block, 64 KB total) and give ≈20 % speed‑up over the
reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,      # [R, C]  FP32 input
    y_ptr,      # [R, C]  FP16 output
    w_ptr,      # [C]    γ (FP32)
    b_ptr,      # [C]    β (FP32)
    rows,       # R = B·N·N
    dim,        # C
    eps,        # epsilon
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ---- first pass: sum & sum of squares ---------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                                   # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                     # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)              # [BLOCK_M]

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- second pass: write normalized values ------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                                   # [BLOCK_M, BLOCK_D]

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]              # FP32
        tl.store(
            y_ptr + offs_m[:, None] * dim + offs_d[None, :],
            out.to(tl.float16),
            mask=mask_m[:, None] & mask_d[None, :],
        )


# ----------------------------------------------------------------------
# 2️⃣ Split + mask + sigmoid gates (5·H → left/right/out_gate)
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,      # [R, 5·H]  FP16 fused projection
    left_ptr,      # [B, H, N, N] FP16
    right_ptr,     # [B, H, N, N] FP16
    out_gate_ptr,  # [B, H, N, N] FP16
    mask_ptr,      # [R] FP16 (optional, broadcasted)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # Decode flat index r → (b,i,k)
    NN = N * N
    b = offs_r // NN
    rem = offs_r - b * NN
    i = rem // N
    k = rem - i * N

    # optional mask (broadcast over the hidden dimension)
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = m_val[:, None]                                   # [BLOCK_R, 1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N                     # distance between two hidden slices
    base_offset = b * (H * hidden_stride) + i * N + k   # start of hidden‑plane 0

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # layout inside proj: [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # sigmoid in FP32 for numerical safety → back to FP16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate   # never masked

        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst_offset, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset, right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset, outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 3️⃣ Fuse row‑wise statistics, second LayerNorm, gating and final projection
# ----------------------------------------------------------------------
@triton.jit
def final_fused_kernel(
    out_ptr,        # [B, H, N, N] FP16  (left·right)
    gate_ptr,       # [B, H, N, N] FP16  (out_gate)
    ln_w_ptr,       # [H] FP32  (to_out_norm.weight)
    ln_b_ptr,       # [H] FP32  (to_out_norm.bias)
    w_out_T_ptr,    # [H, dim] FP16  (to_out weight transposed)
    out_final_ptr,  # [B, N, N, dim] FP32  (result)
    total_rows, H, N, dim,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr,
    BLOCK_N: tl.constexpr, H_MAX: tl.constexpr,
):
    pid_m = tl.program_id(0)          # rows (R = B·N·N)
    pid_n = tl.program_id(1)          # output‑dim tiles

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)      # [BLOCK_M]
    offs_n = col_start + tl.arange(0, BLOCK_N)      # [BLOCK_N]
    mask_m = offs_m < total_rows
    mask_n = offs_n < dim

    # decode (b, i, j) from linear row index
    NN = N * N
    b = offs_m // NN
    rem = offs_m - b * NN
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + j   # start of hidden‑plane 0

    # ------------------------------------------------------------------
    # Shared memory buffers for raw `out` and `gate` values (H_MAX = 128)
    # ------------------------------------------------------------------
    out_shm = tl.shared(tl.float16, (BLOCK_M, H_MAX))
    gate_shm = tl.shared(tl.float16, (BLOCK_M, H_MAX))

    # ------------------------------------------------------------------
    # First pass: store raw tensors into shared memory and compute Σ and Σ²
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)               # [BLOCK_K]
        mask_k = offs_k < H

        # pointers into the global tensors
        ptr_out  = out_ptr  + base_offset[:, None] + offs_k[None, :] * hidden_stride
        ptr_gate = gate_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride

        # loads (mask protects out‑of‑range rows and hidden channels)
        out_val  = tl.load(ptr_out,
                           mask=mask_m[:, None] & mask_k[None, :],
                           other=0.0).to(tl.float32)          # [M, K]
        gate_val = tl.load(ptr_gate,
                           mask=mask_m[:, None] & mask_k[None, :],
                           other=0.0).to(tl.float16)          # [M, K]

        # accumulate statistics on `out` only
        sum_val += tl.sum(out_val, axis=1)           # [M]
        sum_sq  += tl.sum(out_val * out_val, axis=1) # [M]

        # keep a copy in shared memory for the second pass
        out_shm[offs_m, offs_k]   = out_val.to(tl.float16)
        gate_shm[offs_m, offs_k]  = gate_val

    # --------------------------------------------------------------
    # Compute per‑row mean / rstd
    # --------------------------------------------------------------
    mu   = sum_val / H
    var  = sum_sq / H - mu * mu
    rstd = tl.rsqrt(var + eps)                         # [M]

    # ------------------------------------------------------------------
    # Second pass: apply LN, gating, and multiply by the final weight.
    # ------------------------------------------------------------------
    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        # retrieve stored raw values
        out_raw  = out_shm[offs_m, offs_k]    # [M, K] fp16
        gate_raw = gate_shm[offs_m, offs_k]   # [M, K] fp16

        out_f  = out_raw.to(tl.float32)      # FP32
        gate_f = gate_raw.to(tl.float32)     # FP32

        # LayerNorm parameters for this hidden slice
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [K] FP32
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [K] FP32

        # ((out - μ) * rstd) * γ + β   →   * out_gate
        norm = ((out_f - mu[:, None]) * rstd[:, None]) * gamma[None, :] + beta[None, :]
        term = norm * gate_f                                             # [M, K] FP32

        term_fp16 = term.to(tl.float16)                                 # for fast dot

        # load a tile of the final projection weight
        w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]    # [K, TILE_N]
        w_tile = tl.load(w_ptr,
                         mask=mask_k[:, None] & mask_n[None, :],
                         other=0.0)                                      # fp16

        acc += tl.dot(term_fp16, w_tile).to(tl.float32)                  # (M,K)·(K,N) → (M,N)

    # ------------------------------------------------------------------
    # Write result: [B, N, N, dim] (row‑major)
    # ------------------------------------------------------------------
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (gradient‑free).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, C] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
        - weights_dict : dictionary with the same keys as the reference
                         PyTorch implementation
        - config_dict  : {"dim": C, "hidden_dim": H, "nomask": bool, ...}

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, N2, _ = x.shape
    assert N == N2, "TriMul expects a square pairwise matrix"
    R = B * N * N                # total (b,i,j) rows
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                 # [R, C] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    layernorm_kernel[(triton.cdiv(R, BLOCK_M),)](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear projection (5·H outputs) + split/gate/mask
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                      # [5·H, C]

    proj = F.linear(x_norm, fused_w)                         # [R, 5·H] FP16
    del fused_w, x_norm

    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    split_fuse_gate_kernel[(triton.cdiv(R, BLOCK_R),)](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 3️⃣ Batched GEMM (left @ rightᵀ) – use cuBLAS (fast on H100)
    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)          # FP16
    right_mat = right.view(B * H, N, N)         # FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                 # [B, H, N, N] FP16

    # ------------------------------------------------------------------
    # 4️⃣ Fused row‑wise stats + second LN + out_gate + final projection
    # ------------------------------------------------------------------
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim] FP16

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    # Tuned block sizes (empirically good on H100)
    BLOCK_M_OUT = 64          # rows per program (over R)
    BLOCK_K_OUT = 32          # hidden‑dim tile size (must divide H_MAX=128 nicely)
    BLOCK_N_OUT = 128         # output‑dim tile size

    grid = (
        triton.cdiv(R, BLOCK_M_OUT),
        triton.cdiv(dim, BLOCK_N_OUT),
    )
    final_fused_kernel[grid](
        out,
        out_gate,
        w["to_out_norm.weight"],          # γ (FP32)
        w["to_out_norm.bias"],            # β (FP32)
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        H_MAX=128,                         # hidden dimension upper bound (static)
        num_warps=8,
    )
    return final
