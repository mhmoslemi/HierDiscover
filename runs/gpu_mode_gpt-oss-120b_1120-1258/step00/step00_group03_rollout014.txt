We'll implement a simple batched matmul kernel:

```
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch_stride_a, batch_stride_b, batch_stride_c,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # compute program ids
    pid = tl.program_id(0)
    num_pid_m = tl.cdiv(M, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)
    batch = pid // (num_pid_m * num_pid_n)
    pid_rem = pid % (num_pid_m * num_pid_n)
    pid_m = pid_rem // num_pid_n
    pid_n = pid_rem % num_pid_n

    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask out-of-bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # ptrs for A and B
    a_ptrs = a_ptr + batch * batch_stride_a + offs_m[:, None] * stride_am + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    b_ptrs = b_ptr + batch * batch_stride_b + tl.arange(0, BLOCK_K)[:, None] * stride_bk + offs_n[None, :] * stride_bn

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # load
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (k * BLOCK_K + tl.arange(0, BLOCK_K) < K), other=0.0)
        b = tl.load(b_ptrs, mask=(k * BLOCK_K + tl.arange(0, BLOCK_K)[:, None] < K) & mask_n[None, :], other=0.0)
        # compute dot
        acc += tl.dot(a, b)  # a: (BM x BK), b: (BK x BN) -> (BM x BN)
        # increment pointers
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # write back
    c_ptrs = c_ptr + batch * batch_stride_c + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])
```

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config['dim']
    hidden_dim = config['hidden_dim']
    eps = 1e-5
    device = input_tensor.device

    # Get weights
    norm_weight = weights['norm.weight']
    norm_bias = weights['norm.bias']
    left_proj_weight = weights['left_proj.weight']
    right_proj_weight = weights['right_proj.weight']
    left_gate_weight = weights['left_gate.weight']
    right_gate_weight = weights['right_gate.weight']
    out_gate_weight = weights['out_gate.weight']
    to_out_norm_weight = weights['to_out_norm.weight']
    to_out_norm_bias = weights['to_out_norm.bias']
    to_out_weight = weights['to_out.weight']
    
    # LayerNorm on input dim
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), norm_weight, norm_bias, eps=eps)
    
    # Linear projections (no bias)
    left = torch.nn.functional.linear(x, left_proj_weight)  # ... hidden_dim
    right = torch.nn.functional.linear(x, right_proj_weight)

    # Apply mask if present (config may have nomask)
    if not config.get('nomask', True):
        # mask shape [bs, N, N]; broadcast to hidden_dim
        mask_exp = mask.unsqueeze(-1)  # (B,N,N,1)
        left = left * mask_exp
        right = right * mask_exp

    # Gates
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()
    
    left = left * left_gate
    right = right * right_gate

    # Convert to float16 for matmul
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    B, N, _, H = left.shape
    # reshape to batched matrices
    left_flat = left.permute(0,3,1,2).reshape(B*H, N, N)
    right_flat = right.permute(0,3,1,2).reshape(B*H, N, N)
    # transposed view for right
    right_t = right_flat.transpose(1,2).contiguous()

    out_flat = torch.empty_like(left_flat)

    # compute strides
    a_ptr = left_flat
    b_ptr = right_t
    c_ptr = out_flat

    stride_ak = a_ptr.stride(2)
    stride_bk = b_ptr.stride(1)
    stride_bn = b_ptr.stride(2)
    stride_cm = c_ptr.stride(1)
    stride_cn = c_ptr.stride(2)

    batch_stride_a = a_ptr.stride(0)
    batch_stride_b = b_ptr.stride(0)
    batch_stride_c = c_ptr.stride(0)

    import triton
    import triton.language as tl

    @triton.jit
    def batched_matmul_kernel(
        a_ptr, b_ptr, c_ptr,
        batch_stride_a, batch_stride_b, batch_stride_c,
        stride_am, stride_ak,
        stride_bk, stride_bn,
        stride_cm, stride_cn,
        M, N, K,
        BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
    ):
        pid = tl.program_id(0)
        num_pid_m = tl.cdiv(M, BLOCK_M)
        num_pid_n = tl.cdiv(N, BLOCK_N)
        batch = pid // (num_pid_m * num_pid_n)
        pid_rem = pid % (num_pid_m * num_pid_n)
        pid_m = pid_rem // num_pid_n
        pid_n = pid_rem % num_pid_n

        offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

        mask_m = offs_m < M
        mask_n = offs_n < N

        # batch base pointers
        a_batch_ptr = a_ptr + batch * batch_stride_a
        b_batch_ptr = b_ptr + batch * batch_stride_b
        c_batch_ptr = c_ptr + batch * batch_stride_c

        acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

        for k in range(0, tl.cdiv(K, BLOCK_K)):
            offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
            # mask for k range
            mask_k = offs_k < K

            a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_am) + (offs_k[None, :] * stride_ak)
            b_ptrs = b_batch_ptr + (offs_k[:, None] * stride_bk) + (offs_n[None, :] * stride_bn)

            a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
            b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

            acc += tl.dot(a, b)  # (BLOCK_M, BLOCK_K) x (BLOCK_K, BLOCK_N)

        c_ptrs = c_batch_ptr + (offs_m[:, None] * stride_cm) + (offs_n[None, :] * stride_cn)
        tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

    # Determine grid
    # Total blocks = batch * ceil(M/BM) * ceil(N/BN)
    M = N
    N_val = N
    K = N

    total_blocks = (B*H) * ((N + 64 - 1)//64) * ((N + 64 - 1)//64)
    # Use grid as a 1D launch
    grid = (total_blocks,)

    batched_matmul_kernel[grid](
        a_ptr, b_ptr, c_ptr,
        batch_stride_a, batch_stride_b, batch_stride_c,
        stride_am, stride_ak,
        stride_bk, stride_bn,
        stride_cm, stride_cn,
        M, N_val, K,
        BLOCK_M=64, BLOCK_N=64, BLOCK_K=32
    )

    # reshape back
    out = out_flat.reshape(B, H, N, N).permute(0,2,3,1).to(torch.float32)

    out = torch.nn.functional.layer_norm(out, (hidden_dim,), to_out_norm_weight, to_out_norm_bias, eps=eps)

    # Multiply by out_gate
    out = out * out_gate

    # Final linear
    to_out_weight = to_out_weight
    out = torch.nn.functional.linear(out, to_out_weight)

    return out

import torch
import triton
import triton.language as tl

@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch_stride_a, batch_stride_b, batch_stride_c,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)
    num_pid_m = tl.cdiv(M, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)
    batch = pid // (num_pid_m * num_pid_n)
    pid_rem = pid % (num_pid_m * num_pid_n)
    pid_m = pid_rem // num_pid_n
    pid_n = pid_rem % num_pid_n

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch
    a_batch_ptr = a_ptr + batch * batch_stride_a
    b_batch_ptr = b_ptr + batch * batch_stride_b
    c_batch_ptr = c_ptr + batch * batch_stride_c

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_am) + (offs_k[None, :] * stride_ak)
        b_ptrs = b_batch_ptr + (offs_k[:, None] * stride_bk) + (offs_n[None, :] * stride_bn)

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)

    c_ptrs = c_batch_ptr + (offs_m[:, None] * stride_cm) + (offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])
    

def custom_kernel(data):
    """
    Fast forward implementation of the TriMul "outgoing" operator.
    The heavy N×N×N reduction is performed with a custom Triton kernel
    that computes a batched matrix multiplication
        out[b, i, j, h] = sum_k left[b, i, k, h] * right[b, j, k, h]
    for each hidden dimension h. All other operations (layer‑norms,
    linear projections and gating) are done with PyTorch.
    """
    # Unpack inputs
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # -----------------------------------------------------------------
    # 1. LayerNorm on the input tensor (dim axis)
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device)
    norm_bias = weights["norm.bias"].to(device)
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, dim]

    # -----------------------------------------------------------------
    # 2. Linear projections (no bias)
    # -----------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"].to(device)
    right_proj_weight = weights["right_proj.weight"].to(device)

    left = torch.nn.functional.linear(x, left_proj_weight)   # [B,N,N,hidden]
    right = torch.nn.functional.linear(x, right_proj_weight) # [B,N,N,hidden]

    # -----------------------------------------------------------------
    # 3. Optional mask (mask shape: [B,N,N])
    # -----------------------------------------------------------------
    if not config.get("nomask", True) and mask is not None:
        mask = mask.to(dtype).unsqueeze(-1)  # broadcast over hidden dim
        left = left * mask
        right = right * mask

    # -----------------------------------------------------------------
    # 4. Compute gating scalars and apply
    # -----------------------------------------------------------------
    left_gate_weight = weights["left_gate.weight"].to(device)
    right_gate_weight = weights["right_gate.weight"].to(device)
    out_gate_weight = weights["out_gate.weight"].to(device)

    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Batched matrix multiplication via Triton
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # -----------------------------------------------------------------
    B, N, _, H = left.shape  # B=batch, N=seq_len, H=hidden_dim

    # Move hidden dimension to leading batch axis – a contiguous view is required
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    # Collapse the (B, H) dimensions so each hidden‑channel matrix is a
    # separate batch for the kernel.
    left_flat = left_perm.view(B * H, N, N)   # [B*H, N, N]
    right_flat = right_perm.view(B * H, N, N) # [B*H, N, N]
    # We need the second operand transposed: (N,N) -> (N,N) with axes (k,j)
    right_t = right_flat.transpose(1, 2).contiguous()  # [B*H, N, N]

    # Allocate output buffer
    out_flat = torch.empty_like(left_flat)

    # Strides (in elements) – all tensors are now contiguous
    stride_am = left_flat.stride(1)  # rows stride
    stride_ak = left_flat.stride(2)  # columns stride (should be 1)
    stride_bk = right_t.stride(1)   # rows stride of B (k)
    stride_bn = right_t.stride(2)   # cols stride of B (j)
    stride_cm = out_flat.stride(1)
    stride_cn = out_flat.stride(2)

    batch_stride_a = left_flat.stride(0)
    batch_stride_b = right_t.stride(0)
    batch_stride_c = out_flat.stride(0)

    # Kernel launch configuration
    total_batches = B * H
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = lambda meta: (
        total_batches
        * triton.cdiv(N, meta["BLOCK_M"])
        * triton.cdiv(N, meta["BLOCK_N"]),
    )

    batched_matmul_kernel[grid](
        left_flat,
        right_t,
        out_flat,
        batch_stride_a,
        batch_stride_b,
        batch_stride_c,
        stride_am,
        stride_ak,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        N,
        N,
        N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to the original layout
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1)  # [B,N,N,hidden]

    # -----------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate, final projection
    # -----------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"].to(device)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device)

    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    out = out * out_gate

    to_out_weight = weights["to_out.weight"].to(device)
    out = torch.nn.functional.linear(out, to_out_weight)  # final shape [B,N,N,dim]

    return out

"""
Fast forward implementation of the “outgoing” TriMul operator.

The heavy N×N×N reduction
    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
is expressed as a batched matrix multiplication (one matrix per hidden
channel) and executed with a custom Triton kernel. All other
operations (LayerNorm, linear projections and gating) are performed with
PyTorch for simplicity and correctness.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch_stride_a, batch_stride_b, batch_stride_c,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute many independent matrix products:
        C_batch = A_batch @ B_batch.T
    where each batch has shape (M, K) and (K, N) respectively.
    The kernel handles the batch dimension by flattening it into the
    program id space.
    """
    pid = tl.program_id(0)                     # 1‑D launch grid
    num_pid_m = tl.cdiv(M, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)

    batch = pid // (num_pid_m * num_pid_n)      # which matrix in the batch
    pid_rem = pid % (num_pid_m * num_pid_n)
    pid_m = pid_rem // num_pid_n                # block row
    pid_n = pid_rem % num_pid_n                 # block col

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for the selected batch
    a_batch = a_ptr + batch * batch_stride_a
    b_batch = b_ptr + batch * batch_stride_b
    c_batch = c_ptr + batch * batch_stride_c

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        a_ptrs = a_batch + (offs_m[:, None] * stride_am) + (offs_k[None, :] * stride_ak)
        b_ptrs = b_batch + (offs_k[:, None] * stride_bk) + (offs_n[None, :] * stride_bn)

        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)                 # (BM×BK)·(BK×BN) → (BM×BN)

    c_ptrs = c_batch + (offs_m[:, None] * stride_cm) + (offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” module using a Triton kernel.
    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
            input_tensor : torch.Tensor, shape [B, N, N, dim]
            mask_tensor  : torch.Tensor, shape [B, N, N] (may be None)
            weights_dict : mapping of weight names to torch.Tensors
            config_dict  : contains 'dim', 'hidden_dim', 'nomask' (bool)
    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # -----------------------------------------------------------------
    # unpack arguments
    # -----------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (normalises over the last dimension = dim)
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device)
    norm_bias   = weights["norm.bias"].to(device)
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, dim]

    # -----------------------------------------------------------------
    # 2. Linear projections (no bias)
    # -----------------------------------------------------------------
    left_proj_weight  = weights["left_proj.weight"].to(device)   # [hidden, dim]
    right_proj_weight = weights["right_proj.weight"].to(device)  # [hidden, dim]

    left  = torch.nn.functional.linear(x, left_proj_weight)   # [B, N, N, hidden]
    right = torch.nn.functional.linear(x, right_proj_weight)  # [B, N, N, hidden]

    # -----------------------------------------------------------------
    # 3. Optional mask (mask shape [B, N, N])
    # -----------------------------------------------------------------
    if (not config.get("nomask", True)) and mask is not None:
        mask = mask.to(dtype).unsqueeze(-1)       # [B, N, N, 1]
        left  = left * mask
        right = right * mask

    # -----------------------------------------------------------------
    # 4. Gating scalars (sigmoid outputs) and application
    # -----------------------------------------------------------------
    left_gate_weight  = weights["left_gate.weight"].to(device)
    right_gate_weight = weights["right_gate.weight"].to(device)
    out_gate_weight   = weights["out_gate.weight"].to(device)

    left_gate  = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate   = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    left  = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Batched matrix multiplication via Triton
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # -----------------------------------------------------------------
    B, N, _, H = left.shape   # B=batch, N=seq_len, H=hidden_dim

    # Bring the hidden dimension to the leading batch axis and make the
    # data contiguous – this yields a tensor of shape (B*H, N, N)
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    left_flat  = left_perm.view(B * H, N, N)             # [B*H, N, N]
    right_flat = right_perm.view(B * H, N, N)            # [B*H, N, N]

    # Right operand must be transposed on the K axis (k ↔ j)
    right_t = right_flat.transpose(1, 2).contiguous()   # [B*H, N, N]

    # Allocate output buffer
    out_flat = torch.empty_like(left_flat)               # [B*H, N, N]

    # Strides for the contiguous tensors (in element units)
    stride_am = left_flat.stride(1)      # = N   (row stride)
    stride_ak = left_flat.stride(2)      # = 1   (col stride)
    stride_bk = right_t.stride(1)        # = N
    stride_bn = right_t.stride(2)        # = 1
    stride_cm = out_flat.stride(1)
    stride_cn = out_flat.stride(2)

    batch_stride_a = left_flat.stride(0)  # = N*N
    batch_stride_b = right_t.stride(0)
    batch_stride_c = out_flat.stride(0)

    # -----------------------------------------------------------------
    # Kernel launch configuration
    # -----------------------------------------------------------------
    total_batches = B * H
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = lambda meta: (
        total_batches
        * ((N + meta["BLOCK_M"] - 1) // meta["BLOCK_M"])
        * ((N + meta["BLOCK_N"] - 1) // meta["BLOCK_N"]),
    )

    batched_matmul_kernel[grid](
        left_flat,
        right_t,
        out_flat,
        batch_stride_a,
        batch_stride_b,
        batch_stride_c,
        stride_am,
        stride_ak,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        N,            # M
        N,            # N
        N,            # K
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, hidden]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1)   # [B, N, N, hidden]

    # -----------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate, final linear projection
    # -----------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"].to(device)
    to_out_norm_bias   = weights["to_out_norm.bias"].to(device)

    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    out = out * out_gate                                 # element‑wise gate

    to_out_weight = weights["to_out.weight"].to(device)   # [dim, hidden]
    out = torch.nn.functional.linear(out, to_out_weight)  # [B, N, N, dim]

    return out
