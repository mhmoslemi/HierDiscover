    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_k, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid0 = tl.program_id(0) # Batch*Hidden id
    pid1 = tl.program_id(1) # block row (i)
    pid2 = tl.program_id(2) # block col (j)

    # compute batch and hidden
    b = pid0 // H
    h = pid0 % H

    # offsets
    i = pid1 * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid2 * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_j = j < N

"""
TriMul (outgoing) forward implementation with a fused Triton kernel.

Algorithm
---------
1. Layer‑norm the input tensor `x` over the last dimension (dim).
2. Linear projections (no bias) → left_proj, right_proj  (dim → hidden_dim).
3. Optional pairwise mask (shape [B, N, N]) is applied to both projections.
4. Compute three sigmoid‑gates (left, right, out) from the same normalized `x`.
5. Fuse the gates with the projections:
       left  = left_proj  * left_gate
       right = right_proj * right_gate
6. Cast `left` and `right` to FP16 and evaluate the core O(N³) operation:
       out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
   This is a batch of H independent matrix‑multiplications of size N×N.
   The reduction is performed by a custom Triton kernel (`trimul_kernel`)
   which loops over the K‑axis in blocks and accumulates in FP32.
7. Layer‑norm the result over the hidden dimension, multiply by the output
   gate, and project back to the original dimension (dim) with a final linear.
8. Return the tensor of shape [B, N, N, dim] (float32).

The heavy O(N³) contraction is the only part written in Triton; all
other linear / activation / norm steps use PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: computes the pairwise multiplication + sum over k.
#   out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
# ----------------------------------------------------------------------
@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,               # pointers
    B, N, H,                                    # problem sizes
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,   # left strides
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,  # right strides
    stride_out_b, stride_out_i, stride_out_k, stride_out_h,        # out strides
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # ------------------------------------------------------------------
    # Program IDs:
    #   pid0 = batch * hidden index (range B*H)
    #   pid1 = block row   (i)
    #   pid2 = block col   (j)
    # ------------------------------------------------------------------
    pid0 = tl.program_id(0)
    pid1 = tl.program_id(1)
    pid2 = tl.program_id(2)

    # Resolve batch `b` and hidden channel `h`
    b = pid0 // H
    h = pid0 % H

    # Offsets for the i / j tiles
    i = pid1 * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid2 * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for out‑of‑bounds rows / columns
    mask_i = i < N
    mask_j = j < N
    mask_ij = mask_i[:, None] & mask_j[None, :]

    # Base pointers for the selected (b, h)
    left_base  = left_ptr  + b * stride_left_b  + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base   = out_ptr   + b * stride_out_b   + h * stride_out_h

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction axis k
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for block_k in range(0, num_k_blocks):
        k = block_k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # ------------------------------------------------------------------
        # Load a BLOCK_M × BLOCK_K tile from left  (i, k)
        # Load a BLOCK_K × BLOCK_N tile from right (k, j)
        # ------------------------------------------------------------------
        left_ptrs = left_base + i[:, None] * stride_left_i + k[None, :] * stride_left_k
        right_ptrs = right_base + k[:, None] * stride_right_k + j[None, :] * stride_right_i

        left_tile = tl.load(left_ptrs,
                            mask=mask_i[:, None] & mask_k[None, :],
                            other=0.0)               # FP16
        right_tile = tl.load(right_ptrs,
                             mask=mask_k[:, None] & mask_j[None, :],
                             other=0.0)              # FP16

        # FP16 × FP16 → accumulate in FP32
        acc += tl.dot(left_tile, right_tile)

    # ------------------------------------------------------------------
    # Write the result (FP32) back to global memory
    # ------------------------------------------------------------------
    out_ptrs = out_base + i[:, None] * stride_out_i + j[None, :] * stride_out_k
    tl.store(out_ptrs, acc, mask=mask_ij)


# ----------------------------------------------------------------------
# Entry point used by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, D] (float32)
        - mask        : torch.Tensor [B, N, N] (float32 or bool) or None
        - weights     : dict of torch.Tensor containing learned parameters
        - config      : dict with at least 'dim' and 'hidden_dim'

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, D] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # LayerNorm on the input (over the last dimension)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_weight  = weights["left_proj.weight"]      # (hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"]     # (hidden_dim, dim)

    left  = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # Optional mask (broadcasted on the hidden dimension)
    # ------------------------------------------------------------------
    if mask is not None:
        # ensure mask is float and on the same device
        mask_f = mask.to(dtype=dtype, device=device).unsqueeze(-1)   # [B,N,N,1]
        left  = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # Gating networks (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate_weight   = weights["left_gate.weight"]
    right_gate_weight  = weights["right_gate.weight"]
    out_gate_weight    = weights["out_gate.weight"]

    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))   # [B,N,N,hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))

    # ------------------------------------------------------------------
    # Fuse gates into the projected tensors
    # ------------------------------------------------------------------
    left  = left  * left_gate
    right = right * right_gate

    # Cast to half‑precision for the O(N³) kernel (memory & bandwidth)
    left_h  = left.to(torch.float16).contiguous()
    right_h = right.to(torch.float16).contiguous()

    # ------------------------------------------------------------------
    # Allocate output buffer for the kernel (float32 accumulator)
    # ------------------------------------------------------------------
    out_h = torch.empty_like(left_h, dtype=torch.float32, device=device)

    # ------------------------------------------------------------------
    # Strides (required by the Triton kernel)
    # ------------------------------------------------------------------
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left_h.stride()
    stride_right_b, stride_right_i, stride_right_k, stride_right_h = right_h.stride()
    stride_out_b, stride_out_i, stride_out_k, stride_out_h = out_h.stride()

    # ------------------------------------------------------------------
    # Kernel launch configuration
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B * hidden_dim,                         # batch * hidden channel
        (N + BLOCK_M - 1) // BLOCK_M,           # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,           # j‑tiles
    )

    trimul_kernel[grid](
        left_h,
        right_h,
        out_h,
        B,
        N,
        hidden_dim,
        stride_left_b,
        stride_left_i,
        stride_left_k,
        stride_left_h,
        stride_right_b,
        stride_right_i,
        stride_right_k,
        stride_right_h,
        stride_out_b,
        stride_out_i,
        stride_out_k,
        stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # Output LayerNorm, gating, and final projection
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]          # (dim, hidden_dim)

    # Layer‑norm over the hidden dimension
    out = F.layer_norm(out_h, (hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias)

    # Apply the output gate (converted to float32)
    out = out * out_gate.to(torch.float32)

    # Final linear projection back to `dim`
    out = F.linear(out, to_out_weight)   # [B, N, N, dim]

    return out
```

The code follows the reference PyTorch implementation while off‑loading the dominant O(N³) contraction to a custom Triton kernel. The kernel processes a batch‑hidden slice (`b, h`) as an independent matrix multiplication, loops over the reduction axis `k` in blocks, accumulates in FP32, and writes the result back. All other operations (layer‑norms, linear layers, sigmoids, masking) are performed with PyTorch which already runs at near‑optimal speed on H100. This satisfies the requirement of implementing *part* of the operator in Triton and should meet the target runtime on the provided test configurations.
