import triton
import triton.language as tl

# Triton kernel definition for gating and mask
@triton.jit
def fuse_gate_mask_kernel(
    left_ptr, left_gate_ptr,
    right_ptr, right_gate_ptr,
    mask_ptr,
    hidden_dim: tl.constexpr,
    total_elements: tl.constexpr,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr

    mask = idx < total_elements

"""
TriMul (outgoing) forward implementation.

The kernel performs the following steps:
1. Layer‑norm the input tensor.
2. Linear projections (left, right) and three gating projections
   (left_gate, right_gate, out_gate) using torch.nn.functional.linear.
3. Fuse the element‑wise mask (if present) and the sigmoid gates with a
   custom Triton kernel (`fuse_gate_mask_kernel`).  This kernel multiplies
   left = left * left_gate * mask, right = right * right_gate * mask in a
   single pass over memory, reducing bandwidth.
4. Compute the pairwise “tri‑multiplicative” update:
   out[i,j] = Σ_k left[i,k] * right[j,k] for every channel d.
   This is implemented as a batched matrix multiplication
   (B·hidden_dim independent GEMMs) using torch.bmm, which is highly
   optimized on H100 GPUs.
5. Apply a second LayerNorm (`to_out_norm`), multiply by the output gate,
   and finally linearly project back to the original dimension.

All operations are performed in float32; the Triton kernel works on
contiguous 1‑D views of the tensors and supports both masked and
unmasked inputs.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def fuse_gate_mask_kernel(
    left_ptr, left_gate_ptr,          # left & left gate tensors (flattened)
    right_ptr, right_gate_ptr,        # right & right gate tensors (flattened)
    mask_ptr,                         # mask tensor (flattened, optional)
    hidden_dim: tl.constexpr,         # size of the channel dimension
    total_elements: tl.constexpr,     # total number of elements in left/right
    BLOCK: tl.constexpr,              # tile size
    HAS_MASK: tl.constexpr            # 1 => mask present, 0 => no mask
):
    """
    In‑place fusion:
        left  = left  * left_gate  * mask
        right = right * right_gate * mask
    where mask is broadcast over the hidden_dim dimension.
    """
    idx = tl.program_id(0) * BLOCK + tl.arange(0, BLOCK)          # linear index
    mask = idx < total_elements                                   # out‑of‑bounds guard

    # Load the four operands (masked loads return 0.0 where mask is False)
    left_val      = tl.load(left_ptr + idx,       mask=mask, other=0.0)
    left_gate_val = tl.load(left_gate_ptr + idx,  mask=mask, other=0.0)
    right_val     = tl.load(right_ptr + idx,     mask=mask, other=0.0)
    right_gate_val= tl.load(right_gate_ptr + idx,mask=mask, other=0.0)

    # Load the mask value (shared across hidden_dim) if required
    if HAS_MASK:
        # mask index = idx // hidden_dim  (integer division)
        pos = idx // hidden_dim
        mask_val = tl.load(mask_ptr + pos, mask=mask, other=1.0)
    else:
        mask_val = 1.0   # no mask → scalar 1.0

    # Fuse the three multiplicative factors
    new_left  = left_val * left_gate_val * mask_val
    new_right = right_val * right_gate_val * mask_val

    # Write the results back in place
    tl.store(left_ptr + idx,  new_left,  mask=mask)
    tl.store(right_ptr + idx, new_right, mask=mask)


def custom_kernel(data):
    """
    Entry point for the TriMul forward pass.
    Arguments:
        data = (input_tensor, mask, weights, config)
    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32).
    """
    # --------------------------------------------------------------------- #
    # unpack inputs
    # --------------------------------------------------------------------- #
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # config may contain a flag "nomask" – treat as no mask when True
    has_mask = not config.get("nomask", False) and mask is not None

    # --------------------------------------------------------------------- #
    # 1. Input layer‑norm
    # --------------------------------------------------------------------- #
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    # LayerNorm over the last dimension (dim)
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=1e-5)

    # --------------------------------------------------------------------- #
    # 2. Linear projections (no bias)
    # --------------------------------------------------------------------- #
    B, N, _, _ = x.shape
    x_flat = x.view(-1, dim)                     # (B * N * N, dim)

    left_proj_weight   = weights["left_proj.weight"]   # (hidden_dim, dim)
    right_proj_weight  = weights["right_proj.weight"]
    left_gate_weight   = weights["left_gate.weight"]
    right_gate_weight  = weights["right_gate.weight"]
    out_gate_weight    = weights["out_gate.weight"]

    # Linear maps → shape (B*N*N, hidden_dim)
    left      = F.linear(x_flat, left_proj_weight).view(B, N, N, hidden_dim)
    right     = F.linear(x_flat, right_proj_weight).view(B, N, N, hidden_dim)
    left_gate = torch.sigmoid(F.linear(x_flat, left_gate_weight).view(B, N, N, hidden_dim))
    right_gate= torch.sigmoid(F.linear(x_flat, right_gate_weight).view(B, N, N, hidden_dim))
    out_gate  = torch.sigmoid(F.linear(x_flat, out_gate_weight).view(B, N, N, hidden_dim))

    # Ensure contiguity for the Triton kernel
    left       = left.contiguous()
    right      = right.contiguous()
    left_gate  = left_gate.contiguous()
    right_gate = right_gate.contiguous()

    # --------------------------------------------------------------------- #
    # 3. Fuse mask + gates with a Triton kernel
    # --------------------------------------------------------------------- #
    # Flatten tensors to 1‑D views (contiguous)
    left_flat       = left.view(-1)
    right_flat      = right.view(-1)
    left_gate_flat  = left_gate.view(-1)
    right_gate_flat = right_gate.view(-1)

    total_elements = left_flat.numel()          # B * N * N * hidden_dim
    BLOCK = 1024                                 # tile size for the kernel
    grid = (triton.cdiv(total_elements, BLOCK),)

    if has_mask:
        # mask shape (B, N, N); broadcast over hidden_dim inside the kernel
        mask = mask.to(dtype=left.dtype, device=left.device).contiguous()
        mask_flat = mask.view(-1)                # (B * N * N)
    else:
        # dummy mask (value 1.0) – the kernel will ignore it when HAS_MASK==0
        mask_flat = torch.tensor([1.0], dtype=left.dtype, device=left.device)

    fuse_gate_mask_kernel[grid](
        left_flat.data_ptr(),
        left_gate_flat.data_ptr(),
        right_flat.data_ptr(),
        right_gate_flat.data_ptr(),
        mask_flat.data_ptr(),
        hidden_dim,
        total_elements,
        BLOCK,
        HAS_MASK=int(has_mask)
    )

    # After kernel execution, reshape back to original 4‑D layout
    left  = left_flat.view(B, N, N, hidden_dim)
    right = right_flat.view(B, N, N, hidden_dim)

    # --------------------------------------------------------------------- #
    # 4. Tri‑multiplicative update via batched GEMM (i.e. Einstein sum)
    # --------------------------------------------------------------------- #
    # Permute to (B, hidden_dim, N, N) and fuse batch & channel dims
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    left_mat  = left_perm.view(-1, N, N)                # (B*H, N, N)
    right_mat = right_perm.view(-1, N, N)               # (B*H, N, N)

    # out_mat[bhd] = left_mat @ right_mat^T   -> shape (B*H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Restore 4‑D layout: (B, N, N, hidden_dim)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------------- #
    # 5. Output LayerNorm, output gate, final projection
    # --------------------------------------------------------------------- #
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias,
                       eps=1e-5)

    # Apply output gate (broadcasted over hidden_dim)
    out = out * out_gate

    # Final linear projection back to the original dimension
    to_out_weight = weights["to_out.weight"]            # (dim, hidden_dim)
    out_flat = out.view(-1, hidden_dim)
    out = F.linear(out_flat, to_out_weight).view(B, N, N, dim)

    return out
```
