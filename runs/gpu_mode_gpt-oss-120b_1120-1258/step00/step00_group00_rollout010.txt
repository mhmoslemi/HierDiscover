def custom_kernel(data):
    """
    Triton-accelerated forward pass for the Triangle Multiplicative Update (outgoing) operation.
    This implementation follows the AlphaFold3 TriMul module:
    1. LayerNorm on input.
    2. Linear projections (left/right) and gating.
    3. Optional pairwise mask.
    4. Core O(N^3) operation: for each hidden dimension h,
       out[b,i,j,h] = sum_k left[b,i,k,h] * right[b,j,k,h].
       Implemented as a batched GEMM using a custom Triton kernel.
    5. LayerNorm + output gating.
    6. Final linear projection back to input dimension.
    """
    import torch
    import triton
    import triton.language as tl

    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # Unpack weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]
    
    # 1. LayerNorm on input
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # 2. Projections
    left_proj = torch.nn.functional.linear(x, left_proj_weight)  # [B,N,N,H]
    right_proj = torch.nn.functional.linear(x, right_proj_weight)

    # 3. Optional mask
    if mask_tensor is not None and mask_tensor.numel() > 0:
        mask = mask_tensor.unsqueeze(-1).to(left_proj.dtype)
        left_proj = left_proj * mask
        right_proj = right_proj * mask

    # 4. Gates
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    left = left_proj * left_gate
    right = right_proj * right_gate

    # Cast to half for Triton kernel (saves bandwidth)
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    B, N, _, _ = left_fp16.shape

    # Allocate output tensor for the intermediate hidden representation
    out_inter = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=input_tensor.device)

    # Triton kernel
    @triton.jit
    def tri_mul_out_kernel(
        left_ptr, right_ptr, out_ptr,
        B, N, H,
        stride_left_batch, stride_left_i, stride_left_k, stride_left_h,
        stride_right_batch, stride_right_j, stride_right_k, stride_right_h,
        stride_out_batch, stride_out_i, stride_out_j, stride_out_h,
        BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
    ):
        # Compute program ids -------------------------------------------------
        pid0 = tl.program_id(0)  # combined (batch*head*block_i)
        pid1 = tl.program_id(1)  # block_j
        # Number of blocks along M dimension
        num_blocks_m = tl.cdiv(N, BLOCK_M)

        bh_idx = pid0 // num_blocks_m               # index into (batch, head)
        block_i = pid0 % num_blocks_m                # block row index
        block_j = pid1                               # block column index

        # Decode batch and head (hidden) indices
        b = bh_idx // H
        h = bh_idx % H

        # Compute base offsets for each tensor
        left_base = left_ptr + b * stride_left_batch + h * stride_left_h
        right_base = right_ptr + b * stride_right_batch + h * stride_right_h
        out_base = out_ptr + b * stride_out_batch + h * stride_out_h

        # Compute start indices for the block
        i = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
        j = block_j * BLOCK_N + tl.arange(0, BLOCK_N)

        # Masks for row/col bounds
        mask_i = i < N
        mask_j = j < N

        # Initialize accumulator in fp32
        acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

        # Loop over k dimension
        num_blocks_k = tl.cdiv(N, BLOCK_K)
        for kk in range(num_blocks_k):
            k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
            mask_k = k < N

            # Load left tile: shape (BLOCK_M, BLOCK_K)
            a_ptrs = left_base + i[:, None] * stride_left_i + k[None, :] * stride_left_k
            a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
            a = a.to(tl.float16)

            # Load right tile transposed: shape (BLOCK_K, BLOCK_N)
            b_ptrs = right_base + k[:, None] * stride_right_k + j[None, :] * stride_right_j
            b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)
            b = b.to(tl.float16)

            # Fused dot product; tl.dot accumulates in fp32
            acc += tl.dot(a, b)

        # Write back results (only where both i and j are in bounds)
        c_ptrs = out_base + i[:, None] * stride_out_i + j[None, :] * stride_out_j
        tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

    # Choose block sizes (tuned for H100, half-precision)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Compute grid dimensions
    grid_m = (B * hidden_dim * ((N + BLOCK_M - 1) // BLOCK_M))
    grid_n = ((N + BLOCK_N - 1) // BLOCK_N)
    # Launch kernel
    tri_mul_out_kernel[(grid_m, grid_n)](
        left_fp16, right_fp16, out_inter,
        B, N, hidden_dim,
        left_fp16.stride(0), left_fp16.stride(1), left_fp16.stride(2), left_fp16.stride(3),
        right_fp16.stride(0), right_fp16.stride(1), right_fp16.stride(2), right_fp16.stride(3),
        out_inter.stride(0), out_inter.stride(1), out_inter.stride(2), out_inter.stride(3),
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )
    # Ensure kernel finishes
    torch.cuda.synchronize()

    # 5. Output layernorm
    out_norm = torch.nn.functional.layer_norm(out_inter, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    # 6. Apply output gate
    out = out_norm * out_gate

    # 7. Final linear projection back to dim
    output = torch.nn.functional.linear(out, to_out_weight)

    return output
```

"""
TriMul (outgoing) forward kernel – Triton‑accelerated core O(N³) update.

Algorithm
---------
1. Layer‑norm the input tensor `x`   →  shape [B, N, N, C].
2. Linear projections:
       left  = x @ W_left   (shape [B, N, N, H])
       right = x @ W_right  (shape [B, N, N, H])
3. Optional pairwise mask (broadcast over H).
4. Gating (sigmoid) on the *pre‑norm* input:
       left  *= sigmoid(x @ W_left_gate)
       right *= sigmoid(x @ W_right_gate)
   (gates are computed in FP32 for stability.)
5. Cast `left` and `right` to FP16 and compute the
   tensor‑wise multiplication
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   which is a batched GEMM: for each (b,h) we multiply
   the (N×N) matrix `left[b,:, :,h]` with the transpose of
   `right[b,:, :,h]`.  This step is performed by a custom
   Triton kernel (`tri_mul_out_kernel`) that iterates over
   the K‑dimension in tiles.
6. Layer‑norm over the hidden dimension, apply the output gate
   (sigmoid(x @ W_out_gate)), and a final linear projection back
   to the original channel size `C`.

The kernel works for arbitrary batch size (1‑2), sequence lengths
up to 1024 and hidden dimensions up to 128.  It executes in
FP16 for the heavy N³ work and accumulates in FP32, returning a
float‑32 tensor as required.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel for the core tensor‑wise multiplication (step 5)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_out_kernel(
    left_ptr,            # *half*   [B, N, N, H]
    right_ptr,           # *half*   [B, N, N, H]
    out_ptr,             # float32  [B, N, N, H]

    B, N, H,                         # problem sizes
    stride_l_batch, stride_l_i, stride_l_k, stride_l_h,
    stride_r_batch, stride_r_j, stride_r_k, stride_r_h,
    stride_o_batch, stride_o_i, stride_o_j, stride_o_h,

    # compile‑time tile sizes
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Computes, for every (b, h) pair:
        out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
    The computation is tiled in M (i), N (j) and K (k) dimensions.
    """
    pid0 = tl.program_id(0)      # combines (batch*head) and M‑tile
    pid1 = tl.program_id(1)      # N‑tile

    # number of tiles along the M dimension
    num_m_tiles = tl.cdiv(N, BLOCK_M)

    # decode batch, head and tile indices
    bh_idx   = pid0 // num_m_tiles               # (batch * head) index
    block_i  = pid0 %  num_m_tiles                # tile row index
    block_j  = pid1                               # tile column index

    b = bh_idx // H                # batch index
    h = bh_idx %  H                # hidden‑dim index

    # base pointers for the current (b, h) slice
    left_base  = left_ptr  + b * stride_l_batch + h * stride_l_h
    right_base = right_ptr + b * stride_r_batch + h * stride_r_h
    out_base   = out_ptr   + b * stride_o_batch + h * stride_o_h

    # start indices for the current tile
    i = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = block_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # bounds masks
    mask_i = i < N
    mask_j = j < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension in tiles
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for kk in range(num_k_tiles):
        k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # ------- load left tile (M x K) -------
        a_ptrs = left_base + i[:, None] * stride_l_i + k[None, :] * stride_l_k
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)               # a is FP16

        # ------- load right tile transposed (K x N) -------
        b_ptrs = right_base + k[:, None] * stride_r_k + j[None, :] * stride_r_j
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)               # b is FP16

        # fused dot: accumulates in FP32
        acc += tl.dot(a, b)

    # write back only the valid region
    c_ptrs = out_base + i[:, None] * stride_o_i + j[None, :] * stride_o_j
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[:, None])


# ----------------------------------------------------------------------
# Entry point called by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weight_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, C] (float32).
    """
    input_tensor, mask_tensor, weights, config = data

    dim        = config["dim"]            # C  (input channel dim)
    hidden_dim = config["hidden_dim"]    # H  (projection dim)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )   # shape [B, N, N, C]

    # ------------------------------------------------------------------
    # 2. Linear projections (float32)
    # ------------------------------------------------------------------
    left_proj  = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right_proj = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if mask_tensor is not None:
        # mask shape [B, N, N] → [B, N, N, 1] for broadcasting
        mask = mask_tensor.unsqueeze(-1).to(left_proj.dtype)
        left_proj  = left_proj * mask
        right_proj = right_proj * mask

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid) – computed in FP32 for stability
    # ------------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate   = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left  = left_proj  * left_gate   # shape [B, N, N, H]
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 5. Core O(N³) multiplication – Triton kernel
    # ------------------------------------------------------------------
    # Cast to half‑precision to feed the kernel
    left_fp16  = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()

    B, N, _, _ = left_fp16.shape   # underscores: second N and H dimensions

    # Allocate output for the hidden‑dim tensor (float32)
    out_hidden = torch.empty((B, N, N, hidden_dim),
                             dtype=torch.float32,
                             device=input_tensor.device)

    # Tile sizes – tuned for H100 / FP16
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Grid: (batch*head*ceil(N/BLOCK_M), ceil(N/BLOCK_N))
    grid_m = B * hidden_dim * ((N + BLOCK_M - 1) // BLOCK_M)
    grid_n = (N + BLOCK_N - 1) // BLOCK_N

    # Launch the kernel
    tri_mul_out_kernel[(grid_m, grid_n)](
        left_fp16,
        right_fp16,
        out_hidden,
        B, N, hidden_dim,
        left_fp16.stride(0), left_fp16.stride(1), left_fp16.stride(2), left_fp16.stride(3),
        right_fp16.stride(0), right_fp16.stride(1), right_fp16.stride(2), right_fp16.stride(3),
        out_hidden.stride(0), out_hidden.stride(1), out_hidden.stride(2), out_hidden.stride(3),
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=8   # good default for H100
    )
    # Wait for kernel completion
    torch.cuda.synchronize()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm + gating
    # ------------------------------------------------------------------
    out_norm = torch.nn.functional.layer_norm(
        out_hidden,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out_norm * out_gate   # element‑wise gate

    # ------------------------------------------------------------------
    # 7. Final linear projection back to C channels
    # ------------------------------------------------------------------
    output = torch.nn.functional.linear(out, weights["to_out.weight"])

    return output
```

gates) are performed with PyTorch which keeps the code simple yet competitive.
