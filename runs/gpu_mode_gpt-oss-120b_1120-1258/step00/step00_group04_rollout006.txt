"""
Trion‑accelerated implementation of the “outgoing” Triangle‑Multiplicative Update
(TriMul) from AlphaFold‑3.

The kernel fuses the N‑by‑N‑by‑hidden_dim batch matrix multiplication:
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

This is equivalent to a batched GEMM where each head (hidden dimension) has its
own matrix multiply (left @ rightᵀ).  The Triton kernel computes all heads in a
single launch, using 128×128 tiles and a 32‑element K‑tile.  All other linear
layers, layer‑norms and gating are done with PyTorch for simplicity.

The forward pass does **not** compute gradients.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _batch_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    BATCH, M, N, K,
    stride_ap, stride_ai, stride_ak,
    stride_bp, stride_bk, stride_bn,
    stride_cp, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute a batch of GEMMs:
        C[p, i, j] = Σ_k A[p, i, k] * B[p, j, k]   (B is accessed transposed)

    Parameters
    ----------
    a_ptr, b_ptr, c_ptr : pointers to the three tensors
    BATCH, M, N, K      : integer dimensions
    stride_*            : strides (in elements) for each tensor
    BLOCK_*             : compile‑time tile sizes
    """
    pid_batch = tl.program_id(2)   # which head / batch
    pid_m = tl.program_id(0)       # tile row
    pid_n = tl.program_id(1)       # tile column

    # offsets within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # accumulator in FP32 regardless of input dtype
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension in BLOCK_K chunks
    for k in range(0, K, BLOCK_K):
        cur_k = k + tl.arange(0, BLOCK_K)            # [BLOCK_K]
        mask_k = cur_k < K

        # Load a block of A : shape (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_ptr
            + pid_batch * stride_ap
            + offs_m[:, None] * stride_ai
            + cur_k[None, :] * stride_ak,
            mask=(offs_m < M)[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load a block of Bᵀ : shape (BLOCK_K, BLOCK_N)
        # B is stored as (batch, N, N); we treat it as transposed by swapping
        # its row‑/col‑strides.
        b = tl.load(
            b_ptr
            + pid_batch * stride_bp
            + cur_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & (offs_n < N)[None, :],
            other=0.0,
        )
        # Accumulate dot product (automatic up‑cast to FP32)
        acc += tl.dot(a, b)

    # Write the result back to C
    c_ptrs = (
        c_ptr
        + pid_batch * stride_cp
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc,
        mask=(offs_m < M)[:, None] & (offs_n < N)[None, :],
    )


def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator using a Triton‑accelerated
    batched matrix multiplication.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : [B, N, N, dim]  (float32)
        - mask_tensor   : [B, N, N] (bool / float) – element‑wise mask
        - weights_dict  : model parameters (see code for required keys)
        - config_dict   : {"dim": int, "hidden_dim": int, "nomask": bool, ...}

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """

    # ----------------------------------------------------------------------
    # Unpack inputs
    # ----------------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    eps = 1e-5  # layer‑norm epsilon

    # ----------------------------------------------------------------------
    # 1️⃣ Input LayerNorm
    # ----------------------------------------------------------------------
    # nn.LayerNorm over the last dimension (dim)
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]

    # ----------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # ----------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])      # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])    # [B, N, N, hidden_dim]

    # ----------------------------------------------------------------------
    # 3️⃣ Optional mask (applied before gating)
    # ----------------------------------------------------------------------
    if not nomask:
        # mask may be bool; cast to the same dtype as the tensors
        mask_f = mask.to(x.dtype).unsqueeze(-1)          # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ----------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid of linear on the *original* normed input)
    # ----------------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ----------------------------------------------------------------------
    # 5️⃣ Batched GEMM via Triton:
    #     out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #     This is equivalent to (left_h @ right_hᵀ) for every head h.
    # ----------------------------------------------------------------------
    # Rearrange to (B, H, N, N) and flatten the first two dims.
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    B, H, N, _ = left_perm.shape
    batch = B * H

    # Cast to fp16 for higher throughput; Triton accumulates in fp32.
    left_mat = left_perm.view(batch, N, N).to(torch.float16)
    right_mat = right_perm.view(batch, N, N).to(torch.float16)

    # Output buffer (fp16, will be up‑cast later)
    out_mat = torch.empty((batch, N, N), dtype=torch.float16, device=x.device)

    # Tile sizes – chosen to fit H100 tensor‑core resources.
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    # Compute launch grid: (tiles_M, tiles_N, batch)
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        batch,
    )

    # Strides (in elements) for a contiguous [batch, M, K] layout.
    stride_ap = N * N          # batch stride
    stride_ai = N              # row stride
    stride_ak = 1              # col stride

    # For B we want a transposed view: original stride(row)=N, stride(col)=1.
    stride_bp = N * N          # batch stride (same as A)
    stride_bk = 1              # row‑stride of the transposed view (original col)
    stride_bn = N              # col‑stride of the transposed view (original row)

    # Output strides (contiguous)
    stride_cp = N * N
    stride_ci = N
    stride_cj = 1

    _batch_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        batch,
        N,
        N,
        N,
        stride_ap,
        stride_ai,
        stride_ak,
        stride_bp,
        stride_bk,
        stride_bn,
        stride_cp,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Convert back to FP32 and restore original layout
    out = out_mat.to(torch.float32).view(B, H, N, N).permute(0, 2, 3, 1)  # [B, N, N, hidden_dim]

    # ----------------------------------------------------------------------
    # 6️⃣ Output layer‑norm, gating and final projection
    # ----------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])  # [B, N, N, dim]

    return out
