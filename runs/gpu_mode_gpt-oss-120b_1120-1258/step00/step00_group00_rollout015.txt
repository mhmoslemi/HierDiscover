    right_t = right.permute(0,3,2,1).contiguous() # (B, hidden, N, N)
    # flatten batch and hidden
    bh = bs * hidden_dim
    left_mat = left_t.view(bh, N, N)   # (B*hidden, N, N)
    right_mat = right_t.view(bh, N, N) # (B*hidden, N, N)
    out_mat = torch.bmm(left_mat, right_mat)  # (B*hidden, N, N)
    out = out_mat.view(bs, hidden_dim, N, N).permute(0,2,3,1)  # (B,N,N,hidden)

    # Apply to_out_norm
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    # apply out_gate
    out = out * out_gate

    # final linear to_out (weight shape [dim, hidden_dim]) - no bias
    to_out_weight = weights["to_out.weight"]
    # out shape (B,N,N,hidden), we want (B,N,N,dim)
    out = torch.nn.functional.linear(out, to_out_weight) # uses shape (out_features=dim, in_features=hidden)

    return out
```

"""
TriMul (outgoing) forward pass – highly optimized implementation.

Algorithm
---------
1. Layer‑norm the input tensor (shape [B,N,N,C]).
2. Fuse the five linear projections (left_proj, left_gate, right_proj,
   right_gate, out_gate) into a single mat‑mul by concatenating the
   weight matrices (shape (5*H, C)).  This yields five tensors of shape
   [B,N,N,H] after a single call to `torch.nn.functional.linear`.
3. Apply the sigmoid to the three gate tensors.
4. Apply the pairwise mask and the two element‑wise gates (`left_gate`,
   `right_gate`) inside a tiny Triton kernel – this fuses a few
   element‑wise operations and avoids an extra Python‑level pass.
5. For each hidden dimension `h` compute the pairwise product
   `out[...,i,j,h] = Σ_k left[...,i,k,h] * right[...,j,k,h]`.
   This is a batch of H independent matrix‑multiplications:
   we reshape `left`/`right` to (B*H, N, N) and call `torch.bmm`.
6. Layer‑norm the result (`to_out_norm`), multiply by `out_gate`,
   and a final linear projection back to the original dimension `C`.

Only the mask‑gate multiplication (step 4) is performed with a Triton
kernel; all heavy linear algebra uses highly‑optimized PyTorch kernels.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: apply mask and element‑wise gates to the projected tensors
# ----------------------------------------------------------------------
@triton.jit
def _apply_mask_gate_kernel(
    # tensors
    left_proj_ptr, left_gate_ptr, right_proj_ptr, right_gate_ptr,
    mask_ptr,
    left_out_ptr, right_out_ptr,
    # sizes
    B, N, H,
    # strides for the tensors (four dimensions: B, i, j, h)
    stride_lp_b, stride_lp_i, stride_lp_j, stride_lp_h,
    stride_lg_b, stride_lg_i, stride_lg_j, stride_lg_h,
    stride_rp_b, stride_rp_i, stride_rp_j, stride_rp_h,
    stride_rg_b, stride_rg_i, stride_rg_j, stride_rg_h,
    stride_m_b, stride_m_i, stride_m_j, stride_m_h,
    stride_lo_b, stride_lo_i, stride_lo_j, stride_lo_h,
    stride_ro_b, stride_ro_i, stride_ro_j, stride_ro_h,
    # compile‑time block size
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    total = B * N * N * H
    valid = offs < total

    # 1‑D index → (b,i,j,h)
    idx = offs
    h = idx % H
    idx = idx // H
    j = idx % N
    idx = idx // N
    i = idx % N
    b = idx // N

    # compute linear offsets for each tensor
    lp_off = b * stride_lp_b + i * stride_lp_i + j * stride_lp_j + h * stride_lp_h
    lg_off = b * stride_lg_b + i * stride_lg_i + j * stride_lg_j + h * stride_lg_h
    rp_off = b * stride_rp_b + i * stride_rp_i + j * stride_rp_j + h * stride_rp_h
    rg_off = b * stride_rg_b + i * stride_rg_i + j * stride_rg_j + h * stride_rg_h
    m_off  = b * stride_m_b  + i * stride_m_i  + j * stride_m_j  + h * stride_m_h
    lo_off = b * stride_lo_b + i * stride_lo_i + j * stride_lo_j + h * stride_lo_h
    ro_off = b * stride_ro_b + i * stride_ro_i + j * stride_ro_j + h * stride_ro_h

    # load values (mask is broadcasted along H → stride_m_h is 0)
    lp = tl.load(left_proj_ptr + lp_off, mask=valid)
    lg = tl.load(left_gate_ptr + lg_off, mask=valid)
    rp = tl.load(right_proj_ptr + rp_off, mask=valid)
    rg = tl.load(right_gate_ptr + rg_off, mask=valid)
    m  = tl.load(mask_ptr + m_off, mask=valid)

    # reference order: left = (proj * mask) * gate
    left_val  = lp * m
    left_val  = left_val * lg
    right_val = rp * m
    right_val = right_val * rg

    tl.store(left_out_ptr + lo_off, left_val, mask=valid)
    tl.store(right_out_ptr + ro_off, right_val, mask=valid)


def _apply_mask_gate(left_proj, left_gate, right_proj, right_gate, mask):
    """
    Run the Triton kernel that fuses mask multiplication and the two
    element‑wise gates.  Returns the gated tensors `left` and `right`,
    both of shape [B, N, N, H].
    """
    B, N, _, H = left_proj.shape
    device = left_proj.device
    dtype = left_proj.dtype

    # output buffers
    left_out = torch.empty_like(left_proj)
    right_out = torch.empty_like(right_proj)

    total_elements = B * N * N * H
    BLOCK_SIZE = 1024  # ~1 KB of work per program (tunable)

    grid = (triton.cdiv(total_elements, BLOCK_SIZE),)

    # gather strides
    lp_s = left_proj.stride()
    lg_s = left_gate.stride()
    rp_s = right_proj.stride()
    rg_s = right_gate.stride()
    m_s  = mask.stride()          # mask is (B,N,N,1) → stride_h == 0
    lo_s = left_out.stride()
    ro_s = right_out.stride()

    _apply_mask_gate_kernel[grid](
        left_proj, left_gate,
        right_proj, right_gate,
        mask,
        left_out, right_out,
        B, N, H,
        # strides for all tensors
        lp_s[0], lp_s[1], lp_s[2], lp_s[3],
        lg_s[0], lg_s[1], lg_s[2], lg_s[3],
        rp_s[0], rp_s[1], rp_s[2], rp_s[3],
        rg_s[0], rg_s[1], rg_s[2], rg_s[3],
        m_s[0], m_s[1], m_s[2], m_s[3],
        lo_s[0], lo_s[1], lo_s[2], lo_s[3],
        ro_s[0], ro_s[1], ro_s[2], ro_s[3],
        BLOCK_SIZE=BLOCK_SIZE,
    )
    return left_out, right_out


# ----------------------------------------------------------------------
# Main entry point – the custom Triton‑accelerated TriMul kernel
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, C] (C == config["dim"])
    """
    # --------------------------------------------------------------
    # unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]                     # input feature dimension C
    hidden_dim = config["hidden_dim"]      # projection dimension H

    B, N, _, _ = input_tensor.shape
    device = input_tensor.device
    dtype = input_tensor.dtype

    # --------------------------------------------------------------
    # 1. Input layer‑norm
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # shape [B,N,N,dim]

    # --------------------------------------------------------------
    # 2. Fuse five linear maps (proj + gate + out_gate)
    # --------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]      # (H, dim)
    right_proj_weight = weights["right_proj.weight"]    # (H, dim)
    left_gate_weight = weights["left_gate.weight"]      # (H, dim)
    right_gate_weight = weights["right_gate.weight"]    # (H, dim)
    out_gate_weight = weights["out_gate.weight"]        # (H, dim)

    # concatenate along the output‑feature dimension
    weight_concat = torch.cat(
        [
            left_proj_weight,
            left_gate_weight,
            right_proj_weight,
            right_gate_weight,
            out_gate_weight,
        ],
        dim=0,
    )  # shape (5*H, dim)

    # perform a single large linear transform
    x_flat = x.view(-1, dim)                     # (B*N*N, dim)
    proj_out = torch.nn.functional.linear(x_flat, weight_concat)  # (B*N*N, 5*H)
    proj_out = proj_out.view(B, N, N, 5, hidden_dim)  # (B,N,N,5,H)

    # slice the concatenated result
    left_proj   = proj_out[..., 0, :]   # (B,N,N,H)
    left_gate   = proj_out[..., 1, :]   # (B,N,N,H)
    right_proj  = proj_out[..., 2, :]   # (B,N,N,H)
    right_gate  = proj_out[..., 3, :]   # (B,N,N,H)
    out_gate    = proj_out[..., 4, :]   # (B,N,N,H)

    # --------------------------------------------------------------
    # 3. Apply sigmoid to the three gate tensors
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    # --------------------------------------------------------------
    # 4. Mask handling
    # --------------------------------------------------------------
    if config.get("nomask", False):
        mask = torch.ones(B, N, N, dtype=dtype, device=device)
    else:
        mask = mask_tensor.to(dtype)  # [B,N,N]
    # expand to broadcast over the hidden dimension (stride_h will be 0)
    mask_exp = mask.unsqueeze(-1)  # (B,N,N,1)

    # --------------------------------------------------------------
    # 5. Fuse mask + gate multiplication via Triton
    # --------------------------------------------------------------
    left, right = _apply_mask_gate(left_proj, left_gate, right_proj, right_gate, mask_exp)

    # --------------------------------------------------------------
    # 6. Pairwise multiplication: einsum → batched matmul per hidden dim
    # --------------------------------------------------------------
    # left:  (B,N,N,H)   -> (B,H,N,N)
    # right: (B,N,N,H)   -> (B,H,N,N) with second axes swapped
    left_t = left.permute(0, 3, 1, 2).contiguous()   # (B,H,N,N)
    right_t = right.permute(0, 3, 2, 1).contiguous() # (B,H,N,N)

    BH = B * hidden_dim
    left_mat = left_t.view(BH, N, N)    # (B*H, N, N)
    right_mat = right_t.view(BH, N, N)  # (B*H, N, N)

    out_mat = torch.bmm(left_mat, right_mat)        # (B*H, N, N)

    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # (B,N,N,H)

    # --------------------------------------------------------------
    # 7. Output layer‑norm, gating and final linear
    # --------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )  # (B,N,N,H)

    out = out * out_gate  # apply out gate

    # final projection back to input dimension
    to_out_weight = weights["to_out.weight"]   # (dim, hidden_dim)
    out = torch.nn.functional.linear(out, to_out_weight)  # (B,N,N,dim)

    return out
