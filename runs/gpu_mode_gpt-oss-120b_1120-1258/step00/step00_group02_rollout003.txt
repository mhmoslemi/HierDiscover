@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ab,
    stride_bk, stride_bn, stride_bb,
    stride_cm, stride_cn, stride_cb,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    EPS: tl.constexpr=0
):
    pid_m = tl.program_id(axis=0)
    pid_n = tl.program_id(axis=1)
    pid_b = tl.program_id(axis=2)

    # compute block start offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out-of-bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # pointer offsets for A and B
    a_ptrs = a_ptr + pid_b * stride_ab + offs_m[:, None] * stride_am + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    b_ptrs = b_ptr + pid_b * stride_bb + tl.arange(0, BLOCK_K)[:, None] * stride_bk + offs_n[None, :] * stride_bn

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # load blocks
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (tl.arange(0, BLOCK_K) < K - k*BLOCK_K), other=0.0) # shape [BLOCK_M, BLOCK_K]
        b = tl.load(b_ptrs, mask=(tl.arange(0, BLOCK_K) < K - k*BLOCK_K)[:, None] & mask_n[None, :], other=0.0) # shape [BLOCK_K, BLOCK_N]

        # dot product
        acc += tl.dot(a, b)

        # increment pointers
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # store result
    c_ptrs = c_ptr + pid_b * stride_cb + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B = x.shape[0]
    N = x.shape[1]

    # layernorm over last dim
    x_norm = torch.nn.functional.layer_norm(x, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=1e-5)

    # linear projections
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    # mask
    if mask is not None:
        # mask can be boolean, cast to float
        mask_f = mask.to(x.dtype).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f

    # gates
    left_gate = torch.nn.functional.linear(x_norm, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Preparation for kernel: permute and flatten
    # left: [B, N, N, H] -> [B, H, N, N]
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()

    batch_h = B * hidden_dim
    left_flat = left.view(batch_h, N, N)
    right_flat = right.view(batch_h, N, N)

    out_flat = torch.empty_like(left_flat)

    # select suitable block sizes based on N, hidden_dim
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    # grid
    grid = ( (N + BLOCK_M - 1) // BLOCK_M,
             (N + BLOCK_N - 1) // BLOCK_N,
             batch_h )
    # launch kernel
    tri_mul_gemm_kernel[grid](
        left_flat, right_flat, out_flat,
        N, N, N,
        left_flat.stride(1), left_flat.stride(2), left_flat.stride(0),
        right_flat.stride(2), right_flat.stride(1), right_flat.stride(0),
        out_flat.stride(1), out_flat.stride(2), out_flat.stride(0),
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )
    # reshape result to original layout
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1) # shape [B,N,N,H]

    out = out_normed * out_gate

    # final linear
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

"""
TriMul “outgoing” kernel (AlphaFold‑3 style).

The heavy N³ work is the contraction
    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
which is mathematically a batch of hidden_dim matrix‑multiplications:
    out_d = left_d @ right_dᵀ   (left_d shape N×N, right_dᵀ shape N×N).

The kernel below performs a fused batched GEMM for all (batch × hidden_dim)
slices without materialising the transposed right tensor.  The surrounding
operations (LayerNorms, Linear projections, gating, mask handling and the
final linear) are executed in PyTorch – they are cheap compared to the
N³ contraction.  The kernel works with arbitrary batch‑size, sequence length
and hidden dimension and falls back gracefully when dimensions are not a
multiple of the block sizes.

All tensors are expected to be on the GPU.
"""
import torch
import triton
import triton.language as tl
from torch.nn import functional as F


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the einsum contraction
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_gemm_kernel(
    a_ptr,                     # left  [B*H, N, N]  (i,k)
    b_ptr,                     # right [B*H, N, N]  (j,k) – we read it as (k,j)
    c_ptr,                     # out   [B*H, N, N]  (i,j)
    M, N, K,                   # all equal to sequence length
    stride_am, stride_ak, stride_ab,   # strides for A (row i, col k, batch)
    stride_bk, stride_bn, stride_bb,   # strides for B (row k, col j, batch) – note transposed view
    stride_cm, stride_cn, stride_cb,   # strides for C (row i, col j, batch)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C = A @ Bᵀ   (A,B: [batch, N, N])"""

    pid_m = tl.program_id(0)          # block row
    pid_n = tl.program_id(1)          # block col
    pid_b = tl.program_id(2)          # which batch*hidden slice

    # ----- block start indices -----
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # ----- accumulator -----
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Base pointers for this batch slice
    a_base = a_ptr + pid_b * stride_ab
    b_base = b_ptr + pid_b * stride_bb

    # ----- loop over K dimension -----
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        cur_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # A block: (M x K)
        a_ptrs = a_base + offs_m[:, None] * stride_am + cur_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # B block interpreted as (K x N) – we read rows = k, cols = j
        b_ptrs = b_base + cur_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # Multiply‑add
        acc += tl.dot(a, b)

    # ----- write out -----
    c_ptrs = c_ptr + pid_b * stride_cb + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    """
    Forward pass for the TriMul “outgoing” module.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]
        - mask         : torch.Tensor [B, N, N] (may be all‑ones)
        - weights      : dict of model weights
        - config       : dict with keys "dim", "hidden_dim", "nomask"

    Returns
    -------
    torch.Tensor : [B, N, N, dim]
    """
    # --------------------------------------------------------------
    # unpack & basic shapes
    # --------------------------------------------------------------
    x, mask, weights, config = data
    device = x.device
    B, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # --------------------------------------------------------------
    # 1. LayerNorm on the input (dim‑wise)
    # --------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # --------------------------------------------------------------
    # 2. Linear projections (no bias)
    # --------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3. Optional mask (broadcasted on the hidden dimension)
    # --------------------------------------------------------------
    if not config.get("nomask", False) and mask is not None:
        mask_f = mask.to(x.dtype).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # --------------------------------------------------------------
    # 4. Gating (sigmoid of linear projections)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Cast to FP16 for the heavy N³ contraction
    # --------------------------------------------------------------
    left = left.half()
    right = right.half()

    # Rearrange to [B, hidden_dim, N, N] so we can fuse batch & channel
    left = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    # Flatten the (batch, hidden) dimension → a single batch for the kernel
    BH = B * hidden_dim
    left_flat = left.view(BH, N, N)
    right_flat = right.view(BH, N, N)

    # Allocate output buffer in FP16
    out_flat = torch.empty_like(left_flat)

    # --------------------------------------------------------------
    # 6. Triton GEMM kernel launch (C = A @ Bᵀ)
    # --------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        BH,
    )

    tri_mul_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        N,
        N,
        # strides for A
        left_flat.stride(1),   # stride_am : step to next row (i)
        left_flat.stride(2),   # stride_ak : step to next col (k)
        left_flat.stride(0),   # stride_ab : step to next batch*hidden slice
        # strides for B (interpreted as K×N via transposition)
        right_flat.stride(2),  # stride_bk : original col‑stride (k)
        right_flat.stride(1),  # stride_bn : original row‑stride (j) → becomes col‑stride after transpose
        right_flat.stride(0),  # stride_bb : batch‑slice stride
        # strides for C
        out_flat.stride(1),    # stride_cm
        out_flat.stride(2),    # stride_cn
        out_flat.stride(0),    # stride_cb
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------
    # 7. Restore original layout and dtype (FP32)
    # --------------------------------------------------------------
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).float()   # [B,N,N,hidden_dim]

    # --------------------------------------------------------------
    # 8. Output LayerNorm, out‑gate and final linear projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate

    out = F.linear(out, weights["to_out.weight"])   # final projection to dim

    return out
```
