@triton.jit
def batched_matmul_kernel(a_ptr, b_ptr, c_ptr,
                          M, N, K,
                          stride_az, stride_ay, stride_ax,
                          stride_bz, stride_by, stride_bx,
                          stride_cz, stride_cy, stride_cx,
                          BLOCK_M: tl.constexpr,
                          BLOCK_N: tl.constexpr,
                          BLOCK_K: tl.constexpr):
    batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # mask for boundary
    mask_m = offs_m < M
    mask_n = offs_n < N

    # loads for output tile
    c_ptrs = c_ptr + batch * stride_cz + offs_m[:, None] * stride_cy + offs_n[None, :] * stride_cx
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # compute K offset
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)

        # mask for K dimension
        mask_k = offs_k < K

        # a tile: shape BLOCK_M x BLOCK_K
        a_ptrs = a_ptr + batch * stride_az + offs_m[:, None] * stride_ay + offs_k[None, :] * stride_ax
        # b tile: shape BLOCK_K x BLOCK_N
        b_ptrs = b_ptr + batch * stride_bz + offs_k[:, None] * stride_by + offs_n[None, :] * stride_bx

        # load with zero for out-of-bounds
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)
        # accumulate
        acc += tl.dot(a, b)
    # write back
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

"""
TriMul (outgoing) forward pass implemented with a custom Triton kernel.

The kernel performs the heavy O(N³·C_hidden) operation:
    out[b, i, j, d] = Σₖ left[b, i, k, d] * right[b, j, k, d]

We flatten the batch and hidden dimensions (B·C_hidden) and run a
batched matrix‑multiply (A @ Bᵀ) using Triton.  All other
operations (LayerNorm, Linear projections, gating, mask handling and the
final linear projection) are performed with regular PyTorch for
convenience.
"""

import torch
import triton
import triton.language as tl


# ----------------------------------------------------------
# Triton kernel: batched matmul (A @ B) where A, B, C have shape
# (batch, M, K), (batch, K, N) and (batch, M, N) respectively.
# ----------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_az, stride_ay, stride_ax,   # A: (batch, M, K)
    stride_bz, stride_by, stride_bx,   # B: (batch, K, N)
    stride_cz, stride_cy, stride_cx,   # C: (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # ------------------------------------------------------
    # program ids: (batch, tile_m, tile_n)
    # ------------------------------------------------------
    batch_id = tl.program_id(0)
    tile_m = tl.program_id(1)
    tile_n = tl.program_id(2)

    # range of rows/cols this program works on
    offs_m = tile_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = tile_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # boundary masks
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointer to the output tile
    c_ptrs = (
        c_ptr
        + batch_id * stride_cz
        + offs_m[:, None] * stride_cy
        + offs_n[None, :] * stride_cx
    )
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    #-------------------------------------------------------
    # Loop over K dimension in blocks of size BLOCK_K
    #-------------------------------------------------------
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = kk * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile: (BLOCK_M, BLOCK_K)
        a_ptrs = (
            a_ptr
            + batch_id * stride_az
            + offs_m[:, None] * stride_ay
            + offs_k[None, :] * stride_ax
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )

        # B tile: (BLOCK_K, BLOCK_N)
        b_ptrs = (
            b_ptr
            + batch_id * stride_bz
            + offs_k[:, None] * stride_by
            + offs_n[None, :] * stride_bx
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )

        # accumulate partial result
        acc += tl.dot(a, b)

    # write back the result
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------
# Main entry point used by the evaluation harness
# ----------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # Helper to fetch weight tensors
    # ------------------------------------------------------------------
    def _w(name):
        return weights[name]

    # ------------------------------------------------------------------
    # LayerNorm over the channel dimension (dim)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=_w("norm.weight"),
        bias=_w("norm.bias"),
        eps=eps,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, _w("left_proj.weight"))   # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, _w("right_proj.weight")) # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # Optional masking (broadcasted over hidden_dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask = mask_tensor.unsqueeze(-1).to(left.dtype)           # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # Gating (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, _w("left_gate.weight")).sigmoid()
    right_gate = torch.nn.functional.linear(x, _w("right_gate.weight")).sigmoid()
    out_gate = torch.nn.functional.linear(x, _w("out_gate.weight")).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Prepare data for the Triton kernel
    #   left : (B, N, N, hidden_dim)  ->  (B, hidden_dim, N, N)
    #   right: (B, N, N, hidden_dim)  ->  (B, hidden_dim, N, N) with dims (k, j)
    #   flatten batch & hidden_dim into a single batch dimension.
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape

    # permute to bring hidden_dim next to batch
    left_perm = left.permute(0, 3, 1, 2).contiguous()          # [B, hidden_dim, N, N]
    right_perm = right.permute(0, 3, 2, 1).contiguous()        # [B, hidden_dim, N, N] (k,j)

    # merge (B, hidden_dim) -> batch_h
    batch_h = B * hidden_dim
    left_flat = left_perm.view(batch_h, N, N)
    right_flat = right_perm.view(batch_h, N, N)

    # allocate output tensor (same dtype)
    out_flat = torch.empty_like(left_flat)

    # --------------------------------------------------------------
    # Triton launch configuration
    # --------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # strides in element units
    stride_az = left_flat.stride(0)
    stride_ay = left_flat.stride(1)
    stride_ax = left_flat.stride(2)

    stride_bz = right_flat.stride(0)
    stride_by = right_flat.stride(1)
    stride_bx = right_flat.stride(2)

    stride_cz = out_flat.stride(0)
    stride_cy = out_flat.stride(1)
    stride_cx = out_flat.stride(2)

    # grid dimensions: (batch_h, tiles_M, tiles_N)
    num_tiles_m = (N + BLOCK_M - 1) // BLOCK_M
    num_tiles_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (batch_h, num_tiles_m, num_tiles_n)

    _batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,                # M
        N,                # N
        N,                # K (inner dimension)
        stride_az,
        stride_ay,
        stride_ax,
        stride_bz,
        stride_by,
        stride_bx,
        stride_cz,
        stride_cy,
        stride_cx,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------
    # Reshape back to original layout: [B, N, N, hidden_dim]
    # --------------------------------------------------------------
    out = (
        out_flat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # shape [B, N, N, hidden_dim]

    # --------------------------------------------------------------
    # Post‑processing: LayerNorm → out_gate gating → final Linear
    # --------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=_w("to_out_norm.weight"),
        bias=_w("to_out_norm.bias"),
        eps=eps,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, _w("to_out.weight"))  # -> [B, N, N, dim]

    return out
