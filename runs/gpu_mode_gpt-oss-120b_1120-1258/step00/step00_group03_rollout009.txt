@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, D,
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_D: tl.constexpr,
    MASK_VAL: tl.constexpr,
):
    pid_b = tl.program_id(0)  # batch axis
    pid_i = tl.program_id(1)  # i block
    pid_j = tl.program_id(2)  # j block

    # Compute offsets for output block
    i_base = pid_i * BLOCK_I
    j_base = pid_j * BLOCK_J

    # Create range vectors
    i_offsets = i_base + tl.arange(0, BLOCK_I)
    j_offsets = j_base + tl.arange(0, BLOCK_J)
    d_offsets = tl.arange(0, BLOCK_D)

    # Mask out-of-bounds
    i_mask = i_offsets < N
    j_mask = j_offsets < N
    d_mask = d_offsets < D

    # Initialize accumulator
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_D), dtype=tl.float32)

    # Loop over K dimension
    for k_block in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k_block * BLOCK_K
        k_offsets = k_base + tl.arange(0, BLOCK_K)
        k_mask = k_offsets < N

        # Load left: shape (BLOCK_I, BLOCK_K, BLOCK_D)
        # Broadcast i_offsets, k_offsets, d_offsets
        left_ptrs = left_ptr + (
            pid_b * stride_left_b
            + i_offsets[:, None, None] * stride_left_i
            + k_offsets[None, :, None] * stride_left_k
            + d_offsets[None, None, :] * stride_left_d
        )

        # Load right: shape (BLOCK_J, BLOCK_K, BLOCK_D)

            pid_b * stride_right_b
            + j_offsets[:, None, None] * stride_right_j

            + d_offsets[None, None, :] * stride_right_d

        # left shape: I x K x D, right shape: J x K x D
        # We need to sum over K: out[i,j,d] += sum_k left[i,k,d] * right[j,k,d]
        # We can perform elementwise multiplication for each k and reduce.
        # Vectorized over K using broadcasting: produce left[:, :, :] * right[:, :, :] ??? Need to align dimensions.
        # left[i,k,d], right[j,k,d] -> we need outer product across i and j with same k and d.
        # Approach: For each k in block, we can multiply left[:,k,:] with right[:,k,:] via broadcasting.
        # left_k = left[:, k, :] shape (I, D)
        # right_k = right[:, k, :] shape (J, D)
        # We need to add outer product across i and j across D dimension elementwise: out[i,j,d] += left[i,k,d] * right[j,k,d]
        # So for each k, we can compute left_k[:, None, :] * right_k[None, :, :] (shape I,J,D) and accumulate.
        # We can implement this with a loop over k within block to avoid large tensor expansions.

        for kk in range(0, BLOCK_K):

        pid_b * stride_out_b
        + i_offsets[:, None, None] * stride_out_i
        + j_offsets[None, :, None] * stride_out_j
        + d_offsets[None, None, :] * stride_out_d

```
def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config['dim']
    hidden_dim = config['hidden_dim']
    # retrieve weights tensors (they are on device)
    norm_weight = weights['norm.weight']
    norm_bias = weights['norm.bias']
    left_proj_weight = weights['left_proj.weight']
    right_proj_weight = weights['right_proj.weight']
    left_gate_weight = weights['left_gate.weight']
    right_gate_weight = weights['right_gate.weight']
    out_gate_weight = weights['out_gate.weight']
    to_out_norm_weight = weights['to_out_norm.weight']
    to_out_norm_bias = weights['to_out_norm.bias']
    to_out_weight = weights['to_out.weight']

    # Step 1: LayerNorm on input
    x_norm = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # step 2: linear projections
    left_proj = torch.nn.functional.linear(x_norm, left_proj_weight)  # shape [B,N,N,hidden_dim]
    right_proj = torch.nn.functional.linear(x_norm, right_proj_weight)

    # step 3: gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, out_gate_weight))

    # step 4: mask preparation
    if config.get('nomask', False) or mask is None:
        mask_exp = torch.ones_like(left_proj, dtype=left_proj.dtype, device=device)
    else:
        # mask shape [B,N,N]; unsqueeze dim and expand across hidden_dim
        mask = mask.to(dtype=left_proj.dtype, device=device)
        mask_exp = mask.unsqueeze(-1).expand(-1, -1, -1, hidden_dim).contiguous()
    # step 5: elementwise gating and mask via Triton kernel
    left = torch.empty_like(left_proj)
    right = torch.empty_like(right_proj)
    # Ensure contiguous
    left_proj = left_proj.contiguous()
    left_gate = left_gate.contiguous()
    mask_exp = mask_exp.contiguous()
    right_proj = right_proj.contiguous()
    right_gate = right_gate.contiguous()
    # Triton kernels
    total = left_proj.numel()
    BLOCK_SIZE = 1024  # can tune based on size
    grid = (triton.cdiv(total, BLOCK_SIZE),)
    ternary_mul_kernel[grid](
        left_proj, left_gate, mask_exp, left,
        N_ELEMS=total,
        BLOCK_SIZE=BLOCK_SIZE
    )
    ternary_mul_kernel[grid](
        right_proj, right_gate, mask_exp, right,
        N_ELEMS=total,
        BLOCK_SIZE=BLOCK_SIZE
    )
    # step 6: compute the pairwise multiplication reduction
    # out shape [B,N,N,hidden_dim]
    out = torch.einsum('... i k d, ... j k d -> ... i j d', left, right)

    # step 7: to_out_norm
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    # step 8: apply out_gate
    out = out * out_gate

    # step 9: final linear
    output = torch.nn.functional.linear(out, to_out_weight)  # shape [B,N,N,dim]

    return output
```

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def ternary_mul_kernel(a_ptr, b_ptr, c_ptr, out_ptr,
                       N_ELEMS: tl.constexpr,
                       BLOCK_SIZE: tl.constexpr):
    """
    Compute elementwise product of three tensors: out = a * b * c.
    All tensors must be 1‑D contiguous views of the same size.
    """
    pid = tl.program_id(0)
    offset = pid * BLOCK_SIZE
    offs = offset + tl.arange(0, BLOCK_SIZE)
    mask = offs < N_ELEMS

    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    c = tl.load(c_ptr + offs, mask=mask, other=1.0)
    out = a * b * c
    tl.store(out_ptr + offs, out, mask=mask)

def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul module.
    The costly element‑wise fusion of projection, gate and mask is performed
    by a small Triton kernel; the N²×N reduction is carried out with a batched
    matrix multiplication (torch.bmm), which on an H100 is highly optimized.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    # ----------------------------------------------------------------------
    # unpack weights
    # ----------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]
    # ----------------------------------------------------------------------
    # 1. LayerNorm on the input (over the last dim)
    # ----------------------------------------------------------------------
    x = F.layer_norm(input_tensor,
                     (dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=1e-5)
    # ----------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ----------------------------------------------------------------------
    left_proj = F.linear(x, left_proj_weight)   # [B,N,N,hidden_dim]
    right_proj = F.linear(x, right_proj_weight)
    # ----------------------------------------------------------------------
    # 3. Gating values (sigmoid)
    # ----------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))
    # ----------------------------------------------------------------------
    # 4. Prepare mask (broadcast to hidden_dim)
    # ----------------------------------------------------------------------
    if config.get("nomask", False) or mask is None:
        mask_exp = torch.ones_like(left_proj, dtype=left_proj.dtype, device=device)
    else:

        mask = mask.to(dtype=left_proj.dtype, device=device)
        mask_exp = mask.unsqueeze(-1).expand(-1, -1, -1, hidden_dim).contiguous()
    # ----------------------------------------------------------------------
    # 5. Fuse projection, gate and mask with a Triton kernel
    # ----------------------------------------------------------------------
    left_proj = left_proj.contiguous()
    right_proj = right_proj.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    mask_exp = mask_exp.contiguous()

    right = torch.empty_like(right_proj)

    left_bmm = left.permute(0, 3, 1, 2).reshape(bs * hidden_dim, N, N)
    right_bmm = right.permute(0, 3, 1, 2).reshape(bs * hidden_dim, N, N)
    out_bmm = torch.bmm(left_bmm, right_bmm.transpose(1, 2))
    out = out_bmm.reshape(bs, hidden_dim, N, N).permute(0, 2, 3, 1)  # [B,N,N,hidden_dim]
    # ----------------------------------------------------------------------

"""
TriMul (outgoing) forward pass with a small Triton kernel.

The heavy element‑wise fusion of projection, gating and optional mask is performed
by a Triton kernel (`ternary_mul_kernel`).  The N‑by‑N pairwise reduction
required by the einsum is rewritten as a batched matrix multiplication
(L_d @ R_dᵀ) using `torch.bmm`, which is highly optimized on modern GPUs
(e.g. H100).  All other linear / normalization steps are executed with
standard PyTorch ops for simplicity and numerical stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def ternary_mul_kernel(a_ptr, b_ptr, c_ptr, out_ptr,
                       N_ELEMS: tl.constexpr,
                       BLOCK_SIZE: tl.constexpr):
    """Element‑wise product of three flat tensors: out = a * b * c.

    All tensors must be 1‑D contiguous views of the same length.
    """
    pid = tl.program_id(0)
    off = pid * BLOCK_SIZE
    offs = off + tl.arange(0, BLOCK_SIZE)
    mask = offs < N_ELEMS

    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    # c may be a mask; use 1.0 as the neutral element when out‑of‑range.
    c = tl.load(c_ptr + offs, mask=mask, other=1.0)

    out = a * b * c
    tl.store(out_ptr + offs, out, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]
        - mask         : torch.Tensor [B, N, N]  (may be None)
        - weights      : dict of torch.Parameter tensors
        - config       : dict with keys ``dim`` and ``hidden_dim`` and ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # weight tensors
    # ------------------------------------------------------------------
    norm_weight        = weights["norm.weight"]
    norm_bias          = weights["norm.bias"]
    left_proj_weight   = weights["left_proj.weight"]
    right_proj_weight  = weights["right_proj.weight"]
    left_gate_weight   = weights["left_gate.weight"]
    right_gate_weight  = weights["right_gate.weight"]
    out_gate_weight    = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1) LayerNorm over the last ``dim`` dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(input_tensor,
                     (dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=1e-5)

    # ------------------------------------------------------------------
    # 2) Linear projections (bias‑free)
    # ------------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right_proj = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3) Gates (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))

    # ------------------------------------------------------------------
    # 4) Prepare mask (broadcast to hidden_dim)
    # ------------------------------------------------------------------
    if config.get("nomask", False) or mask is None:
        mask_exp = torch.ones_like(left_proj,
                                   dtype=left_proj.dtype,
                                   device=device)
    else:
        # mask: [B, N, N] -> [B, N, N, 1] -> broadcast to hidden_dim
        mask = mask.to(dtype=left_proj.dtype, device=device)
        mask_exp = mask.unsqueeze(-1).expand(-1, -1, -1, hidden_dim).contiguous()

    # ------------------------------------------------------------------
    # 5) Fuse projection, gate and mask with Triton
    # ------------------------------------------------------------------
    # Ensure contiguity for flat indexing inside the kernel
    left_proj  = left_proj.contiguous()
    right_proj = right_proj.contiguous()
    left_gate  = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    mask_exp   = mask_exp.contiguous()

    left  = torch.empty_like(left_proj)
    right = torch.empty_like(right_proj)

    total_elems = left_proj.numel()                     # B * N * N * hidden_dim
    BLOCK_SIZE = 1024                                   # fine‑grained enough for H100
    grid = (triton.cdiv(total_elems, BLOCK_SIZE),)      # 1‑D grid

    ternary_mul_kernel[grid](
        left_proj,
        left_gate,
        mask_exp,
        left,
        N_ELEMS=total_elems,
        BLOCK_SIZE=BLOCK_SIZE,
    )
    ternary_mul_kernel[grid](
        right_proj,
        right_gate,
        mask_exp,
        right,
        N_ELEMS=total_elems,
        BLOCK_SIZE=BLOCK_SIZE,
    )

    # ------------------------------------------------------------------
    # 6) Pairwise multiplicative update (einsum) via batched GEMM
    #    out[..., i, j, d] = sum_k left[..., i, k, d] * right[..., j, k, d]
    #    This is exactly L_d @ R_dᵀ for each hidden channel d.
    # ------------------------------------------------------------------
    # reshape to (B * hidden_dim, N, N) and use torch.bmm
    left_bmm  = left.permute(0, 3, 1, 2).reshape(bs * hidden_dim, N, N)
    right_bmm = right.permute(0, 3, 1, 2).reshape(bs * hidden_dim, N, N)

    out_bmm = torch.bmm(left_bmm, right_bmm.transpose(1, 2))
    # reshape back to [B, N, N, hidden_dim]
    out = out_bmm.reshape(bs, hidden_dim, N, N).permute(0, 2, 3, 1)

    # ------------------------------------------------------------------
    # 7) LayerNorm over hidden_dim
    # ------------------------------------------------------------------
    out = F.layer_norm(out,
                       (hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias,
                       eps=1e-5)

    # ------------------------------------------------------------------
    # 8) Apply the output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 9) Final linear projection back to ``dim``
    # ------------------------------------------------------------------
    output = F.linear(out, to_out_weight)   # [B, N, N, dim]

    return output
