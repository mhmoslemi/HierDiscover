import triton.language as tl

@triton.jit
def gate_mask_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_j, stride_left_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    BLOCK_SIZE: tl.constexpr
):
    pid = tl.program_id(0)
    offset = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    total = B * N * N * H
    mask = offset < total
    # compute indices
    # flatten offsets relative to total
    offset = tl.where(mask, offset, 0)
    # compute b,i,j,h
    # b = offset // (N*N*H)
    # remainder = offset % (N*N*H)
    # i = remainder // (N*H)
    # remainder2 = remainder % (N*H)
    # j = remainder2 // H
    # h = remainder2 % H
    # Use tl.div/mod
    NMH = N * N * H
    NH = N * H
    b = offset // NMH
    off = offset - b * NMH
    i = off // NH
    off2 = off - i * NH
    j = off2 // H
    h = off2 - j * H

    right_offset = left_offset
    left_gate_offset = b * stride_gate_b + i * stride_gate_i + j * stride_gate_j + h * stride_gate_h
    right_gate_offset = left_gate_offset
    mask_offset = b * stride_mask_b + i * stride_mask_i + j * stride_mask_j

    right = tl.load(right_ptr + right_offset, mask=mask, other=0.0)

    right_gate = tl.load(right_gate_ptr + right_gate_offset, mask=mask, other=0.0)

import torch
import triton
import triton.language as tl

@triton.jit
def gate_mask_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_j, stride_left_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    BLOCK_SIZE: tl.constexpr
):
    """
    In-place kernel that multiplies left/right tensors with their sigmoid gates and a
    binary mask. The tensors have shape (B, N, N, H) for left/right/gates and (B,N,N)
    for mask.
    """
    pid = tl.program_id(0)
    # linear index of the element
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    total = B * N * N * H
    mask = offs < total
    offs = tl.where(mask, offs, 0)

    # decompose linear index into (b, i, j, h)
    nmh = N * N * H          # product of dimensions after the first
    nh = N * H               # product of the last two dimensions
    b = offs // nmh
    off = offs - b * nmh
    i = off // nh
    off2 = off - i * nh
    j = off2 // H
    h = off2 - j * H

    # compute the flat offsets for each tensor
    left_offset  = b * stride_left_b + i * stride_left_i + j * stride_left_j + h * stride_left_h
    right_offset = left_offset
    left_gate_offset  = b * stride_gate_b + i * stride_gate_i + j * stride_gate_j + h * stride_gate_h
    right_gate_offset = left_gate_offset
    mask_offset = b * stride_mask_b + i * stride_mask_i + j * stride_mask_j

    # load values
    left      = tl.load(left_ptr      + left_offset,      mask=mask, other=0.0)
    right     = tl.load(right_ptr     + right_offset,     mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + left_gate_offset, mask=mask, other=0.0)
    right_gate= tl.load(right_gate_ptr+ right_gate_offset,mask=mask, other=0.0)
    mval      = tl.load(mask_ptr      + mask_offset,      mask=mask, other=0.0)

    # apply mask and gate (element‑wise)
    left  = left  * left_gate  * mval
    right = right * right_gate * mval

    # write back
    tl.store(left_ptr  + left_offset,  left,  mask=mask)
    tl.store(right_ptr + right_offset, right, mask=mask)

    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.
    The implementation follows the reference PyTorch module but fuses the
    mask‑gate multiplication in a custom kernel and uses a batched GEMM
    (torch.bmm) for the N³ reduction.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # Linear projections (no bias) and gate logits
    # ------------------------------------------------------------------
    # shape: (B, N, N, hidden_dim)
    left      = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right     = torch.nn.functional.linear(x, weights["right_proj.weight"])
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate= torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate  = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    # ------------------------------------------------------------------
    # Apply mask and gates. If a mask is present we fuse the three element‑wise
    # multiplications (mask, left_gate, left) / (mask, right_gate, right) in a
    # Triton kernel; otherwise we simply multiply by the gates.
    # ------------------------------------------------------------------
    if nomask:
        left  = left  * left_gate
        right = right * right_gate
    else:
        # ensure mask is floating point and has shape (B,N,N)
        mask = mask.to(dtype)
        # Ensure all tensors are contiguous for the kernel
        left      = left.contiguous()
        right     = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate= right_gate.contiguous()
        mask      = mask.contiguous()

        B, N, _, _ = left.shape
        H = hidden_dim

        # strides (number of elements, not bytes) for each tensor
        stride_left_b, stride_left_i, stride_left_j, stride_left_h = left.stride()
        stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h = left_gate.stride()
        stride_mask_b, stride_mask_i, stride_mask_j = mask.stride()

        total_elements = B * N * N * H
        BLOCK_SIZE = 1024   # tuneable; 1024 works well for most shapes
        grid = lambda meta: (triton.cdiv(total_elements, meta["BLOCK_SIZE"]),)

        gate_mask_kernel[grid](
            left, right,
            left_gate, right_gate,
            mask,
            B, N, H,
            stride_left_b, stride_left_i, stride_left_j, stride_left_h,
            stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
            stride_mask_b, stride_mask_i, stride_mask_j,
            BLOCK_SIZE=BLOCK_SIZE,
        )
        # after the kernel `left` and `right` already contain the gated values

    # ------------------------------------------------------------------
    # N³ reduction: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # Implemented as a batched matrix multiplication for each channel d.
    # ------------------------------------------------------------------
    # Permute to (B, hidden_dim, N, N) and view as (B*hidden_dim, N, N)
    left_perm  = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    B_H = left_perm.shape[0] * left_perm.shape[1]  # B * hidden_dim
    left_flat  = left_perm.view(B_H, N, N)
    right_flat = right_perm.view(B_H, N, N)

    # batched GEMM: (B*H, N, N) @ (B*H, N, N)^T  → (B*H, N, N)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to (B, N, N, hidden_dim)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # Output LayerNorm, gating and final linear projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

"""
TriMul (outgoing) forward pass with a small Triton kernel.

Algorithm
---------
1. Layer‑norm the input over the last dimension (dim).
2. Linear projections (no bias) to hidden_dim:
      left  = x @ W_left      right = x @ W_right
   Gate logits (sigmoid) to hidden_dim:
      left_gate  = sigmoid(x @ W_left_gate)
      right_gate = sigmoid(x @ W_right_gate)
      out_gate   = sigmoid(x @ W_out_gate)
3. Fuse mask, left_gate and right_gate with a Triton kernel:
      left  = left  * mask * left_gate
      right = right * mask * right_gate
   (If ``nomask`` is True the mask‑gate multiplication is done with plain
   PyTorch element‑wise operations.)
4. N³ reduction:
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   Implemented as a batched GEMM – we reshape the tensors to
   (B·hidden_dim, N, N) and call ``torch.bmm``.
5. Apply a second LayerNorm, multiply by ``out_gate`` and a final
   linear projection back to ``dim``.
The kernel in step 3 is the only part written in Triton; the heavy
matrix‑multiplications are delegated to cuBLAS via ``torch.bmm`` for
maximum performance on an H100.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def gate_mask_kernel(
    left_ptr, right_ptr,                     # [B,N,N,H] mutable tensors
    left_gate_ptr, right_gate_ptr,           # [B,N,N,H] gate tensors
    mask_ptr,                                # [B,N,N] mask tensor
    B, N, H,                                 # compile‑time constants
    stride_left_b, stride_left_i, stride_left_j, stride_left_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    BLOCK_SIZE: tl.constexpr,
):
    """
    In‑place fusion of mask and sigmoid gates.
    For each element (b,i,j,h) we compute:
        left  = left  * left_gate  * mask
        right = right * right_gate * mask
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    total = B * N * N * H
    mask = offs < total
    offs = tl.where(mask, offs, 0)

    # Decompose linear index into (b,i,j,h)
    nmh = N * N * H          # product of dimensions after batch
    nh  = N * H
    b   = offs // nmh
    off = offs - b * nmh
    i   = off // nh
    off2 = off - i * nh
    j   = off2 // H
    h   = off2 - j * H

    # Compute flat offsets for all tensors
    left_off   = b * stride_left_b + i * stride_left_i + j * stride_left_j + h * stride_left_h
    right_off  = left_off
    left_gate_off  = b * stride_gate_b + i * stride_gate_i + j * stride_gate_j + h * stride_gate_h
    right_gate_off = left_gate_off
    mask_off   = b * stride_mask_b + i * stride_mask_i + j * stride_mask_j

    # Load values
    left      = tl.load(left_ptr      + left_off,      mask=mask, other=0.0)
    right     = tl.load(right_ptr     + right_off,     mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + left_gate_off, mask=mask, other=0.0)
    right_gate= tl.load(right_gate_ptr+ right_gate_off,mask=mask, other=0.0)
    mval      = tl.load(mask_ptr      + mask_off,      mask=mask, other=0.0)

    # Apply mask and gates (element‑wise)
    left  = left  * left_gate  * mval
    right = right * right_gate * mval

    # Store back
    tl.store(left_ptr  + left_off,  left,  mask=mask)
    tl.store(right_ptr + right_off, right, mask=mask)

        torch.Tensor of shape [B, N, N, dim] – the module output.
    """

    device = input_tensor.device
    dtype  = input_tensor.dtype

    dim        = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask     = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. First LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # 2. Linear projections (no bias) and gate logits
    # ------------------------------------------------------------------
    # shape → (B, N, N, hidden_dim)

    right     = torch.nn.functional.linear(x, weights["right_proj.weight"])
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate= torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate  = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    # ------------------------------------------------------------------
    # 3. Apply mask and gates (fused in Triton when a mask exists)
    # ------------------------------------------------------------------
    if nomask:
        # No mask → only gate multiplication
        left  = left  * left_gate
        right = right * right_gate

        mask = mask.to(dtype).contiguous()
        left      = left.contiguous()
        right     = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate= right_gate.contiguous()

        B, N, _, _ = left.shape
        H = hidden_dim

        # Strides (in elements, not bytes)
        stride_l_b, stride_l_i, stride_l_j, stride_l_h = left.stride()
        stride_g_b, stride_g_i, stride_g_j, stride_g_h = left_gate.stride()
        stride_m_b, stride_m_i, stride_m_j = mask.stride()

        total_elements = B * N * N * H
        BLOCK_SIZE = 1024                     # Empirically good for the tested sizes

        gate_mask_kernel[grid](

            stride_g_b, stride_g_i, stride_g_j, stride_g_h,

            BLOCK_SIZE=BLOCK_SIZE,
        )
        # After the kernel, ``left`` and ``right`` already contain the masked &
        # gated values.

    # ------------------------------------------------------------------
    # 4. N³ reduction via batched GEMM (out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d])
    # ------------------------------------------------------------------
    # Rearrange to (B, H, N, N) and flatten the first two dims → (B*H, N, N)
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    BH = left_perm.shape[0] * left_perm.shape[1]        # B * hidden_dim
