TriMul “outgoing” implementation (AlphaFold‑3 style) with a fused Triton kernel.

The heavy O(N³·H) part:
    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
is performed by a custom batched GEMM kernel where the hidden dimension `h`
is folded into the batch dimension (batch * hidden).  The kernel computes a
tile of the (i,j) matrix for each (batch, hidden) pair using
tl.dot over K‑tiles, accumulating in float32.  Inputs and outputs are stored
as fp16 for bandwidth, but the accumulation is float32 for numerical safety.
All other operations (LayerNorm, linear projections, gating, final LayerNorm
and projection) are carried out with native PyTorch ops.

The function `custom_kernel(data)` complies with the required signature and
returns a tensor of shape [B, N, N, dim] (float32).

import torch
import triton
import triton.language as tl
from torch.nn.functional import linear, layer_norm, sigmoid

@triton.jit
def trimul_outgoing_kernel(
    left_ptr,          # *fp16  [B,N,N,H]  (i,k) layout
    right_ptr,         # *fp16  [B,N,N,H]  (j,k) layout, we will read it as (k,j)
    out_ptr,           # *fp16  [B,N,N,H]  result (i,j)
    B, N, H,           # problem sizes
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_k, stride_right_j, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):

    Batched GEMM for the TriMul outer product.
    Batch index (including hidden) = program_id(2)
    i‑tile index = program_id(0)
    j‑tile index = program_id(1)
    """

    pid_j = tl.program_id(1)   # j‑tile
    pid_h = tl.program_id(2)   # (batch, hidden) flattened

    b = pid_h // H
    h = pid_h % H

    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # out‑of‑bounds masks for i and j
    mask_i = i < N
    mask_j = j < N

    left_base  = left_ptr  + b * stride_left_b  + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base   = out_ptr   + b * stride_out_b   + h * stride_out_h

    # accumulator in float32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over the reduction dimension k
    for cur_k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = cur_k * BLOCK_K
        k_range  = tl.arange(0, BLOCK_K)
        k_valid  = (k_offset + k_range) < N                     # mask for k

        # offsets for left:  (i, k)
        left_off = i[:, None] * stride_left_i + (k_offset + k_range)[None, :] * stride_left_k
        # offsets for right (read as (k, j)):
        right_off = (k_offset + k_range)[:, None] * stride_right_k + j[None, :] * stride_right_j

        # masked loads – convert to fp32 for the dot product
        a = tl.load(left_base + left_off,
                    mask=mask_i[:, None] & k_valid[None, :],
                    other=0.0).to(tl.float32)   # (BLOCK_M, BLOCK_K)
        b_mat = tl.load(right_base + right_off,
                        mask=k_valid[:, None] & mask_j[None, :],
                        other=0.0).to(tl.float32)   # (BLOCK_K, BLOCK_N)

        # accumulate block‑wise matrix product
        acc += tl.dot(a, b_mat)

    # write the result (cast back to fp16)
    out_off = i[:, None] * stride_out_i + j[None, :] * stride_out_j
    tl.store(out_base + out_off,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    """
    Forward pass for the “outgoing” TriMul module.
    Args:
        data: (input_tensor, mask_tensor, weights_dict, config_dict)
    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]                # input channel dimension
    hidden_dim = config["hidden_dim"]  # hidden dimension (H)

    # ------------------------------------------------------------------
    # Helper to fetch linear weights (biases are None for all proj layers)
    # ------------------------------------------------------------------
    # Linear weight shapes:
    #   [out, in]  – torch.nn.functional.linear expects (out_features, in_features)
    # Biases are omitted in the reference implementation.
    w_norm           = weights["norm.weight"]
    b_norm           = weights["norm.bias"]
    w_left_proj      = weights["left_proj.weight"]
    w_right_proj     = weights["right_proj.weight"]
    w_left_gate      = weights["left_gate.weight"]
    w_right_gate     = weights["right_gate.weight"]
    w_out_gate       = weights["out_gate.weight"]
    w_to_out_norm     = weights["to_out_norm.weight"]
    b_to_out_norm     = weights["to_out_norm.bias"]
    w_to_out          = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # Stage 1 – LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    x = layer_norm(input_tensor,
                   normalized_shape=(dim,),
                   weight=w_norm,
                   bias=b_norm,
                   eps=1e-5)

    # ------------------------------------------------------------------
    # Stage 2 – Linear projections (no bias)
    # ------------------------------------------------------------------
    # shape after projection: [B, N, N, hidden_dim]
    left  = linear(x, w_left_proj)    # (B,N,N,hidden_dim)
    right = linear(x, w_right_proj)

    # ------------------------------------------------------------------
    # Stage 3 – Optional masking (mask shape: [B,N,N])
    # ------------------------------------------------------------------
    if mask is not None:
        mask_f = mask.float().unsqueeze(-1)          # (B,N,N,1)
        left  = left  * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # Stage 4 – Gating terms (produced at hidden_dim size)
    # ------------------------------------------------------------------
    left_gate  = sigmoid(linear(x, w_left_gate))
    right_gate = sigmoid(linear(x, w_right_gate))
    out_gate   = sigmoid(linear(x, w_out_gate))

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Stage 5 – Heavy O(N³·H) Einsum via Triton
    #   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    # Cast to fp16 for memory bandwidth, keep a fp32 copy for the final
    # LayerNorm/Linear that follow.
    left_fp16  = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    B, N, _, H = left_fp16.shape
    out_fp16 = torch.empty_like(left_fp16, dtype=torch.float16)   # (B,N,N,H)

    # Strides (already in the correct order for the kernel)
    sl_b, sl_i, sl_k, sl_h = left_fp16.stride()
    sr_b, sr_i, sr_k, sr_h = right_fp16.stride()
    so_b, so_i, so_j, so_h = out_fp16.stride()

    # Tile sizes – chosen to fit H100 shared‑memory / register budget
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),          # i‑tiles
        triton.cdiv(N, BLOCK_N),          # j‑tiles
        B * H                             # (batch, hidden) flattened
    )

    trimul_outgoing_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        B, N, H,
        sl_b, sl_i, sl_k, sl_h,
        sr_b, sr_k, sr_i, sr_h,   # note: stride_right_k, stride_right_j (j is sr_i)
        so_b, so_i, so_j, so_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # Convert back to fp32 for the remaining FP32‑only layers
    out = out_fp16.to(torch.float32)   # (B,N,N,H)

    # ------------------------------------------------------------------
    # Stage 6 – Post‑processing (LayerNorm, gating, final linear)
    # ------------------------------------------------------------------
    out = layer_norm(out,
                     normalized_shape=(hidden_dim,),
                     weight=w_to_out_norm,
                     bias=b_to_out_norm,
                     eps=1e-5)
    out = out * out_gate                       # broadcast over hidden dim
    out = linear(out, w_to_out)                # (B,N,N,dim)

    return out
