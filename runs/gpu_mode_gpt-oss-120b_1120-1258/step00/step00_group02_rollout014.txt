def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # Cast to float16 maybe for speed
    x = input_tensor.to(torch.float16)
    if mask is None or config.get("nomask", True):
        mask = torch.ones_like(input_tensor[...,0], dtype=torch.float16)  # shape B,N,N
    else:
        mask = mask.to(torch.float16)

    # LayerNorm: compute mean and var across dim
    # Using torch.nn.functional.layer_norm
    x = torch.nn.functional.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # Linear projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # Apply mask
    mask_unsq = mask.unsqueeze(-1)  # shape B,N,N,1
    left = left * mask_unsq
    right = right * mask_unsq

    # Gating
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Move to half for kernel
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # Allocate output tensor
    out = torch.empty_like(left, dtype=torch.float32, device=left.device)  # shape B,N,N,hidden

@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr
):
    pid = tl.program_id(0)
    # compute number of tiles
    num_i = tl.cdiv(N, BLOCK_I)
    num_j = tl.cdiv(N, BLOCK_J)
    # compute batch index and tile indices
    b = pid // (num_i * num_j)
    pid_ij = pid % (num_i * num_j)
    i_start = (pid_ij // num_j) * BLOCK_I
    j_start = (pid_ij % num_j) * BLOCK_J

    # Clamp i_start/j_start to N (not necessary)
    # Number of hidden tiles
    num_h = tl.cdiv(H, BLOCK_H)

    # Loop over hidden tile
    for h_tile in range(num_h):
        h_start = h_tile * BLOCK_H
        # accumulator
        acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)
        # Loop over reduction dimension k
        num_k = tl.cdiv(N, BLOCK_K)
        for k_tile in range(num_k):
            k_start = k_tile * BLOCK_K

            # offsets for left: i, k, h
            i_offs = i_start + tl.arange(0, BLOCK_I)[:, None, None]
            k_offs = k_start + tl.arange(0, BLOCK_K)[None, :, None]
            h_offs = h_start + tl.arange(0, BLOCK_H)[None, None, :]
            # mask for in-bounds
            mask_i = i_offs < N
            mask_k = k_offs < N
            mask_h = h_offs < H
            left_ptrs = left_ptr + b*stride_left_b + i_offs*stride_left_i + k_offs*stride_left_k + h_offs*stride_left_h
            left_block = tl.load(left_ptrs, mask=mask_i & mask_k & mask_h, other=0.0)
            # load right block transposed: k as first, j as second
            j_offs = j_start + tl.arange(0, BLOCK_J)[None, :, None]
            mask_j = j_offs < N

            # For each hidden slice, compute dot product
            for dh in range(0, BLOCK_H):
                l = left_block[:, :, dh]   # (BLOCK_I, BLOCK_K)
                r = right_block[:, :, dh]   # (BLOCK_K, BLOCK_J)
                # accumulate the matrix multiplication
                acc[:, :, dh] += tl.dot(l, r)

        # store accumulator
        i_store = i_start + tl.arange(0, BLOCK_I)[:, None, None]
        j_store = j_start + tl.arange(0, BLOCK_J)[None, :, None]
        h_store = h_start + tl.arange(0, BLOCK_H)[None, None, :]
        mask_i = i_store < N
        mask_j = j_store < N
        mask_h = h_store < H
        out_ptrs = out_ptr + b*stride_out_b + i_store*stride_out_i + j_store*stride_out_j + h_store*stride_out_h

assistantfinal**Solution Overview**

The *outgoing* triangular multiplicative update (TriMul) computes  

\[
\text{out}_{b,i,j,:}= \sum_{k=0}^{N-1}
      \bigl(\text{left}_{b,i,k,:}\odot \text{right}_{b,j,k,:}\bigr)
\]

where `left` and `right` are linear projections of the input tensor,
optionally masked and gated.  
The contraction is the only cubic‑cost part ( ∝ \(N^{2}\!\times\!H\) ) and
is the part we accelerate with Triton.  

The remaining operations (LayerNorm, linear projections, gating,
final LayerNorm + linear) are performed with regular PyTorch – this
keeps the implementation short while still satisfying the requirement
that *part* of the work is done in a Triton kernel.

--------------------------------------------------------------------

### Triton kernel design

* **Tiling** – the three free indices `(i, j, h)` are tiled:
  * `BLOCK_I` × `BLOCK_J` tiles cover the pair‑wise 2‑D plane.
  * `BLOCK_H` tiles cover the hidden dimension `H`.
* **Reduction over `k`** – the inner dimension is processed in blocks
  `BLOCK_K`. For every `(i‑tile, j‑tile, h‑tile)` we keep a
  `float32` accumulator `acc[i, j, h]`.  
* **Matrix‑multiply per hidden slice** – after loading a
  `(i, k, h)` block of `left` and a transposed `(k, j, h)` block of
  `right` we compute

  ```python
  acc[i, j, h] += dot( left[i, :, h] , right[:, j, h] )
  ```

  With `tl.dot` each hidden slice is reduced over `k` with a small
  GEMM (`BLOCK_I × BLOCK_K` × `BLOCK_J`).  
* **Mask handling** – out‑of‑bounds loads are masked with zeros
  (`other=0.0`). The same mask works for `i`, `j`, `k` and the hidden
  tile, guaranteeing correct results for any `N` and `H`.
* **Launch grid** – one program (`pid`) processes a single `(i‑tile,
  j‑tile, batch)` pair. The grid size is  
  `B * ceil(N/BLOCK_I) * ceil(N/BLOCK_J)`.

--------------------------------------------------------------------

### Full implementation

```python
"""
TriMul (outgoing) forward pass – Triton‐accelerated contraction.

The kernel evaluates
    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]

where `left`/`right` are already masked and gated.
All other operations (LayerNorm, linear projections, gating,
final LayerNorm + linear) are performed with PyTorch.
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F


# ----------------------------------------------------------------------
# Triton kernel
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr,              # const * half
    right_ptr,             # const * half
    out_ptr,               # * float32
    B,                     # batch size (runtime)
    N: tl.constexpr,       # sequence length (compile‑time)
    H: tl.constexpr,       # hidden dimension (compile‑time)

    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,

    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    Computes the outgoing TriMul contraction.
    """
    pid = tl.program_id(0)

    # ------------------------------------------------------------------
    # 1️⃣  Compute tile indices (batch, i‑tile, j‑tile)
    # ------------------------------------------------------------------
    num_i_tiles = tl.cdiv(N, BLOCK_I)
    num_j_tiles = tl.cdiv(N, BLOCK_J)

    batch_idx = pid // (num_i_tiles * num_j_tiles)
    pid_ij   = pid % (num_i_tiles * num_j_tiles)

    i_start = (pid_ij // num_j_tiles) * BLOCK_I
    j_start = (pid_ij %  num_j_tiles) * BLOCK_J

    # ------------------------------------------------------------------
    # 2️⃣  Loop over hidden tiles
    # ------------------------------------------------------------------
    num_h_tiles = tl.cdiv(H, BLOCK_H)
    for h_tile in range(num_h_tiles):
        h_start = h_tile * BLOCK_H

        # accumulator for the current (i,j,h‑tile)
        acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)

        # ----------------------------------------------------------------
        # 3️⃣  Loop over the reduction dimension k
        # ----------------------------------------------------------------
        num_k_tiles = tl.cdiv(N, BLOCK_K)
        for k_tile in range(num_k_tiles):
            k_start = k_tile * BLOCK_K

            # ----- offsets -----
            i_offs = i_start + tl.arange(0, BLOCK_I)[:, None, None]          # (I,1,1)
            j_offs = j_start + tl.arange(0, BLOCK_J)[None, :, None]          # (1,J,1)
            k_offs = k_start + tl.arange(0, BLOCK_K)[None, :, None]          # (1,K,1)
            h_offs = h_start + tl.arange(0, BLOCK_H)[None, None, :]          # (1,1,H)

            # ----- masks for out‑of‑bounds loads -----
            mask_i = i_offs < N
            mask_j = j_offs < N
            mask_k = k_offs < N
            mask_h = h_offs < H

            # ----- left[i, k, h] (shape: I × K × H) -----
            left_ptrs = (left_ptr
                         + batch_idx * stride_left_b
                         + i_offs * stride_left_i
                         + k_offs * stride_left_k
                         + h_offs * stride_left_h)
            left = tl.load(left_ptrs,
                           mask=mask_i & mask_k & mask_h,
                           other=0.0).to(tl.float16)

            # ----- right[j, k, h] transposed to (K × J × H) -----
            # we load with k as the leading dimension so that we can use tl.dot later
            right_ptrs = (right_ptr
                          + batch_idx * stride_right_b
                          + k_offs * stride_right_k
                          + j_offs * stride_right_j
                          + h_offs * stride_right_h)
            right = tl.load(right_ptrs,
                            mask=mask_k & mask_j & mask_h,
                            other=0.0).to(tl.float16)

            # ----- per‑hidden‑slice matrix multiplication -----
            # Unroll over the hidden tile (compile‑time constant)
            for dh in tl.static_range(0, BLOCK_H):
                # left slice   : (I, K)
                # right slice  : (K, J)   (already transposed)
                l = left[:, :, dh]          # (I, K)
                r = right[:, :, dh]          # (K, J)
                # accumulate in float32
                acc[:, :, dh] += tl.dot(l, r).to(tl.float32)

        # ----------------------------------------------------------------
        # 4️⃣  Write the tile back to global memory
        # ----------------------------------------------------------------
        i_store = i_start + tl.arange(0, BLOCK_I)[:, None, None]
        j_store = j_start + tl.arange(0, BLOCK_J)[None, :, None]
        h_store = h_start + tl.arange(0, BLOCK_H)[None, None, :]

        mask_i = i_store < N
        mask_j = j_store < N
        mask_h = h_store < H

        out_ptrs = (out_ptr
                    + batch_idx * stride_out_b
                    + i_store * stride_out_i
                    + j_store * stride_out_j
                    + h_store * stride_out_h)

        tl.store(out_ptrs,
                 acc,
                 mask=mask_i & mask_j & mask_h)


# ----------------------------------------------------------------------
# Python entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.
    Arguments
    ----------
    data : Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
        input    – (B, N, N, dim)   float32 tensor
        mask     – (B, N, N)        float32 or bool tensor (may be all‑ones)
        weights  – dictionary with all linear / layer‑norm parameters
        config   – dict containing:
                     * dim          – input feature dimension
                     * hidden_dim   – hidden dimension H
                     * bs, seqlen   – batch size and sequence length
                     * nomask       – True → ignore mask (treat as all‑ones)
    Returns
    -------
    torch.Tensor
        (B, N, N, dim) float32 tensor – the TriMul output
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    bs = config["bs"]
    N = config["seqlen"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣  LayerNorm on the input (dim‑wise)
    # ------------------------------------------------------------------
    # Cast to float16 for the heavy part later
    x = input_tensor.to(torch.float16)
    x = F.layer_norm(x,
                     normalized_shape=(dim,),
                     weight=weights["norm.weight"],
                     bias=weights["norm.bias"],
                     eps=eps)

    # ------------------------------------------------------------------
    # 2️⃣  Linear projections and gating (still in fp16)
    # ------------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])   # (B,N,N,H)
    right = F.linear(x, weights["right_proj.weight"])

    # mask handling ----------------------------------------------------
    if config.get("nomask", True):
        mask = torch.ones_like(input_tensor[..., 0], dtype=torch.float16, device=x.device)
    else:
        # mask may be bool – convert to the same dtype as the tensors
        mask = mask_tensor.to(torch.float16)
    mask = mask.unsqueeze(-1)        # (B,N,N,1)

    left  = left  * mask
    right = right * mask

    # ------------------- gating ---------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # (B,N,N,H)
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))   # (B,N,N,H)

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 3️⃣  Triton kernel – cubic contraction over k
    # ------------------------------------------------------------------
    left = left.contiguous()   # ensure contiguous layout
    right = right.contiguous()
    # output of the contraction lives in float32 (as required later)
    out = torch.empty((bs, N, N, hidden_dim), dtype=torch.float32,
                      device=left.device)

    # launch configuration ------------------------------------------------
    BLOCK_I = 16
    BLOCK_J = 16
    BLOCK_K = 32
    BLOCK_H = 32

    grid = (bs * ((N + BLOCK_I - 1) // BLOCK_I) *
                ((N + BLOCK_J - 1) // BLOCK_J),)

    tri_mul_outgoing_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=bs,
        N=N,
        H=hidden_dim,
        stride_left_b=left.stride(0),
        stride_left_i=left.stride(1),
        stride_left_k=left.stride(2),
        stride_left_h=left.stride(3),
        stride_right_b=right.stride(0),
        stride_right_j=right.stride(1),   # j‑dimension (second dim)
        stride_right_k=right.stride(2),
        stride_right_h=right.stride(3),
        stride_out_b=out.stride(0),
        stride_out_i=out.stride(1),
        stride_out_j=out.stride(2),
        stride_out_h=out.stride(3),
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
    )

    # ------------------------------------------------------------------
    # 4️⃣  Post‑processing (LayerNorm, gating, final Linear)
    # ------------------------------------------------------------------
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=weights["to_out_norm.weight"],
                       bias=weights["to_out_norm.bias"],
                       eps=eps)

    # out_gate was computed in fp16 – cast to fp32 before the elementwise mul
    out_gate = out_gate.to(torch.float32)
    out = out * out_gate

    # final projection back to `dim`
    out = F.linear(out, weights["to_out.weight"])

    return out
```
