@triton.jit
def trimul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k,
    stride_right_b, stride_right_j, stride_right_k,
    stride_out_b, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid0 = tl.program_id(0)  # combined batch and hidden channel
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # Compute batch and hidden channel
    b = pid0 // H
    d = pid0 % H

    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # Mask for out-of-range indices
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k
    for k in range(0, N, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left.tile: shape (M, K)
        # Left is [B, N, N, H] with dimensions i,k,d.
        left = tl.load(left_ptr + b*stride_left_b 
                     + offs_i[:, None] * stride_left_i
                     + offs_k[None, :] * stride_left_k
                     + d * stride_left_h,
                     mask=mask_i[:, None] & mask_k[None, :],
                     other=0.0)

        # Load right.tile: we need right[j,k,d] thus shape (N,K)
        # Right is [B, N, N, H] with dimensions j,k,d.
        # We'll load as (K,N) by swapping i/j? Actually we want right_T shape (K,N) for dot.
        # So we load right_transpose = right[b, :, :, d] transposed across j and k:
        # Let offs_j be column, offs_k be row after transpose.
        right = tl.load(right_ptr + b*stride_right_b 
                     + offs_k[:, None] * stride_right_k   # row index = k
                     + offs_j[None, :] * stride_right_j   # column index = j
                     + d * stride_right_h,
                     mask=mask_k[:, None] & mask_j[None, :],
                     other=0.0)

        # Multiply-accumulate
        acc += tl.dot(left, right)  # (M,K)*(K,N) -> (M,N)
    # Store output
    out = acc.to(tl.float16)  # maybe convert later?

    # Write out
    tl.store(out_ptr + b*stride_out_b 
           + offs_i[:, None] * stride_out_i
           + offs_j[None, :] * stride_out_j
           + d * stride_out_h,
           out,

def trimul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid0 = tl.program_id(0)  # batch * hidden channel
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # Decode batch and hidden index
    b = pid0 // H
    d = pid0 % H

    # Compute start offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, N, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left tile (M, K)
        left = tl.load(
            left_ptr
            + b * stride_left_b
            + offs_i[:, None] * stride_left_i
            + offs_k[None, :] * stride_left_k
            + d * stride_left_h,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
            dtype=tl.float16,
        )
        left = left.to(tl.float32)

        # Load right tile transposed (K, N)
        right = tl.load(
            right_ptr
            + b * stride_right_b
            + offs_k[:, None] * stride_right_k
            + offs_j[None, :] * stride_right_j
            + d * stride_right_h,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
            dtype=tl.float16,
        )
        right = right.to(tl.float32)

        acc += tl.dot(left, right)  # (M,K)@(K,N) -> (M,N)

    # Store result
    tl.store(
        out_ptr
        + b * stride_out_b
        + offs_i[:, None] * stride_out_i
        + offs_j[None, :] * stride_out_j

"""
TriMul “outgoing” kernel (AlphaFold3) – forward only.

Algorithm
---------
1. Layer‑norm the input tensor `x` over the last dimension.
2. Linear projections give left/right tensors of shape [B,N,N,H].
3. Optional pairwise mask (broadcast on H) is applied.
4. Gating values (sigmoid of linear projections) are computed and
   multiplied element‑wise into left/right.
5. The expensive contraction
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   is performed by a Triton kernel that treats each (b,h) pair as a
   separate program and carries out a tiled batched matrix‑multiply
   (left @ rightᵀ) in float32 while loading the inputs as float16.
6. The result (still in FP32) is Layer‑normed, multiplied by the
   output‑gate, and finally projected back to the original dimension.

The kernel fuses the O(N³·H) core operation and provides a large speedup
compared with a naïve PyTorch `einsum`. All other arithmetic (norms,
gates, final linear) stays in PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _trimul_matmul_kernel(
    left_ptr,               # *float16  [B,N,N,H] (i,k,h)
    right_ptr,              # *float16  [B,N,N,H] (j,k,h)
    out_ptr,                # *float32  [B,N,N,H] (i,j,h)
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # 3‑D grid: (b×h) × i‑tile × j‑tile
    pid0 = tl.program_id(0)          # encodes batch * hidden channel
    pid_i = tl.program_id(1)         # i‑tile id
    pid_j = tl.program_id(2)         # j‑tile id

    # decode batch index and hidden‑channel index
    b = pid0 // H
    d = pid0 % H

    # tile offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over the reduction axis k
    for k in range(0, N, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ----------- left tile (M x K) -------------
        left = tl.load(
            left_ptr
            + b * stride_l_b
            + offs_i[:, None] * stride_l_i
            + offs_k[None, :] * stride_l_k
            + d * stride_l_h,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )
        # cast to FP32 for accumulation
        left = left.to(tl.float32)

        # ----------- right tile (K x N) – transposed ----------
        # we load right[j,k,h] as (k,j) to avoid an explicit transpose
        right = tl.load(
            right_ptr
            + b * stride_r_b
            + offs_k[:, None] * stride_r_k   # rows = k
            + offs_j[None, :] * stride_r_j   # cols = j
            + d * stride_r_h,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )
        right = right.to(tl.float32)

        # accumulate dot product over the K‑dimension
        acc += tl.dot(left, right)   # (M,K) @ (K,N) -> (M,N)

    # store the block
    tl.store(
        out_ptr
        + b * stride_o_b
        + offs_i[:, None] * stride_o_i
        + offs_j[None, :] * stride_o_j
        + d * stride_o_h,
        acc,
        mask=mask_i[:, None] & mask_j[None, :],
    )


def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, dim]
        - mask          : torch.Tensor of shape [B, N, N] or None
        - weights       : dict of model weights (see code)
        - config        : dict containing `dim` and `hidden_dim`

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim]
    """
    # unpack
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    B, N, _, _ = x.shape

    # -----------------------------------------------------------------
    # 1. Input layer‑norm
    # -----------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # -----------------------------------------------------------------
    # 2. Linear projections
    # -----------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3. Optional mask (broadcast on hidden dim)
    # -----------------------------------------------------------------
    if mask is not None:
        mask = mask.to(x.dtype).unsqueeze(-1)          # [B,N,N,1]
        left = left * mask
        right = right * mask

    # -----------------------------------------------------------------
    # 4. Gating (sigmoid of linear projections)
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Half‑precision tensors for the heavy matmul
    # -----------------------------------------------------------------
    left_h = left.contiguous().to(torch.float16)
    right_h = right.contiguous().to(torch.float16)

    # free FP32 intermediates early to keep memory usage low
    del left, right, left_gate, right_gate, x
    torch.cuda.empty_cache()

    # -----------------------------------------------------------------
    # 6. Triton batched matmul over the reduction dimension k
    # -----------------------------------------------------------------
    out = torch.empty(
        (B, N, N, hidden_dim), dtype=torch.float32, device=left_h.device
    )

    # strides are expressed in units of elements (not bytes)
    s_lb, s_li, s_lk, s_lh = left_h.stride()
    s_rb, s_rj, s_rk, s_rh = right_h.stride()
    s_ob, s_oi, s_oj, s_oh = out.stride()

    # Tunable block sizes – chosen to keep the number of threads ≤1024
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B * hidden_dim,                     # program 0 : batch × hidden channel
        (N + BLOCK_M - 1) // BLOCK_M,       # program 1 : i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,       # program 2 : j‑tiles
    )

    _trimul_matmul_kernel[
        grid
    ](
        left_h,
        right_h,
        out,
        B,
        N,
        hidden_dim,
        s_lb,
        s_li,
        s_lk,
        s_lh,
        s_rb,
        s_rj,
        s_rk,
        s_rh,
        s_ob,
        s_oi,
        s_oj,
        s_oh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,        # heuristic – good on H100
    )

    # -----------------------------------------------------------------
    # 7. Post‑matmul normalisation, output‑gate and final projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
