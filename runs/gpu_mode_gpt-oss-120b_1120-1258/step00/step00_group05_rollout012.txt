@triton.jit
def tri_mul_kernel(left_ptr, right_ptr, out_ptr,
                   M, N, K,
                   stride_left_batch, stride_left_m, stride_left_k,
                   stride_right_batch, stride_right_k, stride_right_n,
                   stride_out_batch, stride_out_m, stride_out_n,
                   BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid = tl.program_id(0)
    num_pid_m = tl.cdiv(M, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)
    batch = tl.program_id(1)  # batch dimension (B*D)
    # Determine block start
    pid_m = pid // num_pid_n
    pid_n = pid % num_pid_n
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # Mask offsets beyond shape
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)  # accumulate fp32

    # Compute pointers for left and right blocks
    # left: shape (batch, M, K)
    # right: shape (batch, K, N) - due to transposed.
    # Loop over K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # compute k offsets
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # left tile pointer
        left_ptrs = left_ptr + batch * stride_left_batch + offs_m[:, None] * stride_left_m + offs_k[None, :] * stride_left_k
        left = tl.load(left_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # right tile pointer (since right is (batch, K, N) layout)
        right_ptrs = right_ptr + batch * stride_right_batch + offs_k[:, None] * stride_right_k + offs_n[None, :] * stride_right_n
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        acc += tl.dot(left, right)  # (BLOCK_M, BLOCK_K) dot (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)

        # increment pointers for next k-block: but we compute offsets for each iteration, no pointer inc needed

    # Write back results
    out_ptrs = out_ptr + batch * stride_out_batch + offs_m[:, None] * stride_out_m + offs_n[None, :] * stride_out_n
    tl.store(out_ptrs, acc.to(tl.float32), mask=mask_m[:, None] & mask_n[None, :])

"""
TriMul custom kernel (outgoing version)

The forward pass consists of:
1️⃣ Layer‑norm on the input tensor x   (shape [B, N, N, dim]).
2️⃣ Two linear projections (left & right) + three “gate” linear layers,
   followed by a sigmoid and optional element‑wise masking.
3️⃣ A batched matrix‑multiply over the sequence dimension:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is the expensive N³ operation.  We implement it with a
   Triton kernel that treats each (b,d) pair as an independent batch
   and performs a tiled GEMM (A @ Bᵀ) in fp16 with fp32 accumulation.
4️⃣ Layer‑norm on the hidden dimension, gating, and a final linear
   projection back to the original dimensionality.

All heavy linear work (steps 2 and 3) is executed in fp16 to exploit the
Tensor‑cores of an H100; accumulate in fp32 for numerical stability.
The final result is returned as float32.
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F


@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    M, N, K,                                 # all equal to seq_len
    stride_left_batch, stride_left_m, stride_left_k,
    stride_right_batch, stride_right_k, stride_right_n,
    stride_out_batch, stride_out_m, stride_out_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Batched GEMM for the TriMul contraction.
    left  : [batch, M, K]  (M = i, K = k)
    right : [batch, K, N]  (K = k, N = j) – note the transposed view
    out   : [batch, M, N]  (M = i, N = j)
    """
    pid = tl.program_id(0)          # combined (i‑block, j‑block) id
    batch = tl.program_id(1)        # flat index over (B * hidden_dim)

    # block coordinates
    num_pid_n = tl.cdiv(N, BLOCK_N)
    pid_m = pid // num_pid_n
    pid_n = pid % num_pid_n

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # ---- load left tile (M x K) -------------------------------------------------
        left_ptrs = left_ptr + \
                    batch * stride_left_batch + \
                    offs_m[:, None] * stride_left_m + \
                    offs_k[None, :] * stride_left_k
        left = tl.load(left_ptrs,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # ---- load right tile (K x N) – transposed view --------------------------------
        # right is stored as (batch, N, K).  By swapping the strides we interpret it as (K, N)
        right_ptrs = right_ptr + \
                     batch * stride_right_batch + \
                     offs_k[:, None] * stride_right_k + \
                     offs_n[None, :] * stride_right_n
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_n[None, :],
                        other=0.0).to(tl.float32)

        # ---- accumulate --------------------------------------------------------------
        acc += tl.dot(left, right)

    # ---- write back ---------------------------------------------------------------
    out_ptrs = out_ptr + \
               batch * stride_out_batch + \
               offs_m[:, None] * stride_out_m + \
               offs_n[None, :] * stride_out_n
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N]  (bool/float) or None
            - weights_dict : dict with keys
                'norm.weight', 'norm.bias',
                'left_proj.weight', 'right_proj.weight',
                'left_gate.weight', 'right_gate.weight',
                'out_gate.weight',
                'to_out_norm.weight', 'to_out_norm.bias',
                'to_out.weight'
            - config_dict  : dict with at least
                'dim' (int), 'hidden_dim' (int), optionally 'nomask' (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # unpack
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device

    # -------------------------------------------------------------------------
    # 1. LayerNorm on the input (float32)
    # -------------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x_norm = F.layer_norm(input_tensor, (dim,),
                          weight=norm_weight,
                          bias=norm_bias)          # [B, N, N, dim]  (FP32)

    # Cast to fp16 for the heavy work
    x = x_norm.to(torch.float16)

    # -------------------------------------------------------------------------
    # 2. Linear projections + gates (all fp16)
    # -------------------------------------------------------------------------
    # Cast all weight tensors to fp16 once
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # projections (no bias)
    left_proj  = F.linear(x, left_proj_w)          # [B, N, N, hidden_dim]
    right_proj = F.linear(x, right_proj_w)

    # optional mask (mask is broadcast on the hidden dimension)
    if not nomask and mask is not None:
        # mask may be bool or float – cast to fp16 and add a trailing dim
        mask_fp = mask.to(torch.float16).unsqueeze(-1)   # [B, N, N, 1]
        left_proj  = left_proj * mask_fp
        right_proj = right_proj * mask_fp

    # gates (sigmoid)
    left_gate  = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate   = F.linear(x, out_gate_w).sigmoid()

    # apply gates
    left  = left_proj * left_gate   # [B, N, N, hidden_dim]
    right = right_proj * right_gate

    # -------------------------------------------------------------------------
    # 3. Triton kernel: the N³ contraction
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # -------------------------------------------------------------------------
    #   a) bring hidden dimension to leading position for easy flattening
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, D, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, D, N, N]

    B, D, N, _ = left_perm.shape
    batch_flat = B * D                                   # treated as a single batch dim

    #   b) flatten the (B,D) dimension → shape [batch_flat, N, N]
    left_flat  = left_perm.view(batch_flat, N, N)
    right_flat = right_perm.view(batch_flat, N, N)
    out_flat   = torch.empty_like(left_flat)            # fp16 output buffer

    #   c) strides (all tensors are contiguous, row‑major)
    stride_left_batch = left_flat.stride(0)   # N * N
    stride_left_m     = left_flat.stride(1)   # N
    stride_left_k     = left_flat.stride(2)   # 1

    stride_right_batch = right_flat.stride(0)  # N * N
    # To treat right as (K, N) we swap the last two strides
    stride_right_k = right_flat.stride(1)      # N   (row stride becomes K‑stride)
    stride_right_n = right_flat.stride(2)      # 1   (col stride becomes N‑stride)

    stride_out_batch = out_flat.stride(0)
    stride_out_m     = out_flat.stride(1)
    stride_out_n     = out_flat.stride(2)

    #   d) launch Triton kernel
    BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 32
    grid_x = (N + BLOCK_M - 1) // BLOCK_M
    grid_y = (N + BLOCK_N - 1) // BLOCK_N
    grid = (grid_x * grid_y, batch_flat)

    tri_mul_kernel[grid](
        left_flat, right_flat, out_flat,
        N, N, N,
        stride_left_batch, stride_left_m, stride_left_k,
        stride_right_batch, stride_right_k, stride_right_n,
        stride_out_batch, stride_out_m, stride_out_n,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    #   e) reshape back to [B, N, N, hidden_dim]
    out_hidden = out_flat.view(B, D, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,D]

    # -------------------------------------------------------------------------
    # 4. Final LayerNorm, gating and output projection (still fp16)
    # -------------------------------------------------------------------------
    out_norm = F.layer_norm(out_hidden,
                            (hidden_dim,),
                            weight=to_out_norm_w,
                            bias=to_out_norm_b)                # fp16

    out_gated = out_norm * out_gate                           # broadcast over dim

    # final linear back to original dim
    output = F.linear(out_gated, to_out_w)                    # [B,N,N,dim] fp16

    # convert to float32 for the caller / loss calculation
    return output.to(torch.float32)
```
