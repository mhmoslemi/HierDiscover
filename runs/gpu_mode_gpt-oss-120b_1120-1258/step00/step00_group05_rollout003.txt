@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    EPS: tl.constexpr = 1e-5
):
    pid_bh = tl.program_id(2)
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)

    b = pid_bh // H
    h = pid_bh % H

    i_base = pid_i * BLOCK_M
    j_base = pid_j * BLOCK_N

    # range indices
    offs_i = i_base + tl.arange(0, BLOCK_M)
    offs_j = j_base + tl.arange(0, BLOCK_N)
    # mask for i and j
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k_iter in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k_iter * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load left (M x K) tile
        left_ptrs = left_ptr + b*stride_left_b + offs_i[:,None]*stride_left_i + offs_k[None,:]*stride_left_k + h*stride_left_h
        left = tl.load(left_ptrs, mask=mask_i[:,None] & mask_k[None,:], other=0.0)

        # load right (K x N) tile; right indices: row = k, col = j
        right_ptrs = right_ptr + b*stride_right_b + offs_j[None,:]*stride_right_i + offs_k[:,None]*stride_right_k + h*stride_right_h
        right = tl.load(right_ptrs, mask=mask_k[:,None] & mask_j[None,:], other=0.0)

        # dot
        acc += tl.dot(left, right)

    # store result
    out_ptrs = out_ptr + b*stride_out_b + offs_i[:,None]*stride_out_i + offs_j[None,:]*stride_out_j + h*stride_out_h
    tl.store(out_ptrs, acc, mask=mask_i[:,None] & mask_j[None,:])

import torch
import triton
import triton.language as tl
import math

def custom_kernel(data):
    """
    Triton-accelerated forward pass of the AlphaFold3 ``TriMul`` (outgoing) block.
    
    The implementation follows the reference PyTorch code:
      1. Layer‑norm on the input pair representation.
      2. Linear projections ``left`` and ``right`` (shape [B,N,N,H]).
      3. Optional masking, then element‑wise gates (sigmoid of linear layers).
      4. Contraction ``out[i,j] = Σ_k left[i,k] * right[j,k]`` for every batch
         and hidden channel – realised as a batched GEMM over the ``k`` dimension.
      5. Layer‑norm on the hidden dimension, gating, and final linear projection
         back to the original channel dimension.
    
    The heavy O(N³·H) contraction (step 4) is performed by a custom Triton kernel.
    All other operations stay in native PyTorch for simplicity.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights, config) where
        - input_tensor : torch.Tensor of shape [B, N, N, D] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (float32 or bool) or ``None``.
        - weights      : dict of pre‑loaded model parameters.
        - config       : dict containing ``dim`` (D) and ``hidden_dim`` (H) as well as
                         optional ``nomask`` flag.

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, D] – the result of the TriMul block.
    """
    # ----------------------------------------------------------------------
    # unpack arguments
    # ----------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ----------------------------------------------------------------------
    # convenience helpers – all ops stay on GPU, gradients disabled
    # ----------------------------------------------------------------------
    torch.set_grad_enabled(False)

    # ----------------------------------------------------------------------
    # 1. Layer‑norm on the input (per last dimension)
    # ----------------------------------------------------------------------
    eps = 1e-5
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )   # shape [B, N, N, D]

    # ----------------------------------------------------------------------
    # 2. Linear projections to hidden dimension (no bias)
    # ----------------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]          # [H, D]
    right_proj_weight = weights["right_proj.weight"]        # [H, D]
    left = torch.nn.functional.linear(x, left_proj_weight)   # [B,N,N,H]
    right = torch.nn.functional.linear(x, right_proj_weight) # [B,N,N,H]

    # ----------------------------------------------------------------------
    # 3. Optional mask (applied before gating)
    # ----------------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        # mask shape: [B,N,N] → broadcast to hidden dimension
        mask = mask_tensor.to(dtype).unsqueeze(-1)          # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ----------------------------------------------------------------------
    # 4. Gating (sigmoid of linear layers)
    # ----------------------------------------------------------------------
    left_gate_weight = weights["left_gate.weight"]          # [H, D]
    right_gate_weight = weights["right_gate.weight"]        # [H, D]
    out_gate_weight = weights["out_gate.weight"]            # [H, D]

    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ----------------------------------------------------------------------
    # 5. Contraction – the N³ operation (batched GEMM per hidden channel)
    # ----------------------------------------------------------------------
    B, N, _, _ = left.shape      # N is sequence length, dim == hidden_dim now
    # Make sure memory layout is contiguous for Triton
    left = left.contiguous()
    right = right.contiguous()

    # output tensor in hidden space
    out_hidden = torch.empty_like(left)   # [B,N,N,H], float32

    # strides (in elements, not bytes) – needed by the Triton kernel
    l_stride_b, l_stride_i, l_stride_k, l_stride_h = left.stride()
    r_stride_b, r_stride_i, r_stride_k, r_stride_h = right.stride()
    o_stride_b, o_stride_i, o_stride_j, o_stride_h = out_hidden.stride()

    # ----------------------------------------------------------------------
    # Triton kernel launch parameters
    # ----------------------------------------------------------------------
    BLOCK_M = 64   # rows (i)
    BLOCK_N = 64   # cols (j)
    BLOCK_K = 32   # reduction tile

    # grid dimensions:
    #   axis 0 -> i‑tiles, axis 1 -> j‑tiles, axis 2 -> (batch * hidden)
    grid_i = math.ceil(N / BLOCK_M)
    grid_j = math.ceil(N / BLOCK_N)
    grid_bh = B * hidden_dim
    grid = (grid_i, grid_j, grid_bh)

    # ----------------------------------------------------------------------
    # Triton kernel definition
    # ----------------------------------------------------------------------
    @triton.jit
    def tri_mul_outgoing_kernel(
        left_ptr, right_ptr, out_ptr,
        B, N, H,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_i, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    ):
        """
        Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

        Launch layout:
            pid_i  -> tile of the i (first N) dimension
            pid_j  -> tile of the j (second N) dimension
            pid_bh -> combined batch * hidden index (b * H + h)
        """
        pid_i = tl.program_id(0)   # i‑tile
        pid_j = tl.program_id(1)   # j‑tile
        pid_bh = tl.program_id(2)  # batch * hidden

        # decode batch and hidden index
        b = pid_bh // H
        h = pid_bh % H

        # Starting offsets of the current tile
        i_base = pid_i * BLOCK_M
        j_base = pid_j * BLOCK_N

        # Linear offsets for i, j and k
        offs_i = i_base + tl.arange(0, BLOCK_M)
        offs_j = j_base + tl.arange(0, BLOCK_N)

        mask_i = offs_i < N
        mask_j = offs_j < N

        # accumulator
        acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

        # iterate over k‑dimension in tiles
        num_k_tiles = tl.cdiv(N, BLOCK_K)
        for kk in range(0, num_k_tiles):
            k_base = kk * BLOCK_K
            offs_k = k_base + tl.arange(0, BLOCK_K)
            mask_k = offs_k < N

            # ------------------------------------------------------------------
            # Load left tile: shape (BLOCK_M, BLOCK_K)
            # left layout  (b, i, k, h)
            # ------------------------------------------------------------------
            left_ptrs = (left_ptr
                         + b * stride_l_b
                         + offs_i[:, None] * stride_l_i
                         + offs_k[None, :] * stride_l_k
                         + h * stride_l_h)
            left = tl.load(left_ptrs,
                           mask=(mask_i[:, None] & mask_k[None, :]),
                           other=0.0)

            # ------------------------------------------------------------------
            # Load right tile transposed: shape (BLOCK_K, BLOCK_N)
            # right layout (b, j, k, h) – we need (k, j) ordering for GEMM
            # ------------------------------------------------------------------
            right_ptrs = (right_ptr
                          + b * stride_r_b
                          + offs_j[None, :] * stride_r_i   # j dimension uses stride_i
                          + offs_k[:, None] * stride_r_k   # k dimension uses stride_k
                          + h * stride_r_h)
            right = tl.load(right_ptrs,
                            mask=(mask_k[:, None] & mask_j[None, :]),
                            other=0.0)

            # ------------------------------------------------------------------
            # Matrix multiply: (M x K)·(K x N) → (M x N)
            # ------------------------------------------------------------------
            acc += tl.dot(left, right)

        # ----------------------------------------------------------------------
        # Store the result
        # ----------------------------------------------------------------------
        out_ptrs = (out_ptr
                    + b * stride_o_b
                    + offs_i[:, None] * stride_o_i
                    + offs_j[None, :] * stride_o_j
                    + h * stride_o_h)
        tl.store(out_ptrs, acc,
                 mask=(mask_i[:, None] & mask_j[None, :]))

    # launch the kernel
    tri_mul_outgoing_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out_hidden,
        B=B,
        N=N,
        H=hidden_dim,
        stride_l_b=l_stride_b,
        stride_l_i=l_stride_i,
        stride_l_k=l_stride_k,
        stride_l_h=l_stride_h,
        stride_r_b=r_stride_b,
        stride_r_i=r_stride_i,
        stride_r_k=r_stride_k,
        stride_r_h=r_stride_h,
        stride_o_b=o_stride_b,
        stride_o_i=o_stride_i,
        stride_o_j=o_stride_j,
        stride_o_h=o_stride_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ----------------------------------------------------------------------
    # 6. Post‑processing (layer‑norm over hidden dim, gating, final projection)
    # ----------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # Layer‑norm on hidden dim
    out_norm = torch.nn.functional.layer_norm(
        out_hidden,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )

    # Apply output gate
    out_gated = out_norm * out_gate

    # Final linear back to `dim`
    out = torch.nn.functional.linear(out_gated, to_out_weight)  # [B,N,N,dim]

    return out

        )

        # ------------------------------------------------------------------
        # Load right tile transposed – we need (k, j) order for GEMM.
        # Original layout of `right` is (b, i, k, h); we treat `i` as the
        # j‑dimension here.
        # ------------------------------------------------------------------
        right_ptrs = (
            right_ptr
            + b * stride_r_b
            + offs_j[None, :] * stride_r_i   # j uses the `i` stride of the original tensor
            + offs_k[:, None] * stride_r_k   # k uses the `k` stride
            + h * stride_r_h
        )
        right = tl.load(
            right_ptrs,
            mask=(mask_k[:, None] & mask_j[None, :]),
            other=0.0,
        )

        # ------------------------------------------------------------------
        # Multiply‑accumulate
        # ------------------------------------------------------------------
        acc += tl.dot(left, right)   # (M×K)·(K×N) → (M×N)

    # ----------------------------------------------------------------------
    # Write the result back to global memory
    # ----------------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + b * stride_o_b
        + offs_i[:, None] * stride_o_i
        + offs_j[None, :] * stride_o_j
        + h * stride_o_h
    )
    tl.store(out_ptrs, acc, mask=(mask_i[:, None] & mask_j[None, :]))


def custom_kernel(data):
    """
    Forward pass of the TriMul block with a Triton kernel for the
    O(N³·H) contraction.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights, config) where
          - input_tensor : torch.Tensor [B, N, N, D] (float32)
          - mask_tensor  : torch.Tensor [B, N, N] (float32/bool) or None
          - weights      : dict of model parameters
          - config       : dict with keys ``dim`` (D) and ``hidden_dim`` (H);
                           may also contain ``nomask`` (bool).

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, D] – the processed pair representation.
    """
    # ----------------------------------------------------------------------
    # Unpack inputs
    # ----------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ----------------------------------------------------------------------
    # Helper constants
    # ----------------------------------------------------------------------
    eps = 1e-5
    torch.set_grad_enabled(False)

    # ----------------------------------------------------------------------
    # 1. Input layer‑norm over the last dimension
    # ----------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )   # shape [B, N, N, D]

    # ----------------------------------------------------------------------
    # 2. Linear projections to hidden dimension (no bias)
    # ----------------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ----------------------------------------------------------------------
    # 3. Optional masking (applied before gating)
    # ----------------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        mask = mask_tensor.to(dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ----------------------------------------------------------------------
    # 4. Gating (sigmoid of linear layers)
    # ----------------------------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ----------------------------------------------------------------------
    # 5. Contraction via Triton (batched GEMM over the k dimension)
    # ----------------------------------------------------------------------
    B, N, _, _ = left.shape   # N = sequence length
    # Ensure contiguous layout for pointer arithmetic
    left = left.contiguous()
    right = right.contiguous()

    # Output in hidden space
    out_hidden = torch.empty_like(left)   # [B, N, N, H]

    # Strides (in element units, not bytes)
    l_stride_b, l_stride_i, l_stride_k, l_stride_h = left.stride()
    r_stride_b, r_stride_i, r_stride_k, r_stride_h = right.stride()
    o_stride_b, o_stride_i, o_stride_j, o_stride_h = out_hidden.stride()

    # ----------------------------------------------------------------------
    # Triton launch configuration
    # ----------------------------------------------------------------------
    BLOCK_M = 64   # rows (i‑axis)
    BLOCK_N = 64   # cols (j‑axis)
    BLOCK_K = 32   # reduction tile size

    grid_i = math.ceil(N / BLOCK_M)
    grid_j = math.ceil(N / BLOCK_N)
    grid_bh = B * hidden_dim
    grid = (grid_i, grid_j, grid_bh)

    # Launch the kernel
    tri_mul_outgoing_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out_hidden,
        B=B,
        N=N,
        H=hidden_dim,
        stride_l_b=l_stride_b,
        stride_l_i=l_stride_i,
        stride_l_k=l_stride_k,
        stride_l_h=l_stride_h,
        stride_r_b=r_stride_b,
        stride_r_i=r_stride_i,
        stride_r_k=r_stride_k,
        stride_r_h=r_stride_h,
        stride_o_b=o_stride_b,
        stride_o_i=o_stride_i,
        stride_o_j=o_stride_j,
        stride_o_h=o_stride_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ----------------------------------------------------------------------
    # 6. Post‑processing: hidden‑dim layer‑norm, output gate, final linear
    # ----------------------------------------------------------------------
    out_norm = torch.nn.functional.layer_norm(
        out_hidden,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out_gated = out_norm * out_gate
    out = torch.nn.functional.linear(out_gated, weights["to_out.weight"])

    return out
