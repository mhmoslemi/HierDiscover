@triton.jit
def layer_norm_forward_kernel(
    x_ptr, y_ptr, weight_ptr, bias_ptr,
    B, N, C,
    stride_xb, stride_xi, stride_xj, stride_xc,
    stride_yb, stride_yi, stride_yj, stride_yc,
    stride_w, stride_b,
    eps,
    BLOCK_C: tl.constexpr
):
    b = tl.program_id(0)
    i = tl.program_id(1)
    j = tl.program_id(2)

    c_offset = tl.arange(0, BLOCK_C)
    # compute mask for out-of-range c dimension
    mask = c_offset < C

    x_ptrs = x_ptr + b*stride_xb + i*stride_xi + j*stride_xj + c_offset*stride_xc

    # load x values
    x = tl.load(x_ptrs, mask=mask, other=0.0).to(tl.float32)  # convert to fp32

    # Compute mean
    mean = tl.sum(x, axis=0) / C

    # Compute variance
    var = tl.sum((x - mean) * (x - mean), axis=0) / C

    # normalize
    eps = tl.full([1], eps, tl.float32)  # constant
    inv_std = 1.0 / tl.sqrt(var + eps)

    # weight & bias load
    w = tl.load(weight_ptr + c_offset*stride_w, mask=mask, other=0.0).to(tl.float32)
    b_ = tl.load(bias_ptr + c_offset*stride_b, mask=mask, other=0.0).to(tl.float32)

    y = (x - mean) * inv_std * w + b_

    # store result as float16 (or original dtype)
    tl.store(y_ptr + b*stride_yb + i*stride_yi + j*stride_yj + c_offset*stride_yc, y.to(tl.float16), mask=mask)

"""
custom_kernel:  Outgoing TriMul (AlphaFold‑3) forward pass.

The implementation follows the reference PyTorch module but moves the
following parts to Triton for speed:

*   Layer‑norm before the projections (LayerNorm over the last “dim” axis).
*   Fused mask‑gate multiplication for the left and right projected tensors.
*   Layer‑norm after the N³ reduction (LayerNorm over the hidden dimension).

The expensive N³ contraction is carried out as a batched GEMM using
half‑precision torch.bmm, which is heavily optimized on H100 GPUs.
All intermediate results are kept in FP16 and the final output is cast back
to FP32.

The kernel signatures use program_id(0/1/2) for (batch, i‑index, j‑index)
so that each Triton program works on a single (b,i,j) slice and loops over
the channel dimension (C or hidden_dim) in blocks of size BLOCK.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernels
# ----------------------------------------------------------------------

@triton.jit
def layer_norm_kernel(
    x_ptr, y_ptr, weight_ptr, bias_ptr,
    B, N, C,
    stride_xb, stride_xi, stride_xj, stride_xc,
    stride_yb, stride_yi, stride_yj, stride_yc,
    stride_wb,  # weight stride (always 1 for 1‑D weight)
    stride_bb,  # bias stride (always 1 for 1‑D bias)
    eps,
    BLOCK_C: tl.constexpr,
):
    """LayerNorm over the last dimension C.

    One program processes a single (b,i,j) slice.  The mean/variance are
    accumulated in FP32, the normalized result is written back in FP16.
    """
    b = tl.program_id(0)
    i = tl.program_id(1)
    j = tl.program_id(2)

    # Base pointers for this (b,i,j) slice
    x_base = x_ptr + b * stride_xb + i * stride_xi + j * stride_xj
    y_base = y_ptr + b * stride_yb + i * stride_yi + j * stride_yj

    # ------------------------------------------------------------------
    # 1) Compute mean
    # ------------------------------------------------------------------
    mean = tl.float32(0.0)
    offset = 0
    while offset < C:
        c = tl.arange(0, BLOCK_C) + offset
        mask = c < C
        vals = tl.load(x_base + c * stride_xc, mask=mask, other=0.0).to(tl.float32)
        mean += tl.sum(vals, axis=0)
        offset += BLOCK_C
    mean = mean / C

    # ------------------------------------------------------------------
    # 2) Compute variance
    # ------------------------------------------------------------------
    var = tl.float32(0.0)
    offset = 0
    while offset < C:
        c = tl.arange(0, BLOCK_C) + offset
        mask = c < C
        vals = tl.load(x_base + c * stride_xc, mask=mask, other=0.0).to(tl.float32)
        diff = vals - mean
        var += tl.sum(diff * diff, axis=0)
        offset += BLOCK_C
    var = var / C
    inv_std = 1.0 / tl.sqrt(var + eps)

    # ------------------------------------------------------------------
    # 3) Write normalized output
    # ------------------------------------------------------------------
    offset = 0
    while offset < C:
        c = tl.arange(0, BLOCK_C) + offset
        mask = c < C
        x = tl.load(x_base + c * stride_xc, mask=mask, other=0.0).to(tl.float32)
        w = tl.load(weight_ptr + c * stride_wb, mask=mask, other=0.0).to(tl.float32)
        b_ = tl.load(bias_ptr + c * stride_bb, mask=mask, other=0.0).to(tl.float32)

        out = (x - mean) * inv_std * w + b_
        tl.store(y_base + c * stride_yc, out.to(tl.float16), mask=mask)
        offset += BLOCK_C


@triton.jit
def mask_gate_fuse_kernel(
    tensor_ptr, gate_ptr, mask_ptr, out_ptr,
    B, N, C,
    stride_tb, stride_ti, stride_tj, stride_tc,
    stride_gb, stride_gi, stride_gj, stride_gc,
    stride_mb, stride_mi, stride_mj,
    BLOCK_C: tl.constexpr,
):
    """Fused (tensor * gate * mask) where mask is a scalar per (b,i,j)."""
    b = tl.program_id(0)
    i = tl.program_id(1)
    j = tl.program_id(2)

    # Base pointers for the current (b,i,j) slice
    t_base = tensor_ptr + b * stride_tb + i * stride_ti + j * stride_tj
    g_base = gate_ptr   + b * stride_gb + i * stride_gi + j * stride_gj
    out_base = out_ptr   + b * stride_tb + i * stride_ti + j * stride_tj  # same strides as tensor

    # Load the mask scalar (broadcasted over the channel dimension)
    m = tl.load(mask_ptr + b * stride_mb + i * stride_mi + j * stride_mj)
    m = m.to(tl.float16)

    offset = 0
    while offset < C:
        c = tl.arange(0, BLOCK_C) + offset
        mask = c < C
        t = tl.load(t_base + c * stride_tc, mask=mask, other=0.0)
        g = tl.load(g_base + c * stride_gc, mask=mask, other=0.0)
        out = t * g * m
        tl.store(out_base + c * stride_tc, out, mask=mask)
        offset += BLOCK_C

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)       # True -> no mask is applied
    eps = 1e-5

    # Cast everything to half precision for the heavy part
    x = input_tensor.to(torch.float16)

    # ------------------------------------------------------------------
    # LayerNorm (pre‑projection)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(torch.float16)
    norm_bias   = weights["norm.bias"].to(torch.float16)

    x_norm = torch.empty_like(x, dtype=torch.float16)

    layer_norm_kernel[
        (bs, seq_len, seq_len)
    ](
        x_ptr=x,
        y_ptr=x_norm,
        weight_ptr=norm_weight,
        bias_ptr=norm_bias,
        B=bs,
        N=seq_len,
        C=dim,
        stride_xb=x.stride(0), stride_xi=x.stride(1),
        stride_xj=x.stride(2), stride_xc=x.stride(3),
        stride_yb=x_norm.stride(0), stride_yi=x_norm.stride(1),
        stride_yj=x_norm.stride(2), stride_yc=x_norm.stride(3),
        stride_wb=norm_weight.stride(0), stride_bb=norm_bias.stride(0),
        eps=eps,
        BLOCK_C=64,
    )

    # ------------------------------------------------------------------
    # Linear projections (no bias) and gate linear layers
    # ------------------------------------------------------------------
    left_proj_weight  = weights["left_proj.weight"].to(torch.float16)
    right_proj_weight = weights["right_proj.weight"].to(torch.float16)

    left_gate_weight  = weights["left_gate.weight"].to(torch.float16)
    right_gate_weight = weights["right_gate.weight"].to(torch.float16)
    out_gate_weight   = weights["out_gate.weight"].to(torch.float16)

    # [B, N, N, hidden_dim]
    left = F.linear(x_norm, left_proj_weight)          # proj
    right = F.linear(x_norm, right_proj_weight)        # proj

    left_gate  = torch.sigmoid(F.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x_norm, right_gate_weight))
    out_gate   = torch.sigmoid(F.linear(x_norm, out_gate_weight))

    # ------------------------------------------------------------------
    # Apply mask + gate (fused Triton kernel) if a mask is present
    # ------------------------------------------------------------------
    if not nomask:
        # ensure mask is float16 and broadcastable
        mask = mask_tensor.to(torch.float16)
        # left: (B, i, k, hidden)   -> mask dims (B,i,k)
        mask_gate_fuse_kernel[
            (bs, seq_len, seq_len)
        ](
            tensor_ptr=left,
            gate_ptr=left_gate,
            mask_ptr=mask,
            out_ptr=left,
            B=bs,
            N=seq_len,
            C=hidden_dim,
            stride_tb=left.stride(0), stride_ti=left.stride(1),
            stride_tj=left.stride(2), stride_tc=left.stride(3),
            stride_gb=left_gate.stride(0), stride_gi=left_gate.stride(1),
            stride_gj=left_gate.stride(2), stride_gc=left_gate.stride(3),
            stride_mb=mask.stride(0), stride_mi=mask.stride(1), stride_mj=mask.stride(2),
            BLOCK_C=64,
        )
        # right: (B, j, k, hidden)   -> mask dims (B,j,k)
        mask_gate_fuse_kernel[
            (bs, seq_len, seq_len)
        ](
            tensor_ptr=right,

            mask_ptr=mask,
            out_ptr=right,
