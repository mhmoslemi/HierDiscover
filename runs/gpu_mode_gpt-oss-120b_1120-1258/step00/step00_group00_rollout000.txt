import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_batch, stride_l_i, stride_l_k, stride_l_d,
    stride_r_batch, stride_r_j, stride_r_k, stride_r_d,
    stride_o_batch, stride_o_i, stride_o_j, stride_o_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # head index (batch * hidden_dim + d)
    pid_head = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    batch_idx = pid_head // H
    d_idx = pid_head % H

    # starting offsets for this block
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out-of-bounds
    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointers for this batch and head
    left_base = left_ptr + batch_idx * stride_l_batch + d_idx * stride_l_d
    right_base = right_ptr + batch_idx * stride_r_batch + d_idx * stride_r_d

"""
custom_kernel
==============

A Triton‑accelerated forward implementation of the **outgoing** TriMul
operator used in AlphaFold‑3 and related protein‑structure models.

Algorithm
---------
1. **LayerNorm** over the channel dimension `dim`.
2. Linear projections `left_proj` / `right_proj` and gating projections.
3. Optional masking of `left` / `right`.
4. Apply element‑wise sigmoid gates on `left` and `right`.
5. **Core computation** (`einsum('... i k d, ... j k d -> ... i j d')`):
   for each batch and hidden channel `d` we compute a matrix product
   `left[..., :, :, d] @ right[..., :, :, d].T`.  This is implemented
   as a tiled GEMM in Triton (block‑wise accumulation over the `k`
   dimension) and the result is accumulated in FP32.
6. LayerNorm over the hidden dimension, multiply by the output gate,
   and a final linear projection back to the original `dim`.

Only the heavy N³ multiplication is done in Triton; all other
operations stay in PyTorch for simplicity and numerical stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_batch, stride_l_i, stride_l_k, stride_l_d,
    stride_r_batch, stride_r_j, stride_r_k, stride_r_d,
    stride_o_batch, stride_o_i, stride_o_j, stride_o_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute, for each (batch, hidden) pair:
        out[b, i, j, d] = sum_k left[b, i, k, d] * right[b, j, k, d]

    `left` and `right` are stored as [B, N, N, H] (fp16).
    The accumulator `out` is fp32.
    """
    pid_head = tl.program_id(0)   # batch * H + hidden
    pid_i = tl.program_id(1)      # block row index
    pid_j = tl.program_id(2)      # block col index

    # decode batch and hidden index
    batch_idx = pid_head // H
    d_idx = pid_head % H

    # compute absolute offsets for this block
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # out‑of‑bounds masks for the output tile
    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointers for this (batch, hidden) slice
    left_base  = left_ptr  + batch_idx * stride_l_batch + d_idx * stride_l_d
    right_base = right_ptr + batch_idx * stride_r_batch + d_idx * stride_r_d
    out_base   = out_ptr   + batch_idx * stride_o_batch + d_idx * stride_o_d

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension k
    for kb in range(0, tl.cdiv(N, BLOCK_K)):
        k_off = kb * BLOCK_K
        offs_k = k_off + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ----- load left tile (M x K) -----
        a_ptrs = left_base + offs_i[:, None] * stride_l_i + offs_k[None, :] * stride_l_k
        a_mask = mask_i[:, None] & mask_k[None, :]
        a = tl.load(a_ptrs, mask=a_mask, other=0.0)   # fp16

        # ----- load right tile (K x N) -----
        # right is stored as (j, k); we need (k, j) for the multiplication,
        # i.e. we treat it as transposed on‑the‑fly.
        b_ptrs = right_base + offs_k[:, None] * stride_r_k + offs_j[None, :] * stride_r_j
        b_mask = mask_k[:, None] & mask_j[None, :]
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)   # fp16

        # accumulate dot product (fp16 inputs, fp32 accumulator)
        acc += tl.dot(a, b)

    # ----- store the result tile -----
    c_ptrs = out_base + offs_i[:, None] * stride_o_i + offs_j[None, :] * stride_o_j
    c_mask = mask_i[:, None] & mask_j[:, None]
    tl.store(c_ptrs, acc, mask=c_mask)


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, dim]
        - mask         : torch.Tensor of shape [B, N, N] or None
        - weights      : dict of torch.Tensors containing the model parameters
        - config       : dict with keys ``dim`` and ``hidden_dim`` and optional
                         ``nomask`` flag

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, on the same device).
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) Input layer‑norm (over the last dimension)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (float32)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Optional masking (mask shape [B,N,N])
    # ------------------------------------------------------------------
    if mask is not None and not config.get("nomask", False):
        mask = mask.to(dtype=x.dtype).unsqueeze(-1)          # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Triton kernel for the core N³ multiplication
    # ------------------------------------------------------------------
    # Cast to half for the kernel (more bandwidth friendly)
    left_fp16  = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()

    B, N, _, _ = left_fp16.shape
    # allocate output tensor (fp32 accumulator)
    out_fp32 = torch.empty((B, N, N, hidden_dim), dtype=torch.float32,
                           device=left_fp16.device)

    # strides (in elements, not bytes)
    stride_l_b, stride_l_i, stride_l_k, stride_l_d = left_fp16.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_d = right_fp16.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_d = out_fp32.stride()

    # tile sizes – tuned for H100
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    # launch grid: (num_heads, rows, cols)
    num_heads = B * hidden_dim
    grid = (
        num_heads,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _tri_mul_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp32,
        B,
        N,
        hidden_dim,
        stride_l_b,
        stride_l_i,
        stride_l_k,
        stride_l_d,
        stride_r_b,
        stride_r_i,          # j‑stride (second spatial dimension)
        stride_r_k,
        stride_r_d,
        stride_o_b,
        stride_o_i,
        stride_o_j,
        stride_o_d,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6) Output post‑processing (layer‑norm, gating, final linear)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out_fp32,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate                      # element‑wise gate
    out = F.linear(out, weights["to_out.weight"])  # [B,N,N,dim]

    return out
