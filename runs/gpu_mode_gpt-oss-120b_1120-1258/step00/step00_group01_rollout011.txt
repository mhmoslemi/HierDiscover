@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    B, H, N,
    stride_a_batch, stride_a_h, stride_a_i, stride_a_k,
    stride_b_batch, stride_b_h, stride_b_k, stride_b_j,
    stride_c_batch, stride_c_h, stride_c_i, stride_c_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_bh = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)
    # compute combined batch+hidden indices
    b_idx = pid_bh // H
    h_idx = pid_bh % H

    # Calculate pointers base offset for each via strides
    a_base = a_ptr + b_idx * stride_a_batch + h_idx * stride_a_h
    b_base = b_ptr + b_idx * stride_b_batch + h_idx * stride_b_h
    c_base = c_ptr + b_idx * stride_c_batch + h_idx * stride_c_h

    # Offsets in M,N dims
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # masks
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # pointers to A and B tiles
        a_ptrs = a_base + (offs_m[:, None] * stride_a_i) + (offs_k[None, :] * stride_a_k)
        b_ptrs = b_base + (offs_k[:, None] * stride_b_k) + (offs_n[None, :] * stride_b_j)

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)

    # Write back
    c_ptrs = c_base + (offs_m[:, None] * stride_c_i) + (offs_n[None, :] * stride_c_j)
    c = acc.to(tl.float16)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])
```

"""
custom_kernel: high‑performance implementation of the “outgoing” TriMul operation
used in AlphaFold‑3 and related protein‑structure models.

The kernel fuses the N³ inner product:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
where left/right are the linearly projected and gated pairwise tensors.
We treat each (batch, hidden) pair as an independent matrix multiplication
(left : N×N) × (rightᵀ : N×N) and compute it with a Triton matmul that
uses FP16 inputs and FP32 accumulation (Tensor‑Core friendly).  All other
operations (LayerNorm, Linear, gating, final projection) are performed in
PyTorch; only the N³‑dominant computation is off‑loaded to the custom kernel.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    # pointers
    a_ptr, b_ptr, c_ptr,
    # problem size & batch info
    B, H, N,
    # strides for A (left)   : (batch, hidden, i, k)
    stride_a_batch, stride_a_h, stride_a_i, stride_a_k,
    # strides for B (rightᵀ) : (batch, hidden, k, j)
    stride_b_batch, stride_b_h, stride_b_k, stride_b_j,
    # strides for C (output) : (batch, hidden, i, j)
    stride_c_batch, stride_c_h, stride_c_i, stride_c_j,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched matmul for each (batch, hidden) → (N×N)·(N×N)ᵀ."""
    pid_bh = tl.program_id(0)   # combined batch * hidden index
    pid_m  = tl.program_id(1)   # tile row
    pid_n  = tl.program_id(2)   # tile column

    # split combined index back to batch / hidden
    b_idx = pid_bh // H
    h_idx = pid_bh % H

    # base pointers for this (b,h)
    a_base = a_ptr + b_idx * stride_a_batch + h_idx * stride_a_h
    b_base = b_ptr + b_idx * stride_b_batch + h_idx * stride_b_h
    c_base = c_ptr + b_idx * stride_c_batch + h_idx * stride_c_h

    # offsets within the N×N matrix
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension K (= N)
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # pointers to the current tiles
        a_ptrs = a_base + (offs_m[:, None] * stride_a_i) + (offs_k[None, :] * stride_a_k)
        b_ptrs = b_base + (offs_k[:, None] * stride_b_k) + (offs_n[None, :] * stride_b_j)

        # load tiles with boundary masking
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)          # (BLOCK_M, BLOCK_K)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)          # (BLOCK_K, BLOCK_N)

        # tile‑wise matrix multiply, accumulation in FP32
        acc += tl.dot(a, b)

    # write the result (cast back to FP16)
    c_ptrs = c_base + (offs_m[:, None] * stride_c_i) + (offs_n[None, :] * stride_c_j)
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weight_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # --------------------------------------------------------------------- #
    # Unpack arguments
    # --------------------------------------------------------------------- #
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]          # corresponds to `hidden_dim` in the paper

    # --------------------------------------------------------------------- #
    # Helper aliases for weight tensors (all are FP32 and bias‑free unless noted)
    # --------------------------------------------------------------------- #
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    left_proj_weight  = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]

    # --------------------------------------------------------------------- #
    # 1) Input LayerNorm over the channel dimension `dim`
    # --------------------------------------------------------------------- #
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias)                         # [B, N, N, dim]  FP32

    # --------------------------------------------------------------------- #
    # 2) Linear projections (no bias)
    # --------------------------------------------------------------------- #
    left = F.linear(x, left_proj_weight)      # [B, N, N, hidden_dim]    FP32
    right = F.linear(x, right_proj_weight)    # same shape

    # --------------------------------------------------------------------- #
    # 3) Optional mask (mask is broadcast on the hidden dimension)
    # --------------------------------------------------------------------- #
    if mask_tensor is not None:
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # --------------------------------------------------------------------- #
    # 4) Gating (sigmoid of linear transforms of the *norm‑ed* input)
    # --------------------------------------------------------------------- #
    left_gate  = F.linear(x, left_gate_weight).sigmoid()   # [B, N, N, hidden_dim]
    right_gate = F.linear(x, right_gate_weight).sigmoid()
    out_gate   = F.linear(x, out_gate_weight).sigmoid()

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------------- #
    # 5) Cast projections to FP16 for the heavy N³ matmul
    # --------------------------------------------------------------------- #
    left_h  = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # --------------------------------------------------------------------- #
    # 6) Reshape / permute so that each (batch, hidden) pair is a separate
    #    N×N matrix.  The kernel will treat the hidden dimension as an extra
    #    batch axis.
    #       left_perm : [B, H, N, N]   with layout (b, h, i, k)
    #       right_perm_T : [B, H, N, N] layout (b, h, k, j)
    # --------------------------------------------------------------------- #
    left_perm = left_h.permute(0, 3, 1, 2)           # (B, H, N, N)
    right_perm_T = right_h.permute(0, 3, 2, 1)      # (B, H, N, N)
    out_perm = torch.empty_like(left_perm)         # FP16 output buffer

    # --------------------------------------------------------------------- #
    # 7) Launch Triton kernel
    # --------------------------------------------------------------------- #
    B, H, N, _ = left_perm.shape          # B, hidden, seq_len, seq_len (last dim == N)
    # Tile sizes – chosen to map well to Tensor Cores on H100.
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 128

    grid = (
        B * H,                              # one program per (batch, hidden) pair
        (N + BLOCK_M - 1) // BLOCK_M,       # tiles over rows   (i dimension)
        (N + BLOCK_N - 1) // BLOCK_N,       # tiles over cols   (j dimension)
    )

    # Strides for the permuted tensors (all int64)
    stride_a_batch, stride_a_h, stride_a_i, stride_a_k = left_perm.stride()
    stride_b_batch, stride_b_h, stride_b_k, stride_b_j = right_perm_T.stride()
    stride_c_batch, stride_c_h, stride_c_i, stride_c_j = out_perm.stride()

    tri_mul_kernel[grid](
        left_perm,
        right_perm_T,
        out_perm,
        B, H, N,
        stride_a_batch, stride_a_h, stride_a_i, stride_a_k,
        stride_b_batch, stride_b_h, stride_b_k, stride_b_j,
        stride_c_batch, stride_c_h, stride_c_i, stride_c_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )

    # --------------------------------------------------------------------- #
    # 8) Bring tensor back to original layout [B, N, N, hidden_dim]
    # --------------------------------------------------------------------- #
    out = out_perm.permute(0, 2, 3, 1).contiguous()   # [B, N, N, hidden_dim]  FP16
    out = out.to(torch.float32)                       # cast back for the rest of the pipeline

    # --------------------------------------------------------------------- #
    # 9) Output LayerNorm (over hidden_dim), gate, and final linear projection
    # --------------------------------------------------------------------- #
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias)          # FP32
    out = out * out_gate                                 # element‑wise gating
    out = F.linear(out, to_out_weight)                  # final projection → [B, N, N, dim]

    return out
