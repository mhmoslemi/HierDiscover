def batched_matmul_kernel(A_ptr, B_ptr, C_ptr,
                          stride_am, stride_ak, stride_bk, stride_bn,
                          stride_cm, stride_cn,
                          batch_stride_a, batch_stride_b, batch_stride_c,
                          M, N, K,
                          BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):

    pid = tl.program_id(0)
    batch = tl.program_id(1)
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    pid_m = pid // num_blocks_n
    pid_n = pid % num_blocks_n

    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

"""
custom_kernel for the TriMul “outgoing” operation (AlphaFold‑3).

Algorithm
---------
1. Layer‑norm the 4‑D input (shape B×N×N×dim) using the provided
   `norm.{weight,bias}`.
2. Compute two linear projections `left` and `right` (no bias) from the
   normalized tensor using `left_proj.weight` and `right_proj.weight`.
3. If a mask is supplied (or `nomask == False` in the config) multiply
   `left` and `right` by the mask (broadcast on the hidden dimension).
4. Compute three sigmoid gates from the original normalized tensor:
   `left_gate`, `right_gate`, `out_gate`.
5. Apply the gates: `left ← left * left_gate`, `right ← right * right_gate`.
6. The core “outgoing” multiplication:
        out[i,j,d] = Σ_k left[i,k,d] * right[j,k,d]
   i.e. for each hidden channel `d` a matrix multiplication of
   `left[...,d]` with the transpose of `right[...,d]`.
   This N×N matrix product is executed with a custom Triton kernel that
   batches over the `(batch * hidden_dim)` dimension.
7. Layer‑norm the result with `to_out_norm.{weight,bias}`,
   multiply by `out_gate` and apply the final linear projection
   `to_out.weight` (dim ← hidden_dim).
8. Return the tensor of shape B×N×N×dim (float32).

The kernel is a classic tiled GEMM written in Triton and operates on
float‑32 data.  Only the heavy N³ multiplication is off‑loaded to Triton;
all other operations stay in PyTorch for simplicity.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication  (A @ Bᵀ)
#   A : (P, M, K)   – row‑major
#   B : (P, K, N)   – row‑major (already transposed, i.e. original rightᵀ)
#   C : (P, M, N)
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,                     # pointers
    stride_am, stride_ak,                    # A strides (M‑stride, K‑stride)
    stride_bk, stride_bn,                    # B strides (K‑stride, N‑stride)
    stride_cm, stride_cn,                    # C strides (M‑stride, N‑stride)
    batch_stride_a, batch_stride_b, batch_stride_c,  # stride between batches
    M, N, K,                                 # matrix sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # --------------------------------------------------------------
    # tile indices
    # --------------------------------------------------------------
    pid = tl.program_id(0)          # tile over (M,N)
    batch = tl.program_id(1)        # which (batch, hidden) we are processing

    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)

    pid_m = pid // num_blocks_n
    pid_n = pid % num_blocks_n

    # --------------------------------------------------------------
    # compute offsets for this tile
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers to the first element of the tile (including batch offset)
    a_ptrs = A_ptr + batch * batch_stride_a + offs_m[:, None] * stride_am + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    b_ptrs = B_ptr + batch * batch_stride_b + tl.arange(0, BLOCK_K)[:, None] * stride_bk + offs_n[None, :] * stride_bn

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # loop over K dimension in tiling fashion
    # --------------------------------------------------------------
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (tl.arange(0, BLOCK_K) < K)[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=(tl.arange(0, BLOCK_K) < K)[:, None] & mask_n[None, :],
                    other=0.0)
        acc += tl.dot(a, b)

        # advance K‑tiles
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # --------------------------------------------------------------
    # write back the result tile
    # --------------------------------------------------------------
    c_ptrs = C_ptr + batch * batch_stride_c + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point expected by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” block.
    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape (B, N, N, dim) on the same device as the inputs.
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    # shapes
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1. Input layer‑norm
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])   # (B,N,N,hidden_dim)
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask
    # ------------------------------------------------------------------
    if mask is not None and not config.get("nomask", False):
        # mask shape: (B, N, N) -> (B, N, N, 1) to broadcast over hidden_dim
        mask_exp = mask.to(dtype).unsqueeze(-1)
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4. Gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core TriMul (batched GEMM over hidden dimension)
    # ------------------------------------------------------------------
    # bring hidden dimension to the 2nd axis for contiguity
    left_mat = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    # transpose right so that we compute A @ Bᵀ with row‑major matrices
    right_T = right_mat.transpose(-2, -1).contiguous()  # (B, H, N, N)

    # flatten the (batch, hidden) pair into a single batch dimension
    P = B * hidden_dim
    left_flat = left_mat.view(P, N, N)      # (P, M=N, K=N)
    right_flat = right_T.view(P, N, N)      # (P, K=N, N)

    # allocate output
    out_flat = torch.empty_like(left_flat)  # (P, N, N)

    # --------------------------------------------------------------
    # launch Triton kernel
    # --------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # strides for the flattened tensors (they are contiguous in row‑major)
    stride_am = left_flat.stride(1)   # N
    stride_ak = left_flat.stride(2)   # 1
    stride_bk = right_flat.stride(1)  # N
    stride_bn = right_flat.stride(2)  # 1
    stride_cm = out_flat.stride(1)    # N
    stride_cn = out_flat.stride(2)    # 1

    batch_stride_a = left_flat.stride(0)   # N * N
    batch_stride_b = right_flat.stride(0)
    batch_stride_c = out_flat.stride(0)

    grid = (
        tl.cdiv(N, BLOCK_M) * tl.cdiv(N, BLOCK_N),   # tiles over (M,N)
        P                                            # one program per (batch, hidden)
    )

    batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        stride_am,
        stride_ak,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        batch_stride_a,
        batch_stride_b,
        batch_stride_c,
        N,          # M
        N,          # N
        N,          # K
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, hidden_dim)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # (B,N,N,hidden_dim)

    # ------------------------------------------------------------------
    # 6. Output norm, out‑gate and final linear projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])  # (B,N,N,dim)

    return out
