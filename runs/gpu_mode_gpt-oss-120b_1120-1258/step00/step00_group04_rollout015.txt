def custom_kernel(data):
    input_tensor, mask, weights, config = data
    # Unpack config
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # Load weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    eps = 1e-5

    # LayerNorm over dim
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps
    )

    # Linear projections (no bias)
    # shape: [B,N,N,hidden_dim]
    left = torch.nn.functional.linear(x, left_proj_weight)
    right = torch.nn.functional.linear(x, right_proj_weight)

    # Apply mask (if any)
    if not nomask:
        # mask shape [B,N,N]; broadcast
        mask_exp = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_exp
        right = right * mask_exp
    else:
        # No mask -> nothing
        pass

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Prepare for Triton multiplication: permute dims to put hidden_dim as second axis
    # left/right shape [B,N,N,H] -> [B, H, N, N]
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    B, H, N, _ = left_perm.shape
    M_total = B * H  # Number of independent matrix multiplications

    # Flatten batch and channel
    left_flat = left_perm.view(M_total, N, N)
    right_flat = right_perm.view(M_total, N, N)

    # Allocate output
    out_flat = torch.empty_like(left_flat)

    # Triton matmul kernel for sum over k (N) and produce result shape (M_total, N, N)
    # Define block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # Get strides
    stride_am = left_flat.stride(0)
    stride_ai = left_flat.stride(1)
    stride_ak = left_flat.stride(2)

    stride_bm = right_flat.stride(0)
    stride_bj = right_flat.stride(1)  # j dimension (original second)
    stride_bk = right_flat.stride(2)  # k dimension (original third)

    stride_cm = out_flat.stride(0)
    stride_ci = out_flat.stride(1)
    stride_cj = out_flat.stride(2)

    # Launch kernel
    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), M_total)
    tri_mul_outgoing_kernel[grid](
        left_flat, right_flat, out_flat,
        N,
        stride_am, stride_ai, stride_ak,
        stride_bm, stride_bj, stride_bk,
        stride_cm, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # Reshape back
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # to_out_norm over hidden_dim
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps
    )

    # apply out_gate
    out = out * out_gate

    # final linear projection to dim
    output = torch.nn.functional.linear(out, to_out_weight)

    return output

@triton.jit
def tri_mul_outgoing_kernel(
    a_ptr, b_ptr, c_ptr,
    N,
    stride_am, stride_ai, stride_ak,
    stride_bm, stride_bj, stride_bk,
    stride_cm, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)  # tile row
    pid_n = tl.program_id(1)  # tile col
    pid_b = tl.program_id(2)  # batch*channel index

    # compute offsets
    i_base = pid_m * BLOCK_M
    j_base = pid_n * BLOCK_N

    # ranges
    i = i_base + tl.arange(0, BLOCK_M)
    j = j_base + tl.arange(0, BLOCK_N)

    # masks for out-of-range rows/cols
    mask_i = i < N
    mask_j = j < N

    # pointer to output
    c_ptrs = c_ptr + pid_b * stride_cm + i[:, None] * stride_ci + j[None, :] * stride_cj

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k_base in range(0, N, BLOCK_K):
        k = k_base + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # Load A tile (row i, col k)
        a_ptrs = a_ptr + pid_b * stride_am + i[:, None] * stride_ai + k[None, :] * stride_ak

```python
import torch
import triton
import triton.language as tl
import math

@triton.jit
def tri_mul_outgoing_kernel(
    a_ptr, b_ptr, c_ptr,
    N,
    stride_am, stride_ai, stride_ak,
    stride_bm, stride_bj, stride_bk,
    stride_cm, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_m = tl.program_id(0)  # i tile
    pid_n = tl.program_id(1)  # j tile
    pid_b = tl.program_id(2)  # batch*hidden index

    i0 = pid_m * BLOCK_M
    j0 = pid_n * BLOCK_N

    i = i0 + tl.arange(0, BLOCK_M)
    j = j0 + tl.arange(0, BLOCK_N)

    mask_i = i < N
    mask_j = j < N

    # Compute pointer to output tile
    c_ptrs = c_ptr + pid_b * stride_cm + i[:, None] * stride_ci + j[None, :] * stride_cj

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    for k0 in range(0, N, BLOCK_K):
        k = k0 + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # Load A[i, k]
        a_ptrs = a_ptr + pid_b * stride_am + i[:, None] * stride_ai + k[None, :] * stride_ak
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load B[k, j] i.e. transposed tile from original B[j, k]
        b_ptrs = b_ptr + pid_b * stride_bm + k[:, None] * stride_bk + j[None, :] * stride_bj
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        acc += tl.dot(a, b)

    # Store result
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

```python
def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    # load weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # LayerNorm over dim
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps
    )

    # Linear projections (no bias)
    left = torch.nn.functional.linear(x, left_proj_weight)   # [B,N,N,H]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # Apply mask if present
    if not nomask:
        # mask: [B,N,N], broadcast to last dim
        mask_exp = mask.to(left.dtype).unsqueeze(-1)
        left = left * mask_exp
        right = right * mask_exp

    # Compute gates (sigmoid)
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Rearrange for batched GEMM: (B, H, N, N) -> (M=N*H*B, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    B, H, N, _ = left_perm.shape
    M_total = B * H

    left_flat = left_perm.view(M_total, N, N)
    right_flat = right_perm.view(M_total, N, N)

    out_flat = torch.empty_like(left_flat)

    # block sizes (tunable)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # strides for kernel (int)
    stride_am = left_flat.stride(0)
    stride_ai = left_flat.stride(1)
    stride_ak = left_flat.stride(2)

    stride_bm = right_flat.stride(0)
    stride_bj = right_flat.stride(1)   # j dimension (original second)
    stride_bk = right_flat.stride(2)   # k dimension (original third)

    stride_cm = out_flat.stride(0)
    stride_ci = out_flat.stride(1)
    stride_cj = out_flat.stride(2)

    # launch kernel
    grid = ( (N + BLOCK_M - 1) // BLOCK_M,
             (N + BLOCK_N - 1) // BLOCK_N,
             M_total )
    tri_mul_outgoing_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,

"""
Triangular Multiplicative Update (outgoing) custom kernel.

The implementation follows the AlphaFold3 reference:

1. Layer‑norm over the input feature dimension.
2. Linear projections for “left”, “right” and three sigmoid gates.
3. Optional masking of the projected tensors.
4. Gating of left/right tensors.
5. Heavy O(N³·H) contraction
      out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   is fused into a Triton kernel that treats each (batch, hidden) pair as an
   independent GEMM C = A @ Bᵀ with A,B ∈ ℝ^{N×N}.  The kernel tiles the i,
   j and k dimensions (BLOCK_M×BLOCK_N×BLOCK_K) and accumulates with
   `tl.dot`, achieving high occupancy on an H100.
6. Layer‑norm over the hidden dimension, element‑wise multiplication with the
   output gate, and a final linear projection back to the original feature size.
"""

import math
import torch
import triton
import triton.language as tl


@triton.jit
def tri_mul_outgoing_kernel(
    a_ptr,               # left  (M, N, K)   – already contiguous
    b_ptr,               # right (M, N, K)   – we read it transposed as (K, N)
    c_ptr,               # output (M, N, N)
    N,                   # sequence length (both operands are N×N)
    stride_am, stride_ai, stride_ak,   # A strides: batch, i, k
    stride_bm, stride_bj, stride_bk,   # B strides: batch, j, k   (read as k,j)
    stride_cm, stride_ci, stride_cj,   # C strides: batch, i, j
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)            # tile along i  (rows)
    pid_n = tl.program_id(1)            # tile along j  (cols)
    pid_b = tl.program_id(2)            # batch × hidden index

    i0 = pid_m * BLOCK_M
    j0 = pid_n * BLOCK_N

    i = i0 + tl.arange(0, BLOCK_M)
    j = j0 + tl.arange(0, BLOCK_N)

    mask_i = i < N
    mask_j = j < N

    # output pointer for this (i,j) tile
    c_ptrs = c_ptr + pid_b * stride_cm + i[:, None] * stride_ci + j[None, :] * stride_cj
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over the reduction dimension k
    for k0 in range(0, N, BLOCK_K):
        k = k0 + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # ----- load A[i, k] -------------------------------------------------
        a_ptrs = a_ptr + pid_b * stride_am + i[:, None] * stride_ai + k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # ----- load B[k, j]  (B is originally stored as B[j, k]) ------------
        b_ptrs = b_ptr + pid_b * stride_bm + k[:, None] * stride_bk + j[None, :] * stride_bj
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0).to(tl.float32)

        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

    # write back the tile
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module using a Triton kernel for the
    cubic contraction.  Arguments are packed into `data` as
    (input_tensor, mask, weights, config).

    Returns:
        torch.Tensor of shape [bs, seq_len, seq_len, dim] (float32, on CUDA).
    """
    input_tensor, mask, weights, config = data

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    # --------------------------------------------------------------------- #
    # unpack weights (all are torch.Tensor, already on the correct device) #
    # --------------------------------------------------------------------- #
    norm_weight      = weights["norm.weight"]
    norm_bias        = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight= weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight= weights["right_gate.weight"]
    out_gate_weight  = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]

    # -------------------------------------------------------------- #
    # 1. LayerNorm over the input feature dimension (dim)            #
    # -------------------------------------------------------------- #
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, dim]

    # -------------------------------------------------------------- #
    # 2. Linear projections (bias‑free)                              #
    # -------------------------------------------------------------- #
    left  = torch.nn.functional.linear(x, left_proj_weight)   # [B, N, N, H]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # -------------------------------------------------------------- #
    # 3. Optional mask (broadcasted on the hidden dimension)       #
    # -------------------------------------------------------------- #
    if not nomask:
        mask_exp = mask.to(left.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left  = left * mask_exp
        right = right * mask_exp

    # -------------------------------------------------------------- #
    # 4. Gating (sigmoid)                                           #
    # -------------------------------------------------------------- #
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    left  = left * left_gate
    right = right * right_gate

    # -------------------------------------------------------------- #
    # 5. Triton kernel for the cubic contraction                     #
    #    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]           #
    # -------------------------------------------------------------- #
    # Rearrange tensors so that each (batch, hidden) pair becomes a
    # separate matrix multiplication: (M, N, N) where M = B * H.
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    B, H, N, _ = left_perm.shape
    M_total = B * H

    left_flat  = left_perm.view(M_total, N, N)   # [M, N, N]
    right_flat = right_perm.view(M_total, N, N)  # [M, N, N]
    out_flat   = torch.empty_like(left_flat)     # will hold [M, N, N]

    # Block sizes – can be tuned; these values fit comfortably on H100.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # Strides for the kernel (they are Python ints)
    stride_am = left_flat.stride(0)
    stride_ai = left_flat.stride(1)
    stride_ak = left_flat.stride(2)

    stride_bm = right_flat.stride(0)
    # right_flat layout is (M, j, k).  For the transposed view we need:
    #   stride over k (third dim)      = 1
    #   stride over j (second dim)     = N
    stride_bj = right_flat.stride(1)   # stride for j  (original second dim)
    stride_bk = right_flat.stride(2)   # stride for k  (original third dim)

    stride_cm = out_flat.stride(0)
    stride_ci = out_flat.stride(1)
    stride_cj = out_flat.stride(2)

    # Launch the Triton kernel
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # tiles along i
        (N + BLOCK_N - 1) // BLOCK_N,   # tiles along j
        M_total                         # one kernel per (batch, hidden) pair
    )
    tri_mul_outgoing_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        stride_am, stride_ai, stride_ak,
        stride_bm, stride_bj, stride_bk,
        stride_cm, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -------------------------------------------------------------- #
    # 6. LayerNorm over the hidden dimension, gating, final linear   #
    # -------------------------------------------------------------- #
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    out = out * out_gate
    output = torch.nn.functional.linear(out, to_out_weight)   # [B, N, N, dim]

    return output
