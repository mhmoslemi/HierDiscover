@triton.jit
def trilu_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_bc = tl.program_id(2)

    # compute batch and channel
    b = pid_bc // H
    h = pid_bc % H

    # compute offsets within N
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    # mask out of bounds
    mask_i = offs_i < N
    mask_j = offs_j < N

    # initialize accumulator
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # Loop over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # start offset k
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # compute pointer offsets for left and right tile
        # left tile: shape (BLOCK_I, BLOCK_K)
        left_ptrs = left_ptr + b * stride_l_b + h * stride_l_h + offs_i[:, None] * stride_l_i + offs_k[None, :] * stride_l_k
        # right tile: shape (BLOCK_K, BLOCK_J) because we load as (k,j) for multiplication
        right_ptrs = right_ptr + b * stride_r_b + h * stride_r_h + offs_k[:, None] * stride_r_k + offs_j[None, :] * stride_r_i

        # load
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        # accumulate dot product across k dimension
        # left: (BLOCK_I, BLOCK_K) ; right: (BLOCK_K, BLOCK_J)
        acc += tl.dot(left, right)  # shape (BLOCK_I, BLOCK_J)

    # store result to out
    out_ptrs = out_ptr + b * stride_o_b + h * stride_o_h + offs_i[:, None] * stride_o_i + offs_j[None, :] * stride_o_j
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

"""
Triangle Multiplicative Update (outgoing) – Triton implementation.

The kernel computes, for every batch b and hidden channel h,
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

where left / right are linear projections of the input tensor (with gating and optional masking).
The computation is a batched matrix multiply over the sequence dimension,
implemented as a custom Triton kernel that tiles the i‑j plane and reduces over k.
All other operations (LayerNorm, Linear projections, gating) are performed with
torch.nn.functional which already use highly‑optimized CUDA kernels.
The final output is passed through a second LayerNorm, gated by `out_gate`,
and projected back to the original feature dimension.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ---------------------------------------------------------
# Triton kernel: batched matmul over the sequence dimension
# ---------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,               # pointers
    B, N, H,                                     # B=batch, N=seq_len, H=hidden_dim
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Matrix multiply: (B*H, N, N) @ (B*H, N, N)^T -> (B*H, N, N)"""
    pid_i = tl.program_id(0)          # i‑block index
    pid_j = tl.program_id(1)          # j‑block index
    pid_bc = tl.program_id(2)         # combined batch*channel index

    # de‑compose batch and channel
    b = pid_bc // H
    h = pid_bc % H

    # offsets within the i / j dimensions
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator (fp32)
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # loop over the reduction axis k
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for kb in range(0, num_k_blocks):
        offs_k = kb * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile (i, k) – shape (BLOCK_I, BLOCK_K)
        left_ptrs = left_ptr + \
            b * stride_l_b + h * stride_l_h + \
            offs_i[:, None] * stride_l_i + offs_k[None, :] * stride_l_k
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # right tile (k, j) – shape (BLOCK_K, BLOCK_J)
        right_ptrs = right_ptr + \
            b * stride_r_b + h * stride_r_h + \
            offs_k[:, None] * stride_r_k + offs_j[None, :] * stride_r_i
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0).to(tl.float32)

        # accumulate the block‑matrix product
        acc += tl.dot(left, right)          # (BLOCK_I, BLOCK_J)

    # write the result
    out_ptrs = out_ptr + \
        b * stride_o_b + h * stride_o_h + \
        offs_i[:, None] * stride_o_i + offs_j[None, :] * stride_o_j
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

    # convenience aliases for weight tensors
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # LayerNorm on the input feature dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=1e-5)

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)          # [B, N, N, H]
    right = F.linear(x, right_proj_weight)        # [B, N, N, H]

    # ------------------------------------------------------------------
    # Mask handling (if a mask is provided and not disabled)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask shape -> [B, N, N, 1] for broadcasting
        mask_ = mask.unsqueeze(-1).to(x.dtype)
        left = left * mask_
        right = right * mask_
    # otherwise mask == 1, nothing to do

    # ------------------------------------------------------------------
    # Gating tensors (sigmoid of linear transforms)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # ------------------------------------------------------------------
    # Apply gates
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Ensure contiguous layout and switch to FP32 for the kernel
    # ------------------------------------------------------------------
    left = left.contiguous()
    right = right.contiguous()
    # keep FP32 for maximum numerical stability
    if left.dtype != torch.float32:
        left = left.to(torch.float32)
        right = right.to(torch.float32)

    B, N, _, _ = left.shape   # left.shape == (B, N, N, H)

    # ------------------------------------------------------------------
    # Allocate output of the multiplicative update
    # ------------------------------------------------------------------
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32,
                      device=left.device)

    # ------------------------------------------------------------------
    # Prepare stride arguments for Triton (must be Python integers)
    # ------------------------------------------------------------------
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # ------------------------------------------------------------------
    # Launch Triton kernel
    # ------------------------------------------------------------------
    BLOCK_I = 64   # tile size for i
    BLOCK_J = 64   # tile size for j
    BLOCK_K = 32   # reduction tile for k

    grid = (
        (N + BLOCK_I - 1) // BLOCK_I,
        (N + BLOCK_J - 1) // BLOCK_J,
        B * hidden_dim,                # one program per (batch, hidden channel)
    )

    tri_mul_outgoing_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        out.data_ptr(),
        B, N, hidden_dim,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_i, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
        num_warps=4,                # good balance for H100
    )

    # ------------------------------------------------------------------
    # Second LayerNorm over the hidden dimension, gating and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias,
                       eps=1e-5)

    # Apply the output gate (shape broadcasted)
    out = out * out_gate

    # Final linear projection back to `dim` (no bias)
    out = F.linear(out, to_out_weight)

    return out
```
