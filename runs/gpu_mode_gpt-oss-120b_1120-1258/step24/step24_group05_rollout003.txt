    x, mask, weights, cfg = data
    device = x.device

    B,N,_,dim = x.shape
    hidden_dim = cfg["hidden_dim"]

    # flatten
    R = B*N*N
    x_flat = x.view(R, dim)

    # LayerNorm: FP32
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm_fp32 = F.layer_norm(x_flat, (dim,), weight=norm_weight, bias=norm_bias, eps=eps) # FP32
    x_norm = x_norm_fp32.to(torch.float16)

    # fused projection (5*hidden_dim, dim)
    fused_w = torch.cat([...], dim=0).to(torch.float16) # shape (5H, dim)
    proj = F.linear(x_norm, fused_w) # (R,5H) FP16

    # allocate left,right,out_gate
    left = torch.empty(B, hidden_dim, N, N, dtype=torch.float16, device=device)
    right = torch.empty_like(left)
    out_gate = torch.empty_like(left)

    # mask handling
    has_mask = 0
    if not cfg.get("nomask", True):
        if mask is not None:
            # mask shape [B,N,N] -> flatten R
            mask_flat = mask.to(torch.float16).view(R)  # shape R
            has_mask = 1
        else:
            mask_flat = torch.empty(0, dtype=torch.float16, device=device)
            has_mask = 0
    else:
        mask_flat = torch.empty(0, dtype=torch.float16, device=device)

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_flat,
        R,
        hidden_dim,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj

    # main einsum via bmm
    # left: B,H,N,N; right: B,H,N,N
    left_mat = left.view(B*hidden_dim, N, N)      # [B*H,N,N]
    right_mat = right.view(B*hidden_dim, N, N)
    out = torch.bmm(left_mat, right_mat.transpose(1,2))  # [B*H,N,N] FP16
    out = out.view(B, hidden_dim, N, N)  # [B,H,N,N] FP16

    # second layernorm & out_gate
    # compute mu & rstd via torch
    out_f32 = out.float()
    mu = out_f32.mean(dim=1, keepdim=True)  # shape (B,1,N,N)
    var = out_f32.var(dim=1, unbiased=False, keepdim=True)  # shape (B,1,N,N)
    rstd = torch.rsqrt(var + eps)
    # to_out_norm weight and bias
    ln_w = weights["to_out_norm.weight"]  # (hidden_dim) FP32
    ln_b = weights["to_out_norm.bias"]    # (hidden_dim) FP32

    # Apply layernorm

    # Using broadcasting
    out_norm = (out_f32 - mu) * rstd

    # apply out_gate multiplication
    out_norm = out_norm * out_gate.float()

    # final projection: hidden_dim -> dim
    w_out = weights["to_out.weight"]  # (dim, hidden_dim) FP32

    result = result_flat.view(B, N, N, dim)

Algorithm (high‑level):
1️⃣  Row‑wise LayerNorm (FP32) on the flattened pair tensor → FP16.
2️⃣  Fuse the five linear maps (left_proj, right_proj, left_gate,
    right_gate, out_gate) into a single GEMM → shape (R,5·H) FP16.
3️⃣  Triton kernel splits the fused projection, optionally masks the
    left/right tensors, applies sigmoid gates and writes three tensors
    (left, right, out_gate) in layout [B, H, N, N] (FP16).
4️⃣  Batched GEMM (torch.bmm) computes the core einsum:
        out[b, h, i, j] = Σₖ left[b, h, i, k] * right[b, h, j, k]
    producing a tensor `[B, H, N, N]` (FP16).
5️⃣  Row‑wise mean/variance of `out` over the hidden dimension H
    (torch.mean / torch.var) → μ, σ⁻¹.
6️⃣  Fuse the second LayerNorm, the out‑gate multiplication and the
    final projection (hidden→dim) using pure PyTorch (+one Triton kernel
    for gating).  The result is `[B, N, N, dim]` in FP32.
All heavy compute steps (matmuls & reductions) use Torch’s highly
optimised kernels; only the mask‑gate split is done in Triton.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: split the fused projection, apply optional mask and
# compute sigmoid gates.  Writes left, right and out_gate tensors.
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # [R, 5·H]  FP16 fused projection (input)
    left_ptr,          # [B, H, N, N] FP16 output
    right_ptr,         # [B, H, N, N] FP16 output
    out_gate_ptr,      # [B, H, N, N] FP16 output (gate)
    mask_ptr,          # [R]        FP16 mask (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index r → (b,i,k)
    nn = N * N
    b   = offs_r // nn
    rem = offs_r - b * nn
    i   = rem // N
    k   = rem - i * N

    # optional scalar mask per (b,i,k)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)  # FP16
        mask_val = mask_val[:, None]                                 # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # stride for a single hidden plane in the 4‑D layout
    hidden_stride = N * N
    # base offset for plane h=0 (flattened index in the left/right tensors)
    base_offset = b * (H * hidden_stride) + i * N + k

    # iterate over the hidden dimension in tiles
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # view into the fused projection:
        #   [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]  # [R, H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # sigmoid in FP32 → back to FP16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # apply mask and gates
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate

        # destination offsets in the flattened left/right/out_gate buffers
        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride
        tl.store(left_ptr + dst_offset, left,  mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset, right, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset, outg, mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no backward).

    Args:
        data (tuple):
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
                (all float32)
            - config_dict  : contains "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim         = cfg["dim"]
    hidden_dim  = cfg["hidden_dim"]
    nomask      = cfg.get("nomask", True)
    eps         = 1e-5
    device      = x.device

    B, N, _, _ = x.shape
    R = B * N * N                     # total pairwise positions
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()            # [R, dim] FP32
    # use torch's highly‑optimised layer_norm kernel
    x_norm_fp32 = F.layer_norm(
        x_flat,
        normalized_shape=(dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    )                                                   # FP32
    x_norm = x_norm_fp32.to(torch.float16)              # FP16

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear projection (5·hidden_dim)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)   # shape (5·H, dim)

    proj = F.linear(x_norm, fused_w)   # (R, 5·H) FP16
    del fused_w, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ Split, optional mask, sigmoid gates → left/right/out_gate
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty_like(left)
    out_gate  = torch.empty_like(left)

    if nomask or mask is None:
        mask_flat = torch.empty(0, dtype=torch.float16, device=device)
        has_mask  = 0
    else:
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask  = 1

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)

    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_flat,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_flat

    # ------------------------------------------------------------------
    # 4️⃣ Core einsum via batched GEMM (torch.bmm)
    # ------------------------------------------------------------------
    # left/right: [B, H, N, N] → reshape to [B·H, N, N]
    left_mat  = left.view(B * H, N, N)          # [B·H, N, N]
    right_mat = right.view(B * H, N, N)         # [B·H, N, N]
    # out[b, h, i, j] = Σₖ left[b, h, i, k] * right[b, h, j, k]
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                             # [B, H, N, N] FP16

    # ------------------------------------------------------------------
    # 5️⃣ Row‑wise mean & rsqrt‑variance of `out` (over H)
    # ------------------------------------------------------------------
    out_f32 = out.float()                     # promote to FP32 for stable stats
    mu   = out_f32.mean(dim=1, keepdim=True)  # [B,1,N,N]
    var  = out_f32.var(dim=1, unbiased=False, keepdim=True)  # [B,1,N,N]
    rstd = torch.rsqrt(var + eps)             # [B,1,N,N]

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm, out_gate multiplication and final projection
    # ------------------------------------------------------------------
    # LayerNorm parameters for the hidden dimension H
    ln_w = w["to_out_norm.weight"]  # (H,) FP32
    ln_b = w["to_out_norm.bias"]    # (H,) FP32

    # Apply LayerNorm: (x - μ) * rstd * γ + β
    out_norm = (out_f32 - mu) * rstd
    out_norm = out_norm * ln_w.view(1, H, 1, 1) + ln_b.view(1, H, 1, 1)

    # Apply the out‑gate (already sigmoid‑ed in the kernel)
    out_norm = out_norm * out_gate.float()

    # Final projection hidden → dim (no bias)
    w_out = w["to_out.weight"]  # (dim, hidden_dim) FP32
    # reshape to (R, hidden_dim) where hidden_dim == H
    out_flat = out_norm.permute(0, 2, 3, 1).contiguous().view(R, H)   # (R, H)
    out_proj = F.linear(out_flat, w_out)                             # (R, dim) FP32
    final = out_proj.view(B, N, N, dim)                              # (B,N,N,dim)

    return final
