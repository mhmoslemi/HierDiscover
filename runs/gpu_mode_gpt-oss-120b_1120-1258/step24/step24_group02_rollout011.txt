def custom_kernel(data):
    x, mask, w, cfg = data
    dim = cfg['dim']
    hidden_dim = cfg['hidden_dim']
    nomask = cfg.get('nomask', True)
    eps = 1e-5
    B,N,_,_ = x.shape
    R = B * N * N
    device = x.device

    # 1. LayerNorm on x (shape [B,N,N,dim]) -> flatten.
    x_flat = x.reshape(R, dim).contiguous()
    x_norm_fp16 = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid](
         x_flat,
         x_norm_fp16,
         w["norm.weight"],
         w["norm.bias"],
         R,
         dim,
         eps,
         BLOCK_M=BLOCK_M,
         BLOCK_D=BLOCK_D,
         num_warps=8,
    )
    del x_flat

    # 2. Compute left, right, out_gate, and left/right gates via separate linear.
    # Use fp16 linear: weights cast to FP16.
    # left_proj_weight, right_proj_weight shape (hidden_dim, dim)
    # left_gate_weight etc.
    # cast weights to fp16.
    w_left = w["left_proj.weight"].contiguous().to(torch.float16)
    w_right = w["right_proj.weight"].contiguous().to(torch.float16)
    w_left_gate = w["left_gate.weight"].contiguous().to(torch.float16)
    w_right_gate = w["right_gate.weight"].contiguous().to(torch.float16)
    w_out_gate = w["out_gate.weight"].contiguous().to(torch.float16)

    # compute left
    left = F.linear(x_norm_fp16, w_left)  # shape (R, hidden_dim)
    left = left.view(B, N, N, hidden_dim)  # (B,N,N,H)
    # compute right
    right = F.linear(x_norm_fp16, w_right)
    right = right.view(B, N, N, hidden_dim)
    # compute gates
    left_gate = torch.sigmoid(F.linear(x_norm_fp16, w_left_gate)).view(B, N, N, hidden_dim)
    right_gate = torch.sigmoid(F.linear(x_norm_fp16, w_right_gate)).view(B, N, N, hidden_dim)
    out_gate = torch.sigmoid(F.linear(x_norm_fp16, w_out_gate)).view(B, N, N, hidden_dim)

    del x_norm_fp16

    if not nomask and mask is not None:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)  # shape (B,N,N,1)
        left = left * mask_fp16
        right = right * mask_fp16
    # apply gates
    left = left * left_gate
    right = right * right_gate

    # reshape for batched matmul: left_mat shape (B*H, N, N)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)  # (B*H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)  # (B*H, N, N)
    # compute contraction
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))  # shape (B*H, N, N)
    out = out.view(B, hidden_dim, N, N)  # (B,H,N,N)
    # No need to use out_gate yet.

    # 5. Second LayerNorm + out_gate + final projection
    # flatten out on (B,N,N) rows: shape (R, hidden_dim)

    # apply layernorm (torch.nn.functional.layer_norm)
    out_norm = torch.nn.functional.layer_norm(
        out_flat,
        (hidden_dim,),
        weight=w["to_out_norm.weight"],  # shape (H,)
        bias=w["to_out_norm.bias"],
        eps=eps,
    )  # returns same dtype (preserve fp16)

    final = final_flat_fp16.to(torch.float32).view(B, N, N, dim)

    # return final
    return final

the second LayerNorm, out‑gate multiplication and the final projection)
are carried out with highly‑optimized PyTorch primitives
(matmul, torch.nn.functional.layer_norm, torch.nn.functional.linear,
etc.).  This design keeps the memory footprint modest,
avoids the O(R·5·H) temporary buffer used in the reference
implementation and typically exceeds the target runtime on an H100.

Input:
    data = (x, mask, weights, config)
        x        : torch.Tensor  [B, N, N, C]  (float32)
        mask     : torch.Tensor  [B, N, N]    (bool/float) or None
        weights  : dict of model parameters (float32)
        config   : {"dim": C, "hidden_dim": H, "nomask": bool}

Output:
    torch.Tensor  [B, N, N, C]  (float32)
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16) – Triton implementation
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # [R, dim]   FP32 input
    y_ptr,          # [R, dim]   FP16 output
    w_ptr,          # [dim]      γ (FP32)
    b_ptr,          # [dim]      β (FP32)
    rows, dim, eps,                     # scalar arguments
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # --------- first pass: compute mean & variance --------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------- second pass: write normalized values ------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * dim + offs_d[None, :],
            out.to(tl.float16),
            mask=mask_m[:, None] & mask_d[None, :],
        )


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no backward).

    Args:
        data (tuple):
            - x            : torch.Tensor [B, N, N, C] (float32)
            - mask         : torch.Tensor [B, N, N] (bool/float) or None
            - weights      : dict with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
                (all float32)
            - config       : dict containing "dim", "hidden_dim", "nomask"

    Returns:
        torch.Tensor: [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim          = cfg["dim"]          # C
    hidden_dim   = cfg["hidden_dim"]   # H
    nomask       = cfg.get("nomask", True)
    eps          = 1e-5
    device       = x.device

    B, N, _, _ = x.shape
    R = B * N * N                      # total pairwise positions

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()               # [R, C] float32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat                                            # free FP32 input

    # ------------------------------------------------------------------
    # 2️⃣ Five fused linear maps (left, right, left_gate, right_gate, out_gate)
    # ------------------------------------------------------------------
    # Cast all weight matrices once to FP16
    w_left      = w["left_proj.weight"].contiguous().to(torch.float16)   # [H, C]
    w_right     = w["right_proj.weight"].contiguous().to(torch.float16)
    w_left_gate = w["left_gate.weight"].contiguous().to(torch.float16)
    w_right_gate= w["right_gate.weight"].contiguous().to(torch.float16)
    w_out_gate  = w["out_gate.weight"].contiguous().to(torch.float16)

    # ---- left projection ------------------------------------------------
    left = F.linear(x_norm, w_left)                     # (R, H) FP16
    left = left.view(B, N, N, hidden_dim)              # (B, N, N, H)

    # ---- left gate ------------------------------------------------------
    left_gate = torch.sigmoid(
        F.linear(x_norm, w_left_gate)
    ).view(B, N, N, hidden_dim)                        # (B, N, N, H)

    # ---- right projection ------------------------------------------------
    right = F.linear(x_norm, w_right).view(B, N, N, hidden_dim)

    # ---- right gate ------------------------------------------------------
    right_gate = torch.sigmoid(
        F.linear(x_norm, w_right_gate)
    ).view(B, N, N, hidden_dim)

    # ---- out gate --------------------------------------------------------
    out_gate = torch.sigmoid(
        F.linear(x_norm, w_out_gate)
    ).view(B, N, N, hidden_dim)

    # No longer need the normalized input
    del x_norm

    # ------------------------------------------------------------------
    # 3️⃣ Optional pair‑wise mask (applies only to left/right)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask shape (B, N, N) -> (B, N, N, 1)
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)
        left  = left  * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 4️⃣ Apply sigmoid gates to left/right
    # ------------------------------------------------------------------
    left  = left  * left_gate
    right = right * right_gate

    # Free gate tensors that are no longer required
    del left_gate, right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Core N³ contraction via batched GEMM
    #     out[b, h, i, j] = Σₖ left[b, h, i, k] * right[b, h, j, k]
    # ------------------------------------------------------------------
    # Reshape to (B·H, N, N) for torch.bmm
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)   # (B·H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)   # (B·H, N, N)

    # MatMul over the shared dimension k
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B·H, N, N) FP16
    out = out.view(B, hidden_dim, N, N)                    # (B, H, N, N) FP16

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm + out_gate + final projection (all torch kernels)
    # ------------------------------------------------------------------
    # Bring dimensions to (B, N, N, H) for easier flattening
    out = out.permute(0, 2, 3, 1).contiguous()            # (B, N, N, H)

    # Flatten the (B,N,N) axis → R rows
    out_flat = out.view(R, hidden_dim)                     # (R, H) FP16

    # LayerNorm over the hidden dimension (uses internally FP32 for statistics)
    out_norm = F.layer_norm(
        out_flat,
        (hidden_dim,),
        weight=w["to_out_norm.weight"],   # (H,) FP32
        bias=w["to_out_norm.bias"],        # (H,) FP32
        eps=eps,
    )                                                          # (R, H) FP16

    # Apply out_gate (broadcasted over the hidden dimension)
    out_gate_flat = out_gate.permute(0, 2, 3, 1).contiguous().view(R, hidden_dim)  # (R, H) FP16
    out_norm = out_norm * out_gate_flat

    # Final linear projection to original feature dimension
    w_out = w["to_out.weight"].contiguous().to(torch.float16)   # (C, H)
    final_fp16 = F.linear(out_norm, w_out)                     # (R, C) FP16
    final = final_fp16.to(torch.float32).view(B, N, N, dim)    # (B, N, N, C) float32

    return final
```
