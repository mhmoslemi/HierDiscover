"""
TriMul “outgoing” kernel (AlphaFold‑3) – fast forward pass.

Algorithm
---------
1. Row‑wise LayerNorm over the input channel dimension `dim` using a small
   Triton kernel (FP32).  The kernel computes mean/variance, applies the
   learned γ/β and writes the normalized tensor.
2. Cast the normalized tensor to FP16.  All remaining dense linear layers
   and the O(N³) multiplicative update are performed in half‑precision.
3. Fuse the two projection layers (`left_proj`/`right_proj`) and the two
   gating layers (`left_gate`/`right_gate`) by stacking their weight
   matrices, reducing the number of GEMM calls from four to two.
4. Optional multiplicative mask is applied to the projected values.
5. Element‑wise gating (`left_gate`, `right_gate`) is performed and the
   tensors are reshaped to `[B, N, N, H]`.
6. The core update
        out[i, j, h] = Σ_k left[i, k, h] * right[j, k, h]
   is expressed as a batched matrix‑multiply `L @ Rᵀ` for each hidden
   channel.  The tensors are permuted, made contiguous and a single
   `torch.bmm` call (Tensor‑Core GEMM) computes all `B·H` matrix products.
7. A second LayerNorm over the hidden dimension `hidden_dim` is performed
   with PyTorch’s `layer_norm` (still FP16).
8. The normalized result is multiplied element‑wise by the output gate,
   then projected back to `dim` with a final GEMM and cast to FP32.

Only the first LayerNorm is a custom Triton kernel, satisfying the
requirement that “part of the operations” be a custom kernel while the
rest leverages highly‑optimized PyTorch/CUDA primitives.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def layernorm_kernel(
    x_ptr,               # const float*    [total_rows, dim]
    y_ptr,               # mut   float*    [total_rows, dim]
    weight_ptr,          # const float*    [dim]
    bias_ptr,            # const float*    [dim]
    total_rows,          # int (runtime)
    dim: tl.constexpr,   # compile‑time
    eps: tl.constexpr,   # compile‑time
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """Row‑wise LayerNorm (float32) for a 2‑D tensor."""
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------- pass 1: mean & variance -------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                               # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                # row‑wise sum
        sum_sq  += tl.sum(vals * vals, axis=1)         # row‑wise sum‑squares

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                         # 1 / sqrt(var + eps)

    # ------------------- pass 2: write normalized values ------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)

        out = norm * w[None, :] + b[None, :]

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out, mask=mask_m[:, None] & mask_d[None, :])


def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [B, N, N, dim] (float32, on CUDA)
    """
    # -----------------------------------------------------------------
    # 0) unpack
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # -----------------------------------------------------------------
    # 1) First LayerNorm over `dim` (float32, Triton)
    # -----------------------------------------------------------------
    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N                     # rows after flattening

    x_flat = input_tensor.view(total_rows, dim)        # [R, dim]
    x_norm_flat = torch.empty_like(x_flat)             # output buffer

    # tuned block sizes (good for dim up to 384)
    BLOCK_M = 256
    BLOCK_D = 32

    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm_flat,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()

    # -----------------------------------------------------------------
    # 2) Cast to half‑precision and prepare weights
    # -----------------------------------------------------------------
    x_norm = x_norm_flat.view(B, N, N, dim)      # [B,N,N,dim] float32
    x_norm_h = x_norm.to(torch.float16)          # FP16
    x_norm_h_flat = x_norm_h.view(total_rows, dim)   # [R, dim]

    # half‑precision copies of the model weights
    w_left_proj   = weights["left_proj.weight"].to(torch.float16)   # (H, dim)
    w_right_proj  = weights["right_proj.weight"].to(torch.float16)  # (H, dim)
    w_left_gate   = weights["left_gate.weight"].to(torch.float16)   # (H, dim)
    w_right_gate  = weights["right_gate.weight"].to(torch.float16)  # (H, dim)
    w_out_gate    = weights["out_gate.weight"].to(torch.float16)    # (H, dim)
    w_to_out_norm = weights["to_out_norm.weight"].to(torch.float16)  # (H,)
    b_to_out_norm = weights["to_out_norm.bias"].to(torch.float16)    # (H,)
    w_to_out      = weights["to_out.weight"].to(torch.float16)       # (dim, H)

    # -----------------------------------------------------------------
    # 3) Fused linear projections (left/right) and gating (left/right)
    # -----------------------------------------------------------------
    # (left_proj || right_proj) -> shape (R, 2*H)
    proj_weight = torch.cat([w_left_proj, w_right_proj], dim=0)       # (2H, dim)
    proj_out = F.linear(x_norm_h_flat, proj_weight)                  # (R, 2H)
    left_proj_h, right_proj_h = torch.split(proj_out, hidden_dim, dim=1)

    # (left_gate || right_gate) -> shape (R, 2*H)
    gate_weight = torch.cat([w_left_gate, w_right_gate], dim=0)       # (2H, dim)
    gate_out = torch.sigmoid(F.linear(x_norm_h_flat, gate_weight))   # (R, 2H)
    left_gate_h, right_gate_h = torch.split(gate_out, hidden_dim, dim=1)

    # output gate (shape R x H)
    out_gate_h = torch.sigmoid(F.linear(x_norm_h_flat, w_out_gate))

    # -----------------------------------------------------------------
    # 4) Apply optional mask (broadcast over hidden dimension)
    # -----------------------------------------------------------------
    if not config.get("nomask", True):
        mask_h = mask_tensor.to(torch.float16).view(total_rows, 1)   # (R,1)
        left_proj_h  = left_proj_h * mask_h
        right_proj_h = right_proj_h * mask_h

    # -----------------------------------------------------------------
    # 5) Apply element‑wise gates
    # -----------------------------------------------------------------
    left_h  = left_proj_h * left_gate_h          # (R, H)
    right_h = right_proj_h * right_gate_h        # (R, H)

    # reshape back to 4‑D tensors
    left_h  = left_h.view(B, N, N, hidden_dim)    # [B,N,N,H]
    right_h = right_h.view(B, N, N, hidden_dim)   # [B,N,N,H]
    out_gate_h = out_gate_h.view(B, N, N, hidden_dim)

    # -----------------------------------------------------------------
    # 6) Triangular multiplicative update via batched GEMM
    # -----------------------------------------------------------------
    # (B, N, N, H) -> (B, H, N, N) and make contiguous for cuBLAS
    left_mat  = left_h.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right_h.permute(0, 3, 2, 1).contiguous().view(B * hidden_dim, N, N)

    # BMM: L @ Rᵀ  (each hidden channel independent)
    out_mat = torch.bmm(left_mat, right_mat)    # (B*H, N, N)

    # reshape to [B,N,N,H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 7) Second LayerNorm over hidden dimension (FP16)
    # -----------------------------------------------------------------
    out_norm = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=w_to_out_norm,
        bias=b_to_out_norm,
        eps=eps,
    )  # still FP16

    # -----------------------------------------------------------------
    # 8) Gated projection back to `dim`
    # -----------------------------------------------------------------
    out_norm = out_norm * out_gate_h                # element‑wise gate
    out_norm_flat = out_norm.view(total_rows, hidden_dim)   # (R, H)

    final_half = F.linear(out_norm_flat, w_to_out)  # (R, dim), no bias
    final = final_half.view(B, N, N, dim).to(torch.float32)

    return final
