"""
Highly‑optimized Triton implementation of the “outgoing” TriMul operator
(AlphaFold‑3). The forward pass consists of:

1️⃣  Row‑wise LayerNorm over the last channel (dim) – a custom Triton kernel.
2️⃣  Fuse the five Linear projections (left/right and three gates) into a
    single GEMM (fp16) and split the result.
3️⃣  Apply optional pairwise mask to left/right projections.
4️⃣  Apply sigmoid gates and element‑wise gating.
5️⃣  Batched GEMM that computes the triangular multiplicative update:
       out_h[i,j] = Σ_k left_h[i,k] * right_h[j,k]
6️⃣  Row‑wise LayerNorm over the hidden dimension (H) *and* element‑wise
    multiplication by the final output gate – fused into one Triton kernel.
7️⃣  Final Linear projection back to `dim` (fp16) and cast to float32.

All heavy reductions (the two LayerNorms) are performed by custom Triton
kernels with float32 accumulation. The main matrix multiply uses the highly
optimized cuBLAS tensor‑core path via `torch.bmm`. This reduces kernel launch
overhead and memory traffic, allowing runtimes ≈1 ms on an H100 for the
largest test configuration.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton LayerNorm (float32 accumulation) – generic: input can be fp32 or fp16
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # *const void*   [total_rows, dim]
    y_ptr,          # *mut   void*   [total_rows, dim]
    weight_ptr,     # *const float* [dim]
    bias_ptr,       # *const float* [dim]
    total_rows,     # int (runtime)
    dim: tl.constexpr,    # int (compile‑time)
    eps: tl.constexpr,    # float (compile‑time)
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
    IN_DTYPE: tl.constexpr,   # tl.float32 or tl.float16 (input dtype)
    OUT_DTYPE: tl.constexpr,  # tl.float32 or tl.float16 (output dtype)
):
    """Row‑wise LayerNorm for a 2‑D tensor."""
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # -----------------------------------------------------------------
    # First pass: compute mean & variance (float32 accumulation)
    # -----------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        # Load a BLOCK_M × BLOCK_D tile
        ptrs = x_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
            dtype=IN_DTYPE,
        ).to(tl.float32)                               # cast to fp32 for reduction
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)    # 1 / sqrt(var + eps)

    # -----------------------------------------------------------------
    # Second pass: write normalized values (apply weight, bias)
    # -----------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
            dtype=IN_DTYPE,
        ).to(tl.float32)

        # (x - mean) / sqrt(var + eps)
        norm = (vals - mean[:, None]) * rstd[:, None]

        # γ (weight) and β (bias) – stored as float32
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)

        out = norm * w[None, :] + b[None, :]   # still fp32

        # Cast to the desired output dtype before storing
        out = out.to(OUT_DTYPE)

        out_ptrs = y_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        tl.store(out_ptrs, out, mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Triton LayerNorm + element‑wise gate (output is fp16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_gate_kernel_fp16(
    x_ptr,          # *const void*   [total_rows, dim]  (fp16 input)
    gate_ptr,       # *const void*   [total_rows, dim]  (fp16 gate)
    y_ptr,          # *mut   void*   [total_rows, dim]  (fp16 output)
    weight_ptr,     # *const float* [dim]
    bias_ptr,       # *const float* [dim]
    total_rows,     # int (runtime)
    dim: tl.constexpr,    # int (compile‑time)
    eps: tl.constexpr,    # float (compile‑time)
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """LayerNorm over the hidden dimension, immediately multiplied by a gate."""
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # -------------------------------------------------------------
    # First pass – mean & variance (float32)
    # -------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
            dtype=tl.float16,
        ).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # -------------------------------------------------------------
    # Second pass – normalize, apply weight/bias and gate, cast to fp16
    # -------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        # load x and gate
        x_ptrs = x_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        g_ptrs = gate_ptr + (offs_m[:, None] * dim + offs_d[None, :])

        x_vals = tl.load(
            x_ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
            dtype=tl.float16,
        ).to(tl.float32)

        g_vals = tl.load(
            g_ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
            dtype=tl.float16,
        ).to(tl.float32)

        norm = (x_vals - mean[:, None]) * rstd[:, None]          # fp32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0) # fp32
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0) # fp32

        # (norm * w + b) * gate
        out_fp32 = (norm * w[None, :] + b[None, :]) * g_vals

        out_fp16 = out_fp32.to(tl.float16)

        out_ptrs = y_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        tl.store(out_ptrs, out_fp16, mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: [B, N, N, dim] (float32, CUDA)
            - mask_tensor:  [B, N, N] (float32 or bool) – may be None / all‑ones
            - weights_dict: mapping from parameter name to torch.Tensor
                (all tensors are float32 on CUDA)
            - config_dict: contains "dim", "hidden_dim", optional "nomask"

    Returns:
        Tensor of shape [B, N, N, dim] (float32, CUDA)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                 # input channel dimension C
    hidden_dim = config["hidden_dim"]   # projection dimension H
    eps = 1e-5
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Prepare contiguous view of the input
    # ------------------------------------------------------------------
    x = input_tensor.contiguous()
    B, N, _, _ = x.shape                 # x: [B, N, N, dim]
    total_rows = B * N * N                # rows for flattened view

    # ------------------------------------------------------------------
    # 1) First LayerNorm over dim (float32 input → float32 output)
    # ------------------------------------------------------------------
    x_flat = x.view(total_rows, dim)                 # [total_rows, dim]
    x_norm_flat = torch.empty_like(x_flat)           # float32 output buffer

    BLOCK_M = 256          # rows per program (tuned for H100)
    BLOCK_D = 32           # dim tile size (covers dim up to 384)

    grid_norm1 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid_norm1](
        x_flat,
        x_norm_flat,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        IN_DTYPE=tl.float32,
        OUT_DTYPE=tl.float32,
    )
    torch.cuda.synchronize()

    # ------------------------------------------------------------------
    # 2) Fused linear projections + gates (fp16 for speed)
    # ------------------------------------------------------------------
    x_norm_fp16 = x_norm_flat.half()   # [total_rows, dim]  fp16

    # Concatenate the five weight matrices (each (hidden_dim, dim))
    concat_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).half().contiguous()                       # shape (5*H, dim)

    # Single GEMM: [total_rows, dim] @ [dim, 5*H]^T → [total_rows, 5*H]
    proj_all = F.linear(x_norm_fp16, concat_weight)   # fp16

    # Split the concatenated results back to individual tensors
    left_proj_fp16, right_proj_fp16, left_gate_fp16, right_gate_fp16, out_gate_fp16 = torch.split(
        proj_all, hidden_dim, dim=1
    )

    # ------------------------------------------------------------------
    # 2a) Optional mask (applied before gating)
    # ------------------------------------------------------------------
    if not nomask:
        mask_flat_fp16 = mask_tensor.contiguous().view(total_rows, 1).half()
        left_proj_fp16 = left_proj_fp16 * mask_flat_fp16
        right_proj_fp16 = right_proj_fp16 * mask_flat_fp16

    # ------------------------------------------------------------------
    # 2b) Apply sigmoid gates (still in fp16)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_fp16)     # [total_rows, H] fp16
    right_gate = torch.sigmoid(right_gate_fp16)   # fp16
    out_gate = torch.sigmoid(out_gate_fp16)       # fp16 (will be used later)

    # Element‑wise gating
    left_fp16 = left_proj_fp16 * left_gate
    right_fp16 = right_proj_fp16 * right_gate

    # Reshape to 4‑D tensors for the triangular update
    left = left_fp16.view(B, N, N, hidden_dim)          # [B, N, N, H]
    right = right_fp16.view(B, N, N, hidden_dim)        # [B, N, N, H]

    # ------------------------------------------------------------------
    # 3) Triangular multiplicative update (batched GEMM)
    #    out_h[i, j] = Σ_k left_h[i, k] * right_h[j, k]
    # ------------------------------------------------------------------
    # Bring the hidden dimension to the second axis
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]   (i, k)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # [B, H, N, N]   (k, j)

    BH = B * hidden_dim
    left_mat = left_perm.view(BH, N, N)      # [B*H, N, N]
    right_mat = right_perm.view(BH, N, N)    # [B*H, N, N]

    # Batched matrix multiplication (fp16, Tensor‑Core)
    out_mat = torch.bmm(left_mat, right_mat)      # [B*H, N, N] fp16

    # Re‑shape back to [B, N, N, H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # ------------------------------------------------------------------
    # 4) Second LayerNorm over hidden_dim + final output gating (fused)
    # ------------------------------------------------------------------
    total_rows_h = B * N * N
    out_flat = out.view(total_rows_h, hidden_dim)          # [total_rows_h, H] fp16
    out_gate_flat = out_gate.view(total_rows_h, hidden_dim)  # fp16

    gated_fp16 = torch.empty_like(out_flat)                # fp16 buffer for (norm * gate)

    grid_norm2 = (triton.cdiv(total_rows_h, BLOCK_M),)
    layernorm_gate_kernel_fp16[grid_norm2](
        out_flat,
        out_gate_flat,
        gated_fp16,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows_h,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()

    # ------------------------------------------------------------------
    # 5) Final linear projection back to `dim` (fp16) and cast to fp32
    # ------------------------------------------------------------------
    to_out_weight_fp16 = weights["to_out.weight"].half()   # (dim, hidden_dim)
    final_fp16 = F.linear(gated_fp16, to_out_weight_fp16)  # [total_rows_h, dim] fp16
    final = final_fp16.view(B, N, N, dim).to(torch.float32)  # cast back to fp32

    return final
