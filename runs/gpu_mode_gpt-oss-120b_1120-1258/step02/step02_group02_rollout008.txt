"""
custom_kernel – outgoing TriMul (AlphaFold‑3) optimized for H100.

Algorithm
---------
1️⃣  FP16 LayerNorm over the last axis (dim) using a tiny Triton kernel.
    *Input is cast to float16 to halve memory traffic.
    *Mean/variance are accumulated in float32 for numeric stability,
     the normalized value is written back as float16.

2️⃣  Fuse the two projection matrices (proj + gate) per side:
    *left  = [left_proj , left_gate]   → a single torch.nn.functional.linear
    *right = [right_proj, right_gate] → a single linear
    This reduces the number of GEMMs from six to four.

3️⃣  (Optional) mask is applied once (broadcast over hidden dim).

4️⃣  Element‑wise sigmoid gates are applied.

5️⃣  Triangular multiplicative update:
        out_h[i,j] = Σ_k left_h[i,k] · right_h[j,k]   (∈ ℝᴴ)
    Implemented as a single strided‑batched half‑precision GEMM
    (torch.bmm) over the flattened batch dimension B·H.

6️⃣  Second FP16 LayerNorm over the hidden dimension (H) with the same
    Triton kernel.

7️⃣  Apply the output‑gate (sigmoid) and a final half‑precision linear
    projection back to `dim`.  The result is cast to FP32 for the
    caller.

All heavy layers (both LayerNorms) run in custom Triton kernels, while
the dense linear algebra (projections, batched GEMM, final projection)
uses cuBLAS FP16 kernels.  This halves the memory bandwidth compared to
the reference FP32 implementation and typically brings the forward pass
well below the 1 ms target on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# FP16 row‑wise LayerNorm (triton)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,          # const fp16*  [total_rows, dim]
    y_ptr,          #       fp16*  [total_rows, dim]
    w_ptr,          # const fp16*  [dim]   (γ)
    b_ptr,          # const fp16*  [dim]   (β)
    total_rows,     # int (runtime)
    dim: tl.constexpr,          # static dim size
    eps: tl.constexpr,          # static epsilon
    BLOCK_M: tl.constexpr,      # rows per program (multiple of 32)
    BLOCK_D: tl.constexpr,      # columns per inner loop
):
    """
    Row‑wise LayerNorm for a 2‑D tensor stored in row‑major order.
    Input/Output are FP16, accumulation uses FP32.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------- reduction (mean / variance) -------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals_f16 = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        )
        vals = vals_f16.to(tl.float32)                 # promote
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                         # 1 / √(var+eps)

    # ------------------- write normalized values -----------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals_f16 = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        )
        vals = vals_f16.to(tl.float32)

        # (x - mean) / sqrt(var + eps)
        norm = (vals - mean[:, None]) * rstd[:, None]

        # γ and β (broadcasted)
        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)

        out_fp16 = (norm * w[None, :] + b[None, :]).to(tl.float16)

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim]  (float32)
            - mask_tensor  : torch.Tensor [B, N, N]   (float32/bool)
            - weights_dict : dict of torch.Tensor containing all module
                             weights (float32)
            - config_dict  : dict with keys:
                * dim         : int   (model dimension)
                * hidden_dim  : int   (inner dimension)
                * nomask      : bool  (True → no mask)
                * eps         : float (optional, default 1e-5)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32) – the TriMul output.
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Prepare FP16 tensors
    # ------------------------------------------------------------------
    x = input_tensor.contiguous()
    B, N, _, _ = x.shape                    # [B, N, N, dim]
    total_rows = B * N * N

    # Cast input to half for the first LayerNorm
    x_flat_h = x.view(total_rows, dim).to(torch.float16)

    # Output buffer for the normalized tensor (half)
    x_norm_h = torch.empty_like(x_flat_h)

    # Convert LayerNorm parameters to half
    norm_weight_h = weights["norm.weight"].to(torch.float16)
    norm_bias_h   = weights["norm.bias"].to(torch.float16)

    # ------------------------------------------------------------------
    # 1️⃣  FP16 LayerNorm over `dim`
    # ------------------------------------------------------------------
    BLOCK_M = 256          # rows per program (tune‑able)
    BLOCK_D = 64           # columns per inner loop (tune‑able)

    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        x_flat_h,
        x_norm_h,
        norm_weight_h,
        norm_bias_h,
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    # No explicit sync – later kernels will automatically synchronize.

    # Restore 4‑D shape (still FP16)
    x_norm = x_norm_h.view(B, N, N, dim)      # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2️⃣  Fuse projection + gate (two linear layers each side)
    # ------------------------------------------------------------------
    # Concatenate projection & gate weights for the left side
    left_concat_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)                       # [2*H, dim]

    # Concatenate for the right side
    right_concat_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)                       # [2*H, dim]

    # Linear + split
    left_concat  = F.linear(x_norm.view(total_rows, dim), left_concat_w)
    left_proj_h  = left_concat[:, :hidden_dim]          # proj part
    left_gate_h  = left_concat[:, hidden_dim:]          # pre‑sigmoid gate

    right_concat = F.linear(x_norm.view(total_rows, dim), right_concat_w)
    right_proj_h = right_concat[:, :hidden_dim]
    right_gate_h = right_concat[:, hidden_dim:]

    # Output‑gate (no bias)
    out_gate_h = torch.sigmoid(
        F.linear(x_norm.view(total_rows, dim),
                 weights["out_gate.weight"].to(torch.float16))
    )                                            # [total_rows, hidden_dim]

    # ------------------------------------------------------------------
    # 3️⃣  Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask_h = mask_tensor.view(total_rows, 1).to(torch.float16)
        left_proj_h  = left_proj_h * mask_h
        right_proj_h = right_proj_h * mask_h

    # ------------------------------------------------------------------
    # 4️⃣  Apply sigmoid gates
    # ------------------------------------------------------------------
    left_gate_h  = torch.sigmoid(left_gate_h)
    right_gate_h = torch.sigmoid(right_gate_h)

    left_h  = left_proj_h * left_gate_h    # [total_rows, hidden_dim]
    right_h = right_proj_h * right_gate_h

    # ------------------------------------------------------------------
    # 5️⃣  Triangular multiplicative update (batched GEMM)
    # ------------------------------------------------------------------
    # reshape to [B, N, N, H]
    left  = left_h.view(B, N, N, hidden_dim)
    right = right_h.view(B, N, N, hidden_dim)

    # bring hidden dim to the batch axis: (B, H, N, N)
    left_perm  = left.permute(0, 3, 1, 2)          # [B, H, N, N]
    right_perm = right.permute(0, 3, 2, 1)         # [B, H, N, N] (k ↔ j)

    BH = B * hidden_dim
    left_mat  = left_perm.reshape(BH, N, N)       # [BH, N, N] (FP16)
    right_mat = right_perm.reshape(BH, N, N)      # [BH, N, N]

    # Batched matrix multiplication in FP16 (cuBLAS)
    out_mat = torch.bmm(left_mat, right_mat)      # [BH, N, N]

    # back to [B, N, N, H] (still FP16)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6️⃣  Second FP16 LayerNorm over hidden_dim
    # ------------------------------------------------------------------
    total_rows_h = B * N * N
    out_flat_h = out.view(total_rows_h, hidden_dim)
    out_norm_flat_h = torch.empty_like(out_flat_h)

    to_out_norm_w_h = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b_h = weights["to_out_norm.bias"].to(torch.float16)

    grid2 = (triton.cdiv(total_rows_h, BLOCK_M),)
    layernorm_fp16_kernel[grid2](
        out_flat_h,
        out_norm_flat_h,
        to_out_norm_w_h,
        to_out_norm_b_h,
        total_rows_h,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    out_norm_h = out_norm_flat_h.view(B, N, N, hidden_dim)   # FP16

    # ------------------------------------------------------------------
    # 7️⃣  Apply output gate and final projection
    # ------------------------------------------------------------------
    out_norm_h = out_norm_h * out_gate_h.view(B, N, N, hidden_dim)

    # Final linear projection back to `dim`
    final_h = F.linear(
        out_norm_h.view(total_rows_h, hidden_dim),
        weights["to_out.weight"].to(torch.float16)
    )                                 # [total_rows_h, dim] (FP16)

    # Cast back to FP32 for the user
    final = final_h.view(B, N, N, dim).to(torch.float32)

    return final
