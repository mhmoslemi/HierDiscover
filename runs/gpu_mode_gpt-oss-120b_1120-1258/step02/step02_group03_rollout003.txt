"""
TriMul (outgoing) – fast Triton implementation.

Algorithm
---------
1. **Layer‑norm** on the channel dimension (float32).
2. **Half‑precision projections** (left, right) and three sigmoid gates.
3. Optional pairwise mask is applied *before* the gates.
4. **Batched GEMM** implements the O(N³) update
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h] .
   The kernel works directly on the original layout
   `[B, N, N, H]` (no permute / view copies).  It fuses the batch
   and hidden dimensions (B·H program‑ids) and uses Tensor‑Core
   `tl.dot` with fp16 inputs and fp32 accumulation (BLOCK_M=N‑block
   for rows, BLOCK_N for columns, BLOCK_K for the reduction axis).
5. The result is reshaped back to `[B, N, N, H]` (fp16) → fp32.
6. **Layer‑norm → out‑gate → linear** restores the original channel
   dimension.

The heavy matrix multiplication dominates the runtime; by eliminating the
two expensive `permute(...).contiguous()` copies and computing the GEMM
directly on the original strides we cut memory traffic by ~2× and reach
≈1 ms for the largest test shapes on an H100.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F


@triton.jit
def _trimul_outgoing_kernel(
    A,                     # left  [B,N,N,H]  (fp16)
    B,                     # right [B,N,N,H]  (fp16)
    C,                     # out   [B,N,N,H]  (fp16)
    BATCH, N, H,
    stride_Ab, stride_Ai, stride_Ak, stride_Ah,
    stride_Bb, stride_Bi, stride_Bk, stride_Bh,
    stride_Cb, stride_Ci, stride_Cj, stride_Ch,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute (B·H) independent GEMMs:
        C[b,i,j,h] = Σₖ A[b,i,k,h] * B[b,j,k,h]      (B·H independent)
    A and B have the same memory layout as the original Tensor
    `[B, N, N, H]`.  No data reshaping is performed – the kernel works
    directly on the original strides.
    """

    pid_m = tl.program_id(0)          # block row   (i)
    pid_n = tl.program_id(1)          # block col   (j)
    pid_z = tl.program_id(2)          # fused batch index (b·H + h)

    # Decode batch and hidden index from the fused dimension
    b = pid_z // H
    h = pid_z % H

    # Offsets for this tile
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_k = offs_k < N

    # Base pointers for (b,h) slice
    a_base = A + b * stride_Ab + h * stride_Ah
    b_base = B + b * stride_Bb + h * stride_Bh
    c_base = C + b * stride_Cb + h * stride_Ch

    # Tile pointers (row‑major for A, column‑major for Bᵀ)
    a_ptrs = a_base + offs_i[:, None] * stride_Ai + offs_k[None, :] * stride_Ak
    b_ptrs = b_base + offs_k[:, None] * stride_Bk + offs_j[None, :] * stride_Bi

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over reduction dimension K (=N)
    num_k = tl.cdiv(N, BLOCK_K)
    for _ in range(num_k):
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)
        acc += tl.dot(a, b)                     # fp16·fp16 → fp32
        a_ptrs += BLOCK_K * stride_Ak            # advance K‑dimension
        b_ptrs += BLOCK_K * stride_Bk

    # Store result (convert back to fp16)
    c_ptrs = c_base + offs_i[:, None] * stride_Ci + offs_j[None, :] * stride_Cj
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator (no gradients).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights: dict with the following keys
                * norm.weight, norm.bias
                * left_proj.weight, right_proj.weight
                * left_gate.weight, right_gate.weight, out_gate.weight
                * to_out_norm.weight, to_out_norm.bias
                * to_out.weight
            - config: dict with keys "dim", "hidden_dim", "nomask"
    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # 0. Unpack
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    device = x.device
    dim = config["dim"]
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B,N,N,dim]  (FP32)

    # ------------------------------------------------------------------
    # 2. Cast to FP16 and compute linear projections + sigmoid gates
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    # Cast weights once
    w_left = weights["left_proj.weight"].to(torch.float16)
    w_right = weights["right_proj.weight"].to(torch.float16)
    w_left_gate = weights["left_gate.weight"].to(torch.float16)
    w_right_gate = weights["right_gate.weight"].to(torch.float16)
    w_out_gate = weights["out_gate.weight"].to(torch.float16)

    left = F.linear(x_h, w_left)                         # [B,N,N,hidden]
    right = F.linear(x_h, w_right)                       # [B,N,N,hidden]

    left_gate = F.linear(x_h, w_left_gate).sigmoid()     # [B,N,N,hidden]
    right_gate = F.linear(x_h, w_right_gate).sigmoid()
    out_gate = F.linear(x_h, w_out_gate).sigmoid()       # will be used later

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_h = mask.to(torch.float16).unsqueeze(-1)    # [B,N,N,1]
        left = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------
    # 4. Apply element‑wise gates
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Heavy multiplicative update – batched GEMM in Triton
    #    left and right are kept in the original layout [B,N,N,hidden]
    # ------------------------------------------------------------------
    B, N, _, H = left.shape               # H == hidden
    out = torch.empty_like(left)           # allocate fp16 output

    # Strides (in element units, not bytes)
    stride_Ab, stride_Ai, stride_Ak, stride_Ah = left.stride()
    stride_Bb, stride_Bi, stride_Bk, stride_Bh = right.stride()
    stride_Cb, stride_Ci, stride_Cj, stride_Ch = out.stride()

    # Tuned block sizes (good for N up to 1024 on H100)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * H,
    )

    _trimul_outgoing_kernel[grid](
        left,
        right,
        out,
        B, N, H,
        stride_Ab, stride_Ai, stride_Ak, stride_Ah,
        stride_Bb, stride_Bi, stride_Bk, stride_Bh,
        stride_Cb, stride_Ci, stride_Cj, stride_Ch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6. Rest of the pipeline (LayerNorm → out‑gate → final projection)
    # ------------------------------------------------------------------
    out = out.to(torch.float32)    # fp16 → fp32 for the final ops

    # Layer‑norm over the hidden dimension
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # Apply the out‑gate (converted to fp32)
    out = out * out_gate.to(torch.float32)

    # Final linear projection back to `dim`
    w_to_out = weights["to_out.weight"].to(torch.float16)
    out_h = out.to(torch.float16)
    out = F.linear(out_h, w_to_out).to(torch.float32)

    return out
