@triton.jit
def proj_layernorm_kernel(
    x_ptr,            # const float* [total_rows, dim]
    pre_weight_ptr,   # const half* [dim, out_dim] (transposed)
    weight_sum_ptr,   # const float* [out_dim]
    bias_ptr,         # const float* [out_dim]
    out_ptr,          # *mut half* [total_rows, out_dim]
    total_rows: tl.constexpr,
    dim: tl.constexpr,
    out_dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
    BLOCK_O: tl.constexpr,
):
    pid_m = tl.program_id(0)
    pid_o = tl.program_id(1)

    row_start = pid_m * BLOCK_M
    col_start = pid_o * BLOCK_O

    offs_m = row_start + tl.arange(0, BLOCK_M)
    offs_o = col_start + tl.arange(0, BLOCK_O)

    mask_m = offs_m < total_rows
    mask_o = offs_o < out_dim

    # accumulators
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)
    dot_acc = tl.zeros([BLOCK_M, BLOCK_O], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        d_start = d * BLOCK_D
        offs_d = d_start + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        # load x values [BLOCK_M, BLOCK_D]
        x_ptrs = x_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        x = tl.load(x_ptrs, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)

        # accumulate sums for layernorm
        sum_val += tl.sum(x, axis=1)
        sum_sq += tl.sum(x * x, axis=1)

        # load weight block [BLOCK_D, BLOCK_O] (transposed)
        w_ptrs = pre_weight_ptr + (offs_d[:, None] * out_dim + offs_o[None, :])
        w = tl.load(w_ptrs, mask=mask_d[:, None] & mask_o[None, :], other=0.0).to(tl.float32)

        # dot product x @ w  => [BLOCK_M, BLOCK_O]
        dot_acc += tl.dot(x, w)   # (M,K)*(K,N) = (M,N)
    # compute mean, var, rstd
    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # load weight_sum and bias for the current output tile
    wsum = tl.load(weight_sum_ptr + offs_o, mask=mask_o, other=0.0)   # [BLOCK_O]
    b = tl.load(bias_ptr + offs_o, mask=mask_o, other=0.0)          # [BLOCK_O]

    # broadcast mean and rstd
    mean_tile = mean[:, None]   # [BLOCK_M,1]
    rstd_tile = rstd[:, None]   # [BLOCK_M,1]

    # compute final projection
    # proj = ((dot - mean * wsum) * rstd) + b
    proj = ((dot_acc - mean_tile * wsum[None, :]) * rstd_tile) + b[None, :]

    # cast to half and store
    proj_fp16 = proj.to(tl.float16)

    out_ptrs = out_ptr + (offs_m[:, None] * out_dim + offs_o[None, :])
    tl.store(out_ptrs, proj_fp16, mask=mask_m[:, None] & mask_o[None, :])
```

Note: here `pre_weight_ptr` points to pre_weight_T shape (dim, out_dim). We loaded w block with shape (BLOCK_D, BLOCK_O). So `tl.dot(x, w)` works: x shape (BLOCK_M, BLOCK_D) and w shape (BLOCK_D, BLOCK_O). Good.

Now `tl.dot` returns shape (BLOCK_M, BLOCK_O) as float32.

@triton.jit
def layernorm_gate_kernel(
    x_ptr,          # const half* [total_rows, dim]
    gate_ptr,       # const half* [total_rows, dim]
    y_ptr,          # *mut half* [total_rows, dim]
    weight_ptr,     # const half* [dim]   (to_out_norm.weight)
    bias_ptr,       # const half* [dim]   (to_out_norm.bias)
    total_rows: tl.constexpr,
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # pass 1: compute mean and var
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        d_start = d * BLOCK_D
        offs_d = d_start + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        x = tl.load(x_ptr + (offs_m[:, None] * dim + offs_d[None, :]),
                    mask=mask_m[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)

        sum_val += tl.sum(x, axis=1)
        sum_sq += tl.sum(x * x, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)   # [BLOCK_M]

    # load weight and bias for LN
    w = tl.load(weight_ptr + tl.arange(0, BLOCK_D), mask=mask_d, other=0.0).to(tl.float32)   # shape [BLOCK_D] (will be reloaded in second pass)
    b = tl.load(bias_ptr + tl.arange(0, BLOCK_D), mask=mask_d, other=0.0).to(tl.float32)

    # second pass: apply LN + gate
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        d_start = d * BLOCK_D
        offs_d = d_start + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        x = tl.load(x_ptr + (offs_m[:, None] * dim + offs_d[None, :]),
                    mask=mask_m[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)
        g = tl.load(gate_ptr + (offs_m[:, None] * dim + offs_d[None, :]),
                    mask=mask_m[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)

        w_tile = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0).to(tl.float32)   # load per block
        b_tile = tl.load(bias_ptr + offs_d, mask=mask_d, other=0.0).to(tl.float32)

        normed = (x - mean[:, None]) * rstd[:, None]
        # apply weight and bias
        ys = normed * w_tile[None, :] + b_tile[None, :]
        # apply gate
        ys = ys * g

        # store result
        y_ptrs = y_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        tl.store(y_ptrs, ys.to(tl.float16), mask=mask_m[:, None] & mask_d[None, :])

TriMul (outgoing) – fused, low‑memory Triton implementation.
The kernel fuses the first LayerNorm, the five linear projections and the
pre‑gate bias addition into a single pass over the input tensor.
A second Triton kernel fuses the second LayerNorm (to_out_norm) with the
element‑wise output‑gate, eliminating an intermediate buffer.
All heavy work is performed in fp16 (Tensor‑Core) while mean/variance
are accumulated in fp32 for numerical stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1) First LayerNorm + 5 projections (concatenated) – one Triton kernel
# ----------------------------------------------------------------------
@triton.jit
def proj_norm_linear_kernel(
    x_ptr,                # *const float*   [total_rows, dim]
    w_ptr,                # *const half*    [dim, out_dim] (weight.T)
    wsum_ptr,             # *const float*   [out_dim]    (row‑sum of scaled weight)
    bias_ptr,             # *const float*   [out_dim]
    out_ptr,              # *mut half*      [total_rows, out_dim]
    total_rows: tl.constexpr,
    dim:        tl.constexpr,
    out_dim:    tl.constexpr,
    eps:        tl.constexpr,
    BLOCK_M:    tl.constexpr,   # rows per program (threads per block)
    BLOCK_D:    tl.constexpr,   # tile over input channel dimension
    BLOCK_O:    tl.constexpr,   # tile over output (5*hidden_dim)
):
    """Fuse LayerNorm (γ,β) and the concatenated linear projection."""
    pid = tl.program_id(0)      # row tile
    pid_o = tl.program_id(1)    # output‑tile

    # -----------------------------------------------------------------
    #  thread‑wise offsets
    # -----------------------------------------------------------------
    row = pid * BLOCK_M + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    col = pid_o * BLOCK_O + tl.arange(0, BLOCK_O)       # [BLOCK_O]

    mask_row = row < total_rows
    mask_col = col < out_dim

    # -----------------------------------------------------------------
    #  Accumulators (float32)
    # -----------------------------------------------------------------
    sum_val = tl.float32(0.0)        # Σ x       – scalar per thread
    sum_sq  = tl.float32(0.0)        # Σ x²      – scalar per thread
    dot     = tl.zeros([BLOCK_O], dtype=tl.float32)   # (x·w) per output tile

    # -----------------------------------------------------------------
    #  Loop over the input channel dimension
    # -----------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        d_start = d * BLOCK_D
        d_off  = d_start + tl.arange(0, BLOCK_D)
        mask_d = d_off < dim

        # load a tile of the input row  – shape (BLOCK_D,)
        x = tl.load(
            x_ptr + (row[:, None] * dim + d_off[None, :]),
            mask=mask_row[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                # -> (BLOCK_D,)

        # update moments
        sum_val += tl.sum(x)
        sum_sq  += tl.sum(x * x)

        # load the matching slice of the (transposed) weight matrix
        #   w_ptr : [dim, out_dim]  → we need a block (BLOCK_D, BLOCK_O)
        w = tl.load(
            w_ptr + (d_off[:, None] * out_dim + col[None, :]),
            mask=mask_d[:, None] & mask_col[None, :],
            other=0.0,
        ).to(tl.float32)                # -> (BLOCK_D, BLOCK_O)

        # vector‑matrix multiplication for this tile
        dot += tl.dot(x, w)             # (BLOCK_O,)

    # -----------------------------------------------------------------
    #  Compute LayerNorm statistics
    # -----------------------------------------------------------------
    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)         # 1/√(var+eps)

    # -----------------------------------------------------------------
    #  Load row‑sum of the scaled weight and the bias for this tile
    # -----------------------------------------------------------------
    wsum = tl.load(wsum_ptr + col, mask=mask_col, other=0.0)   # (BLOCK_O,)
    b    = tl.load(bias_ptr + col, mask=mask_col, other=0.0)   # (BLOCK_O,)

    # -----------------------------------------------------------------
    #  Final projection:
    #      proj = ((dot - mean*wsum) * rstd) + b
    # -----------------------------------------------------------------
    proj = ((dot - mean * wsum) * rstd) + b
    proj_fp16 = proj.to(tl.float16)

    # -----------------------------------------------------------------
    #  Store result
    # -----------------------------------------------------------------
    out_ptrs = out_ptr + row * out_dim + col
    tl.store(out_ptrs, proj_fp16, mask=mask_row & mask_col)


# ----------------------------------------------------------------------
# 2) Second LayerNorm + output‑gate – one Triton kernel
# ----------------------------------------------------------------------
@triton.jit
def layernorm_gate_kernel(
    x_ptr,            # *const half*   [total_rows, dim]
    gate_ptr,         # *const half*   [total_rows, dim]
    y_ptr,            # *mut half*     [total_rows, dim]
    gamma_ptr,        # *const half*   [dim]   (to_out_norm.weight)
    beta_ptr,         # *const half*   [dim]   (to_out_norm.bias)
    total_rows: tl.constexpr,
    dim:        tl.constexpr,
    eps:        tl.constexpr,
    BLOCK_M:    tl.constexpr,
    BLOCK_D:    tl.constexpr,
):
    """Row‑wise LayerNorm (γ,β) followed by element‑wise gating."""
    pid = tl.program_id(0)
    row = pid * BLOCK_M + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_row = row < total_rows

    # ---------------------------------------------------------------
    #  First pass: compute mean & var of the raw tensor
    # ---------------------------------------------------------------
    sum_val = tl.float32(0.0)
    sum_sq  = tl.float32(0.0)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        d_start = d * BLOCK_D
        d_off  = d_start + tl.arange(0, BLOCK_D)
        mask_d = d_off < dim

        x = tl.load(
            x_ptr + (row[:, None] * dim + d_off[None, :]),
            mask=mask_row[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                # (BLOCK_D,)

        sum_val += tl.sum(x)
        sum_sq  += tl.sum(x * x)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------------------------------------------------------------
    #  Second pass: apply LayerNorm, γ/β, and gate
    # ---------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        d_start = d * BLOCK_D
        d_off  = d_start + tl.arange(0, BLOCK_D)
        mask_d = d_off < dim

        # load input and gate values
        x = tl.load(
            x_ptr + (row[:, None] * dim + d_off[None, :]),
            mask=mask_row[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                # (BLOCK_D,)
        g = tl.load(
            gate_ptr + (row[:, None] * dim + d_off[None, :]),
            mask=mask_row[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                # (BLOCK_D,)

        # γ & β for this channel block
        gamma = tl.load(gamma_ptr + d_off, mask=mask_d, other=0.0).to(tl.float32)
        beta  = tl.load(beta_ptr  + d_off, mask=mask_d, other=0.0).to(tl.float32)

        # normalized value
        norm = (x - mean) * rstd
        # LayerNorm + bias, then gate
        y = (norm * gamma + beta) * g

        # store back as fp16
        y_ptr = y_ptr + (row[:, None] * dim + d_off[None, :])
        tl.store(y_ptr, y.to(tl.float16), mask=mask_row[:, None] & mask_d[None, :])


# ======================================================================
# Entry point – the custom forward pass
# ======================================================================
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator (AlphaFold‑3).

    The implementation fuses both LayerNorms with the surrounding linear
    projections into two Triton kernels, uses fp16 Tensor‑Core GEMMs for
    the heavy matrix multiplications and keeps the final output in fp32.
    """
    # ------------------------------------------------------------------
    #  Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    #  Flattened view for the first block (B·N·N rows)
    # ------------------------------------------------------------------
    total_rows = B * N * N
    x_flat = input_tensor.contiguous().view(total_rows, dim)   # fp32

    # ------------------------------------------------------------------
    #  Pre‑process weights for the fused kernel
    # ------------------------------------------------------------------
    # LayerNorm params (first LN)
    norm_w = weights["norm.weight"]         # [dim] (fp32)
    norm_b = weights["norm.bias"]           # [dim]

    # Concatenate the five projection matrices (shape: [5*H, dim])
    w_left        = weights["left_proj.weight"]
    w_right       = weights["right_proj.weight"]
    w_left_gate   = weights["left_gate.weight"]
    w_right_gate  = weights["right_gate.weight"]
    w_out_gate    = weights["out_gate.weight"]
    concat_w = torch.cat(
        [w_left, w_right, w_left_gate, w_right_gate, w_out_gate],
        dim=0,
    ).to(torch.float32)                     # (out_dim, dim)

    out_dim = 5 * hidden_dim                 # = 5·H

    # Scale the concatenated weight by the LayerNorm γ (norm_w)
    #   pre_w = concat_w * norm_w            (broadcast over columns)
    pre_w = concat_w * norm_w[None, :]       # (out_dim, dim)

    # Bias contributed by the LayerNorm β (norm_b) before projection:
    #   pre_b = (norm_b * concat_w).sum(dim=1)   # shape (out_dim)
    pre_b = (norm_b * concat_w).sum(dim=1)    # fp32

    # Row‑sum of the scaled weight – needed for the mean‑correction term
    pre_w_sum = pre_w.sum(dim=1)              # (out_dim,)

    # Transpose for the kernel (dim, out_dim) and cast to fp16
    pre_w_T = pre_w.t().contiguous().half()   # (dim, out_dim)

    # ------------------------------------------------------------------
    #  Kernel 1: fused LayerNorm + 5 linear projections
    # ------------------------------------------------------------------
    proj_all = torch.empty((total_rows, out_dim), dtype=torch.float16, device=device)

    BLOCK_M = 64            # rows per program (multiple of 32)
    BLOCK_D = 32            # input‑channel tile
    BLOCK_O = 32            # output‑tile (covers hidden_dim chunks)

    grid = (triton.cdiv(total_rows, BLOCK_M),
            triton.cdiv(out_dim,   BLOCK_O))
    proj_norm_linear_kernel[grid](
        x_ptr=x_flat,
        w_ptr=pre_w_T,
        wsum_ptr=pre_w_sum,
        bias_ptr=pre_b,
        out_ptr=proj_all,
        total_rows=total_rows,
        dim=dim,
        out_dim=out_dim,
        eps=eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        BLOCK_O=BLOCK_O,
    )
    # No explicit sync needed – later PyTorch ops will be serialized.

    # ------------------------------------------------------------------
    #  Split concatenated projection into the five tensors
    # ------------------------------------------------------------------
    splits = torch.split(proj_all, hidden_dim, dim=1)
    left_proj_fp16, right_proj_fp16, left_gate_fp16, right_gate_fp16, out_gate_fp16 = splits

    # ------------------------------------------------------------------
    #  Mask (if present) – applied after the linear projection
    # ------------------------------------------------------------------
    if not nomask:
        mask_flat = mask_tensor.contiguous().view(total_rows, 1).to(torch.float16)
        left_proj_fp16  = left_proj_fp16 * mask_flat
        right_proj_fp16 = right_proj_fp16 * mask_flat

    # ------------------------------------------------------------------
    #  Sigmoid gates (still fp16) and element‑wise gating
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate_fp16)
    right_gate = torch.sigmoid(right_gate_fp16)
    out_gate   = torch.sigmoid(out_gate_fp16)          # will be used later

    left  = left_proj_fp16  * left_gate                # (total_rows, H)
    right = right_proj_fp16 * right_gate               # (total_rows, H)

    # ------------------------------------------------------------------
    #  Reshape to 4‑D tensors for the triangular update
    # ------------------------------------------------------------------
    left  = left .view(B, N, N, hidden_dim)
    right = right.view(B, N, N, hidden_dim)
    out_gate = out_gate.view(B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    #  Triangular multiplicative update (batched GEMM)
    # ------------------------------------------------------------------
    # bring the hidden dim to the batch axis
    left_perm  = left .permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()   # (B, H, N, N) – transposed on k
    BH = B * hidden_dim
    left_mat  = left_perm .view(BH, N, N)                 # (B·H, N, N)
    right_mat = right_perm.view(BH, N, N)                 # (B·H, N, N)

    out_mat = torch.bmm(left_mat, right_mat)              # (B·H, N, N) fp16
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # ------------------------------------------------------------------
    #  Second LayerNorm + output‑gate (fused)
    # ------------------------------------------------------------------
    total_rows_h = B * N * N
    out_flat = out.view(total_rows_h, hidden_dim)        # fp16

    # to_out_norm parameters (fp16)
    to_out_norm_w = weights["to_out_norm.weight"].half()
    to_out_norm_b = weights["to_out_norm.bias"].half()

    gated_out = torch.empty_like(out_flat)                # fp16 buffer

    BLOCK_M2 = 128
    BLOCK_D2 = 32
    grid2 = (triton.cdiv(total_rows_h, BLOCK_M2),)
    layernorm_gate_kernel[grid2](
        x_ptr=out_flat,
        gate_ptr=out_gate.view(total_rows_h, hidden_dim),
        y_ptr=gated_out,
        gamma_ptr=to_out_norm_w,
        beta_ptr=to_out_norm_b,
        total_rows=total_rows_h,
        dim=hidden_dim,
        eps=eps,
        BLOCK_M=BLOCK_M2,
        BLOCK_D=BLOCK_D2,
    )

    # ------------------------------------------------------------------
    #  Final linear projection back to `dim`
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].half()           # (dim, hidden_dim)
    final_fp16 = F.linear(gated_out, to_out_w)           # (total_rows_h, dim) fp16

    final = final_fp16.view(B, N, N, dim).to(torch.float32)  # fp32 output
    return final
