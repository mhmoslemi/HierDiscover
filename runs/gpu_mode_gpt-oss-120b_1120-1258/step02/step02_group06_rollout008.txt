"""
TriMul (outgoing) – fast Triton implementation

This kernel computes the cubic contraction
    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
where
    left  = (x·W_left ) * sigmoid(x·W_left_gate )
    right = (x·W_right) * sigmoid(x·W_right_gate)

All heavy arithmetic (the N³ hidden‑dim matrix multiply) is performed with a
batched GEMM written in Triton.  The kernel tiles the output with
BLOCK_M × BLOCK_N output elements and reduces over K in steps of BLOCK_K.
Tiles are sized 128 × 64 (or 64 × 64 for very small sequences) so that the
Tensor‑Core‑accelerated `tl.dot` can keep the H100 fully occupied while
staying within the 64 KB shared‑memory limit per program.

Only the contraction is done in Triton; the surrounding LayerNorms,
linear projections and sigmoid gates are expressed with regular PyTorch
operations (they are memory‑bound and already highly optimised).  The final
output is returned in float32.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ   (A,B : M×N×N,  M = B·hidden_dim)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    a_ptr,               # left  matrix (M, N, N)  – fp16
    b_ptr,               # right matrix (M, N, N)  – fp16, read transposed
    c_ptr,               # output  (M, N, N)      – fp32
    N,                   # sequence length
    stride_am, stride_ai, stride_ak,   # strides of A: (M, i, k)
    stride_bm, stride_bj, stride_bk,   # strides of B: (M, j, k) – we index as (k, j)
    stride_cm, stride_ci, stride_cj,   # strides of C: (M, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # --------------------------------------------------------------
    #  Program identifiers
    # --------------------------------------------------------------
    pid_m = tl.program_id(0)          # tile along i  (row)
    pid_n = tl.program_id(1)          # tile along j  (col)
    pid_b = tl.program_id(2)          # batch * hidden index

    # --------------------------------------------------------------
    #  Offsets for the tile
    # --------------------------------------------------------------
    i0 = pid_m * BLOCK_M
    j0 = pid_n * BLOCK_N

    i = i0 + tl.arange(0, BLOCK_M)
    j = j0 + tl.arange(0, BLOCK_N)

    mask_i = i < N
    mask_j = j < N

    # --------------------------------------------------------------
    #  Pointer to the output tile C[i, j]
    # --------------------------------------------------------------
    c_ptrs = c_ptr + pid_b * stride_cm + i[:, None] * stride_ci + j[None, :] * stride_cj
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    #  Reduction over the k dimension
    # --------------------------------------------------------------
    for k0 in range(0, N, BLOCK_K):
        k = k0 + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # A[i, k]   – left side (fp16)
        a_ptrs = a_ptr + pid_b * stride_am + i[:, None] * stride_ai + k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)

        # B[k, j]   – right side read transposed (fp16)
        b_ptrs = b_ptr + pid_b * stride_bm + k[:, None] * stride_bk + j[None, :] * stride_bj
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)

        # Tensor‑core dot – accumulate in fp32
        acc += tl.dot(a, b)

    # --------------------------------------------------------------
    #  Write the result (fp32)
    # --------------------------------------------------------------
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
#  Public entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module.

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
            - mask         : torch.Tensor, shape [B, N, N] (bool or float)
            - weights      : dict of model parameters (already on the target device)
            - config       : dict containing at least the keys
                             "dim", "hidden_dim", "nomask" (optional)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data

    # ------------------------------------------------------------------
    #  Hyper‑parameters
    # ------------------------------------------------------------------
    dim        = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask     = config.get("nomask", True)
    eps        = 1e-5

    # ------------------------------------------------------------------
    #  Unpack weights (they are already on the correct device)
    # ------------------------------------------------------------------
    norm_weight          = weights["norm.weight"]
    norm_bias            = weights["norm.bias"]
    left_proj_weight     = weights["left_proj.weight"]
    right_proj_weight    = weights["right_proj.weight"]
    left_gate_weight     = weights["left_gate.weight"]
    right_gate_weight    = weights["right_gate.weight"]
    out_gate_weight      = weights["out_gate.weight"]
    to_out_norm_weight   = weights["to_out_norm.weight"]
    to_out_norm_bias     = weights["to_out_norm.bias"]
    to_out_weight        = weights["to_out.weight"]

    # ------------------------------------------------------------------
    #  1. Layer‑norm over the channel dimension (dim)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )                                                # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    #  Work in half‑precision for the heavy ops
    # ------------------------------------------------------------------
    x_h = x.half()

    # ------------------------------------------------------------------
    #  Cast tiny weight matrices to fp16 once
    # ------------------------------------------------------------------
    left_proj_w_h   = left_proj_weight.half()
    right_proj_w_h  = right_proj_weight.half()
    left_gate_w_h   = left_gate_weight.half()
    right_gate_w_h  = right_gate_weight.half()
    out_gate_w_h    = out_gate_weight.half()

    # ------------------------------------------------------------------
    #  2. Linear projections (bias‑free) → left, right  (fp16)
    # ------------------------------------------------------------------
    left  = torch.nn.functional.linear(x_h, left_proj_w_h)   # [B, N, N, H]
    right = torch.nn.functional.linear(x_h, right_proj_w_h)  # [B, N, N, H]

    # ------------------------------------------------------------------
    #  3. Optional mask (broadcasted on H)
    # ------------------------------------------------------------------
    if not nomask:
        # mask may be bool or float – cast to fp16 and broadcast
        mask_exp = mask.to(x_h.dtype).unsqueeze(-1)        # [B, N, N, 1]
        left  = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    #  4. Gating (sigmoid) – still in fp16
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x_h, left_gate_w_h))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_h, right_gate_w_h))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x_h, out_gate_w_h))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    #  5. Batched GEMM:  out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    #                  (implemented with Triton)
    # ------------------------------------------------------------------
    #   Rearrange to (B, H, N, N) and flatten the first two dims so that
    #   each (b, h) pair becomes an independent “batch” for the kernel.
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    B, H, N, _ = left_perm.shape
    M_total = B * H

    left_flat  = left_perm.view(M_total, N, N)   # [M, N, N]  fp16
    right_flat = right_perm.view(M_total, N, N)  # [M, N, N]  fp16
    out_flat   = torch.empty((M_total, N, N), dtype=torch.float32,
                             device=left_flat.device)

    # --------------------------------------------------------------
    #  Choose block sizes (empirically good for H100)
    # --------------------------------------------------------------
    if N <= 256:
        BLOCK_M = 64
        BLOCK_N = 64
    else:
        # 128×64 tiles give high occupancy while staying < 64 KB shared mem
        BLOCK_M = 128
        BLOCK_N = 64
    BLOCK_K = 64               # a power‑of‑two that divides most seq‑lens

    # --------------------------------------------------------------
    #  Strides (as Python integers) – fall back to .stride()
    # --------------------------------------------------------------
    stride_am = left_flat.stride(0)
    stride_ai = left_flat.stride(1)
    stride_ak = left_flat.stride(2)

    stride_bm = right_flat.stride(0)
    stride_bj = right_flat.stride(1)   # stride over the “j” axis
    stride_bk = right_flat.stride(2)   # stride over the “k” axis

    stride_cm = out_flat.stride(0)
    stride_ci = out_flat.stride(1)
    stride_cj = out_flat.stride(2)

    # --------------------------------------------------------------
    #  Grid layout: (tiles_i, tiles_j, batch*hidden)
    # --------------------------------------------------------------
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        M_total,
    )

    # ------------------------------------------------------------------
    #  Launch the Triton kernel
    # ------------------------------------------------------------------
    tri_mul_outgoing_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        stride_am, stride_ai, stride_ak,
        stride_bm, stride_bj, stride_bk,
        stride_cm, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    #  Reshape back to [B, N, N, H]
    # ------------------------------------------------------------------
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # fp32

    # ------------------------------------------------------------------
    #  6. Hidden‑dim layer‑norm, output gating, final projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )                                                   # fp32

    out = out * out_gate.to(out.dtype)                 # broadcast gate (fp32)

    # final linear back to the original dimension (float32)
    output = torch.nn.functional.linear(out, to_out_weight)

    return output
```
