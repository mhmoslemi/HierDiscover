"""
TriMul (outgoing) – Triton‑accelerated forward pass
===================================================

This implementation follows the PyTorch reference in AlphaFold‑3 but
places the cubic contraction

    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]

into a custom Triton kernel.  For every (batch, head) pair we compute a
matrix multiplication C = A @ Bᵀ where A and B have shape (N, N) and the
accumulation is performed in FP32 while the inputs are FP16, thus fully
exploiting Tensor‑cores on H100.

Algorithm
---------
1. Layer‑norm the input tensor on the feature dimension (float32).
2. Cast to FP16 and apply bias‑free linear projections for *left* and *right*,
   together with three sigmoid gates.  An optional mask is applied before
   gating.
3. Reshape to (M, N, N) with M = batch × hidden_dim and launch the Triton
   kernel that computes the batched GEMM C = A @ Bᵀ for each M.
4. Reshape the result back to [B, N, N, hidden_dim] (float32).
5. Second Layer‑norm, multiply by the output gate and apply a bias‑free
   linear projection back to the original feature dimension.

The kernel is intentionally simple (no shared memory) but uses
`tl.dot` which maps to Tensor‑core MMA: inputs are FP16, the accumulator
is FP32.  Block sizes (128×128×64) have been chosen to give high occupancy
on an H100 while keeping register pressure modest.

"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_tri_gemm_kernel(
    A,               # left  tensor (M, N, N)  – FP16
    B,               # right tensor (M, N, N)  – FP16 (read as Bᵀ)
    C,               # output (M, N, N)       – FP32
    N,               # sequence length
    stride_am, stride_ai, stride_ak,   # strides of A: batch, row(i), col(k)
    stride_bm, stride_bj, stride_bk,   # strides of B: batch, col(j), row(k) (Bᵀ)
    stride_cm, stride_ci, stride_cj,   # strides of C: batch, row(i), col(j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM: C = A @ Bᵀ for each entry in the first dimension.
    All tensors are contiguous, A/B are FP16, C is FP32.
    """
    batch_id = tl.program_id(0)          # index over M = batch * hidden_dim
    pid_m    = tl.program_id(1)          # tile row (i)
    pid_n    = tl.program_id(2)          # tile col (j)

    # Compute global row / column indices for this tile
    i0 = pid_m * BLOCK_M
    j0 = pid_n * BLOCK_N
    i = i0 + tl.arange(0, BLOCK_M)
    j = j0 + tl.arange(0, BLOCK_N)

    # Masks for the tail of the matrix
    mask_i = i < N
    mask_j = j < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over reduction axis k
    for k0 in range(0, N, BLOCK_K):
        k = k0 + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # Load A[batch, i, k]  → shape (BLOCK_M, BLOCK_K)
        a_ptr = A + batch_id * stride_am + i[:, None] * stride_ai + k[None, :] * stride_ak
        a = tl.load(
            a_ptr,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )          # fp16

        # Load Bᵀ → B[batch, k, j]  → shape (BLOCK_K, BLOCK_N)
        b_ptr = B + batch_id * stride_bm + k[:, None] * stride_bk + j[None, :] * stride_bj
        b = tl.load(
            b_ptr,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )          # fp16

        # Tensor‑core dot (fp16 * fp16 → fp32)
        acc += tl.dot(a, b)

    # Write the tile back to C
    c_ptr = C + batch_id * stride_cm + i[:, None] * stride_ci + j[None, :] * stride_cj
    tl.store(
        c_ptr,
        acc,
        mask=mask_i[:, None] & mask_j[None, :],
    )


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module using a Triton kernel for the
    cubic contraction.  All tensors are assumed to be on the CUDA device.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
            - mask          : torch.Tensor of shape [B, N, N]   (any dtype)
            - weights       : dict with model parameters (float32)
            - config        : dict containing "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------------- #
    # 0. Unpack inputs
    # --------------------------------------------------------------------- #
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    # --------------------------------------------------------------------- #
    # 1. First LayerNorm (float32)
    # --------------------------------------------------------------------- #
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  float32

    # --------------------------------------------------------------------- #
    # 2. Cast to FP16 and compute bias‑free projections + sigmoid gates
    # --------------------------------------------------------------------- #
    x_h = x.half()   # fp16 view of the normalized tensor

    # Linear projections (bias‑free)
    left = F.linear(x_h, weights["left_proj.weight"].half())   # [B, N, N, H] fp16
    right = F.linear(x_h, weights["right_proj.weight"].half())

    # Optional mask (broadcast over hidden dimension)
    if not nomask and mask is not None:
        # Ensure mask is fp16 and has a trailing singleton dimension
        mask_exp = mask.to(left.dtype).unsqueeze(-1)          # [B, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    # Sigmoid gates (still fp16)
    left_gate = torch.sigmoid(F.linear(x_h, weights["left_gate.weight"].half()))
    right_gate = torch.sigmoid(F.linear(x_h, weights["right_gate.weight"].half()))
    out_gate = torch.sigmoid(F.linear(x_h, weights["out_gate.weight"].half()))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------------- #
    # 3. Batched GEMM: out[b, h, i, j] = Σ_k left[b, i, k, h] * right[b, j, k, h]
    # --------------------------------------------------------------------- #
    B, N, _, _ = left.shape                     # B, N, N, H
    # Bring hidden dim to second axis and make tensors contiguous:
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    M_total = B * hidden_dim                     # number of independent GEMMs
    left_flat = left_perm.view(M_total, N, N)    # [M, N, N]  fp16
    right_flat = right_perm.view(M_total, N, N)  # [M, N, N]  fp16

    # Output buffer in FP32
    out_flat = torch.empty((M_total, N, N), dtype=torch.float32, device=left_flat.device)

    # --------------------------------------------------------------------- #
    # 4. Triton kernel launch
    # --------------------------------------------------------------------- #
    # Block sizes – tuned for H100 Tensor‑cores
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    # Strides (int64 scalars)
    stride_am = left_flat.stride(0)
    stride_ai = left_flat.stride(1)
    stride_ak = left_flat.stride(2)

    stride_bm = right_flat.stride(0)
    stride_bj = right_flat.stride(1)   # row (j) stride
    stride_bk = right_flat.stride(2)   # col (k) stride

    stride_cm = out_flat.stride(0)
    stride_ci = out_flat.stride(1)
    stride_cj = out_flat.stride(2)

    # Grid: (batch*hidden, tile_rows, tile_cols)
    grid = (
        M_total,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )
    batched_tri_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bj,
        stride_bk,
        stride_cm,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------------- #
    # 5. Reshape back to [B, N, N, hidden_dim] (float32)
    # --------------------------------------------------------------------- #
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H]

    # --------------------------------------------------------------------- #
    # 6. Second LayerNorm, output‑gate multiplication, final projection
    # --------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate.float()
    output = F.linear(out, weights["to_out.weight"])

    return output
