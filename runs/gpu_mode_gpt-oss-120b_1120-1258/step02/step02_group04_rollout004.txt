"""
Highly‑optimized Triton implementation of the *outgoing* TriMul operator
(AlphaFold‑3 style).

Algorithm
---------
1. Layer‑norm the 4‑D input (float32 → float16 for the heavy work).
2. Fuse the five linear layers (left_proj, left_gate, right_proj,
   right_gate, out_gate) into a *single* half‑precision GEMM by concatenating
   their weight matrices.  The result is sliced into the five tensors.
3. Apply the optional mask (broadcast on the hidden dimension) and the
   sigmoid gates.  This yields the gated tensors ``left`` and ``right`` of
   shape ``[B, N, N, H]`` (H = hidden_dim) in fp16.
4. Compute the N³ contraction
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   as a **batched half‑precision GEMM** ``C = A @ Bᵀ`` for each hidden
   slice.  The kernel works on 128×128 tiles with a 64‑element K‑tile and
   uses FP32 accumulation (Tensor‑cores) on H100.
5. Reshape the result back to ``[B, N, N, H]``, cast to float32 and apply the
   output LayerNorm, the ``out_gate`` and the final linear projection back to
   the original feature dimension ``dim``.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Batched GEMM kernel: C[b] = A[b] @ B[b]ᵀ  (A: M×K, B: N×K)
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr,                    # fp16 *  (batch, M, K)
    b_ptr,                    # fp16 *  (batch, K, N)   ← accessed as (K,N)
    c_ptr,                    # fp16 *  (batch, M, N)
    M, N, K,                  # int32 matrix dimensions
    stride_az, stride_am, stride_ak,  # strides for A
    stride_bz, stride_bk, stride_bn,  # strides for B
    stride_cz, stride_cm, stride_cn,  # strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched half‑precision GEMM with FP32 accumulation.
    The first program‑id dimension packs (batch, tile_m), the second packs tile_n.
    """
    pid0 = tl.program_id(0)          # batch_id * tiles_M + tile_m
    pid1 = tl.program_id(1)          # tile_n

    # -------------------------------------------------------------
    # Decode batch index and tile coordinates
    # -------------------------------------------------------------
    num_tiles_m = tl.cdiv(M, BLOCK_M)
    batch_id   = pid0 // num_tiles_m
    tile_m     = pid0 %  num_tiles_m
    tile_n     = pid1

    # -------------------------------------------------------------
    # Offsets inside the current tile
    # -------------------------------------------------------------
    offs_m = tile_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = tile_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Pointers to the start of the tile (without the k‑loop offset)
    a_ptrs = a_ptr + batch_id * stride_az \
                  + offs_m[:, None] * stride_am \
                  + offs_k[None, :] * stride_ak
    b_ptrs = b_ptr + batch_id * stride_bz \
                  + offs_k[:, None] * stride_bk \
                  + offs_n[None, :] * stride_bn

    # -------------------------------------------------------------
    # Accumulator (FP32 for numerical safety)
    # -------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------------------------------------------------
    # Main K‑loop
    # -------------------------------------------------------------
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        mask_a = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        mask_b = (offs_k[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=mask_a, other=0.0)   # [BLOCK_M, BLOCK_K] fp16
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)   # [BLOCK_K, BLOCK_N] fp16

        # fp16⋅fp16 → fp32 accumulation (uses Tensor‑cores)
        acc += tl.dot(a, b)

        # Advance pointers to next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # -------------------------------------------------------------
    # Store the result (back to fp16)
    # -------------------------------------------------------------
    c_ptrs = c_ptr + batch_id * stride_cz \
                    + offs_m[:, None] * stride_cm \
                    + offs_n[None, :] * stride_cn
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)

    """
    Forward pass of the *outgoing* TriMul operator (no gradients).

    Args
    ----
    data : tuple

    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim        = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask     = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1) Input LayerNorm (keep in FP32 for stability)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x_fp16 = x.to(torch.float16)          # heavy work in fp16

    # -----------------------------------------------------------------
    # 2) Fuse the five linear layers into a single GEMM
    # -----------------------------------------------------------------
    fused_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                   # shape [(5*H), dim]

    H = hidden_dim
    left_proj = fused[..., :H]
    left_gate = fused[..., H:2 * H]
    right_proj = fused[..., 2 * H:3 * H]
    right_gate = fused[..., 3 * H:4 * H]
    out_gate = fused[..., 4 * H:5 * H]

    # -----------------------------------------------------------------

    # -----------------------------------------------------------------
    if not nomask and mask is not None:

    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    left  = left_proj * left_gate
    right = right_proj * right_gate

    # -----------------------------------------------------------------
    # 5) N³ contraction via a fused batched GEMM (A @ Bᵀ)
    # -----------------------------------------------------------------
    B, N, _, _ = left.shape          # left/right: [B,N,N,H]

    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 2, 1).contiguous().view(B * H, N, N)

    out_mat = torch.empty_like(left_mat, dtype=torch.float16, device=device)

    # -----------------------------------------------------------------
    # Triton launch configuration (tuned for H100)
    # -----------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid_m = triton.cdiv(N, BLOCK_M)
    grid_n = triton.cdiv(N, BLOCK_N)

    # pid0 packs (batch*hidden, tile_m)
    grid = (B * H * grid_m, grid_n)

    batched_matmul_kernel[grid](
        a_ptr=left_mat,
        b_ptr=right_mat,
        c_ptr=out_mat,
        M=N,
        N=N,
        K=N,
        stride_az=left_mat.stride(0),
        stride_am=left_mat.stride(1),
        stride_ak=left_mat.stride(2),
        stride_bz=right_mat.stride(0),
        stride_bk=right_mat.stride(1),
        stride_bn=right_mat.stride(2),
        stride_cz=out_mat.stride(0),
        stride_cm=out_mat.stride(1),
        stride_cn=out_mat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )

    # -----------------------------------------------------------------
    # 6) Reshape back to [B, N, N, H] and cast to FP32
    # -----------------------------------------------------------------
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    return out
