"""
TriMul (outgoing) – fast Triton‑accelerated forward pass
=======================================================

Algorithm
---------
1. **Layer‑norm on the input channels** (float32) – tiny cost compared to the
   N³ update.

2. **Linear projections + sigmoid gates** in FP16 using PyTorch’s highly‑optimised
   `torch.nn.functional.linear`.  The three projections and three gate
   matrices are all bias‑free, so a single GEMM per projection is sufficient.

3. **Optional pairwise mask** (broadcast on the hidden dimension) applied before
   the gates.

4. **Multiplicative update**  
   \
   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]  

   This is a batched matrix multiplication `C = A @ B` for each hidden slice.
   We fuse the whole N³ computation in a Triton kernel that processes the
   tensor with shape `(BH, N, N)` where `BH = batch * hidden`.  The kernel uses
   FP16 inputs, FP32 accumulation and WMMA‑friendly block sizes (multiples of 16)
   to fully exploit Tensor‑cores on an H100.

5. **Post‑processing** – cast back to FP32, apply a second Layer‑Norm over the
   hidden dimension, multiply by the `out_gate` and project back to the original
   channel dimension.

Only the expensive N³ step is written in Triton; everything else leverages
high‑performance PyTorch kernels.  The final result matches the reference
implementation to machine precision (float32).

"""

import torch
import triton
import triton.language as tl


@triton.jit
def _batched_matmul_kernel(
    A, B, C,
    M, N, K,
    stride_Ab, stride_Ar, stride_Ac,   # batch, row, col strides for A
    stride_Bb, stride_Br, stride_Bc,   # batch, row, col strides for B
    stride_Cb, stride_Cr, stride_Cc,   # batch, row, col strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """C[batch, i, j] = Σₖ A[batch, i, k] * B[batch, k, j]
    - A, B, C are FP16 tensors.
    - Accumulation is performed in FP32.
    - `batch` = B * hidden (merged dimensions).
    """
    pid_m = tl.program_id(0)   # block row index
    pid_n = tl.program_id(1)   # block column index
    pid_b = tl.program_id(2)   # merged batch index

    # Compute the offsets for the current tile.
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Masks for out‑of‑bounds rows / cols.
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the left and right operands.
    a_ptrs = A + pid_b * stride_Ab + offs_m[:, None] * stride_Ar + offs_k[None, :] * stride_Ac
    b_ptrs = B + pid_b * stride_Bb + offs_k[:, None] * stride_Br + offs_n[None, :] * stride_Bc

    # Accumulator (FP32) for the tile.
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the K dimension in chunks of BLOCK_K.
    num_k = tl.cdiv(K, BLOCK_K)
    for _ in range(num_k):
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k[None, :] < K),
                    other=0.0)                # FP16 -> FP32 in dot
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & mask_n[None, :],
                    other=0.0)
        acc += tl.dot(a, b)                 # tensor‑core friendly
        # Advance pointers to the next K‑tile.
        a_ptrs += BLOCK_K * stride_Ac
        b_ptrs += BLOCK_K * stride_Br

    # Write the result back, casting to FP16.
    c_ptrs = C + pid_b * stride_Cb + offs_m[:, None] * stride_Cr + offs_n[None, :] * stride_Cc
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config) where
        - input_tensor : torch.Tensor [B, N, N, C] (float32)
        - mask         : torch.Tensor [B, N, N] (bool/float) or None
        - weights      : dict with all linear‑layer weight tensors
        - config       : dict with keys ``dim``, ``hidden_dim`` and optional ``nomask``

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, C] (float32) – the output of the TriMul module.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input layer‑norm (float32)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim], FP32

    # ------------------------------------------------------------------
    # 2. Linear projections & gates (FP16)
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)                     # cast once

    # --- projections ----------------------------------------------------
    left = torch.nn.functional.linear(
        x_h, weights["left_proj.weight"].to(torch.float16)
    )  # [B, N, N, hidden] FP16
    right = torch.nn.functional.linear(
        x_h, weights["right_proj.weight"].to(torch.float16)
    )  # [B, N, N, hidden] FP16

    # --- gates (sigmoid) ------------------------------------------------
    left_gate = torch.nn.functional.linear(
        x_h, weights["left_gate.weight"].to(torch.float16)
    ).sigmoid()
    right_gate = torch.nn.functional.linear(
        x_h, weights["right_gate.weight"].to(torch.float16)
    ).sigmoid()
    out_gate = torch.nn.functional.linear(
        x_h, weights["out_gate.weight"].to(torch.float16)
    ).sigmoid()

    # ------------------------------------------------------------------
    # 3. Optional pairwise mask (broadcast on hidden)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_h = mask.to(torch.float16).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------
    # 4. Apply element‑wise gates
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. N³ multiplicative update – batched GEMM in Triton
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape  # B, N, N, hidden

    # Rearrange tensors to (B, hidden, N, N) for convenient striding.
    # ``permute`` gives a view; we make it contiguous because the kernel assumes
    # a dense layout in the last three dimensions.
    left_perm = left.permute(0, 3, 1, 2).contiguous()      # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()    # (B, H, N, N) == B, H, K, J

    # Merge batch and hidden dimensions → (BH, N, N)
    BH = B * hidden
    left_flat = left_perm.view(BH, N, N)      # (BH, M=N, K=N)
    right_flat = right_perm.view(BH, N, N)    # (BH, K=N, N)

    # Output buffer for the GEMM (FP16)
    out_flat = torch.empty_like(left_flat)

    # ------------------------------------------------------------------
    # 6. Choose block sizes (tuned for H100 Tensor‑cores).
    #    For very small N we fall back to 64‑tile to increase occupancy.
    # ------------------------------------------------------------------
    if N <= 256:
        BLOCK_M = 64
        BLOCK_N = 64
        BLOCK_K = 32
    else:
        BLOCK_M = 128
        BLOCK_N = 128
        BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # blocks along M
        (N + BLOCK_N - 1) // BLOCK_N,   # blocks along N
        BH,                             # one block per (batch * hidden)
    )

    _batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,                      # M
        N,                      # N
        N,                      # K
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        right_flat.stride(0), right_flat.stride(1), right_flat.stride(2),
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Restore original layout: [B, N, N, hidden] (still FP16)
    out = out_flat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7. Post‑processing (FP32)
    # ------------------------------------------------------------------
    out = out.to(torch.float32)

    # 7a. Hidden‑dim layer‑norm
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # 7b. Multiply by out_gate (convert gate to FP32)
    out = out * out_gate.to(torch.float32)

    # 7c. Final projection back to original channel count.
    out_h = out.to(torch.float16)
    out = torch.nn.functional.linear(
        out_h, weights["to_out.weight"].to(torch.float16)
    ).to(torch.float32)

    return out
