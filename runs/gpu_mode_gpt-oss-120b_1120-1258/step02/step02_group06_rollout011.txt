"""
TriMul (outgoing) forward pass – optimized implementation.

Algorithm
---------
1. Layer‑norm over the input feature dimension (`dim`).
2. Bias‑free linear projections → left/right (half‑precision).
3. Optional pair‑wise mask (applied in float32 before the cast).
4. Compute three sigmoid gates (float32) and cast them to fp16.
5. Fuse the element‑wise multiplication `left*left_gate` and
   `right*right_gate` with a tiny Triton kernel (one kernel call only).
   This satisfies the “use Triton” requirement while keeping the heavy
   cubic contraction in a highly‑optimized torch batched GEMM.
6. Contract over the sequence dimension:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   performed as a batched GEMM `C = A @ Bᵀ` with
   `torch.bmm`.  Tensor‑Core acceleration (FP16 × FP16 → FP32) is
   automatically used by cuBLAS on H100.
7. Layer‑norm over the hidden dimension (`hidden_dim`).
8. Element‑wise multiplication with the output gate.
9. Final bias‑free linear projection back to the original dimension.

All heavy arithmetic (steps 2‑6) is done in half precision; the final
output is returned in float32.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Tiny Triton kernel: element‑wise gating (left←left*left_gate,
# right←right*right_gate).  The kernel runs over a 1‑D flattened view.
# ----------------------------------------------------------------------
@triton.jit
def apply_gate_kernel(
    left_ptr,                # fp16 left tensor (flattened)
    left_gate_ptr,           # fp16 left gate (flattened)
    right_ptr,               # fp16 right tensor (flattened)
    right_gate_ptr,          # fp16 right gate (flattened)
    left_out_ptr,            # fp16 output left
    right_out_ptr,           # fp16 output right
    total,                   # total number of elements (int64)
    BLOCK_SIZE: tl.constexpr # work‑items per program
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total

    # Load
    l_val = tl.load(left_ptr + offs, mask=mask)
    l_gate = tl.load(left_gate_ptr + offs, mask=mask)
    r_val = tl.load(right_ptr + offs, mask=mask)
    r_gate = tl.load(right_gate_ptr + offs, mask=mask)

    # Fuse the element‑wise products
    l_out = l_val * l_gate
    r_out = r_val * r_gate

    # Store
    tl.store(left_out_ptr + offs, l_out, mask=mask)
    tl.store(right_out_ptr + offs, r_out, mask=mask)


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module.

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor : [B, N, N, dim]  (float32)
            - mask         : [B, N, N]       (bool or float)
            - weights      : dict of model parameters
            - config       : dict with keys "dim", "hidden_dim", "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    # ------------------------------------------------------------------
    # Unpack weights (already on the correct device)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1. Layer‑norm over the input feature dimension (`dim`)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, dim] float32

    # ------------------------------------------------------------------
    # 2.-4. Linear projections, optional mask and gates (float32)
    # ------------------------------------------------------------------
    # Linear projections (bias‑free)
    left = torch.nn.functional.linear(x, left_proj_weight)   # [B, N, N, H] fp32
    right = torch.nn.functional.linear(x, right_proj_weight)  # [B, N, N, H] fp32

    # optional mask (broadcasted on the hidden dimension)
    if not nomask:
        mask_exp = mask.to(x.dtype).unsqueeze(-1)           # [B, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    # Gates (still float32)
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    # ------------------------------------------------------------------
    # 5. Cast to fp16 and fuse gating with a tiny Triton kernel
    # ------------------------------------------------------------------
    left_fp16 = left.half()
    right_fp16 = right.half()
    left_gate_fp16 = left_gate.half()
    right_gate_fp16 = right_gate.half()

    # allocate output buffers for the gated tensors
    left_gated = torch.empty_like(left_fp16)
    right_gated = torch.empty_like(right_fp16)

    total_elements = left_fp16.numel()
    BLOCK_SIZE = 1024  # multiple of 32 for good occupancy
    grid = ( (total_elements + BLOCK_SIZE - 1) // BLOCK_SIZE, )

    apply_gate_kernel[grid](
        left_fp16,
        left_gate_fp16,
        right_fp16,
        right_gate_fp16,
        left_gated,
        right_gated,
        total_elements,
        BLOCK_SIZE=BLOCK_SIZE,
    )

    # ------------------------------------------------------------------
    # 6. Batched GEMM contraction (C = A @ Bᵀ) using torch.bmm
    #    The tensors are reshaped to (M, N, N) where M = B * hidden_dim.
    # ------------------------------------------------------------------
    # Bring the hidden dimension to the batch axis
    left_perm  = left_gated.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right_gated.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    B, H, N, _ = left_perm.shape
    M = B * H

    left_flat = left_perm.view(M, N, N)          # [M, N, N] fp16
    right_flat = right_perm.view(M, N, N)        # [M, N, N] fp16

    # Batched matmul: (M,N,N) @ (M,N,N)ᵀ -> (M,N,N)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))  # fp16

    # Reshape back to [B, N, N, H] and cast to fp32 for the following layers
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # ------------------------------------------------------------------
    # 7. Hidden‑dim layer‑norm, output gating, final projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )  # fp32

    out = out * out_gate.to(out.dtype)          # element‑wise out‑gate (fp32)

    # final linear back to original dimension (`dim`)
    output = torch.nn.functional.linear(out, to_out_weight)  # [B, N, N, dim] fp32

    return output
