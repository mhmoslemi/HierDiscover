"""
custom_kernel - outgoing TriMul (AlphaFold‑3)

Algorithm
---------
1. Row‑wise LayerNorm over the channel dimension `dim` (FP32) – Triton kernel.
2. Cast to fp16 and fuse the two linear projections + their gates for the left and
   right branches (concatenated weight matrix → 2 GEMMs instead of 4).
3. Optional mask multiplication (broadcast over the hidden dimension).
4. Apply the sigmoid‑gates to the projected tensors.
5. Tri‑multiplicative update:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented with a single `torch.einsum` that avoids the costly transposes /
   copies required for a batched GEMM.
6. Second LayerNorm over the hidden dimension (FP32) – Triton kernel.
7. Apply the output‑gate (sigmoid) and a final fp16 linear projection back to `dim`.
   Cast the result to fp32 and return.

All heavy reductions (layernorms) are performed by Triton.  The rest of the
computation uses fused fp16 GEMM kernels from PyTorch, which on an H100
comfortably fits the target ≤ 1 ms runtime for the benchmark configurations.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def layernorm_kernel(
    x_ptr, y_ptr, w_ptr, b_ptr,
    total_rows, dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr
):
    """Row‑wise LayerNorm (float32)."""
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # --- first pass: mean & variance ---
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)       # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --- second pass: write normalized result ---
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)

        out = norm * w[None, :] + b[None, :]
        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out, mask=mask_m[:, None] & mask_d[None, :])

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor: shape [bs, seq_len, seq_len, dim] (float32, CUDA)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32) ---------------------------------------
    # ------------------------------------------------------------------
    x = input_tensor.contiguous()
    B, N, _, _ = x.shape                     # [B, N, N, dim]
    total_rows = B * N * N

    # flatten to 2‑D for the kernel
    x_flat = x.view(total_rows, dim)          # [total_rows, dim]
    x_norm_flat = torch.empty_like(x_flat)

    BLOCK_M = 128   # rows per program
    BLOCK_D = 128   # columns per inner loop (dim <= 384)

    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm_flat,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()

    # reshape back
    x_norm = x_norm_flat.view(B, N, N, dim)   # FP32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 and fuse projection + gate linear layers ---------
    # ------------------------------------------------------------------
    x_norm_h = x_norm.view(total_rows, dim).to(torch.float16)

    # Concatenate projection & gate weights (out_features, in_features)
    left_concat_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)   # [2*H, dim]

    right_concat_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)   # [2*H, dim]

    # Linear + split
    left_concat = F.linear(x_norm_h, left_concat_w)   # [total_rows, 2*H]
    right_concat = F.linear(x_norm_h, right_concat_w)

    left_proj_h = left_concat[:, :hidden_dim]
    left_gate_pre_h = left_concat[:, hidden_dim:]

    right_proj_h = right_concat[:, :hidden_dim]
    right_gate_pre_h = right_concat[:, hidden_dim:]

    # Output gate (no bias)
    out_gate_h = torch.sigmoid(
        F.linear(x_norm_h, weights["out_gate.weight"].to(torch.float16))
    )   # [total_rows, H]

    # ------------------------------------------------------------------
    # 3️⃣ Mask (optional) ------------------------------------------------
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] → [total_rows, 1]
        mask_h = mask_tensor.view(total_rows, 1).to(torch.float16)
        left_proj_h = left_proj_h * mask_h
        right_proj_h = right_proj_h * mask_h

    # ------------------------------------------------------------------
    # 4️⃣ Apply gates ----------------------------------------------------
    # ------------------------------------------------------------------
    left_gate_h  = torch.sigmoid(left_gate_pre_h)
    right_gate_h = torch.sigmoid(right_gate_pre_h)

    left_h  = left_proj_h * left_gate_h     # [total_rows, H]
    right_h = right_proj_h * right_gate_h

    # ------------------------------------------------------------------
    # 5️⃣ Tri‑multiplicative update (einsum) ----------------------------
    # out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    left_h_4d  = left_h.view(B, N, N, hidden_dim)   # [B, N, N, H]
    right_h_4d = right_h.view(B, N, N, hidden_dim)

    # einsum avoids the two costly transposes+contiguous copies needed for
    # a batched GEMM.
    out_h = torch.einsum('b i k h, b j k h -> b i j h', left_h_4d, right_h_4d)  # FP16

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm (FP32) ---------------------------------------
    # ------------------------------------------------------------------
    out_f32 = out_h.to(torch.float32)
    total_rows_h = B * N * N
    out_flat = out_f32.view(total_rows_h, hidden_dim)
    out_norm_flat = torch.empty_like(out_flat)

    grid2 = (triton.cdiv(total_rows_h, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm_flat,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows_h,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()

    out_norm = out_norm_flat.view(B, N, N, hidden_dim)   # FP32

    # ------------------------------------------------------------------
    # 7️⃣ Output gate & final projection ---------------------------------
    # ------------------------------------------------------------------
    out_gate_f32 = out_gate_h.view(B, N, N, hidden_dim).to(torch.float32)
    out_norm = out_norm * out_gate_f32

    out_norm_h = out_norm.to(torch.float16)

    final_h = F.linear(
        out_norm_h.view(total_rows_h, hidden_dim),
        weights["to_out.weight"].to(torch.float16)
    )   # [total_rows_h, dim] (FP16)

    final = final_h.view(B, N, N, dim).to(torch.float32)

    return final

    left_gate_h  = torch.sigmoid(left_gate_pre)
    right_gate_h = torch.sigmoid(right_gate_pre)

    left_h  = left_proj_h * left_gate_h      # [total_rows, H]
    right_h = right_proj_h * right_gate_h

    # ------------------------------------------------------------------
    # 5️⃣ Tri‑multiplicative update (einsum)
    # ------------------------------------------------------------------
    left_h_4d  = left_h.view(B, N, N, hidden_dim)   # [B, N, N, H]
    right_h_4d = right_h.view(B, N, N, hidden_dim)

    # out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    out_h = torch.einsum('b i k h, b j k h -> b i j h', left_h_4d, right_h_4d)  # FP16

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm (FP32) over hidden_dim
    # ------------------------------------------------------------------
    out_f32 = out_h.to(torch.float32)
    total_rows_h = B * N * N
    out_flat = out_f32.view(total_rows_h, hidden_dim)
    out_norm_flat = torch.empty_like(out_flat)

    grid2 = (triton.cdiv(total_rows_h, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm_flat,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows_h,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()

    out_norm = out_norm_flat.view(B, N, N, hidden_dim)   # FP32

    # ------------------------------------------------------------------
    # 7️⃣ Apply output gate and final projection back to `dim`
    # ------------------------------------------------------------------
    out_gate_f32 = out_gate_h.view(B, N, N, hidden_dim).to(torch.float32)
    out_norm = out_norm * out_gate_f32

    out_norm_h = out_norm.to(torch.float16)

    final_h = F.linear(
        out_norm_h.view(total_rows_h, hidden_dim),
        weights["to_out.weight"].to(torch.float16)
    )   # [total_rows_h, dim] (FP16)

    final = final_h.view(B, N, N, dim).to(torch.float32)

    return final
```
