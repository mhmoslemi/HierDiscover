"""
custom_kernel – “outgoing” TriMul (AlphaFold‑3) – highly‑fused implementation.

Algorithm
---------
1️⃣  Row‑wise LayerNorm (FP32) on the input tensor of shape [B,N,N,C].
2️⃣  Cast the normalized tensor to FP16 and perform ONE fused linear
    projection that produces **five** tensors in a single GEMM:
        * left_proj, left_gate,
        * right_proj, right_gate,
        * out_gate
    (weights are concatenated along the output dimension).
3️⃣  Apply the optional mask (only to the two projection tensors) and the
    sigmoid‑gates.
4️⃣  Reshape the two gated tensors to [B, H, N, N] (H = hidden_dim) and
    compute the *tri‑multiplicative* update by a batched GEMM:
        out_h = left_h  @  right_hᵀ      (shape [B·H, N, N]).
    This is the same as the reference einsum
        out[...,i,j,:] = Σₖ left[...,i,k,:]·right[...,j,k,:].
5️⃣  Second LayerNorm (FP32) over the hidden dimension.
6️⃣  Multiply by the previously computed output‑gate (sigmoid) and
    apply the final linear projection back to the original channel dimension.
7️⃣  Return the result in FP32.

All heavy dense linear algebra uses PyTorch’s cuBLAS FP16 kernels.
Both LayerNorms are implemented in Triton, keeping the reduction
over the channel dimension on‑GPU without any extra synchronisation.
The fused projection reduces three GEMMs (left/right + out_gate) to a
single GEMM, cutting memory traffic and kernel launch overhead.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (float32)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const float*  [total_rows, dim]
    y_ptr,          #       float*  [total_rows, dim]
    w_ptr,          # const float*  [dim]   (γ)
    b_ptr,          # const float*  [dim]   (β)
    total_rows,     # int (runtime)
    dim: tl.constexpr,          # static dimension size
    eps: tl.constexpr,          # static epsilon
    BLOCK_M: tl.constexpr,      # rows per program
    BLOCK_D: tl.constexpr,      # columns per inner loop
):
    """
    Row‑wise LayerNorm for a 2‑D tensor stored in row‑major order.
    Computes:
        y = (x - μ) / √(σ² + eps) * γ + β
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------- 1) mean & variance -------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                               # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                # row‑wise sum
        sum_sq  += tl.sum(vals * vals, axis=1)         # row‑wise sum‑of‑squares

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                         # 1 / √(var+eps)

    # ------------------- 2) write normalized values ---------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        # (x - mean) / sqrt(var + eps)
        norm = (vals - mean[:, None]) * rstd[:, None]

        # γ and β (broadcasted over rows)
        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)

        out = norm * w[None, :] + b[None, :]
        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out, mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Triton‑enabled TriMul (forward only)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

        * input_tensor : torch.Tensor, shape [B, N, N, C] (float32, CUDA)
        * mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float, CUDA)
        * weights_dict : mapping str → torch.Tensor (all float32, CUDA)
        * config_dict  : contains at least
                - "dim"        : C
                - "hidden_dim": H
                - "nomask"    : bool (True → ignore mask)

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32, CUDA)
    """
    # --------------------------------------------------------------
    # 0️⃣ Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]               # = C
    hidden_dim = config["hidden_dim"] # = H
    eps = 1e-5
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32) – flatten to 2‑D and run Triton
    # --------------------------------------------------------------
    x = input_tensor.contiguous()
    B, N, _, _ = x.shape                # [B, N, N, C]
    total_rows = B * N * N               # rows for the flattened view

    x_flat = x.view(total_rows, dim)     # [total_rows, C]
    x_norm_flat = torch.empty_like(x_flat)

    # Tunable block sizes – work well for typical C∈{128,384}
    BLOCK_M = 128
    BLOCK_D = 32
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm_flat,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()

    # Back to 4‑D layout (still FP32)
    x_norm = x_norm_flat.view(B, N, N, dim)

    # --------------------------------------------------------------
    # 2️⃣ Cast to FP16 and perform a **single** fused projection that
    #    produces left/right projections, gates and the output gate.
    # --------------------------------------------------------------
    # flatten to [total_rows, dim] and cast to half
    x_norm_h = x_norm.view(total_rows, dim).to(torch.float16)

    # Concatenate the five weight matrices (out_features, in_features)
    #   0 … H‑1               → left_proj
    #   H … 2H‑1              → left_gate
    #   2H … 3H‑1             → right_proj
    #   3H … 4H‑1             → right_gate
    #   4H … 5H‑1             → out_gate
    w_combined = torch.cat(
        [
            weights["left_proj.weight"],   # (H, C)
            weights["left_gate.weight"],   # (H, C)
            weights["right_proj.weight"],  # (H, C)
            weights["right_gate.weight"],  # (H, C)
            weights["out_gate.weight"],    # (H, C)
        ],
        dim=0,
    ).to(torch.float16)                     # (5*H, C)

    # One fused GEMM: shape [total_rows, 5*H]
    combined = F.linear(x_norm_h, w_combined)   # FP16

    # Split the concatenated result
    left_proj_h       = combined[:, :hidden_dim]                     # (T, H)
    left_gate_pre_h   = combined[:, hidden_dim:2 * hidden_dim]       # (T, H)
    right_proj_h      = combined[:, 2 * hidden_dim:3 * hidden_dim]   # (T, H)
    right_gate_pre_h  = combined[:, 3 * hidden_dim:4 * hidden_dim]   # (T, H)
    out_gate_pre_h    = combined[:, 4 * hidden_dim:]                 # (T, H)

    # --------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over hidden dimension)
    # --------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] → flatten → [total_rows, 1]
        mask_h = mask_tensor.view(total_rows, 1).to(torch.float16)
        left_proj_h  = left_proj_h * mask_h
        right_proj_h = right_proj_h * mask_h

    # --------------------------------------------------------------
    # 4️⃣ Sigmoid gates and element‑wise gating
    # --------------------------------------------------------------
    left_gate_h  = torch.sigmoid(left_gate_pre_h)    # (T, H)
    right_gate_h = torch.sigmoid(right_gate_pre_h)   # (T, H)
    out_gate_h   = torch.sigmoid(out_gate_pre_h)     # (T, H)

    left_h  = left_proj_h * left_gate_h    # (T, H)
    right_h = right_proj_h * right_gate_h  # (T, H)

    # --------------------------------------------------------------
    # 5️⃣ Tri‑multiplicative update = batched matmul over the “k” axis
    # --------------------------------------------------------------
    # Reshape to 4‑D tensors: [B, N, N, H]
    left_h_4d  = left_h.view(B, N, N, hidden_dim)
    right_h_4d = right_h.view(B, N, N, hidden_dim)

    # Bring hidden dim to the batch axis (B·H, N, N)
    # left  : (b,i,k,h) → (b·h, i, k)
    # right : (b,j,k,h) → (b·h, j, k) – we need its transpose on the last two axes
    left_mat  = left_h_4d.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right_h_4d.permute(0, 3, 1, 2).view(B * hidden_dim, N, N)   # non‑contig ok

    # Batched GEMM: out = left @ rightᵀ  (shape [B·H, N, N])
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Restore 4‑D layout [B, N, N, H] (still FP16)
    out_h = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm over the hidden dimension (FP32)
    # --------------------------------------------------------------
    out_f32 = out_h.to(torch.float32)                       # cast for LN
    total_rows_h = B * N * N
    out_flat = out_f32.view(total_rows_h, hidden_dim)
    out_norm_flat = torch.empty_like(out_flat)

    grid2 = (triton.cdiv(total_rows_h, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm_flat,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows_h,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()

    out_norm = out_norm_flat.view(B, N, N, hidden_dim)   # FP32

    # --------------------------------------------------------------
    # 7️⃣ Apply output gate and final projection back to C channels
    # --------------------------------------------------------------
    out_gate_f32 = out_gate_h.view(B, N, N, hidden_dim).to(torch.float32)
    out_norm = out_norm * out_gate_f32                 # element‑wise

    # Final linear (H → C) in FP16 for speed, then cast back to FP32
    out_norm_h = out_norm.to(torch.float16)
    final_h = F.linear(
        out_norm_h.view(total_rows_h, hidden_dim),
        weights["to_out.weight"].to(torch.float16)
    )   # (total_rows_h, C)  FP16
    final = final_h.view(B, N, N, dim).to(torch.float32)

    return final
