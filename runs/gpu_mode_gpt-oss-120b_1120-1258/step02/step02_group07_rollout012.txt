@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_bl, stride_bi, stride_bk, stride_bd,
    stride_rl, stride_ri, stride_rk, stride_rd,
    stride_ol, stride_oi, stride_oj, stride_od,
    MASK_PTR,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(axis=0)  # combine batch * tile_i * tile_j into 1D maybe
    # Using 2D grid: pid_m = program id for i tile, pid_n for j tile.
    # But we also need to handle batch and hidden dimension.
    # We'll use a 3D grid: program_id(0) for batch_hidden (b*H + d), program_id(1) for block_i, program_id(2) for block_j
    # In Triton, grid can be tuple of ints for each axis.

    pid_batch = tl.program_id(axis=0)
    pid_i = tl.program_id(axis=1)
    pid_j = tl.program_id(axis=2)

    # compute block coordinates
    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # compute batch and hidden indices
    b = pid_batch // H
    d = pid_batch % H

    # offsets for left and right base for this (b,d)
    left_base = left_ptr + b * stride_bl + d * stride_bd
    right_base = right_ptr + b * stride_rl + d * stride_rd

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # left tile: rows i, cols k_offset
        left_ptrs = left_base + i[:, None] * stride_bi + k_offset[None, :] * stride_bk
        left = tl.load(left_ptrs, mask=(i[:, None] < N) & (k_offset[None, :] < N), other=0.0).to(tl.float32)

        # right tile: rows j, cols k_offset (but note we need right[j,k]) - same stride dims.
        right_ptrs = right_base + j[:, None] * stride_ri + k_offset[None, :] * stride_rk
        right = tl.load(right_ptrs, mask=(j[:, None] < N) & (k_offset[None, :] < N), other=0.0).to(tl.float32)

        # optional mask: if mask provided, we apply mask on left and right before multiplication.
        # mask shape [B,N,N]. This mask corresponds to (b,i,k) and (b,j,k). We can load mask similarly.
        if MASK_PTR:
            # left mask: mask[b,i,k]
            mask_left_ptrs = mask_ptr + b * stride_mask_b + i[:, None] * stride_mask_i + k_offset[None, :] * stride_mask_k
            mask_left = tl.load(mask_left_ptrs, mask=(i[:, None] < N) & (k_offset[None, :] < N), other=0.0).to(tl.float32)  # broadcast over d

            # right mask: mask[b,j,k]
            mask_right_ptrs = mask_ptr + b * stride_mask_b + j[:, None] * stride_mask_i + k_offset[None, :] * stride_mask_k
            mask_right = tl.load(mask_right_ptrs, mask=(j[:, None] < N) & (k_offset[None, :] < N), other=0.0).to(tl.float32)

            left = left * mask_left
            right = right * mask_right

        # accumulate
        acc += tl.dot(left, right)  # shape (BLOCK_M, BLOCK_N)

    # store acc to out
    out_ptrs = out_ptr + b * stride_ol + d * stride_od + i[:, None] * stride_oi + j[None, :] * stride_oj
    tl.store(out_ptrs, acc.to(tl.float32), mask=(i[:, None] < N) & (j[None, :] < N))

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def trimul_outgoing_kernel(left_ptr, right_ptr, out_ptr,
                           B, N, H,
                           stride_bl, stride_bi, stride_bk, stride_bd,
                           stride_rl, stride_ri, stride_rk, stride_rd,
                           stride_ol, stride_oi, stride_oj, stride_od,
                           BLOCK_M: tl.constexpr,
                           BLOCK_N: tl.constexpr,
                           BLOCK_K: tl.constexpr):
    pid_batch = tl.program_id(axis=0)
    pid_m = tl.program_id(axis=1)
    pid_n = tl.program_id(axis=2)

    # derive batch and hidden dimension
    b = pid_batch // H
    d = pid_batch % H

    # offsets for this batch/hid
    left_batch = left_ptr + b * stride_bl + d * stride_bd
    right_batch = right_ptr + b * stride_rl + d * stride_rd
    out_batch = out_ptr + b * stride_ol + d * stride_od

    # create block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # create mask for bounds
    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)

        mask_k = offs_k < N

        # load left tile: shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_batch + offs_m[:, None] * stride_bi + offs_k[None, :] * stride_bk
        left = tl.load(left_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # load right tile (transposed): shape (BLOCK_N, BLOCK_K)
        right_ptrs = right_batch + offs_n[:, None] * stride_ri + offs_k[None, :] * stride_rk
        right = tl.load(right_ptrs, mask=mask_n[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # accumulate
        acc += tl.dot(left, right)

    # write back accumulator
    out_ptrs = out_batch + offs_m[:, None] * stride_oi + offs_n[None, :] * stride_oj
    tl.store(out_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])
```

"""
TriMul (outgoing) – Triton‑accelerated contraction.

Algorithm
---------
1. Layer‑norm the input across the last channel (dim).
2. Fuse the five linear projections (left_proj, right_proj,
   left_gate, right_gate, out_gate) into a single GEMM:
        Y = X_flat @ W_fused.T   (X_flat = [B·N·N, dim]),
   where W_fused = concat([W_left, W_right,
                           W_left_gate, W_right_gate, W_out_gate]).
   The result is reshaped to (B, N, N, 5·hidden_dim) and split.
3. Apply the optional pair‑wise mask and the three sigmoid gates.
   The gated tensors ``left`` and ``right`` remain in FP16.
4. Compute the heavy N³ contraction with a custom Triton kernel:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   The kernel works directly on the original (B,N,N,hidden) layout,
   avoiding the costly ``permute``‑``contiguous`` copies required by
   a naïve ``torch.bmm`` implementation.
5. Layer‑norm the result over the hidden dimension, apply the output
   gate and a final linear projection back to ``dim``.
6. Return a FP32 tensor as required.

The kernel is launched with a 3‑D grid:
   (B·hidden_dim, ceil(N/BLOCK_M), ceil(N/BLOCK_N)).
BLOCK_M = BLOCK_N = 64 and BLOCK_K = 64 give good occupancy on
H100 while keeping shared‑memory usage well below the limit.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel for the N³ contraction (outgoing direction)
# ----------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    left_ptr,               # const fp16*  [B, N, N, H]
    right_ptr,              # const fp16*  [B, N, N, H]
    out_ptr,                #       fp32*  [B, N, N, H]
    B: tl.int32,
    N: tl.int32,
    H: tl.int32,
    stride_bl, stride_bi, stride_bk, stride_bd,   # left strides
    stride_rl, stride_ri, stride_rk, stride_rd,   # right strides
    stride_ol, stride_oi, stride_oj, stride_od,   # out strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # --------------------------------------------------------------
    # Grid layout
    #   axis 0 : batch * hidden  (pid_bh)
    #   axis 1 : i‑tile          (pid_m)
    #   axis 2 : j‑tile          (pid_n)
    # --------------------------------------------------------------
    pid_bh = tl.program_id(0)
    pid_m  = tl.program_id(1)
    pid_n  = tl.program_id(2)

    # decode batch index (b) and hidden index (d)
    b = pid_bh // H
    d = pid_bh % H

    # base pointers for this (b,d) pair
    left_base  = left_ptr  + b * stride_bl + d * stride_bd
    right_base = right_ptr + b * stride_rl + d * stride_rd
    out_base   = out_ptr   + b * stride_ol + d * stride_od

    # ------------------------------------------------------------------
    # Tile indices in the i (row) and j (col) dimensions
    # ------------------------------------------------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over the reduction axis k
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_off = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_off < N

        # -------- left tile:  [i, k] ---------------------------------
        left_ptrs = left_base + offs_i[:, None] * stride_bi + k_off[None, :] * stride_bk
        left = tl.load(left_ptrs,
                       mask=(mask_i[:, None] & mask_k[None, :]),
                       other=0.0).to(tl.float32)

        # -------- right tile: [j, k] ---------------------------------
        right_ptrs = right_base + offs_j[:, None] * stride_ri + k_off[None, :] * stride_rk
        right = tl.load(right_ptrs,
                        mask=(mask_j[:, None] & mask_k[None, :]),
                        other=0.0).to(tl.float32)

        # -------- accumulate -----------------------------------------
        acc += tl.dot(left, right)          # (BLOCK_M, BLOCK_N) in FP32

    # ------------------------------------------------------------------
    # Write back the result
    # ------------------------------------------------------------------
    out_ptrs = out_base + offs_i[:, None] * stride_oi + offs_j[None, :] * stride_oj
    tl.store(out_ptrs, acc,
             mask=(mask_i[:, None] & mask_j[None, :]))


# ----------------------------------------------------------------------
# Entry point used by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator with a Triton‑based
    N³ contraction.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor[B, N, N, dim] (float32)
        - mask         : torch.Tensor[B, N, N] (bool/float) or None
        - weights      : dict of parameters (float32)
        - config       : dict containing "dim", "hidden_dim", "nomask"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    torch.backends.cuda.matmul.allow_tf32 = True   # H100 excels with TF32
    torch.backends.cudnn.allow_tf32 = True

    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Extract weights (already on the correct device)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    left_proj_weight   = weights["left_proj.weight"]
    right_proj_weight  = weights["right_proj.weight"]
    left_gate_weight   = weights["left_gate.weight"]
    right_gate_weight  = weights["right_gate.weight"]
    out_gate_weight    = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # All heavy arithmetic in mixed precision (FP16), final output FP32
    # ------------------------------------------------------------------
    with torch.no_grad(), torch.cuda.amp.autocast():
        # ---- LayerNorm over the model dimension -----------------------
        x = F.layer_norm(input_tensor,
                         (dim,),
                         weight=norm_weight,
                         bias=norm_bias)

        B, N, _, _ = x.shape

        # ---- Fuse the five linear projections -------------------------
        # Flatten to (B*N*N, dim)
        x_flat = x.reshape(-1, dim)                     # [B·N·N, dim]

        # Concatenate weight matrices (shape (5*hidden_dim, dim))
        fused_weight = torch.cat(
            [left_proj_weight,
             right_proj_weight,
             left_gate_weight,
             right_gate_weight,
             out_gate_weight],
            dim=0,
        )                                                  # fp32 -> autocast ⇒ fp16

        # One large GEMM
        fused_out = F.linear(x_flat, fused_weight)        # [B·N·N, 5*hidden_dim]
        fused_out = fused_out.view(B, N, N, 5 * hidden_dim)

        # ---- Split and apply mask / sigmoid gates --------------------
        left      = fused_out[..., 0:hidden_dim]
        right     = fused_out[..., hidden_dim:2 * hidden_dim]
        left_gate = fused_out[..., 2 * hidden_dim:3 * hidden_dim]
        right_gate= fused_out[..., 3 * hidden_dim:4 * hidden_dim]
        out_gate  = fused_out[..., 4 * hidden_dim:5 * hidden_dim]

        if not nomask and mask is not None:
            mask_f = mask.to(x.dtype).unsqueeze(-1)      # [B, N, N, 1]
            left = left * mask_f
            right = right * mask_f

        left_gate  = torch.sigmoid(left_gate)
        right_gate = torch.sigmoid(right_gate)
        out_gate   = torch.sigmoid(out_gate)

        # In‑place gating to avoid extra allocations
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # Triton kernel: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # Strides (in elements) for each tensor
    stride_bl, stride_bi, stride_bk, stride_bd = left.stride()
    stride_rl, stride_ri, stride_rk, stride_rd = right.stride()
    stride_ol, stride_oi, stride_oj, stride_od = out.stride()

    # Tile sizes – chosen to keep shared memory < 64 KB on H100
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (B * hidden_dim,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    trimul_outgoing_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=hidden_dim,
        stride_bl=stride_bl,
        stride_bi=stride_bi,
        stride_bk=stride_bk,
        stride_bd=stride_bd,
        stride_rl=stride_rl,
        stride_ri=stride_ri,
        stride_rk=stride_rk,
        stride_rd=stride_rd,
        stride_ol=stride_ol,
        stride_oi=stride_oi,
        stride_oj=stride_oj,
        stride_od=stride_od,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # Remaining cheap post‑processing steps (FP32)
    # ------------------------------------------------------------------
    out = F.layer_norm(out,
                       (hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias)

    # Apply the output gate (out_gate is still FP16)
    out = out * out_gate.to(out.dtype)

    # Final linear projection back to model dimension
    out = F.linear(out, to_out_weight)   # shape [B, N, N, dim]

    return out.to(torch.float32)
