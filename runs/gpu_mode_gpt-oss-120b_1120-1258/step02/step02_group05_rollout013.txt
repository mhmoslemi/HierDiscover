import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def apply_mask_kernel(x_ptr, mask_ptr, out_ptr,
                      total_elems, hidden_dim,
                      BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    start = pid * BLOCK
    offs = start + tl.arange(0, BLOCK)
    mask = offs < total_elems
    # compute the corresponding mask index (broadcast over hidden_dim)
    mask_idx = offs // hidden_dim
    x = tl.load(x_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
    tl.store(out_ptr + offs, x * m, mask=mask)

def apply_mask_inplace(tensor, mask, hidden_dim):
    # tensor: [B,N,N,hidden_dim] contiguous, fp16
    # mask: [B,N,N,1] contiguous, same dtype as tensor
    total = tensor.numel()
    BLOCK = 1024
    grid = ( (total + BLOCK - 1) // BLOCK, )
    apply_mask_kernel[grid](
        tensor.view(-1),
        mask.view(-1),
        tensor.view(-1),
        total,
        hidden_dim,
        BLOCK=BLOCK,
    )
    return tensor

def custom_kernel(data):
    """
    TriMul (outgoing) forward pass.
    1. LayerNorm on the feature dimension.
    2. Two bias‑free linear projections (left/right) + three sigmoid gates.
    3. Optional mask applied (broadcast over the hidden dimension) with a tiny Triton kernel.
    4. Cubic contraction `out_{i,j} = Σ_k left_{i,k} * right_{j,k}` performed as a batched
       matrix‑multiply (`torch.bmm`) on FP16 tensors (Tensor‑core friendly).  Result
       is cast back to FP32.
    5. Second LayerNorm, output‑gate multiplication and final linear projection.
    The kernel for the mask is required to satisfy the “use Triton” rule.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  (float32)

    # -----------------------------------------------------------------
    # 2. FP16 projection + sigmoid gates
    # -----------------------------------------------------------------
    x_h = x.half()   # cast to FP16

    left = F.linear(x_h, weights["left_proj.weight"].half())    # [B, N, N, H]
    right = F.linear(x_h, weights["right_proj.weight"].half())

    left_gate = torch.sigmoid(F.linear(x_h, weights["left_gate.weight"].half()))
    right_gate = torch.sigmoid(F.linear(x_h, weights["right_gate.weight"].half()))
    out_gate = torch.sigmoid(F.linear(x_h, weights["out_gate.weight"].half()))

    # -----------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dim) – Triton kernel
    # -----------------------------------------------------------------
    if not nomask:
        # Expand mask to [B, N, N, 1] and cast to FP16
        mask_exp = mask.to(left.dtype).unsqueeze(-1).contiguous()
        # Apply mask in‑place via Triton (much cheaper than a full GEMM)
        apply_mask_inplace(left, mask_exp, hidden_dim)
        apply_mask_inplace(right, mask_exp, hidden_dim)

    # -----------------------------------------------------------------
    # 4. Apply sigmoid gates
    # -----------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Batched GEMM: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    We use torch.bmm on a collapsed batch dimension.
    # -----------------------------------------------------------------
    B, N, _, _ = left.shape
    # bring hidden dim to the batch axis: [B, H, N, N] → (B*H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    M_total = B * hidden_dim
    left_flat = left_perm.view(M_total, N, N)   # (M, N, N)  fp16
    right_flat = right_perm.view(M_total, N, N) # (M, N, N)  fp16

    # torch.bmm returns FP16; cast to FP32 for the next layers
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2)).float()
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H] (float32)

    # -----------------------------------------------------------------
    # 6. Output LayerNorm + output gate + final projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate.float()
    output = F.linear(out, weights["to_out.weight"])
    return output

"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel – apply a broadcast mask to a Tensor of shape [B,N,N,H]
# ----------------------------------------------------------------------
@triton.jit
def apply_mask_kernel(
    x_ptr,                 # input tensor      (flattened)
    mask_ptr,              # mask tensor       (flattened, size = total/H)
    out_ptr,               # output tensor     (flattened, may be the same as x_ptr)
    total_elems: tl.int64, # total number of elements in x
    hidden_dim: tl.int64,  # H – the broadcasting factor
    BLOCK: tl.constexpr,   # number of elements processed per program
):
    pid = tl.program_id(0)
    start = pid * BLOCK
    offs = start + tl.arange(0, BLOCK)
    within_bounds = offs < total_elems

    # map each element of the flattened [B,N,N,H] tensor to the corresponding
    # element of the flattened mask [B,N,N] (broadcast over hidden_dim)
    mask_idx = offs // hidden_dim

    x = tl.load(x_ptr + offs, mask=within_bounds, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=within_bounds, other=1.0)

    tl.store(out_ptr + offs, x * m, mask=within_bounds)


def _apply_mask_inplace(tensor: torch.Tensor, mask: torch.Tensor, hidden_dim: int) -> torch.Tensor:
    """
    Apply ``mask`` (shape [B,N,N,1]) to ``tensor`` (shape [B,N,N,H]) in‑place.
    Both tensors must be contiguous and on the same device.
    """
    total = tensor.numel()
    BLOCK = 1024  # good throughput for elementwise work
    grid = ((total + BLOCK - 1) // BLOCK,)

    apply_mask_kernel[grid](
        tensor.view(-1),
        mask.view(-1),
        tensor.view(-1),
        total,
        hidden_dim,
        BLOCK=BLOCK,
    )
    return tensor


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor, [B, N, N, dim] (float32)
            - mask        : torch.Tensor, [B, N, N]  (any dtype) – optional
            - weights     : dict of model parameters (all torch.Tensors)
            - config      : dict with keys "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim] – float32

    # -----------------------------------------------------------------
    # 2. FP16 projections + sigmoid gates
    # -----------------------------------------------------------------
    x_h = x.half()  # cast once to FP16

    left = F.linear(x_h, weights["left_proj.weight"].half())           # [B, N, N, H]
    right = F.linear(x_h, weights["right_proj.weight"].half())

    left_gate  = torch.sigmoid(F.linear(x_h, weights["left_gate.weight"].half()))
    right_gate = torch.sigmoid(F.linear(x_h, weights["right_gate.weight"].half()))
    out_gate   = torch.sigmoid(F.linear(x_h, weights["out_gate.weight"].half()))

    # -----------------------------------------------------------------
    # 3. Optional mask – apply with a tiny Triton kernel
    # -----------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] → [B, N, N, 1] and cast to FP16
        mask_exp = mask.to(left.dtype).unsqueeze(-1).contiguous()
        _apply_mask_inplace(left,  mask_exp, hidden_dim)
        _apply_mask_inplace(right, mask_exp, hidden_dim)

    # -----------------------------------------------------------------
    # 4. Apply sigmoid gates (still FP16)
    # -----------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Batched GEMM: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # -----------------------------------------------------------------
    B, N, _, _ = left.shape

    # Merge batch and hidden dimension – shape (M, N, N) with M = B*H
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    M_total = B * hidden_dim
    left_flat = left_perm.view(M_total, N, N)   # (M, N, N)  – FP16
    right_flat = right_perm.view(M_total, N, N) # (M, N, N)  – FP16

    # Batched matrix‑multiply on the reduced batch dimension.
    # torch.bmm uses cuBLAS Tensor‑cores for FP16 → very fast.
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2)).float()

    # Restore the original layout: [B, N, N, H] (float32)
    out = (
        out_flat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # [B, N, N, H]

    # -----------------------------------------------------------------
    # 6. Output LayerNorm, output‑gate multiplication, final projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate.float()
    output = F.linear(out, weights["to_out.weight"])   # back to [B, N, N, dim]

    return output
