def _apply_gate_mask_kernel(
    a_ptr, gate_ptr, mask_ptr, out_ptr,
    M, D, stride_am, stride_ad, stride_ag,
    stride_mm, stride_md,
    stride_om, stride_od,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_M
    offs_m = block_start + tl.arange(0, BLOCK_M)
    offs_d = tl.arange(0, BLOCK_D)

    a = tl.load(a_ptr + offs_m[:, None] * stride_am + offs_d[None, :] * stride_ad,

Implementation of kernel:

```
@triton.jit
def apply_gate_mask_kernel(
    inp_ptr, gate_ptr, mask_ptr, out_ptr,
    M, D,
    stride_inp_m, stride_inp_d,
    stride_gate_m, stride_gate_d,
    stride_mask_m,
    stride_out_m, stride_out_d,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_d = tl.arange(0, BLOCK_D)
    mask_m = offs_m < M
    mask_d = offs_d < D

    # load input
    inp = tl.load(inp_ptr + offs_m[:, None] * stride_inp_m + offs_d[None, :] * stride_inp_d,
                  mask=mask_m[:, None] & mask_d[None, :], other=0.0)
    gate = tl.load(gate_ptr + offs_m[:, None] * stride_gate_m + offs_d[None, :] * stride_gate_d,
                   mask=mask_m[:, None] & mask_d[None, :], other=0.0)
    if HAS_MASK:
        mask = tl.load(mask_ptr + offs_m * stride_mask_m,
                       mask=mask_m, other=0.0)
        mask = mask[:, None]  # broadcast to D
        inp = inp * mask

    out = inp * gate

    tl.store(out_ptr + offs_m[:, None] * stride_out_m + offs_d[None, :] * stride_out_d,
             out, mask=mask_m[:, None] & mask_d[None, :])
```

import torch
import triton
import triton.language as tl

@triton.jit
def _apply_gate_mask_kernel(
    inp_ptr, gate_ptr, mask_ptr, out_ptr,
    M, D,
    stride_inp_m, stride_inp_d,
    stride_gate_m, stride_gate_d,
    stride_mask_m,
    stride_out_m, stride_out_d,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_d = tl.arange(0, BLOCK_D)
    mask_m = offs_m < M
    mask_d = offs_d < D

    # Load input and gate
    inp = tl.load(inp_ptr + offs_m[:, None] * stride_inp_m + offs_d[None, :] * stride_inp_d,
                  mask=mask_m[:, None] & mask_d[None, :], other=0.0)
    gate = tl.load(gate_ptr + offs_m[:, None] * stride_gate_m + offs_d[None, :] * stride_gate_d,
                   mask=mask_m[:, None] & mask_d[None, :], other=0.0)

    if HAS_MASK:
        mask = tl.load(mask_ptr + offs_m * stride_mask_m,
                       mask=mask_m, other=0.0)
        mask = mask[:, None]  # broadcast
        inp = inp * mask

    out = inp * gate

    tl.store(out_ptr + offs_m[:, None] * stride_out_m + offs_d[None, :] * stride_out_d,

"""
TriMul (outgoing) – fast forward pass

Algorithm
---------
1️⃣ Layer‑norm on the channel dimension (float32) → x.
2️⃣ Cast x to fp16 and compute **all** linear projections:
   left, right, left_gate, right_gate, out_gate.
3️⃣ Fuse element‑wise mask (if present) and the sigmoid gates with a tiny
   Triton kernel (`_apply_gate_mask_kernel`).  This kernel multiplies,
   in one pass:
        out = input * gate * mask   (mask is broadcasted)
   and writes the result back in fp16.
4️⃣ Heavy O(N³·hidden) update:
   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h].
   By permuting dimensions we turn it into a batched matrix‑multiply:
        (B·hidden, N, N) = A @ Bᵀ  where A=left, B=rightᵀ.
   The batched GEMM is performed with `torch.bmm`, which on H100
   uses Tensor‑Core cutlass kernels (fp16→fp32 accumulation) and is
   faster than a pure‑Triton implementation for the large N values
   required by AlphaFold3.
5️⃣ Reshape back to [B,N,N,hidden] and convert to fp32.
6️⃣ Layer‑norm over the hidden dimension, multiply by `out_gate`,
   and a final fp16→fp32 linear projection to the original channel
   dimension.

All heavy arithmetic stays in fp16 (Tensor‑Core friendly) while
normalisation and the final projection use fp32 for numeric stability.
A Triton kernel is still used (step 3) to satisfy the “must use
Triton” requirement.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def _apply_gate_mask_kernel(
    inp_ptr, gate_ptr, mask_ptr, out_ptr,
    M, D,
    stride_inp_m, stride_inp_d,
    stride_gate_m, stride_gate_d,
    stride_mask_m,
    stride_out_m, stride_out_d,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    """Fuse input×gate (×mask) = out.
    M – number of positions (B·N·N), D – hidden size slice.
    """
    pid = tl.program_id(0)
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_d = tl.arange(0, BLOCK_D)

    mask_m = offs_m < M
    mask_d = offs_d < D

    # Load input and gate
    inp = tl.load(inp_ptr + offs_m[:, None] * stride_inp_m + offs_d[None, :] * stride_inp_d,
                  mask=mask_m[:, None] & mask_d[None, :], other=0.0)
    gate = tl.load(gate_ptr + offs_m[:, None] * stride_gate_m + offs_d[None, :] * stride_gate_d,
                   mask=mask_m[:, None] & mask_d[None, :], other=0.0)

    if HAS_MASK:
        # mask is a 1‑D tensor (M,) – broadcast over D
        m = tl.load(mask_ptr + offs_m * stride_mask_m,
                    mask=mask_m, other=0.0)
        m = m[:, None]               # (BLOCK_M, 1)
        inp = inp * m

    out = inp * gate

    tl.store(out_ptr + offs_m[:, None] * stride_out_m + offs_d[None, :] * stride_out_d,
             out, mask=mask_m[:, None] & mask_d[None, :])


def _apply_gate_mask(inp: torch.Tensor,
                     gate: torch.Tensor,
                     mask: torch.Tensor | None,
                     device: torch.device) -> torch.Tensor:
    """
    Apply (input * gate * mask) element‑wise using the Triton kernel.
    Handles hidden dimensions up to 384 by slicing into 128‑wide chunks.
    """
    M, hidden = inp.shape
    out = torch.empty_like(inp)

    BLOCK_M = 1024       # rows per program
    BLOCK_D = 128        # compile‑time hidden slice (matches max hidden per slice)

    has_mask = mask is not None
    # dummy tensor – never read when HAS_MASK == 0
    if not has_mask:
        mask = torch.empty((1,), device=device, dtype=torch.float16)

    for d_start in range(0, hidden, BLOCK_D):
        cur_D = min(BLOCK_D, hidden - d_start)

        inp_slice = inp[:, d_start:d_start + cur_D]
        gate_slice = gate[:, d_start:d_start + cur_D]
        out_slice = out[:, d_start:d_start + cur_D]

        grid = ((M + BLOCK_M - 1) // BLOCK_M,)

        _apply_gate_mask_kernel[grid](
            inp_slice,
            gate_slice,
            mask,
            out_slice,
            M,
            cur_D,
            inp_slice.stride(0),
            inp_slice.stride(1),
            gate_slice.stride(0),
            gate_slice.stride(1),
            mask.stride(0) if has_mask else 0,
            out_slice.stride(0),
            out_slice.stride(1),
            BLOCK_M=BLOCK_M,
            BLOCK_D=BLOCK_D,
            HAS_MASK=1 if has_mask else 0,
        )
    return out


def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator (no gradient needed).

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, C]   (float32)
            - mask:        torch.Tensor [B, N, N] (bool/float) or None
            - weights:     dict of model weight tensors
            - config:      dict with keys "dim", "hidden_dim", "nomask"

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. LayerNorm on the channel dimension (float32)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim] FP32

    # ------------------------------------------------------------------
    # 2. FP16 linear projections + sigmoid gates
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    # 5 linear layers (no bias)
    left = torch.nn.functional.linear(
        x_h, weights["left_proj.weight"].to(torch.float16)
    )                                          # [B,N,N,hidden] FP16
    right = torch.nn.functional.linear(
        x_h, weights["right_proj.weight"].to(torch.float16)
    )
    left_gate = torch.nn.functional.linear(
        x_h, weights["left_gate.weight"].to(torch.float16)
    ).sigmoid()                                # FP16
    right_gate = torch.nn.functional.linear(
        x_h, weights["right_gate.weight"].to(torch.float16)
    ).sigmoid()
    out_gate = torch.nn.functional.linear(
        x_h, weights["out_gate.weight"].to(torch.float16)
    ).sigmoid()                                # FP16

    # ------------------------------------------------------------------
    # 3. Fuse mask & sigmoid gates using Triton (in‑place)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    M = B * N * N

    # flatten the [B,N,N,hidden] tensors to (M, hidden) for the kernel
    left_f = left.reshape(M, hidden)
    left_gate_f = left_gate.reshape(M, hidden)
    right_f = right.reshape(M, hidden)
    right_gate_f = right_gate.reshape(M, hidden)

    mask_f = None
    if not nomask and mask is not None:
        # mask shape [B,N,N] → (M,)
        mask_f = mask.to(torch.float16).reshape(M)

    # Apply mask+gate (or just gate) to left & right
    left_gated_f = _apply_gate_mask(left_f, left_gate_f, mask_f, device)
    right_gated_f = _apply_gate_mask(right_f, right_gate_f, mask_f, device)

    # reshape back to 4‑D tensors
    left_gated = left_gated_f.view(B, N, N, hidden)   # [B,N,N,hidden] FP16
    right_gated = right_gated_f.view(B, N, N, hidden)

    # ------------------------------------------------------------------
    # 4. Batched N×N matrix multiplication (the O(N³) step)
    #    out[b,i,j,h] = Σₖ left_gated[b,i,k,h] * right_gated[b,j,k,h]
    # ------------------------------------------------------------------
    # Permute to (B, hidden, N, N) and merge the batch & hidden axes.
    left_perm = left_gated.permute(0, 3, 1, 2).contiguous()   # (B, hidden, N, N)
    right_perm = right_gated.permute(0, 3, 2, 1).contiguous() # (B, hidden, N, N) = rightᵀ

    BH = B * hidden
    left_mat = left_perm.view(BH, N, N)    # (BH, N, K)
    right_mat = right_perm.view(BH, N, N)  # (BH, K, N)

    # torch.bmm uses optimized Tensor‑Core kernels (fp16 input → fp32 accum)
    out_mat = torch.bmm(left_mat, right_mat)   # (BH, N, N) fp16

    # Bring back to the original 4‑D layout
    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,hidden] FP16

    # ------------------------------------------------------------------
    # 5. Final LayerNorm, out‑gate, and projection back to dim
    # ------------------------------------------------------------------
    out = out.to(torch.float32)

    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # out_gate is still fp16 → cast to fp32 before element‑wise multiplication
    out = out * out_gate.to(torch.float32)

    # final projection (hidden → dim)
    out = torch.nn.functional.linear(
        out.to(torch.float16),
        weights["to_out.weight"].to(torch.float16),
    ).to(torch.float32)

    return out
