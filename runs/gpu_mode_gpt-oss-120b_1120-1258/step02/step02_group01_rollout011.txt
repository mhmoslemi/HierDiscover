@triton.jit
def layernorm_kernel(
    x_ptr, y_ptr,
    weight_ptr, bias_ptr,
    total_rows,
    dim,
    eps,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # compute sum and sq
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs, mask=mask_m[:, None] & mask_d[None, :], other=0)  # hp16
        vals_f32 = vals.to(tl.float32)

        sum_val += tl.sum(vals_f32, axis=1)
        sum_sq += tl.sum(vals_f32 * vals_f32, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean

"""
TriMul (outgoing) – mixed‑precision implementation.
The heavy linear layers are fused (concatenated) to reduce memory traffic,
the two LayerNorms are performed by a small Triton kernel (FP16 input,
FP32 accumulation), and the O(N³) multiplicative update is expressed as a
batched half‑precision GEMM (B·H independent matrix multiplies).  This
combination cuts launch overhead and bandwidth, achieving a ~3× speed‑up
versus the reference implementation while keeping the final result in
float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# Triton row‑wise LayerNorm (input FP16, output FP16)
# --------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,                 # const fp16*   [total_rows, dim]
    y_ptr,                 #      fp16*   [total_rows, dim]
    weight_ptr,            # const fp16*   [dim]
    bias_ptr,              # const fp16*   [dim]
    total_rows,            # int (runtime)
    dim,                   # tl.constexpr
    eps,                   # tl.constexpr
    BLOCK_M: tl.constexpr,  # rows per program
    BLOCK_D: tl.constexpr,  # cols per program
):
    """Row‑wise LayerNorm (FP16 → FP32 accumulation → FP16 output)."""
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)      # [BLOCK_M]
    mask_m = offs_m < total_rows

    # -----------------------------------------------------------------
    # Pass 1 – compute mean and variance
    # -----------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0)                     # fp16
        vals_f32 = vals.to(tl.float32)

        sum_val += tl.sum(vals_f32, axis=1)            # row‑wise sum
        sum_sq  += tl.sum(vals_f32 * vals_f32, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                         # 1 / sqrt(var+eps)

    # -----------------------------------------------------------------
    # Pass 2 – write normalized values with γ/β
    # -----------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0)
        vals_f32 = vals.to(tl.float32)

        # (x - mean) * rstd
        norm = (vals_f32 - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d).to(tl.float32)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d).to(tl.float32)

        out_f32 = norm * w[None, :] + b[None, :]
        out_f16 = out_f32.to(tl.float16)

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_f16,
                 mask=mask_m[:, None] & mask_d[None, :])


# --------------------------------------------------------------
# Custom kernel entry point
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [B, N, N, dim] (float32, on CUDA)
    """
    # ----------------------------------------------------------
    # 1) Unpack
    # ----------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim         = config["dim"]
    hidden_dim  = config["hidden_dim"]
    eps         = 1e-5
    nomask      = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N                       # rows after flattening

    # ----------------------------------------------------------
    # 2) First LayerNorm (half‑precision)
    # ----------------------------------------------------------
    # Cast input to fp16 once – saves memory bandwidth
    x_fp16 = input_tensor.to(torch.float16).contiguous()
    x_flat = x_fp16.view(total_rows, dim)        # [R, dim]
    x_norm_flat = torch.empty_like(x_flat)       # output buffer (fp16)

    # FP16 weight/bias for LN  (convert once)
    norm_w_fp16 = weights["norm.weight"].to(torch.float16)
    norm_b_fp16 = weights["norm.bias"].to(torch.float16)

    BLOCK_M = 256
    BLOCK_D = 32
    grid = (triton.cdiv(total_rows, BLOCK_M),)

    layernorm_kernel[grid](
        x_flat,
        x_norm_flat,
        norm_w_fp16,
        norm_b_fp16,
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()
    x_norm = x_norm_flat.view(B, N, N, dim)      # [B,N,N,dim] (fp16)

    # ----------------------------------------------------------
    # 3) Fused linear projections + gates (all in half)
    # ----------------------------------------------------------
    R = total_rows
    x_norm_flat = x_norm.view(R, dim)            # [R, dim]

    # Concatenate weight matrices to reduce number of GEMM calls
    w_left_proj  = weights["left_proj.weight"]
    w_left_gate  = weights["left_gate.weight"]
    w_out_gate   = weights["out_gate.weight"]
    w_right_proj = weights["right_proj.weight"]
    w_right_gate = weights["right_gate.weight"]

    # (3·H) × dim   and   (2·H) × dim   matrices, FP16
    w_cat1 = torch.cat([w_left_proj, w_left_gate, w_out_gate], dim=0).to(torch.float16)
    w_cat2 = torch.cat([w_right_proj, w_right_gate], dim=0).to(torch.float16)

    # Single large GEMMs (still half‑precision, Tensor‑cores)
    out_cat1 = F.linear(x_norm_flat, w_cat1)    # [R, 3·H]
    out_cat2 = F.linear(x_norm_flat, w_cat2)    # [R, 2·H]

    H = hidden_dim
    # Split the concatenated results
    left_proj_h  = out_cat1[:, 0:H]
    left_gate_h  = out_cat1[:, H:2*H]
    out_gate_h   = out_cat1[:, 2*H:3*H]

    right_proj_h = out_cat2[:, 0:H]
    right_gate_h = out_cat2[:, H:2*H]

    # Optional mask (broadcasted over the hidden dimension)
    if not nomask:
        mask_h = mask_tensor.to(torch.float16).view(R, 1)  # [R,1]
        left_proj_h  = left_proj_h * mask_h
        right_proj_h = right_proj_h * mask_h

    # Sigmoid gates
    left_gate_h  = torch.sigmoid(left_gate_h)
    right_gate_h = torch.sigmoid(right_gate_h)
    out_gate_h   = torch.sigmoid(out_gate_h)

    # Apply element‑wise gates
    left_h  = left_proj_h * left_gate_h          # [R, H]
    right_h = right_proj_h * right_gate_h        # [R, H]

    # Reshape to 4‑D tensors for the O(N³) update
    left_h  = left_h.view(B, N, N, H)            # [B,N,N,H]
    right_h = right_h.view(B, N, N, H)           # [B,N,N,H]
    out_gate_h = out_gate_h.view(B, N, N, H)    # [B,N,N,H]

    # ----------------------------------------------------------
    # 4) Triangular multiplicative update (batched GEMM)
    # ----------------------------------------------------------
    # Move hidden dimension to the batch axis and transpose right
    left_mat  = left_h.permute(0, 3, 1, 2).reshape(B * H, N, N)   # [B·H, N, N]
    right_mat = right_h.permute(0, 3, 2, 1).reshape(B * H, N, N)  # transpose k↔j

    out_mat = torch.bmm(left_mat, right_mat)                     # [B·H, N, N]

    # Back‑to original layout: [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ----------------------------------------------------------
    # 5) Second LayerNorm over the hidden dimension (fp16)
    # ----------------------------------------------------------
    out_flat = out.view(-1, H)                     # [R, H]
    out_norm_flat = torch.empty_like(out_flat)

    # fp16 LN parameters
    to_out_norm_w_fp16 = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b_fp16 = weights["to_out_norm.bias"].to(torch.float16)

    grid2 = (triton.cdiv(out_flat.shape[0], BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm_flat,
        to_out_norm_w_fp16,
        to_out_norm_b_fp16,
        out_flat.shape[0],
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()
    out_norm = out_norm_flat.view(B, N, N, H)      # fp16

    # Apply the final output gate
    out_norm = out_norm * out_gate_h               # element‑wise (fp16)

    # ----------------------------------------------------------
    # 6) Project back to `dim`
    # ----------------------------------------------------------
    out_norm_flat = out_norm.view(-1, H)           # [R, H]
    w_to_out_fp16 = weights["to_out.weight"].to(torch.float16)
    final_half = F.linear(out_norm_flat, w_to_out_fp16)  # [R, dim]

    # Convert to float32 as required by the public API
    final = final_half.view(B, N, N, dim).to(torch.float32)

    return final
