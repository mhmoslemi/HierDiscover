def layernorm_f16_kernel(
    x_ptr,         # *float16
    w_ptr,         # *float16 weight
    b_ptr,         # *float16 bias
    total_rows,
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # Second pass: write normalized values in-place
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0).to(tl.float32)
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0).to(tl.float32)

        out = norm * w[None, :] + b[None, :]
        # cast back to fp16
        out_fp16 = out.to(tl.float16)
        tl.store(ptrs, out_fp16, mask=mask_m[:, None] & mask_d[None, :])

"""
custom_kernel – “outgoing” TriMul (AlphaFold‑3) forward pass.

Algorithm (half‑precision everywhere except the tiny reductions that stay in FP32):
1️⃣  LayerNorm over the channel dimension (dim) – Triton kernel that works
    in‑place on a FP16 tensor (statistics computed in FP32 for stability).
2️⃣  Cast the normalized tensor to FP16 and fuse the four linear layers
    (left_proj+left_gate, right_proj+right_gate) by concatenating the
    weight matrices.  This reduces from 6 linear layers to 4.
3️⃣  Apply the optional mask, then element‑wise multiply each projection
    with its sigmoid‑gate.
4️⃣  Triangular multiplicative update:
        out[i,j,d] = Σ_k left[i,k,d] · right[j,k,d]
    performed with a single ``torch.einsum`` which avoids the costly
    ``permute+contiguous`` copies required by a batched GEMM.
5️⃣  Second LayerNorm over the hidden dimension (again FP16, in‑place).
6️⃣  Apply the output‑gate (sigmoid) and a final FP16 linear projection
    back to ``dim``.  The result is cast back to FP32 before returning.

All heavy reductions (the two LayerNorms) are executed in Triton,
the dense linear algebra uses PyTorch’s highly‑optimised FP16 kernels.
The implementation supports every configuration described in the prompt
(and works with or without a mask)."""

    b_ptr,            # *float16  [dim]             (β)
    total_rows,       # int   (runtime)
    dim,              # int   (runtime)
    eps,              # float (runtime)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per inner loop
):
    """
    Row‑wise LayerNorm on a FP16 matrix.
    Reads a block of rows, computes mean/var in FP32,
    writes the normalized result back to the same buffer (in‑place).
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # Pass 1 – compute mean and variance (FP32 accumulation)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]   # linear offset
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                               # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                # row‑wise sum
        sum_sq  += tl.sum(vals * vals, axis=1)         # row‑wise sum of squares

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                         # 1/√(var+eps)

    # ------------------------------------------------------------------
    # Pass 2 – write normalized values (still in‑place)
    # ------------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        # (x - mean) / sqrt(var+eps)
        norm = (vals - mean[:, None]) * rstd[:, None]

        # γ (weight) and β (bias) – loaded as FP16 and cast to FP32
        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0).to(tl.float32)
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0).to(tl.float32)

        out = norm * w[None, :] + b[None, :]   # still FP32
        out_fp16 = out.to(tl.float16)          # cast back to half
        tl.store(ptrs, out_fp16, mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32, on CUDA)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim         = config["dim"]
    hidden_dim  = config["hidden_dim"]
    eps         = 1e-5
    nomask      = config.get("nomask", True)

    # ------------------------------------------------------------------
    # ------------------------------------------------------------------
    # 1️⃣  First LayerNorm (FP16, in‑place)
    # ------------------------------------------------------------------
    B, N, _, _ = input_tensor.shape          # input shape [B, N, N, dim]
    total_rows  = B * N * N

    # Cast input to FP16 (creates a new buffer)
    x_fp16 = input_tensor.to(torch.float16).contiguous()
    x_flat = x_fp16.view(total_rows, dim)    # [total_rows, dim]

    # Prepare LayerNorm parameters in FP16
    norm_w_fp16 = weights["norm.weight"].to(torch.float16)
    norm_b_fp16 = weights["norm.bias"].to(torch.float16)

    BLOCK_M = 128          # rows handled per program
    BLOCK_D = 64           # columns per inner loop (tunable)

    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_f16_inplace_kernel[grid](
        x_flat,
        norm_w_fp16,
        norm_b_fp16,
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    # No explicit sync needed – following torch ops will wait on the same stream.

    # ------------------------------------------------------------------
    # 2️⃣  Fuse projection + gate for left & right paths (all FP16)
    # ------------------------------------------------------------------
    # Concatenate projection and gate weight matrices to reduce GEMM count
    left_concat_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)                                    # [2*H, dim]

    right_concat_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)                                    # [2*H, dim]

    # Linear projections (half‑precision)
    left_concat = F.linear(x_flat, left_concat_w)                # [total_rows, 2*H]
    right_concat = F.linear(x_flat, right_concat_w)              # [total_rows, 2*H]

    left_proj_h   = left_concat[:, :hidden_dim]                  # [total_rows, H]
    left_gate_pre = left_concat[:, hidden_dim:]                  # [total_rows, H]

    right_proj_h  = right_concat[:, :hidden_dim]
    right_gate_pre = right_concat[:, hidden_dim:]

    # Output gate (computed once)
    out_gate_pre = F.linear(
        x_flat, weights["out_gate.weight"].to(torch.float16)
    )   # [total_rows, H]

    # ------------------------------------------------------------------
    # 3️⃣  Optional mask (broadcast over the hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape: [B, N, N] → flatten → [total_rows, 1]
        mask_h = mask_tensor.view(total_rows, 1).to(torch.float16)
        left_proj_h  = left_proj_h * mask_h
        right_proj_h = right_proj_h * mask_h

    # ------------------------------------------------------------------
    # 4️⃣  Apply sigmoid‑gates (gate pre‑activations cast to FP32 for accuracy)
    # ------------------------------------------------------------------
    left_gate_h  = torch.sigmoid(left_gate_pre.float()).to(torch.float16)
    right_gate_h = torch.sigmoid(right_gate_pre.float()).to(torch.float16)
    out_gate_h   = torch.sigmoid(out_gate_pre.float()).to(torch.float16)

    # Gated projections
    left_h  = left_proj_h * left_gate_h                # [total_rows, H]
    right_h = right_proj_h * right_gate_h

    # ------------------------------------------------------------------
    # 5️⃣  Triangular multiplicative update (batched contraction via einsum)
    # ------------------------------------------------------------------
    # Reshape to 4‑D tensors (no extra copies – just a view)
    left_h_4d  = left_h.view(B, N, N, hidden_dim)      # [B, N, N, H]
    right_h_4d = right_h.view(B, N, N, hidden_dim)     # [B, N, N, H]

    # Einstein sum over the shared "k" dimension
    # out shape: [B, N, N, H]
    out_h = torch.einsum('b i k d, b j k d -> b i j d', left_h_4d, right_h_4d)

    # ------------------------------------------------------------------
    # 6️⃣  Second LayerNorm over the hidden dimension (FP16, in‑place)
    # ------------------------------------------------------------------
    total_rows_h = B * N * N
    out_flat = out_h.view(total_rows_h, hidden_dim)   # [total_rows_h, H]

    to_out_norm_w_fp16 = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b_fp16 = weights["to_out_norm.bias"].to(torch.float16)

    grid2 = (triton.cdiv(total_rows_h, BLOCK_M),)
    layernorm_f16_inplace_kernel[grid2](
        out_flat,
        to_out_norm_w_fp16,
        to_out_norm_b_fp16,
        total_rows_h,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 7️⃣  Apply output gate and final linear projection
    # ------------------------------------------------------------------
    # Multiply by output gate (both FP16)
    out_norm_gated = out_flat * out_gate_h   # still [total_rows_h, H]

    # Final projection back to `dim`
    final_h = F.linear(
        out_norm_gated,
        weights["to_out.weight"].to(torch.float16)   # [dim, H]
    )   # [total_rows_h, dim]  (FP16)

    # Reshape to original 4‑D shape and cast to FP32 for the API contract
    final = final_h.view(B, N, N, dim).to(torch.float32)

    return final
