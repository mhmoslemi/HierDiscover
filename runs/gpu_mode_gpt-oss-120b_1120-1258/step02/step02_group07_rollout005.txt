@triton.jit
def batched_matmul_kernel(A, B, C, M, N, K,
                          stride_abl, stride_ami, stride_ak,
                          stride_bbk, stride_bk, stride_bn,
                          stride_ccl, stride_cmi, stride_cn,
                          BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    # program IDs
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute start offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # Compute offset for the K tile
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)

        # Load A and B tiles with mask of bounds
        a_ptrs = A + pid_batch*stride_abl + offs_m[:, None] * stride_ami + offs_k[None, :] * stride_ak
        b_ptrs = B + pid_batch*stride_bbk + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

        a = tl.load(a_ptrs, mask=offs_m[:, None] < M and offs_k[None, :] < K, other=0.0)
        b = tl.load(b_ptrs, mask=offs_k[:, None] < K and offs_n[None, :] < N, other=0.0)

        # Convert to fp16? Already half. Use tl.dot to compute acc
        acc += tl.dot(a, b)  # a: (BLOCK_M,BLOCK_K), b: (BLOCK_K,BLOCK_N)

    # Write back to C
    c_ptrs = C + pid_batch*stride_ccl + offs_m[:, None] * stride_cmi + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16), mask=offs_m[:, None] < M and offs_n[None, :] < N)

                          stride_abl, stride_ami, stride_ak,
                          stride_bbk, stride_bk, stride_bn,
                          stride_ccl, stride_cmi, stride_cn,
                          BLOCK_M: tl.constexpr,
                          BLOCK_N: tl.constexpr,
                          BLOCK_K: tl.constexpr):

    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Initialise accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):

        offs_k = k_offset + tl.arange(0, BLOCK_K)

import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def batched_matmul_kernel(A, B, C,
                          M, N, K,
                          stride_abl, stride_ami, stride_ak,
                          stride_bbk, stride_bk, stride_bn,
                          stride_ccl, stride_cmi, stride_cn,
                          BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets within the block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Accumulator in float32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for pid_k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = pid_k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)

        # Pointers
        a_ptrs = A + pid_batch * stride_abl + offs_m[:, None] * stride_ami + offs_k[None, :] * stride_ak
        b_ptrs = B + pid_batch * stride_bbk + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

        # Masks for out‑of‑bounds loads
        mask_a = (offs_m[:, None] < M) & (offs_k[None, :] < K)

        a = tl.load(a_ptrs, mask=mask_a, other=0.0)

"""
TriMul (outgoing) – fast forward pass.

The core N³ contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is realised as a batched matrix‑multiplication
    out_d = left_d @ right_dᵀ
where the hidden dimension `d` is folded into the batch axis.
The heavy contraction is executed by a custom Triton kernel that
uses FP16 inputs with FP32 accumulation (Tensor‑core friendly).
All LayerNorm, Linear and gating operations run under torch.cuda.amp
in mixed‑precision; only the matmul is off‑loaded to Triton.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_matmul_kernel(
    A, B, C,
    M, N, K,
    stride_abl, stride_ami, stride_ak,
    stride_bbk, stride_bk, stride_bn,
    stride_ccl, stride_cmi, stride_cn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched GEMM: C = A @ B   (A: [B, M, K], B: [B, K, N], C: [B, M, N]).

    All tensors are stored in row‑major order.
    """
    batch_id = tl.program_id(0)      # batch * hidden_dim
    pid_m    = tl.program_id(1)      # block row
    pid_n    = tl.program_id(2)      # block col

    # block coordinates
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)

        a_ptrs = A + batch_id * stride_abl \
                 + offs_m[:, None] * stride_ami \
                 + offs_k[None, :] * stride_ak
        b_ptrs = B + batch_id * stride_bbk \
                 + offs_k[:, None] * stride_bk \
                 + offs_n[None, :] * stride_bn

        mask_a = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        mask_b = (offs_k[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=mask_a, other=0.0)
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)

        acc += tl.dot(a, b)                     # FP32 accumulation

    c_ptrs = C + batch_id * stride_ccl \
             + offs_m[:, None] * stride_cmi \
             + offs_n[None, :] * stride_cn
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask         : torch.Tensor [B, N, N] (bool/float) or None
        - weights      : dict of model parameters (float32) with keys
                         ["norm.weight", "norm.bias",
                          "left_proj.weight", "right_proj.weight",
                          "left_gate.weight", "right_gate.weight",
                          "out_gate.weight",
                          "to_out_norm.weight", "to_out_norm.bias",
                          "to_out.weight"]
        - config       : dict containing at least
                         {"dim": int, "hidden_dim": int, "nomask": bool}

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim        = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask     = config.get("nomask", True)

    # --------------------------------------------------------------
    # load model parameters (they are already on the correct device)
    # --------------------------------------------------------------
    norm_weight         = weights["norm.weight"]
    norm_bias           = weights["norm.bias"]
    left_proj_weight    = weights["left_proj.weight"]
    right_proj_weight   = weights["right_proj.weight"]
    left_gate_weight    = weights["left_gate.weight"]
    right_gate_weight   = weights["right_gate.weight"]
    out_gate_weight     = weights["out_gate.weight"]
    to_out_norm_weight  = weights["to_out_norm.weight"]
    to_out_norm_bias    = weights["to_out_norm.bias"]
    to_out_weight       = weights["to_out.weight"]

    # --------------------------------------------------------------
    # mixed‑precision forward (everything under AMP)
    # --------------------------------------------------------------
    with torch.cuda.amp.autocast():
        # 1) LayerNorm over the input channel dimension
        x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

        # 2) Linear projections (no bias) -> shape [B, N, N, hidden_dim]
        left  = F.linear(x, left_proj_weight)   # [B, N, N, H]
        right = F.linear(x, right_proj_weight)  # [B, N, N, H]

        # 3) optional mask (broadcast over hidden dim)
        if not nomask and mask is not None:
            mask_f = mask.to(x.dtype).unsqueeze(-1)   # [B, N, N, 1]
            left  = left * mask_f
            right = right * mask_f

        # 4) gating scalars (sigmoid)
        left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))
        right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
        out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))

        # 5) apply gates
        left  = left  * left_gate
        right = right * right_gate

        # ----------------------------------------------------------
        # 6) Batched matmul (our custom Triton kernel)
        # ----------------------------------------------------------
        # ensure FP16 for the kernel
        left  = left.to(torch.float16)
        right = right.to(torch.float16)

        # reshape to (B, H, N, N) and then fold B*H into a single batch axis
        left_mat  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
        right_mat = right.permute(0, 3, 2, 1).contiguous()  # [B, H, N, N] (k,j)

        B, H, N, _ = left_mat.shape
        total_batch = B * H

        # flatten leading two dimensions → (B*H, N, N)
        left_mat  = left_mat.view(total_batch, N, N)
        right_mat = right_mat.view(total_batch, N, N)

        # allocate output buffer
        out_mat = torch.empty((total_batch, N, N), dtype=torch.float16, device=device)

        # strides (row‑major)
        s_a_batch, s_a_m, s_a_k = left_mat.stride()
        s_b_batch, s_b_k, s_b_n = right_mat.stride()
        s_c_batch, s_c_m, s_c_n = out_mat.stride()

        # launch Triton kernel
        BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 32
        grid = (total_batch,
                triton.cdiv(N, BLOCK_M),
                triton.cdiv(N, BLOCK_N))
        batched_matmul_kernel[grid](
            left_mat, right_mat, out_mat,
            N, N, N,
            s_a_batch, s_a_m, s_a_k,
            s_b_batch, s_b_k, s_b_n,
            s_c_batch, s_c_m, s_c_n,
            BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        )

        # reshape back to the original layout [B, N, N, H]
        out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

        # 7) post‑projection LayerNorm & final gating
        out = F.layer_norm(out, (hidden_dim,),
                           weight=to_out_norm_weight,
                           bias=to_out_norm_bias)
        out = out * out_gate

        # 8) final linear projection back to `dim`
        out = F.linear(out, to_out_weight)

    # convert to float32 for the API contract
    return out.to(torch.float32)
