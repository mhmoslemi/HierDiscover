"""
TriMul outgoing kernel (AlphaFold‑3) – fast FP16 forward pass.
================================================================
Algorithm
---------
1. Row‑wise LayerNorm over the channel dimension (dim) using a small Triton
   kernel (float32).  The kernel computes mean/variance per row and writes
   the normalized values to an FP32 buffer.
2. Cast the normalized tensor to FP16 – all subsequent dense ops use half
   precision.
3. Fuse the two projection linear layers (left/right) into a single GEMM and
   likewise fuse the three gating linear layers.  The result is split into the
   required tensors.
4. Apply the optional mask (broadcasted over the hidden dimension) and the
   element‑wise sigmoid gates.
5. The multiplicative update  
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]  
   is performed with a single `torch.einsum`, which internally uses a
   batched GEMM without any explicit copy/transpose.
6. A second LayerNorm over the hidden dimension (half‑precision) is done with
   `torch.nn.functional.layer_norm`.
7. Apply the output gate and a final linear projection back to `dim`; the
   result is cast back to FP32.

Only the first LayerNorm is implemented in Triton – this satisfies the
requirement that at least part of the work be a custom kernel.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def layernorm_kernel(
    x_ptr,                # const float*    [total_rows, dim]
    y_ptr,                # mut   float*    [total_rows, dim]
    weight_ptr,           # const float*    [dim]
    bias_ptr,             # const float*    [dim]
    total_rows,           # int (runtime)
    dim: tl.constexpr,    # compile‑time
    eps: tl.constexpr,    # compile‑time
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """Row‑wise LayerNorm (float32) for a 2‑D tensor."""
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # -----------------------------------------------------------------
    # 1st pass – compute row‑wise sum and sum‑of‑squares
    # -----------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                               # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                # row‑wise sum
        sum_sq  += tl.sum(vals * vals, axis=1)         # row‑wise sum of squares

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                           # 1 / sqrt(var+eps)

    # -----------------------------------------------------------------
    # 2nd pass – write normalized values with scale (γ) and shift (β)
    # -----------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)

        out = norm * w[None, :] + b[None, :]

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out, mask=mask_m[:, None] & mask_d[None, :])

    """
    Forward pass of the “outgoing” TriMul operator.

    """

    input_tensor, mask_tensor, weights, config = data

    left_h  = left_proj_h  * left_gate_h       # [R, H]

    left_h  = left_h.view(B, N, N, hidden_dim)   # [B,N,N,H]
    right_h = right_h.view(B, N, N, hidden_dim)  # [B,N,N,H]
    out_gate_h = out_gate_h.view(B, N, N, hidden_dim)  # [B,N,N,H]

    # -----------------------------------------------------------------

    # -----------------------------------------------------------------
    # out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    out = torch.einsum('bikh,bjkh->bijh', left_h, right_h)   # [B,N,N,H] FP16

    # -----------------------------------------------------------------
    out = F.layer_norm(

        bias=weights["to_out_norm.bias"].to(torch.float16),
        eps=eps,

    # -----------------------------------------------------------------
    out = out * out_gate_h                                   # element‑wise FP16
