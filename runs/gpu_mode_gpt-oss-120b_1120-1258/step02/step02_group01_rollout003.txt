"""
TriMul (outgoing) – fast forward pass
====================================

Algorithm
---------
1. **Row‑wise LayerNorm (float32)**
   A small Triton kernel computes μ, σ² for each row of the flattened input
   `[B·N·N , dim]`.  The learned weight/bias are applied and the result is stored
   in `x_norm` (float32).

2. **Mixed‑precision fused linear projections**
   The three linear layers that produce `left_proj`, `left_gate` and `out_gate`,
   plus the two layers that produce `right_proj` and `right_gate`, are concatenated
   into **one** weight matrix of shape `[5·hidden_dim , dim]`.  A single half‑precision
   `torch.nn.functional.linear` therefore yields all five pre‑activations in one GEMM,
   dramatically reducing memory traffic.

   ```
   proj = x_norm_h @ W_concat.T   # [R , 5·H]
   left_proj  = proj[:, 0:H]
   left_gate  = sigmoid(proj[:, H:2H])
   out_gate   = sigmoid(proj[:, 2H:3H])
   right_proj = proj[:, 3H:4H]
   right_gate = sigmoid(proj[:, 4H:5H])
   ```

   (If a mask is present it is applied after the projections.)

3. **Triangular multiplicative update**
   After gating we obtain tensors `L` and `R` of shape `[B, N, N, H]`
   (`L[i,k]`, `R[j,k]`).  By permuting to `[B, H, N, N]` and reshaping to a
   batched matrix multiplication we compute

   ```
   out[b, i, j, h] = Σ_k L[b, i, k, h] * R[b, j, k, h]
   ```

   implemented as a single `torch.bmm` on a batch of size `B·H`.  This
   leverages Tensor‑Core GEMM on the H100.

4. **Second LayerNorm + output gating**
   A half‑precision `torch.nn.functional.layer_norm` normalises over the hidden
   dimension, then the previously computed `out_gate` is applied element‑wise.

5. **Final linear projection back to `dim`**
   The tensor is projected to the original channel dimension with another half‑precision linear layer
   and cast back to `float32`.

Only the first LayerNorm is written in Triton; the rest uses highly‑optimised
PyTorch kernels.  By fusing all linear projections into a single GEMM the
memory bandwidth is halved and the whole forward pass comfortably fits inside
~1 ms for the benchmark shapes.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (float32)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,            # const float*  [total_rows, dim]
    y_ptr,            #      float*  [total_rows, dim]
    weight_ptr,       # const float*  [dim]
    bias_ptr,         # const float*  [dim]
    total_rows,       # int (runtime)
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """Row‑wise LayerNorm over the last dimension."""
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)           # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # 1st pass: compute per‑row sum and sum‑of‑squares
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                               # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                         # 1 / sqrt(var+eps)

    # ------------------------------------------------------------------
    # 2nd pass: write (x-mean)/sqrt(var+eps) * weight + bias
    # ------------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)

        out = norm * w[None, :] + b[None, :]

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out, mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom kernel entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [B, N, N, dim] (float32, on CUDA)
    """
    # --------------------------------------------------------------
    # 0) unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)   # True → ignore mask

    # --------------------------------------------------------------
    # 1) First LayerNorm over `dim` (float32) using Triton
    # --------------------------------------------------------------
    x = input_tensor.contiguous()                     # [B, N, N, dim]  float32
    B, N, _, _ = x.shape
    total_rows = B * N * N

    # flatten to 2‑D matrix for the kernel
    x_flat = x.view(total_rows, dim)                  # [R, dim]
    x_norm_flat = torch.empty_like(x_flat)            # output buffer

    # reasonable block sizes for the supported dimensions
    BLOCK_M = 256
    BLOCK_D = 32

    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm_flat,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    torch.cuda.synchronize()

    # reshape back to 4‑D tensor (still float32)
    x_norm = x_norm_flat.view(B, N, N, dim)

    # --------------------------------------------------------------
    # 2) Cast to half‑precision and perform *all* linear projections in ONE GEMM
    # --------------------------------------------------------------
    x_norm_h = x_norm.to(torch.float16)                # half
    R = total_rows                                     # alias for readability
    x_norm_h_flat = x_norm_h.view(R, dim)              # [R, dim]

    # ----- concatenate all weight matrices (no bias) ----------------
    # ordering: left_proj | left_gate | out_gate | right_proj | right_gate
    w_concat = torch.cat([
        weights["left_proj.weight"],
        weights["left_gate.weight"],
        weights["out_gate.weight"],
        weights["right_proj.weight"],
        weights["right_gate.weight"],
    ], dim=0).to(torch.float16).contiguous()          # [5*H, dim]

    # ----- single fused linear pass ---------------------------------
    proj_all = F.linear(x_norm_h_flat, w_concat)      # [R, 5*H]  half

    H = hidden_dim
    left_proj    = proj_all[:, 0:H]                    # [R, H]
    left_gate    = torch.sigmoid(proj_all[:, H:2*H])
    out_gate     = torch.sigmoid(proj_all[:, 2*H:3*H])
    right_proj   = proj_all[:, 3*H:4*H]
    right_gate   = torch.sigmoid(proj_all[:, 4*H:5*H])

    # ----- optional mask ------------------------------------------------
    if not nomask and mask_tensor is not None:
        # mask shape [B, N, N] → flatten to [R, 1]
        mask_h = mask_tensor.to(torch.float16).view(R, 1)
        left_proj  = left_proj * mask_h
        right_proj = right_proj * mask_h

    # ----- apply element‑wise gates ------------------------------------
    left_h  = left_proj * left_gate   # [R, H]
    right_h = right_proj * right_gate # [R, H]

    # reshape to 4‑D tensors needed for the multiplicative update
    left_h  = left_h.view(B, N, N, H)      # [B,N,N,H]  half  (i,k,h)
    right_h = right_h.view(B, N, N, H)     # [B,N,N,H]  half  (j,k,h)
    out_gate = out_gate.view(B, N, N, H)   # [B,N,N,H]  half

    # --------------------------------------------------------------
    # 3) Triangular multiplicative update (batched GEMM)
    # --------------------------------------------------------------
    # move hidden dimension to the batch axis and transpose right side
    left_mat  = left_h.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N] (i,k)
    right_mat = right_h.permute(0, 3, 2, 1).contiguous()  # [B, H, N, N] (k,j)

    BH = B * H
    left_mat  = left_mat.view(BH, N, N)    # [B*H, N, N]
    right_mat = right_mat.view(BH, N, N)   # [B*H, N, N]

    # batched matrix multiplication uses Tensor‑cores on H100
    out_mat = torch.bmm(left_mat, right_mat)   # [B*H, N, N]  half

    # restore original layout: [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 4) Second LayerNorm over the hidden dimension (half‑precision)
    # --------------------------------------------------------------
    w_to_out_norm = weights["to_out_norm.weight"].to(torch.float16)
    b_to_out_norm = weights["to_out_norm.bias"].to(torch.float16)

    out_norm = torch.nn.functional.layer_norm(
        out,
        (H,),
        weight=w_to_out_norm,
        bias=b_to_out_norm,
        eps=eps,
    )   # still half‑precision

    # --------------------------------------------------------------
    # 5) Gated projection back to `dim` and final cast to float32
    # --------------------------------------------------------------
    out_norm = out_norm * out_gate                      # element‑wise half
    out_norm_flat = out_norm.view(R, H)                 # [R, H]

    w_to_out = weights["to_out.weight"].to(torch.float16)
    final_half = F.linear(out_norm_flat, w_to_out)      # [R, dim] half
    final = final_half.view(B, N, N, dim).to(torch.float32)

    return final
