import torch
from torch import nn, einsum
import math

# Reference code in PyTorch
class TriMul(nn.Module):
    def __init__(
        self,
        dim: int,
        hidden_dim: int,
    ):
        super().__init__()

        self.norm = nn.LayerNorm(dim)

        self.left_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.right_proj = nn.Linear(dim, hidden_dim, bias=False)

        self.left_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.right_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.out_gate = nn.Linear(dim, hidden_dim, bias=False)

        self.to_out_norm = nn.LayerNorm(hidden_dim)
        self.to_out = nn.Linear(hidden_dim, dim, bias=False)

    def forward(self, x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
        """
        x: [bs, seq_len, seq_len, dim]
        mask: [bs, seq_len, seq_len]

        Returns:
            output: [bs, seq_len, seq_len, dim]
        """
        batch_size, seq_len, _, dim = x.shape

        x = self.norm(x)

        left = self.left_proj(x)
        right = self.right_proj(x)

        mask = mask.unsqueeze(-1)
        left = left * mask
        right = right * mask

        left_gate = self.left_gate(x).sigmoid()
        right_gate = self.right_gate(x).sigmoid()
        out_gate = self.out_gate(x).sigmoid()

        left = left * left_gate
        right = right * right_gate

        out = einsum('... i k d, ... j k d -> ... i j d', left, right)
        # This einsum is the same as the following:
        # out = torch.zeros(batch_size, seq_len, seq_len, dim, device=x.device)
        
        # # Compute using nested loops
        # for b in range(batch_size):
        #     for i in range(seq_len):
        #         for j in range(seq_len):
        #             # Compute each output element
        #             for k in range(seq_len):
        #                 out[b, i, j] += left[b, i, k, :] * right[b, j, k, :]

        out = self.to_out_norm(out)
        out = out * out_gate
        return self.to_out(out)

"""
custom_kernel for the “outgoing” TriMul operator (AlphaFold‑3).

Algorithm (all tensors reside on the same CUDA device):
1. Row‑wise LayerNorm over the last channel (dim) – Triton kernel.
2. Fuse the five linear projections (left/right and three gates) into a single
   GEMM by concatenating the weight matrices.  Compute them in fp16 for
   tensor‑core speed, then split the result.
3. (Optional) apply the pairwise mask to the left/right projections.
4. Apply sigmoid gates and element‑wise gating.
5. Batched matrix‑multiply per hidden channel:
       out_h[i,j] = Σ_k left_h[i,k] * right_h[j,k]
   which is an ordinary batched GEMM (torch.bmm) on tensors of shape
   (B·H, N, N) in fp16.
6. Row‑wise LayerNorm over the hidden dimension (H) – Triton kernel.
7. Apply the final output gate, then a linear projection back to `dim`
   (also in fp16).  The result is cast back to fp32 before returning.

All heavy reductions (the two LayerNorms) are performed by a custom Triton
kernel; the dense linear algebra uses fused fp16 GEMMs (cuBLAS/Tensor‑Core),
reducing kernel launch overhead from nine (in the reference code) to four.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton Row‑wise LayerNorm (float32 accumulation, works for fp16 data)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # *const float*   [total_rows, dim]
    y_ptr,          # *mut   float*   [total_rows, dim]
    weight_ptr,     # *const float*   [dim]
    bias_ptr,       # *const float*   [dim]
    total_rows,     # int (runtime)
    dim: tl.constexpr,    # int (compile‑time)
    eps: tl.constexpr,    # float (compile‑time)
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """Row‑wise LayerNorm for a 2‑D tensor."""
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # -----------------------------------------------------------------
    # First pass: compute mean & variance (float32)
    # -----------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                               # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)    # 1 / sqrt(var + eps)

    # -----------------------------------------------------------------
    # Second pass: write normalized values
    # -----------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        # (x - mean) / sqrt(var + eps)
        norm = (vals - mean[:, None]) * rstd[:, None]

        # γ (weight) and β (bias) – may be fp16, promoted to fp32 automatically
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)

        out = norm * w[None, :] + b[None, :]

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out, mask=mask_m[:, None] & mask_d[None, :])


def custom_kernel(data):
    """
    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32, on CUDA)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]               # input channel dimension (C)
    hidden_dim = config["hidden_dim"] # projection dimension (H)
    eps = 1e-5
    nomask = config.get("nomask", True)   # True ⇢ no mask

    # ------------------------------------------------------------------
    # Prepare contiguous input
    # ------------------------------------------------------------------
    x = input_tensor.contiguous()
    B, N, _, _ = x.shape                # x: [B, N, N, dim]
    total_rows = B * N * N               # rows for flatten view

    # ------------------------------------------------------------------
    # 1) First LayerNorm over dim
    # ------------------------------------------------------------------
    x_flat = x.view(total_rows, dim)                 # [total_rows, dim]
    x_norm_flat = torch.empty_like(x_flat)           # fp32 output buffer

    BLOCK_M = 128
    BLOCK_D = 32

    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm_flat,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()

    # ------------------------------------------------------------------
    # 2) Fused linear projections + gates (fp16 for speed)
    # ------------------------------------------------------------------
    # Convert normalized input to fp16
    x_norm_fp16 = x_norm_flat.half()   # [total_rows, dim]

    # Concatenate all five weight matrices (shape: (5*H, dim))
    w_left_proj   = weights["left_proj.weight"]
    w_right_proj  = weights["right_proj.weight"]
    w_left_gate   = weights["left_gate.weight"]
    w_right_gate  = weights["right_gate.weight"]
    w_out_gate    = weights["out_gate.weight"]

    # All weight tensors are fp32; cast once to fp16
    concat_weight = torch.cat(
        [w_left_proj, w_right_proj, w_left_gate, w_right_gate, w_out_gate],
        dim=0,
    ).half().contiguous()                     # shape (5*H, dim)

    # Single GEMM:   [total_rows, dim] @ [dim, 5*H]ᵀ  -> [total_rows, 5*H]
    proj_all = F.linear(x_norm_fp16, concat_weight)   # fp16

    # Split back to individual tensors
    splits = torch.split(proj_all, hidden_dim, dim=1)
    left_proj_fp16, right_proj_fp16, left_gate_fp16, right_gate_fp16, out_gate_fp16 = splits

    # ------------------------------------------------------------------
    # 2a) Optional mask (applied before gating)
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape: [B, N, N] -> flatten -> [total_rows, 1]
        mask_flat = mask_tensor.contiguous().view(total_rows, 1).to(x.dtype)
        # Cast mask to fp16 to match the projections
        mask_flat_fp16 = mask_flat.half()
        left_proj_fp16 = left_proj_fp16 * mask_flat_fp16
        right_proj_fp16 = right_proj_fp16 * mask_flat_fp16

    # ------------------------------------------------------------------
    # 2b) Apply sigmoid gates (still fp16)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_fp16)   # [total_rows, H]
    right_gate = torch.sigmoid(right_gate_fp16)
    out_gate = torch.sigmoid(out_gate_fp16)

    # Element‑wise gating
    left_fp16 = left_proj_fp16 * left_gate       # [total_rows, H]
    right_fp16 = right_proj_fp16 * right_gate    # [total_rows, H]

    # Reshape to 4‑D tensors for the triangular update
    left = left_fp16.view(B, N, N, hidden_dim)          # [B, N, N, H]
    right = right_fp16.view(B, N, N, hidden_dim)        # [B, N, N, H]
    out_gate = out_gate.view(B, N, N, hidden_dim)      # [B, N, N, H]

    # ------------------------------------------------------------------
    # 3) Triangular multiplicative update (batched GEMM)
    #    out_h[i, j] = Σ_k left_h[i, k] * right_h[j, k]
    # ------------------------------------------------------------------
    # Permute to put hidden dimension in front: [B, H, N, N]
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 2, 1).contiguous() # [B, H, N, N] (transposed on k)

    BH = B * hidden_dim
    left_mat = left_perm.view(BH, N, N)      # [B*H, N, N]
    right_mat = right_perm.view(BH, N, N)   # [B*H, N, N]

    # Batched matrix multiply in fp16 (Tensor‑Core)
    out_mat = torch.bmm(left_mat, right_mat)  # [B*H, N, N] fp16

    # Reshape back to [B, N, N, H] (fp16)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 4) Second LayerNorm over hidden_dim (fp16 input, fp32 accumulation)
    # ------------------------------------------------------------------
    total_rows_h = B * N * N
    out_flat = out.view(total_rows_h, hidden_dim)   # [total_rows_h, H]
    out_norm_flat = torch.empty_like(out_flat)      # fp16 output buffer

    grid2 = (triton.cdiv(total_rows_h, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm_flat,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows_h,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()

    out_norm = out_norm_flat.view(B, N, N, hidden_dim)   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 5) Apply final output gate and linear projection back to `dim`
    # ------------------------------------------------------------------
    gated = out_norm * out_gate          # fp16 element‑wise

    # Final linear projection: hidden_dim -> dim (fp16)
    to_out_weight_fp16 = weights["to_out.weight"].half()
    gated_flat = gated.view(total_rows_h, hidden_dim)   # [total_rows_h, H]
    final_fp16 = F.linear(gated_flat, to_out_weight_fp16)   # [total_rows_h, dim]

    final = final_fp16.view(B, N, N, dim).to(torch.float32)  # cast to fp32

    return final
