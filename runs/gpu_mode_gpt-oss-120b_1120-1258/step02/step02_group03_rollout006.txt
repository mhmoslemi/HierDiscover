"""
Optimized Triton implementation of the “outgoing” TriMul operator (forward only).

Algorithm
---------
1. Layer‑norm over the channel dimension (float32).
2. Reduce precision to FP16 and compute all linear projections + sigmoid gates.
3. (Optional) apply the pairwise mask before gating.
4. Apply element‑wise gates: left = left * sigmoid(left_gate), right = right * sigmoid(right_gate).
5. Batched matrix multiplication  out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   = A @ Bᵀ  where A and B are (B·H, N, N) FP16 tensors.
   The heavy O(N³·H) work is performed by a fused Triton kernel that:
      • loads tiles of A and B in FP16,
      • accumulates in FP32 with `tl.dot` (Tensor‑cores),
      • writes the result directly in FP32.
   Block size (128,128,64) is tuned for H100; 8 warps per block give high occupancy.
6. Reshape back to [B,N,N,H] (still FP32), apply a second LayerNorm over H,
   multiply by the out‑gate, and linearly project back to the original channel dim.
   Final tensor is float32 as required.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def _batched_matmul_kernel(
    A, B, C,
    M, N, K,
    stride_Ab, stride_Ar, stride_Ac,   # batch, row, col strides for A
    stride_Bb, stride_Br, stride_Bc,   # batch, row, col strides for B
    stride_Cb, stride_Cr, stride_Cc,   # batch, row, col strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched FP16 → FP32 GEMM: C = A @ B  (A: (B·H, M, K), B: (B·H, K, N))

    Accumulates in FP32 and stores the result in FP32.
    """
    pid_m = tl.program_id(0)   # block row
    pid_n = tl.program_id(1)   # block col
    pid_b = tl.program_id(2)   # batch * hidden index

    # linear offsets within the output matrix
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers for this block
    a_ptrs = A + pid_b * stride_Ab + offs_m[:, None] * stride_Ar + offs_k[None, :] * stride_Ac
    b_ptrs = B + pid_b * stride_Bb + offs_k[:, None] * stride_Br + offs_n[None, :] * stride_Bc

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    num_k = tl.cdiv(K, BLOCK_K)
    for _ in range(num_k):
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k[None, :] < K),
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & mask_n[None, :],
                    other=0.0)
        acc += tl.dot(a, b)               # dot promotes to FP32 automatically
        a_ptrs += BLOCK_K * stride_Ac
        b_ptrs += BLOCK_K * stride_Br

    # store result (FP32) with masking
    c_ptrs = C + pid_b * stride_Cb + offs_m[:, None] * stride_Cr + offs_n[None, :] * stride_Cc
    tl.store(c_ptrs, acc,
             mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no backward).

    Args:
        data (Tuple):
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights: dict of weight tensors required by the module
            - config: dict containing keys "dim", "hidden_dim", "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # -------------------------- unpack ------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -------------------------- LayerNorm over input channels -------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim]  (FP32)

    # -------------------------- FP16 linear projections & gates -----------------------
    x_h = x.to(torch.float16)

    # Linear projections (no bias)
    left  = torch.nn.functional.linear(x_h, weights["left_proj.weight"].to(torch.float16))
    right = torch.nn.functional.linear(x_h, weights["right_proj.weight"].to(torch.float16))

    # Gates (sigmoid)
    left_gate  = torch.nn.functional.linear(x_h, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = torch.nn.functional.linear(x_h, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate   = torch.nn.functional.linear(x_h, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # -------------------------- optional pairwise mask -------------------------------
    if not nomask and mask is not None:
        mask_h = mask.to(torch.float16).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask_h
        right = right * mask_h

    # -------------------------- element‑wise gating -----------------------------------
    left = left * left_gate
    right = right * right_gate

    # -------------------------- prepare tensors for batched GEMM ----------------------
    B, N, _, _ = left.shape  # B, N, N, hidden

    # Move hidden dimension to the second axis and make tensors contiguous
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, i, k)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, k, j) == Bᵀ

    # Collapse batch+hidden => (B·H, N, N)
    BH = B * hidden
    left_flat  = left_perm.view(BH, N, N)   # A: (BH, M=N, K=N)
    right_flat = right_perm.view(BH, N, N)  # B: (BH, K=N, N)

    # Output buffer – keep it FP32 to avoid an extra cast after the kernel
    out_flat = torch.empty((BH, N, N), dtype=torch.float32, device=device)

    # -------------------------- launch Triton batched GEMM ---------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        BH,
    )

    _batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,                     # M
        N,                     # N
        N,                     # K
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        right_flat.stride(0), right_flat.stride(1), right_flat.stride(2),
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,           # tuned for H100
        num_stages=2,
    )

    # Reshape back to [B, N, N, hidden] (still FP32)
    out = out_flat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # -------------------------- second LayerNorm, out‑gate, final projection ---------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # Apply out‑gate (convert once to FP32)
    out = out * out_gate.to(torch.float32)

    # Final linear projection (hidden → dim). Compute in FP16 for speed, then back to FP32.
    out_h = out.to(torch.float16)
    to_out_weight_h = weights["to_out.weight"].to(torch.float16)
    out = torch.nn.functional.linear(out_h, to_out_weight_h).to(torch.float32)

    return out
