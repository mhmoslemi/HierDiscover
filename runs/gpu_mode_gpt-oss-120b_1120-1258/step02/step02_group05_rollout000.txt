    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]; hidden_dim = config["hidden_dim"]; nomask = config.get("nomask", True)
    
    eps = 1e-5

    # LayerNorm
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps)

    # cast to half
    x_h = x.half()
    # linear weights half
    left_proj_weight_h = weights["left_proj.weight"].half()
    right_proj_weight_h = weights["right_proj.weight"].half()
    left_gate_weight_h = weights["left_gate.weight"].half()
    right_gate_weight_h = weights["right_gate.weight"].half()
    out_gate_weight_h = weights["out_gate.weight"].half()

    left = F.linear(x_h, left_proj_weight_h)
    right = F.linear(x_h, right_proj_weight_h)

    left_gate = torch.sigmoid(F.linear(x_h, left_gate_weight_h))
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_weight_h))
    out_gate = torch.sigmoid(F.linear(x_h, out_gate_weight_h))

    left = left * left_gate
    right = right * right_gate

    if not nomask:
        mask_exp = mask.to(left.dtype).unsqueeze(-1) # shape [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # shape permutations + contiguity
    B, N, _, _ = left.shape

"""
TriMul (outgoing) – Triton‑accelerated forward pass
===================================================

The module implements the “outgoing” variant from the AlphaFold3 paper:

    out[b,i,j] = Σₖ left[b,i,k] * right[b,j,k]          (hidden dim = H)

where ``left``/``right`` are linear projections of the input tensor (with
sigmoid gates and optional masking).  The costly O(N³·H) contraction is
performed by a custom Triton kernel that carries out a batched GEMM
C = A @ Bᵀ for each (batch, hidden) pair.

Key ideas
---------
1. **Layer‑norms** are left in PyTorch (float32) – they are cheap and
   numerically stable.
2. **Projection + gating** are done in FP16 to halve memory traffic.
3. **Masking** (when present) is fused into the projected tensors.
4. **Cubic contraction**  
   * The 4‑D tensors are reshaped to (M = B·H, N, N) and processed by a
     single Triton kernel.  Each program instance works on a tile
     (BLOCK_M × BLOCK_N) of the output and loops over K‑tiles (BLOCK_K).
   * A‑ and B‑tiles are loaded as FP16, the dot product uses Tensor‑core
     instructions (fp16 × fp16 → fp32 accumulation).  The result is stored
     in FP32.
   * The kernel is autotuned for three sensible block‑size configurations.
5. The result is reshaped back, a second Layer‑norm + output‑gate is
   applied, and a final bias‑free FP32 linear projection restores the
   original feature dimension.

The implementation follows the required signature::

    custom_kernel(data) -> torch.Tensor
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel – batched GEMM  C = A @ Bᵀ
# ----------------------------------------------------------------------
@triton.autotune(
    configs=[
        triton.Config({'BLOCK_M': 64,  'BLOCK_N': 64,  'BLOCK_K': 32},  num_warps=4),
        triton.Config({'BLOCK_M': 128, 'BLOCK_N': 128, 'BLOCK_K': 64},  num_warps=8),
        triton.Config({'BLOCK_M': 96,  'BLOCK_N': 96,  'BLOCK_K': 48},  num_warps=8),
    ],
    key=['N'],
)
@triton.jit
def batched_tri_gemm_kernel(
    A,                # left tensor  (M, N, N)   – fp16
    B,                # right tensor (M, N, N)   – fp16  (read as Bᵀ)
    C,                # output tensor (M, N, N) – fp32
    N,                # sequence length
    stride_am, stride_ai, stride_ak,   # strides of A : batch, row(i), col(k)
    stride_bm, stride_bj, stride_bk,   # strides of B : batch, row(j), col(k)
    stride_cm, stride_ci, stride_cj,   # strides of C : batch, row(i), col(j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute   C[batch, i, j] = Σₖ A[batch, i, k] * B[batch, j, k]
    for all batches (M = B·H).  A,B are fp16, C is fp32.
    """
    batch_id = tl.program_id(0)           # 0‑th grid dimension  →  batch*hidden
    pid_i    = tl.program_id(1)           # tile row (i)
    pid_j    = tl.program_id(2)           # tile col (j)

    # Global coordinates of the tile
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N
    i = i0 + tl.arange(0, BLOCK_M)
    j = j0 + tl.arange(0, BLOCK_N)

    mask_i = i < N
    mask_j = j < N

    # accumulator in fp32 (Tensor‑core dot returns fp32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Reduction over k
    # ------------------------------------------------------------------
    for k0 in range(0, N, BLOCK_K):
        k = k0 + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # A[batch, i, k]   → shape (BLOCK_M, BLOCK_K)
        a_ptr = A + batch_id * stride_am + i[:, None] * stride_ai + k[None, :] * stride_ak
        a = tl.load(a_ptr,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0,
                    dtype=tl.float16)

        # Bᵀ → B[batch, k, j]   → shape (BLOCK_K, BLOCK_N)
        b_ptr = B + batch_id * stride_bm + k[:, None] * stride_bk + j[None, :] * stride_bj
        b = tl.load(b_ptr,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0,
                    dtype=tl.float16)

        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) → fp32

    # ------------------------------------------------------------------
    # Write tile back to C
    # ------------------------------------------------------------------
    c_ptr = C + batch_id * stride_cm + i[:, None] * stride_ci + j[None, :] * stride_cj
    tl.store(c_ptr, acc,
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Custom forward kernel – entry point used by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor[B, N, N, dim]   (float32)
            - mask         : torch.Tensor[B, N, N] or None
            - weights      : dict of model parameters
            - config       : dict with keys
                * dim        – input feature dimension
                * hidden_dim – hidden dimension used in the cubic contraction
                * nomask     – bool, True ⇒ ignore mask even if provided

    Returns:
        torch.Tensor[B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim        = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask     = config.get("nomask", True)
    eps        = 1e-5

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32, stable)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )   # [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------
    # 2) FP16 projection + sigmoid gates
    # ------------------------------------------------------------------
    x_h = x.half()                      # [B, N, N, dim] (fp16)

    # Linear weights in fp16 (bias‑free)
    left_proj_w  = weights["left_proj.weight"].half()
    right_proj_w = weights["right_proj.weight"].half()
    left_gate_w  = weights["left_gate.weight"].half()
    right_gate_w = weights["right_gate.weight"].half()
    out_gate_w   = weights["out_gate.weight"].half()

    left  = F.linear(x_h, left_proj_w)          # [B, N, N, H]
    right = F.linear(x_h, right_proj_w)         # [B, N, N, H]

    left_gate  = torch.sigmoid(F.linear(x_h, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x_h, out_gate_w))

    # Apply gates
    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 3) Optional mask – broadcast over the hidden channel
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask: [B, N, N] → [B, N, N, 1] (fp16)
        mask_exp = mask.to(left.dtype).unsqueeze(-1)
        left  = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4) Cubic contraction:  out[b, h, i, j] = Σₖ left[b, i, k, h] * right[b, j, k, h]
    #    Implemented as a batched GEMM with A @ Bᵀ.
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape                     # batch size & sequence length

    # Move hidden dim to the second axis and make contiguous → (B, H, N, N)
    left_perm  = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    M_total = B * hidden_dim                     # number of independent GEMMs
    left_flat  = left_perm.view(M_total, N, N)   # (M, N, N)   fp16
    right_flat = right_perm.view(M_total, N, N)  # (M, N, N)   fp16

    # Output accumulator in FP32
    out_flat = torch.empty((M_total, N, N), dtype=torch.float32, device=device)

    # --------------------------------------------------------------
    # Kernel launch
    # --------------------------------------------------------------
    # Strides (int64)
    stride_am = left_flat.stride(0)
    stride_ai = left_flat.stride(1)
    stride_ak = left_flat.stride(2)

    stride_bm = right_flat.stride(0)
    stride_bj = right_flat.stride(1)   # row (j) stride
    stride_bk = right_flat.stride(2)   # col (k) stride

    stride_cm = out_flat.stride(0)
    stride_ci = out_flat.stride(1)
    stride_cj = out_flat.stride(2)

    # Grid dimensions: (M_total, tiles_i, tiles_j)
    grid = (
        M_total,
        (N + 127) // 128,   # generous upper bound – actual BLOCK values are chosen by autotune
        (N + 127) // 128,
    )

    batched_tri_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bj,
        stride_bk,
        stride_cm,
        stride_ci,
        stride_cj,
        # The following kwargs are overridden by @autotune when possible.
        BLOCK_M=128,
        BLOCK_N=128,
        BLOCK_K=64,
    )

    # Reshape back to original 4‑D layout: [B, N, N, H] (float32)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 5) Output LayerNorm, gating, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate.float()                 # broadcast over H
    output = F.linear(out, weights["to_out.weight"])   # back to dim, bias‑free

    return output
```
