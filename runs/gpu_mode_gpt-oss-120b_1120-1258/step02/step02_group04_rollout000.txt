"""
Highly‑optimized Triton implementation of the *outgoing* TriMul operator
(AlphaFold‑3 style).

Algorithm
---------
1. Layer‑norm the 4‑D input (`[B,N,N,dim]`) in FP32.
2. Cast to FP16 and apply the four linear projections (left/right + gates).
   Optional masking is fused by an element‑wise multiply.
3. Fuse the N³ contraction
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   as a **batched GEMM**  C = A @ B  where for each head `h`
   A = left[b,:,:,h] (M=N, K=N) and B = right[b,:,:,h] (K=N, N=N).
   The kernel processes tiles (BLOCK_M×BLOCK_N) and streams K‑tiles
   (BLOCK_K) through shared memory, using FP16 for the matrix elements
   and FP32 for the accumulator (Tensor‑cores on H100).
4. Reshape back to `[B,N,N,hidden_dim]`, apply an output LayerNorm,
   the output gate and a final linear projection back to `dim`.

Only the N³ contraction is written in Triton – the surrounding
operations use PyTorch’s highly‑optimized kernels.  Block sizes are
chosen at run‑time (64×64 for small sequences, 128×128 otherwise) to
hit the sweet spot of occupancy vs. launch overhead on H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Batched GEMM kernel: C[b] = A[b] @ B[b]   (A: M×K, B: K×N)
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr,  # fp16 * (batch, M, K)
    b_ptr,  # fp16 * (batch, K, N)
    c_ptr,  # fp16 * (batch, M, N)
    M, N, K,                      # matrix sizes (int32)
    stride_az, stride_am, stride_ak,  # strides for A
    stride_bz, stride_bk, stride_bn,  # strides for B
    stride_cz, stride_cm, stride_cn,  # strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched half‑precision GEMM with FP32 accumulation.
    Grid: (batch*tiles_M, tiles_N)
    """
    pid0 = tl.program_id(0)          # batch_id * tiles_M + tile_m
    pid1 = tl.program_id(1)          # tile_n

    # -------------------------------------------------------------
    # Decode batch index and tile coordinates
    # -------------------------------------------------------------
    num_tiles_m = tl.cdiv(M, BLOCK_M)
    batch_id   = pid0 // num_tiles_m
    tile_m     = pid0 %  num_tiles_m
    tile_n     = pid1

    # -------------------------------------------------------------
    # Offsets inside the current tile
    # -------------------------------------------------------------
    offs_m = tile_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = tile_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Base pointers for this tile (no K‑loop offset yet)
    a_ptrs = a_ptr + batch_id * stride_az \
                  + offs_m[:, None] * stride_am \
                  + offs_k[None, :] * stride_ak
    b_ptrs = b_ptr + batch_id * stride_bz \
                  + offs_k[:, None] * stride_bk \
                  + offs_n[None, :] * stride_bn

    # -------------------------------------------------------------
    # Accumulator (FP32 for numerical safety)
    # -------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------------------------------------------------
    # Main K‑loop
    # -------------------------------------------------------------
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        # Load tiles with out‑of‑bounds masking
        mask_a = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        mask_b = (offs_k[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=mask_a, other=0.0)   # [BLOCK_M, BLOCK_K] fp16
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)   # [BLOCK_K, BLOCK_N] fp16

        # Tensor‑core dot (fp16×fp16 → fp32)
        acc += tl.dot(a, b)

        # Advance K‑dimension pointers
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # -------------------------------------------------------------
    # Write result back (store as fp16)
    # -------------------------------------------------------------
    c_ptrs = c_ptr + batch_id * stride_cz \
                    + offs_m[:, None] * stride_cm \
                    + offs_n[None, :] * stride_cn
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)


def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask         : torch.Tensor [B, N, N]      (bool/float) or None
        - weights      : dict of model parameters (float32)
        - config       : dict with keys {"dim", "hidden_dim", "nomask"}

    Returns
    -------
    torch.Tensor : [B, N, N, dim] (float32)
    """
    # -----------------------------------------------------------------
    # Unpack
    # -----------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim        = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask     = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1) Input LayerNorm (FP32 for stability)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # [B,N,N,dim]  float32

    # -----------------------------------------------------------------
    # 2) Cast to FP16 for the heavy work and prepare weights
    # -----------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # Linear weights → FP16 (no bias)
    left_proj_w  = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w  = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w   = weights["out_gate.weight"].to(torch.float16)

    # -----------------------------------------------------------------
    # 3) Projections
    # -----------------------------------------------------------------
    left  = F.linear(x_fp16, left_proj_w)    # [B,N,N,hidden]
    right = F.linear(x_fp16, right_proj_w)   # [B,N,N,hidden]

    # -----------------------------------------------------------------
    # 4) Optional mask (broadcast on hidden dim)
    # -----------------------------------------------------------------
    if not nomask and mask is not None:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)   # [B,N,N,1]
        left  = left * mask_fp16
        right = right * mask_fp16

    # -----------------------------------------------------------------
    # 5) Gating scalars (sigmoid)
    # -----------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_fp16, left_gate_w))   # [B,N,N,hidden]
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_w))  # [B,N,N,hidden]
    out_gate   = torch.sigmoid(F.linear(x_fp16, out_gate_w))    # [B,N,N,hidden]

    left  = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 6) N³ contraction via batched GEMM (A @ B)
    # -----------------------------------------------------------------
    B, N, _, _ = left.shape   # [B, N, N, hidden]

    # Re‑order to (batch*head, M, K) / (batch*head, K, N)
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 2, 1).contiguous().view(B * hidden_dim, N, N)

    out_mat = torch.empty_like(left_mat, dtype=torch.float16, device=device)

    # -----------------------------------------------------------------
    # Choose tile sizes based on sequence length (empirically good)
    # -----------------------------------------------------------------
    if N <= 256:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 64
        num_warps = 4
    else:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64
        num_warps = 8

    grid_m = triton.cdiv(N, BLOCK_M)   # tiles along M dimension
    grid_n = triton.cdiv(N, BLOCK_N)   # tiles along N dimension

    # First grid dimension packs (batch*head) * tiles_M
    grid = (B * hidden_dim * grid_m, grid_n)

    batched_matmul_kernel[grid](
        a_ptr=left_mat,
        b_ptr=right_mat,
        c_ptr=out_mat,
        M=N,
        N=N,
        K=N,
        stride_az=left_mat.stride(0),
        stride_am=left_mat.stride(1),
        stride_ak=left_mat.stride(2),
        stride_bz=right_mat.stride(0),
        stride_bk=right_mat.stride(1),
        stride_bn=right_mat.stride(2),
        stride_cz=out_mat.stride(0),
        stride_cm=out_mat.stride(1),
        stride_cn=out_mat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=num_warps,
        num_stages=3,               # good latency hiding on H100
    )

    # -----------------------------------------------------------------
    # 7) Reshape back to [B, N, N, hidden] (float32)
    # -----------------------------------------------------------------
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)

    # -----------------------------------------------------------------
    # 8) Output LayerNorm, out‑gate and final projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # Apply the output gate (broadcast over hidden dim)
    out = out * out_gate.to(torch.float32)

    # Final linear projection back to `dim`
    out = F.linear(out, weights["to_out.weight"])   # bias = False

    return out
```
