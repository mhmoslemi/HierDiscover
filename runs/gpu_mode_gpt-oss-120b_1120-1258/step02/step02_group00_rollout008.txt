"""
Highly‑optimized Triton implementation of the “outgoing” TriMul operator
used in AlphaFold‑3 (and related models).

Algorithm (all tensors live on a CUDA device):
1️⃣ Row‑wise LayerNorm on the input channel *dim* (float32 reduction,
    fp16 output) – Triton kernel.
2️⃣ Fuse the five linear projections (left/right + three gates) into a
    single fp16 GEMM (torch.nn.functional.linear) by concatenating the
    weight matrices.
3️⃣ If a mask is present, fuse mask‑application and sigmoid‑gating for
    *left* and *right* projections in one Triton kernel:
       left  = (proj_left  * sigmoid(gate_left ))  * mask
       right = (proj_right * sigmoid(gate_right))  * mask
   The output‑gate is sigmoided with a fast torch call.
4️⃣ Batched GEMM (B·H , N , N) on fp16 tensors using cuBLAS:
       out_h[i,j] = Σ_k left_h[i,k] * right_h[j,k]
   This is exactly a strided‑batched matrix‑multiply.
5️⃣ Row‑wise LayerNorm on the hidden dimension *H* (again Triton, fp16
   output).
6️⃣ In‑place multiply by the output‑gate, then a final fp16 linear
   projection back to *dim*.  The result is cast to float32 before
   returning.

The implementation reduces kernel launches (two Triton kernels instead of
many) and eliminates several intermediate copies (mask+gate fusion,
fp16‑only LayerNorm outputs).  It works for all configurations listed
in the problem statement and supports optional masking.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton Row‑wise LayerNorm (fp32 reduction, fp16 output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_half_kernel(
    x_ptr,          # *const float*   [total_rows, dim]
    y_ptr,          # *mut   half*    [total_rows, dim]
    weight_ptr,     # *const float*   [dim]
    bias_ptr,       # *const float*   [dim]
    total_rows,     # int (runtime)
    dim: tl.constexpr,    # int (compile‑time)
    eps: tl.constexpr,    # float (compile‑time)
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """Row‑wise LayerNorm for a 2‑D tensor, producing fp16 output."""
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)                 # [BLOCK_M]
    mask_m = offs_m < total_rows

    # -----------------------------------------------------------------
    # 1️⃣ Compute mean & variance (float32 accumulation)
    # -----------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)           # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]  # [M, D]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                                      # fp32

        sum_val += tl.sum(vals, axis=1)                        # [M]
        sum_sq  += tl.sum(vals * vals, axis=1)                 # [M]

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                                 # 1 / sqrt(var+eps)

    # -----------------------------------------------------------------
    # 2️⃣ Normalize, apply γ/β and write fp16 result
    # -----------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]          # (x-μ)/σ

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # fp32
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # fp32

        out = norm * w[None, :] + b[None, :]                    # fp32
        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out.to(tl.float16), mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Triton kernel that fuses mask‑application and sigmoid‑gate for left/right
# ----------------------------------------------------------------------
@triton.jit
def gate_mask_fused_kernel(
    left_proj_ptr, left_gate_ptr,          # *const half* [total_rows, hidden]
    right_proj_ptr, right_gate_ptr,        # *const half* [total_rows, hidden]
    mask_ptr,                              # *const float* [total_rows]
    left_out_ptr, right_out_ptr,           # *mut half*    [total_rows, hidden]
    total_rows: tl.constexpr,
    hidden: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """Apply sigmoid to gates, mask the projections and write fp16 results."""
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)               # [BLOCK_M]
    mask_m = offs_m < total_rows

    # Load mask (float32) once per row
    mask_val = tl.load(mask_ptr + offs_m, mask=mask_m, other=0.0)   # [BLOCK_M]
    mask_val = mask_val[:, None]                                   # broadcast

    for d in range(0, tl.cdiv(hidden, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)               # [BLOCK_D]
        mask_d = offs_d < hidden

        # ----- left -----
        lp_ptr = left_proj_ptr + offs_m[:, None] * hidden + offs_d[None, :]
        lg_ptr = left_gate_ptr  + offs_m[:, None] * hidden + offs_d[None, :]
        lp = tl.load(lp_ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)
        lg = tl.load(lg_ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)
        lgate = 1.0 / (1.0 + tl.exp(-lg))                         # sigmoid

        # ----- right -----
        rp_ptr = right_proj_ptr + offs_m[:, None] * hidden + offs_d[None, :]
        rg_ptr = right_gate_ptr + offs_m[:, None] * hidden + offs_d[None, :]
        rp = tl.load(rp_ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)
        rg = tl.load(rg_ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)
        rgate = 1.0 / (1.0 + tl.exp(-rg))

        # Apply mask (broadcasted) and gating
        left_res  = lp * lgate * mask_val
        right_res = rp * rgate * mask_val

        # Store results as half
        lo_ptr = left_out_ptr  + offs_m[:, None] * hidden + offs_d[None, :]
        ro_ptr = right_out_ptr + offs_m[:, None] * hidden + offs_d[None, :]
        tl.store(lo_ptr, left_res.to(tl.float16),  mask=mask_m[:, None] & mask_d[None, :])
        tl.store(ro_ptr, right_res.to(tl.float16), mask=mask_m[:, None] & mask_d[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (float32 or bool) or None
            - weights_dict : dict with the exact keys used in the reference model
            - config_dict  : {"dim": int, "hidden_dim": int, "nomask": bool, ...}

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    torch.cuda.empty_cache()                     # ensure clean start
    # ------------------------------------------------------------------
    # 1️⃣ Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]               # input channel dimension (C)
    hidden = config["hidden_dim"]     # projection dimension (H)
    eps = 1e-5
    nomask = config.get("nomask", True)   # True → no mask

    # ------------------------------------------------------------------
    # 2️⃣ Prepare contiguous tensors
    # ------------------------------------------------------------------
    x = input_tensor.contiguous()
    B, N, _, _ = x.shape               # x: [B, N, N, dim]
    total_rows = B * N * N              # rows for a flattened view

    # ------------------------------------------------------------------
    # 3️⃣ First LayerNorm (float32->fp16)
    # ------------------------------------------------------------------
    x_flat = x.view(total_rows, dim)                   # [M, dim] fp32
    x_norm_fp16 = torch.empty(total_rows, dim,
                              dtype=torch.float16,
                              device=device)   # fp16 buffer

    BLOCK_M = 256      # rows per program
    BLOCK_D = 32       # channel tile

    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_half_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    # No explicit sync needed; subsequent kernels will depend on the data.

    # ------------------------------------------------------------------
    # 4️⃣ Fused linear projections for left/right + three gates
    # ------------------------------------------------------------------
    # Concatenate the five weight matrices (shape: (5*H, dim))
    concat_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).half().contiguous()                               # (5*H, dim)

    # Single GEMM (fp16) → shape (total_rows, 5*hidden)
    proj_all = F.linear(x_norm_fp16, concat_weight)     # fp16

    # Split back to the five components
    left_proj, right_proj, left_gate, right_gate, out_gate = torch.split(
        proj_all, hidden, dim=1
    )                                                   # each [M, H] fp16

    # ------------------------------------------------------------------
    # 5️⃣ Optional mask + sigmoid gating (fused Triton kernel)
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] → flatten → float32 (0/1)
        mask_flat = mask_tensor.contiguous().view(total_rows).to(torch.float32)
        # Output buffers for gated left/right
        left_gated = torch.empty_like(left_proj)
        right_gated = torch.empty_like(right_proj)

        grid_mask = (triton.cdiv(total_rows, BLOCK_M),)
        gate_mask_fused_kernel[grid_mask](
            left_proj,
            left_gate,
            right_proj,
            right_gate,
            mask_flat,
            left_gated,
            right_gated,
            total_rows,
            hidden,
            BLOCK_M=BLOCK_M,
            BLOCK_D=BLOCK_D,
        )
        left_fp16 = left_gated
        right_fp16 = right_gated
    else:
        # No mask: just sigmoid‑gate the projections with regular torch ops
        left_fp16 = left_proj * torch.sigmoid(left_gate)
        right_fp16 = right_proj * torch.sigmoid(right_gate)

    # Sigmoid for the output‑gate (always needed)
    out_gate_fp16 = torch.sigmoid(out_gate)             # [M, H] fp16

    # ------------------------------------------------------------------
    # 6️⃣ Reshape for the triangular multiplication
    # ------------------------------------------------------------------
    # From [M, H] → [B, N, N, H]
    left = left_fp16.view(B, N, N, hidden)
    right = right_fp16.view(B, N, N, hidden)
    out_gate = out_gate_fp16.view(B, N, N, hidden)

    # Permute to [B, H, N, N] so that we can use a single strided‑batched GEMM
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 2, 1).contiguous() # [B, H, N, N] (k‑dim transposed)

    BH = B * hidden
    left_mat = left_perm.view(BH, N, N)                 # [BH, N, N] fp16
    right_mat = right_perm.view(BH, N, N)               # [BH, N, N] fp16

    # ------------------------------------------------------------------
    # 7️⃣ Batched matrix multiply (fp16) → out shape [BH, N, N]
    # ------------------------------------------------------------------
    out_mat = torch.bmm(left_mat, right_mat)            # fp16

    # Reshape back to [B, N, N, H]
    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H]

    # ------------------------------------------------------------------
    # 8️⃣ Second LayerNorm over hidden dimension (fp16 output)
    # ------------------------------------------------------------------
    total_rows_h = B * N * N
    out_flat = out.view(total_rows_h, hidden)           # [M_h, H] fp16
    out_norm_fp16 = torch.empty_like(out_flat)          # fp16 buffer

    grid2 = (triton.cdiv(total_rows_h, BLOCK_M),)
    layernorm_half_kernel[grid2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows_h,
        hidden,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    # ------------------------------------------------------------------
    # 9️⃣ Apply final output gate (in‑place) and project back to `dim`
    # ------------------------------------------------------------------
    # out_gate already has shape [B, N, N, H] → flatten for broadcast
    out_gate_flat = out_gate.view(total_rows_h, hidden)
    # In‑place element‑wise product
    out_norm_fp16.mul_(out_gate_flat)

    # Final linear projection H → dim  (fp16)
    to_out_weight_fp16 = weights["to_out.weight"].half().contiguous()
    final_fp16 = F.linear(out_norm_fp16, to_out_weight_fp16)  # [M_h, dim] fp16

    # Cast back to float32 and reshape to the original 4‑D shape
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
