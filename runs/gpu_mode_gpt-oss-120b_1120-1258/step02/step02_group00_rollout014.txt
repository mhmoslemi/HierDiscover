@triton.jit
def layernorm_gate_kernel(x_ptr, gate_ptr, y_ptr, weight_ptr, bias_ptr, total_rows, hidden_dim, eps, BLOCK_M, BLOCK_D):
  pid = tl.program_id(0)
  row_start = pid * BLOCK_M
  offs_m = row_start + tl.arange(0, BLOCK_M)
  mask_m = offs_m < total_rows

  sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
  sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

  # first pass: compute mean and var
  for d in range(0, tl.cdiv(hidden_dim, BLOCK_D)):
    offs_d = d*BLOCK_D + tl.arange(0, BLOCK_D)
    mask_d = offs_d < hidden_dim
    ptrs = x_ptr + offs_m[:, None] * hidden_dim + offs_d[None, :] # flatten
    vals = tl.load(ptrs, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)
    sum_val += tl.sum(vals, axis=1)
    sum_sq += tl.sum(vals * vals, axis=1)

  mean = sum_val / hidden_dim
  var = sum_sq / hidden_dim - mean*mean
  rstd = tl.rsqrt(var + eps)

  for d in range(0, tl.cdiv(hidden_dim, BLOCK_D)):
    offs_d = d*BLOCK_D + tl.arange(0, BLOCK_D)
    mask_d = offs_d < hidden_dim
    base_ptr = x_ptr + offs_m[:, None] * hidden_dim + offs_d[None, :]
    x_vals = tl.load(base_ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)
    gate_vals = tl.load(gate_ptr + offs_m[:, None] * hidden_dim + offs_d[None, :],
                        mask=mask_m[:, None] & mask_d[None, :],
                        other=0.0).to(tl.float32)

    norm = (x_vals - mean[:, None]) * rstd[:, None]
    w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # float
    b = tl.load(bias_ptr + offs_d, mask=mask_d, other=0.0)
    out = (norm * w + b) * gate_vals

    # cast to half and store
    tl.store(y_ptr + offs_m[:, None] * hidden_dim + offs_d[None, :],
             out.to(tl.float16),
             mask=mask_m[:, None] & mask_d[None, :])

                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)  # cast to float32
        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # second pass: write normalized half values
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]

                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr + offs_d, mask=mask_d, other=0.0)

def layernorm_gate_fp16_kernel(x_ptr, gate_ptr, y_ptr,
                               weight_ptr, bias_ptr,
                               total_rows, dim: tl.constexpr, eps: tl.constexpr,

    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)

    # second pass: normalize, apply weight/bias, then element‑wise gate
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        base_ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]

        b = tl.load(bias_ptr + offs_d, mask=mask_d, other=0.0)

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],

                 mask=mask_m[:, None] & mask_d[None, :])

def custom_kernel(data):
    """
    Triton‑accelerated "outgoing" TriMul forward pass.
    The heavy LayerNorms are fused kernels; all heavy matrix‑multiplications
    (projections, pairwise update, final projection) use fp16 Tensor‑Core GEMM.
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = torch.float16

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N

    # ------------------------------------------------------------------
    # 1) First LayerNorm (input is fp32, output half)
    # ------------------------------------------------------------------
    x_norm = torch.empty((total_rows, dim), dtype=dtype, device=device)

    BLOCK_M = 128
    BLOCK_D = 32
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        input_tensor.view(total_rows, dim),
        x_norm,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()

    # ------------------------------------------------------------------
    # 2) Fused linear projections + gates
    # ------------------------------------------------------------------
    # Concatenate weight matrices for the 5 projections and cast to half
    concat_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).half().contiguous()  # (5*H, dim)

    proj_all = F.linear(x_norm, concat_weight)  # [total_rows, 5*H] half
    left_proj, right_proj, left_gate, right_gate, out_gate = torch.split(
        proj_all, hidden_dim, dim=1
    )

    # ------------------------------------------------------------------
    # 3) Optional mask
    # ------------------------------------------------------------------
    if not nomask:
        mask_flat = mask_tensor.contiguous().view(total_rows, 1).to(dtype)
        left_proj = left_proj * mask_flat
        right_proj = right_proj * mask_flat

    # ------------------------------------------------------------------
    # 4) Apply sigmoid gates (still fp16)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate1 = torch.sigmoid(out_gate)  # keep separate for later

    left = left_proj * left_gate   # [total_rows, H]
    right = right_proj * right_gate

    # Reshape to 4‑D tensors
    left = left.view(B, N, N, hidden_dim)
    right = right.view(B, N, N, hidden_dim)
    out_gate1 = out_gate1.view(B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 5) Pairwise update: batched GEMM
    # ------------------------------------------------------------------
    # Move hidden dimension to batch axis and make matrices contiguous
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    BH = B * hidden_dim
    left_mat = left_perm.view(BH, N, N)    # [B*H, N, N]
    right_mat = right_perm.view(BH, N, N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [B*H, N, N] half
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 6) Second LayerNorm + output gate (fused Triton kernel)
    # ------------------------------------------------------------------
    total_rows_h = B * N * N
    out_flat = out.view(total_rows_h, hidden_dim)
    out_gate_flat = out_gate1.view(total_rows_h, hidden_dim)
    gated_norm = torch.empty_like(out_flat)  # half

    grid2 = (triton.cdiv(total_rows_h, BLOCK_M),)
    layernorm_gate_fp16_kernel[grid2](
        out_flat,
        out_gate_flat,
        gated_norm,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows_h,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()

    # ------------------------------------------------------------------
    # 7) Final linear projection back to `dim`
    # ------------------------------------------------------------------
    to_out_weight = weights["to_out.weight"].half()
    final_half = F.linear(gated_norm, to_out_weight)  # [total_rows_h, dim] half
    final = final_half.view(B, N, N, dim).to(torch.float32)

    return final

    # 6) Second LayerNorm + output gate (fused Triton kernel)
    # --------------------------------------------------------------

    layernorm_gate_fp16_kernel[grid2](

    final = final_half.view(B, N, N, dim).to(torch.float32)
