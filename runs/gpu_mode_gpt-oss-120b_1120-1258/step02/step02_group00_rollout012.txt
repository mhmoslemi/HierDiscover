```python
@triton.jit
def layernorm_kernel(
    x_ptr, y_ptr,
    weight_ptr, bias_ptr,
    total_rows,
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # compute sum and sum_sq
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr + offs_d, mask=mask_d, other=0.0)

        out = norm * w[None, :] + b[None, :]   # w,b are fp32; result fp32
        out_half = out.to(tl.float16)

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_half, mask=mask_m[:, None] & mask_d[None, :])
```

"""
custom_kernel – “outgoing” TriMul (AlphaFold‑3)

Algorithm
---------
1️⃣  Row‑wise LayerNorm on the last dimension (dim) → half‑precision.
2️⃣  Concatenate the five weight matrices (left/right + three gates) and
    perform a single fused GEMM (fp16) on the normalized tensor.
3️⃣  (Optional) apply the pairwise mask to the left/right projections.
4️⃣  Compute sigmoid gates (fp16) and element‑wise gating.
5️⃣  Batched matrix multiplication per hidden channel:
        out_h[i,j] = Σ_k left_h[i,k] * right_h[j,k]
    Implemented as a batched GEMM (torch.bmm) on tensors of shape
    (B·H, N, N) in fp16.
6️⃣  Row‑wise LayerNorm over the hidden dimension (H) → half‑precision.
7️⃣  Multiply by the output gate, then a final linear projection (fp16) back to
    the original channel size (dim).  Cast the result to float32.

Only the two LayerNorms are written in Triton; all dense linear algebra uses
high‑throughput cuBLAS/Tensor‑Core kernels.  The implementation avoids
unnecessary copies (no extra float‑32↔float‑16 casts) and reduces kernel‑launch
overhead by using a large BLOCK_M (1024) for the LayerNorm kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton row‑wise LayerNorm (outputs fp16, accumulates in fp32)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # *const float*   [total_rows, dim]   (input fp32)
    y_ptr,          # *mut   half*   [total_rows, dim]   (output fp16)
    weight_ptr,     # *const float*   [dim]               (γ)
    bias_ptr,       # *const float*   [dim]               (β)
    total_rows,     # int (runtime)
    dim: tl.constexpr,       # int (compile‑time)
    eps,            # float (runtime)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # cols per tile
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M

    # -----------------------------------------------------------------
    # 1️⃣ Compute mean & variance (float32)
    # -----------------------------------------------------------------
    offs_m = row_start + tl.arange(0, BLOCK_M)            # [BLOCK_M]
    mask_m = offs_m < total_rows

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < dim

        # Load a tile [BLOCK_M, BLOCK_D]
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                                   # fp32

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                            # 1 / sqrt(var+eps)

    # -----------------------------------------------------------------
    # 2️⃣ Write normalized values (fp16)
    # -----------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]   # (x-μ)/σ

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β

        out_fp32 = norm * w[None, :] + b[None, :]
        out_fp16 = out_fp32.to(tl.float16)

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


def custom_kernel(data):
    """
    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32, on CUDA)
    """
    # ------------------------------------------------------------------
    # 0️⃣ Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]               # input channel dimension C
    hidden_dim = config["hidden_dim"] # projection dimension H
    eps = 1e-5
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape               # [B, N, N, C]
    total_rows = B * N * N                         # rows for the flattened view

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    x_flat = input_tensor.view(total_rows, dim).contiguous()          # fp32
    x_norm = torch.empty((total_rows, dim), dtype=torch.float16,
                         device=device)                               # fp16 output

    BLOCK_M = 1024          # larger block → fewer program IDs
    BLOCK_D = 32

    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"].contiguous(),
        weights["norm.bias"].contiguous(),
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear projections + gates (all in FP16)
    # ------------------------------------------------------------------
    # Concatenate the five weight matrices (shape: (5*H, dim))
    concat_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16).contiguous()                     # (5*H, dim)

    # Single GEMM: [total_rows, dim] @ [dim, 5*H]^T → [total_rows, 5*H]
    proj_all = F.linear(x_norm, concat_weight)           # fp16

    # Split back
    split_tensors = torch.split(proj_all, hidden_dim, dim=1)
    left_proj, right_proj, left_gate, right_gate, out_gate = split_tensors

    # ------------------------------------------------------------------
    # 2a️⃣ Optional pairwise mask (applied after projection)
    # ------------------------------------------------------------------
    if not nomask:
        mask_f = mask_tensor.view(total_rows, 1).to(torch.float16)
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # ------------------------------------------------------------------
    # 2b️⃣ Sigmoid gates (still FP16)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    # Element‑wise gating
    left = left_proj * left_gate          # [total_rows, H]
    right = right_proj * right_gate       # [total_rows, H]

    # ------------------------------------------------------------------
    # 3️⃣ Reshape to 4‑D tensors
    # ------------------------------------------------------------------
    left = left.view(B, N, N, hidden_dim)          # [B, N, N, H]
    right = right.view(B, N, N, hidden_dim)        # [B, N, N, H]
    out_gate = out_gate.view(B, N, N, hidden_dim) # [B, N, N, H]

    # ------------------------------------------------------------------
    # 4️⃣ Triangular multiplicative update (batched GEMM)
    # ------------------------------------------------------------------
    # Permute to (B, H, N, N) and transpose right on the K‑axis
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 2, 1).contiguous() # [B, H, N, N]

    BH = B * hidden_dim
    left_mat = left_perm.view(BH, N, N)      # [B·H, N, N]
    right_mat = right_perm.view(BH, N, N)   # [B·H, N, N]

    # Batched matrix‑multiply, fp16, Tensor‑Core
    out_mat = torch.bmm(left_mat, right_mat)   # [B·H, N, N]

    # Back to [B, N, N, H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm over hidden dimension (fp16 output)
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, hidden_dim).contiguous()      # [rows, H]
    out_norm = torch.empty((total_rows, hidden_dim), dtype=torch.float16,
                           device=device)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm,
        weights["to_out_norm.weight"].contiguous(),
        weights["to_out_norm.bias"].contiguous(),
        total_rows,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    out_norm = out_norm.view(B, N, N, hidden_dim)   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 6️⃣ Gating + final projection back to dim
    # ------------------------------------------------------------------
    gated = out_norm * out_gate                      # fp16

    gated_flat = gated.view(total_rows, hidden_dim)  # [rows, H]
    to_out_w_fp16 = weights["to_out.weight"].to(torch.float16).contiguous()
    final_fp16 = F.linear(gated_flat, to_out_w_fp16)  # fp16 → [rows, dim]

    # Cast to float32 for the official output type
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
