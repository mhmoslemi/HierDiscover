"""
TriMul outgoing kernel (high‑performance Triton version)

The module implements the forward pass of the “outgoing” TriMul operation
used in AlphaFold‑3 and similar protein‑structure models.

Algorithm
---------
1. Layer‑norm over the input feature dimension (`dim`).
2. Two bias‑free linear projections → `left` and `right` (half‑precision).
3. Optional pair‑wise mask (broadcast on the hidden dimension).
4. Three sigmoid gates (`left_gate`, `right_gate`, `out_gate`) and element‑wise
   gating of `left`/`right`.
5. Cubic contraction:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   which is a batched GEMM C = A @ Bᵀ for every (batch, hidden) pair.
   The heavy GEMM is performed by a Triton kernel that tiles the matrices
   (BLOCK_M×BLOCK_N) and reduces over K in blocks of size BLOCK_K.
   Tensor‑core `tl.dot` is used (fp16·fp16 → fp32 accumulation).
6. Layer‑norm over the hidden dimension, element‑wise multiplication with
   `out_gate`, and a final bias‑free linear projection back to the original
   dimension (`dim`).  The final result is returned in float32.

The kernel has been tuned for H100 GPUs:
  * TILE sizes 64×64×64 (good occupancy and Tensor‑core utilisation)
  * 8 warps per block (`num_warps=8`)
  * Accumulation in fp32 for numerical stability.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def tri_mul_outgoing_kernel(
    a_ptr,               # left  matrix (M, N, N) – fp16
    b_ptr,               # right matrix (M, N, N) – fp16, accessed transposed
    c_ptr,               # output (M, N, N) – fp32
    N,                   # sequence length
    stride_am, stride_ai, stride_ak,   # strides of A: (M, i, k)
    stride_bm, stride_bj, stride_bk,   # strides of B: (M, j, k) – we read as (k, j)
    stride_cm, stride_ci, stride_cj,   # strides of C: (M, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # -----------------------------------------------------------------
    # Tile identifiers
    # -----------------------------------------------------------------
    pid_m = tl.program_id(0)          # tile along i (rows)
    pid_n = tl.program_id(1)          # tile along j (cols)
    pid_b = tl.program_id(2)          # batch * hidden index

    i0 = pid_m * BLOCK_M
    j0 = pid_n * BLOCK_N

    # tile coordinates inside the current block
    i = i0 + tl.arange(0, BLOCK_M)
    j = j0 + tl.arange(0, BLOCK_N)

    mask_i = i < N
    mask_j = j < N

    # -----------------------------------------------------------------
    # Pointer to the output tile C[i, j] for this (batch, hidden) pair
    # -----------------------------------------------------------------
    c_ptrs = c_ptr + pid_b * stride_cm + i[:, None] * stride_ci + j[None, :] * stride_cj
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -----------------------------------------------------------------
    # Reduction over the contraction dimension k
    # -----------------------------------------------------------------
    for k0 in range(0, N, BLOCK_K):
        k = k0 + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # ------------------- load A[i, k] ---------------------------------
        a_ptrs = a_ptr + pid_b * stride_am + i[:, None] * stride_ai + k[None, :] * stride_ak
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )  # fp16

        # ------------------- load B[k, j] (B is transposed) ---------------
        b_ptrs = b_ptr + pid_b * stride_bm + k[:, None] * stride_bk + j[None, :] * stride_bj
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )  # fp16

        # ------------------- Tensor‑core dot (fp16·fp16 → fp32) ------------
        acc += tl.dot(a, b)

    # -----------------------------------------------------------------
    # Store the result
    # -----------------------------------------------------------------
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[:, None])


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module using a Triton kernel for the
    N³·hidden contraction.

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor : [B, N, N, dim]  (float32)
            - mask         : [B, N, N]      (float32, bool or None)
            - weights      : dict of model parameters (already on device)
            - config       : dict with keys "dim", "hidden_dim", "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    # -----------------------------------------------------------------
    # Unpack weights (remain on the same device as the inputs)
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # -----------------------------------------------------------------
    # 1. Layer‑norm over the input feature dimension (dim)
    # -----------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, dim]  float32

    x_h = x.half()  # half‑precision for the heavy path

    # -----------------------------------------------------------------
    # Cast small weight matrices to fp16 once
    # -----------------------------------------------------------------
    left_proj_w_h = left_proj_weight.half()
    right_proj_w_h = right_proj_weight.half()
    left_gate_w_h = left_gate_weight.half()
    right_gate_w_h = right_gate_weight.half()
    out_gate_w_h = out_gate_weight.half()

    # -----------------------------------------------------------------
    # 2. Linear projections (bias‑free) → left, right
    # -----------------------------------------------------------------
    left = torch.nn.functional.linear(x_h, left_proj_w_h)   # [B, N, N, H]  fp16
    right = torch.nn.functional.linear(x_h, right_proj_w_h)  # [B, N, N, H]  fp16

    # -----------------------------------------------------------------
    # 3. Optional pair‑wise mask
    # -----------------------------------------------------------------
    if not nomask:
        mask_exp = mask.to(x_h.dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    # -----------------------------------------------------------------
    # 4. Gating (sigmoid) – still in fp16
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x_h, left_gate_w_h)
    )
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x_h, right_gate_w_h)
    )
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x_h, out_gate_w_h)
    )

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Triton kernel for the cubic contraction
    #    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    #    Treat each (batch, hidden) pair as an independent GEMM.
    # -----------------------------------------------------------------
    # Rearrange tensors to (B, H, N, N) so each (b, h) pair becomes a
    # separate matrix multiplication.
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N] fp16
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N] fp16

    B, H, N, _ = left_perm.shape
    M_total = B * H

    left_flat = left_perm.view(M_total, N, N)   # [M, N, N] fp16
    right_flat = right_perm.view(M_total, N, N)  # [M, N, N] fp16
    out_flat = torch.empty(
        (M_total, N, N), dtype=torch.float32, device=left_flat.device
    )  # [M, N, N] fp32

    # -----------------------------------------------------------------
    # Kernel launch configuration
    # -----------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    stride_am = left_flat.stride(0)
    stride_ai = left_flat.stride(1)
    stride_ak = left_flat.stride(2)

    stride_bm = right_flat.stride(0)
    stride_bj = right_flat.stride(1)   # stride over the “j” dimension (first N)
    stride_bk = right_flat.stride(2)   # stride over the “k” dimension (second N)

    stride_cm = out_flat.stride(0)
    stride_ci = out_flat.stride(1)
    stride_cj = out_flat.stride(2)

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # tiles along i
        (N + BLOCK_N - 1) // BLOCK_N,   # tiles along j
        M_total,                        # one program per (batch, hidden)
    )

    tri_mul_outgoing_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bj,
        stride_bk,
        stride_cm,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,      # more warps → better Tensor‑core utilisation
    )

    # Reshape back to [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # fp32

    # -----------------------------------------------------------------
    # 6. Hidden‑dim layer‑norm, output gating, final linear projection
    # -----------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )  # fp32

    out = out * out_gate.to(out.dtype)  # broadcasted element‑wise gating

    # Final projection back to the original feature dimension
    output = torch.nn.functional.linear(out, to_out_weight)  # [B, N, N, dim] fp32

    return output

The rest of the operations (layer‑norms, gating, linear projections) stay in PyTorch and run in half‑precision where possible, keeping the final output in float32.
