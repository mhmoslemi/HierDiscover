import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def batched_tri_gemm_kernel(
    A, B, C,
    N,
    stride_am, stride_ai, stride_ak,
    stride_bm, stride_bk, stride_bj,
    stride_cm, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    i0 = pid_m * BLOCK_M
    j0 = pid_n * BLOCK_N
    i = i0 + tl.arange(0, BLOCK_M)
    j = j0 + tl.arange(0, BLOCK_N)

    mask_i = i < N
    mask_j = j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_K)
    for k_iter in range(num_k):
        k0 = k_iter * BLOCK_K
        k = k0 + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # A: (BLOCK_M, BLOCK_K)
        a_ptr = A + batch_id * stride_am + i[:, None] * stride_ai + k[None, :] * stride_ak
        a = tl.load(a_ptr, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # B transposed: (BLOCK_K, BLOCK_N) referencing B[batch, k, j]
        b_ptr = B + batch_id * stride_bm + k[:, None] * stride_bk + j[None, :] * stride_bj
        b = tl.load(b_ptr, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) → (BLOCK_M, BLOCK_N)

    c_ptr = C + batch_id * stride_cm + i[:, None] * stride_ci + j[None, :] * stride_cj
    tl.store(c_ptr, acc, mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    """
    Outgoing TriMul forward pass accelerated by a Triton kernel.

    Arguments
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
        - mask: torch.Tensor of shape [B, N, N] (any dtype) or None
        - weights: dict of model parameters (all tensors are float32)
        - config: dict containing
            * "dim" (int) – input feature dim
            * "hidden_dim" (int) – projection dim
            * "nomask" (bool, optional) – whether mask is omitted

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    # --------------------------------------------------------------- #
    # 1. Input LayerNorm (float32)
    # --------------------------------------------------------------- #
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )   # [B, N, N, dim], float32

    # --------------------------------------------------------------- #
    # 2. FP16 projections and sigmoid gates
    # --------------------------------------------------------------- #
    x_h = x.half()   # cast once

    # bias‑free linear layers, weight only (no bias)
    left = F.linear(x_h, weights["left_proj.weight"].half())   # [B, N, N, H]
    right = F.linear(x_h, weights["right_proj.weight"].half())

    left_gate = torch.sigmoid(F.linear(x_h, weights["left_gate.weight"].half()))
    right_gate = torch.sigmoid(F.linear(x_h, weights["right_gate.weight"].half()))
    out_gate = torch.sigmoid(F.linear(x_h, weights["out_gate.weight"].half()))

    # apply gates
    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------- #

"""
Optimized outgoing TriMul forward pass (AlphaFold3).

The heavy O(N³·H) contraction is performed by a custom Triton kernel that
computes, for each hidden slice h of each batch b,

    out[b, h, i, j] = Σ_k left[b, i, k, h] * right[b, j, k, h]

using FP16 arithmetic with FP32 accumulation (Tensor‑core friendly). All
remaining operations (LayerNorm, projections, sigmoid gates, final projection)
are executed in PyTorch. The implementation supports optional masking and
runs efficiently on H100 GPUs.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_tri_gemm_kernel(
    A,                     # left  tensor (M, N, N)  – fp16
    B,                     # right tensor (M, N, N)  – fp16 (read transposed)
    C,                     # output tensor (M, N, N) – fp32
    N,                     # sequence length (scalar)
    stride_am, stride_ai, stride_ak,   # strides of A: batch, row(i), col(k)
    stride_bm, stride_bk, stride_bj,   # strides of B: batch, row(k), col(j)
    stride_cm, stride_ci, stride_cj,   # strides of C: batch, row(i), col(j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM: C = A @ Bᵀ for each independent slice.
    A and B are fp16, accumulation is fp32.
    """
    batch_id = tl.program_id(0)   # index over M = B * hidden_dim
    pid_m    = tl.program_id(1)   # tile row
    pid_n    = tl.program_id(2)   # tile col

    i0 = pid_m * BLOCK_M
    j0 = pid_n * BLOCK_N
    i = i0 + tl.arange(0, BLOCK_M)
    j = j0 + tl.arange(0, BLOCK_N)

    mask_i = i < N
    mask_j = j < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of k‑tiles
    num_k = tl.cdiv(N, BLOCK_K)

    for k_iter in range(num_k):
        k0 = k_iter * BLOCK_K
        k = k0 + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # A[batch, i, k]  → (BLOCK_M, BLOCK_K)
        a_ptr = A + batch_id * stride_am + i[:, None] * stride_ai + k[None, :] * stride_ak
        a = tl.load(
            a_ptr,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float16)

        # Bᵀ → B[batch, k, j]  (BLOCK_K, BLOCK_N)
        b_ptr = B + batch_id * stride_bm + k[:, None] * stride_bk + j[None, :] * stride_bj
        b = tl.load(
            b_ptr,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        ).to(tl.float16)

        # Tensor‑core dot, fp32 accumulation
        acc += tl.dot(a, b)

    # write tile back
    c_ptr = C + batch_id * stride_cm + i[:, None] * stride_ci + j[None, :] * stride_cj
    tl.store(
        c_ptr,
        acc,
        mask=mask_i[:, None] & mask_j[:, None],
    )


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module using a Triton kernel for the cubic
    contraction.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask        : torch.Tensor [B, N, N]       (any dtype, may be None)
        - weights     : dict of torch.Tensor model parameters
        - config      : dict with keys
            * "dim" (int) – input feature dimension
            * "hidden_dim" (int) – projection dimension
            * "nomask" (bool, optional) – True → ignore mask

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    # ------------------------------------------------------------------ #
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------ #
    x = F.layer_norm(

        bias=weights["norm.bias"],

    )  # [B, N, N, dim]  float32

    right_gate = torch.sigmoid(F.linear(x_h, weights["right_gate.weight"].half()))
    out_gate   = torch.sigmoid(F.linear(x_h, weights["out_gate.weight"].half()))

    left  = left * left_gate
    right = right * right_gate

    stride_ak = left_flat.stride(2)

    stride_bm = right_flat.stride(0)
    stride_bk = right_flat.stride(1)   # k‑dimension stride

    stride_cm = out_flat.stride(0)
    stride_ci = out_flat.stride(1)
    stride_cj = out_flat.stride(2)

    grid = (
        M_total,
        (N + BLOCK_M - 1) // BLOCK_M,

    out = out_flat.view(Bsz, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
