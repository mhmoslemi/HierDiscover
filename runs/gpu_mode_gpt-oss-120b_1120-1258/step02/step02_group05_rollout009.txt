@triton.jit
def tri_gemm_outgoing_kernel(A, B, C, N, M_total,
                              stride_a_batch, stride_a_i, stride_a_k,
                              stride_b_batch, stride_b_k, stride_b_j,
                              stride_c_batch, stride_c_i, stride_c_j,
                              BLOCK_M: tl.constexpr,
                              BLOCK_N: tl.constexpr,
                              BLOCK_K: tl.constexpr):
    batch_id = tl.program_id(0) # from 0..M_total-1
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    i = i0 + tl.arange(0, BLOCK_M)
    j = j0 + tl.arange(0, BLOCK_N)
    mask_i = i < N
    mask_j = j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k0 in range(0, N, BLOCK_K):
        k = k0 + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # Load A: shape (BLOCK_M, BLOCK_K) at (batch_id, i, k)
        a_ptr = A + batch_id * stride_a_batch + i[:, None] * stride_a_i + k[None, :] * stride_a_k
        a = tl.load(a_ptr, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        # Load B as transposed: B[batch_id, k, j]
        b_ptr = B + batch_id * stride_b_batch + k[:, None] * stride_b_k + j[None, :] * stride_b_j
        b = tl.load(b_ptr, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        acc += tl.dot(a, b)

    c_ptr = C + batch_id * stride_c_batch + i[:, None] * stride_c_i + j[None, :] * stride_c_j
    tl.store(c_ptr, acc, mask=mask_i[:, None] & mask_j[None, :])

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def tri_mul_outgoing_kernel(
    A,
    B,
    C,
    N,
    stride_am, stride_ai, stride_ak,
    stride_bm, stride_bk, stride_bj,
    stride_cm, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)   # index over M_total = B * hidden_dim
    pid_i = tl.program_id(1)      # tile row index
    pid_j = tl.program_id(2)      # tile col index

    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    i = i0 + tl.arange(0, BLOCK_M)
    j = j0 + tl.arange(0, BLOCK_N)

    mask_i = i < N
    mask_j = j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over reduction dimension k
    for k0 in range(0, N, BLOCK_K):
        k = k0 + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # Load A[batch_id, i, k]   shape (BLOCK_M, BLOCK_K)
        a_ptr = A + batch_id * stride_am + i[:, None] * stride_ai + k[None, :] * stride_ak
        a = tl.load(a_ptr, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # Load B transposed: B[batch_id, k, j]   shape (BLOCK_K, BLOCK_N)
        b_ptr = B + batch_id * stride_bm + k[:, None] * stride_bk + j[None, :] * stride_bj
        b = tl.load(b_ptr, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        acc += tl.dot(a, b)

    # Write back tile C[batch_id, i, j]
    c_ptr = C + batch_id * stride_cm + i[:, None] * stride_ci + j[None, :] * stride_cj
    tl.store(c_ptr, acc, mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    """
    Triton‑accelerated outgoing TriMul forward pass.

    1. Layer‑norm on the input (float32).
    2. Half‑precision linear projections and sigmoid gates.
    3. Optional masking of the gated tensors.
    4. Batched GEMM C = A @ Bᵀ computed by a Triton kernel that
       processes each (batch, hidden) pair as an independent GEMM.
    5. Output layer‑norm, output‑gate multiplication and final projection.

    All heavy cubic contraction work (O(N³·hidden_dim)) is performed in the
    Triton kernel. The remaining element‑wise work stays in PyTorch.
    """
    input_tensor, mask, weights, config = data

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    # 1. Input LayerNorm (float32)
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  float32

    # 2. Cast to fp16 for projections / gates
    x_h = x.half()       # [B,N,N,dim]  fp16

    # bias‑free linear projections (FP16 weights)
    left  = F.linear(x_h, weights["left_proj.weight"].half())   # [B,N,N, hidden_dim]
    right = F.linear(x_h, weights["right_proj.weight"].half())

    # gates (sigmoid, FP16)
    left_gate  = torch.sigmoid(F.linear(x_h, weights["left_gate.weight"].half()))
    right_gate = torch.sigmoid(F.linear(x_h, weights["right_gate.weight"].half()))
    out_gate   = torch.sigmoid(F.linear(x_h, weights["out_gate.weight"].half()))

    # apply elementwise gates
    left  = left * left_gate
    right = right * right_gate

    # 3. Optional mask (broadcast over hidden_dim)
    if not nomask:
        # mask shape [B,N,N] -> [B,N,N,1]
        mask_exp = mask.to(left.dtype).unsqueeze(-1)
        left = left * mask_exp
        right = right * mask_exp

    # 4. Batched GEMM using Triton kernel
    B_, N_, _, _ = left.shape  # batch, seq_len, seq_len, hidden_dim
    # move hidden_dim to second axis and make contiguous for efficient stride pattern
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    M_total = B_ * hidden_dim

    # shape (M_total, N, N) each in fp16
    left_flat = left_perm.view(M_total, N_, N_)
    right_flat = right_perm.view(M_total, N_, N_)
    out_flat = torch.empty((M_total, N_, N_), dtype=torch.float32, device=left_flat.device)

    # Block sizes tuned for H100 registers
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    stride_am = left_flat.stride(0)
    stride_ai = left_flat.stride(1)
    stride_ak = left_flat.stride(2)

    stride_bm = right_flat.stride(0)
    stride_bk = right_flat.stride(2)   # column (k) stride after transpose
    stride_bj = right_flat.stride(1)   # row (j) stride

    stride_cm = out_flat.stride(0)
    stride_ci = out_flat.stride(1)
    stride_cj = out_flat.stride(2)

    grid = (
        M_total,
        (N_ + BLOCK_M - 1) // BLOCK_M,
        (N_ + BLOCK_N - 1) // BLOCK_N,
    )
    tri_mul_outgoing_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N_,
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bk,
        stride_bj,
        stride_cm,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to [B, N, N, hidden_dim] (float32)
    out = out_flat.view(B_, hidden_dim, N_, N_).permute(0, 2, 3, 1).contiguous()

    # 5. Output LayerNorm, output gate and final projection
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate.float()
    output = F.linear(out, weights["to_out.weight"])

    return output

"""
Outgoing TriMul (AlphaFold‑3) – Triton‐accelerated cubic contraction.

Algorithm
---------
1.  Layer‑norm the input tensor (float32).
2.  Cast to fp16 and apply bias‑free linear projections + sigmoid gates.

3.  Optional mask (broadcast over the hidden dimension) is applied after the

    result.

        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    is performed by a custom Triton kernel.  Each pair (batch, hidden) is
    treated as an independent GEMM A @ Bᵀ with A,B in fp16 and the accumulator
    in fp32 (Tensor‑core friendly).  Small tile sizes (64×64×32) keep the
    register pressure low and give high occupancy on an H100.
5.  The result is reshaped back to [B,N,N,H], a second Layer‑norm is applied,
    the output‑gate is multiplied, and a final bias‑free linear projection
    returns a tensor of shape [B,N,N,dim] (float32).

All heavy arithmetic (step 4) lives in the Triton kernel; the surrounding
elementwise work stays in PyTorch for clarity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------------- #
# Triton kernel: batched GEMM C = A @ Bᵀ   (A,B : fp16, C : fp32)
# -----------------------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    A,               # left  tensor   (M, N, N)  – fp16
    B,               # right tensor   (M, N, N)  – fp16 (read as Bᵀ)
    C,               # output tensor  (M, N, N)  – fp32
    N,               # sequence length
    stride_am, stride_ai, stride_ak,   # strides of A: batch, row(i), col(k)
    stride_bm, stride_bj, stride_bk,   # strides of B: batch, row(j), col(k)
    stride_cm, stride_ci, stride_cj,   # strides of C: batch, row(i), col(j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched matrix multiplication C = A @ Bᵀ.

    Each Triton program (identified by (batch_id, pid_m, pid_n)) computes a
    BLOCK_M × BLOCK_N tile of the output for a single (batch, hidden) pair.
    """
    batch_id = tl.program_id(0)   # 0 … M‑1   (M = batch * hidden_dim)
    pid_m    = tl.program_id(1)   # tile row index
    pid_n    = tl.program_id(2)   # tile column index

    # --------------------------------------------------------------------- #
    # tile coordinates
    # --------------------------------------------------------------------- #
    i0 = pid_m * BLOCK_M
    j0 = pid_n * BLOCK_N
    i = i0 + tl.arange(0, BLOCK_M)
    j = j0 + tl.arange(0, BLOCK_N)

    mask_i = i < N
    mask_j = j < N

    # accumulator in FP32 (Tensor‑core dot returns FP32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------------- #
    # reduction over k
    # --------------------------------------------------------------------- #
    for k0 in range(0, N, BLOCK_K):
        k = k0 + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # A[batch_id, i, k]  → (BLOCK_M, BLOCK_K)
        a_ptr = A + batch_id * stride_am + i[:, None] * stride_ai + k[None, :] * stride_ak
        a = tl.load(
            a_ptr,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )                                 # fp16 → stays fp16

        # Bᵀ → we need B[batch_id, j, k]; reading B as (row=j, col=k)
        b_ptr = B + batch_id * stride_bm + k[:, None] * stride_bk + j[None, :] * stride_bj
        b = tl.load(
            b_ptr,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )                                 # fp16

        # Tensor‑core dot (FP16 × FP16 → FP32)
        acc += tl.dot(a, b)

    # --------------------------------------------------------------------- #
    # write tile back
    # --------------------------------------------------------------------- #
    c_ptr = C + batch_id * stride_cm + i[:, None] * stride_ci + j[None, :] * stride_cj
    tl.store(
        c_ptr,
        acc,
        mask=mask_i[:, None] & mask_j[None, :],
    )


# ----------------------------------------------------------------------------- #
# Public entry point used by the benchmark harness
# ----------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module with a Triton kernel for the
    cubic contraction.

    Args
    ----
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        - mask         : torch.Tensor, shape [B, N, N] (any dtype) – may be ignored
        - weights      : dict of model parameters
        - config       : dict, must contain ``dim``, ``hidden_dim`` and
                         optional ``nomask`` (bool)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data

    dim         = config["dim"]
    hidden_dim  = config["hidden_dim"]
    nomask      = config.get("nomask", True)
    eps         = 1e-5

    # ----------------------------------------------------------------- #
    # 1. Input LayerNorm (float32)
    # ----------------------------------------------------------------- #
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )   # [B, N, N, dim]  (float32)

    # ----------------------------------------------------------------- #
    # 2. Half‑precision linear projections + sigmoid gates
    # ----------------------------------------------------------------- #
    x_h = x.half()      # [B, N, N, dim] (fp16)

    # bias‑free linear layers – weight matrix is (out_features, in_features)
    left  = F.linear(x_h, weights["left_proj.weight"].to(x_h.dtype))   # [B, N, N, H]
    right = F.linear(x_h, weights["right_proj.weight"].to(x_h.dtype))

    left_gate  = torch.sigmoid(F.linear(x_h, weights["left_gate.weight"].to(x_h.dtype)))
    right_gate = torch.sigmoid(F.linear(x_h, weights["right_gate.weight"].to(x_h.dtype)))
    out_gate   = torch.sigmoid(F.linear(x_h, weights["out_gate.weight"].to(x_h.dtype)))

    # element‑wise gating
    left  = left * left_gate
    right = right * right_gate

    # ----------------------------------------------------------------- #
    # 3. Optional mask (broadcast over hidden dimension)
    # ----------------------------------------------------------------- #
    if not nomask:
        # mask: [B, N, N] → [B, N, N, 1] → fp16
        mask_h = mask.to(left.dtype).unsqueeze(-1)
        left  = left * mask_h
        right = right * mask_h

    # ----------------------------------------------------------------- #
    # 4. Batched GEMM  out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    # ----------------------------------------------------------------- #
    B, N_seq, _, _ = left.shape                     # B, N, N, H
    # move hidden dimension next to batch for a flat GEMM batch axis
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    M_total = B * hidden_dim                         # number of independent GEMMs
    left_flat  = left_perm.view(M_total, N_seq, N_seq)   # [M, N, N]  fp16
    right_flat = right_perm.view(M_total, N_seq, N_seq)  # [M, N, N]  fp16
    out_flat   = torch.empty((M_total, N_seq, N_seq), dtype=torch.float32,
                             device=left_flat.device)

    # ----------------------------------------------------------------- #
    # kernel launch configuration (tuned for H100)
    # ----------------------------------------------------------------- #
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    stride_am = left_flat.stride(0)
    stride_ai = left_flat.stride(1)
    stride_ak = left_flat.stride(2)

    stride_bm = right_flat.stride(0)
    stride_bj = right_flat.stride(1)   # row (j) stride after transpose
    stride_bk = right_flat.stride(2)   # column (k) stride

    stride_cm = out_flat.stride(0)
    stride_ci = out_flat.stride(1)
    stride_cj = out_flat.stride(2)

    grid = (
        M_total,
        (N_seq + BLOCK_M - 1) // BLOCK_M,
        (N_seq + BLOCK_N - 1) // BLOCK_N,
    )
    tri_mul_outgoing_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N_seq,
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bj,
        stride_bk,
        stride_cm,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ----------------------------------------------------------------- #
    # 5. Reshape, output LayerNorm, output‑gate, final projection
    # ----------------------------------------------------------------- #
    out = out_flat.view(B, hidden_dim, N_seq, N_seq).permute(0, 2, 3, 1).contiguous()

    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate.float()                     # broadcast multiplication
    output = F.linear(out, weights["to_out.weight"]) # final bias‑free projection

    return output
```
