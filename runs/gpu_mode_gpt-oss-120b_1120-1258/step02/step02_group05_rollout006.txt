def batched_tri_gemm_masked_kernel(
    A, B, C, mask,
    N, hidden_dim,
    stride_am, stride_ai, stride_ak,
    stride_bm, stride_bj, stride_bk,
    stride_cm, stride_ci, stride_cj,
    stride_mb, stride_mi, stride_mk,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    i0 = pid_m * BLOCK_M
    j0 = pid_n * BLOCK_N

    i = i0 + tl.arange(0, BLOCK_M)
    j = j0 + tl.arange(0, BLOCK_N)

    mask_i = i < N
    mask_j = j < N

    # compute batch index for mask
    batch = batch_id // hidden_dim
    # offset for mask for this batch
    mask_batch_offset = batch * stride_mb

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k0 in range(0, N, BLOCK_K):
        k = k0 + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # load A (left)
        a_ptr = A + batch_id * stride_am + i[:, None] * stride_ai + k[None, :] * stride_ak
        a = tl.load(a_ptr, mask=mask_i[:, None] & mask_k[None,:], other=0.0)

        # load mask for left
        mask_a_ptr = mask + mask_batch_offset + i[:, None] * stride_mi + k[None, :] * stride_mk
        mask_a = tl.load(mask_a_ptr, mask=mask_i[:, None] & mask_k[None,:], other=0.0)

        a = a * mask_a

        # load B (right). B reads as B^T: B[batch_id, k, j]
        b_ptr = B + batch_id * stride_bm + k[:, None] * stride_bk + j[None, :] * stride_bj
        b = tl.load(b_ptr, mask=mask_k[:, None] & mask_j[None,:], other=0.0)

        # load mask for right (mask(b,j,k))
        mask_b_ptr = mask + mask_batch_offset + j[None,:] * stride_mi + k[:, None] * stride_mk
        mask_b = tl.load(mask_b_ptr, mask=mask_k[:, None] & mask_j[None,:], other=0.0)

        b = b * mask_b

        acc += tl.dot(a.to(tl.float16), b.to(tl.float16))

    c_ptr = C + batch_id * stride_cm + i[:, None] * stride_ci + j[None, :] * stride_cj
    tl.store(c_ptr, acc, mask=mask_i[:, None] & mask_j[None,:])
```

"""
TriMul (outgoing) forward pass accelerated with Triton.

Algorithm
---------
1. Layer‑norm the input (float32) → cast to fp16.
2. Bias‑free linear projections for the “left” and “right” tensors and three
   sigmoid‑gates (all in fp16).  Multiply the projections by their gates.
3. If a mask is supplied, incorporate it inside the Triton kernel instead of a
   separate element‑wise pass (saves a O(B·N·N·H) operation).
4. Perform the cubic contraction
        out[b,h,i,j] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   as a batched GEMM C = A @ Bᵀ where A,B∈[M,N,N] with M = B·H.
   The kernel computes the GEMM tile‑wise with fp16 inputs and fp32
   accumulation (tensor‑core friendly).  Two kernels are provided:
   – `batched_tri_gemm_kernel`  (no mask)
   – `batched_tri_gemm_masked_kernel` (with mask)
5. Reshape the result back to [B,N,N,H] (float32), apply a second
   LayerNorm on the hidden dimension, multiply by the output‑gate and a final
   bias‑free linear projection back to the original feature dimension.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernels
# ----------------------------------------------------------------------
@triton.jit
def batched_tri_gemm_kernel(
    A,  # left  [M, N, N]  (fp16)
    B,  # right [M, N, N]  (fp16) – read as Bᵀ
    C,  # out   [M, N, N]  (fp32)
    N,
    stride_am, stride_ai, stride_ak,   # A strides
    stride_bm, stride_bj, stride_bk,   # B strides (Bᵀ layout)
    stride_cm, stride_ci, stride_cj,   # C strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C = A @ Bᵀ   (no mask)"""
    batch_id = tl.program_id(0)          # 0‑dim: M = batch * hidden
    pid_m    = tl.program_id(1)          # tile‑row (i)
    pid_n    = tl.program_id(2)          # tile‑col (j)

    i0 = pid_m * BLOCK_M
    j0 = pid_n * BLOCK_N
    i = i0 + tl.arange(0, BLOCK_M)
    j = j0 + tl.arange(0, BLOCK_N)

    mask_i = i < N
    mask_j = j < N

    # accumulator (fp32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over reduction axis k
    for k0 in range(0, N, BLOCK_K):
        k = k0 + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # ---- A[i, k] -------------------------------------------------
        a_ptr = A + batch_id * stride_am + i[:, None] * stride_ai + k[None, :] * stride_ak
        a = tl.load(
            a_ptr,
            mask=mask_i[:, None] & mask_k[None, :],
            other=tl.float16(0.0)
        )  # (BLOCK_M, BLOCK_K), fp16

        # ---- Bᵀ[k, j] (i.e. B[j, k]) --------------------------------
        b_ptr = B + batch_id * stride_bm + k[:, None] * stride_bk + j[None, :] * stride_bj
        b = tl.load(
            b_ptr,
            mask=mask_k[:, None] & mask_j[None, :],
            other=tl.float16(0.0)
        )  # (BLOCK_K, BLOCK_N), fp16

        # ---- accumulate ------------------------------------------------
        acc += tl.dot(a, b)   # fp16·fp16 → fp32

    # ---- write back ----------------------------------------------------
    c_ptr = C + batch_id * stride_cm + i[:, None] * stride_ci + j[None, :] * stride_cj
    tl.store(
        c_ptr,
        acc,
        mask=mask_i[:, None] & mask_j[None, :]
    )


@triton.jit
def batched_tri_gemm_masked_kernel(
    A, B, C, mask,
    N,
    hidden_dim,                      # = H   (compile‑time constant is not required)
    stride_am, stride_ai, stride_ak,
    stride_bm, stride_bj, stride_bk,
    stride_cm, stride_ci, stride_cj,
    stride_mb, stride_mi, stride_mk,   # mask strides (B,N,N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C = (A * mask) @ (B * mask)ᵀ  (mask broadcasted over H)"""
    batch_id = tl.program_id(0)          # 0‑dim over M = B·H
    pid_m    = tl.program_id(1)          # tile‑row (i)
    pid_n    = tl.program_id(2)          # tile‑col (j)

    i0 = pid_m * BLOCK_M
    j0 = pid_n * BLOCK_N
    i = i0 + tl.arange(0, BLOCK_M)
    j = j0 + tl.arange(0, BLOCK_N)

    mask_i = i < N
    mask_j = j < N

    # batch index for the mask (mask does not depend on H)
    batch = batch_id // hidden_dim
    mask_batch_off = batch * stride_mb

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k0 in range(0, N, BLOCK_K):
        k = k0 + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # ---- A[i, k] -------------------------------------------------
        a_ptr = A + batch_id * stride_am + i[:, None] * stride_ai + k[None, :] * stride_ak
        a = tl.load(
            a_ptr,
            mask=mask_i[:, None] & mask_k[None, :],
            other=tl.float16(0.0)
        )
        # mask for left (mask[b,i,k])
        m_a_ptr = mask + mask_batch_off + i[:, None] * stride_mi + k[None, :] * stride_mk
        m_a = tl.load(
            m_a_ptr,
            mask=mask_i[:, None] & mask_k[None, :],
            other=tl.float16(0.0)
        )
        a = a * m_a

        # ---- Bᵀ[k, j] (i.e. B[j, k]) --------------------------------
        b_ptr = B + batch_id * stride_bm + k[:, None] * stride_bk + j[None, :] * stride_bj
        b = tl.load(
            b_ptr,
            mask=mask_k[:, None] & mask_j[None, :],
            other=tl.float16(0.0)
        )
        # mask for right (mask[b,j,k])
        m_b_ptr = mask + mask_batch_off + j[None, :] * stride_mi + k[:, None] * stride_mk
        m_b = tl.load(
            m_b_ptr,
            mask=mask_k[:, None] & mask_j[None, :],
            other=tl.float16(0.0)
        )
        b = b * m_b

        acc += tl.dot(a, b)

    # ---- write back ----------------------------------------------------
    c_ptr = C + batch_id * stride_cm + i[:, None] * stride_ci + j[None, :] * stride_cj
    tl.store(
        c_ptr,
        acc,
        mask=mask_i[:, None] & mask_j[None, :]
    )

    Arguments
    ---------
    data : tuple (input_tensor, mask_tensor, weights_dict, config_dict)
        input_tensor : [B, N, N, dim]  (float32)
        mask_tensor  : [B, N, N]      (any dtype; may be None)
        weights_dict : model parameters
        config_dict  : {"dim": int, "hidden_dim": int, "nomask": bool, ...}
    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32) → fp16
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )                       # [B, N, N, dim]  float32
    x_h = x.half()          # fp16 version for all linear ops

    # ------------------------------------------------------------------
    # 2. Projections + gates (fp16)
    # ------------------------------------------------------------------
    # bias‑free linear projections
    left  = F.linear(x_h, weights["left_proj.weight"].half())   # [B,N,N,H]
    right = F.linear(x_h, weights["right_proj.weight"].half())

    # gates (sigmoid)
    left_gate  = torch.sigmoid(F.linear(x_h, weights["left_gate.weight"].half()))
    right_gate = torch.sigmoid(F.linear(x_h, weights["right_gate.weight"].half()))
    out_gate   = torch.sigmoid(F.linear(x_h, weights["out_gate.weight"].half()))

    # apply gates (mask will be applied inside the kernel if needed)
    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 3. Prepare tensors for the cubic contraction
    # ------------------------------------------------------------------
    # bring hidden dimension to the 2‑nd axis and make them contiguous
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    B, N, _, _ = left.shape                     # B, seq_len, seq_len, dim
    M_total = B * hidden_dim                    # number of independent GEMMs

    # flatten to (M_total, N, N) – this layout matches the kernel expectations
    left_flat  = left_perm.view(M_total, N, N)      # fp16
    right_flat = right_perm.view(M_total, N, N)     # fp16
    out_flat   = torch.empty((M_total, N, N), dtype=torch.float32, device=left_flat.device)

    # ------------------------------------------------------------------
    # 4. Choose kernel (masked / unmasked) and launch
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # strides (in element units)
    stride_am = left_flat.stride(0)
    stride_ai = left_flat.stride(1)
    stride_ak = left_flat.stride(2)

    stride_bm = right_flat.stride(0)
    stride_bj = right_flat.stride(1)   # j‑stride (second N)
    stride_bk = right_flat.stride(2)   # k‑stride (first N)

    stride_cm = out_flat.stride(0)
    stride_ci = out_flat.stride(1)
    stride_cj = out_flat.stride(2)

    grid = (
        M_total,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    if nomask or mask_tensor is None:
        # --------------------------------------------------------------
        # No mask – use the simple kernel
        # --------------------------------------------------------------
        batched_tri_gemm_kernel[grid](
            left_flat,
            right_flat,
            out_flat,
            N,
            stride_am,
            stride_ai,
            stride_ak,
            stride_bm,
            stride_bj,
            stride_bk,
            stride_cm,
            stride_ci,
            stride_cj,
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
        )
    else:
        # --------------------------------------------------------------
        # Masked version – mask is broadcasted over hidden dimension
        # --------------------------------------------------------------
        mask_h = mask_tensor.to(torch.float16).contiguous()     # [B,N,N] fp16
        stride_mb = mask_h.stride(0)
        stride_mi = mask_h.stride(1)
        stride_mk = mask_h.stride(2)

        batched_tri_gemm_masked_kernel[grid](
            left_flat,
            right_flat,
            out_flat,
            mask_h,
            N,
            hidden_dim,
            stride_am,
            stride_ai,
            stride_ak,
            stride_bm,
            stride_bj,
            stride_bk,
            stride_cm,
            stride_ci,
            stride_cj,
            stride_mb,
            stride_mi,
            stride_mk,
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
        )

    # ------------------------------------------------------------------
    # 5. Reshape back to [B, N, N, H] (float32)
    # ------------------------------------------------------------------
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()   # [B,N,N,H]

    # ------------------------------------------------------------------
    # 6. Output LayerNorm → out_gate → final linear (float32)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate.float()                         # gate multiplication
    output = F.linear(out, weights["to_out.weight"].half())   # final proj, back to dim

    return output
