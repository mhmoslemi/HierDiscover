The heavy O(N³) contraction

accumulation) with torch.cuda.amp for speed.  The kernel processes one
pair (batch, hidden) at a time and tiles the i‑ and j‑axes (BLOCK_M,
BLOCK_N) while iterating over k in chunks (BLOCK_K).  The final result
is converted back to FP32 as required by the API.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    # strides for left  [B, N, N, H]
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    # strides for right [B, N, N, H] (same layout, but we keep i/j separate)
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    # strides for out   [B, N, N, H]
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    # runtime parameters
    B, N, H,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

    Grid:
        pid_i  -> tile over the i‑axis (rows)
        pid_j  -> tile over the j‑axis (cols)
        pid_bh -> combined batch‑hidden index (B*H tiles)
    """
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_bh = tl.program_id(2)

    # Decode batch and hidden head from the combined index
    b = pid_bh // H
    h = pid_bh % H

    # Offsets for the current tile
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    left_base = left_ptr + b * stride_left_b + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base = out_ptr + b * stride_out_b + h * stride_out_h

        )
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
            dtype=tl.float16,
        )

        # -------------------------------------------------------------
        # Load right tile  : shape (BLOCK_K, BLOCK_N)  -> (k, j)
        #   (note the order: rows = k, cols = j)
        # -------------------------------------------------------------
        right_ptrs = (
            right_base
            + offs_k[:, None] * stride_right_k       # k‑stride (row)
            + offs_j[None, :] * stride_right_i       # j‑stride (col)
        )
        right = tl.load(
            right_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
            dtype=tl.float16,
        )

        # Matrix multiplication (M×K)·(K×N) → (M×N)
        acc += tl.dot(left, right)   # FP32 accumulation

    # -------------------------------------------------------------
    # Write the result (cast back to fp16)
    # -------------------------------------------------------------
    out_ptrs = (
        out_base
        + offs_i[:, None] * stride_out_i
        + offs_j[None, :] * stride_out_j
    )
    tl.store(
        out_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask         : torch.Tensor [B, N, N] (bool/float) or None
        - weights      : dict of model parameters (float32)
        - config       : dict with keys "dim", "hidden_dim", "nomask"

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, dim] (float32)
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # Load model parameters (they are already on the correct device)
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # -----------------------------------------------------------------
    # Mixed‑precision forward pass up to the N³ contraction
    # -----------------------------------------------------------------
    with torch.cuda.amp.autocast():
        # LayerNorm on the input channel dimension
        x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

        # Linear projections (no bias) → [B, N, N, hidden_dim]
        left = F.linear(x, left_proj_weight)   # shape: (B,N,N,hidden)
        right = F.linear(x, right_proj_weight)

        # Optional mask (broadcast over the hidden dimension)
        if not nomask and mask is not None:
            mask_f = mask.to(x.dtype).unsqueeze(-1)   # [B,N,N,1]
            left = left * mask_f
            right = right * mask_f

        # Gating scalars (sigmoid)
        left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
        right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
        out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

        # Apply gates element‑wise
        left = left * left_gate
        right = right * right_gate

        # -----------------------------------------------------------------
        # Triton kernel:    out = Σ_k left[...,i,k,d] * right[...,j,k,d]
        # -----------------------------------------------------------------
        B, N, _, _ = left.shape  # N is the sequence length
        # Ensure contiguous layout for the kernel
        left = left.contiguous()
        right = right.contiguous()
        # Allocate output buffer (FP16 for the kernel, will be cast later)
        out = torch.empty((B, N, N, hidden_dim), dtype=left.dtype, device=device)

        # Strides (as python ints)
        s_lb, s_li, s_lk, s_lh = left.stride()
        s_rb, s_ri, s_rk, s_rh = right.stride()
        s_ob, s_oi, s_oj, s_oh = out.stride()

        # Triton launch configuration
        BLOCK_M = 64
        BLOCK_N = 64
        BLOCK_K = 64
        grid = (
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N,
            B * hidden_dim,
        )

        trimul_outgoing_kernel[grid](
            left,
            right,
            out,
            # strides
            s_lb,
            s_li,
            s_lk,
            s_lh,
            s_rb,
            s_ri,
            s_rk,
            s_rh,
            s_ob,
            s_oi,
            s_oj,
            s_oh,
            # runtime params
            B,
            N,
            hidden_dim,
            # tile sizes
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
        )

        # -----------------------------------------------------------------
        # Post‑contraction processing (LayerNorm → gating → final projection)
        # -----------------------------------------------------------------
        out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)
        out = out * out_gate
        out = F.linear(out, to_out_weight)          # back to `dim`

        # Cast final result to FP32 as required by the spec
        out = out.to(torch.float32)

    # -----------------------------------------------------------------
    # Return the final tensor
    # -----------------------------------------------------------------
    return out
