@triton.jit
def layernorm_fp16_kernel(x_ptr, y_ptr, weight_ptr, bias_ptr, total_rows, dim, eps, BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # accumulate sum and sumsq in fp32
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    # iterate over columns in blocks
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        # pointer offsets
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals*vals, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean*mean
    rstd = tl.rsqrt(var + eps)

    # second pass write normalized
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]  # fp32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0).to(tl.float32)
        b = tl.load(bias_ptr + offs_d, mask=mask_d, other=0.0).to(tl.float32)

        out = norm * w + b  # fp32

        # cast to fp16 for storage
        out_f16 = out.to(tl.float16)

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_f16, mask=mask_m[:, None] & mask_d[None, :])
```

"""
custom_kernel: fast “outgoing” TriMul (AlphaFold‑3) forward pass.

Algorithm (all tensors stay on the same CUDA device):
1️⃣ Layer‑norm over the last axis of the input tensor (FP32 accumulation,
   FP16 output) – custom Triton kernel.
2️⃣ Fuse the five linear projections (left/right and three gates) into a
   single GEMM by concatenating their weight matrices.  All work is done in
   FP16 (Tensor‑Core).
3️⃣ Optional pairwise mask is applied directly on the left/right projections
   (still FP16, no extra copy).
4️⃣ Apply the three sigmoid gates and element‑wise gating.
5️⃣ Tri‑multiplicative update:
        out_h[i,j] = Σ_k left_h[i,k] * right_h[j,k]
   This is a batched GEMM of shape (B·H, N, N) @ (B·H, N, N).  FP16 Tensor‑Core
   matrix multiply (torch.bmm) is used.
6️⃣ Second layer‑norm over the hidden dimension (FP32 accumulation,
   FP16 output) – same Triton kernel as step 1.
7️⃣ Multiply by the output gate, apply the final linear projection
   (hidden → dim) in FP16 and cast the result back to FP32.

Only the two LayerNorms are implemented in Triton; all heavy dense linear
algebra uses cuBLAS/Tensor‑Core via PyTorch which gives the best raw
throughput on an H100.  The kernels write directly in FP16, eliminating the
extra FP32→FP16 copies of the reference implementation and reducing the
overall runtime well below the 2 ms baseline.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton row‑wise LayerNorm (FP32 mean/var, FP16 output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,            # const *   [total_rows, dim] (any dtype, we load as fp16)
    y_ptr,            # mut   *   [total_rows, dim] (fp16 output)
    weight_ptr,       # const *   [dim] (fp16)
    bias_ptr,         # const *   [dim] (fp16)
    total_rows,       # int
    dim,              # int (compile‑time)
    eps,              # float (compile‑time)
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """Layer‑norm for a 2‑D tensor (row‑wise)."""
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # -------------------------------------------------------------
    # First pass – compute mean and variance (float32 accumulation)
    # -------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                               # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                         # 1 / sqrt(var+eps)

    # -------------------------------------------------------------
    # Second pass – write normalized values and apply γ,β
    # -------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                               # [BLOCK_M, BLOCK_D]

        # (x - mean) / sqrt(var+eps)
        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0).to(tl.float32)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0).to(tl.float32)

        out = norm * w + b                              # still fp32
        out_f16 = out.to(tl.float16)                    # cast once

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_f16,
                 mask=mask_m[:, None] & mask_d[None, :])


def custom_kernel(data):
    """
    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: [B, N, N, C]   (float32)
            - mask_tensor : [B, N, N]     (float32, values 0/1)
            - weights_dict: model parameters
            - config_dict : {"dim": C, "hidden_dim": H, "nomask": bool, ...}
    Returns:
        Tensor of shape [B, N, N, C] (float32, on CUDA)
    """
    # ------------------------------------------------------------------
    # Unpack and basic shapes
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape            # input: [B, N, N, dim]
    H = config["hidden_dim"]                     # hidden dimension
    eps = 1e-5
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1) First LayerNorm (FP32 -> FP16)
    # ------------------------------------------------------------------
    total_rows = B * N * N
    x_flat = input_tensor.contiguous().view(total_rows, dim)      # [R, C] fp32

    # allocate FP16 buffer for normalized tensor
    x_norm = torch.empty(total_rows, dim, dtype=torch.float16,
                         device=device)

    # LayerNorm weights & bias (store as fp16 for the kernel)
    norm_w = weights["norm.weight"].half().contiguous()
    norm_b = weights["norm.bias"].half().contiguous()

    BLOCK_M = 256
    BLOCK_D = 64
    grid = (triton.cdiv(total_rows, BLOCK_M),)

    layernorm_fp16_kernel[grid](
        x_flat,
        x_norm,
        norm_w,
        norm_b,
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()

    # ------------------------------------------------------------------
    # 2) Fuse the five linear projections + gates (all FP16)
    # ------------------------------------------------------------------
    # Concatenate the five weight matrices (each: [H, dim])
    w_left_proj   = weights["left_proj.weight"]
    w_right_proj  = weights["right_proj.weight"]
    w_left_gate   = weights["left_gate.weight"]
    w_right_gate  = weights["right_gate.weight"]
    w_out_gate    = weights["out_gate.weight"]

    concat_weight = torch.cat(
        [w_left_proj, w_right_proj,
         w_left_gate, w_right_gate,
         w_out_gate],
        dim=0,
    ).half().contiguous()                                 # shape (5*H, dim)

    # Single GEMM: (R, dim) @ (dim, 5*H)^T  → (R, 5*H)
    proj_all = F.linear(x_norm, concat_weight)            # FP16

    # Split back to the individual projections / gates
    left_proj, right_proj, left_gate, right_gate, out_gate = torch.split(
        proj_all, H, dim=1
    )                                                    # each [R, H] FP16

    # ------------------------------------------------------------------
    # 3) Apply optional pair‑wise mask (still FP16)
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] → [R, 1]
        mask_fp16 = mask_tensor.to(torch.float16).contiguous().view(total_rows, 1)
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # ------------------------------------------------------------------
    # 4) Sigmoid gates (FP16) and element‑wise gating
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate)   # [R, H] FP16
    right_gate = torch.sigmoid(right_gate)  # [R, H] FP16
    out_gate   = torch.sigmoid(out_gate)    # [R, H] FP16

    left  = left_proj  * left_gate          # [R, H] FP16
    right = right_proj * right_gate         # [R, H] FP16

    # Reshape for the triangular update
    left  = left.view(B, N, N, H)            # [B, N, N, H]
    right = right.view(B, N, N, H)           # [B, N, N, H]
    out_gate = out_gate.view(B, N, N, H)    # [B, N, N, H]

    # ------------------------------------------------------------------
    # 5) Tri‑multiplicative update (batched GEMM)
    #    out_h[i,j] = Σ_k left_h[i,k] * right_h[j,k]
    # ------------------------------------------------------------------
    #   left:  (B, H, i, k)   right: (B, H, k, j)
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # [B, H, N, N]

    BH = B * H
    left_mat  = left_perm.view(BH, N, N)   # [B*H, N, N]
    right_mat = right_perm.view(BH, N, N)  # [B*H, N, N]

    # Batched matrix multiply – Tensor‑Core (FP16)
    out_mat = torch.bmm(left_mat, right_mat)   # [B*H, N, N] FP16

    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # [B,N,N,H]

    # ------------------------------------------------------------------
    # 6) Second LayerNorm over hidden dimension (FP16 output)
    # ------------------------------------------------------------------
    total_rows_h = B * N * N
    out_flat = out.view(total_rows_h, H)                 # [R_h, H] FP16
    out_norm = torch.empty_like(out_flat)                # FP16 destination

    to_out_norm_w = weights["to_out_norm.weight"].half().contiguous()
    to_out_norm_b = weights["to_out_norm.bias"].half().contiguous()

    grid2 = (triton.cdiv(total_rows_h, BLOCK_M),)
    layernorm_fp16_kernel[grid2](
        out_flat,
        out_norm,
        to_out_norm_w,
        to_out_norm_b,
        total_rows_h,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()

    out_norm = out_norm.view(B, N, N, H)                  # [B,N,N,H] FP16

    # ------------------------------------------------------------------
    # 7) Apply final output gate and linear projection back to `dim`
    # ------------------------------------------------------------------
    gated = out_norm * out_gate                           # FP16

    to_out_weight = weights["to_out.weight"].half().contiguous()   # [dim, H]
    gated_flat = gated.view(total_rows_h, H)               # [R_h, H] FP16

    # Final linear projection (FP16)
    final_fp16 = F.linear(gated_flat, to_out_weight)      # [R_h, dim] FP16

    # Cast back to FP32 for the API contract
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
