"""
TriMul (outgoing) – fast Triton implementation
================================================

The module implements the *outgoing* version of the TriMul operator
described in the AlphaFold‑3 paper.  The forward pass consists of

1. **Layer‑norm** over the input feature dimension (dim).
2. **Bias‑free linear projections** → left, right (hidden_dim features).
   The two projections are fused into a single matmul to reduce memory traffic.
3. **Optional mask** broadcast on the hidden dimension.
4. **Three sigmoid gates** (left, right, out).  The three gate
   linear layers are also fused.
5. **Cubic contraction**  

   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]  

   – realised as a batched GEMM C = A @ Bᵀ with A,B ∈ ℝ^{N×N}
     for each (batch, hidden) pair.  The heavy N³·hidden work is
     performed with a custom Triton kernel that loads FP16 values,
     uses Tensor‑Core `tl.dot`, and accumulates in FP32.
6. **Layer‑norm** over the hidden dimension, element‑wise multiplication
   with the output gate, and a final bias‑free linear projection back to the
   original dimension.

All heavy arithmetic (steps 2‑5) runs in half‑precision; the final result
is returned as FP32.  The kernel tiles the output (i,j) plane with
square 64 × 64 or 128 × 128 blocks (chosen automatically from the
sequence length) and iterates over the reduction dimension k in equally
sized blocks.  This layout gives good occupancy on an H100 and keeps
launch overhead low by flattening the (batch, hidden) axes into a single
grid dimension.
"""

import math
from typing import Tuple, Dict

import torch
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ
#   A,B : (M_total, N, N),  A – left,  B – right
#   C   : (M_total, N, N) , accumulates in fp32
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    a_ptr,               # left  matrix (M, N, N)  – fp16
    b_ptr,               # right matrix (M, N, N)  – fp16, will be read transposed
    c_ptr,               # output   (M, N, N)    – fp32
    N,                   # sequence length (int)
    stride_am, stride_ai, stride_ak,   # strides of A: (M, i, k)
    stride_bm, stride_bj, stride_bk,   # strides of B: (M, j, k) – accessed as (k, j)
    stride_cm, stride_ci, stride_cj,   # strides of C: (M, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # ------------------------------------------------------------------
    # 1) tile identifiers
    # ------------------------------------------------------------------
    pid_m = tl.program_id(0)          # row‑tile index (i)
    pid_n = tl.program_id(1)          # col‑tile index (j)
    pid_b = tl.program_id(2)          # flattened (batch * hidden) index

    i0 = pid_m * BLOCK_M
    j0 = pid_n * BLOCK_N

    # ------------------------------------------------------------------
    # 2) offsets inside the current tile
    # ------------------------------------------------------------------
    i = i0 + tl.arange(0, BLOCK_M)
    j = j0 + tl.arange(0, BLOCK_N)

    mask_i = i < N
    mask_j = j < N

    # ------------------------------------------------------------------
    # 3) output pointer & accumulator (fp32)
    # ------------------------------------------------------------------
    c_ptrs = (
        c_ptr
        + pid_b * stride_cm
        + i[:, None] * stride_ci
        + j[None, :] * stride_cj
    )
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # 4) reduction over the k dimension
    # ------------------------------------------------------------------
    for k0 in range(0, N, BLOCK_K):
        k = k0 + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # A[i, k]  (fp16)
        a_ptrs = (
            a_ptr
            + pid_b * stride_am
            + i[:, None] * stride_ai
            + k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )   # (BLOCK_M, BLOCK_K), fp16

        # B[k, j] = right[j, k]  (fp16) – we read transposed
        b_ptrs = (
            b_ptr
            + pid_b * stride_bm
            + k[:, None] * stride_bk
            + j[None, :] * stride_bj
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )   # (BLOCK_K, BLOCK_N), fp16

        # Tensor‑core dot, accumulation in fp32
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # 5) write back
    # ------------------------------------------------------------------
    tl.store(
        c_ptrs,
        acc,
        mask=mask_i[:, None] & mask_j[None, :],
    )


def custom_kernel(data: Tuple[torch.Tensor,
                               torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module with a Triton kernel.

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor : [B, N, N, dim]   (float32)
            - mask         : [B, N, N]        (float32 / bool) – ignored if config["nomask"] is True
            - weights      : dict of model parameters
            - config       : dict with keys
                               "dim" (int), "hidden_dim" (int), "nomask" (bool, optional)

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    # ------------------------------------------------------------------
    # unpack weight tensors (already on the correct device)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]

    left_proj_weight = weights["left_proj.weight"]          # (hidden, dim)
    right_proj_weight = weights["right_proj.weight"]        # (hidden, dim)

    left_gate_weight = weights["left_gate.weight"]          # (hidden, dim)
    right_gate_weight = weights["right_gate.weight"]        # (hidden, dim)
    out_gate_weight = weights["out_gate.weight"]            # (hidden, dim)

    to_out_norm_weight = weights["to_out_norm.weight"]      # (hidden,)
    to_out_norm_bias = weights["to_out_norm.bias"]          # (hidden,)

    to_out_weight = weights["to_out.weight"]                # (dim, hidden)

    # ------------------------------------------------------------------
    # 1. Input layer‑norm over the feature dimension (dim)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, dim] – FP32

    # ------------------------------------------------------------------
    # Cast to FP16 for the heavy parts
    # ------------------------------------------------------------------
    x_h = x.half()   # [B, N, N, dim] – FP16

    # ------------------------------------------------------------------
    # 2. Fuse the two linear projections (left/right) into one matmul
    # ------------------------------------------------------------------
    #   concat_weights : (2*hidden_dim, dim)
    proj_weight = torch.cat([left_proj_weight, right_proj_weight], dim=0).half()   # FP16
    proj = torch.nn.functional.linear(x_h, proj_weight)   # [B, N, N, 2*hidden]
    left, right = torch.split(proj, hidden_dim, dim=-1)   # each [B, N, N, hidden]

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape [B, N, N] -> [B, N, N, 1] -> same dtype as left/right
        mask_exp = mask.to(x_h.dtype).unsqueeze(-1)
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4. Fuse the three gate linear layers and apply sigmoid
    # ------------------------------------------------------------------
    gate_weight = torch.cat(
        [left_gate_weight, right_gate_weight, out_gate_weight], dim=0
    ).half()                               # (3*hidden, dim)

    gates = torch.nn.functional.linear(x_h, gate_weight)   # [B, N, N, 3*hidden]
    left_gate, right_gate, out_gate = torch.split(gates, hidden_dim, dim=-1)
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)        # [B, N, N, hidden]  FP16

    # ------------------------------------------------------------------
    # 5. Apply gates to the projected tensors
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6. Triton kernel for the cubic contraction:
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    We treat each (b, h) pair as a separate GEMM.
    # ------------------------------------------------------------------
    # Bring hidden dim to the second axis for easy flattening:
    #   [B, hidden, N, N] -> view as (M_total, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, hidden, N, N] FP16
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, hidden, N, N] FP16

    B, H, N, _ = left_perm.shape
    M_total = B * H

    left_flat = left_perm.view(M_total, N, N)    # [M, N, N] FP16
    right_flat = right_perm.view(M_total, N, N)  # [M, N, N] FP16
    out_flat = torch.empty(
        (M_total, N, N), dtype=torch.float32, device=device
    )                                            # [M, N, N] FP32

    # ------------------------------------------------------------------
    # Choose tile sizes: larger tiles for long sequences improve occupancy.
    # ------------------------------------------------------------------
    if N <= 256:
        BLOCK_M = BLOCK_N = BLOCK_K = 64
    else:
        BLOCK_M = BLOCK_N = BLOCK_K = 128

    # Strides (in elements) for the flattened tensors
    stride_am = left_flat.stride(0)
    stride_ai = left_flat.stride(1)
    stride_ak = left_flat.stride(2)

    stride_bm = right_flat.stride(0)
    stride_bj = right_flat.stride(1)   # stride over j (second dim)
    stride_bk = right_flat.stride(2)   # stride over k (third dim)

    stride_cm = out_flat.stride(0)
    stride_ci = out_flat.stride(1)
    stride_cj = out_flat.stride(2)

    # Grid dimensions: (tiles_i, tiles_j, batch*hidden)
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        M_total,
    )

    # Launch kernel
    tri_mul_outgoing_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bj,
        stride_bk,
        stride_cm,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,          # H100 Tensor‑Core friendly
        num_stages=2,
    )

    # ------------------------------------------------------------------
    # 7. Reshape back to the original layout (float32)
    # ------------------------------------------------------------------
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # [B, N, N, hidden] FP32

    # ------------------------------------------------------------------
    # 8. Hidden‑dim layer‑norm, output gating, and final projection back to dim
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )  # FP32

    # Apply the output gate (cast to output dtype)
    out = out * out_gate.to(out.dtype)

    # Final linear projection (bias‑free) back to original dimension
    output = torch.nn.functional.linear(out, to_out_weight)   # [B, N, N, dim] FP32

    return output
```
