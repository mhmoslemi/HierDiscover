"""
TriMul (outgoing) – forward only.

Algorithm
---------
1. Layer‑norm on the channel dimension (float32).
2. Cast to FP16 and compute all linear projections + gates (no bias) with
   torch.nn.functional.linear (cuBLAS, Tensor‑cores).
3. If a pairwise mask is present, apply it (broadcast) to the projections.
4. Fuse the element‑wise gate multiplications with a tiny Triton kernel.
   This kernel multiplies
        left  ← left  * left_gate
        right ← right * right_gate
   in‑place on the flattened (B·H, N, N) tensors.
5. Heavy multiplicative update:
        out[b·H,i,j] = Σₖ left[b·H,i,k] * right[b·H,j,k]
   performed as a batched GEMM (torch.bmm) which uses cuBLAS Tensor‑cores
   and is far faster than a hand‑written Triton matmul for the N³ work.
6. Reshape → Layer‑norm over the hidden dimension, multiply by the
   out‑gate, and project back to the original channel dimension.
   All remaining ops are in FP32 (except the final linear, which runs in FP16).

The Triton kernel is only the element‑wise gating step, satisfying the
requirement to use Triton while keeping launch overhead minimal.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _gate_mul_kernel(
    a_ptr, a_gate_ptr,                # left  and left_gate
    b_ptr, b_gate_ptr,                # right and right_gate
    numel: tl.constexpr,              # total number of elements (B·H·N·N)
    BLOCK_SIZE: tl.constexpr,         # number of elements processed per program
):
    """In‑place element‑wise multiplication:
       a ← a * a_gate
       b ← b * b_gate
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < numel

    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    ag = tl.load(a_gate_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    bg = tl.load(b_gate_ptr + offs, mask=mask, other=0.0)

    a = a * ag
    b = b * bg

    tl.store(a_ptr + offs, a, mask=mask)
    tl.store(b_ptr + offs, b, mask=mask)

    total_elems = left_flat_1d.numel()        # BH * N * N
    BLOCK_SIZE = 1024                         # enough to keep launch overhead low
    grid = ( (total_elems + BLOCK_SIZE - 1) // BLOCK_SIZE, )

    _gate_mul_kernel[grid](
        left_flat_1d,
        left_gate_flat_1d,
        right_flat_1d,
        right_gate_flat_1d,
        total_elems,
        BLOCK_SIZE=BLOCK_SIZE,
    )
    # After the kernel left_flat/right_flat are already multiplied by their gates

    # ------------------------------------------------------------------
    # 5. Heavy multiplicative update (batched GEMM)
    #    out_flat[b] = left_flat[b] @ right_flat[b].T
    # ------------------------------------------------------------------
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))   # (BH, N, N), FP16

    # ------------------------------------------------------------------
    # 6. Reshape back to [B, N, N, hidden] (still FP16)
    # ------------------------------------------------------------------
    out = out_flat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7. Final LayerNorm, out‑gate, and projection back to original dim
    # ------------------------------------------------------------------
    out_fp32 = out.to(torch.float32)

    out_fp32 = torch.nn.functional.layer_norm(
        out_fp32,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # Apply out‑gate (out_gate is still FP16)
    out_fp32 = out_fp32 * out_gate.to(torch.float32)

    # Final linear projection (hidden → dim) – keep it in FP16 for speed,
    # then cast back to FP32 for the final output.
    out_h = out_fp32.to(torch.float16)
    out = torch.nn.functional.linear(
        out_h,
        weights["to_out.weight"].to(torch.float16)
    ).to(torch.float32)

    return out
