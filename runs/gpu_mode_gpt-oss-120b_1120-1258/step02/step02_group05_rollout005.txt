"""
TriMul (outgoing) forward pass – Triton‑accelerated cubic contraction.

Algorithm
---------
1️⃣ Layer‑norm the input on the feature axis (float32, torch.nn.functional).
2️⃣ Cast to FP16 and apply bias‑free linear projections for the left/right
   tensors together with three sigmoid gates (left/right/out).
3️⃣ (Optional) mask the projected tensors – the mask is broadcasted over the
   hidden dimension.
4️⃣ Reshape to (M, N, N) where M = batch × hidden_dim and launch a Triton
   kernel that computes for each (m,i,j):
        C[m,i,j] = Σₖ A[m,i,k]·B[m,k,j]      (A = left, B = right, Bᵀ used)
   The kernel performs a batched GEMM with FP16 inputs, FP32 accumulation
   (Tensor‑core friendly).  Autotuning picks the best BLOCK_M/N/K
   configuration and number of warps for the given N.
5️⃣ Reshape the result back to [B,N,N,hidden_dim] (float32) and apply a second
   Layer‑norm, the output‑gate, and a final bias‑free linear projection back
   to the original feature dimension.

The heavy O(N³·hidden_dim) work is the Triton kernel; the remaining ops stay
in PyTorch for simplicity and numerical stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# -------------------------------------------------------------------------
# Triton kernel – batched GEMM C = A @ Bᵀ for each batch*hidden slice.
# -------------------------------------------------------------------------
@triton.autotune(
    # Configs tuned for H100 Tensor‑cores (FP16 × FP16 → FP32).
    # The key `N` allows the runtime to pick the best block size for the
    # sequence length of the current launch.
    configs=[
        triton.Config({"BLOCK_M": 64, "BLOCK_N": 64, "BLOCK_K": 32}, num_warps=4, num_stages=2),
        triton.Config({"BLOCK_M": 128, "BLOCK_N": 128, "BLOCK_K": 64}, num_warps=8, num_stages=3),
        triton.Config({"BLOCK_M": 256, "BLOCK_N": 256, "BLOCK_K": 64}, num_warps=8, num_stages=4),
    ],
    key=["N"],
)
@triton.jit
def batched_tri_gemm_kernel(
    A,               # left tensor  (M, N, N) – FP16
    B,               # right tensor (M, N, N) – FP16 (read as Bᵀ)
    C,               # output      (M, N, N) – FP32
    N,               # sequence length (scalar)
    stride_am, stride_ai, stride_ak,   # strides of A: batch, row(i), col(k)
    stride_bm, stride_bj, stride_bk,   # strides of B: batch, row(j), col(k)
    stride_cm, stride_ci, stride_cj,   # strides of C: batch, row(i), col(j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM C = A @ Bᵀ.

    Each program instance computes a (BLOCK_M × BLOCK_N) tile of C for a
    particular batch‑hidden index `batch_id`.  The reduction over K (the
    sequence dimension) is tiled with size BLOCK_K.  All arithmetic is done
    in FP32 while the loads are FP16, enabling Tensor‑core utilization.
    """
    batch_id = tl.program_id(0)          # index over M = batch * hidden_dim
    pid_m    = tl.program_id(1)          # tile row   (i)
    pid_n    = tl.program_id(2)          # tile col   (j)

    # -------------------------------------------------
    # Tile offsets
    # -------------------------------------------------
    i0 = pid_m * BLOCK_M
    j0 = pid_n * BLOCK_N
    i = i0 + tl.arange(0, BLOCK_M)
    j = j0 + tl.arange(0, BLOCK_N)

    mask_i = i < N
    mask_j = j < N

    # -------------------------------------------------
    # Accumulator (FP32)
    # -------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------------------------------------
    # Main K‑loop
    # -------------------------------------------------
    for k0 in range(0, N, BLOCK_K):
        k = k0 + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # A[batch, i, k]   → shape (BLOCK_M, BLOCK_K)
        a_ptr = A + batch_id * stride_am + i[:, None] * stride_ai + k[None, :] * stride_ak
        a = tl.load(
            a_ptr,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )   # already FP16

        # Bᵀ → B[batch, k, j]   → shape (BLOCK_K, BLOCK_N)
        b_ptr = B + batch_id * stride_bm + k[:, None] * stride_bk + j[None, :] * stride_bj
        b = tl.load(
            b_ptr,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )   # FP16

        # Tensor‑core dot (FP16 × FP16 → FP32)
        acc += tl.dot(a, b)

    # -------------------------------------------------
    # Write back the tile
    # -------------------------------------------------
    c_ptr = C + batch_id * stride_cm + i[:, None] * stride_ci + j[None, :] * stride_cj
    tl.store(
        c_ptr,
        acc,
        mask=mask_i[:, None] & mask_j[None, :],
    )


# -------------------------------------------------------------------------
# Public entry point
# -------------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Args:
        data (tuple): (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask:         torch.Tensor of shape [B, N, N] (any dtype) – optional
            - weights:      dict of torch.Tensors matching the module parameters
            - config:       dict containing:
                * "dim"         – input feature dimension
                * "hidden_dim" – hidden dimension of the TriMul projection
                * "nomask"      – bool, True means ignore the mask argument

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # -----------------------------------------------------------------
    # 0️⃣ Unpack
    # -----------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    # -----------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim] (float32)

    # -----------------------------------------------------------------
    # 2️⃣ FP16 projection + gates
    # -----------------------------------------------------------------
    x_fp16 = x.half()  # cast once

    # Linear projections (bias‑free)
    left  = F.linear(x_fp16, weights["left_proj.weight"].half())   # [B, N, N, H]
    right = F.linear(x_fp16, weights["right_proj.weight"].half())

    # Sigmoid gates
    left_gate  = torch.sigmoid(F.linear(x_fp16, weights["left_gate.weight"].half()))
    right_gate = torch.sigmoid(F.linear(x_fp16, weights["right_gate.weight"].half()))
    out_gate   = torch.sigmoid(F.linear(x_fp16, weights["out_gate.weight"].half()))

    # Apply gates
    left  = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over hidden dim)
    # -----------------------------------------------------------------
    if (mask is not None) and (not nomask):
        # mask: [B, N, N] → [B, N, N, 1] and cast to fp16
        mask_exp = mask.to(dtype=left.dtype).unsqueeze(-1)
        left  = left * mask_exp
        right = right * mask_exp

    # -----------------------------------------------------------------
    # 4️⃣ Batched GEMM: out[b, h, i, j] = Σₖ left[b, i, k, h] * right[b, j, k, h]
    # -----------------------------------------------------------------
    B, N, _, _ = left.shape                     # B, seq_len, seq_len, hidden_dim
    # Move the hidden dimension to the leading axis (batch*hidden)
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    M_total = B * hidden_dim                     # number of independent GEMMs
    left_flat  = left_perm.view(M_total, N, N)   # [M, N, N]  (FP16)
    right_flat = right_perm.view(M_total, N, N)  # [M, N, N]  (FP16)

    out_flat = torch.empty((M_total, N, N), dtype=torch.float32, device=left_flat.device)

    # -------------------------------------------------
    # Strides (int64 scalars)
    # -------------------------------------------------
    stride_am = left_flat.stride(0)
    stride_ai = left_flat.stride(1)
    stride_ak = left_flat.stride(2)

    stride_bm = right_flat.stride(0)
    # NOTE: In the kernel we read Bᵀ, therefore:
    stride_bj = right_flat.stride(1)   # row (j) stride
    stride_bk = right_flat.stride(2)   # col (k) stride

    stride_cm = out_flat.stride(0)
    stride_ci = out_flat.stride(1)
    stride_cj = out_flat.stride(2)

    # -------------------------------------------------
    # Kernel launch grid
    # -------------------------------------------------
    # The autotuned kernel decides the block sizes; we only supply the grid.
    grid = (
        M_total,                         # axis 0  → batch*hidden
        (N + 127) // 128,                # axis 1  → tile rows (i)
        (N + 127) // 128,                # axis 2  → tile cols (j)
    )
    # Launch.  Triton will pick the best BLOCK_* configuration based on `N`.
    batched_tri_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bj,
        stride_bk,
        stride_cm,
        stride_ci,
        stride_cj,
        # No need to pass BLOCK_* – autotune supplies them.
    )
    # -------------------------------------------------
    # Reshape back to [B, N, N, H] (float32)
    # -------------------------------------------------
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H]

    # -----------------------------------------------------------------
    # 5️⃣ Output LayerNorm, output‑gate, final projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate.float()                     # broadcast multiplication
    output = F.linear(out, weights["to_out.weight"]) # final bias‑free linear

    return output
