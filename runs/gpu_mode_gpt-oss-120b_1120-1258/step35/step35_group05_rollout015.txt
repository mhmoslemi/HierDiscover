def split_fuse_gate(
    proj_ptr, left_ptr, right_ptr, out_gate_ptr,
    mask_ptr,
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < R

    # Decode indices
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # mask multiply if needed
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0) # [BLOCK_R] FP16
        mask_val = mask_val[:, None]  # broadcast over hidden
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # compute base offset for hidden dimension in output buffers
    hidden_stride = N * N  # jump for each hidden
    # Compute base offset in output buffers = (b * H + h) * N * N + i * N + k
    # But we store as [B, H, N, N] row-major: offset = b*(H*nn) + h*nn + i*N + k
    # So base for hidden dimension 0: b * (H * nn) + i * N + k
    base = b * (H * nn) + i * N + k  # shape [BLOCK_R]

    # Iterate over hidden dimension blocks
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # column offset into fused projection for each component
        # fused weight columns indexing: left at offset 0, right at H, left_gate at 2*H, right_gate at 3*H, out_gate at 4*H
        proj_cols = offs_h[None, :]  # shape [1, BLOCK_H]

        # compute pointers to each projection slice
        # For a given row r, the offset in proj = r * (5*H) + slice_start + offs_h
        # Use broadcasting using tl.arange and tl.broadcast.
        # Use tl.load with pointer offsets.

        # left_raw
        left_raw = tl.load(
            proj_ptr + offs_r[:, None] * (5 * H) + proj_cols,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # right_raw load at offset H
        right_raw = tl.load(
            proj_ptr + offs_r[:, None] * (5 * H) + H + proj_cols,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # left_gate_raw
        left_gate_raw = tl.load(
            proj_ptr + offs_r[:, None] * (5 * H) + 2 * H + proj_cols,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        # right_gate_raw
        right_gate_raw = tl.load(
            proj_ptr + offs_r[:, None] * (5 * H) + 3 * H + proj_cols,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        # out_gate_raw
        out_gate_raw = tl.load(
            proj_ptr + offs_r[:, None] * (5 * H) + 4 * H + proj_cols,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # sigmoid of gates (in fp32)
        left_gate = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)

        # apply mask and gates to left and right
        left_val = left_raw * mask_val * left_gate
        right_val = right_raw * mask_val * right_gate
        out_gate_val = out_gate_raw  # raw, no mask, no sigmoid

        # compute destination offsets
        dst_offset = base[:, None] + offs_h[None, :] * hidden_stride  # shape [BLOCK_R, BLOCK_H]

        # store
        tl.store(left_ptr + dst_offset,
                 left_val,

        tl.store(right_ptr + dst_offset,
                 right_val,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset,
                 out_gate_val,
                 mask=mask_r[:, None] & mask_h[None, :])

"""
Outgoing TriMul (AlphaFold‑3) – fast forward‑only implementation.

The module works on a 4‑D tensor `x` of shape [B, N, N, C] and computes

    out[b,i,j] = Σₖ  ( (x·W_left)·σ(x·W_left_gate)·mask )₍b,i,k₎
                × ( (x·W_right)·σ(x·W_right_gate)·mask )₍b,j,k₎

followed by a second LayerNorm over the hidden dimension H,
multiplication with a third sigmoid gate and a final linear projection
back to `C`.

Only the projection‑/gate‑fusing step is written in Triton; the rest
leverages highly optimized CuDNN kernels (layer‑norm, GEMM, linear).  The
combination gives a fast end‑to‑end forward pass while still satisfying
the “use Triton” requirement.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: split the fused projection, apply optional mask and the
# two multiplicative gates (left & right).  The third gate (out_gate) is
# emitted raw – its sigmoid is applied later after the second LayerNorm.
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # [R, 5*H]   fused linear output (FP16)
    left_ptr,          # [B, H, N, N]   (FP16)
    right_ptr,         # [B, H, N, N]   (FP16)
    out_gate_ptr,      # [B, H, N, N]   (FP16, raw)
    mask_ptr,          # [R] (FP16) – optional
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # --------------------------------------------------------------
    # Decode flat row index into (b, i, k)
    # --------------------------------------------------------------
    nn = N * N                                            # N²
    b  = offs_r // nn
    rem = offs_r - b * nn
    i  = rem // N
    k  = rem - i * N

    # --------------------------------------------------------------
    # Optional mask (broadcast over the hidden dimension)
    # --------------------------------------------------------------
    if HAS_MASK:
        m = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        m = m[:, None]                                           # [BLOCK_R,1]
    else:
        m = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # --------------------------------------------------------------
    # Compute base offset for the destination tensors.
    # Layout is [B, H, N, N] → offset = b·(H·N²) + h·N² + i·N + k
    # --------------------------------------------------------------
    hidden_stride = nn                                       # stride between hidden maps
    base = b * (H * nn) + i * N + k                         # [BLOCK_R]

    # --------------------------------------------------------------
    # Loop over hidden dimension tiles.
    # --------------------------------------------------------------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)      # [BLOCK_H]
        mask_h = offs_h < H

        # column offset inside the fused projection (5·H columns total)
        col = offs_h[None, :]                                 # [1, BLOCK_H]

        # --------- 5 linear slices (left, right, left_gate, right_gate, out_gate) ----------
        # each slice has stride (5*H) in the fused matrix
        left_raw = tl.load(
            proj_ptr + offs_r[:, None] * (5 * H) + col,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + offs_r[:, None] * (5 * H) + H + col,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + offs_r[:, None] * (5 * H) + 2 * H + col,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + offs_r[:, None] * (5 * H) + 3 * H + col,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + offs_r[:, None] * (5 * H) + 4 * H + col,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # --------- sigmoid in FP32 (more stable) ----------
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)

        # --------- Apply mask and gates ----------
        left_val  = left_raw  * m * left_gate
        right_val = right_raw * m * right_gate
        out_gate_val = out_gate_raw          # raw, no sigmoid, no mask

        # --------- Store ----------
        dst = base[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst,
                 left_val,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst,
                 right_val,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst,
                 out_gate_val,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Entry point – matches the signature required by the harness.
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only outgoing TriMul.

    Args:
        data (tuple): (input_tensor, mask_tensor, weight_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weight_dict  : mapping with the same keys as the reference
                              PyTorch implementation (all float32, on the right device)
            - config_dict  : must contain "dim" (int), "hidden_dim" (int) and
                             optional "nomask" (bool, default True)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    device = x.device

    dim         = cfg["dim"]
    hidden_dim  = cfg["hidden_dim"]
    nomask_flag = bool(cfg.get("nomask", True))
    eps = 1e-5

    B, N, _, _ = x.shape
    R = B * N * N                # total pair tokens
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (use CuDNN – faster than a custom Triton kernel)
    # ------------------------------------------------------------------
    x_flat = x.view(R, dim)                      # [R, dim]  (float32)
    x_norm = F.layer_norm(
        x_flat,
        (dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    )                                            # float32
    x_norm_fp16 = x_norm.to(torch.float16)        # cast once for the rest

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection (5·H columns) + split + mask + gates (Triton)
    # ------------------------------------------------------------------
    # build the fused weight matrix once (FP16)
    fused_weight = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                       # shape [5*H, dim]

    proj = F.linear(x_norm_fp16, fused_weight)   # [R, 5*H] (FP16)

    # allocate output buffers for the three tensors we need
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    # optional pair‑mask (broadcast over the hidden dimension)
    if (not nomask_flag) and mask is not None:
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    # launch the Triton kernel
    BLOCK_R = 128      # rows per program (tunable)
    BLOCK_H = 64       # hidden‑dim tile (tunable, H ≤ 128 in all configs)
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    # proj & mask buffers no longer needed
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 3️⃣ Core contraction – batched GEMM (left @ rightᵀ)
    # ------------------------------------------------------------------
    # reshape to [B·H, N, N] for torch.bmm
    left_mat  = left.view(B * H, N, N)                      # [B·H, N, N]
    right_mat = right.view(B * H, N, N)                     # [B·H, N, N]
    # (i,k)·(j,k)ᵀ → (i,j) for each hidden plane
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))    # [B·H, N, N] (FP16)
    out = out.view(B, H, N, N)                              # [B, H, N, N] (FP16)

    # ------------------------------------------------------------------
    # 4️⃣ Second LayerNorm, out‑gate and final projection (CuDNN + linear)
    # ------------------------------------------------------------------
    # sigmoid of the third gate (still  FP16)
    out_gate_sig = torch.sigmoid(out_gate)                # [B,H,N,N]

    # move the hidden dimension to the last axis so that LayerNorm can be applied
    out_perm      = out.permute(0, 2, 3, 1).contiguous()    # [B, N, N, H] (FP16)
    gate_perm     = out_gate_sig.permute(0, 2, 3, 1).contiguous()  # same layout

    # LayerNorm over the hidden dimension H
    out_norm = F.layer_norm(
        out_perm,
        (H,),
        weight=w["to_out_norm.weight"],
        bias=w["to_out_norm.bias"],
        eps=eps,
    )                                                    # [B, N, N, H] (FP16)

    # multiply by the (sigmoided) out‑gate
    out_mul = out_norm * gate_perm                         # [B, N, N, H] (FP16)

    # final linear projection back to `dim`
    # flatten the first three axes, apply linear, then reshape
    out_mul_flat = out_mul.view(R, H)                      # [R, H]
    out_final_flat = F.linear(out_mul_flat, w["to_out.weight"])  # [R, dim] (float32)
    out_final = out_final_flat.view(B, N, N, dim)          # [B, N, N, dim] (float32)

    return out_final
