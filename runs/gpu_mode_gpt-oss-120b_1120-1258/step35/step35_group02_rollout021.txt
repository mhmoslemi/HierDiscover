"""
Outgoing TriMul (AlphaFold‑3) – fast forward‑only implementation.

Algorithm
---------
1️⃣ Layer‑norm on the flattened pairwise tensor (FP32 → FP16) via
   ``torch.nn.functional.layer_norm``.
2️⃣ Fuse the five linear maps (left, right, left_gate, right_gate, out_gate)
   into a single FP16 GEMM.
3️⃣ Split the fused projection, apply the optional mask, compute the
   sigmoid gates and store ``left``, ``right`` and ``out_gate`` tensors
   (shape [B, H, N, N], FP16) with a Triton kernel.
4️⃣ Core einsum is computed as a batched matrix‑multiply:
   ``out[b,h,i,j] = Σₖ left[b,h,i,k]·right[b,h,j,k]`` → torch.bmm.
5️⃣ Second LayerNorm (over the hidden dimension) + out‑gate multiplication
   are performed with ``torch.nn.functional.layer_norm``.
6️⃣ Final linear projection (H → dim) uses ``torch.nn.functional.linear``.
   The result is cast back to FP32.

Only the split‑and‑gate step is written in Triton – this satisfies the
requirement of having a custom kernel while keeping the heavy linear
algebra on cuBLAS/Tensor‑Core for maximum throughput.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: split fused projection, apply mask & sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,       # [R, 5*H]   FP16 fused projection
    left_ptr,       # [B, H, N, N] FP16
    right_ptr,      # [B, H, N, N] FP16
    out_gate_ptr,   # [B, H, N, N] FP16
    mask_ptr,       # [R]        FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)            # [BLOCK_R]
    mask_r = offs_r < R

    # Decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # Optional mask (broadcast over hidden dim)
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)      # [BLOCK_R]
        m_val = m_val[:, None]                                         # [BLOCK_R, 1]
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = nn                                   # stride between (b,i,k) slices for different h

    # Base address for the (b,i,k) position at h = 0
    base = b * (H * hidden_stride) + i * N + k          # [BLOCK_R]

    # Iterate over hidden dimension in chunks of BLOCK_H
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # column offsets inside the fused projection
        proj_col = offs_h[None, :]                         # [1, BLOCK_H]

        # left   : columns [0      , H)
        left_raw = tl.load(
            proj_ptr + offs_r[:, None] * (5 * H) + proj_col,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        # right  : columns [H      , 2H)
        right_raw = tl.load(
            proj_ptr + offs_r[:, None] * (5 * H) + proj_col + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        # left_gate  : columns [2H , 3H)
        left_gate_raw = tl.load(
            proj_ptr + offs_r[:, None] * (5 * H) + proj_col + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        # right_gate : columns [3H , 4H)
        right_gate_raw = tl.load(
            proj_ptr + offs_r[:, None] * (5 * H) + proj_col + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        # out_gate   : columns [4H , 5H)
        out_gate_raw = tl.load(
            proj_ptr + offs_r[:, None] * (5 * H) + proj_col + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # sigmoid (in fp32 for accuracy) → back to fp16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # Apply mask & gates
        left  = left_raw  * m_val * left_gate
        right = right_raw * m_val * right_gate
        outg  = out_gate

        # Destination offsets for left/right/out_gate tensors
        dst = base[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst, right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst, outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Entry‑point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only outgoing TriMul implementation.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
        - weights_dict : mapping with the following keys (all torch.Tensor):
            "norm.weight", "norm.bias",
            "left_proj.weight", "right_proj.weight",
            "left_gate.weight", "right_gate.weight", "out_gate.weight",
            "to_out_norm.weight", "to_out_norm.bias",
            "to_out.weight"
        - config_dict  : must contain "dim", "hidden_dim", optional "nomask" (bool)

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = bool(cfg.get("nomask", True))
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                      # number of pairwise tokens
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm over the last dimension (FP32 → FP16)
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                     # [R, dim] float32
    # builtin layernorm (FP32) → cast to FP16 for the rest of the pipeline
    x_norm_fp32 = F.layer_norm(
        x_flat,
        (dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    )
    x_norm = x_norm_fp32.to(torch.float16)

    # ------------------------------------------------------------------
    # 2️⃣ Fuse the five linear maps (left, right, left_gate, right_gate, out_gate)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(device=device, dtype=torch.float16)                 # [5*H, dim]

    proj = F.linear(x_norm, fused_w)                           # (R, 5*H) FP16
    del fused_w, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ Split, apply mask and compute sigmoid gates
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if not nomask and mask is not None:
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    # reasonable block sizes (H ≤ 128 in all test configs)
    BLOCK_R = 128
    BLOCK_H = 128
    grid = (triton.cdiv(R, BLOCK_R),)

    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Core einsum via batched GEMM (Tensor‑Core)
    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)                 # FP16
    right_mat = right.view(B * H, N, N)                # FP16

    # out[b, h, i, j] = Σₖ left[b, h, i, k] * right[b, h, j, k]
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))
    out = out.view(B, H, N, N)                        # [B, H, N, N] FP16

    # free memory that is no longer needed
    del left, right

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm (over hidden dim) + out‑gate multiplication
    # ------------------------------------------------------------------
    # move hidden dimension to the tail for the builtin LayerNorm
    out_perm = out.permute(0, 2, 3, 1).contiguous()    # [B, N, N, H] FP16

    to_out_norm_weight = w["to_out_norm.weight"].to(torch.float16)
    to_out_norm_bias   = w["to_out_norm.bias"].to(torch.float16)

    out_norm = F.layer_norm(
        out_perm,
        (H,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )                                                   # [B, N, N, H] FP16

    out_norm = out_norm.permute(0, 3, 1, 2)            # back to [B, H, N, N]

    # element‑wise multiplication with the gate tensor
    out_gated = out_norm * out_gate
    del out_norm, out_gate

    # ------------------------------------------------------------------
    # 6️⃣ Final linear projection (H → dim)
    # ------------------------------------------------------------------
    # flatten the first three dimensions and apply a normal Linear layer
    out_flat = out_gated.permute(0, 2, 3, 1).reshape(R, H)   # [R, H] FP16
    to_out_weight = w["to_out.weight"].to(torch.float16)    # [dim, H]

    final_fp16 = F.linear(out_flat, to_out_weight)          # [R, dim] FP16
    final = final_fp16.view(B, N, N, dim).to(torch.float32) # cast to FP32

    return final
