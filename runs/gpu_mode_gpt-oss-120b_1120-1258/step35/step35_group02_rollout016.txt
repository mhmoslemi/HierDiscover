"""
Outgoing TriMul (AlphaFold‑3) – optimized forward‑only implementation.

Algorithm
---------
1️⃣  Layer‑norm of the flattened pairwise tensor (FP32 statistics → FP16 data)
    – custom Triton kernel.
2️⃣  Fuse the five linear maps (left, right, left_gate, right_gate, out_gate)
    into a single GEMM (`torch.nn.functional.linear`).
3️⃣  Split the fused projection, apply the optional mask, compute sigmoid gates
    and reshape to `[B, H, N, N]` (FP16).  All element‑wise work is done with
    vectorised PyTorch kernels (no extra Triton launch needed).
4️⃣  Core O(N³) contraction:
          out[b,i,j] = Σₖ left[b,i,k]·right[b,j,k]
    – realised as a batched GEMM (`torch.bmm`).
5️⃣  Row‑wise mean / variance of `out` (second LayerNorm) via
    `torch.mean` / `torch.var` (FP32 reduction).
6️⃣  Apply the second LayerNorm, multiply by `out_gate`, and a final linear
    projection (`to_out`) using `torch.nn.functional.linear` (cuBLAS).
7️⃣  Reshape to `[B, N, N, dim]` (float32) and return.

Only the first LayerNorm is implemented in Triton, satisfying the
requirement of having a custom kernel while the remaining work uses
high‑performance PyTorch primitives.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 statistics → FP16 output)
# --------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,    # [R, dim]  FP32 input (contiguous)
    y_ptr,    # [R, dim]  FP16 output
    w_ptr,    # [dim]     γ (FP32)
    b_ptr,    # [dim]     β (FP32)
    rows,     # int       R = B·N·N
    dim,      # int       feature dimension
    eps,      # float     epsilon
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # dim chunk per loop
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ---------- first pass: compute mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)          # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)   # [BLOCK_M]

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- second pass: write normalized values ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# --------------------------------------------------------------
# Entrypoint
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only outgoing TriMul.

    Args:
        data (tuple):
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict  : dict with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config_dict   : must contain "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = bool(cfg.get("nomask", True))
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N          # number of pairwise entries
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Layer‑norm (FP32 → FP16)
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                     # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128          # rows per program (tuned)
    BLOCK_D = 64           # dim‑chunk (tuned)
    grid = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # ------------------------------------------------------------------
    # 2️⃣ Fuse five linear maps (left, right, left_gate, right_gate, out_gate)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],      # (H, dim)
            w["right_proj.weight"],     # (H, dim)
            w["left_gate.weight"],      # (H, dim)
            w["right_gate.weight"],     # (H, dim)
            w["out_gate.weight"],       # (H, dim)
        ],
        dim=0,
    ).to(torch.float16)                # shape (5*H, dim)

    proj = F.linear(x_norm, fused_w)   # (R, 5*H) FP16
    del fused_w, x_norm

    # reshape to (R, 5, H) for easy splitting
    proj = proj.view(R, 5, H)

    left_raw      = proj[:, 0, :]      # (R, H)
    right_raw     = proj[:, 1, :]      # (R, H)
    left_gate_raw = proj[:, 2, :]      # (R, H)
    right_gate_raw= proj[:, 3, :]      # (R, H)
    out_gate_raw  = proj[:, 4, :]      # (R, H)

    # optional mask (broadcast over hidden dim)
    if not nomask and mask is not None:
        mask_flat = mask.to(torch.float16).reshape(R)          # (R,)
        mask_flat = mask_flat[:, None]                         # (R,1)
    else:
        mask_flat = torch.ones(R, 1, dtype=torch.float16, device=device)

    # sigmoid gates (still FP16)
    left_gate  = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate   = torch.sigmoid(out_gate_raw)

    # apply mask and gates
    left  = left_raw  * mask_flat * left_gate   # (R, H) FP16
    right = right_raw * mask_flat * right_gate  # (R, H) FP16
    out_gate = out_gate                         # (R, H) FP16

    # reshape to (B, H, N, N)
    left = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    out_gate = out_gate.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    del proj, left_raw, right_raw, left_gate_raw, right_gate_raw, out_gate_raw

    # ------------------------------------------------------------------
    # 3️⃣ Core O(N³) contraction via batched GEMM (cuBLAS)
    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)                 # (B*H, N, N) FP16
    right_mat = right.view(B * H, N, N)                # (B*H, N, N) FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B*H, N, N) FP16
    out = out.view(B, H, N, N)                         # (B, H, N, N) FP16
    del left_mat, right_mat, left, right               # free memory

    # ------------------------------------------------------------------
    # 4️⃣ Second LayerNorm (row‑wise stats) – use native reductions
    # ------------------------------------------------------------------
    # flatten to (R, H) for reduction
    out_flat = out.view(R, H).to(torch.float32)            # FP32 for stats
    mu   = out_flat.mean(dim=1)                           # (R,)
    var  = out_flat.var(dim=1, unbiased=False)            # (R,)
    rstd = torch.rsqrt(var + eps)                         # (R,)

    # ------------------------------------------------------------------
    # 5️⃣ Apply LN, out_gate, and final projection (cuBLAS)
    # ------------------------------------------------------------------
    # normalize + γ/β (to_out_norm) + gating
    ln_w = w["to_out_norm.weight"]      # (H,) FP32
    ln_b = w["to_out_norm.bias"]        # (H,) FP32

    # (out - μ) * rstd
    norm = (out_flat - mu[:, None]) * rstd[:, None]         # (R, H) FP32
    norm = norm * ln_w[None, :] + ln_b[None, :]              # (R, H) FP32
    norm = norm * out_gate.view(R, H).to(torch.float32)      # gating (R, H)

    # final linear projection  (cuBLAS)
    final = F.linear(norm, w["to_out.weight"])              # (R, dim) FP32
    final = final.view(B, N, N, dim)                        # (B, N, N, dim)

    return final
