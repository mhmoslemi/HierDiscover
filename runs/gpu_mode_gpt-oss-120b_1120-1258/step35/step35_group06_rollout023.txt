def rownorm_stats_kernel(x_ptr, mu_ptr, rstd_ptr, rows, dim, eps, BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0)
        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)
    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)
    tl.store(mu_ptr + offs_m, mean, mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)

"""
Outgoing TriMul (AlphaFold‑3) – highly‑optimized forward pass.

Algorithm
---------
1️⃣ Row‑wise statistics (mean & rsqrt‑variance) of the raw pair tensor.
2️⃣ Fuse the 5 linear maps (left, right, left_gate, right_gate, out_gate) into a
   single GEMM; the LayerNorm γ (weight) is merged into the weights, and the
   β (bias) is pre‑computed as an extra bias term.
3️⃣ One Triton kernel (split_scale_gate) reads the GEMM output, applies the
   per‑row (µ,σ) scaling, adds the pre‑computed β, multiplies the optional
   pair‑mask, evaluates the three sigmoid gates and writes `left`, `right`,
   `out_gate` in layout `[B, H, N, N]` (FP16).
4️⃣ Batched matrix‑multiply (cuBLAS) computes the core einsum:
       out[b, h, i, j] = Σₖ left[b, h, i, k] · right[b, h, j, k]
5️⃣ Row‑wise mean & rsqrt‑variance of `out` (over the hidden dim H).
6️⃣ Final Triton kernel fuses the second LayerNorm, the `out_gate` gating and
   the projection to the output dimension.  The result is `[B, N, N, dim]`
   (FP32).

All kernels use mixed‑precision (FP32 statistics, FP16 bulk) and target
Triton 3.3.1 on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ---------------------------------------------------------------
# 1️⃣ Row‑wise mean / rsqrt‑variance of the raw input (no write‑back)
# ---------------------------------------------------------------
@triton.jit
def rownorm_stats_kernel(
    x_ptr,        # [R, dim] FP32
    mu_ptr,       # [R]    FP32
    rstd_ptr,     # [R]    FP32
    rows, dim,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0)                       # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)          # [BLOCK_M]

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_m, mean,  mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)


# ---------------------------------------------------------------
# 2️⃣ Split, apply mask, scaling, sigmoid‑gates, write left/right/out_gate
# ---------------------------------------------------------------
@triton.jit
def split_scale_gate_kernel(
    proj_ptr,      # [R, 5*H]  FP16   (raw GEMM output)
    left_ptr,      # [B, H, N, N] FP16
    right_ptr,     # [B, H, N, N] FP16
    out_gate_ptr,  # [B, H, N, N] FP16
    mu_ptr,        # [R] FP32
    rstd_ptr,      # [R] FP32
    B_vec_ptr,     # [5*H] FP32  ( Σ_d γ_d·W )
    C_vec_ptr,     # [5*H] FP32  ( Σ_d β_d·W )
    mask_ptr,      # [R] FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)               # [BLOCK_R]
    mask_r = offs_r < R

    # decode (b,i,k) from flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional pair‑mask (broadcast over hidden dim)
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        m_val = m_val[:, None]                                      # broadcast
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # common offsets for the three output tensors
    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + k                     # start of h‑plane 0

    # load per‑row statistics once (float32)
    mu   = tl.load(mu_ptr   + offs_r, mask=mask_r, other=0.0)      # [BLOCK_R]
    rstd = tl.load(rstd_ptr + offs_r, mask=mask_r, other=0.0)      # [BLOCK_R]

    # -------------------------------------------------------------------
    # Process hidden dimension in tiles
    # -------------------------------------------------------------------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)         # [BLOCK_H]
        mask_h = offs_h < H

        # ----------- load the five raw projection slices -----------------
        # layout: [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw      = tl.load(proj_ptr + proj_base,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_raw     = tl.load(proj_ptr + proj_base + H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        left_gate_raw = tl.load(proj_ptr + proj_base + 2 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_gate_raw = tl.load(proj_ptr + proj_base + 3 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        out_gate_raw  = tl.load(proj_ptr + proj_base + 4 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)

        # ----------- load B / C vectors for the current hidden tile -----
        # (these are the pre‑computed Σ_d γ·W and Σ_d β·W )
        off = 0
        B_l = tl.load(B_vec_ptr + off + offs_h, mask=mask_h, other=0.0)
        B_r = tl.load(B_vec_ptr + H   + offs_h, mask=mask_h, other=0.0)
        B_lg = tl.load(B_vec_ptr + 2*H + offs_h, mask=mask_h, other=0.0)
        B_rg = tl.load(B_vec_ptr + 3*H + offs_h, mask=mask_h, other=0.0)
        B_og = tl.load(B_vec_ptr + 4*H + offs_h, mask=mask_h, other=0.0)

        C_l = tl.load(C_vec_ptr + off + offs_h, mask=mask_h, other=0.0)
        C_r = tl.load(C_vec_ptr + H   + offs_h, mask=mask_h, other=0.0)
        C_lg = tl.load(C_vec_ptr + 2*H + offs_h, mask=mask_h, other=0.0)
        C_rg = tl.load(C_vec_ptr + 3*H + offs_h, mask=mask_h, other=0.0)
        C_og = tl.load(C_vec_ptr + 4*H + offs_h, mask=mask_h, other=0.0)

        # ----------------------------------------------------------------
        # Apply the (µ,σ) scaling:   out = (raw - µ·B)·σ + C
        # ----------------------------------------------------------------
        mu_b  = mu[:, None]    # [BLOCK_R,1]
        rstd_b = rstd[:, None] # [BLOCK_R,1]

        left_fp32      = left_raw.to(tl.float32)
        right_fp32     = right_raw.to(tl.float32)
        left_gate_fp32 = left_gate_raw.to(tl.float32)
        right_gate_fp32= right_gate_raw.to(tl.float32)
        out_gate_fp32  = out_gate_raw.to(tl.float32)

        left_scaled      = (left_fp32      - mu_b * B_l ) * rstd_b + C_l
        right_scaled     = (right_fp32     - mu_b * B_r ) * rstd_b + C_r
        left_gate_scaled = (left_gate_fp32- mu_b * B_lg) * rstd_b + C_lg
        right_gate_scaled= (right_gate_fp32- mu_b * B_rg) * rstd_b + C_rg
        out_gate_scaled  = (out_gate_fp32 - mu_b * B_og) * rstd_b + C_og

        # ---------------------------------------------------------------
        # Sigmoid for the three gates (computed in FP32, stored as FP16)
        # ---------------------------------------------------------------
        sigmoid = lambda x: (1.0 / (1.0 + tl.exp(-x))).to(tl.float16)

        left_gate  = sigmoid(left_gate_scaled)
        right_gate = sigmoid(right_gate_scaled)
        out_gate   = sigmoid(out_gate_scaled)

        # ---------------------------------------------------------------
        # Apply mask (only to left/right) and combine with gates
        # ---------------------------------------------------------------
        left  = left_scaled.to(tl.float16) * m_val * left_gate
        right = right_scaled.to(tl.float16) * m_val * right_gate
        outg  = out_gate

        # ----------------------------------------------------------------
        # Store results in layout [B, H, N, N]
        # ----------------------------------------------------------------
        dst = base[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr     + dst, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr    + dst, right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst, outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 3️⃣ Row‑wise mean & rsqrt‑variance of the contracted tensor (out)
# ----------------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr,   # [B, H, N, N]  FP16
    mu_ptr,    # [R]            FP32
    rstd_ptr,  # [R]            FP32
    total_rows, H, N,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + j          # start of hidden plane 0

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        ptr = out_ptr + base[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)        # [BLOCK_M, BLOCK_K]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_m, mean,  mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)


# ----------------------------------------------------------------------
# 4️⃣ Fuse second LayerNorm, out_gate, and final projection
# ----------------------------------------------------------------------
@triton.jit
def final_output_kernel(
    out_ptr,        # [B, H, N, N]  FP16
    gate_ptr,       # [B, H, N, N]  FP16 (out_gate)
    mu_ptr,         # [R]            FP32
    rstd_ptr,       # [R]            FP32
    ln_w_ptr,       # [H]            FP32 (to_out_norm.weight)
    ln_b_ptr,       # [H]            FP32 (to_out_norm.bias)
    w_out_T_ptr,    # [H, dim]       FP16 (to_out weight transposed)
    out_final_ptr,  # [B, N, N, dim] FP32
    total_rows, H, N, dim,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr,
):
    pid_m = tl.program_id(0)   # rows (R)
    pid_n = tl.program_id(1)   # output‑dim tiles (dim)

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)   # [BLOCK_M]
    offs_n = col_start + tl.arange(0, BLOCK_N)   # [BLOCK_N]

    mask_m = offs_m < total_rows
    mask_n = offs_n < dim

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + j   # start of hidden plane 0

    # per‑row stats
    mu   = tl.load(mu_ptr   + offs_m, mask=mask_m, other=0.0)   # [BLOCK_M]
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)   # [BLOCK_M]

    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)               # [BLOCK_K]
        mask_k = offs_k < H

        # load contracted tensor and gate
        ptr_out  = out_ptr  + base[:, None] + offs_k[None, :] * hidden_stride
        ptr_gate = gate_ptr + base[:, None] + offs_k[None, :] * hidden_stride

        vals_out = tl.load(ptr_out,
                           mask=mask_m[:, None] & mask_k[None, :],
                           other=0.0).to(tl.float32)          # [M, K]
        vals_gate = tl.load(ptr_gate,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)         # [M, K]

        # LayerNorm params (γ,β) for the second LN
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)  # [K]
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)  # [K]

        # (out - µ)·rstd → γ· + β → * out_gate
        diff = (vals_out - mu[:, None]) * rstd[:, None]               # [M, K]
        norm = diff * gamma[None, :] + beta[None, :]                  # [M, K]
        term = norm * vals_gate                                       # [M, K]

        # Load slice of final projection weight (H, dim) – already transposed
        w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]   # [K, TILE_N]
        w_tile = tl.load(w_ptr,
                         mask=mask_k[:, None] & mask_n[None, :],
                         other=0.0)                                     # [K, TILE_N]

        acc += tl.dot(term.to(tl.float16), w_tile).to(tl.float32)

    # write final output
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entrypoint
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul operator (no gradients).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : dict with the following keys (float32 tensors, on device)
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config_dict must contain keys "dim", "hidden_dim", "nomask"
    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    device = x.device
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = bool(cfg.get("nomask", True))
    eps = 1e-5

    B, N, _, _ = x.shape
    R = B * N * N                 # total number of pairwise tokens
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ Row‑wise mean & rsqrt‑variance of the raw input (no write‑back)
    # --------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()          # [R, dim] FP32
    mu   = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 128
    BLOCK_D_STATS = 64
    grid_stats = (triton.cdiv(R, BLOCK_M_STATS),)
    rownorm_stats_kernel[grid_stats](
        x_flat,
        mu,
        rstd,
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_D=BLOCK_D_STATS,
        num_warps=4,
    )
    del x_flat

    # --------------------------------------------------------------
    # 2️⃣ Fuse five linear maps (including LayerNorm γ) into a single GEMM
    # --------------------------------------------------------------
    # concatenate original weights (float32)
    orig_fused = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float32)                                 # [5·H, dim]

    # incorporate LayerNorm γ (weight) → scaled weight
    gamma = w["norm.weight"]                             # [dim]
    scaled_fused = orig_fused * gamma[None, :]            # [5·H, dim]

    # pre‑compute Σ_d γ·W  (used for µ‑scaling) and Σ_d β·W (bias term)
    B_vec = scaled_fused.sum(dim=1).contiguous()         # [5·H] float32
    beta = w["norm.bias"]                                # [dim]
    C_vec = (orig_fused * beta[None, :]).sum(dim=1).contiguous()  # [5·H] float32

    # move scaled weight to FP16 for the GEMM
    scaled_fused_fp16 = scaled_fused.to(torch.float16).contiguous()

    # GEMM:  (R, dim) @ (dim, 5·H)^T  → (R, 5·H)   (output FP16)
    # using FP32 input (avoids an extra FP16 copy of x)
    proj = torch.nn.functional.linear(x, scaled_fused_fp16, bias=None)  # FP16
    del x, scaled_fused, orig_fused

    # --------------------------------------------------------------
    # 3️⃣ Split, apply mask, µ‑/σ‑scaling, sigmoid‑gates
    # --------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    # optional mask handling
    if not nomask and mask is not None:
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)

    split_scale_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mu,
        rstd,
        B_vec,
        C_vec,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mu, rstd, B_vec, C_vec, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣ Batched matrix‑multiply (cuBLAS) over the hidden dimension
    # --------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)               # FP16
    right_mat = right.view(B * H, N, N)              # FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))
    out = out.view(B, H, N, N)                       # [B, H, N, N] FP16

    # --------------------------------------------------------------
    # 5️⃣ Row‑wise mean & rsqrt‑variance of `out`
    # --------------------------------------------------------------
    mu_out   = torch.empty(R, dtype=torch.float32, device=device)
    rstd_out = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 256
    BLOCK_K_OUT = 128                     # H ≤ 128 → single tile
    grid_stats2 = (triton.cdiv(R, BLOCK_M_OUT),)
    rowwise_stats_kernel[grid_stats2](
        out,
        mu_out,
        rstd_out,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm + out_gate + final projection
    # --------------------------------------------------------------
    # to_out weight transposed to [H, dim] for easy strided loading
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()  # [H, dim]

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_FINAL = 128   # rows per program
    BLOCK_K_FINAL = 32    # hidden‑dim tile size
    BLOCK_N_FINAL = 128   # output‑dim (dim) tile size

    grid_final = (
        triton.cdiv(R, BLOCK_M_FINAL),
        triton.cdiv(dim, BLOCK_N_FINAL),
    )
    final_output_kernel[grid_final](
        out,
        out_gate,
        mu_out,
        rstd_out,
        w["to_out_norm.weight"],   # γ (FP32)
        w["to_out_norm.bias"],     # β (FP32)
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_FINAL,
        BLOCK_K=BLOCK_K_FINAL,
        BLOCK_N=BLOCK_N_FINAL,
        num_warps=8,
    )
    return final
