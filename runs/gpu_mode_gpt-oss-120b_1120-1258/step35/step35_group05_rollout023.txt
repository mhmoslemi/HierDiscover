"""
Outgoing TriMul (AlphaFold‑3) – fused‑kernel Triton implementation.

The operator works on a 4‑D tensor `x` of shape [B, N, N, C] and computes

    out[b,i,j] = Σₖ  ( (x·W_left)·σ(x·W_left_gate)·mask )₍b,i,k₎
                × ( (x·W_right)·σ(x·W_right_gate)·mask )₍b,j,k₎

followed by a second LayerNorm over the hidden dimension H,
an output‑gate, and a linear projection back to the original channel dimension C.

Key optimisations
-----------------
* ① Row‑wise LayerNorm (FP32→FP16) – one Triton kernel.
* ② Fused linear projection for the 5·H maps (left, right, three gates) – one
   kernel that also applies the pair‑mask and sigmoid gates.
* ③ Core N×N×N contraction performed by a single batched GEMM
   (`torch.bmm`).  This uses cuBLAS which is optimal on an H100.
* ④ Row‑wise mean / variance and the final LayerNorm, output‑gate, and
   projection are **fused** into a single Triton kernel.  The contracted
   tensor `out` is read only once, saved in shared memory, statistics are
   accumulated, and the final output is produced with a tiled dot‑product.
   This reduces global‑memory traffic by ~50 % compared with the previous
   two‑kernel design.

All kernels run in mixed‑precision (FP16 for bulk data, FP32 for statistics)
and are tuned for NVIDIA H100 (Triton 3.3.1).

The entry‑point `custom_kernel` follows the signature required by the test harness.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,    # [R, dim]   FP32 input
    y_ptr,    # [R, dim]   FP16 output
    w_ptr,    # [dim]      γ (FP32)
    b_ptr,    # [dim]      β (FP32)
    rows,     # scalar R
    dim,      # scalar dim
    eps,      # scalar eps
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # dim chunk per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)            # [BLOCK_M]
    mask_m = offs_m < rows

    # ---------- first pass: mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- second pass: write normalised values ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, apply mask and sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,      # [R, 5·H]  FP16 fused projection
    left_ptr,      # [B, H, N, N] FP16
    right_ptr,     # [B, H, N, N] FP16
    out_gate_ptr,  # [B, H, N, N] FP16
    mask_ptr,      # [R]       FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)           # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask (broadcast over hidden dimension)
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r,
                        mask=mask_r,
                        other=1.0)                     # [BLOCK_R]
        m_val = m_val[:, None]                          # [BLOCK_R, 1]
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N                      # stride between hidden planes
    base = b * (H * hidden_stride) + i * N + k  # start of hidden‑plane 0

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # column offset inside fused projection (5·H columns)
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # sigmoid (FP32 for stability, back to FP16)
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # Apply mask and gates
        left = left_raw * m_val * left_gate
        right = right_raw * m_val * right_gate
        outg = out_gate          # out_gate is NOT masked

        dst = base[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst,
                 outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 4️⃣ Fuse 2nd LayerNorm, out_gate and final projection (single read of `out`)
# ----------------------------------------------------------------------
@triton.jit
def fused_final_output_kernel(
    out_ptr,          # [B, H, N, N]            FP16   contracted tensor
    gate_ptr,         # [B, H, N, N]            FP16   out_gate
    ln_w_ptr,         # [H]                     FP32   to_out_norm.weight (γ)
    ln_b_ptr,         # [H]                     FP32   to_out_norm.bias   (β)
    w_out_T_ptr,      # [H, dim]                FP16   (to_out weight transposed)
    out_final_ptr,    # [B, N, N, dim]          FP32   final output
    total_rows, H, N, dim,
    eps,
    BLOCK_M: tl.constexpr,   # rows per program (flattened B·N·N)
    BLOCK_K: tl.constexpr,   # hidden‑dim tile (≤ H)
    BLOCK_N: tl.constexpr,   # output‑dim tile
):
    pid_m = tl.program_id(0)   # rows
    pid_n = tl.program_id(1)   # output‑dim tiles

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    offs_n = col_start + tl.arange(0, BLOCK_N)          # [BLOCK_N]

    mask_m = offs_m < total_rows
    mask_n = offs_n < dim

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + j         # start of hidden‑plane 0

    # ------------------------------------------------------------------
    # 1) Load the contracted tensor and the out‑gate into shared memory,
    #    simultaneously accumulate per‑row sum / sum‑sq for mean & var.
    # ------------------------------------------------------------------
    MAX_H = 128                         # compile‑time upper bound for hidden dim

    out_sh = tl.shared([BLOCK_M, MAX_H], dtype=tl.float16)
    gate_sh = tl.shared([BLOCK_M, MAX_H], dtype=tl.float16)

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for h_start in range(0, H, BLOCK_K):
        offs_h = h_start + tl.arange(0, BLOCK_K)           # [BLOCK_K]
        mask_h = offs_h < H

        # global pointers for this hidden slice
        out_ptrs  = out_ptr  + base[:, None] + offs_h[None, :] * hidden_stride
        gate_ptrs = gate_ptr + base[:, None] + offs_h[None, :] * hidden_stride

        vals_out  = tl.load(out_ptrs,
                            mask=mask_m[:, None] & mask_h[None, :],
                            other=0.0).to(tl.float32)
        vals_gate = tl.load(gate_ptrs,
                            mask=mask_m[:, None] & mask_h[None, :],
                            other=0.0).to(tl.float32)

        # store into shared memory (FP16)
        row_off = offs_m * MAX_H                         # [BLOCK_M]
        sh_ptr_out  = out_sh  + (row_off[:, None] + offs_h[None, :])
        sh_ptr_gate = gate_sh + (row_off[:, None] + offs_h[None, :])

        tl.store(sh_ptr_out,
                 vals_out.to(tl.float16),
                 mask=mask_m[:, None] & mask_h[None, :])
        tl.store(sh_ptr_gate,
                 vals_gate.to(tl.float16),
                 mask=mask_m[:, None] & mask_h[None, :])

        sum_val += tl.sum(vals_out, axis=1)
        sum_sq  += tl.sum(vals_out * vals_out, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------------------------------------------------------
    # 2) Second pass: compute (LN(out) * out_gate) and multiply with w_out_T.
    # ------------------------------------------------------------------
    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    # broadcast the per‑row statistics for the hidden‑tile loops
    mean_b = mean[:, None]    # [BLOCK_M,1]
    rstd_b = rstd[:, None]    # [BLOCK_M,1]

    for h_start in range(0, H, BLOCK_K):
        offs_h = h_start + tl.arange(0, BLOCK_K)           # [BLOCK_K]
        mask_h = offs_h < H

        # load slices from shared memory
        row_off = offs_m * MAX_H
        sh_out_ptr  = out_sh  + (row_off[:, None] + offs_h[None, :])
        sh_gate_ptr = gate_sh + (row_off[:, None] + offs_h[None, :])

        out_slice  = tl.load(sh_out_ptr,
                             mask=mask_m[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float32)   # [M,K]
        gate_slice = tl.load(sh_gate_ptr,
                             mask=mask_m[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float32)   # [M,K]

        # ----- LayerNorm parameters (γ,β) -----
        gamma = tl.load(ln_w_ptr + offs_h, mask=mask_h, other=0.0)   # [K]  FP32
        beta  = tl.load(ln_b_ptr + offs_h, mask=mask_h, other=0.0)   # [K]  FP32

        # ----- Normalisation + output‑gate -----
        # (out - μ) * rstd
        norm = (out_slice - mean_b) * rstd_b                 # [M,K]
        # γ·norm + β
        norm = norm * gamma[None, :] + beta[None, :]         # [M,K]
        # multiply by out_gate
        term_fp16 = (norm * gate_slice).to(tl.float16)       # [M,K] FP16

        # ----- Dot with final weight (H×dim) -----
        w_ptr = w_out_T_ptr + offs_h[:, None] * dim + offs_n[None, :]   # [K, TILE_N]
        w_tile = tl.load(w_ptr,
                         mask=mask_h[:, None] & mask_n[None, :],
                         other=0.0)                                    # [K, TILE_N] FP16

        # accumulate
        acc += tl.dot(term_fp16, w_tile).to(tl.float32)

    # ------------------------------------------------------------------
    # 3) Write final output  (flattened rows × dim)
    # ------------------------------------------------------------------
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, acc,
             mask=mask_m[:, None] & mask_n[None, :])

# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only outgoing TriMul.

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping with keys
                  "norm.weight", "norm.bias",
                  "left_proj.weight", "right_proj.weight",
                  "left_gate.weight", "right_gate.weight", "out_gate.weight",
                  "to_out_norm.weight", "to_out_norm.bias",
                  "to_out.weight"
              (all float32, already on the same device as inputs)
            - config_dict must contain:
                  "dim" (int), "hidden_dim" (int), "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # ---------------------------------------------------------------
    # unpack
    # ---------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = bool(cfg.get("nomask", True))
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                 # total pair tokens
    H = hidden_dim

    # ---------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32 → FP16)
    # ---------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                       # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 128   # larger tile for the dimension (max dim = 384)
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # ---------------------------------------------------------------
    # 2️⃣ Fuse five linear maps (left, right, left_gate, right_gate, out_gate)
    # ---------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                         # [5·H, dim]

    proj = F.linear(x_norm, fused_w)            # (R, 5·H) FP16
    del fused_w, x_norm

    # ---------------------------------------------------------------
    # 3️⃣ Split, apply optional mask and sigmoid gates
    # ---------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64
    BLOCK_H = 128            # tile the hidden dimension in one pass
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ---------------------------------------------------------------
    # 4️⃣ Core einsum via batched GEMM (left @ rightᵀ)
    # ---------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)                     # [B·H, N, N]
    right_mat = right.view(B * H, N, N)                    # [B·H, N, N]
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                             # [B, H, N, N] FP16

    # ---------------------------------------------------------------
    # 5️⃣ Fuse 2nd LayerNorm, out_gate, and final projection
    # ---------------------------------------------------------------
    # to_out.weight : [dim, hidden_dim] → transpose → [H, dim]
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim]

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 128    # rows per program (flattened B·N·N)
    BLOCK_K_OUT = 32     # hidden‑dim tile (must divide H≤128)
    BLOCK_N_OUT = 128    # output‑dim tile

    grid_out = (
        triton.cdiv(R, BLOCK_M_OUT),
        triton.cdiv(dim, BLOCK_N_OUT),
    )
    fused_final_output_kernel[grid_out](
        out,
        out_gate,
        w["to_out_norm.weight"],   # γ  FP32
        w["to_out_norm.bias"],     # β  FP32
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=8,
    )

    return final
