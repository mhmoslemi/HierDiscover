"""
Outgoing TriMul (AlphaFold‑3) – fast forward‑only implementation.

Algorithm
---------
1️⃣  Input pairwise tensor (`x`) → LayerNorm (FP32) → cast to FP16.
2️⃣  Fuse the five linear maps (left, right, left_gate, right_gate, out_gate)
    into a single FP16 GEMM (`torch.nn.functional.linear`).
3️⃣  Triton kernel `split_fuse_gate_kernel`:
    * splits the fused projection into the five parts,
    * applies the optional mask on *left*/*right*,
    * computes sigmoid gates,
    * writes `left`, `right` and `out_gate` in layout [B, H, N, N] (FP16).
4️⃣  Core einsum ⇢ batched GEMM (`torch.bmm`) :
    `out[b, h, i, j] = Σₖ left[b, h, i, k]·right[b, h, j, k]`.
5️⃣  Second LayerNorm is performed with standard PyTorch reductions
    (mean / variance) on `out` (FP32), followed by the learned
    scale/shift (`to_out_norm`), then element‑wise multiplication with
    `out_gate`.
6️⃣  Final linear projection (`to_out`) is a single GEMM on the
    flattened tensor (`torch.nn.functional.linear`).

Only step 3 is a custom Triton kernel – the rest uses highly‑optimized
PyTorch/cuBLAS kernels.  The implementation works for all test
configurations (varying batch size, sequence length, hidden/feature
dimensions and optional masks).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: split fused projection, apply mask, compute sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,      # [R, 5·H]  FP16 fused projection
    left_ptr,      # [B, H, N, N] FP16
    right_ptr,     # [B, H, N, N] FP16
    out_gate_ptr,  # [B, H, N, N] FP16
    mask_ptr,      # [R] FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)           # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index r = b·N·N + i·N + k   (k is the summed‑over axis)
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask (float16 0/1) broadcast over hidden dimension
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r,
                        mask=mask_r,
                        other=1.0)                     # [BLOCK_R]
        m_val = m_val[:, None]                         # [BLOCK_R,1]
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N                # stride between hidden slices

    # base address of hidden‑plane h = 0 for (b,i,k)
    base = b * (H * hidden_stride) + i * N + k

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # layout of proj: [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # sigmoid in FP32 → back to FP16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * m_val * left_gate
        right = right_raw * m_val * right_gate
        outg  = out_gate

        dst = base[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst,
                 outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Entrypoint
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only outgoing TriMul.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor   [B, N, N, dim]  (float32)
        - mask_tensor  : torch.Tensor   [B, N, N]       (bool/float) or None
        - weights_dict : dict with the exact keys used by the reference module
        - config_dict  : must contain ``dim`` and ``hidden_dim``, optional ``nomask``

    Returns
    -------
    torch.Tensor : [B, N, N, dim] (float32)
    """
    # unpack
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = bool(cfg.get("nomask", True))
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                # number of pairwise tokens
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣  LayerNorm on the original pairwise tensor (FP32 → FP16)
    # ------------------------------------------------------------------
    # torch.nn.functional.layer_norm is highly optimized on H100
    x_norm = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    )                               # [B, N, N, dim]  float32
    x_norm = x_norm.to(torch.float16)               # cast once for linear proj

    # ------------------------------------------------------------------
    # 2️⃣  Fuse the five linear maps into a single GEMM
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                      # [5·H, dim]

    x_flat = x_norm.view(R, dim)                # [R, dim]
    proj = F.linear(x_flat, fused_w)            # [R, 5·H]  FP16
    del x_flat, fused_w

    # ------------------------------------------------------------------
    # 3️⃣  Split, mask & compute sigmoid gates  (Triton kernel)
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣  Core einsum via batched GEMM (cuBLAS)
    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)                # FP16
    right_mat = right.view(B * H, N, N)               # FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                               # [B, H, N, N] FP16
    del left_mat, right_mat

    # ------------------------------------------------------------------
    # 5️⃣  Second LayerNorm (mean / var) – pure PyTorch (FP32)
    # ------------------------------------------------------------------
    out_fp32 = out.view(R, H).float()               # [R, H] FP32
    mu  = out_fp32.mean(dim=1)                     # [R]
    var = out_fp32.var(dim=1, unbiased=False)     # [R]
    rstd = torch.rsqrt(var + eps)                  # [R]

    # ------------------------------------------------------------------
    # 6️⃣  Apply LN(weight/bias), out_gate and final projection
    # ------------------------------------------------------------------
    #   LN:   (out - μ) * rstd  →  *γ + β
    #   Gating:  * out_gate
    #   Projection:  linear with `to_out.weight` (no bias)
    # ------------------------------------------------------------------
    # normalize
    out_norm = (out_fp32 - mu[:, None]) * rstd[:, None]      # [R, H]

    # to_out_norm parameters (γ, β) – FP32
    gamma = w["to_out_norm.weight"]          # [H]
    beta  = w["to_out_norm.bias"]            # [H]
    out_norm = out_norm * gamma[None, :] + beta[None, :]      # [R, H]

    # multiply with out_gate (casted to FP32)
    out_gate_fp32 = out_gate.view(R, H).float()
    out_norm_gate = out_norm * out_gate_fp32                     # [R, H]

    # final linear projection (FP32)
    w_out = w["to_out.weight"]                # [dim, H]  FP32, bias=False
    final = F.linear(out_norm_gate, w_out)    # [R, dim]

    # reshape to the original 4‑D layout
    final = final.view(B, N, N, dim)          # [B, N, N, dim]  float32
    return final
