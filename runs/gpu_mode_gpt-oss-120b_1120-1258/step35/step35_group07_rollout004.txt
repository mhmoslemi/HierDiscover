"""
TriMul (outgoing) – fused Triton‑accelerated forward pass (no backward).

Algorithm (all operations are fused as much as possible):
1️⃣  Row‑wise LayerNorm on the flattened [B·N·N, C] input (FP32→FP16).
2️⃣  One fused linear projection produces five matrices (left, right,
    left‑gate, right‑gate, out‑gate) of shape (R, H) where R = B·N·N,
    H = hidden_dim.  The projection is performed with a regular torch
    `F.linear` (FP16) to leverage cuBLAS.
3️⃣  A Triton kernel splits the fused projection, applies the optional
    mask, computes sigmoid gates and writes three tensors `left`,
    `right` and `out_gate` in layout [B, H, N, N] (FP16).
4️⃣  A batched GEMM (torch.bmm) computes the core 4‑dimensional einsum:
       out[b, h, i, j] = Σₖ left[b, h, i, k]·right[b, h, j, k]     → [B, H, N, N] (FP16).
5️⃣  Row‑wise mean & rsqrt‑variance of `out` over the hidden dimension H
    (used by the second LayerNorm) are computed by a Triton kernel.
6️⃣  The final Triton kernel fuses the second LayerNorm, the per‑position
    `out_gate` multiplication and the output linear projection
    (to_out.weight).  The result is written as [B, N, N, C] in FP32.

All kernels target Triton 3.3.1 on an H100 and use mixed‑precision
arithmetic (FP32 for reductions, FP16 for storage/computation).
"""

import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # [R, dim]   FP32 input
    y_ptr,          # [R, dim]   FP16 output
    w_ptr,          # [dim]      LN weight γ (FP32)
    b_ptr,          # [dim]      LN bias   β (FP32)
    rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)                     # [BLOCK_M]
    mask_m = offs_m < rows

    # ---- first pass: compute mean and variance (FP32) ----
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)           # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]      # [M, D]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)                # [M, D]
        sum_val += tl.sum(vals, axis=1)                           # [M]
        sum_sq  += tl.sum(vals * vals, axis=1)                    # [M]
    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- second pass: write normalized values (FP16) ----
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)           # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]      # [M, D]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]              # [M, D]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)    # [D]  FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)    # [D]  FP32

        out = norm * gamma[None, :] + beta[None, :]                # [M, D]
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, apply optional mask, compute sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,      # [R, 5·H]  FP16 fused projection
    left_ptr,      # [B, H, N, N] FP16
    right_ptr,     # [B, H, N, N] FP16
    out_gate_ptr,  # [B, H, N, N] FP16
    mask_ptr,      # [R] FP16 (optional, broadcast over H)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)            # [BLOCK_R]
    mask_r = offs_r < R

    # Decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask broadcasted over H
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                   # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N                     # stride for one hidden plane in the 4‑D layout
    base_offset = b * (H * hidden_stride) + i * N + k                # start of hidden plane 0

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)            # [BLOCK_H]
        mask_h = offs_h < H

        # offsets inside the fused projection: [left | right | l_gate | r_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]      # [BLOCK_R, BLOCK_H]

        left_raw      = tl.load(proj_ptr + proj_base,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_raw     = tl.load(proj_ptr + proj_base + H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        left_gate_raw = tl.load(proj_ptr + proj_base + 2 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_gate_raw= tl.load(proj_ptr + proj_base + 3 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        out_gate_raw  = tl.load(proj_ptr + proj_base + 4 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)

        # sigmoid in FP32 then cast back to FP16 (stable)
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        # out_gate is *not* masked
        outg  = out_gate

        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst_offset,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset,
                 outg,
                 mask=mask_r[:, None] & mask_h[None, :])

# 3️⃣ Row‑wise statistics of the contracted tensor (mean & rsqrt‑variance)
# ----------------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr,   # [B, H, N, N] FP16
    mu_ptr,    # [R] FP32   (mean over H)
    rstd_ptr,  # [R] FP32   (rsqrt(var+eps) over H)
    total_rows, H, N,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + j    # start of hidden plane 0

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)        # [BLOCK_K]
        mask_k = offs_k < H

        ptr = out_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)      # [M, K]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_m, mean,  mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)


# ----------------------------------------------------------------------
# 4️⃣ Fuse second LayerNorm, out_gate and final projection
# ----------------------------------------------------------------------
@triton.jit
def final_output_kernel(
    out_ptr,        # [B, H, N, N] FP16
    gate_ptr,       # [B, H, N, N] FP16 (out_gate)
    mu_ptr,         # [R] FP32    (mean over H)
    rstd_ptr,       # [R] FP32    (rstd over H)
    ln_w_ptr,       # [H] FP32    (to_out_norm.weight)
    ln_b_ptr,       # [H] FP32    (to_out_norm.bias)
    w_out_T_ptr,    # [H, dim] FP16 (transposed to_out weight)
    out_final_ptr,  # [B, N, N, dim] FP32
    total_rows, H, N, dim,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr,
):
    pid_m = tl.program_id(0)   # rows (flattened B·N·N)
    pid_n = tl.program_id(1)   # output dim tiles

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)   # [BLOCK_M]
    offs_n = col_start + tl.arange(0, BLOCK_N)   # [BLOCK_N]

    mask_m = offs_m < total_rows
    mask_n = offs_n < dim

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + j    # start of hidden plane 0

    # Load per‑row statistics (mean & rstd)
    mu   = tl.load(mu_ptr   + offs_m, mask=mask_m, other=0.0)   # [M]
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)   # [M]

    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)      # accumulator

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)               # [K]
        mask_k = offs_k < H

        # Load hidden‑plane values (out) and out‑gate
        ptr_out  = out_ptr  + base_offset[:, None] + offs_k[None, :] * hidden_stride
        ptr_gate = gate_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride

        vals_out = tl.load(ptr_out,
                           mask=mask_m[:, None] & mask_k[None, :],
                           other=0.0).to(tl.float32)            # [M, K]
        vals_gate = tl.load(ptr_gate,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)           # [M, K]

        # LayerNorm over hidden dimension (γ,β)
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [K] FP32
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [K] FP32

        # (out - μ) * rstd  → normalized hidden vector
        diff = (vals_out - mu[:, None]) * rstd[:, None]              # [M, K]
        norm = diff * gamma[None, :] + beta[None, :]                 # [M, K]

        # Multiply by out_gate
        term = norm * vals_gate                                      # [M, K]

        # Cast to FP16 for the dot with final weight (better bandwidth)
        term_fp16 = term.to(tl.float16)                              # [M, K]

        # Load a tile of the final projection weight (H, dim) → (K, BLOCK_N)
        w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]  # [K, BLOCK_N]
        w_tile = tl.load(w_ptr,
                         mask=mask_k[:, None] & mask_n[None, :],
                         other=0.0)                                    # [K, BLOCK_N]

        # Accumulate matmul term_fp16·w_tile
        acc += tl.dot(term_fp16, w_tile).to(tl.float32)            # [M, BLOCK_N]

    # Store final tensor (already FP32)
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])

    """

    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data

    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)

    x_flat = x.reshape(R, dim).contiguous()                 # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat   # free memory

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear projection (five heads → 5·H)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                         # shape [5·H, dim]

    proj = F.linear(x_norm, fused_w)             # (R, 5·H) FP16
    del fused_w, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ Split, mask, sigmoid gates → left / right / out_gate
    # ------------------------------------------------------------------
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        # dummy tensor – never accessed when HAS_MASK==0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM (cuBLAS) – core einsum
    # ------------------------------------------------------------------
    # Reshape to [B·H, N, N] for torch.bmm
    left_mat  = left.view(B * H, N, N)               # FP16
    right_mat = right.view(B * H, N, N)              # FP16

    # out[b, h, i, j] = Σₖ left[b, h, i, k] * right[b, h, j, k]
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))    # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                              # [B, H, N, N] FP16

    # ------------------------------------------------------------------
    # 5️⃣ Row‑wise mean & rsqrt‑variance of `out` (over hidden dim H)
    # ------------------------------------------------------------------
    mu   = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 256
    BLOCK_K_STATS = 128                     # H ≤ 128 → fits in one tile
    grid_stats = (triton.cdiv(R, BLOCK_M_STATS),)
    rowwise_stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm + out_gate + final projection (fused)
    # ------------------------------------------------------------------
    # Transpose final linear weight once: shape (hidden_dim, dim)
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 128       # rows per program (tiles over R)
    BLOCK_K_OUT = 32        # hidden‑dim chunk (fits in registers)
    BLOCK_N_OUT = 128       # output‑dim chunk

    grid_out = (
        triton.cdiv(R, BLOCK_M_OUT),
        triton.cdiv(dim, BLOCK_N_OUT),
    )
    final_output_kernel[grid_out](
        out,
        out_gate,
        mu,
        rstd,
        w["to_out_norm.weight"],   # γ (FP32)
        w["to_out_norm.bias"],     # β (FP32)
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=8,
    )

    return final
```
