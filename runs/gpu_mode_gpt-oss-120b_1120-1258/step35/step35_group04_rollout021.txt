def custom_kernel(data):
    x, mask, weights, config = data
    dim = config["dim"]; hidden_dim = config["hidden_dim"]; nomask = config.get("nomask", True)
    eps = 1e-5

    B, N, _, _ = x.shape
    R = B * N * N
    H = hidden_dim

    # 1: LayerNorm (FP32 to FP16)
    x_norm = torch.nn.functional.layer_norm(x, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps).to(torch.float16)

    # 2: Fused projection
    fused_weight = torch.cat([weights["left_proj.weight"],
                              weights["right_proj.weight"],
                              weights["left_gate.weight"],
                              weights["right_gate.weight"],
                              weights["out_gate.weight"]], dim=0).to(torch.float16)
    # (5*H, dim)
    x_norm_flat = x_norm.view(R, dim)  # contiguous
    proj = torch.nn.functional.linear(x_norm_flat, fused_weight)  # (R, 5*H) fp16

    # 3: Split and apply mask/gates via triton kernel
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=x.device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=x.device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=x.device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).contiguous().view(R)
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=x.device)

    BLOCK_R = 128
    BLOCK_H = 64  # maybe 128? Determine limit for registers

    split_fuse_gate_kernel[(triton.cdiv(R, BLOCK_R),)](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )

    # Clean up proj to free memory
    del proj

    # 4: Compute pairwise product via batched matmul
    left_mat = left.view(B*H, N, N)
    right_mat = right.view(B*H, N, N)
    out = torch.bmm(left_mat, right_mat.transpose(1,2))  # (B*H,N,N) fp16
    out = out.view(B, H, N, N)

    # free left,right
    del left, right

    # 5: Apply second LayerNorm and out_gate
    # reshape for layernorm
    out_perm = out.permute(0,2,3,1).contiguous()  # (B,N,N,H)
    out_flat = out_perm.view(R, H).float()
    out_norm = torch.nn.functional.layer_norm(out_flat, (H,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"], eps=eps)  # float32

    # Apply out_gate
    gate_perm = out_gate.permute(0,2,3,1).contiguous()
    gate_flat = gate_perm.view(R, H).float()

    # 6: Final linear projection
    final = torch.nn.functional.linear(gated, weights["to_out.weight"], bias=None)  # (R,dim) float32

    # reshape to original shape
    final = final.view(B, N, N, dim)
    return final

   FP32 → FP16).  This replaces the custom Triton LayerNorm and is highly optimised.

2️⃣ Fuse the five linear maps (left_proj, right_proj, left_gate, right_gate, out_gate)
   into a single GEMM (cuBLAS).  The weight matrix has shape (5·H, dim) and the
   result has shape (R, 5·H) where R = B·N·N.

3️⃣ Triton kernel ``split_fuse_gate_kernel`` extracts the three tensors
   (left, right, out_gate) from the fused projection, applies the optional mask
   and the sigmoid gates, and stores them in layout [B, H, N, N] (FP16).

4️⃣ Batched GEMM (torch.bmm) computes the core einsum:
       out[b,h,i,j] = Σₖ left[b,h,i,k] · right[b,h,j,k]   (FP16).

5️⃣ Second LayerNorm on the hidden dimension (torch.nn.functional.layer_norm)
   followed by element‑wise multiplication with ``out_gate`` (both cast to FP32).

6️⃣ Final linear projection (torch.nn.functional.linear) maps the hidden
   dimension back to the original channel dimension (float32).

All heavy linear algebra uses FP16 for bandwidth and tensor‑core utilisation,
while the final output is returned in float32 as required by the reference model.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: split fused projection, apply mask, compute sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,      # [R, 5·H]  FP16 fused projection
    left_ptr,      # [B, H, N, N] FP16
    right_ptr,     # [B, H, N, N] FP16
    out_gate_ptr,  # [B, H, N, N] FP16
    mask_ptr,      # [R]       FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # Decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # Optional per‑pair mask (broadcast across hidden dim)
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        m_val = m_val[:, None]                                      # [BLOCK_R, 1]
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # Base offset for hidden‑plane 0 (layout B, H, N, N → stride H·NN, NN, N, 1)
    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + k               # [BLOCK_R]

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)     # [BLOCK_H]
        mask_h = offs_h < H

        # Offsets inside the fused projection row (5 blocks of H)
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # Sigmoid (computed in FP32 for stability, then cast back)
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # Apply mask and gates
        left  = left_raw  * m_val * left_gate
        right = right_raw * m_val * right_gate
        outg  = out_gate   # no mask on out_gate

        # Destination offsets in B,H,N,N layout
        dst = base[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst, right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst, outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only outgoing TriMul.

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping with keys
                  "norm.weight", "norm.bias",
                  "left_proj.weight", "right_proj.weight",
                  "left_gate.weight", "right_gate.weight", "out_gate.weight",
                  "to_out_norm.weight", "to_out_norm.bias",
                  "to_out.weight"
              (all float32, already on the same device as inputs)
            - config_dict must contain:
                  "dim" (int), "hidden_dim" (int), "nomask" (bool, optional)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = bool(cfg.get("nomask", True))
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                     # total pairwise tokens
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16) using torch's highly‑optimised kernel
    # --------------------------------------------------------------
    # LayerNorm over the last dimension (dim)
    x_norm = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    ).to(torch.float16)               # [B, N, N, dim]  FP16

    # --------------------------------------------------------------
    # 2️⃣ Fuse five linear maps into a single GEMM (cuBLAS)
    # --------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                # shape (5·H, dim)

    # Flatten to (R, dim) – contiguity is important for cuBLAS
    x_flat = x_norm.view(R, dim).contiguous()
    proj = F.linear(x_flat, fused_w)    # (R, 5·H)  FP16
    del x_norm, x_flat, fused_w        # free memory early

    # --------------------------------------------------------------
    # 3️⃣ Split, mask, sigmoid, store left / right / out_gate (Triton)
    # --------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).contiguous().view(R)
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy

    BLOCK_R = 128          # rows per program (tunable)
    BLOCK_H = 64           # hidden channels per program
    grid = (triton.cdiv(R, BLOCK_R),)

    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣ Batched GEMM (cuBLAS) – core einsum
    # --------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)          # (B·H, N, N)  FP16
    right_mat = right.view(B * H, N, N)         # (B·H, N, N)  FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B·H, N, N)  FP16
    out = out.view(B, H, N, N)                  # (B, H, N, N)  FP16
    del left_mat, right_mat, left, right        # free memory

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm + out_gate multiplication (torch)
    # --------------------------------------------------------------
    # Move hidden dim to the last axis so LayerNorm can operate on it
    out_perm = out.permute(0, 2, 3, 1).contiguous()          # (B, N, N, H)
    out_flat = out_perm.view(R, H).float()                  # FP32 for stability

    out_norm = F.layer_norm(
        out_flat,
        normalized_shape=(H,),
        weight=w["to_out_norm.weight"],
        bias=w["to_out_norm.bias"],
        eps=eps,
    )                                                         # (R, H) FP32

    gate_perm = out_gate.permute(0, 2, 3, 1).contiguous()    # (B, N, N, H)
    gate_flat = gate_perm.view(R, H).float()                # FP32

    gated = out_norm * gate_flat                              # (R, H) FP32
    del out_norm, gate_flat, out_perm, gate_perm, out, out_gate

    # --------------------------------------------------------------
    # 6️⃣ Final linear projection back to `dim` (cuBLAS)
    # --------------------------------------------------------------
    # to_out.weight has shape [dim, hidden_dim] (float32)
    final = F.linear(gated, w["to_out.weight"], bias=None)   # (R, dim) FP32

    # Reshape to the original 4‑D layout
    final = final.view(B, N, N, dim)

    return final
```
