def split_fuse_gate_kernel(proj_ptr, left_ptr, right_ptr, out_gate_ptr, mask_ptr,
                           R, H, N,
                           HAS_MASK: tl.constexpr,
                           BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr):
    pid = tl.program_id(0)
    block_r_start = pid * BLOCK_R
    offs_r = block_r_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < R

    # decode b,i,k from r
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # compute base offset in left/right etc (starting at h=0)
    hidden_stride = N * N
    # offset of left/right/out for h=0 plane: b*H*hidden_stride + i*N + k
    base_off = b * (H * hidden_stride) + i * N + k

    # compute mask if needed:
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # fp16
        mask_val = mask_val[:, None]  # broadcast over H
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # compute offset to proj: each row r offset = r * (5*H) + h
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]  # shape [R,H]

        left_raw = tl.load(proj_ptr + proj_base,
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0)
        right_raw = tl.load(proj_ptr + proj_base + H,
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0)
        left_gate_raw = tl.load(proj_ptr + proj_base + 2*H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_gate_raw = tl.load(proj_ptr + proj_base + 3*H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        out_gate_raw = tl.load(proj_ptr + proj_base + 4*H,
                               mask=mask_r[:, None] & mask_h[None, :],

"""
TriMul (outgoing) – fast forward‑only Triton implementation.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the pairwise tensor (FP32 → FP16) using
   torch.nn.functional.layer_norm (highly tuned).

2️⃣ Fuse the five linear maps (left_proj, right_proj, left_gate,
   right_gate, out_gate) into one matrix multiplication:
   proj = Linear(x_norm, W_fused)    # [R, 5·H]  (FP16).

3️⃣ Triton kernel `split_fuse_gate_kernel` splits the fused projection,
   applies the optional pairwise mask, computes sigmoid gates and writes
   three tensors `left`, `right`, `out_gate` in layout [B, H, N, N] (FP16).

4️⃣ Core einsum is a batched GEMM:
        out[b,h,i,j] = Σₖ left[b,h,i,k]·right[b,h,j,k]
   performed with `torch.bmm`.

5️⃣ Row‑wise mean/variance over the hidden dimension H is computed with
   `torch.mean/var` (FP32) → μ and rstd per (b,i,j).

6️⃣ 2nd LayerNorm (γ,β) across H, multiplication with `out_gate`,
   and final projection to the original dimension are done with
   pure PyTorch kernels (FP32).  This satisfies the requirement of
   having at least one custom Triton kernel while keeping the overall
   runtime low.

The implementation supports optional masking, arbitrary batch size
(1‑2), sequence length up to 1024 and hidden dimensions up to 128.
All tensors are assumed to reside on the same CUDA device.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 3️⃣ Split fused projection, apply mask & sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # [R, 5·H]    (FP16)
    left_ptr,          # [B, H, N, N] (FP16)
    right_ptr,         # [B, H, N, N] (FP16)
    out_gate_ptr,      # [B, H, N, N] (FP16)
    mask_ptr,          # [R] (FP16) – optional
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [R]
    mask_r = offs_r < R

    # Decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # Optional mask (broadcast over hidden dim)
    if HAS_MASK:
        mval = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [R]
        mval = mval[:, None]                                       # [R,1]
    else:
        mval = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N               # distance between consecutive h‑planes
    # base offset for h = 0 plane of (b,i,k)
    base_off = b * (H * hidden_stride) + i * N + k

    # Process H in tiles
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # linear view: [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [R, H]

        left_raw = tl.load(proj_ptr + proj_base,
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0)

        right_raw = tl.load(proj_ptr + proj_base + H,
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0)

        left_gate_raw = tl.load(proj_ptr + proj_base + 2 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)

        right_gate_raw = tl.load(proj_ptr + proj_base + 3 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)

        out_gate_raw = tl.load(proj_ptr + proj_base + 4 * H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)

        # Sigmoid in FP32 → back to FP16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # Apply mask and gates
        left  = left_raw  * mval * left_gate
        right = right_raw * mval * right_gate
        outg  = out_gate

        dst_off = base_off[:, None] + offs_h[None, :] * hidden_stride   # [R, H]

        tl.store(left_ptr + dst_off,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])

        tl.store(right_ptr + dst_off,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])

        tl.store(out_gate_ptr + dst_off,
                 outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (gradient‑free).

    Args:
        data (tuple):
            input_tensor : torch.Tensor [B, N, N, dim] (float32)
            mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            weights_dict : dict with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
                (all float32)
            config_dict : {"dim": int, "hidden_dim": int, "nomask": bool}
    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                      # total pairwise tokens
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16) using torch's highly tuned kernel
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim)                     # [R, dim] FP32
    x_norm = F.layer_norm(
        x_flat,
        (dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    )                                                # FP32
    x_norm_fp16 = x_norm.to(torch.float16)           # FP16 for the next linear

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection of the five linear maps (weight matrix 5·H × dim)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                               # [5·H, dim]

    proj = F.linear(x_norm_fp16, fused_w)             # [R, 5·H] FP16
    del fused_w, x_norm_fp16, x_flat

    # ------------------------------------------------------------------
    # 3️⃣ Split projection, apply mask & sigmoid gates (custom Triton kernel)
    # ------------------------------------------------------------------
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty_like(left)
    out_gate = torch.empty_like(left)

    if (not nomask) and (mask is not None):
        # mask broadcasted over hidden dim; convert to 0/1 float16
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64   # rows per program (<= 1024 / BLOCK_H)
    BLOCK_H = 16   # hidden dim tiles (64*16 = 1024 threads)
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Core einsum as batched GEMM over the hidden dimension
    # ------------------------------------------------------------------
    left_mat = left.view(B * H, N, N)               # [B·H, N, N] FP16
    right_mat = right.view(B * H, N, N)             # [B·H, N, N] FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                      # [B, H, N, N] FP16

    # ------------------------------------------------------------------
    # 5️⃣ Row‑wise mean & rsqrt‑var over hidden dim (FP32)
    # ------------------------------------------------------------------
    out_fp32 = out.to(torch.float32)                # promote to FP32 for stats
    mu = out_fp32.mean(dim=1, keepdim=True)         # [B,1,N,N]
    var = out_fp32.var(dim=1, unbiased=False, keepdim=True)  # [B,1,N,N]
    rstd = torch.rsqrt(var + eps)                   # [B,1,N,N]

    # ------------------------------------------------------------------
    # 6️⃣ Final LayerNorm + out‑gate + projection (pure PyTorch)
    # ------------------------------------------------------------------
    # LayerNorm parameters (γ, β) over hidden dimension H
    ln_w = w["to_out_norm.weight"].view(1, H, 1, 1)   # [1,H,1,1]
    ln_b = w["to_out_norm.bias"].view(1, H, 1, 1)     # [1,H,1,1]

    # LN over hidden dim
    out_norm = ((out_fp32 - mu) * rstd) * ln_w + ln_b   # [B,H,N,N] FP32

    # Multiply by out_gate (converted to FP32)
    gated = out_norm * out_gate.to(torch.float32)       # [B,H,N,N] FP32

    # Final linear projection to `dim`
    # reshape to (B·N·N, H) for torch.nn.functional.linear
    gated_reshaped = gated.permute(0, 2, 3, 1).contiguous()   # [B,N,N,H]
    gated_flat = gated_reshaped.view(-1, H)                    # [(B·N·N), H]

    final_flat = F.linear(gated_flat, w["to_out.weight"])     # [(B·N·N), dim]
    final = final_flat.view(B, N, N, dim)                     # [B,N,N,dim] (FP32)

    return final
