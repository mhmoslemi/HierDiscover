"""
Outgoing TriMul (AlphaFold‑3) – fast forward‑only implementation.

The operator works on a 4‑D tensor `x` of shape [B, N, N, C] and computes

    out[b,i,j] = Σₖ  ( (x·W_left)·σ(x·W_left_gate)·mask )₍b,i,k₎
                × ( (x·W_right)·σ(x·W_right_gate)·mask )₍b,j,k₎

followed by a second LayerNorm over the hidden dimension H, multiplication with a
third sigmoid gate and a final linear projection back to `C`.

All heavy element‑wise work (the five fused linear maps + the three gates) is
performed by a single Triton kernel (`split_fuse_gate_kernel`).  The large
contractions are delegated to cuBLAS (`torch.bmm` and `torch.nn.functional.linear`);
the two LayerNorms are expressed with the highly‑optimised
`torch.nn.functional.layer_norm`.  This design dramatically reduces memory traffic
while still keeping a custom Triton kernel in the critical path.

The entry‑point `custom_kernel` follows the signature required by the
evaluation harness.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Split‑fused projection + mask + three sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # [R, 5·H]  fused linear output,  FP16
    left_ptr,          # [B, H, N, N]  output tensor,   FP16
    right_ptr,         # [B, H, N, N]  output tensor,   FP16
    out_gate_ptr,      # [B, H, N, N]  output tensor,   FP16
    mask_ptr,          # [R]                mask (optional), FP16
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    """
    For each “pair token’’ (b,i,k) compute:
        left      = (x·W_left)      * mask * σ(x·W_left_gate)
        right     = (x·W_right)     * mask * σ(x·W_right_gate)
        out_gate  = σ(x·W_out_gate)               (no mask)

    The three gating layers and the two projections are fused into a single
    load/store pass to minimise traffic.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)              # [BLOCK_R]
    mask_r = offs_r < R

    # Decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # Optional pair‑mask (broadcast over the hidden dimension)
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        m_val = m_val[:, None]                                      # [BLOCK_R, 1]
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # Stride between consecutive hidden planes inside the output buffers
    hidden_stride = N * N                     # jump from h to h+1
    base = b * (H * hidden_stride) + i * N + k   # start index of hidden‑plane 0

    # Loop over hidden dimension in tiles
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # Column offset inside the fused projection (5·H columns)
        proj_off = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        # Load the five consecutive slices
        left_raw      = tl.load(proj_ptr + proj_off,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)                       # [R, H]
        right_raw     = tl.load(proj_ptr + proj_off + H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        left_gate_raw = tl.load(proj_ptr + proj_off + 2 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_gate_raw= tl.load(proj_ptr + proj_off + 3 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        out_gate_raw  = tl.load(proj_ptr + proj_off + 4 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)

        # Sigmoid (compute in FP32 for numeric stability)
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw .to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw .to(tl.float32)))).to(tl.float16)

        # Apply mask and gates
        left  = left_raw  * m_val * left_gate
        right = right_raw * m_val * right_gate
        outg  = out_gate                       # already a sigmoid

        # Destination offsets inside the three output buffers
        dst = base[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr      + dst, left,      mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr     + dst, right,     mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr  + dst, outg,      mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only outgoing TriMul.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        – input_tensor : torch.Tensor[ B, N, N, C ]  (float32)
        – mask_tensor  : torch.Tensor[ B, N, N ]    (bool/float) or ``None``
        – weights_dict : {
              "norm.weight", "norm.bias",
              "left_proj.weight", "right_proj.weight",
              "left_gate.weight", "right_gate.weight", "out_gate.weight",
              "to_out_norm.weight", "to_out_norm.bias",
              "to_out.weight"
          }  all float32 and already on the correct device
        – config_dict must contain:
            * ``dim``         – input channel dimension C
            * ``hidden_dim``  – hidden dimension H
            * ``nomask``      – ``bool`` (True → no mask)

    Returns
    -------
    torch.Tensor[ B, N, N, C ] (float32) – result of the TriMul forward pass
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = bool(cfg.get("nomask", True))
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                         # total number of (b,i,k) tokens
    H = hidden_dim

    # ------------------------------------------------------------------
    # 0️⃣ Input LayerNorm  (use the highly‑optimised torch kernel)
    # ------------------------------------------------------------------
    # Normalise over the last dimension ``dim`` for every pair token.
    x_norm = F.layer_norm(x,
                          normalized_shape=(dim,),
                          weight=w["norm.weight"],
                          bias=w["norm.bias"],
                          eps=eps)                # [B,N,N,dim] float32
    x_norm = x_norm.to(torch.float16)                 # cast once for the linear

    # ------------------------------------------------------------------
    # 1️⃣ Fuse the five linear maps (left, right, three gates)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],   # [H, dim]
            w["right_proj.weight"],  # [H, dim]
            w["left_gate.weight"],   # [H, dim]
            w["right_gate.weight"],  # [H, dim]
            w["out_gate.weight"],    # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)                               # [5·H, dim]

    proj = F.linear(x_norm, fused_w)                  # (R, 5·H)  fp16
    del x_norm, fused_w

    # ------------------------------------------------------------------
    # 2️⃣ Split, mask and apply the three sigmoid gates
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64          # rows per program (see analysis above)
    BLOCK_H = 64          # hidden‑dim tile
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 3️⃣ Core contraction  (left @ rightᵀ)   – batched GEMM
    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)                     # [B·H, N, N]  fp16
    right_mat = right.view(B * H, N, N)                    # [B·H, N, N]  fp16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [B·H, N, N]  fp16
    out = out.view(B, H, N, N)                            # [B, H, N, N]  fp16
    del left_mat, right_mat

    # ------------------------------------------------------------------
    # 4️⃣ Second LayerNorm (over the hidden dimension H)
    #    + element‑wise out‑gate multiplication
    # ------------------------------------------------------------------
    # Move H to the last axis so that `layer_norm` can act on it directly.
    out_perm = out.permute(0, 2, 3, 1).contiguous()          # (B, N, N, H)  fp16
    out_norm = F.layer_norm(out_perm,
                            normalized_shape=(H,),
                            weight=w["to_out_norm.weight"],
                            bias=w["to_out_norm.bias"],
                            eps=eps)                     # (B, N, N, H)  float32
    # Restore original layout and fuse the out‑gate.
    out_norm = out_norm.permute(0, 3, 1, 2)                   # (B, H, N, N)  float32
    out_gate_f = out_gate.float()                           # cast once
    gated = out_norm * out_gate_f                            # (B, H, N, N)  float32

    # ------------------------------------------------------------------
    # 5️⃣ Final linear projection back to ``dim``  (large GEMM)
    # ------------------------------------------------------------------
    # Flatten the (B,N,N) positions → R rows, each of length H.
    gated_flat = gated.view(R, H)                            # (R, H)  float32
    final_flat = F.linear(gated_flat, w["to_out.weight"])   # (R, dim) float32

    final = final_flat.view(B, N, N, dim)                    # (B, N, N, dim)

    return final
