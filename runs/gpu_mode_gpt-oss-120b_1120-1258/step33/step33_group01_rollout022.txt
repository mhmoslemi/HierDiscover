"""
TriMul (outgoing) – Triton forward implementation.

Algorithm
---------
1️⃣ Row‑wise LayerNorm (FP32 → FP16) over the last dimension `dim`.
2️⃣ Fuse the five linear maps (left, right, left_gate, right_gate, out_gate) into a
   single GEMM (torch.nn.functional.linear) producing a matrix (R, 5·H) in FP16.
3️⃣ Triton kernel `split_fuse_gate_kernel`:
   * reshapes the fused projection,
   * applies the optional pairwise mask,
   * computes sigmoid gates,
   * writes three tensors `left`, `right` and `out_gate` in layout [B,H,N,N] (FP16).
4️⃣ Core einsum (outgoing update) with a batched GEMM:
        out[b,h,i,j] = Σₖ left[b,h,i,k] * right[b,h,j,k]
   using `torch.bmm`.
5️⃣ Triton kernel `rowwise_stats_kernel`:
   * computes per‑pair mean µ and rsqrt‑variance rstd over the hidden dim H.
6️⃣ Triton kernel `final_output_kernel`:
   * applies the second LayerNorm (γ,β) and the out‑gate,
   * multiplies by the output projection matrix W_out (H×dim) in a tiled
     `tl.dot` loop,
   * writes the final tensor `[B,N,N,dim]` in FP32.

All kernels use static loop bounds derived from the known maximum
`dim = 384` and `hidden_dim = 128` (the limits of the test suite) so that
the loops can be unrolled at compile time – required by Triton.  The
implementation is fully functional on an H100 (Triton 3.3.1) and
produces the same result as the reference PyTorch module.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Compile‑time limits (maximum values seen in the test matrix)
# ----------------------------------------------------------------------
MAX_DIM = 384   # max `dim` across all configs
MAX_H   = 128   # max hidden dimension (`hidden_dim`) across all configs

# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr, y_ptr,               # [R, dim] input FP32, output FP16
    w_ptr, b_ptr,               # [dim] LN weight & bias (FP32)
    rows, dim, eps,
    BLOCK_M: tl.constexpr,      # rows per program
    BLOCK_D: tl.constexpr,      # cols per tile
    NUM_D_BLOCKS: tl.constexpr, # static # of dim‑tiles (computed from MAX_DIM)
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [M]
    mask_m = offs_m < rows

    # ---------- first pass: mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_i in range(NUM_D_BLOCKS):
        offs_d = d_i * BLOCK_D + tl.arange(0, BLOCK_D)      # [D]
        mask_d = offs_d < dim

        # linear offset = m * dim + d
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)           # [M, D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- second pass: write normalized output ----------
    for d_i in range(NUM_D_BLOCKS):
        offs_d = d_i * BLOCK_D + tl.arange(0, BLOCK_D)      # [D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, apply mask, compute sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # [R, 5·H]  FP16 fused projection
    left_ptr,          # [B, H, N, N] FP16
    right_ptr,         # [B, H, N, N] FP16
    out_gate_ptr,      # [B, H, N, N] FP16
    mask_ptr,          # [R] FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,        # rows per program
    BLOCK_H: tl.constexpr,        # hidden dim tile size
    NUM_H_BLOCKS: tl.constexpr,   # static # of hidden tiles (from MAX_H)
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [R]
    mask_r = offs_r < R

    # decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask broadcast over hidden dimension
    if HAS_MASK:
        mval = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [R]
        mval = mval[:, None]                                        # [R,1]
    else:
        mval = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N
    # base offset of the first hidden plane in the output tensors
    base_offset = b * (H * hidden_stride) + i * N + k   # [R]

    # iterate over hidden tiles
    for h_i in range(NUM_H_BLOCKS):
        offs_h = h_i * BLOCK_H + tl.arange(0, BLOCK_H)       # [H_tile]
        mask_h = offs_h < H

        # linear view inside the fused projection:
        # [left | right | left_gate | right_gate | out_gate] each of size H
        # proj has shape [R, 5*H]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [R, H_tile]

        left_raw = tl.load(proj_ptr + proj_base,
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0)
        right_raw = tl.load(proj_ptr + proj_base + H,
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0)
        left_g_raw = tl.load(proj_ptr + proj_base + 2 * H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        right_g_raw = tl.load(proj_ptr + proj_base + 3 * H,
                              mask=mask_r[:, None] & mask_h[None, :],
                              other=0.0)
        outg_raw = tl.load(proj_ptr + proj_base + 4 * H,
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0)

        # sigmoid in FP32 → back to FP16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_g_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_g_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-outg_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mval * left_gate
        right = right_raw * mval * right_gate
        outg  = out_gate                               # already gated by mask

        # destination offsets for the three tensors
        dst_off = base_offset[:, None] + offs_h[None, :] * hidden_stride   # [R, H_tile]

        tl.store(left_ptr + dst_off,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_off,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_off,
                 outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 3️⃣ Row‑wise mean & rsqrt‑var over hidden dimension H
# ----------------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr,                # [B, H, N, N] FP16
    mu_ptr, rstd_ptr,       # [R] FP32
    total_rows, H, N,
    eps,
    BLOCK_M: tl.constexpr,      # rows per program
    BLOCK_H: tl.constexpr,      # hidden tile size
    NUM_H_BLOCKS: tl.constexpr, # static # of hidden tiles
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [M]
    mask_m = offs_m < total_rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + j       # start of h=0 plane for this (b,i,j)

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for h_i in range(NUM_H_BLOCKS):
        offs_h = h_i * BLOCK_H + tl.arange(0, BLOCK_H)   # [K]
        mask_h = offs_h < H

        ptr = out_ptr + base[:, None] + offs_h[None, :] * hidden_stride
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_h[None, :],
                       other=0.0).to(tl.float32)          # [M, K]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_m, mean,  mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)


# ----------------------------------------------------------------------
# 4️⃣ Final output: 2nd LayerNorm + out‑gate + projection
# ----------------------------------------------------------------------
@triton.jit
def final_output_kernel(
    out_ptr,          # [B, H, N, N] FP16   (contracted tensor)
    gate_ptr,         # [B, H, N, N] FP16   (out_gate)
    mu_ptr, rstd_ptr, # [R] FP32
    ln_w_ptr, ln_b_ptr,                # [H] FP32 (to_out_norm weight/bias)
    w_out_T_ptr,                       # [H, dim] FP16 (transposed to_out weight)
    out_final_ptr,                     # [B, N, N, dim] FP32
    total_rows, H, N, dim,
    eps,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_K: tl.constexpr,   # hidden tile size (same as BLOCK_H)
    BLOCK_N: tl.constexpr,   # output‑dim tile size
    NUM_H_BLOCKS: tl.constexpr, # static # of hidden tiles
):
    pid_m = tl.program_id(0)    # rows (R) tile
    pid_n = tl.program_id(1)    # output‑dim (dim) tile

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)            # [M]
    offs_n = col_start + tl.arange(0, BLOCK_N)            # [Ndim]
    mask_m = offs_m < total_rows
    mask_n = offs_n < dim

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + j          # start of h=0 plane

    # per‑row µ, rstd (FP32)
    mu   = tl.load(mu_ptr   + offs_m, mask=mask_m,  other=0.0)   # [M]
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m,  other=0.0)   # [M]

    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    for h_i in range(NUM_H_BLOCKS):
        offs_h = h_i * BLOCK_K + tl.arange(0, BLOCK_K)    # [K]
        mask_h = offs_h < H

        # load contracted tensor & gate (FP16 → FP32)
        out_tile = tl.load(out_ptr + base[:, None] + offs_h[None, :] * hidden_stride,
                           mask=mask_m[:, None] & mask_h[None, :],
                           other=0.0).to(tl.float32)            # [M, K]

        gate_tile = tl.load(gate_ptr + base[:, None] + offs_h[None, :] * hidden_stride,
                            mask=mask_m[:, None] & mask_h[None, :],
                            other=0.0).to(tl.float32)            # [M, K]

        # LayerNorm parameters (γ, β)
        gamma = tl.load(ln_w_ptr + offs_h, mask=mask_h, other=0.0)   # [K]
        beta  = tl.load(ln_b_ptr + offs_h, mask=mask_h, other=0.0)   # [K]

        # ((out‑µ)·rstd·γ + β)
        norm = ((out_tile - mu[:, None]) * rstd[:, None]) * gamma[None, :] + beta[None, :]

        # apply out‑gate
        term = norm * gate_tile                               # [M, K]
        term_fp16 = term.to(tl.float16)                       # back to FP16 for tensor‑core matmul

        # weight slice (K, BLOCK_N) – transposed weight stored as [H, dim]
        w_ptr = w_out_T_ptr + offs_h[:, None] * dim + offs_n[None, :]   # [K, Ndim]
        w_tile = tl.load(w_ptr,
                         mask=mask_h[:, None] & mask_n[None, :],
                         other=0.0)                                     # [K, Ndim]

        # accumulate
        acc += tl.dot(term_fp16, w_tile).to(tl.float32)                # [M, Ndim]
    # store final FP32 result
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry‑point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
              (all float32)
            - config_dict must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim        = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask     = cfg.get("nomask", True)
    eps        = 1e-5
    device     = x.device

    B, N, _, _ = x.shape
    R = B * N * N                # total pairwise tokens
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # --------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                     # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    NUM_D_BLOCKS = (MAX_DIM + BLOCK_D - 1) // BLOCK_D            # = 6 for MAX_DIM=384
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        NUM_D_BLOCKS=NUM_D_BLOCKS,
        num_warps=8,
    )
    del x_flat  # free memory early

    # --------------------------------------------------------------
    # 2️⃣ Fused linear projection for the five heads (5·H)
    # --------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                              # [5·H, dim]

    proj = F.linear(x_norm, fused_w)                 # (R, 5·H) FP16
    del fused_w, x_norm

    # --------------------------------------------------------------
    # 3️⃣ Split, mask and sigmoid gates → left / right / out_gate
    # --------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        # Convert mask to 0/1 FP16 and flatten to [R]
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK  = 1
        mask_ptr  = mask_flat
    else:
        HAS_MASK  = 0
        mask_ptr  = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64
    BLOCK_H = 32                                 # tile size for hidden dimension
    NUM_H_BLOCKS = (MAX_H + BLOCK_H - 1) // BLOCK_H  # = 4 for MAX_H=128
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        NUM_H_BLOCKS=NUM_H_BLOCKS,
        num_warps=8,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣ Batched GEMM (left·rightᵀ) – core einsum
    # --------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)               # [B·H, N, N] FP16
    right_mat = right.view(B * H, N, N)              # [B·H, N, N] FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                       # [B, H, N, N] FP16

    # --------------------------------------------------------------
    # 5️⃣ Row‑wise mean & rsqrt‑var over hidden dim
    # --------------------------------------------------------------
    mu   = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 256
    BLOCK_H_STATS = BLOCK_H                     # reuse same hidden tile size
    NUM_H_BLOCKS_STATS = NUM_H_BLOCKS
    grid_stats = (triton.cdiv(R, BLOCK_M_STATS),)
    rowwise_stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_H=BLOCK_H_STATS,
        NUM_H_BLOCKS=NUM_H_BLOCKS_STATS,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 6️⃣ Final LayerNorm + out_gate + to_out projection
    # --------------------------------------------------------------
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim] (transposed)

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 128
    BLOCK_K_OUT = BLOCK_H                     # hidden tile size
    BLOCK_N_OUT = 128
    NUM_H_BLOCKS_OUT = NUM_H_BLOCKS

    grid_out = (
        triton.cdiv(R, BLOCK_M_OUT),
        triton.cdiv(dim, BLOCK_N_OUT),
    )
    final_output_kernel[grid_out](
        out,
        out_gate,
        mu,
        rstd,
        w["to_out_norm.weight"],   # γ  (FP32)
        w["to_out_norm.bias"],     # β  (FP32)
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        NUM_H_BLOCKS=NUM_H_BLOCKS_OUT,
        num_warps=8,
    )
    return final
