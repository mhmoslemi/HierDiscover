"""
TriMul (outgoing) – fast forward‑only implementation.

The kernel performs the forward pass of the “outgoing” Triangle
Multiplicative Update from AlphaFold‑3.  The high‑cost operations
(LayerNorms, the 5 linear projections, masking & gating, and the
final output projection) are fused where possible, while the heavy
pairwise contraction is delegated to cuBLAS (`torch.bmm`).  Only the
first LayerNorm and the fused projection+mask+gate step are written
in Triton; the second LayerNorm and the final projection use the
highly‑optimised PyTorch kernels.

Input
-----
x          : torch.Tensor[ B, N, N, C ]  (float32)
mask       : torch.Tensor[ B, N, N ]    (bool/float) or None
weights    : dict with the same keys as the reference PyTorch module
cfg        : {"dim": C, "hidden_dim": H, "nomask": bool, ...}

Output
------
y : torch.Tensor[ B, N, N, C ] (float32)
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # [R, C]  FP32 input
    y_ptr,          # [R, C]  FP16 output
    w_ptr,          # [C]    γ  FP32
    b_ptr,          # [C]    β  FP32
    rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # ---- compute mean / variance -------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- write normalized values --------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, apply mask & sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,      # [R, 5·H]  FP16 fused projection
    left_ptr,      # [B, H, N, N] FP16
    right_ptr,     # [B, H, N, N] FP16
    out_gate_ptr,  # [B, H, N, N] FP16
    mask_ptr,      # [R] (optional) FP16
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)   # [BLOCK_R]
    mask_r = offs_r < R

    NN = N * N
    b = offs_r // NN
    rem = offs_r - b * NN
    i = rem // N
    k = rem - i * N

    # mask broadcast over hidden dimension
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = m_val[:, None]                                   # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = NN
    base_offset = b * (H * hidden_stride) + i * N + k      # offset for h=0

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # layout inside proj: [left|right|left_gate|right_gate|out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw       = tl.load(proj_ptr + proj_base,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        right_raw      = tl.load(proj_ptr + proj_base + H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        left_gate_raw  = tl.load(proj_ptr + proj_base + 2 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        right_gate_raw = tl.load(proj_ptr + proj_base + 3 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        out_gate_raw   = tl.load(proj_ptr + proj_base + 4 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)

        # sigmoid in FP32 then back to FP16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate   # already sigmoid, no mask

        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst_offset,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset,
                 outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
#  Entrypoint
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, C] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
        - weights_dict : contains all linear / LN parameters
        - config_dict  : {"dim": C, "hidden_dim": H, "nomask": bool, ...}

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, C] (float32)
    """
    # ------------------------------ unpack ---------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]                # C
    hidden_dim = cfg["hidden_dim"]  # H
    nomask = cfg.get("nomask", True)
    eps = 1e-5

    device = x.device
    B, N, N2, _ = x.shape
    assert N == N2, "TriMul expects square pairwise tensor"
    R = B * N * N                    # number of (b,i,k) rows for the first projection

    # -------------------------- 1️⃣ LayerNorm -------------------------------
    x_flat = x.reshape(R, dim).contiguous()               # [R, C] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    layernorm_kernel[(triton.cdiv(R, BLOCK_M),)](
        x_flat,
        x_norm,
        weights["norm.weight"],
        weights["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # -------------------------- 2️⃣ Fused projection ------------------------
    # Fuse left/right and the three gates into a single matmul
    fused_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                   # [5*H, C]

    proj = F.linear(x_norm, fused_w)                       # [R, 5*H]  FP16
    del fused_w, x_norm

    # -------------------------- 3️⃣ Split, mask, sigmoid -------------------
    left      = torch.empty(B, hidden_dim, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, hidden_dim, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, hidden_dim, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    split_fuse_gate_kernel[(triton.cdiv(R, BLOCK_R),)](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        hidden_dim,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # -------------------------- 4️⃣ Batched GEMM (cuBLAS) -------------------
    # left/right : [B, H, N, N] → reshape to batch for bmm
    left_mat  = left.view(B * hidden_dim, N, N)                     # [B·H, N, N]
    right_mat = right.view(B * hidden_dim, N, N)                    # [B·H, N, N]

    # out[b,h,i,j] = Σ_k left[b,h,i,k] * right[b,h,j,k]
    contracted = torch.bmm(left_mat, right_mat.transpose(1, 2))    # [B·H, N, N]

    # reshape to (B,N,N,H) for the second LayerNorm
    out = contracted.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()   # [B,N,N,H]

    del left_mat, right_mat, contracted, left, right

    # -------------------------- 5️⃣ Second LayerNorm (torch) ---------------
    # torch.layer_norm is heavily tuned; we operate on flattened rows.
    out_flat = out.view(R, hidden_dim).contiguous()               # [R, H]
    out_norm = F.layer_norm(
        out_flat,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )                                                             # FP16

    # -------------------------- 6️⃣ Gate + final projection ---------------
    # out_gate has shape [B, H, N, N]; bring it to the same layout as out_norm
    out_gate_flat = out_gate.permute(0, 2, 3, 1).contiguous().view(R, hidden_dim)  # [R, H]

    gated = out_norm * out_gate_flat                               # element‑wise FP16

    # final linear projection H → C
    w_out = weights["to_out.weight"].to(torch.float16)            # [C, H]
    final_fp16 = F.linear(gated, w_out)                           # [R, C]  FP16
    final = final_fp16.to(torch.float32).view(B, N, N, dim)       # [B,N,N,C] FP32

    return final
