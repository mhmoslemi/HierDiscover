def custom_kernel(data):
    import torch
    import torch.nn.functional as F
    import triton
    import triton.language as tl

    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device
    B, N, _, _ = x.shape
    R = B * N * N
    H = hidden_dim

    # 1. LayerNorm over dim (input)
    # Use torch.nn.functional.layer_norm
    x_norm = torch.nn.functional.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    ).to(torch.float16)  # to half for projection

    # 2. Fused projection (5*H)
    fused_weight = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)  # shape [5*H, dim]

    proj = F.linear(x_norm, fused_weight)  # shape [R, 5*H] FP16

    # 3. Split into left, right, out_gate with mask + sigmoid
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if not nomask and mask is not None:
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)

    # kernel launch
    BLOCK_R = 128
    BLOCK_H = 64
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj

    # 4. Compute contraction out = left @ right^T per batch*hidden
    left_mat = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [B*H,N,N] FP16
    out = out.view(B, H, N, N)  # [B,H,N,N] FP16

    # 5. Second LayerNorm over hidden dim H
    # Permute to [B,N,N,H] for layer_norm
    out_perm = out.permute(0, 2, 3, 1).contiguous()  # [B,N,N,H]
    out_norm = torch.nn.functional.layer_norm(
        out_perm,
        normalized_shape=(H,),
        weight=w["to_out_norm.weight"],
        bias=w["to_out_norm.bias"],
        eps=eps,
    )  # FP16

    # 6. Multiply by out_gate (permute to match shape)
    out_gate_perm = out_gate.permute(0, 2, 3, 1).contiguous()  # [B,N,N,H]
    out_norm = out_norm * out_gate_perm

    # 7. Final linear projection to dim
    # Flatten batch & N,N
    out_flat = out_norm.view(-1, H)  # shape [B*N*N, H]
    w_out = w["to_out.weight"].to(torch.float16).t()  # shape [H, dim]
    final = F.linear(out_flat, w_out)  # [B*N*N, dim] FP16
    final = final.view(B, N, N, dim).to(torch.float32)
    return final

"""
TriMul (outgoing) – fast forward pass for AlphaFold‑3 style triangle multiplicative update.

Algorithm
---------
1️⃣ First LayerNorm (over the feature dim `dim`) – use PyTorch’s highly‑optimised
    `torch.nn.functional.layer_norm` on the FP32 input and cast to FP16.

2️⃣ Fuse the five linear projections (left, right, left‑gate, right‑gate, out‑gate)
    into a single GEMM (`torch.nn.functional.linear`).  The fused weight has shape
    `[5·hidden_dim, dim]` (FP16) and produces a tensor `[R, 5·hidden_dim]`,
    where `R = B·N·N`.

3️⃣ Custom Triton kernel `split_fuse_gate_kernel`:
       – decodes the flat index (`b,i,k`) → the three‑dimensional view.
       – extracts the five slices, applies the optional mask,
         computes sigmoid gates and stores
         `left`, `right` and `out_gate` in layout `[B, hidden_dim, N, N]` (FP16).

4️⃣ Batched GEMM (`torch.bmm`) computes the core contraction
       `out[b,h,i,j] = Σ_k left[b,h,i,k] * right[b,h,j,k]`
    (still in FP16).

5️⃣ Second LayerNorm over the hidden dimension `hidden_dim` – again via
    `torch.nn.functional.layer_norm` after permuting to `[B,N,N,hidden_dim]`.

6️⃣ Element‑wise multiplication with the `out_gate` (already sigmoid‑ed).

7️⃣ Final Linear projection to the original feature dimension `dim`
    (`torch.nn.functional.linear` with weight `[dim, hidden_dim]`).

Only the split‑and‑gate step is written in Triton; all large GEMMs and
LayerNorms use highly‑optimised PyTorch/CuBLAS kernels, dramatically reducing
kernel‑launch overhead and overall runtime.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# 3️⃣ Split fused projection, apply optional mask, compute sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,      # [R, 5*H]   FP16 fused projection
    left_ptr,      # [B, H, N, N] FP16
    right_ptr,     # [B, H, N, N] FP16
    out_gate_ptr,  # [B, H, N, N] FP16
    mask_ptr,      # [R]        FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional per‑row mask (broadcast over the hidden dimension)
    if HAS_MASK:
        m = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = m[:, None]                                   # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # base offset inside the (B,H,N,N) tensors for the (b,i,k) plane
    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + k   # start of h=0

    # process the hidden dimension in tiles
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # layout inside `proj` : 5 blocks of size H per row
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw       = tl.load(proj_ptr + proj_base,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        right_raw      = tl.load(proj_ptr + proj_base + H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        left_gate_raw  = tl.load(proj_ptr + proj_base + 2 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        right_gate_raw = tl.load(proj_ptr + proj_base + 3 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        out_gate_raw   = tl.load(proj_ptr + proj_base + 4 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)

        # stable sigmoid in FP32 → cast back to FP16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate

        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst_offset,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset,
                 out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Entry point – forward pass of outgoing TriMul (no gradient)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]          (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
        - weights_dict : mapping with all model weights (float32)
        - config_dict  : contains "dim", "hidden_dim", "nomask"

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    torch._C._log_api_usage_once("custom_kernel.triton_trimul")  # silence warnings
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                 # total positions (b,i,k)
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm over the feature dimension (dim)
    # ------------------------------------------------------------------
    # torch.nn.functional.layer_norm works on FP32; we cast to FP16 afterwards.
    x_norm_fp32 = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    )
    x_norm = x_norm_fp32.to(torch.float16)               # [B,N,N,dim] → FP16

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection to 5·hidden_dim (still FP16)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                 # [5·H, dim]

    proj = F.linear(x_norm.view(R, dim), fused_w)       # [R, 5·H] FP16
    del x_norm  # free memory

    # ------------------------------------------------------------------
    # 3️⃣ Split‑fuse‑gate kernel (mask handling + sigmoid)
    # ------------------------------------------------------------------
    left  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy

    BLOCK_R = 128
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Core contraction via batched GEMM (BMM)
    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)        # [B·H, N, N] FP16
    right_mat = right.view(B * H, N, N)       # [B·H, N, N] FP16

    # out[b,h,i,j] = Σ_k left[b,h,i,k] * right[b,h,j,k]
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                               # [B, H, N, N] FP16
    del left_mat, right_mat

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm over the hidden dim (H)
    # ------------------------------------------------------------------
    # Permute to bring the hidden dimension to the last axis (LayerNorm works on that)
    out_perm = out.permute(0, 2, 3, 1).contiguous()          # [B, N, N, H] FP16
    out_norm = F.layer_norm(
        out_perm,
        normalized_shape=(H,),
        weight=w["to_out_norm.weight"],
        bias=w["to_out_norm.bias"],
        eps=eps,
    )  # FP16

    # ------------------------------------------------------------------
    # 6️⃣ Apply out_gate (after permuting it to the same layout)
    # ------------------------------------------------------------------
    out_gate_perm = out_gate.permute(0, 2, 3, 1).contiguous()  # [B, N, N, H] FP16
    out_norm = out_norm * out_gate_perm                         # element‑wise FP16

    # ------------------------------------------------------------------
    # 7️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    # flatten batch & spatial dims for the final matmul
    out_flat = out_norm.view(-1, H)                        # [B·N·N, H] FP16
    w_out = w["to_out.weight"].to(torch.float16)           # [dim, H] FP16
    final_fp16 = F.linear(out_flat, w_out)                 # [B·N·N, dim] FP16
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
```
