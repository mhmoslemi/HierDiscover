@triton.jit
def layernorm_kernel(
    x_ptr, y_ptr,
    w_ptr, b_ptr,
    rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # compute sum, sum_sq
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # second pass: write normalized values
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]
        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        beta = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :], out, mask=mask_m[:, None] & mask_d[None, :])
```

"""
TriMul (outgoing) – mixed‑precision forward pass with a Triton
row‑wise LayerNorm kernel.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the flattened pairwise tensor (FP32 → FP32).
2️⃣ Fuse the five linear maps (left, right, left_gate, right_gate, out_gate)
   into a single GEMM (cuBLAS) producing a matrix (R, 5·H) in FP32.
3️⃣ Split the fused projection, broadcast the optional mask, apply the
   sigmoid gates and obtain three tensors `left`, `right` and `out_gate`
   in layout [B, H, N, N] (FP32).
4️⃣ Batched GEMM over the hidden dimension evaluates the core einsum:
        out[b, h, i, j] = Σₖ left[b, h, i, k]·right[b, h, j, k]
   (torch.bmm) → [B, H, N, N] (FP32).
5️⃣ Row‑wise mean & rsqrt‑variance of `out` over the hidden dimension H
   (torch.mean / torch.var) → μ, rstd per (b,i,j).
6️⃣ Final LayerNorm + out‑gate multiplication + output projection
   (all FP32) → `[B, N, N, dim]` (float32).

All kernels target Triton 3.3.1 (H100) and use mixed‑precision arithmetic
where it does not affect numerical correctness.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP32)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,                # [R, dim] input (float32)
    y_ptr,                # [R, dim] output (float32)
    w_ptr, b_ptr,         # [dim] weight & bias (float32)
    rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # ---------- first pass: mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        # Load a tile of the input (row‑major layout)
        vals = tl.load(
            x_ptr + offs_m[:, None] * dim + offs_d[None, :],
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- second pass: write normalized output ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        vals = tl.load(
            x_ptr + offs_m[:, None] * dim + offs_d[None, :],
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * dim + offs_d[None, :],
            out,
            mask=mask_m[:, None] & mask_d[None, :],
        )


# ----------------------------------------------------------------------
# Entry point ---------------------------------------------------------
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul operator (no gradients).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
              (all float32)
            - config_dict must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim          = cfg["dim"]
    hidden_dim   = cfg["hidden_dim"]
    nomask       = cfg.get("nomask", True)
    eps          = 1e-5
    device       = x.device

    B, N, _, _ = x.shape
    R = B * N * N               # total pairwise tokens
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP32) via Triton
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                     # [R, dim] FP32
    x_norm = torch.empty_like(x_flat)                           # [R, dim] FP32
    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    # ---------------------------------------------------------------
    # 2️⃣ Fused projection for the five heads (left, right, gates)
    # ---------------------------------------------------------------
    # Stack the five weight matrices (each [H, dim]) → [5·H, dim]
    fused_weight = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float32)                                          # [5·H, dim]

    # Linear: (R, dim) @ (dim, 5·H)ᵀ → (R, 5·H)
    proj = F.linear(x_norm, fused_weight)                        # FP32
    del fused_weight, x_norm  # free memory early

    # ---------------------------------------------------------------
    # 3️⃣ Split projection, apply mask, sigmoid gates
    # ---------------------------------------------------------------
    # Split into five tensors of shape (R, H)
    left_raw, right_raw, left_gate_raw, right_gate_raw, out_gate_raw = torch.split(
        proj, H, dim=1
    )
    # Reshape to [B, N, N, H]
    left_raw  = left_raw.view(B, N, N, H)
    right_raw = right_raw.view(B, N, N, H)
    left_gate_raw  = left_gate_raw.view(B, N, N, H)
    right_gate_raw = right_gate_raw.view(B, N, N, H)
    out_gate_raw   = out_gate_raw.view(B, N, N, H)

    # Optional mask handling
    if (mask is not None) and (not nomask):
        # mask: [B, N, N] → [B, N, N, 1] (float32)
        mask_f = mask.to(dtype=x.dtype).unsqueeze(-1)
        left  = left_raw  * mask_f
        right = right_raw * mask_f
    else:
        left  = left_raw
        right = right_raw

    # Apply sigmoid gates
    left_gate  = torch.sigmoid(left_gate_raw)   # [B,N,N,H]
    right_gate = torch.sigmoid(right_gate_raw)  # [B,N,N,H]
    out_gate   = torch.sigmoid(out_gate_raw)    # [B,N,N,H]

    # Final gated projections
    left  = left  * left_gate
    right = right * right_gate

    # ---------------------------------------------------------------
    # 4️⃣ Core einsum via batched GEMM (left·rightᵀ)
    # ---------------------------------------------------------------
    # Rearrange to [B*H, N, N] for torch.bmm
    left_mat  = left.permute(0, 3, 1, 2).reshape(B * H, N, N)   # [B·H, N, N]
    right_mat = right.permute(0, 3, 1, 2).reshape(B * H, N, N)  # [B·H, N, N]

    # Batched matrix multiplication: (B·H, N, N) @ (B·H, N, N)ᵀ
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))       # [B·H, N, N]
    out = out.view(B, H, N, N)                                 # [B, H, N, N]

    # ---------------------------------------------------------------
    # 5️⃣ Per‑location LayerNorm over the hidden dimension H
    # ---------------------------------------------------------------
    # μ, σ² across dim=1 (the hidden dimension)
    mu    = out.mean(dim=1, keepdim=True)                      # [B,1,N,N]
    var   = out.var(dim=1, unbiased=False, keepdim=True)      # [B,1,N,N]
    rstd  = torch.rsqrt(var + eps)                             # [B,1,N,N]

    # Normalization + learned γ (weight) & β (bias)
    gamma = w["to_out_norm.weight"].view(1, H, 1, 1)   # [1,H,1,1]
    beta  = w["to_out_norm.bias"].view(1, H, 1, 1)    # [1,H,1,1]

    out_norm = ((out - mu) * rstd) * gamma + beta     # [B,H,N,N]

    # ---------------------------------------------------------------
    # 6️⃣ Multiply by out_gate and final linear projection
    # ---------------------------------------------------------------
    out_norm = out_norm * out_gate                     # [B,H,N,N]

    # Move hidden dimension to the last axis, flatten batch & seq dims
    out_perm = out_norm.permute(0, 2, 3, 1).reshape(R, H)   # [R, H]

    # Final projection: (R, H) @ (H, dim)ᵀ → (R, dim)
    final = F.linear(out_perm, w["to_out.weight"])          # FP32, shape [R, dim]

    # Reshape to the original 4‑D layout
    final = final.view(B, N, N, dim)

    return final
```

The implementation follows the reference algorithm:

* A custom Triton kernel implements the row‑wise LayerNorm used as the first normalization step.
* The five linear maps are fused into a single GEMM, then split and combined with the optional mask and sigmoid gates.
* The core pairwise contraction is carried out with a batched `torch.bmm`.
* A second per‑location LayerNorm is performed in pure PyTorch (mean / var) followed by the out‑gate multiplication and the final output projection.

All operations run on the same CUDA device as the inputs, the output is a `torch.float32` tensor of shape `[bs, seq_len, seq_len, dim]`.
