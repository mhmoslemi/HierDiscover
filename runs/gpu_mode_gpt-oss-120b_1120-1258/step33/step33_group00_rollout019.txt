"""
TriMul (outgoing) – fast forward‑only implementation.

The operator works on a 4‑D tensor X ∈ ℝ^{B×N×N×C} and produces an output of the same
shape.  The heavy parts (two LayerNorms, the gating logic, and the final projection)
are fused into three Triton kernels:

1️⃣  Row‑wise LayerNorm on the input (FP32 → FP16).
2️⃣  Split/fuse the five linear maps (left, right, left‑gate, right‑gate, out‑gate)
    while applying the optional binary mask and the sigmoid gates.
3️⃣  Compute the second LayerNorm, multiply by the out‑gate and write a
    flattened representation Y ∈ ℝ^{(B·N·N)×H}.  Y is then projected to the
    output dimension C with a single cuBLAS GEMM (torch.nn.functional.linear).

All other dense “pairwise” work (left·right) is performed with a batched
matrix‑multiply (torch.bmm) which already exploits Tensor‑cores on H100.
The design removes the intermediate µ/σ buffers of the reference
implementation and lets cuBLAS handle the final large linear layer,
yielding lower memory traffic and higher throughput.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # [R, C]  FP32 input
    y_ptr,          # [R, C]  FP16 output
    w_ptr,          # [C]    γ  FP32
    b_ptr,          # [C]    β  FP32
    rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # ---------- first pass: mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)   # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- second pass: write normalized values ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2️⃣ Split/fuse projection, apply mask and sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,      # [R, 5·H]  FP16 fused projection
    left_ptr,      # [B, H, N, N] FP16
    right_ptr,     # [B, H, N, N] FP16
    out_gate_ptr,  # [B, H, N, N] FP16
    mask_ptr,      # [R] (optional) FP16
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)   # [BLOCK_R]
    mask_r = offs_r < R

    NN = N * N
    b = offs_r // NN
    rem = offs_r - b * NN
    i = rem // N
    k = rem - i * N

    # optional mask broadcast over hidden dimension
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = m_val[:, None]                                   # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = NN
    base_offset = b * (H * hidden_stride) + i * N + k   # offset of h=0

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # Layout inside proj: [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw       = tl.load(proj_ptr + proj_base,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        right_raw      = tl.load(proj_ptr + proj_base + H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        left_gate_raw  = tl.load(proj_ptr + proj_base + 2 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        right_gate_raw = tl.load(proj_ptr + proj_base + 3 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        out_gate_raw   = tl.load(proj_ptr + proj_base + 4 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)

        # Sigmoid (FP32 → FP16)
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate   # no mask

        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst_offset,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset,
                 outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 3️⃣ Second LayerNorm + out‑gate → flattened Y (R × H)
# ----------------------------------------------------------------------
@triton.jit
def norm_gate_kernel(
    out_ptr,           # [B, H, N, N] FP16 (contracted tensor)
    gate_ptr,          # [B, H, N, N] FP16 (out_gate after sigmoid)
    ln_w_ptr,          # [H] FP32   (to_out_norm.weight, γ)
    ln_b_ptr,          # [H] FP32   (to_out_norm.bias,   β)
    Y_ptr,             # [R, H] FP16 output (flattened LN(out)*gate)
    R, H, N,
    eps,
    BLOCK_R: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    NN = N * N
    b = offs_r // NN
    rem = offs_r - b * NN
    i = rem // N
    j = rem - i * N

    hidden_stride = NN
    base_offset = b * (H * hidden_stride) + i * N + j   # start of hidden=0

    # --------- first pass : compute mean & rstd ----------
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_R], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        ptr = out_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(ptr,
                       mask=mask_r[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mu = sum_val / H
    var = sum_sq / H - mu * mu
    rstd = tl.rsqrt(var + eps)

    # --------- second pass : apply LN, multiply by out_gate and store Y ----------
    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        # load raw out and out_gate
        out_vals = tl.load(out_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride,
                           mask=mask_r[:, None] & mask_k[None, :],
                           other=0.0).to(tl.float32)
        gate_vals = tl.load(gate_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride,
                            mask=mask_r[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)

        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # FP32
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # FP32

        # LN(out)
        norm = ((out_vals - mu[:, None]) * rstd[:, None]) * gamma[None, :] + beta[None, :]

        # multiply by out_gate (already sigmoid)
        gated = norm * gate_vals

        # store into Y (row‑major: [R, H])
        Y_ptr + (offs_r[:, None] * H + offs_k[None, :])
        tl.store(Y_ptr + (offs_r[:, None] * H + offs_k[None, :]),
                 gated.to(tl.float16),
                 mask=mask_r[:, None] & mask_k[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, C] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
        - weights_dict : mapping of all weight tensors required by the module
        - config_dict  : {"dim": C, "hidden_dim": H, "nomask": bool, ...}

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]                # C
    hidden_dim = cfg["hidden_dim"]  # H
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, N2, _ = x.shape
    assert N == N2, "TriMul expects a square pairwise matrix"
    R = B * N * N                  # rows = (b,i,k) for the per‑pair projections
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                # [R, C] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    layernorm_kernel[(triton.cdiv(R, BLOCK_M),)](
        x_flat,
        x_norm,
        weights["norm.weight"],
        weights["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear projection (5·H outputs)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                    # [5·H, C]
    proj = F.linear(x_norm, fused_w)       # [R, 5·H]  FP16
    del fused_w, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ Split, apply optional mask, and compute sigmoid gates
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    split_fuse_gate_kernel[(triton.cdiv(R, BLOCK_R),)](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM over the sequence dimension (cuBLAS)
    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)                     # [B·H, N, N] FP16
    right_mat = right.view(B * H, N, N)                    # [B·H, N, N] FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                            # [B, H, N, N] FP16
    del left, right   # no longer needed

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm + out_gate (produces flattened Y)
    # ------------------------------------------------------------------
    Y = torch.empty(R, H, dtype=torch.float16, device=device)

    BLOCK_R_STATS = 128
    BLOCK_K_STATS = 32                         # H ≤ 128 ⇒ 32‑wide tiles are fine
    norm_gate_kernel[(triton.cdiv(R, BLOCK_R_STATS),)](
        out,
        out_gate,
        weights["to_out_norm.weight"],   # γ  (FP32)
        weights["to_out_norm.bias"],     # β  (FP32)
        Y,
        R,
        H,
        N,
        eps,
        BLOCK_R=BLOCK_R_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=8,
    )
    del out, out_gate

    # ------------------------------------------------------------------
    # 6️⃣ Final projection H → C with cuBLAS (torch.linear)
    # ------------------------------------------------------------------
    # weight shape for torch.linear: (out_features, in_features) = (C, H)
    w_out = weights["to_out.weight"].to(torch.float16)   # [C, H]
    final = F.linear(Y, w_out)      # [R, C]  FP16
    final = final.to(torch.float32).view(B, N, N, dim)   # float32 output

    return final
```

The implementation follows the reference TriMul module, but:

* **LayerNorm** and **gate‑splitting** are executed in custom Triton kernels.
* **The second LayerNorm and out‑gate multiplication** are fused into a single kernel that emits a flattened tensor **Y**.
* **The final H→C projection** is delegated to cuBLAS via `torch.nn.functional.linear`, which is highly optimized on H100.
* Intermediate statistics (`μ`, `σ`) are never written to global memory, reducing traffic and improving the overall runtime.
