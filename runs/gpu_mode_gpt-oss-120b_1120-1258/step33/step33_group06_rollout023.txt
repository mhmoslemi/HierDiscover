def split_fuse_gate_kernel(
    proj_ptr, left_ptr, right_ptr, out_gate_ptr, mask_ptr,
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)

    mask_r = offs_r < R

    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

```python
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

def custom_kernel(data):
    """
    Outgoing TriMul (AlphaFold‑3) – forward only.
    -------------------------------------------------
    1. Row‑wise LayerNorm (`F.layer_norm`) on the flattened pair tensor.
    2. Fuse the five linear maps (left/right projections + three gates) into a single
       GEMM (`F.linear`) producing a tensor of shape [R, 5·H] (R = B·N·N).
    3. Triton kernel ``split_fuse_gate_kernel`` extracts the five slices,
       applies the optional pairwise mask, evaluates the sigmoid gates and writes
       ``left``, ``right`` and ``out_gate`` tensors in layout [B, H, N, N] (FP16).
    4. Batched matrix‑multiply (cuBLAS) computes the core einsum:
               out[b, h, i, j] = Σₖ left[b, h, i, k] · right[b, h, j, k]
       resulting in a tensor ``out`` of shape [B, H, N, N] (FP16).
    5. Triton kernel ``rowwise_stats_kernel`` computes per‑row mean and
       rsqrt‑variance of ``out`` (over the hidden dimension H) → ``mu`` and ``rstd``.
    6. Triton kernel ``final_output_kernel`` fuses the second LayerNorm,
       the ``out_gate`` multiplication and the final linear projection, emitting
       the final tensor of shape [B, N, N, dim] (FP32).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    device = x.device
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = bool(cfg.get("nomask", True))
    eps = 1e-5

    B, N, _, _ = x.shape
    R = B * N * N
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm ( FP32 → FP16 ) using cuDNN for speed
    # ------------------------------------------------------------------
    # flatten to [R, dim]
    x_flat = x.reshape(R, dim).contiguous()
    # torch layer_norm operates on the last dimension; we supply per‑row weight/bias
    x_norm_fp32 = F.layer_norm(
        x_flat,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )
    x_norm = x_norm_fp32.to(torch.float16)

    # ------------------------------------------------------------------
    # 2️⃣ Fuse five linear maps (left, right, left_gate, right_gate, out_gate)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)  # [5·H, dim]

    # Linear projection on the fused weight matrix
    proj = F.linear(x_norm, fused_w)  # [R, 5·H] in FP16
    # free memory early
    del x_norm, fused_w

    # ------------------------------------------------------------------
    # 3️⃣ Split, mask, sigmoid gates → left/right/out_gate
    # ------------------------------------------------------------------
    left = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if not nomask and mask is not None:
        mask_fp16 = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_fp16
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy

    @triton.jit
    def split_fuse_gate_kernel(
        proj_ptr, left_ptr, right_ptr, out_gate_ptr, mask_ptr,
        R, H, N,
        HAS_MASK: tl.constexpr,
        BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
    ):
        pid = tl.program_id(0)
        row_start = pid * BLOCK_R
        offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
        mask_r = offs_r < R

        # decode (b,i,k) from flat index
        nn = N * N
        b = offs_r // nn
        rem = offs_r - b * nn
        i = rem // N
        k = rem - i * N

        # optional mask (broadcast over H)
        if HAS_MASK:
            m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)  # [BLOCK_R]
            m_val = m_val[:, None]  # [BLOCK_R, 1]
        else:
            m_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

        hidden_stride = N * N
        base = b * (H * hidden_stride) + i * N + k   # start of hidden plane 0

        # iterate over hidden dimension tiles
        for h_start in range(0, H, BLOCK_H):
            cur_h = h_start + tl.arange(0, BLOCK_H)                 # [BLOCK_H]
            mask_h = cur_h < H

            # offset into proj: each row has 5*H elements
            proj_off = offs_r[:, None] * (5 * H) + cur_h[None, :]   # left segment start

            left_raw = tl.load(proj_ptr + proj_off,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)
            right_raw = tl.load(proj_ptr + proj_off + H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
            left_gate_raw = tl.load(proj_ptr + proj_off + 2 * H,
                                    mask=mask_r[:, None] & mask_h[None, :],
                                    other=0.0)
            right_gate_raw = tl.load(proj_ptr + proj_off + 3 * H,
                                     mask=mask_r[:, None] & mask_h[None, :],
                                     other=0.0)
            out_gate_raw = tl.load(proj_ptr + proj_off + 4 * H,
                                   mask=mask_r[:, None] & mask_h[None, :],
                                   other=0.0)

            # sigmoid (FP32->FP16)
            left_gate = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
            right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
            outg = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

            left_val = left_raw * m_val * left_gate
            right_val = right_raw * m_val * right_gate

            dst = base[:, None] + cur_h[None, :] * hidden_stride  # [BLOCK_R, BLOCK_H]

            tl.store(left_ptr + dst, left_val,
                     mask=mask_r[:, None] & mask_h[None, :])
            tl.store(right_ptr + dst, right_val,
                     mask=mask_r[:, None] & mask_h[None, :])
            tl.store(out_gate_ptr + dst, outg,
                     mask=mask_r[:, None] & mask_h[None, :])

    # launch split/fuse kernel
    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM over hidden dimension (cuBLAS)
    # ------------------------------------------------------------------
    # left/right are stored as [B, H, N, N].  View as (B*H, N, N) for bmm.
    left_mat = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)

    # Compute out = left @ right^T for each slice.
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))
    out = out.view(B, H, N, N)   # [B, H, N, N] FP16

    # ------------------------------------------------------------------
    # 5️⃣ Row‑wise mean & rsqrt‑variance of `out`
    # ------------------------------------------------------------------
    mu = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    @triton.jit
    def rowwise_stats_kernel(
        out_ptr, mu_ptr, rstd_ptr,
        total_rows, H, N,
        eps,
        BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr,
    ):
        pid = tl.program_id(0)
        row_start = pid * BLOCK_M
        offs_m = row_start + tl.arange(0, BLOCK_M)
        mask_m = offs_m < total_rows

        nn = N * N
        b = offs_m // nn
        rem = offs_m - b * nn
        i = rem // N
        j = rem - i * N

        hidden_stride = N * N
        base = b * (H * hidden_stride) + i * N + j

        sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
        sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

        for k_start in range(0, H, BLOCK_K):
            cur_k = k_start + tl.arange(0, BLOCK_K)
            mask_k = cur_k < H

            ptr = out_ptr + base[:, None] + cur_k[None, :] * hidden_stride
            vals = tl.load(ptr,
                           mask=mask_m[:, None] & mask_k[None, :],
                           other=0.0).to(tl.float32)   # [BLOCK_M, BLOCK_K]

            sum_val += tl.sum(vals, axis=1)
            sum_sq += tl.sum(vals * vals, axis=1)

        mean = sum_val / H
        var = sum_sq / H - mean * mean
        rstd_val = tl.rsqrt(var + eps)

        tl.store(mu_ptr + offs_m, mean, mask=mask_m)
        tl.store(rstd_ptr + offs_m, rstd_val, mask=mask_m)

    BLOCK_M_STATS = 256
    BLOCK_K_STATS = H   # one tile covers whole hidden dim (H ≤ 128)
    grid_stats = (triton.cdiv(R, BLOCK_M_STATS),)
    rowwise_stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Fuse second LayerNorm, out_gate, and final projection
    # ------------------------------------------------------------------
    # transpose final projection weight for column‑wise loading
    w_out_T = weights["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim]

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    @triton.jit
    def final_output_kernel(
        out_ptr, gate_ptr, mu_ptr, rstd_ptr,
        ln_w_ptr, ln_b_ptr,
        w_out_T_ptr, out_final_ptr,
        total_rows, H, N, dim,
        eps,
        BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr,
    ):
        pid_m = tl.program_id(0)   # rows (R)
        pid_n = tl.program_id(1)   # output‑dim tiles

        row_start = pid_m * BLOCK_M
        col_start = pid_n * BLOCK_N

        offs_m = row_start + tl.arange(0, BLOCK_M)   # [BLOCK_M]
        offs_n = col_start + tl.arange(0, BLOCK_N)   # [BLOCK_N]

        mask_m = offs_m < total_rows
        mask_n = offs_n < dim

        nn = N * N
        b = offs_m // nn
        rem = offs_m - b * nn
        i = rem // N
        j = rem - i * N

        hidden_stride = N * N
        base = b * (H * hidden_stride) + i * N + j

        # load per‑row statistics
        mu = tl.load(mu_ptr + offs_m, mask=mask_m, other=0.0)          # [BLOCK_M]
        rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)    # [BLOCK_M]

        acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

        for k_start in range(0, H, BLOCK_K):
            cur_k = k_start + tl.arange(0, BLOCK_K)
            mask_k = cur_k < H

            # load hidden plane values
            ptr_out = out_ptr + base[:, None] + cur_k[None, :] * hidden_stride
            ptr_gate = gate_ptr + base[:, None] + cur_k[None, :] * hidden_stride

            vals_out = tl.load(ptr_out,
                               mask=mask_m[:, None] & mask_k[None, :],
                               other=0.0).to(tl.float32)   # [M, K]
            vals_gate = tl.load(ptr_gate,
                                mask=mask_m[:, None] & mask_k[None, :],
                                other=0.0).to(tl.float32)   # [M, K]

            # LayerNorm params for this hidden slice
            gamma = tl.load(ln_w_ptr + cur_k, mask=mask_k, other=0.0)   # [K]
            beta = tl.load(ln_b_ptr + cur_k, mask=mask_k, other=0.0)    # [K]

            # normalize + gate
            diff = (vals_out - mu[:, None]) * rstd[:, None]          # [M, K]
            norm = diff * gamma[None, :] + beta[None, :]             # [M, K]
            term = norm * vals_gate                                 # [M, K]
            term_fp16 = term.to(tl.float16)

            # load final projection weight slice (H, dim)
            w_ptr = w_out_T_ptr + cur_k[:, None] * dim + offs_n[None, :]   # [K, TILE_N]
            w_tile = tl.load(w_ptr,
                             mask=mask_k[:, None] & mask_n[None, :],
                             other=0.0)                                 # [K, TILE_N]

            # accumulate dot product (FP16·FP16 → FP32)
            acc += tl.dot(term_fp16, w_tile).to(tl.float32)

        dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
        tl.store(dst_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])

    # Choose tile sizes based on dimensions (compile‑time constants)
    BLOCK_M_OUT = 128
    BLOCK_K_OUT = min(64, H)  # H ≤ 128 ⇒ 64 gives ≤2 tiles
    BLOCK_N_OUT = 128

    grid_out = (
        triton.cdiv(R, BLOCK_M_OUT),
        triton.cdiv(dim, BLOCK_N_OUT),
    )
    final_output_kernel[grid_out](
        out,
        out_gate,
        mu,
        rstd,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=8,
    )

    return final

"""
Outgoing TriMul (AlphaFold‑3) – forward only.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the flattened pair tensor (FP32 → FP16) using cuDNN.
2️⃣ Fuse the five linear maps (left, right, left_gate, right_gate, out_gate) into a
   single GEMM (`F.linear`).  Result shape = [R, 5·H] where R = B·N·N.
3️⃣ Triton kernel ``split_fuse_gate_kernel`` extracts the five slices, applies the
   optional pair‑mask, evaluates three sigmoid gates and writes three tensors
   `left`, `right`, `out_gate` in layout [B, H, N, N] (FP16).
4️⃣ Batched GEMM (cuBLAS) computes the core einsum:
       out[b, h, i, j] = Σₖ left[b, h, i, k] · right[b, h, j, k]
   → [B, H, N, N] (FP16).
5️⃣ Triton kernel ``rowwise_stats_kernel`` computes per‑row mean and rsqrt‑variance
   of `out` over the hidden dimension H → `mu` and `rstd`.
6️⃣ Triton kernel ``final_output_kernel`` fuses the second LayerNorm, the
   `out_gate` multiplication and the final projection (`to_out`) into a single
   kernel that writes the final tensor `[B, N, N, dim]` (FP32).

All kernels target Triton 3.3.1 on an H100 and use mixed precision
(FP32 for statistics, FP16 for the bulk).  No gradients are computed.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 3️⃣ Split fused projection, apply mask and sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # [R, 5·H]   FP16 fused projection
    left_ptr,          # [B, H, N, N]  FP16 (output)
    right_ptr,         # [B, H, N, N]  FP16 (output)
    out_gate_ptr,      # [B, H, N, N]  FP16 (output)
    mask_ptr,          # [R]           FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # Decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # Optional pair‑mask (broadcast over H)
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)          # [BLOCK_R]
        m_val = m_val[:, None]                                            # [BLOCK_R, 1]
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + k                    # start of hidden plane 0

    # Iterate over hidden tiles
    for h_start in range(0, H, BLOCK_H):
        cur_h = h_start + tl.arange(0, BLOCK_H)                      # [BLOCK_H]
        mask_h = cur_h < H

        # Offsets inside the fused projection (layout: [left|right|lg|rg|og])
        proj_off = offs_r[:, None] * (5 * H) + cur_h[None, :]        # left segment

        left_raw      = tl.load(proj_ptr + proj_off,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_raw     = tl.load(proj_ptr + proj_off + H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        left_gate_raw = tl.load(proj_ptr + proj_off + 2 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_gate_raw= tl.load(proj_ptr + proj_off + 3 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        out_gate_raw  = tl.load(proj_ptr + proj_off + 4 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)

        # Sigmoid – compute in FP32 then cast back
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left_val  = left_raw  * m_val * left_gate
        right_val = right_raw * m_val * right_gate

        dst = base[:, None] + cur_h[None, :] * hidden_stride      # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr     + dst, left_val,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr    + dst, right_val,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst, out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 5️⃣ Row‑wise mean & rsqrt‑variance of the contracted tensor
# ----------------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr,   # [B, H, N, N]  FP16
    mu_ptr,    # [R]            FP32
    rstd_ptr,  # [R]            FP32
    total_rows, H, N,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + j          # start of hidden plane 0

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        cur_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = cur_k < H

        ptr = out_ptr + base[:, None] + cur_k[None, :] * hidden_stride
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)        # [BLOCK_M, BLOCK_K]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_m, mean,  mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)


# ----------------------------------------------------------------------
# 6️⃣ Fuse second LayerNorm, out_gate and final projection
# ----------------------------------------------------------------------
@triton.jit
def final_output_kernel(
    out_ptr,        # [B, H, N, N]  FP16
    gate_ptr,       # [B, H, N, N]  FP16
    mu_ptr,         # [R]            FP32
    rstd_ptr,       # [R]            FP32
    ln_w_ptr,       # [H]            FP32 (to_out_norm.weight)
    ln_b_ptr,       # [H]            FP32 (to_out_norm.bias)
    w_out_T_ptr,    # [H, dim]       FP16 (to_out weight transposed)
    out_final_ptr,  # [B, N, N, dim] FP32
    total_rows, H, N, dim,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr,
):
    pid_m = tl.program_id(0)   # rows (R)
    pid_n = tl.program_id(1)   # output‑dim tiles

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)   # [BLOCK_M]
    offs_n = col_start + tl.arange(0, BLOCK_N)   # [BLOCK_N]

    mask_m = offs_m < total_rows
    mask_n = offs_n < dim

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + j   # start of hidden plane 0

    # Load per‑row statistics
    mu   = tl.load(mu_ptr   + offs_m, mask=mask_m, other=0.0)          # [BLOCK_M]
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)          # [BLOCK_M]

    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        cur_k = k_start + tl.arange(0, BLOCK_K)               # [BLOCK_K]
        mask_k = cur_k < H

        # Load hidden‑plane values and gate
        ptr_out  = out_ptr  + base[:, None] + cur_k[None, :] * hidden_stride
        ptr_gate = gate_ptr + base[:, None] + cur_k[None, :] * hidden_stride

        vals_out = tl.load(ptr_out,
                           mask=mask_m[:, None] & mask_k[None, :],
                           other=0.0).to(tl.float32)          # [M, K]
        vals_gate= tl.load(ptr_gate,
                           mask=mask_m[:, None] & mask_k[None, :],
                           other=0.0).to(tl.float32)          # [M, K]

        # LayerNorm parameters for this hidden slice
        gamma = tl.load(ln_w_ptr + cur_k, mask=mask_k, other=0.0)   # [K]
        beta  = tl.load(ln_b_ptr + cur_k, mask=mask_k, other=0.0)   # [K]

        # (out - µ) * rstd → γ· + β → * out_gate
        diff = (vals_out - mu[:, None]) * rstd[:, None]               # [M, K]
        norm = diff * gamma[None, :] + beta[None, :]                  # [M, K]
        term = norm * vals_gate                                       # [M, K]
        term_fp16 = term.to(tl.float16)                               # back to FP16

        # Load slice of final projection weight (H, dim)
        w_ptr = w_out_T_ptr + cur_k[:, None] * dim + offs_n[None, :]   # [K, TILE_N]
        w_tile = tl.load(w_ptr,
                         mask=mask_k[:, None] & mask_n[None, :],
                         other=0.0)                                     # [K, TILE_N]

        # Accumulate (M,K)·(K,N) → (M,N)   (fp16·fp16 → fp32)
        acc += tl.dot(term_fp16, w_tile).to(tl.float32)

    # Store final tensor
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entrypoint
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : dict with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"               (all float32, already on device)
            - config_dict must contain
                "dim" (int), "hidden_dim" (int), "nomask" (bool)
    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    device = x.device
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = bool(cfg.get("nomask", True))
    eps = 1e-5

    B, N, _, _ = x.shape
    R = B * N * N                     # number of pairwise tokens
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16) – use cuDNN for speed
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                     # [R, dim] FP32
    x_norm_fp32 = F.layer_norm(
        x_flat,
        normalized_shape=(dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    )
    x_norm = x_norm_fp32.to(torch.float16)                      # [R, dim] FP16
    del x_flat, x_norm_fp32

    # ------------------------------------------------------------------
    # 2️⃣ Fuse five linear maps (left, right, left_gate, right_gate, out_gate)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                          # [5·H, dim]

    proj = F.linear(x_norm, fused_w)                             # [R, 5·H] FP16
    del x_norm, fused_w

    # ------------------------------------------------------------------
    # 3️⃣ Split, mask, sigmoid gates → left/right/out_gate
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if not nomask and mask is not None:
        mask_fp16 = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_fp16
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension (cuBLAS)
    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)               # FP16
    right_mat = right.view(B * H, N, N)              # FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))
    out = out.view(B, H, N, N)                       # [B, H, N, N] FP16

    # ------------------------------------------------------------------
    # 5️⃣ Row‑wise mean & rsqrt‑variance of `out`
    # ------------------------------------------------------------------
    mu   = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 256
    BLOCK_K_STATS = H          # H ≤ 128, so a single tile suffices
    grid_stats = (triton.cdiv(R, BLOCK_M_STATS),)
    rowwise_stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm + out_gate + final projection
    # ------------------------------------------------------------------
    # Transpose final linear weight for column‑wise loading
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim]

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 128
    BLOCK_K_OUT = min(64, H)      # tile size over hidden dimension
    BLOCK_N_OUT = 128

    grid_out = (
        triton.cdiv(R, BLOCK_M_OUT),
        triton.cdiv(dim, BLOCK_N_OUT),
    )
    final_output_kernel[grid_out](
        out,
        out_gate,
        mu,
        rstd,
        w["to_out_norm.weight"],   # γ (FP32)
        w["to_out_norm.bias"],     # β (FP32)
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=8,
    )

    return final
