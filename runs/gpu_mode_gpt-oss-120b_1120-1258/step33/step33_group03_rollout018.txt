import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# Triton kernel definitions (split_fuse_gate_kernel)

@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,      # [R, 5*H] FP16
    left_ptr,      # [B, H, N, N] FP16
    right_ptr,     # [B, H, N, N] FP16
    out_gate_ptr,  # [B, H, N, N] FP16
    mask_ptr,      # [R] FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask broadcasted over H
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                   # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + k               # start of plane h=0

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # offsets within fused projection:
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_raw     = tl.load(proj_ptr + proj_base + H,
                                mask=mask_r[:, None] & mask_h[None, :],

                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_raw     = tl.load(proj_ptr + proj_base + H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        left_gate_raw = tl.load(proj_ptr + proj_base + 2 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_gate_raw= tl.load(proj_ptr + proj_base + 3 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        out_gate_raw  = tl.load(proj_ptr + proj_base + 4 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)

        # Stable sigmoid in FP32, then cast back to FP16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # Apply mask & gates
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate

        # Destination offset for this hidden tile
        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst_offset, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset, right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset, outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Triton‑enabled TriMul forward (no gradient)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradient).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, C] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
                (all float32)
            - config_dict must contain:
                "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, C] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N               # total number of (b,i,k) positions
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm on the input (torch, FP16 output)
    # --------------------------------------------------------------
    # torch.layer_norm is heavily tuned on H100 and is faster than a custom kernel
    x_norm = F.layer_norm(
        x,
        (dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    ).to(torch.float16)        # [B, N, N, dim]  FP16

    # --------------------------------------------------------------
    # 2️⃣ Fused linear projection (5·H) → split + mask + sigmoid gates
    # --------------------------------------------------------------
    # Flatten to [R, dim] for the fused linear
    x_flat = x_norm.view(R, dim).contiguous()                     # [R, dim]
    # Fuse the five projection weights (no bias)
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                            # [5·H, dim]

    proj = F.linear(x_flat, fused_w)                               # [R, 5·H]  FP16

    # Allocate output tensors for left/right/out_gate (FP16)
    left     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right    = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    # Prepare mask pointer for the kernel
    if (not nomask) and (mask is not None):
        mask_fp16 = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_fp16
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    # Triton launch – split, mask, sigmoid, and store left/right/out_gate
    BLOCK_R = 64   # rows per program (tuned for typical B,N)
    BLOCK_H = 64   # hidden‑dim tiles
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    # Clean up temporary tensors
    del proj, x_flat, mask_fp16 if 'mask_fp16' in locals() else None, fused_w

    # --------------------------------------------------------------
    # 3️⃣ Core contraction – batched GEMM over the hidden dimension
    # --------------------------------------------------------------
    # Convert to [B·H, N, N] layout for torch.bmm
    left_mat  = left.permute(0, 1, 2, 3).contiguous().view(B * H, N, N)   # [B·H, N, N]
    right_mat = right.permute(0, 1, 2, 3).contiguous().view(B * H, N, N)  # [B·H, N, N]

    # out[b,h,i,j] = Σₖ left[b,h,i,k] · right[b,h,j,k]
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))                 # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                                            # [B, H, N, N] FP16

    # --------------------------------------------------------------
    # 4️⃣ Second LayerNorm (over hidden dim), gate, final linear
    # --------------------------------------------------------------
    # Bring tensors to shape [B, N, N, H] for torch.layer_norm
    out_perm     = out.permute(0, 2, 3, 1).contiguous()                 # [B, N, N, H]
    out_gate_perm= out_gate.permute(0, 2, 3, 1).contiguous()            # [B, N, N, H]

    # LayerNorm over the H dimension (γ,β are float32)
    out_norm = F.layer_norm(
        out_perm,
        (H,),
        weight=w["to_out_norm.weight"],
        bias=w["to_out_norm.bias"],
        eps=eps,
    )   # still FP16 because input is FP16

    # Apply the output gate (element‑wise)
    gated = out_norm * out_gate_perm                                     # FP16

    # Final linear projection back to C (hidden_dim → dim)
    w_out_fp16 = w["to_out.weight"].to(torch.float16)                    # [dim, hidden_dim]
    final_fp16 = F.linear(gated, w_out_fp16, bias=None)                  # [B, N, N, dim] FP16

    # Cast to float32 as required by the interface
    final = final_fp16.to(torch.float32)

    return final
