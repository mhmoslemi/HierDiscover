"""
TriMul (outgoing) – fast forward‑only implementation.

The kernel pipeline follows the reference PyTorch module but
fuses the three linear projections (left, right and the three
gates) into a single matrix‑multiply and uses a custom Triton
kernel only for the split / mask / sigmoid step.  All LayerNorms
and the final projection are delegated to the highly‑optimised
PyTorch (cuBLAS/cuDNN) kernels.

Steps
-----
1️⃣  LayerNorm on the last dimension C (torch.nn.functional.layer_norm,
    FP32 → FP16).
2️⃣  Fuse the five linear maps (left, right, left‑gate, right‑gate,
    out‑gate) into one matmul:  X_norm @ W_fused  →  [R, 5·H].
3️⃣  Custom Triton kernel (`split_fuse_gate_kernel`) splits the fused
    projection, applies the optional binary mask, computes three
    sigmoids and writes three tensors `left`, `right`, `out_gate`
    (FP16, shape [B, H, N, N]).
4️⃣  Batched GEMM (torch.bmm) over the sequence dimension:
        out[b,h,i,j] = Σₖ left[b,h,i,k] * right[b,h,j,k]   (FP16).
5️⃣  Row‑wise LayerNorm on the hidden dimension H (torch.nn.functional.layer_norm).
6️⃣  Multiply by `out_gate`, project H → C with a final linear layer
    (torch.nn.functional.linear) and cast the result back to FP32.

Only the split/fuse/gate step is written in Triton; the rest
leverages the fastest existing CUDA kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 3️⃣ Split fused projection, apply mask & sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,        # [R, 5·H]   FP16 fused projection
    left_ptr,        # [B, H, N, N]  FP16
    right_ptr,       # [B, H, N, N]  FP16
    out_gate_ptr,    # [B, H, N, N]  FP16
    mask_ptr,        # [R] (optional) FP16
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    """
    Each program works on a contiguous block of rows (R = B·N·N)
    and processes a block of hidden dimensions.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    NN = N * N
    b = offs_r // NN
    rem = offs_r - b * NN
    i = rem // N          # i-index in the pairwise matrix
    k = rem - i * N       # k-index (summation axis)

    # optional mask (broadcast along hidden dim)
    if HAS_MASK:
        # mask is 0/1 stored as FP16
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)  # [BLOCK_R]
        mask_val = m_val[:, None]                                 # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # stride of a single (i,k) element over hidden dim
    hidden_stride = NN

    # Base offset for the first hidden slice (h = 0)
    base_offset = b * (H * hidden_stride) + i * N + k   # [BLOCK_R]

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # ---- layout inside proj: [left | right | left_gate | right_gate | out_gate] ----
        # proj has shape [R, 5·H]; we compute the per‑row offset for each hidden slice
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]      # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0)

        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0)

        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0)

        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0)

        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0)

        # sigmoid in FP32 → back to FP16
        left_gate = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate   # no mask

        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst_offset,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_offset,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset,
                 outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, C] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
        - weights_dict : dict with the following keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
        - config_dict  : {"dim": C, "hidden_dim": H, "nomask": bool, ...}

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    B, N, N2, C = x.shape
    assert N == N2, "TriMul expects a square pairwise matrix"

    dim = cfg["dim"]               # output channel C
    hidden_dim = cfg["hidden_dim"] # H
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32 → FP16) using the highly‑optimised torch kernel
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x_norm = F.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)   # FP32
    x_norm_fp16 = x_norm.to(torch.float16)                                          # FP16

    # ------------------------------------------------------------------
    # 2️⃣ Fuse the five linear maps (left, right, left_gate, right_gate, out_gate)
    # ------------------------------------------------------------------
    R = B * N * N                                   # flattened rows = (b,i,k)
    x_flat = x_norm_fp16.view(R, dim).contiguous() # [R, C]

    fused_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                             # [5·H, C]

    # torch.linear performs (x_flat @ fused_w.T) → [R, 5·H]  (FP16)
    proj = F.linear(x_flat, fused_w)                # [R, 5·H]

    # ------------------------------------------------------------------
    # 3️⃣ Split, mask and apply sigmoid gates (custom Triton kernel)
    # ------------------------------------------------------------------
    left     = torch.empty(B, hidden_dim, N, N, dtype=torch.float16, device=device)
    right    = torch.empty(B, hidden_dim, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, hidden_dim, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64   # rows per program (tuned for typical R)
    BLOCK_H = 64   # hidden‑dim tiles per program

    split_fuse_gate_kernel[
        (triton.cdiv(R, BLOCK_R),)
    ](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        hidden_dim,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM over the sequence dimension (cuBLAS)
    # ------------------------------------------------------------------
    # shape for bmm: [B·H, N, N] × [B·H, N, N]^T
    left_mat  = left.view(B * hidden_dim, N, N)                     # FP16
    right_mat = right.view(B * hidden_dim, N, N)                    # FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))           # [B·H, N, N] FP16
    out = out.view(B, hidden_dim, N, N)                            # [B, H, N, N] FP16

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm (row‑wise over H) + out_gate + final projection
    # ------------------------------------------------------------------
    # Bring hidden dimension to the last axis so we can fuse the LayerNorm
    # with torch.nn.functional.layer_norm (which is very fast).
    out_perm      = out.permute(0, 2, 3, 1).contiguous()          # [B, N, N, H]
    out_gate_perm = out_gate.permute(0, 2, 3, 1).contiguous()    # [B, N, N, H]

    out_flat = out_perm.view(-1, hidden_dim)          # [(B·N·N), H]  FP16
    gate_flat = out_gate_perm.view(-1, hidden_dim)    # same shape

    # LayerNorm on hidden dimension H
    ln_w = weights["to_out_norm.weight"].to(torch.float16)
    ln_b = weights["to_out_norm.bias"].to(torch.float16)
    out_norm_flat = F.layer_norm(
        out_flat,
        (hidden_dim,),
        weight=ln_w,
        bias=ln_b,
        eps=eps,
    )                                                  # FP16

    # Apply the out‑gate
    gated_flat = out_norm_flat * gate_flat               # FP16

    # Final projection H → C
    w_out = weights["to_out.weight"].to(torch.float16)   # [C, H]
    final_flat = F.linear(gated_flat, w_out)            # [B·N·N, C] FP16

    # Reshape to the expected output layout and cast back to FP32
    final = final_flat.view(B, N, N, dim).to(torch.float32)

    return final
```
