def fused_proj_gate_kernel(
    x_ptr, left_ptr, right_ptr, out_gate_ptr,
    mask_ptr,
    left_proj_ptr, right_proj_ptr,
    left_gate_ptr, right_gate_ptr,
    out_gate_w_ptr,
    R, H, N, dim,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)  # shape BLOCK_R
    mask_r = offs_r < R

    # decode b,i,k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # base offset for hidden dimension 0
    hidden_stride = nn
    base = b * (H * hidden_stride) + i * N + k

    # Optionally load mask per row
    if HAS_MASK:
        m_val_row = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)  # fp16
        # broadcast over hidden dims later
    else:
        m_val_row = tl.full([BLOCK_R], 1.0, dtype=tl.float16)

    # Loop over hidden dim blocks
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)  # [BLOCK_H]
        mask_h = offs_h < H

        # Compute destination offset for each hidden index
        dst = base[:, None] + offs_h[None, :] * hidden_stride  # [BLOCK_R, BLOCK_H]

        # Initialize accumulators in fp32
        acc_left_raw = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        acc_right_raw = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        acc_left_gate = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        acc_right_gate = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        acc_out_gate = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)

        # Loop over dim blocks
        for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
            offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)  # [BLOCK_D]
            mask_d = offs_d < dim

            # Load input vector
            x_ptrs = x_ptr + offs_r[:, None] * dim + offs_d[None, :]  # shape [BLOCK_R, BLOCK_D]
            x = tl.load(x_ptrs, mask=mask_r[:, None] & mask_d[None, :], other=0.0).to(tl.float32)  # fp32

            # Load weight blocks (transposed) for each projection
            # left_proj weight transposed: shape [BLOCK_D, BLOCK_H]
            left_w_ptrs = left_proj_ptr + (offs_h[None, :] * dim) + offs_d[:, None]  # [BLOCK_D, BLOCK_H]
            left_w = tl.load(left_w_ptrs, mask=mask_h[None, :] & mask_d[:, None], other=0.0).to(tl.float32)

            # similarly for other weights
            right_w_ptrs = right_proj_ptr + (offs_h[None, :] * dim) + offs_d[:, None]
            right_w = tl.load(right_w_ptrs, mask=mask_h[None, :] & mask_d[:, None], other=0.0).to(tl.float32)

            left_gate_w_ptrs = left_gate_ptr + (offs_h[None, :] * dim) + offs_d[:, None]
            left_gate_w = tl.load(left_gate_w_ptrs, mask=mask_h[None, :] & mask_d[:, None], other=0.0).to(tl.float32)

            right_gate_w_ptrs = right_gate_ptr + (offs_h[None, :] * dim) + offs_d[:, None]
            right_gate_w = tl.load(right_gate_w_ptrs, mask=mask_h[None, :] & mask_d[:, None], other=0.0).to(tl.float32)

            out_gate_w_ptrs = out_gate_w_ptr + (offs_h[None, :] * dim) + offs_d[:, None]
            out_gate_w = tl.load(out_gate_w_ptrs, mask=mask_h[None, :] & mask_d[:, None], other=0.0).to(tl.float32)

            # Compute dot products for each map
            # x shape [BLOCK_R, BLOCK_D]; weight_T shape [BLOCK_D, BLOCK_H] -> dot => [BLOCK_R, BLOCK_H]
            acc_left_raw   += tl.dot(x, left_w)          # left_proj
            acc_right_raw  += tl.dot(x, right_w)         # right_proj
            acc_left_gate  += tl.dot(x, left_gate_w)    # left_gate
            acc_right_gate += tl.dot(x, right_gate_w)   # right_gate
            acc_out_gate   += tl.dot(x, out_gate_w)     # out_gate

        # after dim loop, apply sigmoid and mask
        # compute gate sigmoids in fp32 then cast
        left_gate_sig = (1.0 / (1.0 + tl.exp(-acc_left_gate))).to(tl.float16)
        right_gate_sig = (1.0 / (1.0 + tl.exp(-acc_right_gate))).to(tl.float16)
        out_gate_sig = (1.0 / (1.0 + tl.exp(-acc_out_gate))).to(tl.float16)

        # mask value broadcast to hidden dimension
        # m_val_row is fp16 shape [BLOCK_R]; broadcast to [BLOCK_R, BLOCK_H]
        m_val = m_val_row[:, None]   # shape [BLOCK_R,1]
        # apply mask and gates
        left_val = (acc_left_raw.to(tl.float16) * left_gate_sig) * m_val
        right_val = (acc_right_raw.to(tl.float16) * right_gate_sig) * m_val

        # store results
        tl.store(left_ptr + dst, left_val, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst, right_val, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst, out_gate_sig, mask=mask_r[:, None] & mask_h[None, :])
```

"""
Outgoing TriMul (AlphaFold‑3 style) – forward only.

Algorithm
---------
1️⃣ Row‑wise LayerNorm over the flattened pair‑wise tensor
   (shape [B·N·N, dim]) – FP32 → FP16.

2️⃣ Fuse the five linear maps (left_proj, right_proj,
   left_gate, right_gate, out_gate) into a single Triton kernel.
   The kernel:
        * reads a FP16‑normalized row `x`  (shape [dim])
        * computes five mat‑vec products with the corresponding
          weight matrices (weight layout = [H, dim])
        * applies the optional pair‑mask to `left` / `right`
        * applies sigmoid to the three gate vectors
        * writes three tensors – `left`, `right` and `out_gate`
          – both of shape [B, H, N, N] (FP16).

   This eliminates the huge intermediate tensor
   `proj` = [R, 5·H] used in the reference implementation
   and cuts memory traffic dramatically.

3️⃣ Batched GEMM (cuBLAS) contracts over the sequence dimension
   `k`:
        out[b, h, i, j] = Σₖ left[b, h, i, k]·right[b, h, j, k]
   → tensor [B, H, N, N] (FP16).

4️⃣ Row‑wise mean / rsqrt‑variance of `out` over the hidden dimension H
   (Triton kernel) → μ, rstd per (b,i,j).

5️⃣ Final kernel fuses the second LayerNorm, the output‑gate,
   and the projection to the original `dim`:
        norm   = (out − μ)·rstd ∘ γ + β
        gated  = norm ∘ out_gate
        out    = gated @ W_out.T   (FP32 accumulator)
   → final tensor `[B, N, N, dim]` (FP32).

All kernels are written for Triton 3.3.1 and target an NVIDIA H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr, y_ptr, w_ptr, b_ptr, rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # -------- first pass: mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # -------- second pass: write normalized values ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
# 2️⃣ Fused projection + mask + sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def fused_proj_gate_kernel(
    x_ptr,               # [R, dim]         FP16 (already layer‑normed)
    left_ptr, right_ptr, out_gate_ptr,  # [B, H, N, N] FP16
    mask_ptr,            # [R]              FP16 (optional)
    left_proj_ptr, right_proj_ptr,
    left_gate_ptr, right_gate_ptr,
    out_gate_w_ptr,      # each: [H, dim]   FP16
    R, H, N, dim,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # decode (b,i,k) from the flat index
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    hidden_stride = nn
    base = b * (H * hidden_stride) + i * N + k          # offset for hidden = 0

    # optional pair mask (broadcast over hidden dim)
    if HAS_MASK:
        m_row = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0).to(tl.float16)   # [BLOCK_R]
    else:
        m_row = tl.full([BLOCK_R], 1.0, dtype=tl.float16)                           # [BLOCK_R]

    # ------------------------------------------------------------------
    # loop over hidden‑dimension blocks
    # ------------------------------------------------------------------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)    # [BLOCK_H]
        mask_h = offs_h < H

        # destination offset for the current hidden block
        dst = base[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        # accumulators (fp32) for the five linear maps
        acc_left      = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        acc_right     = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        acc_left_gate = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        acc_right_gate= tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        acc_out_gate  = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)

        # ------------------------------------------------------------------
        # loop over the input feature dimension
        # ------------------------------------------------------------------
        for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
            offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
            mask_d = offs_d < dim

            # ----- load input row block (fp32) -----
            x = tl.load(x_ptr + offs_r[:, None] * dim + offs_d[None, :],
                        mask=mask_r[:, None] & mask_d[None, :],
                        other=0.0).to(tl.float32)               # [BLOCK_R, BLOCK_D]

            # ----- load weight blocks (transposed, fp32) -----
            # left_proj
            w_l = tl.load(left_proj_ptr + offs_d[:, None] + offs_h[None, :] * dim,
                          mask=mask_d[:, None] & mask_h[None, :],
                          other=0.0).to(tl.float32)               # [BLOCK_D, BLOCK_H]
            # right_proj
            w_r = tl.load(right_proj_ptr + offs_d[:, None] + offs_h[None, :] * dim,
                          mask=mask_d[:, None] & mask_h[None, :],
                          other=0.0).to(tl.float32)
            # left_gate
            w_lg = tl.load(left_gate_ptr + offs_d[:, None] + offs_h[None, :] * dim,
                           mask=mask_d[:, None] & mask_h[None, :],
                           other=0.0).to(tl.float32)
            # right_gate
            w_rg = tl.load(right_gate_ptr + offs_d[:, None] + offs_h[None, :] * dim,
                           mask=mask_d[:, None] & mask_h[None, :],
                           other=0.0).to(tl.float32)
            # out_gate
            w_og = tl.load(out_gate_w_ptr + offs_d[:, None] + offs_h[None, :] * dim,
                           mask=mask_d[:, None] & mask_h[None, :],
                           other=0.0).to(tl.float32)

            # ----- accumulate dot products -----
            acc_left      += tl.dot(x, w_l)          # [BLOCK_R, BLOCK_H]
            acc_right     += tl.dot(x, w_r)
            acc_left_gate += tl.dot(x, w_lg)
            acc_right_gate+= tl.dot(x, w_rg)
            acc_out_gate  += tl.dot(x, w_og)

        # ------ post‑processing: sigmoid gates & mask ------
        # sigmoid (fp16 for compact storage)
        left_gate_sig  = (1.0 / (1.0 + tl.exp(-acc_left_gate))).to(tl.float16)
        right_gate_sig = (1.0 / (1.0 + tl.exp(-acc_right_gate))).to(tl.float16)
        out_gate_sig   = (1.0 / (1.0 + tl.exp(-acc_out_gate))).to(tl.float16)

        # broadcast mask over the hidden dimension
        m = m_row[:, None]                               # [BLOCK_R, 1]

        left_val  = (acc_left .to(tl.float16) * left_gate_sig  ) * m
        right_val = (acc_right.to(tl.float16) * right_gate_sig ) * m
        # out_gate is *not* masked
        out_gate_val = out_gate_sig

        # ----- store results -----
        tl.store(left_ptr      + dst, left_val,      mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr     + dst, right_val,     mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr  + dst, out_gate_val,  mask=mask_r[:, None] & mask_h[None, :])

# ----------------------------------------------------------------------
# 3️⃣ Row‑wise mean & rsqrt‑variance of contracted tensor (over H)
# ----------------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr, mu_ptr, rstd_ptr,
    total_rows, H, N,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = nn
    base = b * (H * hidden_stride) + i * N + j

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        ptr = out_ptr + base[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_m, mean,   mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd,   mask=mask_m)

# ----------------------------------------------------------------------
# 4️⃣ Fuse second LayerNorm, out_gate, and final projection
# ----------------------------------------------------------------------
@triton.jit
def final_output_kernel(
    out_ptr,                # [B, H, N, N] FP16   (contracted tensor)
    gate_ptr,               # [B, H, N, N] FP16   (out_gate)
    mu_ptr, rstd_ptr,       # [R] FP32
    ln_w_ptr, ln_b_ptr,     # [H] FP32   (to_out_norm)
    w_out_T_ptr,            # [H, dim]  FP16   (to_out.weight.T)
    out_final_ptr,          # [B, N, N, dim] FP32
    R, H, N, dim,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr,
):
    pid_m = tl.program_id(0)   # rows (R)
    pid_n = tl.program_id(1)   # output‑dim tiles

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)   # [BLOCK_M]
    offs_n = col_start + tl.arange(0, BLOCK_N)   # [BLOCK_N]

    mask_m = offs_m < R
    mask_n = offs_n < dim

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = nn
    base = b * (H * hidden_stride) + i * N + j   # offset for hidden = 0

    # Load per‑row LN statistics
    mu   = tl.load(mu_ptr   + offs_m, mask=mask_m, other=0.0)   # [M]
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)   # [M]

    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        # Load hidden slice of contracted tensor & out_gate
        ptr_out  = out_ptr  + base[:, None] + offs_k[None, :] * hidden_stride
        ptr_gate = gate_ptr + base[:, None] + offs_k[None, :] * hidden_stride

        vals_out = tl.load(ptr_out,
                           mask=mask_m[:, None] & mask_k[None, :],
                           other=0.0).to(tl.float32)   # [M,K]
        vals_gate = tl.load(ptr_gate,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)   # [M,K]

        # LayerNorm over hidden dim
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [K]
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [K]

        diff = (vals_out - mu[:, None]) * rstd[:, None]            # [M,K]
        norm = diff * gamma[None, :] + beta[None, :]               # [M,K]

        term = norm * vals_gate                                   # [M,K]
        term_fp16 = term.to(tl.float16)                           # fp16 for GEMM

        # Load final projection weight slice (H, dim)
        w_ptr = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]   # [K, TILE_N]
        w_tile = tl.load(w_ptr,
                         mask=mask_k[:, None] & mask_n[None, :],
                         other=0.0)                                      # [K, TILE_N]

        # Accumulate (M,K)·(K,N) → (M,N)
        acc += tl.dot(term_fp16, w_tile).to(tl.float32)

    # Store final tensor – rows are flattened (B·N·N)
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
# Overall entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only outgoing TriMul.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : dict of model weights (torch.Tensor)
        - config_dict  : must contain ``dim``, ``hidden_dim`` and ``nomask``

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim        = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask     = bool(cfg.get("nomask", True))
    eps        = 1e-5
    device     = x.device

    B, N, _, _ = x.shape
    R = B * N * N                 # total pairwise tokens
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()          # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection + gates (no large intermediate tensor)
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    # cast weights to fp16 once (they are small)
    left_proj_w   = w["left_proj.weight"].to(torch.float16).contiguous()
    right_proj_w  = w["right_proj.weight"].to(torch.float16).contiguous()
    left_gate_w   = w["left_gate.weight"].to(torch.float16).contiguous()
    right_gate_w  = w["right_gate.weight"].to(torch.float16).contiguous()
    out_gate_w    = w["out_gate.weight"].to(torch.float16).contiguous()

    if not nomask and mask is not None:
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    BLOCK_D = 32
    grid_proj = (triton.cdiv(R, BLOCK_R),)

    fused_proj_gate_kernel[grid_proj](
        x_norm,
        left,
        right,
        out_gate,
        mask_ptr,
        left_proj_w,
        right_proj_w,
        left_gate_w,
        right_gate_w,
        out_gate_w,
        R,
        H,
        N,
        dim,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_norm, left_proj_w, right_proj_w, left_gate_w, right_gate_w, out_gate_w

    # ------------------------------------------------------------------
    # 3️⃣ Batched GEMM over hidden dim (cuBLAS)
    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)                     # FP16
    right_mat = right.view(B * H, N, N)                    # FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                            # [B, H, N, N] FP16

    # ------------------------------------------------------------------
    # 4️⃣ Row‑wise mean & rsqrt‑variance of the contracted tensor
    # ------------------------------------------------------------------
    mu   = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 256
    BLOCK_K_STATS = 128
    grid_stats = (triton.cdiv(R, BLOCK_M_STATS),)

    rowwise_stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm + out_gate + final projection
    # ------------------------------------------------------------------
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim]

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 128   # rows per program
    BLOCK_K_OUT = 32    # hidden‑dim chunk
    BLOCK_N_OUT = 128   # output‑dim tile

    grid_out = (
        triton.cdiv(R, BLOCK_M_OUT),
        triton.cdiv(dim, BLOCK_N_OUT),
    )
    final_output_kernel[grid_out](
        out,
        out_gate,
        mu,
        rstd,
        w["to_out_norm.weight"],   # γ  FP32
        w["to_out_norm.bias"],     # β  FP32
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=8,
    )

    return final
