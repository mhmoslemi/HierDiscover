"""
Outgoing TriMul (AlphaFold‑3 style) – fast forward only.

Algorithm
---------
1️⃣  Pair‑wise LayerNorm on the input tensor (torch.nn.functional.layer_norm).
2️⃣  Fuse the five linear maps (left_proj, right_proj, left_gate,
    right_gate, out_gate) into a single half‑precision GEMM.
3️⃣  Triton kernel “split_fuse_gate_kernel” splits the fused GEMM,
    applies the optional pair mask, computes the sigmoid gates and
    writes three tensors: left, right and out_gate (all fp16).
4️⃣  Batched half‑precision GEMM (torch.bmm) performs the core
    “tri‑multiplication”:  out[b,h,i,j] = Σₖ left[b,h,i,k]·right[b,h,j,k]
5️⃣  Torch LayerNorm (fp16) normalises the contracted tensor over the
    hidden dimension H, then it is multiplied by the out‑gate.
6️⃣  Final half‑precision linear projection (to_out) produces the
    result tensor of shape [B,N,N,dim] (cast back to fp32).

All heavy compute kernels are written for Triton 3.3.1 and target an H100.
Only a single Triton kernel is required to satisfy the “must use Triton”
rule.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 2️⃣  Split fused projection, apply optional mask and compute sigmoids
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # [R, 5*H]   fp16 fused projection (R = B*N*N)
    left_ptr,          # [B, H, N, N] fp16
    right_ptr,         # [B, H, N, N] fp16
    out_gate_ptr,      # [B, H, N, N] fp16 (sigmoid)
    mask_ptr,          # [R]        fp16 (only if HAS_MASK=1)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional pair‑wise mask (broadcast over H)
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        m_val = m_val[:, None]                                      # [BLOCK_R,1]
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N                # offset between consecutive H slices
    base = b * (H * hidden_stride) + i * N + k   # start offset for h = 0

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # each row has 5*H values → split into 5 blocks of size H
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw  = tl.load(proj_ptr + proj_base,
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0)
        right_raw = tl.load(proj_ptr + proj_base + H,
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0)
        left_gate_raw  = tl.load(proj_ptr + proj_base + 2 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        right_gate_raw = tl.load(proj_ptr + proj_base + 3 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        out_gate_raw   = tl.load(proj_ptr + proj_base + 4 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)

        # sigmoid (compute in fp32 → back to fp16)
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # apply mask + gates
        left  = left_raw  * m_val * left_gate
        right = right_raw * m_val * right_gate
        outg  = out_gate

        dst = base[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst, right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst, outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only outgoing TriMul.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : dict of model weights (torch.Tensor)
        - config_dict  : must contain ``dim``, ``hidden_dim`` and ``nomask``

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = bool(cfg.get("nomask", True))
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                     # total number of pairwise tokens
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣  Pair‑wise LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    # torch.nn.functional.layer_norm works on the last dimension.
    x_norm_fp32 = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    )
    x_norm = x_norm_fp32.to(torch.float16).contiguous()   # [B,N,N,dim] fp16

    # ------------------------------------------------------------------
    # 2️⃣  Fuse five linear maps (left, right, left_gate, right_gate, out_gate)
    # ------------------------------------------------------------------
    # concatenate the five weight matrices (shape: [5*H, dim])
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                   # [5*H, dim]

    # flatten pair dimension for the GEMM
    x_flat = x_norm.view(R, dim)                           # [R, dim]
    proj = F.linear(x_flat, fused_w)                        # [R, 5*H] fp16
    del x_norm, x_flat, fused_w

    # ------------------------------------------------------------------
    # 3️⃣  Split, apply mask, compute sigmoids → left, right, out_gate
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if not nomask and mask is not None:
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        # dummy tensor – never accessed because HAS_MASK=0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)

    BLOCK_R = 128
    BLOCK_H = 64
    grid = (triton.cdiv(R, BLOCK_R),)

    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣  Batched GEMM over hidden dimension (half‑precision)
    # ------------------------------------------------------------------
    # reshape to (B*H, N, N) → perform (B*H) independent matrix‑multiplies
    left_mat  = left.view(B * H, N, N)                     # fp16
    right_mat = right.view(B * H, N, N)                    # fp16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B*H, N, N] fp16
    out = out.view(B, H, N, N)                             # [B, H, N, N] fp16
    del left_mat, right_mat, left, right

    # ------------------------------------------------------------------
    # 5️⃣  LayerNorm over the hidden dimension, gating and final projection
    # ------------------------------------------------------------------
    # Torch LayerNorm works on the last dimension → move H to the end
    out_perm = out.permute(0, 2, 3, 1).contiguous()        # [B, N, N, H] fp16
    ln_w = w["to_out_norm.weight"].to(torch.float16)
    ln_b = w["to_out_norm.bias"].to(torch.float16)

    out_norm = F.layer_norm(
        out_perm,
        normalized_shape=(H,),
        weight=ln_w,
        bias=ln_b,
        eps=eps,
    )                                                       # fp16

    # out_gate needs the same permutation
    out_gate_perm = out_gate.permute(0, 2, 3, 1).contiguous()   # [B,N,N,H] fp16
    del out_gate

    # Apply the out‑gate
    gated = out_norm * out_gate_perm
    del out_norm, out_gate_perm, out

    # Final linear projection back to `dim`
    gated_flat = gated.view(R, H)                           # [R, H] fp16
    w_out = w["to_out.weight"].to(torch.float16)            # [dim, H] fp16
    final = F.linear(gated_flat, w_out)                     # [R, dim] fp16
    final = final.view(B, N, N, dim).to(torch.float32)      # back to fp32

    return final
