"""
Outgoing TriMul (AlphaFold‑3) – fast forward‑only Triton implementation.

Algorithm
---------
1️⃣  Input LayerNorm (FP32 → FP16) using torch.nn.functional.layer_norm.
2️⃣  Fuse the five linear maps (left, right, left_gate, right_gate, out_gate)
    into a single half‑precision GEMM (cuBLAS) → tensor [R, 5·H] where
    R = B·N·N and H = hidden_dim.
3️⃣  Triton kernel – split the fused projection, apply the optional pair mask,
    compute sigmoid gates and write three tensors `left`, `right` and `out_gate`
    in layout [B, H, N, N] (FP16).
4️⃣  Batched matrix‑multiply (cuBLAS) computes the core einsum:
       out[b, h, i, j] = Σₖ left[b, h, i, k]·right[b, h, j, k].
5️⃣  Second LayerNorm, gating and final linear projection are performed with
    torch kernels (layer_norm + element‑wise + linear) entirely in half‑precision
    with FP32 accumulation, which is much faster than a custom Triton kernel.
6️⃣  The result is reshaped to [B, N, N, dim] (float32).

Only the split‑&‑gate step is written in Triton; the remaining heavy work
leverages highly‑optimized cuBLAS / torch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,      # [R, 5·H]   FP16 fused projection
    left_ptr,      # [B, H, N, N] FP16
    right_ptr,     # [B, H, N, N] FP16
    out_gate_ptr,  # [B, H, N, N] FP16
    mask_ptr,      # [R]        FP16 (optional)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)           # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index  r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask (broadcast across hidden dimension)
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        m_val = m_val[:, None]                                       # [BLOCK_R, 1]
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N                       # stride between hidden planes
    base = b * (H * hidden_stride) + i * N + k # offset for hidden‑plane 0

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # column offset inside the fused projection (5·H columns)
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw = tl.load(
            proj_ptr + proj_base,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_raw = tl.load(
            proj_ptr + proj_base + H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        left_gate_raw = tl.load(
            proj_ptr + proj_base + 2 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        right_gate_raw = tl.load(
            proj_ptr + proj_base + 3 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )
        out_gate_raw = tl.load(
            proj_ptr + proj_base + 4 * H,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0,
        )

        # sigmoid in FP32 for stability, then cast back to FP16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # apply mask & gates
        left  = left_raw  * m_val * left_gate
        right = right_raw * m_val * right_gate
        outg  = out_gate

        dst = base[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr + dst, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst, right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst, outg,
                 mask=mask_r[:, None] & mask_h[None, :])


def custom_kernel(data):
    """
    Forward‑only outgoing TriMul.

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping with keys
                  "norm.weight", "norm.bias",
                  "left_proj.weight", "right_proj.weight",
                  "left_gate.weight", "right_gate.weight", "out_gate.weight",
                  "to_out_norm.weight", "to_out_norm.bias",
                  "to_out.weight"
              (all float32, already on the correct device)
            - config_dict must contain:
                  "dim" (int), "hidden_dim" (int), "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = bool(cfg.get("nomask", True))
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                     # total pairwise tokens
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    # flatten to [R, dim] for a single layer‑norm call
    x_flat = x.reshape(R, dim).contiguous()                     # FP32
    x_norm_fp32 = F.layer_norm(
        x_flat,
        normalized_shape=(dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    )                                                            # FP32
    x_norm = x_norm_fp32.to(torch.float16)                       # FP16

    # ------------------------------------------------------------------
    # 2️⃣ Fuse five linear maps (left, right, left_gate, right_gate, out_gate)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                 # [5·H, dim]   FP16
    proj = F.linear(x_norm, fused_w)    # (R, 5·H)    FP16
    del fused_w, x_norm, x_flat, x_norm_fp32

    # ------------------------------------------------------------------
    # 3️⃣ Split, mask, sigmoid, store left/right/out_gate
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Core einsum via batched GEMM (FP16)
    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)                # [B·H, N, N]  FP16
    right_mat = right.view(B * H, N, N)               # [B·H, N, N]  FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                        # [B, H, N, N] FP16

    # free memory we no longer need
    del left_mat, right_mat, left, right

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm + out_gate + final projection (all torch)
    # ------------------------------------------------------------------
    # reshape tensors to [B, N, N, H] for per‑token LayerNorm over H
    out_perm      = out.permute(0, 2, 3, 1).contiguous()          # [B, N, N, H] FP16
    out_gate_perm = out_gate.permute(0, 2, 3, 1).contiguous()    # [B, N, N, H] FP16

    # LayerNorm in FP32 (statistics in FP32, result kept FP32)
    out_perm_fp32 = out_perm.to(torch.float32)
    out_norm_fp32 = F.layer_norm(
        out_perm_fp32,
        normalized_shape=(H,),
        weight=w["to_out_norm.weight"],
        bias=w["to_out_norm.bias"],
        eps=eps,
    )                                                             # FP32

    # Gate (already FP16), promote to FP32 for the multiplication
    out_gate_fp32 = out_gate_perm.to(torch.float32)

    term_fp32 = out_norm_fp32 * out_gate_fp32               # FP32
    term_fp16 = term_fp32.to(torch.float16)                 # FP16 for GEMM

    # Final linear projection (half‑precision GEMM, accumulation in FP32)
    w_out_fp16 = w["to_out.weight"].to(torch.float16)       # [dim, H] FP16
    final_fp16 = F.linear(term_fp16.view(R, H), w_out_fp16)  # [R, dim] FP16
    final = final_fp16.to(torch.float32).view(B, N, N, dim)  # FP32

    return final
```
