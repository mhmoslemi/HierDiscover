"""
Outgoing TriMul (AlphaFold‑3) – fast forward‑only implementation.

Algorithm
---------
1️⃣  Row‑wise LayerNorm (FP32 → FP16) on the flattened pairwise tensor.
2️⃣  Fuse the five linear maps (left, right, left_gate, right_gate, out_gate)
    into a single FP16 GEMM (`torch.nn.functional.linear`).
3️⃣  Reshape & split the fused projection, apply optional mask and sigmoid gates
    using vectorised torch ops (no Triton kernel needed here).
4️⃣  Batched GEMM (`torch.bmm`) computes the core einsum:
       out[b, h, i, j] = Σₖ left[b, h, i, k]·right[b, h, j, k]   → [B, H, N, N] (FP16)
5️⃣  LayerNorm over the hidden dimension, multiply by `out_gate`,
    and final projection back to `dim` using a fused linear layer.
    (All steps after 1️⃣ use highly‑optimised torch kernels.)

Only the first LayerNorm is written in Triton – the rest relies on
torch's cuBLAS/ATen kernels, which provides the best performance on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,      # [R, D]   FP32 input (contiguous)
    y_ptr,      # [R, D]   FP16 output
    w_ptr,      # [D]      γ (FP32)
    b_ptr,      # [D]      β (FP32)
    rows,       # int      R = B·N·N
    dim,        # int      D  (input channel dim)
    eps,        # float    epsilon
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # dim‑chunk per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ------------------------------------------------------------------
    # First pass: compute mean & variance (FP32)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        # load a tile [BLOCK_M, BLOCK_D]
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)          # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)   # [BLOCK_M]

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------------------------------------------------------
    # Second pass: write normalized values (FP16)
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Entrypoint
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only outgoing TriMul.

    Args:
        data (tuple):
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict  : dict with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config_dict   : {"dim": int, "hidden_dim": int,
                               "nomask": bool (optional)}
    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = bool(cfg.get("nomask", True))
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N          # number of pairwise tokens
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm on the flattened input (FP32 → FP16)
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                     # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid = ( (R + BLOCK_M - 1) // BLOCK_M, )
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat  # free memory

    # ------------------------------------------------------------------
    # 2️⃣ Fuse five linear maps into a single GEMM (FP16)
    # ------------------------------------------------------------------
    # concatenate weights: each is (H, dim) → result (5*H, dim)
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # [5*H, dim]

    proj = F.linear(x_norm, fused_w)        # (R, 5*H) FP16
    del fused_w, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ Split, apply optional mask and sigmoid gates (pure torch)
    # ------------------------------------------------------------------
    # reshape to [B, N, N, 5, H]
    proj = proj.view(B, N, N, 5, H)

    left_raw      = proj[..., 0, :]   # [B,N,N,H]
    right_raw     = proj[..., 1, :]   # [B,N,N,H]
    left_gate_raw = proj[..., 2, :]   # [B,N,N,H]
    right_gate_raw= proj[..., 3, :]   # [B,N,N,H]
    out_gate      = torch.sigmoid(proj[..., 4, :])   # [B,N,N,H]

    # optional mask (float16 0/1)
    if (mask is not None) and (not nomask):
        mask_fp = mask.to(torch.float16).unsqueeze(-1)   # [B,N,N,1]
        left_raw  = left_raw * mask_fp
        right_raw = right_raw * mask_fp

    left_gate  = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)

    left  = left_raw  * left_gate   # [B,N,N,H]
    right = right_raw * right_gate  # [B,N,N,H]
    del left_raw, right_raw, left_gate_raw, right_gate_raw, proj

    # ------------------------------------------------------------------
    # 4️⃣ Core einsum via batched GEMM (cuBLAS)
    # ------------------------------------------------------------------
    # bring hidden dimension to the second axis: [B, H, N, N]
    left  = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()

    BH = B * H
    left_mat  = left.view(BH, N, N)                       # FP16
    right_mat = right.view(BH, N, N)                      # FP16

    # out = Σₖ left[...,i,k] * right[...,j,k]   (B*H, N, N)
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))  # FP16
    out = out.view(B, H, N, N)                           # [B, H, N, N]

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm + out_gate + final projection (pure torch)
    # ------------------------------------------------------------------
    # bring hidden dim to tail for LayerNorm: [B, N, N, H]
    out_perm = out.permute(0, 2, 3, 1).contiguous()       # [B,N,N,H]

    # LayerNorm parameters (cast to fp16 for compatibility)
    ln_w_fp16 = w["to_out_norm.weight"].to(torch.float16)
    ln_b_fp16 = w["to_out_norm.bias"].to(torch.float16)

    out_norm = F.layer_norm(out_perm, (H,),
                            weight=ln_w_fp16,
                            bias=ln_b_fp16,
                            eps=eps)                # FP16

    # multiply by out_gate (same layout)
    out_gate = out_gate.to(torch.float16)                 # ensure same dtype
    out_gated = out_norm * out_gate                        # FP16

    # final linear projection: H → dim
    to_out_w = w["to_out.weight"].to(torch.float16)       # [dim, H]
    final = F.linear(out_gated.view(-1, H), to_out_w)      # [B*N*N, dim] FP16
    final = final.view(B, N, N, dim).to(torch.float32)    # cast to FP32

    return final
