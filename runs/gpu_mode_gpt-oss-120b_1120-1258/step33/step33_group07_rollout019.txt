"""
Outgoing TriMul (AlphaFold‑3 style) – forward only.

Optimisations
-------------
1️⃣  Row‑wise LayerNorm (FP32 → FP16) – one Triton kernel.
2️⃣  Fuse the five linear maps (left, right, left_gate, right_gate, out_gate)
    into a *single* GEMM, then split the result and apply the optional mask
    and sigmoid gates (one Triton kernel, no hidden‑dim loop).
3️⃣  Batched GEMM (cuBLAS) for the core “tri‑multiplication”.
4️⃣  Fuse the second LayerNorm with the out‑gate in a single Triton kernel;
    the kernel computes mean / variance, normalises, multiplies by the gate
    and writes a packed hidden matrix `H × dim` (`[R, H]` where `R = B·N·N`).
5️⃣  Final linear projection is performed by a highly‑optimised cuBLAS GEMM
    (`[R, H] @ [H, dim] → [R, dim]`).  The result is cast back to FP32.

The implementation reduces the number of kernel launches (3 → 2) and
eliminates an intermediate `μ/σ` tensor, while keeping all heavy
computations in FP16 so the H100 can utilise Tensor‑cores.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,   # [R, dim] FP32 input
    y_ptr,   # [R, dim] FP16 output
    w_ptr,   # [dim]    γ (FP32)
    b_ptr,   # [dim]    β (FP32)
    rows,    # scalar R
    dim,     # scalar dim
    eps,     # scalar eps
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)            # [BLOCK_M]
    mask_m = offs_m < rows

    # ------------ first pass: mean & var -------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)  # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------ second pass: write norm ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])

    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)               # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index  r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional pair mask (broadcast over hidden dim)
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        m_val = m_val[:, None]                                      # [BLOCK_R,1]
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + k   # offset for hidden = 0

    # ------------------------------------------------------------------
    # Load the five blocks (left, right, left_gate, right_gate, out_gate)
    # ------------------------------------------------------------------
    offs_h = tl.arange(0, BLOCK_H)                    # [BLOCK_H]
    mask_h = offs_h < H

    proj_base = offs_r[:, None] * (5 * H)              # [BLOCK_R,1]

    # left projection
    left_raw = tl.load(
        proj_ptr + proj_base + offs_h[None, :],
        mask=mask_r[:, None] & mask_h[None, :],
        other=0.0,
    )
    # right projection
    right_raw = tl.load(
        proj_ptr + proj_base + H + offs_h[None, :],
        mask=mask_r[:, None] & mask_h[None, :],
        other=0.0,
    )
    # left gate (pre‑sigmoid)
    left_gate_raw = tl.load(
        proj_ptr + proj_base + 2 * H + offs_h[None, :],
        mask=mask_r[:, None] & mask_h[None, :],
        other=0.0,
    )
    # right gate (pre‑sigmoid)
    right_gate_raw = tl.load(
        proj_ptr + proj_base + 3 * H + offs_h[None, :],
        mask=mask_r[:, None] & mask_h[None, :],
        other=0.0,
    )
    # out gate (pre‑sigmoid)
    out_gate_raw = tl.load(
        proj_ptr + proj_base + 4 * H + offs_h[None, :],
        mask=mask_r[:, None] & mask_h[None, :],
        other=0.0,
    )

    # sigmoid – compute in FP32 then cast back to FP16
    left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
    right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
    out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

    # apply mask then gates
    left  = left_raw  * m_val * left_gate
    right = right_raw * m_val * right_gate
    outg  = out_gate                                 # shape [BLOCK_R, BLOCK_H]

    dst = base[:, None] + offs_h[None, :] * hidden_stride   # [BLOCK_R, BLOCK_H]

    tl.store(left_ptr + dst,
             left,
             mask=mask_r[:, None] & mask_h[None, :])
    tl.store(right_ptr + dst,
             right,
             mask=mask_r[:, None] & mask_h[None, :])
    tl.store(out_gate_ptr + dst,
             outg,
             mask=mask_r[:, None] & mask_h[None, :])

# ----------------------------------------------------------------------
# 3️⃣ Fuse second LayerNorm, out_gate → packed hidden matrix
# ----------------------------------------------------------------------
@triton.jit
def ln_gate_kernel(
    out_ptr,          # [B, H, N, N] FP16
    gate_ptr,         # [B, H, N, N] FP16 (out_gate)
    hidden_out_ptr,   # [R, H]       FP16  – packed, ready for final GEMM
    ln_w_ptr,         # [H] FP32     (to_out_norm.weight)
    ln_b_ptr,         # [H] FP32     (to_out_norm.bias)
    total_rows, H, N, eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)               # [BLOCK_M]
    mask_m = offs_m < total_rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base = b * (H * hidden_stride) + i * N + j               # offset for hidden = 0

    # ----------- first pass : mean & var -------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)               # [BLOCK_K]
        mask_k = offs_k < H

        ptr = out_ptr + base[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)               # [M, K]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ----------- second pass : LN + out_gate ----------
    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)               # [BLOCK_K]
        mask_k = offs_k < H

        # load raw values
        ptr_out  = out_ptr  + base[:, None] + offs_k[None, :] * hidden_stride
        ptr_gate = gate_ptr + base[:, None] + offs_k[None, :] * hidden_stride

        vals_out  = tl.load(ptr_out,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)          # [M,K]
        vals_gate = tl.load(ptr_gate,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)          # [M,K]

        # LayerNorm parameters (γ,β) – FP32
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [K]
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [K]

        # Normalise → (x - μ) * rstd
        norm = (vals_out - mean[:, None]) * rstd[:, None]           # [M,K]
        # Apply γ,β  →  γ·norm + β
        norm = norm * gamma[None, :] + beta[None, :]                # [M,K]

        # Multiply by out‑gate
        gated = norm * vals_gate                                    # [M,K]

        # Store into packed hidden matrix [R, H]
        dst_ptr = hidden_out_ptr + offs_m[:, None] * H + offs_k[None, :]
        tl.store(dst_ptr,
                 gated.to(tl.float16),
                 mask=mask_m[:, None] & mask_k[None, :])

# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only outgoing TriMul.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : dict of model weights (torch.Tensor)
        - config_dict  : must contain ``dim``, ``hidden_dim`` and ``nomask``

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim         = cfg["dim"]
    hidden_dim  = cfg["hidden_dim"]
    nomask      = bool(cfg.get("nomask", True))
    eps         = 1e-5
    device      = x.device

    B, N, _, _ = x.shape
    R = B * N * N                     # total number of pairwise tokens
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()               # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16,
                         device=device)                # [R, dim] FP16

    BLOCK_M = 256
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # ------------------------------------------------------------------
    # 2️⃣ Fuse five linear maps (left, right, left_gate, right_gate, out_gate)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                   # [5·H, dim]

    # (R, 5·H) FP16
    proj = F.linear(x_norm, fused_w)
    del fused_w, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ Split, apply mask, compute sigmoids → left, right, out_gate
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_flat
    else:
        HAS_MASK = 0
        # dummy tensor – never dereferenced when HAS_MASK == 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)

    BLOCK_R = 128          # rows per program
    BLOCK_H = hidden_dim   # we load the whole hidden dimension at once
    grid_split = (triton.cdiv(R, BLOCK_R),)

    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM over hidden dim (cuBLAS) : left × rightᵀ
    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)                     # FP16
    right_mat = right.view(B * H, N, N)                    # FP16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                            # [B, H, N, N] FP16

    # ------------------------------------------------------------------
    # 5️⃣ Fuse second LayerNorm and out_gate → packed hidden matrix
    # ------------------------------------------------------------------
    hidden_out = torch.empty(R, H, dtype=torch.float16, device=device)  # [R, H]

    BLOCK_M_STATS = 128    # rows per program for the LN+gate kernel
    BLOCK_K_STATS = 32     # hidden‑dim tile (128 will be processed in 4 iterations)
    grid_ln_gate = (triton.cdiv(R, BLOCK_M_STATS),)

    ln_gate_kernel[grid_ln_gate](
        out,
        out_gate,
        hidden_out,
        w["to_out_norm.weight"],   # γ  FP32
        w["to_out_norm.bias"],     # β  FP32
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=8,
    )
    # we no longer need the raw `out` nor `out_gate`
    del out, out_gate

    # ------------------------------------------------------------------
    # 6️⃣ Final linear projection (cuBLAS)  →  [R, dim] FP32
    # ------------------------------------------------------------------
    # weight: [dim, H] → transpose to [H, dim] (FP16)
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim]

    # (R, H) @ (H, dim)  → (R, dim) (still FP16)
    final_fp16 = torch.matmul(hidden_out, w_out_T)                     # FP16
    final = final_fp16.to(torch.float32)                               # FP32

    # reshape to [B, N, N, dim]
    out_tensor = final.view(B, N, N, dim)

    return out_tensor
