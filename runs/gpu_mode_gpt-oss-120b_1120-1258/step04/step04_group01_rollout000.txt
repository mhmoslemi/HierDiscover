"""
Outgoing TriMul (AlphaFold‑3) – Triton‑accelerated forward pass
================================================================
The function `custom_kernel` implements the *outgoing* TriMul operator
used in AlphaFold‑3.  The algorithm follows the reference PyTorch
implementation but replaces the two LayerNorms with custom Triton
kernels and keeps most heavy work in FP16 (Tensor‑Core friendly)
to hit the sub‑millisecond target on an H100.

Steps
-----
1. **Row‑wise LayerNorm** on the input 4‑D tensor (B,N,N,C).  The
   Triton kernel works on FP32 input and writes FP16 results to cut
   memory traffic.

2. **Fused linear projection** – the five dense layers
   (left_proj, right_proj, left_gate, right_gate, out_gate) are
   concatenated into a single weight matrix of shape
   (5·H, C) (H = hidden_dim).  A single half‑precision GEMM
   (`F.linear`) produces all five tensors at once.

3. **Mask & gating** – the optional pair‑wise mask is applied only to
   the two projections; the three sigmoid gates are computed in FP16.

4. **Triangular multiplicative update** – the core O(N³·H) operation
   is expressed as an einsum:
   `out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]`.
   With tensors laid out as (B,N,N,H) this avoids the costly
   `permute‑contiguous` copy required for a batched `matmul`.

5. **Second LayerNorm** – identical Triton kernel as in step 1.

6. **Output gate & final projection** – element‑wise gating followed
   by a half‑precision linear layer back to the model dimension.
   The result is cast to float32 for downstream use.

All heavy kernels run on FP16, exploiting Tensor‑Cores, while the
LayerNorms are fully fused in Triton to minimise memory traffic.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton LayerNorm (row‑wise, FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,            # const   *  [total_rows, dim]   (fp32 or fp16)
    y_ptr,            # mutable *  [total_rows, dim]   (fp16)
    weight_ptr,       # const   *  [dim]               (fp32)
    bias_ptr,         # const   *  [dim]               (fp32)
    total_rows,       # int (runtime)
    dim: tl.constexpr,    # compile‑time
    eps: tl.constexpr,    # compile‑time
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """Row‑wise layer‑norm.
    Input and output are stored row‑major.  The kernel computes per‑row
    mean/variance in the first pass and writes the normalized result
    (γ·x̂+β) in FP16 in the second pass."""
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # -----------------------------------------------------------------
    # 1st pass – compute mean & variance per row
    # -----------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        # Load as FP16 when possible, then promote to FP32
        val = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0)
        val_f32 = val.to(tl.float32)

        sum_val += tl.sum(val_f32, axis=1)
        sum_sq  += tl.sum(val_f32 * val_f32, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)               # 1/√(var+eps)

    # -----------------------------------------------------------------
    # 2nd pass – write normalized value * gamma + beta (store FP16)
    # -----------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        val = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0)
        val_f32 = val.to(tl.float32)

        norm = (val_f32 - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]       # FP32

        out_fp16 = out.to(tl.float16)
        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Outgoing TriMul forward pass
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32, on the same CUDA device).
    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]                         # model dimension (e.g. 128 or 384)
    hidden_dim = config["hidden_dim"]           # hidden dimension (always 128 in tests)
    eps = 1e-5
    device = input_tensor.device

    # --------------------------------------------------------------
    # 1) First LayerNorm (row‑wise) – output in FP16
    # --------------------------------------------------------------
    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N                      # R = B·N·N

    # flatten to 2‑D [R, dim]; ensure contiguous layout
    x_flat = input_tensor.view(total_rows, dim).contiguous()
    x_norm_flat = torch.empty(total_rows, dim,
                              dtype=torch.float16, device=device)

    BLOCK_M = 256
    BLOCK_D = 64                 # works for dim up to 384 (64*6 = 384)
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,                     # input pointer (FP32)
        x_norm_flat,                # output pointer (FP16)
        weights["norm.weight"],     # gamma (float32)
        weights["norm.bias"],       # beta  (float32)
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # --------------------------------------------------------------
    # 2) Fuse 5 linear layers (proj + 3 gates) into a single GEMM
    # --------------------------------------------------------------
    # Concatenate weight matrices: shape [5*H, dim]
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],   # (H, dim)
            weights["right_proj.weight"],  # (H, dim)
            weights["left_gate.weight"],   # (H, dim)
            weights["right_gate.weight"],  # (H, dim)
            weights["out_gate.weight"],    # (H, dim)
        ],
        dim=0,
    ).to(torch.float16)                     # FP16 for Tensor‑Core speed

    # Single half‑precision linear layer: [R, dim] @ [dim, 5*H]^T -> [R, 5*H]
    proj_out = F.linear(x_norm_flat, w_concat)   # shape (R, 5*H)

    H = hidden_dim
    left_proj   = proj_out[:, 0:H]                     # (R, H)
    right_proj  = proj_out[:, H:2*H]
    left_gate_p = proj_out[:, 2*H:3*H]
    right_gate_p= proj_out[:, 3*H:4*H]
    out_gate_p  = proj_out[:, 4*H:5*H]

    # --------------------------------------------------------------
    # 3) Apply sigmoid gates (FP16) and optional mask
    # --------------------------------------------------------------
    left_gate   = torch.sigmoid(left_gate_p)   # (R, H)
    right_gate  = torch.sigmoid(right_gate_p)
    out_gate    = torch.sigmoid(out_gate_p)   # (R, H)

    if not config.get("nomask", True):
        # mask shape [B, N, N] -> broadcast to [R, 1]
        mask_h = mask_tensor.to(torch.float16).view(total_rows, 1)
        left_proj  = left_proj * mask_h
        right_proj = right_proj * mask_h

    # Gated projections
    left  = left_proj * left_gate          # (R, H)
    right = right_proj * right_gate        # (R, H)

    # Reshape back to 4‑D tensors
    left  = left.view(B, N, N, H)          # (B, N, N, H)
    right = right.view(B, N, N, H)
    out_gate = out_gate.view(B, N, N, H)   # (B, N, N, H)

    # --------------------------------------------------------------
    # 4) Triangular multiplicative update (einsum on the native layout)
    #    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    # --------------------------------------------------------------
    # This avoids the expensive permute+contiguous that a batched matmul
    # would require.
    out = torch.einsum('b i k h, b j k h -> b i j h', left, right)   # (B,N,N,H)

    # --------------------------------------------------------------
    # 5) Second LayerNorm over the hidden dimension H
    # --------------------------------------------------------------
    out_flat = out.view(total_rows, H).contiguous()
    out_norm_flat = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm_flat,
        weights["to_out_norm.weight"],   # gamma (float32)
        weights["to_out_norm.bias"],     # beta  (float32)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    out_norm = out_norm_flat.view(B, N, N, H)   # (B,N,N,H)

    # --------------------------------------------------------------
    # 6) Apply output gate and final projection back to `dim`
    # --------------------------------------------------------------
    out_norm = out_norm * out_gate               # element‑wise FP16

    out_norm_flat2 = out_norm.view(total_rows, H)   # (R, H)

    final_half = F.linear(
        out_norm_flat2,
        weights["to_out.weight"].to(torch.float16)   # (dim, H)
    )                                                # (R, dim)

    # Cast back to FP32 (the rest of the model expects FP32)
    final = final_half.view(B, N, N, dim).to(torch.float32)

    return final
