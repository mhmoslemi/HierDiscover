"""
Highly‑optimized outgoing TriMul (AlphaFold‑3) forward pass.

Algorithm
---------
1️⃣ Row‑wise LayerNorm (FP32 → FP16) on the input tensor.
2️⃣ Fuse the five linear layers (2 projections + 3 gates) into a single
   half‑precision GEMM (shape [R, dim] × [5·H, dim]ᵀ → [R, 5·H]).
3️⃣ Apply the optional mask to the two projected tensors and the three
   sigmoid gates.
4️⃣ Triangular multiplicative update:
   For every head `h` we need   Cₕ = Aₕ @ Bₕᵀ   where
   Aₕ, Bₕ ∈ ℝ^{N×N} (FP16).  This is done with a custom Triton
   kernel that performs a batched GEMM over the (B·H) heads
   (no extra copies, FP32 accumulation, FP16 output).
5️⃣ Second LayerNorm over the hidden dimension (FP32 → FP16).
6️⃣ Multiply by the output gate, apply the final linear projection and
   cast the result back to FP32.

All heavy per‑row normalisations are performed in a custom Triton kernel,
the N³ step is a fused batched GEMM also written in Triton, and the
remaining linear work uses half‑precision PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,            # const   * [total_rows, dim]   (float32 or fp16)
    y_ptr,            # mutable * [total_rows, dim]   (fp16 output)
    w_ptr,            # const   * [dim]               (float32)
    b_ptr,            # const   * [dim]               (float32)
    total_rows,       # int  (runtime)
    dim,              # tl.constexpr
    eps,              # tl.constexpr
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """Row‑wise LayerNorm with FP32 statistics, FP16 output."""
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # 1️⃣ compute mean / variance per row
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------------------------------------------------------
    # 2️⃣ write normalized output (γ·(x−μ)·rstd + β) as FP16
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]                 # fp32

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Triton kernel: batched A @ Bᵀ for many heads (TriMul core)
# ----------------------------------------------------------------------
@triton.jit
def trimul_kernel(
    A_ptr,            # const * [BH, N, N]   (fp16)
    B_ptr,            # const * [BH, N, N]   (fp16)
    C_ptr,            # mutable * [BH, N, N] (fp16)
    BH,               # int (B * hidden_dim)
    N,                # int (sequence length)
    stride_batch,     # int (stride between different heads)
    stride_i,         # int (row stride = N)
    stride_k,         # int (col stride = 1)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """C[b,i,j] = Σₖ A[b,i,k] * B[b,j,k]   (batched GEMM with Bᵀ)"""
    pid_b = tl.program_id(2)                     # which head (B·H)
    pid_m = tl.program_id(0)                     # row tile
    pid_n = tl.program_id(1)                     # col tile

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # A block:  (M, K)
        a_ptrs = (A_ptr
                  + pid_b * stride_batch
                  + offs_m[:, None] * stride_i
                  + offs_k[None, :] * stride_k)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # Bᵀ block: (K, N)  – we load B transposed by swapping k & n in the address
        b_ptrs = (B_ptr
                  + pid_b * stride_batch
                  + offs_k[:, None] * stride_i   # k as “row”
                  + offs_n[None, :] * stride_k)  # n as “col”
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)

        acc += tl.dot(a, b)                       # (M,K) @ (K,N) → (M,N)

    # write result
    c_ptrs = (C_ptr
              + pid_b * stride_batch
              + offs_m[:, None] * stride_i
              + offs_n[None, :] * stride_k)
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [B, N, N, dim] (float32, on the same CUDA device).
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    device = input_tensor.device

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N                     # rows for row‑wise LN

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → fp16
    # --------------------------------------------------------------
    x_flat = input_tensor.view(total_rows, dim).contiguous()
    x_norm_flat = torch.empty(total_rows, dim,
                              dtype=torch.float16,
                              device=device)

    BLOCK_M = 256
    BLOCK_D = 32
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm_flat,
        weights["norm.weight"],   # γ (fp32)
        weights["norm.bias"],     # β (fp32)
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # --------------------------------------------------------------
    # 2️⃣ Fuse five linear layers (2 proj + 3 gates) into one GEMM
    # --------------------------------------------------------------
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)          # [5·H, dim]
    proj_concat = F.linear(x_norm_flat, w_concat)   # [R, 5·H] (fp16)

    H = hidden_dim
    left_proj   = proj_concat[:, 0:H]                       # [R, H]
    right_proj  = proj_concat[:, H:2*H]                     # [R, H]
    left_gate_p = proj_concat[:, 2*H:3*H]                   # [R, H]
    right_gate_p = proj_concat[:, 3*H:4*H]                  # [R, H]
    out_gate_p  = proj_concat[:, 4*H:5*H]                   # [R, H]

    # sigmoid gates (still fp16)
    left_gate   = torch.sigmoid(left_gate_p)
    right_gate  = torch.sigmoid(right_gate_p)
    out_gate    = torch.sigmoid(out_gate_p)

    # --------------------------------------------------------------
    # 3️⃣ Optional mask (applied only to the projected tensors)
    # --------------------------------------------------------------
    if not config.get("nomask", True):
        mask_f = mask_tensor.to(torch.float16).reshape(total_rows, 1)
        left_proj  = left_proj * mask_f
        right_proj = right_proj * mask_f

    # element‑wise gating
    left  = left_proj * left_gate
    right = right_proj * right_gate

    # --------------------------------------------------------------
    # 4️⃣ Triangular multiplicative update (batched GEMM) via Triton
    # --------------------------------------------------------------
    # Reshape to [B, N, N, H] → permute → [B, H, N, N] → contiguous
    left = left.view(B, N, N, H)
    right = right.view(B, N, N, H)
    out_gate = out_gate.view(B, N, N, H)

    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    BH = B * H
    # reshape to [BH, N, N] (contiguous memory)
    left_mat = left_perm.view(BH, N, N)
    right_mat = right_perm.view(BH, N, N)

    out_mat = torch.empty_like(left_mat)               # fp16 output buffer

    # strides for the contiguous [BH, N, N] tensors
    stride_batch = left_mat.stride(0)   # N * N
    stride_i = left_mat.stride(1)       # N
    stride_k = left_mat.stride(2)       # 1

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (
        triton.cdiv(N, BLOCK_M),          # rows
        triton.cdiv(N, BLOCK_N),          # cols
        BH,                               # batch / head dimension
    )
    trimul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        BH,
        N,
        stride_batch,
        stride_i,
        stride_k,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm over hidden dimension (fp16)
    # --------------------------------------------------------------
    out_flat = out.view(total_rows, H).contiguous()
    out_norm_flat = torch.empty_like(out_flat, dtype=torch.float16)

    grid_ln2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid_ln2](
        out_flat,
        out_norm_flat,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    out_norm = out_norm_flat.view(B, N, N, H)   # [B, N, N, H] fp16

    # --------------------------------------------------------------
    # 6️⃣ Apply output gate and final projection back to `dim`
    # --------------------------------------------------------------
    out_norm = out_norm * out_gate                # element‑wise (fp16)

    final_half = F.linear(
        out_norm.view(total_rows, H),
        weights["to_out.weight"].to(torch.float16)   # [dim, H]
    )   # [R, dim] (fp16)

    final = final_half.view(B, N, N, dim).to(torch.float32)

    return final
