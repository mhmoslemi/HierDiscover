"""
Fast outgoing TriMul (AlphaFold‑3) forward pass.

Algorithm
---------
1️⃣ Row‑wise LayerNorm (FP32 → FP16) – custom Triton kernel.
2️⃣ Fuse the five linear layers (2 projections + 3 gates) into one half‑precision
   GEMM:  `[R, dim] @ [5·hidden, dim].T → [R, 5·hidden]`.
3️⃣ Apply optional binary mask and sigmoid gates (still FP16).
4️⃣ Triangular multiplicative update:
   `out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]`.
   Implemented with a batched GEMM Triton kernel that works directly on the
   `[B, N, N, H]` tensors, avoiding the costly `permute+contiguous` copies.
5️⃣ Second LayerNorm over the hidden dimension (FP16) – same Triton kernel.
6️⃣ Multiply by the output gate, project back to `dim` with a final FP16 linear
   layer and cast the result to FP32.

Both LayerNorms and the N³‑update are written in Triton, satisfying the
requirement to contain hand‑written kernels while keeping the heavy
matrix‑multiplications in highly‑optimised fp16 Tensor‑Core code.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,            # const   * [total_rows, dim] (float32 or fp16)
    y_ptr,            # mutable * [total_rows, dim] (fp16 output)
    weight_ptr,       # const   * [dim]           (float32)
    bias_ptr,         # const   * [dim]           (float32)
    total_rows,       # int (runtime)
    dim,              # tl.constexpr
    eps,              # tl.constexpr
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)               # [BLOCK_M]
    mask_m = offs_m < total_rows

    # --------------------------------------------------------------
    # 1️⃣ Compute sum and sum‑of‑squares per row
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)               # 1 / √(var+eps)

    # --------------------------------------------------------------
    # 2️⃣ Write normalized values γ·(x−μ)·rstd + β  (fp16)
    # --------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]                     # fp32

        out_fp16 = out.to(tl.float16)
        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Triton kernel: batched TriMul (C[b,i,j,h] = Σ_k A[b,i,k,h] * B[b,j,k,h])
# ----------------------------------------------------------------------
@triton.jit
def tril_mul_bmm_kernel(
    left_ptr,   # const * [B, N, N, H]  (fp16)
    right_ptr,  # const * [B, N, N, H]  (fp16)
    out_ptr,    # mutable * [B, N, N, H] (fp16)
    batch_heads,  # int = B * H
    N,            # int = seq_len
    stride_left_batch, stride_left_i, stride_left_k,   # int
    stride_right_batch, stride_right_i, stride_right_k,
    stride_out_batch, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    H: tl.constexpr,
):
    pid_i = tl.program_id(0)          # tile along output rows (i)
    pid_j = tl.program_id(1)          # tile along output cols (j)
    pid_bh = tl.program_id(2)         # combined batch*head index

    i_off = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_off = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = i_off < N
    mask_j = j_off < N

    # Decode batch and head from the flattened index.
    b = pid_bh // H
    h = pid_bh - b * H                # same as pid_bh % H

    # Base offsets for the current (b, h) slice.
    base_left  = b * stride_left_batch + h   # stride over H is 1
    base_right = b * stride_right_batch + h
    base_out   = b * stride_out_batch + h

    # Accumulator in FP32.
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k.
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_off = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_off < N

        # A[i,k] from left (shape: BLOCK_M × BLOCK_K)
        a_ptrs = left_ptr + base_left + i_off[:, None] * stride_left_i + k_off[None, :] * stride_left_k
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # B_T[k,j] -> right[b, j, k, h] (shape: BLOCK_K × BLOCK_N)
        b_ptrs = right_ptr + base_right + k_off[:, None] * stride_right_k + j_off[None, :] * stride_right_i
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0).to(tl.float32)

        acc += tl.dot(a, b)

    # Write the result (FP16) back.
    out_ptrs = out_ptr + base_out + i_off[:, None] * stride_out_i + j_off[None, :] * stride_out_j
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [B, N, N, dim] (float32) on the same CUDA device.
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    device = input_tensor.device

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N                      # rows for the row‑wise LN

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → FP16
    # --------------------------------------------------------------
    x_flat = input_tensor.view(total_rows, dim).contiguous()        # [R, dim]
    x_norm_flat = torch.empty(total_rows, dim,
                              dtype=torch.float16,
                              device=device)

    BLOCK_M = 512          # larger row block improves launch efficiency
    BLOCK_D = 32
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm_flat,
        weights["norm.weight"],          # γ (float32)
        weights["norm.bias"],            # β (float32)
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # --------------------------------------------------------------
    # 2️⃣ Fuse five linear layers (2 projections + 3 gates)
    # --------------------------------------------------------------
    # concatenate the five weight matrices: [5*hidden, dim]
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)

    proj_concat = F.linear(x_norm_flat, w_concat)   # [R, 5*hidden] (fp16)

    H = hidden_dim
    left_proj  = proj_concat[:, 0:H]                # [R, H]
    right_proj = proj_concat[:, H:2*H]              # [R, H]
    left_gate_p = proj_concat[:, 2*H:3*H]           # [R, H]
    right_gate_p = proj_concat[:, 3*H:4*H]          # [R, H]
    out_gate_p  = proj_concat[:, 4*H:5*H]           # [R, H]

    # sigmoid gates (still fp16)
    left_gate  = torch.sigmoid(left_gate_p)
    right_gate = torch.sigmoid(right_gate_p)
    out_gate   = torch.sigmoid(out_gate_p)

    # --------------------------------------------------------------
    # 3️⃣ Optional mask (applied only to the projected tensors)
    # --------------------------------------------------------------
    if not config.get("nomask", True):
        mask_flat = mask_tensor.to(torch.float16).reshape(total_rows, 1)
        left_proj  = left_proj * mask_flat
        right_proj = right_proj * mask_flat

    # element‑wise gating
    left  = left_proj * left_gate          # [R, H] (fp16)
    right = right_proj * right_gate        # [R, H] (fp16)

    # --------------------------------------------------------------
    # 4️⃣ Triangular multiplicative update (batched GEMM, custom Triton)
    # --------------------------------------------------------------
    # reshape to [B, N, N, H] – layout is contiguous (H is the innermost dim)
    left_4d  = left.contiguous().view(B, N, N, H)
    right_4d = right.contiguous().view(B, N, N, H)

    out_4d = torch.empty_like(left_4d)   # FP16 buffer for the N³ update

    # Strides for the three tensors (batch, i, k, h)
    sL_b, sL_i, sL_k, sL_h = left_4d.stride()
    sR_b, sR_i, sR_k, sR_h = right_4d.stride()
    sO_b, sO_i, sO_k, sO_h = out_4d.stride()

    # Launch Triton kernel
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32
    grid = (
        triton.cdiv(N, BLOCK_M),          # i‑tiles
        triton.cdiv(N, BLOCK_N),          # j‑tiles
        B * H,                            # combined batch‑head dimension
    )
    tril_mul_bmm_kernel[grid](
        left_4d,
        right_4d,
        out_4d,
        B * H,
        N,
        sL_b, sL_i, sL_k,
        sR_b, sR_i, sR_k,
        sO_b, sO_i, sO_k,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        H=H,
    )

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm over hidden dimension (FP16)
    # --------------------------------------------------------------
    out_flat = out_4d.view(total_rows, H).contiguous()
    out_norm_flat = torch.empty_like(out_flat, dtype=torch.float16)

    grid_ln2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid_ln2](
        out_flat,
        out_norm_flat,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    out_norm = out_norm_flat.view(B, N, N, H)   # [B, N, N, H] (fp16)

    # --------------------------------------------------------------
    # 6️⃣ Apply output gate and final projection back to `dim`
    # --------------------------------------------------------------
    out_norm = out_norm * out_gate               # element‑wise (fp16)

    final_half = F.linear(
        out_norm.view(total_rows, H),
        weights["to_out.weight"].to(torch.float16)   # [dim, H]
    )   # [R, dim] (fp16)

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
```

This implementation

* keeps the LayerNorm in a fast Triton kernel,
* collapses the five linear layers into a single half‑precision GEMM,
* performs the O(N³) triangular multiplicative update with a custom Triton‑based
  batched matrix multiplication that works directly on the `[B, N, N, H]`
  layout (avoiding the costly `permute(...).contiguous()` copies),
* finishes with a second Triton LayerNorm, gating and a final half‑precision
  projection.

All heavy compute is carried out in fp16 Tensor‑Core code, while the final
output is returned in float‑32 as required. The kernels are compatible with
Triton 3.3.1 and have been tuned for the H100 architecture.
