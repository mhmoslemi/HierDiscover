"""
TriMul (outgoing) – AlphaFold‑3 style forward pass

Algorithm
---------
1️⃣  Layer‑Norm over the input channel dimension `dim` (fp32 reduction,
    fp16 output) – Triton kernel.
2️⃣  Fuse the five linear layers (two projections + three gates) into one
    GEMM by concatenating their weights.  All ops are performed in fp16.
3️⃣  Optional pairwise mask is applied to the two projections before gating.
4️⃣  Apply sigmoid to the three gate tensors and element‑wise gating.
5️⃣  Triangular multiplicative update:
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    realized as a batched GEMM on tensors reshaped to (B·H, N, N).
6️⃣  Second Layer‑Norm over the hidden dimension `hidden_dim` (fp32 reduction,
    fp16 output) – same Triton kernel.
7️⃣  Multiply by the output‑gate, project back to `dim` with a final linear
    layer and cast the result to float32 (API contract).

All tensors are kept in fp16 while the heavy reductions of the two LayerNorms
are done in fp32 inside the custom Triton kernel, yielding good Tensor‑Core
utilisation on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: LayerNorm → fp16 output (mean/var computed in fp32)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,          # *const float*  [total_rows, dim]  (input, fp32 or fp16)
    y_ptr,          # *mut   half*   [total_rows, dim] (output fp16)
    weight_ptr,     # *const float*  [dim]           (γ)
    bias_ptr,       # *const float*  [dim]           (β)
    total_rows,     # int (runtime)
    dim: tl.constexpr,                 # int (compile‑time)
    eps: tl.constexpr,                 # float (compile‑time)
    BLOCK_M: tl.constexpr,             # rows per program
    BLOCK_D: tl.constexpr,             # columns per tile
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # --------------------------------------------------------------
    # Pass 1 – compute mean & variance (fp32 accumulation)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim

        # Load a tile of x (fp32 or fp16 → convert to fp32)
        ptrs = x_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                               # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------------------------------------------------------------
    # Pass 2 – write normalized + affine (fp16)
    # --------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptrs = x_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                               # fp32 raw data

        norm = (vals - mean[:, None]) * rstd[:, None]   # fp32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # γ, fp32
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # β, fp32

        out_fp32 = norm * w + b
        out_fp16 = out_fp32.to(tl.float16)

        out_ptrs = y_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        tl.store(out_ptrs, out_fp16, mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    TriMul (outgoing) forward pass.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, C], dtype torch.float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : mapping of layer‑norm and linear weights (float32)
        - config_dict  : contains "dim", "hidden_dim", optional "nomask"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, C], dtype torch.float32,
        residing on the same device as `input_tensor`.
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                # input channel C
    hidden_dim = config["hidden_dim"]  # hidden dimension H
    eps = 1e-5
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Shape utilities
    # ------------------------------------------------------------------
    B, N, _, C = input_tensor.shape   # C == dim
    assert C == dim, "dim mismatch between config and input tensor"
    total_rows = B * N * N             # rows for the flattened view

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (fp32 reduction → fp16 result)
    # ------------------------------------------------------------------
    # Flatten to [total_rows, dim] and launch kernel
    x_flat = input_tensor.view(total_rows, dim).contiguous()
    x_norm_fp16 = torch.empty((total_rows, dim), dtype=torch.float16, device=device)

    # Block sizes – good defaults for H100
    BLOCK_M = 256
    BLOCK_D = 64
    grid = (triton.cdiv(total_rows, BLOCK_M),)

    layernorm_fp16_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fuse the five linear layers (projections + gates) into one GEMM
    # ------------------------------------------------------------------
    # Concatenate weight matrices:   (5 * hidden_dim, dim)
    concat_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).half().contiguous()  # (5*H, C) in fp16

    # Linear: [total_rows, C] @ [C, 5*H]^T → [total_rows, 5*H]  (still fp16)
    proj_all = F.linear(x_norm_fp16, concat_w)  # (total_rows, 5*H)

    # Split back into the five tensors
    left_proj, right_proj, left_gate, right_gate, out_gate = torch.split(
        proj_all, hidden_dim, dim=1
    )  # each: (total_rows, H)

    # ------------------------------------------------------------------
    # 2a️⃣ Optional mask (applied before gating)
    # ------------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        mask_flat = mask_tensor.view(total_rows, 1).to(dtype=torch.float16)
        left_proj = left_proj * mask_flat
        right_proj = right_proj * mask_flat

    # ------------------------------------------------------------------
    # 2b️⃣ Sigmoid gates
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)   # (total_rows, H)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    # Apply gates element‑wise
    left_fp16 = left_proj * left_gate      # (total_rows, H)
    right_fp16 = right_proj * right_gate   # (total_rows, H)

    # Reshape to 4‑D tensors for the triangular update
    left = left_fp16.view(B, N, N, hidden_dim)    # (B, N, N, H)
    right = right_fp16.view(B, N, N, hidden_dim)  # (B, N, N, H)
    out_gate = out_gate.view(B, N, N, hidden_dim)  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 3️⃣ Triangular multiplicative update (batched GEMM)
    # ------------------------------------------------------------------
    # Permute so that the summed‑over dimension (k) is the innermost
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N)

    # Fuse batch and channel dimensions: (B*H, N, N) × (B*H, N, N)
    B_H = B * hidden_dim
    left_mat = left_perm.view(B_H, N, N)
    right_mat = right_perm.view(B_H, N, N)

    out_mat = torch.matmul(left_mat, right_mat)  # (B*H, N, N), fp16

    # Restore original layout: (B, N, N, H)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 4️⃣ Second LayerNorm over the hidden dimension (fp32 reduction → fp16)
    # ------------------------------------------------------------------
    total_rows_h = B * N * N
    out_flat = out.view(total_rows_h, hidden_dim).contiguous()
    out_norm_fp16 = torch.empty_like(out_flat)   # fp16 buffer

    grid2 = (triton.cdiv(total_rows_h, BLOCK_M),)
    layernorm_fp16_kernel[grid2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows_h,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    out_norm = out_norm_fp16.view(B, N, N, hidden_dim)  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5️⃣ Output gate + final projection back to `dim`
    # ------------------------------------------------------------------
    gated = out_norm * out_gate                     # (B, N, N, H) fp16

    # Final linear: H → C (still fp16)
    to_out_w_fp16 = weights["to_out.weight"].half()   # (C, H)
    gated_flat = gated.view(total_rows_h, hidden_dim)  # (total_rows_h, H)
    final_fp16 = F.linear(gated_flat, to_out_w_fp16)   # (total_rows_h, C)

    # Cast back to fp32 as required by the API contract
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
