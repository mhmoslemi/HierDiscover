@triton.jit
def fused_norm_linear_kernel(
    x_ptr,   # const half* [total_rows, hidden]
    gate_ptr, # const half* [total_rows, hidden] (sigmoid)
    weight_ptr, # const half* [dim, hidden]
    out_ptr, # mut float* [total_rows, dim]
    ln_weight_ptr, # const float* [hidden]
    ln_bias_ptr,   # const float* [hidden]
    total_rows: tl.int32,
    hidden: tl.constexpr,
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # First pass: compute mean&var across hidden (dim H)
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for h in range(0, tl.cdiv(hidden, BLOCK_H)):
        offs_h = h * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < hidden
        x = tl.load(x_ptr + offs_m[:, None] * hidden + offs_h[None, :],
                    mask=mask_m[:, None] & mask_h[None, :],
                    other=0.0).to(tl.float32)  # [BLOCK_M, BLOCK_H]
        sum_val += tl.sum(x, axis=1)
        sum_sq += tl.sum(x * x, axis=1)

    mean = sum_val / hidden
    var = sum_sq / hidden - mean * mean
    rstd = tl.rsqrt(var + eps)

    # Second pass: per dim tile accumulation
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        # result accumulator per row
        acc = tl.zeros([BLOCK_M, BLOCK_D], dtype=tl.float32)

        for h in range(0, tl.cdiv(hidden, BLOCK_H)):
            offs_h = h * BLOCK_H + tl.arange(0, BLOCK_H)
            mask_h = offs_h < hidden

            # Load x and compute normalized
            x = tl.load(x_ptr + offs_m[:, None] * hidden + offs_h[None, :],
                        mask=mask_m[:, None] & mask_h[None, :],
                        other=0.0).to(tl.float32)  # [BLOCK_M, BLOCK_H]

            norm = (x - mean[:, None]) * rstd[:, None]

            # Apply layernorm weight & bias
            w = tl.load(ln_weight_ptr + offs_h, mask=mask_h, other=0.0)  # [BLOCK_H]
            b = tl.load(ln_bias_ptr + offs_h, mask=mask_h, other=0.0)
            norm = norm * w[None, :] + b[None, :]  # [BLOCK_M, BLOCK_H]

            # Apply output gate
            gate = tl.load(gate_ptr + offs_m[:, None] * hidden + offs_h[None, :],
                           mask=mask_m[:, None] & mask_h[None, :],
                           other=0.0).to(tl.float32)   # [BLOCK_M, BLOCK_H]
            gated = norm * gate

            # Load weight transpose tile (hidden, dim)
            wgt = tl.load(weight_ptr + offs_h[:, None] + offs_d[None, :] * hidden,
                          mask=mask_h[:, None] & mask_d[None, :],

            acc += tl.dot(gated, wgt)   # (BLOCK_M, BLOCK_D)

        # Store the accumulated output for this dim tile

"""
Highly‑optimized Triton + PyTorch forward pass for the **outgoing** TriMul
operator (AlphaFold‑3).

Main ideas
---------
1️⃣  FP32‑reduced Row‑wise LayerNorm (input tensor) – Triton kernel.
2️⃣  Fuse the five linear projections (left/right + 3 gates) into a single
    FP16 GEMM (`torch.nn.functional.linear`).  This reads the normalized
    tensor only once.
3️⃣  Apply the optional binary mask and the sigmoid gates (still FP16).
4️⃣  Batched matrix‑multiply across the hidden dimension
    (left @ rightᵀ) using `torch.bmm`.  The batch size is
    `B * hidden_dim` (≤ 128 × 2 = 256) – cuBLAS/Tensor‑Cores on H100 are
    optimal for this.
5️⃣  Fuse the second LayerNorm, the output gate and the final linear
    projection (`hidden → dim`) into **one** Triton kernel.  This removes the
    intermediate `out_norm` tensor and cuts the total memory traffic by
    roughly 50 % compared to the reference implementation.

The result is a fast forward pass that stays completely in FP16
(except the final LayerNorm statistics) and returns the required
`float32` tensor.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1) Row‑wise LayerNorm (FP32 reduction → FP16 output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,               # const float*  [total_rows, dim]   (FP32 input)
    y_ptr,               #       half*  [total_rows, dim]   (FP16 output)
    weight_ptr,          # const float*  [dim]              (γ)
    bias_ptr,            # const float*  [dim]              (β)
    total_rows: tl.int32,
    dim: tl.constexpr,   # static (≤ 384)
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # ---------------------------------------------------------------
    # First pass: mean & variance (FP32)
    # ---------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        # load a block of the input
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------------------------------------------------------------
    # Second pass: normalise, apply γ, β and write as FP16
    # ---------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]   # (x-µ)/σ

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]                      # FP32

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 5) Fused LayerNorm (`to_out_norm`) + out‑gate + final linear proj.
# ----------------------------------------------------------------------
@triton.jit
def fused_norm_gate_linear_kernel(
    x_ptr,               # const half*   [total_rows, hidden]   (pre‑norm)
    gate_ptr,            # const half*   [total_rows, hidden]   (out_gate, sigmoid‑ed)
    weight_ptr,          # const half*   [dim, hidden]         (to_out.weight)
    y_ptr,               #       float*  [total_rows, dim]     (final output)
    ln_weight_ptr,       # const float*  [hidden]              (to_out_norm.γ)
    ln_bias_ptr,         # const float*  [hidden]              (to_out_norm.β)
    total_rows: tl.int32,
    hidden: tl.constexpr,
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ---------------------------------------------------------------
    # 1️⃣  Compute mean / variance across the hidden dimension (FP32)
    # ---------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for h in range(0, tl.cdiv(hidden, BLOCK_H)):
        offs_h = h * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < hidden

        x = tl.load(x_ptr + offs_m[:, None] * hidden + offs_h[None, :],
                    mask=mask_m[:, None] & mask_h[None, :],
                    other=0.0).to(tl.float32)         # [BLOCK_M, BLOCK_H]

        sum_val += tl.sum(x, axis=1)
        sum_sq  += tl.sum(x * x, axis=1)

    mean = sum_val / hidden
    var = sum_sq / hidden - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------------------------------------------------------------
    # 2️⃣  Normalise, apply LN γ/β, multiply by out_gate and GEMM→final.
    # ---------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        acc = tl.zeros([BLOCK_M, BLOCK_D], dtype=tl.float32)   # per‑row accumulator

        for h in range(0, tl.cdiv(hidden, BLOCK_H)):
            offs_h = h * BLOCK_H + tl.arange(0, BLOCK_H)
            mask_h = offs_h < hidden

            # ----- Load the pre‑norm values -----
            x = tl.load(x_ptr + offs_m[:, None] * hidden + offs_h[None, :],
                        mask=mask_m[:, None] & mask_h[None, :],
                        other=0.0).to(tl.float32)          # [BM, BH]

            # ----- LayerNorm normalisation -----
            norm = (x - mean[:, None]) * rstd[:, None]

            # ----- γ / β (float32) -----
            wln = tl.load(ln_weight_ptr + offs_h, mask=mask_h, other=0.0)   # γ
            bln = tl.load(ln_bias_ptr   + offs_h, mask=mask_h, other=0.0)   # β
            norm = norm * wln[None, :] + bln[None, :]                       # FP32

            # ----- Apply out‑gate (half → FP32) -----
            gate = tl.load(gate_ptr + offs_m[:, None] * hidden + offs_h[None, :],
                           mask=mask_m[:, None] & mask_h[None, :],
                           other=0.0).to(tl.float32)                # [BM, BH]
            gated = norm * gate                                   # [BM, BH]

            # ----- Load a tile of the final weight (transpose) -----
            # weight shape : (dim, hidden) → we need (hidden, dim) for dot.
            # element weight[d, h] has offset d * hidden + h.
            w = tl.load(weight_ptr + offs_h[:, None] + offs_d[None, :] * hidden,
                        mask=mask_h[:, None] & mask_d[None, :],
                        other=0.0).to(tl.float32)               # [BH, BD]

            # ----- Accumulate GEMM over the hidden tiles -----
            acc += tl.dot(gated, w)                               # (BM, BD)

        # ----- Store the finished output tile (FP32) -----
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 acc,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the **outgoing** TriMul operator (AlphaFold‑3).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, C] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (float32/bool)
        - weights_dict : model parameters (float32)
        - config_dict  : {
              "dim"        : C,
              "hidden_dim" : H,
              "nomask"     : bool,
              "eps"        : float (optional)
          }

    Returns
    -------
    torch.Tensor of shape [B, N, N, C] (float32) on the same CUDA device.
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    inp, mask, weights, cfg = data
    device = inp.device

    dim = cfg["dim"]
    hidden = cfg["hidden_dim"]
    eps = cfg.get("eps", 1e-5)
    nomask = cfg.get("nomask", True)

    B, N, _, _ = inp.shape
    total_rows = B * N * N                    # rows for all flattened tensors

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP16 output)
    # --------------------------------------------------------------
    inp_flat = inp.view(total_rows, dim).contiguous()          # [R, C] FP32
    norm_fp16 = torch.empty(total_rows, dim,
                            dtype=torch.half, device=device)  # [R, C] FP16

    BLOCK_M = 128   # rows per program –  fits nicely in shared memory
    BLOCK_D = 64    # columns per iteration (dim ≤ 384)

    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        inp_flat,
        norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # --------------------------------------------------------------
    # 2️⃣ Fused projection & gate computation (single FP16 GEMM)
    # --------------------------------------------------------------
    # Concatenate the 5 projection / gate weight matrices (each [hidden, dim])
    concat_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).half().contiguous()                                    # (5*H, C)

    # (R, 5*H)  –  FP16 GEMM
    proj_all = F.linear(norm_fp16, concat_w)                 # (R, 5*H)

    # Split the result
    left_proj, right_proj, left_gate, right_gate, out_gate = torch.split(
        proj_all, hidden, dim=1
    )                                                        # each (R, H)

    # --------------------------------------------------------------
    # 2a) Apply binary mask (if required)
    # --------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask.view(total_rows, 1).to(torch.float16)   # (R,1)
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # --------------------------------------------------------------
    # 2b) Sigmoid gates & element‑wise gating
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)        # (R, H)  FP16
    right_gate = torch.sigmoid(right_gate)      # (R, H)  FP16
    out_gate = torch.sigmoid(out_gate)          # (R, H)  FP16 (used later)

    left = left_proj * left_gate                # (R, H)  FP16
    right = right_proj * right_gate             # (R, H)  FP16

    # --------------------------------------------------------------
    # 3️⃣ Triangular multiplicative update  (batched GEMM)
    # --------------------------------------------------------------
    # Reshape to 4‑D tensors
    left = left.view(B, N, N, hidden)           # (B, N, N, H)
    right = right.view(B, N, N, hidden)         # (B, N, N, H)

    # Bring hidden into the batch dimension for cuBLAS
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden, N, N)   # (B*H, N, N)
    right_mat = right.permute(0, 3, 2, 1).contiguous().view(B * hidden, N, N) # (B*H, N, N)

    # Batched matrix multiplication – Tensor‑Core, FP16
    out_mat = torch.bmm(left_mat, right_mat)    # (B*H, N, N)

    # Restore original 4‑D layout
    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()   # (B, N, N, H)

    # --------------------------------------------------------------
    # 4️⃣ Fuse second LayerNorm, out‑gate and final linear projection
    # --------------------------------------------------------------
    out_flat = out.view(total_rows, hidden).contiguous()          # (R, H)  FP16
    out_gate_flat = out_gate.view(total_rows, hidden).contiguous()# (R, H)  FP16

    final_fp32 = torch.empty(total_rows, dim,
                             dtype=torch.float32, device=device)   # (R, dim)

    BLOCK_H = 64      # hidden‑dim tile size (≤128)
    BLOCK_D = 64      # output‑dim tile size (≤384)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    fused_norm_gate_linear_kernel[grid2](
        out_flat,
        out_gate_flat,
        weights["to_out.weight"].half().contiguous(),   # (dim, hidden)  FP16
        final_fp32,
        weights["to_out_norm.weight"],                  # γ  (float32)
        weights["to_out_norm.bias"],                    # β  (float32)
        total_rows,
        hidden,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_H=BLOCK_H,
        BLOCK_D=BLOCK_D,
    )

    # --------------------------------------------------------------
    # 5️⃣ Reshape to the original 4‑D layout and cast back to FP32
    # --------------------------------------------------------------
    out = final_fp32.view(B, N, N, dim)       # (B, N, N, C)  FP32
    return out
