@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    batch, M, N, K,
    stride_ab, stride_a_m, stride_a_k,
    stride_bb, stride_b_k, stride_b_n,
    stride_cb, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_idx = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    a_ptrs = A_ptr + batch_idx * stride_ab + offs_m[:, None] * stride_a_m + tl.arange(0, BLOCK_K)[None,:] * stride_a_k
    b_ptrs = B_ptr + batch_idx * stride_bb + tl.arange(0, BLOCK_K)[:, None] * stride_b_k + offs_n[None,:] * stride_b_n

    # mask out-of-bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (tl.arange(0, BLOCK_K) < K - k*BLOCK_K)[None, :], other=0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask=(tl.arange(0, BLOCK_K) < K - k*BLOCK_K)[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        acc += tl.dot(a, b)

        # offset pointers for next K block
        a_ptrs += BLOCK_K * stride_a_k
        b_ptrs += BLOCK_K * stride_b_k

    c = acc.to(tl.float16)

    # store
    c_ptrs = C_ptr + batch_idx * stride_cb + offs_m[:, None] * stride_c_m + offs_n[None,:] * stride_c_n
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None,:])

@triton.jit
def batched_matmul_fp16_kernel(
    A, B, C,
    batch, M, N, K,
    stride_Ab, stride_Am, stride_Ak,
    stride_Bb, stride_Bk, stride_Bn,
    stride_Cb, stride_Cm, stride_Cn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    batch_idx = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute start offsets for this tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for out-of-bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Pointers for the sub-tiles
        a_ptrs = A + batch_idx * stride_Ab + offs_m[:, None] * stride_Am + offs_k[None, :] * stride_Ak
        b_ptrs = B + batch_idx * stride_Bb + offs_k[:, None] * stride_Bk + offs_n[None, :] * stride_Bn

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        acc += tl.dot(a, b)

    # Write back
    c = acc.to(tl.float16)
    c_ptrs = C + batch_idx * stride_Cb + offs_m[:, None] * stride_Cm + offs_n[None, :] * stride_Cn
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])

"""
Highly‑optimized forward pass for the **outgoing TriMul** operator (AlphaFold‑3).

Algorithm
---------
1️⃣ **Row‑wise LayerNorm + fused projection**  
   A custom Triton kernel (`layernorm_kernel`) normalises the input
   and writes the result as FP16.  The normalised tensor is then fed to a
   single half‑precision GEMM (`F.linear`) whose weight matrix concatenates
   the five linear layers (left/right projections + three gates).  The GEMM
   yields the five tensors `left_proj`, `right_proj`, `left_gate`,
   `right_gate` and `out_gate` in one pass.

2️⃣ **Optional masking** – applied only to the two projected tensors.

3️⃣ **Batched half‑precision matrix multiply** – the core N³ update:
   `out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]`.  
   After reshaping, the operation is a batched mat‑mul of shape
   `(B·H, N, N) @ (B·H, N, N)ᵀ`.  It is executed by a custom Triton
   kernel (`batched_matmul_fp16_kernel`) that launches a single kernel for
   the whole batch, avoiding the overhead of cuBLAS’s many‑small‑GEMM
   launches.

4️⃣ **Second LayerNorm + gating + final projection** – the output of the
   mat‑mul is normalised across the hidden dimension (`H`) with another
   Triton `layernorm_kernel`, multiplied element‑wise by the sigmoid‑gated
   `out_gate` and projected back to the original feature dimension (`dim`)
   with a half‑precision linear layer.

All heavy compute is performed in FP16, while the two layer‑norm
operations use FP32 accumulation for numerical stability.  The final
output is cast back to FP32 to match the reference implementation.

The function `custom_kernel` follows the specification required by the
evaluation harness.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise (or hidden‑wise) LayerNorm (float32 → float16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,               # const *  [R, D]   (float32 or fp16 input)
    y_ptr,               #   *      [R, D]   (fp16 output)
    weight_ptr,          # const *  [D]      (float32)   γ
    bias_ptr,            # const *  [D]      (float32)   β
    total_rows,          # int (runtime)
    D,                   # tl.constexpr
    eps,                 # tl.constexpr
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """
    Row‑wise LayerNorm.
    y = (x - μ) / √(σ² + eps) * γ + β   (output FP16)
    """
    pid = tl.program_id(0)                     # row block id
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M) # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # 1️⃣ Reduce over the feature dimension to obtain mean & variance
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(D, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < D

        # Load a tile of the input (float32 after conversion)
        x = tl.load(
            x_ptr + offs_m[:, None] * D + offs_d[None, :],
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        sum_val += tl.sum(x, axis=1)
        sum_sq  += tl.sum(x * x, axis=1)

    mean = sum_val / D
    var  = sum_sq / D - mean * mean
    rstd = tl.rsqrt(var + eps)                     # 1 / √(var+eps)

    # ------------------------------------------------------------------
    # 2️⃣ Normalise, apply γ/β and write as FP16
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(D, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < D

        x = tl.load(
            x_ptr + offs_m[:, None] * D + offs_d[None, :],
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (x - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β

        y = (norm * w[None, :] + b[None, :]).to(tl.float16)

        tl.store(
            y_ptr + offs_m[:, None] * D + offs_d[None, :],
            y,
            mask=mask_m[:, None] & mask_d[None, :],
        )


# ----------------------------------------------------------------------
# Triton kernel: batched FP16 GEMM (A @ Bᵀ) for the N³ update
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_fp16_kernel(
    A_ptr, B_ptr, C_ptr,                     # * const, * const, * mutable (fp16)
    batch, M, N, K,                          # int runtime parameters
    stride_Ab, stride_Am, stride_Ak,          # strides for A (batch, M, K)
    stride_Bb, stride_Bk, stride_Bn,          # strides for B (batch, K, N) – B already transposed
    stride_Cb, stride_Cm, stride_Cn,          # strides for C (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched half‑precision matrix multiplication:
        C[b, i, j] = Σ_k A[b, i, k] * B[b, k, j]
    The kernel processes all batches in a single launch.
    """
    batch_idx = tl.program_id(0)               # which matrix in the batch
    pid_m = tl.program_id(1)                   # tile row index
    pid_n = tl.program_id(2)                   # tile column index

    # offsets inside the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32 for accuracy
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k_tile in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # pointers for the current sub‑tiles
        a_ptr = (
            A_ptr
            + batch_idx * stride_Ab
            + offs_m[:, None] * stride_Am
            + offs_k[None, :] * stride_Ak
        )
        b_ptr = (
            B_ptr
            + batch_idx * stride_Bb
            + offs_k[:, None] * stride_Bk
            + offs_n[None, :] * stride_Bn
        )

        a = tl.load(
            a_ptr,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)
        b = tl.load(
            b_ptr,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        ).to(tl.float32)

        acc += tl.dot(a, b)                     # (BLOCK_M, BLOCK_K)·(BLOCK_K, BLOCK_N)

    # store result as FP16
    c = acc.to(tl.float16)
    c_ptr = (
        C_ptr
        + batch_idx * stride_Cb
        + offs_m[:, None] * stride_Cm
        + offs_n[None, :] * stride_Cn
    )
    tl.store(c_ptr, c, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# High‑level custom kernel
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradient needed).

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [B, N, N, dim] (float32, on the same CUDA device).
    """
    # --------------------------------------------------------------
    # Unpack
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim         = config["dim"]          # input feature dimension
    hidden_dim  = config["hidden_dim"]   # hidden dimension H
    eps         = 1e-5
    nomask      = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N                     # rows for the row‑wise LN

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → FP16
    # ------------------------------------------------------------------
    x_flat = input_tensor.view(total_rows, dim).contiguous()
    x_ln = torch.empty(total_rows, dim, dtype=torch.float16, device=device)

    BLOCK_M = 256
    BLOCK_D = 32
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_ln,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fuse the five linear layers (left/right proj + 3 gates)
    # ------------------------------------------------------------------
    # concatenate the five weight matrices [5*H, dim]  (bias‑less)
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # (5*H, dim)

    # single GEMM: (R, dim) @ (5*H, dim).T → (R, 5*H)
    proj = F.linear(x_ln, w_concat)        # shape (R, 5*H), FP16

    H = hidden_dim
    left_proj   = proj[:, 0:H]                       # (R, H)
    right_proj  = proj[:, H:2*H]                     # (R, H)
    left_gate   = proj[:, 2*H:3*H]                   # (R, H)
    right_gate  = proj[:, 3*H:4*H]                   # (R, H)
    out_gate    = proj[:, 4*H:5*H]                   # (R, H)

    # 3️⃣ Sigmoid gates (still FP16)
    left_gate   = torch.sigmoid(left_gate)
    right_gate  = torch.sigmoid(right_gate)
    out_gate    = torch.sigmoid(out_gate)

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask (applied only on the projected tensors)
    # ------------------------------------------------------------------
    if not nomask:
        mask_h = mask_tensor.to(torch.float16).reshape(total_rows, 1)   # (R,1)
        left_proj  = left_proj * mask_h
        right_proj = right_proj * mask_h

    # element‑wise gating
    left  = left_proj * left_gate      # (R, H)
    right = right_proj * right_gate    # (R, H)

    # reshape back to 4‑D tensors
    left  = left.view(B, N, N, H)          # (B, N, N, H)
    right = right.view(B, N, N, H)         # (B, N, N, H)
    out_gate = out_gate.view(B, N, N, H)   # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5️⃣ Batched matmul: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    # bring hidden dimension into the batch axis
    left_perm  = left.permute(0, 3, 1, 2).contiguous()     # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()    # (B, H, N, N)

    # fuse batch & head dimensions → (B*H, N, N)
    left_mat  = left_perm.view(B * H, N, N)               # (BH, N, N)
    right_mat = right_perm.view(B * H, N, N)              # (BH, N, N)

    # transpose right for the gemm
    right_mat_T = right_mat.transpose(1, 2).contiguous()  # (BH, N, N) where last two dims are (K,N)

    out_mat = torch.empty_like(left_mat, dtype=torch.float16)   # (BH, N, N)

    # Triton launch configuration (tuned for H100 TensorCores)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (
        B * H,                                      # batch dimension
        triton.cdiv(N, BLOCK_M),                    # M‑tiles
        triton.cdiv(N, BLOCK_N),                    # N‑tiles
    )
    batched_matmul_fp16_kernel[grid](
        left_mat,
        right_mat_T,
        out_mat,
        B * H, N, N, N,
        left_mat.stride(0), left_mat.stride(1), left_mat.stride(2),
        right_mat_T.stride(0), right_mat_T.stride(1), right_mat_T.stride(2),
        out_mat.stride(0), out_mat.stride(1), out_mat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # restore (B, N, N, H) layout
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # (B, N, N, H)

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm over hidden dimension (fp16)
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, H).contiguous()
    out_norm_flat = torch.empty_like(out_flat, dtype=torch.float16)

    grid_ln2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid_ln2](
        out_flat,
        out_norm_flat,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    out_norm = out_norm_flat.view(B, N, N, H)   # (B, N, N, H) fp16

    # ------------------------------------------------------------------
    # 7️⃣ Output gate and final projection back to `dim`
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate                     # element‑wise (fp16)

    final_half = F.linear(
        out_norm.view(total_rows, H),
        weights["to_out.weight"].to(torch.float16),   # (dim, H)
    )   # (R, dim) fp16

    # Cast to FP32 as required by the reference implementation
    final = final_half.view(B, N, N, dim).to(torch.float32)

    return final
