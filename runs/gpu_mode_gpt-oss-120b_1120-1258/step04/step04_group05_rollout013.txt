@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr,
    right_ptr,
    out_ptr,
    BHM,
    N,
    K,
    stride_lbm,
    stride_li,
    stride_lk,
    stride_rbm,
    stride_ri,
    stride_rk,
    stride_obm,
    stride_oi,
    stride_oj,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    # Compute total tiles per batch
    num_tile_i = tl.cdiv(N, BLOCK_M)
    num_tile_j = tl.cdiv(N, BLOCK_N)
    tiles_per_batch = num_tile_i * num_tile_j

    batch = pid // tiles_per_batch
    tile_idx = pid % tiles_per_batch
    tile_i = tile_idx // num_tile_j
    tile_j = tile_idx % num_tile_j

    # Compute actual coordinate offsets
    i_off = tile_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_off = tile_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks
    mask_i = i_off < N
    mask_j = j_off < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k_iter in range(0, tl.cdiv(K, BLOCK_K)):
        k_off = k_iter * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_off < K

        # Load left tile (BLOCK_M, BLOCK_K)
        left_ptrs = left_ptr + batch * stride_lbm + i_off[:, None] * stride_li + k_off[None, :] * stride_lk
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load right tile transposed (BLOCK_K, BLOCK_N)
        right_ptrs = right_ptr + batch * stride_rbm + k_off[:, None] * stride_rk + j_off[None, :] * stride_ri
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        # Multiply-accumulate
        acc += tl.dot(left, right)

    # Write out
    out_ptrs = out_ptr + batch * stride_obm + i_off[:, None] * stride_oi + j_off[None, :] * stride_oj
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

#!/usr/bin/env python
"""
Optimized outgoing TriMul kernel (AlphaFold‑3 style).

Algorithm
---------
1️⃣  Row‑wise LayerNorm on the input channel dimension (fp32 reduction,
    fp16 output) – implemented in Triton.

2️⃣  Fuse the five linear projections (left/right + 3 gates) into a single
    GEMM by concatenating the weight matrices.

3️⃣  Apply the optional mask (before gating) and sigmoid gates.

4️⃣  Triangular multiplicative update (outgoing version):
        out_h[i, j] = Σ_k left_h[i, k] * right_h[j, k]
    Implemented as a custom batched GEMM kernel (`tri_mul_outgoing_kernel`).
    The kernel performs a block‑wise matrix multiplication for every
    hidden‑dimension batch (B·H) using fp16 → fp32 accumulation.

5️⃣  Row‑wise LayerNorm over the hidden dimension (fp32 reduction,
    fp16 output).

6️⃣  Gating with the output‑gate and a final linear projection back to the
    original channel dimension.

All heavy compute steps are executed in FP16 with FP32 accumulation for
numerical stability; the final result is returned in FP32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1. Row‑wise LayerNorm (fp16 output, fp32 reduction)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,          # *const  float*   [total_rows, dim] (fp32 or fp16)
    y_ptr,          # *mut    float16* [total_rows, dim] (fp16 output)
    weight_ptr,     # *const  float*   [dim]          (γ)
    bias_ptr,       # *const  float*   [dim]          (β)
    total_rows,     # i32
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns processed per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)               # [BLOCK_M]
    mask_m = offs_m < total_rows

    # -----------------------------------------------------------------
    # 1. Compute mean & variance (fp32)
    # -----------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)       # [BLOCK_D]
        mask_d = offs_d < dim

        # Load a tile [BLOCK_M, BLOCK_D]
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                                   # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                     # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)              # [BLOCK_M]

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                               # 1 / sqrt(var+eps)

    # -----------------------------------------------------------------
    # 2. Write normalized output (fp16)
    # -----------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                                   # fp32

        # (x - mean) / sqrt(var+eps)
        norm = (vals - mean[:, None]) * rstd[:, None]      # fp32

        # γ (weight) and β (bias) are fp32
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)

        out_fp16 = (norm * w[None, :] + b[None, :]).to(tl.float16)

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2. Batched triangular multiplicative update (outgoing version)
#    Computes out[b*h,i,j] = Σ_k left[b*h,i,k] * right[b*h,j,k]
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr,    # *const  half*  [BHM, N, K]   (fp16)
    right_ptr,   # *const  half*  [BHM, N, K]   (fp16)
    out_ptr,     # *mut    half*  [BHM, N, N]   (fp16)
    BHM,         # i32  = B * hidden_dim
    N,           # i32
    K,           # i32  (K == N)
    stride_lbm, stride_li, stride_lk,   # strides for left
    stride_rbm, stride_ri, stride_rk,   # strides for right
    stride_obm, stride_oi, stride_oj,   # strides for out
    BLOCK_M: tl.constexpr,   # tile size on i dimension
    BLOCK_N: tl.constexpr,   # tile size on j dimension
    BLOCK_K: tl.constexpr,   # tile size on reduction dimension
):
    pid = tl.program_id(0)

    # -------------------------------------------------------------
    # Decompose the 1‑D program id into (batch, i‑tile, j‑tile)
    # -------------------------------------------------------------
    num_tile_i = tl.cdiv(N, BLOCK_M)
    num_tile_j = tl.cdiv(N, BLOCK_N)
    tiles_per_batch = num_tile_i * num_tile_j

    batch   = pid // tiles_per_batch
    tile_id = pid % tiles_per_batch
    tile_i  = tile_id // num_tile_j
    tile_j  = tile_id % num_tile_j

    # -------------------------------------------------------------
    # Offsets for the current tile
    # -------------------------------------------------------------
    i_off = tile_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_off = tile_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = i_off < N
    mask_j = j_off < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------------------------------------------------
    # Loop over the reduction dimension K in BLOCK_K chunks
    # -------------------------------------------------------------
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        k_off = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_off < K

        # left tile   : [BLOCK_M, BLOCK_K]
        left_ptrs = left_ptr + \
            batch * stride_lbm + \
            i_off[:, None] * stride_li + \
            k_off[None, :] * stride_lk
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # right tile transposed : [BLOCK_K, BLOCK_N]
        # we load right as (k, j) so that tl.dot can be used directly.
        right_ptrs = right_ptr + \
            batch * stride_rbm + \
            k_off[:, None] * stride_rk + \
            j_off[None, :] * stride_ri
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0).to(tl.float32)

        # accumulate: left (M x K) dot right (K x N)
        acc += tl.dot(left, right)

    # -------------------------------------------------------------
    # Write the result back (fp16)
    # -------------------------------------------------------------
    out_ptrs = out_ptr + \
        batch * stride_obm + \
        i_off[:, None] * stride_oi + \
        j_off[None, :] * stride_oj
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    Arguments:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, C] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool / float) – may be None
            - weights_dict : mapping of weight names to tensors (float32)
            - config_dict  : {"dim": C, "hidden_dim": H, "nomask": bool}
    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # 0. Unpack & basic shapes
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                # input channel dimension C
    hidden_dim = config["hidden_dim"]  # hidden dimension H
    eps = 1e-5
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N            # rows for the flattened view used by LayerNorm

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (fp16 output)
    # ------------------------------------------------------------------
    x_flat = input_tensor.contiguous().view(total_rows, dim)    # fp32
    x_norm_fp16 = torch.empty(total_rows, dim,
                              dtype=torch.float16,
                              device=device)                # fp16 buffer

    BLOCK_M = 256          # rows per program (tuned for the given workloads)
    BLOCK_D = 64           # cols per reduction chunk
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fuse the five linear projections (left/right + 3 gates)
    # ------------------------------------------------------------------
    # all weight tensors are [hidden_dim, dim] (no bias)
    fused_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).half().contiguous()                               # shape (5*H, C)

    # fused GEMM: (total_rows, C) @ (C, 5*H).T -> (total_rows, 5*H)
    proj_all = F.linear(x_norm_fp16, fused_weight)     # fp16

    # split back
    left_proj, right_proj, left_gate, right_gate, out_gate = torch.split(
        proj_all, hidden_dim, dim=1
    )                                                   # each [total_rows, H] fp16

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (applied before gating)
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] -> [total_rows, 1]
        mask_fp16 = mask_tensor.contiguous().view(total_rows, 1).to(torch.float16)
        left_proj  = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # ------------------------------------------------------------------
    # 4️⃣ Apply sigmoid gates and element‑wise gating
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    left = left_proj * left_gate          # [total_rows, H] fp16
    right = right_proj * right_gate       # [total_rows, H] fp16

    # reshape to 4‑D tensors
    left = left.view(B, N, N, hidden_dim)     # [B, N, N, H]
    right = right.view(B, N, N, hidden_dim)   # [B, N, N, H]
    out_gate = out_gate.view(B, N, N, hidden_dim)  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 5️⃣ Triangular multiplicative update (batched GEMM)
    # ------------------------------------------------------------------
    # bring hidden dimension into the batch dimension
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    BHM = B * hidden_dim                                 # B * H
    left_mat = left_perm.view(BHM, N, N)                 # [BHM, N, N]
    right_mat = right_perm.view(BHM, N, N)               # [BHM, N, N]

    out_mat = torch.empty_like(left_mat)                 # fp16 output buffer

    # launch the custom kernel
    BLOCK_M2 = 64      # tile size in the i‑dimension
    BLOCK_N2 = 64      # tile size in the j‑dimension
    BLOCK_K2 = 32      # reduction tile size
    ti = (N + BLOCK_M2 - 1) // BLOCK_M2
    tj = (N + BLOCK_N2 - 1) // BLOCK_N2
    total_progs = BHM * ti * tj
    grid = (total_progs,)
    tri_mul_outgoing_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        BHM,
        N,
        N,
        left_mat.stride(0), left_mat.stride(1), left_mat.stride(2),
        right_mat.stride(0), right_mat.stride(1), right_mat.stride(2),
        out_mat.stride(0), out_mat.stride(1), out_mat.stride(2),
        BLOCK_M=BLOCK_M2,
        BLOCK_N=BLOCK_N2,
        BLOCK_K=BLOCK_K2,
    )

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm over the hidden dimension (fp16 output)
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, hidden_dim)          # [total_rows, H]
    out_norm_fp16 = torch.empty_like(out_flat,
                                     dtype=torch.float16,
                                     device=device)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    out_norm = out_norm_fp16.view(B, N, N, hidden_dim)   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 7️⃣ Final gating + linear projection back to C
    # ------------------------------------------------------------------
    gated = out_norm * out_gate                   # fp16 element‑wise
    # final projection (H -> C)
    final_fp16 = F.linear(gated.view(total_rows, hidden_dim),
                          weights["to_out.weight"].half())   # fp16

    # ------------------------------------------------------------------
    # 8️⃣ Cast to fp32 and return
    # ------------------------------------------------------------------
    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final
