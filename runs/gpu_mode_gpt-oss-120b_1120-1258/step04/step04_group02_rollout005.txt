"""
Highly‑optimized forward pass for the **outgoing** TriMul operator (AlphaFold‑3).

Algorithm
---------
1️⃣  Input LayerNorm – performed with the native PyTorch `layer_norm`
    (FP32 reduction → FP16 output).  This is much faster than a hand‑written
    Triton kernel on H100.

2️⃣  Fuse the five linear projections (left/right + three gates) into a single
    FP16 GEMM (`torch.nn.functional.linear`).  The weight matrix is the
    concatenation of the five `[hidden, dim]` matrices.

3️⃣  (Optional) apply the binary mask to the left/right projections.

4️⃣  Compute the three sigmoid gates (still FP16) and apply the element‑wise
    gating to the left and right projections.

5️⃣  Triangular multiplicative update – a batched matrix multiplication:
    for each batch and hidden channel we compute
    ``out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]``.
    This is realized with a single `torch.bmm` on the reshaped tensors,
    allowing cuBLAS/Tensor‑Core to handle the whole batch efficiently.

6️⃣  Fuse the second LayerNorm and the output‑gate into a single Triton kernel
    (`fused_layernorm_gate_kernel`).  It computes the per‑row mean/variance in
    FP32, applies the learned γ/β, multiplies by the pre‑computed `out_gate`,
    and writes the result as FP16.

7️⃣  Final linear projection back to `dim` (`torch.nn.functional.linear`)
    and cast the result to FP32.

Only the two LayerNorm steps require Triton kernels; the remaining dense
linear algebra uses cuBLAS which is already optimal on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 6) Fused LayerNorm + output‑gate (both FP16)
# ----------------------------------------------------------------------
@triton.jit
def fused_layernorm_gate_kernel(
    x_ptr,               # *const half*  [total_rows, dim]   (pre‑norm)
    y_ptr,               # *mut   half*  [total_rows, dim]   (post‑norm⋅gate)
    weight_ptr,          # *const float* [dim]               (γ)
    bias_ptr,            # *const float* [dim]               (β)
    gate_ptr,            # *const half*  [total_rows, dim]   (out_gate, sigmoid‑ed)
    total_rows: tl.int32,
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per program
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ---------------------------------------------------------------
    # First pass: mean & variance (FP32 accumulation)
    # ---------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)      # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                         # 1 / sqrt(var+eps)

    # ---------------------------------------------------------------
    # Second pass: normalise, apply γ, β, multiply by gate, store FP16
    # ---------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # FP32

        # (x - μ) / σ
        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]                      # FP32

        # multiply by the already‑sigmoided out_gate (half → FP32)
        gate_vals = tl.load(gate_ptr + offs_m[:, None] * dim + offs_d[None, :],
                            mask=mask_m[:, None] & mask_d[None, :],
                            other=0.0).to(tl.float32)          # FP32
        out = out * gate_vals

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the **outgoing** TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, C]      (float32)
        - mask_tensor  : torch.Tensor [B, N, N]       (float32 or bool); ignored if config["nomask"]=True
        - weights_dict : model parameters (float32)
        - config_dict  : {
              "dim"        : int   (C),
              "hidden_dim" : int   (H),
              "nomask"    : bool,
              "eps"       : float (optional)
          }

    Returns
    -------
    torch.Tensor [B, N, N, C] (float32) on the same CUDA device.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    inp, mask, weights, cfg = data
    device = inp.device

    dim    = cfg["dim"]                # input & output channel dimension (C)
    hidden = cfg["hidden_dim"]         # hidden dimension (H)
    eps    = cfg.get("eps", 1e-5)
    nomask = cfg.get("nomask", True)

    B, N, _, _ = inp.shape                # [B, N, N, C]
    total_rows = B * N * N                 # rows for flattened tensors

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    # Native PyTorch LayerNorm is highly‑optimised on H100.
    inp_norm = torch.nn.functional.layer_norm(
        inp,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )
    # Cast to FP16 for the rest of the pipeline
    x_norm_fp16 = inp_norm.half()   # [B, N, N, C] in half

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear projections + gates (single FP16 GEMM)
    # ------------------------------------------------------------------
    # Concatenate the five projection / gate weight matrices (each [H, C])
    concat_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).half()                               # shape (5*H, C)

    # Flatten to (R, C) where R = B*N*N
    x_norm_flat = x_norm_fp16.view(total_rows, dim).contiguous()
    # One big linear projection producing all five tensors at once
    proj_all = F.linear(x_norm_flat, concat_w)   # (R, 5*H)

    # Split into the individual tensors (each (R, H))
    left_proj_fp16, right_proj_fp16, left_gate_fp16, right_gate_fp16, out_gate_fp16 = torch.split(
        proj_all, hidden, dim=1
    )

    # ------------------------------------------------------------------
    # 2a) Apply mask if required (mask broadcasted on the hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask_flat = mask.view(total_rows, 1).to(torch.float16)   # (R,1)
        left_proj_fp16  = left_proj_fp16 * mask_flat
        right_proj_fp16 = right_proj_fp16 * mask_flat

    # ------------------------------------------------------------------
    # 2b) Sigmoid gates (still FP16) and element‑wise gating
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate_fp16)   # (R, H)
    right_gate = torch.sigmoid(right_gate_fp16)  # (R, H)
    out_gate   = torch.sigmoid(out_gate_fp16)    # (R, H) – will be used later

    left_fp16  = left_proj_fp16 * left_gate
    right_fp16 = right_proj_fp16 * right_gate

    # ------------------------------------------------------------------
    # 3️⃣ Triangular multiplicative update (batched GEMM)
    # ------------------------------------------------------------------
    # Reshape back to (B, N, N, H)
    left  = left_fp16.view(B, N, N, hidden)   # (B, N, N, H)
    right = right_fp16.view(B, N, N, hidden)  # (B, N, N, H)

    # Rearrange to (B*H, N, N) where rows = i, cols = k  and rows = k, cols = j respectively
    left_mat  = left.permute(0, 3, 1, 2).reshape(B * hidden, N, N)   # (B*H, N, N)
    right_mat = right.permute(0, 3, 2, 1).reshape(B * hidden, N, N)   # (B*H, N, N)

    # Batched matrix multiplication – takes full advantage of Tensor‑Core
    out_mat = torch.bmm(left_mat, right_mat)   # (B*H, N, N)

    # Reshape back to (B, N, N, H)
    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 4️⃣ Second LayerNorm + output‑gate (fused Triton kernel)
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, hidden).contiguous()   # (R, H) half
    out_norm_gate_fp16 = torch.empty_like(out_flat)        # (R, H) half

    BLOCK_M = 64   # rows per program (tuned for H100)
    BLOCK_D = 64   # columns per program
    grid = (triton.cdiv(total_rows, BLOCK_M),)

    fused_layernorm_gate_kernel[grid](
        out_flat,
        out_norm_gate_fp16,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        out_gate,                                 # (R, H) half, already sigmoid‑ed
        total_rows,
        hidden,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 5️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    to_out_w_fp16 = weights["to_out.weight"].half()       # (dim, H)
    final_fp16 = F.linear(out_norm_gate_fp16, to_out_w_fp16)   # (R, dim) half

    # Cast back to FP32 and restore original 4‑D shape
    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final
