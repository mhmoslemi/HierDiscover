"""
TriMul (outgoing) forward kernel – highly‑optimized Triton + PyTorch implementation.

Algorithm
---------
1️⃣  Layer‑norm over the channel dimension `dim` (float32 mean/var, fp16 output).
2️⃣  Fuse the five linear layers (left/right projections + three gates) into a
    single half‑precision GEMM by concatenating their weight matrices.
3️⃣  Apply the optional pairwise mask (only to the left/right projections).
4️⃣  Sigmoid the three gate tensors and multiply them element‑wise with the
    projected values.
5️⃣  Triangular multiplicative update:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    realized as a batched half‑precision matrix‑multiply
    (B·H, N, N) × (B·H, N, N) → (B·H, N, N).
6️⃣  Second Layer‑norm over the hidden dimension `hidden_dim` (fp16 output).
7️⃣  Multiply by the output gate, project back to `dim` with a final half‑precision
    linear layer, and cast the result to float32.

The heavy reductions (two LayerNorms) are performed by a custom Triton kernel,
while dense linear algebra is delegated to cuBLAS/Tensor‑Core via regular
PyTorch ops.  The implementation works for all supported batch sizes,
sequence lengths and channel dimensions.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: LayerNorm that reads fp32, computes mean/var in fp32,
# and writes the normalized+affine result in fp16.
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,          # const float*   [total_rows, dim]  (input)
    y_ptr,          #       half*   [total_rows, dim]  (output)
    weight_ptr,     # const float*   [dim]             (γ)
    bias_ptr,       # const float*   [dim]             (β)
    total_rows,     # int32 (runtime)
    dim: tl.constexpr,          # compile‑time constant
    eps: tl.constexpr,          # compile‑time constant
    BLOCK_M: tl.constexpr,      # rows per program
    BLOCK_D: tl.constexpr,      # columns per tile
):
    pid = tl.program_id(0)                     # one program per row‑block
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M) # [BLOCK_M]
    mask_m = offs_m < total_rows                # mask for out‑of‑bounds rows

    # -----------------------------------------------------------------
    # Pass 1 – compute mean and variance (fp32 accumulation)
    # -----------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        # Load a (BLOCK_M x BLOCK_D) tile of x (fp32)
        ptrs = x_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                              # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)               # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)        # [BLOCK_M]

    mean = sum_val / tl.float32(dim)
    var  = sum_sq / tl.float32(dim) - mean * mean
    rstd = tl.rsqrt(var + eps)

    # -----------------------------------------------------------------
    # Pass 2 – write (x - mean) / sqrt(var+eps) * γ + β   (fp16 output)
    # -----------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        # Load raw fp32 values again
        ptrs = x_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                              # [BLOCK_M, BLOCK_D]

        # Normalize
        norm = (vals - mean[:, None]) * rstd[:, None] # fp32

        # Load γ, β (fp32)
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)

        out_fp32 = norm * w + b                         # fp32
        out_fp16 = out_fp32.to(tl.float16)

        out_ptrs = y_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    TriMul forward (outgoing) – returns a float32 tensor of shape
    [bs, seq_len, seq_len, dim].

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        * input_tensor : torch.Tensor, shape [B, N, N, C], dtype float32
        * mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        * weights_dict : mapping of all LayerNorm/Linear parameters
                         (float32 tensors)
        * config_dict  : contains "dim", "hidden_dim", optional "nomask"

    Returns
    -------
    torch.Tensor
        The transformed tensor, dtype torch.float32, on the same device.
    """
    # ------------------------------------------------------------------
    # 0️⃣ Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                # channel dimension C
    hidden_dim = config["hidden_dim"]  # hidden dimension H
    eps = 1e-5
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (fp32 mean/var, fp16 output)
    # ------------------------------------------------------------------
    B, N, _, _ = input_tensor.shape          # B, N, N, C
    total_rows = B * N * N                    # rows of the flattened view

    # Flatten to (total_rows, dim) and ensure contiguous layout
    x_flat = input_tensor.view(total_rows, dim).contiguous()
    x_norm_fp16 = torch.empty((total_rows, dim), dtype=torch.float16,
                              device=device)

    # Block sizes – tuned for H100, work well for C up to 384
    BLOCK_M = 256
    BLOCK_D = 64
    grid = (triton.cdiv(total_rows, BLOCK_M),)

    layernorm_fp16_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear projections + three gate linear layers
    #    (all in fp16, no bias)
    # ------------------------------------------------------------------
    # Concatenate the five weight matrices → shape (5*H, C)
    concat_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).half().contiguous()                     # (5*H, C)

    # Linear: [total_rows, C] @ [C, 5*H]^T → [total_rows, 5*H] (fp16)
    proj_all = F.linear(x_norm_fp16, concat_w)   # (total_rows, 5*H)

    # Split back into the five tensors
    left_proj, right_proj, left_gate, right_gate, out_gate = torch.split(
        proj_all, hidden_dim, dim=1
    )  # each: (total_rows, H)

    # ------------------------------------------------------------------
    # 2a️⃣ Optional pairwise mask (applied only to the projections)
    # ------------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        # mask: [B, N, N] → [total_rows, 1] broadcast over hidden dim
        mask_flat = mask_tensor.view(total_rows, 1).to(dtype=torch.float16, device=device)
        left_proj = left_proj * mask_flat
        right_proj = right_proj * mask_flat

    # ------------------------------------------------------------------
    # 2b️⃣ Sigmoid gates (half‑precision)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    # Element‑wise gating
    left_fp16 = left_proj * left_gate     # (total_rows, H)
    right_fp16 = right_proj * right_gate  # (total_rows, H)

    # ------------------------------------------------------------------
    # 3️⃣ Reshape for the triangular update
    # ------------------------------------------------------------------
    left = left_fp16.view(B, N, N, hidden_dim)   # (B, N, N, H)
    right = right_fp16.view(B, N, N, hidden_dim) # (B, N, N, H)
    out_gate = out_gate.view(B, N, N, hidden_dim)  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 4️⃣ Triangular multiplicative update (batched GEMM)
    #     out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    # Rearrange to (B, H, N, N) where the summed‑over dimension is last for
    # left and first for right.
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, i=k)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, k, j)

    # Merge batch and hidden dimensions → (B*H, N, N)
    B_H = B * hidden_dim
    left_mat = left_perm.view(B_H, N, N)   # (B*H, N, N)
    right_mat = right_perm.view(B_H, N, N) # (B*H, N, N)

    # Batched half‑precision matrix multiplication
    out_mat = torch.matmul(left_mat, right_mat)  # (B*H, N, N), fp16

    # Reshape back to (B, N, N, H)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm over the hidden dimension (fp16 output)
    # ------------------------------------------------------------------
    total_rows_h = B * N * N
    out_flat = out.view(total_rows_h, hidden_dim).contiguous()
    out_norm_fp16 = torch.empty_like(out_flat)   # fp16 buffer

    grid2 = (triton.cdiv(total_rows_h, BLOCK_M),)
    layernorm_fp16_kernel[grid2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows_h,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    out_norm = out_norm_fp16.view(B, N, N, hidden_dim)  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 6️⃣ Output gate + final projection back to `dim`
    # ------------------------------------------------------------------
    gated = out_norm * out_gate                               # (B, N, N, H) fp16

    # Final linear projection H → C (still half‑precision)
    to_out_w_fp16 = weights["to_out.weight"].half()
    gated_flat = gated.view(total_rows_h, hidden_dim)          # (total_rows_h, H)
    final_fp16 = F.linear(gated_flat, to_out_w_fp16)           # (total_rows_h, C)

    # Cast to float32 as required by the public API
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
