"""
custom_kernel: highly‑optimized “outgoing” TriMul forward (AlphaFold‑3).

Algorithm
---------
1️⃣  Row‑wise LayerNorm on the input tensor (FP32 → FP16) – a tiny Triton
    kernel calculates the mean/variance in FP32 and writes normalized
    values as FP16 to cut memory traffic.

2️⃣  Fuse the two linear projections (proj + gate) for the left and right
    paths into a single matrix‑multiply (dim → 2·hidden_dim).  The result
    is split and the gate part gets a sigmoid.

3️⃣  Optional pair‑wise mask is applied (broadcast over the hidden dim).

4️⃣  After gating, the core “tri‑multiplicative” update is performed by a
    single `torch.einsum`:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    This avoids the costly `permute(...).contiguous()` copies that the
    reference implementation performs.  The einsum is internally mapped
    to a batched GEMM (B·hidden_dim matrices of size N×N) and runs on
    Tensor‑Cores in FP16.

5️⃣  Second LayerNorm (hidden_dim) – same Triton kernel as step 1.

6️⃣  Apply the output‑gate (sigmoid) and the final linear projection
    (hidden_dim → dim).  All linear layers use FP16 weights and inputs,
    the final result is cast back to FP32.

The kernel only implements the two LayerNorms; all other heavy work is
delegated to the highly‑optimized PyTorch/ATen kernels, giving a
sub‑millisecond forward pass on an H100 for the target configurations.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm that writes FP16 output
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,          # const *   [total_rows, dim] (float32 or half)
    y_ptr,          #       *   [total_rows, dim] (output, half)
    w_ptr,          # const *   [dim]   (γ)
    b_ptr,          # const *   [dim]   (β)
    total_rows,     # int (runtime)
    dim: tl.constexpr,          # static: number of columns
    eps: tl.constexpr,          # static epsilon
    BLOCK_M: tl.constexpr,      # rows per program
    BLOCK_D: tl.constexpr,      # columns per inner loop
):
    """
    Computes LayerNorm over the last dimension of a 2‑D tensor.
    Mean/variance are accumulated in FP32, the normalized values are
    written as FP16.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # Pass 1: compute per‑row mean and variance
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)      # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                         # 1/√(var+ε)

    # ------------------------------------------------------------------
    # Pass 2: write normalized values
    # ------------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # β

        out = norm * w[None, :] + b[None, :]                # FP32
        out_fp16 = out.to(tl.float16)

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

        input_tensor : torch.Tensor [B, N, N, dim] (float32)
        mask_tensor  : torch.Tensor [B, N, N]     (float32, may be unused)
        weights_dict : {
            "norm.weight":        Tensor[dim],
            "norm.bias":          Tensor[dim],
            "left_proj.weight":   Tensor[hidden_dim, dim],
            "right_proj.weight":  Tensor[hidden_dim, dim],
            "left_gate.weight":   Tensor[hidden_dim, dim],
            "right_gate.weight":  Tensor[hidden_dim, dim],
            "out_gate.weight":    Tensor[hidden_dim, dim],
            "to_out_norm.weight": Tensor[hidden_dim],
            "to_out_norm.bias":   Tensor[hidden_dim],
            "to_out.weight":      Tensor[dim, hidden_dim],
        }
        config_dict  : {
            "dim": int,
            "hidden_dim": int,
            "nomask": bool (optional, default True),
        }

    Returns:
        torch.Tensor [B, N, N, dim] (float32, same device as input)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32 → FP16)
    # --------------------------------------------------------------
    total_rows = B * N * N
    # flatten to [rows, dim] (float32)
    x_flat = input_tensor.view(total_rows, dim).contiguous()
    # allocate fp16 buffer for normalized tensor
    x_norm_h = torch.empty((total_rows, dim), dtype=torch.float16,
                           device=device)

    # launch Triton LayerNorm kernel
    BLOCK_M = 256          # Tunable: rows per program
    BLOCK_D = 64           # Covers dim ≤ 384
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        x_flat,
        x_norm_h,
        weights["norm.weight"].to(device),
        weights["norm.bias"].to(device),
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    # reshape back to [B, N, N, dim] (still fp16)
    x_norm_h = x_norm_h.view(B, N, N, dim)

    # --------------------------------------------------------------
    # 2️⃣ Fuse (proj + gate) for left/right paths
    # --------------------------------------------------------------
    # concat projection and gate weights for a single GEMM
    left_concat_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16, device=device, non_blocking=True)   # [2H, dim]

    right_concat_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16, device=device, non_blocking=True)  # [2H, dim]

    rows = total_rows
    # Linear: (rows, dim) @ (2H, dim).T → (rows, 2H)
    left_concat = F.linear(x_norm_h.view(rows, dim), left_concat_w)   # [rows, 2H]
    right_concat = F.linear(x_norm_h.view(rows, dim), right_concat_w) # [rows, 2H]

    # split into projection and pre‑gate tensors
    left_proj_h      = left_concat[:, :hidden_dim]          # [rows, H]
    left_gate_pre_h  = left_concat[:, hidden_dim:]          # [rows, H]
    right_proj_h     = right_concat[:, :hidden_dim]
    right_gate_pre_h = right_concat[:, hidden_dim:]

    # out‑gate (shared across i/j positions)
    out_gate_w = weights["out_gate.weight"].to(torch.float16, device=device,
                                                non_blocking=True)
    out_gate_h = torch.sigmoid(
        F.linear(x_norm_h.view(rows, dim), out_gate_w)
    )   # [rows, H] (FP16)

    # --------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over hidden dimension)
    # --------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] → [rows, 1] (FP16)
        mask_h = mask_tensor.view(rows, 1).to(torch.float16,
                                              device=device,
                                              non_blocking=True)
        left_proj_h = left_proj_h * mask_h
        right_proj_h = right_proj_h * mask_h

    # --------------------------------------------------------------
    # 4️⃣ Apply sigmoid‑gates
    # --------------------------------------------------------------
    left_gate_h  = torch.sigmoid(left_gate_pre_h)   # [rows, H]
    right_gate_h = torch.sigmoid(right_gate_pre_h) # [rows, H]

    left_h  = left_proj_h * left_gate_h   # [rows, H]
    right_h = right_proj_h * right_gate_h # [rows, H]

    # reshape to 4‑D tensors for the tri‑multiplicative step
    left_h  = left_h.view(B, N, N, hidden_dim)   # [B, N, N, H]
    right_h = right_h.view(B, N, N, hidden_dim)  # [B, N, N, H]

    # --------------------------------------------------------------
    # 5️⃣ Tri‑multiplicative update (batched GEMM) using einsum
    # --------------------------------------------------------------
    # out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    # This avoids the costly permute‑contiguous copies.
    out = torch.einsum('b i k h, b j k h -> b i j h', left_h, right_h)  # [B,N,N,H]

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm (on hidden_dim)
    # --------------------------------------------------------------
    rows_h = B * N * N
    out_flat = out.view(rows_h, hidden_dim).contiguous()
    out_norm_h = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(rows_h, BLOCK_M),)
    layernorm_fp16_kernel[grid2](
        out_flat,
        out_norm_h,
        weights["to_out_norm.weight"].to(device),
        weights["to_out_norm.bias"].to(device),
        rows_h,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    out_norm = out_norm_h.view(B, N, N, hidden_dim)   # FP16

    # --------------------------------------------------------------
    # 7️⃣ Output‑gate and final projection (hidden_dim → dim)
    # --------------------------------------------------------------
    out_gate = out_gate_h.view(B, N, N, hidden_dim)   # FP16
    gated = out_norm * out_gate                        # FP16

    final_h = F.linear(
        gated.view(rows_h, hidden_dim),
        weights["to_out.weight"].to(torch.float16, device=device,
                                    non_blocking=True)
    )   # [rows_h, dim] (FP16)

    # Cast back to FP32 as required by the original module
    final = final_h.view(B, N, N, dim).to(torch.float32)

    return final
