@triton.jit
def ln_out_linear_kernel(
    out_ptr, gate_ptr,
    weight_ptr,
    ln_weight_ptr, ln_bias_ptr,
    total_rows, hidden_dim, dim,
    eps,
    stride_out_R, stride_out_H,
    stride_gate_R, stride_gate_H,
    stride_w_dim, stride_w_hidden,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr, BLOCK_O: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < total_rows

    # First pass: compute mean & var
    sum_ = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_R], dtype=tl.float32)
    for h in range(0, tl.cdiv(hidden_dim, BLOCK_H)):
        offs_h = h * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < hidden_dim
        vals = tl.load(out_ptr + offs_r[:, None] * stride_out_R + offs_h[None,:] * stride_out_H,
                       mask=mask_r[:,None] & mask_h[None,:], other=0.0).to(tl.float32)
        sum_ += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    mean = sum_ / hidden_dim
    var = sum_sq / hidden_dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # Iterate over output dim blocks
    for o in range(0, tl.cdiv(dim, BLOCK_O)):
        offs_o = o * BLOCK_O + tl.arange(0, BLOCK_O)
        mask_o = offs_o < dim

        acc = tl.zeros([BLOCK_R, BLOCK_O], dtype=tl.float32)

        # Loop over hidden dimension blocks again
        for h in range(0, tl.cdiv(hidden_dim, BLOCK_H)):
            offs_h = h * BLOCK_H + tl.arange(0, BLOCK_H)
            mask_h = offs_h < hidden_dim

            # Load out and gate
            out_vals = tl.load(out_ptr + offs_r[:, None] * stride_out_R + offs_h[None,:] * stride_out_H,
                               mask=mask_r[:,None] & mask_h[None,:], other=0.0).to(tl.float32)
            gate_vals = tl.load(gate_ptr + offs_r[:, None] * stride_gate_R + offs_h[None,:] * stride_gate_H,
                               mask=mask_r[:,None] & mask_h[None,:], other=0.0).to(tl.float32)

            # Normalize
            norm = (out_vals - mean[:, None]) * rstd[:, None]

            # Apply LN weight and bias
            gamma = tl.load(ln_weight_ptr + offs_h, mask=mask_h, other=0.0).to(tl.float32)
            beta = tl.load(ln_bias_ptr + offs_h, mask=mask_h, other=0.0).to(tl.float32)
            ln = norm * gamma[None, :] + beta[None, :]

            # Apply out gate
            gated = ln * gate_vals   # shape (BLOCK_R, BLOCK_H)

            # Load final weight slice (BLOCK_H, BLOCK_O)
            w = tl.load(weight_ptr + offs_h[:, None] * stride_w_hidden + offs_o[None, :] * stride_w_dim,
                        mask=mask_h[:, None] & mask_o[None, :], other=0.0)  # shape (BLOCK_H, BLOCK_O)

            acc += tl.dot(gated, w)   # (BLOCK_R, BLOCK_O)

        # Store results
        out_ptr_final = weight_ptr  # not needed, we have separate final output pointer

@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    N,
    stride_a_batch, stride_a_i, stride_a_k, stride_a_h,
    stride_b_batch, stride_b_j, stride_b_k, stride_b_h,
    stride_c_batch, stride_c_i, stride_c_j, stride_c_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    H: tl.constexpr,
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_head = tl.program_id(2)

    batch_idx = pid_head // H
    head_idx = pid_head % H

    a_base = a_ptr + batch_idx * stride_a_batch + head_idx * stride_a_h
    b_base = b_ptr + batch_idx * stride_b_batch + head_idx * stride_b_h
    c_base = c_ptr + batch_idx * stride_c_batch + head_idx * stride_c_h

    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        a = tl.load(a_base + offs_i[:, None] * stride_a_i + offs_k[None, :] * stride_a_k,
                    mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
        b = tl.load(b_base + offs_k[:, None] * stride_b_k + offs_j[None, :] * stride_b_j,
                    mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        acc += tl.dot(a, b)
    # store
    c_ptrs = c_base + offs_i[:, None] * stride_c_i + offs_j[None, :] * stride_c_j
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

@triton.jit
def ln_out_linear_kernel(
    hidden_ptr, gate_ptr,
    weight_ptr,
    ln_weight_ptr, ln_bias_ptr,
    output_ptr,
    total_rows, hidden_dim, dim,
    eps,
    stride_hidden_R, stride_hidden_H,
    stride_gate_R, stride_gate_H,
    stride_w_dim, stride_w_hidden,
    stride_out_R, stride_out_O,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr, BLOCK_O: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < total_rows

    # mean & var
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_R], dtype=tl.float32)

    for h in range(0, tl.cdiv(hidden_dim, BLOCK_H)):
        offs_h = h * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < hidden_dim
        v = tl.load(hidden_ptr + offs_r[:, None] * stride_hidden_R + offs_h[None, :] * stride_hidden_H,
                    mask=mask_r[:, None] & mask_h[None, :], other=0.0).to(tl.float32)
        sum_val += tl.sum(v, axis=1)
        sum_sq += tl.sum(v * v, axis=1)

    mean = sum_val / hidden_dim
    var = sum_sq / hidden_dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # process output dim tiles
    for o in range(0, tl.cdiv(dim, BLOCK_O)):
        offs_o = o * BLOCK_O + tl.arange(0, BLOCK_O)
        mask_o = offs_o < dim
        acc = tl.zeros([BLOCK_R, BLOCK_O], dtype=tl.float32)

        # hidden loop
        for h in range(0, tl.cdiv(hidden_dim, BLOCK_H)):
            offs_h = h * BLOCK_H + tl.arange(0, BLOCK_H)
            mask_h = offs_h < hidden_dim

            # Load hidden and gate
            hidden_vals = tl.load(hidden_ptr + offs_r[:, None] * stride_hidden_R + offs_h[None, :] * stride_hidden_H,
                                  mask=mask_r[:, None] & mask_h[None, :], other=0.0).to(tl.float32)
            gate_vals = tl.load(gate_ptr + offs_r[:, None] * stride_gate_R + offs_h[None, :] * stride_gate_H,
                                mask=mask_r[:, None] & mask_h[None, :], other=0.0).to(tl.float32)

            # Normalize
            norm = (hidden_vals - mean[:, None]) * rstd[:, None]

            # Apply LN weight & bias
            gamma = tl.load(ln_weight_ptr + offs_h, mask=mask_h, other=0.0).to(tl.float32)
            beta = tl.load(ln_bias_ptr + offs_h, mask=mask_h, other=0.0).to(tl.float32)
            ln = norm * gamma[None, :] + beta[None, :]

            # Apply out gate
            gated = ln * gate_vals   # shape (BLOCK_R, BLOCK_H)

            # Load final weight slice (BLOCK_H, BLOCK_O)
            w = tl.load(weight_ptr + offs_h[:, None] * stride_w_hidden + offs_o[None, :] * stride_w_dim,
                        mask=mask_h[:, None] & mask_o[None, :], other=0.0)

            acc += tl.dot(gated, w)  # (BLOCK_R, BLOCK_O)

        # Store to output

"""
Fast Triton implementation of the *outgoing* TriMul operator (AlphaFold‑3).

Algorithm
---------
1️⃣  Row‑wise LayerNorm (FP32 → FP16) on the input tensor.
2️⃣  Five linear projections are fused into three GEMMs:
       * left  = Linear(x)  →  proj + gate (2·H columns)
       * right = Linear(x)  →  proj + gate (2·H columns)
       * out_gate = Linear(x)                (H columns)
   All GEMMs are performed in FP16 using cuBLAS.
3️⃣  Optional pairwise mask (broadcasted on the hidden dimension) and the
    three sigmoid gates.
4️⃣  **TriMul** – batched GEMM over the hidden dimension:
       out[b,i,j,h] = Σₖ left[b,i,k,h]·right[b,j,k,h] .
   Implemented as a custom Triton kernel that avoids the costly
   ``permute``/``contiguous`` copies required by `torch.matmul`.
5️⃣  The second LayerNorm, the output‑gate and the final projection are
    fused into one Triton kernel: for each row (b,i,j) we compute the
    mean/variance, normalize, apply the LN weight & bias, multiply by the
    output‑gate and finally perform a GEMM with the ``to_out`` weight.
    This eliminates the materialisation of the intermediate normalized
    tensor and reduces memory traffic dramatically.
6️⃣  Cast the result back to FP32 for downstream use.

All heavy kernels run in FP16 with float‑32 accumulation, exploiting
Tensor‑Cores on an H100.  The implementation fits the required signature
``custom_kernel`` and works for all test configurations listed in the
problem statement.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1) Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr, y_ptr,      # [total_rows, dim]  (input FP32, output FP16)
    weight_ptr, bias_ptr,  # [dim] (FP32)
    total_rows,
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # -----------------------------------------------------------------
    # First pass – compute mean & variance
    # -----------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        val = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0)
        val_f32 = val.to(tl.float32)

        sum_val += tl.sum(val_f32, axis=1)
        sum_sq  += tl.sum(val_f32 * val_f32, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # -----------------------------------------------------------------
    # Second pass – write normalized values (γ·x̂+β) in FP16
    # -----------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        val = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0)
        val_f32 = val.to(tl.float32)

        norm = (val_f32 - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]       # FP32

        out_fp16 = out.to(tl.float16)
        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

# 2) TriMul kernel  (batched GEMM left @ rightᵀ for each head)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    N,
    stride_a_batch, stride_a_i, stride_a_k, stride_a_h,
    stride_b_batch, stride_b_j, stride_b_k, stride_b_h,
    stride_c_batch, stride_c_i, stride_c_j, stride_c_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    H: tl.constexpr,                     # hidden dimension (compile‑time)
):
    pid_i = tl.program_id(0)      # tile along rows  (i)
    pid_j = tl.program_id(1)      # tile along cols  (j)
    pid_h = tl.program_id(2)      # fused (batch * head)

    # de‑compose into batch index and head index
    batch_idx = pid_h // H
    head_idx  = pid_h % H

    # base pointers for the chosen batch‑head
    a_base = a_ptr + batch_idx * stride_a_batch + head_idx * stride_a_h
    b_base = b_ptr + batch_idx * stride_b_batch + head_idx * stride_b_h
    c_base = c_ptr + batch_idx * stride_c_batch + head_idx * stride_c_h

    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # A: shape (BLOCK_M, BLOCK_K)
        a = tl.load(a_base + offs_i[:, None] * stride_a_i
                              + offs_k[None, :] * stride_a_k,
                    mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Bᵀ: we want shape (BLOCK_K, BLOCK_N)
        b = tl.load(b_base + offs_k[:, None] * stride_b_k
                              + offs_j[None, :] * stride_b_j,
                    mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        acc += tl.dot(a, b)        # FP32 accumulation

    c_ptrs = c_base + offs_i[:, None] * stride_c_i + offs_j[None, :] * stride_c_j
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])

# 3) fused LN (second) + output‑gate + final linear

@triton.jit
def ln_out_linear_kernel(
    hidden_ptr, gate_ptr,                     # [R, H]  (FP16)
    weight_ptr,                               # [dim, H] (FP16)
    ln_weight_ptr, ln_bias_ptr,               # [H] (FP32)
    out_ptr,                                  # [R, dim] (FP16)
    total_rows, hidden_dim, dim,
    eps,
    stride_hidden_R, stride_hidden_H,
    stride_gate_R,   stride_gate_H,
    stride_w_dim,    stride_w_hidden,
    stride_out_R,    stride_out_O,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr, BLOCK_O: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < total_rows

    # -----------------------------------------------------------------
    # Compute per‑row mean / variance (first pass)
    # -----------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_R], dtype=tl.float32)

    for h in range(0, tl.cdiv(hidden_dim, BLOCK_H)):
        offs_h = h * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < hidden_dim
        v = tl.load(hidden_ptr + offs_r[:, None] * stride_hidden_R
                               + offs_h[None, :] * stride_hidden_H,
                    mask=mask_r[:, None] & mask_h[None, :],
                    other=0.0).to(tl.float32)
        sum_val += tl.sum(v, axis=1)
        sum_sq  += tl.sum(v * v, axis=1)

    mean = sum_val / hidden_dim
    var  = sum_sq / hidden_dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # -----------------------------------------------------------------
    # Process output dimension tiles
    # -----------------------------------------------------------------
    for o in range(0, tl.cdiv(dim, BLOCK_O)):
        offs_o = o * BLOCK_O + tl.arange(0, BLOCK_O)
        mask_o = offs_o < dim
        acc = tl.zeros([BLOCK_R, BLOCK_O], dtype=tl.float32)

        for h in range(0, tl.cdiv(hidden_dim, BLOCK_H)):
            offs_h = h * BLOCK_H + tl.arange(0, BLOCK_H)
            mask_h = offs_h < hidden_dim

            # hidden values & gate
            hidden_v = tl.load(hidden_ptr + offs_r[:, None] * stride_hidden_R
                                            + offs_h[None, :] * stride_hidden_H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0).to(tl.float32)
            gate_v   = tl.load(gate_ptr   + offs_r[:, None] * stride_gate_R
                                            + offs_h[None, :] * stride_gate_H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0).to(tl.float32)

            # normalize
            norm = (hidden_v - mean[:, None]) * rstd[:, None]

            # LN weight / bias (γ, β)
            gamma = tl.load(ln_weight_ptr + offs_h, mask=mask_h, other=0.0).to(tl.float32)
            beta  = tl.load(ln_bias_ptr   + offs_h, mask=mask_h, other=0.0).to(tl.float32)

            ln = norm * gamma[None, :] + beta[None, :]           # (BLOCK_R, BLOCK_H)
            gated = ln * gate_v                                   # element‑wise

            # Load weight slice (BLOCK_H, BLOCK_O) – transposed for dot
            w = tl.load(weight_ptr + offs_h[:, None] * stride_w_hidden
                                          + offs_o[None, :] * stride_w_dim,
                        mask=mask_h[:, None] & mask_o[None, :],
                        other=0.0)                               # FP16 → FP32 in dot

            acc += tl.dot(gated, w)                               # (BLOCK_R, BLOCK_O)

        # store the final tile
        out_tile_ptr = out_ptr + offs_r[:, None] * stride_out_R + offs_o[None, :] * stride_out_O
        tl.store(out_tile_ptr, acc.to(tl.float16),
                 mask=mask_r[:, None] & mask_o[None, :])

# 4) Overall forward entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    Input:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
    Output:
        Tensor of shape [B, N, N, dim] (float32, on CUDA)
    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]                 # e.g. 128 or 384
    hidden_dim = config["hidden_dim"]   # always 128 in the tests
    eps = 1e-5
    device = input_tensor.device

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N                        # R = B·N·N

    # --------------------------------------------------------------
    # 1) First LayerNorm (row‑wise) – result in FP16
    # --------------------------------------------------------------
    x_flat = input_tensor.view(total_rows, dim).contiguous()          # [R, dim] (FP32)
    x_norm = torch.empty(total_rows, dim, dtype=torch.float16, device=device)

    BLOCK_M = 256
    BLOCK_D = 128                               # works for dim up to 384
    grid_ln = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # --------------------------------------------------------------
    # 2) Linear projections + gates (fused in three GEMMs)
    # --------------------------------------------------------------
    # left_proj + left_gate  (2·H columns)
    w_left_cat = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)                         # [2·H, dim]
    left_cat = F.linear(x_norm, w_left_cat)     # [R, 2·H]
    left_proj = left_cat[:, :hidden_dim]
    left_gate_pre = left_cat[:, hidden_dim:]
    left_gate = torch.sigmoid(left_gate_pre)

    # right_proj + right_gate (2·H columns)
    w_right_cat = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)
    right_cat = F.linear(x_norm, w_right_cat)
    right_proj = right_cat[:, :hidden_dim]
    right_gate_pre = right_cat[:, hidden_dim:]
    right_gate = torch.sigmoid(right_gate_pre)

    # out_gate (H columns)
    out_gate = torch.sigmoid(
        F.linear(x_norm, weights["out_gate.weight"].to(torch.float16))
    )

    # --------------------------------------------------------------
    # 3) Apply optional pairwise mask and the three sigmoid gates
    # --------------------------------------------------------------
    if not config.get("nomask", True):
        mask_flat = mask_tensor.to(torch.float16).view(total_rows, 1)   # [R,1]
        left_proj  = left_proj * mask_flat
        right_proj = right_proj * mask_flat

    left  = left_proj * left_gate
    right = right_proj * right_gate

    # --------------------------------------------------------------
    # 4) Reshape to 4‑D tensors for the TriMul kernel
    # --------------------------------------------------------------
    left_4d  = left.view(B, N, N, hidden_dim)
    right_4d = right.view(B, N, N, hidden_dim)

    # --------------------------------------------------------------
    # 5) TriMul (batched GEMM) – custom Triton kernel
    # --------------------------------------------------------------
    out_4d = torch.empty_like(left_4d)        # [B,N,N,H] FP16

    BLOCK_M_T = 64
    BLOCK_N_T = 64
    BLOCK_K_T = 32
    grid_mul = (
        triton.cdiv(N, BLOCK_M_T),
        triton.cdiv(N, BLOCK_N_T),
        B * hidden_dim,
    )
    stride_a = left_4d.stride()   # (batch, i, k, h)
    stride_b = right_4d.stride()
    stride_c = out_4d.stride()
    tri_mul_kernel[grid_mul](
        left_4d,
        right_4d,
        out_4d,
        N,
        stride_a[0], stride_a[1], stride_a[2], stride_a[3],
        stride_b[0], stride_b[1], stride_b[2], stride_b[3],
        stride_c[0], stride_c[1], stride_c[2], stride_c[3],
        BLOCK_M=BLOCK_M_T,
        BLOCK_N=BLOCK_N_T,
        BLOCK_K=BLOCK_K_T,
        H=hidden_dim,
    )

    # --------------------------------------------------------------
    # 6) Fuse second LayerNorm, output gate and final projection
    # --------------------------------------------------------------
    out_flat = out_4d.view(total_rows, hidden_dim).contiguous()      # [R, H]
    out_gate_flat = out_gate.view(total_rows, hidden_dim).contiguous()

    # final linear weight  [dim, H]  (FP16)
    w_to_out = weights["to_out.weight"].to(torch.float16)

    final_flat = torch.empty(total_rows, dim, dtype=torch.float16, device=device)

    BLOCK_R = 128
    BLOCK_H = 32
    BLOCK_O = 64
    grid_final = (triton.cdiv(total_rows, BLOCK_R),)
    ln_out_linear_kernel[grid_final](
        out_flat,
        out_gate_flat,
        w_to_out,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        final_flat,
        total_rows,
        hidden_dim,
        dim,
        eps,
        # strides
        out_flat.stride(0),  out_flat.stride(1),
        out_gate_flat.stride(0), out_gate_flat.stride(1),
        w_to_out.stride(0), w_to_out.stride(1),
        final_flat.stride(0), final_flat.stride(1),
        # block sizes
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        BLOCK_O=BLOCK_O,
    )

    # --------------------------------------------------------------
    # 7) Reshape to [B, N, N, dim] and cast to float32
    # --------------------------------------------------------------
    final = final_flat.view(B, N, N, dim).to(torch.float32)

    return final
