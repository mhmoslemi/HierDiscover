@triton.jit
def batched_matmul_kernel(
     a_ptr, b_ptr, c_ptr,
     batch_stride_a, batch_stride_b, batch_stride_c,
     N,
     BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)  # each batch dimension (B*H)
    pid_m = tl.program_id(1)  # block row
    pid_n = tl.program_id(2)  # block col

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    a_ptrs = a_ptr + batch_id * batch_stride_a + offs_m[:, None] * N + tl.arange(0, BLOCK_K)[None, :]
    b_ptrs = b_ptr + batch_id * batch_stride_b + tl.arange(0, BLOCK_K)[:, None] * N + offs_n[None, :]

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)  # accumulate in fp32 for precision

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (tl.arange(0, BLOCK_K) < N - k*BLOCK_K), other=0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask=(tl.arange(0, BLOCK_K) < N - k*BLOCK_K)[:, None] & mask_n[None, :], other=0.0).to(tl.float32)
        acc += tl.dot(a, b)
        # increment pointers
        a_ptrs += BLOCK_K
        b_ptrs += BLOCK_K * N
    # write back
    c = acc.to(tl.float16)
    c_ptrs = c_ptr + batch_id * batch_stride_c + offs_m[:, None] * N + offs_n[None, :]
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])

```
@triton.jit
def batched_gemm_transpose_kernel(
    a_ptr, b_ptr, c_ptr,
    stride_batch_a, stride_batch_b, stride_batch_c,
    N,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    eps: tl.constexpr,
):
    # program ids
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # pointers to start of batch
    a_batch_ptr = a_ptr + batch_id * stride_batch_a
    b_batch_ptr = b_ptr + batch_id * stride_batch_b
    c_batch_ptr = c_ptr + batch_id * stride_batch_c

    # accumulate in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # compute pointers
        # A: row index offs_m, col index offs_k
        a_ptrs = a_batch_ptr + offs_m[:, None] * N + offs_k[None, :]
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # B_T: row index offs_k, col index offs_n
        b_ptrs = b_batch_ptr + offs_k[:, None] * N + offs_n[None, :]
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        # dot
        acc += tl.dot(a, b)

    # write result
    c = acc.to(tl.float16)
    c_ptrs = c_batch_ptr + offs_m[:, None] * N + offs_n[None, :]
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])
```

"""
custom_kernel – Triton‑accelerated outgoing TriMul (Triangular Multiplicative Update)

Algorithm
---------
1. **Row‑wise LayerNorm** on the input [B, N, N, C] using a custom Triton
   kernel.  The normalized tensor is stored in fp16 to keep the whole
   pipeline in half precision.

2. **Fuse five linear layers** (left_proj, right_proj, left_gate, right_gate,
   out_gate) into a single GEMM:
      X_norm (R×C) @ W_concat.T (C×5H)  →  (R×5H)   with R = B·N·N.
   The result is sliced into the three projected tensors and three gates.
   All tensors stay in fp16.

3. **Mask** (if present) is applied to the two projected tensors.

4. **Gating** – element‑wise multiplication with sigmoid gates.
   The tensors are reshaped to [B, N, N, H] (H = hidden_dim).

5. **Batched matrix multiplication** implementing the O(N³) update:
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h].
   This is a batched GEMM `left @ rightᵀ` for every hidden channel.
   A custom Triton kernel computes all B·H matmuls in a single launch,
   avoiding the many small cuBLAS kernels of `torch.bmm`.

6. **Second LayerNorm** over the hidden dimension, again with a Triton
   kernel.

7. **Output gating & projection** – the gated hidden tensor is multiplied
   by the output gate, linearly projected back to the original dimension,
   and cast to float32.

All heavy arithmetic runs in fp16 (accumulated in fp32 for numerical
stability).  The only explicit Triton kernels are the two LayerNorms and
the batched matrix multiplication.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def layernorm_kernel(
    x_ptr,          # const   *[total_rows, dim]   (fp32 input)
    y_ptr,          # mutable *[total_rows, dim]   (fp16 output)
    weight_ptr,     # const   *[dim]               (fp32 γ)
    bias_ptr,       # const   *[dim]               (fp32 β)
    total_rows,     # int (runtime)
    dim,            # int (compile‑time)
    eps,            # float (compile‑time)
    BLOCK_M: tl.constexpr,   # rows processed per program
    BLOCK_D: tl.constexpr,   # columns processed per inner loop
):
    """
    Row‑wise LayerNorm.
    Computes:
        y = ((x - μ) / sqrt(var + eps)) * γ + β
    and writes y in fp16.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ---------- 1st pass: mean & variance ---------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        x = tl.load(
            x_ptr + offs_m[:, None] * dim + offs_d[None, :],
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        )
        x_f32 = x.to(tl.float32)
        sum_val += tl.sum(x_f32, axis=1)
        sum_sq  += tl.sum(x_f32 * x_f32, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)        # 1 / sqrt(var + eps)

    # ---------- 2nd pass: write normalized result -------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        x = tl.load(
            x_ptr + offs_m[:, None] * dim + offs_d[None, :],
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        )
        x_f32 = x.to(tl.float32)

        # (x - μ) / sqrt(var+eps)
        norm = (x_f32 - mean[:, None]) * rstd[:, None]

        # apply learned scale & bias
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ (fp32)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β (fp32)

        out = norm * w[None, :] + b[None, :]      # fp32
        out_fp16 = out.to(tl.float16)

        tl.store(
            y_ptr + offs_m[:, None] * dim + offs_d[None, :],
            out_fp16,
            mask=mask_m[:, None] & mask_d[None, :],
        )


@triton.jit
def batched_matmul_kernel(
    a_ptr,              # const  *[batch, N, N]   (fp16)
    b_ptr,              # const  *[batch, N, N]   (fp16)  (already transposed)
    c_ptr,              # mutable*[batch, N, N]   (fp16)
    stride_batch_a,     # int  (row‑major stride to next batch for A)
    stride_batch_b,     # int
    stride_batch_c,     # int
    N,                  # int (matrix size)
    BLOCK_M: tl.constexpr,   # tile size for rows of C
    BLOCK_N: tl.constexpr,   # tile size for cols of C
    BLOCK_K: tl.constexpr,   # reduction tile size
):
    """
    Batched GEMM computing   C[b] = A[b] @ B[b]ᵀ
    where each A and B have shape (N, N) in row‑major layout.
    The kernel processes all batches in one launch.
    """
    batch_id = tl.program_id(0)
    pid_m    = tl.program_id(1)
    pid_n    = tl.program_id(2)

    # offsets inside the N×N matrix
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # pointers to the beginning of the current batch
    a_batch = a_ptr + batch_id * stride_batch_a
    b_batch = b_ptr + batch_id * stride_batch_b
    c_batch = c_ptr + batch_id * stride_batch_c

    # accumulator (use FP32 for accuracy)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # A[batch, i, k]
        a = tl.load(
            a_batch + offs_m[:, None] * N + offs_k[None, :],
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # B_transposed[batch, k, j]  (i.e. original rightᵀ)
        b = tl.load(
            b_batch + offs_k[:, None] * N + offs_n[None, :],
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # dot product over the K‑tile
        acc += tl.dot(a.to(tl.float32), b.to(tl.float32))

    # write result back as FP16
    c = acc.to(tl.float16)
    tl.store(
        c_batch + offs_m[:, None] * N + offs_n[None, :],
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no backward).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor of shape [B, N, N, C] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (bool/float)
        - weights_dict : mapping str → torch.Tensor (float32)
        - config_dict  : {
                "dim"       : int,
                "hidden_dim": int,
                "nomask"    : bool (optional)
          }

    Returns
    -------
    torch.Tensor of shape [B, N, N, C] (float32) on the same device.
    """
    # --------------------------------------------------------------
    # Unpack
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    device = input_tensor.device
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N                         # R = B·N·N

    # --------------------------------------------------------------
    # 1) First LayerNorm (row‑wise) – output in fp16
    # --------------------------------------------------------------
    # flatten to [R, dim]
    x_flat = input_tensor.view(total_rows, dim).contiguous()
    x_norm = torch.empty(total_rows, dim, dtype=torch.float16, device=device)

    # launch parameters – dim ≤ 384 so BLOCK_D = 64 is a good fit
    BLOCK_M_LN = 256
    BLOCK_D_LN = 64
    grid_ln1 = (triton.cdiv(total_rows, BLOCK_M_LN),)
    layernorm_kernel[grid_ln1](
        x_flat,
        x_norm,
        weights["norm.weight"],   # γ  (fp32)
        weights["norm.bias"],     # β  (fp32)
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M_LN,
        BLOCK_D=BLOCK_D_LN,
    )

    # --------------------------------------------------------------
    # 2) Fuse five linear layers (projections + gates) into one GEMM
    # --------------------------------------------------------------
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
            weights["out_gate.weight"],    # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)                     # [5H, dim]

    # fused projection (fp16)
    proj = F.linear(x_norm, w_concat)       # [R, 5H] (fp16)

    H = hidden_dim
    left_proj   = proj[:, 0:H]               # [R, H]
    right_proj  = proj[:, H:2 * H]
    left_gate_p = proj[:, 2 * H:3 * H]
    right_gate_p= proj[:, 3 * H:4 * H]
    out_gate_p  = proj[:, 4 * H:5 * H]

    # --------------------------------------------------------------
    # 3) Gates (sigmoid) – still fp16
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate_p)
    right_gate = torch.sigmoid(right_gate_p)
    out_gate   = torch.sigmoid(out_gate_p)

    # --------------------------------------------------------------
    # 4) Optional mask (broadcasted over hidden dim)
    # --------------------------------------------------------------
    if not nomask:
        mask_h = mask_tensor.to(torch.float16).view(total_rows, 1)
        left_proj  = left_proj * mask_h
        right_proj = right_proj * mask_h

    # --------------------------------------------------------------
    # 5) Apply gates and reshape to [B, N, N, H]
    # --------------------------------------------------------------
    left  = left_proj * left_gate            # [R, H]
    right = right_proj * right_gate          # [R, H]

    left  = left.view(B, N, N, H).contiguous()
    right = right.view(B, N, N, H).contiguous()
    out_gate = out_gate.view(B, N, N, H).contiguous()   # will be used later

    # --------------------------------------------------------------
    # 6) Batched GEMM: out = left @ rightᵀ  (TriMul core)
    # --------------------------------------------------------------
    # reshape to [B*H, N, N]   – A rows = i, cols = k
    batch = B * H
    left_batch = left.permute(0, 3, 1, 2).contiguous().view(batch, N, N)          # (i,k)
    # rightᵀ → rows = k, cols = j
    right_batch_T = right.permute(0, 3, 2, 1).contiguous().view(batch, N, N)     # (k,j)

    out_batch = torch.empty(batch, N, N, dtype=torch.float16, device=device)

    # kernel launch parameters (tuned for N up to 1024)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid_mul = (
        batch,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    batched_matmul_kernel[grid_mul](
        left_batch,
        right_batch_T,
        out_batch,
        left_batch.stride(0),        # stride to next batch for A
        right_batch_T.stride(0),     # stride to next batch for B
        out_batch.stride(0),         # stride to next batch for C
        N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to [B, N, N, H]
    out = out_batch.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # (B,N,N,H)

    # --------------------------------------------------------------
    # 7) Second LayerNorm over the hidden dimension
    # --------------------------------------------------------------
    out_flat = out.view(total_rows, H).contiguous()
    out_norm = torch.empty_like(out_flat, dtype=torch.float16)

    grid_ln2 = (triton.cdiv(total_rows, BLOCK_M_LN),)
    layernorm_kernel[grid_ln2](
        out_flat,
        out_norm,
        weights["to_out_norm.weight"],   # γ (fp32)
        weights["to_out_norm.bias"],     # β (fp32)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M_LN,
        BLOCK_D=BLOCK_D_LN,
    )
    out_norm = out_norm.view(B, N, N, H)   # (B,N,N,H) fp16

    # --------------------------------------------------------------
    # 8) Apply output gate and final projection back to `dim`
    # --------------------------------------------------------------
    out_norm = out_norm * out_gate                               # element‑wise fp16
    out_norm_flat = out_norm.view(total_rows, H).contiguous()    # [R, H]

    final_half = F.linear(
        out_norm_flat,
        weights["to_out.weight"].to(torch.float16)               # [dim, H]
    )                                                            # [R, dim] fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)

    return final
