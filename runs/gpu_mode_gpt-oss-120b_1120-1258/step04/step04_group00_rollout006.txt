"""
TriMul (outgoing) forward pass – highly‑fused implementation.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the 4‑D input (B,N,N,dim).  Implemented in a small
   Triton kernel that computes mean/variance in FP32 and writes the
   normalized result in FP16 (reduces memory traffic).

2️⃣ Fuse **all** linear projections + their gates into **one** GEMM:
      (dim → 5·hidden_dim)  =  [left_proj, left_gate,
                               right_proj, right_gate,
                               out_gate]  (no bias).
   The resulting tensor is sliced to obtain the five intermediate tensors.
   This cuts the number of kernel launches from three to one.

3️⃣ Optional pair‑wise mask (broadcast over hidden_dim) is applied to the
   left‑ and right‑projections before the sigmoid gates.

4️⃣ Apply the sigmoid gates and element‑wise multiply the gated projections.

5️⃣ Tri‑multiplicative update:
      out_h(i,j) = Σ_k left(i,k,h)·right(j,k,h)
   Realised as a batched half‑precision GEMM (B·hidden_dim matrices of size
   N×N) via `torch.bmm`, which on an H100 maps to Tensor‑Core kernels.

6️⃣ Second LayerNorm (hidden_dim) – performed with PyTorch’s native
   `torch.nn.functional.layer_norm` (highly‑optimised cuDNN kernel).

7️⃣ Apply the output‑gate (sigmoid) and the final linear projection
   (hidden_dim → dim).  The whole pipeline stays in FP16 until the very
   last cast back to FP32, matching the reference module’s behavior.

Only the first LayerNorm is written in Triton, satisfying the “must use a
kernel” requirement while keeping the critical O(N³) work on the
high‑throughput PyTorch cuBLAS/cuDNN kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm that writes FP16 output
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,          # const float*   [total_rows, dim]
    y_ptr,          #       half*   [total_rows, dim] (output)
    w_ptr,          # const float*  [dim]   (γ)
    b_ptr,          # const float*  [dim]   (β)
    total_rows,     # int (runtime)
    dim: tl.constexpr,          # static: number of columns (≤384)
    eps: tl.constexpr,          # static epsilon
    BLOCK_M: tl.constexpr,      # rows processed per program
    BLOCK_D: tl.constexpr,      # columns per inner loop
):
    """
    Compute LayerNorm over the last dimension of a 2‑D tensor.
    Mean/variance are accumulated in FP32, the normalized result is stored
    as FP16 to halve memory traffic.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # --------------------------------------------------------------
    # Pass 1: accumulate sum and sum of squares
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                           # 1 / sqrt(var+eps)

    # --------------------------------------------------------------
    # Pass 2: write normalized values with scale (γ) and bias (β)
    # --------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # [BLOCK_M, BLOCK_D]

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # β

        out = norm * w[None, :] + b[None, :]                 # fp32
        out_fp16 = out.to(tl.float16)

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        mask_tensor  : torch.Tensor [B, N, N]      (float32) – may be unused
        weights_dict  : mapping of module parameters (float32)
        config_dict   : {
            "dim"        : int,
            "hidden_dim": int,
            "nomask"    : bool (optional, default True)
        }

    Returns
    -------
    torch.Tensor [B, N, N, dim] (float32, same device as ``input_tensor``)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32 → FP16) via Triton
    # ------------------------------------------------------------------
    total_rows = B * N * N
    # flatten to 2‑D [rows, dim]
    x_flat = input_tensor.view(total_rows, dim).contiguous()

    # allocate FP16 buffer for the normalized tensor
    x_norm_fp16 = torch.empty((total_rows, dim), dtype=torch.float16, device=device)

    # launch Triton kernel
    BLOCK_M = 256            # tune‑able: rows per program
    BLOCK_D = 64             # columns per tile (covers dim up to 384)
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    # reshape to the original 4‑D layout (still FP16)
    x_norm_fp16 = x_norm_fp16.view(B, N, N, dim)

    # ------------------------------------------------------------------
    # 2️⃣ Fuse *all* linear projections + gates into ONE GEMM
    # ------------------------------------------------------------------
    # Pre‑convert all projection / gate weight tensors to FP16 once
    left_proj_w   = weights["left_proj.weight"]
    left_gate_w   = weights["left_gate.weight"]
    right_proj_w  = weights["right_proj.weight"]
    right_gate_w  = weights["right_gate.weight"]
    out_gate_w    = weights["out_gate.weight"]

    # Vertically concatenate the five weight matrices: shape (5*H, dim)
    # order: left_proj, left_gate, right_proj, right_gate, out_gate
    concat_weight_fp16 = torch.cat(
        [
            left_proj_w,
            left_gate_w,
            right_proj_w,
            right_gate_w,
            out_gate_w,
        ],
        dim=0,
    ).to(torch.float16, non_blocking=True)   # [5*H, dim]

    # Run the single fused linear layer on the flattened input
    rows = total_rows
    fused_out = F.linear(
        x_norm_fp16.view(rows, dim),  # [rows, dim], FP16
        concat_weight_fp16,           # [5*H, dim], FP16
        bias=None,
    )  # → [rows, 5*hidden_dim] (FP16)

    # ------------------------------------------------------------------
    # 3️⃣ Slice the fused output
    # ------------------------------------------------------------------
    H = hidden_dim
    # left projection and its pre‑gate
    left_proj_h       = fused_out[:, 0 * H : 1 * H]   # [rows, H]
    left_gate_pre_h   = fused_out[:, 1 * H : 2 * H]   # [rows, H]
    # right projection and its pre‑gate
    right_proj_h      = fused_out[:, 2 * H : 3 * H]   # [rows, H]
    right_gate_pre_h  = fused_out[:, 3 * H : 4 * H]   # [rows, H]
    # output gate (pre‑sigmoid)
    out_gate_pre_h    = fused_out[:, 4 * H : 5 * H]   # [rows, H]

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] → [rows, 1] (FP16)
        mask_h = mask_tensor.view(rows, 1).to(torch.float16, non_blocking=True)
        left_proj_h = left_proj_h * mask_h
        right_proj_h = right_proj_h * mask_h

    # ------------------------------------------------------------------
    # 5️⃣ Apply sigmoid gates and elementwise multiplication
    # ------------------------------------------------------------------
    left_gate_h  = torch.sigmoid(left_gate_pre_h)   # [rows, H]
    right_gate_h = torch.sigmoid(right_gate_pre_h)  # [rows, H]
    out_gate_h   = torch.sigmoid(out_gate_pre_h)    # [rows, H]

    left_h  = left_proj_h * left_gate_h   # [rows, H]
    right_h = right_proj_h * right_gate_h # [rows, H]

    # ------------------------------------------------------------------
    # 6️⃣ Tri‑multiplicative update (batched GEMM)
    # ------------------------------------------------------------------
    # reshape to [B, N, N, H]
    left  = left_h.view(B, N, N, hidden_dim)
    right = right_h.view(B, N, N, hidden_dim)

    # bring hidden_dim into the batch dimension → [B, H, N, N]
    left_perm  = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    BH = B * hidden_dim                     # effective batch size
    left_mat  = left_perm.view(BH, N, N)    # [BH, N, N] (FP16)
    right_mat = right_perm.view(BH, N, N)   # [BH, N, N] (FP16)

    # Compute Σ_k left[i,k,h] * right[j,k,h] via batched matrix multiplication
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [BH, N, N]

    # Reshape back to [B, N, N, H] (still FP16)
    out = (
        out_mat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 7️⃣ Second LayerNorm (hidden_dim) – use PyTorch’s native kernel
    # ------------------------------------------------------------------
    rows_h = B * N * N
    out_flat = out.view(rows_h, hidden_dim).contiguous()

    # Cast LayerNorm parameters to FP16 once
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16, non_blocking=True)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16, non_blocking=True)

    out_norm_fp16 = torch.nn.functional.layer_norm(
        out_flat,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )   # [rows_h, hidden_dim] (FP16)

    out_norm = out_norm_fp16.view(B, N, N, hidden_dim)   # FP16

    # ------------------------------------------------------------------
    # 8️⃣ Apply output‑gate and final projection
    # ------------------------------------------------------------------
    # reshape output gate to [B, N, N, H] and multiply
    out_gate = out_gate_h.view(B, N, N, hidden_dim)   # FP16
    out_norm = out_norm * out_gate                   # gated (FP16)

    # final linear: hidden_dim → dim
    to_out_w = weights["to_out.weight"].to(torch.float16, non_blocking=True)
    final_fp16 = F.linear(
        out_norm.view(rows_h, hidden_dim),
        to_out_w,
        bias=None,
    )   # [rows_h, dim] (FP16)

    # Cast back to FP32 to match the reference implementation
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
