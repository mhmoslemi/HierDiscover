"""
TriMul (outgoing) – Triton‑accelerated forward pass.

Algorithm (fast, low‑memory):
1. Row‑wise LayerNorm on the input 4‑D tensor X ∈ [B,N,N,C] (C = dim).
   Implemented with a custom Triton kernel that writes the normalized
   tensor in fp16.

2. Fuse the five linear layers (left_proj, right_proj, left_gate,
   right_gate, out_gate) into a single half‑precision GEMM:
   X_norm @ W_concat.T  →  [B·N·N, 5·H]   (H = hidden_dim).

3. Slice the concatenated result, apply the optional mask (broadcast),
   compute the three sigmoids and obtain the gated projected tensors
   `left` and `right` (shape [B,N,N,H]) – all still in fp16.

4. Triangular multiplicative update (the heart of the operator):
   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d].
   This is a batched matrix multiplication:
       (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N)
   performed with torch.bmm which internally uses Tensor‑Core GEMM.

5. Second LayerNorm (over the hidden dimension H) – again the Triton
   kernel, writing fp16.

6. Multiply by the output‑gate, project back to the original dimension
   with a final half‑precision linear layer, and cast the result to fp32.

All heavy arithmetic runs in fp16, maximizing Tensor‑Core throughput on
an H100.  The only custom Triton code needed is the row‑wise LayerNorm,
which is the part that torch.compile cannot fuse automatically.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def layernorm_kernel(
    x_ptr,          # const   *[total_rows, dim]   (fp32 or fp16)
    y_ptr,          # mutable *[total_rows, dim]   (fp16 output)
    weight_ptr,     # const   *[dim]               (fp32)
    bias_ptr,       # const   *[dim]               (fp32)
    total_rows,     # int (runtime)
    dim,            # int (compile‑time)
    eps,            # float (compile‑time)
    BLOCK_M: tl.constexpr,   # rows processed per program
    BLOCK_D: tl.constexpr,   # columns processed per inner loop
):
    """Row‑wise LayerNorm (μ, σ²) → (x‑μ)·rsqrt(σ²+eps)·γ+β  (fp16 out)."""
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ---- 1st pass: compute mean and variance ---------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0)
        vals_f32 = vals.to(tl.float32)
        sum_val += tl.sum(vals_f32, axis=1)
        sum_sq  += tl.sum(vals_f32 * vals_f32, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)          # 1 / sqrt(var + eps)

    # ---- 2nd pass: write normalized result ----------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0)
        vals_f32 = vals.to(tl.float32)

        # (x - μ) * rstd
        norm = (vals_f32 - mean[:, None]) * rstd[:, None]

        # γ and β (float32)
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)

        out = norm * w[None, :] + b[None, :]               # fp32
        out_fp16 = out.to(tl.float16)

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor, [B, N, N] (bool/float) – may be None
        - weights_dict : dict of torch.Tensor (float32) containing
                         all model parameters
        - config_dict  : {"dim": int,
                         "hidden_dim": int,
                         "nomask": bool (optional)}

    Returns
    -------
    torch.Tensor
        Output tensor, shape [B, N, N, dim] (float32, same device)
    """
    # ------------------------------------------------------------------
    # 0) unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5
    device = input_tensor.device

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N                     # R = B·N·N

    # ------------------------------------------------------------------
    # 1) First LayerNorm (fp16 output)
    # ------------------------------------------------------------------
    x_flat = input_tensor.view(total_rows, dim).contiguous()
    x_norm = torch.empty(total_rows, dim,
                         dtype=torch.float16,
                         device=device)

    # Triton launch config – tuned for dim ≤ 384
    BLOCK_M = 256
    BLOCK_D = 64
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 2) Fuse the five linear layers (proj + 3 gates) into one GEMM
    # ------------------------------------------------------------------
    # Concatenate weight matrices -> shape [5*H, dim]
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],    # [H, dim]
            weights["right_proj.weight"],   # [H, dim]
            weights["left_gate.weight"],    # [H, dim]
            weights["right_gate.weight"],   # [H, dim]
            weights["out_gate.weight"],     # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)

    # Single half‑precision linear: [R, dim] @ [dim, 5*H]ᵀ → [R, 5*H]
    out_concat = F.linear(x_norm, w_concat)          # fp16

    H = hidden_dim
    left_proj_raw = out_concat[:, 0:H]                # [R, H]
    right_proj_raw = out_concat[:, H:2*H]
    left_gate_raw = out_concat[:, 2*H:3*H]
    right_gate_raw = out_concat[:, 3*H:4*H]
    out_gate_raw = out_concat[:, 4*H:5*H]

    # Gates – still fp16
    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate = torch.sigmoid(out_gate_raw)

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcasted over hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape [B,N,N] → flatten to [R,1] as half‑precision
        mask_h = mask_tensor.to(torch.float16).view(total_rows, 1)
        left_proj_raw = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    # ------------------------------------------------------------------
    # 4) Apply gates and reshape to [B,N,N,H]
    # ------------------------------------------------------------------
    # left = left_proj * left_gate ; right = right_proj * right_gate
    left = (left_proj_raw * left_gate).view(B, N, N, H)
    right = (right_proj_raw * right_gate).view(B, N, N, H)
    out_gate = out_gate.view(B, N, N, H)   # keep for later

    # ------------------------------------------------------------------
    # 5) Triangular multiplicative update:  out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    # Bring hidden dim into the batch dimension: (B*H, N, N)
    left_batch = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_batch = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Batched GEMM (Tensor‑Core):  (B*H, N, N) @ (B*H, N, N)ᵀ
    out_batch = torch.bmm(left_batch, right_batch.transpose(1, 2))

    # Back to [B, N, N, H]
    out = out_batch.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6) Second LayerNorm over hidden dimension (fp16)
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, H).contiguous()
    out_ln = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_ln,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    out_ln = out_ln.view(B, N, N, H)               # fp16

    # ------------------------------------------------------------------
    # 7) Apply output gate (in‑place) then final projection back to `dim`
    # ------------------------------------------------------------------
    out_ln.mul_(out_gate)                          # fp16, in‑place

    # Final linear: [R, H] @ [H, dim]ᵀ → [R, dim]
    final_half = F.linear(
        out_ln.view(total_rows, H),
        weights["to_out.weight"].to(torch.float16)   # [dim, H]
    )                                                # fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
