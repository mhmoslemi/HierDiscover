"""
Highly‑optimized “outgoing” TriMul forward pass (AlphaFold‑3).

Algorithm (see paper)
---------------------
1️⃣ Row‑wise LayerNorm over the input channel dimension `C`.  
   → fp32 accumulation, fp16 output (native Triton kernel).

2️⃣ Fuse the five linear projections (left/right + three gates) into a single
   half‑precision GEMM by concatenating the weight matrices.

3️⃣ (Optional) Apply the pair‑wise mask to the left/right projections
   before the gates.  Mask is broadcasted over the hidden dimension `H`.

4️⃣ Apply sigmoid gates and element‑wise gating (still fp16).

5️⃣ Triangular multiplicative update (outgoing version)  
   `out_h[i, j] = Σ_k left_h[i, k] * right_h[j, k]`  
   Implemented as a batched half‑precision GEMM `left @ rightᵀ`
   using `torch.bmm` (cuBLAS Tensor‑Core kernels).

6️⃣ Row‑wise LayerNorm over the hidden dimension `H` (fp32 reduce, fp16 output).

7️⃣ Multiply by the output gate, then a final half‑precision linear projection
   back to the original channel dimension `C`.

8️⃣ Cast the result to float32 and return.

Both LayerNorms are written in Triton; the dense linear algebra uses
high‑performance Torch kernels that run on H100 Tensor‑Cores.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (fp32 statistics, fp16 output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,          # *const  void*   [total_rows, dim]   (fp16 or fp32)
    y_ptr,          # *mut    fp16*   [total_rows, dim]   (output)
    weight_ptr,     # *const  fp32*   [dim]
    bias_ptr,       # *const  fp32*   [dim]
    total_rows,     # i32
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # -----------------------------------------------------------------
    # 1. Compute mean & variance in fp32
    # -----------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim

        # pointer to element (row, col)
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]   # [BLOCK_M, BLOCK_D]
        # Load as fp16 or fp32, then cast to fp32 for accumulation
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)          # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)   # [BLOCK_M]

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                  # 1/sqrt(var+eps)

    # -----------------------------------------------------------------
    # 2. Write normalized output (fp16)
    # -----------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]   # fp32

        # γ & β are fp32
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)

        out_fp16 = (norm * w[None, :] + b[None, :]).to(tl.float16)

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
# Custom TriMul kernel
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Outgoing TriMul forward pass.

    Args:
        data (tuple):
            - input_tensor : torch.Tensor [B, N, N, C]   (float32)
            - mask_tensor  : torch.Tensor [B, N, N]     (bool/float, optional)
            - weights_dict : mapping name → torch.Tensor (float32)
            - config_dict  : {"dim": C, "hidden_dim": H, "nomask": bool}
    Returns:
        torch.Tensor [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack & basic shapes
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    assert dtype == torch.float32, "Input must be FP32"
    dim = config["dim"]               # C
    hidden_dim = config["hidden_dim"] # H
    eps = 1e-5
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N            # flattened batch*seq*seq dimension

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (fp16 output)
    # ------------------------------------------------------------------
    x_flat = input_tensor.contiguous().view(total_rows, dim)  # FP32
    x_norm_fp16 = torch.empty(total_rows, dim,
                              dtype=torch.float16,
                              device=device)

    BLOCK_M = 256
    BLOCK_D = 64
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fuse the five linear projections (left/right + three gates)
    # ------------------------------------------------------------------
    # All projection weight matrices are of shape [hidden_dim, dim] (no bias)
    w_left    = weights["left_proj.weight"]      # [H, C]
    w_right   = weights["right_proj.weight"]     # [H, C]
    w_lgate   = weights["left_gate.weight"]      # [H, C]
    w_rgate   = weights["right_gate.weight"]     # [H, C]
    w_outgate = weights["out_gate.weight"]       # [H, C]

    # Concatenate along out‑features dimension → shape (5*H, C)
    concat_w = torch.cat([w_left, w_right, w_lgate,
                          w_rgate, w_outgate], dim=0)
    concat_w = concat_w.to(torch.float16).contiguous()

    # Fused GEMM: (total_rows, C) @ (C, 5*H)^T → (total_rows, 5*H)
    proj_all = F.linear(x_norm_fp16, concat_w)   # fp16 output

    # Split back
    left_proj_fp16, right_proj_fp16, left_gate_fp16, \
        right_gate_fp16, out_gate_fp16 = torch.split(
            proj_all, hidden_dim, dim=1
        )   # each [total_rows, H] (fp16)

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape [B,N,N] → [total_rows, 1] (float16)
        mask_fp16 = mask_tensor.view(total_rows, 1).to(torch.float16)
        left_proj_fp16  = left_proj_fp16 * mask_fp16
        right_proj_fp16 = right_proj_fp16 * mask_fp16

    # ------------------------------------------------------------------
    # 4️⃣ Apply sigmoid gates & element‑wise gating
    # ------------------------------------------------------------------
    left_gate   = torch.sigmoid(left_gate_fp16)        # [total_rows, H]
    right_gate  = torch.sigmoid(right_gate_fp16)
    out_gate    = torch.sigmoid(out_gate_fp16)

    left_fp16   = left_proj_fp16 * left_gate
    right_fp16  = right_proj_fp16 * right_gate

    # Reshape to original 4‑D layout
    left  = left_fp16.view(B, N, N, hidden_dim)   # [B, N, N, H]
    right = right_fp16.view(B, N, N, hidden_dim)  # [B, N, N, H]
    out_gate = out_gate.view(B, N, N, hidden_dim) # [B, N, N, H]

    # ------------------------------------------------------------------
    # 5️⃣ Triangular multiplicative update (batched GEMM)
    #    out_h[i, j] = Σ_k left_h[i, k] * right_h[j, k]
    # ------------------------------------------------------------------
    # Move hidden dim into batch dimension for a single bmm call:
    #   left  : [B, N, N, H] → [B*H, N, N]  (i as rows, k as cols)
    #   right : same layout, but we need rightᵀ on the last two dims.
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    BH = B * hidden_dim
    left_mat  = left_perm.view(BH, N, N)                 # [BH, N, N]
    right_mat = right_perm.view(BH, N, N)                # [BH, N, N]

    # Batched matmul: left @ rightᵀ  →  [BH, N, N]   (fp16)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Back to [B, N, N, H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm over hidden_dim (fp16 output)
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, hidden_dim)          # [total_rows, H]
    out_norm_fp16 = torch.empty_like(out_flat,
                                     dtype=torch.float16,
                                     device=device)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid2](
        out_flat,               # fp16 input (cast to fp32 inside)
        out_norm_fp16,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    out_norm = out_norm_fp16.view(B, N, N, hidden_dim)   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 7️⃣ Final gate + linear projection back to C
    # ------------------------------------------------------------------
    gated = out_norm * out_gate                # fp16 element‑wise

    to_out_w_fp16 = weights["to_out.weight"].to(torch.float16).contiguous()
    gated_flat = gated.view(total_rows, hidden_dim)      # [total_rows, H]
    final_fp16 = F.linear(gated_flat, to_out_w_fp16)    # [total_rows, C]

    # ------------------------------------------------------------------
    # 8️⃣ Cast to FP32 and reshape to original layout
    # ------------------------------------------------------------------
    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final
