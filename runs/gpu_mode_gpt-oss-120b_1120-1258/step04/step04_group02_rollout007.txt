def gating_mask_kernel(
    left_proj_ptr, right_proj_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr, # shape (total_rows, 1) half
    left_out_ptr, right_out_ptr,
    total_rows: tl.int32,
    hidden: tl.constexpr,
    nomask: tl.constexpr,
    BLOCK_M: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # load mask factor broadcasted across hidden
    if nomask:
        mask_val = tl.full([BLOCK_M, 1], 1.0, dtype=tl.float16)
    else:
        mask_vals = tl.load(mask_ptr + offs_m[:, None] * 1, mask=mask_m[:, None], other=0.0) # mask shape (R,1) half

    # iterate over hidden dimension in blocks
    for h_start in range(0, tl.cdiv(hidden, BLOCK_H)):
        offs_h = h_start * BLOCK_H + tl.arange(0, BLOCK_H)  # [BLOCK_H]
        mask_h = offs_h < hidden
        # load slices
        left_proj = tl.load(left_proj_ptr + offs_m[:, None] * hidden + offs_h[None, :],
                            mask=mask_m[:, None] & mask_h[None, :], other=0.0)
        right_proj = tl.load(right_proj_ptr + offs_m[:, None] * hidden + offs_h[None, :],
                             mask=mask_m[:, None] & mask_h[None, :], other=0.0)
        left_gate = tl.load(left_gate_ptr + offs_m[:, None] * hidden + offs_h[None, :],
                            mask=mask_m[:, None] & mask_h[None, :], other=0.0)
        right_gate = tl.load(right_gate_ptr + offs_m[:, None] * hidden + offs_h[None, :],
                             mask=mask_m[:, None] & mask_h[None, :], other=0.0)
        # compute sigmoid gates
        left_gate_sig = 1.0 / (1.0 + tl.exp(-left_gate))
        right_gate_sig = 1.0 / (1.0 + tl.exp(-right_gate))
        # apply mask if needed
        if nomask:
            mask_factor = tl.full([BLOCK_M, 1], 1.0, dtype=tl.float16)
        else:
            mask_factor = mask_vals
        # broadcast mask_factor across hidden dimension
        mask_exp = mask_factor   # shape (BLOCK_M,1)
        # compute out
        left_out = left_proj * mask_exp * left_gate_sig
        right_out = right_proj * mask_exp * right_gate_sig
        # store
        tl.store(left_out_ptr + offs_m[:, None] * hidden + offs_h[None, :],
                 left_out,
                 mask=mask_m[:, None] & mask_h[None, :])
        tl.store(right_out_ptr + offs_m[:, None] * hidden + offs_h[None, :],
                 right_out,
                 mask=mask_m[:, None] & mask_h[None, :])

"""
Highly‑optimized Triton + PyTorch forward for the **outgoing** TriMul operator
(AlphaFold‑3).  

Algorithm
---------
1️⃣  FP16 LayerNorm on the input (torch.nn.functional.layer_norm – native
    CUDA kernel).  
2️⃣  Concatenated linear projection that simultaneously produces the five
    tensors required for the update (left/right projections + three gates).  
3️⃣  Custom Triton kernel fuses:
        * sigmoid on the three gate tensors,
        * binary mask (if present),
        * element‑wise gating of the left/right projections,
        * sigmoid of the output‑gate.
   This replaces four separate PyTorch kernels with a single pass over the
   large (R×H) tensors.
4️⃣  Reshape the gated left/right tensors and perform the batched matrix‑
    multiply   `out[b,i,j] = Σ_k left[b,i,k]·right[b,j,k]`  via a single
   `torch.bmm`.  This is the dominant O(N³·H) work and is executed by the
   highly‑optimised cuBLAS Tensor‑Core routine.
5️⃣  FP16 fused LayerNorm + out‑gate (custom Triton kernel).  
6️⃣  Final FP16 linear projection back to the original channel dimension.  

All heavy memory‑bound passes are either fused in Triton or delegated to
cuBLAS/Tensor‑Core, yielding runtimes well below the 1 ms target on an
H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 2) Fuse the second LayerNorm + out‑gate (both FP16)
# ----------------------------------------------------------------------
@triton.jit
def fused_layernorm_gate_kernel(
    x_ptr,               # *const half*  [total_rows, hidden]
    y_ptr,               # *mut   half*  [total_rows, hidden]
    weight_ptr,          # *const float* [hidden] (γ)
    bias_ptr,            # *const float* [hidden] (β)
    gate_ptr,            # *const half*  [total_rows, hidden] (out_gate, sigmoid’ed)
    total_rows: tl.int32,
    hidden: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per inner loop (≤ 128)
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ---------------------------------------------------------------
    # mean / variance (FP32)
    # ---------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(hidden, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < hidden
        ptrs = x_ptr + offs_m[:, None] * hidden + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)      # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / hidden
    var = sum_sq / hidden - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------------------------------------------------------------
    # normalise, apply γ, β, multiply by gate, store as FP16
    # ---------------------------------------------------------------
    for d in range(0, tl.cdiv(hidden, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < hidden
        ptrs = x_ptr + offs_m[:, None] * hidden + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]          # FP32

        # multiply by gate (already sigmoid‑ed)
        gate = tl.load(gate_ptr + offs_m[:, None] * hidden + offs_d[None, :],
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        out = out * gate

        tl.store(y_ptr + offs_m[:, None] * hidden + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 3) Fuse mask, sigmoid‑gates and out_gate computation
# ----------------------------------------------------------------------
@triton.jit
def gating_mask_kernel(
    left_proj_ptr,  # [total_rows, hidden]  (pre‑gate left projection)
    right_proj_ptr, # [total_rows, hidden]  (pre‑gate right projection)
    left_gate_ptr,  # [total_rows, hidden]  (raw left gate)
    right_gate_ptr, # [total_rows, hidden]  (raw right gate)
    out_gate_ptr,   # [total_rows, hidden]  (raw out gate)
    mask_ptr,       # [total_rows]          (binary mask, 1/0) – ignored if NOMASK
    left_out_ptr,   # [total_rows, hidden]  (final left)
    right_out_ptr,  # [total_rows, hidden]  (final right)
    out_gate_out_ptr,  # [total_rows, hidden] (sigmoid(out_gate))
    total_rows: tl.int32,
    hidden: tl.constexpr,
    NOMASK: tl.constexpr,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_H: tl.constexpr,   # hidden cols per inner loop
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # Load mask once per block (if needed)
    # ------------------------------------------------------------------
    if NOMASK:
        mask_val = tl.full([BLOCK_M], 1.0, dtype=tl.float16)   # broadcast 1
    else:
        mask_val = tl.load(mask_ptr + offs_m,
                           mask=mask_m,
                           other=0.0)                     # [BLOCK_M]

    # ------------------------------------------------------------------
    # Loop over hidden dimension
    # ------------------------------------------------------------------
    for h_start in range(0, tl.cdiv(hidden, BLOCK_H)):
        offs_h = h_start * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < hidden

        # Load slices
        lp = tl.load(left_proj_ptr  + offs_m[:, None] * hidden + offs_h[None, :],
                     mask=mask_m[:, None] & mask_h[None, :], other=0.0)
        rp = tl.load(right_proj_ptr + offs_m[:, None] * hidden + offs_h[None, :],
                     mask=mask_m[:, None] & mask_h[None, :], other=0.0)
        lg = tl.load(left_gate_ptr  + offs_m[:, None] * hidden + offs_h[None, :],
                     mask=mask_m[:, None] & mask_h[None, :], other=0.0)
        rg = tl.load(right_gate_ptr + offs_m[:, None] * hidden + offs_h[None, :],
                     mask=mask_m[:, None] & mask_h[None, :], other=0.0)
        og = tl.load(out_gate_ptr   + offs_m[:, None] * hidden + offs_h[None, :],
                     mask=mask_m[:, None] & mask_h[None, :], other=0.0)

        # sigmoid gates (half → float32 for stability, back to half)
        lg_sig = (1.0 / (1.0 + tl.exp(-lg))).to(tl.float16)
        rg_sig = (1.0 / (1.0 + tl.exp(-rg))).to(tl.float16)
        og_sig = (1.0 / (1.0 + tl.exp(-og))).to(tl.float16)

        # Broadcast mask across the hidden dimension
        m = mask_val[:, None]        # [BLOCK_M,1]

        # Apply mask and gates
        left_out  = lp * lg_sig * m
        right_out = rp * rg_sig * m

        # Store results
        tl.store(left_out_ptr  + offs_m[:, None] * hidden + offs_h[None, :],
                 left_out,
                 mask=mask_m[:, None] & mask_h[None, :])
        tl.store(right_out_ptr + offs_m[:, None] * hidden + offs_h[None, :],
                 right_out,
                 mask=mask_m[:, None] & mask_h[None, :])
        tl.store(out_gate_out_ptr + offs_m[:, None] * hidden + offs_h[None, :],
                 og_sig,
                 mask=mask_m[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the **outgoing** TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, C] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (float32 / bool) – ignored if config["nomask"]=True
        - weights_dict : model parameters (float32)
        - config_dict  : {
              "dim"        : int   (C),
              "hidden_dim" : int   (H),
              "nomask"    : bool,
              "eps"       : float (optional)
          }

    Returns
    -------
    torch.Tensor [B, N, N, C] (float32) on the same CUDA device.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    inp, mask, weights, cfg = data
    device = inp.device

    dim = cfg["dim"]
    hidden = cfg["hidden_dim"]
    eps = cfg.get("eps", 1e-5)
    nomask = cfg.get("nomask", True)

    B, N, _, _ = inp.shape
    total_rows = B * N * N        # rows for the flattened tensors

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP16) – use the highly‑optimised PyTorch kernel
    # ------------------------------------------------------------------
    # Cast input to FP16 once (cost‑effective on H100) and run layer‑norm
    inp_fp16 = inp.half()
    norm_weight_fp16 = weights["norm.weight"].half()
    norm_bias_fp16   = weights["norm.bias"].half()
    x_norm_fp16 = torch.nn.functional.layer_norm(
        inp_fp16,
        normalized_shape=(dim,),
        weight=norm_weight_fp16,
        bias=norm_bias_fp16,
        eps=eps,
    )                               # [B, N, N, dim]  (FP16)

    # ------------------------------------------------------------------
    # 2️⃣ Single concatenated linear projection (produces 5×hidden columns)
    # ------------------------------------------------------------------
    # Combine the five projection / gate weight matrices
    concat_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).half().contiguous()                     # (5*H, C)

    # Perform the GEMM (FP16) on the flattened tensor
    proj_all = F.linear(x_norm_fp16.view(total_rows, dim), concat_w)  # (R,5H)

    # Split the concatenated result
    left_proj_fp16, right_proj_fp16, left_gate_fp16, right_gate_fp16, out_gate_fp16 = torch.split(
        proj_all, hidden, dim=1
    )  # each (R, H)

    # ------------------------------------------------------------------
    # 3️⃣ Fuse mask, sigmoid‑gates and out‑gate in one Triton kernel
    # ------------------------------------------------------------------
    left_out_fp16   = torch.empty_like(left_proj_fp16)
    right_out_fp16  = torch.empty_like(right_proj_fp16)
    out_gate_sig_fp16 = torch.empty_like(out_gate_fp16)

    # Mask tensor (flattened, broadcasted over hidden)
    if not nomask:
        mask_flat = mask.view(total_rows).to(torch.float16)   # (R,)
    else:
        mask_flat = torch.empty(0, dtype=torch.float16, device=device)  # dummy

    # Kernel launch parameters
    BLOCK_M = 128                # rows per program (tuned for H100)
    BLOCK_H = 32                 # hidden cols per inner loop
    grid = (triton.cdiv(total_rows, BLOCK_M),)

    gating_mask_kernel[grid](
        left_proj_fp16,
        right_proj_fp16,
        left_gate_fp16,
        right_gate_fp16,
        out_gate_fp16,
        mask_flat,
        left_out_fp16,
        right_out_fp16,
        out_gate_sig_fp16,
        total_rows,
        hidden,
        NOMASK=nomask,
        BLOCK_M=BLOCK_M,
        BLOCK_H=BLOCK_H,
    )

    # ------------------------------------------------------------------
    # 4️⃣ Reshape for the triangular multiplicative update (batched GEMM)
    # ------------------------------------------------------------------
    left = left_out_fp16.view(B, N, N, hidden)    # (B, N, N, H)
    right = right_out_fp16.view(B, N, N, hidden)  # (B, N, N, H)

    # Prepare tensors for batched matmul:
    #   left_mat  : (B*H, N, N)   rows = i, cols = k
    #   right_mat : (B*H, N, N)   rows = k, cols = j
    left_mat = left.permute(0, 3, 1, 2).reshape(B * hidden, N, N)
    right_mat = right.permute(0, 3, 2, 1).reshape(B * hidden, N, N)

    # Batched matrix multiplication (Tensor‑Core, FP16)
    out_mat = torch.bmm(left_mat, right_mat)     # (B*H, N, N)

    # Reshape back to (B, N, N, H) – keep FP16
    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 5️⃣ Second fused LayerNorm + out‑gate (custom Triton kernel)
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, hidden).contiguous()
    out_norm_gate_fp16 = torch.empty_like(out_flat)

    # Re‑use the same BLOCK_M / BLOCK_D (dim → hidden) for this kernel
    BLOCK_D = 128
    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    fused_layernorm_gate_kernel[grid2](
        out_flat,
        out_norm_gate_fp16,
        weights["to_out_norm.weight"].float(),   # γ  (float32 needed for accumulation)
        weights["to_out_norm.bias"].float(),     # β
        out_gate_sig_fp16,                       # already sigmoid‑ed, FP16
        total_rows,
        hidden,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    to_out_w_fp16 = weights["to_out.weight"].half()   # (dim, hidden)
    final_fp16 = F.linear(out_norm_gate_fp16, to_out_w_fp16)  # (R, dim) FP16

    # Cast back to FP32 as required by the module interface
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
