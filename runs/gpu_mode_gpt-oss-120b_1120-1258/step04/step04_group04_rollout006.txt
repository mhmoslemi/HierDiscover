import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def batched_matmul_fp16_kernel(
    a_ptr, b_ptr, c_ptr,
    BATCH, M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute start indices
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    a_batch_off = batch_id * stride_a_batch
    b_batch_off = batch_id * stride_b_batch
    c_batch_off = batch_id * stride_c_batch

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        a_ptrs = a_ptr + a_batch_off + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        b_ptrs = b_ptr + b_batch_off + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n

        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)

        acc += tl.dot(a, b)

    # Write result
    c_ptrs = c_ptr + c_batch_off + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operator.

    The implementation follows the reference PyTorch code but
    - uses torch.nn.functional.layer_norm for the two LayerNorms
      (still float32 / fp16 stable)
    - fuses the five linear projections into a single GEMM
    - performs the O(N³) “triangular multiplicative update” with a
      custom batched fp16 GEMM written in Triton (the only required
      Triton kernel in this solution).

    All heavy tensor arithmetic runs in fp16; the final result is
    returned as float32.
    """
    # ------------------------------------------------------------------
    # 0️⃣ unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    eps = 1e-5
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape  # [B, N, N, C]
    total_rows = B * N * N            # flatten over (B,N,N)

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 → fp16)
    # ------------------------------------------------------------------
    # torch.layer_norm works on the last dimension
    x_norm = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # float32
    x_norm = x_norm.half()  # fp16 for the rest

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection + gating (single GEMM)
    # ------------------------------------------------------------------
    # concatenate weight matrices: (5*H, C)
    concat_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).half()  # (5H, C)
    # Linear: (R, C) @ (C, 5H)^T → (R, 5H)
    proj_all = F.linear(x_norm, concat_weight)  # fp16, shape (R, 5H)

    # split
    left_proj, right_proj, left_gate, right_gate, out_gate = torch.split(
        proj_all, hidden_dim, dim=1
    )  # each (R, H)

    # ------------------------------------------------------------------
    # 2a️⃣ Optional mask on left/right projections
    # ------------------------------------------------------------------
    if not nomask:
        mask_flat = mask_tensor.view(total_rows, 1).to(dtype=torch.float16)
        left_proj = left_proj * mask_flat
        right_proj = right_proj * mask_flat

    # ------------------------------------------------------------------
    # 2b️⃣ Sigmoid gates + element‑wise gating
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    left = left_proj * left_gate      # (R, H)
    right = right_proj * right_gate   # (R, H)

    # ------------------------------------------------------------------
    # 3️⃣ Reshape to [B, N, N, H] for triangular multiplication
    # ------------------------------------------------------------------
    left = left.view(B, N, N, hidden_dim)   # [B, N, N, H]
    right = right.view(B, N, N, hidden_dim) # [B, N, N, H]
    out_gate = out_gate.view(B, N, N, hidden_dim)  # [B, N, N, H]

    # Bring hidden dim to batch axis
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N] (i,k)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # [B, H, N, N] (k,j)

    BH = B * hidden_dim
    left_mat = left_perm.view(BH, N, N).contiguous()
    right_mat = right_perm.view(BH, N, N).contiguous()

    # ------------------------------------------------------------------
    # 4️⃣ Batched matrix multiplication in Triton (i,k)·(k,j) → (i,j)
    # ------------------------------------------------------------------
    out_mat = torch.empty_like(left_mat)  # (BH, N, N), fp16

    # Compute strides (in elements, not bytes)
    stride_a_batch = left_mat.stride(0)
    stride_a_m = left_mat.stride(1)
    stride_a_k = left_mat.stride(2)

    stride_b_batch = right_mat.stride(0)
    stride_b_k = right_mat.stride(1)
    stride_b_n = right_mat.stride(2)

    stride_c_batch = out_mat.stride(0)
    stride_c_m = out_mat.stride(1)
    stride_c_n = out_mat.stride(2)

    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (BH, triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N))
    batched_matmul_fp16_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        BH,
        N,
        N,
        N,
        stride_a_batch,
        stride_a_m,
        stride_a_k,
        stride_b_batch,
        stride_b_k,
        stride_b_n,
        stride_c_batch,
        stride_c_m,
        stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # Reshape back to [B, N, N, H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm (over hidden_dim)
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"].half()
    to_out_norm_bias = weights["to_out_norm.bias"].half()
    out_norm = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )  # fp16

    # ------------------------------------------------------------------
    # 6️⃣ Gating + final projection back to dim
    # ------------------------------------------------------------------
    gated = out_norm * out_gate  # fp16

    to_out_weight = weights["to_out.weight"].half()
    gated_flat = gated.view(B * N * N, hidden_dim)
    final_fp16 = F.linear(gated_flat, to_out_weight)  # fp16, shape (B*N*N, dim)

    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final

# Triton batched matrix‑multiply (fp16 → fp16) used for the O(N³) TriMul step.
# Input  A : [BATCH, M, K]   (fp16)
# Input  B : [BATCH, K, N]   (fp16)
# Output C : [BATCH, M, N]   (fp16)
# ---------------------------------------------------------------------------
@triton.jit
def batched_matmul_fp16_kernel(
    a_ptr, b_ptr, c_ptr,                      # pointers
    BATCH, M, N, K,                           # dimensions
    stride_a_batch, stride_a_m, stride_a_k,   # strides of A
    stride_b_batch, stride_b_k, stride_b_n,   # strides of B
    stride_c_batch, stride_c_m, stride_c_n,   # strides of C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM kernel (fp16 → fp16).  One program handles a TILE of C."""
    batch_id = tl.program_id(0)
    pid_m    = tl.program_id(1)
    pid_n    = tl.program_id(2)

    # --------------------------------------------------------------
    # Tile offsets
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Batch offsets
    a_batch_off = batch_id * stride_a_batch
    b_batch_off = batch_id * stride_b_batch
    c_batch_off = batch_id * stride_c_batch

    # --------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.view(total_rows, 1).to(dtype=torch.float16)
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # --------------------------------------------------------------
    # 2b️⃣ Apply sigmoid gates and element‑wise gating
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate   = torch.sigmoid(out_gate)

    left  = left_proj  * left_gate   # [R, H]
    right = right_proj * right_gate # [R, H]

    # --------------------------------------------------------------
    # 3️⃣ Reshape to [B, N, N, H] for the triangular update
    # --------------------------------------------------------------
    left  = left .view(B, N, N, hidden_dim)   # [B, N, N, H]
    right = right.view(B, N, N, hidden_dim)   # [B, N, N, H]
    out_gate = out_gate.view(B, N, N, hidden_dim)

    # Bring the hidden dimension into the batch axis
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]   (i,k)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # [B, H, N, N]   (k,j)

    BH = B * hidden_dim
    left_mat  = left_perm .view(BH, N, N).contiguous()   # [BH, N, N]  (i,k)
    right_mat = right_perm.view(BH, N, N).contiguous()   # [BH, N, N]  (k,j)

    # --------------------------------------------------------------
    # 4️⃣ Batched GEMM via Triton (the heavy O(N³) part)
    # --------------------------------------------------------------
    out_mat = torch.empty_like(left_mat)                  # [BH, N, N], fp16

    # Compute strides in elements (not bytes)
    stride_a_batch = left_mat.stride(0)
    stride_a_m     = left_mat.stride(1)
    stride_a_k     = left_mat.stride(2)

    stride_b_batch = right_mat.stride(0)
    stride_b_k     = right_mat.stride(1)
    stride_b_n     = right_mat.stride(2)

    stride_c_batch = out_mat.stride(0)
    stride_c_m     = out_mat.stride(1)
    stride_c_n     = out_mat.stride(2)

    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (BH,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))

    batched_matmul_fp16_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        BH,
        N,
        N,
        N,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # Reshape back to the 4‑D tensor [B, N, N, H] (still fp16)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm over the hidden dimension
    # --------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].half(),
        bias=weights["to_out_norm.bias"].half(),
        eps=eps,
    )                                            # fp16

    # --------------------------------------------------------------
    # 6️⃣ Gating by out_gate and final projection back to `dim`
    # --------------------------------------------------------------
    gated = out_norm * out_gate                     # fp16

    final_fp16 = F.linear(
        gated.view(B * N * N, hidden_dim),          # flatten
        weights["to_out.weight"].half(),           # (C, H) -> (H, C) needed, linear does weight^T internally
    )                                              # fp16, shape (B·N·N, C)

    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
