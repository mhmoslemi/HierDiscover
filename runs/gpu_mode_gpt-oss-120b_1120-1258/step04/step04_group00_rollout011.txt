import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def layernorm_fp16_kernel(x_ptr, y_ptr, w_ptr, b_ptr, total_rows, dim,
                          eps, BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0)
        vals_f = vals.to(tl.float32)
        sum_val += tl.sum(vals_f, axis=1)
        sum_sq  += tl.sum(vals_f * vals_f, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        norm = (vals - mean[:, None]) * rstd[:, None]
        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)
        out = norm * w[None, :] + b[None, :]
        out_fp16 = out.to(tl.float16)
        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    x_norm_h = x_norm_h.view(B, N, N, dim)   # FP16

    # Concatenated weight for left/right proj and gates + out_gate
    # prepare weight matrix: [5*hidden_dim, dim]
    left_proj_w = weights["left_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_proj_w = weights["right_proj.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    # cat along out_features dimension (0)
    cat_weight = torch.cat(
        [left_proj_w, left_gate_w, right_proj_w, right_gate_w, out_gate_w], dim=0
    ).to(torch.float16, non_blocking=True)   # shape [5*hidden_dim, dim]

    # Linear transformation (rows, dim) @ cat_weight.t -> (rows, 5*hidden_dim)
    rows = total_rows
    x_norm_flat = x_norm_h.view(rows, dim)
    cat_out = F.linear(x_norm_flat, cat_weight)   # (rows, 5*hidden_dim) in FP16

    # split outputs
    H = hidden_dim
    left_proj_h = cat_out[:, :H]
    left_gate_pre_h = cat_out[:, H:2*H]
    right_proj_h = cat_out[:, 2*H:3*H]
    right_gate_pre_h = cat_out[:, 3*H:4*H]
    out_gate_pre_h = cat_out[:, 4*H:5*H]

    # Apply optional mask
    if not nomask:
        mask_h = mask_tensor.view(rows, 1).to(torch.float16, non_blocking=True)
        left_proj_h = left_proj_h * mask_h
        right_proj_h = right_proj_h * mask_h

    # Gates (sigmoid)
    left_gate_h = torch.sigmoid(left_gate_pre_h)
    right_gate_h = torch.sigmoid(right_gate_pre_h)
    out_gate_h = torch.sigmoid(out_gate_pre_h)

    left_h = left_proj_h * left_gate_h   # shape rows x H
    right_h = right_proj_h * right_gate_h

    # Reshape to [B,N,N,H]
    left = left_h.view(B, N, N, H)
    right = right_h.view(B, N, N, H)

    # Batched GEMM: (B,H,N,N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    BH = B * H
    left_mat = left_perm.view(BH, N, N)
    right_mat = right_perm.view(BH, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [BH,N,N]

    # Reshape back to [B,N,N,H] FP16
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # Second LayerNorm on hidden dim
    rows_h = B * N * N
    out_flat = out.view(rows_h, H).contiguous()
    out_norm_h = torch.empty_like(out_flat, dtype=torch.float16)
    BLOCK_M2 = 256
    BLOCK_D2 = 64 if H <= 128 else 128
    grid2 = (triton.cdiv(rows_h, BLOCK_M2),)
    layernorm_fp16_kernel[grid2](

        out_norm_h,

        weights["to_out_norm.bias"],
        rows_h,
        H,
        eps,
        BLOCK_M=BLOCK_M2,
        BLOCK_D=BLOCK_D2,
    )
    out_norm = out_norm_h.view(B, N, N, H)   # FP16

    # Apply output gate
    out_norm = out_norm * out_gate_h.view(B, N, N, H)

    # Final linear (hidden_dim -> dim)
    final_h = F.linear(
        out_norm.view(rows_h, H),
        weights["to_out.weight"].to(torch.float16, non_blocking=True)
    )   # [rows_h, dim] FP16

    final = final_h.view(B, N, N, dim).to(torch.float32)
    return final

"""
Highly‑optimized Triton‑fused forward for the “outgoing” TriMul module.
The algorithm:
1. Row‑wise LayerNorm over the last axis (FP32 statistics → FP16 output) via a Triton kernel.
2. Fuse all dense linear maps (left_proj, left_gate, right_proj, right_gate, out_gate)
   into a single FP16 matmul (rows × dim) × (5·hidden_dim × dim)ᵀ.
3. Split the result, apply the optional pair‑wise mask and sigmoid gates.
4. Batched GEMM that implements the triangular multiplicative update:
   out_h(i,j)=Σ_k left_h(i,k)·right_h(j,k)  → torch.bmm on [B·H, N, N] tensors.
5. Second LayerNorm (hidden_dim) via the same Triton kernel.
6. Apply the output gate, a final FP16 linear projection and cast back to FP32.
All heavy linear algebra is delegated to cuBLAS Tensor‑Core kernels; the two
LayerNorms are the only custom Triton kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def layernorm_fp16_kernel(
    x_ptr,          # const float*   [total_rows, dim]
    y_ptr,          #       half*   [total_rows, dim] (output)
    w_ptr,          # const float*  [dim]   (γ)
    b_ptr,          # const float*  [dim]   (β)
    total_rows,     # int (runtime)
    dim: tl.constexpr,          # static: number of columns
    eps: tl.constexpr,          # static epsilon
    BLOCK_M: tl.constexpr,      # rows per program
    BLOCK_D: tl.constexpr,      # columns per inner loop
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # --------------------------------------------------------------
    # Pass 1: mean / variance (FP32)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)        # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                         # 1/√(var+eps)

    # --------------------------------------------------------------
    # Pass 2: write normalized data (FP16)
    # --------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]   # (x-μ)/σ

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)

        out = norm * w[None, :] + b[None, :]            # FP32
        out_fp16 = out.to(tl.float16)

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module (AlphaFold‑3 style).

    Args:
        data: tuple containing
            - input_tensor : torch.Tensor[B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor[B, N, N]      (float32)
            - weights_dict : mapping of weight tensors
            - config_dict  : {"dim": int, "hidden_dim": int, "nomask": bool (optional)}

    Returns:
        torch.Tensor[B, N, N, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32 statistics → FP16 values)
    # ------------------------------------------------------------------
    total_rows = B * N * N
    x_flat = input_tensor.view(total_rows, dim).contiguous()   # [rows, dim] fp32
    x_norm_h = torch.empty((total_rows, dim), dtype=torch.float16, device=device)

    # Choose block sizes (tuned for dim ≤ 384)
    BLOCK_M = 256
    BLOCK_D = 64 if dim <= 128 else 128   # covers all supported dimensions

    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        x_flat,
        x_norm_h,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    # reshape back to 4‑D (FP16)
    x_norm_h = x_norm_h.view(B, N, N, dim)

    # ------------------------------------------------------------------
    # 2️⃣ Fuse all linear maps (left/right projections & gates + out_gate)
    # ------------------------------------------------------------------
    # Stack weights: (5*hidden_dim, dim)
    cat_weight = torch.cat(
        [
            weights["left_proj.weight"],   # (H, dim)
            weights["left_gate.weight"],   # (H, dim)
            weights["right_proj.weight"],  # (H, dim)
            weights["right_gate.weight"],  # (H, dim)
            weights["out_gate.weight"],    # (H, dim)
        ],
        dim=0,
    ).to(torch.float16, non_blocking=True)   # FP16

    rows = total_rows
    x_norm_flat = x_norm_h.view(rows, dim)   # (rows, dim) FP16

    # Single fused matmul (rows × dim) × (5*H × dim)ᵀ → (rows, 5*H)
    fused_out = F.linear(x_norm_flat, cat_weight)   # FP16

    H = hidden_dim
    left_proj_h      = fused_out[:, :H]               # (rows, H)
    left_gate_pre_h  = fused_out[:, H:2*H]
    right_proj_h     = fused_out[:, 2*H:3*H]
    right_gate_pre_h = fused_out[:, 3*H:4*H]
    out_gate_pre_h   = fused_out[:, 4*H:5*H]

    # ------------------------------------------------------------------
    # 3️⃣ Optional pair‑wise mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        mask_h = mask_tensor.view(rows, 1).to(torch.float16, non_blocking=True)
        left_proj_h = left_proj_h * mask_h
        right_proj_h = right_proj_h * mask_h

    # ------------------------------------------------------------------
    # 4️⃣ Gate activations (sigmoid) and gated projections
    # ------------------------------------------------------------------
    left_gate_h  = torch.sigmoid(left_gate_pre_h)
    right_gate_h = torch.sigmoid(right_gate_pre_h)
    out_gate_h   = torch.sigmoid(out_gate_pre_h)  # (rows, H)

    left_h  = left_proj_h * left_gate_h   # (rows, H)
    right_h = right_proj_h * right_gate_h

    # ------------------------------------------------------------------
    # 5️⃣ Tri‑multiplicative update (batched GEMM over the hidden dimension)
    # ------------------------------------------------------------------
    left  = left_h.view(B, N, N, H)
    right = right_h.view(B, N, N, H)

    # Bring H into the batch axis → [B·H, N, N]
    left_perm  = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    BH = B * H
    left_mat  = left_perm.view(BH, N, N)
    right_mat = right_perm.view(BH, N, N)

    # out_mat[b*h, i, j] = Σ_k left_mat[b*h, i, k] * right_mat[b*h, j, k]
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (BH, N, N)

    # reshape back to [B, N, N, H] (still FP16)
    out = (
        out_mat.view(B, H, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )   # (B, N, N, H)

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm on the hidden dimension
    # ------------------------------------------------------------------
    rows_h = B * N * N
    out_flat = out.view(rows_h, H).contiguous()
    out_norm_h = torch.empty_like(out_flat, dtype=torch.float16)

    BLOCK_D2 = 64 if H <= 128 else 128
    grid2 = (triton.cdiv(rows_h, BLOCK_M),)
    layernorm_fp16_kernel[grid2](
        out_flat,
        out_norm_h,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        rows_h,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D2,
    )
    out_norm = out_norm_h.view(B, N, N, H)   # FP16

    # ------------------------------------------------------------------
    # 7️⃣ Apply output gate and final linear projection
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate_h.view(B, N, N, H)

    final_h = F.linear(
        out_norm.view(rows_h, H),
        weights["to_out.weight"].to(torch.float16, non_blocking=True)
    )   # (rows_h, dim) FP16

    # Cast back to FP32 as expected by the original module
    final = final_h.view(B, N, N, dim).to(torch.float32)
    return final

"""
Outgoing TriMul (AlphaFold‑3) forward pass – highly optimized.

Algorithm
---------
1️⃣  Row‑wise LayerNorm on the [B,N,N,dim] tensor (FP32 stats → FP16) using
    a tiny Triton kernel.
2️⃣  Fuse **all** linear maps (left/right projections, their gates and the
    output gate) into a **single** FP16 matmul:
        (rows, dim) × (5·hidden_dim, dim)ᵀ  →  (rows, 5·hidden_dim)
3️⃣  Split the fused result, broadcast the optional pair‑wise mask,
    apply sigmoid gates and obtain the gated left/right tensors.
4️⃣  Tri‑multiplicative update:
        out_h(i,j) = Σ_k left_h(i,k)·right_h(j,k)
    is a batched GEMM performed with `torch.bmm` on tensors of shape
    [B·hidden_dim, N, N].
5️⃣  Second LayerNorm on the hidden dimension (again via the Triton kernel).
6️⃣  Apply the output‑gate, a final FP16 linear projection
    (hidden_dim → dim) and cast the result back to FP32.

Only the two LayerNorms are implemented in Triton; all dense linear
operations and the core N×N batched GEMM are delegated to cuBLAS Tensor‑Core
kernels, which gives the best throughput on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (output in FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,          # const float*   [total_rows, dim]
    y_ptr,          #       half*   [total_rows, dim] (output)
    w_ptr,          # const float*  [dim]   (γ)
    b_ptr,          # const float*  [dim]   (β)
    total_rows,     # int (runtime)
    dim: tl.constexpr,          # static: number of columns (used for strides)
    eps: tl.constexpr,          # static epsilon
    BLOCK_M: tl.constexpr,      # rows processed per program
    BLOCK_D: tl.constexpr,      # columns processed per inner loop
):
    """
    Compute LayerNorm over the last dimension of a flat 2‑D tensor.
    Mean / variance are accumulated in FP32, the normalized result is written
    as FP16 to reduce memory traffic.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # --------------------------------------------------------------
    # Pass 1 – compute mean and variance (FP32)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)        # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                         # 1 / sqrt(var+eps)

    # --------------------------------------------------------------
    # Pass 2 – write normalized values (FP16)
    # --------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        # (x - μ) / σ
        norm = (vals - mean[:, None]) * rstd[:, None]

        # γ and β (LayerNorm parameters)
        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)

        out = norm * w[None, :] + b[None, :]          # FP32
        out_fp16 = out.to(tl.float16)

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Triton‑fused forward for TriMul (outgoing version)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        input_tensor : torch.Tensor [B, N, N, dim] (float32)
        mask_tensor  : torch.Tensor [B, N, N]      (float32) – may be unused
        weights_dict : {
            "norm.weight":        Tensor[dim],
            "norm.bias":          Tensor[dim],
            "left_proj.weight":   Tensor[hidden_dim, dim],
            "right_proj.weight":  Tensor[hidden_dim, dim],
            "left_gate.weight":   Tensor[hidden_dim, dim],
            "right_gate.weight":  Tensor[hidden_dim, dim],
            "out_gate.weight":    Tensor[hidden_dim, dim],
            "to_out_norm.weight": Tensor[hidden_dim],
            "to_out_norm.bias":   Tensor[hidden_dim],
            "to_out.weight":      Tensor[dim, hidden_dim],
        }
        config_dict : {
            "dim":         int,
            "hidden_dim":  int,
            "nomask":      bool (optional, default True),
        }

    Returns
    -------
    torch.Tensor [B, N, N, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣  First LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    total_rows = B * N * N                       # rows of the flattened tensor
    x_flat = input_tensor.view(total_rows, dim).contiguous()   # [rows, dim] FP32
    x_norm_h = torch.empty((total_rows, dim),
                           dtype=torch.float16,
                           device=device)

    # Block sizes (tuned for the supported dimensions)
    BLOCK_M = 256
    BLOCK_D = 64 if dim <= 128 else 128

    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        x_flat,
        x_norm_h,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    # Reshape to 4‑D (still FP16)
    x_norm_h = x_norm_h.view(B, N, N, dim)

    # ------------------------------------------------------------------
    # 2️⃣  Fuse all linear maps (projections + gates + output gate)
    # ------------------------------------------------------------------
    # Stack the five weight matrices along the output‑feature dimension:
    #   left_proj, left_gate, right_proj, right_gate, out_gate
    fused_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16, non_blocking=True)      # shape [5*H, dim]

    rows = total_rows
    x_norm_flat = x_norm_h.view(rows, dim)       # (rows, dim) FP16
    fused_out = F.linear(x_norm_flat, fused_weight)  # (rows, 5*H) FP16

    H = hidden_dim
    # Split fused output
    left_proj_h      = fused_out[:, :H]               # (rows, H)
    left_gate_pre_h  = fused_out[:, H:2*H]
    right_proj_h     = fused_out[:, 2*H:3*H]
    right_gate_pre_h = fused_out[:, 3*H:4*H]
    out_gate_pre_h   = fused_out[:, 4*H:5*H]

    # ------------------------------------------------------------------
    # 3️⃣  Optional mask (broadcast over the hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        mask_h = mask_tensor.view(rows, 1).to(torch.float16, non_blocking=True)
        left_proj_h = left_proj_h * mask_h
        right_proj_h = right_proj_h * mask_h

    # ------------------------------------------------------------------
    # 4️⃣  Sigmoid gates and gated projections
    # ------------------------------------------------------------------
    left_gate_h  = torch.sigmoid(left_gate_pre_h)
    right_gate_h = torch.sigmoid(right_gate_pre_h)
    out_gate_h   = torch.sigmoid(out_gate_pre_h)          # (rows, H)

    left_h  = left_proj_h * left_gate_h                   # (rows, H)
    right_h = right_proj_h * right_gate_h

    # ------------------------------------------------------------------
    # 5️⃣  Tri‑multiplicative update (batched GEMM)
    # ------------------------------------------------------------------
    left  = left_h.view(B, N, N, H)
    right = right_h.view(B, N, N, H)

    # Bring the hidden dimension into the batch axis -> [B·H, N, N]
    left_perm  = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    BH = B * H
    left_mat  = left_perm.view(BH, N, N)
    right_mat = right_perm.view(BH, N, N)

    # out_mat[b*h, i, j] = Σ_k left_mat[b*h, i, k] * right_mat[b*h, j, k]
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (BH, N, N)

    # Back to [B, N, N, H] (FP16)
    out = (
        out_mat.view(B, H, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )   # (B, N, N, H)

    # ------------------------------------------------------------------
    # 6️⃣  Second LayerNorm on the hidden dimension
    # ------------------------------------------------------------------
    rows_h = B * N * N
    out_flat = out.view(rows_h, H).contiguous()
    out_norm_h = torch.empty_like(out_flat, dtype=torch.float16)

    BLOCK_D2 = 64 if H <= 128 else 128
    grid2 = (triton.cdiv(rows_h, BLOCK_M),)
    layernorm_fp16_kernel[grid2](
        out_flat,
        out_norm_h,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        rows_h,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D2,
    )
    out_norm = out_norm_h.view(B, N, N, H)   # FP16

    # ------------------------------------------------------------------
    # 7️⃣  Output gate and final projection
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate_h.view(B, N, N, H)

    final_h = F.linear(
        out_norm.view(rows_h, H),
        weights["to_out.weight"].to(torch.float16, non_blocking=True)
    )   # (rows_h, dim) FP16

    # Cast back to FP32 (the original module returns FP32)
    final = final_h.view(B, N, N, dim).to(torch.float32)
    return final
