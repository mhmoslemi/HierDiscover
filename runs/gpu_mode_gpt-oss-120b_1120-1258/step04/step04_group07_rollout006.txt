"""
custom_kernel – Outgoing TriMul (AlphaFold‑3 style)

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the 4‑D input → fp16.
2️⃣ Fuse the five linear layers (left_proj, right_proj, left_gate,
   right_gate, out_gate) into a single GEMM:  X_norm @ W_concat.T.
   The result is sliced into the five tensors and the three gate
   tensors are passed through a sigmoid.
3️⃣ (Optional) apply the binary mask to the two projected tensors.
4️⃣ Apply the three gates → obtain `left` and `right` tensors 
   of shape [B, N, N, H] (fp16).
5️⃣ **Tri‑multiplicative update** – custom Triton kernel that computes
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]  (i‑j matrix
   multiplication for each head).  The kernel works on a flattened
   head dimension (B·H matrices of size N×N) and uses Tensor‑Core
   `tl.dot` with fp16 inputs / fp32 accumulation.
6️⃣ Second LayerNorm (fp16) on the hidden dimension.
7️⃣ Multiply by the output‑gate, project back to the original dimension
   with a half‑precision linear layer and cast the result to float32.

All heavy arithmetic is performed in fp16 (except the accumulator of
the GEMM which is fp32 to retain numerical stability).  Two custom
Triton kernels are used: a fast row‑wise LayerNorm and the batched
matrix‑multiplication required by the triangular update.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (kernel)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,  # const *[total_rows, dim]  (fp32)
    y_ptr,  # mutable *[total_rows, dim] (fp16)
    weight_ptr,  # const *[dim] (fp32)
    bias_ptr,    # const *[dim] (fp32)
    total_rows,  # int
    dim,         # int (compile‑time)
    eps,         # float (compile‑time)
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """Rowwise layer‑norm: y = (x-μ)/√(σ²+ε) * γ + β   (output fp16)"""
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ---------- pass 1: compute mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0)
        vals_f32 = vals.to(tl.float32)
        sum_val += tl.sum(vals_f32, axis=1)
        sum_sq  += tl.sum(vals_f32 * vals_f32, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)               # 1/√(var+ε)

    # ---------- pass 2: write normalised values ----------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0)
        vals_f32 = vals.to(tl.float32)

        norm = (vals_f32 - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)
        out = norm * w[None, :] + b[None, :]  # fp32

        out_fp16 = out.to(tl.float16)
        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 5️⃣ Tri‑multiplicative update (batched GEMM) – custom kernel
# ----------------------------------------------------------------------
@triton.jit
def trimul_kernel(
    left_ptr,   # const *[B, N, N, H]  (fp16)
    right_ptr,  # const *[B, N, N, H]  (fp16)
    out_ptr,    # mutable *[B, N, N, H] (fp16)

    B, N, H,                # runtime ints
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,   # int64
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,

    BLOCK_M: tl.constexpr,   # tile size for i dimension
    BLOCK_N: tl.constexpr,   # tile size for j dimension
    BLOCK_K: tl.constexpr,   # tile size for k dimension
):
    """
    Compute, for every head h of each batch b:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    This is a batched matrix multiplication left @ rightᵀ.
    """
    head_id = tl.program_id(0)          # 0 … B*H-1
    block_i = tl.program_id(1)          # i‑tile
    block_j = tl.program_id(2)          # j‑tile

    # recover batch and head indices
    batch_id = head_id // H
    h_id     = head_id % H

    # offsets inside the N×N matrices
    i_base = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_base = block_j * BLOCK_N + tl.arange(0, BLOCK_N)
    k_base = tl.arange(0, BLOCK_K)

    mask_i = i_base < N
    mask_j = j_base < N
    mask_k = k_base < N

    # base pointers for this (batch,head)
    left_base  = left_ptr  + batch_id * stride_l_b + h_id * stride_l_h
    right_base = right_ptr + batch_id * stride_r_b + h_id * stride_r_h
    out_base   = out_ptr   + batch_id * stride_o_b + h_id * stride_o_h

    # accumulator in fp32 for numerical stability
    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    # loop over the reduction axis k
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for kk in range(num_k_blocks):
        k_start = kk * BLOCK_K
        cur_k   = k_start + k_base
        cur_mask_k = cur_k < N

        # left[i,k]  → shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_base + i_base[:, None] * stride_l_i + cur_k[None, :] * stride_l_k
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & cur_mask_k[None, :],
                    other=0.0).to(tl.float32)

        # right[j,k] (need transposed layout) → shape (BLOCK_K, BLOCK_N)
        b_ptrs = right_base + cur_k[:, None] * stride_r_k + j_base[None, :] * stride_r_j
        b = tl.load(b_ptrs,
                    mask=cur_mask_k[:, None] & mask_j[None, :],
                    other=0.0).to(tl.float32)

        # GEMM step
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_N)

    # write the result (cast back to fp16)
    out_ptrs = out_base + i_base[:, None] * stride_o_i + j_base[None, :] * stride_o_j
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Entrypoint – the user‑visible function
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor, shape [B, N, N]   (bool/float)
        - weights_dict : dict {str: torch.Tensor}  (float32)
        - config_dict  : {"dim": int, "hidden_dim": int, "nomask": bool (opt.)}

    Returns
    -------
    torch.Tensor
        Output tensor, shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5
    device = input_tensor.device

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N            # R = B·N·N

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → fp16
    # --------------------------------------------------------------
    x_flat = input_tensor.view(total_rows, dim).contiguous()
    x_norm = torch.empty(total_rows, dim, dtype=torch.float16, device=device)

    BLOCK_M = 256            # rows processed per program
    BLOCK_D = 64             # cols processed per inner loop (tune for dim ≤ 384)

    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # --------------------------------------------------------------
    # 2️⃣ Fuse the five linear layers into a single GEMM
    # --------------------------------------------------------------
    # concatenate the five weight matrices (shape: [5*H, dim])
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
            weights["out_gate.weight"],    # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)   # fp16 for Tensor‑Core

    # fused linear: (R, dim) @ (5*H, dim).T → (R, 5*H)
    proj_all = F.linear(x_norm, w_concat)   # fp16, bias omitted

    H = hidden_dim
    # slice the concatenated result
    left_proj   = proj_all[:, 0:H]                    # (R, H)
    right_proj  = proj_all[:, H:2*H]
    left_gate_p = proj_all[:, 2*H:3*H]
    right_gate_p= proj_all[:, 3*H:4*H]
    out_gate_p  = proj_all[:, 4*H:5*H]

    # gates → sigmoid (still fp16)
    left_gate  = torch.sigmoid(left_gate_p)
    right_gate = torch.sigmoid(right_gate_p)
    out_gate   = torch.sigmoid(out_gate_p)

    # --------------------------------------------------------------
    # 3️⃣ Optional masking (broadcast over hidden dim)
    # --------------------------------------------------------------
    if not nomask:
        # mask: [B,N,N] → [R,1] (fp16) and broadcast
        mask_h = mask_tensor.to(torch.float16).view(total_rows, 1)
        left_proj  = left_proj * mask_h
        right_proj = right_proj * mask_h

    # --------------------------------------------------------------
    # 4️⃣ Apply projection‑gates
    # --------------------------------------------------------------
    left  = left_proj * left_gate     # (R, H)
    right = right_proj * right_gate   # (R, H)

    # reshape to 4‑D tensors required by the triangular update
    left  = left.view(B, N, N, H).contiguous()
    right = right.view(B, N, N, H).contiguous()
    out_gate = out_gate.view(B, N, N, H).contiguous()

    # --------------------------------------------------------------
    # 5️⃣ Tri‑multiplicative update (custom Triton GEMM)
    # --------------------------------------------------------------
    out = torch.empty_like(left, dtype=torch.float16)   # [B,N,N,H]

    # --- strides for the kernels (row‑major layout) ---
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_j, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # launch configuration
    BLOCK_M_T = 64      # tile size for i   (tuned for N up to 1024)
    BLOCK_N_T = 64      # tile size for j
    BLOCK_K_T = 32      # tile size for k (shared‑mem friendly)

    grid = (
        B * H,                              # head dimension (batch * hidden)
        triton.cdiv(N, BLOCK_M_T),          # i‑tiles
        triton.cdiv(N, BLOCK_N_T),          # j‑tiles
    )
    trimul_kernel[grid](
        left,
        right,
        out,
        B, N, H,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_j, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M=BLOCK_M_T,
        BLOCK_N=BLOCK_N_T,
        BLOCK_K=BLOCK_K_T,
    )

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm over the hidden dimension (fp16)
    # --------------------------------------------------------------
    out_flat = out.view(total_rows, H).contiguous()
    out_norm = torch.empty_like(out_flat, dtype=torch.float16)

    grid_ln2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid_ln2](
        out_flat,
        out_norm,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    out_norm = out_norm.view(B, N, N, H)   # fp16

    # --------------------------------------------------------------
    # 7️⃣ Final gating + projection back to `dim`
    # --------------------------------------------------------------
    out_norm = out_norm * out_gate                     # element‑wise fp16
    out_norm_flat = out_norm.view(total_rows, H).contiguous()

    final_fp16 = F.linear(
        out_norm_flat,
        weights["to_out.weight"].to(torch.float16)     # [dim, H] (fp16)
    )  # (R, dim) fp16

    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final
