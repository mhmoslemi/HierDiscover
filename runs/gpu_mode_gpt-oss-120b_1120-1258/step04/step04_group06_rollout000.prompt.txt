import torch
from torch import nn, einsum
import math

# Reference code in PyTorch
class TriMul(nn.Module):
    def __init__(
        self,
        dim: int,
        hidden_dim: int,
    ):
        super().__init__()

        self.norm = nn.LayerNorm(dim)

        self.left_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.right_proj = nn.Linear(dim, hidden_dim, bias=False)

        self.left_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.right_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.out_gate = nn.Linear(dim, hidden_dim, bias=False)

        self.to_out_norm = nn.LayerNorm(hidden_dim)
        self.to_out = nn.Linear(hidden_dim, dim, bias=False)

    def forward(self, x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
        """
        x: [bs, seq_len, seq_len, dim]
        mask: [bs, seq_len, seq_len]

        Returns:
            output: [bs, seq_len, seq_len, dim]
        """
        batch_size, seq_len, _, dim = x.shape

        x = self.norm(x)

        left = self.left_proj(x)
        right = self.right_proj(x)

        mask = mask.unsqueeze(-1)
        left = left * mask
        right = right * mask

        left_gate = self.left_gate(x).sigmoid()
        right_gate = self.right_gate(x).sigmoid()
        out_gate = self.out_gate(x).sigmoid()

        left = left * left_gate
        right = right * right_gate

        out = einsum('... i k d, ... j k d -> ... i j d', left, right)
        # This einsum is the same as the following:
        # out = torch.zeros(batch_size, seq_len, seq_len, dim, device=x.device)
        
        # # Compute using nested loops
        # for b in range(batch_size):
        #     for i in range(seq_len):
        #         for j in range(seq_len):
        #             # Compute each output element
        #             for k in range(seq_len):
        #                 out[b, i, j] += left[b, i, k, :] * right[b, j, k, :]

        out = self.to_out_norm(out)
        out = out * out_gate
        return self.to_out(out)

"""
Fast outgoing TriMul (AlphaFold‑3) forward pass.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the input (shape B·N·N × dim) – implemented in
   Triton (`layernorm_kernel`).  Output is fp16.

2️⃣ Fuse the five linear layers (left/right projections + three gates)
   into a single half‑precision GEMM:
   `[R, dim] @ [5*hidden, dim].T → [R, 5*hidden]`.
   Slicing yields `left_proj`, `right_proj` and the three gate vectors.

3️⃣ Apply the optional mask (only on the two projected tensors) and the
   three sigmoid gates.

4️⃣ Triangular multiplicative update:
   `out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]`.
   This is exactly a batched matrix multiplication of shape
   `(B*hidden, N, N) @ (B*hidden, N, N).T`.  We use `torch.bmm`,
   which calls cuBLAS and fully exploits Tensor‑Cores on H100.

5️⃣ Second LayerNorm over the hidden dimension (again Triton).

6️⃣ Multiply by the output gate, project back to `dim` with a final
   half‑precision linear layer and cast to float32.

All heavy per‑row normalisations are done in a custom Triton kernel,
while the large N³ step uses the highly‑optimised cuBLAS batched GEMM.
The implementation supports both masked/unmasked inputs and works for
all test configurations listed in the problem statement.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (fp32 → fp16 output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,            # const *   [total_rows, dim]  (float32 or fp16)
    y_ptr,            # mutable * [total_rows, dim]  (fp16 output)
    weight_ptr,       # const *   [dim]              (float32)
    bias_ptr,         # const *   [dim]              (float32)
    total_rows,       # int (runtime)
    dim,              # tl.constexpr
    eps,              # tl.constexpr
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """
    Row‑wise LayerNorm.
    Computes:  y = (x - μ) / √(σ²+eps) * γ + β   (output fp16).
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # --------------------------------------------------------------
    # 1️⃣  Compute sum and sum‑of‑squares per row
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        # Load as fp32 regardless of the original precision
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)               # 1 / √(var+eps)

    # --------------------------------------------------------------
    # 2️⃣  Write normalized values γ·(x−μ)·rstd + β  (fp16)
    # --------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # β
        out = norm * w[None, :] + b[None, :]                     # fp32

        out_fp16 = out.to(tl.float16)
        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [B, N, N, dim] (float32, on the same CUDA device).
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    device = input_tensor.device

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N                     # rows for row‑wise LN

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → fp16
    # --------------------------------------------------------------
    # flatten to [R, dim]
    x_flat = input_tensor.view(total_rows, dim).contiguous()
    # allocate fp16 buffer for normalized data
    x_norm_flat = torch.empty(total_rows, dim,
                              dtype=torch.float16,
                              device=device)

    BLOCK_M = 256
    BLOCK_D = 32
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,                     # input pointer (fp32)
        x_norm_flat,                # output pointer (fp16)
        weights["norm.weight"],     # γ (fp32)
        weights["norm.bias"],       # β (fp32)
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # --------------------------------------------------------------
    # 2️⃣ Fuse five linear layers (2 projections + 3 gates)
    # --------------------------------------------------------------
    # concatenate the five weight matrices: [5*hidden, dim]
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)

    # single GEMM: [R, dim] @ [5*hidden, dim].T → [R, 5*hidden] (half‑precision)
    proj_concat = F.linear(x_norm_flat, w_concat)          # [R, 5*hidden]

    H = hidden_dim
    left_proj   = proj_concat[:, 0:H]                       # [R, H]
    right_proj  = proj_concat[:, H:2*H]                     # [R, H]
    left_gate_p = proj_concat[:, 2*H:3*H]                   # [R, H]
    right_gate_p = proj_concat[:, 3*H:4*H]                  # [R, H]
    out_gate_p  = proj_concat[:, 4*H:5*H]                   # [R, H]

    # sigmoid gates (still fp16)
    left_gate   = torch.sigmoid(left_gate_p)
    right_gate  = torch.sigmoid(right_gate_p)
    out_gate    = torch.sigmoid(out_gate_p)

    # --------------------------------------------------------------
    # 3️⃣ Optional mask (applied only to the projected tensors)
    # --------------------------------------------------------------
    if not config.get("nomask", True):
        # mask shape: [B, N, N] → flatten to [R, 1] (fp16)
        mask_h = mask_tensor.to(torch.float16).reshape(total_rows, 1)
        left_proj  = left_proj * mask_h
        right_proj = right_proj * mask_h

    # element‑wise gating
    left  = left_proj * left_gate
    right = right_proj * right_gate

    # reshape back to 4‑D tensors for the N³ update
    left  = left.view(B, N, N, H)          # [B, N, N, H]
    right = right.view(B, N, N, H)         # [B, N, N, H]
    out_gate = out_gate.view(B, N, N, H)   # [B, N, N, H]

    # --------------------------------------------------------------
    # 4️⃣ Triangular multiplicative update (batched GEMM)
    # --------------------------------------------------------------
    # permute to bring H to the batch dimension: [B, H, N, N]
    left_perm  = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    # fuse batch and head dimensions → shape [B*H, N, N]
    left_mat  = left_perm.view(B * H, N, N)
    right_mat = right_perm.view(B * H, N, N)

    # out_mat = left_mat @ right_matᵀ   (batched GEMM, uses cuBLAS Tensor‑Cores)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # restore original order: [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm over hidden dimension (fp16)
    # --------------------------------------------------------------
    out_flat = out.view(total_rows, H).contiguous()
    out_norm_flat = torch.empty_like(out_flat, dtype=torch.float16)

    grid_ln2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid_ln2](
        out_flat,
        out_norm_flat,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    out_norm = out_norm_flat.view(B, N, N, H)   # [B, N, N, H] (fp16)

    # --------------------------------------------------------------
    # 6️⃣ Apply output gate and final projection back to `dim`
    # --------------------------------------------------------------
    out_norm = out_norm * out_gate                # element‑wise (fp16)

    # final linear: [R, H] → [R, dim]   (half‑precision)
    final_half = F.linear(
        out_norm.view(total_rows, H),
        weights["to_out.weight"].to(torch.float16)   # [dim, H]
    )   # [R, dim] fp16

    # reshape and cast to float32 (as in the reference implementation)
    final = final_half.view(B, N, N, dim).to(torch.float32)

    return final
