"""
Highly‑optimized outgoing TriMul (AlphaFold‑3) forward pass.

Algorithm
---------
1️⃣ Row‑wise LayerNorm over the input channel dimension C.
   • 32‑bit accumulation for numerical stability.
   • 16‑bit output to reduce traffic (implemented in Triton).

2️⃣ Fuse the five linear projections (left/right + three gates) into a single
   half‑precision GEMM by concatenating the weight matrices.

3️⃣ Apply an optional pair‑wise mask to the left/right projections (broadcast).

4️⃣ Sigmoid on the three gate tensors and element‑wise gating of left/right.

5️⃣ Triangular multiplicative update (outgoing version):
      out_h[i, j] = Σ_k left_h[i, k] * right_h[j, k] .
   Implemented as a batched GEMM `left @ rightᵀ` where hidden
   dimension H is folded into the batch axis (BH = B·H).

6️⃣ Row‑wise LayerNorm over the hidden dimension H (again Triton).

7️⃣ Multiply by the output gate, then a final half‑precision linear projection
   back to the original channel dimension C.

8️⃣ Cast the result to float32 (the model expects fp32 output).

Only the LayerNorms are written in Triton; the dense linear algebra is
performed with PyTorch’s highly‑optimized Tensor‑Core kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (fp32 accumulate → fp16 output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,          # *const  T*   [total_rows, dim]   (input, fp16 or fp32)
    y_ptr,          # *mut    fp16* [total_rows, dim]   (output)
    weight_ptr,     # *const  fp32* [dim]               (γ)
    bias_ptr,       # *const  fp32* [dim]               (β)
    total_rows,     # i32
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # channel block size
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)              # [BLOCK_M]
    mask_m = offs_m < total_rows

    # --------------------------------------------------------------
    # 1. Compute mean & variance (fp32)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    # iterate over dim in chunks of BLOCK_D
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)        # [BLOCK_D]
        mask_d = offs_d < dim

        # 2‑D pointer: [row, col]
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        # load as the same dtype as the input tensor (fp16 or fp32) and cast to fp32
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                                   # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                     # [BLOCK_M]
        sum_sq  += tl.sum(vals * vals, axis=1)              # [BLOCK_M]

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                             # 1 / sqrt(var+eps)

    # --------------------------------------------------------------
    # 2. Write normalized output (fp16)
    # --------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        # (x - mean) * rsqrt(var+eps)
        norm = (vals - mean[:, None]) * rstd[:, None]      # fp32

        # γ and β are stored as fp32
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)

        out_fp16 = (norm * w[None, :] + b[None, :]).to(tl.float16)

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

def _choose_layernorm_block_sizes(dim: int):
    """Select sensible block sizes for the LayerNorm kernel."""
    # Larger BLOCK_M reduces grid launch overhead; keep it ≤1024 for register pressure.
    BLOCK_M = 1024 if dim <= 256 else 512
    # Choose BLOCK_D that evenly covers dim but stays ≤128 for good coalescing.
    if dim <= 64:
        BLOCK_D = dim
    elif dim <= 128:
        BLOCK_D = 128
    else:
        BLOCK_D = 64
    return BLOCK_M, BLOCK_D

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, C]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]    (bool/float, optional)
        - weights_dict : dict of fp32 weight tensors
        - config_dict  : {"dim": C, "hidden_dim": H, "nomask": bool}

    Returns
    -------
    torch.Tensor [B, N, N, C] (float32)
    """
    # --------------------------------------------------------------
    # Unpack and basic constants
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                # C
    hidden_dim = config["hidden_dim"]  # H
    eps = 1e-5
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 0. No‑grad context – we never need autograd for the forward only.
    # ------------------------------------------------------------------
    with torch.no_grad():
        B, N, _, _ = input_tensor.shape
        total_rows = B * N * N                     # flatten over (B,i,k)

        # --------------------------------------------------------------
        # 1️⃣  First LayerNorm (fp16 output)
        # --------------------------------------------------------------
        # flatten to (total_rows, C) – view does not copy memory
        x_flat = input_tensor.view(total_rows, dim).contiguous()

        # allocate fp16 buffer for the normalized tensor
        x_norm_fp16 = torch.empty(total_rows, dim,
                                  dtype=torch.float16,
                                  device=device)

        BLOCK_M, BLOCK_D = _choose_layernorm_block_sizes(dim)
        grid = (triton.cdiv(total_rows, BLOCK_M),)
        layernorm_fp16_kernel[grid](
            x_flat,
            x_norm_fp16,
            weights["norm.weight"],   # γ (fp32)
            weights["norm.bias"],     # β (fp32)
            total_rows,
            dim,
            eps,
            BLOCK_M=BLOCK_M,
            BLOCK_D=BLOCK_D,
        )

        # --------------------------------------------------------------
        # 2️⃣  Fuse the five linear projections (left/right + 3 gates)
        # --------------------------------------------------------------
        # Gather weight matrices (all are [hidden_dim, dim] and have no bias)
        w_left   = weights["left_proj.weight"]
        w_right  = weights["right_proj.weight"]
        w_lgate  = weights["left_gate.weight"]
        w_rgate  = weights["right_gate.weight"]
        w_outgate= weights["out_gate.weight"]

        # Concatenate along the output dimension → shape (5*H, C)
        # Store in half‑precision to enable Tensor‑Core GEMMs.
        concat_w = torch.cat(
            [w_left, w_right, w_lgate, w_rgate, w_outgate], dim=0
        ).half().contiguous()                       # (5*H, C)

        # Fused GEMM: (total_rows, C) @ (C, 5*H)ᵀ → (total_rows, 5*H)
        proj_all = F.linear(x_norm_fp16, concat_w)  # fp16 output

        # Split the result back into the five components
        left_proj_fp16, right_proj_fp16, left_gate_fp16, \
            right_gate_fp16, out_gate_fp16 = torch.split(
            proj_all, hidden_dim, dim=1
        )                                            # each [total_rows, H]

        # --------------------------------------------------------------
        # 3️⃣ Optional mask (only for left/right before gating)
        # --------------------------------------------------------------
        if not nomask:
            # mask shape [B, N, N] -> flatten to [total_rows, 1] and cast to fp16
            mask_fp16 = mask_tensor.view(total_rows, 1).to(torch.float16)
            left_proj_fp16  = left_proj_fp16 * mask_fp16
            right_proj_fp16 = right_proj_fp16 * mask_fp16

        # --------------------------------------------------------------
        # 4️⃣ Sigmoid gates (still half‑precision) and element‑wise gating
        # --------------------------------------------------------------
        left_gate   = torch.sigmoid(left_gate_fp16)      # [total_rows, H]
        right_gate  = torch.sigmoid(right_gate_fp16)
        out_gate    = torch.sigmoid(out_gate_fp16)

        left_fp16   = left_proj_fp16 * left_gate
        right_fp16  = right_proj_fp16 * right_gate

        # Reshape tensors back to 4‑D layout for the triangular update
        left  = left_fp16.view(B, N, N, hidden_dim)   # [B, N, N, H]
        right = right_fp16.view(B, N, N, hidden_dim)
        out_gate = out_gate.view(B, N, N, hidden_dim)

        # --------------------------------------------------------------
        # 5️⃣ Triangular multiplicative update (batched GEMM)
        #    out_h[i, j] = Σ_k left_h[i, k] * right_h[j, k]
        # --------------------------------------------------------------
        # Move hidden dimension into the batch axis: [B, H, N, N] → (B·H, N, N)
        left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
        right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

        BH = B * hidden_dim
        left_mat  = left_perm.view(BH, N, N)                 # [BH, N, N]
        right_mat = right_perm.view(BH, N, N)

        # Batched half‑precision GEMM: left @ rightᵀ  →  [BH, N, N]
        out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

        # Restore original 4‑D layout
        out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
        # out shape: [B, N, N, H]

        # --------------------------------------------------------------
        # 6️⃣ Second LayerNorm over the hidden dimension H (fp16 output)
        # --------------------------------------------------------------
        out_flat = out.view(total_rows, hidden_dim).contiguous()
        out_norm_fp16 = torch.empty_like(out_flat,
                                         dtype=torch.float16,
                                         device=device)

        BLOCK_M2, BLOCK_D2 = _choose_layernorm_block_sizes(hidden_dim)
        grid2 = (triton.cdiv(total_rows, BLOCK_M2),)
        layernorm_fp16_kernel[grid2](
            out_flat,
            out_norm_fp16,
            weights["to_out_norm.weight"],   # γ (fp32)
            weights["to_out_norm.bias"],     # β (fp32)
            total_rows,
            hidden_dim,
            eps,
            BLOCK_M=BLOCK_M2,
            BLOCK_D=BLOCK_D2,
        )

        out_norm = out_norm_fp16.view(B, N, N, hidden_dim)   # [B, N, N, H]

        # --------------------------------------------------------------
        # 7️⃣ Final gating + linear projection back to C
        # --------------------------------------------------------------
        gated = out_norm * out_gate                       # element‑wise fp16

        # Final projection weight (C, H) → we store as half‑precision
        to_out_w_fp16 = weights["to_out.weight"].half()

        gated_flat = gated.view(total_rows, hidden_dim)   # [M, H]
        final_fp16 = F.linear(gated_flat, to_out_w_fp16) # [M, C]

        # --------------------------------------------------------------
        # 8️⃣ Cast back to float32 and reshape to [B, N, N, C]
        # --------------------------------------------------------------
        final = final_fp16.view(B, N, N, dim).to(torch.float32)

        return final
