"""
custom_kernel – optimized “outgoing” TriMul forward pass (AlphaFold3).

Algorithm
---------
1️⃣ First LayerNorm (fp32 → fp16) with a custom Triton kernel.
2️⃣ Fuse the five linear projections (left/right + three gates) into a
   single half‑precision GEMM by concatenating their weight matrices.
3️⃣ Apply optional pair‑wise mask (broadcast to the hidden_dim).
4️⃣ Apply sigmoid gates and element‑wise gating (still fp16).
5️⃣ Triangular multiplicative update:
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   realized as a batched half‑precision GEMM (torch.bmm) on tensors of
   shape (B·hidden_dim, N, N).
6️⃣ Second LayerNorm over the hidden dimension (fp16) – same Triton kernel.
7️⃣ Multiply by the output gate, project back to `dim` with a final half‑
   precision linear layer and cast the result to float32 (required output).

All heavy reductions (the two LayerNorms) are performed by Triton;
dense linear algebra uses fp16 Tensor‑Core‑friendly cuBLAS kernels.
The implementation works for every test configuration (various
sequence lengths, batch sizes, optional mask, and different input
distributions).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton LayerNorm – fp32/fp16 input → fp16 output
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,          # *const   fp32/fp16   [total_rows, dim]
    y_ptr,          # *mut     fp16       [total_rows, dim]
    weight_ptr,     # *const   fp32       [dim]   (γ)
    bias_ptr,       # *const   fp32       [dim]   (β)
    total_rows,     # i32 (runtime)
    dim: tl.constexpr,          # i32 (compile‑time)
    eps: tl.constexpr,          # fp32 (compile‑time)
    BLOCK_M: tl.constexpr,      # rows per program
    BLOCK_D: tl.constexpr,      # cols per tile
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # --------------------------------------------------------------
    # Pass 1: compute per‑row mean & variance in fp32
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        # load as fp32 no matter the original precision
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------------------------------------------------------------
    # Pass 2: write normalized half‑precision values
    # --------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]     # (x-μ)/σ

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)

        out_fp32 = norm * w + b
        out_fp16 = out_fp32.to(tl.float16)

        out_ptr = y_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor, shape [B, N, N, C] (float32)
            - mask_tensor : torch.Tensor, shape [B, N, N] (bool/float) – may be ignored.
            - weights_dict : mapping with the ten weight tensors required by TriMul.
            - config_dict  : {"dim": C, "hidden_dim": H, "nomask": bool, ...}

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32) on the same CUDA device.
    """
    # --------------------------------------------------------------
    # 0️⃣ unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                 # C   (input & output channel size)
    hidden_dim = config["hidden_dim"]   # H   (internal projection size)
    eps = 1e-5
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (fp32 → fp16)
    # --------------------------------------------------------------
    B, N, _, _ = input_tensor.shape                 # B, N, N, C
    total_rows = B * N * N                           # flatten over (B,N,N)

    x_flat = input_tensor.view(total_rows, dim).contiguous()
    x_norm_fp16 = torch.empty((total_rows, dim),
                              dtype=torch.float16,
                              device=device)

    # launch parameters – enough to cover all dims up to 384
    BLOCK_M = 128
    BLOCK_D = 64
    grid = (triton.cdiv(total_rows, BLOCK_M),)

    layernorm_fp16_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # --------------------------------------------------------------
    # 2️⃣ Fused projections & gating (all in fp16)
    # --------------------------------------------------------------
    # Concatenate the five weight matrices (each [hidden_dim, dim]) → [5*H, C]
    concat_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).half().contiguous()                         # shape (5*H, C)

    # Single half‑precision GEMM: [R, C] @ [C, 5*H]^T → [R, 5*H]
    proj_all = F.linear(x_norm_fp16, concat_w)    # fp16

    # Split back into the five components
    left_proj, right_proj, left_gate, right_gate, out_gate = torch.split(
        proj_all, hidden_dim, dim=1
    )  # each [R, H] (fp16)

    # --------------------------------------------------------------
    # 2a️⃣ Optional pair‑wise mask (broadcast over hidden_dim)
    # --------------------------------------------------------------
    if not nomask:
        # mask shape [B, N, N] -> [R, 1] and promoted to fp16
        mask_flat = mask_tensor.view(total_rows, 1).to(dtype=torch.float16)
        left_proj = left_proj * mask_flat
        right_proj = right_proj * mask_flat

    # --------------------------------------------------------------
    # 2b️⃣ Sigmoid gates & element‑wise gating (still fp16)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    left_fp16 = left_proj * left_gate       # [R, H]
    right_fp16 = right_proj * right_gate    # [R, H]

    # --------------------------------------------------------------
    # 3️⃣ Reshape for the triangular multiplicative update
    # --------------------------------------------------------------
    left = left_fp16.view(B, N, N, hidden_dim)          # [B, N, N, H]
    right = right_fp16.view(B, N, N, hidden_dim)        # [B, N, N, H]
    out_gate = out_gate.view(B, N, N, hidden_dim)      # [B, N, N, H]

    # Prepare tensors for batched GEMM: (i,k) @ (k,j) → (i,j)
    # left_perm:  [B, H, i, k]   ; right_perm: [B, H, k, j]
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 2, 1).contiguous()

    BH = B * hidden_dim
    left_mat = left_perm.view(BH, N, N)    # [BH, N, N]  (i,k)
    right_mat = right_perm.view(BH, N, N)  # [BH, N, N]  (k,j)

    # Batched half‑precision matrix multiply – uses Tensor Cores
    out_mat = torch.bmm(left_mat, right_mat)   # [BH, N, N] (i,j), fp16

    # Restore 4‑D layout
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    # --------------------------------------------------------------
    # 4️⃣ Second LayerNorm over the hidden dimension (fp16)
    # --------------------------------------------------------------
    total_rows_h = B * N * N
    out_flat = out.view(total_rows_h, hidden_dim).contiguous()
    out_norm_fp16 = torch.empty_like(out_flat)   # fp16 buffer

    grid2 = (triton.cdiv(total_rows_h, BLOCK_M),)
    layernorm_fp16_kernel[grid2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows_h,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    out_norm = out_norm_fp16.view(B, N, N, hidden_dim)   # fp16

    # --------------------------------------------------------------
    # 5️⃣ Final gating + projection back to `dim`
    # --------------------------------------------------------------
    gated = out_norm * out_gate                     # fp16 element‑wise

    # Linear projection H -> C (no bias) – half‑precision weight
    to_out_w = weights["to_out.weight"].half()
    gated_flat = gated.view(total_rows_h, hidden_dim)   # [R_h, H]
    final_fp16 = F.linear(gated_flat, to_out_w)        # [R_h, C]  fp16

    # Cast to float32 as required by the specification
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
