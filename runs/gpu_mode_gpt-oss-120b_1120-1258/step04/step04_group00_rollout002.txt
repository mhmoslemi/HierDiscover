"""
custom_kernel – TriMul “outgoing” (AlphaFold‑3) forward pass.

Algorithm Overview
------------------
1️⃣ Row‑wise LayerNorm (FP32 → FP16) on the 4‑D input.
2️⃣ One fused linear layer produces all projections, gate pre‑activations and the
   output‑gate in a single matmul (weight shape (5·H, D)).  The tensor is then
   sliced into
   * left_proj, left_gate, right_proj, right_gate, out_gate.
3️⃣ Optional pair‑wise mask is applied only to the two projections.
4️⃣ Sigmoid gates are computed, and the masked projections are multiplied by
   their gates → ``left`` and ``right`` (FP16, shape [B,N,N,H]).
5️⃣ Tri‑multiplicative update: for each hidden channel we perform a batched
   matrix‑multiply ``left @ rightᵀ`` (B·H independent GEMMs, FP16 Tensor‑Core).
6️⃣ Second LayerNorm (over the hidden dimension) – again FP32 → FP16.
7️⃣ Apply the output‑gate, a final linear projection (FP16), and cast the result
   back to FP32.

Only the two LayerNorms are written in Triton; all dense linear algebra and
the core N×N batch GEMM use PyTorch’s highly‑optimised FP16 kernels.  This
fused‑linear‑plus‑gate formulation reduces kernel launch overhead and memory
traffic, meeting the sub‑microsecond runtime target on H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (FP32 mean/var) → FP16 output
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,          # const float*   [total_rows, dim]
    y_ptr,          #       half*   [total_rows, dim] (output)
    w_ptr,          # const float*  [dim]   (γ)
    b_ptr,          # const float*  [dim]   (β)
    total_rows,     # int (runtime)
    dim: tl.constexpr,          # static: number of columns
    eps: tl.constexpr,          # static epsilon
    BLOCK_M: tl.constexpr,      # rows per program
    BLOCK_D: tl.constexpr,      # columns per inner loop
):
    """Row‑wise LayerNorm over a 2‑D tensor. Computes mean/var in FP32,
    writes normalized FP16 result."""
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------ Pass 1: mean & variance -------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)      # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------ Pass 2: write normalized values ------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)      # [BLOCK_M, BLOCK_D]

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # β

        out = norm * w[None, :] + b[None, :]                 # FP32
        out_fp16 = out.to(tl.float16)

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    TriMul “outgoing” forward pass.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        input_tensor : torch.Tensor [B, N, N, D] (float32)
        mask_tensor  : torch.Tensor [B, N, N]   (float32) – may be unused
        weights_dict : mapping weight name → torch.Tensor
        config_dict  : {
            "dim"        : int,
            "hidden_dim": int,
            "nomask"    : bool (optional, default True)
        }

    Returns
    -------
    torch.Tensor [B, N, N, D] (float32, same device as input)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    H = config["hidden_dim"]          # hidden dimension
    eps = 1e-5
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    total_rows = B * N * N
    x_flat = input_tensor.view(total_rows, dim).contiguous()                # [R, D]
    x_norm_h = torch.empty((total_rows, dim), dtype=torch.float16,
                           device=device)                                   # [R, D]

    BLOCK_M = 256  # rows per program (tuned)
    BLOCK_D = 64   # columns per inner loop (covers D ≤ 384)
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        x_flat,
        x_norm_h,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # reshape to 4‑D (still FP16)
    x_norm_h = x_norm_h.view(B, N, N, dim)

    # ------------------------------------------------------------------
    # 2️⃣ One fused linear layer for left/right projections, gates and out‑gate
    # ------------------------------------------------------------------
    # Assemble combined weight: [5*H, dim]
    combined_w = torch.cat(
        [
            weights["left_proj.weight"],          # [H, dim]
            weights["left_gate.weight"],          # [H, dim]
            weights["right_proj.weight"],         # [H, dim]
            weights["right_gate.weight"],         # [H, dim]
            weights["out_gate.weight"],           # [H, dim]
        ],
        dim=0,
    ).to(torch.float16, non_blocking=True)        # [5H, dim]

    rows = total_rows
    # F.linear: (rows, dim) @ (5H, dim).T -> (rows, 5H)
    combined_h = F.linear(x_norm_h.view(rows, dim), combined_w)  # FP16

    # ------------------------------------------------------------------
    # 3️⃣ Slice and apply mask + sigmoid gates
    # ------------------------------------------------------------------
    # Slicing indices
    h0 = 0
    h1 = H
    h2 = 2 * H
    h3 = 3 * H
    h4 = 4 * H
    h5 = 5 * H

    left_proj_h      = combined_h[:, h0:h1]      # [R, H]
    left_gate_pre_h  = combined_h[:, h1:h2]      # [R, H]
    right_proj_h     = combined_h[:, h2:h3]      # [R, H]
    right_gate_pre_h = combined_h[:, h3:h4]      # [R, H]
    out_gate_pre_h   = combined_h[:, h4:h5]      # [R, H]

    if not nomask:
        # mask: [B,N,N] → [R,1] in FP16
        mask_h = mask_tensor.view(rows, 1).to(torch.float16, non_blocking=True)
        left_proj_h = left_proj_h * mask_h
        right_proj_h = right_proj_h * mask_h

    # gates
    left_gate_h  = torch.sigmoid(left_gate_pre_h)    # [R, H]
    right_gate_h = torch.sigmoid(right_gate_pre_h)   # [R, H]
    out_gate_h   = torch.sigmoid(out_gate_pre_h)     # [R, H]

    # apply gates
    left_h  = left_proj_h * left_gate_h    # [R, H]
    right_h = right_proj_h * right_gate_h  # [R, H]

    # ------------------------------------------------------------------
    # 4️⃣ Tri‑multiplicative update (batched GEMM over hidden dim)
    # ------------------------------------------------------------------
    # reshape for batched GEMM: [B, H, N, N]
    left_perm  = left_h.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    right_perm = right_h.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()

    BH = B * H                         # effective batch size
    left_mat  = left_perm.view(BH, N, N)        # [BH, N, N]
    right_mat = right_perm.view(BH, N, N)       # [BH, N, N]

    # out_mat = left @ rightᵀ  (sum over k)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [BH, N, N]

    # reshape back to [B, N, N, H] (still FP16)
    out = (
        out_mat.view(B, H, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm (over hidden dimension)
    # ------------------------------------------------------------------
    rows_h = B * N * N
    out_flat = out.view(rows_h, H).contiguous()
    out_norm_h = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(rows_h, BLOCK_M),)
    layernorm_fp16_kernel[grid2](
        out_flat,
        out_norm_h,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        rows_h,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    out_norm = out_norm_h.view(B, N, N, H)   # FP16

    # ------------------------------------------------------------------
    # 6️⃣ Apply output‑gate and final projection
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate_h.view(B, N, N, H)   # gated output

    final_h = F.linear(
        out_norm.view(rows_h, H),
        weights["to_out.weight"].to(torch.float16, non_blocking=True)
    )   # [R, dim]  (FP16)

    # Cast back to FP32 as required by the original module
    final = final_h.view(B, N, N, dim).to(torch.float32)

    return final
