"""
Highly‑optimized Triton + PyTorch forward pass for the **outgoing** TriMul
operator (AlphaFold‑3).

Algorithm
---------
1️⃣  Input LayerNorm (FP16) – Triton kernel that computes the mean/var in
    FP32, writes a half‑precision normalised tensor.

2️⃣  **Two** fused linear + gating passes (half‑precision):
    * left‑side:  proj + gate  →  left = proj * sigmoid(gate)
    * right‑side: proj + gate  →  right = proj * sigmoid(gate)
   Each pass reads the normalised tensor once, avoiding the large
   temporary `[R,5*H]` buffer from the reference implementation.

3️⃣  (Optional) binary mask – applied after gating (mask is a scalar per
    row, cheap to broadcast).

4️⃣  Triangular multiplicative update – batched matrix‑multiply
    `out[b,i,j,h] = Σk left[b,i,k,h] * right[b,j,k,h]`.  Implemented with a
    single `torch.bmm` on the reshaped tensors (cuBLAS/Tensor‑Core).

5️⃣  Second LayerNorm + output‑gate – a fused Triton kernel that combines
    the per‑row mean/var computation, the learned scale/bias, and the
    element‑wise multiplication with the previously computed `out_gate`.

6️⃣  Final linear projection (hidden → dim) – a regular FP16 GEMM
    (`torch.nn.functional.linear`).  The result is cast back to FP32.

The implementation reduces memory traffic (only two reads of the
normalised tensor, no `[R,5*H]` intermediate) and keeps all heavy
computations in FP16/Tensor‑Core while preserving FP32 accuracy for the
LayerNorm reductions.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1) Row‑wise LayerNorm (FP16 output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,               # *const float*  [total_rows, dim] (FP32)
    y_ptr,               # *mut   half*   [total_rows, dim] (FP16)
    weight_ptr,          # *const float*  [dim]          (γ)
    bias_ptr,            # *const float*  [dim]          (β)
    total_rows: tl.int32,
    dim: tl.constexpr,   # static (≤ 384)
    eps: tl.constexpr,   # epsilon
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ---------------------------------------------------------------
    # First pass: mean & variance (FP32 accumulation)
    # ---------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)      # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                         # 1/√(var+eps)

    # ---------------------------------------------------------------
    # Second pass: normalise, apply γ, β and store as FP16
    # ---------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]            # (x-µ)/σ

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]                      # FP32

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 6) Fused LayerNorm + out‑gate (both FP16)
# ----------------------------------------------------------------------
@triton.jit
def fused_layernorm_gate_kernel(
    x_ptr,               # *const half*  [total_rows, hidden] (pre‑norm)
    y_ptr,               # *mut   half*  [total_rows, hidden] (post‑norm⋅gate)

    bias_ptr,            # *const float* [hidden] (β)
    gate_ptr,            # *const half*  [total_rows, hidden] (out_gate, already sigmoid’ed)
    total_rows: tl.int32,
    hidden: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ---------------------------------------------------------------
    # First pass: mean / variance (FP32)
    # ---------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(hidden, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < hidden
        ptrs = x_ptr + offs_m[:, None] * hidden + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # FP32

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / hidden
    var  = sum_sq / hidden - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------------------------------------------------------------
    # Second pass: normalise, apply γ, β, multiply by out_gate, store FP16
    # ---------------------------------------------------------------
    for d in range(0, tl.cdiv(hidden, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < hidden
        ptrs = x_ptr + offs_m[:, None] * hidden + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # FP32

        norm = (vals - mean[:, None]) * rstd[:, None]   # (x‑µ)/σ

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # β
        out = norm * w[None, :] + b[None, :]            # FP32

        gate_vals = tl.load(gate_ptr + offs_m[:, None] * hidden + offs_d[None, :],
                            mask=mask_m[:, None] & mask_d[None, :],
                            other=0.0).to(tl.float32)          # FP32
        out = out * gate_vals

        tl.store(y_ptr + offs_m[:, None] * hidden + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the **outgoing** TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, C] (float32)
        - mask_tensor  : torch.Tensor [B, N, N]   (float32/bool) – ignored if config["nomask"]=True
        - weights_dict : model parameters (float32)
        - config_dict  : {"dim":C, "hidden_dim":H, "nomask":bool, "eps":float (opt.)}

    Returns
    -------
    torch.Tensor [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    inp, mask, weights, cfg = data
    device = inp.device

    dim    = cfg["dim"]                   # C
    hidden = cfg["hidden_dim"]            # H
    eps    = cfg.get("eps", 1e-5)
    nomask = cfg.get("nomask", True)

    B, N, _, _ = inp.shape
    total_rows = B * N * N                      # rows for flattened tensors

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP16 output)
    # ------------------------------------------------------------------
    inp_flat = inp.view(total_rows, dim).contiguous()          # [R, C] float32
    x_norm_fp16 = torch.empty(total_rows, dim, dtype=torch.half,
                              device=device)                # [R, C] half

    # Choose block sizes that fit the given dimension without wasting threads.
    BLOCK_M = 128                              # rows per program (more rows → fewer launches)
    BLOCK_D = 128 if dim > 256 else 64        # columns per iteration (covers dim efficiently)

    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        inp_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Two fused linear‑+‑gate passes (left & right)
    #    Each pass produces: proj (half) and gate (half) → gated output.
    # ------------------------------------------------------------------
    # Cast all weight matrices to FP16 once (half‑precision GEMM)
    left_proj_w  = weights["left_proj.weight"].half().contiguous()
    left_gate_w  = weights["left_gate.weight"].half().contiguous()
    right_proj_w = weights["right_proj.weight"].half().contiguous()
    right_gate_w = weights["right_gate.weight"].half().contiguous()
    out_gate_w   = weights["out_gate.weight"].half().contiguous()

    # ------------------------ left side ------------------------
    left_comb_w = torch.cat([left_proj_w, left_gate_w], dim=0)      # (2*H, C)
    left_comb   = F.linear(x_norm_fp16, left_comb_w)               # (R, 2*H)
    left_proj_raw, left_gate_raw = torch.split(left_comb, hidden, dim=1)
    left_gate = torch.sigmoid(left_gate_raw)                       # (R, H)
    left = left_proj_raw * left_gate                               # (R, H)

    # ------------------------ right side ------------------------
    right_comb_w = torch.cat([right_proj_w, right_gate_w], dim=0)    # (2*H, C)
    right_comb   = F.linear(x_norm_fp16, right_comb_w)              # (R, 2*H)
    right_proj_raw, right_gate_raw = torch.split(right_comb, hidden, dim=1)
    right_gate = torch.sigmoid(right_gate_raw)                      # (R, H)
    right = right_proj_raw * right_gate                            # (R, H)

    # ------------------------ out‑gate -------------------------
    out_gate_raw = F.linear(x_norm_fp16, out_gate_w)               # (R, H)
    out_gate = torch.sigmoid(out_gate_raw)                         # (R, H) – used later

    # ------------------------------------------------------------------
    # 2a) Apply binary mask (if present) – broadcast on hidden dim.
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape (B,N,N) → (R,1) as half
        mask_flat = mask.view(total_rows, 1).to(torch.float16)
        left = left * mask_flat
        right = right * mask_flat

    # ------------------------------------------------------------------
    # 3️⃣ Triangular multiplicative update (batched matrix‑multiply)
    # ------------------------------------------------------------------
    # Reshape to [B, N, N, H]
    left  = left.view(B, N, N, hidden)    # (B, N, N, H)
    right = right.view(B, N, N, hidden)   # (B, N, N, H)

    # Prepare (B*H, N, N) tensors for the batched GEMM
    left_mat  = left.permute(0, 3, 1, 2).reshape(B * hidden, N, N)   # (B*H, N, N)
    right_mat = right.permute(0, 3, 2, 1).reshape(B * hidden, N, N)  # (B*H, N, N)

    # Batched matmul → (B*H, N, N)
    out_mat = torch.bmm(left_mat, right_mat)

    # Reshape back to [B, N, N, H]
    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 4️⃣ Second LayerNorm + output‑gate (fused Triton kernel)
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, hidden).contiguous()   # (R, H) half
    out_norm_gate = torch.empty_like(out_flat)            # (R, H) half

    # Same block configuration works well for the hidden dimension
    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    fused_layernorm_gate_kernel[grid2](
        out_flat,
        out_norm_gate,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        out_gate,                                       # already sigmoid’ed, half
        total_rows,
        hidden,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 5️⃣ Final linear projection back to `dim` (FP16 GEMM)
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].half()            # (dim, H)
    final_fp16 = F.linear(out_norm_gate, to_out_w)        # (R, dim) half
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
