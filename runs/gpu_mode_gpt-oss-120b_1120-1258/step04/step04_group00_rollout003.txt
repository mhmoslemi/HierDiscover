"""
TriMul “outgoing” forward pass (AlphaFold‑3) – Triton + PyTorch implementation.

Algorithm
---------
1️⃣  Layer‑norm on the input tensor (shape [B,N,N,dim]) → FP16.
2️⃣  Fuse each projection with its gate:
        (dim → hidden) + (dim → hidden) → (dim → 2·hidden)  via a single `F.linear`.
3️⃣  Optional pair‑wise mask (broadcast over the hidden dimension) is applied
    to the projected tensors before gating.
4️⃣  Apply the sigmoid gates.
5️⃣  Tri‑multiplicative update:
        out_h(i,j) = Σ_k left(i,k,h)·right(j,k,h)
    is realized as a batched GEMM over the hidden dimension
    (`torch.bmm`).  This is the dominant compute kernel and runs on the
    Tensor‑Core‑accelerated cuBLAS implementation.
6️⃣  Second Layer‑norm (over the hidden dimension) → FP16 (tiny Triton kernel).
7️⃣  Multiply by the output gate (sigmoid) and a final linear projection
    (hidden → dim).  The result is cast back to FP32.

Only the two LayerNorms are hand‑written in Triton; everything else uses
high‑throughput PyTorch kernels (FP16 matmuls, element‑wise ops, etc.).
The implementation works for any batch size ≤2, sequence length up to 1024,
and hidden/feature dimensions used in the test matrix.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm that reads FP32, computes mean/var in
# FP32, and writes the normalized result in FP16.
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,          # float*   [total_rows, dim]   (input)
    y_ptr,          # half*    [total_rows, dim]   (output)
    gamma_ptr,      # float*   [dim]               (scale)
    beta_ptr,       # float*   [dim]               (bias)
    total_rows,     # int (runtime)
    dim: tl.constexpr,          # static: number of columns
    eps: tl.constexpr,          # static epsilon
    BLOCK_M: tl.constexpr,      # rows processed per program
    BLOCK_D: tl.constexpr,      # column block size
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)              # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # Pass 1 – compute mean and variance for each row
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)        # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)           # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                     # sum over dim
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                               # 1 / sqrt(var+eps)

    # ------------------------------------------------------------------
    # Pass 2 – write normalized values (γ·x̂+β) as FP16
    # ------------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        # (x - μ) / σ
        norm = (vals - mean[:, None]) * rstd[:, None]

        # γ, β
        w = tl.load(gamma_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(beta_ptr  + offs_d, mask=mask_d, other=0.0)

        out = norm * w[None, :] + b[None, :]                # FP32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

        input_tensor : torch.Tensor[float32]  [B, N, N, dim]
        mask_tensor  : torch.Tensor[float32] [B, N, N]   (may be unused)
        weights_dict : {
            "norm.weight":           Tensor[dim],
            "norm.bias":             Tensor[dim],
            "left_proj.weight":      Tensor[hidden_dim, dim],
            "right_proj.weight":     Tensor[hidden_dim, dim],
            "left_gate.weight":      Tensor[hidden_dim, dim],
            "right_gate.weight":     Tensor[hidden_dim, dim],
            "out_gate.weight":       Tensor[hidden_dim, dim],
            "to_out_norm.weight":    Tensor[hidden_dim],
            "to_out_norm.bias":      Tensor[hidden_dim],
            "to_out.weight":         Tensor[dim, hidden_dim],
        }
        config_dict  : {
            "dim": int,
            "hidden_dim": int,
            "nomask": bool (optional, default True),
        }

    Returns:
        torch.Tensor[float32]  [B, N, N, dim]
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32 → FP16) on the 4‑D tensor
    # ------------------------------------------------------------------
    total_rows = B * N * N
    x_flat = input_tensor.view(total_rows, dim).contiguous()          # [rows, dim]

    # allocate output buffer (FP16)
    x_norm_h = torch.empty((total_rows, dim), dtype=torch.float16,
                           device=device)

    # kernel launch configuration
    BLOCK_M = 128          # rows per program – 128 fits well in shared memory
    BLOCK_D = 64           # column block (covers dim ≤ 384)
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        x_flat,
        x_norm_h,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # reshape back to [B, N, N, dim] (still FP16)
    x_norm_h = x_norm_h.view(B, N, N, dim)

    # ------------------------------------------------------------------
    # 2️⃣ Fuse projection + gate for the left and right paths
    # ------------------------------------------------------------------
    # Concatenate projection and gate weights (dim → 2·hidden_dim)
    left_concat_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16, non_blocking=True)   # [2·H, dim]

    right_concat_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16, non_blocking=True)   # [2·H, dim]

    rows = total_rows
    # Linear: (rows, dim) @ (2·H, dim).T  → (rows, 2·H)
    left_concat = F.linear(x_norm_h.view(rows, dim), left_concat_w)   # [rows, 2·H]
    right_concat = F.linear(x_norm_h.view(rows, dim), right_concat_w)

    # Split projection / pre‑gate tensors
    left_proj_h      = left_concat[:, :hidden_dim]          # [rows, H]
    left_gate_pre_h  = left_concat[:, hidden_dim:]          # [rows, H]
    right_proj_h     = right_concat[:, :hidden_dim]
    right_gate_pre_h = right_concat[:, hidden_dim:]

    # Out‑gate (shared across positions)
    out_gate_h = torch.sigmoid(
        F.linear(
            x_norm_h.view(rows, dim),
            weights["out_gate.weight"].to(torch.float16, non_blocking=True)
        )
    )   # [rows, H]

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over hidden_dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask_h = mask_tensor.view(rows, 1).to(torch.float16, non_blocking=True)
        left_proj_h = left_proj_h * mask_h
        right_proj_h = right_proj_h * mask_h

    # ------------------------------------------------------------------
    # 4️⃣ Apply sigmoid gates
    # ------------------------------------------------------------------
    left_gate_h  = torch.sigmoid(left_gate_pre_h)
    right_gate_h = torch.sigmoid(right_gate_pre_h)

    left_h  = left_proj_h * left_gate_h    # [rows, H]
    right_h = right_proj_h * right_gate_h  # [rows, H]

    # ------------------------------------------------------------------
    # 5️⃣ Tri‑multiplicative update (batched GEMM over hidden_dim)
    # ------------------------------------------------------------------
    # reshape to [B, N, N, H]
    left  = left_h.view(B, N, N, hidden_dim)
    right = right_h.view(B, N, N, hidden_dim)

    # bring H into batch dimension → [B·H, N, N]
    left_perm  = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    BH = B * hidden_dim
    left_mat  = left_perm.view(BH, N, N)          # [BH, N, N]
    right_mat = right_perm.view(BH, N, N)         # [BH, N, N]

    # GEMM: (N×N) @ (N×N)^T  → sum over k dimension
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [BH, N, N]

    # reshape back to [B, N, N, H] (still FP16)
    out = (
        out_mat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm (over hidden_dim) → FP16
    # ------------------------------------------------------------------
    rows_h = B * N * N
    out_flat = out.view(rows_h, hidden_dim).contiguous()
    out_norm_h = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(rows_h, BLOCK_M),)
    layernorm_fp16_kernel[grid2](
        out_flat,
        out_norm_h,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        rows_h,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    out_norm = out_norm_h.view(B, N, N, hidden_dim)   # FP16

    # ------------------------------------------------------------------
    # 7️⃣ Apply output‑gate and final projection
    # ------------------------------------------------------------------
    out_gate = out_gate_h.view(B, N, N, hidden_dim)   # FP16
    gated = out_norm * out_gate                       # element‑wise FP16

    final_h = F.linear(
        gated.view(rows_h, hidden_dim),
        weights["to_out.weight"].to(torch.float16, non_blocking=True)
    )   # [rows_h, dim]  (FP16)

    # Cast back to FP32 to match the original module's output dtype
    final = final_h.view(B, N, N, dim).to(torch.float32)

    return final
