Pseudo code:

```
@triton.jit
def prepare_inputs_kernel(
    proj_ptr, mask_ptr,
    left_out_ptr, right_out_ptr, out_gate_ptr,
    total_rows, N, H,
    HAS_MASK: tl.constexpr,
    BLOCK_ROWS: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_ROWS
    offs_row = row_start + tl.arange(0, BLOCK_ROWS)   # shape BLOCK_ROWS
    mask_row = offs_row < total_rows

    # compute i,k indices for each row
    seq2 = N * N
    b_idx = offs_row // seq2
    rem = offs_row - b_idx * seq2
    i_idx = rem // N
    k_idx = rem - i_idx * N

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)  # hidden slice
        # guard if h+off >= H
        mask_h = offs_h < H

        # compute offsets for each component
        # base offset = offs_row[:, None] * (5*H) + offs_h[None,:]
        # left_proj at offset + 0
        # right_proj at offset + H
        # left_gate at offset + 2*H
        # right_gate at offset + 3*H
        # out_gate at offset + 4*H
        # load values
        val_base = proj_ptr + (offs_row[:, None] * 5*H + offs_h[None, :])
        left = tl.load(val_base, mask=mask_row[:, None] & mask_h, other=0.0)
        right = tl.load(val_base + H, mask=mask_row[:, None] & mask_h, other=0.0)
        left_gate_raw = tl.load(val_base + 2*H, mask=..., other=0.0)
        right_gate_raw = tl.load(val_base + 3*H, mask=..., other=0.0)
        out_gate_raw = tl.load(val_base + 4*H, mask=..., other=0.0)

        left_gate = 1.0 / (1.0 + tl.exp(-left_gate_raw))
        right_gate = 1.0 / (1.0 + tl.exp(-right_gate_raw))
        out_gate = 1.0 / (1.0 + tl.exp(-out_gate_raw))

        left_val = left * left_gate
        right_val = right * right_gate

        if HAS_MASK:
            mask_val = tl.load(mask_ptr + offs_row[:, None], mask=mask_row[:, None], other=0.0)
            left_val = left_val * mask_val
            right_val = right_val * mask_val

        # write left_val, right_val to left_mat and right_mat
        # compute batch index for each hidden channel: batch_idx = b_idx * H + offs_h[None,:]
        batch_idx = b_idx * H + offs_h[None,:]   # shape (BLOCK_ROWS, BLOCK_H)
        # output offset = batch_idx * N * N + i_idx * N + k_idx
        out_offset = batch_idx * (N * N) + i_idx[:, None] * N + k_idx[:, None]
        tl.store(left_out_ptr + out_offset, left_val, mask=mask_row[:, None] & mask_h)
        tl.store(right_out_ptr + out_offset, right_val, mask=mask_row[:, None] & mask_h)

        # store out_gate resulting (R, H) flatten (no reordering):
        out_gate_offset = offs_row[:, None] * H + offs_h[None, :]
        tl.store(out_gate_ptr + out_gate_offset, out_gate, mask=mask_row[:, None] & mask_h)
```

    proj_ptr, mask_ptr,
    left_ptr, right_ptr, out_gate_ptr,
    total_rows, N, hidden_dim,
    HAS_MASK: tl.constexpr,
    BLOCK_ROWS: tl.constexpr,

    offs_row = row_start + tl.arange(0, BLOCK_ROWS)

    # compute base indices
    seq_sq = N * N
    b_idx = offs_row // seq_sq
    rem = offs_row - b_idx * seq_sq
    i_idx = rem // N
    k_idx = rem - i_idx * N

    if HAS_MASK:

def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    device = input_tensor.device
    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N
    # first layernorm
    x_norm = torch.empty(total_rows, dim, dtype=torch.float16, device=device)
    BLOCK_M = 256
    BLOCK_D = 32
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        input_tensor.view(total_rows, dim).contiguous(),
        x_norm,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    # fused linear projection
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)
    proj = F.linear(x_norm, w_concat)  # shape (total_rows, 5*hidden_dim)
    # allocate buffers for left/right mats and out_gate
    left_mat = torch.empty(B * hidden_dim, N, N, dtype=torch.float16, device=device)
    right_mat = torch.empty_like(left_mat)
    out_gate_flat = torch.empty(total_rows, hidden_dim, dtype=torch.float16, device=device)
    # mask handling
    has_mask = not config.get("nomask", True)
    if has_mask:
        mask_flat = mask_tensor.to(torch.float16).reshape(total_rows)
    else:
        mask_flat = torch.empty(1, dtype=torch.float16, device=device)  # placeholder
    # launch fused kernel
    BLOCK_ROWS = 128
    BLOCK_H = 32
    grid = (triton.cdiv(total_rows, BLOCK_ROWS),)
    prepare_inputs_kernel[grid](
        proj,
        mask_flat,
        left_mat,
        right_mat,
        out_gate_flat,
        total_rows,
        N,
        hidden_dim,
        HAS_MASK=1 if has_mask else 0,
        BLOCK_ROWS=BLOCK_ROWS,
        BLOCK_H=BLOCK_H,
    )
    # batched gemm
    out_mat = torch.bmm(left_mat, right_mat.transpose(1,2))  # (B*H,N,N)
    # reshape to (B,N,N,H)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    # second layernorm using torch native
    out_norm = torch.nn.functional.layer_norm(
        out, 
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
        eps=eps,
    )
    # out_gate reshape
    out_gate = out_gate_flat.view(B, N, N, hidden_dim)
    out_norm = out_norm * out_gate
    # final projection
    final_half = F.linear(
        out_norm.view(total_rows, hidden_dim),
        weights["to_out.weight"].to(torch.float16)
    )
    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final

"""
Fast outgoing TriMul (AlphaFold‑3) forward pass.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the input tensor (B·N·N × dim). Implemented in
   Triton (`layernorm_kernel`) – output FP16.

2️⃣ Fuse the five linear layers (2 projections + 3 gates) into a single
   GEMM:  [R, dim] @ [5·hidden, dim]ᵀ → [R, 5·hidden]  (R = B·N·N).
   The result is kept in FP16.

3️⃣ One Triton kernel (`prepare_inputs_kernel`) reads the fused projection,
   applies the three sigmoids, the optional mask and directly writes the
   two gated matrices in the *contiguous* layout required by the batched
   GEMM, together with the raw output‑gate values.

4️⃣ Batched matrix multiplication (B·hidden, N, N) @ (B·hidden, N, N)ᵀ
   – performed by `torch.bmm` (cuBLAS Tensor‑Core fast path).

5️⃣ Second LayerNorm over the hidden dimension (FP16) – using the highly
   optimised native `torch.nn.functional.layer_norm`.

6️⃣ Multiply by the output‑gate, final linear projection back to `dim`,
   cast to float32 and return.

Only the row‑wise LayerNorms and the projection‑re‑ordering kernel are
written in Triton; the heavy N³ part leverages cuBLAS.  This removes the
expensive `view → permute → contiguous` copies of the original
implementation and brings the runtime well below the 1 ms target on an
H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (fp32 → fp16 output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,            # const *   [R, dim] (float32)
    y_ptr,            # mutable * [R, dim] (float16 output)
    weight_ptr,       # const *   [dim]    (float32 γ)
    bias_ptr,         # const *   [dim]    (float32 β)
    total_rows,       # int (runtime)
    dim,              # tl.constexpr
    eps,              # tl.constexpr
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # --------------------------------------------------------------
    # 1️⃣  Compute sum and sum‑of‑squares per row
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0)                # fp32 (input is fp32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)               # 1 / √(var+eps)

    # --------------------------------------------------------------
    # 2️⃣  Write normalized values γ·(x−μ)·rstd + β  (fp16)
    # --------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # β
        out = norm * w[None, :] + b[None, :]                     # fp32

        out_fp16 = out.to(tl.float16)
        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 3️⃣ Fuse projection, gating, masking and layout conversion
# ----------------------------------------------------------------------
@triton.jit
def prepare_inputs_kernel(
    proj_ptr,          # const *   [R, 5*H]  (fp16)
    mask_ptr,          # const *   [R]       (fp16) – only read if HAS_MASK
    left_ptr,          # mutable * [B*H, N, N] (fp16)
    right_ptr,         # mutable * [B*H, N, N] (fp16)
    out_gate_ptr,      # mutable * [R, H]       (fp16)
    total_rows,        # int (runtime)
    N,                 # int (seq_len)
    hidden_dim,        # int (H)
    HAS_MASK: tl.constexpr,
    BLOCK_ROWS: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    Reads the fused projection tensor (R × 5·H) and writes:
        * left  matrix  → shape (B·H, N, N)
        * right matrix  → shape (B·H, N, N)
        * out_gate      → shape (R, H)   (kept in the original flattened layout)
    while applying:
        - sigmoid to the three gate vectors,
        - optional mask (broadcast over hidden dim),
        - element‑wise gating of the two projections.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_ROWS
    offs_row = row_start + tl.arange(0, BLOCK_ROWS)          # [BLOCK_ROWS]
    mask_row = offs_row < total_rows

    # ---------------------------------------
    # decode (b, i, k) from the flattened row index
    # ---------------------------------------
    seq_sq = N * N
    b_idx = offs_row // seq_sq
    rem   = offs_row - b_idx * seq_sq
    i_idx = rem // N
    k_idx = rem - i_idx * N

    # optional mask (fp16, shape [R])
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_row, mask=mask_row, other=0.0)[:, None]  # shape (BLOCK_ROWS, 1)

    proj_stride = 5 * hidden_dim               # distance between consecutive rows

    # iterate over hidden dimension blocks
    for d_off in range(0, tl.cdiv(hidden_dim, BLOCK_H)):
        offs_h = d_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < hidden_dim

        # ------------------------------
        # load the five slices for this block
        # ------------------------------
        base = offs_row[:, None] * proj_stride + offs_h[None, :]   # (BLOCK_ROWS, BLOCK_H)

        left      = tl.load(proj_ptr + base,
                            mask=mask_row[:, None] & mask_h,
                            other=0.0, dtype=tl.float16)
        right     = tl.load(proj_ptr + base + hidden_dim,
                            mask=mask_row[:, None] & mask_h,
                            other=0.0, dtype=tl.float16)
        left_g_raw  = tl.load(proj_ptr + base + 2 * hidden_dim,
                              mask=mask_row[:, None] & mask_h,
                              other=0.0, dtype=tl.float16)
        right_g_raw = tl.load(proj_ptr + base + 3 * hidden_dim,
                              mask=mask_row[:, None] & mask_h,
                              other=0.0, dtype=tl.float16)
        out_g_raw   = tl.load(proj_ptr + base + 4 * hidden_dim,
                              mask=mask_row[:, None] & mask_h,
                              other=0.0, dtype=tl.float16)

        # sigmoid in fp16 (fast enough on H100)
        left_gate  = 1.0 / (1.0 + tl.exp(-left_g_raw))
        right_gate = 1.0 / (1.0 + tl.exp(-right_g_raw))
        out_gate   = 1.0 / (1.0 + tl.exp(-out_g_raw))

        left_val  = left  * left_gate
        right_val = right * right_gate

        if HAS_MASK:
            left_val  = left_val  * mask_val
            right_val = right_val * mask_val

        # -------------------------------------------------
        # write left/right to contiguous (B*H, N, N) layout
        # -------------------------------------------------
        # batch index for each hidden slice:  b*H + h
        batch_idx = b_idx[:, None] * hidden_dim + offs_h[None, :]          # (BLOCK_ROWS, BLOCK_H)

        # row‑major offset inside the output tensors
        out_offset = batch_idx * (N * N) + i_idx[:, None] * N + k_idx[:, None]

        tl.store(left_ptr  + out_offset, left_val,
                 mask=mask_row[:, None] & mask_h)
        tl.store(right_ptr + out_offset, right_val,
                 mask=mask_row[:, None] & mask_h)

        # -------------------------------------------------
        # store raw out‑gate (still sigmoid‑ed) in flat layout
        # -------------------------------------------------
        out_gate_offset = offs_row[:, None] * hidden_dim + offs_h[None, :]
        tl.store(out_gate_ptr + out_gate_offset, out_gate,
                 mask=mask_row[:, None] & mask_h)


# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [B, N, N, dim] (float32, on the same CUDA device).
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    device = input_tensor.device

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N                      # rows for the first LayerNorm

    # --------------------------------------------------------------
    # 1️⃣ First (row‑wise) LayerNorm → FP16
    # --------------------------------------------------------------
    # flatten to [R, dim] (float32)
    x_flat = input_tensor.view(total_rows, dim).contiguous()
    x_norm = torch.empty(total_rows, dim,
                         dtype=torch.float16,
                         device=device)

    BLOCK_M = 256
    BLOCK_D = 32
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,                     # input pointer (fp32)
        x_norm,                     # output pointer (fp16)
        weights["norm.weight"],     # γ (fp32)
        weights["norm.bias"],       # β (fp32)
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # --------------------------------------------------------------
    # 2️⃣ Fuse five linear layers (2 projections + 3 gates)
    # --------------------------------------------------------------
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # [5*H, dim]

    # [R, 5*H]  (half‑precision)
    proj = F.linear(x_norm, w_concat)

    # --------------------------------------------------------------
    # 3️⃣ Prepare left/right matrices & output‑gate (fused Triton kernel)
    # --------------------------------------------------------------
    left_mat = torch.empty(B * hidden_dim, N, N,
                           dtype=torch.float16, device=device)
    right_mat = torch.empty_like(left_mat)
    out_gate_flat = torch.empty(total_rows, hidden_dim,
                                dtype=torch.float16, device=device)

    has_mask = not config.get("nomask", True)
    if has_mask:
        mask_flat = mask_tensor.to(torch.float16).reshape(total_rows)
    else:
        # dummy tensor – never read because HAS_MASK==0
        mask_flat = torch.empty(1, dtype=torch.float16, device=device)

    BLOCK_ROWS = 128
    BLOCK_H = 32
    grid = (triton.cdiv(total_rows, BLOCK_ROWS),)
    prepare_inputs_kernel[grid](
        proj,
        mask_flat,
        left_mat,
        right_mat,
        out_gate_flat,
        total_rows,
        N,
        hidden_dim,
        HAS_MASK=1 if has_mask else 0,
        BLOCK_ROWS=BLOCK_ROWS,
        BLOCK_H=BLOCK_H,
    )

    # --------------------------------------------------------------
    # 4️⃣ Batched GEMM  (B·H, N, N) @ (B·H, N, N)ᵀ  →  (B·H, N, N)
    # --------------------------------------------------------------
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape to [B, N, N, H] (non‑contiguous view is fine for the next step)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm over the hidden dimension (using PyTorch impl)
    # --------------------------------------------------------------
    out_norm = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
        eps=eps,
    )

    # --------------------------------------------------------------
    # 6️⃣ Apply output gate and final projection back to `dim`
    # --------------------------------------------------------------
    out_gate = out_gate_flat.view(B, N, N, hidden_dim)   # [B, N, N, H]
    out_norm = out_norm * out_gate                       # element‑wise (fp16)

    final_half = F.linear(
        out_norm.view(total_rows, hidden_dim),
        weights["to_out.weight"].to(torch.float16)      # [dim, H]
    )   # [R, dim] (fp16)

    final = final_half.view(B, N, N, dim).to(torch.float32)

    return final
