import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def layernorm_fp16_kernel(x_ptr, y_ptr, weight_ptr, bias_ptr,
                         total_rows, dim: tl.constexpr, eps: tl.constexpr,
                         BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)   # [BLOCK_M]
    mask_m = offs_m < total_rows

    # Accumulate mean & var in fp32
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)  # [BLOCK_D]
        mask_d = offs_d < dim

        ptrs = x_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)   # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # Write normalized + affine (fp16)
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptrs = x_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]   # fp32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # β

        out_fp32 = norm * w + b
        out_fp16 = out_fp32.to(tl.float16)

        out_ptrs = y_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

"""
TriMul (outgoing) – fast forward pass
=====================================

Algorithm (all ops are done in fp16 except the final cast to fp32):
1️⃣  Layer‑Norm over the input channel dimension (dim) – custom Triton kernel
    that computes mean/variance in fp32 and writes the normalized result in fp16.

2️⃣  Fuse the five linear layers (left/right projections + three gates) into a
    single GEMM by concatenating their weight matrices.  No bias is used,
    so the whole operation is a single `F.linear` on the fp16‑normalized tensor.

3️⃣  If a mask is present, apply it to the two projection tensors
    (`left_proj`, `right_proj`) before the gating step.

4️⃣  Apply a sigmoid to the three gate tensors and perform element‑wise gating.

5️⃣  Triangular multiplicative update:
       out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
    realised as a batched matrix multiplication with shape
    (B·H, N, N) × (B·H, N, N) → (B·H, N, N).

6️⃣  Second Layer‑Norm over the hidden dimension (H) – same Triton kernel.

7️⃣  Multiply by the output gate, project back to the original `dim` with a
    final linear layer, and return the result as FP32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: LayerNorm → fp16 output (mean/var in fp32)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,          # *const float*   [total_rows, dim]   (input fp32)
    y_ptr,          # *mut   half*    [total_rows, dim]   (output fp16)
    weight_ptr,     # *const float*   [dim]              (γ)
    bias_ptr,       # *const float*   [dim]              (β)
    total_rows,     # int (runtime)
    dim: tl.constexpr,          # int (compile‑time)
    eps: tl.constexpr,          # float (compile‑time)
    BLOCK_M: tl.constexpr,      # rows processed per program
    BLOCK_D: tl.constexpr,      # channel tile size
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # --------------------------------------------------------------
    # Pass 1 – compute per‑row mean & variance (fp32 accumulation)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    num_blocks_d = tl.cdiv(dim, BLOCK_D)
    for d in range(0, num_blocks_d):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim

        # Load a tile of the input (fp32)
        ptrs = x_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                               # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                # sum over dim
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                         # 1 / sqrt(var+eps)

    # --------------------------------------------------------------
    # Pass 2 – write (x‑mean)/sqrt(var+eps) * γ + β  (fp16)
    # --------------------------------------------------------------
    for d in range(0, num_blocks_d):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        # Reload the raw values (fp32)
        ptrs = x_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]   # fp32

        # Load γ, β (fp32)
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)

        out_fp32 = norm * w + b
        out_fp16 = out_fp32.to(tl.float16)

        out_ptrs = y_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, C], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : mapping from parameter name to torch.Tensor (float32)
        - config_dict  : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, C], dtype float32, on the same device.
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                # C
    hidden_dim = config["hidden_dim"]  # H
    eps = 1e-5
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Shapes
    # ------------------------------------------------------------------
    B, N, _, _ = input_tensor.shape               # [B, N, N, C]
    total_rows = B * N * N                         # flatten for LN1

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (fp16 output)
    # ------------------------------------------------------------------
    x_flat = input_tensor.view(total_rows, dim).contiguous()  # fp32
    x_norm_fp16 = torch.empty((total_rows, dim),
                              dtype=torch.float16,
                              device=device)

    # Tuned launch parameters (good for all supported dims)
    BLOCK_M = 256
    BLOCK_D = 128                     # cover up to dim=384 in ≤3 tiles
    grid = (triton.cdiv(total_rows, BLOCK_M),)

    layernorm_fp16_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear projections + gates (still fp16)
    # ------------------------------------------------------------------
    # Concatenate the five weight matrices (shape: (5*H, C))
    concat_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).half().contiguous()   # (5*H, C)

    # Linear transformation – fp16 GEMM (cuBLAS / Tensor‑Core)
    proj_all = F.linear(x_norm_fp16, concat_w)   # (total_rows, 5*H)

    # Split back to the five distinct tensors
    left_proj, right_proj, left_gate, right_gate, out_gate = torch.split(
        proj_all, hidden_dim, dim=1
    )  # each: (total_rows, H)

    # ------------------------------------------------------------------
    # 2a️⃣ Optional mask (applied before gating)
    # ------------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        # mask shape [B,N,N] → [total_rows,1] broadcastable to projections
        mask_flat = mask_tensor.view(total_rows, 1).to(dtype=torch.float16)
        left_proj  = left_proj * mask_flat
        right_proj = right_proj * mask_flat

    # ------------------------------------------------------------------
    # 2b️⃣ Sigmoid gates
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate   = torch.sigmoid(out_gate)

    # Element‑wise gating of the projections
    left_fp16  = left_proj * left_gate   # (total_rows, H)
    right_fp16 = right_proj * right_gate  # (total_rows, H)

    # Reshape for the triangular update
    left  = left_fp16.view(B, N, N, hidden_dim)   # (B, N, N, H)
    right = right_fp16.view(B, N, N, hidden_dim) # (B, N, N, H)
    out_gate = out_gate.view(B, N, N, hidden_dim)  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 3️⃣ Triangular multiplicative update (batched GEMM)
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    # Bring the summed‑over dimension k to the innermost axis for both
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N)

    B_H = B * hidden_dim
    left_mat  = left_perm.view(B_H, N, N)   # (B*H, N, N)
    right_mat = right_perm.view(B_H, N, N)  # (B*H, N, N)

    # Batched matrix multiplication – fp16, Tensor‑Core friendly
    out_mat = torch.matmul(left_mat, right_mat)  # (B*H, N, N)

    # Restore the original layout: (B, N, N, H)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 4️⃣ Second LayerNorm over the hidden dimension (fp16 output)
    # ------------------------------------------------------------------
    total_rows_h = B * N * N
    out_flat = out.view(total_rows_h, hidden_dim).contiguous()
    out_norm_fp16 = torch.empty_like(out_flat)   # fp16 buffer

    grid2 = (triton.cdiv(total_rows_h, BLOCK_M),)
    layernorm_fp16_kernel[grid2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows_h,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    out_norm = out_norm_fp16.view(B, N, N, hidden_dim)  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 5️⃣ Output gate + final projection back to dim
    # ------------------------------------------------------------------
    gated = out_norm * out_gate                     # (B,N,N,H) fp16

    # Final linear projection H → C (still fp16)
    to_out_w_fp16 = weights["to_out.weight"].half()
    gated_flat = gated.view(total_rows_h, hidden_dim)  # (total_rows_h, H)
    final_fp16 = F.linear(gated_flat, to_out_w_fp16)   # (total_rows_h, C)

    # Cast back to FP32 as required by the API contract
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
