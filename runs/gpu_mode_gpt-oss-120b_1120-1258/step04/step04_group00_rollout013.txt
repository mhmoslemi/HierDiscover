        out_norm_h,

        hidden_dim,

    )   # output fp32

"""
TriMul “outgoing” (AlphaFold‑3) forward pass – Triton‑accelerated version.

Algorithm
---------
1️⃣ Compute per‑row mean and 1/σ (rstd) of the 4‑D input tensor
   (shape [B,N,N,C]) with a tiny Triton kernel – only two reductions
   (sum & sum‑of‑squares).  The normalized values are **not** stored,
   we only keep the scalar statistics.

2️⃣ Fuse the first LayerNorm and all four linear projections + their
   gates into **two** dense GEMMs:
   * A single GEMM (rows × C) × (4·H × C)^T gives the four
     pre‑gate tensors (left‑proj, left‑gate, right‑proj, right‑gate)
     for every position.
   * A second GEMM yields the output‑gate pre‑activation.
   Using the identity
       (x‑μ)·rstd · w = rstd·(x·w ‑ μ·Σw)
   we can turn each dense GEMM on the *raw* input into a LayerNorm‑
   aware GEMM by broadcasting the per‑row statistics and the per‑
   output‑channel weight sums (pre‑computed on the host).

3️⃣ Apply sigmoid to the gate tensors and multiply with the
   corresponding projections.  The optional pair‑wise mask is applied
   after gating – it is a scalar per position, so order does not
   matter.

4️⃣ The core triangular multiplicative update:
   out_{i,j,h} = Σ_k left_{i,k,h}·right_{j,k,h}
   is realised as a batched GEMM (torch.bmm) on tensors of shape
   [B·H, N, N] (half‑precision, Tensor‑Core friendly).

5️⃣ A second LayerNorm (with learned γ/β) is performed on the
   intermediate result with a Triton kernel that writes FP16
   directly to minimise traffic.

6️⃣ Apply the output‑gate, a final linear projection and cast the
   result back to FP32.

All heavy workloads (reductions, LayerNorms, and the batched GEMM)
run on the GPU; the fused linear layers avoid three separate passes
over the input, reducing memory bandwidth and improving runtime.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: compute per‑row mean and rstd (1/σ) of a 2‑D tensor
# ----------------------------------------------------------------------
@triton.jit
def row_mean_rstd_kernel(
    x_ptr,          # const float*  [total_rows, dim]
    mean_ptr,       #       float*  [total_rows]
    rstd_ptr,       #       float*  [total_rows]
    total_rows,     # int
    dim: tl.constexpr,          # static: number of columns
    eps: tl.constexpr,          # static epsilon
    BLOCK_M: tl.constexpr,      # rows per program
    BLOCK_D: tl.constexpr,      # columns per inner loop
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)        # [BLOCK_M]
    mask_m = offs_m < total_rows

    # Accumulate sum and sum‑of‑squares for each row
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)  # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                # 1 / sqrt(var+eps)

    # Store results
    tl.store(mean_ptr + offs_m, mean, mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (γ/β) that writes FP16 output
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,          # const float*   [total_rows, dim]
    y_ptr,          #       half*    [total_rows, dim] (output)
    w_ptr,          # const float*  [dim]   (γ)
    b_ptr,          # const float*  [dim]   (β)
    total_rows,     # int (runtime)
    dim: tl.constexpr,          # static: number of columns
    eps: tl.constexpr,          # static epsilon
    BLOCK_M: tl.constexpr,      # rows processed per program
    BLOCK_D: tl.constexpr,      # columns per inner loop
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # --------------------------------------------------------------
    # Pass 1 – compute mean and variance
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)        # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                 # ∑_d x
        sum_sq  += tl.sum(vals * vals, axis=1)          # ∑_d x²

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                         # 1 / σ

    # --------------------------------------------------------------
    # Pass 2 – write normalized values with γ/β
    # --------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]   # (x‑μ)/σ

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # β

        out = norm * w[None, :] + b[None, :]            # fp32
        out_fp16 = out.to(tl.float16)

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
            input_tensor : torch.Tensor [B, N, N, C] (float32)
            mask_tensor  : torch.Tensor [B, N, N]   (float32) – may be unused
            weights_dict : {
                "norm.weight":        Tensor[C],
                "norm.bias":          Tensor[C],
                "left_proj.weight":   Tensor[H, C],
                "right_proj.weight":  Tensor[H, C],
                "left_gate.weight":   Tensor[H, C],
                "right_gate.weight":  Tensor[H, C],
                "out_gate.weight":    Tensor[H, C],
                "to_out_norm.weight": Tensor[H],
                "to_out_norm.bias":   Tensor[H],
                "to_out.weight":      Tensor[C, H],
            }
            config_dict  : {
                "dim": int,
                "hidden_dim": int,
                "nomask": bool (optional, default True),
            }

    Returns:
        torch.Tensor [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    H = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    total_rows = B * N * N                     # number of (b,i,k) positions
    rows = total_rows

    # ------------------------------------------------------------------
    # 1️⃣ Compute per‑row mean and rstd (fast Triton reduction)
    # ------------------------------------------------------------------
    x_flat = input_tensor.view(rows, dim).contiguous()   # [rows, dim] FP32

    mean = torch.empty(rows, dtype=torch.float32, device=device)
    rstd = torch.empty(rows, dtype=torch.float32, device=device)

    BLOCK_M = 256    # rows per program – empirically good on H100
    BLOCK_D = 64     # columns per inner loop
    grid = (triton.cdiv(rows, BLOCK_M),)

    row_mean_rstd_kernel[grid](
        x_flat,
        mean,
        rstd,
        rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    # ------------------------------------------------------------------
    # 2️⃣ Fuse the first LayerNorm + all projections + gates
    # ------------------------------------------------------------------
    # Concatenate the four weight matrices (proj + gate for left/right)
    left_right_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16, non_blocking=True)          # shape [4*H, dim]

    out_gate_weight = weights["out_gate.weight"].to(
        torch.float16, non_blocking=True
    )                                               # shape [H, dim]

    # Pre‑compute per‑output‑channel weight sums Σw (needed for the
    # “norm‑aware” GEMM trick)
    left_right_wsum = left_right_weight.sum(dim=1).to(torch.float32)   # [4*H]
    out_gate_wsum = out_gate_weight.sum(dim=1).to(torch.float32)      # [H]

    # ----- 2a: left & right projections + their gates (single GEMM) -----
    # (rows, dim) @ (4*H, dim).T → (rows, 4*H)   – result is FP32
    lr_concat = torch.nn.functional.linear(x_flat, left_right_weight)   # [rows, 4*H]

    # Normalize the linear results using the previously computed statistics
    mean_b = mean[:, None]          # [rows,1]
    rstd_b = rstd[:, None]          # [rows,1]
    lr_norm = (lr_concat - mean_b * left_right_wsum[None, :]) * rstd_b   # FP32

    # Split into the four pieces
    l_proj = lr_norm[:, 0:H]                 # [rows, H]
    l_gate_pre = lr_norm[:, H:2 * H]         # [rows, H]
    r_proj = lr_norm[:, 2 * H:3 * H]         # [rows, H]
    r_gate_pre = lr_norm[:, 3 * H:]          # [rows, H]

    # Apply sigmoid gates
    l_gate = torch.sigmoid(l_gate_pre)
    r_gate = torch.sigmoid(r_gate_pre)

    # Final gated tensors (store as FP16 to save bandwidth)
    left_h = (l_proj * l_gate).to(torch.float16)    # [rows, H]
    right_h = (r_proj * r_gate).to(torch.float16)   # [rows, H]

    # Clean up intermediate tensors we no longer need
    del lr_concat, lr_norm, l_proj, l_gate_pre, r_proj, r_gate_pre, l_gate, r_gate

    # ----- 2b: output‑gate (separate GEMM) -----
    out_gate_lin = torch.nn.functional.linear(x_flat, out_gate_weight)    # [rows, H]
    out_gate_norm = (out_gate_lin - mean_b * out_gate_wsum[None, :]) * rstd_b
    out_gate = torch.sigmoid(out_gate_norm).to(torch.float16)            # [rows, H]

    del out_gate_lin, out_gate_norm, mean, rstd, mean_b, rstd_b

    # ------------------------------------------------------------------
    # 3️⃣ Apply optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        mask_flat = mask_tensor.view(rows, 1).to(torch.float16, non_blocking=True)
        left_h = left_h * mask_flat
        right_h = right_h * mask_flat

    # ------------------------------------------------------------------
    # 4️⃣ Reshape for the batched GEMM (Tri‑multiplicative update)
    # ------------------------------------------------------------------
    left_h = left_h.view(B, N, N, H)               # [B, N, N, H]  FP16
    right_h = right_h.view(B, N, N, H)             # [B, N, N, H]  FP16

    # Move hidden dimension to the batch axis → [B·H, N, N]
    left_perm = left_h.permute(0, 3, 1, 2).contiguous()
    right_perm = right_h.permute(0, 3, 1, 2).contiguous()
    BH = B * H

    left_mat = left_perm.view(BH, N, N)            # FP16
    right_mat = right_perm.view(BH, N, N)          # FP16

    # Core Tri‑multiplicative update (batched GEMM)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [BH, N, N] FP16

    # Back‑to shape [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm on the hidden dimension (γ/β) → FP16
    # ------------------------------------------------------------------
    rows_h = B * N * N
    out_flat = out.view(rows_h, H).contiguous()
    out_norm_fp16 = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(rows_h, BLOCK_M),)
    layernorm_fp16_kernel[grid2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        rows_h,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    out_norm = out_norm_fp16.view(B, N, N, H)    # FP16

    # ------------------------------------------------------------------
    # 6️⃣ Apply output‑gate and final projection
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate                # gated output (FP16)

    final = torch.nn.functional.linear(
        out_norm.view(rows_h, H),
        weights["to_out.weight"].to(torch.float16, non_blocking=True),
    )                                             # → FP32

    final = final.view(B, N, N, dim).to(torch.float32)

    return final
