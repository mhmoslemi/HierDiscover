"""
custom_kernel – “outgoing” TriMul (AlphaFold‑3) forward pass.

Algorithm (all heavy‑computations are in fp16):
1️⃣  Layer‑Norm over the input channel dimension (dim) → fp16.
2️⃣  Fuse the five linear projections/gates (left/right proj + three gates)
    into a single GEMM by concatenating their weight matrices.
3️⃣  (Optional) apply the pair‑wise mask after the two projections.
4️⃣  Sigmoid the three gate tensors and element‑wise gate the projected tensors.
5️⃣  Triangular multiplicative update:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    realised as a batched matrix‑multiply per head (`torch.matmul`).
6️⃣  Second Layer‑Norm over the hidden dimension (hidden_dim) → fp16.
7️⃣  Multiply by the output‑gate, final linear projection (hidden_dim → dim),
    and cast the result back to fp32.

The only custom Triton code is the fused Layer‑Norm (mean/var computed in fp32,
output stored in fp16).  All dense linear algebra uses cuBLAS/Tensor‑Core via
PyTorch which is already optimal on H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: LayerNorm that reads fp32, computes mean/var in fp32,
# writes normalized+affine result in fp16.
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,          # *const float*   [total_rows, dim]   (input)
    y_ptr,          # *mut   half*    [total_rows, dim]   (output)
    weight_ptr,     # *const float*   [dim]              (gamma)
    bias_ptr,       # *const float*   [dim]              (beta)
    total_rows,     # int (runtime)
    dim: tl.constexpr,          # int (compile‑time)
    eps: tl.constexpr,          # float (compile‑time)
    BLOCK_M: tl.constexpr,      # rows per program
    BLOCK_D: tl.constexpr,      # columns per tile
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------- Pass 1: mean & variance --------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim

        ptrs = x_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                               # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------- Pass 2: write normalized output -------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptrs = x_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                               # fp32 raw

        norm = (vals - mean[:, None]) * rstd[:, None]   # fp32

        # γ and β (float32)
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)

        out_fp32 = norm * w + b                         # fp32
        out_fp16 = out_fp32.to(tl.float16)              # fp16

        out_ptrs = y_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        tl.store(out_ptrs, out_fp16, mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        * input_tensor – torch.Tensor, shape [B, N, N, C], dtype float32
        * mask_tensor  – torch.Tensor, shape [B, N, N] (bool/float) or None
        * weights_dict – dict with the following keys (all float32):
              norm.weight, norm.bias,
              left_proj.weight, right_proj.weight,
              left_gate.weight, right_gate.weight, out_gate.weight,
              to_out_norm.weight, to_out_norm.bias,
              to_out.weight
        * config_dict  – {"dim": C, "hidden_dim": H, "nomask": bool (optional)}

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, C], dtype float32, on the same device.
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device

    dim = config["dim"]                 # C
    hidden_dim = config["hidden_dim"]   # H
    eps = 1e-5
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Shape helpers
    # ------------------------------------------------------------------
    B, N, _, _ = input_tensor.shape          # [B, N, N, C]
    total_rows = B * N * N                    # rows for flattened view

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (fp16 output)
    # ------------------------------------------------------------------
    x_flat = input_tensor.view(total_rows, dim).contiguous()
    x_norm_fp16 = torch.empty(
        (total_rows, dim), dtype=torch.float16, device=device
    )

    # Launch parameters – tuned for large H100 threads
    BLOCK_M = 256          # rows per program
    BLOCK_D = 64           # channel tile
    grid = (triton.cdiv(total_rows, BLOCK_M),)

    layernorm_fp16_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection + three gates (single GEMM)
    # ------------------------------------------------------------------
    # Concatenate the five weight matrices → (5*H, C)
    concat_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).half().contiguous()  # FP16 for Tensor‑Core

    # Linear transformation in FP16 (no bias)
    proj_all = F.linear(x_norm_fp16, concat_w)   # shape (total_rows, 5*H)

    # Split back to individual tensors
    left_proj, right_proj, left_gate, right_gate, out_gate = torch.split(
        proj_all, hidden_dim, dim=1
    )  # each: (total_rows, H)

    # ------------------------------------------------------------------
    # 2a️⃣ Optional mask (applied after the two projections)
    # ------------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        mask_flat = mask_tensor.view(total_rows, 1).to(dtype=torch.float16)
        left_proj = left_proj * mask_flat
        right_proj = right_proj * mask_flat

    # ------------------------------------------------------------------
    # 2b️⃣ Sigmoid gates
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    # Gating
    left_fp16 = left_proj * left_gate   # (total_rows, H)
    right_fp16 = right_proj * right_gate

    # ------------------------------------------------------------------
    # Reshape to 4‑D tensors for the triangular update
    # ------------------------------------------------------------------
    left = left_fp16.view(B, N, N, hidden_dim)   # (B, N, N, H)
    right = right_fp16.view(B, N, N, hidden_dim) # (B, N, N, H)
    out_gate = out_gate.view(B, N, N, hidden_dim)  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 3️⃣ Triangular multiplicative update (batched GEMM per head)
    # ------------------------------------------------------------------
    # Bring the hidden dimension in front so we can treat it as batch.
    # left  : (B, H, N, N)   – value at (i,k)
    # right : (B, H, N, N)   – value at (j,k)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, N, N) == rightᵀ

    # Merge batch and head dimensions for a single matmul call.
    BH = B * hidden_dim
    left_mat = left_perm.view(BH, N, N)        # (B*H, N, N)
    right_mat = right_perm.view(BH, N, N)      # (B*H, N, N)

    # Batched matrix multiplication – Tensor‑Core friendly (FP16)
    out_mat = torch.matmul(left_mat, right_mat)   # (B*H, N, N), fp16

    # Restore original layout
    out = (
        out_mat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)   # (B, N, N, H)
        .contiguous()
    )

    # ------------------------------------------------------------------
    # 4️⃣ Second LayerNorm over the hidden dimension (fp16 output)
    # ------------------------------------------------------------------
    total_rows_h = B * N * N
    out_flat = out.view(total_rows_h, hidden_dim).contiguous()
    out_norm_fp16 = torch.empty_like(out_flat)

    grid2 = (triton.cdiv(total_rows_h, BLOCK_M),)
    layernorm_fp16_kernel[grid2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows_h,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    out_norm = out_norm_fp16.view(B, N, N, hidden_dim)  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 5️⃣ Output gate + final projection back to `dim`
    # ------------------------------------------------------------------
    gated = out_norm * out_gate               # (B, N, N, H)  fp16

    # Final linear projection H → C
    to_out_w_fp16 = weights["to_out.weight"].half()
    gated_flat = gated.view(total_rows_h, hidden_dim)
    final_fp16 = F.linear(gated_flat, to_out_w_fp16)  # (total_rows_h, C)

    # Cast back to FP32 as required by the API contract
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
