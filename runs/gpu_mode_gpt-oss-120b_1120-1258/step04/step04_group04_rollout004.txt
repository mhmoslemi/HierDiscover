"""
TriMul (outgoing) – fast forward pass in Triton + PyTorch.

Algorithm
---------
1️⃣ First LayerNorm (fp16) – custom Triton kernel.
2️⃣ Fuse the five linear projections (left/right & three gates) by a
   single torch.nn.functional.linear using a concatenated weight matrix.
3️⃣ Fuse mask application, sigmoid‑gates and element‑wise gating in one
   Triton kernel (left, right and out_gate).
4️⃣ Triangular multiplicative update:
      out[b,i,j,:] = Σ_k left[b,i,k,:] * right[b,j,k,:]
   performed as a batched half‑precision GEMM (torch.bmm) on tensors
   of shape (B·H, N, N) where H = hidden_dim.
5️⃣ Second LayerNorm (fp16) – custom Triton kernel.
6️⃣ Multiply by the output gate, final linear projection back to `dim`,
   and cast to float32.

All heavy element‑wise work (norm, gating, masking) is done in
Triton kernels; dense linear algebra uses cuBLAS Tensor‑Core kernels.
The implementation works for all test configurations (various N, B,
dim, hidden_dim and optional masks).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton LayerNorm (fp32 → fp16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,          # *const   fp32*   [total_rows, dim]
    y_ptr,          # *mut     half*   [total_rows, dim]
    weight_ptr,     # *const   fp32*   [dim]   (γ)
    bias_ptr,       # *const   fp32*   [dim]   (β)
    total_rows,     # int (runtime)
    dim: tl.constexpr,          # int (compile‑time)
    eps: tl.constexpr,          # float (compile‑time)
    BLOCK_M: tl.constexpr,      # rows per program
    BLOCK_D: tl.constexpr,      # columns per tile
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # --------------------------------------------------------------
    # Pass 1 – compute mean and variance (fp32 accumulator)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------------------------------------------------------------
    # Pass 2 – write normalized values (fp16)
    # --------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]   # (x‑μ)/σ

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)

        out_fp32 = norm * w + b
        out_fp16 = out_fp32.to(tl.float16)

        out_ptrs = y_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Triton fused gating + optional mask kernel
# ----------------------------------------------------------------------
@triton.jit
def fused_gating_kernel(
    left_proj_ptr,   # *const   half* [R, H]
    left_gate_ptr,   # *const   half* [R, H]
    right_proj_ptr,  # *const   half* [R, H]
    right_gate_ptr,  # *const   half* [R, H]
    out_gate_ptr,    # *const   half* [R, H]
    mask_ptr,        # *const   half* [R]   (1.0 if no mask)
    left_out_ptr,    # *mut     half* [R, H]
    right_out_ptr,   # *mut     half* [R, H]
    out_gate_out_ptr,# *mut     half* [R, H]
    total_rows,      # int (runtime)
    hidden_dim: tl.constexpr,   # int (compile‑time, = H)
    BLOCK_M: tl.constexpr,      # rows per program
    BLOCK_H: tl.constexpr,      # hidden‑dim tile size
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # --------------------------------------------------------------
    # Load per‑row mask (broadcast later). For a “nomask” case a tensor
    # of ones is passed, so we don’t need a branch in the kernel.
    # --------------------------------------------------------------
    mask_val = tl.load(mask_ptr + offs_m, mask=mask_m, other=1.0).to(tl.float16)  # [BLOCK_M]

    for h in range(0, tl.cdiv(hidden_dim, BLOCK_H)):
        offs_h = h * BLOCK_H + tl.arange(0, BLOCK_H)    # [BLOCK_H]
        mask_h = offs_h < hidden_dim

        # Load slices
        base = offs_m[:, None] * hidden_dim + offs_h[None, :]
        left_proj  = tl.load(left_proj_ptr  + base,
                             mask=mask_m[:, None] & mask_h[None, :],
                             other=0.0)
        left_gate  = tl.load(left_gate_ptr  + base,
                             mask=mask_m[:, None] & mask_h[None, :],
                             other=0.0)
        right_proj = tl.load(right_proj_ptr + base,
                             mask=mask_m[:, None] & mask_h[None, :],
                             other=0.0)
        right_gate = tl.load(right_gate_ptr + base,
                             mask=mask_m[:, None] & mask_h[None, :],
                             other=0.0)
        out_gate   = tl.load(out_gate_ptr   + base,
                             mask=mask_m[:, None] & mask_h[None, :],
                             other=0.0)

        # Sigmoids
        left_gate_sig  = tl.sigmoid(left_gate)
        right_gate_sig = tl.sigmoid(right_gate)
        out_gate_sig   = tl.sigmoid(out_gate)

        # Broadcast mask to hidden dimension
        mask_tile = mask_val[:, None]            # [BLOCK_M, 1]

        # Element‑wise gated values
        left  = left_proj  * left_gate_sig  * mask_tile
        right = right_proj * right_gate_sig * mask_tile
        # out_gate is only sigmoided, not masked
        out_gate_res = out_gate_sig

        # Store results
        tl.store(left_out_ptr  + base,
                 left,
                 mask=mask_m[:, None] & mask_h[None, :])
        tl.store(right_out_ptr + base,
                 right,
                 mask=mask_m[:, None] & mask_h[None, :])
        tl.store(out_gate_out_ptr + base,
                 out_gate_res,
                 mask=mask_m[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Custom kernel entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: [B, N, N, C] (float32)
            - mask_tensor : [B, N, N] (bool/float, optional)
            - weights_dict: model parameters (float32)
            - config_dict : {"dim": C, "hidden_dim": H, "nomask": bool, ...}

    Returns:
        Tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]               # C
    hidden_dim = config["hidden_dim"] # H (always 128 in the tests)
    eps = 1e-5
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape          # [B, N, N, C]
    total_rows = B * N * N                    # R = B·N·N

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (fp16 output)
    # ------------------------------------------------------------------
    x_flat = input_tensor.view(total_rows, dim).contiguous()  # [R, C] fp32
    x_norm_fp16 = torch.empty((total_rows, dim),
                              dtype=torch.float16,
                              device=device)

    BLOCK_M = 128          # rows per Triton program
    BLOCK_D = 128          # columns per tile (covers dim up to 384)
    grid = (triton.cdiv(total_rows, BLOCK_M),)

    layernorm_fp16_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection (five linear layers) – torch linear
    # ------------------------------------------------------------------
    # Concatenate all five weight matrices (each [H, C])
    concat_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).half().contiguous()                     # (5H, C)

    # Linear projection: [R, C] @ [C, 5H]^T → [R, 5H] (fp16)
    proj_all = F.linear(x_norm_fp16, concat_weight)   # fp16

    # Split into the five components
    left_proj, right_proj, left_gate, right_gate, out_gate = torch.split(
        proj_all, hidden_dim, dim=1
    )  # each [R, H]

    # ------------------------------------------------------------------
    # 3️⃣ Fuse mask, sigmoids and gating in a Triton kernel
    # ------------------------------------------------------------------
    if nomask:
        # A tensor full of ones – mask_val = 1 for every position
        mask_flat = torch.ones(total_rows,
                               dtype=torch.float16,
                               device=device)
    else:
        # mask_tensor: [B, N, N] → flatten & cast to half
        mask_flat = mask_tensor.view(total_rows).to(torch.float16)

    # Allocate outputs for the fused kernel
    left_fused  = torch.empty_like(left_proj)   # [R, H]
    right_fused = torch.empty_like(right_proj) # [R, H]
    out_gate_fused = torch.empty_like(out_gate) # [R, H]

    # Triton launch parameters
    BLOCK_H = 64          # tile size for the hidden dimension
    grid_fuse = (triton.cdiv(total_rows, BLOCK_M),)

    fused_gating_kernel[grid_fuse](
        left_proj,
        left_gate,
        right_proj,
        right_gate,
        out_gate,
        mask_flat,
        left_fused,
        right_fused,
        out_gate_fused,
        total_rows,
        hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_H=BLOCK_H,
    )

    # ------------------------------------------------------------------
    # 4️⃣ Reshape for the triangular multiplicative update
    # ------------------------------------------------------------------
    left = left_fused.view(B, N, N, hidden_dim)          # [B, N, N, H]
    right = right_fused.view(B, N, N, hidden_dim)        # [B, N, N, H]
    out_gate = out_gate_fused.view(B, N, N, hidden_dim)  # [B, N, N, H]

    # Prepare tensors for batched GEMM
    # left : (i,k)   → (B, H, N, N)
    # right: (k,j)   → (B, H, N, N) with axes (k,j) swapped
    left_perm = left.permute(0, 3, 1, 2).contiguous()    # [B, H, N, N]
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # [B, H, N, N]

    BH = B * hidden_dim
    left_mat = left_perm.view(BH, N, N)   # [B·H, N, N]  (i,k)
    right_mat = right_perm.view(BH, N, N) # [B·H, N, N]  (k,j)

    # Batched GEMM (fp16) → [B·H, N, N] (i,j)
    out_mat = torch.bmm(left_mat, right_mat)   # fp16

    # Back to 4‑D tensor
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H] fp16

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    total_rows_h = B * N * N                 # same as total_rows
    out_flat = out.view(total_rows_h, hidden_dim).contiguous()
    out_norm_fp16 = torch.empty_like(out_flat)  # fp16 buffer

    grid2 = (triton.cdiv(total_rows_h, BLOCK_M),)

    layernorm_fp16_kernel[grid2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows_h,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_H,            # hidden_dim ≤ 128 → BLOCK_H=64 works well
    )

    out_norm = out_norm_fp16.view(B, N, N, hidden_dim)   # [B, N, N, H] fp16

    # ------------------------------------------------------------------
    # 6️⃣ Apply output gate and final projection back to `dim`
    # ------------------------------------------------------------------
    gated = out_norm * out_gate                           # element‑wise fp16

    to_out_weight_fp16 = weights["to_out.weight"].half()
    gated_flat = gated.view(total_rows_h, hidden_dim)      # [R_h, H]
    final_fp16 = F.linear(gated_flat, to_out_weight_fp16) # [R_h, C] fp16

    # Cast back to float32 (the expected output dtype)
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
