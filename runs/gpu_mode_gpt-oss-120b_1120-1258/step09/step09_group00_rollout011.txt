@triton.jit
def final_norm_gate_linear_kernel(
    out_ptr, gate_ptr, y_ptr,
    ln_weight_ptr, ln_bias_ptr,
    final_wt_ptr, # shape [H, dim_out] float16 transposed weight
    rows: tl.int32,
    H: tl.int32,
    dim_out: tl.int32,
    eps: tl.float32,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_O: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < rows

    # Compute mean & var for each row
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_R], dtype=tl.float32)
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H
        out_vals = tl.load(out_ptr + offs_r[:, None] * H + offs_h[None, :],
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0).to(tl.float32)
        sum_val += tl.sum(out_vals, axis=1)
        sum_sq += tl.sum(out_vals * out_vals, axis=1)
    mean = sum_val / H
    var = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # Loop over output dim blocks
    for o_off in range(0, tl.cdiv(dim_out, BLOCK_O)):
        offs_o = o_off * BLOCK_O + tl.arange(0, BLOCK_O)
        mask_o = offs_o < dim_out
        # accumulator for result
        acc = tl.zeros([BLOCK_R, BLOCK_O], dtype=tl.float32)
        # loop over hidden dim blocks
        for h_off in range(0, tl.cdiv(H, BLOCK_H)):
            offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
            mask_h = offs_h < H
            # load out and gate
            out_vals = tl.load(out_ptr + offs_r[:, None] * H + offs_h[None, :],
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0).to(tl.float32)
            gate_vals = tl.load(gate_ptr + offs_r[:, None] * H + offs_h[None, :],
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0).to(tl.float32)
            # load ln weight and bias
            ln_w = tl.load(ln_weight_ptr + offs_h, mask=mask_h, other=0.0)
            ln_b = tl.load(ln_bias_ptr + offs_h, mask=mask_h, other=0.0)
            # normalize
            norm = (out_vals - mean[:, None]) * rstd[:, None]   # fp32
            # affine
            norm = norm * ln_w[None, :] + ln_b[None, :]          # fp32
            # gate multiplication
            gated = norm * gate_vals                            # fp32
            # load final weight slice: shape [BLOCK_H, BLOCK_O]
            wt = tl.load(final_wt_ptr + offs_h[:, None] * dim_out + offs_o[None, :],
                         mask=mask_h[:, None] & mask_o[None, :],
                         other=0.0).to(tl.float32)
            # compute dot: (BLOCK_R, BLOCK_H) x (BLOCK_H, BLOCK_O) -> (BLOCK_R, BLOCK_O)
            acc += tl.dot(gated, wt)
        # write result
        y_ptr_off = y_ptr + offs_r[:, None] * dim_out + offs_o[None, :]
        y_fp16 = acc.to(tl.float16)
        tl.store(y_ptr_off, y_fp16,
                 mask=mask_r[:, None] & mask_o[None, :])
```

"""
High‑performance Triton implementation of the *outgoing* TriMul update.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the 4‑D input (B·N·N × dim) → fp16.
2️⃣ Fused Linear (5·H) on the normalized tensor (torch cublas, fp16).
3️⃣ Triton kernel ‑ split the fused projection, apply optional pair‑wise mask,
   compute sigmoid gates and produce three tensors:
        left   = (x·W_left)   * mask * σ(x·W_left_gate)
        right  = (x·W_right)  * mask * σ(x·W_right_gate)
        out_gate = σ(x·W_out_gate)
4️⃣ Batched GEMM (B·H) of shape (N,N) @ (N,N)ᵀ  →  (B·H,N,N)  (torch bmm, Tensor‑cores).
5️⃣ Triton kernel – row‑wise LayerNorm on the GEMM result, multiply by the
   previously computed *out_gate* and immediately apply the final linear
   projection (H→dim).  This fuses two LayerNorms, the gate multiplication and
   the final projection into a single pass over memory.

All kernels operate in fp16 except the reductions needed for mean/variance,
which are accumulated in fp32 for numerical stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (float32 → float16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   *[rows, dim]   (float32)
    y_ptr,          # mutable *[rows, dim]   (float16)
    weight_ptr,     # const   *[dim]        (float32)   γ
    bias_ptr,       # const   *[dim]        (float32)   β
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ---------- mean / variance (fp32) ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)          # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]    # [M,D]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- write normalized values (γ·x̂+β) ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]           # fp32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ   fp32
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β   fp32

        out = norm * w[None, :] + b[None, :]                     # fp32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 3️⃣ Split proj, optional mask, sigmoid gates (left/right/out_gate)
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[rows, 5*H] (float16)
    left_ptr,          # mutable *[rows, H]   (float16)
    right_ptr,         # mutable *[rows, H]   (float16)
    out_gate_ptr,      # mutable *[rows, H]   (float16)
    mask_ptr,          # const   *[rows]    (float16)   (ignored if HAS_MASK=0)
    rows: tl.int32,
    H: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)               # [BLOCK_R]
    mask_r = offs_r < rows

    # optional pair‑wise mask (broadcast later)
    if HAS_MASK:
        m = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        m = m[:, None]                                          # [BLOCK_R,1]
    else:
        m = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # loop over hidden dimension in chunks
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)      # [BLOCK_H]
        mask_h = offs_h < H

        # layout: [left, right, left_gate, right_gate, out_gate]
        base = offs_r[:, None] * (5 * H) + offs_h[None, :]    # [R,H] offset for seg0

        left_raw_ptr   = proj_ptr + base
        right_raw_ptr  = proj_ptr + base + H
        left_g_ptr     = proj_ptr + base + 2 * H
        right_g_ptr    = proj_ptr + base + 3 * H
        out_g_ptr      = proj_ptr + base + 4 * H

        left_raw   = tl.load(left_raw_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        right_raw  = tl.load(right_raw_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        left_g_raw = tl.load(left_g_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        right_g_raw= tl.load(right_g_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        out_g_raw  = tl.load(out_g_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)

        # sigmoid (computed in fp32 for accuracy)
        lg_f32 = left_g_raw.to(tl.float32)
        left_gate = (1.0 / (1.0 + tl.exp(-lg_f32))).to(tl.float16)

        rg_f32 = right_g_raw.to(tl.float32)
        right_gate = (1.0 / (1.0 + tl.exp(-rg_f32))).to(tl.float16)

        og_f32 = out_g_raw.to(tl.float32)
        out_gate = (1.0 / (1.0 + tl.exp(-og_f32))).to(tl.float16)

        # apply mask, gates and store
        left  = left_raw  * m * left_gate
        right = right_raw * m * right_gate

        left_dst_ptr    = left_ptr    + offs_r[:, None] * H + offs_h[None, :]
        right_dst_ptr   = right_ptr   + offs_r[:, None] * H + offs_h[None, :]
        out_gate_dst_ptr= out_gate_ptr+ offs_r[:, None] * H + offs_h[None, :]

        tl.store(left_dst_ptr,    left,    mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_dst_ptr,   right,   mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_dst_ptr,out_gate,mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 5️⃣ Final LayerNorm + out_gate * linear projection (fused)
# ----------------------------------------------------------------------
@triton.jit
def final_norm_gate_linear_kernel(
    out_ptr,           # const   *[rows, H]   (float16)
    gate_ptr,          # const   *[rows, H]   (float16)
    y_ptr,             # mutable *[rows, dim_out] (float16)
    ln_weight_ptr,     # const   *[H]       (float32)   γ
    ln_bias_ptr,       # const   *[H]       (float32)   β
    weight_T_ptr,      # const   *[H, dim_out] (float16)   transposed final weight
    rows: tl.int32,
    H: tl.int32,
    dim_out: tl.int32,
    eps: tl.float32,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_O: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)               # [BLOCK_R]
    mask_r = offs_r < rows

    # --------------------------------------------------------------
    # 1) mean / variance over H (fp32)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_R], dtype=tl.float32)

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H
        vals = tl.load(out_ptr + offs_r[:, None] * H + offs_h[None, :],
                       mask=mask_r[:, None] & mask_h[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------------------------------------------------------------
    # 2) For each output‐dim tile:  (norm → affine → gate) • W_T
    # --------------------------------------------------------------
    for o_off in range(0, tl.cdiv(dim_out, BLOCK_O)):
        offs_o = o_off * BLOCK_O + tl.arange(0, BLOCK_O)
        mask_o = offs_o < dim_out

        acc = tl.zeros([BLOCK_R, BLOCK_O], dtype=tl.float32)   # accumulator

        for h_off in range(0, tl.cdiv(H, BLOCK_H)):
            offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
            mask_h = offs_h < H

            # load out and gate
            out_vals  = tl.load(out_ptr + offs_r[:, None] * H + offs_h[None, :],
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0).to(tl.float32)
            gate_vals = tl.load(gate_ptr + offs_r[:, None] * H + offs_h[None, :],
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0).to(tl.float32)

            # load LN affine parameters
            w = tl.load(ln_weight_ptr + offs_h, mask=mask_h, other=0.0)   # γ fp32
            b = tl.load(ln_bias_ptr   + offs_h, mask=mask_h, other=0.0)   # β fp32

            # (out - μ) * rstd  →  fp32
            norm = (out_vals - mean[:, None]) * rstd[:, None]

            # affine transformation of LayerNorm
            norm = norm * w[None, :] + b[None, :]

            # multiply by out‑gate
            gated = norm * gate_vals                               # fp32

            # load slice of final weight (shape [BLOCK_H, BLOCK_O])
            wgt = tl.load(weight_T_ptr + offs_h[:, None] * dim_out + offs_o[None, :],
                          mask=mask_h[:, None] & mask_o[None, :],
                          other=0.0).to(tl.float32)               # [Hblk,Oblk]

            # dot: (BLOCK_R, BLOCK_H)·(BLOCK_H, BLOCK_O) → (BLOCK_R, BLOCK_O)
            acc += tl.dot(gated, wgt)

        # write result
        y_ptr_off = y_ptr + offs_r[:, None] * dim_out + offs_o[None, :]
        y_fp16 = acc.to(tl.float16)
        tl.store(y_ptr_off, y_fp16,
                 mask=mask_r[:, None] & mask_o[None, :])

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : mapping with all Linear/LayerNorm parameters (float32)
        - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N            # total rows for row‑wise ops
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (fp32 → fp16)
    # ------------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()                # [R, dim]  fp32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 256
    BLOCK_D = 32
    grid = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],   # γ  (float32)
        weights["norm.bias"],     # β  (float32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection: 5·H outputs (left/right + 3 gates)
    # ------------------------------------------------------------------
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # [5·H, dim]

    proj = F.linear(x_norm, w_concat)        # [R, 5·H]  fp16
    del w_concat, x_norm                     # free memory early

    # ------------------------------------------------------------------
    # 3️⃣ Split, mask, sigmoid gates (single Triton kernel)
    # ------------------------------------------------------------------
    left      = torch.empty(R, H, dtype=torch.float16, device=device)
    right     = torch.empty(R, H, dtype=torch.float16, device=device)
    out_gate  = torch.empty(R, H, dtype=torch.float16, device=device)

    if cfg.get("nomask", True):
        # dummy mask – kernel will ignore it
        has_mask = 0
        dummy_mask = torch.empty(0, dtype=torch.float16, device=device)
        mask_ptr = dummy_mask
    else:
        # mask: [B,N,N] → [R]   (1.0 for True, 0.0 for False)
        mask_fp16 = mask.to(torch.float16).reshape(R)
        has_mask = 1
        mask_ptr = mask_fp16

    BLOCK_R = 128
    BLOCK_H = 32
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension (B·H, N, N) @ (B·H, N, N)ᵀ
    # ------------------------------------------------------------------
    left = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    left_mat  = left.view(B * H, N, N)      # [B·H, N, N]
    right_mat = right.view(B * H, N, N)     # [B·H, N, N]

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B·H, N, N)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]
    del left_mat, right_mat, left, right, out_mat

    out_flat = out.view(R, H).contiguous()   # [R, H]
    del out

    # ------------------------------------------------------------------
    # 5️⃣ Final LayerNorm + out_gate multiplication + projection (fused)
    # ------------------------------------------------------------------
    final = torch.empty(R, dim, dtype=torch.float16, device=device)

    # transpose final weight once (H × dim) for better memory layout
    w_out_T = weights["to_out.weight"].t().contiguous().to(torch.float16)   # [H, dim]

    BLOCK_R2 = 128
    BLOCK_H2 = 32
    BLOCK_O  = 32                         # dim‑tile size
    grid = (triton.cdiv(R, BLOCK_R2),)
    final_norm_gate_linear_kernel[grid](
        out_flat,
        out_gate,
        final,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        w_out_T,
        R,
        H,
        dim,
        eps,
        BLOCK_R=BLOCK_R2,
        BLOCK_H=BLOCK_H2,
        BLOCK_O=BLOCK_O,
        num_warps=4,
    )
    del out_flat, out_gate, w_out_T

    # ------------------------------------------------------------------
    # reshape to [B, N, N, dim] and cast to float32
    # ------------------------------------------------------------------
    out_float = final.view(B, N, N, dim).to(torch.float32)
    return out_float
