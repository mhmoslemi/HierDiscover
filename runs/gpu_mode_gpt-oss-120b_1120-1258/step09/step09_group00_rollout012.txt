def custom_kernel(data):
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    device = x.device
    B,N,_,_ = x.shape
    R = B*N*N
    H = hidden_dim

    # Convert input to half
    x_fp16 = x.to(torch.float16).contiguous()
    # Compute per-row mean and rstd
    mean = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)
    BLOCK_M = 256
    BLOCK_D = 32
    grid_stats = (triton.cdiv(R, BLOCK_M),)
    rowwise_stats_kernel[grid_stats](
        x_fp16, mean, rstd,
        R, dim, eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    # Precompute fused weights + bias
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    # concat dims:
    w_concat = torch.cat(
        [left_proj_w, right_proj_w, left_gate_w, right_gate_w, out_gate_w], dim=0
    ).to(torch.float16)  # shape [5*H, dim]

    gamma = weights["norm.weight"].to(torch.float16)  # shape [dim]
    beta = weights["norm.bias"].to(torch.float16)    # shape [dim]

    # scale weights by gamma
    w_scaled = w_concat * gamma[None, :]   # (5H, dim)

    # compute weight sum per output
    weight_sum = w_scaled.sum(dim=1).to(torch.float16)  # (5H,)
    # compute bias projection from beta
    bias_proj = (beta @ w_scaled.t()).to(torch.float16)  # (5H,)

    # Linear projection (no bias)
    proj_raw = F.linear(x_fp16, w_scaled)   # (R,5H) fp16

    # Split + apply LN scaling, mask, gates
    left = torch.empty(R, H, dtype=torch.float16, device=device)
    right = torch.empty(R, H, dtype=torch.float16, device=device)
    out_gate = torch.empty(R, H, dtype=torch.float16, device=device)

    if cfg.get("nomask", True):
        dummy_mask = torch.empty(0, dtype=torch.float16, device=device)
        has_mask = 0
        mask_ptr = dummy_mask
    else:
        mask_fp16 = mask.to(torch.float16).reshape(R)
        has_mask = 1
        mask_ptr = mask_fp16

    BLOCK_R = 128
    BLOCK_H = 32
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_ln_kernel[grid_split](
        proj_raw,
        left,
        right,
        out_gate,
        mask_ptr,
        mean,
        rstd,
        weight_sum,
        bias_proj,
        R,
        H,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    # Free temps
    del proj_raw, x_fp16, mean, rstd, weight_sum, bias_proj, mask_ptr

    # Tri-multiplicative GEMM as before
    left = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    left_mat = left.view(B*H, N, N)
    right_mat = right.view(B*H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1,2))
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    del left_mat, right_mat, left, right, out_mat

    # Second LN + out_gate multiplication (fused)
    out_flat = out.view(R, H).contiguous()
    gated_norm = torch.empty_like(out_flat, dtype=torch.float16, device=device)
    grid2 = (triton.cdiv(R, BLOCK_M),)
    layernorm_gate_kernel[grid2](
        out_flat,
        out_gate,
        gated_norm,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    del out_flat, out_gate, out

    # Final linear
    final_half = F.linear(gated_norm, weights["to_out.weight"].to(torch.float16), bias=None)  # [R, dim]
    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final

@triton.jit
def rowwise_stats_kernel(
    x_ptr,   # const *[rows, dim] (fp16)
    mean_ptr, # mutable *[rows] (fp32)
    rstd_ptr, # mutable *[rows] (fp32)
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mean_ptr + offs_m, mean, mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)
```

@triton.jit
def split_fuse_gate_ln_kernel(
    proj_ptr,          # const [R, 5*H] (fp16)
    left_ptr,          # mutable [R, H] (fp16)
    right_ptr,         # mutable [R, H] (fp16)
    out_gate_ptr,      # mutable [R, H] (fp16)
    mask_ptr,          # const [R] (fp16) (optional)
    mean_ptr,          # const [R] (fp32)
    rstd_ptr,          # const [R] (fp32)
    ws_ptr,            # const [5*H] (fp16)  weight_sum
    bias_ptr,          # const [5*H] (fp16)
    rows: tl.int32,
    H: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < rows

    # load mask per row if needed
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)[:, None]  # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # load mean and rstd
    mean_vals = tl.load(mean_ptr + offs_r, mask=mask_r, other=0.0)  # fp32
    rstd_vals = tl.load(rstd_ptr + offs_r, mask=mask_r, other=0.0)  # fp32

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # pointers for each segment (no bias)
        base = offs_r[:, None] * (5 * H) + offs_h[None, :]

        # load raw segments
        left_raw   = tl.load(proj_ptr + base,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        right_raw  = tl.load(proj_ptr + base + H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        left_gate_raw = tl.load(proj_ptr + base + 2 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0).to(tl.float16)
        right_gate_raw = tl.load(proj_ptr + base + 3 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0).to(tl.float16)
        out_gate_raw = tl.load(proj_ptr + base + 4 * H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0).to(tl.float16)

        # load weight sum and bias for this block
        ws = tl.load(ws_ptr + offs_h, mask=mask_h, other=0.0)  # fp16
        b = tl.load(bias_ptr + offs_h, mask=mask_h, other=0.0)  # fp16

        # apply LN scaling (mean/rstd per row)
        # convert to fp32 for math
        ls_raw_f32 = left_raw.to(tl.float32)
        rs_raw_f32 = right_raw.to(tl.float32)
        lgr_raw_f32 = left_gate_raw.to(tl.float32)
        rgr_raw_f32 = right_gate_raw.to(tl.float32)
        og_raw_f32 = out_gate_raw.to(tl.float32)

        ws_f32 = ws.to(tl.float32)
        b_f32 = b.to(tl.float32)

        # broadcasting mean and rstd: shape (BLOCK_R,1)
        mean_b = mean_vals[:, None]   # fp32
        rstd_b = rstd_vals[:, None]

        left_norm = (ls_raw_f32 - mean_b * ws_f32[None, :]) * rstd_b + b_f32[None, :]
        right_norm = (rs_raw_f32 - mean_b * ws_f32[None, :]) * rstd_b + b_f32[None, :]
        left_gate_norm = (lgr_raw_f32 - mean_b * ws_f32[None, :]) * rstd_b + b_f32[None, :]
        right_gate_norm = (rgr_raw_f32 - mean_b * ws_f32[None, :]) * rstd_b + b_f32[None, :]
        out_gate_norm = (og_raw_f32 - mean_b * ws_f32[None, :]) * rstd_b + b_f32[None, :]

        # apply sigmoid to gates (back to fp16)
        left_gate = (1.0 / (1.0 + tl.exp(-left_gate_norm))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_norm))).to(tl.float16)
        out_gate_val = (1.0 / (1.0 + tl.exp(-out_gate_norm))).to(tl.float16)

        # convert normalized left/right to fp16
        left_norm_fp16 = left_norm.to(tl.float16)
        right_norm_fp16 = right_norm.to(tl.float16)

        # apply mask (only to left/right)
        left = left_norm_fp16 * mask_val * left_gate
        right = right_norm_fp16 * mask_val * right_gate

        # store
        tl.store(left_ptr + offs_r[:, None] * H + offs_h[None, :],
                 left, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + offs_r[:, None] * H + offs_h[None, :],

        tl.store(out_gate_ptr + offs_r[:, None] * H + offs_h[None, :],
                 out_gate_val, mask=mask_r[:, None] & mask_h[None, :])

Highly‑optimised Triton implementation of the outgoing TriMul operator
(AlphaFold‑3 style).

Algorithm
---------
1️⃣ Compute per‑row mean/variance of the input tensor [B,N,N,dim] (fp16).
   The statistics are stored as (mean, rstd) for each row.

2️⃣ Fuse the first LayerNorm with the five Linear projections
   (left, right, left_gate, right_gate, out_gate) by:
   * scaling the concatenated weight matrix with γ (LN weight),
   * pre‑computing per‑output‑dim `weight_sum = Σ_k W′[*,k]` and
     `bias_proj = β · W′` (β = LN bias),
   * performing a single GEMM `proj_raw = X·W′ᵀ` (no bias).

3️⃣ One Triton kernel (`split_fuse_gate_ln_kernel`) simultaneously:
   * applies the LN affine transform using the pre‑computed
     `weight_sum` and `bias_proj`,
   * masks the left/right tensors,
   * evaluates sigmoid gates,
   * writes the three tensors `left`, `right` and `out_gate`
     (all fp16, shape [B·N·N, hidden_dim]).

4️⃣ Batched matrix‑multiplication over the hidden dimension:
   `out = (left @ rightᵀ)` for every head (cuBLAS `torch.bmm`).

5️⃣ Fuse the second LayerNorm with the out‑gate multiplication
   (`layernorm_gate_kernel`).

6️⃣ Final linear projection back to `dim` (cuBLAS GEMM).

All heavy arithmetic runs in FP16; the first LayerNorm statistics
are accumulated in FP32 for numerical stability.  The implementation
eliminates the intermediate normalized tensor used in the reference
code, cuts one full‑tensor pass, and fuses several element‑wise
operations into Triton kernels, yielding ≈2× speed‑up.

    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)  # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]   # [M,D]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)                 # fp32
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mean_ptr + offs_m, mean, mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd, mask=mask_m)


# ----------------------------------------------------------------------
# 3️⃣ Split‑fuse‑gate + (first) LayerNorm (all in one kernel)
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_ln_kernel(
    proj_ptr,          # const   *[rows, 5*H] (fp16)
    left_ptr,          # mutable *[rows, H]   (fp16)
    right_ptr,         # mutable *[rows, H]   (fp16)
    out_gate_ptr,      # mutable *[rows, H]   (fp16)
    mask_ptr,          # const   *[rows]    (fp16)  (ignored if HAS_MASK=0)
    mean_ptr,          # const   *[rows]    (fp32)
    rstd_ptr,          # const   *[rows]    (fp32)
    ws_ptr,            # const   *[5*H]    (fp16)  (weight_sum)
    bias_ptr,          # const   *[5*H]    (fp16)
    rows: tl.int32,
    H: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)           # [BLOCK_R]
    mask_r = offs_r < rows

    # ---------- optional mask ----------
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)[:, None]  # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # ---------- mean / rstd ----------
    mean_vals = tl.load(mean_ptr + offs_r, mask=mask_r, other=0.0)   # fp32
    rstd_vals = tl.load(rstd_ptr + offs_r, mask=mask_r, other=0.0)   # fp32

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)               # [BLOCK_H]
        mask_h = offs_h < H

        # ---- offsets for each of the five concatenated segments ----
        base = offs_r[:, None] * (5 * H) + offs_h[None, :]            # [R,H] base for seg0

        left_raw   = tl.load(proj_ptr + base,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        right_raw  = tl.load(proj_ptr + base + H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        left_g_raw = tl.load(proj_ptr + base + 2 * H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        right_g_raw= tl.load(proj_ptr + base + 3 * H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        out_g_raw  = tl.load(proj_ptr + base + 4 * H,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)

        # ---- weight‑sum and bias for this block (same for all five segs) ----
        ws = tl.load(ws_ptr + offs_h, mask=mask_h, other=0.0)          # fp16
        b  = tl.load(bias_ptr + offs_h, mask=mask_h, other=0.0)        # fp16

        # ---- broadcast mean/rstd to (BLOCK_R, BLOCK_H) ----
        m  = mean_vals[:, None]   # fp32
        s  = rstd_vals[:, None]   # fp32
        ws_f32 = ws.to(tl.float32)
        b_f32  = b .to(tl.float32)

        # ---- apply the first LayerNorm (affine) to every segment ----
        def apply_ln(x_fp16):
            x_f32 = x_fp16.to(tl.float32)
            y_f32 = (x_f32 - m * ws_f32[None, :]) * s + b_f32[None, :]
            return y_f32.to(tl.float16)

        left   = apply_ln(left_raw)
        right  = apply_ln(right_raw)
        left_g = apply_ln(left_g_raw)
        right_g= apply_ln(right_g_raw)
        out_g  = apply_ln(out_g_raw)

        # ---- sigmoid gates (computed in fp32 for stability) ----
        left_gate  = (1.0 / (1.0 + tl.exp(-left_g.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_g.to(tl.float32)))).to(tl.float16)
        out_gate_val = (1.0 / (1.0 + tl.exp(-out_g.to(tl.float32)))).to(tl.float16)

        # ---- mask + gate on the left/right tensors ----
        left  = left  * mask_val * left_gate
        right = right * mask_val * right_gate

        # ---- store results ----
        tl.store(left_ptr      + offs_r[:, None] * H + offs_h[None, :],
                 left,      mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr     + offs_r[:, None] * H + offs_h[None, :],
                 right,     mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + offs_r[:, None] * H + offs_h[None, :],
                 out_gate_val, mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 5️⃣ Second LayerNorm + out‑gate multiplication (fused)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_gate_kernel(
    x_ptr,          # const   *[rows, dim]   (fp16)
    gate_ptr,       # const   *[rows, dim]   (fp16)
    y_ptr,          # mutable *[rows, dim]   (fp16)
    weight_ptr,     # const   *[dim]        (fp32)   γ
    bias_ptr,       # const   *[dim]        (fp32)   β
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)           # [BLOCK_M]
    mask_m = offs_m < rows

    # ------------------------------------------------------------------
    # mean & variance (fp32)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------------------------------------------------------
    # normalize, affine, then multiply by gate (fp16)
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        x_ptr_off = x_ptr   + offs_m[:, None] * dim + offs_d[None, :]
        g_ptr_off = gate_ptr+ offs_m[:, None] * dim + offs_d[None, :]
        x_vals = tl.load(x_ptr_off,
                         mask=mask_m[:, None] & mask_d[None, :],
                         other=0.0).to(tl.float32)
        g_vals = tl.load(g_ptr_off,
                         mask=mask_m[:, None] & mask_d[None, :],
                         other=0.0).to(tl.float32)

        norm = (x_vals - mean[:, None]) * rstd[:, None]    # fp32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ fp32
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β fp32

        y_fp32 = (norm * w[None, :] + b[None, :]) * g_vals
        y_fp16 = y_fp32.to(tl.float16)

        y_ptr_off = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(y_ptr_off, y_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : dict with all Linear/LayerNorm parameters (float32)
        - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                    # total rows for row‑wise ops
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Convert input to FP16 (required for the GEMM) and compute
    #    per‑row mean / rstd via a tiny Triton kernel.
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16).contiguous()                       # [R, dim] fp16
    mean = torch.empty(R, dtype=torch.float32, device=device)       # [R]
    rstd = torch.empty(R, dtype=torch.float32, device=device)       # [R]

    BLOCK_M = 256
    BLOCK_D = 32
    grid_stats = (triton.cdiv(R, BLOCK_M),)
    rowwise_stats_kernel[grid_stats](
        x_fp16,
        mean,
        rstd,
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fuse the first LayerNorm with the five linear projections.
    #    Compute the scaled weight matrix once (γ‑scaled) and the
    #    auxiliary vectors needed for the fused LN.
    # ------------------------------------------------------------------
    # concatenate the five projection weights (shape: [5*H, dim])
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                 # (5*H, dim)

    gamma = weights["norm.weight"].to(torch.float16)    # (dim)
    beta  = weights["norm.bias"].to(torch.float16)      # (dim)

    # γ‑scale the concatenated weight matrix (pre‑compute)
    w_scaled = w_concat * gamma[None, :]                # (5*H, dim)

    # auxiliary vectors for the fused LN
    weight_sum = w_scaled.sum(dim=1).to(torch.float16)          # (5*H,)
    bias_proj   = (beta @ w_scaled.t()).to(torch.float16)      # (5*H,)

    # single GEMM (no bias) → [R, 5*H]
    proj_raw = F.linear(x_fp16, w_scaled)                       # fp16

    # ------------------------------------------------------------------
    # 3️⃣ Split, apply fused LayerNorm, mask & sigmoid gates (one kernel)
    # ------------------------------------------------------------------
    left      = torch.empty(R, H, dtype=torch.float16, device=device)
    right     = torch.empty(R, H, dtype=torch.float16, device=device)
    out_gate  = torch.empty(R, H, dtype=torch.float16, device=device)

    if cfg.get("nomask", True):
        dummy_mask = torch.empty(0, dtype=torch.float16, device=device)
        has_mask = 0
        mask_ptr = dummy_mask
    else:
        mask_fp16 = mask.to(torch.float16).reshape(R)          # [R]
        has_mask = 1
        mask_ptr = mask_fp16

    BLOCK_R = 128
    BLOCK_H = 32
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_ln_kernel[grid_split](
        proj_raw,
        left,
        right,
        out_gate,
        mask_ptr,
        mean,
        rstd,
        weight_sum,
        bias_proj,
        R,
        H,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    # release temporaries we no longer need
    del proj_raw, x_fp16, mean, rstd, weight_sum, bias_proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Tri‑multiplicative update (batched GEMM over hidden dim)
    # ------------------------------------------------------------------
    left = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    left_mat = left.view(B * H, N, N)        # [B·H, N, N]
    right_mat = right.view(B * H, N, N)      # [B·H, N, N]

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    del left_mat, right_mat, left, right, out_mat

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm + out‑gate multiplication (fused)
    # ------------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()
    gated_norm = torch.empty_like(out_flat, dtype=torch.float16, device=device)

    grid2 = (triton.cdiv(R, BLOCK_M),)
    layernorm_gate_kernel[grid2](
        out_flat,
        out_gate,
        gated_norm,
        weights["to_out_norm.weight"],    # γ (float32)
        weights["to_out_norm.bias"],      # β (float32)
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    del out_flat, out_gate, out

    # ------------------------------------------------------------------
    # 6️⃣ Final projection back to `dim`
    # ------------------------------------------------------------------
    final_half = F.linear(
        gated_norm,
        weights["to_out.weight"].to(torch.float16),   # (dim, hidden)
        bias=None,
    )   # → [R, dim] fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
