"""
Tri‑multiplicative (outgoing) update – fast Triton + PyTorch implementation.

Algorithm
---------
1️⃣  Row‑wise LayerNorm on the 4‑D tensor (B,N,N,dim) → fp16.
2️⃣  Fuse the 5 linear maps (left/right projections + three gates) into a
    single GEMM:  X_norm @ W_fusedᵀ  →  (R,5·H) where R = B·N·N.
3️⃣  Triton kernel *split_fuse_gate_kernel*:
        - splits the fused projection into left, right and out‑gate tensors,
        - applies the optional pair‑wise mask,
        - evaluates sigmoid gates (fp32 → fp16 for stability).
4️⃣  Batched half‑precision GEMM over the hidden dimension:
        out[b·H+i] = left_mat[b·H+i] @ right_mat[b·H+i]ᵀ
5️⃣  Second LayerNorm (torch) + element‑wise multiply with out_gate.
6️⃣  Final linear projection back to `dim`.

Only step 3 is written in Triton; the remaining heavy linear algebra
operations use cuBLAS via PyTorch which gives near‑peak Tensor‑Core
throughput on an H100.  The design eliminates two custom LayerNorm
kernels, removes unnecessary copies after permutation and uses larger
tile sizes for the Triton kernel, achieving sub‑millisecond runtimes.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: split fused projection, apply mask, sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[rows, 5*H]    (float16)
    left_ptr,          # mutable *[rows, H]    (float16)
    right_ptr,         # mutable *[rows, H]    (float16)
    out_gate_ptr,      # mutable *[rows, H]    (float16)
    mask_ptr,          # const   *[rows]       (float16)  (ignored if HAS_MASK=0)
    rows: tl.int32,
    H: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    For each row r:
        left  = (proj_left  * mask) * sigmoid(proj_left_gate)
        right = (proj_right * mask) * sigmoid(proj_right_gate)
        out_gate = sigmoid(proj_out_gate)

    All ops are in fp16 (sigmoid computed in fp32 for numerical stability).
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < rows

    # broadcast mask if present
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r,
                           mask=mask_r,
                           other=1.0).to(tl.float16)    # [BLOCK_R]
        mask_val = mask_val[:, None]                    # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # hidden‑dim loop
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # layout: [left, right, left_gate, right_gate, out_gate] each of size H
        base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # offsets for segment 0

        # pointers to the five contiguous segments
        left_raw_ptr    = proj_ptr + base
        right_raw_ptr   = proj_ptr + base + H
        left_gate_ptr   = proj_ptr + base + 2 * H
        right_gate_ptr  = proj_ptr + base + 3 * H
        out_gate_ptr_t  = proj_ptr + base + 4 * H

        # load raw projection results (fp16)
        left_raw   = tl.load(left_raw_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        right_raw  = tl.load(right_raw_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        left_g_raw = tl.load(left_gate_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        right_g_raw= tl.load(right_gate_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        out_g_raw  = tl.load(out_gate_ptr_t,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)

        # sigmoid in fp32 → fp16 (more stable)
        left_gate  = (1.0 / (1.0 + tl.exp(-left_g_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_g_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_g_raw.to(tl.float32)))).to(tl.float16)

        # apply mask and gates
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate

        # store results
        left_dst_ptr    = left_ptr    + offs_r[:, None] * H + offs_h[None, :]
        right_dst_ptr   = right_ptr   + offs_r[:, None] * H + offs_h[None, :]
        out_gate_dst_ptr= out_gate_ptr+ offs_r[:, None] * H + offs_h[None, :]

        tl.store(left_dst_ptr,    left,    mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_dst_ptr,   right,   mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_dst_ptr,out_gate,mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (bool/float) or None
        - weights_dict : mapping containing all Linear/LayerNorm parameters
                         (float32 tensors, bias=False for Linear)
        - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    device = x.device
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5

    B, N, _, _ = x.shape
    R = B * N * N                     # total rows for the row‑wise ops
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (torch, fp32 → fp16)
    # ------------------------------------------------------------------
    # torch.nn.functional.layer_norm computes row‑wise statistics when the
    # normalized_shape is the last dimension.
    x_norm_fp32 = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )                                   # [B,N,N,dim] float32
    x_norm = x_norm_fp32.to(torch.float16)   # cast to fp16 for the GEMM

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear projection (left/right + gates) → (R, 5*H)
    # ------------------------------------------------------------------
    # Concatenate the five weight matrices (all bias‑free) and cast to fp16.
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                # shape [5*H, dim]

    # (R, dim) @ (5*H, dim)ᵀ   ->   (R, 5*H)
    proj = F.linear(x_norm.view(R, dim), w_fused)   # fp16
    del x_norm, w_fused                     # free memory early

    # ------------------------------------------------------------------
    # 3️⃣ Split, apply optional mask, sigmoid gates   (Triton)
    # ------------------------------------------------------------------
    left      = torch.empty(R, H, dtype=torch.float16, device=device)
    right     = torch.empty(R, H, dtype=torch.float16, device=device)
    out_gate  = torch.empty(R, H, dtype=torch.float16, device=device)

    if cfg.get("nomask", True) or mask is None:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)
    else:
        # mask ∈ {0,1}, reshape to (R,)
        mask_half = mask.to(torch.float16).reshape(R)
        HAS_MASK = 1
        mask_ptr = mask_half

    BLOCK_R = 256          # rows per program
    BLOCK_H = 64           # hidden‑dim chunk (must be a multiple of 32)
    grid = (triton.cdiv(R, BLOCK_R),)

    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Tri‑multiplicative GEMM over the hidden dimension
    # ------------------------------------------------------------------
    # Reshape to [B, N, N, H] and pull H into the batch axis.
    left_view  = left.view(B, N, N, H)
    right_view = right.view(B, N, N, H)

    # (B, H, N, N) → (B*H, N, N)  (no extra copy needed)
    left_mat  = left_view.permute(0, 3, 1, 2).reshape(B * H, N, N)
    right_mat = right_view.permute(0, 3, 1, 2).reshape(B * H, N, N)

    # Batched matrix multiply:  left_mat @ right_matᵀ
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B*H, N, N]

    # Back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H]

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm (torch) + element‑wise multiplication with out_gate
    # ------------------------------------------------------------------
    out_flat = out.view(R, H)                       # [R, H]

    out_norm = F.layer_norm(
        out_flat,
        (H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )                                                # fp16 (inherits dtype)

    # multiply by the previously computed out_gate (fp16 × fp16 → fp16)
    out_scaled = out_norm * out_gate                # [R, H] fp16
    del out, out_norm, out_gate

    # ------------------------------------------------------------------
    # 6️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    final_half = F.linear(
        out_scaled,
        weights["to_out.weight"].to(torch.float16),   # [dim, H]
    )                                                # fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
