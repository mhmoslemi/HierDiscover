def custom_kernel(data):
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    device = x.device

    B,N,_,_ = x.shape
    R = B*N*N
    H = hidden_dim

    # first LayerNorm (use torch.nn.functional.layer_norm)
    # x shape [B,N,N,dim], LN over dim
    # output dtype fp32, then cast to fp16
    x_norm_fp16 = torch.nn.functional.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    ).to(torch.float16)
    # flatten for linear
    x_norm_flat = x_norm_fp16.view(R, dim)

    # fused linear
    w_fused = torch.cat([...], dim=0).to(torch.float16)
    proj = torch.nn.functional.linear(x_norm_flat, w_fused)  # [R,5*H] fp16

    # split and gating (kernel)
    left = torch.empty(R, H, dtype=torch.float16, device=device)
    right = torch.empty(R, H, dtype=torch.float16, device=device)
    out_gate = torch.empty(R, H, dtype=torch.float16, device=device)

    if cfg["nomask"]:
        dummy_mask = torch.empty(0, dtype=torch.float16, device=device)
        mask_ptr = dummy_mask
        HAS_MASK = 0
    else:
        mask_flat = mask.to(torch.float16).reshape(R)  # [R]
        mask_ptr = mask_flat
        HAS_MASK = 1

    # launch kernel
    BLOCK_R = 256
    BLOCK_H = 32
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj

    # pack left right matrices for bmm
    # shape to [B*H,N,N]
    left = left.view(B,N,N,H).permute(0,3,1,2).contiguous()
    right = right.view(B,N,N,H).permute(0,3,1,2).contiguous()
    left_mat = left.view(B*H, N, N)
    right_mat = right.view(B*H, N, N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1,2))  # [B*H,N,N]
    out = out_mat.view(B, H, N, N).permute(0,2,3,1).contiguous()  # [B,N,N,H] fp16

    # second LN
    out_norm = torch.nn.functional.layer_norm(
        out,
        (H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )  # default dtype same as input (fp16). returns fp16

    # apply out_gate
    # out_gate shape [R, H] → reshape to [B,N,N,H]
    out_gate_reshaped = out_gate.view(B,N,N,H)
    out_norm = out_norm * out_gate_reshaped

    # final linear
    final = torch.nn.functional.linear(
        out_norm.view(R, H),
        weights["to_out.weight"].to(torch.float16),
    )  # fp16

    final = final.view(B,N,N,dim).to(torch.float32)
    return final

"""
custom_kernel – high‑performance “outgoing” TriMul (AlphaFold3)

Algorithm
---------
1️⃣  Row‑wise LayerNorm on the 4‑D input (B·N·N × dim) – use
    torch.nn.functional.layer_norm which is cuDNN‑accelerated.

2️⃣  Fuse the five linear projections (left, right, left_gate,
    right_gate, out_gate) into a single GEMM:
        X_norm @ W_fusedᵀ → (R, 5·H)  (R = B·N·N, H = hidden_dim).

3️⃣  Triton kernel “split_fuse_gate_kernel”:
       * reads the fused projection,
       * (optionally) multiplies by the pair‑wise mask,
       * applies sigmoid gates (fp32 → fp16),
       * writes three tensors: left, right, out_gate.
    All operations are performed in fp16 except the sigmoid for
    numerical stability.

4️⃣  Reshape left/right to (B·H, N, N) and compute the
    batched matrix multiplication
        out = left @ rightᵀ               (torch.bmm, Tensor‑cores).

5️⃣  Second LayerNorm on the hidden‑dim tensor – again use the cudnn
    implementation.  Multiply the result element‑wise with the previously
    computed `out_gate`.

6️⃣  Final linear projection back to `dim` (fp16 → fp32).

Only the split‑gate step is custom Triton code; everything else
leverages highly‑optimised PyTorch / cuBLAS kernels, yielding
sub‑millisecond runtimes on an H100 for the target configurations.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: split fused projection, apply mask & sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[rows, 5*H] (float16)
    left_ptr,          # mutable *[rows, H]   (float16)
    right_ptr,         # mutable *[rows, H]   (float16)
    out_gate_ptr,      # mutable *[rows, H]   (float16)
    mask_ptr,          # const   *[rows]    (float16)   (ignored if HAS_MASK=0)
    rows: tl.int32,
    H: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    For each row r:
        left  = (proj_left  * mask) * sigmoid(proj_left_gate)
        right = (proj_right * mask) * sigmoid(proj_right_gate)
        out_gate = sigmoid(proj_out_gate)
    All ops use fp16, sigmoid evaluated in fp32.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)               # [BLOCK_R]
    mask_r = offs_r < rows

    # ------------------------------------------------------------------
    # optional scalar mask per row, broadcasted over the hidden dim
    # ------------------------------------------------------------------
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                   # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # ------------------------------------------------------------------
    # iterate over hidden dimension
    # ------------------------------------------------------------------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)          # [BLOCK_H]
        mask_h = offs_h < H

        # layout of fused tensor:
        #   0 .. H-1               -> left
        #   H .. 2H-1              -> right
        #   2H .. 3H-1             -> left_gate
        #   3H .. 4H-1             -> right_gate
        #   4H .. 5H-1             -> out_gate
        row_offset = offs_r[:, None] * (5 * H)                    # [R,1]
        col_offset = offs_h[None, :]                             # [1,H]

        left_raw   = tl.load(proj_ptr + row_offset + col_offset,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        right_raw  = tl.load(proj_ptr + row_offset + H + col_offset,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        left_g_raw = tl.load(proj_ptr + row_offset + 2 * H + col_offset,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        right_g_raw= tl.load(proj_ptr + row_offset + 3 * H + col_offset,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        out_g_raw  = tl.load(proj_ptr + row_offset + 4 * H + col_offset,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)

        # sigmoid in fp32 → back to fp16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_g_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_g_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_g_raw.to(tl.float32)))).to(tl.float16)

        # apply mask and gates
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate

        # store
        tl.store(left_ptr    + offs_r[:, None] * H + offs_h[None, :],
                 left,  mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr   + offs_r[:, None] * H + offs_h[None, :],
                 right, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr+ offs_r[:, None] * H + offs_h[None, :],
                 out_gate, mask=mask_r[:, None] & mask_h[None, :])

# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : mapping with all linear / layer‑norm parameters
        - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim], dtype float32
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N          # total number of rows (B·N·N)
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (use cuDNN implementation)
    # ------------------------------------------------------------------
    # normalized over the last dimension `dim`
    x_norm = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    ).to(torch.float16)               # [B, N, N, dim] fp16

    x_norm_flat = x_norm.view(R, dim)  # [R, dim]

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection (five linear layers in one GEMM)
    # ------------------------------------------------------------------
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)               # [5*H, dim]

    # (R, dim) @ (5*H, dim).T  → (R, 5*H)
    proj = F.linear(x_norm_flat, w_fused)  # fp16
    del w_fused, x_norm, x_norm_flat

    # ------------------------------------------------------------------
    # 3️⃣ Split fused tensor, apply mask & sigmoid gates (Triton)
    # ------------------------------------------------------------------
    left      = torch.empty(R, H, dtype=torch.float16, device=device)
    right     = torch.empty(R, H, dtype=torch.float16, device=device)
    out_gate  = torch.empty(R, H, dtype=torch.float16, device=device)

    if cfg.get("nomask", True):
        # no mask → compile‑time constant HAS_MASK=0
        dummy_mask = torch.empty(0, dtype=torch.float16, device=device)
        mask_ptr = dummy_mask
        HAS_MASK = 0
    else:
        mask_flat = mask.to(torch.float16).reshape(R)   # [R]
        mask_ptr = mask_flat
        HAS_MASK = 1

    # launch parameters (tuned for H ≤ 128)
    BLOCK_R = 256
    BLOCK_H = 32
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Tri‑multiplicative update (batched GEMM over hidden dim)
    # ------------------------------------------------------------------
    # reshape to [B, H, N, N] for batched matmul
    left  = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    left_mat  = left.view(B * H, N, N)   # [B·H, N, N]
    right_mat = right.view(B * H, N, N)  # [B·H, N, N]

    # (B·H, N, N) @ (B·H, N, N)ᵀ  →  (B·H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm + gate multiplication (cuDNN)
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        normalized_shape=(H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )  # still fp16

    # reshape out_gate to match out_norm shape and multiply
    out_gate_reshaped = out_gate.view(B, N, N, H)
    out_norm = out_norm * out_gate_reshaped
    del out_gate, out, left, right, left_mat, right_mat

    # ------------------------------------------------------------------
    # 6️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    final_fp16 = F.linear(
        out_norm.view(R, H),
        weights["to_out.weight"].to(torch.float16),   # [dim, H]
    )   # fp16

    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final
