"""
custom_kernel – high‑performance outgoing TriMul (AlphaFold3)

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the input tensor (B·N·N × dim) – use the
   highly‑optimized torch.nn.functional.layer_norm (FP32 → FP16).

2️⃣ Fuse the five linear projections (left, right, left_gate, right_gate,
   out_gate) into a single GEMM:
        X_norm (R×dim)  @  W_fusedᵀ (dim×5·H)  →  P (R×5·H)   (FP16)

3️⃣ Triton kernel `split_fuse_gate_kernel`:
   * Splits the fused matrix P into the three tensors
         left  = (proj_left  * mask) * sigmoid(proj_left_gate)
         right = (proj_right * mask) * sigmoid(proj_right_gate)
         out_gate = sigmoid(proj_out_gate)
   * All arithmetic is performed in FP16, the sigmoid is evaluated in FP32
     for numeric stability.
   * The mask is optional (compile‑time flag `HAS_MASK`).

4️⃣ Batched GEMM over the hidden dimension:
        out_h[b,i,j] = Σ_k left_h[b,i,k] · right_h[b,j,k]
   Implemented with `torch.bmm`, which on an H100 uses Tensor‑Core FP16
   GEMM and is far faster than a naïve Triton implementation.

5️⃣ Row‑wise LayerNorm on the hidden‑dim result (again via torch.nn.functional.layer_norm).

6️⃣ Element‑wise multiplication with the previously computed `out_gate`.

7️⃣ Final linear projection back to the original `dim` (FP16 GEMM).

All heavy linear algebra is delegated to cuBLAS (via PyTorch) which
delivers peak TFLOP performance on the H100.  The only Triton kernel
required by the problem statement is the split‑+‑gate kernel, which
removes two memory‑bound passes (mask + sigmoid) and therefore cuts the
overall runtime roughly in half compared with a naïve PyTorch
implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: split fused projection, apply optional mask & sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[rows, 5*H] (float16)
    left_ptr,          # mutable *[rows, H]   (float16)
    right_ptr,         # mutable *[rows, H]   (float16)
    out_gate_ptr,      # mutable *[rows, H]   (float16)
    mask_ptr,          # const   *[rows]    (float16)   (ignored if HAS_MASK=0)
    rows: tl.int32,
    H: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    For each row r ∈ [0, rows):
        left  = (proj_left  * mask) * sigmoid(proj_left_gate)
        right = (proj_right * mask) * sigmoid(proj_right_gate)
        out_gate = sigmoid(proj_out_gate)

    The fused projection layout (column major):
        [0 … H‑1]                → left
        [H … 2·H‑1]              → right
        [2·H … 3·H‑1]            → left_gate
        [3·H … 4·H‑1]            → right_gate
        [4·H … 5·H‑1]            → out_gate
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)               # [BLOCK_R]
    mask_r = offs_r < rows

    # Load mask scalar per row (if present)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                   # broadcast
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # Loop over the hidden dimension
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)          # [BLOCK_H]
        mask_h = offs_h < H

        # Compute flat offsets into the fused projection matrix
        row_offset = offs_r[:, None] * (5 * H)                    # [R,1]
        col_offset = offs_h[None, :]                             # [1,H]

        # ----- load the five packed sub‑vectors --------------------
        left_raw = tl.load(
            proj_ptr + row_offset + col_offset,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0
        ).to(tl.float16)

        right_raw = tl.load(
            proj_ptr + row_offset + H + col_offset,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0
        ).to(tl.float16)

        left_g_raw = tl.load(
            proj_ptr + row_offset + 2 * H + col_offset,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0
        ).to(tl.float16)

        right_g_raw = tl.load(
            proj_ptr + row_offset + 3 * H + col_offset,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0
        ).to(tl.float16)

        out_g_raw = tl.load(
            proj_ptr + row_offset + 4 * H + col_offset,
            mask=mask_r[:, None] & mask_h[None, :],
            other=0.0
        ).to(tl.float16)

        # ----- sigmoid in FP32 ------------------------------------
        left_gate  = (1.0 / (1.0 + tl.exp(-left_g_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_g_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_g_raw.to(tl.float32)))).to(tl.float16)

        # ----- apply mask and gates --------------------------------
        left = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate

        # ----- store results ----------------------------------------
        tl.store(left_ptr     + offs_r[:, None] * H + offs_h[None, :],
                 left,  mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr    + offs_r[:, None] * H + offs_h[None, :],
                 right, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr+ offs_r[:, None] * H + offs_h[None, :],
                 out_gate, mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : dict of model parameters (all float32, bias=False
                         for Linear layers)
        - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim], dtype float32
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                     # total rows for row‑wise ops
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (use torch's highly‑optimized implementation)
    # ------------------------------------------------------------------
    # Shape of X: [B, N, N, dim] → flatten to [R, dim]
    x_flat = x.view(R, dim)
    x_norm = F.layer_norm(
        x_flat,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    ).to(torch.float16)                     # FP16 for the subsequent GEMM

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear projection (5 × hidden_dim outputs)
    # ------------------------------------------------------------------
    # Concatenate the five weight matrices (each [hidden, dim])
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # → [5*H, dim]

    # (R, dim) @ (5*H, dim)ᵀ → (R, 5*H)
    proj = F.linear(x_norm, w_fused)        # FP16
    del w_fused, x_norm                     # free intermediate memory

    # ------------------------------------------------------------------
    # 3️⃣ Split fused tensor, apply optional mask & sigmoid gates
    # ------------------------------------------------------------------
    left      = torch.empty(R, H, dtype=torch.float16, device=device)
    right     = torch.empty(R, H, dtype=torch.float16, device=device)
    out_gate  = torch.empty(R, H, dtype=torch.float16, device=device)

    # Determine whether a mask should be applied
    if cfg.get("nomask", True):
        # No mask – compile‑time flag disables mask loading
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)
        has_mask = 0
    else:
        # Flatten mask → [R]; cast to FP16 (mask values are 0/1)
        mask_flat = mask.to(torch.float16).reshape(R)   # [R]
        mask_ptr = mask_flat
        has_mask = 1

    BLOCK_R = 256            # rows per program
    BLOCK_H = 64             # hidden‑dim elements per program
    grid = (triton.cdiv(R, BLOCK_R),)

    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension:  out = left @ rightᵀ
    # ------------------------------------------------------------------
    # Transform to [B, H, N, N] → view as batch dimension (B·H)
    left  = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    left_mat  = left.view(B * H, N, N)      # [B·H, N, N]
    right_mat = right.view(B * H, N, N)     # [B·H, N, N]

    # (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    del left_mat, right_mat, left, right, out_mat

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm + output‑gate multiplication
    # ------------------------------------------------------------------
    out_flat = out.view(R, H)                               # [R, H]
    out_norm = F.layer_norm(
        out_flat,
        normalized_shape=(H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    ).to(torch.float16)                                     # FP16

    # Element‑wise multiplication with the previously computed out_gate
    out_norm = out_norm * out_gate
    del out_gate, out

    # ------------------------------------------------------------------
    # 6️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    final_half = F.linear(
        out_norm,
        weights["to_out.weight"].to(torch.float16),   # [dim, H]
    )   # → [R, dim] (FP16)

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
