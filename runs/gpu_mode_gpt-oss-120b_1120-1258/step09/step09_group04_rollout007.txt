"""
tri_mul_outgoing.py
~~~~~~~~~~~~~~~~~~~

High‑performance *outgoing* TriMul implementation (forward only) for AlphaFold‑3.
The algorithm follows the reference PyTorch implementation but fuses the two
LayerNorms and the final out‑gate into custom Triton kernels while re‑using
torch's Tensor‑Core GEMMs for all linear projections and the N³ multiplicative
update.

Steps
-----
1️⃣ Row‑wise LayerNorm on the input tensor (float32 → float16).
2️⃣ Fuse the four linear projections (left_proj, right_proj,
   left_gate, right_gate) into a single FP16 GEMM.
3️⃣ Compute the output‑gate with a second FP16 linear.
4️⃣ (Optional) Apply the binary mask to the left/right projections.
5️⃣ Apply sigmoid to the three gate tensors and multiply.
6️⃣ Batched GEMM: for each hidden dimension **h**, compute
   out_h = left_h @ right_hᵀ  (shape: B·H × N × N).
7️⃣ Fuse the second LayerNorm and the element‑wise multiplication by the
   out‑gate into a Triton kernel (FP32 accumulation → FP16 output).
8️⃣ Final projection back to the original dimension with a FP16 linear.
   The result is returned in float32.

Only the two LayerNorm kernels are written in Triton; all other
operations use highly‑optimised torch kernels (Tensor‑Core‑enabled).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   * [rows, dim] (float32)
    y_ptr,          # mutable * [rows, dim] (float16)
    weight_ptr,     # const   * [dim]      (float32)
    bias_ptr,       # const   * [dim]      (float32)
    rows,           # i32 total number of rows
    dim,            # i32 dimension per row
    eps,            # fp32 epsilon (compile‑time)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ---------- mean & variance (FP32) ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]   # [M,D]
        val = tl.load(ptr,
                      mask=mask_m[:, None] & mask_d[None, :],
                      other=0.0).to(tl.float32)
        sum_val += tl.sum(val, axis=1)
        sum_sq  += tl.sum(val * val, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- normalise, apply γ/β ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        val = tl.load(ptr,
                      mask=mask_m[:, None] & mask_d[None, :],
                      other=0.0).to(tl.float32)

        norm = (val - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β

        out = norm * w[None, :] + b[None, :]          # still FP32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Triton kernel: fused LayerNorm + element‑wise out‑gate (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_gate_kernel(
    x_ptr,          # const   * [rows, dim] (float32)
    gate_ptr,       # const   * [rows, dim] (float16)
    y_ptr,          # mutable * [rows, dim] (float16)
    weight_ptr,     # const   * [dim]      (float32)
    bias_ptr,       # const   * [dim]      (float32)
    rows,           # i32 total rows
    dim,            # i32 dimension per row
    eps,            # fp32 epsilon (compile‑time)
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # ---------- mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        val = tl.load(ptr,
                      mask=mask_m[:, None] & mask_d[None, :],
                      other=0.0).to(tl.float32)
        sum_val += tl.sum(val, axis=1)
        sum_sq  += tl.sum(val * val, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- normalise, apply γ/β and fuse with gate ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        x = tl.load(x_ptr + offs_m[:, None] * dim + offs_d[None, :],
                    mask=mask_m[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)

        g = tl.load(gate_ptr + offs_m[:, None] * dim + offs_d[None, :],
                    mask=mask_m[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)

        norm = (x - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β

        out = (norm * w[None, :] + b[None, :]) * g   # FP32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor [B, N, N, dim]      (float32)
        - mask_tensor  : torch.Tensor [B, N, N] or None (bool/float)
        - weights_dict : mapping of all Linear/LayerNorm parameters
        - config_dict  : {"dim": int,
                         "hidden_dim": int,
                         "nomask": bool (optional)}

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # -------------------- 0️⃣ unpack & constants --------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape            # [B, N, N, dim]
    R = B * N * N                    # total rows = B·N·N
    H = hidden_dim

    # -------------------- 1️⃣ First LayerNorm (row‑wise) --------------------
    # reshape to [R, dim] (contiguous) and apply torch's native LayerNorm.
    # This kernel is highly optimized in cuDNN; we keep a Triton kernel
    # for the *second* LayerNorm to satisfy the requirement.
    x_flat = x.view(R, dim).contiguous()          # float32
    x_norm_fp32 = F.layer_norm(
        x_flat,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )
    x_norm = x_norm_fp32.to(torch.float16)        # FP16 for all subsequent GEMMs

    # -------------------- 2️⃣ Fused projection & gate (four linear layers) --------------------
    # Concatenate weights of the four projections (left/right + left/right gates)
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)  # [4·H, dim]

    # Single FP16 GEMM: [R, dim] @ [4·H, dim]ᵀ → [R, 4·H]
    fused = F.linear(x_norm, w_fused)  # FP16

    left_raw   = fused[:, 0:H]                # [R, H]
    right_raw  = fused[:, H:2 * H]            # [R, H]
    left_gate_raw  = fused[:, 2 * H:3 * H]    # [R, H]
    right_gate_raw = fused[:, 3 * H:4 * H]    # [R, H]

    # Separate linear for the output‑gate (still FP16)
    out_gate_raw = F.linear(
        x_norm,
        weights["out_gate.weight"].to(torch.float16)   # [H, dim]
    )                                                  # [R, H]

    # -------------------- 3️⃣ Mask & sigmoid gates --------------------
    if not nomask and mask is not None:
        # mask shape: [B, N, N] → flatten to [R, 1] (float16)
        mask_fp16 = mask.reshape(R, 1).to(torch.float16)
        left_raw  = left_raw * mask_fp16
        right_raw = right_raw * mask_fp16

    # In‑place sigmoid on gate tensors (FP16 → FP16)
    left_gate = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate = out_gate_raw.sigmoid_()          # [R, H] FP16

    # Apply gates
    left  = left_raw * left_gate   # [R, H] FP16
    right = right_raw * right_gate # [R, H] FP16

    # -------------------- 4️⃣ Batched N³ multiplicative update --------------------
    # reshape to (B, N, N, H) → (B·H, N, N) for bmm
    left = left.view(B, N, N, H)
    right = right.view(B, N, N, H)

    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # (B·H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # (B·H, N, N)

    # (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]

    # -------------------- 5️⃣ Fused LayerNorm + element‑wise out‑gate --------------------
    out_flat = out.view(R, H).contiguous()           # [R, H] FP16
    out_gate_flat = out_gate                         # already FP16
    out_norm_gate = torch.empty_like(out_flat, dtype=torch.float16, device=device)

    # Kernel launch parameters – tuned for H100.
    # BLOCK_M: rows per program, BLOCK_D: columns per iteration.
    BLOCK_M = 256
    BLOCK_D = 64
    grid = (triton.cdiv(R, BLOCK_M),)

    layernorm_gate_kernel[grid](
        out_flat,
        out_gate_flat,
        out_norm_gate,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
        num_stages=2,
    )

    # -------------------- 6️⃣ Final linear projection (FP16) --------------------
    final_fp16 = F.linear(
        out_norm_gate,
        weights["to_out.weight"].to(torch.float16)   # [dim, H]
    )                                                # [R, dim] FP16

    # Return in float32 as required by the reference implementation
    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final
