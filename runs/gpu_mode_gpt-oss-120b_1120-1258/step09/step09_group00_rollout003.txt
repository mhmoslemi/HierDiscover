"""
custom_kernel: fast outgoing Tri‑Multiplicative update (TriMul) for AlphaFold‑3.

Algorithm (B – batch size, N – sequence length, D – input dim, H – hidden dim)
---------------------------------------------------------------------------
1️⃣ Row‑wise LayerNorm on the flattened [B·N·N, D] tensor (FP16)
   – uses torch.nn.functional.layer_norm (cuDNN) for maximal throughput.

2️⃣ One fused Linear projection:
        X_norm @ W_fusedᵀ    →   [R, 5·H]   (R = B·N·N)
   – W_fused concatenates the five weight matrices (left, right,
     left_gate, right_gate, out_gate).  No bias (as in the reference).

3️⃣ Triton kernel **split_fuse_gate_kernel**:
   * Splits the fused projection into left, right and three gate tensors.
   * (Optional) applies the pair‑wise mask.
   * Applies sigmoid to the three gates (computed in FP32) and
     stores   left * mask * left_gate,
              right * mask * right_gate,
              out_gate   (all FP16).

4️⃣ Batched GEMM over the hidden dimension:
        (B·H, N, N) @ (B·H, N, N)ᵀ   →   (B·H, N, N)
   – performed with torch.bmm on FP16 tensors (tensor‑core friendly).

5️⃣ Second LayerNorm + out‑gate multiplication.
   – cuDNN LayerNorm on the hidden‑dim tensor (FP16) with its
     own γ/β parameters, then element‑wise multiply by the out‑gate.

6️⃣ Final Linear projection back to the original dimension D
   – FP16 GEMM (torch.nn.functional.linear).

Only step 3 is written in Triton; the rest rely on highly‑optimized
cuDNN / cuBLAS kernels.  The implementation supports optional masking,
both `nomask=True` and `nomask=False`, and works for the full range of
test shapes described in the problem statement.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: split fused projection, apply mask & sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[rows, 5*H]   (fp16)
    left_ptr,          # mutable *[rows, H]   (fp16)
    right_ptr,         # mutable *[rows, H]   (fp16)
    out_gate_ptr,      # mutable *[rows, H]   (fp16)
    mask_ptr,          # const   *[rows]      (fp16)   (ignored if HAS_MASK=0)
    rows: tl.int32,
    H: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)      # [BLOCK_R]
    mask_r = offs_r < rows

    # ------------------------------------------------------------------
    # Optional mask: broadcast later to the hidden dimension
    # ------------------------------------------------------------------
    if HAS_MASK:
        # One mask value per row (pair (i,k))
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)  # [BLOCK_R]
        mask_val = mask_val[:, None]                                 # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # ------------------------------------------------------------------
    # Loop over hidden dimension
    # ------------------------------------------------------------------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # layout inside proj: [left, right, left_gate, right_gate, out_gate]
        base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw_ptr    = proj_ptr + base
        right_raw_ptr   = proj_ptr + base + H
        left_gate_ptr   = proj_ptr + base + 2 * H
        right_gate_ptr  = proj_ptr + base + 3 * H
        out_gate_raw_ptr= proj_ptr + base + 4 * H

        left_raw    = tl.load(left_raw_ptr,
                              mask=mask_r[:, None] & mask_h[None, :],
                              other=0.0).to(tl.float16)
        right_raw   = tl.load(right_raw_ptr,
                              mask=mask_r[:, None] & mask_h[None, :],
                              other=0.0).to(tl.float16)
        left_g_raw  = tl.load(left_gate_ptr,
                              mask=mask_r[:, None] & mask_h[None, :],
                              other=0.0).to(tl.float16)
        right_g_raw = tl.load(right_gate_ptr,
                              mask=mask_r[:, None] & mask_h[None, :],
                              other=0.0).to(tl.float16)
        out_g_raw   = tl.load(out_gate_raw_ptr,
                              mask=mask_r[:, None] & mask_h[None, :],
                              other=0.0).to(tl.float16)

        # sigmoid in FP32 for numerical stability
        lg = left_g_raw.to(tl.float32)
        rg = right_g_raw.to(tl.float32)
        og = out_g_raw.to(tl.float32)

        left_gate  = (1.0 / (1.0 + tl.exp(-lg))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-rg))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-og))).to(tl.float16)

        # apply mask and gates
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate

        # store results
        tl.store(left_ptr    + offs_r[:, None] * H + offs_h[None, :],
                 left,  mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr   + offs_r[:, None] * H + offs_h[None, :],
                 right, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr+ offs_r[:, None] * H + offs_h[None, :],
                 out_gate, mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
        - weights_dict : dict of all linear / layer‑norm parameters (float32)
        - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    x, mask, weights, cfg = data
    device = x.device
    dim = cfg["dim"]               # input channel dimension D
    hidden_dim = cfg["hidden_dim"] # hidden dimension H
    eps = 1e-5

    B, N, _, _ = x.shape
    R = B * N * N                 # total number of pairwise rows
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP16) – cuDNN implementation
    # --------------------------------------------------------------
    # flatten & cast to FP16 for bandwidth‑optimal LN
    x_flat_fp16 = x.view(R, dim).to(torch.float16)
    norm_w_fp16 = weights["norm.weight"].to(torch.float16)
    norm_b_fp16 = weights["norm.bias"].to(torch.float16)
    x_norm = F.layer_norm(x_flat_fp16,
                          (dim,),
                          weight=norm_w_fp16,
                          bias=norm_b_fp16,
                          eps=eps)                     # [R, dim]  fp16

    # --------------------------------------------------------------
    # 2️⃣ Fused linear projection producing (5·H) features per row
    # --------------------------------------------------------------
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],   # [H, D]
            weights["right_proj.weight"],  # [H, D]
            weights["left_gate.weight"],   # [H, D]
            weights["right_gate.weight"],  # [H, D]
            weights["out_gate.weight"],    # [H, D]
        ],
        dim=0,
    ).to(torch.float16)                                 # [5·H, D]

    proj = F.linear(x_norm, w_fused)                     # [R, 5·H]  fp16
    del w_fused, x_norm                                   # free memory early

    # --------------------------------------------------------------
    # 3️⃣ Split, mask & sigmoid (single Triton kernel)
    # --------------------------------------------------------------
    left   = torch.empty(R, H, dtype=torch.float16, device=device)
    right  = torch.empty(R, H, dtype=torch.float16, device=device)
    out_gate = torch.empty(R, H, dtype=torch.float16, device=device)

    if cfg.get("nomask", True):
        has_mask = 0
        # Triton still requires a pointer; pass an empty tensor (won't be read)
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)
    else:
        has_mask = 1
        # mask: bool/float → FP16 (1.0 for kept, 0.0 for masked) and flatten
        mask_fp16 = mask.to(torch.float16).reshape(R)
        mask_ptr = mask_fp16

    BLOCK_R = 128   # rows per program
    BLOCK_H = 32    # hidden dim chunk
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, mask_ptr                               # free memory

    # --------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension (B·H, N, N) @ (B·H, N, N)^T
    # --------------------------------------------------------------
    # reshape to [B, N, N, H] → permute to [B, H, N, N] for batched matmul
    left = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    left_mat = left.view(B * H, N, N)    # [B·H, N, N]
    right_mat = right.view(B * H, N, N)  # [B·H, N, N]

    # (B·H, N, N) @ (B·H, N, N)ᵀ   →   (B·H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]
    del left_mat, right_mat, left, right, out_mat

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm + out‑gate multiplication (cuDNN LN)
    # --------------------------------------------------------------
    out_flat_fp16 = out.view(R, H).contiguous()
    to_out_norm_w_fp16 = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b_fp16 = weights["to_out_norm.bias"].to(torch.float16)

    # LayerNorm on hidden dim (FP16)
    out_norm = F.layer_norm(out_flat_fp16,
                            (H,),
                            weight=to_out_norm_w_fp16,
                            bias=to_out_norm_b_fp16,
                            eps=eps)                # [R, H] fp16

    # Apply out‑gate (element‑wise) before the final projection
    gated = out_norm * out_gate                                 # fp16
    del out_flat_fp16, out_norm, out_gate, out

    # --------------------------------------------------------------
    # 6️⃣ Final linear projection back to `dim` (FP16 GEMM)
    # --------------------------------------------------------------
    final_half = F.linear(gated,
                         weights["to_out.weight"].to(torch.float16),  # [dim, H]
                         bias=None)                                 # [R, dim] fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)        # fp32 output
    return final
```
