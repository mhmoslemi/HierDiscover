"""
custom_kernel: high‑performance forward of the *outgoing* TriMul operator.

Algorithm
---------
1️⃣ Row‑wise LayerNorm over the last channel dimension (`dim`).
   Implemented in Triton (`layernorm_kernel`), FP32 reduction → FP16 output.

2️⃣ Fuse the four linear projections (left/right + their gates) into a single
   GEMM:   X_norm @ W_fused.T  →  [R, 4·H]   (R = B·N·N, H = hidden_dim).
   Split the result into left_raw, right_raw, left_gate_raw, right_gate_raw.

3️⃣ Compute the output‑gate linear projection on the same normalized input.

4️⃣ If a mask is present, apply it to left/right (but **not** to the gates),
   then apply the sigmoid gates:
        left  = (left_raw  * mask) * sigmoid(left_gate_raw)
        right = (right_raw * mask) * sigmoid(right_gate_raw)

5️⃣ Batched N³ multiplicative update.
   Reshape left/right → (B·H, N, N) and perform a batched matrix‑multiply:
        out_mat = (B·H, N, N) @ (B·H, N, N)ᵀ
   Result shape → (B, N, N, H).

6️⃣ Fuse the second LayerNorm (over hidden dim `H`) with the element‑wise
   multiplication by the output‑gate in a single Triton kernel
   (`layernorm_gate_kernel`).

7️⃣ Final linear projection back to the original channel dimension `dim`
   (FP16 GEMM).  Cast the result back to FP32 for the model output.

The implementation mixes Triton kernels for the two LayerNorms (the only
operations that cannot be expressed as a single cuBLAS call) with
highly‑optimised PyTorch GEMMs for the linear layers and the batched
matrix‑multiply.  All heavy compute is done in FP16, while the LayerNorms
use FP32 accumulation for numerical stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   * [total_rows, dim] (float32)
    y_ptr,          # mutable * [total_rows, dim] (float16)
    weight_ptr,     # const   * [dim]            (float32)  γ
    bias_ptr,       # const   * [dim]            (float32)  β
    total_rows,     # int32   runtime
    dim,            # int32   runtime
    eps,            # float   runtime (epsilon)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # cols per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)               # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ----- compute row‑wise mean & variance (FP32) --------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)            # FP32 load
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                               # 1/√(var+eps)

    # ----- write normalised values: out = γ·(x−μ)·rstd + β (FP16) -----
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]                       # FP32

        out_fp16 = out.to(tl.float16)
        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

# Triton kernel: fused LayerNorm + element‑wise gate (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_gate_kernel(
    x_ptr,          # const   * [total_rows, dim] (float16)
    gate_ptr,       # const   * [total_rows, dim] (float16)
    y_ptr,          # mutable * [total_rows, dim] (float16)
    weight_ptr,     # const   * [dim]            (float32)  γ
    bias_ptr,       # const   * [dim]            (float32)  β
    total_rows,     # int32   runtime
    dim,            # int32   runtime
    eps,            # float   runtime (epsilon)
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)               # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ----- compute mean & variance (ignore gate) ----------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ----- normalise, apply γ/β and multiply by gate (FP32 → FP16) ----
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        # load activation and gate
        x_vals = tl.load(x_ptr + offs_m[:, None] * dim + offs_d[None, :],
                         mask=mask_m[:, None] & mask_d[None, :],
                         other=0.0).to(tl.float32)
        g_vals = tl.load(gate_ptr + offs_m[:, None] * dim + offs_d[None, :],
                         mask=mask_m[:, None] & mask_d[None, :],
                         other=0.0).to(tl.float32)

        norm = (x_vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β

        out = (norm * w[None, :] + b[None, :]) * g_vals
        out_fp16 = out.to(tl.float16)

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

    # --------------------------------------------------------------

    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    total_rows = B * N * N          # R = B·N·N
    H = hidden_dim

    # 1️⃣ Row‑wise LayerNorm over channel dim (FP32 → FP16)
    # --------------------------------------------------------------

                              dtype=torch.float16,

        x_flat,
        x_norm_fp16,
        weights["norm.weight"],   # γ (float32)
        weights["norm.bias"],     # β (float32)
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,

            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]

    left_raw       = fused[:, 0:H]                # [R, H]
    right_raw      = fused[:, H:2*H]              # [R, H]
    left_gate_raw  = fused[:, 2*H:3*H]            # [R, H]
    right_gate_raw = fused[:, 3*H:4*H]            # [R, H]

    # Output‑gate (still on normalized input)
    out_gate_raw = F.linear(
        x_norm_fp16,
        weights["out_gate.weight"].to(torch.float16)   # [H, dim]
    )                                                   # [R, H]

    # --------------------------------------------------------------
    # 3️⃣ Apply mask (if any) and sigmoid gates
    # --------------------------------------------------------------
    if (not nomask) and (mask is not None):
        # mask: [B, N, N] → [R, 1] (FP16)
        mask_fp16 = mask.to(torch.float16).reshape(total_rows, 1)
        left_raw  = left_raw * mask_fp16
        right_raw = right_raw * mask_fp16

    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()      # FP16

    left  = left_raw  * left_gate   # [R, H] FP16
    right = right_raw * right_gate  # [R, H] FP16

    # --------------------------------------------------------------
    # 4️⃣ Batched N³ multiplicative update (left @ rightᵀ)
    # --------------------------------------------------------------
    # reshape to (B, N, N, H)
    left  = left.view(B, N, N, H)
    right = right.view(B, N, N, H)

    # move hidden dim into batch axis → (B·H, N, N)
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # [B,N,N,H]

    # --------------------------------------------------------------
    # 5️⃣ Fuse second LayerNorm (over H) with out_gate multiplication
    # --------------------------------------------------------------
    out_flat = out.view(total_rows, H).contiguous()      # [R, H] FP16
    out_gate_flat = out_gate                             # [R, H] FP16
    out_norm_gate_fp16 = torch.empty_like(out_flat, dtype=torch.float16, device=device)

    layernorm_gate_kernel[grid](
        out_flat,
        out_gate_flat,
        out_norm_gate_fp16,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    # --------------------------------------------------------------
    # 6️⃣ Final linear projection back to original dim (FP16 GEMM)
    # --------------------------------------------------------------
    final_fp16 = F.linear(
        out_norm_gate_fp16,
        weights["to_out.weight"].to(torch.float16)   # [dim, H]
    )                                                # [R, dim] FP16

    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final
