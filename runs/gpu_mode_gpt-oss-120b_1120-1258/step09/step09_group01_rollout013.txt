def layernorm_proj_kernel(
    x_ptr, proj_ptr, weight_T_ptr,
    rows, dim, out_dim ( = 5*H ), eps,
    BLOCK_R, BLOCK_D, BLOCK_O
):
   pid = tl.program_id(0)
   row_start = pid * BLOCK_R
   offs_r = row_start + tl.arange(0, BLOCK_R)
   mask_r = offs_r < rows
   # first pass: compute mean,var
   sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
   sum_sq = tl.zeros([BLOCK_R], dtype=tl.float32)
   for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off*BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        x_ptrs = x_ptr + offs_r[:,None] * dim + offs_d[None,:]
        vals = tl.load(x_ptrs, mask=mask_r[:,None] & mask_d[None,:], other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals*vals, axis=1)

   mean = sum_val / dim
   var = sum_sq / dim - mean*mean
   rstd = tl.rsqrt(var + eps)

   # second pass: compute proj in output blocks
   for o_off in range(0, tl.cdiv(out_dim, BLOCK_O)):
        offs_o = o_off*BLOCK_O + tl.arange(0, BLOCK_O)
        mask_o = offs_o < out_dim
        acc = tl.zeros([BLOCK_R, BLOCK_O], dtype=tl.float32)
        for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
            offs_d = d_off*BLOCK_D + tl.arange(0, BLOCK_D)
            mask_d = offs_d < dim
            # load x

            vals = tl.load(x_ptrs, mask=mask_r[:,None] & mask_d[None,:], other=0.0).to(tl.float32)
            # normalize
            x_norm = (vals - mean[:, None]) * rstd[:, None] # shape [B_R, B_D]
            # load weight block (dim, out_dim) transposed already
            w_ptrs = weight_T_ptr + offs_d[:, None] * out_dim + offs_o[None, :]  # shape [B_D, B_O]

    x_ptr, proj_ptr, weight_ptr,
    rows, dim, out_dim, eps,
    BLOCK_R: tl.constexpr, BLOCK_D: tl.constexpr, BLOCK_O: tl.constexpr
):

    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < rows

    # 1st pass: compute mean,var
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_R], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        x_ptrs = x_ptr + offs_r[:, None] * dim + offs_d[None, :]
        vals = tl.load(x_ptrs, mask=mask_r[:, None] & mask_d[None, :], other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # 2nd pass: compute projection (including normalization)
    for o_off in range(0, tl.cdiv(out_dim, BLOCK_O)):
        offs_o = o_off * BLOCK_O + tl.arange(0, BLOCK_O)
        mask_o = offs_o < out_dim

        acc = tl.zeros([BLOCK_R, BLOCK_O], dtype=tl.float32)

        for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
            offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
            mask_d = offs_d < dim

        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # column offsets for each slice
        col_left = offs_h
        col_right = H + offs_h
        col_lgate = 2 * H + offs_h
        col_rgate = 3 * H + offs_h
        col_ogate = 4 * H + offs_h

        # load raw tensors
        left_raw = tl.load(proj_ptr + offs_r[:, None] * (5 * H) + col_left[None, :],
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0).to(tl.float16)
        right_raw = tl.load(proj_ptr + offs_r[:, None] * (5 * H) + col_right[None, :],
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0).to(tl.float16)
        left_gate_raw = tl.load(proj_ptr + offs_r[:, None] * (5 * H) + col_lgate[None, :],
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0).to(tl.float16)
        right_gate_raw = tl.load(proj_ptr + offs_r[:, None] * (5 * H) + col_rgate[None, :],
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0).to(tl.float16)
        out_gate_raw = tl.load(proj_ptr + offs_r[:, None] * (5 * H) + col_ogate[None, :],
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0).to(tl.float16)

        # sigmoid in fp32
        left_gate = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # apply mask and gates
        left = left_raw * mask_val * left_gate
        right = right_raw * mask_val * right_gate

        # store
        left_ptr + offs_r[:, None] * H + offs_h[None, :]  # compute pointer offset
        tl.store(left_ptr + offs_r[:, None] * H + offs_h[None, :],
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + offs_r[:, None] * H + offs_h[None, :],
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + offs_r[:, None] * H + offs_h[None, :],
                 out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])

# Kernel3: second LayerNorm + element‑wise gate multiplication
@triton.jit
def layernorm_mul_kernel(
    x_ptr, gate_ptr, y_ptr,
    weight_ptr, bias_ptr,
    rows, dim, eps,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # mean / variance
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # normalize, apply γ/β, multiply gate
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        # load x
        x_vals = tl.load(x_ptr + offs_m[:, None] * dim + offs_d[None, :],
                         mask=mask_m[:, None] & mask_d[None, :],
                         other=0.0).to(tl.float32)

        # normalize
        norm = (x_vals - mean[:, None]) * rstd[:, None]

        # γ and β
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0).to(tl.float32)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0).to(tl.float32)
        norm = norm * w[None, :] + b[None, :]

        # gate value
        gate = tl.load(gate_ptr + offs_m[:, None] * dim + offs_d[None, :],
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        out = norm * gate
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])

def custom_kernel(data):
    """
    Outgoing TriMul forward pass (no backward).
    Steps:
      1) Row‑wise LayerNorm on the input + fused linear projection
         (produces a tensor of shape [R,5*H] where R = B·N·N).
      2) Split the projected tensor into left, right and three gates,
         apply optional pair‑wise mask and sigmoid gates.
      3) Batched matrix‑multiply over the hidden dimension:
         out[b,i,j,k] = Σₗ left[b,i,l,k] * right[b,j,l,k].
      4) Row‑wise LayerNorm on the result and element‑wise multiplication
         with the output‑gate.
      5) Final linear projection back to `dim`.
    All heavy element‑wise work is done in Triton kernels; the two GEMMs
    (projection and final linear) are delegated to cuBLAS for maximal
    Tensor‑Core throughput.
    """
    # unpack
    x, mask, weights, cfg = data
    device = x.device
    eps = 1e-5

    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)

    B, N, _, _ = x.shape
    R = B * N * N
    H = hidden_dim
    out_dim = 5 * H

    # -----------------------------------------------------------------
    # 1️⃣  Row‑wise LayerNorm + fused projection
    # -----------------------------------------------------------------
    # flatten input
    x_flat = x.view(R, dim).contiguous()               # fp32
    proj = torch.empty(R, out_dim, dtype=torch.float16, device=device)

    # build transposed fused weight matrix: (dim,5*H)
    weight_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                 # (5H, dim)
    weight_T = weight_concat.t().contiguous()          # (dim, 5H)

    BLOCK_R = 64
    BLOCK_D = 32
    BLOCK_O = 64
    grid = (triton.cdiv(R, BLOCK_R),)
    layernorm_proj_kernel[grid](
        x_flat,
        proj,
        weight_T,
        R,
        dim,
        out_dim,
        eps,
        BLOCK_R=BLOCK_R,
        BLOCK_D=BLOCK_D,
        BLOCK_O=BLOCK_O,
        num_warps=4,
    )
    del x_flat, weight_concat, weight_T

    # -----------------------------------------------------------------
    # 2️⃣ Split, mask, sigmoid gates
    # -----------------------------------------------------------------
    left  = torch.empty(R, H, dtype=torch.float16, device=device)
    right = torch.empty(R, H, dtype=torch.float16, device=device)
    out_gate = torch.empty(R, H, dtype=torch.float16, device=device)

    if nomask or mask is None:
        HAS_MASK = 0
        dummy_mask = torch.empty(0, dtype=torch.float16, device=device)
        mask_ptr = dummy_mask
    else:
        HAS_MASK = 1
        mask_half = mask.to(torch.float16).reshape(R)
        mask_ptr = mask_half

    BLOCK_R2 = 64
    BLOCK_H = 32
    grid_split = (triton.cdiv(R, BLOCK_R2),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R2,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, mask_ptr

    # -----------------------------------------------------------------
    # 3️⃣ Tri‑multiplicative update (batched GEMM)
    # -----------------------------------------------------------------
    # reshape to [B, N, N, H] then permute to [B, H, N, N]
    left = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()

    left_mat = left.view(B * H, N, N)                    # (B*H, N, N)
    right_mat = right.view(B * H, N, N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B*H, N, N)

    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    del left, right, out_mat, left_mat, right_mat

    # -----------------------------------------------------------------
    # 4️⃣ Second LayerNorm + out‑gate multiplication (fused)
    # -----------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()
    out_scaled = torch.empty_like(out_flat, dtype=torch.float16, device=device)

    BLOCK_M2 = 64
    BLOCK_D2 = 32
    grid2 = (triton.cdiv(R, BLOCK_M2),)
    layernorm_mul_kernel[grid2](
        out_flat,
        out_gate.view(R, H),
        out_scaled,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M2,
        BLOCK_D=BLOCK_D2,
        num_warps=4,
    )
    del out, out_gate, out_flat

    # -----------------------------------------------------------------
    # 5️⃣ Final linear projection back to dim
    # -----------------------------------------------------------------
    final_half = F.linear(
        out_scaled,
        weights["to_out.weight"].to(torch.float16),   # (dim, H)
    )   # -> (R, dim) fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)

    return final

"""
Tri‑multiplicative (outgoing) update – high‑performance Triton + cuBLAS implementation.

Algorithm Overview
------------------
1️⃣ Row‑wise LayerNorm on the input tensor (FP32) **and** a fused linear
   projection (`W_fused`) that simultaneously produces the 5×hidden‑dim
   tensors needed for the left/right projections and their three gates.
   The fused kernel (`layernorm_proj_kernel`) reads each row twice
   (once to compute mean/var, a second time to compute the normalized
   values and the matrix‑vector product) and writes the result directly
   in FP16, eliminating the intermediate `x_norm` buffer.

2️⃣ A second Triton kernel (`split_fuse_gate_kernel`) splits the fused
   projection into `left`, `right` and `out_gate`, applies the optional
   pair‑wise mask and the sigmoid gates – all in FP16 (sigmoid computed
   in FP32 for stability).

3️⃣ Batched GEMM (cuBLAS) computes the core tri‑multiplicative update:
   `out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]`.
   Implemented as `torch.bmm` on tensors reshaped to (B·H,N,N).

4️⃣ Row‑wise LayerNorm on the result + element‑wise multiplication with
   `out_gate` in a fused kernel (`layernorm_mul_kernel`).

5️⃣ Final linear projection back to the original channel dimension using
   cuBLAS (`torch.nn.functional.linear`).  The final tensor is converted
   to FP32 before being returned.

Only the three element‑wise kernels are written in Triton; the two
large GEMMs are delegated to cuBLAS (Tensor‑Core) for maximal throughput.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Kernel 1: Row‑wise LayerNorm + fused projection (Wᵀ : dim → 5·H)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_proj_kernel(
    x_ptr,            # const   *[rows, dim]          (float32)
    proj_ptr,         # mutable *[rows, out_dim]     (float16)
    weight_ptr,       # const   *[dim, out_dim]      (float16)  = (W_fused)ᵀ
    rows: tl.int32,
    dim: tl.int32,
    out_dim: tl.int32,   # = 5 * hidden_dim
    eps: tl.float32,
    BLOCK_R: tl.constexpr,
    BLOCK_D: tl.constexpr,
    BLOCK_O: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < rows

    # --------------------------------------------------------------
    # 1️⃣ Mean & variance (accumulate in fp32)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_R], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        x_ptrs = x_ptr + offs_r[:, None] * dim + offs_d[None, :]
        vals = tl.load(x_ptrs,
                       mask=mask_r[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------------------------------------------------------------
    # 2️⃣ Projection: y = (x‑μ)·rstd @ Wᵀ   (output size = out_dim)
    # --------------------------------------------------------------
    for o_off in range(0, tl.cdiv(out_dim, BLOCK_O)):
        offs_o = o_off * BLOCK_O + tl.arange(0, BLOCK_O)   # [BLOCK_O]
        mask_o = offs_o < out_dim
        acc = tl.zeros([BLOCK_R, BLOCK_O], dtype=tl.float32)

        for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
            offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
            mask_d = offs_d < dim

            # load and normalize a slice of x
            x_ptrs = x_ptr + offs_r[:, None] * dim + offs_d[None, :]
            x_vals = tl.load(x_ptrs,
                             mask=mask_r[:, None] & mask_d[None, :],
                             other=0.0).to(tl.float32)
            x_norm = (x_vals - mean[:, None]) * rstd[:, None]   # [R,D]

            # load the corresponding block of Wᵀ : shape (BLOCK_D, BLOCK_O)
            w_ptrs = weight_ptr + offs_d[:, None] * out_dim + offs_o[None, :]
            w_vals = tl.load(w_ptrs,
                             mask=mask_d[:, None] & mask_o[None, :],
                             other=0.0).to(tl.float32)

            # accumulate dot‑product
            acc += tl.dot(x_norm, w_vals)   # (BLOCK_R, BLOCK_O)

        out_ptr = proj_ptr + offs_r[:, None] * out_dim + offs_o[None, :]
        tl.store(out_ptr, acc.to(tl.float16),
                 mask=mask_r[:, None] & mask_o[None, :])

                 mask=mask_r[:, None] & mask_h[None, :])

# Kernel 3: Second LayerNorm + element‑wise multiplication with out_gate

    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,

    BLOCK_D: tl.constexpr,

    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # 1️⃣ Compute mean / variance (fp32)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]

                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # 2️⃣ Normalize, apply γ/β, multiply with gate, store (fp16)
    # --------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        # load feature slice
        x_vals = tl.load(x_ptr + offs_m[:, None] * dim + offs_d[None, :],
                         mask=mask_m[:, None] & mask_d[None, :],
                         other=0.0).to(tl.float32)

        # (x‑μ)·rstd
        norm = (x_vals - mean[:, None]) * rstd[:, None]  # fp32

        # γ / β
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0).to(tl.float32)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0).to(tl.float32)
        norm = norm * w[None, :] + b[None, :]

        # gate value (out_gate)
        gate = tl.load(gate_ptr + offs_m[:, None] * dim + offs_d[None, :],
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        out = norm * gate
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul operator.

    Arguments
    ---------
    data = (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : mapping of all Linear/LayerNorm parameters
                         (float32 tensors, bias=False for Linear)
        - config_dict  : must contain "dim", "hidden_dim" and optionally "nomask"

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # -----------------------------------------------------------------
    # Unpack arguments
    # -----------------------------------------------------------------
    x, mask, w, cfg = data
    device = x.device
    eps = 1e-5

    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)

    B, N, _, _ = x.shape
    R = B * N * N               # total number of rows for row‑wise ops
    H = hidden_dim
    out_dim = 5 * H

    # -----------------------------------------------------------------
    # 1️⃣ LayerNorm + fused projection (produces [R,5*H] tensor)
    # -----------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()                     # FP32, [R,dim]
    proj = torch.empty(R, out_dim, dtype=torch.float16, device=device)

    # concatenate the five projection / gate weight matrices and transpose
    weight_concat = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # (5*H, dim)
    weight_T = weight_concat.t().contiguous()   # (dim, 5*H)

    BLOCK_R = 64
    BLOCK_D = 32
    BLOCK_O = 64
    grid = (triton.cdiv(R, BLOCK_R),)
    layernorm_proj_kernel[grid](
        x_flat,
        proj,
        weight_T,
        R,
        dim,
        out_dim,
        eps,
        BLOCK_R=BLOCK_R,
        BLOCK_D=BLOCK_D,
        BLOCK_O=BLOCK_O,
        num_warps=4,
    )
    del x_flat, weight_concat, weight_T

    # -----------------------------------------------------------------
    # 2️⃣ Split, apply mask and sigmoid gates
    # -----------------------------------------------------------------
    left  = torch.empty(R, H, dtype=torch.float16, device=device)
    right = torch.empty(R, H, dtype=torch.float16, device=device)
    out_gate = torch.empty(R, H, dtype=torch.float16, device=device)

    if nomask or mask is None:
        HAS_MASK = 0
        dummy_mask = torch.empty(0, dtype=torch.float16, device=device)
        mask_ptr = dummy_mask
    else:
        HAS_MASK = 1
        mask_ptr = mask.to(torch.float16).reshape(R)

    BLOCK_R2 = 64
    BLOCK_H = 32
    grid2 = (triton.cdiv(R, BLOCK_R2),)
    split_fuse_gate_kernel[grid2](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R2,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, mask_ptr

    # -----------------------------------------------------------------
    # 3️⃣ Tri‑multiplicative update (batched GEMM over hidden dim)
    # -----------------------------------------------------------------
    # reshape to [B, N, N, H] and bring H into batch axis
    left = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    left_mat = left.view(B * H, N, N)      # [B·H, N, N]
    right_mat = right.view(B * H, N, N)    # [B·H, N, N]

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))      # [B·H, N, N]

    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]
    del left, right, left_mat, right_mat, out_mat

    # -----------------------------------------------------------------
    # 4️⃣ Second LayerNorm + multiplication with out_gate
    # -----------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()
    out_fused = torch.empty_like(out_flat, dtype=torch.float16, device=device)

    BLOCK_M = 64
    BLOCK_D2 = 32
    grid3 = (triton.cdiv(R, BLOCK_M),)
    layernorm_mul_kernel[grid3](
        out_flat,
        out_gate.view(R, H),
        out_fused,
        w["to_out_norm.weight"],   # γ (float32)
        w["to_out_norm.bias"],     # β (float32)
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D2,
        num_warps=4,
    )
    del out, out_gate, out_flat

    # -----------------------------------------------------------------
    # 5️⃣ Final linear projection back to `dim`
    # -----------------------------------------------------------------
    final_half = F.linear(
        out_fused,
        w["to_out.weight"].to(torch.float16),   # (dim, H)
    )   # fp16, shape [R, dim]

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
