"""
High‑performance outgoing TriMul (AlphaFold3) forward pass.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the flattened input (FP32 → FP16) using a
   custom Triton kernel (`layernorm_kernel`).  The kernel computes the
   mean/variance in FP32, applies the learned γ/β, and writes the
   normalized tensor in FP16.

2️⃣ Fuse the four linear projections (left/right + their gates) into a
   single GEMM:  X_norm @ [W_left; W_right; W_left_gate; W_right_gate]ᵀ.
   The result is split into `left_raw`, `right_raw`, `left_gate_raw`,
   `right_gate_raw`.

3️⃣ Compute the output gate with a second GEMM:
   `out_gate_raw = X_norm @ W_out_gateᵀ`.

4️⃣ Optional binary mask (broadcasted over the hidden dimension) is
   applied to `left_raw` and `right_raw`.  Sigmoid gates are taken in‑place.

5️⃣ Cubic multiplicative update:
   for each head `h` compute `L_h @ R_hᵀ`.  This is realised as a
   strided‑batched GEMM of shape `(B·H, N, N)` where `H = hidden_dim`.

6️⃣ Second LayerNorm (including the output‑gate) is fused into a single
   Triton kernel (`layernorm_mul_kernel`).  It normalises `out`,
   applies the learned γ/β, then multiplies element‑wise with the
   previously computed `out_gate`.

7️⃣ Final linear projection back to `dim` and cast to FP32.

Only the two LayerNorm kernels are hand‑written; all other
heavy‑weight operations leverage highly‑optimised PyTorch fused GEMMs.
The implementation works for batch size 1‑2, sequence length up to 1024,
`dim ∈ {128,384,768}` and `hidden_dim = 128` on an H100 GPU.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton row‑wise LayerNorm (fp32 → fp16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const  * [total_rows, dim] (fp32)
    y_ptr,          # mutable* [total_rows, dim] (fp16)
    w_ptr,          # const  * [dim]          (fp32) – γ
    b_ptr,          # const  * [dim]          (fp32) – β
    total_rows,     # int (runtime)
    dim,            # int (runtime)
    eps,            # float (runtime)
    BLOCK_M: tl.constexpr,  # rows per program
    BLOCK_D: tl.constexpr,  # cols per tile
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # --------------------------------------------------------------
    # 1️⃣ mean & variance (fp32 accumulation)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # fp32 load
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)        # 1 / sqrt(var+eps)

    # --------------------------------------------------------------
    # 2️⃣ write normalized + γ/β (cast to fp16)
    # --------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]   # (x‑μ)/σ

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # γ (fp32)
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # β (fp32)

        out = norm * w[None, :] + b[None, :]                 # fp32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

    gate_ptr,     # const  * [total_rows, dim] (fp16)
    y_ptr,        # mutable* [total_rows, dim] (fp16)
    w_ptr,        # const  * [dim]          (fp32) – γ
    b_ptr,        # const  * [dim]          (fp32) – β
    total_rows,   # int (runtime)
    dim,          # int (runtime)
    eps,          # float (runtime)
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):

    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # --------------------------------------------------------------
    # 1️⃣ mean & variance (fp32)
    # --------------------------------------------------------------

    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    rstd = tl.rsqrt(var + eps)

    # --------------------------------------------------------------
    # 2️⃣ normalized, γ/β, then multiply with gate
    # --------------------------------------------------------------

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : dict with keys
              "norm.weight", "norm.bias",
              "left_proj.weight", "right_proj.weight",
              "left_gate.weight", "right_gate.weight",
              "out_gate.weight", "to_out_norm.weight",
              "to_out_norm.bias", "to_out.weight"
          All tensors are float32.
        - config_dict  : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # 0️⃣ unpack & constants
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    total_rows = B * N * N          # R = B·N·N
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (fp32 → fp16) via Triton
    # ------------------------------------------------------------------
    x_flat = x.view(total_rows, dim).contiguous()   # [R, dim] fp32
    x_norm_fp16 = torch.empty(total_rows, dim,
                              dtype=torch.float16,
                              device=device)

    BLOCK_M = 128   # rows per program – tuned for H100
    BLOCK_D = 64    # columns per tile
    grid = (triton.cdiv(total_rows, BLOCK_M),)

    layernorm_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],   # γ (fp32)
        weights["norm.bias"],     # β (fp32)
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fuse left/right projections & their gates into a single GEMM
    # ------------------------------------------------------------------
    # Concatenate weight matrices: [4·H, dim] (FP16)
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)                     # shape [4·H, dim]

    fused = F.linear(x_norm_fp16, w_fused)   # [R, 4·H] (FP16)

    left_raw       = fused[:, 0:H]
    right_raw      = fused[:, H:2*H]
    left_gate_raw  = fused[:, 2*H:3*H]
    right_gate_raw = fused[:, 3*H:4*H]

    # ------------------------------------------------------------------
    # 3️⃣ Output‑gate (separate GEMM)
    # ------------------------------------------------------------------
    out_gate_raw = F.linear(
        x_norm_fp16,
        weights["out_gate.weight"].to(torch.float16)   # [H, dim]
    )   # [R, H] (FP16)

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask & sigmoid gates (in‑place)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask: [B, N, N] → [R, 1] → broadcast over hidden dim
        mask_fp16 = mask.reshape(total_rows, 1).to(torch.float16)
        left_raw  = left_raw * mask_fp16
        right_raw = right_raw * mask_fp16

    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    left  = left_raw * left_gate      # [R, H] (FP16)
    right = right_raw * right_gate    # [R, H] (FP16)
    # out_gate stays as FP16 for later use

    # ------------------------------------------------------------------
    # 5️⃣ Batched cubic update (left @ rightᵀ) – FP16 GEMM
    # ------------------------------------------------------------------
    # reshape to [B, N, N, H]
    left  = left.view(B, N, N, H)
    right = right.view(B, N, N, H)

    # bring H into batch dimension -> [B·H, N, N]
    left_perm  = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    left_mat   = left_perm.view(B * H, N, N)
    right_mat  = right_perm.view(B * H, N, N)

    # (N×N) @ (N×N)ᵀ  →  [B·H, N, N]  (Tensor‑Core FP16)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back → [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm + output‑gate (fused Triton kernel)
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, H).contiguous()          # [R, H] (FP16)
    out_gate_flat = out_gate.view(total_rows, H).contiguous()# [R, H] (FP16)

    out_preproj_fp16 = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_mul_kernel[grid2](
        out_flat,
        out_gate_flat,
        out_preproj_fp16,
        weights["to_out_norm.weight"],   # γ (fp32)
        weights["to_out_norm.bias"],     # β (fp32)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 7️⃣ Final linear projection & cast back to FP32
    # ------------------------------------------------------------------
    final_fp16 = F.linear(
        out_preproj_fp16,
        weights["to_out.weight"].to(torch.float16)   # [dim, H]
    )   # [R, dim] (FP16)

    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final
```
