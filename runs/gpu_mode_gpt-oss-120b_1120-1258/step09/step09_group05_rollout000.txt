"""
TriMul (outgoing) – mixed‑precision Triton‑accelerated forward pass.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the flattened 4‑D tensor (FP32 → FP16) using a
   custom Triton kernel (no synchronisation across rows).

2️⃣ Fuse the four linear projections (left/right + their gates) into one
   FP16 GEMM:
        X_norm @ [W_left; W_right; W_left_gate; W_right_gate]ᵀ  →  [R, 4·H]
   where R = B·N·N and H = hidden_dim.  Split the result into the four
   components.

3️⃣ Compute the output‑gate with a second FP16 GEMM.

4️⃣ (Optional) Apply the binary mask to the raw left/right tensors and
   apply sigmoid to all three gate tensors (in‑place to save memory).

5️⃣ Reshape left/right to [B, H, N, N] and perform the cubic multiplicative
   update as a batched matrix multiplication:
        out[b·H] = left[b·H] @ right[b·H]ᵀ      (Tensor‑Core FP16).

6️⃣ Second LayerNorm over the hidden dimension (FP16) using the same
   Triton kernel.

7️⃣ Multiply by the output‑gate, apply the final FP16 linear projection and
   cast back to FP32.

The implementation works for any
    B ∈ {1,2},   N ∈ {128…1024},   dim ∈ {128,384,768},   hidden_dim = 128
and supports both masked and un‑masked inputs.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (fp32 → fp16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const  * [total_rows, dim]  (fp32)
    y_ptr,          # mutable* [total_rows, dim]  (fp16)
    w_ptr,          # const  * [dim]               (fp32)  – γ
    b_ptr,          # const  * [dim]               (fp32)  – β
    total_rows,     # int (runtime)
    dim,            # int (runtime)
    eps,            # compile‑time float
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per tile
):
    """
    Row‑wise LayerNorm.
    y = ((x - μ) / sqrt(σ² + eps)) * γ + β
    The kernel processes BLOCK_M rows at a time and iterates over the
    channel dimension in BLOCK_D‑wide tiles.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # --------------------------------------------------------------
    # 1. Compute mean and variance for each row
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # fp32 load
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                            # 1 / sqrt(var+eps)

    # --------------------------------------------------------------
    # 2. Write normalized result (fp16)
    # --------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]     # (x-μ)/σ

        # γ and β are fp32 – keep computation in fp32, cast at the end
        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)
        out = norm * w[None, :] + b[None, :]              # fp32

        out_fp16 = out.to(tl.float16)
        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom Triton‑accelerated TriMul kernel entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass (outgoing) of the TriMul operator.

    Parameters
    ----------
    data : tuple
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (bool/float) or None
        - weights_dict : dict of torch.Parameters (float32) with keys:
            "norm.weight", "norm.bias",
            "left_proj.weight", "right_proj.weight",
            "left_gate.weight", "right_gate.weight",
            "out_gate.weight",
            "to_out_norm.weight", "to_out_norm.bias",
            "to_out.weight"
        - config_dict  : dict with keys:
            "dim"         – input channel dimension
            "hidden_dim" – hidden dimension (H)
            "nomask"      – bool, True → ignore mask

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # 0️⃣ unpack & constants
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    total_rows = B * N * N                 # R = B·N·N
    H = hidden_dim                         # hidden dimension (H)

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) – fp32 → fp16
    # ------------------------------------------------------------------
    # Flatten to [R, dim]
    x_flat = x.view(total_rows, dim).contiguous()
    # Output buffer (fp16)
    x_norm_fp16 = torch.empty(total_rows, dim,
                              dtype=torch.float16,
                              device=device)

    # Tuned block sizes for H100 (fits in shared memory, good occupancy)
    BLOCK_M = 256       # rows per program
    BLOCK_D = 64        # columns per tile
    grid = (triton.cdiv(total_rows, BLOCK_M),)

    layernorm_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],      # γ (float32)
        weights["norm.bias"],        # β (float32)
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fuse left/right projections + gates into a single FP16 GEMM
    # ------------------------------------------------------------------
    # Concatenate weights → [4·H, dim] (FP16)
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)                     # -> FP16

    # Single GEMM: X_norm @ W_fusedᵀ → [R, 4·H]  (FP16)
    fused = F.linear(x_norm_fp16, w_fused)  # (fp16)

    left_raw        = fused[:, 0:H]
    right_raw       = fused[:, H:2 * H]
    left_gate_raw   = fused[:, 2 * H:3 * H]
    right_gate_raw  = fused[:, 3 * H:4 * H]

    # ------------------------------------------------------------------
    # 3️⃣ Output‑gate (separate GEMM)
    # ------------------------------------------------------------------
    out_gate_raw = F.linear(
        x_norm_fp16,
        weights["out_gate.weight"].to(torch.float16)   # [H, dim]
    )   # [R, H] fp16

    # ------------------------------------------------------------------
    # 4️⃣ Apply mask (if any) and sigmoid gates
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask: [B, N, N] → [R, 1] (fp16) for broadcasting
        mask_fp16 = mask.reshape(total_rows, 1).to(torch.float16)
        left_raw  = left_raw * mask_fp16
        right_raw = right_raw * mask_fp16

    # In‑place sigmoid to keep memory usage low
    left_gate = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate = out_gate_raw.sigmoid_()

    # Apply gates
    left  = left_raw  * left_gate   # [R, H] fp16
    right = right_raw * right_gate  # [R, H] fp16

    # ------------------------------------------------------------------
    # 5️⃣ Reshape for cubic update & batched matmul
    # ------------------------------------------------------------------
    # Shape back to [B, N, N, H]
    left = left.view(B, N, N, H)
    right = right.view(B, N, N, H)
    out_gate = out_gate.view(B, N, N, H)

    # Move H to the batch dimension → [B, H, N, N]
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()

    # Collapse batch & head → [B·H, N, N]
    left_mat = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)

    # Batched matmul: (N×N) @ (N×N)ᵀ   →   [B·H, N, N]  (FP16, Tensor‑Core)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Restore original layout → [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm over the hidden dimension (FP16)
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, H).contiguous()
    out_norm_fp16 = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 7️⃣ Apply output‑gate, final projection, cast to FP32
    # ------------------------------------------------------------------
    # Element‑wise gating (still FP16)
    out_norm_fp16 = out_norm_fp16 * out_gate.view(total_rows, H)

    # Final linear projection: [R, H] @ [dim, H]ᵀ → [R, dim] (FP16)
    final_fp16 = F.linear(
        out_norm_fp16,
        weights["to_out.weight"].to(torch.float16)   # [dim, H]
    )   # (fp16)

    # Reshape & cast back to FP32
    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final
