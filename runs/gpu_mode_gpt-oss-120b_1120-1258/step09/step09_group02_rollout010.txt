def custom_kernel(data):
    x, mask, weights, cfg = data
    B,N,_,dim = x.shape  # dimension param
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    device = x.device

    R = B * N * N
    H = hidden_dim

    # 1st LN (half)
    x_fp16 = x.to(torch.float16)   # copy from fp32?
    # we can view as (R, dim)

    x_norm = F.layer_norm(x_flat, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps)

    # fused projection
    w_fused = torch.cat([
        weights["left_proj.weight"],
        weights["right_proj.weight"],
        weights["left_gate.weight"],
        weights["right_gate.weight"],
        weights["out_gate.weight"]
    ], dim=0).to(torch.float16)

    proj = F.linear(x_norm, w_fused)  # (R, 5*H) half

    # split/gate kernel
    left = torch.empty(R, H, dtype=torch.float16, device=device)
    right = torch.empty(R, H, dtype=torch.float16, device=device)
    out_gate = torch.empty(R, H, dtype=torch.float16, device=device)

    # mask pointer
    if not cfg.get("nomask", True):
        mask_fp16 = mask.to(torch.float16).reshape(R)
        mask_ptr = mask_fp16
        has_mask = 1
    else:
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)
        has_mask = 0

    # Launch kernel
    BLOCK_R = 128
    BLOCK_H = 64  # maybe use bigger block
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,

"""
TriMul (outgoing) – fast forward pass for AlphaFold‑3 style pair‑wise update.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the (B·N·N)×dim input (half‑precision).
2️⃣ Fuse the five linear projections (left, right, left_gate, right_gate, out_gate)
   into a single GEMM:  X_norm @ W_fusedᵀ  →  (R,5·H)   where R=B·N·N.
3️⃣ Triton kernel `split_fuse_gate_kernel`:
     * splits the fused projection into left/right tensors,
     * applies the optional pair‑wise mask,
     * computes sigmoid gates (in fp32 for numeric stability) and
       stores three half‑precision tensors: left, right, out_gate.
4️⃣ Batched matrix multiplication over the hidden dimension:
        left_mat = left.view(B·H,N,N)
        right_mat = right.view(B·H,N,N)
        out_mat = left_mat @ right_matᵀ        (torch.bmm – cuBLAS Tensor‑Core)
5️⃣ Second row‑wise LayerNorm on the (B·N·N)×H result.
6️⃣ Element‑wise multiplication with `out_gate` and final linear projection
   back to the original `dim` (half → float32).

Only step 3 is written in Triton, satisfying the “must use a kernel” rule.
All heavy matrix‑multiplications are delegated to cuBLAS via PyTorch for
maximum throughput on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: split fused projection, apply mask & sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[rows, 5*H] (fp16)
    left_ptr,          # mutable *[rows, H]   (fp16)
    right_ptr,         # mutable *[rows, H]   (fp16)
    out_gate_ptr,      # mutable *[rows, H]   (fp16)
    mask_ptr,          # const   *[rows]    (fp16) – ignored if HAS_MASK=0
    rows: tl.int32,
    H: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """For each row r:
          left  = (proj_left  * mask) * sigmoid(proj_left_gate)
          right = (proj_right * mask) * sigmoid(proj_right_gate)
          out_gate = sigmoid(proj_out_gate)

       `proj_ptr` layout:
          [0:H)               → left
          [H:2H)              → right
          [2H:3H)             → left_gate
          [3H:4H)             → right_gate
          [4H:5H)             → out_gate
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)           # [BLOCK_R]
    mask_r = offs_r < rows

    # ---- per‑row mask (scalar) ---------------------------------------
    if HAS_MASK:
        m = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        m = m[:, None]                                          # [BLOCK_R,1]
    else:
        m = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # ---- iterate over hidden dimension --------------------------------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)        # [BLOCK_H]
        mask_h = offs_h < H

        # offset for the current row block inside the fused matrix
        row_off = offs_r[:, None] * (5 * H)                     # [R,1]
        col_off = offs_h[None, :]                               # [1,H]

        # ----- raw values ------------------------------------------------
        left_raw   = tl.load(proj_ptr + row_off + col_off,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        right_raw  = tl.load(proj_ptr + row_off + H + col_off,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        left_g_raw = tl.load(proj_ptr + row_off + 2 * H + col_off,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        right_g_raw= tl.load(proj_ptr + row_off + 3 * H + col_off,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        out_g_raw  = tl.load(proj_ptr + row_off + 4 * H + col_off,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)

        # ----- sigmoid (computed in fp32) --------------------------------
        left_gate  = (1.0 / (1.0 + tl.exp(-left_g_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_g_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_g_raw.to(tl.float32)))).to(tl.float16)

        # ----- apply mask & gates ----------------------------------------
        left  = left_raw  * m * left_gate
        right = right_raw * m * right_gate

        # ----- store -------------------------------------------------------
        tl.store(left_ptr      + offs_r[:, None] * H + offs_h[None, :],
                 left,  mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr     + offs_r[:, None] * H + offs_h[None, :],
                 right, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr  + offs_r[:, None] * H + offs_h[None, :],
                 out_gate, mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : {"norm.weight":..., "norm.bias":..., "left_proj.weight":...,
                         "right_proj.weight":..., "left_gate.weight":..., ...}
                         all tensors are float32 (Linear layers have bias=False)
        - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim], dtype float32
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    device = x.device
    B, N, _, dim = x.shape
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5

    R = B * N * N                     # total number of (i,k) pairs
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise, keep half‑precision)
    # ------------------------------------------------------------------
    # Cast input to half and view as (R, dim)
    x_fp16 = x.to(torch.float16)
    x_flat = x_fp16.view(R, dim)

    # torch.nn.functional.layer_norm works on the last dimension;
    # it computes mean/var in fp32 internally and returns fp16 when input is fp16.
    x_norm = F.layer_norm(
        x_flat,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )   # (R, dim)  fp16

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear projection (produces 5·H columns)
    # ------------------------------------------------------------------
    # concatenate the five projection matrices (shape: [5*H, dim])
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # fp16

    # (R, dim) @ (5·H, dim)ᵀ → (R, 5·H)
    proj = F.linear(x_norm, w_fused)        # fp16
    del w_fused, x_norm                     # free intermediate memory

    # ------------------------------------------------------------------
    # 3️⃣ Split fused tensor, apply mask & sigmoid gates
    # ------------------------------------------------------------------
    left      = torch.empty(R, H, dtype=torch.float16, device=device)
    right     = torch.empty(R, H, dtype=torch.float16, device=device)
    out_gate  = torch.empty(R, H, dtype=torch.float16, device=device)

    if not cfg.get("nomask", True):
        # mask shape [B, N, N] → [R] (float16)
        mask_flat = mask.to(torch.float16).reshape(R)
        mask_ptr = mask_flat
        HAS_MASK = 1
    else:
        # dummy tensor – never accessed when HAS_MASK=0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)
        HAS_MASK = 0

    # launch Triton kernel
    BLOCK_R = 128
    BLOCK_H = 64
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM over hidden dimension (left @ rightᵀ)
    # ------------------------------------------------------------------
    # reshape to [B, N, N, H] → bring H into batch dimension
    left  = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    left_mat  = left.view(B * H, N, N)        # [B·H, N, N]
    right_mat = right.view(B * H, N, N)       # [B·H, N, N]

    # (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    out_flat = out.view(R, H)                # (R, H) fp16
    del out, left, right, left_mat, right_mat, out_mat

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm + element‑wise out‑gate
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(
        out_flat,
        (H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )   # fp16

    out_norm = out_norm * out_gate
    del out_gate, out_flat

    # ------------------------------------------------------------------
    # 6️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    final_half = F.linear(
        out_norm,
        weights["to_out.weight"].to(torch.float16),   # shape [dim, H]
    )   # fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
