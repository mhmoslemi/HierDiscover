@triton.jit
def final_norm_gate_linear_kernel(
    out_ptr,          # const *[rows, H] (fp16)
    gate_ptr,         # const *[rows, H] (fp16)
    ln_weight_ptr,    # const *[H] (fp32)
    ln_bias_ptr,      # const *[H] (fp32)
    weight_out_ptr,   # const *[H, dim] (fp16)   # pre-transposed weight
    out_ptr_final,    # mutable *[rows, dim] (fp32)
    rows: tl.int32,
    H: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid_row = tl.program_id(0)
    pid_dim = tl.program_id(1)
    row_start = pid_row * BLOCK_R
    dim_start = pid_dim * BLOCK_D

    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < rows
    offs_d = dim_start + tl.arange(0, BLOCK_D)
    mask_d = offs_d < dim

    # -------------------------------------------------
    # 1️⃣ Compute mean & variance (fp32) per row
    # -------------------------------------------------
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_R], dtype=tl.float32)

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H
        ptr = out_ptr + offs_r[:, None] * H + offs_h[None, :]
        vals = tl.load(ptr,
                       mask=mask_r[:, None] & mask_h[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # -------------------------------------------------
    # 2️⃣ Accumulate output across hidden dim
    # -------------------------------------------------
    acc = tl.zeros([BLOCK_R, BLOCK_D], dtype=tl.float32)

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # load out and gate
        ptr_out = out_ptr + offs_r[:, None] * H + offs_h[None, :]
        out_vals = tl.load(ptr_out,
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0).to(tl.float32)

        ptr_gate = gate_ptr + offs_r[:, None] * H + offs_h[None, :]
        gate_vals = tl.load(ptr_gate,
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0).to(tl.float32)

        # layernorm scaling
        ln_w = tl.load(ln_weight_ptr + offs_h, mask=mask_h, other=0.0)  # fp32
        ln_b = tl.load(ln_bias_ptr   + offs_h, mask=mask_h, other=0.0)  # fp32

        # normalize, apply LN weight/bias, then gate
        normed = (out_vals - mean[:, None]) * rstd[:, None]          # fp32
        weighted = normed * ln_w[None, :] + ln_b[None, :]           # fp32
        gated = weighted * gate_vals                                 # fp32

        # load output weight slice [H, dim]  (transposed)
        # weight_out_ptr points to weight_out_T (H, dim)
        ptr_w = weight_out_ptr + offs_h[:, None] * dim + offs_d[None, :]  # [BLOCK_H, BLOCK_D]
        w_vals = tl.load(ptr_w,
                         mask=mask_h[:, None] & mask_d[None, :],
                         other=0.0).to(tl.float32)                       # fp32

        # accumulate product across H
        acc += tl.dot(gated, w_vals)   # (BLOCK_R, BLOCK_H) dot (BLOCK_H, BLOCK_D) -> (BLOCK_R, BLOCK_D)

    # -------------------------------------------------
    # 3️⃣ Store result
    # -------------------------------------------------
    out_ptr_final = out_ptr_final + offs_r[:, None] * dim + offs_d[None, :]
    tl.store(out_ptr_final,
             acc,
             mask=mask_r[:, None] & mask_d[None, :])
```

Note: weight_out_T is of shape (H, dim). For each h block we load contiguous slice of dim. offsets: offset = h * dim + d.

import triton.language as tl


# ----------------------------------------------------------------------
# Kernel 1: row‑wise LayerNorm (fp32 → fp16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   *[rows, dim]   (float32)
    y_ptr,          # mutable *[rows, dim]   (float16)
    w_ptr,          # const   *[dim]        (float32)   γ
    b_ptr,          # const   *[dim]        (float32)   β
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,   # rows processed per program
    BLOCK_D: tl.constexpr,   # columns processed per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ---------- mean & variance (fp32) ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]   # [M, D]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)               # fp32
        sum_val += tl.sum(vals, axis=1)                     # [M]
        sum_sq  += tl.sum(vals * vals, axis=1)              # [M]

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- normalize, apply γ/β, store fp16 ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]       # fp32

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0) # γ
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0) # β
        out = norm * w[None, :] + b[None, :]                # fp32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Kernel 2: split fused projection, apply mask & sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[rows, 5*H] (float16)
    left_ptr,          # mutable *[rows, H]   (float16)
    right_ptr,         # mutable *[rows, H]   (float16)
    out_gate_ptr,      # mutable *[rows, H]   (float16)
    mask_ptr,          # const   *[rows]    (float16)   (ignored if HAS_MASK=0)
    rows: tl.int32,
    H: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    For each row r:
        left      = (proj_left  * mask) * sigmoid(proj_left_gate)
        right     = (proj_right * mask) * sigmoid(proj_right_gate)
        out_gate  = sigmoid(proj_out_gate)

    All ops are in fp16; sigmoid evaluated in fp32.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)               # [BLOCK_R]
    mask_r = offs_r < rows

    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                   # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)          # [BLOCK_H]
        mask_h = offs_h < H

        # fused layout   0..H-1 -> left
        #                 H..2H-1 -> right
        #                 2H..3H-1 -> left_gate
        #                 3H..4H-1 -> right_gate
        #                 4H..5H-1 -> out_gate
        row_offset = offs_r[:, None] * (5 * H)                   # [R,1]
        col_offset = offs_h[None, :]                             # [1,H]

        left_raw   = tl.load(proj_ptr + row_offset + col_offset,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        right_raw  = tl.load(proj_ptr + row_offset + H + col_offset,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        left_g_raw = tl.load(proj_ptr + row_offset + 2 * H + col_offset,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        right_g_raw= tl.load(proj_ptr + row_offset + 3 * H + col_offset,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        out_g_raw  = tl.load(proj_ptr + row_offset + 4 * H + col_offset,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)

        # sigmoid in fp32 for stability
        left_gate  = (1.0 / (1.0 + tl.exp(-left_g_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_g_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_g_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate

        tl.store(left_ptr    + offs_r[:, None] * H + offs_h[None, :],
                 left,  mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr   + offs_r[:, None] * H + offs_h[None, :],
                 right, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr+ offs_r[:, None] * H + offs_h[None, :],
                 out_gate, mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Kernel 3: second LayerNorm + out‑gate multiplication + final linear
# ----------------------------------------------------------------------
@triton.jit
def final_norm_gate_linear_kernel(
    out_ptr,          # const   *[rows, H]   (fp16)   – result of the GEMM
    gate_ptr,         # const   *[rows, H]   (fp16)   – out_gate
    ln_w_ptr,         # const   *[H]        (fp32)   – to_out_norm.weight
    ln_b_ptr,         # const   *[H]        (fp32)   – to_out_norm.bias
    w_out_ptr,        # const   *[H, dim]   (fp16)   – transposed to_out.weight (H×dim)
    out_final_ptr,    # mutable *[rows, dim] (fp32)
    rows: tl.int32,
    H: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """
    For each flattened position r (r = b·N² + i·N + j):
        1️⃣ Compute row‑wise mean & variance of `out`.
        2️⃣ Normalize, apply LN weight/bias, then element‑wise multiply by `gate`.
        3️⃣ Project to `dim` via a fused GEMV (H → dim) using Tensor‑Core dot.
    The whole sequence is fused to avoid materialising intermediate tensors.
    """
    pid_row = tl.program_id(0)          # block over rows
    pid_dim = tl.program_id(1)          # block over output dimension
    row_start = pid_row * BLOCK_R
    dim_start = pid_dim * BLOCK_D

    offs_r = row_start + tl.arange(0, BLOCK_R)           # [BLOCK_R]
    mask_r = offs_r < rows
    offs_d = dim_start + tl.arange(0, BLOCK_D)           # [BLOCK_D]
    mask_d = offs_d < dim

    # ------------------------------------------------------------------
    # 1️⃣ mean & variance (fp32)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_R], dtype=tl.float32)

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H
        ptr = out_ptr + offs_r[:, None] * H + offs_h[None, :]
        vals = tl.load(ptr,
                       mask=mask_r[:, None] & mask_h[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------------------------------------------------------
    # 2️⃣ accumulate final projection (fp32)
    # ------------------------------------------------------------------
    acc = tl.zeros([BLOCK_R, BLOCK_D], dtype=tl.float32)

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # load raw out and gate values
        ptr_out = out_ptr + offs_r[:, None] * H + offs_h[None, :]
        out_vals = tl.load(ptr_out,
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0).to(tl.float32)

        ptr_gate = gate_ptr + offs_r[:, None] * H + offs_h[None, :]
        gate_vals = tl.load(ptr_gate,
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0).to(tl.float32)

        # LayerNorm weight / bias (per hidden dim)
        w_ln = tl.load(ln_w_ptr + offs_h, mask=mask_h, other=0.0)   # fp32
        b_ln = tl.load(ln_b_ptr + offs_h, mask=mask_h, other=0.0)   # fp32

        # normalize → apply LN γ/β → multiply by out_gate
        normed   = (out_vals - mean[:, None]) * rstd[:, None]               # fp32
        ln_applied = normed * w_ln[None, :] + b_ln[None, :]                 # fp32
        gated   = ln_applied * gate_vals                                     # fp32

        # load slice of final‑projection weight (H×dim, transposed)
        ptr_w = w_out_ptr + offs_h[:, None] * dim + offs_d[None, :]          # [BLOCK_H, BLOCK_D]
        w_vals = tl.load(ptr_w,
                         mask=mask_h[:, None] & mask_d[None, :],
                         other=0.0).to(tl.float32)                        # fp32

        # accumulate dot product over the hidden dimension
        acc += tl.dot(gated, w_vals)   # (BLOCK_R,BLOCK_H)·(BLOCK_H,BLOCK_D) → (BLOCK_R,BLOCK_D)

    # ------------------------------------------------------------------
    # 3️⃣ store result (fp32)
    # ------------------------------------------------------------------
    out_ptr_final = out_final_ptr + offs_r[:, None] * dim + offs_d[None, :]
    tl.store(out_ptr_final,
             acc,
             mask=mask_r[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (bool/float) or None
        - weights_dict : mapping with all linear / layer‑norm parameters
                         (float32 tensors, bias=False for Linear)
        - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                     # total rows for row‑wise ops
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (fp32 → fp16)
    # ------------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()                     # [R, dim] fp32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 256
    BLOCK_D = 32
    grid_ln1 = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln1](
        x_flat,
        x_norm,
        weights["norm.weight"],    # γ  (float32)
        weights["norm.bias"],      # β  (float32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection (left, right, gates)
    # ------------------------------------------------------------------
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # (5·H, dim)

    proj = F.linear(x_norm, w_fused)        # (R, 5·H)  fp16
    del w_fused, x_norm                     # free early

    # ------------------------------------------------------------------
    # 3️⃣ Split proj, apply mask, sigmoid‑gates
    # ------------------------------------------------------------------
    left      = torch.empty(R, H, dtype=torch.float16, device=device)
    right     = torch.empty(R, H, dtype=torch.float16, device=device)
    out_gate  = torch.empty(R, H, dtype=torch.float16, device=device)

    if cfg.get("nomask", True):
        # kernel will treat HAS_MASK=0
        dummy_mask = torch.empty(0, dtype=torch.float16, device=device)
        mask_ptr = dummy_mask
        has_mask = 0
    else:
        mask_flat = mask.to(torch.float16).reshape(R)   # [R]
        mask_ptr = mask_flat
        has_mask = 1

    BLOCK_R = 128
    BLOCK_H = 32
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Batched matrix multiplication (B·H) × (N×N) → (R, H)
    # ------------------------------------------------------------------
    # reshape to [B, N, N, H] → permute to [B, H, N, N] for cuBLAS
    left_mat = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_mat = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    left_mat = left_mat.view(B * H, N, N)      # [B·H, N, N]
    right_mat = right_mat.view(B * H, N, N)    # [B·H, N, N]

    out = torch.bmm(left_mat, right_mat.transpose(1, 2))      # (B·H, N, N)
    out = out.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)
    out_flat = out.view(R, H)                                      # (R, H)

    del left, right, left_mat, right_mat, out

    # ------------------------------------------------------------------
    # 5️⃣ Fuse second LayerNorm, out‑gate multiplication and final projection
    # ------------------------------------------------------------------
    # prepare transposed final linear weight (H × dim) for fast dot
    w_out_T = weights["to_out.weight"].t().contiguous().to(torch.float16)  # (H, dim)

    final = torch.empty(R, dim, dtype=torch.float32, device=device)

    BLOCK_D = 32
    grid_final = (triton.cdiv(R, BLOCK_R), triton.cdiv(dim, BLOCK_D))
    final_norm_gate_linear_kernel[grid_final](
        out_flat,                     # out (R, H) fp16
        out_gate,                     # gate (R, H) fp16
        weights["to_out_norm.weight"],   # LN γ (fp32)
        weights["to_out_norm.bias"],     # LN β (fp32)
        w_out_T,                      # weight_out_T (H, dim) fp16
        final,                        # output (R, dim) fp32
        R,
        H,
        dim,
        eps,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    del out_flat, out_gate, w_out_T

    # ------------------------------------------------------------------
    # 6️⃣ Reshape to the original layout
    # ------------------------------------------------------------------
    out_tensor = final.view(B, N, N, dim)          # [B, N, N, dim] (float32)
    return out_tensor
