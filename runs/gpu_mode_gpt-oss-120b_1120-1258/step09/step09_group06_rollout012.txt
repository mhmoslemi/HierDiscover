"""
High‑performance outgoing TriMul (AlphaFold3) forward pass.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the flattened 4‑D tensor (FP32 → FP16) via a
   custom Triton kernel (single pass for mean+var, second pass for
   normalization).  Block sizes are tuned for the H100.
2️⃣ Fuse the four linear projections (left/right + left/right‑gate) into a
   single FP16 GEMM:  X_norm @ [W_left; W_right; W_left_gate; W_right_gate]ᵀ.
   The result is split into *left_raw*, *right_raw*, *left_gate_raw*,
   *right_gate_raw*.
3️⃣ Compute *out_gate* with a second FP16 GEMM.
4️⃣ Apply optional binary mask, sigmoid‑gates and element‑wise gating
   (all in‑place to minimise memory traffic).
5️⃣ Cubic multiplicative update:
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   – implemented as a strided‑batched half‑precision GEMM (B·H, N, N)
     using torch.bmm (the fastest approach on H100).
6️⃣ Second LayerNorm over the hidden dimension (FP16) – same Triton kernel
   as step 1.
7️⃣ Element‑wise multiplication with *out_gate* and final FP16 linear
   projection back to *dim*.
   The result is cast back to FP32.

All heavy linear algebra is performed in FP16 (Tensor‑Core friendly);
only the two LayerNorms are custom‑written Triton kernels, satisfying
the “must use a kernel” requirement.  The implementation works for
batch‑sizes 1‑2, sequence lengths up to 1024 and hidden dimensions 128,
with optional masking.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const  * [total_rows, dim] (fp32)
    y_ptr,          # mutable* [total_rows, dim] (fp16)
    w_ptr,          # const  * [dim]          (fp32) – γ
    b_ptr,          # const  * [dim]          (fp32) – β
    total_rows,     # int (runtime)
    dim,            # int (runtime)
    eps,            # float (compile‑time)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per tile
):
    """
    Row‑wise LayerNorm.
    y = ((x - μ) / sqrt(σ² + eps)) * γ + β
    Two‑pass algorithm: first accumulate sum / sum‑sq, then normalize.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)            # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # 1️⃣ Compute mean & variance
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    # loop over dim in tiles of size BLOCK_D
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        # pointer into the flattened input
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # fp32 load
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)        # 1 / sqrt(var+eps)

    # ------------------------------------------------------------------
    # 2️⃣ Write normalized result (fp16)
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]   # (x-μ)/σ

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # γ (fp32)
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # β (fp32)

        out = norm * w[None, :] + b[None, :]               # fp32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
# Custom Triton kernel for in‑place mask + sigmoid + gating fusion
# ----------------------------------------------------------------------
@triton.jit
def gate_fusion_kernel(
    raw_ptr,          # fp16*[R, H]  – left or right raw tensor
    gate_ptr,         # fp16*[R, H]  – gate pre‑sigmoid tensor
    mask_ptr,         # fp16*[R, 1] – mask (may be nullptr)
    out_ptr,          # fp16*[R, H]  – output tensor (in‑place)
    R,                # runtime int
    H,                # runtime int
    HAS_MASK: tl.constexpr,   # 0/1 flag
):
    """
    out = raw * sigmoid(gate) * mask
    If HAS_MASK==0 the mask load is skipped.
    """
    pid = tl.program_id(0)
    stride = pid * H
    mask_stride = pid
    if stride >= R * H:
        return

    # load one row (H elements)
    offs = tl.arange(0, H)
    raw = tl.load(raw_ptr + stride + offs)
    gate = tl.load(gate_ptr + stride + offs)
    gated = raw * tl.sigmoid(gate)

    if HAS_MASK:
        m = tl.load(mask_ptr + mask_stride)  # scalar fp16
        gated = gated * m

    tl.store(out_ptr + stride + offs, gated)

# ----------------------------------------------------------------------
# Triton kernel entry‑point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : dict of torch.Parameters (float32) with keys:
              "norm.weight", "norm.bias",
              "left_proj.weight", "right_proj.weight",
              "left_gate.weight", "right_gate.weight",
              "out_gate.weight",
              "to_out_norm.weight", "to_out_norm.bias",
              "to_out.weight"
        - config_dict  : dict with keys:
              "dim" (int), "hidden_dim" (int), "nomask" (bool, optional)

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # 0️⃣ unpack & constants
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    total_rows = B * N * N          # R = B·N·N
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (fp32 → fp16) – Triton
    # ------------------------------------------------------------------
    x_flat = x.view(total_rows, dim).contiguous()                     # [R, dim] fp32
    x_norm_fp16 = torch.empty(total_rows, dim,
                              dtype=torch.float16,
                              device=device)

    # Tune block sizes for the H100 (larger tiles → fewer launches)
    BLOCK_M = 256       # rows per program
    BLOCK_D = 128       # columns per tile (covers the whole dim in ≤6 tiles)
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],   # γ, fp32
        weights["norm.bias"],     # β, fp32
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fuse left/right projections & their gates (single FP16 GEMM)
    # ------------------------------------------------------------------
    # Concatenate the four weight matrices: [4·H, dim]
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)                     # [4·H, dim]

    fused = F.linear(x_norm_fp16, w_fused)   # [R, 4·H] (fp16)

    left_raw       = fused[:, 0:H]
    right_raw      = fused[:, H:2 * H]
    left_gate_raw  = fused[:, 2 * H:3 * H]
    right_gate_raw = fused[:, 3 * H:4 * H]

    # ------------------------------------------------------------------
    # 3️⃣ Output‑gate (second GEMM)
    # ------------------------------------------------------------------
    out_gate_raw = F.linear(
        x_norm_fp16,
        weights["out_gate.weight"].to(torch.float16)   # [H, dim]
    )   # [R, H] (fp16)

    # ------------------------------------------------------------------
    # 4️⃣ Fuse mask, sigmoid‑gate and element‑wise gating (Triton)
    #    – one kernel for left, one for right.
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_fp16 = mask.view(total_rows, 1).to(torch.float16)
        HAS_MASK = 1
    else:
        mask_fp16 = torch.empty(0, device=device, dtype=torch.float16)  # dummy
        HAS_MASK = 0

    # left
    out_left = torch.empty_like(left_raw)
    grid_gate = (triton.cdiv(total_rows, 1),)   # one program per row
    gate_fusion_kernel[grid_gate](
        left_raw,
        left_gate_raw,
        mask_fp16,
        out_left,
        total_rows,
        H,
        HAS_MASK=HAS_MASK,
        num_warps=4,
    )
    # right
    out_right = torch.empty_like(right_raw)
    gate_fusion_kernel[grid_gate](
        right_raw,
        right_gate_raw,
        mask_fp16,
        out_right,
        total_rows,
        H,
        HAS_MASK=HAS_MASK,
        num_warps=4,
    )

    # sigmoid for out_gate (in‑place)
    out_gate = out_gate_raw.sigmoid_()   # [R, H]

    # ------------------------------------------------------------------
    # 5️⃣ Batched cubic update  (left @ rightᵀ) – FP16 Tensor‑Core GEMM
    # ------------------------------------------------------------------
    # reshape to [B, N, N, H]
    left = out_left.view(B, N, N, H)
    right = out_right.view(B, N, N, H)

    # bring H into the batch dimension → [B·H, N, N]
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # (N×N) @ (N×N)ᵀ  →  [B·H, N, N]   (FP16, Tensor‑Core)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back → [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm over hidden dim (FP16) – Triton
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, H).contiguous()
    out_norm_fp16 = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],   # γ, fp32
        weights["to_out_norm.bias"],     # β, fp32
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 7️⃣ Apply output‑gate, final projection, cast back to fp32
    # ------------------------------------------------------------------
    # element‑wise gating (still fp16)
    out_norm_fp16 = out_norm_fp16 * out_gate.view(total_rows, H)

    # final projection: [R, H] @ [dim, H]ᵀ → [R, dim]  (fp16)
    final_fp16 = F.linear(
        out_norm_fp16,
        weights["to_out.weight"].to(torch.float16)   # [dim, H]
    )   # [R, dim] (fp16)

    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final
