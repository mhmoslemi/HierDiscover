@triton.jit
def layernorm_gate_kernel(x_ptr, y_ptr, weight_ptr, bias_ptr, gate_ptr,
                          rows, dim, eps,
                          BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr):
   pid = tl.program_id(0)
   row_start = pid * BLOCK_M
   offs_m = row_start + tl.arange(0, BLOCK_M)
   mask_m = offs_m < rows

   # compute mean, var
   sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
   sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

   for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

   mean = sum_val / dim
   var = sum_sq / dim - mean * mean
   rstd = tl.rsqrt(var + eps)

   # normalize and apply weight,bias,gate
   for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)
        norm = (vals - mean[:, None]) * rstd[:, None]  # fp32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # γ
        b = tl.load(bias_ptr + offs_d, mask=mask_d, other=0.0)   # β

        out = norm * w[None, :] + b[None, :]   # fp32

        gate = tl.load(gate_ptr + offs_m[:, None] * dim + offs_d[None, :],
                        mask=mask_m[:, None] & mask_d[None, :],
                        other=0.0)                # fp16

        out = out * gate  # fp32 * fp16 -> promotion? gate as fp16 -> cast to fp32 automatically? Convert gate to fp32
        out_fp16 = out.to(tl.float16)
        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16, mask=mask_m[:, None] & mask_d[None, :])
```

"""
custom_kernel – high‑performance outgoing TriMul (AlphaFold3)

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the 4‑D input (B·N·N, dim) → fp16.
2️⃣ Fuse the five linear projections (left, right, left‑gate, right‑gate,
   out‑gate) into one GEMM: X_norm @ W_fusedᵀ → (R, 5·H)  (R = B·N·N).
3️⃣ Triton kernel `split_fuse_gate_kernel` splits the fused tensor,
   applies the optional pair‑wise mask, computes sigmoid gates (fp32→fp16)
   and writes three tensors: left, right, out_gate  (all fp16).
4️⃣ Batched GEMM over the hidden dimension:
        out[b·H+i] = left[b·H+i] @ right[b·H+i]ᵀ
   (implemented with torch.bmm → Tensor Cores).
5️⃣ Triton kernel `layernorm_gate_kernel` performs a second LayerNorm on
   the hidden‑dim tensor **and** multiplies by `out_gate` in a single
   pass (reducing memory traffic).
6️⃣ Final linear projection back to the original `dim` (fp16 → fp32).

Only the two LayerNorm kernels and the split‑gate kernel are written in
Triton; all heavy GEMMs use cuBLAS via PyTorch for maximal throughput.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (fp32 → fp16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   *[rows, dim]   (float32)
    y_ptr,          # mutable *[rows, dim]   (float16)
    weight_ptr,     # const   *[dim]        (float32)   γ
    bias_ptr,       # const   *[dim]        (float32)   β
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ---- compute mean & variance (fp32 accumulation) ----
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]   # [M, D]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)               # fp32
        sum_val += tl.sum(vals, axis=1)                     # [M]
        sum_sq  += tl.sum(vals * vals, axis=1)              # [M]

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- normalize, apply γ/β and store as fp16 ----
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]        # fp32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # β
        out = norm * w[None, :] + b[None, :]                     # fp32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, apply mask & sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[rows, 5*H] (float16)
    left_ptr,          # mutable *[rows, H]   (float16)
    right_ptr,         # mutable *[rows, H]   (float16)
    out_gate_ptr,      # mutable *[rows, H]   (float16)
    mask_ptr,          # const   *[rows]    (float16)  (ignored if HAS_MASK=0)
    rows: tl.int32,
    H: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    For each row r:
        left  = (proj_left  * mask) * sigmoid(proj_left_gate)
        right = (proj_right * mask) * sigmoid(proj_right_gate)
        out_gate = sigmoid(proj_out_gate)
    All ops in fp16; sigmoid evaluated in fp32.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)               # [BLOCK_R]
    mask_r = offs_r < rows

    # ---- optional mask (broadcast scalar per row) ----
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        m_val = m_val[:, None]                                      # [BLOCK_R,1]
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # ---- iterate over hidden dimension ----
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)          # [BLOCK_H]
        mask_h = offs_h < H

        # layout of fused tensor:
        #   0 .. H-1               -> left
        #   H .. 2H-1              -> right
        #   2H .. 3H-1             -> left_gate
        #   3H .. 4H-1             -> right_gate
        #   4H .. 5H-1             -> out_gate
        row_off = offs_r[:, None] * (5 * H)                # [R,1]
        col_off = offs_h[None, :]                          # [1,H]

        left_raw   = tl.load(proj_ptr + row_off + col_off,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        right_raw  = tl.load(proj_ptr + row_off + H + col_off,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        left_g_raw = tl.load(proj_ptr + row_off + 2 * H + col_off,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        right_g_raw= tl.load(proj_ptr + row_off + 3 * H + col_off,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        out_g_raw  = tl.load(proj_ptr + row_off + 4 * H + col_off,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)

        # sigmoid in fp32 for stability
        left_gate  = (1.0 / (1.0 + tl.exp(-left_g_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_g_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_g_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * m_val * left_gate
        right = right_raw * m_val * right_gate

        tl.store(left_ptr     + offs_r[:, None] * H + offs_h[None, :],
                 left,  mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr    + offs_r[:, None] * H + offs_h[None, :],
                 right, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr+ offs_r[:, None] * H + offs_h[None, :],
                 out_gate, mask=mask_r[:, None] & mask_h[None, :])

# ----------------------------------------------------------------------
# 3️⃣ Second LayerNorm + out_gate multiplication (fused)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_gate_kernel(
    x_ptr,           # const   *[rows, dim]   (float16)
    y_ptr,           # mutable *[rows, dim]   (float16)
    weight_ptr,      # const   *[dim]        (float32)   γ
    bias_ptr,        # const   *[dim]        (float32)   β
    gate_ptr,        # const   *[rows, dim]   (float16)
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ---- mean & variance (fp32) ----
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]   # [M, D]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- normalize, apply γ/β and multiply by gate ----
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]        # fp32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # γ, fp32
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # β, fp32
        out = norm * w[None, :] + b[None, :]                     # fp32

        g = tl.load(gate_ptr + offs_m[:, None] * dim + offs_d[None, :],
                    mask=mask_m[:, None] & mask_d[None, :],
                    other=0.0)                                      # fp16
        out = out * g                                            # fp32 * fp16 → fp32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : mapping with all linear / layer‑norm parameters
                         (float32 tensors, bias=False for Linear)
        - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                         # total rows for row‑wise ops
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (fp32 → fp16) using Triton
    # ------------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()                 # [R, dim] float32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 1024
    BLOCK_D = 32
    grid = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],          # γ (float32)
        weights["norm.bias"],            # β (float32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection: left, right, left_gate, right_gate, out_gate
    # ------------------------------------------------------------------
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # [5*H, dim]

    # (R, dim) @ (5*H, dim)ᵀ → (R, 5*H)   fp16
    proj = F.linear(x_norm, w_fused)        # fp16
    del w_fused, x_norm                     # free memory

    # ------------------------------------------------------------------
    # 3️⃣ Split, mask & apply sigmoid gates (single Triton kernel)
    # ------------------------------------------------------------------
    left = torch.empty(R, H, dtype=torch.float16, device=device)
    right = torch.empty(R, H, dtype=torch.float16, device=device)
    out_gate = torch.empty(R, H, dtype=torch.float16, device=device)

    if cfg.get("nomask", True) or mask is None:
        # No mask – kernel will compile with HAS_MASK=0
        dummy_mask = torch.empty(0, dtype=torch.float16, device=device)
        mask_ptr = dummy_mask
        has_mask = 0
    else:
        mask_flat = mask.to(torch.float16).reshape(R)   # [R]
        mask_ptr = mask_flat
        has_mask = 1

    BLOCK_R = 128
    BLOCK_H = 32
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Tri‑multiplicative update: batched GEMM over hidden dim
    # ------------------------------------------------------------------
    # reshape to [B, N, N, H] → bring H into batch dimension
    left = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    left_mat = left.view(B * H, N, N)      # [B·H, N, N]
    right_mat = right.view(B * H, N, N)    # [B·H, N, N]

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))      # [B·H, N, N]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm + out_gate multiplication (fused)
    # ------------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()                 # [R, H] fp16
    out_norm = torch.empty_like(out_flat, dtype=torch.float16, device=device)

    grid2 = (triton.cdiv(R, BLOCK_M),)                     # reuse BLOCK_M from before
    layernorm_gate_kernel[grid2](
        out_flat,
        out_norm,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        out_gate,                         # gate (fp16)
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    # free temp tensors no longer needed
    del out_flat, out, out_gate

    # ------------------------------------------------------------------
    # 6️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    final_half = F.linear(
        out_norm,
        weights["to_out.weight"].to(torch.float16),   # [dim, H]
    )   # fp16, shape [R, dim]

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
