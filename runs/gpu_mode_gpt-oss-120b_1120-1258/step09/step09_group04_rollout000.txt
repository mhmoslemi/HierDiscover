"""
TriMul (outgoing) – high‑performance forward implementation.

Algorithm
---------
1️⃣ Row‑wise LayerNorm over the last channel dimension (float32 → float16).
2️⃣ Fuse the four linear projections (left_proj, right_proj, left_gate, right_gate)
   into a single GEMM:
        X_norm @ W_fused.T  ->  [R, 4·H]   (R = B·N·N, H = hidden_dim)
   Split the result into the four tensors.
3️⃣ Compute the output‑gate with a second linear (still FP16).
4️⃣ Optional binary mask is applied to *left* and *right* (pre‑gate).
5️⃣ Apply sigmoid to the three gate tensors (in‑place) and multiply:
        left  = left_raw  * sigmoid(left_gate)
        right = right_raw * sigmoid(right_gate)
6️⃣ Perform the N³ multiplicative update.
   Reshape left/right → (B·H, N, N) and use a batched GEMM:
        out_mat[b·H] = left_mat[b·H] @ right_mat[b·H].T
   Reshape back to (B, N, N, H).
7️⃣ Fuse the second LayerNorm and the element‑wise out‑gate into a single
   Triton kernel (float32 → float16).
8️⃣ Final linear projection back to the original dimensionality (FP16 GEMM) and
   cast to float32.

Only the two LayerNorm‑related kernels are written in Triton; the remaining
operations leverage highly‑optimised torch kernels (Tensor‑Core‑enabled FP16
GEMMs and batched GEMM).  This balance yields excellent portability while
meeting the speed target on an H100.

import torch.nn.functional as F
import triton
import triton.language as tl

# Triton kernel: row‑wise LayerNorm (FP32 -> FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   * [total_rows, dim]   (float32)
    y_ptr,          # mutable * [total_rows, dim]   (float16)
    weight_ptr,     # const   * [dim]              (float32)
    bias_ptr,       # const   * [dim]              (float32)
    total_rows,     # int32 (runtime)
    dim,            # int32 (runtime)
    eps,            # fp32  (compile‑time)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # cols per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # -------------------- mean & variance (FP32) --------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]  # [M,D]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)          # 1 / sqrt(var+eps)

    # -------------------- normalise & apply γ,β (FP16) --------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]   # (x-μ)/σ

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β

        out = norm * w[None, :] + b[None, :]            # still FP32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

# Custom forward implementation for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
        - weights_dict : mapping of all Linear/LayerNorm parameters
        - config_dict  : {"dim": int,
                         "hidden_dim": int,
                         "nomask": bool (optional)}

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ---------------------------------------------------------------
    # 0️⃣ unpack & constants
    # ---------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape                        # [B, N, N, dim]
    R = B * N * N                                # total rows (B·N·N)
    H = hidden_dim

    # ---------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm → FP16 (triton)
    # ---------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()                     # [R, dim] FP32
    x_norm_fp16 = torch.empty(R, dim,
                              dtype=torch.float16,
                              device=device)

    # launch configuration – tuned for H100
    BLOCK_M = 128          # rows per CTA
    BLOCK_D = 64           # columns per iteration
    grid = (triton.cdiv(R, BLOCK_M),)

    layernorm_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],          # γ (float32)
        weights["norm.bias"],            # β (float32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
        num_stages=2,
    )

    # ---------------------------------------------------------------
    # 2️⃣ Fuse left/right projections + their gates in a single GEMM
    # ---------------------------------------------------------------
    # Build a fused weight matrix of shape [4·H, dim] (FP16)
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)                 # [4·H, dim]

    # (R, dim) @ (dim, 4·H).T -> (R, 4·H)
    fused = F.linear(x_norm_fp16, w_fused)   # FP16

    left_raw      = fused[:, 0:H]                 # [R, H]
    right_raw     = fused[:, H:2*H]               # [R, H]
    left_gate_raw = fused[:, 2*H:3*H]             # [R, H]
    right_gate_raw= fused[:, 3*H:4*H]             # [R, H]

    # second linear for output‑gate (still FP16)
    out_gate_raw = F.linear(
        x_norm_fp16,
        weights["out_gate.weight"].to(torch.float16)   # [H, dim]
    )   # [R, H]

    # ---------------------------------------------------------------
    # 3️⃣ Apply mask (if present) and sigmoid gates
    # ---------------------------------------------------------------
    if (not nomask) and mask is not None:
        # mask shape: [B, N, N] → flatten → [R, 1] (FP16)
        mask_fp16 = mask.reshape(R, 1).to(torch.float16)
        left_raw  = left_raw * mask_fp16
        right_raw = right_raw * mask_fp16

    # in‑place sigmoid (FP16)
    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()          # [R, H] FP16

    left  = left_raw  * left_gate   # [R, H] FP16
    right = right_raw * right_gate  # [R, H] FP16

    # ---------------------------------------------------------------
    # 4️⃣ Batched N³ multiplicative update (B·H independent GEMMs)
    # ---------------------------------------------------------------
    # reshape to (B, N, N, H) → (B, H, N, N) → (B·H, N, N)
    left = left.view(B, N, N, H)
    right = right.view(B, N, N, H)

    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # batched GEMM: (B·H, N, N) @ (B·H, N, N)^T → (B·H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H] FP16

    # ---------------------------------------------------------------
    # 5️⃣ Fused LayerNorm + element‑wise out‑gate (over hidden dim H)
    # ---------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()          # [R, H] FP16
    out_gate_flat = out_gate                         # already FP16
    out_norm_gate = torch.empty_like(out_flat, dtype=torch.float16, device=device)

    grid2 = (triton.cdiv(R, BLOCK_M),)                # reuse BLOCK_M/D
    layernorm_gate_kernel[grid2](
        out_flat,
        out_gate_flat,
        out_norm_gate,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
        num_stages=2,
    )

    # ---------------------------------------------------------------
    # 6️⃣ Final linear projection back to original dim and cast to fp32
    # ---------------------------------------------------------------
    final_fp16 = F.linear(
        out_norm_gate,
        weights["to_out.weight"].to(torch.float16)   # [dim, H]
    )   # [R, dim] FP16

    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final
