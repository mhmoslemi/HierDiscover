"""
custom_kernel
==============

High‑performance forward pass for the *outgoing* TriMul operator (AlphaFold3).

Algorithm (FP16‑centric):
  1️⃣ Row‑wise LayerNorm on the flattened input (FP32 → FP16) – Triton kernel.
  2️⃣ Fuse the four linear layers (left/right projection + left/right gate) into one
     GEMM:   X_norm @ [W_left; W_right; W_left_gate; W_right_gate]^T.
     Split the result into `left_raw`, `right_raw`, `left_gate_raw`, `right_gate_raw`.
  3️⃣ Compute the output‑gate with a second GEMM.
  4️⃣ (Optional) apply binary mask and sigmoid gates (in‑place).
  5️⃣ Reshape to `[B, N, N, H]` (H = hidden_dim) and perform a batched matrix‑multiply
     `left @ rightᵀ` for each head → cubic multiplicative update (FP16 Tensor‑Core bmm).
  6️⃣ Second LayerNorm over the hidden dimension (FP16) – PyTorch `layer_norm`.
  7️⃣ Element‑wise output‑gate, final linear projection, cast back to FP32.

The kernel works for batch size 1‑2, sequence length up to 1024 and channel sizes
`dim ∈ {128,384,768}`, hidden_dim = 128.  The function returns a `torch.float32`
tensor of shape `[B, N, N, dim]`.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton row‑wise LayerNorm (fp32 → fp16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const  * [total_rows, dim]   (float32)
    y_ptr,          # mutable* [total_rows, dim]   (float16)
    w_ptr,          # const  * [dim]               (float32) – γ
    b_ptr,          # const  * [dim]               (float32) – β
    total_rows,     # int (runtime)
    dim,            # int (runtime)
    eps,            # float (run‑time)
    BLOCK_M: tl.constexpr,  # rows per program
    BLOCK_D: tl.constexpr,  # columns per tile
):
    """
    Row‑wise LayerNorm.
    y = ((x - μ) / sqrt(σ² + eps)) * γ + β
    The kernel processes BLOCK_M rows at a time and iterates over the
    channel dimension in BLOCK_D‑wide tiles.
    """
    pid = tl.program_id(0)                     # 1‑D grid
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------- 1. mean & variance -------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]   # address
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0)                # fp32 load
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                # 1 / sqrt(var + eps)

    # ------------------- 2. write normalized result (fp16) ---------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)   # fp32 -> fp32 (still)

        norm = (vals - mean[:, None]) * rstd[:, None]          # (x-μ)/σ

        # γ, β are fp32 → keep in fp32, cast the final result to fp16
        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)
        out = norm * w[None, :] + b[None, :]                    # fp32

        out_fp16 = out.to(tl.float16)
        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom kernel entry‑point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : dict of torch.Parameters (float32) with keys:
              "norm.weight", "norm.bias",
              "left_proj.weight", "right_proj.weight",
              "left_gate.weight", "right_gate.weight",
              "out_gate.weight",
              "to_out_norm.weight", "to_out_norm.bias",
              "to_out.weight"
        - config_dict  : dict with keys:
              "dim" (int), "hidden_dim" (int), optional "nomask" (bool)

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # 0️⃣ unpack & constants
    # --------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    total_rows = B * N * N                # R = B·N·N
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32 → FP16)
    # --------------------------------------------------------------
    x_flat = x.reshape(total_rows, dim).contiguous()          # [R, dim] (fp32)
    x_norm_fp16 = torch.empty(total_rows, dim,
                              dtype=torch.float16,
                              device=device)

    BLOCK_M = 128     # rows per program – tuned for H100
    BLOCK_D = 64      # cols per tile
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],   # γ, fp32
        weights["norm.bias"],     # β, fp32
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 2️⃣ Fuse left/right projections & their gates into one GEMM (FP16)
    # --------------------------------------------------------------
    # Concatenate the four weight matrices: [4·H, dim]
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],    # [H, dim]
            weights["right_proj.weight"],   # [H, dim]
            weights["left_gate.weight"],    # [H, dim]
            weights["right_gate.weight"],   # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)                     # [4·H, dim]

    # x_norm_fp16 is [R, dim]; fused GEMM → [R, 4·H]
    fused = F.linear(x_norm_fp16, w_fused)   # fp16

    left_raw       = fused[:, 0:H]
    right_raw      = fused[:, H:2 * H]
    left_gate_raw  = fused[:, 2 * H:3 * H]
    right_gate_raw = fused[:, 3 * H:4 * H]

    # --------------------------------------------------------------
    # 3️⃣ Output‑gate (separate GEMM)
    # --------------------------------------------------------------
    out_gate_raw = F.linear(
        x_norm_fp16,
        weights["out_gate.weight"].to(torch.float16)   # [H, dim]
    )   # [R, H] fp16

    # --------------------------------------------------------------
    # 4️⃣ (Optional) mask & sigmoid gates (in‑place)
    # --------------------------------------------------------------
    if not nomask and mask is not None:
        mask_fp16 = mask.reshape(total_rows, 1).to(torch.float16)
        left_raw  = left_raw * mask_fp16
        right_raw = right_raw * mask_fp16

    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    left  = left_raw * left_gate       # [R, H]
    right = right_raw * right_gate     # [R, H]

    # --------------------------------------------------------------
    # 5️⃣ Batched cubic update  (left @ rightᵀ) – FP16 bmm
    # --------------------------------------------------------------
    # Reshape to [B, N, N, H]
    left  = left.view(B, N, N, H)
    right = right.view(B, N, N, H)
    out_gate = out_gate.view(B, N, N, H)

    # Permute to [B, H, N, N] for strided batched GEMM
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    # Collapse (B, H) into a single batch dimension
    left_mat  = left_perm.view(B * H, N, N)               # [BH, N, N]
    right_mat = right_perm.view(B * H, N, N)              # [BH, N, N]

    # (N×N) @ (N×N)ᵀ → [BH, N, N]   (Tensor‑Core GEMM)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Restore original shape → [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm over hidden dimension (FP16) – PyTorch
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )   # still fp16

    # --------------------------------------------------------------
    # 7️⃣ Apply output‑gate, final projection, cast back to FP32
    # --------------------------------------------------------------
    out = out * out_gate                                 # element‑wise gate (fp16)

    final_fp16 = F.linear(
        out,
        weights["to_out.weight"].to(torch.float16)      # [dim, H]
    )   # [B, N, N, dim] (fp16)

    final = final_fp16.to(torch.float32)                 # return fp32
    return final
```
