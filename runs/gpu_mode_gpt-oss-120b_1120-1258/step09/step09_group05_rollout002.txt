"""
TriMul (outgoing) – Triton‑accelerated forward pass
===================================================

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the flattened 4‑D input (fp32 → fp16) via Triton.
2️⃣ Fuse the two projections (left / right) together with the two gate
   linear layers into one FP16 GEMM:
        X_norm @ [W_left; W_right; W_left_gate; W_right_gate]ᵀ → [R, 4·H]
   where R = B·N·N and H = hidden_dim.
   The result is sliced into raw left/right tensors and raw gate tensors.
3️⃣ Compute the output‑gate with a second FP16 GEMM.
4️⃣ (Optional) apply the binary mask to the raw left/right tensors,
   then apply `sigmoid` in‑place to the three gate tensors.
5️⃣ Form the cubic multiplicative update as a batched matrix multiplication:
        out[b·H] = left[b·H] @ right[b·H]ᵀ      (Tensor‑Core FP16)
   This is exactly the einsum used in the reference code.
6️⃣ Second LayerNorm over the hidden dimension (fp16) via the same Triton kernel.
7️⃣ Fuse the output‑gate, a final FP16 linear projection and cast back to fp32.

The implementation works for any batch size B∈{1,2},
sequence length N∈{128…1024},
input dimension dim∈{128,384,768},
hidden_dim H=128 (as used in the tests) and supports both masked and
un‑masked variants.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (fp32 → fp16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,      # const   * [total_rows, dim] (fp32)
    y_ptr,      # mutable * [total_rows, dim] (fp16)
    w_ptr,      # const   * [dim]          (fp32)  – gamma
    b_ptr,      # const   * [dim]          (fp32)  – beta
    total_rows, # i32
    dim,        # i32
    eps,        # f32, compile‑time constant
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # cols per tile
):
    """Row‑wise layer‑norm."""
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------- mean / variance -------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)          # 1 / sqrt(var + eps)

    # ------------------- write normalized output (fp16) -------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]          # (x-μ)/σ

        # γ and β are fp32 → keep the computation in fp32
        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)
        out = norm * w[None, :] + b[None, :]                   # fp32

        out_fp16 = out.to(tl.float16)
        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Triton‑accelerated TriMul forward
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : mapping str → torch.nn.Parameter (float32)
        - config_dict  : {"dim": int,
                          "hidden_dim": int,
                          "nomask": bool (optional)}

    Returns
    -------
    torch.Tensor : shape [B, N, N, dim], dtype float32
    """
    # ------------------- unpack -------------------
    x, mask, w, cfg = data
    device = x.device
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5

    B, N, _, _ = x.shape
    R = B * N * N          # total number of pairwise rows
    H = hidden_dim

    # ------------------- 1️⃣ first LayerNorm (fp32 → fp16) -------------------
    # Flatten to [R, dim] (row‑major)
    x_flat = x.view(R, dim).contiguous()
    x_norm_fp16 = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 256   # rows per program (tuned for H100)
    BLOCK_D = 64    # columns per tile
    grid = (triton.cdiv(R, BLOCK_M),)

    layernorm_kernel[grid](
        x_flat,
        x_norm_fp16,
        w["norm.weight"],      # γ  (float32)
        w["norm.bias"],        # β  (float32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # ------------------- 2️⃣ fused projection + gates -------------------
    # Concatenate the four weight matrices: [W_left; W_right; W_left_gate; W_right_gate]
    w_fused = torch.cat(
        [
            w["left_proj.weight"],   # (H, dim)
            w["right_proj.weight"],  # (H, dim)
            w["left_gate.weight"],   # (H, dim)
            w["right_gate.weight"],  # (H, dim)
        ],
        dim=0,
    ).to(torch.float16)                # (4·H, dim)

    fused = F.linear(x_norm_fp16, w_fused)   # (R, 4·H)  fp16

    left_raw        = fused[:, 0:H]
    right_raw       = fused[:, H:2 * H]
    left_gate_raw   = fused[:, 2 * H:3 * H]
    right_gate_raw  = fused[:, 3 * H:4 * H]

    # ------------------- 3️⃣ output‑gate -------------------
    out_gate_raw = F.linear(
        x_norm_fp16,
        w["out_gate.weight"].to(torch.float16)   # (H, dim)
    )   # (R, H) fp16

    # ------------------- 4️⃣ mask & sigmoid -------------------
    if not nomask and mask is not None:
        # mask: [B, N, N] → [R, 1] broadcast over hidden dim
        mask_fp16 = mask.view(R, 1).to(torch.float16)
        left_raw  = left_raw * mask_fp16
        right_raw = right_raw * mask_fp16

    left_gate   = left_gate_raw.sigmoid_()
    right_gate  = right_gate_raw.sigmoid_()
    out_gate    = out_gate_raw.sigmoid_()

    left  = left_raw  * left_gate   # (R, H) fp16
    right = right_raw * right_gate  # (R, H) fp16

    # ------------------- 5️⃣ cubic multiplicative update (batched GEMM) -------------------
    # reshape to [B, N, N, H] → permute to [B, H, N, N] → view as [B·H, N, N]
    left  = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    out_gate = out_gate.view(B, N, N, H)   # keep for later

    left_mat  = left.view(B * H, N, N)          # (B·H, N, N) fp16
    right_mat = right.view(B * H, N, N)         # (B·H, N, N) fp16

    # (N×N) @ (N×N)ᵀ → (B·H, N, N)   using Tensor‑Core bmm in fp16
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # (B, N, N, H)

    # ------------------- 6️⃣ second LayerNorm over hidden dim -------------------
    out_flat = out.view(R, H).contiguous()
    out_norm_fp16 = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm_fp16,
        w["to_out_norm.weight"],   # γ  (float32)
        w["to_out_norm.bias"],     # β  (float32)
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # ------------------- 7️⃣ final gating, projection, cast back -------------------
    # element‑wise output gate (still fp16)
    out_norm_fp16 = out_norm_fp16 * out_gate.view(R, H)

    # final linear projection: (R, H) @ (dim, H)ᵀ → (R, dim)
    final_fp16 = F.linear(
        out_norm_fp16,
        w["to_out.weight"].to(torch.float16)   # (dim, H)
    )   # (R, dim) fp16

    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final
