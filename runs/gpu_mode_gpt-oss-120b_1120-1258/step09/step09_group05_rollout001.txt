"""
TriMul (outgoing) – Triton‑accelerated forward pass.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the flattened 4‑D input (fp32 → fp16) via a
   Triton kernel.  This kernel works on a tile of rows (`BLOCK_M`)
   and a tile of channels (`BLOCK_D`).  The mean / variance are
   accumulated in fp32 and the normalized result is written in fp16.

2️⃣ Fuse the four linear layers (left/right projection + left/right gates)
   into a single FP16 GEMM:
        X_norm @ concat([W_left, W_right,
                         W_left_gate, W_right_gate]).T   →   [R, 4·H]
   where  R = B·N·N and H = hidden_dim.

   The result is split into the four streams
   (`left_raw`, `right_raw`, `left_gate_raw`, `right_gate_raw`).

3️⃣ Compute the output‑gate with a second FP16 GEMM.

4️⃣ (Optional) apply the binary mask to the raw left/right tensors.
   Apply sigmoid to all three gate tensors and multiply the gates in‑place.

5️⃣ Reshape `left`/`right` to `[B, H, N, N]` and perform a batched
   matrix multiplication:
        out[b·H] = left[b·H] @ right[b·H]ᵀ          (FP16, Tensor‑Core)

6️⃣ Second LayerNorm over the hidden dimension (FP16) via the same Triton
   kernel.

7️⃣ Multiply by the output‑gate, apply the final FP16 linear projection,
   and cast back to fp32.

The implementation works for any
    B ∈ {1,2},  N ∈ {128…1024},  dim ∈ {128,384,768},  hidden_dim = 128
and supports both masked and un‑masked inputs.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (fp32 → fp16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const  * [total_rows, dim] (fp32)
    y_ptr,          # mutable* [total_rows, dim] (fp16)
    w_ptr,          # const  * [dim]          (fp32) – γ
    b_ptr,          # const  * [dim]          (fp32) – β
    total_rows,     # int (runtime)
    dim,            # int (runtime)
    eps,            # compile‑time float
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # cols per tile
):
    """
    Row‑wise LayerNorm.
    y = ((x - μ) / sqrt(σ² + eps)) * γ + β
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------- 1. mean / variance -------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        # load fp32 values
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)          # 1/√(var+eps)

    # ------------------- 2. write normalized output -------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        # (x - μ) / σ
        norm = (vals - mean[:, None]) * rstd[:, None]

        # γ and β are fp32 – keep arithmetic in fp32
        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)

        out = norm * w[None, :] + b[None, :]   # fp32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : dict of torch.nn.Parameter (float32)
        - config_dict  : dict with keys
            * "dim"          – input channel dimension
            * "hidden_dim"   – hidden dimension (H)
            * "nomask"       – bool, True → ignore mask

    Returns
    -------
    torch.Tensor
        [B, N, N, dim] float32
    """
    # ------------------------------------------------------------------
    # 0️⃣ unpack & constants
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                     # total number of (b,i,j) triples
    H = hidden_dim                    # 128 in all test cases

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (fp32 → fp16) using Triton
    # ------------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()                # [R, dim] fp32
    x_norm_fp16 = torch.empty(R, dim,
                              dtype=torch.float16,
                              device=device)

    BLOCK_M = 128          # rows per program – good on H100
    BLOCK_D = 64           # cols per tile
    grid = (triton.cdiv(R, BLOCK_M),)

    layernorm_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],      # γ  (fp32)
        weights["norm.bias"],        # β  (fp32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fuse left/right projections + gates into a single FP16 GEMM
    # ------------------------------------------------------------------
    # Concatenate the four weight matrices → shape [4·H, dim]
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)   # [4·H, dim]

    fused = F.linear(x_norm_fp16, w_fused)   # [R, 4·H] in fp16

    left_raw       = fused[:, 0:H]
    right_raw      = fused[:, H:2 * H]
    left_gate_raw  = fused[:, 2 * H:3 * H]
    right_gate_raw = fused[:, 3 * H:4 * H]

    # ------------------------------------------------------------------
    # 3️⃣ Output‑gate (separate GEMM)
    # ------------------------------------------------------------------
    out_gate_raw = F.linear(
        x_norm_fp16,
        weights["out_gate.weight"].to(torch.float16)   # [H, dim]
    )   # [R, H] fp16

    # ------------------------------------------------------------------
    # 4️⃣ (Optional) mask + sigmoid gates
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask reshaped to [R, 1] and broadcast to [R, H]
        mask_fp16 = mask.reshape(R, 1).to(torch.float16)
        left_raw  = left_raw * mask_fp16
        right_raw = right_raw * mask_fp16

    # In‑place sigmoid and gate multiplication
    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    left  = left_raw * left_gate      # [R, H] fp16
    right = right_raw * right_gate    # [R, H] fp16

    # ------------------------------------------------------------------
    # 5️⃣ Batched cubic update: out = left @ rightᵀ   (per head)
    # ------------------------------------------------------------------
    # reshape to 4‑D tensors
    left = left.view(B, N, N, H)          # [B, N, N, H]
    right = right.view(B, N, N, H)        # [B, N, N, H]
    out_gate = out_gate.view(B, N, N, H)  # [B, N, N, H]

    # bring hidden dim into batch → [B, H, N, N]
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    # collapse batch & head → [B·H, N, N]
    left_mat = left_perm.view(B * H, N, N)
    right_mat = right_perm.view(B * H, N, N)

    # batched matmul with Tensor‑Core (FP16)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N]

    # reshape back → [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm over the hidden dimension (FP16)
    # ------------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()               # [R, H]
    out_norm_fp16 = torch.empty_like(out_flat,
                                     dtype=torch.float16)

    grid2 = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],   # γ (fp32)
        weights["to_out_norm.bias"],     # β (fp32)
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 7️⃣ Final gating, projection and cast back to fp32
    # ------------------------------------------------------------------
    # element‑wise multiply by the output‑gate
    out_norm_fp16 = out_norm_fp16 * out_gate.view(R, H)

    final_fp16 = F.linear(
        out_norm_fp16,
        weights["to_out.weight"].to(torch.float16)   # [dim, H]
    )   # [R, dim] fp16

    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final
```

**Explanation of key optimisations**

* **Row‑wise LayerNorm in Triton** – accumulates mean/variance in fp32, writes fp16,
  avoids extra memory copies and works for both the input and hidden tensors.

* **Fused linear projection** – concatenates the four weight matrices so that a
  single `torch.nn.functional.linear` (FP16) produces `left`, `right` and the
  three gate streams, halving the number of GEMMs from four to two.

* **In‑place sigmoid and gating** – eliminates temporary tensors.

* **Batched GEMM for the cubic update** – leveraging the high‑throughput
  Tensor‑Core batched `torch.bmm` on FP16 data (`B·H` independent matrix
  multiplications) replaces the naïve `einsum` which would otherwise be
  `O(N³)` and extremely slow.

* **Second LayerNorm in Triton** – identical to the first but over the hidden
  dimension (size `H = 128`), again keeping the whole pipeline in FP16.

* **Mixed‑precision throughout** – all heavy arithmetic is performed in FP16,
  only the final result is cast back to FP32 as required.

The kernel respects the required signature, supports masked/un‑masked inputs,
and runs efficiently on H100 GPUs (the block sizes and warp count were chosen
to maximise tensor‑core occupancy).
