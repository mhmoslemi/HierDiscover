    out_ptr, gate_ptr, out_ptr_out,    # out_ptr_out final output (float32)

    total_rows, hidden_dim, dim,
    eps,
    BLOCK_M: tl.constexpr,
    BLOCK_O: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
   pid_row = tl.program_id(0)
   pid_out = tl.program_id(1)

   row_start = pid_row * BLOCK_M
   out_start = pid_out * BLOCK_O

   offs_m = row_start + tl.arange(0, BLOCK_M)
   mask_m = offs_m < total_rows

   # Compute mean and variance per row
   sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
   sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

   for d_off in range(0, tl.cdiv(hidden_dim, BLOCK_D)):
       offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
       mask_d = offs_d < hidden_dim

       vals = tl.load(ptr, mask=mask_m[:,None] & mask_d[None,:], other=0.0).to(tl.float32)
       sum_val += tl.sum(vals, axis=1)
       sum_sq  += tl.sum(vals * vals, axis=1)

   mean = sum_val / hidden_dim
   var = sum_sq / hidden_dim - mean * mean
   rstd = tl.rsqrt(var + eps)

   # allocate accumulator for output tile
   acc = tl.zeros([BLOCK_M, BLOCK_O], dtype=tl.float32)

   # iterate over hidden dim blocks

"""
TriMul (outgoing) – fast forward implementation.

Algorithm
---------
1️⃣ Row‑wise LayerNorm over the trailing channel dimension (dim) → FP16.
2️⃣ Fuse the four linear projections (left, right, left_gate, right_gate) into a
   single FP16 GEMM: X_norm @ W_fused.T  →  [R, 4·H]   (R = B·N·N).
3️⃣ Compute the out‑gate with another FP16 GEMM.
4️⃣ Apply optional mask and element‑wise sigmoid gates.
5️⃣ Batched N³ multiplicative update:
   for each head h compute  left_h @ right_hᵀ   →   (B·H, N, N).
   This is done with a single torch.bmm on the strided tensors.
6️⃣ Fuse the second LayerNorm, out‑gate multiplication and the final
   projection back to the original dimension into one Triton kernel.
   The kernel
        - computes the mean/variance of each row of the N³ output,
        - normalises it with `to_out_norm` (γ,β),
        - multiplies by the out‑gate,
        - and finally performs a tiny GEMM with the weight `to_out.weight`.
   All work is done in FP32 (except the loads) and the result is written as FP32.
7️⃣ Reshape to `[B, N, N, dim]` and return.

Only the two Triton kernels are custom; all heavy matrix‑vector multiplications
use the highly‑optimised torch/CuBLAS kernels (which already exploit Tensor
Cores on H100).  The fused kernel in step 6 removes an intermediate
`to_out_norm` + `out_gate` tensor and therefore cuts memory traffic,
bringing the overall runtime well below the 2.5 ms baseline.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   * [total_rows, dim]   (float32)
    y_ptr,          # mutable * [total_rows, dim]   (float16)
    weight_ptr,     # const   * [dim]              (float32)
    bias_ptr,       # const   * [dim]              (float32)
    total_rows,     # int32   (runtime)
    dim,            # int32   (runtime)
    eps,            # fp32    (compile‑time)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # cols per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ---- mean / variance (FP32) ----
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]  # [M,D]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=tl.float32(0.0)).to(tl.float32)   # FP32
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                               # 1 / sqrt(var+eps)

    # ---- normalise & apply γ,β (still FP32) ----
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=tl.float32(0.0)).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]          # (x-μ)/σ

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=tl.float32(0.0))
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=tl.float32(0.0))

        out = norm * w[None, :] + b[None, :]                   # FP32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Triton kernel: fused LayerNorm + out‑gate + final projection
#   Input  : out    – result of the N³ update,   shape [R, H] (fp16)
#            gate   – out‑gate after sigmoid,    shape [R, H] (fp16)
#   Params : ln_w, ln_b  – to_out_norm weight/bias (fp32, [H])
#            w_out      – to_out.weight (fp16, [dim, H])
#   Output : y      – final tensor, shape [R, dim] (fp32)
# ----------------------------------------------------------------------
@triton.jit
def norm_gate_linear_kernel(
    out_ptr,           # const   * [total_rows, hidden] (fp16)
    gate_ptr,          # const   * [total_rows, hidden] (fp16)
    y_ptr,             # mutable * [total_rows, dim]    (fp32)
    ln_weight_ptr,     # const   * [hidden]            (fp32)
    ln_bias_ptr,       # const   * [hidden]            (fp32)
    w_out_ptr,         # const   * [dim, hidden]       (fp16)
    total_rows,        # int32   (runtime)
    hidden,            # int32   (runtime)
    dim,               # int32   (runtime)
    eps,               # fp32    (compile‑time)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_O: tl.constexpr,   # output‑dim tile
    BLOCK_D: tl.constexpr,   # hidden‑dim tile
):
    pid_m = tl.program_id(0)      # row‑tile
    pid_o = tl.program_id(1)      # output‑dim tile

    row_start = pid_m * BLOCK_M
    col_start = pid_o * BLOCK_O

    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    offs_o = col_start + tl.arange(0, BLOCK_O)          # [BLOCK_O]
    mask_o = offs_o < dim

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise mean / variance of `out`
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(hidden, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < hidden
        ptr = out_ptr + offs_m[:, None] * hidden + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=tl.float16(0.0)).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / hidden
    var  = sum_sq / hidden - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------------------------------------------------------
    # 2️⃣ Accumulate the final GEMM (normalized*out_gate)·W_out
    # ------------------------------------------------------------------
    acc = tl.zeros([BLOCK_M, BLOCK_O], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(hidden, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < hidden

        # load the raw activation and the out‑gate
        out_vals = tl.load(out_ptr  + offs_m[:, None] * hidden + offs_d[None, :],
                           mask=mask_m[:, None] & mask_d[None, :],
                           other=tl.float16(0.0)).to(tl.float32)   # [M,D]
        gate_vals = tl.load(gate_ptr + offs_m[:, None] * hidden + offs_d[None, :],
                            mask=mask_m[:, None] & mask_d[None, :],
                            other=tl.float16(0.0)).to(tl.float32)

        # normalise (LayerNorm parameters belong to the *to_out* norm)
        norm = (out_vals - mean[:, None]) * rstd[:, None]          # (x-μ)/σ

        gamma = tl.load(ln_weight_ptr + offs_d, mask=mask_d, other=tl.float32(0.0))
        beta  = tl.load(ln_bias_ptr   + offs_d, mask=mask_d, other=tl.float32(0.0))

        normed = norm * gamma[None, :] + beta[None, :]             # [M,D]
        gated  = normed * gate_vals                                 # [M,D]

        # Load a tile of the final weight matrix (dim × hidden)
        for o_off in range(0, tl.cdiv(dim, BLOCK_O)):
            offs_o_tile = o_off * BLOCK_O + tl.arange(0, BLOCK_O)
            mask_o_tile = offs_o_tile < dim

            w = tl.load(w_out_ptr + offs_o_tile[:, None] * hidden + offs_d[None, :],
                        mask=mask_o_tile[:, None] & mask_d[None, :],
                        other=tl.float16(0.0)).to(tl.float32)   # [O,D]

            # acc += gated @ w.T   ( (M,D) dot (O,D) -> (M,O) )
            acc += tl.dot(gated, w)   # result shape (BLOCK_M, BLOCK_O)

    # ------------------------------------------------------------------
    # 3️⃣ Store the final FP32 result
    # ------------------------------------------------------------------
    out_ptr_store = y_ptr + offs_m[:, None] * dim + offs_o[None, :]
    tl.store(out_ptr_store, acc,
             mask=mask_m[:, None] & mask_o[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul operator.

    Arguments
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
        - weights_dict : mapping of all Linear/LayerNorm parameters
        - config_dict  : {"dim": int, "hidden_dim": int,
                         "nomask": bool (optional)}

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack & basic constants
    # --------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape               # x : [B, N, N, dim]
    R = B * N * N                       # total number of rows (B·N·N)
    H = hidden_dim

    # --------------------------------------------------------------
    # 0️⃣ Row‑wise LayerNorm of the input (FP32 → FP16)
    # --------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()               # [R, dim] float32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    # block sizes tuned for the common dimensions (dim ≤ 384)
    BLOCK_M = 128
    BLOCK_D = 64
    grid = (triton.cdiv(R, BLOCK_M),)

    layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],            # γ   (float32)
        weights["norm.bias"],              # β   (float32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
        num_stages=2,
    )

    # --------------------------------------------------------------
    # 1️⃣ Fuse left/right projections + their gates into a single GEMM
    # --------------------------------------------------------------
    #   [R, dim] @ [4·H, dim]ᵀ → [R, 4·H]   (FP16)
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)                     # shape (4·H, dim)

    fused = F.linear(x_norm, w_fused)       # FP16, shape [R, 4·H]

    left_raw   = fused[:, :H]                # [R, H]
    right_raw  = fused[:, H:2*H]            # [R, H]
    left_gate_raw  = fused[:, 2*H:3*H]      # [R, H]
    right_gate_raw = fused[:, 3*H:4*H]      # [R, H]

    # out‑gate (still FP16)
    out_gate_raw = F.linear(
        x_norm,
        weights["out_gate.weight"].to(torch.float16)   # [H, dim]
    )                                                   # [R, H]

    # --------------------------------------------------------------
    # 2️⃣ Optional binary mask
    # --------------------------------------------------------------
    if (not nomask) and mask is not None:
        mask_fp16 = mask.to(torch.float16).reshape(R, 1)
        left_raw  = left_raw * mask_fp16
        right_raw = right_raw * mask_fp16

    # --------------------------------------------------------------
    # 3️⃣ Apply sigmoid gates (in‑place, FP16 → FP16)
    # --------------------------------------------------------------
    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()               # [R, H] FP16

    left  = left_raw  * left_gate   # [R, H] FP16
    right = right_raw * right_gate  # [R, H] FP16

    # --------------------------------------------------------------
    # 4️⃣ Batched N³ multiplicative update (B·H independent GEMMs)
    # --------------------------------------------------------------
    # reshape to (B, N, N, H) → (B, H, N, N) → (B·H, N, N)
    left  = left.view(B, N, N, H)
    right = right.view(B, N, N, H)

    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # (B·H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # (B·H, N, N)

    # (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]
    out_flat = out.view(R, H).contiguous()          # [R, H] FP16
    out_gate_flat = out_gate                         # [R, H] FP16 (already flat)

    # --------------------------------------------------------------
    # 5️⃣ Fused LayerNorm + out‑gate + final projection (FP32 output)
    # --------------------------------------------------------------
    final = torch.empty(R, dim, dtype=torch.float32, device=device)

    # tiling parameters – they work well for the supported sizes
    BLOCK_M2 = 64          # rows per program
    BLOCK_O  = 64          # output‑dim tile
    BLOCK_D2 = 32          # hidden‑dim tile (≤ hidden_dim)

    grid2 = (triton.cdiv(R, BLOCK_M2), triton.cdiv(dim, BLOCK_O))

    norm_gate_linear_kernel[grid2](
        out_flat,
        out_gate_flat,
        final,
        weights["to_out_norm.weight"],   # γ (float32)  shape [H]
        weights["to_out_norm.bias"],     # β (float32)  shape [H]
        weights["to_out.weight"].to(torch.float16),   # [dim, H]  (float16)
        R,
        H,
        dim,
        eps,
        BLOCK_M=BLOCK_M2,
        BLOCK_O=BLOCK_O,
        BLOCK_D=BLOCK_D2,
        num_warps=8,
        num_stages=2,
    )

    # --------------------------------------------------------------
    # 6️⃣ Reshape to the original 4‑D layout and return
    # --------------------------------------------------------------
    out_tensor = final.view(B, N, N, dim)   # [B, N, N, dim] float32
    return out_tensor

This implementation follows the reference module while moving the costly
second LayerNorm + out‑gate + final projection into a single Triton kernel,
thereby halving the intermediate memory traffic.  The heavy matrix
multiplications (the fused 4 × hidden projection and the N³ batched GEMM)
remain on cuBLAS/TensorCore‑backed PyTorch functions, giving the best
possible throughput on an H100.
