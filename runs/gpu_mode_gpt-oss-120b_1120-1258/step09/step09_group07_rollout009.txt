@triton.jit
def apply_gate_kernel(
    left_raw_ptr, right_raw_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    left_out_ptr, right_out_ptr,
    total_elems,
    BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    l_raw = tl.load(left_raw_ptr + offs, mask=mask).to(tl.float32)
    r_raw = tl.load(right_raw_ptr + offs, mask=mask).to(tl.float32)
    l_gate = tl.load(left_gate_ptr + offs, mask=mask).to(tl.float32)
    r_gate = tl.load(right_gate_ptr + offs, mask=mask).to(tl.float32)
    m_val = tl.load(mask_ptr + offs, mask=mask).to(tl.float32)

    l = l_raw * m_val * tl.sigmoid(l_gate)
    r = r_raw * m_val * tl.sigmoid(r_gate)

    tl.store(left_out_ptr + offs, l.to(tl.float16), mask=mask)
    tl.store(right_out_ptr + offs, r.to(tl.float16), mask=mask)

def custom_kernel(data):
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    B,N,_,_ = x.shape
    total_rows = B * N * N
    H = hidden_dim

    device = x.device

    # 1st LayerNorm using torch.nn.functional
    x_norm = torch.nn.functional.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    ).to(dtype=torch.float16) # cast after

    # 2. fused linear for left,right,left_gate,right_gate
    w_fused = torch.cat([
        weights["left_proj.weight"],
        weights["right_proj.weight"],
        weights["left_gate.weight"],
        weights["right_gate.weight"]
    ], dim=0).to(torch.float16) # shape (4*H, dim)

    fused = torch.nn.functional.linear(x_norm.view(total_rows, dim), w_fused) # [R, 4*H]

    left_raw = fused[:, :H]
    right_raw = fused[:, H:2*H]
    left_gate_raw = fused[:, 2*H:3*H]
    right_gate_raw = fused[:, 3*H:]

    # out_gate linear
    out_gate = torch.nn.functional.linear(
        x_norm.view(total_rows, dim),
        weights["out_gate.weight"].to(torch.float16)
    ).sigmoid_() # [R, H] FP16

    # Apply mask and gates via kernel
    total_elems = total_rows * H

    # prepare mask flat
    if not nomask and mask is not None:
        mask_flat = mask.view(total_rows, 1).to(torch.float16).expand(-1, H).contiguous().view(-1)
    else:
        mask_flat = torch.ones(total_elems, dtype=torch.float16, device=device)

    left = torch.empty_like(left_raw)
    right = torch.empty_like(right_raw)

    BLOCK = 1024  # choose
    grid = (triton.cdiv(total_elems, BLOCK),)
    apply_gate_kernel[grid](
        left_raw,
        right_raw,
        left_gate_raw,
        right_gate_raw,
        mask_flat,
        left,
        right,
        total_elems,
        BLOCK_SIZE=BLOCK,
        num_warps=4,
    )

    # Reshape back for batched mul
    left = left.view(B, N, N, H)
    right = right.view(B, N, N, H)

    # move hidden dim to batch: (B*H, N, N)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B*H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B*H, N, N)

    # batched matmul
    out_mat = torch.bmm(left_mat, right_mat.transpose(1,2))  # (B*H,N,N)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous() # [B,N,N,H]

    # Second layernorm using torch.nn.functional
    out_norm = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    ).to(dtype=torch.float16)

    # Multiply with out_gate via kernel
    out_norm_flat = out_norm.view(total_rows, H)
    out_gate_flat = out_gate.view(total_rows, H)

    out_norm_gate = torch.empty_like(out_norm_flat)
    total_gate_elems = total_rows * H

    grid_gate = (triton.cdiv(total_gate_elems, BLOCK),)
    gate_mul_kernel[grid_gate](
        out_norm_flat,
        out_gate_flat,
        out_norm_gate,
        total_gate_elems,
        BLOCK_SIZE=BLOCK,
        num_warps=4,
    )

    # Final linear projection to dim
    final_fp16 = torch.nn.functional.linear(
        out_norm_gate,
        weights["to_out.weight"].to(torch.float16)
    ) # [R, dim] FP16

    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final

import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def mul_gate_kernel(out_ptr, gate_ptr, out_ptr_gate, total_elems, BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    out_val = tl.load(out_ptr + offs, mask=mask).to(tl.float32)
    gate_val = tl.load(gate_ptr + offs, mask=mask).to(tl.float32)

    res = out_val * gate_val

    tl.store(out_ptr_gate + offs, res.to(tl.float16), mask=mask)


def custom_kernel(data):
    """
    High‑performance forward of the *outgoing* TriMul operator.

    The implementation:
    1. Row‑wise LayerNorm (torch.nn.functional) on the input.
    2. Four linear projections (left/right + gates) are fused into a single
       matmul with a concatenated weight matrix (FP16).
    3. The optional pair‑wise mask and sigmoid gates are applied with
       inexpensive torch element‑wise ops.
    4. The multiplicative update `einsum('... i k d, ... j k d -> ... i j d')`
       is expressed as a batched matrix‑multiply (`torch.bmm`).
    5. A second LayerNorm is performed (torch.nn.functional) on the result.
    6. The output gate is fused with the second LayerNorm using a tiny
       Triton kernel (`mul_gate_kernel`).
    7. Final linear projection back to the original channel dimension.

    All heavy compute stays in FP16; only the final tensor is cast back
    to FP32 to match the reference implementation.
    """
    # Unpack arguments
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    B, N, _, _ = x.shape
    total_rows = B * N * N
    H = hidden_dim
    device = x.device

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32) – use torch functional for speed
    # ------------------------------------------------------------------
    # Output stays FP32; will cast to FP16 afterwards.
    x_norm = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    ).to(dtype=torch.float16)  # [B,N,N,dim] FP16

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection for left/right and their gates
    # ------------------------------------------------------------------
    # Concatenate the four projection matrices (each [H, dim])
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
        ],
        dim=0,
    ).to(dtype=torch.float16)  # shape (4*H, dim)

    fused = F.linear(x_norm.reshape(total_rows, dim), w_fused)  # [R, 4*H]

    left_raw = fused[:, 0:H]
    right_raw = fused[:, H : 2 * H]
    left_gate_raw = fused[:, 2 * H : 3 * H]
    right_gate_raw = fused[:, 3 * H : 4 * H]

    # out‑gate (separate linear)
    out_gate = F.linear(
        x_norm.reshape(total_rows, dim),
        weights["out_gate.weight"].to(torch.float16),
    ).sigmoid_()  # [R, H] FP16

    # ------------------------------------------------------------------
    # 3️⃣ Apply mask (if any) and sigmoid gates – pure torch ops
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask: [B, N, N] → [R, 1] → broadcast over H
        mask_fp = mask.view(total_rows, 1).to(dtype=torch.float16)
        left = left_raw * mask_fp * left_gate_raw.sigmoid_()
        right = right_raw * mask_fp * right_gate_raw.sigmoid_()
    else:
        left = left_raw * left_gate_raw.sigmoid_()
        right = right_raw * right_gate_raw.sigmoid_()

    # ------------------------------------------------------------------
    # 4️⃣ Batched N³ multiplicative update (einsum) via torch.bmm
    # ------------------------------------------------------------------
    # reshape to (B,N,N,H)
    left = left.view(B, N, N, H)
    right = right.view(B, N, N, H)

    # bring hidden dim into batch axis: (B*H, N, N)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # (B*H, N, N) @ (B*H, N, N)^T → (B*H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H]

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm (on hidden dim) – torch functional
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        normalized_shape=(H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    ).to(dtype=torch.float16)

    # ------------------------------------------------------------------
    # 6️⃣ Fuse multiplication by out_gate using a tiny Triton kernel
    # ------------------------------------------------------------------
    out_norm_flat = out_norm.reshape(-1)          # [R*H]
    out_gate_flat = out_gate.reshape(-1)          # [R*H]
    out_norm_gate = torch.empty_like(out_norm_flat)

    total_elems = out_norm_flat.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total_elems, BLOCK),)

    mul_gate_kernel[grid](
        out_norm_flat,
        out_gate_flat,
        out_norm_gate,
        total_elems,
        BLOCK_SIZE=BLOCK,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 7️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    final_fp16 = F.linear(
        out_norm_gate.view(total_rows, H),
        weights["to_out.weight"].to(torch.float16),
    )  # [R, dim] FP16

    # Cast back to FP32 as required by the model
    final = final_fp16.view(B, N, N, dim).to(dtype=torch.float32)
    return final

"""
custom_kernel – high‑performance forward of the *outgoing* TriMul operator.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the input (torch.nn.functional.layer_norm, FP32).
2️⃣ Fuse the four linear projections (left/right + their gates) into a
   single FP16 GEMM with a concatenated weight matrix.
3️⃣ Compute the output‑gate with a second FP16 GEMM.
4️⃣ Apply the optional pair‑wise mask and the sigmoid gates (pure torch ops).
5️⃣ Perform the multiplicative update
       out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   as a batched matrix‑multiply (torch.bmm).
6️⃣ Second LayerNorm over the hidden dimension (torch.nn.functional.layer_norm).
7️⃣ Fuse the multiplication by the output‑gate with a tiny Triton kernel.
8️⃣ Final linear projection back to the original channel dimension
   (FP16 GEMM) and cast the result to FP32.

Only the final gate‑multiplication is written in Triton; all heavy‑compute
steps stay in FP16 and use highly‑optimized PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: element‑wise multiply of a tensor by the output‑gate.
# Both inputs are FP16, computation is done in FP32 for numerical safety.
# ----------------------------------------------------------------------
@triton.jit
def mul_gate_kernel(
    out_ptr,            # FP16 tensor, result of the second LayerNorm (flattened)
    gate_ptr,           # FP16 tensor, out_gate (flattened)
    out_gate_ptr,       # FP16 tensor where the gated result will be written
    total_elems,        # int32 – number of elements (= B·N·N·H)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # Load, promote to FP32, multiply, cast back to FP16
    a = tl.load(out_ptr + offs, mask=mask).to(tl.float32)
    b = tl.load(gate_ptr + offs, mask=mask).to(tl.float32)
    c = a * b
    tl.store(out_gate_ptr + offs, c.to(tl.float16), mask=mask)

# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : dict with all Linear/LayerNorm parameters
        - config_dict  : {"dim": int, "hidden_dim": int,
                         "nomask": bool (optional)}

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    B, N, _, _ = x.shape
    R = B * N * N                 # total number of (b,i,k) rows
    H = hidden_dim
    device = x.device

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) – torch implementation (FP32)
    # ------------------------------------------------------------------
    # Normalise over the last dimension (`dim`), keep FP32 for numeric
    # stability and cast to FP16 afterwards.
    x_norm_fp16 = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    ).to(dtype=torch.float16)          # [B,N,N,dim] FP16

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection for left/right and their gates (FP16 GEMM)
    # ------------------------------------------------------------------
    # Concatenate the four weight matrices: each has shape [H, dim]
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
        ],
        dim=0,
    ).to(dtype=torch.float16)          # (4*H, dim)

    fused = F.linear(
        x_norm_fp16.view(R, dim),       # [R, dim]
        w_fused,                        # [4*H, dim]
    )                                   # → [R, 4*H]

    left_raw       = fused[:, 0:H]
    right_raw      = fused[:, H:2 * H]
    left_gate_raw  = fused[:, 2 * H:3 * H]
    right_gate_raw = fused[:, 3 * H:4 * H]

    # ------------------------------------------------------------------
    # 3️⃣ Output‑gate (separate linear) – also FP16
    # ------------------------------------------------------------------
    out_gate = F.linear(
        x_norm_fp16.view(R, dim),       # [R, dim]
        weights["out_gate.weight"].to(torch.float16),  # [H, dim]
    ).sigmoid_()                       # [R, H] FP16

    # ------------------------------------------------------------------
    # 4️⃣ Apply mask (if any) and sigmoid gates (pure torch ops)
    # ------------------------------------------------------------------
    left_gate = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()

    if not nomask and mask is not None:

        mask_fp = mask.view(R, 1).to(dtype=torch.float16)
        left = left_raw * mask_fp * left_gate
        right = right_raw * mask_fp * right_gate

        right = right_raw * right_gate

    left = left.view(B, N, N, H)

    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H]

    out_norm = F.layer_norm(

    out_norm_flat = out_norm.view(-1)   # size = B·N·N·H
