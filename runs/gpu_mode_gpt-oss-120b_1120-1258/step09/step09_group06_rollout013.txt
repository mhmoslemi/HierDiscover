"""
High‑performance forward pass for the **outgoing** TriMul operator (AlphaFold3).

Algorithm (FP16 heavy):
1️⃣ Row‑wise LayerNorm (flattened tensor) → FP16.
2️⃣ Fuse the four linear layers (left/right projection + left/right gate) into a
   single GEMM, split the result.
3️⃣ Compute the output‑gate with a second GEMM.
4️⃣ Apply optional binary mask and sigmoid‑gates (in‑place).
5️⃣ Batched matrix‑multiply `C = A @ Bᵀ` where `A`/`B` are the left/right
   tensors reshaped as `[B·H, N, N]`.  This is implemented with a custom
   Triton kernel that loops over the `K`‑dimension and uses `tl.dot`
   (FP32 accumulation, FP16 storage).
6️⃣ Row‑wise LayerNorm on the hidden dimension.
7️⃣ Element‑wise output‑gate, final linear projection, cast back to FP32.

Only the two LayerNorms and the cubic update are written in Triton; all
other linear operations use highly‑optimized torch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,              # const  * [total_rows, dim] (fp32)
    y_ptr,              # mutable* [total_rows, dim] (fp16)
    w_ptr,              # const  * [dim]          (fp32) – γ
    b_ptr,              # const  * [dim]          (fp32) – β
    total_rows,         # int (runtime)
    dim,                # int (runtime)
    eps,                # float (compile‑time)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # cols per tile
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)           # [BLOCK_M]
    mask_m = offs_m < total_rows

    # --------- 1. mean & variance ------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        vals = tl.load(
            x_ptr + offs_m[:, None] * dim + offs_d[None, :],
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                                   # fp32 load

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)        # 1 / sqrt(var+eps)

    # --------- 2. write normalized result (fp16) --------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        vals = tl.load(
            x_ptr + offs_m[:, None] * dim + offs_d[None, :],
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]   # (x-μ)/σ

        # γ & β are fp32 → keep in fp32, cast to fp16 at the end
        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)
        out = norm * w[None, :] + b[None, :]            # fp32

        tl.store(
            y_ptr + offs_m[:, None] * dim + offs_d[None, :],
            out.to(tl.float16),
            mask=mask_m[:, None] & mask_d[None, :],
        )


# ----------------------------------------------------------------------
# Batched (A @ Bᵀ) kernel – computes C = A @ Bᵀ for many heads at once
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_T_kernel(
    a_ptr, b_ptr, c_ptr,                # fp16 tensors
    batch_stride_a, batch_stride_b, batch_stride_c,
    M, N, K,                            # all int scalars
    stride_am, stride_ak,                # A: (batch, M, K)
    stride_bk, stride_bn,                # B (non‑transposed) : (batch, N, K)
    stride_cm, stride_cn,                # C: (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    EPS: tl.constexpr = 0.0,
):
    """
    Computes, for each batch‑head `b`:
        C[b, i, j] = Σ_k A[b, i, k] * B[b, j, k]    (i∈M, j∈N, k∈K)
    Input B is not transposed; the kernel loads it as Bᵀ on‑the‑fly.
    Accumulation is performed in fp32, result stored as fp16.
    """
    pid_batch = tl.program_id(0)          # batch*head dimension
    pid_m     = tl.program_id(1)          # tile over rows (M)
    pid_n     = tl.program_id(2)          # tile over cols (N)

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # batch pointers
    a_batch = a_ptr + pid_batch * batch_stride_a
    b_batch = b_ptr + pid_batch * batch_stride_b
    c_batch = c_ptr + pid_batch * batch_stride_c

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K‑tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A[i, k] – shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch + (offs_m[:, None] * stride_am
                            + offs_k[None, :] * stride_ak)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # Bᵀ[k, j] = B[j, k] – shape (BLOCK_K, BLOCK_N)
        b_ptrs = b_batch + (offs_k[:, None] * stride_bk
                            + offs_n[None, :] * stride_bn)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)

        # matrix‑multiply tile
        acc += tl.dot(a, b)                     # (BM, BK) • (BK, BN) → (BM, BN)

    # write C tile
    c_ptrs = c_batch + (offs_m[:, None] * stride_cm
                        + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])

        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : dict of torch.Parameters (float32). Expected keys:
              "norm.weight", "norm.bias",
              "left_proj.weight", "right_proj.weight",
              "left_gate.weight", "right_gate.weight",
              "out_gate.weight",
              "to_out_norm.weight", "to_out_norm.bias",
              "to_out.weight"
        - config_dict  : dict with keys "dim" (int), "hidden_dim" (int),
                         optional "nomask" (bool)

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ---------------------------------------------------------------
    # 0️⃣ unpack & constants
    # ---------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape               # [B, N, N, dim]
    total_rows = B * N * N              # R = B·N·N
    H = hidden_dim

    # ---------------------------------------------------------------
    # 1️⃣ First LayerNorm (fp32 → fp16)
    # ---------------------------------------------------------------
    x_flat = x.view(total_rows, dim).contiguous()      # [R, dim] fp32
    x_norm_fp16 = torch.empty(total_rows, dim,
                              dtype=torch.float16,
                              device=device)

    BLOCK_M = 128            # good baseline for H100
    BLOCK_D = 64
    grid_ln = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],          # γ, fp32
        weights["norm.bias"],            # β, fp32
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # ---------------------------------------------------------------
    # 2️⃣ Fuse left/right projections & their gates (single GEMM)
    # ---------------------------------------------------------------
    # [4·H, dim] weight matrix (FP16)
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)

    fused = F.linear(x_norm_fp16, w_fused)       # [R, 4·H] (fp16)

    left_raw       = fused[:, 0:H]
    right_raw      = fused[:, H:2 * H]
    left_gate_raw  = fused[:, 2 * H:3 * H]
    right_gate_raw = fused[:, 3 * H:4 * H]

    # ---------------------------------------------------------------
    # 3️⃣ Output‑gate (separate GEMM)
    # ---------------------------------------------------------------
    out_gate_raw = F.linear(
        x_norm_fp16,
        weights["out_gate.weight"].to(torch.float16)   # [H, dim]
    )   # [R, H] (fp16)

    # ---------------------------------------------------------------
    # 4️⃣ optional mask & sigmoid gates (in‑place)
    # ---------------------------------------------------------------
    if not nomask and mask is not None:
        # mask: [B, N, N] → [R, 1] and broadcast
        mask_fp16 = mask.reshape(total_rows, 1).to(torch.float16)
        left_raw  = left_raw * mask_fp16
        right_raw = right_raw * mask_fp16

    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    left  = left_raw * left_gate          # [R, H]
    right = right_raw * right_gate        # [R, H]

    # ---------------------------------------------------------------
    # 5️⃣ Cubic update – batched matmul A @ Bᵀ (FP16 kernel)
    # ---------------------------------------------------------------
    # reshape to [B, N, N, H] and bring H into the batch dimension
    left = left.view(B, N, N, H).contiguous()
    right = right.view(B, N, N, H).contiguous()
    out_gate = out_gate.view(B, N, N, H).contiguous()

    # [B, H, N, N] -> flatten batch×head
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    left_mat = left_perm.view(B * H, N, N)   # [BH, N, N]  (fp16)
    right_mat = right_perm.view(B * H, N, N) # [BH, N, N]  (fp16)

    # allocate output tensor
    out_mat = torch.empty_like(left_mat, dtype=torch.float16)

    # strides (in elements)
    batch_stride_a = left_mat.stride(0)    # BH * N * N
    stride_am = left_mat.stride(1)         # N   (row stride)
    stride_ak = left_mat.stride(2)         # 1   (col stride)

    batch_stride_b = right_mat.stride(0)
    stride_bk = right_mat.stride(2)        # 1   (inner‑K stride)
    stride_bn = right_mat.stride(1)        # N   (row stride)

    batch_stride_c = out_mat.stride(0)
    stride_cm = out_mat.stride(1)          # N
    stride_cn = out_mat.stride(2)          # 1

    # kernel launch configuration
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32
    grid_mat = (
        B * H,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    batched_matmul_T_kernel[grid_mat](
        left_mat,
        right_mat,
        out_mat,
        batch_stride_a,
        batch_stride_b,
        batch_stride_c,
        N,                     # M
        N,                     # N
        N,                     # K
        stride_am,
        stride_ak,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # reshape back to [B, N, N, H]
    out = (
        out_mat.view(B, H, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )   # [B, N, N, H]

    # ---------------------------------------------------------------
    # 6️⃣ Second LayerNorm over the hidden dimension (FP16)
    # ---------------------------------------------------------------
    out_flat = out.view(total_rows, H).contiguous()
    out_norm_fp16 = torch.empty_like(out_flat, dtype=torch.float16)

    grid_ln2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid_ln2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],   # γ, fp32
        weights["to_out_norm.bias"],     # β, fp32
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # ---------------------------------------------------------------
    # 7️⃣ Apply output‑gate, final projection, cast back to fp32
    # ---------------------------------------------------------------
    out_norm_fp16 = out_norm_fp16 * out_gate.view(total_rows, H)

    final_fp16 = F.linear(
        out_norm_fp16,
        weights["to_out.weight"].to(torch.float16)   # [dim, H]
    )   # [R, dim] (fp16)

    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final
```
