    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < rows

    # optional mask values
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0).to(tl.float16)[:, None]  # broadcast to write dimension later
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # compute mean, var
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_R], dtype=tl.float32)

    # first pass over dim to compute stats
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_r[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr, mask=mask_r[:, None] & mask_d[None, :], other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals*vals, axis=1)
    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # We'll compute only left/right/outgate
    # allocate temporary variables for outputs
    # because we have many output channels; we compute in chunks

    # Loop over output channel blocks
    for out_off in range(0, tl.cdiv(5*H, BLOCK_OUT)):
        # Determine segment: left (0:H), right (H:2H), left_gate (2H:3H), right_gate (3H:4H), out_gate (4H:5H)
        seg_start = out_off * BLOCK_OUT
        # compute actual block size
        offs_o = seg_start + tl.arange(0, BLOCK_OUT)
        mask_o = offs_o < 5*H

        # partial accumulation per out channel block
        acc = tl.zeros([BLOCK_R, BLOCK_OUT], dtype=tl.float32)

        # second pass over dim to compute dot products
        for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
            offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
            mask_d = offs_d < dim
            # load x normalized values (x - mean) * rstd
            x_ptr_off = x_ptr + offs_r[:, None] * dim + offs_d[None, :]
            x_vals = tl.load(x_ptr_off,
                                mask=mask_r[:, None] & mask_d[None, :],
                                other=0.0).to(tl.float32)
            norm_vals = (x_vals - mean[:, None]) * rstd[:, None]  # shape [R, Dblock]

            # load weights for current output block segment
            # weight_ptr is (5*H, dim) shape (out_channels, dim). Want weight[offs_o:offs_o+BLOCK_OUT, offs_d...]
            w_ptr = weight_proj_ptr + offs_o[:, None] * dim + offs_d[None, :]   # [O_block, Dblock]
            w = tl.load(w_ptr,
                        mask=mask_o[:, None] & mask_d[None, :],
                        other=0.0).to(tl.float32)
            # compute dot product: sum over d of norm_vals * w
            # using tl.dot for each row, but we have block R rows and O out channels.
            # We can compute via outer product? Actually need to compute for each row r and each out channel o: sum_d norm_vals[r,d] * w[o,d].

"""
High‑performance Triton implementation of the *outgoing* TriMul operator
used in AlphaFold‑3 and related protein‑structure models.

Algorithm
---------
1️⃣  Row‑wise LayerNorm on the 4‑D pair representation (B·N·N × C)
    – performed with the highly‑optimised torch.nn.functional.layer_norm.

2️⃣  Fused linear projection that creates the five hidden tensors
    (left, right, left‑gate, right‑gate, out‑gate) in one CUDA launch.
    The kernel
        split_fuse_gate_kernel
    reads the 5·H projection results, applies the optional pairwise mask,
    evaluates the three sigmoids and writes the three final tensors:
    * left  = (proj_left  * mask)  ⊙ sigmoid(proj_left_gate )
    * right = (proj_right * mask)  ⊙ sigmoid(proj_right_gate)
    * out_gate = sigmoid(proj_out_gate)

3️⃣  Tri‑multiplicative update:  left @ rightᵀ  is a batched GEMM of size
    (B·H, N, N) × (B·H, N, N)ᵀ, executed with torch.bmm (cuBLAS tensor‑core).

4️⃣  Second Row‑wise LayerNorm + out‑gate multiplication (again using the
    built‑in layer_norm for speed) followed by the final linear projection
    back to the original channel dimension.

Only the mask‑gate kernel is written in Triton – the rest uses the
well‑optimised PyTorch/cuBLAS primitives, which gives a good balance
between simplicity and the sub‑millisecond target runtime on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 3️⃣ Split the 5·H projection, apply mask, evaluate sigmoids
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[rows, 5*H] (fp16)
    left_ptr,          # mutable *[rows, H]   (fp16)
    right_ptr,         # mutable *[rows, H]   (fp16)
    out_gate_ptr,      # mutable *[rows, H]   (fp16)
    mask_ptr,          # const   *[rows]    (fp16)   (ignored if HAS_MASK=0)
    rows: tl.int32,
    H: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)               # [BLOCK_R]
    mask_r = offs_r < rows

    # ------------------------------------------------------------------
    # optional mask (broadcast to H dim)
    # ------------------------------------------------------------------
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                   # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # ------------------------------------------------------------------
    # Process H‑dim in chunks
    # ------------------------------------------------------------------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)          # [BLOCK_H]
        mask_h = offs_h < H

        # layout of proj: [ left, right, left_gate, right_gate, out_gate ]
        base = offs_r[:, None] * (5 * H) + offs_h[None, :]        # [R,H]

        # pointers for the five contiguous segments
        left_raw_ptr    = proj_ptr + base
        right_raw_ptr   = proj_ptr + base + H
        left_gate_ptr   = proj_ptr + base + 2 * H
        right_gate_ptr  = proj_ptr + base + 3 * H
        out_gate_raw_ptr= proj_ptr + base + 4 * H

        # --------------------------------------------------------------
        # load raw tensors
        # --------------------------------------------------------------
        left_raw   = tl.load(left_raw_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        right_raw  = tl.load(right_raw_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        left_g_raw = tl.load(left_gate_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        right_g_raw= tl.load(right_gate_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        out_g_raw  = tl.load(out_gate_raw_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)

        # --------------------------------------------------------------
        # sigmoid (computed in fp32 for numerical stability)
        # --------------------------------------------------------------
        lg_f32 = left_g_raw.to(tl.float32)
        left_gate = (1.0 / (1.0 + tl.exp(-lg_f32))).to(tl.float16)

        rg_f32 = right_g_raw.to(tl.float32)
        right_gate = (1.0 / (1.0 + tl.exp(-rg_f32))).to(tl.float16)

        og_f32 = out_g_raw.to(tl.float32)
        out_gate = (1.0 / (1.0 + tl.exp(-og_f32))).to(tl.float16)

        # --------------------------------------------------------------
        # apply mask, gates and store
        # --------------------------------------------------------------
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate

        left_dst_ptr    = left_ptr    + offs_r[:, None] * H + offs_h[None, :]
        right_dst_ptr   = right_ptr   + offs_r[:, None] * H + offs_h[None, :]
        out_gate_dst_ptr= out_gate_ptr+ offs_r[:, None] * H + offs_h[None, :]

        tl.store(left_dst_ptr,    left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_dst_ptr,   right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_dst_ptr,out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])

    # --------------------------------------------------------------

    hidden_dim = cfg["hidden_dim"]

    H = hidden_dim

    # ------------------------------------------------------------------

            weights["out_gate.weight"],

        right,
        out_gate,
        mask_ptr,
        R,
        H,
        HAS_MASK=has_mask,

        BLOCK_H=BLOCK_H,
        num_warps=4,

    # ------------------------------------------------------------------

    left = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    left_mat = left.view(B * H, N, N)      # [B·H, N, N]
    right_mat = right.view(B * H, N, N)    # [B·H, N, N]

    # (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm + out‑gate multiplication (PyTorch)
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        normalized_shape=(H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )                                               # fp32

    # out_gate is fp16 → cast to fp32 for the element‑wise product
    out_gate_fp32 = out_gate.view_as(out_norm).to(torch.float32)
    gated = out_norm * out_gate_fp32                 # [B, N, N, H]

    # ------------------------------------------------------------------
    # 6️⃣ Final projection back to ``dim``
    # ------------------------------------------------------------------
    final = F.linear(
        gated.view(R, H),
        weights["to_out.weight"].to(torch.float16),   # [dim, H]
        bias=None,
    )   # → [R, dim]  fp16

    final = final.view(B, N, N, dim).to(torch.float32)
    return final
```
