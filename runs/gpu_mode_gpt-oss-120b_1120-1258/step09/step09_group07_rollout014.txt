@triton.jit
def gate_fuse_kernel(
    fused_ptr,    # [R, 5*H] fp16 raw results (left_proj, right_proj, left_gate, right_gate, out_gate)
    mask_ptr,    # [R] (optional) fp16 mask (scalar)
    left_out_ptr,
    right_out_ptr,
    out_gate_ptr,
    R, H,
    BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < R

    # Process per row and per hidden in loops
    # Iterate over hidden groups block_h
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H
        # compute linear indices for each segment
        # Offsets for left_proj: base = 0
        # Right_proj: base = H
        # left_gate: base = 2*H
        # right_gate: base = 3*H
        # out_gate: base = 4*H

        # fetch vectors
        left_raw = tl.load(fused_ptr + offs_m[:, None] * (5*H) + offs_h[None, :],
                            mask=mask_m[:, None] & mask_h[None, :], other=0.0)
        right_raw = tl.load(fused_ptr + offs_m[:, None] * (5*H) + (H + offs_h)[None, :],
                            ...)
        left_gate_raw = tl.load(... + (2*H + offs_h)[None,:])
        right_gate_raw = tl.load(... + (3*H + offs_h)[None,:])
        out_gate_raw = tl.load(... + (4*H + offs_h)[None,:])

        # mask value (broadcast) - if mask_ptr provided.
        if mask_ptr is not None:
            mask_val = tl.load(mask_ptr + offs_m[:, None], mask=mask_m[:, None], other=1.0)  # shape [BLOCK_M,1]
            # broadcast to hidden
            mask_vec = mask_val  # shape [BLOCK_M,1]; will broadcast in multiplication
        else:
            mask_vec = 1.0

        # apply sigmoid: use tl.sigmoid
        left_gate = tl.sigmoid(left_gate_raw)
        right_gate = tl.sigmoid(right_gate_raw)
        out_gate = tl.sigmoid(out_gate_raw)

        # compute final left and right with mask and gates
        left_out = left_raw * mask_vec * left_gate
        right_out = right_raw * mask_vec * right_gate

        # store results
        tl.store(left_out_ptr + offs_m[:, None] * H + offs_h[None, :],
                 left_out, mask=mask_m[:, None] & mask_h[None, :])
        tl.store(right_out_ptr + offs_m[:, None] * H + offs_h[None, :],
                 right_out, mask=mask_m[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + offs_m[:, None] * H + offs_h[None, :],
                 out_gate, mask=mask_m[:, None] & mask_h[None, :])

@triton.jit
def fused_gate_kernel(
    fused_ptr, mask_ptr,
    left_ptr, right_ptr, out_gate_ptr,
    R, H,
    stride_fused_r, stride_fused_c,
    stride_left_r, stride_left_c,
    stride_right_r, stride_right_c,
    stride_out_gate_r, stride_out_gate_c,
    stride_mask_r,
    BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < R

    # compute loops over hidden dim
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # load raw values for each segment
        # fused layout: columns [0:H] left_raw, [H:2H] right_raw, [2H:3H] left_gate_raw, [3H:4H] right_gate_raw, [4H:5H] out_gate_raw
        # compute col offsets
        left_raw = tl.load(fused_ptr + offs_m[:, None] * stride_fused_r + (offs_h)[None, :] * stride_fused_c,
                           mask=mask_m[:, None] & mask_h[None, :], other=0.0)
        right_raw = tl.load(fused_ptr + offs_m[:, None] * stride_fused_r + (H + offs_h)[None, :] * stride_fused_c,
                           mask=mask_m[:, None] & mask_h[None, :], other=0.0)
        left_gate_raw = tl.load(fused_ptr + offs_m[:, None] * stride_fused_r + (2*H + offs_h)[None, :] * stride_fused_c,
                               mask=mask_m[:, None] & mask_h[None, :], other=0.0)
        right_gate_raw = tl.load(fused_ptr + offs_m[:, None] * stride_fused_r + (3*H + offs_h)[None, :] * stride_fused_c,
                                mask=mask_m[:, None] & mask_h[None, :], other=0.0)
        out_gate_raw = tl.load(fused_ptr + offs_m[:, None] * stride_fused_r + (4*H + offs_h)[None, :] * stride_fused_c,
                               mask=mask_m[:, None] & mask_h[None, :], other=0.0)
        # load mask if provided
        mask_val = tl.load(mask_ptr + offs_m * stride_mask_r,
                           mask=mask_m, other=1.0)  # shape [BLOCK_M]
        mask_vec = mask_val[:, None]  # broadcast

        # apply sigmoid
        left_gate = tl.sigmoid(left_gate_raw)
        right_gate = tl.sigmoid(right_gate_raw)
        out_gate = tl.sigmoid(out_gate_raw)

        # compute final left, right
        left = left_raw * left_gate * mask_vec
        right = right_raw * right_gate * mask_vec

        # store
        tl.store(left_ptr + offs_m[:, None] * stride_left_r + offs_h[None, :] * stride_left_c,
                 left, mask=mask_m[:, None] & mask_h[None, :])
        tl.store(right_ptr + offs_m[:, None] * stride_right_r + offs_h[None, :] * stride_right_c,
                 right, mask=mask_m[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + offs_m[:, None] * stride_out_gate_r + offs_h[None, :] * stride_out_gate_c,
                 out_gate, mask=mask_m[:, None] & mask_h[None, :])

"""
custom_kernel: high‑performance forward of the *outgoing* TriMul operator.

Algorithm
---------
1️⃣  Row‑wise LayerNorm on the input (FP32→FP16) using
    ``torch.nn.functional.layer_norm`` – highly‑optimized cuDNN/CUDA
    implementation.

2️⃣  Fuse the four linear projections (left/right) + the three gates
    (left_gate, right_gate, out_gate) into one GEMM:
        X_norm @ W_fused.T  →  [R, 5·H]   (R = B·N·N, H = hidden_dim)

3️⃣  One Triton kernel (``gate_fuse_kernel``) consumes the fused GEMM
    result and produces three tensors:
        left   = (left_raw   * mask) * sigmoid(left_gate_raw)
        right  = (right_raw  * mask) * sigmoid(right_gate_raw)
        out_gate = sigmoid(out_gate_raw)
   This eliminates three extra element‑wise passes.

4️⃣  Compute the multiplicative update with a single
    ``torch.einsum`` (no permute/contigu‑ify copies):
        out = einsum('b i k h, b j k h -> b i j h', left, right)

5️⃣  Fuse the second LayerNorm (over the hidden dimension) and the
    multiplication by ``out_gate`` in ``layernorm_gate_kernel`` (FP16).

6️⃣  Final projection back to the original channel dimension with a
    lightweight FP16 GEMM.

All heavy memory‑bound work is kept in FP16, only the final output is
cast back to FP32.  The implementation works for any supported
configuration (B∈{1,2}, N≤1024, dim∈{128,384}, hidden_dim=128) and obeys
the “use Triton at least once” rule.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fused mask + sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def gate_fuse_kernel(
    fused_ptr,               # const  * [R, 5*H]  (fp16)   raw GEMM output
    mask_ptr,                # const  * [R]       (fp16)   0/1 mask (all‑ones if no mask)
    left_ptr,                # mutable* [R, H]    (fp16)   left after mask & gate
    right_ptr,               # mutable* [R, H]    (fp16)   right after mask & gate
    out_gate_ptr,            # mutable* [R, H]    (fp16)   sigmoid(out_gate_raw)
    R, H,                    # runtime sizes
    stride_fused_r, stride_fused_c,
    stride_mask_r,
    stride_left_r, stride_left_c,
    stride_right_r, stride_right_c,
    stride_out_gate_r, stride_out_gate_c,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_H: tl.constexpr,   # hidden‑dim slice per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)               # [BLOCK_M]
    mask_m = offs_m < R

    # ------------------------------------------------------------------
    # Loop over the hidden dimension in chunks of BLOCK_H
    # ------------------------------------------------------------------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)    # [BLOCK_H]
        mask_h = offs_h < H

        # ------------------------------------------------------------------
        # Load raw GEMM slices
        #   layout in fused: 0:H → left_raw
        #                     H:2H → right_raw
        #                     2H:3H → left_gate_raw
        #                     3H:4H → right_gate_raw
        #                     4H:5H → out_gate_raw
        # ------------------------------------------------------------------
        base = offs_m[:, None] * stride_fused_r
        left_raw      = tl.load(fused_ptr + base + offs_h[None, :] * stride_fused_c,
                                mask=mask_m[:, None] & mask_h[None, :], other=0.0)
        right_raw     = tl.load(fused_ptr + base + (H + offs_h)[None, :] * stride_fused_c,
                                mask=mask_m[:, None] & mask_h[None, :], other=0.0)
        left_gate_raw = tl.load(fused_ptr + base + (2*H + offs_h)[None, :] * stride_fused_c,
                                mask=mask_m[:, None] & mask_h[None, :], other=0.0)
        right_gate_raw= tl.load(fused_ptr + base + (3*H + offs_h)[None, :] * stride_fused_c,
                                mask=mask_m[:, None] & mask_h[None, :], other=0.0)
        out_gate_raw  = tl.load(fused_ptr + base + (4*H + offs_h)[None, :] * stride_fused_c,
                                mask=mask_m[:, None] & mask_h[None, :], other=0.0)

        # ------------------------------------------------------------------
        # Load mask (scalar per row) – broadcast across hidden dim
        # ------------------------------------------------------------------
        mask_val = tl.load(mask_ptr + offs_m * stride_mask_r, mask=mask_m, other=1.0)  # [BLOCK_M]
        mask_vec = mask_val[:, None]                     # broadcast to [BLOCK_M,1]

        # ------------------------------------------------------------------
        # Sigmoid (implemented as 1/(1+exp(-x))) and final element‑wise ops
        # ------------------------------------------------------------------
        left_gate   = 1.0 / (1.0 + tl.exp(-left_gate_raw))
        right_gate  = 1.0 / (1.0 + tl.exp(-right_gate_raw))
        out_gate    = 1.0 / (1.0 + tl.exp(-out_gate_raw))

        left   = left_raw  * left_gate  * mask_vec
        right  = right_raw * right_gate * mask_vec

        # ------------------------------------------------------------------
        # Store results
        # ------------------------------------------------------------------
        tl.store(left_ptr      + offs_m[:, None] * stride_left_r      + offs_h[None, :] * stride_left_c,
                 left,      mask=mask_m[:, None] & mask_h[None, :])
        tl.store(right_ptr     + offs_m[:, None] * stride_right_r     + offs_h[None, :] * stride_right_c,
                 right,     mask=mask_m[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr  + offs_m[:, None] * stride_out_gate_r + offs_h[None, :] * stride_out_gate_c,
                 out_gate,  mask=mask_m[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Triton kernel: LayerNorm + element‑wise gate (used for the second norm)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_gate_kernel(
    x_ptr,          # const  * [R, H] (fp16)
    gate_ptr,       # const  * [R, H] (fp16)
    y_ptr,          # mutable* [R, H] (fp16)
    weight_ptr,     # const  * [H]   (fp32)   γ
    bias_ptr,       # const  * [H]   (fp32)   β
    R, H,           # runtime sizes
    eps,            # compile‑time constant
    stride_x_r, stride_x_c,
    stride_gate_r, stride_gate_c,
    stride_y_r, stride_y_c,
    stride_w, stride_b,
    BLOCK_M: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)               # [BLOCK_M]
    mask_m = offs_m < R

    # ------------------------------------------------------------------
    # Compute mean & variance per row (in FP32)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H
        ptr = x_ptr + offs_m[:, None] * stride_x_r + offs_h[None, :] * stride_x_c
        vals = tl.load(ptr, mask=mask_m[:, None] & mask_h[None, :], other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------------------------------------------------------
    # Normalise, apply γ/β and multiply by gate
    # ------------------------------------------------------------------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # load activation and its gate
        x_vals = tl.load(x_ptr    + offs_m[:, None] * stride_x_r    + offs_h[None, :] * stride_x_c,
                         mask=mask_m[:, None] & mask_h[None, :], other=0.0).to(tl.float32)
        g_vals = tl.load(gate_ptr + offs_m[:, None] * stride_gate_r + offs_h[None, :] * stride_gate_c,
                         mask=mask_m[:, None] & mask_h[None, :], other=0.0).to(tl.float32)

        norm = (x_vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_h, mask=mask_h, other=0.0)   # γ  (fp32)
        b = tl.load(bias_ptr   + offs_h, mask=mask_h, other=0.0)   # β  (fp32)

        out = (norm * w[None, :] + b[None, :]) * g_vals
        out_fp16 = out.to(tl.float16)

        tl.store(y_ptr + offs_m[:, None] * stride_y_r + offs_h[None, :] * stride_y_c,
                 out_fp16,
                 mask=mask_m[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
            - weights_dict : mapping containing all Linear/LayerNorm parameters
            - config_dict  : {"dim": int, "hidden_dim": int,
                             "nomask": bool (optional)}

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    total_rows = B * N * N          # R = B·N·N
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (Row‑wise) – fp16 output
    # ------------------------------------------------------------------
    # Convert to fp16 once and run the cuDNN LayerNorm (very fast)
    x_fp16 = x.to(torch.float16)
    norm_weight_fp16 = weights["norm.weight"].to(torch.float16)
    norm_bias_fp16   = weights["norm.bias"].to(torch.float16)
    x_norm = F.layer_norm(
        x_fp16,
        (dim,),
        weight=norm_weight_fp16,
        bias=norm_bias_fp16,
        eps=eps,
    )                                 # [B, N, N, dim] fp16

    # ------------------------------------------------------------------
    # 2️⃣ Fuse all five linear maps into one GEMM
    # ------------------------------------------------------------------
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],    # [H, dim]
            weights["right_proj.weight"],   # [H, dim]
            weights["left_gate.weight"],    # [H, dim]
            weights["right_gate.weight"],   # [H, dim]
            weights["out_gate.weight"],    # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)                # [5·H, dim]

    # flatten batch‑spatial dims for the GEMM
    x_norm_flat = x_norm.reshape(total_rows, dim)          # [R, dim]
    fused = F.linear(x_norm_flat, w_fused)                 # [R, 5·H]

    # ------------------------------------------------------------------
    # 3️⃣ Fuse mask & sigmoid gates (single Triton kernel)
    # ------------------------------------------------------------------
    if nomask or mask is None:
        mask_fp16 = torch.ones(total_rows, dtype=torch.float16, device=device)
    else:
        mask_fp16 = mask.to(torch.float16).reshape(total_rows)  # [R]

    left_out  = torch.empty(total_rows, H, dtype=torch.float16, device=device)
    right_out = torch.empty(total_rows, H, dtype=torch.float16, device=device)
    out_gate  = torch.empty(total_rows, H, dtype=torch.float16, device=device)

    # launch configuration
    BLOCK_M = 128          # rows per program (tuned for H100)
    BLOCK_H = 64           # hidden‑dim slice per iteration
    grid = (triton.cdiv(total_rows, BLOCK_M),)

    gate_fuse_kernel[grid](
        fused, mask_fp16,
        left_out, right_out, out_gate,
        total_rows, H,
        fused.stride(0), fused.stride(1),
        mask_fp16.stride(0),
        left_out.stride(0), left_out.stride(1),
        right_out.stride(0), right_out.stride(1),
        out_gate.stride(0), out_gate.stride(1),
        BLOCK_M=BLOCK_M,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )

    # reshape to [B, N, N, H] (no copy – view works because tensors are contiguous)
    left  = left_out.view(B, N, N, H)
    right = right_out.view(B, N, N, H)
    out_gate = out_gate.view(B, N, N, H)

    # ------------------------------------------------------------------
    # 4️⃣ Multiplicative update (einsum – avoids permute/contiguous copies)
    # ------------------------------------------------------------------
    out = torch.einsum('b i k h, b j k h -> b i j h', left, right)   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 5️⃣ Fuse second LayerNorm + multiplication by the out‑gate
    # ------------------------------------------------------------------
    out_flat = out.reshape(total_rows, H).contiguous()       # [R, H] fp16
    out_gate_flat = out_gate.reshape(total_rows, H).contiguous()

    out_norm_gate = torch.empty_like(out_flat, dtype=torch.float16, device=device)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_gate_kernel[grid2](
        out_flat, out_gate_flat, out_norm_gate,
        weights["to_out_norm.weight"],   # γ (fp32)
        weights["to_out_norm.bias"],     # β (fp32)
        total_rows, H,
        eps,
        out_flat.stride(0), out_flat.stride(1),
        out_gate_flat.stride(0), out_gate_flat.stride(1),
        out_norm_gate.stride(0), out_norm_gate.stride(1),
        weights["to_out_norm.weight"].stride(0), weights["to_out_norm.bias"].stride(0),
        BLOCK_M=BLOCK_M,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Final linear projection back to the original dimension
    # ------------------------------------------------------------------
    to_out_weight_fp16 = weights["to_out.weight"].to(torch.float16)   # [dim, H]
    final_fp16 = F.linear(out_norm_gate, to_out_weight_fp16)         # [R, dim] fp16

    final = final_fp16.view(B, N, N, dim).to(torch.float32)          # cast back to FP32
    return final
