def custom_kernel(data):

    dim = config["dim"]; hidden_dim = config["hidden_dim"]; eps = 1e-5; device = input_tensor.device

    H = hidden_dim

    # 1) first LayerNorm using torch.nn.functional.layer_norm
    x_flat = input_tensor.view(R, dim)
    x_norm_fp32 = F.layer_norm(x_flat, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps)
    x_norm = x_norm_fp32.to(torch.float16)

    # 2) fused projection
    w_fused = torch.cat([...], dim=0).to(torch.float16) # shape (5*H, dim)
    proj = F.linear(x_norm, w_fused) # fp16 shape (R,5H)

    # 3) split and gate using Triton
    left = torch.empty(R, H, dtype=torch.float16, device=device)
    right = torch.empty(R, H, dtype=torch.float16, device=device)
    out_gate = torch.empty(R, H, dtype=torch.float16, device=device)

    if config.get("nomask", True):
        mask_flat = None
        HAS_MASK = 0
    else:
        mask_flat = mask.to(torch.float16).reshape(R) # [R]
        HAS_MASK = 1

    # Determine block sizes
    BLOCK_R = 256
    BLOCK_H = 64
    grid = (triton.cdiv(R, BLOCK_R),)

    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_flat if mask_flat is not None else torch.tensor([], dtype=torch.float16, device=device),
        R,
        H,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    # free

    # 4) batched GEMM
    left = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    left_mat = left.view(B*H, N, N)
    right_mat = right.view(B*H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1,2))
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous() # [B,N,N,H]

    # 5) second layernorm using torch.nn.functional.layer_norm
    out_flat = out.view(R, H)

"""
High‑performance outgoing TriMul (AlphaFold3) forward pass.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the 4‑D tensor (B·N·N × dim) – done with
   torch.nn.functional.layer_norm (fp32 → fp16).

2️⃣ Fuse the five linear projections (left, right, left_gate, right_gate,
   out_gate) into a single GEMM:
        (R, dim) @ (5·H, dim)ᵀ  →  (R, 5·H)   where  R = B·N·N.
   Weight matrix is pre‑concatenated and kept in FP16.

3️⃣ Triton kernel `split_fuse_gate_kernel`:
   * Splits the fused projection into three tensors (left, right, out_gate).
   * Applies the optional pair‑wise mask.
   * Computes sigmoid gates (fp32 → fp16) and multiplies them.
   * All operations are fused and stay in FP16.

4️⃣ Tri‑multiplicative update:
       out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   Implemented as a batched GEMM (B·H, N, N) @ (B·H, N, N)ᵀ
   using torch.bmm – exploits Tensor‑Cores on H100.

5️⃣ Second row‑wise LayerNorm (fp32 → fp16) via torch.nn.functional.layer_norm,
   followed by element‑wise multiplication with `out_gate`.

6️⃣ Final linear projection back to the original `dim` (FP16 → FP32).

Only the split‑&‑gate step is written in Triton (requirement satisfied);
all heavy linear algebra uses cuBLAS/Tensor‑Core kernels for maximal throughput.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: split fused projection, apply mask & sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[rows, 5*H]   (fp16)
    left_ptr,          # mutable *[rows, H]     (fp16)
    right_ptr,         # mutable *[rows, H]     (fp16)
    out_gate_ptr,      # mutable *[rows, H]     (fp16)
    mask_ptr,          # const   *[rows]       (fp16) (ignored if HAS_MASK=0)
    rows: tl.int32,
    H: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,   # rows per program
    BLOCK_H: tl.constexpr,   # hidden dim per iteration
):
    """
    For each row r (r ≡ (b,i,k) in the original tensor):
        left  = (proj_left   * mask) * sigmoid(proj_left_gate)
        right = (proj_right  * mask) * sigmoid(proj_right_gate)
        out_gate = sigmoid(proj_out_gate)

    All tensors are FP16; sigmoid is evaluated in FP32 for numerical stability.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < rows

    # ------------------------------------------------------------------
    # Load per‑row scalar mask (broadcast later) – only when HAS_MASK=1
    # ------------------------------------------------------------------
    if HAS_MASK:
        m = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        m = m[:, None]                                          # [BLOCK_R,1]
    else:
        m = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # ------------------------------------------------------------------
    # Iterate over the hidden dimension
    # ------------------------------------------------------------------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # Offsets within the fused projection (layout described below)
        #   0 .. H-1               -> left
        #   H .. 2H-1              -> right
        #   2H .. 3H-1             -> left_gate
        #   3H .. 4H-1             -> right_gate
        #   4H .. 5H-1             -> out_gate
        row_offset = offs_r[:, None] * (5 * H)             # [R,1]
        col_offset = offs_h[None, :]                       # [1,H]

        left_raw   = tl.load(proj_ptr + row_offset + col_offset,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)

        right_raw  = tl.load(proj_ptr + row_offset + H + col_offset,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)

        left_g_raw = tl.load(proj_ptr + row_offset + 2 * H + col_offset,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)

        right_g_raw = tl.load(proj_ptr + row_offset + 3 * H + col_offset,
                              mask=mask_r[:, None] & mask_h[None, :],
                              other=0.0).to(tl.float16)

        out_g_raw   = tl.load(proj_ptr + row_offset + 4 * H + col_offset,
                              mask=mask_r[:, None] & mask_h[None, :],
                              other=0.0).to(tl.float16)

        # ----- sigmoid in fp32 -----
        left_gate  = (1.0 / (1.0 + tl.exp(-left_g_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_g_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_g_raw.to(tl.float32)))).to(tl.float16)

        # ----- apply mask and gates -----
        left  = left_raw  * m * left_gate
        right = right_raw * m * right_gate

        # ----- store results -----
        tl.store(left_ptr    + offs_r[:, None] * H + offs_h[None, :],
                 left,  mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr   + offs_r[:, None] * H + offs_h[None, :],
                 right, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr+ offs_r[:, None] * H + offs_h[None, :],
                 out_gate, mask=mask_r[:, None] & mask_h[None, :])

# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
            - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
            - weights_dict : mapping containing all linear / layer‑norm parameters
                             (float32 tensors, bias=False for Linear)
            - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim        = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps        = 1e-5
    device     = x.device

    B, N, _, _ = x.shape
    R = B * N * N                    # total rows for row‑wise ops
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32 → FP16) using the highly‑optimized torch kernel
    # ------------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()                 # [R, dim] FP32
    x_norm_fp32 = F.layer_norm(
        x_flat,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )
    x_norm = x_norm_fp32.to(torch.float16)                # [R, dim] FP16

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear projection (left, right, gates) in a single GEMM
    # ------------------------------------------------------------------
    # concatenate the five weight matrices: (5*H, dim)
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                   # FP16

    # (R, dim) @ (5*H, dim)ᵀ → (R, 5*H)
    proj = F.linear(x_norm, w_fused)                      # FP16
    del w_fused, x_norm                                    # free memory early

    # ------------------------------------------------------------------
    # 3️⃣ Split fused tensor, apply mask, compute sigmoid gates (Triton)
    # ------------------------------------------------------------------
    left      = torch.empty(R, H, dtype=torch.float16, device=device)
    right     = torch.empty(R, H, dtype=torch.float16, device=device)
    out_gate  = torch.empty(R, H, dtype=torch.float16, device=device)

    if cfg.get("nomask", True):
        # No mask – compile‑time constant disables the mask path
        mask_flat = torch.empty(0, dtype=torch.float16, device=device)
        HAS_MASK  = 0
    else:
        mask_flat = mask.to(torch.float16).reshape(R)    # [R]
        HAS_MASK  = 1

    BLOCK_R = 256    # rows per program (tuned for occupancy)
    BLOCK_H = 64     # hidden dim per iteration
    grid = (triton.cdiv(R, BLOCK_R),)

    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_flat,
        R,
        H,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,               # H100 can profit from 8 warps per program
    )
    del proj, mask_flat

    # ------------------------------------------------------------------
    # 4️⃣ Tri‑multiplicative update (batched GEMM over hidden dimension)
    # ------------------------------------------------------------------
    # reshape to [B, N, N, H] → [B, H, N, N] for batched matmul
    left = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    left_mat  = left.view(B * H, N, N)   # [B·H, N, N]
    right_mat = right.view(B * H, N, N)  # [B·H, N, N]

    # (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N)   – uses Tensor‑Cores
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to original 4‑D layout (FP16)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm + multiplication with out_gate
    # ------------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()
    out_norm_fp32 = F.layer_norm(
        out_flat,
        normalized_shape=(H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out_norm = out_norm_fp32.to(torch.float16)          # FP16
    del out_flat, out

    # element‑wise gate (FP16)
    out_norm = out_norm * out_gate
    del out_gate

    # ------------------------------------------------------------------
    # 6️⃣ Final projection back to `dim`
    # ------------------------------------------------------------------
    final_half = F.linear(
        out_norm,
        weights["to_out.weight"].to(torch.float16),    # (dim, H)
    )   # FP16, shape (R, dim)

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
