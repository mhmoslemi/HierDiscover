"""
High‑performance outgoing TriMul (AlphaFold3) forward pass.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the flattened [B·N·N, dim] input (float32 → float16)
   – custom Triton kernel (two‑pass: mean/var then normalize).

2️⃣ Fuse the four linear layers (left/right projection + left/right gate) into a
   single half‑precision GEMM:
   X_norm @ [W_left;W_right;W_left_gate;W_right_gate]ᵀ → split into
   left_raw, right_raw, left_gate_raw, right_gate_raw.

3️⃣ Compute the output‑gate with a second half‑precision GEMM.

4️⃣ (Optional) apply binary mask to left/right after projection, then apply
   sigmoid gates (in‑place) and combine: left = left_raw * sigmoid(left_gate),
   right = right_raw * sigmoid(right_gate).

5️⃣ Cubic multiplicative update:
   reshape left/right to [B, H, N, N] → view as (B·H, N, N) and perform a
   batched matrix‑multiply (N×N) @ (N×N)ᵀ with torch.bmm (Tensor‑Core FP16).

6️⃣ Second LayerNorm over the hidden dimension (FP16 → FP16) – same Triton kernel.

7️⃣ Element‑wise multiplication with the output‑gate, final half‑precision
   linear projection to the original dim and cast back to float32.

All heavy compute is done in FP16; two LayerNorms are hand‑written Triton
kernels. The implementation works for batch sizes 1‑2, sequence lengths up
to 1024, dim ∈ {128,384,768} and hidden_dim = 128.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton row‑wise LayerNorm (input may be fp32 or fp16, output is fp16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const  * [total_rows, dim]   (fp32 or fp16)
    y_ptr,          # mutable* [total_rows, dim]   (fp16)
    w_ptr,          # const  * [dim]               (fp32) – gamma
    b_ptr,          # const  * [dim]               (fp32) – beta
    total_rows,     # int (runtime)
    dim,            # int (runtime)
    eps,            # float (compile‑time)
    BLOCK_M: tl.constexpr,  # rows per program
    BLOCK_D: tl.constexpr,  # cols per tile
):
    """
    Row‑wise LayerNorm:
    y = ((x - μ) / sqrt(σ² + eps)) * γ + β
    The kernel processes BLOCK_M rows at a time and iterates over the
    channel dimension in BLOCK_D‑wide tiles.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ----- 1. Compute mean & variance ------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        # load a tile of the input
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # fp32 (or cast from fp16)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)        # 1 / sqrt(var + eps)

    # ----- 2. Write normalized output (fp16) -----------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]      # (x-μ)/σ

        # γ and β are fp32 → keep in fp32, cast final result to fp16
        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)
        out = norm * w[None, :] + b[None, :]               # fp32

        out_fp16 = out.to(tl.float16)
        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom kernel entry‑point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : dict of torch.Parameters (float32) with keys:
              "norm.weight", "norm.bias",
              "left_proj.weight", "right_proj.weight",
              "left_gate.weight", "right_gate.weight",
              "out_gate.weight", "to_out_norm.weight",
              "to_out_norm.bias", "to_out.weight"
        - config_dict  : dict with keys:
              "dim" (int), "hidden_dim" (int), "nomask" (bool, optional)

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # 0️⃣ unpack & constants
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    total_rows = B * N * N          # R = B·N·N
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (fp32/ fp16 → fp16)
    # ------------------------------------------------------------------
    x_flat = x.view(total_rows, dim).contiguous()                     # [R, dim] fp32
    x_norm_fp16 = torch.empty(total_rows, dim,
                              dtype=torch.float16,
                              device=device)

    BLOCK_M = 128       # rows per program (good baseline on H100)
    BLOCK_D = 64        # columns per tile
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],   # γ, fp32
        weights["norm.bias"],     # β, fp32
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fuse left/right projections & their gates into one GEMM (FP16)
    # ------------------------------------------------------------------

        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)                     # [4·H, dim]

    fused = F.linear(x_norm_fp16, w_fused)   # [R, 4·H] (fp16)

    left_raw       = fused[:, 0:H]
    right_raw      = fused[:, H:2*H]
    left_gate_raw  = fused[:, 2*H:3*H]
    right_gate_raw = fused[:, 3*H:4*H]

    # ------------------------------------------------------------------
    # 3️⃣ Output‑gate (separate GEMM)
    # ------------------------------------------------------------------
    out_gate_raw = F.linear(
        x_norm_fp16,
        weights["out_gate.weight"].to(torch.float16)   # [H, dim]
    )   # [R, H] (fp16)

    # ------------------------------------------------------------------
    # 4️⃣ (Optional) mask & sigmoid gates (in‑place)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask: [B, N, N] → [R, 1] → broadcast over hidden dimension
        mask_fp16 = mask.reshape(total_rows, 1).to(torch.float16)
        left_raw = left_raw * mask_fp16
        right_raw = right_raw * mask_fp16

    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    left  = left_raw * left_gate      # [R, H]
    right = right_raw * right_gate    # [R, H]

    # ------------------------------------------------------------------
    # 5️⃣ Batched cubic update  (left @ rightᵀ) – FP16 Tensor‑Core GEMM
    # ------------------------------------------------------------------
    # reshape to [B, N, N, H]
    left = left.view(B, N, N, H)
    right = right.view(B, N, N, H)
    out_gate = out_gate.view(B, N, N, H)

    # bring H into the batch dimension: [B, H, N, N]
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    # fuse B & H → strided‑batched GEMM of shape (B·H, N, N)
    left_mat = left_perm.view(B * H, N, N)
    right_mat = right_perm.view(B * H, N, N)

    # (N×N) @ (N×N)ᵀ → [B·H, N, N]   (FP16, Tensor‑Core)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back → [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm over the hidden dimension (fp16)
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, H).contiguous()
    out_norm_fp16 = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],   # γ, fp32
        weights["to_out_norm.bias"],     # β, fp32
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 7️⃣ Apply output‑gate, final linear projection, cast back to fp32
    # ------------------------------------------------------------------
    # element‑wise gating (still fp16)
    out_norm_fp16 = out_norm_fp16 * out_gate.view(total_rows, H)

    # final projection: [R, H] @ [dim, H]ᵀ → [R, dim]  (fp16)
    final_fp16 = F.linear(
        out_norm_fp16,
        weights["to_out.weight"].to(torch.float16)   # [dim, H]
    )   # [R, dim] (fp16)

    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final
