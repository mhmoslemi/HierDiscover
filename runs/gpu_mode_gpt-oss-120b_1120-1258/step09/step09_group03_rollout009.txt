@triton.jit
def split_fuse_gate_kernel(
    proj_ptr, left_ptr, right_ptr, out_gate_ptr, mask_ptr,
    rows: tl.int32, H: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < rows

    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0).to(tl.float16)
        mask_val = mask_val[:, None]    # [BLOCK_R, 1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        base = offs_r[:, None] * (5 * H) + offs_h[None, :]    # row * width + col

        # load projection chunk for left, right, gates
        left_raw      = tl.load(proj_ptr + base,                     mask=mask_r[:, None] & mask_h[None, :])
        right_raw     = tl.load(proj_ptr + base + H,                 mask=mask_r[:, None] & mask_h[None, :])
        left_gate_raw = tl.load(proj_ptr + base + 2*H,               mask=mask_r[:, None] & mask_h[None, :])
        right_gate_raw= tl.load(proj_ptr + base + 3*H,               mask=mask_r[:, None] & mask_h[None, :])
        out_gate_raw  = tl.load(proj_ptr + base + 4*H,               mask=mask_r[:, None] & mask_h[None, :])

        # sigmoid on fp32
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw * mask_val * left_gate
        right = right_raw * mask_val * right_gate

        dst = offs_r[:, None] * H + offs_h[None, :]

        tl.store(left_ptr + dst,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst,
                 out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])
```

---------
1️⃣  Row‑wise LayerNorm on the input tensor (B,N,N,dim) – use the highly‑optimised
    torch.nn.functional.layer_norm (float32 → float16).

2️⃣  Fuse the five linear projections (left, right, left‑gate, right‑gate,
    out‑gate) into a single matmul:
        proj = x_norm @ W_fused.T    # shape (B,N,N,5*H), FP16

3️⃣  Triton kernel `split_fuse_gate_kernel`:
    * reads the fused projection,
    * applies the optional pair‑wise mask,
    * computes three sigmoid gates in FP32,
    * writes three tensors (left, right, out_gate) in FP16.

4️⃣  Batched GEMM over the sequence dimension:
        left  = left . view(B*H,N,N)
        right = right . view(B*H,N,N)
        out   = left @ rightᵀ               # torch.bmm, FP16
    Reshape back to (B,N,N,H).

5️⃣  Second LayerNorm on the hidden dimension (FP16) using torch.layer_norm,
    followed by element‑wise multiplication with the out‑gate.

6️⃣  Final linear projection back to `dim` (FP16 → FP32).

Only step‑3 is custom Triton code; all heavy dense kernels are delegated
to cuBLAS / torch‑optimized kernels for maximal throughput on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: split fused projection, apply mask, sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[rows, 5*H] (fp16)
    left_ptr,          # mutable *[rows, H]   (fp16)
    right_ptr,         # mutable *[rows, H]   (fp16)
    out_gate_ptr,      # mutable *[rows, H]   (fp16)
    mask_ptr,          # const   *[rows]    (fp16) – ignored if HAS_MASK=0
    rows: tl.int32,    # total rows = B*N*N
    H: tl.int32,       # hidden dimension
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    For each row r (corresponds to a (batch, i, k) pair):
        left  = (proj_left  * mask) * sigmoid(left_gate)
        right = (proj_right * mask) * sigmoid(right_gate)
        out_gate = sigmoid(out_gate_proj)
    All tensors are FP16; sigmoid computed in FP32 for stability.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)           # [BLOCK_R]
    mask_r = offs_r < rows

    # ---- optional mask (broadcast over hidden) ----
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0).to(tl.float16)
        mask_val = mask_val[:, None]                     # [BLOCK_R, 1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # ---- process hidden dimension in tiles ----
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # base offset for the current tile inside the fused projection
        base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [R,H]

        # raw projection values
        left_raw      = tl.load(proj_ptr + base,
                                 mask=mask_r[:, None] & mask_h[None, :])
        right_raw     = tl.load(proj_ptr + base + H,
                                 mask=mask_r[:, None] & mask_h[None, :])
        left_gate_raw = tl.load(proj_ptr + base + 2 * H,
                                 mask=mask_r[:, None] & mask_h[None, :])
        right_gate_raw= tl.load(proj_ptr + base + 3 * H,
                                 mask=mask_r[:, None] & mask_h[None, :])
        out_gate_raw  = tl.load(proj_ptr + base + 4 * H,
                                 mask=mask_r[:, None] & mask_h[None, :])

        # sigmoid in fp32 → fp16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # apply mask & gates
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate

        # write out
        dst = offs_r[:, None] * H + offs_h[None, :]   # linear offset into [rows, H]
        tl.store(left_ptr + dst,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst,
                 out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Custom forward of the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : model parameters (float32)
            - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                     # total rows for row‑wise ops
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 → float16)
    # --------------------------------------------------------------
    # torch.layer_norm is highly optimized on H100, we use it instead of a custom kernel.
    x_norm_fp32 = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],   # (dim,) float32
        bias=weights["norm.bias"],       # (dim,) float32
        eps=eps,
    )
    x_norm = x_norm_fp32.to(torch.float16)    # [B,N,N,dim] FP16

    # --------------------------------------------------------------
    # 2️⃣ Fused linear projection to (5*H) outputs
    # --------------------------------------------------------------
    # Concatenate the five weight matrices and cast to FP16 once.
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                      # shape (5*H, dim)

    # (B,N,N,dim) @ (5*H, dim).T -> (B,N,N,5*H)
    proj = F.linear(x_norm, w_fused)         # FP16
    del x_norm, w_fused                       # free memory early

    # --------------------------------------------------------------
    # 3️⃣ Split projection, apply mask, compute sigmoid gates
    # --------------------------------------------------------------
    proj_flat = proj.view(R, 5 * H).contiguous()   # [R,5H]
    left  = torch.empty(R, H, dtype=torch.float16, device=device)
    right = torch.empty(R, H, dtype=torch.float16, device=device)
    out_gate = torch.empty(R, H, dtype=torch.float16, device=device)

    # Prepare mask (flattened) – if no mask, we pass a dummy tensor.
    if cfg.get("nomask", True) or mask is None:
        has_mask = 0
        mask_flat = torch.empty(0, dtype=torch.float16, device=device)
    else:
        # mask can be bool or float; we interpret non‑zero as 1.
        mask_fp16 = mask.to(torch.float16).reshape(R)
        has_mask = 1
        mask_flat = mask_fp16

    # Triton launch configuration
    BLOCK_R = 128
    BLOCK_H = 32
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj_flat,
        left,
        right,
        out_gate,
        mask_flat,
        R,
        H,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj_flat, mask_flat

    # --------------------------------------------------------------
    # 4️⃣ Tri‑multiplicative update (batched GEMM over sequence dim)
    # --------------------------------------------------------------
    # Reshape to [B, N, N, H] and bring H into the batch axis.
    left = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    left_mat = left.view(B * H, N, N)      # [B*H, N, N]
    right_mat = right.view(B * H, N, N)    # [B*H, N, N]

    # (B*H, N, N) @ (B*H, N, N)^T → (B*H, N, N)  (FP16)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm + output‑gate multiplication
    # --------------------------------------------------------------
    # Cast LayerNorm parameters to FP16 to match the tensor dtype.
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out_norm = F.layer_norm(
        out,
        normalized_shape=(H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )   # FP16

    # Multiply with out_gate (both FP16). Reshape out_gate back to 4‑D.
    out_gate_4d = out_gate.view(B, N, N, H)
    out_norm = out_norm * out_gate_4d
    del out_gate, out_gate_4d

    # --------------------------------------------------------------
    # 6️⃣ Final projection back to `dim`
    # --------------------------------------------------------------
    final_half = F.linear(
        out_norm,
        weights["to_out.weight"].to(torch.float16),   # (dim, H) FP16
    )   # FP16

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
```
