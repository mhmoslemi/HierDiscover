    left_gate, right_gate, out_gate) into a single GEMM:
    `proj = x_norm @ W_fused.T`  (W_fused ∈ ℝ^{5·H×dim}, fp16).
3️⃣  Split the fused tensor, apply an optional pair‑wise mask,
    compute sigmoid gates and write tensors `left`, `right`,
    `out_gate` of shape [B, H, N, N] (fp16).  This is done in a Triton
    kernel (no extra memory copies).
4️⃣  Batched half‑precision matrix multiplication over the summed
    sequence dimension `k`:
        `out[b,h,i,j] = Σ_k left[b,h,i,k] * right[b,h,j,k]`
    (`torch.bmm` on the view [B·H, N, N] → fp16).
5️⃣  Row‑wise mean / rsqrt‑variance of `out` across the hidden dimension
    H (second LayerNorm) – Triton kernel writes `mu` and `rstd`
    (fp32, one value per position b,i,j).
6️⃣  Fuse the second LayerNorm, the `out_gate` element‑wise multiplication
    and the final linear projection (`to_out`) into a single Triton kernel:
        `norm = (out - μ) * rstd * γ + β`
        `term = norm * out_gate`
        `out_final = term @ W_out.T`
    The kernel works on tiles of positions (M) and output features (N) and
    produces the final [B, N, N, dim] tensor in float32.

All kernels are written for Triton 3.3.1 and target an NVIDIA H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr, y_ptr, w_ptr, b_ptr,
    rows: tl.int32, dim: tl.int32, eps: tl.float32,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    """Row‑wise LayerNorm on a 2‑D tensor (rows × dim).

    x_ptr : input   [rows, dim] (float32)
    y_ptr : output  [rows, dim] (float16)
    w_ptr : gamma   [dim]      (float32)
    b_ptr : beta    [dim]      (float32)
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # ---- compute mean & variance (FP32) ----
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- write normalized values ----
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # (FP32)
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # (FP32)

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, optional mask, sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,                # [R, 5*H] fp16 (contiguous)
    left_ptr,                # [B, H, N, N] fp16
    right_ptr,               # [B, H, N, N] fp16
    out_gate_ptr,            # [B, H, N, N] fp16
    mask_ptr,                # [R] fp16 (ignored if HAS_MASK==0)
    R: tl.int32, H: tl.int32, N: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    """Decode the flattened index (b,i,j), apply mask, sigmoid gates and
    write ``left``, ``right`` and ``out_gate`` tensors."""
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)      # [BLOCK_R]
    mask_r = offs_r < R

    # decode (b,i,j) from linear index r
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    j = rem - i * N

    # mask value per position (broadcast over hidden)
    if HAS_MASK:
        m = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # fp16
        m = m[:, None]                                          # [BLOCK_R,1]
    else:
        m = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # base offset (hidden==0) for the destination tensors
    hidden_stride = N * N
    base_offset = b * (H * N * N) + i * N + j   # offset for hidden‑plane 0

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # address inside proj : [r, col] where col = slice*H + h
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [R, H]

        left_raw      = tl.load(proj_ptr + proj_base,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_raw     = tl.load(proj_ptr + proj_base + H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        left_gate_raw = tl.load(proj_ptr + proj_base + 2 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_gate_raw= tl.load(proj_ptr + proj_base + 3 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        out_gate_raw  = tl.load(proj_ptr + proj_base + 4 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)

        # sigmoid in FP32 → cast back to FP16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * m * left_gate
        right = right_raw * m * right_gate
        outg  = out_gate                     # no mask on out_gate

        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride

        tl.store(left_ptr      + dst_offset,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr     + dst_offset,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr  + dst_offset,
                 outg,
                 mask=mask_r[:, None] & mask_h[None, :])

# ----------------------------------------------------------------------
# 3️⃣ Row‑wise mean & rsqrt‑variance of the contracted tensor
# ----------------------------------------------------------------------
@triton.jit
def rowwise_stats_kernel(
    out_ptr, mu_ptr, rstd_ptr,
    total_rows: tl.int32, H: tl.int32, N: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Compute per‑position mean and rsqrt‑variance over the hidden dimension."""
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * N * N) + i * N + j   # hidden==0

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H
        ptr = out_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    tl.store(mu_ptr   + offs_m, mean,  mask=mask_m)
    tl.store(rstd_ptr + offs_m, rstd,  mask=mask_m)

# ----------------------------------------------------------------------
# 4️⃣ Fuse second LayerNorm, out_gate and final linear projection
# ----------------------------------------------------------------------
@triton.jit
def final_output_kernel(
    out_ptr, gate_ptr, mu_ptr, rstd_ptr,
    ln_w_ptr, ln_b_ptr,
    w_out_T_ptr,
    out_final_ptr,
    total_rows: tl.int32, H: tl.int32, N: tl.int32, dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr,
):
    """For each position (b,i,j) compute:

        norm   = ((out - μ) * rstd) * γ + β
        term   = norm * out_gate
        out    = term @ W_out.T

    All in a fused tiled kernel."""

    pid_m = tl.program_id(0)   # row block
    pid_n = tl.program_id(1)   # column block

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)   # [BLOCK_M]
    offs_n = col_start + tl.arange(0, BLOCK_N)   # [BLOCK_N]

    mask_m = offs_m < total_rows
    mask_n = offs_n < dim

    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    hidden_stride = N * N
    base_offset = b * (H * N * N) + i * N + j   # hidden==0

    # load per‑position mean / rstd (FP32)
    mu   = tl.load(mu_ptr   + offs_m, mask=mask_m, other=0.0)   # [M]
    rstd = tl.load(rstd_ptr + offs_m, mask=mask_m, other=0.0)   # [M]

    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        # load contracted tensor & out_gate (fp16 → fp32)
        out_vals = tl.load(out_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride,
                           mask=mask_m[:, None] & mask_k[None, :],
                           other=0.0).to(tl.float32)          # [M, K]
        gate_vals = tl.load(gate_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)          # [M, K]

        # LayerNorm weight & bias for this slice of hidden dim
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # FP32
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # FP32

        # (out - μ) * rstd
        diff = (out_vals - mu[:, None]) * rstd[:, None]            # [M, K]

        # LN
        norm = diff * gamma[None, :] + beta[None, :]               # [M, K]

        # apply out_gate
        term = norm * gate_vals                                    # [M, K]

        # cast back to fp16 for GEMM with W_out_T (fp16)
        term_fp16 = term.to(tl.float16)

        # load slice of W_out_T (shape [H, dim])
        w_tile = tl.load(w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :],
                         mask=mask_k[:, None] & mask_n[None, :],
                         other=0.0)                                 # [K, N]

        # accumulate dot products
        acc += tl.dot(term_fp16, w_tile).to(tl.float32)           # [M, N]

    # write final result (float32)
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul operator (no gradient).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : dict with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim        = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps        = 1e-5
    device     = x.device

    B, N, _, _ = x.shape
    R = B * N * N                # total number of (b,i,j) positions
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                     # [R, dim] float32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 256
    BLOCK_D = 32
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        weights["norm.weight"],
        weights["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    del x_flat

    # ------------------------------------------------------------------
    # 2️⃣ Fuse five linear maps into a single GEMM (fp16)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # [5*H, dim]

    # Linear: (R, dim) @ (dim, 5*H).T  → (R, 5*H)
    proj = F.linear(x_norm, fused_w)        # fp16
    del fused_w, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ Split, apply mask, sigmoid gates → left/right/out_gate
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    # mask handling
    has_mask = 0 if cfg.get("nomask", True) else 1
    if has_mask:
        assert mask is not None, "mask must be supplied when nomask=False"
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        mask_ptr = mask_flat
    else:
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)

    BLOCK_R = 128
    BLOCK_H = 32
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Batched half‑precision matmul over the summed axis `k`
    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)          # [B*H, N, N] fp16
    right_mat = right.view(B * H, N, N)         # [B*H, N, N] fp16
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B*H, N, N] fp16
    out = out.view(B, H, N, N)                 # [B, H, N, N] fp16

    # ------------------------------------------------------------------
    # 5️⃣ Row‑wise mean & rsqrt‑variance of `out` (second LayerNorm)
    # ------------------------------------------------------------------
    mu   = torch.empty(R, dtype=torch.float32, device=device)
    rstd = torch.empty(R, dtype=torch.float32, device=device)

    BLOCK_M_STATS = 256
    BLOCK_K_STATS = 64
    grid_stats = (triton.cdiv(R, BLOCK_M_STATS),)
    rowwise_stats_kernel[grid_stats](
        out,
        mu,
        rstd,
        R,
        H,
        N,
        eps,
        BLOCK_M=BLOCK_M_STATS,
        BLOCK_K=BLOCK_K_STATS,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Fuse second LayerNorm, out_gate and final projection
    # ------------------------------------------------------------------
    w_out_T = weights["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim]

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 64       # rows per program (positions)
    BLOCK_K_OUT = 32       # hidden‑dim chunk
    BLOCK_N_OUT = 64       # output‑dim chunk
    grid_out = (triton.cdiv(R, BLOCK_M_OUT), triton.cdiv(dim, BLOCK_N_OUT))
    final_output_kernel[grid_out](
        out,
        out_gate,
        mu,
        rstd,
        weights["to_out_norm.weight"],   # γ (FP32)
        weights["to_out_norm.bias"],     # β (FP32)
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=4,
    )
    return final
