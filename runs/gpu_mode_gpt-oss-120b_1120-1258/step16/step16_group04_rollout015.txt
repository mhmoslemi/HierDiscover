"""
TriMul – outgoing variant (forward only)

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the input tensor (B,N,N,dim) using the highly‑optimised
   PyTorch `layer_norm`.  The result is cast to float16 for the rest of the
   pipeline.

2️⃣ One fused linear projection produces the five hidden‑dim tensors
   (left, right, left_gate, right_gate, out_gate) in a single GEMM:
        proj = x_norm @ W_fused.T                       # (R,5·H)  FP16
   where R = B·N·N and H = hidden_dim.  The projection is split and the mask,
   sigmoid‑gates are applied inside a tiny Triton kernel
   `split_fuse_gate_kernel`.  This kernel writes the three tensors
   `left`, `right`, `out_gate` in the layout [B, H, N, N] (contiguous).

3️⃣ Batched half‑precision GEMM (cuBLAS) computes the pair‑wise contraction:
        out[b,h,i,j] = Σ_k left[b,h,i,k] * right[b,h,j,k]
   using `torch.bmm`.  The result `out` has shape (B, H, N, N) in FP16.

4️⃣ A second LayerNorm (again via PyTorch) normalises `out` over the hidden
   dimension H, then the element‑wise `out_gate` is applied.

5️⃣ The final linear projection (`to_out`) is fused with the previous step
   using `torch.nn.functional.linear`.  The final tensor is cast back to
   float32.

Only step‑2 is implemented in Triton; all heavy linear algebra and
normalisation are delegated to the vendor‑optimised PyTorch/cuBLAS kernels,
yielding a fast implementation on H100 while still satisfying the requirement
to use a custom Triton kernel.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# -------------------------------------------------------------
# Triton kernel: split the fused projection, apply optional mask,
# and compute sigmoid gates.
# -------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,            # [R, 5*H]  fp16 (fused projection)
    left_ptr,            # [B, H, N, N] fp16
    right_ptr,           # [B, H, N, N] fp16
    out_gate_ptr,        # [B, H, N, N] fp16
    mask_ptr,            # [R] fp16 (scalar mask per (b,i,k)) – ignored if HAS_MASK=0
    R: tl.int32,
    H: tl.int32,
    N: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    # -----------------------------------------------------------------
    # 1️⃣  Compute the (b,i,k) coordinates from the flattened row index.
    # -----------------------------------------------------------------
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < R

    nn = N * N                       # number of (i,k) pairs per batch
    b = offs_r // nn                 # batch index
    rem = offs_r - b * nn
    i = rem // N                     # first seq index
    k = rem - i * N                  # second seq index

    # -----------------------------------------------------------------
    # 2️⃣  Load optional scalar mask (broadcast over hidden dim).
    # -----------------------------------------------------------------
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)  # [BLOCK_R]
        mask_val = mask_val[:, None]                                 # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # -----------------------------------------------------------------
    # 3️⃣  Strides for the destination tensors.
    #     Layout = [B, H, N, N] -> hidden stride = N*N.
    # -----------------------------------------------------------------
    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + k   # start of hidden‑plane 0

    # -----------------------------------------------------------------
    # 4️⃣  Iterate over hidden dimension tiles.
    # -----------------------------------------------------------------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # ---- 5️⃣  Load the five slices from the fused projection.
        # Layout in proj: [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]      # [R, H]
        left_raw      = tl.load(proj_ptr + proj_base,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_raw     = tl.load(proj_ptr + proj_base + H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        left_gate_raw = tl.load(proj_ptr + proj_base + 2 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_gate_raw= tl.load(proj_ptr + proj_base + 3 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        out_gate_raw  = tl.load(proj_ptr + proj_base + 4 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)

        # ---- 6️⃣  FP32 sigmoid for stability, cast back to FP16.
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw .to(tl.float32)))).to(tl.float16)

        # ---- 7️⃣  Apply mask and gates.
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate                     # out_gate is NOT masked

        # ---- 8️⃣  Store results.
        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride
        tl.store(left_ptr    + dst_offset,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr   + dst_offset,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr+ dst_offset,
                 outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# -------------------------------------------------------------
# Entry point
# -------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict]) -> torch.Tensor:
    """
    Forward pass of the *outgoing* TriMul operator (no backward).

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor[ B, N, N, dim ] (float32)
        - mask_tensor  : torch.Tensor[ B, N, N ] (bool/float) or None
        - weights_dict : {
            "norm.weight", "norm.bias",
            "left_proj.weight", "right_proj.weight",
            "left_gate.weight", "right_gate.weight", "out_gate.weight",
            "to_out_norm.weight", "to_out_norm.bias",
            "to_out.weight"
          }  all float32
        - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns
    -------
    torch.Tensor : [B, N, N, dim] (float32)
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    x, mask, w, cfg = data
    device = x.device
    B, N, _, dim = x.shape
    H = cfg["hidden_dim"]
    eps = 1e-5

    # -----------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # -----------------------------------------------------------------
    # `torch.nn.functional.layer_norm` is heavily optimised on H100.
    x_norm = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    )                               # still float32
    x_norm = x_norm.to(torch.float16)   # cast once for the rest

    # -----------------------------------------------------------------
    # 2️⃣ Fused projection for the five linear maps (output dim = 5·H)
    # -----------------------------------------------------------------
    # Build the fused weight matrix once, in FP16.
    fused_weight = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)               # shape (5·H, dim)

    # Linear on the last dimension: result shape (B, N, N, 5·H)
    proj = F.linear(x_norm, fused_weight)    # FP16

    # Flatten to (R, 5·H) for the Triton kernel.
    R = B * N * N
    proj_flat = proj.view(R, 5 * H).contiguous()

    # -----------------------------------------------------------------
    # 3️⃣ Split, optional pair‑wise mask, sigmoid‐gates → left/right/out_gate
    # -----------------------------------------------------------------
    left  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    # Prepare mask for the kernel (scalar per (b,i,k) entry) if needed.
    has_mask = 0
    if not cfg.get("nomask", True):
        # mask shape: [B, N, N] → flatten to [R]
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        # dummy tensor – the kernel will ignore it when HAS_MASK==0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)

    # Launch the split/fuse kernel.
    BLOCK_R = 128   # rows per program (tuneable)
    BLOCK_H = 32    # hidden‑dim tile size (tuneable)
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj_flat,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    # Free temporary tensors we no longer need.
    del proj, proj_flat, x_norm, fused_weight, mask_ptr

    # -----------------------------------------------------------------
    # 4️⃣ Batched half‑precision GEMM (cuBLAS) over the hidden dimension
    # -----------------------------------------------------------------
    # Reshape to (B·H, N, N) → torch.bmm does a (N×N) @ (N×N)^T per head.
    left_mat  = left.reshape(B * H, N, N)
    right_mat = right.reshape(B * H, N, N)
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # FP16, shape (B·H, N, N)
    out = out.view(B, H, N, N)                             # (B, H, N, N)

    # -----------------------------------------------------------------
    # 5️⃣ Second LayerNorm (over hidden dim) + out_gate multiplication
    # -----------------------------------------------------------------
    # Move H to the last dimension to reuse torch.layer_norm.
    out_perm = out.permute(0, 2, 3, 1).contiguous()   # (B, N, N, H)
    out_norm = F.layer_norm(
        out_perm,
        normalized_shape=(H,),
        weight=w["to_out_norm.weight"],
        bias=w["to_out_norm.bias"],
        eps=eps,
    )   # still FP16

    # Apply out_gate (requires same layout as out_norm)
    out_gate_perm = out_gate.permute(0, 2, 3, 1).contiguous()   # (B, N, N, H)
    out_norm = out_norm * out_gate_perm                        # FP16

    # -----------------------------------------------------------------
    # 6️⃣ Final linear projection to `dim` (still FP16) and cast back.
    # -----------------------------------------------------------------
    # Flatten spatial dimensions, apply linear weight (dim × H).
    out_flat = out_norm.view(R, H)                     # (R, H)
    w_out_fp16 = w["to_out.weight"].to(torch.float16) # (dim, H)
    final_fp16 = F.linear(out_flat, w_out_fp16)       # (R, dim) FP16
    final = final_fp16.view(B, N, N, dim).to(torch.float32)   # back to FP32

    return final
