"""
high‑performance outgoing TriMul (Triangular Multiplicative Update).

Algorithm
---------
1️⃣  Row‑wise LayerNorm on the flattened input [B·N·N, dim] (fp32 → fp16,
    Triton kernel).  This replaces `self.norm`.
2️⃣  Fuse the four linear projections (left, right, left‑gate, right‑gate)
    into a single GEMM:
        X_norm @ [W_left; W_right; W_left_gate; W_right_gate]ᵀ   →   [R, 4·H]
    (fp16, cuBLAS).  Split the result.
3️⃣  Compute the output‑gate with a second GEMM:
        X_norm @ W_out_gateᵀ    →   [R, H]   (fp16).
4️⃣  Optional binary mask (broadcasted) and sigmoid on the three gate tensors.
5️⃣  Apply the gates to the projected tensors → `left` and `right` (fp16).
6️⃣  Reshape to [B, H, N, N] and perform the cubic update as a batched
    matrix‑multiply:
        out[b,h] = left[b,h] @ right[b,h]ᵀ                         (fp16,
        Tensor‑Core).
7️⃣  Row‑wise LayerNorm over the hidden dimension (fp16, the same Triton kernel).
8️⃣  Multiply by the output‑gate, final linear projection back to `dim`,
    and cast to float32.

All heavy linear algebra uses FP16 Tensor‑Core kernels; the only
hand‑written code is the row‑wise LayerNorm.  The implementation works for
any batch size B∈{1,2}, sequence length N∈{128…1024},
input‑channel dim∈{128,384,768} and hidden_dim=128.  A mask may be
omitted (config["nomask"]=True) or provided (bool/float tensor).

The function signature is

    custom_kernel(data) → torch.Tensor

where `data` is a tuple (input_tensor, mask_tensor, weights_dict, config_dict)
as described in the problem statement.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def layernorm_kernel(
    x_ptr,          # const  * [total_rows, dim] (fp32)
    y_ptr,          # mutable* [total_rows, dim] (fp16)
    w_ptr,          # const  * [dim] (fp32)   – γ
    b_ptr,          # const  * [dim] (fp32)   – β
    total_rows,     # int (runtime)
    dim,            # int (runtime)
    eps,            # const float (compile‑time)
    BLOCK_M: tl.constexpr,  # rows per program
    BLOCK_D: tl.constexpr,  # columns per tile
):
    """
    Row‑wise LayerNorm.
    y = ((x - μ) / sqrt(σ² + eps)) * γ + β
    x is fp32, y is fp16.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)        # [BLOCK_M]
    mask_m = offs_m < total_rows

    # -----------------------------------------------------------------
    # 1. mean & variance
    # -----------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                # 1 / sqrt(var+eps)

    # -----------------------------------------------------------------
    # 2. normalized + scale+bias → fp16 output
    # -----------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]   # (x-μ)/σ

        # γ and β are fp32 → stay in fp32 while scaling
        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)
        out_fp32 = norm * w[None, :] + b[None, :]       # fp32

        out_fp16 = out_fp32.to(tl.float16)
        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor  : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor   : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict  : dict of torch.Parameters (float32)
        - config_dict   : dict with keys
            * "dim"         – input channel dimension
            * "hidden_dim" – hidden dimension (H)
            * "nomask"      – bool, True → ignore mask

    Returns
    -------
    torch.Tensor
        Shape [B, N, N, dim], dtype float32
    """
    # --------------------------------------------------------------
    # unpack inputs
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    total_rows = B * N * N            # R = B·N·N
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣  First LayerNorm (row‑wise) – fp32 → fp16
    # --------------------------------------------------------------
    x_flat = x.view(total_rows, dim).contiguous()          # [R, dim] fp32
    x_norm_fp16 = torch.empty(total_rows, dim,
                              dtype=torch.float16,
                              device=device)

    BLOCK_M = 128          # rows per program (tuned for H100)
    BLOCK_D = 64           # dim tile size
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm_fp16,
        w["norm.weight"],      # γ  (float32)
        w["norm.bias"],        # β  (float32)
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 2️⃣  Fuse left/right projections + gates (single GEMM)
    # --------------------------------------------------------------
    #   W_fused : [4·H, dim] in fp16
    w_fused = torch.cat(
        [
            w["left_proj.weight"],     # [H, dim]
            w["right_proj.weight"],    # [H, dim]
            w["left_gate.weight"],     # [H, dim]
            w["right_gate.weight"],    # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)

    fused = F.linear(x_norm_fp16, w_fused)      # [R, 4·H]  (fp16)

    left_raw       = fused[:, 0:H]
    right_raw      = fused[:, H:2 * H]
    left_gate_raw  = fused[:, 2 * H:3 * H]
    right_gate_raw = fused[:, 3 * H:4 * H]

    # --------------------------------------------------------------
    # 3️⃣  Output‑gate projection (separate GEMM)
    # --------------------------------------------------------------
    out_gate_raw = F.linear(
        x_norm_fp16,
        w["out_gate.weight"].to(torch.float16)   # [H, dim] → fp16
    )   # [R, H] fp16

    # --------------------------------------------------------------
    # 4️⃣  Mask (if any) + sigmoid gates (in‑place)
    # --------------------------------------------------------------
    if (not nomask) and mask is not None:
        # mask shape [B, N, N] → [R, 1]
        mask_fp16 = mask.reshape(total_rows, 1).to(torch.float16)
        left_raw  = left_raw * mask_fp16
        right_raw = right_raw * mask_fp16

    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    left  = left_raw * left_gate        # [R, H] fp16
    right = right_raw * right_gate      # [R, H] fp16

    # --------------------------------------------------------------
    # 5️⃣  Cubic multiplicative update (batched GEMM)
    # --------------------------------------------------------------
    # reshape to [B, N, N, H]
    left = left.view(B, N, N, H)
    right = right.view(B, N, N, H)
    out_gate = out_gate.view(B, N, N, H)

    # bring hidden dim into batch → [B·H, N, N]
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # batched matmul: left @ rightᵀ  (fp16, Tensor‑Core)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6️⃣  Second LayerNorm over hidden dimension (fp16)
    # --------------------------------------------------------------
    out_flat = out.view(total_rows, H).contiguous()
    out_norm_fp16 = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm_fp16,
        w["to_out_norm.weight"],   # γ (float32)
        w["to_out_norm.bias"],     # β (float32)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 7️⃣  Apply output gate, final projection, cast back to float32
    # --------------------------------------------------------------
    # element‑wise gate (still fp16)
    out_norm_fp16 = out_norm_fp16 * out_gate.view(total_rows, H)

    # final linear: [R, H] @ [dim, H]ᵀ → [R, dim]  (fp16)
    final_fp16 = F.linear(
        out_norm_fp16,
        w["to_out.weight"].to(torch.float16)   # [dim, H] fp16
    )   # [R, dim] fp16

    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final
