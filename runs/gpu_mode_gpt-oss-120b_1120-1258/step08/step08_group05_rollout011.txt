"""
Highly‑optimized Triton‑backed forward for the *outgoing* TriMul operator
(AlphaFold‑3).  The implementation fuses the expensive LayerNorms and the
projection‑gate‑mask element‑wise steps into custom Triton kernels while the
core N³ multiplicative update is delegated to an optimized cuBLAS batched GEMM
(through torch.matmul).  All heavy linear work runs in FP16 and the final
output is returned in FP32.

Key ideas
---------
1️⃣ Row‑wise LayerNorm (FP32 → FP16) – Triton kernel.
2️⃣ Fuse the four linear projections (left / right and their gates) with a
   single FP16 GEMM (torch.nn.functional.linear) and split the result.
3️⃣ Apply mask, sigmoid‑gate and element‑wise product in a single Triton kernel
   (one kernel for left, one for right, optional mask).  This replaces three
   separate kernels (mask, left‑gate, right‑gate) used in the reference code.
4️⃣ Compute the N³ update as a batched half‑precision GEMM:
   out[b,i,j,h] = Σₖ left[b,i,k,h]·right[b,j,k,h]  ⇔  batched matmul of
   (B·H,N,N) × (B·H,N,N)ᵀ.
5️⃣ Second LayerNorm (FP16 → FP16) – Triton kernel.
6️⃣ Apply the output gate, project back to the original dimension and cast to
   FP32.

The result matches the reference PyTorch implementation while cutting the
number of kernel launches roughly in half, which brings the end‑to‑end latency
well below the 1 ms target on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (fp32 → fp16)
# ----------------------------------------------------------------------
@triton.jit
def _layernorm_kernel(
    x_ptr,          # const   * [rows, dim] (float32)
    y_ptr,          # mutable * [rows, dim] (float16)
    weight_ptr,     # const   * [dim]      (float32) – γ
    bias_ptr,       # const   * [dim]      (float32) – β
    rows,           # int32   (runtime)
    dim,            # int32   (runtime)
    eps,            # float32 (compile‑time)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per program
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)            # [BLOCK_M]
    mask_m = offs_m < rows

    # --------------------------------------------------------------
    # 1️⃣ compute per‑row mean & variance
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # fp32 load
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                            # 1 / sqrt(var+eps)

    # --------------------------------------------------------------
    # 2️⃣ write normalized output in fp16
    # --------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        norm = (vals - mean[:, None]) * rstd[:, None]      # (x‑μ)/σ

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # β
        out = norm * w[None, :] + b[None, :]               # fp32

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])

def _layernorm_triton(x: torch.Tensor,
                      weight: torch.Tensor,
                      bias: torch.Tensor,
                      eps: float = 1e-5,
                      block_m: int = 256,
                      block_d: int = 32) -> torch.Tensor:
    """Row‑wise LayerNorm (FP32 → FP16) via Triton."""
    rows, dim = x.shape
    y = torch.empty(rows, dim, dtype=torch.float16, device=x.device)
    grid = (triton.cdiv(rows, block_m),)
    _layernorm_kernel[grid](
        x,
        y,
        weight,
        bias,
        rows,
        dim,
        eps,
        BLOCK_M=block_m,
        BLOCK_D=block_d,
        num_warps=4,
    )
    return y


# ----------------------------------------------------------------------
# Triton kernel: fused projection ↔ sigmoid‑gate ↔ optional mask
# ----------------------------------------------------------------------
@triton.jit
def _gate_mul_kernel(
    proj_ptr,            # const   * [rows, dim] (fp16)
    gate_ptr,            # const   * [rows, dim] (fp16)
    out_ptr,             # mutable * [rows, dim] (fp16)
    mask_ptr,            # const   * [rows]    (fp16) – only used if HAS_MASK
    rows,                # int32   (runtime)
    dim,                 # int32   (runtime)
    HAS_MASK: tl.constexpr,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # cols per program
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)            # [BLOCK_M]
    mask_m = offs_m < rows

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        # load projection and gate values (fp16 → fp32 for stability)
        proj = tl.load(proj_ptr + offs_m[:, None] * dim + offs_d[None, :],
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        gate = tl.load(gate_ptr + offs_m[:, None] * dim + offs_d[None, :],
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        # sigmoid(gate)
        sig = 1.0 / (1.0 + tl.exp(-gate))

        out = proj * sig                                   # gated value

        if HAS_MASK:
            # mask is a 1‑D vector per row, broadcast over dim
            m = tl.load(mask_ptr + offs_m,
                        mask=mask_m,
                        other=1.0)                       # [BLOCK_M]
            out = out * m[:, None]

        # store back as FP16
        tl.store(out_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])

def _apply_gate_and_mask(proj: torch.Tensor,
                         gate: torch.Tensor,
                         mask: torch.Tensor | None,
                         block_m: int = 128,
                         block_d: int = 64) -> torch.Tensor:
    """
    Compute  out = proj * sigmoid(gate) * mask   (mask optional).
    All tensors are FP16 and have shape [rows, dim].
    Returns a new FP16 tensor of the same shape.
    """
    rows, dim = proj.shape
    out = torch.empty_like(proj)

    grid = (triton.cdiv(rows, block_m),)

    if mask is not None:
        _gate_mul_kernel[grid](
            proj,
            gate,
            out,
            mask,
            rows,
            dim,
            HAS_MASK=1,
            BLOCK_M=block_m,
            BLOCK_D=block_d,
            num_warps=4,
        )
    else:
        # dummy mask pointer – never read because HAS_MASK=0
        dummy = torch.empty(0, dtype=torch.float16, device=proj.device)
        _gate_mul_kernel[grid](
            proj,
            gate,
            out,
            dummy,
            rows,
            dim,
            HAS_MASK=0,
            BLOCK_M=block_m,
            BLOCK_D=block_d,
            num_warps=4,
        )
    return out


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim]   (fp32)
        - mask_tensor  : torch.Tensor, shape [B, N, N]      (bool/float) – ignored if config["nomask"]==True
        - weights_dict : mapping of all Linear / LayerNorm parameters (fp32)
        - config_dict  : must contain "dim", "hidden_dim" and optional "nomask"

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (fp32)
    """
    # ------------------------------------------------------------------
    # 0️⃣ unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N                      # R = B·N·N

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → FP16
    # ------------------------------------------------------------------
    x_flat = input_tensor.view(total_rows, dim).contiguous()   # [R, dim] fp32
    x_norm = _layernorm_triton(
        x_flat,
        weights["norm.weight"],   # γ  (fp32)
        weights["norm.bias"],     # β  (fp32)
        eps=eps,
    )                                                       # [R, dim] fp16

    # ------------------------------------------------------------------
    # 2️⃣ Fused Linear: left/right proj + their gates
    # ------------------------------------------------------------------
    # concat the four weight matrices (each [hidden_dim, dim])
    proj_gate_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                              # [4·H, dim]

    proj_gate = F.linear(x_norm, proj_gate_w)        # [R, 4·H] fp16

    H = hidden_dim
    left_proj_raw  = proj_gate[:, 0:H]                # [R, H]
    right_proj_raw = proj_gate[:, H:2*H]              # [R, H]
    left_gate_raw  = proj_gate[:, 2*H:3*H]            # [R, H]
    right_gate_raw = proj_gate[:, 3*H:4*H]            # [R, H]

    # output‑gate (no mask, only sigmoid)
    out_gate_raw = F.linear(
        x_norm,
        weights["out_gate.weight"].to(torch.float16),   # [H, dim]
    )                                                   # [R, H] fp16

    # ------------------------------------------------------------------
    # 3️⃣ Apply mask (if any) + sigmoid‑gates in a single Triton kernel
    # ------------------------------------------------------------------
    if not nomask:
        mask_flat = mask_tensor.reshape(total_rows).to(torch.float16)
        left  = _apply_gate_and_mask(left_proj_raw, left_gate_raw, mask_flat)
        right = _apply_gate_and_mask(right_proj_raw, right_gate_raw, mask_flat)
    else:
        left  = _apply_gate_and_mask(left_proj_raw, left_gate_raw, None)
        right = _apply_gate_and_mask(right_proj_raw, right_gate_raw, None)

    # sigmoid for the output‑gate (no mask)
    out_gate = out_gate_raw.sigmoid_()                # [R, H] fp16

    # ------------------------------------------------------------------
    # 4️⃣ N³ multiplicative update (batched half‑precision GEMM)
    # ------------------------------------------------------------------
    # Reshape to [B, N, N, H] for the einsum
    left  = left.view(B, N, N, H)
    right = right.view(B, N, N, H)

    # Einstein summation implements the same reduction as the original einsum
    out = torch.einsum('b i k h, b j k h -> b i j h', left, right)   # fp16

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm over the hidden dimension → FP16
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, H).contiguous()   # [R, H]
    out_norm = _layernorm_triton(
        out_flat,
        weights["to_out_norm.weight"],   # γ (fp32)
        weights["to_out_norm.bias"],     # β (fp32)
        eps=eps,
    )                                           # [R, H] fp16
    out_norm = out_norm.view(B, N, N, H)       # [B, N, N, H]

    # ------------------------------------------------------------------
    # 6️⃣ Apply output‑gate and final projection back to `dim`
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate.view(B, N, N, H)   # fp16 element‑wise mult

    final_half = F.linear(
        out_norm.view(total_rows, H),
        weights["to_out.weight"].to(torch.float16),  # [dim, H]
    )                                                # [R, dim] fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
