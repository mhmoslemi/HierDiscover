"""
Outgoing TriMul (AlphaFold‑3) – fast Triton‑accelerated forward pass.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the pair representation (FP32 → FP16) via a custom
   Triton kernel.  The kernel works on a 2‑D view of the tensor (R = B·N·N rows,
   C = dim columns) and computes per‑row mean/variance in FP32, then writes the
   normalized FP16 result.

2️⃣ Fuse the four linear maps (left_proj, right_proj, left_gate, right_gate)
   into a single FP16 GEMM:
        W₄h = [Wₗₚ; Wᵣₚ; Wₗg; Wᵣg]   (shape 4·H × dim)
   The GEMM produces the four tensors in one pass, eliminating three extra
   memory reads.

3️⃣ Apply the optional pairwise mask, compute three independent sigmoids and
   gate the projected tensors.

4️⃣ Reshape the gated tensors to (B, N, N, H) and compute the cubic update
   (the costly O(N³·H) part):
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   This is realised as a **batched FP16 matrix multiplication** using
   `torch.bmm`.  With H100 Tensor‑Cores this operation is heavily
   accelerated and dominates the runtime.

5️⃣ Row‑wise LayerNorm over the hidden dimension H (again via Triton).

6️⃣ Multiply by the output‑gate, then a final FP16 linear projection back to
   the original feature dimension `dim`.  The result is cast back to FP32 to
   match the reference implementation.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton row‑wise LayerNorm (float32 → float16)
# ----------------------------------------------------------------------
@triton.jit
def rowwise_layernorm_kernel(
    x_ptr,            # const  * [total_rows, dim] (float32)
    y_ptr,            # mutable* [total_rows, dim] (float16)
    w_ptr,            # const  * [dim] (float32) – γ
    b_ptr,            # const  * [dim] (float32) – β
    total_rows,       # int32
    dim,              # int32
    eps,              # float32
    BLOCK_R: tl.constexpr,   # rows per program (≈128‑256)
    BLOCK_D: tl.constexpr,   # columns per inner loop (<= 384)
):
    """
    Compute per‑row LayerNorm.
      y = ((x-µ)/√(σ²+ϵ)) * γ + β
    Input  : x – FP32
    Output : y – FP16
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)               # [BLOCK_R]
    mask_r = offs_r < total_rows

    # ---------- 1) compute mean & variance (FP32) ----------
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_R], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_r[:, None] * dim + offs_d[None, :]  # stride = dim
        vals = tl.load(ptr,
                       mask=mask_r[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)   # 1 / sqrt(var + eps)

    # ---------- 2) write normalized output (FP16) ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_r[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_r[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)

        out = norm * gamma[None, :] + beta[None, :]
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_r[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_r[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float) – optional
        - weights_dict : model weight tensors (see key names below)
        - config_dict  : must contain "dim", "hidden_dim", "nomask"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device

    dim        = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps        = 1e-5
    nomask     = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape                     # [B, N, N, dim]
    total_rows = B * N * N                              # R = B·N·N
    H = hidden_dim                                      # hidden dim (often 128)

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → fp16
    # --------------------------------------------------------------
    # flatten to [R, dim] in FP32
    x_flat = input_tensor.reshape(total_rows, dim).contiguous()
    # allocate FP16 output buffer
    x_norm = torch.empty(total_rows, dim, dtype=torch.float16,
                         device=device)

    # block sizes – chosen to cover the largest dim (384) while keeping
    # occupancy high on H100.
    BLOCK_R = 256          # rows per program
    BLOCK_D = 128          # columns per inner loop (dim ≤ 384)

    grid = (triton.cdiv(total_rows, BLOCK_R),)
    rowwise_layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],          # γ (FP32)
        weights["norm.bias"],            # β (FP32)
        total_rows,
        dim,
        eps,
        BLOCK_R=BLOCK_R,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    # --------------------------------------------------------------
    # 2️⃣ Fuse four linear maps (proj + gates) into a single GEMM
    # --------------------------------------------------------------
    # Concatenate the four weight matrices (no bias).  Shape → (4·H, dim)
    w_4h = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                 # cast once to fp16

    # Single GEMM: [R, dim] @ [dim, 4·H]ᵀ → [R, 4·H]  (FP16)
    proj_gate = F.linear(x_norm, w_4h)   # nb. bias=None

    # Split the concatenated result
    left_proj_raw  = proj_gate[:, 0:H]
    right_proj_raw = proj_gate[:, H:2*H]
    left_gate_raw  = proj_gate[:, 2*H:3*H]
    right_gate_raw = proj_gate[:, 3*H:4*H]

    # Out‑gate (separate linear map, still FP16)
    out_gate_raw = F.linear(
        x_norm,
        weights["out_gate.weight"].to(torch.float16)   # (H, dim)
    )   # → [R, H] (FP16)

    # --------------------------------------------------------------
    # 3️⃣ Optional mask, sigmoid gates, element‑wise gating
    # --------------------------------------------------------------
    if not nomask:
        # mask: [B,N,N] → [R,1] and cast to fp16
        mask_fp16 = mask_tensor.reshape(total_rows, 1).to(torch.float16)
        left_proj_raw  = left_proj_raw * mask_fp16
        right_proj_raw = right_proj_raw * mask_fp16

    # In‑place sigmoid for speed
    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    # Apply gates
    left  = left_proj_raw * left_gate      # [R, H]  FP16
    right = right_proj_raw * right_gate    # [R, H]  FP16

    # --------------------------------------------------------------
    # 4️⃣ Reshape to (B, N, N, H)
    # --------------------------------------------------------------
    left = left.view(B, N, N, H)
    right = right.view(B, N, N, H)
    out_gate = out_gate.view(B, N, N, H)

    # --------------------------------------------------------------
    # 5️⃣ Cubic update via batched GEMM (Tensor‑Core friendly)
    #    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    # --------------------------------------------------------------
    # Move H into the batch dimension → (B·H, N, N)
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    left_mat  = left_perm.view(B * H, N, N)   # [B·H, N, N]
    right_mat = right_perm.view(B * H, N, N)  # [B·H, N, N]

    # Batched matrix multiply: left @ rightᵀ (FP16)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N]

    # Restore (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm (row‑wise over hidden dim H) → fp16
    # --------------------------------------------------------------
    out_flat = out.reshape(total_rows, H).contiguous()
    out_norm = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows, BLOCK_R),)
    rowwise_layernorm_kernel[grid2](
        out_flat,
        out_norm,
        weights["to_out_norm.weight"],   # γ (FP32)
        weights["to_out_norm.bias"],     # β (FP32)
        total_rows,
        H,
        eps,
        BLOCK_R=BLOCK_R,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    out_norm = out_norm.view(B, N, N, H)

    # --------------------------------------------------------------
    # 7️⃣ Apply out‑gate and final linear projection back to `dim`
    # --------------------------------------------------------------
    gated = out_norm * out_gate          # [B, N, N, H] FP16

    final = F.linear(
        gated.view(total_rows, H),
        weights["to_out.weight"].to(torch.float16)   # (dim, H)
    )   # → [R, dim] (FP16)

    final = final.view(B, N, N, dim).to(torch.float32)

    return final
