"""
TriMul “outgoing” forward pass – Triton‑accelerated row‑wise LayerNorm
====================================================================

Algorithm
---------
1️⃣ Row‐wise LayerNorm (float32 → float16) on the input tensor.
2️⃣ One fused linear projection for the four “proj/gate” matrices
   (left_proj, right_proj, left_gate, right_gate) using torch.nn.functional.linear.
3️⃣ Separate linear for the out‑gate.
4️⃣ Optional masking, sigmoid, and gating.
5️⃣ Batched GEMM (B·H matmuls of shape N×N) for the N³ multiplicative update.
6️⃣ Row‑wise LayerNorm on the hidden tensor.
7️⃣ Apply the out‑gate, final linear projection back to `dim`, and cast to float32.

Only the two LayerNorm passes are written in Triton; all dense matrix
multiplications use cuBLAS (via torch.nn.functional.linear / torch.bmm),
which is optimal on an H100.  The kernels are tuned for the supported
dimension ranges (dim ≤ 768, hidden_dim ≤ 384).  The implementation
handles the optional mask and works for any batch size `B ∈ {1,2}`.

The final output has shape `[B, N, N, dim]` and dtype ``torch.float32``.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (float32 → float16)
# ----------------------------------------------------------------------
@triton.jit
def rowwise_layernorm_kernel(
    x_ptr,          # const  * [total_rows, dim]   (float32)
    y_ptr,          # mutable* [total_rows, dim]   (float16)
    w_ptr,          # const  * [dim]               (float32)  gamma
    b_ptr,          # const  * [dim]               (float32)  beta
    total_rows,     # int
    dim,            # int
    eps,            # float (compile‑time constant)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # cols per tile
):
    """
    Row‑wise LayerNorm:
        y = ((x - μ) / sqrt(σ²+eps)) * γ + β
    Reduction uses fp32, output is fp16.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)                # [BLOCK_M]
    mask_m = offs_m < total_rows

    # --------------------------------------------------------------
    # 1️⃣ Compute mean & variance per row
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    num_d_tiles = tl.cdiv(dim, BLOCK_D)
    for d_tile in range(0, num_d_tiles):
        offs_d = d_tile * BLOCK_D + tl.arange(0, BLOCK_D)       # [BLOCK_D]
        mask_d = offs_d < dim

        # Load a tile of x (float32)
        x = tl.load(
            x_ptr + offs_m[:, None] * dim + offs_d[None, :],
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                                       # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(x, axis=1)                           # [BLOCK_M]
        sum_sq  += tl.sum(x * x, axis=1)                       # [BLOCK_M]

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                                 # 1 / sqrt(var+eps)

    # --------------------------------------------------------------
    # 2️⃣ Write normalized output (apply gamma/beta)
    # --------------------------------------------------------------
    for d_tile in range(0, num_d_tiles):
        offs_d = d_tile * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        x = tl.load(
            x_ptr + offs_m[:, None] * dim + offs_d[None, :],
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        # Normalize
        norm = (x - mean[:, None]) * rstd[:, None]            # fp32

        # Load gamma/beta (fp32)
        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)

        out_fp32 = norm * gamma[None, :] + beta[None, :]     # fp32
        out_fp16 = out_fp32.to(tl.float16)

        tl.store(
            y_ptr + offs_m[:, None] * dim + offs_d[None, :],
            out_fp16,
            mask=mask_m[:, None] & mask_d[None, :],
        )

        (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor       [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor       [B, N, N]    (bool/float) – may be all‑ones.
        - weights_dict : dict of torch.Tensors containing all Linear/LayerNorm params.
        - config_dict  : dict with keys:
              * "dim"         – input channel dimension
              * "hidden_dim"  – hidden dimension (H)
              * "nomask"      – bool, if True mask is ignored (default True)

    Returns
    -------
    torch.Tensor
        [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    device = input_tensor.device
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape
    R = B * N * N                     # total rows for row‑wise operations

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → fp16
    # ------------------------------------------------------------------
    # Flatten to [R, dim] and make sure it's contiguous in memory
    x_flat = input_tensor.view(R, dim).contiguous()          # (R, dim) fp32
    x_norm_fp16 = torch.empty(R, dim, dtype=torch.float16, device=device)

    # Choose block sizes (empirically good across the test space)
    BLOCK_D = 32 if dim <= 256 else 64
    BLOCK_M = 256 if R >= 256 * 256 else 128   # more rows per block when many rows

    grid = (triton.cdiv(R, BLOCK_M),)
    rowwise_layernorm_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],   # gamma, fp32
        weights["norm.bias"],     # beta,  fp32
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fuse four proj/gate linear layers (single GEMM)
    # ------------------------------------------------------------------
    # Concatenate the four projection/gate weight matrices → (4*H, dim)
    w_proj_gate = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)   # fp16 for Tensor‑Core GEMM

    # Linear: [R, dim] @ [4*H, dim]^T → [R, 4*H] (fp16)
    proj_gate = F.linear(x_norm_fp16, w_proj_gate)       # bias = None

    H = hidden_dim
    left_proj_raw  = proj_gate[:, 0:H]               # [R, H]
    right_proj_raw = proj_gate[:, H:2*H]             # [R, H]
    left_gate_raw  = proj_gate[:, 2*H:3*H]           # [R, H]
    right_gate_raw = proj_gate[:, 3*H:4*H]           # [R, H]

    # Separate linear for the output‑gate (still half‑precision)
    out_gate_raw = F.linear(
        x_norm_fp16,
        weights["out_gate.weight"].to(torch.float16)   # [H, dim]
    )   # → [R, H]

    # ------------------------------------------------------------------
    # 3️⃣ Optional masking + sigmoid gates + gating
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape [B, N, N] → flatten to [R, 1] and broadcast
        mask_flat = mask_tensor.to(torch.float16).view(R, 1)   # [R,1]
        left_proj_raw  = left_proj_raw * mask_flat
        right_proj_raw = right_proj_raw * mask_flat

    # In‑place sigmoid (half‑precision)
    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    # Gating
    left  = left_proj_raw * left_gate           # [R, H] fp16
    right = right_proj_raw * right_gate         # [R, H] fp16

    # ------------------------------------------------------------------
    # 4️⃣ Reshape for the N³ multiplicative update (batched GEMM)
    # ------------------------------------------------------------------
    left  = left.view(B, N, N, H)          # [B, N, N, H]
    right = right.view(B, N, N, H)         # [B, N, N, H]
    out_gate = out_gate.view(B, N, N, H)   # [B, N, N, H]

    # Move hidden dimension to batch axis → many independent GEMMs
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    left_mat  = left_perm.view(B * H, N, N)      # [B·H, N, N]  fp16
    right_mat = right_perm.view(B * H, N, N)     # [B·H, N, N]  fp16

    # Batched GEMM: (B·H, N, N) @ (B·H, N, N)^T → (B·H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # fp16

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm over hidden dimension → fp16
    # ------------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()
    out_norm_fp16 = torch.empty_like(out_flat, dtype=torch.float16)

    # Re‑use the same kernel, just with different dimensions
    BLOCK_D2 = 32 if H <= 256 else 64
    BLOCK_M2 = 256 if R >= 256 * 256 else 128
    grid2 = (triton.cdiv(R, BLOCK_M2),)
    rowwise_layernorm_kernel[grid2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],   # gamma (fp32)
        weights["to_out_norm.bias"],     # beta  (fp32)
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M2,
        BLOCK_D=BLOCK_D2,
        num_warps=8,
    )
    out_norm = out_norm_fp16.view(B, N, N, H)   # fp16

    # ------------------------------------------------------------------
    # 6️⃣ Apply output gate and final linear projection back to `dim`
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate                       # [B,N,N,H] fp16

    # Final linear: [R, H] @ [dim, H]^T → [R, dim] (fp16)
    final_half = F.linear(
        out_norm.view(R, H),
        weights["to_out.weight"].to(torch.float16)      # [dim, H]
    )   # → [R, dim] fp16

    # Cast to float32 as required by the API
    final = final_half.view(B, N, N, dim).to(torch.float32)

    return final
