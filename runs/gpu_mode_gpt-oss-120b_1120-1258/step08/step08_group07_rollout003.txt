"""
TriMul “outgoing” forward pass – Triton‑accelerated LayerNorm + fused Linear.
Algorithm
---------
1️⃣  Row‑wise LayerNorm on the input tensor (B·N·N × dim) → fp16.
2️⃣  Fuse the four projection/gate Linear layers *and* the output‐gate Linear
    into a **single** half‑precision GEMM:
        [R, dim] @ [5·H, dim]ᵀ → [R, 5·H]   (R = B·N·N, H = hidden_dim)
    Splits give left_proj, right_proj, left_gate, right_gate, out_gate.
3️⃣  Optional mask (applied only to left/right projections), sigmoid on gates
    and element‑wise gating.
4️⃣  Reshape left/right to [B·H, N, N] and perform the dominant N³ update with
    a batched half‑precision GEMM (torch.bmm).  This computes
        out[b,i,j,d] = Σₖ left[b,i,k,d]·right[b,j,k,d].
5️⃣  Row‑wise LayerNorm on the hidden dimension (B·N·N × H) → fp16.
6️⃣  Apply the output‑gate, a final half‑precision Linear projection back to
    the model dimension and cast the result to float32.

Only the two LayerNorm passes are written in Triton; all dense matrix
multiplications use cuBLAS/Tensor‑Core via torch.nn.functional.linear and
torch.bmm, which is optimal on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (input fp32 → output fp16)
# ----------------------------------------------------------------------
@triton.jit
def rowwise_layernorm_kernel(
    x_ptr,               # const   * [total_rows, dim] (float32)
    y_ptr,               # mutable * [total_rows, dim] (float16)
    w_ptr,               # const   * [dim]           (float32) gamma
    b_ptr,               # const   * [dim]           (float32) beta
    total_rows,          # int
    dim,                 # int
    eps,                 # float32 (compile‑time constant)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # cols per tile
):
    """Row‑wise LayerNorm:
    y = ((x - μ) / sqrt(σ²+eps)) * γ + β
    Reduction is done in fp32, output is fp16.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # --------------------------------------------------------------
    # 1️⃣  Compute mean & variance per row (fp32)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    # iterate over the dim axis in tiles of size BLOCK_D
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        # load a tile of the input (fp32)
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)         # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)          # sum over dim tile
        sum_sq  += tl.sum(vals * vals, axis=1)   # sum of squares

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)   # 1 / sqrt(var + eps)

    # --------------------------------------------------------------
    # 2️⃣  Write normalized output (fp16)
    # --------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        # (x - μ) * rstd  →  normalized value (fp32)
        norm = (vals - mean[:, None]) * rstd[:, None]

        # affine transformation γ·norm + β  (both fp32)
        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)

        out = norm * gamma[None, :] + beta[None, :]   # fp32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

    x_norm_fp16 = torch.empty(R, dim, dtype=torch.float16, device=device)

    # block sizes – chosen to fit well in shared memory for dim ≤ 384
    BLOCK_M = 256
    BLOCK_D = 32
    grid = (triton.cdiv(R, BLOCK_M),)
    rowwise_layernorm_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],   # gamma (fp32)
        weights["norm.bias"],     # beta  (fp32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fuse all five linear projections (left, right, left_gate,
    #    right_gate, out_gate) into a single GEMM.
    # ------------------------------------------------------------------
    # Concatenate weights along the output‑feature dimension:
    #   shape -> [5*H, dim]
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
            weights["out_gate.weight"],    # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)   # fp16 for Tensor‑Core

    # Single half‑precision linear: [R, dim] @ [5*H, dim]^T → [R, 5*H]
    proj_all = F.linear(x_norm_fp16, w_fused)   # bias = None

    # Split the concatenated result
    left_proj_raw, right_proj_raw, left_gate_raw, right_gate_raw, out_gate_raw = torch.split(
        proj_all, H, dim=1
    )   # each [R, H] fp16

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask, sigmoid gates and element‑wise gating
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape [B, N, N] → flatten → broadcast over hidden dim
        mask_flat = mask_tensor.to(torch.float16).view(R, 1)   # [R,1]
        left_proj_raw = left_proj_raw * mask_flat
        right_proj_raw = right_proj_raw * mask_flat

    # In‑place sigmoid (half‑precision)
    left_gate = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate = out_gate_raw.sigmoid_()

    # Gating
    left = left_proj_raw * left_gate          # [R, H] fp16
    right = right_proj_raw * right_gate       # [R, H] fp16

    # ------------------------------------------------------------------
    # 4️⃣ N³ multiplicative update (batched GEMM)
    #    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    # Reshape to 4‑D tensors
    left = left.view(B, N, N, H)          # [B, N, N, H]
    right = right.view(B, N, N, H)        # [B, N, N, H]
    out_gate = out_gate.view(B, N, N, H)  # [B, N, N, H]

    # Move hidden dimension to batch axis for many independent GEMMs
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    left_mat = left_perm.view(B * H, N, N)      # [B·H, N, N]  fp16
    right_mat = right_perm.view(B * H, N, N)    # [B·H, N, N]  fp16

    # Batched matrix multiplication: (B·H, N, N) @ (B·H, N, N)^T → (B·H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Restore original layout: [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # fp16

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm over hidden dimension (row‑wise) → fp16
    # ------------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()
    out_norm_fp16 = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(R, BLOCK_M),)
    rowwise_layernorm_kernel[grid2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],   # gamma (fp32)
        weights["to_out_norm.bias"],     # beta  (fp32)
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    out_norm = out_norm_fp16.view(B, N, N, H)   # fp16

    # ------------------------------------------------------------------
    # 6️⃣ Apply output gate and final linear projection back to `dim`
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate                 # [B,N,N,H] fp16
    final_half = F.linear(
        out_norm.view(R, H),
        weights["to_out.weight"].to(torch.float16),   # [dim, H]
    )   # → [R, dim] fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)

    return final
```
