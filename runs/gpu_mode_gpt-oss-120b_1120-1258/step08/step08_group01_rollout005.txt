import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def layernorm_kernel(
    x_ptr,  # const float32 *
    y_ptr,  # mutable float16 *
    weight_ptr,  # const float32 *
    bias_ptr,    # const float32 *
    total_rows,  # int
    dim,         # int
    eps,         # float constexpr
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """
    Row‑wise LayerNorm. Operates on a contiguous 2‑D view [total_rows, dim].
    Input    : fp32
    Output   : fp16 = ((x-μ)/sqrt(var+eps))*γ+β
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # accumulator for mean & var (fp32)
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    # iterate over d dimension
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0)
        vals_f32 = vals.to(tl.float32)
        sum_val += tl.sum(vals_f32, axis=1)
        sum_sq += tl.sum(vals_f32 * vals_f32, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # write normalized output (fp16)
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0)
        vals_f32 = vals.to(tl.float32)

        norm = (vals_f32 - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr + offs_d, mask=mask_d, other=0.0)
        out = norm * w[None, :] + b[None, :]   # fp32

        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16, mask=mask_m[:, None] & mask_d[None, :])

"""
custom_kernel
==============

Fast Triton‑accelerated forward pass for the **outgoing** TriMul operator
(AlphaFold‑3 style).  The implementation mirrors the reference PyTorch
module but moves the two LayerNorms to a custom Triton kernel while the
large N³ multiplicative update and all dense linear layers are kept in
high‑throughput FP16 torch operations.

Algorithm
---------
1. **First LayerNorm** (row‑wise) on the input tensor (float32 → float16).
2. **Fused linear projection**:
   * Concatenate the four weight matrices (left/right projections and
     their two gates) → shape `[4·H, dim]`.
   * One `F.linear` call produces a tensor `[R, 4·H]` (`R = B·N·N`).
   * Split the result into `{left, right, left_gate, right_gate}`.
   * Compute `out_gate` with a separate linear layer.
3. **Mask & gating** – optional mask multiplication, in‑place sigmoid for
   the three gates, and element‑wise weighting.
4. **N³ multiplicative update**:
   * Reshape to `[B, H, N, N]`, flatten batch+head (`B·H`) and run a
     batched matrix multiplication `left @ rightᵀ` (`torch.bmm`), which
     yields the tensor of shape `[B·H, N, N]`.
   * Restore the original `[B, N, N, H]` layout.
5. **Second LayerNorm** (row‑wise) over the hidden dimension `H`.
6. **Output gating + final linear projection** back to the original
   feature dimension `dim`.  The final result is cast to `float32`.

Only the LayerNorms are written in Triton; all large GEMMs remain in
cuBLAS/Tensor‑Core FP16 for maximal throughput.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (float32 → float16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   * [total_rows, dim]   (float32)
    y_ptr,          # mutable * [total_rows, dim]   (float16)
    weight_ptr,     # const   * [dim]               (float32)
    bias_ptr,       # const   * [dim]               (float32)
    total_rows,     # int (runtime)
    dim,            # int (runtime)
    eps,            # float (runtime)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # cols per tile
):
    """
    Row‑wise LayerNorm:
        y = ((x - μ) / √(σ² + eps)) * γ + β
    Input  : fp32
    Output : fp16
    """
    pid = tl.program_id(0)                     # one program per BLOCK_M rows
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M) # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # 1️⃣ Compute per‑row mean & variance (fp32 accumulation)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                                    # fp32 load
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                         # 1/√(var+eps)

    # ------------------------------------------------------------------
    # 2️⃣ Write normalized output (fp16)
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]          # fp32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # γ fp32
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # β fp32
        out = norm * w[None, :] + b[None, :]                    # fp32

        out_fp16 = out.to(tl.float16)
        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

    Forward pass of the outgoing TriMul operator.

    Parameters

    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : dict with LayerNorm/Linear parameters (float32)
        - config_dict  : dict with keys
            * "dim" (int) – input feature dimension
            * "hidden_dim" (int) – hidden dimension H
            * "nomask" (bool, optional) – if True the mask is ignored

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim], dtype float32
    """

    # 0️⃣ Unpack inputs & configuration
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    device = x.device
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]

    eps = 1e-5

    BLOCK_D = 64

    layernorm_kernel[grid](
        x_flat,
        ln1_out,
        weights["norm.weight"],   # γ (float32)
        weights["norm.bias"],     # β (float32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
        num_stages=2,
    )

        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
        ],
        dim=0,                                 # [4·H, dim]
    ).to(device=device, dtype=torch.float16)

    fused = F.linear(ln1_out, proj_w)   # FP16

    # Slice the four sub‑tensors
    left_raw      = fused[:, 0:H]               # [R, H]
    right_raw     = fused[:, H:2*H]             # [R, H]
    left_gate_raw = fused[:, 2*H:3*H]           # [R, H]
    right_gate_raw= fused[:, 3*H:4*H]           # [R, H]

    # Output‑gate has its own weight matrix
    out_gate_raw = F.linear(
        ln1_out,
        weights["out_gate.weight"].to(device=device, dtype=torch.float16)  # [H, dim]
    )   # [R, H] FP16

    # 3️⃣ Optional mask, sigmoid gating and element‑wise weighting
    # ------------------------------------------------------------------
    if not nomask and mask is not None:

        mask_flat = mask.view(R, 1).to(device=device, dtype=torch.float16)
        left_raw  = left_raw * mask_flat
        right_raw = right_raw * mask_flat

    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    left  = left_raw * left_gate    # [R, H] FP16
    right = right_raw * right_gate  # [R, H] FP16

    # ------------------------------------------------------------------

    left  = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    # Collapse batch+head → (B·H, N, N)
    left_mat  = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)

    # Batched matmul: out[b*h, i, j] = Σ_k left[b*h, i, k] * right[b*h, j, k]
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B·H, N, N)

    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()
    ln2_out = torch.empty_like(out_flat, dtype=torch.float16)

    layernorm_kernel[grid2](

        ln2_out,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
        num_stages=2,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Apply output gate, final projection and cast back to float32
    # ------------------------------------------------------------------
    out_norm = ln2_out.view(B, N, N, H)                # FP16
    out_norm = out_norm * out_gate.view(B, N, N, H)    # element‑wise FP16

    final_fp16 = F.linear(
        out_norm.view(R, H),
        weights["to_out.weight"].to(device=device, dtype=torch.float16)   # [dim, H]
    )   # [R, dim] FP16

    final = final_fp16.view(B, N, N, dim).to(torch.float32)   # fp32 output
    return final
```
