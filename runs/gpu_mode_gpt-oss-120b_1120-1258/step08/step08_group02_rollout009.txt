"""
custom_kernel – high‑performance forward of the *outgoing* TriMul operator
-----------------------------------------------------------------------

Algorithm (all tensors live on the GPU):
1️⃣  Layer‑norm on the last channel (dim) – use the highly‑optimised
    torch.nn.functional.layer_norm (cuDNN) → FP16.
2️⃣  Fuse the two projection matrices and the two gate matrices into one
    GEMM:  X_norm @ W_fused.T  →  [R, 4·H]  where R = B·N·N.
    Slice the result to obtain `left`, `right`, `left_gate` and `right_gate`.
3️⃣  Compute the output‑gate with a separate GEMM.
4️⃣  Apply the optional binary mask (broadcasted) and the three sigmoid
    gates (in‑place, FP16).
5️⃣  Reshape `left`/`right` to (B, N, N, H) → (B·H, N, N) and perform a
    batched GEMM:
        out_mat = (B·H, N, N) @ (B·H, N, N)ᵀ   →   (B·H, N, N)
6️⃣  Fuse the second LayerNorm (over H) *and* the element‑wise multiplication
    by `out_gate` in a single Triton kernel (`layernorm_gate_kernel`).
    All reductions are done in FP32, the result is stored in FP16.
7️⃣  Final linear projection back to the original channel dimension
    (FP16 GEMM) and cast the result to FP32.

Only step 6 is implemented in Triton; the remaining steps use
torch‑CUDA kernels (cuBLAS, cuDNN) which already give TF32/Tensor‑Core
performance.  The implementation works for any batch size (1‑2),
sequence length (128‑1024) and the channel sizes demanded by the
challenge.  The final output is float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fused LayerNorm + element‑wise gate (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_gate_kernel(
    x_ptr,          # const   * [total_rows, dim] (float16)
    gate_ptr,       # const   * [total_rows, dim] (float16)
    y_ptr,          # mutable * [total_rows, dim] (float16)
    weight_ptr,     # const   * [dim]            (float32)
    bias_ptr,       # const   * [dim]            (float32)
    total_rows,     # int32   (runtime)
    dim,            # int32   (runtime)
    eps,            # float   (compile‑time)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # cols per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # 1️⃣ Compute per‑row mean & variance (FP32)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)      # FP32 load
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                             # 1 / sqrt(var+eps)

    # ------------------------------------------------------------------
    # 2️⃣ Normalise, apply γ/β and multiply by gate (all FP32, store FP16)
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        # load raw activation
        x_ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        x_vals = tl.load(x_ptrs,
                         mask=mask_m[:, None] & mask_d[None, :],
                         other=0.0).to(tl.float32)

        # load gate (same shape as x)
        g_ptrs = gate_ptr + offs_m[:, None] * dim + offs_d[None, :]
        g_vals = tl.load(g_ptrs,
                         mask=mask_m[:, None] & mask_d[None, :],
                         other=0.0).to(tl.float32)

        norm = (x_vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # β

        out = (norm * w[None, :] + b[None, :]) * g_vals   # FP32
        out_fp16 = out.to(tl.float16)

        y_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(y_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
# Custom forward for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Args
    ----
    data = (input_tensor, mask_tensor, weights_dict, config_dict)

    - input_tensor : torch.Tensor     [B, N, N, dim] (float32)
    - mask_tensor  : torch.Tensor or None  [B, N, N] (bool/float)
    - weights_dict : dict with all Linear/LayerNorm parameters
    - config_dict  : {"dim":int, "hidden_dim":int, "nomask":bool (opt.)}

    Returns
    -------
    torch.Tensor  [B, N, N, dim] (float32)
    """
    x, mask, weights, config = data
    device = x.device
    eps = 1e-5

    dim        = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask     = config.get("nomask", True)

    B, N, _, _ = x.shape
    total_rows = B * N * N          # R = B·N·N
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm – use cuDNN implementation (fast, FP32)
    # ------------------------------------------------------------------
    x_norm = torch.nn.functional.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )                                   # [B,N,N,dim] float32
    x_norm_fp16 = x_norm.view(total_rows, dim).contiguous().half()  # [R,dim] fp16

    # ------------------------------------------------------------------
    # 2️⃣ Fuse left/right projections + gates into one GEMM
    # ------------------------------------------------------------------
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)                     # [4·H, dim]

    fused = F.linear(x_norm_fp16, w_fused)   # [R, 4·H] fp16

    left_raw       = fused[:, 0:H]               # [R, H]
    right_raw      = fused[:, H:2*H]              # [R, H]
    left_gate_raw  = fused[:, 2*H:3*H]            # [R, H]
    right_gate_raw = fused[:, 3*H:4*H]            # [R, H]

    # Output‑gate (separate GEMM)
    out_gate_raw = F.linear(
        x_norm_fp16,
        weights["out_gate.weight"].to(torch.float16)   # [H, dim]
    )                                                   # [R, H] fp16

    # ------------------------------------------------------------------
    # 3️⃣ Mask (if any) & sigmoid gates
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_fp16 = mask.to(torch.float16).view(total_rows, 1)  # broadcast
        left_raw = left_raw * mask_fp16
        right_raw = right_raw * mask_fp16

    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()          # [R, H] fp16

    left  = left_raw * left_gate    # [R, H] fp16
    right = right_raw * right_gate  # [R, H] fp16

    # ------------------------------------------------------------------
    # 4️⃣ Batched N³ multiplicative update (B·H independent GEMMs)
    # ------------------------------------------------------------------
    left  = left.view(B, N, N, H)
    right = right.view(B, N, N, H)

    # bring hidden dim to batch axis → (B·H, N, N)
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B·H,N,N) fp16

    # reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # [B,N,N,H] fp16

    # ------------------------------------------------------------------
    # 5️⃣ Fused LayerNorm + out_gate multiplication (over hidden dim H)
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, H).contiguous()          # [R, H] fp16
    out_gate_flat = out_gate                                 # already [R, H] fp16

    out_norm_gate_fp16 = torch.empty_like(out_flat, dtype=torch.float16, device=device)

    BLOCK_M = 128      # rows per program (tuned)
    BLOCK_D = 128      # cols per iteration (tuned for H≤384)
    grid = (triton.cdiv(total_rows, BLOCK_M),)

    layernorm_gate_kernel[grid](
        out_flat,
        out_gate_flat,
        out_norm_gate_fp16,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Final linear projection back to original dim
    # ------------------------------------------------------------------
    final_fp16 = F.linear(
        out_norm_gate_fp16,
        weights["to_out.weight"].to(torch.float16)   # [dim, H]
    )                                                # [R, dim] fp16

    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final
