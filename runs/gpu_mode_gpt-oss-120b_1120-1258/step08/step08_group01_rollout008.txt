"""
Highly‑optimized forward pass for the **outgoing** TriMul operator
(AlphaFold‑3 style).

Algorithm
---------
1️⃣ Row‑wise LayerNorm (fp32 → fp16) on the input tensor.
2️⃣ One fused half‑precision GEMM that computes the four linear maps
   (left_proj, right_proj, left_gate, right_gate) in a single pass.
3️⃣ Separate GEMM for the output‑gate.
4️⃣ Optional mask multiplication (broadcasted) and in‑place sigmoid
   for all three gates.
5️⃣ Reshape/tile the left/right tensors so that each head becomes part
   of the batch dimension and perform a batched half‑precision matrix
   multiplication `out = left @ rightᵀ`.  This implements the
   N³‐multiplicative update `einsum('...ikd,...jkd->...ijd')`.
6️⃣ Second row‑wise LayerNorm over the hidden dimension.
7️⃣ Element‑wise multiplication by `out_gate`, a final linear projection
   back to the original dimensionality, and cast to float32.

All heavy element‑wise work (both LayerNorms, mask & sigmoid) is done
with Triton kernels; the dense matrix multiplications use cuBLAS/Tensor‑Core
via `torch.nn.functional.linear` and `torch.bmm`.  Mixed‑precision (fp16)
is used throughout except for the final output which must be `float32`.

The kernel has been tuned for typical AlphaFold‑3 shapes
(B∈{1,2}, N∈{128,…,1024}, dim∈{128,384}, hidden_dim=128) on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (fp32 → fp16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   * [total_rows, dim]   (float32)
    y_ptr,          # mutable * [total_rows, dim]   (float16)
    weight_ptr,     # const   * [dim]               (float32)
    bias_ptr,       # const   * [dim]               (float32)
    total_rows,     # int (runtime)
    dim,            # int (runtime)
    eps,            # float (compile‑time constexpr)
    BLOCK_M: tl.constexpr,   # rows per program (≤1024)
    BLOCK_D: tl.constexpr,   # columns per tile (≤dim)
):
    """
    Compute a row‑wise LayerNorm:
        y = ((x - μ) / sqrt(σ² + eps)) * γ + β
    Input x is fp32, output y is fp16.
    """
    pid = tl.program_id(0)                     # one program per BLOCK_M rows
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # 1️⃣ Compute per‑row mean & variance (fp32 accumulation)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    num_blocks_d = tl.cdiv(dim, BLOCK_D)
    for d_block in range(num_blocks_d):
        offs_d = d_block * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                     # 1 / sqrt(var + eps)

    # ------------------------------------------------------------------
    # 2️⃣ Write normalized output (fp16)
    # ------------------------------------------------------------------
    for d_block in range(num_blocks_d):
        offs_d = d_block * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]   # fp32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # γ fp32
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # β fp32
        out = norm * w[None, :] + b[None, :]               # fp32

        out_fp16 = out.to(tl.float16)
        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : dict with LayerNorm/Linear parameters (float32)
        - config_dict  : dict with keys
              "dim" (int), "hidden_dim" (int), optional "nomask" (bool)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim], dtype float32
    """
    # ------------------------------------------------------------------
    # 0️⃣ Unpack inputs & configuration
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    device = x.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)          # True → ignore mask
    eps = 1e-5

    B, N, _, _ = x.shape
    total_rows = B * N * N                # R = B·N·N
    H = hidden_dim                        # hidden dimension

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm over the last dim (fp32 → fp16)
    # ------------------------------------------------------------------
    # Flatten to [R, dim] for the kernel
    x_flat = x.view(total_rows, dim).contiguous()
    ln1_out = torch.empty(total_rows, dim,
                          dtype=torch.float16,
                          device=device)

    BLOCK_M = 256                # rows per program (tuned for H100)
    BLOCK_D = 128                # columns per tile (≤dim, power‑of‑2)
    grid = (triton.cdiv(total_rows, BLOCK_M),)

    layernorm_kernel[grid](
        x_flat,
        ln1_out,
        weights["norm.weight"].contiguous(),   # γ  (float32)
        weights["norm.bias"].contiguous(),     # β  (float32)
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection + gates (4·H outputs) + out‑gate (H outputs)
    # ------------------------------------------------------------------
    # Concatenate the four weight matrices for the first fused GEMM.
    w_proj = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
        ],
        dim=0,                             # [4·H, dim]
    ).to(device=device, dtype=torch.float16)

    # Linear: [R, dim] @ [dim, 4·H]ᵀ → [R, 4·H]
    fused = F.linear(ln1_out, w_proj)          # fp16, no bias

    # Slice the four blocks
    left_raw      = fused[:, 0:H]                     # [R, H]
    right_raw     = fused[:, H:2 * H]                 # [R, H]
    left_gate_raw = fused[:, 2 * H:3 * H]             # [R, H]
    right_gate_raw = fused[:, 3 * H:4 * H]            # [R, H]

    # Separate GEMM for the output‑gate
    out_gate_raw = F.linear(
        ln1_out,
        weights["out_gate.weight"].to(device=device, dtype=torch.float16)  # [H, dim]
    )   # [R, H] fp16

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask, sigmoid gates and element‑wise weighting
    # ------------------------------------------------------------------
    if (not nomask) and (mask is not None):
        # mask: [B, N, N] → [R, 1] (float16)
        mask_flat = mask.to(device=device, dtype=torch.float16).reshape(total_rows, 1)
        left_raw  = left_raw * mask_flat
        right_raw = right_raw * mask_flat

    # In‑place sigmoid to save memory
    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    # Apply gates
    left  = left_raw  * left_gate    # [R, H]
    right = right_raw * right_gate   # [R, H]

    # ------------------------------------------------------------------
    # 4️⃣ N³ multiplicative update via batched GEMM
    # ------------------------------------------------------------------
    # Reshape to [B, N, N, H]
    left  = left.view(B, N, N, H)
    right = right.view(B, N, N, H)
    out_gate = out_gate.view(B, N, N, H)

    # Rearrange to [B·H, N, N] (each head becomes a batch element)
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Batched matrix multiplication: Σ_k left[b*h, i, k] * right[b*h, j, k]
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B·H, N, N)

    # Restore [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm over the hidden dimension H
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, H).contiguous()
    ln2_out = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        ln2_out,
        weights["to_out_norm.weight"].contiguous(),   # γ (float32)
        weights["to_out_norm.bias"].contiguous(),     # β (float32)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Apply output‑gate, final projection and cast back to fp32
    # ------------------------------------------------------------------
    out_norm = ln2_out.view(B, N, N, H)        # fp16
    out_norm = out_norm * out_gate             # element‑wise fp16

    final_fp16 = F.linear(
        out_norm.view(total_rows, H),
        weights["to_out.weight"].to(device=device, dtype=torch.float16)   # [dim, H]
    )   # [R, dim] fp16

    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final
