"""
Highly‑optimized forward pass for the **outgoing** TriMul operator
(AlphaFold‑3 style). The heavy element‑wise LayerNorms are implemented
in Triton while the dense linear layers and the N³ multiplicative
update use native PyTorch (cuBLAS) for maximum throughput.

Algorithm
---------
1️⃣ Row‑wise LayerNorm in FP32 (custom Triton kernel).
2️⃣ Fuse the four linear projections (left, right, left_gate, right_gate)
   into a single `torch.nn.functional.linear` call by concatenating their
   weight matrices.
3️⃣ Compute the output‑gate with a second linear.
4️⃣ Apply optional mask, Sigmoid gates and element‑wise weighting.
5️⃣ Reshape and perform a batched matrix multiplication
   `out[i,j] = Σ_k left[i,k] * right[j,k]` (N³ update) using `torch.bmm`.
6️⃣ Second LayerNorm over the hidden dimension (Triton).
7️⃣ Apply the output‑gate, a final linear projection and return the result
   as `float32`.

All heavy element‑wise work (LayerNorm) is performed in Triton; the rest
leverages highly‑tuned cuBLAS kernels.  Mixed‑precision is avoided to
preserve numerical fidelity with the reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (FP32 → FP32)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp32_kernel(
    x_ptr,          # const   * [total_rows, dim]   (float32)
    y_ptr,          # mutable * [total_rows, dim]   (float32)
    weight_ptr,     # const   * [dim]               (float32)
    bias_ptr,       # const   * [dim]               (float32)
    total_rows,     # int (runtime)
    dim,            # int (runtime)
    eps,            # float (compile‑time constexpr)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per tile
):
    """
    Row‑wise LayerNorm:
        y = ((x - μ) / sqrt(σ² + eps)) * γ + β
    """
    pid = tl.program_id(0)                     # one program per BLOCK_M rows
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)  # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # 1️⃣ Compute per‑row mean & variance (FP32 accumulation)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]   # row‑major
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        )  # fp32 load
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                     # 1 / sqrt(var + eps)

    # ------------------------------------------------------------------
    # 2️⃣ Write normalized output (FP32)
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        )  # fp32

        # (x - μ) * rstd
        norm = (vals - mean[:, None]) * rstd[:, None]          # fp32

        # γ (weight) and β (bias)
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # fp32
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # fp32

        out = norm * w[None, :] + b[None, :]                    # fp32

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out, mask=mask_m[:, None] & mask_d[None, :])

    """
    Forward pass of the outgoing TriMul operator.

    ----

        - config_dict  : dict with keys

    -------
    torch.Tensor

    """
    # ------------------------------------------------------------------
    # 0️⃣ Unpack inputs & configuration
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    device = x.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)      # True → ignore mask
    eps = 1e-5

    B, N, _, _ = x.shape
    total_rows = B * N * N          # R = B·N·N
    H = hidden_dim                  # hidden dimension

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm: [R, dim] (FP32 → FP32)
    # ------------------------------------------------------------------

    ln1_out = torch.empty(total_rows, dim,
                          dtype=torch.float32,

    BLOCK_D = 64
    grid = (triton.cdiv(total_rows, BLOCK_M),)

        x_flat,
        ln1_out,

        weights["norm.bias"],     # β (float32)
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fuse left/right projections + their gates (4·H outputs)
    # ------------------------------------------------------------------
    # Concatenate the four projection weight matrices (each [H, dim])
    w_proj = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
        ],
        dim=0,                                     # [4·H, dim]
    )   # keep FP32 for numerical fidelity

    # Fused GEMM: [R, dim] @ [4·H, dim].T → [R, 4·H]
    fused = F.linear(ln1_out, w_proj)   # FP32, no bias

    left_raw      = fused[:, 0:H]                     # [R, H]
    right_raw     = fused[:, H:2 * H]                 # [R, H]
    left_gate_raw = fused[:, 2 * H:3 * H]             # [R, H]
    right_gate_raw = fused[:, 3 * H:4 * H]            # [R, H]

    # Separate output‑gate linear
    out_gate_raw = F.linear(
        ln1_out,
        weights["out_gate.weight"]                     # [H, dim]
    )   # [R, H] FP32

    # ------------------------------------------------------------------
    # 3️⃣ Apply optional mask, sigmoid gates and element‑wise weighting
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask: [B, N, N] → [R, 1] (float32)
        mask_flat = mask.to(device=device, dtype=torch.float32).reshape(total_rows, 1)
        left_raw  = left_raw * mask_flat
        right_raw = right_raw * mask_flat

    left_gate  = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate   = torch.sigmoid(out_gate_raw)

    left  = left_raw * left_gate       # [R, H]
    right = right_raw * right_gate     # [R, H]

    # ------------------------------------------------------------------
    # 4️⃣ Reshape for the N³ multiplicative update (batched GEMM)
    # ------------------------------------------------------------------
    left  = left.view(B, N, N, H)       # [B, N, N, H]
    right = right.view(B, N, N, H)      # [B, N, N, H]
    out_gate = out_gate.view(B, N, N, H)

    # Move head dimension to batch axis: [B, H, N, N]
    left_perm  = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    # Collapse batch & heads → [B·H, N, N]
    left_mat  = left_perm.view(B * H, N, N)
    right_mat = right_perm.view(B * H, N, N)

    # Batched matrix multiplication:
    # out_mat[b*h, i, j] = Σ_k left_mat[b*h, i, k] * right_mat[b*h, j, k]
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B·H, N, N)

    # Restore layout: [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm over the hidden dimension (H)
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, H).contiguous()   # [R, H]
    ln2_out = torch.empty_like(out_flat, dtype=torch.float32)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp32_kernel[grid2](
        out_flat,
        ln2_out,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Apply output gate, final projection and cast back to float32
    # ------------------------------------------------------------------
    out_norm = ln2_out.view(B, N, N, H)    # FP32
    out_norm = out_norm * out_gate         # element‑wise FP32

    final = F.linear(
        out_norm.view(total_rows, H),
        weights["to_out.weight"]           # [dim, H]
    )   # [R, dim] FP32

    final = final.view(B, N, N, dim)      # [B, N, N, dim]
    return final
```

**Explanation of key points**

* **LayerNorm** – Implemented as a Triton kernel that computes the mean and variance per row in FP32, normalizes, applies the learned scale (γ) and bias (β), and writes back FP32. This avoids any cross‑row synchronization.

* **Fused projection** – By concatenating the four projection weight matrices we replace four separate linear layers with a single matrix multiplication, reducing memory bandwidth and kernel launch overhead.

* **Mask handling** – If a mask is present (`nomask=False`), it is broadcast to the hidden dimension and multiplied with `left_raw` and `right_raw` before gating, exactly matching the reference implementation.

* **N³ update** – The multiplicative update `out[i,j] = Σ_k left[i,k] * right[j,k]` is performed as a batched `torch.bmm`, which maps to an efficient cuBLAS GEMM on the H100.

* **Second LayerNorm** – Re‑uses the same Triton kernel, now over the hidden dimension `H`.

* **Final projection** – A standard linear layer (`to_out`) brings the hidden representation back to the original dimension.

All operations are performed on the input device, respect the supplied weight tensors, and return a `float32` tensor of shape `[B, N, N, dim]`. The implementation satisfies the requirement of having at least one custom Triton kernel while remaining numerically faithful to the reference PyTorch code.
