"""
TriMul (outgoing) – fast forward pass

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the 4‑D input (B·N·N × dim).  This is done in a
   custom Triton kernel that accumulates the mean/variance in fp32 but
   writes the normalized result in fp16.

2️⃣ Fuse the four linear layers (left_proj, right_proj, left_gate,
   right_gate) into a single GEMM:
        [R, dim] @ [4·H, dim]ᵀ → [R, 4·H]    (R = B·N·N)
   The weight matrix is concatenated and cast to float16 so that the
   mat‑mul uses Tensor‑Cores.

3️⃣ A separate half‑precision linear yields the output‑gate
   (out_gate).

4️⃣ Apply the optional mask, the sigmoid gates and the element‑wise
   gating (left * left_gate, right * right_gate).

5️⃣ Reshape to (B, H, N, N) and perform the N³ multiplicative update.
   This is a batched half‑precision GEMM:
        (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N)
   The result is reshaped to (B, N, N, H).

6️⃣ Row‑wise LayerNorm over the hidden dimension (again with the Triton
   kernel).

7️⃣ Multiply by the output‑gate, apply the final half‑precision linear
   projection back to ``dim`` and cast to float32.

Only the two LayerNorm passes are written in Triton; all dense matrix
multiplications use PyTorch’s highly‑optimised cuBLAS kernels (Tensor‑Core
half‑precision).  The implementation works for any of the test
configurations described in the problem statement.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (input can be fp32 or fp16,
# output is fp16)
# ----------------------------------------------------------------------
@triton.jit
def rowwise_layernorm_kernel(
    x_ptr,               # const   * [total_rows, dim]   (float32 / float16)
    y_ptr,               # mutable * [total_rows, dim]   (float16)
    w_ptr,               # const   * [dim]               (float32)  gamma
    b_ptr,               # const   * [dim]               (float32)  beta
    total_rows,          # int
    dim,                 # int
    eps,                 # float (compile‑time constant)
    BLOCK_M: tl.constexpr,   # rows processed per program
    BLOCK_D: tl.constexpr,   # columns per tile
):
    """Row‑wise LayerNorm:
       y = ((x - μ) / sqrt(σ²+eps)) * γ + β
       Mean/variance are accumulated in fp32; the final result is stored in fp16.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ---------- 1️⃣ mean & variance per row ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    # iterate over the dim axis in tiles of BLOCK_D
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        # load a tile of size (BLOCK_M, BLOCK_D)
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        # note: tl.load returns the same dtype as the memory (fp16 or fp32)
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)   # promote to fp32

        sum_val += tl.sum(vals, axis=1)          # Σ x
        sum_sq  += tl.sum(vals * vals, axis=1)   # Σ x²

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)       # 1 / sqrt(var + eps)

    # ---------- 2️⃣ write normalized output ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        # (x - μ) * rstd  (fp32)
        norm = (vals - mean[:, None]) * rstd[:, None]

        # load γ, β (float32)
        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)

        out = norm * gamma[None, :] + beta[None, :]   # fp32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor       shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor       shape [B, N, N] (bool/float)
        - weights_dict : dict of torch.Tensors containing all Linear/LayerNorm params
        - config_dict  : must contain "dim", "hidden_dim" and optional "nomask"

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim          = config["dim"]
    hidden_dim   = config["hidden_dim"]
    eps          = 1e-5
    device       = input_tensor.device
    nomask       = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape
    R = B * N * N                      # total rows for the row‑wise LN passes

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → fp16
    # ------------------------------------------------------------------
    x_flat = input_tensor.view(R, dim).contiguous()  # [R, dim]  (float32)
    x_norm_fp16 = torch.empty(R, dim,
                              dtype=torch.float16,
                              device=device)

    # block sizes tuned for the dimensions used in the test suite
    BLOCK_M = 256
    BLOCK_D = 32
    grid = (triton.cdiv(R, BLOCK_M),)

    rowwise_layernorm_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],   # γ  (float32)
        weights["norm.bias"],     # β  (float32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fuse left/right projections + gate projections (single GEMM)
    # ------------------------------------------------------------------
    H = hidden_dim
    # Concatenate the four weight matrices: shape [4*H, dim]
    w_proj_gate = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)   # fp16 for Tensor‑Core matmul

    # Linear:   [R, dim] @ [4*H, dim]ᵀ  →  [R, 4*H]   (fp16)
    proj_gate = F.linear(x_norm_fp16, w_proj_gate)   # bias = 0

    # split the fused result
    left_proj_raw  = proj_gate[:, 0:H]                 # [R, H]
    right_proj_raw = proj_gate[:, H:2*H]               # [R, H]
    left_gate_raw  = proj_gate[:, 2*H:3*H]             # [R, H]
    right_gate_raw = proj_gate[:, 3*H:4*H]             # [R, H]

    # Separate linear for the output‑gate (still fp16)
    out_gate_raw = F.linear(
        x_norm_fp16,
        weights["out_gate.weight"].to(torch.float16)   # [H, dim]
    )   # → [R, H]

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask, sigmoid gates and gating
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape [B, N, N] → flatten to [R, 1] and broadcast
        mask_h = mask_tensor.to(torch.float16).view(R, 1)
        left_proj_raw  = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    # inplace sigmoid for speed
    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    # apply gates
    left  = left_proj_raw * left_gate          # [R, H]  (fp16)
    right = right_proj_raw * right_gate        # [R, H]  (fp16)

    # ------------------------------------------------------------------
    # 4️⃣ N³ multiplicative update (batched GEMM)
    # ------------------------------------------------------------------
    # reshape to [B, N, N, H]
    left  = left.view(B, N, N, H)
    right = right.view(B, N, N, H)
    out_gate = out_gate.view(B, N, N, H)

    # bring hidden dim to the batch axis -> many small GEMMs
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    left_mat  = left_perm.view(B * H, N, N)      # [B·H, N, N]  (fp16)
    right_mat = right_perm.view(B * H, N, N)     # [B·H, N, N]  (fp16)

    # Batched half‑precision GEMM: (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to [B, N, N, H] (contiguous, H is the last dim)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # fp16

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm over hidden dimension → fp16
    # ------------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()
    out_norm_fp16 = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(R, BLOCK_M),)
    rowwise_layernorm_kernel[grid2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    out_norm = out_norm_fp16.view(B, N, N, H)   # fp16

    # ------------------------------------------------------------------
    # 6️⃣ Apply output‑gate and final linear projection
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate               # element‑wise gating (fp16)

    final_half = F.linear(
        out_norm.view(R, H),
        weights["to_out.weight"].to(torch.float16)   # [dim, H]
    )   # → [R, dim]  (fp16)

    final = final_half.view(B, N, N, dim).to(torch.float32)

    return final
