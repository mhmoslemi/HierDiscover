@triton.jit
def rowwise_layernorm_gate_kernel(
    x_ptr,               # const * [total_rows, dim] (float16) -- read/write
    w_ptr,               # const * [dim] (float32)
    b_ptr,               # const * [dim] (float32)
    gate_ptr,            # const * [total_rows, dim] (float16)
    total_rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_R: tl.constexpr,
    BLOCK_D: tl.constexpr,
    HAS_GATE: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < total_rows

    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_R], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_r[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_r[:, None] & mask_d[None, :],
                       other=tl.float16(0.0)).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # Write back normalized (and optionally gated)
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_r[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_r[:, None] & mask_d[None, :],
                       other=tl.float16(0.0)).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=tl.float32(0.0))
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=tl.float32(0.0))

        out = norm * gamma[None, :] + beta[None, :]

        out_fp16 = out.to(tl.float16)

        if HAS_GATE:
            gate = tl.load(gate_ptr + offs_r[:, None] * dim + offs_d[None, :],
                           mask=mask_r[:, None] & mask_d[None, :],
                           other=tl.float16(0.0))
            out_fp16 = out_fp16 * gate

        tl.store(ptr, out_fp16,
                 mask=mask_r[:, None] & mask_d[None, :])

"""
Outgoing TriMul (AlphaFold‑3) – fast Triton‑accelerated forward pass
-----------------------------------------------------------------
1. Row‑wise LayerNorm over the last feature dimension (dim) → fp16.
2. Fuse the four linear maps (left_proj, right_proj, left_gate, right_gate)
   into a single GEMM (shape 4·H × dim).  The result is split into the four
   tensors and the three gate tensors are passed through a sigmoid.
3. Optional pair‑wise mask is applied to the two projection tensors.
4. Form the gated left/right tensors and reshape them to (B,N,N,H) (half).
5. Cubic update:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   realised as a batched GEMM (B·H, N, N) @ (B·H, N, N)ᵀ.
6. Second row‑wise LayerNorm over the hidden dimension H **and** element‑wise
   multiplication by the out‑gate are fused into a single Triton kernel,
   operating in‑place on the cubic‑update result.
7. Final linear projection back to the original feature dimension (dim).  
   The whole pipeline stays in fp16 until the very last cast to fp32.

The implementation uses two Triton kernels:
* `rowwise_layernorm_kernel` – standard per‑row LayerNorm.
* `rowwise_layernorm_gate_kernel` – LayerNorm that also multiplies by a
  per‑element gate (used for the second normalisation step).
Both kernels work in‑place on fp16 tensors to minimise memory traffic.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (input FP32 → output FP16)
# ----------------------------------------------------------------------
@triton.jit
def rowwise_layernorm_kernel(
    x_ptr,          # const    * [total_rows, dim] (float32)
    y_ptr,          # mutable  * [total_rows, dim] (float16)
    w_ptr,          # const    * [dim] (float32) – γ
    b_ptr,          # const    * [dim] (float32) – β
    total_rows,     # int32
    dim,            # int32
    eps,            # float32
    BLOCK_R: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # cols per inner loop
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)              # [BLOCK_R]
    mask_r = offs_r < total_rows

    # ---------- compute mean & variance (FP32) ----------
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_R], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_r[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_r[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)               # 1/√(var+eps)

    # ---------- write normalised values (FP16) ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_r[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_r[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)

        out = norm * gamma[None, :] + beta[None, :]
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_r[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_r[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2️⃣ Row‑wise LayerNorm + per‑element gate (in‑place on the tensor)
# ----------------------------------------------------------------------
@triton.jit
def rowwise_layernorm_gate_kernel(
    x_ptr,               # const   * [total_rows, dim] (float16) – also output
    w_ptr,               # const   * [dim] (float32) – γ
    b_ptr,               # const   * [dim] (float32) – β
    gate_ptr,            # const   * [total_rows, dim] (float16) – gating tensor
    total_rows,          # int32
    dim,                 # int32
    eps,                 # float32
    BLOCK_R: tl.constexpr,
    BLOCK_D: tl.constexpr,
    HAS_GATE: tl.constexpr,   # compile‑time bool (1 -> apply gate)
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < total_rows

    # ---- mean & var (FP32) ----
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_R], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_r[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_r[:, None] & mask_d[None, :],
                       other=tl.float16(0.0)).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- write normalised (and gated) values ----
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_r[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_r[:, None] & mask_d[None, :],
                       other=tl.float16(0.0)).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)

        out = norm * gamma[None, :] + beta[None, :]
        out_fp16 = out.to(tl.float16)

        if HAS_GATE:
            gate = tl.load(gate_ptr + offs_r[:, None] * dim + offs_d[None, :],
                           mask=mask_r[:, None] & mask_d[None, :],
                           other=tl.float16(0.0))
            out_fp16 = out_fp16 * gate

        tl.store(ptr, out_fp16,
                 mask=mask_r[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 3️⃣ Custom kernel entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float) – optional
        - weights_dict : model weights (see keys in documentation)
        - config_dict  : must contain "dim", "hidden_dim", "nomask"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N               # R = B·N·N
    H = hidden_dim

    # -----------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32 → FP16)
    # -----------------------------------------------------------------
    x_flat = input_tensor.reshape(total_rows, dim).contiguous()      # [R, dim] FP32
    x_norm = torch.empty(total_rows, dim, dtype=torch.float16,
                         device=device)                               # [R, dim] FP16

    BLOCK_R = 256        # rows per Triton program
    BLOCK_D = 128        # columns per inner loop (covers dim ≤ 384)
    grid = (triton.cdiv(total_rows, BLOCK_R),)

    rowwise_layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],          # γ   (FP32)
        weights["norm.bias"],            # β   (FP32)
        total_rows,
        dim,
        eps,
        BLOCK_R=BLOCK_R,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    # -----------------------------------------------------------------
    # 2️⃣ Fuse the four linear maps (left_proj, right_proj,
    #    left_gate, right_gate) into a single GEMM.
    # -----------------------------------------------------------------
    # Stack weight matrices (no bias) → (4·H, dim)
    w_4h = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # (4·H, dim)

    proj_gate = F.linear(x_norm, w_4h)       # [R, 4·H] FP16

    left_proj_raw  = proj_gate[:, 0:H]
    right_proj_raw = proj_gate[:, H:2 * H]
    left_gate_raw  = proj_gate[:, 2 * H:3 * H]
    right_gate_raw = proj_gate[:, 3 * H:4 * H]

    # Separate out‑gate (still half)
    out_gate_raw = F.linear(
        x_norm,
        weights["out_gate.weight"].to(torch.float16)   # (H, dim)
    )   # [R, H] FP16

    # -----------------------------------------------------------------
    # 3️⃣ Optional mask, sigmoid gates and element‑wise gating
    # -----------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.reshape(total_rows, 1).to(torch.float16)
        left_proj_raw  = left_proj_raw * mask_fp16
        right_proj_raw = right_proj_raw * mask_fp16

    left_gate  = left_gate_raw.sigmoid()
    right_gate = right_gate_raw.sigmoid()
    out_gate   = out_gate_raw.sigmoid()               # [R, H] FP16

    left  = left_proj_raw * left_gate                # [R, H] FP16
    right = right_proj_raw * right_gate              # [R, H] FP16

    # -----------------------------------------------------------------
    # 4️⃣ Reshape to (B, N, N, H)
    # -----------------------------------------------------------------
    left  = left.view(B, N, N, H)
    right = right.view(B, N, N, H)
    out_gate = out_gate.view(B, N, N, H)              # keep for later gating

    # -----------------------------------------------------------------
    # 5️⃣ Cubic update via batched GEMM:
    #     out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    # -----------------------------------------------------------------
    # (B, N, N, H) → (B·H, N, N) for torch.bmm
    left_perm  = left.permute(0, 3, 1, 2).contiguous()    # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]

    left_mat  = left_perm.view(B * H, N, N)    # [B·H, N, N]
    right_mat = right_perm.view(B * H, N, N)   # [B·H, N, N]

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N]

    # Restore (B, N, N, H) – still fp16
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 6️⃣ Second LayerNorm over hidden dimension H and multiply by out‑gate
    #     (fused into a single Triton kernel, in‑place on `out`).
    # -----------------------------------------------------------------
    out_flat = out.view(total_rows, H).contiguous()
    out_gate_flat = out_gate.view(total_rows, H).contiguous()

    grid2 = (triton.cdiv(total_rows, BLOCK_R),)
    rowwise_layernorm_gate_kernel[grid2](
        out_flat,                                    # in‑place tensor
        weights["to_out_norm.weight"],               # γ  (FP32)
        weights["to_out_norm.bias"],                 # β  (FP32)
        out_gate_flat,                               # gate (FP16)
        total_rows,
        H,
        eps,
        BLOCK_R=BLOCK_R,
        BLOCK_D=BLOCK_D,
        HAS_GATE=1,                                  # compile‑time flag
        num_warps=8,
    )
    # out_flat now contains (LayerNorm(out) * out_gate) in FP16.

    # -----------------------------------------------------------------
    # 7️⃣ Final linear projection back to `dim` (still half precision)
    # -----------------------------------------------------------------
    final = F.linear(
        out_flat,
        weights["to_out.weight"].to(torch.float16)   # (dim, H)
    )   # [R, dim] FP16

    final = final.view(B, N, N, dim).to(torch.float32)

    return final
```
