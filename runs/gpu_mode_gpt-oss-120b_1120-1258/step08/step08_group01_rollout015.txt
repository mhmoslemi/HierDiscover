@triton.jit
def layernorm_gate_kernel(x_ptr, gate_ptr, y_ptr, weight_ptr, bias_ptr, total_rows, dim, eps, BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # Compute mean and var per row
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals*vals, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # Process each block of columns: also load gate and compute out
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr_x = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        x_vals = tl.load(ptr_x, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)
        # normalize
        norm = (x_vals - mean[:, None]) * rstd[:, None]
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr + offs_d, mask=mask_d, other=0.0)
        out = norm * w[None, :] + b[None, :]
        # load gate
        ptr_gate = gate_ptr + offs_m[:, None] * dim + offs_d[None, :]
        gate_vals = tl.load(ptr_gate, mask=mask_m[:, None] & mask_d[None, :], other=0.0)  # fp16 if gate is fp16
        out * = gate_vals  # elementwise multiply (need to cast gate_vals to fp32 maybe)
        out_fp16 = out.to(tl.float16)
        ptr_y = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(ptr_y, out_fp16, mask=mask_m[:, None] & mask_d[None, :])
```

def custom_kernel(data):
    # Takes tuple
    input_tensor, mask, weights, config = data
    # Setup
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N
    H = hidden_dim

    # First layernorm
    x_flat = input_tensor.view(total_rows, dim).contiguous()
    ln1_out = torch.empty(total_rows, dim, dtype=torch.float16, device=device)

    BLOCK_M = 256
    BLOCK_D = 64
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        ln1_out,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # fused projection + gates (5*H rows)
    # concatenate all projection weights
    w_proj = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,   # [5*H, dim]
    ).to(device=device, dtype=torch.float16)

    # fused GEMM
    proj = F.linear(ln1_out, w_proj)   # [R, 5*H] fp16

    # split results
    left_raw     = proj[:, 0:H]                # [R, H]
    right_raw    = proj[:, H:2*H]              # [R, H]
    left_gate_raw  = proj[:, 2*H:3*H]          # [R, H]
    right_gate_raw = proj[:, 3*H:4*H]          # [R, H]
    out_gate_raw   = proj[:, 4*H:5*H]          # [R, H]

    # Optional mask
    if not nomask and mask is not None:
        mask_flat = mask.to(device=device, dtype=torch.float16).reshape(total_rows, 1)
        left_raw = left_raw * mask_flat
        right_raw = right_raw * mask_flat

    # gates sigmoid
    left_gate = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate = out_gate_raw.sigmoid_()

    # Apply gates (in-place)
    left = left_raw.mul_(left_gate)   # reuses left_raw storage
    right = right_raw.mul_(right_gate)  # reuses right_raw storage

@triton.jit
def layernorm_gate_kernel(
    x_ptr,         # const * [total_rows, dim] fp16
    gate_ptr,      # const * [total_rows, dim] fp16
    y_ptr,         # mutable * [total_rows, dim] fp16
    weight_ptr,    # const * [dim] fp32
    bias_ptr,      # const * [dim] fp32
    total_rows,    # int
    dim,           # int
    eps,           # float
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        # load original x
        ptr_x = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        x_vals = tl.load(ptr_x,
                         mask=mask_m[:, None] & mask_d[None, :],
                         other=0.0).to(tl.float32)
        # normalize
        norm = (x_vals - mean[:, None]) * rstd[:, None]

        # apply gamma, beta
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # fp32
        b = tl.load(bias_ptr + offs_d, mask=mask_d, other=0.0)    # fp32
        norm = norm * w[None, :] + b[None, :]
        # load gate
        ptr_gate = gate_ptr + offs_m[:, None] * dim + offs_d[None, :]
        gate_vals = tl.load(ptr_gate,
                            mask=mask_m[:, None] & mask_d[None, :],
                            other=0.0)   # fp16
        # multiply gate (convert to fp32)
        out = norm * gate_vals.to(tl.float32)

        # store as fp16
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])

Optimized outgoing TriMul forward pass.

The implementation follows the reference PyTorch module but
- Fuses the four linear projections and their three gates (plus the

    var = sum_sq / dim - mean * mean

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):

        b = tl.load(bias_ptr + offs_d, mask=mask_d, other=0.0)

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (AlphaFold‑3 style).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : {"dim": int, "hidden_dim": int, "nomask": bool (optional)}

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] and dtype float32.
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    inp, mask, weights, cfg = data
    device = inp.device
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5

    B, N, _, _ = inp.shape
    total_rows = B * N * N
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32 -> FP16)
    # ------------------------------------------------------------------
    x_flat = inp.view(total_rows, dim).contiguous()
    ln1_out = torch.empty(total_rows, dim, dtype=torch.float16, device=device)

    BLOCK_M = 256
    BLOCK_D = 64
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        ln1_out,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection + five linear heads (left, right, gates, out_gate)
    # ------------------------------------------------------------------
    w_proj = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,   # [5·H, dim]
    ).to(device=device, dtype=torch.float16)

    proj = F.linear(ln1_out, w_proj)            # [R, 5·H]  fp16

    # split
    left_raw     = proj[:, 0:H]                     # [R, H]
    right_raw    = proj[:, H:2 * H]                 # [R, H]
    left_gate_raw  = proj[:, 2 * H:3 * H]           # [R, H]
    right_gate_raw = proj[:, 3 * H:4 * H]           # [R, H]
    out_gate_raw   = proj[:, 4 * H:5 * H]           # [R, H]

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask + sigmoid gates
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_flat = mask.to(device=device, dtype=torch.float16).reshape(total_rows, 1)
        left_raw = left_raw * mask_flat
        right_raw = right_raw * mask_flat

    left_gate = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate = out_gate_raw.sigmoid_()

    # In‑place gating → reuse memory
    left = left_raw.mul_(left_gate)      # [R, H] (fp16)
    right = right_raw.mul_(right_gate)   # [R, H] (fp16)

    # ------------------------------------------------------------------
    # 4️⃣ N³ multiplicative update (einsum, no extra permute/copy)
    # ------------------------------------------------------------------
    left = left.view(B, N, N, H)
    right = right.view(B, N, N, H)

    out = torch.einsum('b i k h, b j k h -> b i j h', left, right)   # [B, N, N, H] fp16

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm + output‑gate (fused Triton kernel)
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, H).contiguous()
    out_gate_flat = out_gate.view(total_rows, H).contiguous()
    ln2_out = torch.empty_like(out_flat, dtype=torch.float16, device=device)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_gate_kernel[grid2](
        out_flat,
        out_gate_flat,
        ln2_out,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Final linear projection (FP16 → FP32)
    # ------------------------------------------------------------------
    final_fp16 = F.linear(
        ln2_out,
        weights["to_out.weight"].to(device=device, dtype=torch.float16)
    )   # [R, dim] fp16

    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final

Highly‑optimized forward pass for the **outgoing** TriMul operator
(AlphaFold‑3 style).

Key ideas
---------
1️⃣ Row‑wise LayerNorm (FP32 statistics → FP16 output) – custom Triton kernel.
2️⃣ Fuse the four projections *and* the three gate projections *plus* the
   output‑gate projection into a **single** GEMM (5·H rows).
3️⃣ Apply the optional mask and sigmoid gates in‑place to avoid temporaries.
4️⃣ Compute the N³ multiplicative update with a single `torch.einsum`,
   thereby removing the heavyweight `permute+contiguous` copies required for
   the batched `bmm` implementation.
5️⃣ Fuse the second LayerNorm and the element‑wise output‑gate multiplication
   into another Triton kernel, cutting one extra memory pass.
6️⃣ Final linear projection (FP16) and cast back to FP32.

All heavy element‑wise work (LayerNorms, mask & gate handling) is performed
in Triton; dense matrix multiplications use cuBLAS for maximum throughput.
"""

import triton
import triton.language as tl

# ----------------------------------------------------------------------
@triton.jit

    bias_ptr,       # const   * [dim]           (float32)
    total_rows,     # int (runtime)

        norm = (x_vals - mean[:, None]) * rstd[:, None]   # fp32

        # γ, β
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)
        norm = norm * w[None, :] + b[None, :]             # fp32

        # gate (fp16 → fp32)
        g_ptrs = gate_ptr + offs_m[:, None] * dim + offs_d[None, :]
        g_vals = tl.load(
            g_ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        out = norm * g_vals

        tl.store(
            y_ptr + offs_m[:, None] * dim + offs_d[None, :],
            out.to(tl.float16),
            mask=mask_m[:, None] & mask_d[None, :],
        )


# ----------------------------------------------------------------------
# 3️⃣ Custom Triton‑accelerated forward pass
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, [B, N, N] (bool/float) or None
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : {"dim": int, "hidden_dim": int,
                         "nomask": bool (optional)}

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] and dtype float32.
    """
    # ------------------------------------------------------------------
    # unpack & constants
    # ------------------------------------------------------------------
    inp, mask, weights, cfg = data
    device = inp.device
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5

    B, N, _, _ = inp.shape
    total_rows = B * N * N               # R = B·N·N
    H = hidden_dim                       # hidden dimension

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm   (float32 → float16)
    # ------------------------------------------------------------------
    x_flat = inp.view(total_rows, dim).contiguous()          # [R, dim] float32
    ln1_out = torch.empty(total_rows, dim,
                          dtype=torch.float16,
                          device=device)

    BLOCK_M = 256
    BLOCK_D = 64
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        ln1_out,
        weights["norm.weight"],   # γ  (float32)
        weights["norm.bias"],     # β  (float32)
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection + 5 linear heads (left, right, 3 gates, out_gate)
    # ------------------------------------------------------------------
    w_proj = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,                                    # [5·H, dim]
    ).to(device=device, dtype=torch.float16)

    proj = F.linear(ln1_out, w_proj)               # [R, 5·H]  fp16

    # split the fused result
    left_raw      = proj[:, 0:H]                     # [R, H]
    right_raw     = proj[:, H:2 * H]                 # [R, H]
    left_gate_raw = proj[:, 2 * H:3 * H]             # [R, H]
    right_gate_raw = proj[:, 3 * H:4 * H]            # [R, H]
    out_gate_raw   = proj[:, 4 * H:5 * H]            # [R, H]

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask + sigmoid gates (in‑place)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_flat = mask.to(device=device, dtype=torch.float16).reshape(total_rows, 1)
        left_raw  = left_raw * mask_flat
        right_raw = right_raw * mask_flat

    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    # in‑place gating → reuse the same memory buffers
    left  = left_raw.mul_(left_gate)       # [R, H] fp16
    right = right_raw.mul_(right_gate)     # [R, H] fp16

    # ------------------------------------------------------------------
    # 4️⃣ N³ multiplicative update (einsum – no extra permute/copy)
    # ------------------------------------------------------------------
    left  = left.view(B, N, N, H)          # [B, N, N, H]
    right = right.view(B, N, N, H)         # [B, N, N, H]

    # out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    out = torch.einsum('b i k h, b j k h -> b i j h', left, right)   # [B,N,N,H] fp16

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm + output‑gate (fused Triton kernel)
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, H).contiguous()           # [R, H] fp16
    out_gate_flat = out_gate.view(total_rows, H).contiguous()
    ln2_out = torch.empty_like(out_flat, dtype=torch.float16, device=device)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_gate_kernel[grid2](
        out_flat,
        out_gate_flat,
        ln2_out,
        weights["to_out_norm.weight"],   # γ  (float32)
        weights["to_out_norm.bias"],     # β  (float32)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Final linear projection & cast back to FP32
    # ------------------------------------------------------------------
    final_fp16 = F.linear(
        ln2_out,
        weights["to_out.weight"].to(device=device, dtype=torch.float16)  # [dim, H]
    )   # [R, dim] fp16

    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final
