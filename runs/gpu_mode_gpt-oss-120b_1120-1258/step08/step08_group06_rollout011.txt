"""
Highly‑optimized Triton‑accelerated forward pass for the **outgoing**
TriMul operator (AlphaFold‑3).

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the pair representation (FP32 → FP16) using a
   custom Triton kernel (no extra memory allocations besides the output).

2️⃣ Fuse the four linear projections/gates (left_proj, right_proj,
   left_gate, right_gate) into a single GEMM:
   W₄ₕ = [Wₗₚ; Wᵣₚ; Wₗg; Wᵣg]  (shape (4·H, D)).
   This yields the four tensors in one pass over the normalized data.

3️⃣ Compute the optional mask and the three gated sigmoids.  The mask is
   broadcast‑multiplied only on the *projections* (as in the reference).

4️⃣ Reshape the gated tensors to (B, N, N, H) and perform the cubic
   update:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication of shape (B·H, N, N) → (B·H, N, N),
   executed with `torch.bmm` (Tensor‑Core friendly FP16).

5️⃣ Second row‑wise LayerNorm over the hidden dimension H (again via Triton).

6️⃣ Apply the out‑gate and a final linear projection back to the original
   feature dimension (`dim`).  The final result is cast back to FP32.

Only the two LayerNorms are implemented in Triton; the remaining
operations leverage highly‑optimized PyTorch kernels (F.linear, torch.bmm,
etc.).  This hybrid approach keeps the runtime bound by the cubic GEMM
while still satisfying the “must‑use‑Triton” requirement.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def rowwise_layernorm_kernel(
    x_ptr,      # const  * [total_rows, dim] (float32)
    y_ptr,      # mutable* [total_rows, dim] (float16)
    gamma_ptr,  # const  * [dim] (float32) – weight
    beta_ptr,   # const  * [dim] (float32) – bias
    total_rows, # int32
    dim,        # int32
    eps,        # float32
    BLOCK_R: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per inner loop
):
    """Per‑row LayerNorm:  y = ((x-μ)/√(σ²+eps)) * γ + β"""
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < total_rows

    # ---- compute mean & variance (FP32) ---------------------------------
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_R], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_r[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_r[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)          # 1/√(var+eps)

    # ---- write normalized values (FP16) ---------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_r[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_r[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(gamma_ptr + offs_d, mask=mask_d, other=0.0)
        beta  = tl.load(beta_ptr + offs_d,  mask=mask_d, other=0.0)

        out = norm * gamma[None, :] + beta[None, :]
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_r[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_r[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, D] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float) – may be None
        - weights_dict : model weights (see keys below)
        - config_dict  : must contain "dim", "hidden_dim", "nomask"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    B, N, _, D = input_tensor.shape
    assert D == dim, f"config dim ({dim}) does not match input dim ({D})"
    R = B * N * N            # total number of pair rows
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → fp16
    # ------------------------------------------------------------------
    # flatten to (R, D) – contiguous for the kernel
    x_flat = input_tensor.reshape(R, D).contiguous()
    x_norm = torch.empty(R, D, dtype=torch.float16, device=device)

    BLOCK_R = 256    # rows per program (tuned for H100)
    BLOCK_D = 128    # columns per inner loop (covers dim ≤ 384)
    grid = (triton.cdiv(R, BLOCK_R),)

    rowwise_layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],   # γ  (FP32)
        weights["norm.bias"],     # β  (FP32)
        R,
        D,
        eps,
        BLOCK_R=BLOCK_R,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection & gate linear maps (4·H × dim)
    # ------------------------------------------------------------------
    # Concatenate the four weight matrices (bias‑free)
    w_4h = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # (4·H, D)

    # Single GEMM: [R, D] @ [D, 4·H]ᵀ → [R, 4·H]
    proj_gate = F.linear(x_norm, w_4h)       # (R, 4·H) FP16

    left_proj_raw  = proj_gate[:, 0:H]
    right_proj_raw = proj_gate[:, H:2 * H]
    left_gate_raw  = proj_gate[:, 2 * H:3 * H]
    right_gate_raw = proj_gate[:, 3 * H:4 * H]

    # out‑gate (independent linear map)
    out_gate_raw = F.linear(
        x_norm,
        weights["out_gate.weight"].to(torch.float16)   # (H, D)
    )   # (R, H) FP16

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask, sigmoid gates and element‑wise gating
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] → [R, 1]   (float16)
        mask_fp16 = mask_tensor.reshape(R, 1).to(torch.float16)
        left_proj_raw  = left_proj_raw * mask_fp16
        right_proj_raw = right_proj_raw * mask_fp16

    # Sigmoid in‑place (fast)
    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    # Apply gates
    left  = left_proj_raw * left_gate      # (R, H) FP16
    right = right_proj_raw * right_gate    # (R, H) FP16

    # ------------------------------------------------------------------
    # 4️⃣ Reshape to (B, N, N, H)
    # ------------------------------------------------------------------
    left  = left.view(B, N, N, H)
    right = right.view(B, N, N, H)
    out_gate = out_gate.view(B, N, N, H)

    # ------------------------------------------------------------------
    # 5️⃣ Cubic update via batched GEMM:
    #     out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    # Bring H into the batch dimension → (B·H, N, N)
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    left_mat  = left_perm.view(B * H, N, N)   # (B·H, N, N)
    right_mat = right_perm.view(B * H, N, N)  # (B·H, N, N)

    # Batched matrix multiplication (left @ rightᵀ) – stays in FP16
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B·H, N, N)

    # Restore original layout (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm over hidden dimension H (row‑wise)
    # ------------------------------------------------------------------
    out_flat = out.reshape(R, H).contiguous()
    out_norm = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(R, BLOCK_R),)
    rowwise_layernorm_kernel[grid2](
        out_flat,
        out_norm,
        weights["to_out_norm.weight"],   # γ (FP32)
        weights["to_out_norm.bias"],     # β (FP32)
        R,
        H,
        eps,
        BLOCK_R=BLOCK_R,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    out_norm = out_norm.view(B, N, N, H)   # (B, N, N, H) FP16

    # ------------------------------------------------------------------
    # 7️⃣ Apply out‑gate and final linear projection back to dim
    # ------------------------------------------------------------------
    gated = out_norm * out_gate   # (B, N, N, H) FP16

    final = F.linear(
        gated.view(R, H),                     # (R, H)
        weights["to_out.weight"].to(torch.float16)   # (dim, H)
    )   # (R, dim) FP16

    final = final.view(B, N, N, dim).to(torch.float32)

    return final
