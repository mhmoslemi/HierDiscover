"""
custom_kernel: high‑performance forward of the *outgoing* TriMul operator.

Algorithm
---------
1️⃣ Row‑wise LayerNorm over the channel dimension `dim` (FP32 → FP16).
2️⃣ Fuse the four linear layers (left/right projection + left/right gate) into
   a single GEMM:  X_norm @ W_fused.T  →  [R, 4·H]   (R = B·N·N, H = hidden_dim).
   Split the result into left_raw, right_raw, left_gate_raw, right_gate_raw.
3️⃣ Compute the output‑gate (a separate GEMM) from the normalized input.
4️⃣ (Optional) Apply binary mask to the two projected tensors.
   Apply sigmoid to the three gate tensors (in‑place) and fuse them:
        left  = left_raw  * sigmoid(left_gate_raw)
        right = right_raw * sigmoid(right_gate_raw)
5️⃣ Reshape left/right → (B·H, N, N) and perform a batched GEMM:
        out_mat[b·H] = left_mat[b·H] @ right_mat[b·H].T
   Reshape back to (B, N, N, H).
6️⃣ Fuse the second LayerNorm (over the hidden dim H) **and** the
   element‑wise multiplication by `out_gate` into a single Triton kernel.
7️⃣ Final linear projection back to the original `dim` (FP16 GEMM) and cast
   to float32 for the output.

Only the two LayerNorm kernels are written in Triton; the remaining
operations use highly‑optimized torch kernels (FP16 GEMMs, batched matrix
multiplication, etc.).  This design keeps the critical reductions (means /
variances) on‑device while leveraging torch’s fast GEMM implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   * [total_rows, dim] (float32)
    y_ptr,          # mutable * [total_rows, dim] (float16)
    weight_ptr,     # const   * [dim]            (float32)  # γ
    bias_ptr,       # const   * [dim]            (float32)  # β
    total_rows,     # int32   (runtime)
    dim,            # int32   (runtime)
    eps,            # float   (compile‑time)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # cols per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # 1️⃣ Compute per‑row mean & variance (FP32)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)      # FP32 load
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                             # 1 / sqrt(var+eps)

    # ------------------------------------------------------------------
    # 2️⃣ Write normalized values γ·(x−μ)·rstd + β (FP16)
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # β
        out = norm * w[None, :] + b[None, :]                     # still FP32

        out_fp16 = out.to(tl.float16)
        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Triton kernel: fused LayerNorm + element‑wise gate (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_gate_kernel(
    x_ptr,          # const   * [total_rows, dim] (float32)   # raw tensor before LN
    gate_ptr,       # const   * [total_rows, dim] (float16)   # gate (sigmoid already)
    y_ptr,          # mutable * [total_rows, dim] (float16)   # (LN(x) * γ + β) * gate
    weight_ptr,     # const   * [dim]            (float32)  # γ of LN
    bias_ptr,       # const   * [dim]            (float32)  # β of LN
    total_rows,     # int32   (runtime)
    dim,            # int32   (runtime)
    eps,            # float   (compile‑time)
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # Compute per‑row mean & variance (ignoring gate)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------------------------------------------------------
    # Normalise, apply LN γ/β and multiply by gate (all FP32 → store FP16)
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        # load raw activation
        x_vals = tl.load(x_ptr + offs_m[:, None] * dim + offs_d[None, :],
                         mask=mask_m[:, None] & mask_d[None, :],
                         other=0.0).to(tl.float32)

        # load gate (already sigmoid'ed)
        g_vals = tl.load(gate_ptr + offs_m[:, None] * dim + offs_d[None, :],
                         mask=mask_m[:, None] & mask_d[None, :],
                         other=0.0).to(tl.float32)

        norm = (x_vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # β

        # (norm*γ + β) * gate
        out = (norm * w[None, :] + b[None, :]) * g_vals
        out_fp16 = out.to(tl.float16)

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor shape [B, N, N] (bool/float) or None
        - weights_dict : mapping with all Linear/LayerNorm parameters
        - config_dict  : {"dim": int, "hidden_dim": int,
                          "nomask": bool (optional)}

    Returns:
        torch.Tensor shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # 0️⃣ unpack inputs
    # --------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    total_rows = B * N * N          # R = B·N·N
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → FP16
    # --------------------------------------------------------------
    # flatten to (R, dim)
    x_flat = x.view(total_rows, dim).contiguous()           # FP32
    x_norm_fp16 = torch.empty(total_rows, dim,
                              dtype=torch.float16,
                              device=device)

    BLOCK_M = 128   # rows per program (tuned)
    BLOCK_D = 64    # cols per iteration (tuned)
    grid = (triton.cdiv(total_rows, BLOCK_M),)

    layernorm_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],   # γ (float32)
        weights["norm.bias"],     # β (float32)
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    # --------------------------------------------------------------
    # 2️⃣ Fuse left/right projections + their gates (4·H outputs)
    # --------------------------------------------------------------
    # Concatenate the four weight matrices (each [H, dim])
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)  # shape [4·H, dim]

    # Linear: (R, dim) @ (4·H, dim).T → (R, 4·H)
    fused = F.linear(x_norm_fp16, w_fused)  # FP16

    left_raw       = fused[:, 0:H]               # [R, H]
    right_raw      = fused[:, H:2 * H]           # [R, H]
    left_gate_raw  = fused[:, 2 * H:3 * H]       # [R, H]
    right_gate_raw = fused[:, 3 * H:4 * H]       # [R, H]

    # Separate output‑gate linear (still from normalized input)
    out_gate_raw = F.linear(
        x_norm_fp16,
        weights["out_gate.weight"].to(torch.float16)   # [H, dim]
    )                                                   # [R, H]

    # --------------------------------------------------------------
    # 3️⃣ Apply mask (if any) and sigmoid gates
    # --------------------------------------------------------------
    if not nomask and mask is not None:
        # mask shape [B,N,N] → (R,1) broadcast over hidden dim
        mask_fp16 = mask.reshape(total_rows, 1).to(torch.float16)
        left_raw = left_raw * mask_fp16
        right_raw = right_raw * mask_fp16

    # In‑place sigmoid (FP16)
    left_gate = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate = out_gate_raw.sigmoid_()          # [R, H] FP16

    # Gated tensors
    left = left_raw * left_gate   # [R, H]
    right = right_raw * right_gate  # [R, H]

    # --------------------------------------------------------------
    # 4️⃣ Batched N³ multiplicative update (B·H independent GEMMs)
    # --------------------------------------------------------------
    # reshape to (B, N, N, H)
    left = left.view(B, N, N, H)
    right = right.view(B, N, N, H)

    # move hidden dim to batch axis → (B·H, N, N)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # (B·H, N, N) @ (B·H, N, N)^T → (B·H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 5️⃣ Fused LayerNorm (over hidden dim) + multiplication by out_gate
    # --------------------------------------------------------------
    out_flat = out.view(total_rows, H).contiguous()   # FP16
    out_gate_flat = out_gate                           # already FP16
    out_norm_gate_fp16 = torch.empty_like(out_flat, dtype=torch.float16, device=device)

    layernorm_gate_kernel[grid](
        out_flat,
        out_gate_flat,
        out_norm_gate_fp16,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    # --------------------------------------------------------------
    # 6️⃣ Final linear projection back to original dim
    # --------------------------------------------------------------
    final_fp16 = F.linear(
        out_norm_gate_fp16,
        weights["to_out.weight"].to(torch.float16)   # [dim, H]
    )                                                # [R, dim] FP16

    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final
```
