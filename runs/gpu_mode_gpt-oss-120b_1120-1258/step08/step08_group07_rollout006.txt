"""
TriMul “outgoing” forward pass – Triton‑accelerated LayerNorm + fused
linear projections.

Algorithm
---------
1️⃣  Row‑wise LayerNorm on the input tensor (B·N·N × dim).  The kernel
    computes mean / variance in fp32 and writes the normalized result
    in fp16.

2️⃣  Fuse the four  (left_proj, right_proj, left_gate, right_gate)  and
    the output‑gate linear layers into a *single* GEMM:
        [R, dim] @ [5·H, dim]ᵀ → [R, 5·H]   (R = B·N·N, H = hidden_dim)

3️⃣  Apply the optional binary mask, sigmoid gates and element‑wise
    gating for “left” and “right”.

4️⃣  Reshape to (B, H, N, N) and perform the N³ multiplicative update
    with a batched half‑precision GEMM (torch.bmm).

5️⃣  Row‑wise LayerNorm over the hidden dimension (again Triton).

6️⃣  Apply the output‑gate, a final half‑precision linear projection back
    to ‘dim’, and cast the result to float32.

Only the two LayerNorm passes are written in Triton; all dense matrix
multiplications use cuBLAS/Tensor‑Core via torch.nn.functional.linear and
torch.bmm, which is optimal on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (input fp32 → output fp16)
# ----------------------------------------------------------------------
@triton.jit
def rowwise_layernorm_kernel(
    x_ptr,          # const   * [total_rows, dim]   (float32)
    y_ptr,          # mutable * [total_rows, dim]   (float16)
    w_ptr,          # const   * [dim]               (float32)  gamma
    b_ptr,          # const   * [dim]               (float32)  beta
    total_rows,     # int
    dim,            # int
    eps,            # float   (compile‑time constant)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per tile
):
    """Row‑wise LayerNorm:
       y = ((x - μ) / sqrt(σ²+eps)) * γ + β
       Reduction uses fp32, output is fp16.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    # row indices handled by this program
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ---------- 1️⃣ mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    # iterate over dim in tiles of size BLOCK_D
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)               # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)          # 1 / sqrt(var + eps)

    # ---------- 2️⃣ write normalized output ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]   # fp32

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)  # fp32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)  # fp32

        out = norm * gamma[None, :] + beta[None, :]      # fp32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom forward kernel for outgoing TriMul
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor  [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor  [B, N, N]   (bool/float)  (may be None)
        - weights_dict : dict of torch.Tensors containing all Linear/LayerNorm params
        - config_dict  : must contain "dim", "hidden_dim" and optional "nomask"

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    device = input_tensor.device
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape
    R = B * N * N                         # total rows for row‑wise ops
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → fp16
    # --------------------------------------------------------------
    # flatten to (R, dim)
    x_flat = input_tensor.view(R, dim).contiguous()          # fp32
    x_norm_fp16 = torch.empty(R, dim, dtype=torch.float16, device=device)

    # choose block sizes (tuned for H100, FP16/FP32)
    BLOCK_M = 1024               # rows per program
    BLOCK_D = 128                # columns per tile (multiple of 32)
    grid = (triton.cdiv(R, BLOCK_M),)
    rowwise_layernorm_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],   # gamma, fp32
        weights["norm.bias"],     # beta,  fp32
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    # --------------------------------------------------------------
    # 2️⃣ Fuse left/right projections, gates and output‑gate into one GEMM
    # --------------------------------------------------------------
    # Concatenate the five weight matrices: [left_proj, right_proj,
    # left_gate, right_gate, out_gate] -> shape (5*H, dim)
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
            weights["out_gate.weight"],    # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)   # fp16 for Tensor‑Core GEMM

    # Linear: [R, dim] @ [5*H, dim]ᵀ → [R, 5*H] (fp16)
    proj_all = F.linear(x_norm_fp16, w_fused)   # no bias

    # Slice the fused result
    left_proj_raw  = proj_all[:, 0:H]                 # [R, H]
    right_proj_raw = proj_all[:, H:2*H]               # [R, H]
    left_gate_raw  = proj_all[:, 2*H:3*H]             # [R, H]
    right_gate_raw = proj_all[:, 3*H:4*H]             # [R, H]
    out_gate_raw   = proj_all[:, 4*H:5*H]             # [R, H]

    # --------------------------------------------------------------
    # 3️⃣ Optional mask, sigmoid gates and gating
    # --------------------------------------------------------------
    if not nomask:
        # mask shape [B, N, N] → [R, 1] broadcast over hidden dim
        mask_h = mask_tensor.to(torch.float16).view(R, 1)
        left_proj_raw = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    # In‑place sigmoid (half‑precision)
    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    # Gating
    left  = left_proj_raw * left_gate          # [R, H] fp16
    right = right_proj_raw * right_gate        # [R, H] fp16

    # --------------------------------------------------------------
    # 4️⃣ N³ multiplicative update (batched GEMM)
    # --------------------------------------------------------------
    # Reshape to (B, N, N, H)
    left  = left.view(B, N, N, H)
    right = right.view(B, N, N, H)
    out_gate = out_gate.view(B, N, N, H)

    # Move hidden dim to the batch axis: (B, H, N, N)
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    # Collapse B and H → (B*H, N, N)
    left_mat  = left_perm.view(B * H, N, N)      # fp16
    right_mat = right_perm.view(B * H, N, N)     # fp16

    # Batched GEMM: (B*H, N, N) @ (B*H, N, N)ᵀ → (B*H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # fp16

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm over hidden dimension → fp16
    # --------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()
    out_norm_fp16 = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(R, BLOCK_M),)
    rowwise_layernorm_kernel[grid2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],   # gamma, fp32
        weights["to_out_norm.bias"],     # beta,  fp32
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    out_norm = out_norm_fp16.view(B, N, N, H)   # fp16

    # --------------------------------------------------------------
    # 6️⃣ Apply output gate and final projection back to `dim`
    # --------------------------------------------------------------
    out_norm = out_norm * out_gate               # [B,N,N,H] fp16

    final_half = F.linear(
        out_norm.view(R, H),
        weights["to_out.weight"].to(torch.float16)   # [dim, H]
    )   # → [R, dim] fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)

    return final
