"""
Outgoing TriMul (AlphaFold‑3) – highly‑fused forward pass.

Algorithm (fp16‑centric)
------------------------
1️⃣ Row‑wise LayerNorm on the input tensor (B·N·N rows, dim cols) – Triton.
2️⃣ Fuse the four linear projections (left, right, left‑gate, right‑gate)
   into one GEMM using `torch.nn.functional.linear`.  The result is split.
3️⃣ Separate linear for the output‑gate.
4️⃣ Optional mask application (broadcasted) and sigmoid on the three gates.
   The gates are applied in‑place to keep memory traffic low.
5️⃣ Reshape & permute so that each head becomes part of the batch:
   `[B, N, N, H] → [B·H, N, N]`.  Batched matrix multiplication
   (`torch.bmm`) implements the `einsum('... i k d, ... j k d -> ... i j d')`.
6️⃣ Second LayerNorm over the hidden dimension – Triton.
7️⃣ Apply the output‑gate, final linear projection, and cast back to float32.

All heavy element‑wise work (LayerNorm, optional mask) runs in Triton;
dense matrix multiplications use cuBLAS Tensor‑Cores (fp16) for maximum
throughput on H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton row‑wise LayerNorm (fp32 → fp16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   * [rows, dim]   (float32)
    y_ptr,          # mutable * [rows, dim]   (float16)
    weight_ptr,     # const   * [dim]         (float32)
    bias_ptr,       # const   * [dim]         (float32)
    rows,           # int (runtime)
    dim,            # int (runtime)
    eps,            # float (compile‑time constexpr)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per tile
):
    """
    Compute row‑wise LayerNorm:
        y = ((x - μ) / sqrt(σ² + eps)) * γ + β
    where μ,σ² are per‑row mean/variance, γ/β are the learned scale/bias.
    """
    pid = tl.program_id(0)                     # one program per BLOCK_M rows
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M) # [BLOCK_M]
    mask_m = offs_m < rows

    # --------------------------------------------------------------
    # 1️⃣ Compute per‑row mean & variance (float32 accumulation)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    num_iter_d = tl.cdiv(dim, BLOCK_D)
    for d_iter in range(num_iter_d):
        offs_d = d_iter * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)             # fp32 load
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                     # 1 / sqrt(var + eps)

    # --------------------------------------------------------------
    # 2️⃣ Normalise, apply scale/bias and store (fp16)
    # --------------------------------------------------------------
    for d_iter in range(num_iter_d):
        offs_d = d_iter * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        # (x - μ) * rstd  -- still fp32
        norm = (vals - mean[:, None]) * rstd[:, None]

        # γ, β (scale / bias) are fp32
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # [dim]
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # [dim]

        out = norm * w[None, :] + b[None, :]                      # fp32
        out_fp16 = out.to(tl.float16)

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom forward pass for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : dict with keys:
            "norm.weight", "norm.bias",
            "left_proj.weight", "right_proj.weight",
            "left_gate.weight", "right_gate.weight",
            "out_gate.weight",
            "to_out_norm.weight", "to_out_norm.bias",
            "to_out.weight"
          all tensors are float32 and reside on the same device as the input.
        - config_dict : {
              "dim"        : int,
              "hidden_dim": int,
              "nomask"    : bool (optional, default True)
          }

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim], dtype float32.
    """
    # --------------------------------------------------------------
    # 0️⃣ Unpack inputs
    # --------------------------------------------------------------
    x, mask, weights, config = data
    device = x.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    B, N, _, _ = x.shape               # x: [B, N, N, dim]
    total_rows = B * N * N             # R = B·N·N
    H = hidden_dim                     # hidden dimension

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (fp32 → fp16)
    # --------------------------------------------------------------
    x_flat = x.view(total_rows, dim).contiguous()               # [R, dim] fp32
    ln1_out = torch.empty(total_rows, dim,
                          dtype=torch.float16,
                          device=device)                     # [R, dim] fp16

    # Block sizes tuned for dim ≤ 384
    BLOCK_M = 256
    BLOCK_D = 64
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        ln1_out,
        weights["norm.weight"],   # γ  (float32)
        weights["norm.bias"],     # β  (float32)
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 2️⃣ Fuse left/right projections & their gates (4·H outputs)
    # --------------------------------------------------------------
    # Stack the four weight matrices [H, dim] → [4·H, dim]
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
        ],
        dim=0,
    ).to(device=device, dtype=torch.float16)                  # [4·H, dim]

    # Linear: [R, dim] @ [4·H, dim].T → [R, 4·H]  (fp16, no bias)
    fused = F.linear(ln1_out, w_fused)                         # [R, 4·H] fp16

    # Slice the fused tensor
    left_raw      = fused[:, 0:H]               # [R, H]
    right_raw     = fused[:, H:2 * H]           # [R, H]
    left_gate_raw = fused[:, 2 * H:3 * H]       # [R, H]
    right_gate_raw = fused[:, 3 * H:4 * H]      # [R, H]

    # Output‑gate (separate linear)
    out_gate_raw = F.linear(
        ln1_out,
        weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    )   # [R, H] fp16

    # --------------------------------------------------------------
    # 3️⃣ Mask (if present), sigmoid gates, element‑wise weighting
    # --------------------------------------------------------------
    if not nomask and mask is not None:
        # mask: [B, N, N] → [R, 1] (float16)
        mask_flat = mask.to(device=device, dtype=torch.float16).reshape(total_rows, 1)
        left_raw = left_raw * mask_flat
        right_raw = right_raw * mask_flat

    # In‑place sigmoid to avoid extra temporaries
    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    left  = left_raw * left_gate          # [R, H] fp16
    right = right_raw * right_gate        # [R, H] fp16

    # --------------------------------------------------------------
    # 4️⃣ N³ multiplicative update via batched GEMM
    # --------------------------------------------------------------
    # Reshape to [B, N, N, H]
    left  = left.view(B, N, N, H)
    right = right.view(B, N, N, H)
    out_gate = out_gate.view(B, N, N, H)

    # Move head dim into batch: [B, H, N, N]
    left_perm  = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    # Collapse batch & heads → [B·H, N, N]
    left_mat  = left_perm.view(B * H, N, N)
    right_mat = right_perm.view(B * H, N, N)

    # Batched matmul: out[b*h, i, j] = Σ_k left[i,k] * right[j,k]
    # Equivalent to einsum('... i k d, ... j k d -> ... i j d')
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B·H, N, N) fp16

    # Restore layout → [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm over hidden dimension H (fp16)
    # --------------------------------------------------------------
    out_flat = out.view(total_rows, H).contiguous()   # [R, H] fp16
    ln2_out = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        ln2_out,
        weights["to_out_norm.weight"],   # γ  (float32)
        weights["to_out_norm.bias"],     # β  (float32)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 6️⃣ Apply output gate, final projection, cast to float32
    # --------------------------------------------------------------
    out_norm = ln2_out.view(B, N, N, H)       # fp16
    out_norm = out_norm * out_gate            # element‑wise fp16

    final_fp16 = F.linear(
        out_norm.view(total_rows, H),
        weights["to_out.weight"].to(device=device, dtype=torch.float16)   # [dim, H]
    )   # [R, dim] fp16

    # Cast back to float32 as required by the API
    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final
