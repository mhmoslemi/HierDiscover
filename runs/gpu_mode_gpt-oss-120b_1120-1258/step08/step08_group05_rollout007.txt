"""
Optimized forward implementation of the *outgoing* TriMul operator
(AlphaFold‑3).  The algorithm follows the reference PyTorch code but
moves the two LayerNorms to Triton and keeps all heavy linear work in
FP16 to make full use of Tensor‑Core performance on an H100.

Steps
-----
1️⃣  Row‑wise LayerNorm on the input tensor (FP32 → FP16) – Triton.
2️⃣  Fuse the four projection/gate Linear layers (no bias) into a single
    half‑precision GEMM and split the result.
3️⃣  (Optional) apply the pair‑wise mask, compute σ‑gates and multiply.
4️⃣  Compute the N³ update:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    as a batched half‑precision GEMM (B·H, N, N) @ (B·H, N, N)ᵀ.
5️⃣  Second LayerNorm over the hidden dimension – Triton.
6️⃣  Apply the output gate and a final half‑precision Linear projection
    back to the original channel dimension.  The result is returned in
    float32.

Only the two LayerNorms are Triton kernels; the remaining operations
use highly‑optimized PyTorch / cuBLAS kernels, giving a good
performance/complexity trade‑off while satisfying the requirement of
using at least one custom kernel.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (fp32 → fp16)
# ----------------------------------------------------------------------
@triton.jit
def _layernorm_kernel(
    x_ptr,         # const   * [rows, dim] (float32)
    y_ptr,         # mutable * [rows, dim] (float16)
    weight_ptr,    # const   * [dim]      (float32) – γ
    bias_ptr,      # const   * [dim]      (float32) – β
    rows,          # int32   (runtime)
    dim,           # int32   (runtime)
    eps,           # float32 (compile‑time)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per program
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ------------------------------------------------------------------
    # 1️⃣  Reduce over the feature dimension: mean & variance
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    # loop over the dim axis in blocks of BLOCK_D
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # fp32 load
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                           # 1 / sqrt(var+eps)

    # ------------------------------------------------------------------
    # 2️⃣  Write normalized output in fp16 (γ * (x-μ)/σ + β)
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]    # (x‑μ)/σ

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # β
        out = norm * w[None, :] + b[None, :]             # fp32

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


def _layernorm_triton(x: torch.Tensor,
                      weight: torch.Tensor,
                      bias: torch.Tensor,
                      eps: float = 1e-5,
                      block_m: int = 256,
                      block_d: int = 64) -> torch.Tensor:
    """
    Row‑wise LayerNorm on a 2‑D FP32 tensor ``x`` of shape [rows, dim].
    Returns an FP16 tensor of the same shape.
    """
    rows, dim = x.shape
    y = torch.empty(rows, dim, dtype=torch.float16, device=x.device)

    grid = (triton.cdiv(rows, block_m),)
    _layernorm_kernel[grid](
        x,
        y,
        weight,
        bias,
        rows,
        dim,
        eps,
        BLOCK_M=block_m,
        BLOCK_D=block_d,
        num_warps=4,
    )
    return y


# ----------------------------------------------------------------------
# Triton‑accelerated forward of the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) – ignored if
                         config["nomask"] == True
        - weights_dict : mapping of all Linear / LayerNorm parameters (float32)
        - config_dict  : must contain "dim", "hidden_dim" and optionally "nomask"

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # 0️⃣ Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N                     # R = B·N·N

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → FP16
    # --------------------------------------------------------------
    # flatten to [R, dim] for the kernel
    x_flat = input_tensor.view(total_rows, dim).contiguous()
    x_norm = _layernorm_triton(
        x_flat,
        weights["norm.weight"],    # γ, FP32
        weights["norm.bias"],      # β, FP32
        eps=eps,
    )                               # FP16, shape [R, dim]

    # --------------------------------------------------------------
    # 2️⃣ Fused projection & gate Linear layers (no bias)
    # --------------------------------------------------------------
    H = hidden_dim
    # Concatenate the four projection / gate weight matrices (each [H, dim])
    proj_gate_w = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)                     # [4·H, dim]

    # (R, dim) @ (dim, 4·H).T → (R, 4·H)  (FP16 GEMM)
    proj_gate = F.linear(x_norm, proj_gate_w)   # FP16

    # split the fused result
    left_proj_raw  = proj_gate[:, 0:H]                 # [R, H]
    right_proj_raw = proj_gate[:, H:2 * H]             # [R, H]
    left_gate_raw  = proj_gate[:, 2 * H:3 * H]         # [R, H]
    right_gate_raw = proj_gate[:, 3 * H:4 * H]         # [R, H]

    # Separate linear for the output‑gate (no bias)
    out_gate_raw = F.linear(
        x_norm,
        weights["out_gate.weight"].to(torch.float16),   # [H, dim]
    )                                                   # [R, H]

    # --------------------------------------------------------------
    # 3️⃣ Mask (if any) + sigmoid gates + element‑wise product
    # --------------------------------------------------------------
    if not nomask:
        # mask shape [B, N, N] → [R, 1] and cast to FP16
        mask_h = mask_tensor.to(torch.float16).reshape(total_rows, 1)
        left_proj_raw = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    # In‑place sigmoid for speed
    left_gate = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate = out_gate_raw.sigmoid_()

    # Apply gates
    left = left_proj_raw * left_gate      # [R, H]  FP16
    right = right_proj_raw * right_gate   # [R, H]  FP16

    # --------------------------------------------------------------
    # 4️⃣ N³ multiplicative update (batched half‑precision GEMM)
    # --------------------------------------------------------------
    # reshape to [B, N, N, H] for the later transpose
    left = left.view(B, N, N, H)
    right = right.view(B, N, N, H)

    # bring the hidden dim to the batch dimension: (B, H, N, N)
    left_bh = left.permute(0, 3, 1, 2).contiguous()
    right_bh = right.permute(0, 3, 1, 2).contiguous()

    # batched matmul: (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N)
    out_bh = torch.matmul(
        left_bh,
        right_bh.transpose(-2, -1)
    )  # FP16, shape [B, H, N, N]

    # back to [B, N, N, H]
    out = out_bh.permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm over the hidden dimension → FP16
    # --------------------------------------------------------------
    out_flat = out.view(total_rows, H).contiguous()
    out_norm = _layernorm_triton(
        out_flat,
        weights["to_out_norm.weight"],   # γ, FP32
        weights["to_out_norm.bias"],     # β, FP32
        eps=eps,
    )                                   # FP16, [R, H]
    out_norm = out_norm.view(B, N, N, H)   # [B, N, N, H]

    # --------------------------------------------------------------
    # 6️⃣ Apply output‑gate and final projection back to `dim`
    # --------------------------------------------------------------
    out_norm = out_norm * out_gate.view(B, N, N, H)   # FP16 element‑wise

    final_half = F.linear(
        out_norm.view(total_rows, H),
        weights["to_out.weight"].to(torch.float16),   # [dim, H]
    )                                                   # FP16

    # Return float32 tensor as required by the interface
    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
