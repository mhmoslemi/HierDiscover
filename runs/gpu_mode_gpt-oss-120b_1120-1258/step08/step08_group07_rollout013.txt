@triton.jit
def batch_gemm_outgoing_kernel(
    A_ptr, B_ptr, C_ptr,
    batch, M, N, K,
    stride_A_batch, stride_A_i, stride_A_k,
    stride_B_batch, stride_B_k, stride_B_i,
    stride_C_batch, stride_C_i, stride_C_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0) # for batch and rows
    pid_n = tl.program_id(1) # for cols

    # Compute number of row tiles
    num_m_tiles = tl.cdiv(M, BLOCK_M)

    # Decode batch/h index
    batch_idx = pid // num_m_tiles
    tile_m = pid % num_m_tiles

    # Compute offsets
    offs_i = tile_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < M
    mask_j = offs_j < N

    # Compute base pointers for A and B and C for this batch_idx (which encodes (b, h) index)
    # A and B share same strides; we need to compute offset for hidden dimension included in batch_idx.
    # Assuming batch includes hidden dimension i.e., total_batch = B * H.
    a_base = A_ptr + batch_idx * stride_A_batch
    b_base = B_ptr + batch_idx * stride_B_batch
    c_base = C_ptr + batch_idx * stride_C_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        a_ptrs = a_base + offs_i[:, None] * stride_A_i + (k_offset + tl.arange(0, BLOCK_K))[None, :] * stride_A_k
        b_ptrs = b_base + (k_offset + tl.arange(0, BLOCK_K))[:, None] * stride_B_k + offs_j[None, :] * stride_B_i

        a = tl.load(a_ptrs, mask=mask_i[:, None] & (k_offset + tl.arange(0, BLOCK_K) < K), other=0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask=(k_offset + tl.arange(0, BLOCK_K) < K)[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        acc += tl.dot(a, b)

    # Store result
    c_ptrs = c_base + offs_i[:, None] * stride_C_i + offs_j[None, :] * stride_C_j
    out = acc.to(tl.float16)
    tl.store(c_ptrs, out, mask=mask_i[:, None] & mask_j[None, :])

"""
TriMul “outgoing” forward pass – fused layer‑norms + a custom batched GEMM
=======================================================================

Algorithm
---------
1️⃣ Row‑wise LayerNorm over the input tensor (B·N·N × dim).  Implemented in
    Triton (fp32 reductions, fp16 output).

2️⃣ Fuse the four linear projections (left/right + their gates) into a single
    GEMM:  [R, dim] @ [4·H, dim]ᵀ → [R, 4·H] (R = B·N·N, H = hidden_dim).
    The `out_gate` projection is computed separately.

3️⃣ Apply the optional mask, element‑wise sigmoid, and gating:
    left = (x·W_left)   * sigmoid(x·W_left_gate)
    right = (x·W_right) * sigmoid(x·W_right_gate)
    out_gate = sigmoid(x·W_out_gate)

4️⃣ **Custom Triton batched GEMM** – compute the N³ multiplicative update
    for every hidden slice without any expensive permute/contiguous copies:

        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

    The kernel works on the original layout [B, N, N, H] and uses
    per‑batch‑hidden strides, so no extra memory copies are needed.

5️⃣ Second row‑wise LayerNorm over the hidden dimension (R × H).

6️⃣ Element‑wise multiplication with `out_gate` and a final linear projection
    back to `dim`.

All heavy compute (both layer‑norms and the N³ update) run on Triton;
the cheap linear layers use PyTorch’s highly‑optimized `torch.nn.functional.linear`.
The final output is returned as float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm kernel (FP32 reduction → FP16 output)
# ----------------------------------------------------------------------
@triton.jit
def rowwise_layernorm_kernel(
    x_ptr,               # const   * [total_rows, dim] (float32)
    y_ptr,               # mutable * [total_rows, dim] (float16)
    w_ptr,               # const   * [dim]          (float32) gamma
    b_ptr,               # const   * [dim]          (float32) beta
    total_rows,          # int
    dim,                 # int
    eps,                 # float (compile‑time constant)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # cols per tile
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ---------- 1️⃣ mean & variance (two‑pass) ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)   # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- 2️⃣ write normalized result ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)  # fp32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)  # fp32

        out = norm * gamma[None, :] + beta[None, :]   # fp32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
# 4️⃣ Batched GEMM for the N³ update (no permutes, no copies)
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_outgoing_kernel(
    A_ptr, B_ptr, C_ptr,                # fp16 tensors, shape [B, N, N, H]
    B_batch, H_hidden, N_seq,           # int: real batch dim, hidden dim, seq length (K)
    stride_A_batch, stride_A_i, stride_A_k, stride_A_h,
    stride_B_batch, stride_B_i, stride_B_k, stride_B_h,
    stride_C_batch, stride_C_i, stride_C_j, stride_C_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute for every (b, h):
        C[b,i,j,h] = Σ_k A[b,i,k,h] * B[b,j,k,h]
    A, B, C are stored as [B, N, N, H] (contiguous).
    """
    pid = tl.program_id(0)          # encodes (batch·hidden) × row‑tiles
    pid_n = tl.program_id(1)        # column tile

    # number of row tiles for one (b,h) matrix
    num_tile_m = tl.cdiv(N_seq, BLOCK_M)

    # decode the (b, h) pair and the row‑tile index
    batch_hidden_idx = pid // num_tile_m
    tile_m = pid % num_tile_m

    b_idx = batch_hidden_idx // H_hidden
    h_idx = batch_hidden_idx % H_hidden

    # offsets within the N×N matrix
    offs_i = tile_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N_seq
    mask_j = offs_j < N_seq

    # base pointers for the selected (b,h) slice
    a_base = A_ptr + b_idx * stride_A_batch + h_idx * stride_A_h
    b_base = B_ptr + b_idx * stride_B_batch + h_idx * stride_B_h
    c_base = C_ptr + b_idx * stride_C_batch + h_idx * stride_C_h

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N_seq, BLOCK_K)):
        k_off = k * BLOCK_K
        k_valid = (k_off + tl.arange(0, BLOCK_K)) < N_seq

        # A[i, k]
        a_ptrs = a_base + offs_i[:, None] * stride_A_i + (k_off + tl.arange(0, BLOCK_K))[None, :] * stride_A_k
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & k_valid[None, :],
                    other=0.0).to(tl.float32)

        # B[j, k]  (note the transpose – we need B[j,k] not B[k,j])
        b_ptrs = b_base + (k_off + tl.arange(0, BLOCK_K))[:, None] * stride_B_k + offs_j[None, :] * stride_B_i
        b = tl.load(b_ptrs,
                    mask=k_valid[:, None] & mask_j[None, :],
                    other=0.0).to(tl.float32)

        acc += tl.dot(a, b)   # (BLOCK_M × BLOCK_K) × (BLOCK_K × BLOCK_N)

    # write result
    c_ptrs = c_base + offs_i[:, None] * stride_C_i + offs_j[None, :] * stride_C_j
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])

# ----------------------------------------------------------------------
# Custom forward kernel
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args
    ----
    data : tuple (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor    shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor    shape [B, N, N] (bool/float) – optional
        - weights_dict : dict of torch.Tensors containing all Linear/LayerNorm params
        - config_dict  : must contain "dim", "hidden_dim" and optional "nomask"

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # 1️⃣ unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    device = input_tensor.device
    B, N, _, _ = input_tensor.shape
    R = B * N * N                       # rows for the first LayerNorm

    # --------------------------------------------------------------
    # 2️⃣ First LayerNorm (row‑wise) → FP16
    # --------------------------------------------------------------
    x_flat = input_tensor.view(R, dim).contiguous()      # [R, dim] FP32
    x_norm_fp16 = torch.empty(R, dim,
                              dtype=torch.float16,
                              device=device)

    BLOCK_M = 256
    BLOCK_D = 32
    grid_ln1 = (triton.cdiv(R, BLOCK_M),)
    rowwise_layernorm_kernel[grid_ln1](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],   # gamma (fp32)
        weights["norm.bias"],     # beta  (fp32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    # --------------------------------------------------------------
    # 3️⃣ Fuse left/right projections + gates (single GEMM)
    # --------------------------------------------------------------
    H = hidden_dim
    # concatenate weights for the four linear layers: [4·H, dim]
    w_proj_gate = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)   # fp16 weights

    proj_gate = F.linear(x_norm_fp16, w_proj_gate)   # [R, 4·H] (fp16)

    left_proj_raw  = proj_gate[:, 0:H]               # [R, H]
    right_proj_raw = proj_gate[:, H:2*H]             # [R, H]
    left_gate_raw   = proj_gate[:, 2*H:3*H]          # [R, H]
    right_gate_raw  = proj_gate[:, 3*H:4*H]          # [R, H]

    # separate out‑gate projection
    out_gate_raw = F.linear(
        x_norm_fp16,
        weights["out_gate.weight"].to(torch.float16)   # [H, dim]
    )   # → [R, H]

    # --------------------------------------------------------------
    # 4️⃣ Mask, sigmoid, gating
    # --------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask_tensor.to(torch.float16).unsqueeze(-1)   # [B, N, N, 1]
        left_proj_raw  = left_proj_raw.view(B, N, N, H) * mask_fp16
        right_proj_raw = right_proj_raw.view(B, N, N, H) * mask_fp16
    else:
        left_proj_raw  = left_proj_raw.view(B, N, N, H)
        right_proj_raw = right_proj_raw.view(B, N, N, H)

    # out_gate always needs reshaping
    out_gate = out_gate_raw.view(B, N, N, H)

    # sigmoid (in‑place, fp16)
    left_gate = left_gate_raw.view(B, N, N, H).sigmoid_()
    right_gate = right_gate_raw.view(B, N, N, H).sigmoid_()
    out_gate = out_gate.sigmoid_()

    # Apply gates
    left = left_proj_raw * left_gate       # [B, N, N, H] fp16
    right = right_proj_raw * right_gate    # [B, N, N, H] fp16

    # --------------------------------------------------------------
    # 5️⃣ Batched GEMM (N³ update) – no extra permutes / copies
    # --------------------------------------------------------------
    out = torch.empty_like(left)  # [B, N, N, H] fp16

    # strides of the tensors (all are identical in layout)
    stride_A_batch = left.stride(0)   # B‑stride = N·N·H
    stride_A_i     = left.stride(1)   # i‑stride = N·H
    stride_A_k     = left.stride(2)   # k‑stride = H
    stride_A_h     = left.stride(3)   # hidden‑stride = 1

    stride_B_batch = right.stride(0)
    stride_B_i     = right.stride(1)
    stride_B_k     = right.stride(2)
    stride_B_h     = right.stride(3)

    stride_C_batch = out.stride(0)
    stride_C_i     = out.stride(1)
    stride_C_j     = out.stride(2)
    stride_C_h     = out.stride(3)

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    num_tile_m = triton.cdiv(N, BLOCK_M)
    num_tile_n = triton.cdiv(N, BLOCK_N)
    grid_gemm = (B * H * num_tile_m, num_tile_n)

    batched_gemm_outgoing_kernel[grid_gemm](
        left,                                 # A
        right,                                # B
        out,                                  # C
        B, H, N,
        stride_A_batch, stride_A_i, stride_A_k, stride_A_h,
        stride_B_batch, stride_B_i, stride_B_k, stride_B_h,
        stride_C_batch, stride_C_i, stride_C_j, stride_C_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm (over hidden dimension) → FP16
    # --------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()
    out_norm_fp16 = torch.empty_like(out_flat, dtype=torch.float16)

    grid_ln2 = (triton.cdiv(R, BLOCK_M),)
    rowwise_layernorm_kernel[grid_ln2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],   # gamma (fp32)
        weights["to_out_norm.bias"],     # beta  (fp32)
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    out_norm = out_norm_fp16.view(B, N, N, H)   # fp16

    # --------------------------------------------------------------
    # 7️⃣ Apply out‑gate and final projection
    # --------------------------------------------------------------
    gated = out_norm * out_gate                 # [B, N, N, H] fp16

    final_half = F.linear(
        gated.view(R, H),
        weights["to_out.weight"].to(torch.float16)   # [dim, H]
    )   # → [R, dim] fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)

    return final
