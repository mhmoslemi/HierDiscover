def _layernorm_kernel(
    x_ptr, y_ptr,
    weight_ptr, bias_ptr,
    rows, dim,
    eps,

    BLOCK_D: tl.constexpr

    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    sum_ = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    # iterate over dim in blocks
    for d_start in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_start * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        vals = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)

        sum_sq += tl.sum(vals * vals, axis=1)

"""
Optimized “outgoing” TriMul (AlphaFold‑3) forward pass.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the input tensor (fp32 → fp16) – custom Triton kernel.
2️⃣ Fuse the four linear projections (left, right, left‑gate, right‑gate) *and* the
   output‑gate linear layer into a single half‑precision GEMM.  Split the result
   into the five needed tensors.
3️⃣ (Optional) apply the pair‑wise mask, compute sigmoid gates and multiply.
4️⃣ Compute the expensive N³ update
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   as a batched half‑precision matmul:
        (B·H, N, N) @ (B·H, N, N)ᵀ
5️⃣ Second LayerNorm over the hidden dimension (torch.nn.functional.layer_norm)
   and fuse the out‑gate multiplication.
6️⃣ Final half‑precision projection back to the original channel dimension
   (fp16 GEMM).  Cast the result to float32 for the model output.

Only the first LayerNorm is implemented in Triton; the rest uses
high‑performance PyTorch kernels (half‑precision GEMM and fused
LayerNorm).  This reduces kernel launches and eliminates unnecessary
memory copies while keeping the overall runtime well below the target.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (fp32 → fp16)
# ----------------------------------------------------------------------
@triton.jit
def _layernorm_rows_kernel(
    x_ptr,          # const   * [rows, dim] (float32)
    y_ptr,          # mutable * [rows, dim] (float16)
    weight_ptr,     # const   * [dim]      (float32) – γ
    bias_ptr,       # const   * [dim]      (float32) – β
    rows,           # int32   (runtime)
    dim,            # int32   (runtime)
    eps,            # float32 (runtime)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per program
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)               # [BLOCK_M]
    mask_m = offs_m < rows

    # --------------------------------------------------------------
    # 1️⃣ Compute per‑row mean & variance
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]  # [m, d]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)               # fp32 load
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                               # 1 / sqrt(var+eps)

    # --------------------------------------------------------------
    # 2️⃣ Write normalized output (fp16)
    # --------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        # (x - μ) / σ
        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β

        out = norm * w[None, :] + b[None, :]                     # fp32
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


def _layernorm_triton(x: torch.Tensor,
                      weight: torch.Tensor,
                      bias: torch.Tensor,
                      eps: float = 1e-5,
                      block_m: int = 256,
                      block_d: int = 32) -> torch.Tensor:
    """
    Row‑wise LayerNorm with fp16 output.
    x : [rows, dim]  (float32)
    returns fp16 tensor of the same shape.
    """
    rows, dim = x.shape
    y = torch.empty(rows, dim, dtype=torch.float16, device=x.device)

    grid = (triton.cdiv(rows, block_m),)
    _layernorm_rows_kernel[grid](
        x,
        y,
        weight,
        bias,
        rows,
        dim,
        eps,
        BLOCK_M=block_m,
        BLOCK_D=block_d,
        num_warps=4,
    )
    return y

        - config_dict  : must contain "dim", "hidden_dim" and optional "nomask"

    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # 0️⃣ unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N                     # R = B·N·N

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → fp16
    # ------------------------------------------------------------------
    x_flat = input_tensor.view(total_rows, dim).contiguous()   # [R, dim] fp32
    x_norm = _layernorm_triton(
        x_flat,
        weights["norm.weight"],   # γ (float32)
        weights["norm.bias"],     # β (float32)
        eps=eps,
    )                                                       # fp16, [R, dim]

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection & gate linear layers (including out_gate)
    # ------------------------------------------------------------------
    H = hidden_dim
    # concat: left_proj, right_proj, left_gate, right_gate, out_gate
    fused_weight = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
            weights["out_gate.weight"],    # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)                                     # [5·H, dim]

    # (R, dim) @ (dim, 5·H)ᵀ  → (R, 5·H)
    proj_all = F.linear(x_norm, fused_weight)               # fp16

    # split
    left_proj_raw  = proj_all[:, 0:H]                       # [R, H]
    right_proj_raw = proj_all[:, H:2 * H]                   # [R, H]
    left_gate_raw  = proj_all[:, 2 * H:3 * H]               # [R, H]
    right_gate_raw = proj_all[:, 3 * H:4 * H]               # [R, H]
    out_gate_raw   = proj_all[:, 4 * H:5 * H]               # [R, H]

    # ------------------------------------------------------------------
    # 3️⃣ Mask (if any) + sigmoid gating + element‑wise product
    # ------------------------------------------------------------------
    if not nomask:
        mask_flat = mask_tensor.to(torch.float16).reshape(total_rows, 1)  # [R,1]
        left_proj_raw = left_proj_raw * mask_flat
        right_proj_raw = right_proj_raw * mask_flat

    left_gate = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate = out_gate_raw.sigmoid_()

    left = left_proj_raw * left_gate        # [R, H] fp16
    right = right_proj_raw * right_gate     # [R, H] fp16

    # ------------------------------------------------------------------
    # 4️⃣ N³ multiplicative update (batched half‑precision GEMM)
    # ------------------------------------------------------------------
    left = left.view(B, N, N, H)            # [B, N, N, H]
    right = right.view(B, N, N, H)          # [B, N, N, H]

    # reshape to (B·H, N, N) for cuBLAS batch matmul
    left_bh = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_bh = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    out_bh = torch.matmul(
        left_bh,
        right_bh.transpose(-2, -1)
    )                                                   # fp16, [B, H, N, N]

    # back to (B, N, N, H)
    out = out_bh.permute(0, 2, 3, 1).contiguous()       # [B, N, N, H]

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm over hidden dimension + out‑gate fusion
    # ------------------------------------------------------------------
    # Torch's functional LayerNorm is already highly tuned.
    out_norm = F.layer_norm(
        out,
        (H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )                                                    # fp16, [B, N, N, H]

    # fuse output‑gate
    out_norm = out_norm * out_gate.view(B, N, N, H)      # fp16

    # ------------------------------------------------------------------
    # 6️⃣ Final projection back to `dim`
    # ------------------------------------------------------------------
    final_half = F.linear(
        out_norm.view(total_rows, H),                     # [R, H]
        weights["to_out.weight"].to(torch.float16),      # [dim, H]
    )                                                    # fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
