@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    B, M, N, K,
    stride_ab, stride_ak, stride_abatch,
    stride_bk, stride_bn, stride_bbatch,
    stride_cm, stride_cn, stride_cbatch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program id along batch dimension
    batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute start indexes
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for valid M,N
    mask_m = offs_m < M
    mask_n = offs_n < N

    a_ptrs = a_ptr + batch * stride_abatch + offs_m[:, None] * stride_ak + tl.arange(0, BLOCK_K)[None, :] * stride_ab
    b_ptrs = b_ptr + batch * stride_bbatch + offs_n[None, :] * stride_bn + tl.arange(0, BLOCK_K)[:, None] * stride_bk

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        a = tl.load(a_ptrs + k_offset * stride_ab, mask=mask_m[:, None] & (k_offset + tl.arange(0, BLOCK_K) < K), other=0.0)
        b = tl.load(b_ptrs + k_offset * stride_bk, mask=mask_n[None, :] & (k_offset + tl.arange(0, BLOCK_K) < K), other=0.0)

        acc += tl.dot(a, b)

    # write

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# Row-wise LayerNorm kernel: normalizes each row of a 2D tensor.
@triton.jit
def rowwise_layernorm_kernel(
    x_ptr, y_ptr, w_ptr, b_ptr,
    total_rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # accumulate sum and sumsq
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)
    for d0 in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d0 * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # write normalized values
    for d0 in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d0 * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)

        out = norm * gamma[None, :] + beta[None, :]
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

# Batched GEMM kernel for TriMul’s “outer‑product” step.
# Computes C[b,i,j] = Σ_k A[b,i,k] * B[b,j,k]  (B is used transposed)
@triton.jit
def tri_mul_batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    batch_dim, M, N, K,
    stride_a_batch, stride_a_i, stride_a_k,
    stride_b_batch, stride_b_i, stride_b_k,
    stride_c_batch, stride_c_i, stride_c_j,
    H: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    eps: tl.constexpr
):
    # program ids
    pid_batch = tl.program_id(0)   # 0 … B*H-1  (combined batch)
    pid_m = tl.program_id(1)       # block row
    pid_n = tl.program_id(2)       # block column

    # decompose combined batch id into (batch, hidden)
    b = pid_batch // H
    h = pid_batch - b * H   # equivalent to pid_batch % H

    # base pointers for this (batch, hidden) slice
    a_base = a_ptr + b * stride_a_batch + h * tl.constexpr(0)  # hidden stride is 1, handled later
    b_base = b_ptr + b * stride_b_batch + h * tl.constexpr(0)

    # coordinates within the M×N output tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for k_iter in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k_iter * BLOCK_K
        # mask for the K‑tile
        k_mask = k_offset + tl.arange(0, BLOCK_K) < K

        # A: [BLOCK_M, BLOCK_K]  -> rows i, cols k
        a_ptrs = a_base \
                 + offs_m[:, None] * stride_a_i \
                 + (k_offset + tl.arange(0, BLOCK_K))[None, :] * stride_a_k
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & k_mask[None, :],
                    other=0.0).to(tl.float16)

        # B (transposed): we need B[j,k] = original B[k,j]
        b_ptrs = b_base \
                 + (k_offset + tl.arange(0, BLOCK_K))[:, None] * stride_b_i \
                 + offs_n[None, :] * stride_b_k
        b = tl.load(b_ptrs,
                    mask=k_mask[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float16)

        acc += tl.dot(a, b)   # fp32 accumulator

    # write result back as fp16
    c_ptrs = c_ptr + b * stride_c_batch + h * tl.constexpr(0) \
             + offs_m[:, None] * stride_c_i \
             + offs_n[None, :] * stride_c_j
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operation.
    ---------------------------------------------------------------
    1️⃣ Row‑wise LayerNorm on the input tensor (B·N·N × dim) → fp16.
    2️⃣ Fuse the four linear projections (left/right + gates) and the
       output‑gate linear into a single half‑precision GEMM.
    3️⃣ Apply the optional mask, sigmoid gates and element‑wise gating.
    4️⃣ Compute the N³ multiplicative update with a custom batched GEMM
       kernel that works directly on the [B,N,N,hidden] layout (no
       materialising of a contiguous [B·hidden,N,N] tensor).
    5️⃣ Row‑wise LayerNorm over the hidden dimension.
    6️⃣ Multiply by the output‑gate and a final linear projection back
       to ``dim``.  The final tensor is returned in float32.
    """
    # --------------------------------------------------------------
    # unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    device = input_tensor.device
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape
    R = B * N * N

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → fp16
    # --------------------------------------------------------------
    x_flat = input_tensor.view(R, dim).contiguous()   # [R, dim] fp32
    x_norm_fp16 = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 256
    BLOCK_D = 32
    grid = (triton.cdiv(R, BLOCK_M),)
    rowwise_layernorm_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],          # gamma, fp32
        weights["norm.bias"],            # beta,  fp32
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    # --------------------------------------------------------------
    # 2️⃣ Fuse projections (left, right, gates, out_gate) into one GEMM
    # --------------------------------------------------------------
    # concatenate weights: order = left_proj, right_proj, left_gate, right_gate, out_gate
    w_fused = torch.cat([
        weights["left_proj.weight"],    # [H, dim]
        weights["right_proj.weight"],   # [H, dim]
        weights["left_gate.weight"],    # [H, dim]
        weights["right_gate.weight"],   # [H, dim]
        weights["out_gate.weight"],     # [H, dim]
    ], dim=0).to(torch.float16)

    proj_all = F.linear(x_norm_fp16, w_fused)   # [R, 5*H] fp16

    H = hidden_dim
    left_proj_raw  = proj_all[:, 0:H]
    right_proj_raw = proj_all[:, H:2*H]
    left_gate_raw  = proj_all[:, 2*H:3*H]
    right_gate_raw = proj_all[:, 3*H:4*H]
    out_gate_raw   = proj_all[:, 4*H:5*H]

    # --------------------------------------------------------------
    # 3️⃣ Mask & gating
    # --------------------------------------------------------------
    if not nomask:
        mask_h = mask_tensor.to(torch.float16).view(R, 1)   # [R,1]
        left_proj_raw = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    left  = left_proj_raw * left_gate   # [R, H] fp16
    right = right_proj_raw * right_gate

    # --------------------------------------------------------------
    # 4️⃣ Reshape to [B, N, N, H] (still fp16)
    # --------------------------------------------------------------
    left_tensor  = left.view(B, N, N, H)
    right_tensor = right.view(B, N, N, H)
    out_gate_tensor = out_gate.view(B, N, N, H)

    # --------------------------------------------------------------
    # 5️⃣ N³ multiplicative update via custom batched GEMM
    # --------------------------------------------------------------
    out_tensor = torch.empty_like(left_tensor)   # [B, N, N, H] fp16

    # compute strides (in element units)
    stride_a_batch = left_tensor.stride(0)   # N*N*H
    stride_a_i     = left_tensor.stride(1)   # N*H
    stride_a_k     = left_tensor.stride(2)   # H

    stride_b_batch = right_tensor.stride(0)
    stride_b_i     = right_tensor.stride(1)
    stride_b_k     = right_tensor.stride(2)

    stride_c_batch = out_tensor.stride(0)
    stride_c_i     = out_tensor.stride(1)
    stride_c_j     = out_tensor.stride(2)

    # launch parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (B * H,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))

    tri_mul_batched_gemm_kernel[grid](
        left_tensor,
        right_tensor,
        out_tensor,
        B * H,                # batch_dim (flattened)
        N, N, N,              # M,N,K
        stride_a_batch, stride_a_i, stride_a_k,
        stride_b_batch, stride_b_i, stride_b_k,
        stride_c_batch, stride_c_i, stride_c_j,
        H,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm over hidden dimension
    # --------------------------------------------------------------
    out_flat = out_tensor.view(R, H).contiguous()
    out_norm_fp16 = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(R, BLOCK_M),)
    rowwise_layernorm_kernel[grid2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    out_norm = out_norm_fp16.view(B, N, N, H)

    # --------------------------------------------------------------
    # 7️⃣ Apply output gate and final linear projection
    # --------------------------------------------------------------
    out_norm = out_norm * out_gate_tensor   # element‑wise gating, fp16

    final_half = F.linear(
        out_norm.view(R, H),
        weights["to_out.weight"].to(torch.float16)   # [dim, H]
    )   # [R, dim] fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)

    return final

"""
TriMul “outgoing” forward pass – Triton‐accelerated fusion
---------------------------------------------------------

Algorithm
~~~~~~~~~
1️⃣ Row‑wise LayerNorm of the input 4‑D tensor (B·N·N × dim) → fp16.
2️⃣ Fuse the four projection/gate Linear layers *and* the output‑gate
   Linear into a single half‑precision GEMM:
        [R, dim] @ [5·H, dim]ᵀ → [R, 5·H]   (R = B·N·N)
3️⃣ Optional mask, sigmoid gates and element‑wise gating.
4️⃣ Reshape to [B, N, N, H] (still fp16) and compute the
   N³ multiplicative update with a custom batched GEMM kernel that
   works directly on the original layout – no extra copies.
   The kernel computes
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
5️⃣ Second row‑wise LayerNorm over the hidden dimension H → fp16.
6️⃣ Apply the output‑gate, a final half‑precision Linear
   projection back to *dim* and cast to float32.

Only the two LayerNorm passes and the N³ update are written in Triton;
all dense GEMMs (projection, final projection) use cuBLAS via
``torch.nn.functional.linear`` which is already optimal on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (fp32 → fp16)
# ----------------------------------------------------------------------
@triton.jit
def rowwise_layernorm_kernel(
    x_ptr, y_ptr, w_ptr, b_ptr,
    total_rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr
):
    """Row‑wise LayerNorm:
       y = ((x - μ) / sqrt(σ²+eps)) * γ + β
       reduction is in fp32, output in fp16.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # ----- mean / variance (float32) -----
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d0 in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d0 * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ----- write normalized output -----
    for d0 in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d0 * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)

        out = norm * gamma[None, :] + beta[None, :]
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 4️⃣ Batched GEMM kernel for the N³ update
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    batch_dim, M, N, K,                # batch_dim = B*H
    stride_a_batch, stride_a_i, stride_a_k, stride_a_h,
    stride_b_batch, stride_b_i, stride_b_k, stride_b_h,
    stride_c_batch, stride_c_i, stride_c_j, stride_c_h,
    H: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """C[b,i,j] = Σₖ A[b,i,k] * B[b,j,k]   (B is accessed transposed)"""
    pid_batch = tl.program_id(0)          # 0 … B*H-1  (combined batch)
    pid_m     = tl.program_id(1)          # row tile
    pid_n     = tl.program_id(2)          # col tile

    # decompose combined batch id → (batch, hidden)
    b = pid_batch // H
    h = pid_batch - b * H                  # same as pid_batch % H

    # base pointers for this (batch, hidden) slice
    a_base = a_ptr + b * stride_a_batch + h * stride_a_h
    b_base = b_ptr + b * stride_b_batch + h * stride_b_h
    c_base = c_ptr + b * stride_c_batch + h * stride_c_h

    # tile coordinates
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in fp32 (Tensor‑Core friendly)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------- loop over K --------
    for k_iter in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k_iter * BLOCK_K
        k_mask   = k_offset + tl.arange(0, BLOCK_K) < K

        # A tile : rows i, cols k
        a_ptrs = a_base \
                 + offs_m[:, None] * stride_a_i \
                 + (k_offset + tl.arange(0, BLOCK_K))[None, :] * stride_a_k
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & k_mask[None, :],
                    other=0.0)               # fp16

        # B tile (transposed) : rows k, cols j
        b_ptrs = b_base \
                 + (k_offset + tl.arange(0, BLOCK_K))[:, None] * stride_b_i \
                 + offs_n[None, :] * stride_b_k
        b = tl.load(b_ptrs,
                    mask=k_mask[:, None] & mask_n[None, :],
                    other=0.0)               # fp16

        acc += tl.dot(a, b)                # fp32 accumulation

    # -------- write back --------
    c_ptrs = c_base \
             + offs_m[:, None] * stride_c_i \
             + offs_n[None, :] * stride_c_j
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.
    Returns a tensor of shape [bs, seq_len, seq_len, dim] (float32).
    """
    # --------------------------------------------------------------
    # unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim          = config["dim"]
    hidden_dim   = config["hidden_dim"]
    eps          = 1e-5
    device       = input_tensor.device
    nomask       = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape
    R = B * N * N                       # total (batch·seq·seq) rows

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → fp16
    # --------------------------------------------------------------
    x_flat = input_tensor.view(R, dim).contiguous()      # [R, dim] float32
    x_norm_fp16 = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 256
    BLOCK_D = 32
    grid = (triton.cdiv(R, BLOCK_M),)
    rowwise_layernorm_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],   # γ, fp32
        weights["norm.bias"],     # β, fp32
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    # --------------------------------------------------------------
    # 2️⃣ Fuse left/right projections, gates and out_gate into one GEMM
    # --------------------------------------------------------------
    # Order: left_proj, right_proj, left_gate, right_gate, out_gate
    w_fused = torch.cat([
        weights["left_proj.weight"],   # [H, dim]
        weights["right_proj.weight"],  # [H, dim]
        weights["left_gate.weight"],   # [H, dim]
        weights["right_gate.weight"],  # [H, dim]
        weights["out_gate.weight"],    # [H, dim]
    ], dim=0).to(torch.float16)

    proj_all = F.linear(x_norm_fp16, w_fused)           # [R, 5*H] fp16

    H = hidden_dim
    left_proj_raw  = proj_all[:, 0:H]
    right_proj_raw = proj_all[:, H:2*H]
    left_gate_raw  = proj_all[:, 2*H:3*H]
    right_gate_raw = proj_all[:, 3*H:4*H]
    out_gate_raw   = proj_all[:, 4*H:5*H]

    # --------------------------------------------------------------
    # 3️⃣ Mask (if present) and sigmoid gating
    # --------------------------------------------------------------
    if not nomask:
        mask_h = mask_tensor.to(torch.float16).view(R, 1)   # [R,1]
        left_proj_raw  = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    left  = left_proj_raw * left_gate          # [R, H] fp16
    right = right_proj_raw * right_gate        # [R, H] fp16

    # --------------------------------------------------------------
    # 4️⃣ Reshape to [B, N, N, H] (still fp16)
    # --------------------------------------------------------------
    left_tensor  = left.view(B, N, N, H)
    right_tensor = right.view(B, N, N, H)
    out_gate_tensor = out_gate.view(B, N, N, H)

    # --------------------------------------------------------------
    # 5️⃣ N³ multiplicative update via custom batched GEMM
    # --------------------------------------------------------------
    out_tensor = torch.empty_like(left_tensor)           # [B, N, N, H] fp16

    # strides (in element counts, not bytes)
    stride_a_batch = left_tensor.stride(0)   # N*N*H
    stride_a_i     = left_tensor.stride(1)   # N*H
    stride_a_k     = left_tensor.stride(2)   # H
    stride_a_h     = left_tensor.stride(3)   # 1

    stride_b_batch = right_tensor.stride(0)
    stride_b_i     = right_tensor.stride(1)
    stride_b_k     = right_tensor.stride(2)
    stride_b_h     = right_tensor.stride(3)

    stride_c_batch = out_tensor.stride(0)
    stride_c_i     = out_tensor.stride(1)
    stride_c_j     = out_tensor.stride(2)
    stride_c_h     = out_tensor.stride(3)

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    grid = (B * H,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))

    tri_mul_batched_gemm_kernel[grid](
        left_tensor,
        right_tensor,
        out_tensor,
        B * H,                     # total combined batch size
        N, N, N,                  # M, N, K
        stride_a_batch, stride_a_i, stride_a_k, stride_a_h,
        stride_b_batch, stride_b_i, stride_b_k, stride_b_h,
        stride_c_batch, stride_c_i, stride_c_j, stride_c_h,
        H,                         # hidden dimension (compile‑time constant)
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm over hidden dimension → fp16
    # --------------------------------------------------------------
    out_flat = out_tensor.view(R, H).contiguous()
    out_norm_fp16 = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(R, BLOCK_M),)
    rowwise_layernorm_kernel[grid2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    out_norm = out_norm_fp16.view(B, N, N, H)

    # --------------------------------------------------------------
    # 7️⃣ Apply output gate and final projection back to *dim*
    # --------------------------------------------------------------
    out_norm = out_norm * out_gate_tensor               # element‑wise, fp16

    final_half = F.linear(
        out_norm.view(R, H),
        weights["to_out.weight"].to(torch.float16)      # [dim, H]
    )                                                    # [R, dim] fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)

    return final
