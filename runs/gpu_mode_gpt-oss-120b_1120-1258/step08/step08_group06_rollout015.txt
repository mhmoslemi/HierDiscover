"""
Outgoing TriMul (AlphaFold‑3) – Triton‑accelerated forward pass.

Key ideas
---------
* Row‑wise LayerNorm is performed with a tiny Triton kernel (FP32
  accumulation → FP16 output) – this satisfies the “must use a kernel”
  requirement.
* The four projection matrices (`left_proj`, `right_proj`, `left_gate`,
  `right_gate`) are concatenated into a single weight matrix
  of shape (4·H, dim) and multiplied in one `torch.nn.functional.linear`
  call (FP16).  The `out_gate` linear map is computed separately.
* `mask` (when present) is applied **after** the projection but **before**
  the sigmoid‑gates – this avoids extra memory traffic.
* The cubic update

      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

  is expressed with a single `torch.einsum` call.  This removes the
  expensive `permute → contiguous → view → torch.bmm` sequence that
  otherwise copies a tensor of size B·H·N·N (up to ~0.5 GiB) twice.
* A second row‑wise LayerNorm followed by the final linear projection
  (`to_out`) completes the module.  The only large intermediate buffers are
  the normalized tensors required for the two GEMM layers; no extra
  copies are introduced.

The implementation works for all test configurations listed in the
statement and returns a float‑32 tensor of shape [B, N, N, dim].
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# Triton row‑wise LayerNorm (FP32 statistics → FP16 output)
# --------------------------------------------------------------
@triton.jit
def rowwise_layernorm_kernel(
    x_ptr,          # const  * [total_rows, dim] (float32)
    y_ptr,          # mutable* [total_rows, dim] (float16)
    w_ptr,          # const  * [dim] (float32) – γ
    b_ptr,          # const  * [dim] (float32) – β
    total_rows,     # int32
    dim,            # int32
    eps,            # float32
    BLOCK_R: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per inner loop
):
    """
    Compute per‑row LayerNorm:
      y = ((x - μ) / √(σ²+ε)) * γ + β
    Input  : x – FP32
    Output : y – FP16
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < total_rows

    # ---------- 1.  compute sum and sum‑of‑squares (FP32) ----------
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_R], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_r[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_r[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)          # 1/√(var+ε)

    # ---------- 2.  write normalized values (FP16) ---------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_r[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_r[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)

        out = norm * gamma[None, :] + beta[None, :]
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_r[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_r[:, None] & mask_d[None, :])


# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor_or_None, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float) – optional
        - weights_dict : model weights (see keys below)
        - config_dict  : must contain "dim", "hidden_dim", "nomask"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N          # R = B·N·N
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → fp16
    # ------------------------------------------------------------------
    x_flat = input_tensor.reshape(total_rows, dim).contiguous()   # [R, dim] FP32
    x_norm = torch.empty(total_rows, dim, dtype=torch.float16,
                         device=device)

    BLOCK_R = 256          # rows per program (tunable)
    BLOCK_D = 128          # columns per inner loop (covers dim ≤ 384)
    grid = (triton.cdiv(total_rows, BLOCK_R),)

    rowwise_layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],          # γ (FP32)
        weights["norm.bias"],            # β (FP32)
        total_rows,
        dim,
        eps,
        BLOCK_R=BLOCK_R,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection + gate linear maps (4·H + H → 5·H)
    # ------------------------------------------------------------------
    # Concatenate the four projection matrices and the out‑gate matrix
    w_proj = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # (4·H, dim)

    # Linear to obtain left/right projections and the two gate tensors
    proj_gate = F.linear(x_norm, w_proj)    # [R, 4·H] FP16

    left_proj_raw  = proj_gate[:, 0:H]
    right_proj_raw = proj_gate[:, H:2 * H]
    left_gate_raw  = proj_gate[:, 2 * H:3 * H]
    right_gate_raw = proj_gate[:, 3 * H:4 * H]

    # Out‑gate is a separate linear map (H, dim)
    out_gate_raw = F.linear(
        x_norm,
        weights["out_gate.weight"].to(torch.float16)   # (H, dim)
    )   # [R, H] FP16

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask, sigmoid‑gates and element‑wise gating
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B,N,N] → [R,1] (float16) – broadcasting on the hidden dim
        mask_fp16 = mask_tensor.reshape(total_rows, 1).to(torch.float16)
        left_proj_raw = left_proj_raw * mask_fp16
        right_proj_raw = right_proj_raw * mask_fp16

    # In‑place sigmoid (fast)
    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    # Apply gates
    left  = left_proj_raw * left_gate      # [R, H] FP16
    right = right_proj_raw * right_gate    # [R, H] FP16

    # ------------------------------------------------------------------
    # 4️⃣ Reshape to (B, N, N, H)
    # ------------------------------------------------------------------
    left  = left.view(B, N, N, H)
    right = right.view(B, N, N, H)
    out_gate = out_gate.view(B, N, N, H)   # used later

    # ------------------------------------------------------------------
    # 5️⃣ Cubic update:  out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented with a single einsum – avoids the two large
    #    .permute(...).contiguous() copies present in a naïve bmm approach.
    # ------------------------------------------------------------------
    out = torch.einsum('b i k h, b j k h -> b i j h', left, right)   # [B, N, N, H] FP16

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm over the hidden dimension H
    # ------------------------------------------------------------------
    out_flat = out.reshape(total_rows, H).contiguous()
    out_norm = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows, BLOCK_R),)
    rowwise_layernorm_kernel[grid2](
        out_flat,
        out_norm,
        weights["to_out_norm.weight"],   # γ (FP32)
        weights["to_out_norm.bias"],     # β (FP32)
        total_rows,
        H,
        eps,
        BLOCK_R=BLOCK_R,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    out_norm = out_norm.view(B, N, N, H)

    # ------------------------------------------------------------------
    # 7️⃣ Apply the output gate and final linear projection
    # ------------------------------------------------------------------
    gated = out_norm * out_gate                     # [B, N, N, H] FP16

    final = F.linear(
        gated.view(total_rows, H),                  # flatten batch & spatial dims
        weights["to_out.weight"].to(torch.float16)  # (dim, H)
    )   # [R, dim] FP16

    final = final.view(B, N, N, dim).to(torch.float32)

    return final
