@triton.jit
def final_proj_norm_gate_kernel(
    out_ptr, gate_ptr, weight_ptr,
    gamma_ptr, beta_ptr,
    out_ptr_final,
    total_rows, H, dim,
    eps,
    BLOCK_M: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    pid = tl.program_id(0)  # row blocks
    pid_n = tl.program_id(1) # output dim blocks
    row_start = pid * BLOCK_M
    dim_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    offs_n = dim_start + tl.arange(0, BLOCK_N)
    mask_n = offs_n < dim

    # Compute mean & var for each row
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k in range(0, tl.cdiv(H, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        # load out values
        ptrs = out_ptr + offs_m[:, None] * H + offs_k[None, :]
        x = tl.load(ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)  # [M,K]
        sum_val += tl.sum(x, axis=1)
        sum_sq += tl.sum(x * x, axis=1)

    mean = sum_val / H
    var = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)  # fp32

    # Initialize accumulator for output dims
    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    # Second pass: compute normalized * gate * (gamma + beta) and dot with weight
    for k in range(0, tl.cdiv(H, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        # load out values for this tile
        ptrs = out_ptr + offs_m[:, None] * H + offs_k[None, :]
        x = tl.load(ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)  # [M,K]

        # normalize
        norm = (x - mean[:, None]) * rstd[:, None]  # [M,K] fp32

        # load gamma and beta (fp32)
        gamma = tl.load(gamma_ptr + offs_k, mask=mask_k, other=0.0)  # [K]
        beta = tl.load(beta_ptr + offs_k, mask=mask_k, other=0.0)    # [K]

        norm = norm * gamma[None, :] + beta[None, :]

        # load gate values
        gate = tl.load(gate_ptr + offs_m[:, None] * H + offs_k[None, :], mask=mask_m[:, None] & mask_k[None,:], other=0.0).to(tl.float32)  # [M,K]

        # apply gate
        z = norm * gate  # fp32

        # cast to fp16 for dot
        z_fp16 = z.to(tl.float16)

        # load weight tile (dim tiles)
        # weight shape [dim, H] -> we need weight_T slice (H, dim)
        # weight_ptr + offs_k[:, None] + offs_n[None, :] * H
        weight_tile_ptr = weight_ptr + offs_k[:, None] + offs_n[None, :] * H
        weight_tile = tl.load(weight_tile_ptr, mask=mask_k[:, None] & mask_n[None,:], other=0.0).to(tl.float16)  # [K,N]

        # dot product
        acc += tl.dot(z_fp16, weight_tile)  # [M,N]

    # Store result
    out_ptrs = out_ptr_final + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None,:])

"""
Highly‑optimized forward pass of the **outgoing** TriMul operator (AlphaFold‑3).

Algorithm
---------
1️⃣  First LayerNorm on the input tensor – performed with PyTorch
    `torch.nn.functional.layer_norm` (cuDNN).  This is the most
    bandwidth‑intensive step, cuDNN is faster than a hand‑written
    Triton kernel for the large `[R, dim]` tensor (`R = B·N·N`).

2️⃣  Fuse the four linear projections (`left_proj`, `right_proj`,
    `left_gate`, `right_gate`) into a single GEMM (`torch.nn.functional.linear`).

3️⃣  Compute the output gate with a second GEMM.

4️⃣  Apply the optional mask, the sigmoid‑gates and obtain `left` and `right`.

5️⃣  Batched matrix multiplication (`torch.bmm`) implements the
    N³ multiplicative update  
    `out[b,i,j] = Σ_k left[b,i,k] * right[b,j,k]`  in fp16.

6️⃣  **Fusion kernel** (`final_proj_norm_gate_kernel`):
    * Row‑wise LayerNorm over the hidden dimension `H`.
    * Element‑wise multiplication with the already‑computed output gate.
    * Final projection (`to_out`) – all in a single Triton kernel.
    This removes two extra memory passes (second LN, gating and final linear).

The function `custom_kernel` returns a float32 tensor of shape
`[B, N, N, dim]` and uses at least one Triton kernel as required.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fused final LayerNorm + output‑gate + projection
# ----------------------------------------------------------------------
@triton.jit
def final_proj_norm_gate_kernel(
    # pointers
    inp_ptr,          # fp16  * [total_rows, H]          (output of the N³ update)
    gate_ptr,         # fp16  * [total_rows, H]          (out_gate after sigmoid)
    weight_ptr,       # fp16  * [dim, H]                (to_out weight)
    gamma_ptr,        # fp32  * [H]                     (to_out_norm weight)
    beta_ptr,         # fp32  * [H]                     (to_out_norm bias)
    out_ptr,          # fp16  * [total_rows, dim]       (final result)
    # runtime constants
    total_rows,       # i32
    H,                # i32  hidden dimension
    dim,              # i32  output dimension
    eps,              # f32  epsilon for LayerNorm (compile‑time const)
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_K: tl.constexpr,   # reduction dimension (hidden dim)
    BLOCK_N: tl.constexpr,   # output‑dim tile
):
    pid = tl.program_id(0)          # row‑tile id
    pid_n = tl.program_id(1)        # output‑dim tile id

    row_start = pid * BLOCK_M
    dim_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)    # [BLOCK_M]
    offs_n = dim_start + tl.arange(0, BLOCK_N)    # [BLOCK_N]

    mask_m = offs_m < total_rows
    mask_n = offs_n < dim

    # ------------------------------------------------------------------
    # 1️⃣  Compute mean / variance for each row (fp32 reduction)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k in range(0, tl.cdiv(H, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)   # [BLOCK_K]
        mask_k = offs_k < H

        # load a tile of `inp` (fp16 -> fp32)
        inp_tile = tl.load(
            inp_ptr + offs_m[:, None] * H + offs_k[None, :],
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)

        sum_val += tl.sum(inp_tile,  axis=1)
        sum_sq  += tl.sum(inp_tile * inp_tile, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)               # 1 / sqrt(var + eps)

    # ------------------------------------------------------------------
    # 2️⃣  Accumulate the final output:
    #     out = ((inp - mean) / sqrt(var+eps) * γ + β) * gate  @  weight.T
    # ------------------------------------------------------------------
    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    for k in range(0, tl.cdiv(H, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        # ---- load raw input ------------------------------------------------
        inp_tile = tl.load(
            inp_ptr + offs_m[:, None] * H + offs_k[None, :],
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)                     # [M, K]

        # ---- normalize ------------------------------------------------------
        norm = (inp_tile - mean[:, None]) * rstd[:, None]   # fp32

        # ---- apply LayerNorm weight / bias (γ,β) ----------------------------
        gamma = tl.load(gamma_ptr + offs_k, mask=mask_k, other=0.0)   # fp32
        beta  = tl.load(beta_ptr  + offs_k, mask=mask_k, other=0.0)   # fp32
        norm = norm * gamma[None, :] + beta[None, :]

        # ---- apply output gate ---------------------------------------------
        gate_tile = tl.load(
            gate_ptr + offs_m[:, None] * H + offs_k[None, :],
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)                     # fp32
        gated = norm * gate_tile               # fp32

        # ---- cast to fp16 for Tensor‑Core dot --------------------------------
        gated_fp16 = gated.to(tl.float16)      # [M, K]

        # ---- load a tile of the projection weight (K × N) --------------------
        # weight shape = [dim, H]; we need its transpose (H, dim)
        # address = h_offset + d_offset * H
        weight_tile = tl.load(
            weight_ptr + offs_k[:, None] + offs_n[None, :] * H,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        ).to(tl.float16)                     # [K, N]

        # ---- dot product (M×K) @ (K×N) -> (M×N) -----------------------------
        acc += tl.dot(gated_fp16, weight_tile)   # fp32 accumulation

    # ------------------------------------------------------------------
    # 3️⃣  Write the result (fp16 → fp16)
    # ------------------------------------------------------------------
    out_ptrs = out_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Triton kernel used for the *first* LayerNorm (optional – kept for completeness)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr, y_ptr,
    weight_ptr, bias_ptr,
    total_rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)
        out = norm * w[None, :] + b[None, :]
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])

# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : dict of parameters (float32)
        - config_dict  : {"dim": int, "hidden_dim": int,
                          "nomask": bool (optional)}

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim], dtype float32
    """
    # --------------------------------------------------------------
    # 0️⃣ Unpack
    # --------------------------------------------------------------
    x, mask, weights, config = data
    device = x.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    B, N, _, _ = x.shape
    total_rows = B * N * N          # R
    H = hidden_dim                  # hidden dimension

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (use cuDNN – fast)
    # --------------------------------------------------------------
    x_norm = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    ).to(torch.float16)            # [B,N,N,dim] → fp16

    # --------------------------------------------------------------
    # 2️⃣ Fused projection for left/right and their gates
    # --------------------------------------------------------------
    # concatenate the four weight matrices (each [H, dim])
    proj_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
        ],
        dim=0,
    ).to(device=device, dtype=torch.float16)   # [4*H, dim]

    x_flat = x_norm.view(total_rows, dim)       # [R, dim]
    fused = F.linear(x_flat, proj_weight)       # [R, 4*H] fp16

    left_raw      = fused[:, 0:H]
    right_raw     = fused[:, H:2 * H]
    left_gate_raw = fused[:, 2 * H:3 * H]
    right_gate_raw = fused[:, 3 * H:4 * H]

    # output gate (separate GEMM)
    out_gate_raw = F.linear(
        x_flat,
        weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    )   # [R, H] fp16

    # --------------------------------------------------------------
    # 3️⃣ Mask (if present) & sigmoids
    # --------------------------------------------------------------
    if not nomask and mask is not None:
        # mask: [B,N,N] → [R,1]
        mask_flat = mask.to(device=device, dtype=torch.float16).reshape(total_rows, 1)
        left_raw  = left_raw * mask_flat
        right_raw = right_raw * mask_flat

    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    left  = left_raw * left_gate      # [R, H] fp16
    right = right_raw * right_gate    # [R, H] fp16

    # --------------------------------------------------------------
    # 4️⃣ N³ multiplicative update (batched GEMM)
    # --------------------------------------------------------------
    left  = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B,H,N,N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()  # [B,H,N,N]

    left_mat  = left.view(B * H, N, N)      # [B*H, N, N]
    right_mat = right.view(B * H, N, N)     # [B*H, N, N]

    # out_mat[b*h, i, j] = Σ_k left_mat[b*h, i, k] * right_mat[b*h, j, k]
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B·H, N, N) fp16

    # Restore to [B,N,N,H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H] fp16

    out_flat = out.view(total_rows, H).contiguous()          # [R, H] fp16
    out_gate_flat = out_gate.view(total_rows, H).contiguous()  # [R, H] fp16

    # --------------------------------------------------------------
    # 5️⃣ Final fused LayerNorm + out‑gate + projection (Triton)
    # --------------------------------------------------------------
    final_fp16 = torch.empty(total_rows, dim,
                              dtype=torch.float16,
                              device=device)

    # Tile sizes – tuned for H ≤ 128 and dim ≤ 384
    BLOCK_M = 64          # rows per program
    BLOCK_K = 128         # hidden dimension (must be >= H)
    BLOCK_N = 64          # output‑dim tile

    grid = (triton.cdiv(total_rows, BLOCK_M),
            triton.cdiv(dim, BLOCK_N))

    final_proj_norm_gate_kernel[grid](
        out_flat,
        out_gate_flat,
        weights["to_out.weight"].to(device=device, dtype=torch.float16),   # [dim, H]
        weights["to_out_norm.weight"],    # γ  (fp32)
        weights["to_out_norm.bias"],      # β  (fp32)
        final_fp16,
        total_rows,
        H,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_K=BLOCK_K,
        BLOCK_N=BLOCK_N,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 6️⃣ Cast back to float32 and reshape
    # --------------------------------------------------------------
    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final
