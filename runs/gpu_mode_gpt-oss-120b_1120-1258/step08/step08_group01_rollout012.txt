def gate_and_mask_kernel(
    fused_ptr, out_gate_raw_ptr,
    mask_ptr,
    left_ptr, right_ptr, out_gate_ptr,
    total_rows, hidden_dim,
    mask_enabled: tl.constexpr,
    BLOCK_D: tl.constexpr,
):

    # offsets
    base = row * (4 * hidden_dim)
    left_base = base

@triton.jit
def norm_gate_kernel(
    in_ptr,  # out from bmm, fp16
    out_ptr, # final after LN and out_gate
    out_gate_ptr, # sigmoid out_gate, fp16
    weight_ptr, bias_ptr,
    total_rows,
    hidden_dim,
    eps,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # first pass: compute sum, sum_sq
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(hidden_dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < hidden_dim
        ptrs = in_ptr + offs_m[:, None] * hidden_dim + offs_d[None, :]
        vals = tl.load(ptrs, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    mean = sum_val / hidden_dim
    var = sum_sq / hidden_dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # second pass: compute normalized * weight,bias * out_gate
    for d_off in range(0, tl.cdiv(hidden_dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < hidden_dim
        # load in values
        ptrs = in_ptr + offs_m[:, None] * hidden_dim + offs_d[None, :]
        vals = tl.load(ptrs, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)

        # normalisation
        norm = (vals - mean[:, None]) * rstd[:, None]  # fp32

        # scale and bias
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # fp32
        b = tl.load(bias_ptr + offs_d, mask=mask_d, other=0.0)  # fp32
        out_val = norm * w[None, :] + b[None, :]  # fp32

        # multiply by gated out_gate
        out_gate = tl.load(out_gate_ptr + offs_m[:, None] * hidden_dim + offs_d[None, :],
                           mask=mask_m[:, None] & mask_d[None, :],
                           other=0.0).to(tl.float32)

        out_val = out_val * out_gate

        # store as fp16
        out_ptrs = out_ptr + offs_m[:, None] * hidden_dim + offs_d[None, :]
        tl.store(out_ptrs, out_val.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])
```

def custom_kernel(data):
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = x.device
    eps = 1e-5

    B,N,_,_ = x.shape
    total_rows = B * N * N
    H = hidden_dim

    # step 1: first LayerNorm using torch.nn.functional.layer_norm
    # Input x: [B,N,N,dim] float32
    ln1_out_fp32 = torch.nn.functional.layer_norm(
        x, normalized_shape=(dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps
    )
    # Cast to fp16 for following linear
    ln1_out = ln1_out_fp32.to(torch.float16)

    # step 2: fused projection + gates linear
    # combine weights into one large weight matrix (4*H, dim)
    w_proj = torch.cat([
        weights["left_proj.weight"],
        weights["right_proj.weight"],
        weights["left_gate.weight"],
        weights["right_gate.weight"],
    ], dim=0).to(device=device, dtype=torch.float16)
    fused = torch.nn.functional.linear(ln1_out, w_proj)  # [R, 4H] fp16

    # step 3: out_gate linear
    out_gate_raw = torch.nn.functional.linear(
        ln1_out, weights["out_gate.weight"].to(device=device, dtype=torch.float16)  # [H, dim]
    )  # [R, H] fp16

    # step 4: apply mask, sigmoid and gate using Triton
    if not nomask:
        mask_flat = mask.to(device=device, dtype=torch.float16).reshape(total_rows)  # [R]
    else:
        mask_flat = torch.empty(0, device=device)  # unused

    # allocate result buffers
    left = torch.empty(total_rows, H, dtype=torch.float16, device=device)
    right = torch.empty(total_rows, H, dtype=torch.float16, device=device)
    out_gate = torch.empty(total_rows, H, dtype=torch.float16, device=device)

    # Triton kernel launch
    BLOCK_D = 64  # maybe 64 or 128
    grid = (total_rows,)
    mask_enabled = not nomask
    gate_and_mask_kernel[grid](
        fused, out_gate_raw,
        mask_flat,
        left, right, out_gate,
        total_rows,
        H,
        mask_enabled,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # step 5: reshape left/right -> [B,N,N,H]
    left = left.view(B, N, N, H)
    right = right.view(B, N, N, H)

    # move head dim to batch for bmm
    left_perm = left.permute(0,3,1,2).contiguous()
    right_perm = right.permute(0,3,1,2).contiguous()
    left_mat = left_perm.view(B*H, N, N)
    right_mat = right_perm.view(B*H, N, N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1,2))  # (B*H,N,N)

    out_flat = out_mat.view(total_rows, H).contiguous()

    # step 6: second LayerNorm + out_gate multiplication via Triton
    final_ln_out = torch.empty_like(out_flat, dtype=torch.float16)
    grid2 = (triton.cdiv(total_rows, BLOCK_M),)  # will define later variable

    # block sizes set
    BLOCK_M = 256
    BLOCK_D2 = 64

    norm_gate_kernel[grid2](
        out_flat,
        final_ln_out,
        out_gate,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D2,
        num_warps=4,
    )

    # step 7: final linear
    final_fp16 = torch.nn.functional.linear(
        final_ln_out,
        weights["to_out.weight"].to(device=device, dtype=torch.float16)  # shape [dim, H]
    )  # [R, dim] fp16

    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final
```

    fused_ptr, out_gate_raw_ptr,
    mask_ptr,

    total_rows, hidden_dim,

    if mask_enabled:

    else:

    # compute base offsets
    base = row * (4 * hidden_dim)
    left_base = base
    right_base = base + hidden_dim

    right_gate_base = base + 3 * hidden_dim

                           mask=mask_row & mask_d,

                                mask=mask_row & mask_d,

"""
Highly‑optimized forward pass for the **outgoing** TriMul operator
(AlphaFold‑3 style).  The implementation follows the reference PyTorch
module but fuses the heavy element‑wise work in two small Triton
kernels:

1️⃣  `gate_and_mask_kernel` – applies the optional mask, computes the
    sigmoid‑gates and produces the gated tensors *left*, *right* and the
    gated output‑gate.  This replaces three separate element‑wise passes.

2️⃣  `norm_gate_kernel` – combines the second LayerNorm over the hidden
    dimension with the multiplication by the previously computed output‑gate.
    The kernel performs a first pass to obtain per‑row mean/variance and a
    second pass that normalises, applies γ/β and finally multiplies by the
    output‑gate, all in FP32 and writes the result back in FP16.

All large dense GEMMs (projections, the N³ multiplicative update and the
final projection) are executed with `torch.nn.functional.linear` /
`torch.bmm`, which dispatch to cuBLAS/Tensor‑Cores on an H100.  Mixed‑precision
(FP16) is used throughout; the final result is cast back to FP32 as required.

The code works for any of the test configurations listed in the prompt,
handles the optional mask and requires only the forward pass.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel 1: mask + sigmoid gates + element‑wise weighting
# ----------------------------------------------------------------------
@triton.jit
def gate_and_mask_kernel(
    fused_ptr,               # const   * [R, 4*H]   (fp16)
    out_gate_raw_ptr,        # const   * [R, H]     (fp16)
    mask_ptr,                # const   * [R]        (fp16) – optional
    left_ptr,                # mutable * [R, H]     (fp16)
    right_ptr,               # mutable * [R, H]     (fp16)
    out_gate_ptr,            # mutable * [R, H]     (fp16)
    R,                       # int: total rows = B*N*N
    H,                       # int: hidden dimension
    mask_enabled: tl.constexpr,  # bool compile‑time flag
    BLOCK_D: tl.constexpr,       # vector tile size (≤ H)
):
    """
    For each row i ∈ [0, R):
        left_i  = (left_raw_i  * mask_i) * sigmoid(left_gate_raw_i)
        right_i = (right_raw_i * mask_i) * sigmoid(right_gate_raw_i)
        out_gate_i = sigmoid(out_gate_raw_i)
    All tensors are stored in row‑major layout.
    """
    row = tl.program_id(0)                     # one program per row
    active = row < R

    # ------------------------------------------------------------------
    # optional scalar mask (broadcast over H)
    # ------------------------------------------------------------------
    if mask_enabled:
        m = tl.load(mask_ptr + row, mask=active, other=0.0).to(tl.float32)
    else:
        m = 1.0

    # base offsets for the four projection slices inside the fused matrix
    base = row * (4 * H)                # start of the row in the fused tensor
    left_base      = base
    right_base     = base + H
    left_gate_base = base + 2 * H
    right_gate_base= base + 3 * H
    out_gate_raw_base = row * H

    # ------------------------------------------------------------------
    # iterate over the hidden dimension
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(H, BLOCK_D)):
        d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = d < H

        # load raw values (fp16 → fp32 for arithmetic)
        left_raw = tl.load(fused_ptr + left_base + d,
                           mask=active & mask_d,
                           other=0.0).to(tl.float32)
        right_raw = tl.load(fused_ptr + right_base + d,
                            mask=active & mask_d,
                            other=0.0).to(tl.float32)
        left_gate_raw = tl.load(fused_ptr + left_gate_base + d,
                                mask=active & mask_d,
                                other=0.0).to(tl.float32)
        right_gate_raw = tl.load(fused_ptr + right_gate_base + d,
                                 mask=active & mask_d,
                                 other=0.0).to(tl.float32)

        out_gate_raw = tl.load(out_gate_raw_ptr + out_gate_raw_base + d,
                               mask=active & mask_d,
                               other=0.0).to(tl.float32)

        # ------------------------------------------------------------------
        # apply mask (scalar) and sigmoid‑gates
        # ------------------------------------------------------------------
        # mask is a scalar broadcasted over the hidden dimension
        left_raw  = left_raw  * m
        right_raw = right_raw * m

        left_gate  = 1.0 / (1.0 + tl.exp(-left_gate_raw))
        right_gate = 1.0 / (1.0 + tl.exp(-right_gate_raw))
        out_gate   = 1.0 / (1.0 + tl.exp(-out_gate_raw))

        left  = left_raw  * left_gate
        right = right_raw * right_gate

        # ------------------------------------------------------------------
        # write results (fp16)
        # ------------------------------------------------------------------
        tl.store(left_ptr  + left_base  + d,
                 left .to(tl.float16),
                 mask=active & mask_d)
        tl.store(right_ptr + right_base + d,
                 right.to(tl.float16),
                 mask=active & mask_d)
        tl.store(out_gate_ptr + out_gate_raw_base + d,
                 out_gate.to(tl.float16),
                 mask=active & mask_d)


# ----------------------------------------------------------------------
# Triton kernel 2: fused LayerNorm (γ/β) + multiplication by out_gate
# ----------------------------------------------------------------------
@triton.jit
def norm_gate_kernel(
    inp_ptr,           # const   * [R, H] (fp16) – output of the N³ update
    out_ptr,           # mutable * [R, H] (fp16) – LN‑scaled and gated
    out_gate_ptr,      # const   * [R, H] (fp16) – sigmoid(out_gate_raw)
    gamma_ptr,         # const   * [H]    (fp32)
    beta_ptr,          # const   * [H]    (fp32)
    R,                 # int: total rows = B*N*N
    H,                 # int: hidden dimension
    eps,               # float32 constant
    BLOCK_M: tl.constexpr,   # rows per program (default 256)
    BLOCK_D: tl.constexpr,   # hidden‑dim tile (default 64)
):
    """
    Row‑wise LayerNorm followed by element‑wise multiplication with the
    previously computed output‑gate.
        y = ((x - μ) / sqrt(σ²+eps)) * γ + β
        y = y * out_gate
    """
    pid = tl.program_id(0)                     # one program per BLOCK_M rows
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)   # [BLOCK_M]
    mask_m = offs_m < R

    # ------------------------------------------------------------------
    # 1️⃣  Compute mean & variance (FP32 accumulator)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(H, BLOCK_D)):
        d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = d < H
        ptrs = inp_ptr + offs_m[:, None] * H + d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)                     # 1/√(var+eps)

    # ------------------------------------------------------------------
    # 2️⃣  Normalise, apply γ/β and multiply by out_gate
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(H, BLOCK_D)):
        d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = d < H

        # load original values
        ptrs = inp_ptr + offs_m[:, None] * H + d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        # normalisation
        norm = (vals - mean[:, None]) * rstd[:, None]   # FP32

        # scale & bias (γ/β)
        w = tl.load(gamma_ptr + d, mask=mask_d, other=0.0)   # γ  FP32
        b = tl.load(beta_ptr  + d, mask=mask_d, other=0.0)   # β  FP32
        out = norm * w[None, :] + b[None, :]                # FP32

        # multiply by the pre‑sigmoided out_gate
        out_gate_vals = tl.load(out_gate_ptr + offs_m[:, None] * H + d[None, :],
                                mask=mask_m[:, None] & mask_d[None, :],
                                other=0.0).to(tl.float32)
        out = out * out_gate_vals

        # store
        out_ptrs = out_ptr + offs_m[:, None] * H + d[None, :]
        tl.store(out_ptrs,
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : dict containing all LayerNorm & Linear parameters
        - config_dict  : dict with keys "dim", "hidden_dim", "nomask" (bool, optional)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim], dtype float32
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    device = x.device
    eps = 1e-5

    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)          # True → ignore mask
    B, N, _, _ = x.shape
    R = B * N * N                              # total rows (B·N·N)
    H = hidden_dim

    # ------------------------------------------------------------------
    # 0️⃣ first LayerNorm (use highly‑optimised torch implementation)
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )
    # cast to fp16 for the following GEMMs
    x_norm = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 1️⃣ fused projection + gate linear (4·H outputs)
    # ------------------------------------------------------------------
    w_proj = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
        ],
        dim=0,
    ).to(device=device, dtype=torch.float16)   # [4·H, dim]

    fused = F.linear(x_norm, w_proj)            # [R, 4·H] (fp16)

    # ------------------------------------------------------------------
    # 2️⃣ output‑gate linear (H outputs)
    # ------------------------------------------------------------------
    out_gate_raw = F.linear(
        x_norm,
        weights["out_gate.weight"].to(device=device, dtype=torch.float16),  # [H, dim]
    )                                            # [R, H] (fp16)

    # ------------------------------------------------------------------
    # 3️⃣ mask + sigmoid gates (in a single Triton kernel)
    # ------------------------------------------------------------------
    # flatten mask to [R] (float16) – ignored when nomask==True
    if mask is not None and not nomask:
        mask_flat = mask.to(device=device, dtype=torch.float16).reshape(R)
    else:
        # dummy tensor; the kernel will not read it when mask_enabled=False
        mask_flat = torch.empty(0, device=device, dtype=torch.float16)

    left   = torch.empty(R, H, dtype=torch.float16, device=device)
    right  = torch.empty(R, H, dtype=torch.float16, device=device)
    out_gate = torch.empty(R, H, dtype=torch.float16, device=device)

    BLOCK_D = 64  # works for H=128 and H=384
    grid = (R,)  # one program per row
    gate_and_mask_kernel[grid](
        fused,
        out_gate_raw,
        mask_flat,
        left,
        right,
        out_gate,
        R,
        H,
        not nomask,           # compile‑time bool: mask enabled?
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 4️⃣ N³ multiplicative update (batched GEMM)
    # ------------------------------------------------------------------
    left  = left.view(B, N, N, H)          # [B, N, N, H]
    right = right.view(B, N, N, H)         # [B, N, N, H]

    # move head dimension to batch axis → [B·H, N, N]
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # batched matrix multiplication (Tensor‑Core)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B·H, N, N)

    out_flat = out_mat.view(R, H).contiguous()   # [R, H] fp16

    # ------------------------------------------------------------------
    # 5️⃣ second LayerNorm + multiplication by out_gate (fused Triton kernel)
    # ------------------------------------------------------------------
    out_pre = torch.empty_like(out_flat, dtype=torch.float16)  # buffer for LN‑scaled + gate
    BLOCK_M = 256
    BLOCK_D2 = 64
    grid2 = (triton.cdiv(R, BLOCK_M),)
    norm_gate_kernel[grid2](
        out_flat,
        out_pre,
        out_gate,
        weights["to_out_norm.weight"],   # γ  (fp32)
        weights["to_out_norm.bias"],     # β  (fp32)
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D2,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 6️⃣ final projection back to `dim`
    # ------------------------------------------------------------------
    final_fp16 = F.linear(
        out_pre,
        weights["to_out.weight"].to(device=device, dtype=torch.float16)  # [dim, H]
    )   # [R, dim] fp16

    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final
