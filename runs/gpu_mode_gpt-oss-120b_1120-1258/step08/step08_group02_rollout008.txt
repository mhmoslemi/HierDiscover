"""
custom_kernel: high‑performance forward of the *outgoing* TriMul operator.

Algorithm
---------
1️⃣ Row‑wise LayerNorm over the input channel dimension (dim) – Triton kernel.
2️⃣ Fuse the four linear layers (left_proj, right_proj, left_gate, right_gate)
   into a single GEMM:  X_norm @ W_fused.T  →  [R, 4·H]   (R = B·N·N,
   H = hidden_dim).  Split the result into the four tensors.
3️⃣ Compute the gate for the output (out_gate) with a second GEMM.
4️⃣ (Optional) Apply the binary mask to the two projected tensors.
   Apply sigmoid to the three gate tensors (in‑place) and fuse them:
        left  = left_raw  * sigmoid(left_gate)
        right = right_raw * sigmoid(right_gate)
5️⃣ Reshape left/right → (B·H, N, N) and perform a batched GEMM:
        out_mat = left_mat @ right_matᵀ          # shape (B·H, N, N)
   Reshape back to (B, N, N, H).
6️⃣ Fuse the second LayerNorm (over the hidden dimension H) *and* the
   element‑wise multiplication by out_gate into a single Triton kernel.
7️⃣ Final linear projection back to the original dimension (dim) – FP16 GEMM.
   Convert the result back to FP32 and return.

All heavy element‑wise work (the two LayerNorms + the gate‑multiplication)
is done in Triton; the large GEMMs use the highly‑optimized cuBLAS
back‑ends via `torch.nn.functional.linear` and `torch.bmm`.  The code
runs entirely in half‑precision except for the final conversion to float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (float32 → float16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   * [total_rows, dim] (float32)
    y_ptr,          # mutable * [total_rows, dim] (float16)
    weight_ptr,     # const   * [dim]            (float32)   # γ
    bias_ptr,       # const   * [dim]            (float32)   # β
    total_rows,     # int32   (runtime)
    dim,            # int32   (runtime)
    eps,            # float   (compile‑time)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # cols per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # --------------------------------------------------------------
    # 1️⃣ Compute per‑row mean & variance (FP32)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)            # FP32 load
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                             # 1 / sqrt(var+eps)

    # --------------------------------------------------------------
    # 2️⃣ Write normalized values γ·(x−μ)·rstd + β  (store as FP16)
    # --------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]                      # still FP32

        out_fp16 = out.to(tl.float16)
        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
# Triton kernel: fused LayerNorm + element‑wise gate (float32 → float16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_gate_kernel(
    x_ptr,          # const   * [total_rows, dim] (float32)
    gate_ptr,       # const   * [total_rows, dim] (float16)
    y_ptr,          # mutable * [total_rows, dim] (float16)
    weight_ptr,     # const   * [dim]            (float32)   # γ
    bias_ptr,       # const   * [dim]            (float32)   # β
    total_rows,     # int32   (runtime)
    dim,            # int32   (runtime)
    eps,            # float   (compile‑time)
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # --------------------------------------------------------------
    # Compute mean & variance (ignoring gate)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------------------------------------------------------------
    # Normalise, apply γ/β and multiply by gate (store FP16)
    # --------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        # raw activation
        x_ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        x_vals = tl.load(x_ptrs,
                         mask=mask_m[:, None] & mask_d[None, :],
                         other=0.0).to(tl.float32)

        # gate (same shape as x)
        g_ptrs = gate_ptr + offs_m[:, None] * dim + offs_d[None, :]
        g_vals = tl.load(g_ptrs,
                         mask=mask_m[:, None] & mask_d[None, :],
                         other=0.0).to(tl.float32)

        norm = (x_vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β

        out = (norm * w[None, :] + b[None, :]) * g_vals          # FP32
        out_fp16 = out.to(tl.float16)

        y_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(y_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
# Forward entry‑point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : mapping with all Linear/LayerNorm parameters
        - config_dict  : {"dim": int, "hidden_dim": int,
                         "nomask": bool (optional)}

    Returns
    -------
    torch.Tensor, shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    total_rows = B * N * N                 # R = B·N·N
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → FP16
    # ------------------------------------------------------------------
    x_flat = x.view(total_rows, dim).contiguous()        # [R, dim] FP32
    x_norm_fp16 = torch.empty(total_rows, dim,
                              dtype=torch.float16,
                              device=device)

    # Tuned block sizes for H100: 256 rows × 128 cols per iteration
    BLOCK_M = 256
    BLOCK_D = 128
    grid = (triton.cdiv(total_rows, BLOCK_M),)

    layernorm_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],          # γ (float32)
        weights["norm.bias"],            # β (float32)
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fuse left/right projections + their gates (4·H outputs)
    # ------------------------------------------------------------------
    # Concatenate weight matrices along the output dimension (no bias)
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)                                 # [4·H, dim]

    # Fused GEMM :  [R, dim] @ [4·H, dim]ᵀ → [R, 4·H]   (FP16)
    fused = F.linear(x_norm_fp16, w_fused)             # FP16

    left_raw       = fused[:, 0:H]                     # [R, H]
    right_raw      = fused[:, H:2*H]                   # [R, H]
    left_gate_raw  = fused[:, 2*H:3*H]                 # [R, H]
    right_gate_raw = fused[:, 3*H:4*H]                 # [R, H]

    # Output‑gate (separate GEMM)
    out_gate_raw = F.linear(
        x_norm_fp16,
        weights["out_gate.weight"].to(torch.float16)   # [H, dim]
    )                                                   # [R, H]

    # ------------------------------------------------------------------
    # 3️⃣ Mask (if any) and sigmoid gates
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_fp16 = mask.to(torch.float16).reshape(total_rows, 1)
        left_raw  = left_raw  * mask_fp16
        right_raw = right_raw * mask_fp16

    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()                # [R, H]  FP16

    left  = left_raw  * left_gate                       # [R, H]  FP16
    right = right_raw * right_gate                      # [R, H]  FP16

    # ------------------------------------------------------------------
    # 4️⃣ Batched N³ multiplicative update (B·H independent GEMMs)
    # ------------------------------------------------------------------
    # reshape to (B, N, N, H)
    left  = left.view(B, N, N, H)
    right = right.view(B, N, N, H)

    # bring hidden dimension to batch axis → (B·H, N, N)
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 5️⃣ Fused LayerNorm + out_gate multiplication (over hidden dim H)
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, H).contiguous()    # [R, H] FP16
    out_gate_flat = out_gate                           # already FP16
    out_norm_gate_fp16 = torch.empty_like(out_flat, dtype=torch.float16, device=device)

    layernorm_gate_kernel[grid](
        out_flat,
        out_gate_flat,
        out_norm_gate_fp16,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Final linear projection back to original dim
    # ------------------------------------------------------------------
    final_fp16 = F.linear(
        out_norm_gate_fp16,
        weights["to_out.weight"].to(torch.float16)   # [dim, H]
    )                                                # [R, dim] FP16

    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final
