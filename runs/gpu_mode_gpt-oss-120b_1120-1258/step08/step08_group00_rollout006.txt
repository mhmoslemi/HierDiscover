@triton.jit
def layernorm_mul_kernel(
    x_ptr,                 # const *[rows, dim] (fp16)
    out_ptr,               # mutable *[rows, dim] (fp16)
    weight_ptr,            # const *[dim] (fp32)
    bias_ptr,              # const *[dim] (fp32)
    gate_ptr,              # const *[rows, dim] (fp16)
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # Compute mean and var
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # In second pass compute normalized, apply weight, bias, and gate
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        # load input
        ptr_x = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr_x,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        # load gate
        ptr_gate = gate_ptr + offs_m[:, None] * dim + offs_d[None, :]
        gate = tl.load(ptr_gate,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        # normalize
        norm = (vals - mean[:, None]) * rstd[:, None]

        # weight and bias (float32)
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)

        out = norm * w[None, :] + b[None, :]   # fp32
        out = out * gate                         # gated
        out_fp16 = out.to(tl.float16)

        out_ptrs = out_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

"""
TriMul (outgoing) forward pass – high‑throughput Triton + PyTorch implementation.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the 4‑D tensor (B·N·N × dim) – done with
   `torch.nn.functional.layer_norm` (fast CUDNN kernel, fp32 -> fp16).

2️⃣ Fuse the five linear maps (left/right projections + three gates) into a
   single GEMM:
       X_norm @ W_fusedᵀ  →  (R, 5·H)   with R = B·N·N, H = hidden_dim.

3️⃣ Triton kernel `split_fuse_gate_kernel`:
   * splits the fused projection into left, right and three gate tensors,
   * applies the optional pair‑wise mask,
   * evaluates sigmoid(gate) in fp32 for stability,
   * writes `left`, `right` and `out_gate` (all fp16).

4️⃣ Batched matrix multiplication (Tensor‑core GEMM):
       out[b·H+i] = left_mat[b·H+i] @ right_mat[b·H+i]ᵀ
   performed with `torch.bmm` on tensors of shape (B·H, N, N).

5️⃣ Triton kernel `layernorm_mul_kernel`:
   * Row‑wise LayerNorm of the intermediate tensor (R, H),
   * immediately multiplies by the pre‑computed output gate,
   * all in fp16 (mean/var accumulated in fp32).

6️⃣ Final linear projection back to `dim` (fp16 → fp32) with a single GEMM.

Only the two Triton kernels are custom – the heavy GEMMs are delegated to
cuBLAS/Tensor‑cores, giving a fast end‑to‑end implementation that meets the
target sub‑millisecond runtime on H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# Triton kernel: split fused projection, apply mask, sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[rows, 5*H] (fp16)
    left_ptr,          # mutable *[rows, H]   (fp16)
    right_ptr,         # mutable *[rows, H]   (fp16)
    out_gate_ptr,      # mutable *[rows, H]   (fp16)
    mask_ptr,          # const   *[rows]    (fp16)   (ignored if HAS_MASK=0)
    rows: tl.int32,
    H: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)               # [BLOCK_R]
    mask_r = offs_r < rows

    # broadcasted mask (1.0 if no mask)
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)
        mask_val = m_val[:, None]    # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)          # [BLOCK_H]
        mask_h = offs_h < H

        # compute offsets for the five concatenated segments
        base = offs_r[:, None] * (5 * H) + offs_h[None, :]        # [R,HB]

        left_raw_ptr    = proj_ptr + base
        right_raw_ptr   = proj_ptr + base + H
        left_gate_ptr   = proj_ptr + base + 2 * H
        right_gate_ptr  = proj_ptr + base + 3 * H
        out_gate_raw_ptr= proj_ptr + base + 4 * H

        left_raw    = tl.load(left_raw_ptr,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0).to(tl.float16)
        right_raw   = tl.load(right_raw_ptr,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0).to(tl.float16)
        left_g_raw  = tl.load(left_gate_ptr,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0).to(tl.float16)
        right_g_raw = tl.load(right_gate_ptr,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0).to(tl.float16)
        out_g_raw   = tl.load(out_gate_raw_ptr,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0).to(tl.float16)

        # sigmoid in fp32 for numerical stability
        lg_f32 = left_g_raw.to(tl.float32)
        left_gate = (1.0 / (1.0 + tl.exp(-lg_f32))).to(tl.float16)

        rg_f32 = right_g_raw.to(tl.float32)
        right_gate = (1.0 / (1.0 + tl.exp(-rg_f32))).to(tl.float16)

        og_f32 = out_g_raw.to(tl.float32)
        out_gate = (1.0 / (1.0 + tl.exp(-og_f32))).to(tl.float16)

        # apply mask and gates
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate

        # store results
        left_dst_ptr    = left_ptr    + offs_r[:, None] * H + offs_h[None, :]
        right_dst_ptr   = right_ptr   + offs_r[:, None] * H + offs_h[None, :]
        out_gate_dst_ptr= out_gate_ptr+ offs_r[:, None] * H + offs_h[None, :]

        tl.store(left_dst_ptr,    left,    mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_dst_ptr,   right,   mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_dst_ptr,out_gate,mask=mask_r[:, None] & mask_h[None, :])

# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm + element‑wise output‑gate multiplication
# ----------------------------------------------------------------------
@triton.jit
def layernorm_mul_kernel(
    x_ptr,         # const   *[rows, dim] (fp16)
    out_ptr,       # mutable *[rows, dim] (fp16)
    weight_ptr,    # const   *[dim]      (fp32)   γ
    bias_ptr,      # const   *[dim]      (fp32)   β
    gate_ptr,      # const   *[rows, dim] (fp16)   out_gate
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)                 # [BLOCK_M]
    mask_m = offs_m < rows

    # ---------- first pass: mean & variance (fp32) ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)       # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]  # [M,D]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- second pass: normalize, apply γ/β, multiply gate ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        # load input and gate
        x_val = tl.load(x_ptr + offs_m[:, None] * dim + offs_d[None, :],
                        mask=mask_m[:, None] & mask_d[None, :],
                        other=0.0).to(tl.float32)
        g_val = tl.load(gate_ptr + offs_m[:, None] * dim + offs_d[None, :],
                        mask=mask_m[:, None] & mask_d[None, :],
                        other=0.0).to(tl.float32)

        # normalize
        norm = (x_val - mean[:, None]) * rstd[:, None]

        # γ/β (fp32)
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)

        out = norm * w[None, :] + b[None, :]   # fp32
        out = out * g_val                       # apply output gate

        out_fp16 = out.to(tl.float16)

        out_ptrs = out_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.
    Arguments:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
            - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
            - weights_dict : dict of model weights (float32 tensors, bias=False for Linear)
            - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)
    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim          = cfg["dim"]
    hidden_dim   = cfg["hidden_dim"]
    eps          = 1e-5
    device       = x.device

    B, N, _, _ = x.shape
    R = B * N * N                       # total rows for row‑wise ops
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (fp32 → fp16) – use highly‑optimized CUDNN kernel
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    ).half()                                   # [B,N,N,dim] fp16

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear: left/right projections + 3 gates  →  (R, 5·H)
    # ------------------------------------------------------------------
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).half()                                   # [5·H, dim]

    proj = F.linear(x_norm.view(R, dim), w_concat)   # [R,5·H] fp16
    del w_concat, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ Split, mask, sigmoid gates (single Triton kernel)
    # ------------------------------------------------------------------
    left      = torch.empty(R, H, dtype=torch.float16, device=device)
    right     = torch.empty(R, H, dtype=torch.float16, device=device)
    out_gate  = torch.empty(R, H, dtype=torch.float16, device=device)

    if cfg.get("nomask", True):
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy
    else:
        HAS_MASK = 1
        mask_ptr = mask.to(torch.float16).reshape(R)                     # [R]

    BLOCK_R = 256
    BLOCK_H = 64
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM over hidden dimension (B·H, N, N) × (B·H, N, N)ᵀ
    # ------------------------------------------------------------------
    left  = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B,H,N,N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()  # [B,H,N,N]

    left_mat  = left.view(B * H, N, N)        # [B·H, N, N]
    right_mat = right.view(B * H, N, N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [B·H, N, N]

    # reshape to (R, H) for the next LayerNorm
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # [B,N,N,H]
    out_flat = out.view(R, H)                                          # [R,H]
    del out, left_mat, right_mat, left, right

    # ------------------------------------------------------------------
    # 5️⃣ Row‑wise LayerNorm on the hidden‑dim tensor + output‑gate multiply
    # ------------------------------------------------------------------
    out_fused = torch.empty_like(out_flat, dtype=torch.float16, device=device)

    BLOCK_M = 256
    BLOCK_D = 64
    grid2 = (triton.cdiv(R, BLOCK_M),)
    layernorm_mul_kernel[grid2](
        out_flat,
        out_fused,
        weights["to_out_norm.weight"],   # γ (fp32)
        weights["to_out_norm.bias"],     # β (fp32)
        out_gate,
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del out_flat, out_gate

    # ------------------------------------------------------------------
    # 6️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    final_half = F.linear(
        out_fused,
        weights["to_out.weight"].half(),          # [dim, H]
    )                                            # [R, dim] fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
