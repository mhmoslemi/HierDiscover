"""
TriMul (outgoing) – fast Triton implementation
================================================

The forward pass consists of:

1. Layer‑norm over the input channel dimension (dim).
2. Two linear projections (left / right) producing tensors of shape
   [B, S, S, H] (H = hidden_dim).
3. Optional mask applied to the projected tensors.
4. Gating: sigmoid(linear(x)) for left, right and the final output.
5. Core N³ reduction:  out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   – realised as a batched GEMM (left @ rightᵀ) using a custom Triton kernel.
   The GEMM works on FP16 for bandwidth while accumulating in FP32.
6. Layer‑norm on the hidden dimension, multiplication by the output gate,
   and a final linear projection back to the original channel size.

Only step 5 is performed on the GPU with Triton; the remaining operations
are cheap PyTorch calls.  The kernel tiles the multiply with
BLOCK_M × BLOCK_N × BLOCK_K and supports arbitrary sequence lengths via
masking of out‑of‑bounds tiles.

The implementation supports all configurations described in the prompt,
including optional masking and different input/hidden dimensions.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------------------- #
# Triton kernel: batched GEMM for the TriMul reduction (left @ rightᵀ)
# --------------------------------------------------------------------------- #
@triton.jit
def _trimul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,                # pointers
    M, N, K,                            # matrix dimensions (all = seq_len)
    stride_am, stride_ah, stride_ai, stride_ak,   # A strides: batch, hidden, i, k
    stride_ba, stride_bh, stride_bk, stride_bn,   # B strides: batch, hidden, k, j
    stride_cm, stride_ch, stride_ci, stride_cj,   # C strides: batch, hidden, i, j
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    hidden_dim: tl.constexpr,
):
    """
    Compute C = A @ Bᵀ for every (batch, hidden) slice.
    A : (batch, hidden, M, K)      – fp16
    B : (batch, hidden, K, N)      – fp16 (already transposed)
    C : (batch, hidden, M, N)      – fp32
    """
    pid_m = tl.program_id(0)   # tile row
    pid_n = tl.program_id(1)   # tile col
    pid_z = tl.program_id(2)   # flattened (batch * hidden)

    # ------------------------------------------------------------------ #
    # Decode batch and hidden indices from pid_z
    # ------------------------------------------------------------------ #
    batch_idx  = pid_z // hidden_dim
    hidden_idx = pid_z - batch_idx * hidden_dim

    # Base pointers for the chosen (batch, hidden) slice
    a_base = a_ptr + batch_idx * stride_am + hidden_idx * stride_ah
    b_base = b_ptr + batch_idx * stride_ba + hidden_idx * stride_bh
    c_base = c_ptr + batch_idx * stride_cm + hidden_idx * stride_ch

    # -------------------------------------------------------------- #
    # Tile offsets (within the current M,N block)
    # -------------------------------------------------------------- #
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------------------------------------------------- #
    # K‑loop: load tiles of A (M×K) and B (K×N) and accumulate
    # -------------------------------------------------------------- #
    for start_k in range(0, K, BLOCK_K):
        cur_k = start_k + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_base + (offs_m[:, None] * stride_ai
                           + cur_k[None, :] * stride_ak)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # Load B tile: shape (BLOCK_K, BLOCK_N)
        b_ptrs = b_base + (cur_k[:, None] * stride_bk
                           + offs_n[None, :] * stride_bn)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # Accumulate dot product
        acc += tl.dot(a, b)

    # -------------------------------------------------------------- #
    # Write back the result tile (FP32)
    # -------------------------------------------------------------- #
    c_ptrs = c_base + (offs_m[:, None] * stride_ci
                       + offs_n[None, :] * stride_cj)
    tl.store(c_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


# --------------------------------------------------------------------------- #
# Public entry point expected by the evaluation harness
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, S, S, C]  (C == dim)
        - mask_tensor  : torch.Tensor of shape [B, S, S] or ``None``
        - weights_dict : dictionary with all linear / layer‑norm parameters
        - config_dict  : {'dim': C, 'hidden_dim': H, 'nomask': bool, ...}

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, S, S, C] (float32)
    """
    # ------------------------------------------------------------------- #
    # Unpack arguments
    # ------------------------------------------------------------------- #
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, S, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------- #
    # Extract model weights (already on the correct device)
    # ------------------------------------------------------------------- #
    norm_weight           = weights["norm.weight"]
    norm_bias             = weights["norm.bias"]
    left_proj_weight      = weights["left_proj.weight"]
    right_proj_weight     = weights["right_proj.weight"]
    left_gate_weight      = weights["left_gate.weight"]
    right_gate_weight     = weights["right_gate.weight"]
    out_gate_weight       = weights["out_gate.weight"]
    to_out_norm_weight    = weights["to_out_norm.weight"]
    to_out_norm_bias      = weights["to_out_norm.bias"]
    to_out_weight         = weights["to_out.weight"]

    # ------------------------------------------------------------------- #
    # 1. Input LayerNorm over the channel dimension (dim)
    # ------------------------------------------------------------------- #
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )   # [B, S, S, dim] , float32

    # ------------------------------------------------------------------- #
    # 2. Linear projections (no bias) → left / right
    # ------------------------------------------------------------------- #
    left  = F.linear(x, left_proj_weight)    # [B, S, S, H]
    right = F.linear(x, right_proj_weight)   # [B, S, S, H]

    # ------------------------------------------------------------------- #
    # 3. Optional mask (applied before gating)
    # ------------------------------------------------------------------- #
    if not nomask and mask_tensor is not None:
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)   # [B, S, S, 1]
        left  = left * mask
        right = right * mask

    # ------------------------------------------------------------------- #
    # 4. Gating (sigmoid of linear projections)
    # ------------------------------------------------------------------- #
    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))    # [B, S, S, H]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))   # [B, S, S, H]
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))     # [B, S, S, H]

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------- #
    # 5. Core N³ reduction via a batched GEMM (Triton)
    # ------------------------------------------------------------------- #
    # Cast to half‑precision for the GEMM (memory‑bandwidth friendly)
    left_h  = left.to(torch.float16).contiguous()
    right_h = right.to(torch.float16).contiguous()

    # Rearrange tensors so that the hidden dimension is the second axis
    #      left_h  : [B, H, i, k]
    #      right_h_T: [B, H, k, j]   (transpose the last two axes)
    left_h   = left_h.permute(0, 3, 1, 2).contiguous()
    right_h_T = right_h.permute(0, 3, 2, 1).contiguous()

    # Output buffer (FP32 accumulation)
    out_fp32 = torch.empty(
        (B, hidden_dim, S, S), dtype=torch.float32, device=device
    )

    # Choose block sizes (tuned for H100, FP16 dot)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    # Grid: (M‑tiles, N‑tiles, batch*hidden)
    grid = (
        triton.cdiv(S, BLOCK_M),          # rows
        triton.cdiv(S, BLOCK_N),          # cols
        B * hidden_dim,                   # (batch * hidden)
    )

    _trimul_gemm_kernel[grid](
        a_ptr=left_h,
        b_ptr=right_h_T,
        c_ptr=out_fp32,
        M=S,
        N=S,
        K=S,
        stride_am=left_h.stride(0),
        stride_ah=left_h.stride(1),
        stride_ai=left_h.stride(2),
        stride_ak=left_h.stride(3),
        stride_ba=right_h_T.stride(0),
        stride_bh=right_h_T.stride(1),
        stride_bk=right_h_T.stride(2),
        stride_bn=right_h_T.stride(3),
        stride_cm=out_fp32.stride(0),
        stride_ch=out_fp32.stride(1),
        stride_ci=out_fp32.stride(2),
        stride_cj=out_fp32.stride(3),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        hidden_dim=hidden_dim,
        num_warps=4,          # reasonable default for FP16
        num_stages=2,         # double‑buffering
    )

    # ------------------------------------------------------------------- #
    # 6. Restore original layout: [B, S, S, H]
    # ------------------------------------------------------------------- #
    out = out_fp32.permute(0, 2, 3, 1).contiguous()   # [B, S, S, H]

    # ------------------------------------------------------------------- #
    # 7. Output LayerNorm over the hidden dimension
    # ------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )   # [B, S, S, H]

    # ------------------------------------------------------------------- #
    # 8. Apply out‑gate and final linear projection back to dim
    # ------------------------------------------------------------------- #
    out = out * out_gate
    out = F.linear(out, to_out_weight)   # [B, S, S, dim]

    return out
