def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config['dim']; hidden_dim = config['hidden_dim']; nomask = config.get('nomask', True)
    eps = 1e-5

    # layernorm
    x_normed = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights['norm.weight'], bias=weights['norm.bias'], eps=eps)

    # cast to half for projection
    x_half = x_normed.half()

    left = torch.nn.functional.linear(x_half, weights['left_proj.weight'].half())  # shape [B,N,N, hidden_dim]
    right = torch.nn.functional.linear(x_half, weights['right_proj.weight'].half())

    # gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_half, weights['left_gate.weight'].half()))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_half, weights['right_gate.weight'].half()))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_half, weights['out_gate.weight'].half()))

    left = left * left_gate
    right = right * right_gate

    if not nomask:
        mask_exp = mask.to(left.dtype).unsqueeze(-1)
        left = left * mask_exp
        right = right * mask_exp

    # rearrange for batched GEMM
    B,N1,N2,_ = left.shape
    assert N1 == N2
    N = N1
    left_reshape = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_reshape = right.permute(0, 3, 1, 2).contiguous()

    # flatten batch and hidden dims into batch for torch.bmm
    left_flat = left_reshape.view(B*hidden_dim, N, N)   # shape [M, N, N]
    right_flat = right_reshape.view(B*hidden_dim, N, N)

    # compute out using torch.bmm, which uses matmul (A @ B^T)
    # Need to transpose right on second dimension
    out_flat = torch.bmm(left_flat, right_flat.transpose(1,2))  # [M,N,N] in fp16

    # convert out_flat to fp32 (or keep half for later to reduce compute) but final layernorm expects float32; we can compute out_gate (half) multiply, then to_out_norm (fp32) then to_out linear.

    out = out_flat.view(B, hidden_dim, N, N).permute(0,2,3,1).contiguous() # [B,N,N,hidden_dim]

    # out is in half. Cast to float32 before layernorm? maybe layernorm/linear weight cast.

    out_f32 = out.float()  # cast

    # layernorm on hidden dimension
    out_normed = torch.nn.functional.layer_norm(out_f32, (hidden_dim,), weight=weights['to_out_norm.weight'], bias=weights['to_out_norm.bias'], eps=eps)

    # multiply out_gate (which is half) but need broadcast to float
    out_gate_f32 = out_gate.float()   # shape [B,N,N,hidden_dim]
    out_normed = out_normed * out_gate_f32

    # final linear projection
    out_final = torch.nn.functional.linear(out_normed, weights['to_out.weight'].float())  # [B,N,N,dim]

    return out_final

def tri_matmul_batch_kernel(a_ptr, b_ptr, c_ptr,
                            N,
                            stride_am, stride_ak, stride_ai,
                            stride_bm, stride_bk, stride_bj,
                            stride_cm, stride_ck, stride_cj,
                            BLOCK_M: tl.constexpr,
                            BLOCK_N: tl.constexpr,
                            BLOCK_K: tl.constexpr):
    pid_b = tl.program_id(0)  # batch index (M_total)
    pid_m = tl.program_id(1)  # tile row i
    pid_n = tl.program_id(2)  # tile col j

    i0 = pid_m * BLOCK_M
    j0 = pid_n * BLOCK_N

    i = i0 + tl.arange(0, BLOCK_M)
    j = j0 + tl.arange(0, BLOCK_N)

    mask_i = i < N
    mask_j = j < N

def custom_kernel(data):
    """
    Triton‑accelerated outgoing TriMul forward pass.

    Steps:
    1. Layer‑norm on input.
    2. Half‑precision linear projections + sigmoid gates.
    3. Optional masking (broadcasted over the hidden dimension).
    4. Batched GEMM (A @ Bᵀ) for every hidden slice using a custom Triton kernel.
    5. Layer‑norm on the hidden dimension, gating, final projection back to `dim`.

    The heavy O(N³·H) contraction is executed in FP16 on Tensor‑cores,
    while the surrounding operations stay in FP32 for numerical stability.
    """
    import torch.nn.functional as F

    input_tensor, mask, weights, config = data

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    # ---------------------------------------------------------
    # 1. Input layer‑norm (FP32)
    # ---------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )   # [B, N, N, dim]

    # ---------------------------------------------------------
    # 2. Cast to FP16 and compute projections / gates
    # ---------------------------------------------------------
    x_h = x.half()   # [B, N, N, dim] in FP16

    # linear projections (bias‑free)
    left = F.linear(x_h, weights["left_proj.weight"].half())          # [B, N, N, H]
    right = F.linear(x_h, weights["right_proj.weight"].half())

    # sigmoid gates (still FP16)
    left_gate = torch.sigmoid(F.linear(x_h, weights["left_gate.weight"].half()))
    right_gate = torch.sigmoid(F.linear(x_h, weights["right_gate.weight"].half()))
    out_gate = torch.sigmoid(F.linear(x_h, weights["out_gate.weight"].half()))

    # apply gates
    left = left * left_gate
    right = right * right_gate

    # ---------------------------------------------------------
    # 3. Optional mask (broadcasted over hidden dim)
    # ---------------------------------------------------------
    if not nomask:
        # mask shape: [B, N, N]  -->  [B, N, N, 1]
        mask_exp = mask.to(left.dtype).unsqueeze(-1)
        left = left * mask_exp
        right = right * mask_exp

    # ---------------------------------------------------------
    # 4. Batched GEMM:   out[b,h,i,j] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    We reinterpret the tensors as (M, N, N) where M = B*H.
    # ---------------------------------------------------------
    B, N, _, _ = left.shape
    # reshape to [B, H, N, N]
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    M_total = B * hidden_dim

    left_flat = left_perm.view(M_total, N, N)    # fp16
    right_flat = right_perm.view(M_total, N, N)  # fp16

    # output buffer (FP32, because the kernel accumulates in FP32)
    out_flat = torch.empty((M_total, N, N), dtype=torch.float32, device=left_flat.device)

    # -----------------------------
    # kernel launch configuration
    # -----------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    stride_am = left_flat.stride(0)
    stride_ai = left_flat.stride(1)
    stride_ak = left_flat.stride(2)

    stride_bm = right_flat.stride(0)
    stride_bj = right_flat.stride(1)   # row (j) stride
    stride_bk = right_flat.stride(2)   # col (k) stride

    stride_cm = out_flat.stride(0)
    stride_ci = out_flat.stride(1)
    stride_cj = out_flat.stride(2)

    grid = (
        M_total,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )
    batched_tri_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        stride_am, stride_ai, stride_ak,
        stride_bm, stride_bj, stride_bk,
        stride_cm, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to [B, N, N, H] (still FP32)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ---------------------------------------------------------
    # 5. Output layer‑norm, gating and final linear (FP32)
    # ---------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate.float()
    output = F.linear(out, weights["to_out.weight"])

    return output

"""
TriMul (outgoing) forward pass with a Triton‑accelerated cubic contraction.

Algorithm
---------
1. Layer‑norm the input tensor on the feature dimension (float32).
2. Cast to FP16 and apply bias‑free linear projections for the left/right
   tensors together with three sigmoid gates.
3. (Optional) mask the projected tensors – the mask is broadcasted over the
   hidden dimension.
4. Reshape to (M, N, N) where M = batch × hidden_dim and run a custom Triton
   kernel that computes, for each (batch, hidden) pair,
          out = left @ rightᵀ
   i.e. a batched GEMM with A,B in FP16 while accumulating in FP32 (Tensor‑core
   friendly).  This is the O(N³·H) work‑horse of the module.
5. Reshape the result back to [B, N, N, hidden_dim] (float32),
   apply a second Layer‑norm, multiply by the output gate and finally a
   bias‑free linear projection back to the original feature dimension.

The heavy O(N³·H) contraction is performed in the Triton kernel; the remaining
operations stay in PyTorch for simplicity and numerical stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_tri_gemm_kernel(
    A,               # left  tensor (M, N, N)  – FP16
    B,               # right tensor (M, N, N)  – FP16 (will be read as Bᵀ)
    C,               # output (M, N, N)       – FP32
    N,               # sequence length
    stride_am, stride_ai, stride_ak,   # strides of A: batch, row(i), col(k)
    stride_bm, stride_bj, stride_bk,   # strides of B: batch, row(j), col(k)
    stride_cm, stride_ci, stride_cj,   # strides of C: batch, row(i), col(j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM C = A @ Bᵀ for each entry in the first dimension.
    All tensors are contiguous, A/B are FP16, C is FP32.
    """
    batch_id = tl.program_id(0)          # index over M = batch * hidden_dim
    pid_m   = tl.program_id(1)          # tile row (i)
    pid_n   = tl.program_id(2)          # tile col (j)

    i0 = pid_m * BLOCK_M
    j0 = pid_n * BLOCK_N
    i = i0 + tl.arange(0, BLOCK_M)
    j = j0 + tl.arange(0, BLOCK_N)

    mask_i = i < N
    mask_j = j < N

    # accumulator in FP32 (Tensor‑core dot returns FP32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over reduction dimension k
    for k0 in range(0, N, BLOCK_K):
        k = k0 + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # A[batch, i, k]   – shape (BLOCK_M, BLOCK_K)
        a_ptr = A + batch_id * stride_am + i[:, None] * stride_ai + k[None, :] * stride_ak
        a = tl.load(
            a_ptr,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float16)

        # Bᵀ -> B[batch, k, j]   – shape (BLOCK_K, BLOCK_N)
        b_ptr = B + batch_id * stride_bm + k[:, None] * stride_bk + j[None, :] * stride_bj
        b = tl.load(
            b_ptr,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        ).to(tl.float16)

        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

    # write the tile back to C
    c_ptr = C + batch_id * stride_cm + i[:, None] * stride_ci + j[None, :] * stride_cj
    tl.store(
        c_ptr,
        acc,
        mask=mask_i[:, None] & mask_j[None, :],
    )


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module using a Triton kernel for the
    cubic contraction.  All tensors are assumed to be on the CUDA device.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor [B, N, N, dim] (float32)
            - mask          [B, N, N]   (any dtype, optional)
            - weights: dict with the model parameters
            - config: dict containing "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    # ----------------------------------------------------------------- #
    # 1. Input LayerNorm (float32)
    # ----------------------------------------------------------------- #
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  (float32)

    # ----------------------------------------------------------------- #
    # 2. Cast to FP16 and compute projections + sigmoid gates
    # ----------------------------------------------------------------- #
    x_h = x.half()   # [B, N, N, dim]  (fp16)

    # bias‑free linear projections
    left  = F.linear(x_h, weights["left_proj.weight"].half())   # [B, N, N, H]
    right = F.linear(x_h, weights["right_proj.weight"].half())

    # sigmoid gates (still fp16)
    left_gate  = torch.sigmoid(F.linear(x_h, weights["left_gate.weight"].half()))
    right_gate = torch.sigmoid(F.linear(x_h, weights["right_gate.weight"].half()))
    out_gate   = torch.sigmoid(F.linear(x_h, weights["out_gate.weight"].half()))

    # apply gates
    left  = left * left_gate
    right = right * right_gate

    # ----------------------------------------------------------------- #
    # 3. Optional mask (broadcast over hidden dim)
    # ----------------------------------------------------------------- #
    if not nomask:
        # mask: [B, N, N] → [B, N, N, 1] and cast to fp16
        mask_exp = mask.to(left.dtype).unsqueeze(-1)
        left  = left * mask_exp
        right = right * mask_exp

    # ----------------------------------------------------------------- #
    # 4. Batched GEMM: out[b, h, i, j] = Σ_k left[b, i, k, h] * right[b, j, k, h]
    # ----------------------------------------------------------------- #
    B, N, _, _ = left.shape
    # bring hidden dimension to the second axis and make tensors contiguous
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    M_total = B * hidden_dim                     # number of independent GEMMs
    left_flat  = left_perm.view(M_total, N, N)   # [M, N, N]  fp16
    right_flat = right_perm.view(M_total, N, N)  # [M, N, N]  fp16
    out_flat   = torch.empty((M_total, N, N), dtype=torch.float32, device=left_flat.device)

    # block sizes – tuned for H100 Tensor‑cores
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    # strides for the kernel (int64 Python scalars)
    stride_am = left_flat.stride(0)
    stride_ai = left_flat.stride(1)
    stride_ak = left_flat.stride(2)

    stride_bm = right_flat.stride(0)
    stride_bj = right_flat.stride(1)   # row (j) stride
    stride_bk = right_flat.stride(2)   # col (k) stride

    stride_cm = out_flat.stride(0)
    stride_ci = out_flat.stride(1)
    stride_cj = out_flat.stride(2)

    # launch grid: (batch*hidden, tile_rows, tile_cols)
    grid = (
        M_total,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )
    batched_tri_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bj,
        stride_bk,
        stride_cm,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to [B, N, N, H] (float32)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ----------------------------------------------------------------- #
    # 5. Output LayerNorm, output‑gate multiplication, final projection
    # ----------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate.float()
    output = F.linear(out, weights["to_out.weight"])

    return output
