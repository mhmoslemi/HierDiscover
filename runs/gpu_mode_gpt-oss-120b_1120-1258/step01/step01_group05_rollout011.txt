@triton.jit
def apply_mask_and_gate_kernel(
    left_ptr, right_ptr,
    mask_ptr,
    left_gate_ptr, right_gate_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_j, stride_left_h,
    stride_right_b, stride_right_i, stride_right_j, stride_right_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_lgate_b, stride_lgate_i, stride_lgate_j, stride_lgate_h,
    stride_rgate_b, stride_rgate_i, stride_rgate_j, stride_rgate_h,
    BLOCK_H: tl.constexpr
):
    pid = tl.program_id(0)
    total = B * N * N
    # each pid corresponds to a specific (b,i,j)
    b = pid // (N * N)
    rem = pid % (N * N)
    i = rem // N
    j = rem % N

    # offsets for H within each element
    offs_h = tl.arange(0, BLOCK_H)
    mask = (offs_h < H)

    # compute pointers for left/right/gate
    left_ptrs = left_ptr + b * stride_left_b + i * stride_left_i + j * stride_left_j + offs_h * stride_left_h
    right_ptrs = right_ptr + b * stride_right_b + i * stride_right_i + j * stride_right_j + offs_h * stride_right_h
    mask_ptrs = mask_ptr + b * stride_mask_b + i * stride_mask_i + j * stride_mask_j
    left_gate_ptrs = left_gate_ptr + b * stride_lgate_b + i * stride_lgate_i + j * stride_lgate_j + offs_h * stride_lgate_h
    right_gate_ptrs = right_gate_ptr + b * stride_rgate_b + i * stride_rgate_i + j * stride_rgate_j + offs_h * stride_rgate_h

    # load
    left = tl.load(left_ptrs, mask=mask, other=tl.float16(0.0))
    right = tl.load(right_ptrs, mask=mask, other=tl.float16(0.0))
    mask_val = tl.load(mask_ptrs)  # scalar broadcasted
    left_gate = tl.load(left_gate_ptrs, mask=mask, other=tl.float16(0.0))
    right_gate = tl.load(right_gate_ptrs, mask=mask, other=tl.float16(0.0))

    # compute
    left = left * mask_val * left_gate
    right = right * mask_val * right_gate

    # store
    tl.store(left_ptrs, left, mask=mask)
    tl.store(right_ptrs, right, mask=mask)

"""
TriMul (outgoing) forward pass – fast implementation.
----------------------------------------------------
1. Layer‑norm the input (float32).
2. Fuse the two linear projections (left/right) together with the
   corresponding gating linear layers (cat‑weight trick) and cast to
   float16 to halve memory traffic.
3. Compute three sigmoid gates (float16) from the normalized input.
4. A tiny Triton kernel fuses the pair‑wise mask (if present) and the
   left‑/right‑gates into a single in‑place element‑wise multiplication.
5. The heavy O(N³·H) operation is a batched matrix multiplication:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h] .
   By reshaping (B,N,N,H) → (B·H,N,N) we can call `torch.bmm`,
   which uses cuBLAS/WMMA on the H100 and is far faster than a custom
   Triton GEMM.
6. Second layer‑norm, output‑gate, and final linear projection back
   to the original channel dimension.
Only the mask‑gate fusion is performed in Triton; the rest of the
pipeline uses highly‑optimised PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _mask_gate_fusion(
    left_ptr, right_ptr,               # [B,N,N,H]   FP16
    mask_ptr,                          # [B,N,N]     FP16 (broadcasted)
    left_gate_ptr, right_gate_ptr,     # [B,N,N,H]   FP16
    B, N, H,
    stride_left_b, stride_left_i, stride_left_j, stride_left_h,
    stride_right_b, stride_right_i, stride_right_j, stride_right_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_lgate_b, stride_lgate_i, stride_lgate_j, stride_lgate_h,
    stride_rgate_b, stride_rgate_i, stride_rgate_j, stride_rgate_h,
    BLOCK_H: tl.constexpr
):
    """
    In‑place fusion:  left  = left  * mask * left_gate
                     right = right * mask * right_gate
    One program instance processes a (b,i,j) tile and a BLOCK_H slice
    along the hidden dimension.
    """
    pid = tl.program_id(0)                     # linear index for (b,i,j)
    total_ij = B * N * N
    if pid >= total_ij:
        return

    # decode (b,i,j) from linear pid
    b = pid // (N * N)
    rem = pid % (N * N)
    i = rem // N
    j = rem % N

    # offsets for the hidden dimension
    offs_h = tl.arange(0, BLOCK_H)
    mask_h = offs_h < H                       # valid lanes

    # pointers to the current element slice
    left_ptrs = (left_ptr
                 + b * stride_left_b
                 + i * stride_left_i
                 + j * stride_left_j
                 + offs_h * stride_left_h)
    right_ptrs = (right_ptr
                  + b * stride_right_b
                  + i * stride_right_i
                  + j * stride_right_j
                  + offs_h * stride_right_h)

    # mask is a scalar, broadcast to BLOCK_H
    mask_scalar = tl.load(mask_ptr
                          + b * stride_mask_b
                          + i * stride_mask_i
                          + j * stride_mask_j)

    left_gate_ptrs = (left_gate_ptr
                      + b * stride_lgate_b
                      + i * stride_lgate_i
                      + j * stride_lgate_j
                      + offs_h * stride_lgate_h)
    right_gate_ptrs = (right_gate_ptr
                       + b * stride_rgate_b
                       + i * stride_rgate_i
                       + j * stride_rgate_j
                       + offs_h * stride_rgate_h)

    # load (masked) tiles
    left  = tl.load(left_ptrs,
                    mask=mask_h,
                    other=tl.float16(0.0))
    right = tl.load(right_ptrs,
                    mask=mask_h,
                    other=tl.float16(0.0))
    left_gate  = tl.load(left_gate_ptrs,
                         mask=mask_h,
                         other=tl.float16(0.0))
    right_gate = tl.load(right_gate_ptrs,
                         mask=mask_h,
                         other=tl.float16(0.0))

    # fuse mask and gates
    left  = left  * mask_scalar * left_gate
    right = right * mask_scalar * right_gate

    # write back
    tl.store(left_ptrs,  left,  mask=mask_h)
    tl.store(right_ptrs, right, mask=mask_h)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, C] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (float32, bool or None)
            - weights_dict : mapping of all weight tensors (float32)
            - config_dict  : {"dim": C, "hidden_dim": H}
    Returns:
        torch.Tensor [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    eps = 1e-5
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # 1) First LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )                       # [B, N, N, C], float32

    # ------------------------------------------------------------------
    # 2) Fuse proj + gate linear layers (cast to FP16 for speed)
    # ------------------------------------------------------------------
    # cast the normalized input once
    x_h = x.to(torch.float16)

    # ----- left side -----
    left_concat_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)                         # (2*H, C)
    left_concat = F.linear(x_h, left_concat_w)  # [B, N, N, 2*H] (FP16)

    left   = left_concat[..., :hidden_dim]      # [B, N, N, H]
    left_g = left_concat[..., hidden_dim:]      # [B, N, N, H] (pre‑sigmoid)
    left_gate = torch.sigmoid(left_g.float()).to(torch.float16)

    # ----- right side -----
    right_concat_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)                         # (2*H, C)
    right_concat = F.linear(x_h, right_concat_w)  # [B, N, N, 2*H] (FP16)

    right   = right_concat[..., :hidden_dim]     # [B, N, N, H]
    right_g = right_concat[..., hidden_dim:]     # [B, N, N, H] (pre‑sigmoid)
    right_gate = torch.sigmoid(right_g.float()).to(torch.float16)

    # ----- out gate (separate) -----
    out_gate = torch.sigmoid(
        F.linear(x, weights["out_gate.weight"])
    ).to(torch.float16)                          # [B, N, N, H]

    # ------------------------------------------------------------------
    # 3) Apply mask (if any) together with the gates – tiny Triton kernel
    # ------------------------------------------------------------------
    if mask_tensor is None:
        # No mask ⇒ just apply the gates with a regular torch multiplication
        left = left * left_gate
        right = right * right_gate
    else:
        # mask is broadcasted over the hidden dimension
        mask = mask_tensor.to(torch.float16)
        # launch kernel (one program per (b,i,j) entry)
        total_ij = B * N * N
        grid = (total_ij,)

        # strides (int64)
        l_str = left.stride()          # (B*N*N*H, N*N*H, N*H, H)
        r_str = right.stride()
        lg_str = left_gate.stride()
        rg_str = right_gate.stride()
        m_str = mask.stride()          # (N*N, N, 1)

        _mask_gate_fusion[grid](
            left,
            right,
            mask,
            left_gate,
            right_gate,
            B,
            N,
            hidden_dim,
            l_str[0],
            l_str[1],
            l_str[2],
            l_str[3],
            r_str[0],
            r_str[1],
            r_str[2],
            r_str[3],
            m_str[0],
            m_str[1],
            m_str[2],
            lg_str[0],
            lg_str[1],
            lg_str[2],
            lg_str[3],
            rg_str[0],
            rg_str[1],
            rg_str[2],
            rg_str[3],
            BLOCK_H=64,                # works for all hidden_dim ≤ 384
        )

    # ------------------------------------------------------------------
    # 4) Batched matrix multiplication (L @ Rᵀ) using torch.bmm
    # ------------------------------------------------------------------
    # reshape to (B*H, N, N)
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # (B*H, N, N) @ (B*H, N, N)ᵀ  → (B*H, N, N)  (FP16)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # back to (B, N, N, H) and up‑cast to FP32 for the remaining ops
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous().float()

    # ------------------------------------------------------------------
    # 5) Second LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )  # [B, N, N, H] (float32)

    # ------------------------------------------------------------------
    # 6) Output gate
    # ------------------------------------------------------------------
    out = out * out_gate.float()

    # ------------------------------------------------------------------
    # 7) Final projection back to original channel dimension
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])  # [B, N, N, C] (float32)

    return out
