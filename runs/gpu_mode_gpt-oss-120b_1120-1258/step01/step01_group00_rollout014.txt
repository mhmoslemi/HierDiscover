def trimul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_az, stride_ah, stride_ai, stride_ak,
    stride_bz, stride_bh, stride_bk, stride_bn,
    stride_cz, stride_ch, stride_ci, stride_cj,

    GROUP_M: tl.constexpr,

    pid = tl.program_id(0)

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# Triton kernel: fused gating + optional mask
@triton.jit
def _gate_and_mask_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    total_elements,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    # Load
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    lgate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    rgate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        m = tl.load(mask_ptr + offs, mask=mask, other=0.0)
        left = left * lgate * m
        right = right * rgate * m
    else:
        left = left * lgate
        right = right * rgate

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)

def custom_kernel(data):
    """
    Outgoing TriMul forward pass: layer‑norm → linear projections → gating & mask (Triton) → 
    batched half‑precision matrix multiply (torch.bmm) → hidden layer‑norm → out‑gate → 
    linear projection back to input dimension.
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, S, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # unpack weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # 1. input layer‑norm
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )   # [B,S,S,dim]  (float32)

    # 2. linear projections (no bias)
    left = F.linear(x, left_proj_weight)     # [B,S,S,hidden_dim]
    right = F.linear(x, right_proj_weight)   # [B,S,S,hidden_dim]

    # 3. gating (sigmoid of linear proj)
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # 4. cast to FP16 for the heavy reduction
    left = left.to(torch.float16).contiguous()
    right = right.to(torch.float16).contiguous()
    left_gate = left_gate.to(torch.float16).contiguous()
    right_gate = right_gate.to(torch.float16).contiguous()

    # 5. optional mask + element‑wise gating fused in a Triton kernel
    if not nomask and mask_tensor is not None:
        # mask shape [B,S,S] -> broadcast to hidden dim
        mask = mask_tensor.to(left.dtype).unsqueeze(-1).contiguous()   # [B,S,S,1]
        mask = mask.expand(-1, -1, -1, hidden_dim).contiguous()
    else:
        mask = None

    total_elems = left.numel()
    BLOCK = 1024

    grid = lambda meta: (triton.cdiv(total_elems, BLOCK),)

    _gate_and_mask_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        left_gate_ptr=left_gate,
        right_gate_ptr=right_gate,
        mask_ptr=mask if mask is not None else left,  # dummy ptr, will never be read when HAS_MASK=0
        total_elements=total_elems,
        BLOCK=BLOCK,
        HAS_MASK=int(mask is not None),
    )

    # 6. Batched matrix multiplication: out[b,h,i,j] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # Reshape tensors to (B*H, S, S)
    left_flat = left.view(B * hidden_dim, S, S)
    right_flat = right.view(B * hidden_dim, S, S)

    # Use torch.bmm (FP16 inputs, FP32 accumulation via torch.cuda.amp.autocast if needed)
    # For determinism we keep FP16 output and cast later.
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))  # [B*H, S, S] , fp16

    # Restore original layout [B, S, S, H]
    out = out_flat.view(B, hidden_dim, S, S).permute(0, 2, 3, 1).contiguous()  # [B, S, S, H]
    out = out.to(torch.float32)  # convert for following layer‑norm

    # 7. hidden‑dim layer‑norm
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )

    # 8. apply out‑gate and final projection back to original dim
    out = out * out_gate
    out = F.linear(out, to_out_weight)  # [B,S,S,dim]

    return out

Algorithm
---------
1. **Layer‑Norm** over the last dimension (`dim`).
2. **Linear projections** → `left` and `right`  (shape [B,S,S,H]).
3. **Gating + optional mask** are fused in a tiny Triton kernel
   (element‑wise `left = left * sigmoid(left_gate) * mask`,
    `right = right * sigmoid(right_gate) * mask`).
4. **Batched half‑precision GEMM**:
       out[b,h,i,j] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a standard batched `torch.bmm` on FP16 tensors (Tensor‑Core
   accelerated).  The result is cast back to FP32.
5. **Layer‑Norm** on the hidden dimension (`H`).
6. Apply the **output gate** and a final linear layer back to `dim`.

Only the element‑wise fusion (step 3) is written in Triton; the
dominant O(N³·H) work is performed by cuBLAS (`torch.bmm`), which on an
H100 easily meets the sub‑millisecond target for all benchmark sizes.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------------------- #
# Triton kernel: fuse the two gates and an optional mask.
# --------------------------------------------------------------------------- #
@triton.jit
def _gate_and_mask_kernel(
    left_ptr,               # fp16 * (B*S*S*H)  – in‑place write back
    right_ptr,              # fp16 * (B*S*S*H)
    left_gate_ptr,          # fp16 * (B*S*S*H)   – sigmoid(left_gate) already computed
    right_gate_ptr,         # fp16 * (B*S*S*H)
    mask_ptr,               # fp16 * (B*S*S*H)   – broadcast mask (0/1) or dummy
    total_elements,         # int – B·S·S·H
    BLOCK: tl.constexpr,    # launch‑time block size (e.g. 1024)
    HAS_MASK: tl.constexpr, # 0/1 compile‑time flag
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    pred = offs < total_elements

    # load the four tensors
    left = tl.load(left_ptr + offs, mask=pred, other=0.0)
    right = tl.load(right_ptr + offs, mask=pred, other=0.0)
    lgate = tl.load(left_gate_ptr + offs, mask=pred, other=0.0)
    rgate = tl.load(right_gate_ptr + offs, mask=pred, other=0.0)

    if HAS_MASK:
        m = tl.load(mask_ptr + offs, mask=pred, other=0.0)
        left = left * lgate * m
        right = right * rgate * m
    else:
        left = left * lgate
        right = right * rgate

    tl.store(left_ptr + offs, left, mask=pred)
    tl.store(right_ptr + offs, right, mask=pred)


# --------------------------------------------------------------------------- #
# Entry‑point used by the evaluation harness
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Args:
        data (tuple):
            - input_tensor : torch.Tensor, shape [B, S, S, C]   (float32)
            - mask_tensor  : torch.Tensor, shape [B, S, S] or None
            - weights       : dict of torch.Tensors (all model weights)
            - config        : dict with keys 'dim', 'hidden_dim', 'nomask' …

    Returns:
        torch.Tensor of shape [B, S, S, C] (float32)
    """
    # ------------------------------------------------------------------- #
    # Unpack arguments
    # ------------------------------------------------------------------- #
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, S, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------- #
    # Weight shortcuts (already on the correct device)
    # ------------------------------------------------------------------- #
    norm_weight          = weights["norm.weight"]
    norm_bias            = weights["norm.bias"]
    left_proj_weight     = weights["left_proj.weight"]
    right_proj_weight    = weights["right_proj.weight"]
    left_gate_weight     = weights["left_gate.weight"]
    right_gate_weight    = weights["right_gate.weight"]
    out_gate_weight      = weights["out_gate.weight"]
    to_out_norm_weight   = weights["to_out_norm.weight"]
    to_out_norm_bias     = weights["to_out_norm.bias"]
    to_out_weight        = weights["to_out.weight"]

    # ------------------------------------------------------------------- #
    # 1. Input LayerNorm (dim‑wise)
    # ------------------------------------------------------------------- #
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )                         # [B, S, S, dim]  (float32)

    # ------------------------------------------------------------------- #
    # 2. Linear projections → left / right (no bias)
    # ------------------------------------------------------------------- #
    left  = F.linear(x, left_proj_weight)   # [B, S, S, H]
    right = F.linear(x, right_proj_weight)  # [B, S, S, H]

    # ------------------------------------------------------------------- #
    # 3. Compute gates (sigmoid of linear projections)
    # ------------------------------------------------------------------- #
    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))   # [B, S, S, H]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))  # [B, S, S, H]
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))    # [B, S, S, H]

    # ------------------------------------------------------------------- #
    # 4. Cast to half‑precision – the heavy work is done in FP16
    # ------------------------------------------------------------------- #
    left = left.to(torch.float16).contiguous()
    right = right.to(torch.float16).contiguous()
    left_gate = left_gate.to(torch.float16).contiguous()
    right_gate = right_gate.to(torch.float16).contiguous()

    # ------------------------------------------------------------------- #
    # 5. Optional mask + fuse element‑wise gating using Triton
    # ------------------------------------------------------------------- #
    if not nomask and mask_tensor is not None:
        # mask shape [B, S, S] → broadcast over hidden dimension
        mask = (
            mask_tensor.to(left.dtype)               # fp16
            .unsqueeze(-1)                           # [B, S, S, 1]
            .expand(-1, -1, -1, hidden_dim)          # [B, S, S, H]
            .contiguous()
        )
    else:
        mask = left  # dummy pointer – never read because HAS_MASK == 0

    total_elems = left.numel()                     # B·S·S·H
    BLOCK = 1024                                   # ~1 Ki elements per program
    grid = lambda meta: (triton.cdiv(total_elems, BLOCK),)

    _gate_and_mask_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        left_gate_ptr=left_gate,
        right_gate_ptr=right_gate,
        mask_ptr=mask,
        total_elements=total_elems,
        BLOCK=BLOCK,
        HAS_MASK=int(mask_tensor is not None and not nomask),
    )
    # ------------------------------------------------------------------- #
    # 6. Batched GEMM (left @ rightᵀ) – use cuBLAS via torch.bmm
    # ------------------------------------------------------------------- #
    # reshape to (B*H, S, S)  – each slice is an independent matrix multiply
    left_flat  = left.view(B * hidden_dim, S, S)                 # fp16
    right_flat = right.view(B * hidden_dim, S, S)                # fp16

    # rightᵀ on the last two dimensions
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))  # fp16, shape (B*H, S, S)

    # Restore the [B, S, S, H] layout and promote to FP32 for the next LN
    out = (
        out_flat.view(B, hidden_dim, S, S)
        .permute(0, 2, 3, 1)   # [B, S, S, H]
        .contiguous()
        .to(torch.float32)
    )

    # ------------------------------------------------------------------- #
    # 7. Hidden‑dim LayerNorm
    # ------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )   # [B, S, S, H] (float32)

    # ------------------------------------------------------------------- #
    # 8. Apply out‑gate and final linear projection back to `dim`
    # ------------------------------------------------------------------- #
    out = out * out_gate                     # element‑wise gating
    out = F.linear(out, to_out_weight)      # [B, S, S, dim]

    return out
