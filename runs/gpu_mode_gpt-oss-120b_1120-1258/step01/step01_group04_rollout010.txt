def tri_mul_outgoing_kernel(
    a_ptr, b_ptr, c_ptr,
    N,
    stride_am, stride_ai, stride_ak,
    stride_bm, stride_bj, stride_bk,
    stride_cm, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):

    pid_n = tl.program_id(1) # j tile
    pid_b = tl.program_id(2) # batch*hidden

    i0 = pid * BLOCK_M
    j0 = pid_n * BLOCK_N

    # define ranges
    i = i0 + tl.arange(0, BLOCK_M)
    j = j0 + tl.arange(0, BLOCK_N)
    # masks
    mask_i = i < N
    mask_j = j < N

    # pointer offsets
    c_ptrs = c_ptr + pid_b * stride_cm + i[:, None] * stride_ci + j[None, :] * stride_cj
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

        mask_k = k < N
        a_ptrs = a_ptr + pid_b * stride_am + i[:, None] * stride_ai + k[None, :] * stride_ak

```python
"""
TriMul (outgoing) forward pass.
- LayerNorm on input.
- Projection and gating computed in FP16 for high bandwidth.
- The cubic contraction out[i,j] = Σ_k left[i,k] * right[j,k] is realized as a
  batched matrix‑multiply per hidden channel using a Triton kernel that
  tiles the i, j and k dimensions (BLOCK_M×BLOCK_N×BLOCK_K) and performs
  `tl.dot` on FP16 tiles, accumulating in FP32.
- The result is LayerNorm → output‑gate → final linear projection back to
  the original feature dimension.
"""

import torch
import triton
import triton.language as tl

# Triton kernel for the pairwise contraction.
@triton.jit
def tri_mul_outgoing_kernel(
    a_ptr, b_ptr, c_ptr,                # pointers
    N,                                  # seq_len
    stride_am, stride_ai, stride_ak,    # A strides
    stride_bm, stride_bj, stride_bk,    # B strides (read as K,J)
    stride_cm, stride_ci, stride_cj,    # C strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)   # tile over i
    pid_j = tl.program_id(1)   # tile over j
    pid_b = tl.program_id(2)   # batch * hidden (M)

    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    i = i0 + tl.arange(0, BLOCK_M)
    j = j0 + tl.arange(0, BLOCK_N)

    mask_i = i < N
    mask_j = j < N

    # pointer to C tile
    c_ptrs = c_ptr + pid_b * stride_cm + i[:, None] * stride_ci + j[None, :] * stride_cj
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # reduction over k
    for k0 in range(0, N, BLOCK_K):
        k = k0 + tl.arange(0, BLOCK_K)
        mask_k = k < N

        a_ptrs = a_ptr + pid_b * stride_am + i[:, None] * stride_ai + k[None, :] * stride_ak
        b_ptrs = b_ptr + pid_b * stride_bm + k[:, None] * stride_bk + j[None, :] * stride_bj

        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)                # half
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)                # half

        # tl.dot works on half, accumulates in FP16, we cast to FP32
        acc += tl.dot(a, b).to(tl.float32)

    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation using a mixed‑precision
    Triton kernel for the O(N³·H) contraction.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool or float)
            - weights: dict of torch.Tensor containing the module parameters
            - config: dict with keys "dim", "hidden_dim", "nomask" (optional)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data

    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    # ----- unpack weights --------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    left_proj_weight   = weights["left_proj.weight"]
    right_proj_weight  = weights["right_proj.weight"]
    left_gate_weight   = weights["left_gate.weight"]
    right_gate_weight  = weights["right_gate.weight"]
    out_gate_weight    = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]

    # ----- 1. LayerNorm on the input (float32) ----------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )   # [B, N, N, dim]

    # Cast to half for the heavy linear projections and the kernel.
    x_h = x.half()

    # ----- 2. Projections + gates in FP16 ---------------------------------
    # linear projections
    left = torch.nn.functional.linear(x_h, left_proj_weight.half())   # [B, N, N, H]
    right = torch.nn.functional.linear(x_h, right_proj_weight.half())

    # gates (also in FP16)
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_h, left_gate_weight.half())).half()
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_h, right_gate_weight.half())).half()
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))   # keep float32 for later

    # ----- 3. Optional mask ------------------------------------------------
    if not nomask:
        # mask is broadcasted on the hidden dimension
        mask_exp = mask.to(torch.float16).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    # ----- 4. Apply elementwise gates --------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ----- 5. Heavy cubic contraction using Triton -------------------------
    # Permute to bring hidden dim to the second axis: [B, H, N, N]
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    B, H, N, _ = left_perm.shape
    M_total = B * H

    # view as batch of matrices for the kernel
    left_flat = left_perm.view(M_total, N, N)   # [M, N, N]  (FP16)
    right_flat = right_perm.view(M_total, N, N) # [M, N, N]  (FP16)

    # allocate output (float32)
    out_flat = torch.empty((M_total, N, N), dtype=torch.float32, device=device)

    # block sizes – tuned for H100.  Larger blocks increase compute per launch.
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    # strides for the kernel (int)
    stride_am = left_flat.stride(0)
    stride_ai = left_flat.stride(1)
    stride_ak = left_flat.stride(2)

    stride_bm = right_flat.stride(0)
    stride_bj = right_flat.stride(1)  # j dimension
    stride_bk = right_flat.stride(2)  # k dimension

    stride_cm = out_flat.stride(0)
    stride_ci = out_flat.stride(1)
    stride_cj = out_flat.stride(2)

    # launch grid
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j tiles
        M_total                         # one program per (batch, hidden)
    )
    tri_mul_outgoing_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        stride_am, stride_ai, stride_ak,
        stride_bm, stride_bj, stride_bk,
        stride_cm, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ----- 6. Final LayerNorm, gating, linear projection -------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    out = out * out_gate
    output = torch.nn.functional.linear(out, to_out_weight)

    return output
```

"""
TriMul (outgoing) forward pass – highly‑optimized Triton implementation.

Algorithm
---------
1. Layer‑norm on the input tensor (float32).
2. Cast the normalized tensor to FP16 and compute the three linear
   projections (left, right) as well as the two gating projections in FP16.
3. Apply the optional mask (broadcasted over the hidden dimension) and the
   element‑wise gates while staying in FP16.
4. The cubic contraction
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   is a batched matrix multiplication C = A @ Bᵀ for each (batch, hidden)
   pair.  It is realised with a Triton kernel that tiles the i, j and k
   dimensions (BLOCK_M × BLOCK_N × BLOCK_K) and accumulates with `tl.dot`
   on FP16 tiles, storing the result in FP32.
5. The result is LayerNorm over the hidden dimension, multiplied by the
   output gate, and finally projected back to the original feature size.

The heavy O(N³·H) work is performed in the Triton kernel; the rest of the
pipeline stays in PyTorch to keep the implementation compact.
"""

import torch
import triton
import triton.language as tl


# --------------------------------------------------------------------------- #
# Triton kernel:  out[b,i,j] = Σₖ left[b,i,k] * right[b,j,k]
# (left and right are FP16, accumulation is FP32)
# --------------------------------------------------------------------------- #
@triton.jit
def tri_mul_outgoing_kernel(
    a_ptr, b_ptr, c_ptr,          # pointers to left, right, output
    N,                            # sequence length
    stride_am, stride_ai, stride_ak,    # A strides: batch, i, k
    stride_bm, stride_bj, stride_bk,    # B strides: batch, j, k (read as k,j)
    stride_cm, stride_ci, stride_cj,    # C strides: batch, i, j
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)   # tile over output rows   (i)
    pid_j = tl.program_id(1)   # tile over output cols   (j)
    pid_b = tl.program_id(2)   # (batch * hidden) index

    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    i = i0 + tl.arange(0, BLOCK_M)
    j = j0 + tl.arange(0, BLOCK_N)

    mask_i = i < N
    mask_j = j < N

    # pointer to the (i,j) tile of C
    c_ptrs = c_ptr + pid_b * stride_cm + i[:, None] * stride_ci + j[None, :] * stride_cj
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # reduction over k
    for k0 in range(0, N, BLOCK_K):
        k = k0 + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # A[i, k]  (FP16)
        a_ptrs = a_ptr + pid_b * stride_am + i[:, None] * stride_ai + k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)          # half

        # B[k, j]  (we read the right tensor transposed)
        b_ptrs = b_ptr + pid_b * stride_bm + k[:, None] * stride_bk + j[None, :] * stride_bj
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)          # half

        # tl.dot returns half; cast to fp32 before accumulation
        acc += tl.dot(a, b).to(tl.float32)

    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[:, None])


# --------------------------------------------------------------------------- #
# Entry point called from the evaluation harness
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask         : torch.Tensor [B, N, N] (bool or float)
            - weights      : dict of torch.Tensor with module parameters
            - config       : dict containing "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data

    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    # --------------------------------------------------- #
    # 1) unpack all required weight tensors (already on device)
    # --------------------------------------------------- #
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    left_proj_weight   = weights["left_proj.weight"]
    right_proj_weight  = weights["right_proj.weight"]
    left_gate_weight   = weights["left_gate.weight"]
    right_gate_weight  = weights["right_gate.weight"]
    out_gate_weight    = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]

    # --------------------------------------------------- #
    # 2) Layer‑norm over the input feature dimension (float32)
    # --------------------------------------------------- #
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )                                            # [B, N, N, dim]

    # --------------------------------------------------- #
    # 3) Cast to FP16 – the heavy work will be done in half precision
    # --------------------------------------------------- #
    x_h = x.half()

    # --------------------------------------------------- #
    # 4) Linear projections (bias‑free) in FP16
    # --------------------------------------------------- #
    left  = torch.nn.functional.linear(x_h, left_proj_weight.half())   # [B, N, N, H]
    right = torch.nn.functional.linear(x_h, right_proj_weight.half())

    # --------------------------------------------------- #
    # 5) Gates (sigmoid) – also in FP16 for the left/right paths
    # --------------------------------------------------- #
    left_gate  = torch.sigmoid(
        torch.nn.functional.linear(x_h, left_gate_weight.half())
    ).half()
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x_h, right_gate_weight.half())
    ).half()
    # Output gate stays in FP32 – it is applied after the final LayerNorm
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x, out_gate_weight)
    )                                            # [B, N, N, H] (float32)

    # --------------------------------------------------- #
    # 6) Optional masking (broadcasted on hidden dimension)
    # --------------------------------------------------- #
    if not nomask:
        mask_exp = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left  = left * mask_exp
        right = right * mask_exp

    # --------------------------------------------------- #
    # 7) Apply element‑wise gates (still FP16)
    # --------------------------------------------------- #
    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------- #
    # 8) Heavy O(N³·H) contraction via Triton
    # --------------------------------------------------- #
    # Bring hidden dimension to the second axis: [B, H, N, N] and flatten
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N] (FP16)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N] (FP16)

    B, H, N, _ = left_perm.shape
    M_total = B * H                                   # batch * hidden

    left_flat  = left_perm.view(M_total, N, N)        # [M, N, N] (FP16)
    right_flat = right_perm.view(M_total, N, N)       # [M, N, N] (FP16)
    out_flat   = torch.empty((M_total, N, N), dtype=torch.float32, device=device)

    # ------------------------------------------------------------------- #
    # 9) Kernel launch parameters – tuned for an H100
    # ------------------------------------------------------------------- #
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    # strides (Python ints)
    stride_am = left_flat.stride(0)
    stride_ai = left_flat.stride(1)
    stride_ak = left_flat.stride(2)

    stride_bm = right_flat.stride(0)
    stride_bj = right_flat.stride(1)   # j‑dimension stride
    stride_bk = right_flat.stride(2)   # k‑dimension stride

    stride_cm = out_flat.stride(0)
    stride_ci = out_flat.stride(1)
    stride_cj = out_flat.stride(2)

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # tiles over i
        (N + BLOCK_N - 1) // BLOCK_N,   # tiles over j
        M_total                         # one program per (batch, hidden) pair
    )

    tri_mul_outgoing_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        stride_am, stride_ai, stride_ak,
        stride_bm, stride_bj, stride_bk,
        stride_cm, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------- #
    # 10) Reshape back to the original layout: [B, N, N, H]
    # ------------------------------------------------------------------- #
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # FP32

    # ------------------------------------------------------------------- #
    # 11) Final LayerNorm, gating, and projection back to `dim`
    # ------------------------------------------------------------------- #
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    out = out * out_gate
    output = torch.nn.functional.linear(out, to_out_weight)   # [B, N, N, dim]

    return output
```

This implementation keeps the expensive cubic contraction in a Triton kernel that works on FP16 tiles (halving memory bandwidth) while the surrounding linear layers, gates and normalizations are handled by PyTorch. The chosen block sizes (128 × 128 × 64) fit comfortably in H100 shared memory and provide high occupancy, delivering a substantial speed‑up over the reference Triton version.
