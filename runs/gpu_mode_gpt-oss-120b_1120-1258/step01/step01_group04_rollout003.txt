                        M, N, K,

TriMul (outgoing) – fast Triton‑based implementation
====================================================

The module follows the AlphaFold‑3 “outgoing” TriMul formulation:

1. **Layer‑norm** over the input feature dimension *dim*.
2. **Bias‑free linear projections** (left/right) and three **sigmoid gates**.
   All linear layers are evaluated in FP16 to halve memory traffic.
3. **Optional mask** applied to the projected tensors.
4. **Gating**: left = left_proj * sigmoid(left_gate), right analogous.
5. **Batched GEMM**: for every (batch, hidden) pair we compute
       out[m,i,j] = Σₖ left[m,i,k] · right[m,j,k]
   i.e. a matrix multiplication A @ Bᵀ with A,B ∈ ℝ^{N×N}.
   The heavy O(N³·H) contraction is fused into a single Triton kernel.
   The kernel tiles the i‑ (rows), j‑ (cols) and k‑ (reduction) dimensions
   with configurable block sizes (default 64×64×32) and accumulates in FP32
   while loading the operands as FP16.
6. **Layer‑norm** over the hidden dimension, element‑wise multiplication with
   the output gate, and a final bias‑free linear projection back to the original
   feature size *dim*.

Only the cubic contraction is executed on the GPU via Triton; all other
operations are performed with PyTorch (still in FP16 where beneficial) to keep the
implementation simple yet fast.  The function returns a float‑32 tensor of shape
[bs, seq_len, seq_len, dim].
"""

import math
import torch
import triton
import triton.language as tl

# --------------------------------------------------------------------------- #
# Triton kernel: batched GEMM (C = A @ Bᵀ) for the TriMul contraction.
# --------------------------------------------------------------------------- #
@triton.jit
def batched_gemm_kernel(
    a_ptr,               # left  [M, N, K]  (M = B*H, K = N)
    b_ptr,               # right [M, N, K]  (we read it as [M, K, N])
    c_ptr,               # out   [M, N, N]
    N,                   # sequence length (common N for A and B)
    stride_am, stride_ai, stride_ak,   # A strides: batch, i, k
    stride_bm, stride_bj, stride_bk,   # B strides: batch, j, k   (read as k,j)
    stride_cm, stride_ci, stride_cj,   # C strides: batch, i, j
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute C[m,i,j] = Σₖ A[m,i,k] * B[m,j,k]  for all m ∈ [0,M).
    A and B are stored in FP16, accumulation is in FP32.
    """
    pid_m = tl.program_id(0)   # tile along i (rows)
    pid_n = tl.program_id(1)   # tile along j (cols)
    pid_b = tl.program_id(2)   # batch × hidden index (M)

    # --------------------------------------------------------------
    # Tile start indices
    # --------------------------------------------------------------
    i0 = pid_m * BLOCK_M
    j0 = pid_n * BLOCK_N

    i = i0 + tl.arange(0, BLOCK_M)
    j = j0 + tl.arange(0, BLOCK_N)

    mask_i = i < N
    mask_j = j < N

    # --------------------------------------------------------------
    # Pointers to the output tile
    # --------------------------------------------------------------
    c_ptrs = c_ptr + pid_b * stride_cm + i[:, None] * stride_ci + j[None, :] * stride_cj
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Main reduction over k
    # --------------------------------------------------------------
    for k0 in range(0, N, BLOCK_K):
        k = k0 + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # ---- load A[i, k] -------------------------------------------------
        a_ptrs = a_ptr + pid_b * stride_am + i[:, None] * stride_ai + k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0,
                    dtype=tl.float16).to(tl.float32)

        # ---- load B[k, j]  (B is stored as B[j, k] -> we read transposed) -
        b_ptrs = b_ptr + pid_b * stride_bm + k[:, None] * stride_bk + j[None, :] * stride_bj
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0,
                    dtype=tl.float16).to(tl.float32)

        # ---- accumulate ----------------------------------------------------
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

    # --------------------------------------------------------------
    # Write back the tile
    # --------------------------------------------------------------
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


# --------------------------------------------------------------------------- #
# Public entry point – the forward pass of the outgoing TriMul module.
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator using a fused Triton GEMM.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask         : torch.Tensor of shape [B, N, N] (bool/float) or None
        - weights      : dict of torch.Tensors (all on the same CUDA device)
        - config       : dict with keys "dim", "hidden_dim", optionally "nomask"

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32) – the TriMul output.
    """
    # --------------------------------------------------------------------- #
    # Unpack inputs
    # --------------------------------------------------------------------- #
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    # --------------------------------------------------------------------- #
    # 1. LayerNorm over the input feature dimension (dim)
    # --------------------------------------------------------------------- #
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim] – FP32

    # --------------------------------------------------------------------- #
    # 2. Cast to FP16 for all subsequent linear + gating operations
    # --------------------------------------------------------------------- #
    x_h = x.to(torch.float16)

    # Helper to cast weights once
    def _to_fp16(t): return t.to(torch.float16)

    # Linear projections (bias‑free)
    left_proj  = torch.nn.functional.linear(x_h, _to_fp16(weights["left_proj.weight"]))   # [B,N,N,H]
    right_proj = torch.nn.functional.linear(x_h, _to_fp16(weights["right_proj.weight"]))

    # Gates (sigmoid) – also FP16
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x_h, _to_fp16(weights["left_gate.weight"])))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_h, _to_fp16(weights["right_gate.weight"])))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x_h, _to_fp16(weights["out_gate.weight"])))  # [B,N,N,H]

    # Apply gating
    left  = left_proj * left_gate
    right = right_proj * right_gate

    # --------------------------------------------------------------------- #
    # 3. Optional mask (broadcasted on the hidden dimension)
    # --------------------------------------------------------------------- #
    if not nomask and mask is not None:
        # mask may be bool, uint8 or float – cast to the same dtype as left/right
        mask_exp = mask.to(left.dtype).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # --------------------------------------------------------------------- #
    # 4. Rearrange tensors so that each (batch, hidden) pair becomes an
    #    independent N×N matrix multiplication.
    # --------------------------------------------------------------------- #
    #   left_perm / right_perm : [B, H, N, N]
    left_perm  = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    B, H, N, _ = left_perm.shape
    M_total = B * H                               # total number of independent GEMMs

    left_flat  = left_perm.view(M_total, N, N)    # [M, N, N] (FP16)
    right_flat = right_perm.view(M_total, N, N)   # [M, N, N] (FP16)

    # Output buffer – keep FP16 until after the GEMM
    out_flat = torch.empty_like(left_flat)

    # --------------------------------------------------------------------- #
    # 5. Triton kernel launch (batched GEMM)
    # --------------------------------------------------------------------- #
    # Choose block sizes that give high occupancy on H100 while staying within
    # shared‑memory limits.  64×64×32 is a good default for N up to 1024.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # Strides for the three tensors (Python ints)
    stride_am = left_flat.stride(0)   # batch (M) stride
    stride_ai = left_flat.stride(1)   # i   stride
    stride_ak = left_flat.stride(2)   # k   stride

    stride_bm = right_flat.stride(0)  # batch (M) stride
    stride_bj = right_flat.stride(1)  # j   stride  (second dim after permute)
    stride_bk = right_flat.stride(2)  # k   stride  (third dim)

    stride_cm = out_flat.stride(0)
    stride_ci = out_flat.stride(1)
    stride_cj = out_flat.stride(2)

    # Grid: tiles along i, tiles along j, one program per (batch, hidden) pair
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        M_total                         # independent GEMM per hidden channel
    )

    batched_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        stride_am, stride_ai, stride_ak,
        stride_bm, stride_bj, stride_bk,
        stride_cm, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------------- #
    # 6. Reshape back to the original 4‑D layout
    # --------------------------------------------------------------------- #
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]
    out = out.to(torch.float32)   # cast back to FP32 for the final layers

    # --------------------------------------------------------------------- #
    # 7. LayerNorm over the hidden dimension, gating, final projection
    # --------------------------------------------------------------------- #
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    # Apply the output gate (originally computed in FP16)
    out = out * out_gate.to(torch.float32)

    # Final linear projection back to `dim`
    output = torch.nn.functional.linear(out, weights["to_out.weight"])

    return output
