"""
TriMul (outgoing) – fast forward implementation

Algorithm
---------
1. Layer‑norm the 4‑D input  x ∈ ℝ^{B×N×N×C}.
2. Fuse the five linear layers of the reference implementation:
      * One matmul yields left_proj, right_proj and out_gate.
      * A second matmul yields left_gate and right_gate.
   This cuts the number of full‑dimension matrix‑vector products from 5
   to 2, dramatically reducing memory traffic (the input tensor is read only
   twice).
3. Apply the optional pair‑wise mask (broadcast over the hidden dimension)
   and the sigmoid gates.  The result are the tensors *left* and *right*
   of shape [B,N,N,H] (H = hidden_dim).
4. Cast *left* and *right* to float16 and launch a Triton kernel that
   computes the cubic “TriMul” contraction

        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

   This is a batch of GEMM operations (B·H independent matrix‑multiplications
   of size N×N).  The kernel is a classic tiled GEMM that uses `tl.dot`
   (Tensor‑Core friendly) with fp32 accumulation while loading fp16 data.
   Block sizes are selected at runtime according to the sequence length.
5. Reshape the result back to [B,N,N,H] and apply a second LayerNorm.
6. Multiply by the output gate and perform the final linear projection
   back to the original channel dimension C.
7. Return the float‑32 tensor of shape [B,N,N,C].

Only the heavy N³ contraction is performed in Triton; all other
operations use highly‑optimised PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel for the batched GEMM that implements the TriMul contraction.
# Computes:  C[b,i,j] = Σ_k A[b,i,k] * B[b,j,k]     (note the transpose on B)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_gemm_kernel(
    A_ptr,                     # *ptr to left tensor  (batch, M, K)  fp16
    B_ptr,                     # *ptr to right tensor (batch, K, N) fp16 (originally (batch, N, K))
    C_ptr,                     # *ptr to output tensor (batch, M, N) fp32
    BATCH, M, N, K,            # matrix dimensions (int)
    stride_Ab, stride_Am, stride_Ak,   # strides for A
    stride_Bb, stride_Bk, stride_Bn,   # strides for B (B_k over K, B_n over N after transpose)
    stride_Cb, stride_Cm, stride_Cn,   # strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(2)        # batch index (B*H)
    pid_m = tl.program_id(0)            # block row index
    pid_n = tl.program_id(1)            # block col index

    # ----------------- block offsets -----------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----------------- loop over K dimension -----------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # pointers for current tile of A and B
        a_ptrs = (A_ptr + pid_batch * stride_Ab
                  + offs_m[:, None] * stride_Am
                  + offs_k[None, :] * stride_Ak)
        b_ptrs = (B_ptr + pid_batch * stride_Bb
                  + offs_k[:, None] * stride_Bk
                  + offs_n[None, :] * stride_Bn)

        # load tiles (masked)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)           # fp16 – will be promoted to fp32 in dot
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # dot‑product accumulation (uses Tensor‑cores on H100)
        acc += tl.dot(a, b)               # result fp32

    # ----------------- write back -----------------
    c_ptrs = (C_ptr + pid_batch * stride_Cb
              + offs_m[:, None] * stride_Cm
              + offs_n[None, :] * stride_Cn)

    tl.store(c_ptrs, acc,
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point expected by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the "outgoing" TriMul module (no gradients).

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, C] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (bool/float) or None
        - weights_dict : mapping from string names to torch.Tensor (float32)
        - config_dict  : must contain keys "dim" (C) and "hidden_dim" (H)

    Returns
    -------
    torch.Tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]               # input channel dimension C
    hidden_dim = config["hidden_dim"] # hidden dimension H
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) First LayerNorm over the channel dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # shape [B, N, N, C]

    # ------------------------------------------------------------------
    # 2) Fuse the three projection linear layers (left_proj, right_proj,
    #    out_gate) into a single matmul.
    # ------------------------------------------------------------------
    # Stack weights: shape (3*H, C)
    proj_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    )  # (3*H, C)

    proj_out = F.linear(x, proj_weight)  # [B, N, N, 3*H]
    left_proj_raw, right_proj_raw, out_gate_raw = torch.split(
        proj_out, hidden_dim, dim=-1
    )

    # ------------------------------------------------------------------
    # 3) Fuse the two gate linear layers (left_gate, right_gate)
    # ------------------------------------------------------------------
    gate_weight = torch.cat(
        [weights["left_gate.weight"], weights["right_gate.weight"]], dim=0
    )  # (2*H, C)
    gate_out = F.linear(x, gate_weight)  # [B, N, N, 2*H]
    left_gate_raw, right_gate_raw = torch.split(gate_out, hidden_dim, dim=-1)

    # Apply sigmoid to obtain gates
    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate = torch.sigmoid(out_gate_raw)

    # ------------------------------------------------------------------
    # 4) Apply mask (if any) and combine projection + gates
    # ------------------------------------------------------------------
    if mask_tensor is None:
        mask = torch.ones(x.shape[:3], dtype=x.dtype, device=x.device)
    else:
        mask = mask_tensor.to(x.dtype)
    mask = mask.unsqueeze(-1)                     # [B, N, N, 1]

    left = left_proj_raw * left_gate * mask       # [B, N, N, H]
    right = right_proj_raw * right_gate * mask    # [B, N, N, H]

    # ------------------------------------------------------------------
    # 5) Batched GEMM with Triton (the cubic part)
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    # Cast to half‑precision for the kernel (Tensor‑cores)
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, N, _, H = left_h.shape  # H == hidden_dim
    batch = B * H               # fold the hidden dimension into the batch axis

    # reshape to (batch, N, N) : each (b,h) pair becomes a separate batch entry
    left_reshaped = (
        left_h.permute(0, 3, 1, 2).contiguous().view(batch, N, N)
    )  # (B*H, N, N)
    right_reshaped = (
        right_h.permute(0, 3, 1, 2).contiguous().view(batch, N, N)
    )  # (B*H, N, N)

    # output accumulator (fp32)
    out_reshaped = torch.empty_like(left_reshaped, dtype=torch.float32)

    # ------------------------------------------------------------------
    # Choose block sizes based on N (heuristic for H100)
    # ------------------------------------------------------------------
    if N <= 256:
        BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 32
    elif N <= 512:
        BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 32
    else:  # N up to 1024
        BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 64

    # Grid: (rows, cols, batch)
    grid = (
        triton.cdiv(N, BLOCK_M),   # pid_m
        triton.cdiv(N, BLOCK_N),   # pid_n
        batch,                     # pid_batch
    )

    # Strides for A (left)
    stride_Ab = left_reshaped.stride(0)
    stride_Am = left_reshaped.stride(1)
    stride_Ak = left_reshaped.stride(2)

    # Strides for B (right) – note the transposition: we treat B as (batch, K, N)
    stride_Bb = right_reshaped.stride(0)
    stride_Bk = right_reshaped.stride(2)   # stride over original K (column)
    stride_Bn = right_reshaped.stride(1)   # stride over original N (row)

    # Strides for C (output)
    stride_Cb = out_reshaped.stride(0)
    stride_Cm = out_reshaped.stride(1)
    stride_Cn = out_reshaped.stride(2)

    # Launch the Triton kernel
    tri_mul_gemm_kernel[grid](
        left_reshaped,
        right_reshaped,
        out_reshaped,
        batch,
        N,
        N,
        N,
        stride_Ab,
        stride_Am,
        stride_Ak,
        stride_Bb,
        stride_Bk,
        stride_Bn,
        stride_Cb,
        stride_Cm,
        stride_Cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # ------------------------------------------------------------------
    # 6) Reshape back to [B, N, N, H]
    # ------------------------------------------------------------------
    out = (
        out_reshaped.view(B, H, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 7) Second LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 8) Apply the output gate and final projection back to dim C
    # ------------------------------------------------------------------
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])  # [B, N, N, C]

    return out
