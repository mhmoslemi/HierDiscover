"""
custom_kernel
==============

Highly‑optimized forward implementation of the *outgoing* TriMul operator
used in AlphaFold‑3 and related protein‑structure models.

Algorithm
---------
1. **Layer‑norm** the input tensor `x` of shape `[B, N, N, C]`.
2. **Linear projections** to the hidden dimension `H` (no bias).  The
   projected tensors are cast to FP16 for speed.
3. **Mask** (if present) and **gating**:
   * Compute three sigmoid gates from the FP32‑normed input:
        - left_gate, right_gate  – cast to FP16.
        - out_gate               – stays FP32.
   * Apply left/right gates element‑wise to the projected tensors.
4. **Batched GEMM** (core `O(N³)` work):
   * Reshape `left` to `[G, N, N]` with `G = B·H` and permute the
     dimensions so that `left[g, i, k]` and `right[g, k, j]` are ready
     for a matrix multiply.
   * Launch a Triton kernel that computes, for every group `g`,
        `C[g, i, j] = Σ_k left[g, i, k] * right[g, k, j]`
     using FP16 tensor‑core `tl.dot` with FP32 accumulation.
   * Block sizes `BLOCK_M = BLOCK_N = 128`, `BLOCK_K = 32` and
     `num_warps = 8` match the H100 shared‑memory layout.
5. **Post‑processing**:
   * Reshape the result back to `[B, N, N, H]`.
   * Apply a second Layer‑norm over the hidden dimension, multiply by
     `out_gate`, and a final linear projection back to `C`.

All heavy compute (the `N³` term) is performed in the Triton kernel,
while the remaining lightweight ops use high‑performance PyTorch
primitives.  Mixed‑precision (FP16 for the large tensors, FP32 for
normalisations and final projection) provides a large speed‑up with
negligible loss of numerical fidelity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,                     # pointers to A, B, C
    M, N, K,                                 # matrix sizes (M×K)·(K×N) → (M×N)
    stride_ag, stride_am, stride_ak,         # A: group, row, col strides
    stride_bg, stride_bk, stride_bn,         # B: group, row(K), col(N) strides
    stride_cg, stride_cm, stride_cn,         # C: group, row, col strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM: C[g, i, j] = Σ_k A[g, i, k] * B[g, k, j]
    All tensors are stored in row‑major order.
    The kernel processes one (M,N) tile per program instance and loops
    over the K dimension in BLOCK_K chunks.
    """
    pid = tl.program_id(0)      # tile id within the (M,N) plane
    pid_g = tl.program_id(1)    # group id = batch * hidden_dim

    # --------------------------------------------------------------
    # Tile coordinates
    # --------------------------------------------------------------
    num_tiles_m = tl.cdiv(M, BLOCK_M)
    num_tiles_n = tl.cdiv(N, BLOCK_N)

    pid_m = pid // num_tiles_n
    pid_n = pid %  num_tiles_n

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # --------------------------------------------------------------
    # Pointers to the current group
    # --------------------------------------------------------------
    a_batch = a_ptr + pid_g * stride_ag
    b_batch = b_ptr + pid_g * stride_bg
    c_batch = c_ptr + pid_g * stride_cg

    # --------------------------------------------------------------
    # Accumulator (FP32)
    # --------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over K dimension
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile (M×K)
        a = tl.load(
            a_batch
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load B tile (K×N)
        b = tl.load(
            b_batch
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # Dot‑product accumulation (FP16·FP16 → FP32)
        acc += tl.dot(a, b)

    # --------------------------------------------------------------
    # Write result
    # --------------------------------------------------------------
    c = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (dtype float32).
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]                # C  (input channels)
    hidden_dim = config["hidden_dim"]  # H  (projection dimension)
    use_mask = not config.get("nomask", False)

    # --------------------------------------------------------------
    # 1) Input layer‑norm (FP32)
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, C]  FP32

    # --------------------------------------------------------------
    # 2) Linear projections to hidden dim (FP16)
    # --------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    # Cast normalized tensor to half for the projection
    x_fp16 = x.to(torch.float16)

    left = F.linear(x_fp16, left_proj_w)   # [B, N, N, H]  FP16
    right = F.linear(x_fp16, right_proj_w) # [B, N, N, H]  FP16

    # --------------------------------------------------------------
    # 3) Optional mask (broadcast over hidden dim)
    # --------------------------------------------------------------
    if use_mask:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_fp16
        right = right * mask_fp16

    # --------------------------------------------------------------
    # 4) Gating (sigmoid)
    # --------------------------------------------------------------
    # Small linear layers are cheap in FP32; compute gates in FP32
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x, left_gate_w)).to(torch.float16)   # [B,N,N,H] FP16
    right_gate = torch.sigmoid(F.linear(x, right_gate_w)).to(torch.float16) # [B,N,N,H] FP16
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))                      # [B,N,N,H] FP32

    # Apply left/right gates
    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5) Batched GEMM via Triton:
    #    left : (B, H, N, N) with axes (i,k)
    #    right: (B, H, N, N) with axes (k,j)
    # --------------------------------------------------------------
    # Rearrange so that the summed dimension is the inner one
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, i, k]
    right_perm = right.permute(0, 3, 2, 1).contiguous() # [B, H, k, j]

    B, H, N, _ = left_perm.shape
    G = B * H                                            # total groups

    # Collapse batch & head dimensions
    A = left_perm.view(G, N, N)   # (G, M=N, K=N)  FP16
    B_ = right_perm.view(G, N, N) # (G, K=N, N)  FP16

    # Output buffer (FP32 accumulation)
    C = torch.empty_like(A, dtype=torch.float32, device=device)  # (G,N,N)

    # --------------------------------------------------------------
    # Strides (in elements, not bytes)
    # --------------------------------------------------------------
    stride_ag = A.stride(0)
    stride_am = A.stride(1)
    stride_ak = A.stride(2)

    stride_bg = B_.stride(0)
    stride_bk = B_.stride(1)   # K‑dimension stride
    stride_bn = B_.stride(2)   # N‑dimension stride

    stride_cg = C.stride(0)
    stride_cm = C.stride(1)
    stride_cn = C.stride(2)

    M = N
    K = N

    # --------------------------------------------------------------
    # Kernel launch parameters – tuned for H100
    # --------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    num_tiles_m = (M + BLOCK_M - 1) // BLOCK_M
    num_tiles_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (num_tiles_m * num_tiles_n, G)

    _batched_gemm_kernel[grid](
        A,
        B_,
        C,
        M,
        N,
        K,
        stride_ag,
        stride_am,
        stride_ak,
        stride_bg,
        stride_bk,
        stride_bn,
        stride_cg,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # --------------------------------------------------------------
    # 6) Reshape back to [B, N, N, H] (FP32)
    # --------------------------------------------------------------
    out = C.view(B, H, N, N).permute(0, 2, 3, 1)  # [B, N, N, H]  FP32

    # --------------------------------------------------------------
    # 7) Output layer‑norm, out‑gate, final linear projection
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=1e-5,
    )  # [B,N,N,H] FP32

    out = out * out_gate                                 # FP32

    to_out_w = weights["to_out.weight"]
    output = F.linear(out, to_out_w)                     # [B,N,N,dim] FP32

    return output
