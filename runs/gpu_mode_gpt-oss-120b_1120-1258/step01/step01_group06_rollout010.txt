def tri_mul_kernel(
    a_ptr,                     # *fp16   [B*H, N, N]   left  matrix
    b_ptr,                     # *fp16   [B*H, N, N]   right matrix (to be transposed)
    c_ptr,                     # *fp32   [B*H, N, N]   output
    M, N, K,                   # all equal to seq_len
    stride_ah, stride_ai, stride_ak,   # strides for A (batch*H, i, k)
    stride_bh, stride_bi, stride_bk,   # strides for B (batch*H, j, k)
    stride_ch, stride_ci, stride_cj,   # strides for C (batch*H, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    num_stages: tl.constexpr,          # async copy pipeline depth
    num_warps: tl.constexpr,
):
    """
    Compute a batched matrix multiplication for the TriMul operator.
    """
    pid_m = tl.program_id(0)      # block row index
    pid_n = tl.program_id(1)      # block col index
    pid_b = tl.program_id(2)      # batch*hidden index

    # ------------------------------------------------------------------
    # Tile offsets (row/col within the N×N matrix)
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for out‑of‑bounds threads
    mask_m = offs_m < M
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # Pointers to the beginning of the current tile (A: M×K, B: N×K)
    # ------------------------------------------------------------------
    a_ptrs = a_ptr + pid_b * stride_ah + offs_m[:, None] * stride_ai
    b_ptrs = b_ptr + pid_b * stride_bh + offs_n[None, :] * stride_bi

    # ------------------------------------------------------------------
    # Accumulator (fp32) – we sum over the K dimension
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # K‑loop with tiled loads.  We use async copy (num_stages) to hide
    # memory latency – this pattern matches the highly‑tuned matmul
    # kernel from the Triton repo.
    # ------------------------------------------------------------------
    for k in range(0, K, BLOCK_K):
        cur_k = k + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # Load A tile : [BLOCK_M, BLOCK_K]  – fp16
        a = tl.load(
            a_ptrs + cur_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )

        # Load B tile : [BLOCK_K, BLOCK_N]  – fp16
        # Note: we read B as (k, n) which is equivalent to Bᵀ for the matmul.
        b = tl.load(
            b_ptrs + cur_k[:, None] * stride_bk,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )

        # Dot‑product and accumulation (fp32)
        acc += tl.dot(a, b).to(tl.float32)

    # ------------------------------------------------------------------
    # Write the result
    # ------------------------------------------------------------------
    c_ptrs = (
        c_ptr
        + pid_b * stride_ch
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry‑point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    # ------------------------------------------------------------------
    # Unpack config
    # ------------------------------------------------------------------
    dim = config["dim"]               # input channel dimension
    hidden_dim = config["hidden_dim"] # projection dimension (e.g. 128)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (fp16) – heavy tensors
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"]).to(torch.float16)   # [B,N,N,H]
    right = F.linear(x, weights["right_proj.weight"]).to(torch.float16)  # [B,N,N,H]

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcasted on the hidden dimension)
    # ------------------------------------------------------------------
    if not config.get("nomask", False):
        # mask: [B, N, N] → [B, N, N, 1] → fp16
        mask = mask_tensor.to(left.dtype).unsqueeze(-1)
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4) Gating (keep in fp16 to stay in the same precision as the matmul)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"])).to(torch.float16)
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])).to(torch.float16)

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Out‑gate (kept in fp32 because it is applied after the matmul)
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))  # [B,N,N,H]

    # ------------------------------------------------------------------
    # 6) Rearrange tensors so that each (batch, hidden) pair is a
    #    separate matrix of shape [N, N] – this layout is required by the
    #    kernel (contiguous inner dimensions).
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Output buffer for the matmul (fp32)
    out_flat = torch.empty((B * H, N, N), dtype=torch.float32, device=device)

    # ------------------------------------------------------------------
    # 7) Launch the Triton kernel
    # ------------------------------------------------------------------
    # Tuned tile sizes for H100 (tensor‑core friendly)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # tiles in M dimension
        (N + BLOCK_N - 1) // BLOCK_N,   # tiles in N dimension
        B * H,                          # one program per (batch, hidden)
    )

    tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        M=N,
        N=N,
        K=N,
        stride_ah=left_flat.stride(0),
        stride_ai=left_flat.stride(1),
        stride_ak=left_flat.stride(2),
        stride_bh=right_flat.stride(0),
        stride_bi=right_flat.stride(1),
        stride_bk=right_flat.stride(2),
        stride_ch=out_flat.stride(0),
        stride_ci=out_flat.stride(1),
        stride_cj=out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_stages=3,          # async copy pipeline depth
        num_warps=8,           # 8 warps → good occupancy on H100
    )

    # ------------------------------------------------------------------
    # 8) Restore original layout: [B, N, N, H]
    # ------------------------------------------------------------------
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # fp32

    # ------------------------------------------------------------------
    # 9) Final LayerNorm over the hidden dimension, out‑gate, and projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    output = F.linear(out, weights["to_out.weight"])   # bias is omitted in the model

    return output
