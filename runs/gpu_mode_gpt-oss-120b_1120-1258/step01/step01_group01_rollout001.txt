def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # 1. LayerNorm (float32)
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weights["norm.weight"], weights["norm.bias"], eps=1e-5)

    # Cast to half for linear ops
    x_h = x.to(torch.float16)

    # Linear projections (half precision)
    left = torch.nn.functional.linear(x_h, weights["left_proj.weight"].to(torch.float16))
    right = torch.nn.functional.linear(x_h, weights["right_proj.weight"].to(torch.float16))

    # Gates (half)
    left_gate = torch.nn.functional.linear(x_h, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = torch.nn.functional.linear(x_h, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = torch.nn.functional.linear(x_h, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # Apply mask before gating (if needed)
    if not nomask and mask is not None:
        mask_f = mask.to(torch.float16).unsqueeze(-1)  # shape [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Heavy multiplication: (B,H,N,N) [] heavy, using Triton
    B, N, _, _ = left.shape
    # Permute to (B, H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 2, 1).contiguous()   # transposed K<->i: (B, H, N, N) with dims (k,i)
    # Flatten (B*H,N,N)
    BH = B * hidden
    left_flat = left_perm.view(BH, N, N)
    right_flat = right_perm.view(BH, N, N)

    # Output buffer (FP16)
    out_flat = torch.empty_like(left_flat)   # shape (BH,N,N) FP16

    # Block sizes tuned
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    # Determine grid
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        BH,
    )

        left_flat,
        right_flat,
        out_flat,
        N,
        N,
        N,
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        right_flat.stride(0), right_flat.stride(1), right_flat.stride(2),
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B,N,N,hidden] fp16
    out = out_flat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    out_fp32 = out.to(torch.float32)

    # Final LayerNorm over hidden dim

        out_fp32,
        (hidden,),
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        eps=1e-5,
    )

"""
TriMul (outgoing) forward – highly‑optimized Triton implementation.

Algorithm
---------
1. Layer‑norm on the channel dimension (float32).
2. Cast to FP16 and compute **all** linear projections (left, right, left‑gate,
   right‑gate, out‑gate) in a single precision‑reduced pass.
3. Apply the optional pairwise mask (broadcast on the hidden dimension) **before**
   gating, then multiply by the sigmoid gates.
4. Heavy O(N³) update:
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication for each hidden slice.
   We fuse it in a Triton kernel that computes (B·H, N, N) = A @ Bᵀ
   with FP16 inputs and FP32 accumulation.
5. Reshape back to [B,N,N,hidden] and cast to FP32.
6. Layer‑norm over the hidden dimension, multiply by the out‑gate, and
   linearly project back to the original channel dimension (FP16 → FP32).

The Triton kernel uses block sizes (128,128,32) which are well‑tuned for
N up to 1024 on an H100 (full‑warp WMMA tiles, 16‑byte alignment).
All heavy arithmetic is performed in FP16 to halve memory traffic,
while accumulation stays in FP32 for numerical stability.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def _batched_matmul_kernel(
    A, B, C,
    M, N, K,
    stride_Ab, stride_Ar, stride_Ac,   # batch, row, col strides for A
    stride_Bb, stride_Br, stride_Bc,   # batch, row, col strides for B
    stride_Cb, stride_Cr, stride_Cc,   # batch, row, col strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched FP16 matmul C = A @ B (A: (B, M, K), B: (B, K, N)).
    Accumulates in FP32 and writes FP16."""
    pid_m = tl.program_id(0)   # block row
    pid_n = tl.program_id(1)   # block col
    pid_b = tl.program_id(2)   # batch (merged B·H)

    # Offsets for this block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers into A and B
    a_ptrs = A + pid_b * stride_Ab + offs_m[:, None] * stride_Ar + offs_k[None, :] * stride_Ac
    b_ptrs = B + pid_b * stride_Bb + offs_k[:, None] * stride_Br + offs_n[None, :] * stride_Bc

    # Accumulator (FP32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    num_k = tl.cdiv(K, BLOCK_K)
    for _ in range(num_k):
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k[None, :] < K),
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & mask_n[None, :],
                    other=0.0)
        acc += tl.dot(a, b)           # dot promotes to FP32 automatically
        a_ptrs += BLOCK_K * stride_Ac
        b_ptrs += BLOCK_K * stride_Br

    # Write result (FP16) with masking
    c_ptrs = C + pid_b * stride_Cb + offs_m[:, None] * stride_Cr + offs_n[None, :] * stride_Cc
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator (no backward needed).

    Args:
        data (Tuple):
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights: dict of weight tensors required by the module
            - config: dict containing keys "dim", "hidden_dim", "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. LayerNorm on channel dimension (float32)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim]  (FP32)

    # ------------------------------------------------------------------
    # 2. Cast to half and compute all linear projections + gates in FP16
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    # Linear projections (no bias)
    left = torch.nn.functional.linear(
        x_h, weights["left_proj.weight"].to(torch.float16)
    )  # [B,N,N,hidden] FP16
    right = torch.nn.functional.linear(
        x_h, weights["right_proj.weight"].to(torch.float16)
    )  # [B,N,N,hidden] FP16

    # Gates (sigmoid)
    left_gate = torch.nn.functional.linear(
        x_h, weights["left_gate.weight"].to(torch.float16)
    ).sigmoid()                      # [B,N,N,hidden] FP16
    right_gate = torch.nn.functional.linear(
        x_h, weights["right_gate.weight"].to(torch.float16)
    ).sigmoid()
    out_gate = torch.nn.functional.linear(
        x_h, weights["out_gate.weight"].to(torch.float16)
    ).sigmoid()

    # ------------------------------------------------------------------
    # 3. Apply optional mask (broadcast on hidden)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_h = mask.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------
    # 4. Apply element‑wise gates
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Heavy multiplicative update using Triton (batched GEMM)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape  # B, N, N, hidden

    # Rearrange to (B, hidden, N, N) for A and (B, hidden, K, J) for Bᵀ
    left_perm = left.permute(0, 3, 1, 2).contiguous()          # (B, H, N, N)  A[i,k]
    right_perm = right.permute(0, 3, 2, 1).contiguous()        # (B, H, K, J)  Bᵀ[k,j]

    # Merge batch and hidden dimensions → (B·H, N, N)
    BH = B * hidden
    left_flat = left_perm.view(BH, N, N)    # (BH, M=N, K=N)
    right_flat = right_perm.view(BH, N, N)  # (BH, K=N, N)

    # Output buffer (FP16)
    out_flat = torch.empty_like(left_flat)  # (BH, N, N)

    # Tune block sizes for H100 (multiples of 16 for WMMA)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        BH,
    )

    _batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,                      # M
        N,                      # N
        N,                      # K
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        right_flat.stride(0), right_flat.stride(1), right_flat.stride(2),
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, hidden] (still FP16)
    out = out_flat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Final LayerNorm, out‑gate, and projection back to original dim
    # ------------------------------------------------------------------
    out = out.to(torch.float32)   # convert to FP32 for the remaining ops

    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # Apply out‑gate (converted to FP32)
    out = out * out_gate.to(torch.float32)

    # Final linear projection (hidden → dim) in FP16 for speed, then back to FP32
    out_h = out.to(torch.float16)
    to_out_weight_h = weights["to_out.weight"].to(torch.float16)
    out = torch.nn.functional.linear(out_h, to_out_weight_h).to(torch.float32)

    return out
