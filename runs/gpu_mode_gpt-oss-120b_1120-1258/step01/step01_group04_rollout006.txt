"""
TriMul (outgoing) – Triton‑accelerated forward pass.

Algorithm
---------
1. Layer‑norm the input tensor (float32).
2. Cast to float16 and apply three bias‑free linear projections:
      left, right and three sigmoid gates (left_gate, right_gate, out_gate).
   All linear layers use half‑precision weights.
3. Apply gating (element‑wise multiplication) and optional mask 
   (mask is also cast to float16).
4. Perform the cubic contraction
      out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   for every (batch, hidden) pair by launching a Triton kernel that
   computes a batched GEMM C = A @ Bᵀ with A,B ∈ ℝ^{N×N}.
   * Inputs are float16, accumulation is float32 (tensor cores).
   * Tile sizes: BLOCK_M × BLOCK_N × BLOCK_K (tuned for H100).
5. Reshape the result back to [B,N,N,hidden] (float32).
6. Layer‑norm over the hidden dimension, multiply by the output gate,
   and a final bias‑free linear projection back to the original feature size.
   The final projection is performed in float32.

The implementation keeps the memory footprint low by using half‑precision for
the heavy linear projections and the O(N³) contraction, while the surrounding
normalisation layers stay in float32 for numerical stability.
"""

import math
import torch
import triton
import triton.language as tl


@triton.jit
def tri_mul_outgoing_kernel(
    a_ptr,               # left  (M, N, N)   – half‑precision
    b_ptr,               # right (M, N, N)   – half‑precision
    c_ptr,               # output (M, N, N) – float32
    N,                   # sequence length
    stride_am, stride_ai, stride_ak,   # left strides (batch, i, k)
    stride_bm, stride_bj, stride_bk,   # right strides (batch, j, k) – we read k first, j second
    stride_cm, stride_ci, stride_cj,   # output strides (batch, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """C = A @ Bᵀ   where A,B ∈ ℝ^{N×N} per (batch,hidden) pair."""
    pid_m = tl.program_id(0)   # i‑tile
    pid_n = tl.program_id(1)   # j‑tile
    pid_b = tl.program_id(2)   # (batch, hidden) index

    i0 = pid_m * BLOCK_M
    j0 = pid_n * BLOCK_N
    i = i0 + tl.arange(0, BLOCK_M)
    j = j0 + tl.arange(0, BLOCK_N)

    mask_i = i < N
    mask_j = j < N

    # pointer to the output tile
    c_ptrs = c_ptr + pid_b * stride_cm + i[:, None] * stride_ci + j[None, :] * stride_cj
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k
    for k0 in range(0, N, BLOCK_K):
        k = k0 + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # ----- load A[i, k] -------------------------------------------------
        a_ptrs = a_ptr + pid_b * stride_am + i[:, None] * stride_ai + k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=tl.float16(0.0))

        # ----- load B[k, j]   (B is stored as (j, k), we read transposed) --
        b_ptrs = b_ptr + pid_b * stride_bm + k[:, None] * stride_bk + j[None, :] * stride_bj
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=tl.float16(0.0))

        # half‑precision dot, accumulate in fp32
        acc += tl.dot(a, b)

    # write the tile
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module using a Triton kernel for the
    O(N³·H) contraction.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask:        torch.Tensor [B, N, N] (float32 or bool)
            - weights:     dict of model parameters (all torch.Tensors on CUDA)
            - config:      dict with keys "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------ #
    # unpack inputs
    # ------------------------------------------------------------------ #
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    # ------------------------------------------------------------------ #
    # unpack weights (all are already on the correct device)
    # ------------------------------------------------------------------ #
    norm_weight          = weights["norm.weight"]
    norm_bias            = weights["norm.bias"]
    left_proj_weight     = weights["left_proj.weight"]
    right_proj_weight    = weights["right_proj.weight"]
    left_gate_weight     = weights["left_gate.weight"]
    right_gate_weight    = weights["right_gate.weight"]
    out_gate_weight      = weights["out_gate.weight"]
    to_out_norm_weight   = weights["to_out_norm.weight"]
    to_out_norm_bias     = weights["to_out_norm.bias"]
    to_out_weight        = weights["to_out.weight"]

    # ------------------------------------------------------------------ #
    # 1. LayerNorm over the input feature dimension (float32)
    # ------------------------------------------------------------------ #
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------ #
    # 2. Convert to half‑precision for the heavy work
    # ------------------------------------------------------------------ #
    x_h = x.to(torch.float16)

    # cast weights once to half‑precision
    left_proj_weight_h   = left_proj_weight.to(torch.float16)
    right_proj_weight_h  = right_proj_weight.to(torch.float16)
    left_gate_weight_h   = left_gate_weight.to(torch.float16)
    right_gate_weight_h  = right_gate_weight.to(torch.float16)
    out_gate_weight_h    = out_gate_weight.to(torch.float16)

    # ------------------------------------------------------------------ #
    # 3. Linear projections (bias‑free) in half‑precision
    # ------------------------------------------------------------------ #
    left  = torch.nn.functional.linear(x_h, left_proj_weight_h)   # [B, N, N, H]  hf
    right = torch.nn.functional.linear(x_h, right_proj_weight_h)  # [B, N, N, H]

    # ------------------------------------------------------------------ #
    # 4. Gating (sigmoid) also in half‑precision
    # ------------------------------------------------------------------ #
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x_h, left_gate_weight_h))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_h, right_gate_weight_h))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x_h, out_gate_weight_h))

    # apply gates
    left  = left * left_gate
    right = right * right_gate

    # optional mask (broadcasted on the hidden dimension)
    if not nomask:
        mask_h = mask.to(torch.float16).unsqueeze(-1)           # [B, N, N, 1]
        left  = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------ #
    # 5. Rearrange tensors so that each (batch, hidden) pair becomes a
    #    separate matrix multiplication: (M, N, N) where M = B * H.
    # ------------------------------------------------------------------ #
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]  hf
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]  hf

    B, H, N, _ = left_perm.shape
    M_total = B * H

    left_flat  = left_perm.view(M_total, N, N)   # [M, N, N]  hf
    right_flat = right_perm.view(M_total, N, N)  # [M, N, N]  hf
    out_flat   = torch.empty((M_total, N, N), dtype=torch.float32, device=device)

    # ------------------------------------------------------------------ #
    # 6. Triton kernel launch (batched GEMM C = A @ Bᵀ)
    # ------------------------------------------------------------------ #
    # Tile sizes – tuned for H100 (grid fits in shared memory)
    BLOCK_M = 64   # rows processed per program
    BLOCK_N = 64   # cols processed per program
    BLOCK_K = 64   # reduction tile

    # Strides (in element units)
    stride_am = left_flat.stride(0)
    stride_ai = left_flat.stride(1)
    stride_ak = left_flat.stride(2)

    stride_bm = right_flat.stride(0)
    stride_bj = right_flat.stride(1)   # j‑dimension stride
    stride_bk = right_flat.stride(2)   # k‑dimension stride

    stride_cm = out_flat.stride(0)
    stride_ci = out_flat.stride(1)
    stride_cj = out_flat.stride(2)

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        M_total                         # one program per (batch, hidden)
    )

    tri_mul_outgoing_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        stride_am, stride_ai, stride_ak,
        stride_bm, stride_bj, stride_bk,
        stride_cm, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------ #
    # 7. Reshape back to [B, N, N, H] (float32)
    # ------------------------------------------------------------------ #
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]

    # ------------------------------------------------------------------ #
    # 8. Final LayerNorm, output‑gate, and projection back to dim
    # ------------------------------------------------------------------ #
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    # out_gate is still half‑precision; cast to float32 for the multiplication
    out = out * out_gate.to(torch.float32)

    output = torch.nn.functional.linear(out, to_out_weight)   # [B, N, N, dim]  (float32)

    return output
