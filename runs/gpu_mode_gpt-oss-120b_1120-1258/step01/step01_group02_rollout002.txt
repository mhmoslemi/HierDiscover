"""
TriMul (outgoing) – fast forward pass with a fused Triton batched‑GEMM.

The heavy tensor contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is a batched matrix multiply (A @ Bᵀ) for every head *d*.
We implement this N³ part in Triton (half‑precision inputs, FP32
accumulation) while keeping the surrounding LayerNorms,
linear projections and gating in PyTorch for simplicity.
The kernel works for any batch size (1‑2), sequence length up to
1024 and hidden dimensions that fit in memory (the tests use
hidden_dim = 128).  The final result is returned as float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ B   (A: M×K, B: K×N)
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr,            # *fp16   left matrices   (B*H, M, K)
    b_ptr,            # *fp16   right matrices  (B*H, K, N)
    c_ptr,            # *fp16   output matrices (B*H, M, N)
    M, N, K,          # int32  dimensions (all equal to seq_len)
    stride_az, stride_am, stride_ak,   # strides for A
    stride_bz, stride_bk, stride_bn,   # strides for B
    stride_cz, stride_cm, stride_cn,   # strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM using fp16 operands and fp32 accumulation.
    Each program-id encodes (batch_id, tile_m, tile_n).
    """
    pid0 = tl.program_id(0)          # batch_id * num_tiles_m + tile_m
    pid1 = tl.program_id(1)          # tile_n

    # -------------------------------------------------------------
    # Decode batch and tile indices
    # -------------------------------------------------------------
    num_tiles_m = tl.cdiv(M, BLOCK_M)
    batch_id    = pid0 // num_tiles_m
    tile_m      = pid0 %  num_tiles_m
    tile_n      = pid1

    # -------------------------------------------------------------
    # Offsets inside the tile
    # -------------------------------------------------------------
    offs_m = tile_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = tile_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # -------------------------------------------------------------
    # Pointers to the start of the current tile (no k‑loop offset yet)
    # -------------------------------------------------------------
    a_ptrs = a_ptr + batch_id * stride_az \
                 + offs_m[:, None] * stride_am \
                 + offs_k[None, :] * stride_ak
    b_ptrs = b_ptr + batch_id * stride_bz \
                 + offs_k[:, None] * stride_bk \
                 + offs_n[None, :] * stride_bn

    # -------------------------------------------------------------
    # Accumulator (FP32 for numeric safety)
    # -------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------------------------------------------------
    # Main k‑loop
    # -------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # Masks for out‑of‑bounds loads (required when M,N,K not multiples)
        mask_a = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        mask_b = (offs_k[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=mask_a, other=0.0)   # (BLOCK_M, BLOCK_K) fp16
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)   # (BLOCK_K, BLOCK_N) fp16

        # fp16·fp16 → fp32 accumulation
        acc += tl.dot(a, b)

        # Advance pointers to the next k‑block
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # -------------------------------------------------------------
    # Write out the result (convert back to fp16)
    # -------------------------------------------------------------
    c_ptrs = c_ptr + batch_id * stride_cz \
                     + offs_m[:, None] * stride_cm \
                     + offs_n[None, :] * stride_cn
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    Arguments
    ----------
    data : tuple (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask         : torch.Tensor [B, N, N] (bool/float) or None
        - weights      : dict of model parameters (float32)
        - config       : dict with keys:
                         "dim", "hidden_dim", "nomask" (bool)

    Returns
    -------
    torch.Tensor : [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack user inputs
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype  = input_tensor.dtype

    dim        = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask     = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm over the last dimension (`dim`)
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,),
                     weight=norm_weight,
                     bias=norm_bias)

    # --------------------------------------------------------------
    # 2️⃣ Linear projections (no bias) → [B, N, N, hidden_dim]
    # --------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast on hidden dimension)
    # --------------------------------------------------------------
    if (not nomask) and (mask is not None):
        mask_f = mask.to(dtype).unsqueeze(-1)          # [B, N, N, 1]
        left  = left * mask_f
        right = right * mask_f

    # --------------------------------------------------------------
    # 4️⃣ Gating scalars (sigmoid after linear)
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # Apply gates
    left  = left  * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Cast to fp16 for the N³ contraction
    # --------------------------------------------------------------
    left_fp16  = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    B, N, _, _ = left_fp16.shape   # (B, N, N, hidden_dim)

    # --------------------------------------------------------------
    # 6️⃣ Reshape to (B*hidden_dim, N, N)
    #     left  : (M=i, K=k)   → (B*H, M, K)
    #     right : (K=k, N=j)   → (B*H, K, N)   (k/j axes swapped)
    # --------------------------------------------------------------
    left_mat  = left_fp16.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right_fp16.permute(0, 3, 2, 1).contiguous().view(B * hidden_dim, N, N)

    # Allocate output buffer (fp16, accumulation in FP32 inside the kernel)
    out_mat = torch.empty_like(left_mat, dtype=torch.float16, device=device)

    # --------------------------------------------------------------
    # 7️⃣ Triton launch configuration
    # --------------------------------------------------------------
    # Tuned block sizes – safe for all test lengths up to 1024.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # Number of tiles along M/N within one batch*head
    grid_m = triton.cdiv(N, BLOCK_M)   # tile rows
    grid_n = triton.cdiv(N, BLOCK_N)   # tile columns

    # pid0 packs (batch*head, tile_m); pid1 indexes tile_n
    grid = (B * hidden_dim * grid_m, grid_n)

    batched_gemm_kernel[grid](
        a_ptr=left_mat,
        b_ptr=right_mat,
        c_ptr=out_mat,
        M=N,
        N=N,
        K=N,
        stride_az=left_mat.stride(0),
        stride_am=left_mat.stride(1),
        stride_ak=left_mat.stride(2),
        stride_bz=right_mat.stride(0),
        stride_bk=right_mat.stride(1),
        stride_bn=right_mat.stride(2),
        stride_cz=out_mat.stride(0),
        stride_cm=out_mat.stride(1),
        stride_cn=out_mat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,          # good default on H100
        num_stages=2,
    )

    # --------------------------------------------------------------
    # 8️⃣ Restore original tensor shape: [B, N, N, hidden_dim]
    # --------------------------------------------------------------
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)   # back to FP32 for the remaining ops

    # --------------------------------------------------------------
    # 9️⃣ Output LayerNorm, gating and final projection back to `dim`
    # --------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    out = F.layer_norm(out, (hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias)

    out = out * out_gate                       # element‑wise output gating
    out = F.linear(out, weights["to_out.weight"])   # (dim, hidden_dim) → final dim

    return out
```
