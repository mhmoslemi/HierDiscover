```
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    M, N, K,
    stride_fa, stride_fi, stride_fk,
    stride_rb, stride_ri, stride_rk,
    stride_ob, stride_oi, stride_oj,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0) # row block
    pid_n = tl.program_id(1) # col block
    pid_b = tl.program_id(2) # batch*hidden

    # Offsets
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers for A and B
    a_ptrs = left_ptr + pid_b * stride_fa + offs_m[:, None] * stride_fi
    b_ptrs = right_ptr + pid_b * stride_rb + offs_n[None, :] * stride_ri

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        a = tl.load(
            a_ptrs + offs_k[None, :] * stride_fk,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        b = tl.load(
            b_ptrs + offs_k[:, None] * stride_rk,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # dot: a (M,K), b (K,N)
        acc += tl.dot(a, b, out_dtype=tl.float32)

    # Store
    c_ptrs = out_ptr + pid_b * stride_ob + offs_m[:, None] * stride_oi + offs_n[None, :] * stride_oj
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])
```

def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # 1) LayerNorm
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps)

    # 2) Cast to half for linear projections/gates
    # Cast weight to half
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)  # maybe use fp16 then cast later

    # Projections
    left = F.linear(x, left_proj_w)  # fp16
    right = F.linear(x, right_proj_w)

    # Mask
    if not nomask:
        mask = mask_tensor.to(left.dtype).unsqueeze(-1)
        left = left * mask
        right = right * mask

    # Gates
    left_gate = torch.sigmoid(F.linear(x, left_gate_w)).to(torch.float16)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w)).to(torch.float16)

    left = left * left_gate
    right = right * right_gate

    # out_gate (keep fp32 for later multiplication)
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))  # fp32

    # 3) Prepare for kernel
    B, N, _, H = left.shape  # H = hidden_dim
    # permute & contiguous
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    out_flat = torch.empty((B * H, N, N), dtype=torch.float32, device=device)

    # Launch kernel
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = ( (N + BLOCK_M - 1) // BLOCK_M,
             (N + BLOCK_N - 1) // BLOCK_N,
             B * H )

    tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        M=N,
        N=N,
        K=N,
        stride_fa=left_flat.stride(0),
        stride_fi=left_flat.stride(1),
        stride_fk=left_flat.stride(2),
        stride_rb=right_flat.stride(0),
        stride_ri=right_flat.stride(1),
        stride_rk=right_flat.stride(2),
        stride_ob=out_flat.stride(0),
        stride_oi=out_flat.stride(1),
        stride_oj=out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )

    # 4) Reshape back
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # 5) Final LayerNorm, out_gate, projection
    out = F.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"], eps=eps)
    out = out * out_gate
    output = F.linear(out, weights["to_out.weight"])

    return output

"""
Outgoing Triangle‑Multiplicative Update (TriMul) kernel.

The heavy part of the TriMul forward pass is the “einsum”
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is a batch of independent matrix multiplications
    C = A @ Bᵀ   with   A,B ∈ ℝ^{(B·H)×N×N}   (H = hidden_dim).

The kernel below implements this batched GEMM in Triton:
* FP16 inputs (left/right)  →  FP32 accumulation for numerical stability.
* Large tile sizes (128×128×64) reduce kernel launch overhead.
* 8 warps + 3 pipeline stages fully utilise an H100 SM.

All surrounding ops (LayerNorm, linear projections, gating, mask handling,
final LayerNorm, out‑gate, output projection) are performed with
standard PyTorch kernels.  The function `custom_kernel` follows the
signature required by the test harness.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    left_ptr,               # fp16  [B*H, N, N]
    right_ptr,              # fp16  [B*H, N, N]
    out_ptr,                # fp32  [B*H, N, N]
    M, N, K,                # all equal to seq_len
    stride_fa, stride_fi, stride_fk,   # A strides (batch, i, k)
    stride_rb, stride_ri, stride_rk,   # B strides (batch, j, k)
    stride_ob, stride_oi, stride_oj,   # C strides (batch, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM C = A @ Bᵀ  (A,B are FP16, C is FP32)."""
    pid_m = tl.program_id(0)          # row‑tile of C
    pid_n = tl.program_id(1)          # col‑tile of C
    pid_b = tl.program_id(2)          # which (batch*H) matrix

    # ------------------------------------------------------------------
    # Tile offsets and masks
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # Pointers to the start of current tiles of A and Bᵀ
    # ------------------------------------------------------------------
    # A tile: rows = i (offs_m), cols = k (added later)
    a_ptr = left_ptr + pid_b * stride_fa + offs_m[:, None] * stride_fi
    # Bᵀ tile: rows = k (added later), cols = j (offs_n)
    b_ptr = right_ptr + pid_b * stride_rb + offs_n[None, :] * stride_ri

    # ------------------------------------------------------------------
    # Accumulator (FP32)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the K dimension (tiles of size BLOCK_K)
    # ------------------------------------------------------------------
    num_tiles_k = tl.cdiv(K, BLOCK_K)
    for k in range(0, num_tiles_k):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile (M×K)
        a = tl.load(
            a_ptr + offs_k[None, :] * stride_fk,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load Bᵀ tile (K×N) – note the transposed view
        b = tl.load(
            b_ptr + offs_k[:, None] * stride_rk,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # Compute partial product and accumulate in FP32
        acc += tl.dot(a, b, out_dtype=tl.float32)

    # ------------------------------------------------------------------
    # Write the result tile
    # ------------------------------------------------------------------
    c_ptr = out_ptr + pid_b * stride_ob + offs_m[:, None] * stride_oi + offs_n[None, :] * stride_oj
    tl.store(c_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Arguments:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32).
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    dim = config["dim"]               # input channel dimension (c)
    hidden_dim = config["hidden_dim"] # projection dimension (c_z)
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (over the last dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # shape: [B, N, N, dim]  (FP32)

    # ------------------------------------------------------------------
    # 2) Linear projections (still FP32 for numerical fidelity)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast over hidden_dim)
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B,N,N] → [B,N,N,1]
        mask = mask_tensor.to(left.dtype).unsqueeze(-1)
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid) – keep in FP32 for now, then cast to FP16
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))  # FP32, used later

    left = left * left_gate
    right = right * right_gate

    # Cast the gated tensors to FP16 before the heavy matmul
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # ------------------------------------------------------------------
    # 5) Reshape for Triton kernel:  [B, hidden_dim, N, N] → [B*hidden_dim, N, N]
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Output buffer (FP32 accumulation)
    out_flat = torch.empty((B * H, N, N), dtype=torch.float32, device=device)

    # ------------------------------------------------------------------
    # 6) Launch the Triton kernel
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # rows
        (N + BLOCK_N - 1) // BLOCK_N,   # cols
        B * H,                          # batch * hidden
    )

    tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        M=N,
        N=N,
        K=N,
        stride_fa=left_flat.stride(0),
        stride_fi=left_flat.stride(1),
        stride_fk=left_flat.stride(2),
        stride_rb=right_flat.stride(0),
        stride_ri=right_flat.stride(1),
        stride_rk=right_flat.stride(2),
        stride_ob=out_flat.stride(0),
        stride_oi=out_flat.stride(1),
        stride_oj=out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,        # 8 warps → 256 threads per block on H100
        num_stages=3,       # pipeline three stages of loads
    )

    # ------------------------------------------------------------------
    # 7) Reshape the kernel output back to [B, N, N, hidden_dim]
    # ------------------------------------------------------------------
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # FP32

    # ------------------------------------------------------------------
    # 8) Final LayerNorm, out‑gate, and projection back to original dim
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate  # broadcast over hidden_dim
    output = F.linear(out, weights["to_out.weight"])  # [B, N, N, dim]

    return output
