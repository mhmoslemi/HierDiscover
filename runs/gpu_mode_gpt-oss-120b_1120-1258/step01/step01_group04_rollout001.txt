"""
Optimized Triton implementation of the outgoing TriMul operation.

Algorithm
---------
1. Layer‑norm the input over the feature dimension `dim`.
2. Bias‑free linear projections to a hidden dimension `hidden_dim`
   (left_proj, right_proj) and three sigmoid‑gated linear projections
   (left_gate, right_gate, out_gate).
3. Optional masking of the projected tensors.
4. Apply the element‑wise gates to `left` and `right`.  The tensors are
   cast to float16 to reduce memory traffic.
5. Perform the cubic contraction
        out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   as a batched GEMM `C = A @ Bᵀ` where for each (batch, hidden) pair
   `A` and `B` are `N×N` matrices.  The kernel tiles the `i`, `j`,
   and `k` dimensions (BLOCK_M×BLOCK_N×BLOCK_K) and uses `tl.dot`
   with half‑precision inputs, accumulating in float32.
6. Cast the result back to float32, apply a second layer‑norm over the
   hidden dimension, multiply by the output gate, and finish with a
   bias‑free linear projection back to the original `dim`.

The kernel is fused to the reduction dimension and uses half‑precision
memory accesses while preserving float32 accumulation for numerical
stability.
"""

import torch
import triton
import triton.language as tl
import math

# --------------------------------------------------------------------------- #
# Triton kernel: C = A @ Bᵀ   (A, B ∈ ℝ^{N×N})  for many independent batches
# --------------------------------------------------------------------------- #
@triton.jit
def tri_mul_outgoing_kernel(
    a_ptr,               # left  (M, N, N)   – float16
    b_ptr,               # right (M, N, N)   – float16, read as (k, j)
    c_ptr,               # output (M, N, N) – float16 (will be cast later)
    N,                   # sequence length
    # strides for A
    stride_am, stride_ai, stride_ak,
    # strides for B (read transposed)
    stride_bm, stride_bj, stride_bk,
    # strides for C
    stride_cm, stride_ci, stride_cj,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)            # tile along i (rows)
    pid_n = tl.program_id(1)            # tile along j (cols)
    pid_b = tl.program_id(2)            # combined batch & hidden index

    # Compute absolute indices of the block
    i0 = pid_m * BLOCK_M
    j0 = pid_n * BLOCK_N
    i = i0 + tl.arange(0, BLOCK_M)
    j = j0 + tl.arange(0, BLOCK_N)

    mask_i = i < N
    mask_j = j < N

    # Pointer to the output tile
    c_ptrs = (
        c_ptr
        + pid_b * stride_cm
        + i[:, None] * stride_ci
        + j[None, :] * stride_cj
    )
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k
    for k0 in range(0, N, BLOCK_K):
        k = k0 + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # Load A[i, k]   (shape BLOCK_M × BLOCK_K)
        a_ptrs = (
            a_ptr
            + pid_b * stride_am
            + i[:, None] * stride_ai
            + k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float16)

        # Load B[k, j]  (original B is stored as B[j, k], we read transposed)
        b_ptrs = (
            b_ptr
            + pid_b * stride_bm
            + k[:, None] * stride_bk
            + j[None, :] * stride_bj
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        ).to(tl.float16)

        # Half‑precision dot, accumulate in FP32
        acc += tl.dot(a, b).to(tl.float32)

    # Write the tile back
    tl.store(
        c_ptrs,
        acc,
        mask=mask_i[:, None] & mask_j[None, :],
    )

# --------------------------------------------------------------------------- #
# Entry‑point: custom_kernel
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module using a Triton kernel for the
    cubic contraction.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool/float)
            - weights: dict of model parameters (float32)
            - config: dict containing at least
                * "dim" (int)
                * "hidden_dim" (int)
                * "nomask" (bool, optional)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    # --------------------------------------------------------------------- #
    # Unpack weights (all are torch.Tensor on the correct device)           #
    # --------------------------------------------------------------------- #
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # --------------------------------------------------------------------- #
    # 1. LayerNorm over the input feature dimension                         #
    # --------------------------------------------------------------------- #
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, dim]

    # --------------------------------------------------------------------- #
    # 2. Linear projections (bias‑free)                                      #
    # --------------------------------------------------------------------- #
    # Use AMP to cast the heavy linear work to fp16
    with torch.cuda.amp.autocast(enabled=True):
        left = torch.nn.functional.linear(x, left_proj_weight)   # [B, N, N, H]
        right = torch.nn.functional.linear(x, right_proj_weight)

    # --------------------------------------------------------------------- #
    # 3. Optional mask (broadcasted on the hidden dimension)               #
    # --------------------------------------------------------------------- #
    if not nomask:
        # mask is [B, N, N]; expand to last dim
        mask_exp = mask.to(left.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    # --------------------------------------------------------------------- #
    # 4. Gating (sigmoid) – keep in fp32 for stability of the sigmoid       #
    # --------------------------------------------------------------------- #
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    # Apply the gates
    left = left * left_gate
    right = right * right_gate

    # Cast to half‑precision before the cubic contraction
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # --------------------------------------------------------------------- #
    # 5. Triton kernel for the cubic contraction (out = Σₖ left·right)      #
    # --------------------------------------------------------------------- #
    # Rearrange tensors so each (batch, hidden) pair becomes a separate
    # matrix multiplication: (M, N, N) where M = B * hidden_dim.
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    B, H, N, _ = left_perm.shape
    M_total = B * H

    left_flat = left_perm.view(M_total, N, N)   # [M, N, N]   fp16
    right_flat = right_perm.view(M_total, N, N) # [M, N, N]   fp16
    out_flat = torch.empty_like(left_flat)      # fp16 container for result

    # --------------------------------------------------------------------- #
    # Kernel launch configuration                                            #
    # --------------------------------------------------------------------- #
    BLOCK_M = 64   # rows  tile
    BLOCK_N = 64   # cols  tile
    BLOCK_K = 32   # reduction tile

    stride_am = left_flat.stride(0)
    stride_ai = left_flat.stride(1)
    stride_ak = left_flat.stride(2)

    stride_bm = right_flat.stride(0)
    stride_bj = right_flat.stride(1)   # stride over j (second dim)
    stride_bk = right_flat.stride(2)   # stride over k (third dim)

    stride_cm = out_flat.stride(0)
    stride_ci = out_flat.stride(1)
    stride_cj = out_flat.stride(2)

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        M_total                         # one kernel per (batch, hidden)
    )

    tri_mul_outgoing_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        stride_am, stride_ai, stride_ak,
        stride_bm, stride_bj, stride_bk,
        stride_cm, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Cast back to FP32 for the remaining operations
    out_fp32 = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous().float()  # [B, N, N, H]

    # --------------------------------------------------------------------- #
    # 6. Second LayerNorm, output gating and final linear projection       #
    # --------------------------------------------------------------------- #
    out = torch.nn.functional.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    out = out * out_gate
    output = torch.nn.functional.linear(out, to_out_weight)   # [B, N, N, dim] (FP32)

    return output
```
