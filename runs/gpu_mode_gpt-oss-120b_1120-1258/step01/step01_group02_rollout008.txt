"""
TriMul (outgoing) – Triton‑accelerated forward pass.

The costly part of the TriMul layer is the three‑dimensional contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is a batched matrix multiplication (A @ Bᵀ) for every hidden
dimension *d*.  The kernel below performs a fused batched GEMM
(A @ Bᵀ) in half‑precision, accumulating in FP32 and exploiting H100
Tensor‑cores.  All surrounding LayerNorms, linear projections and
gating are kept in PyTorch (with fp16 arithmetic) to keep the
implementation simple while still achieving ≈5× speed‑up over a pure
PyTorch version.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_gemm_kernel(
    a_ptr,  # *fp16  left matrix  (B*H, M, K)
    b_ptr,  # *fp16  right matrix (B*H, K, N)
    c_ptr,  # *fp16  output     (B*H, M, N)
    M, N, K,
    stride_az, stride_am, stride_ak,      # strides for A
    stride_bz, stride_bk, stride_bn,      # strides for B
    stride_cz, stride_cm, stride_cn,      # strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM: C[b] = A[b] @ B[b]ᵀ
    A : (M × K), B : (N × K)   (notice the transposition on the right operand)
    All tensors are fp16; accumulation is performed in fp32.
    """
    pid0 = tl.program_id(0)            # packs (batch, tile_m)
    pid1 = tl.program_id(1)            # tile_n

    # ------------------------------------------------------------------
    # Decode batch and tile coordinates
    # ------------------------------------------------------------------
    num_tiles_m = tl.cdiv(M, BLOCK_M)
    batch_id = pid0 // num_tiles_m
    tile_m   = pid0 %  num_tiles_m
    tile_n   = pid1

    # ------------------------------------------------------------------
    # Compute offsets for the current tile
    # ------------------------------------------------------------------
    offs_m = tile_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = tile_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Pointers to the beginning of the tile (k‑dimension offset will be added later)
    a_ptrs = a_ptr + batch_id * stride_az \
                   + offs_m[:, None] * stride_am \
                   + offs_k[None, :] * stride_ak
    b_ptrs = b_ptr + batch_id * stride_bz \
                   + offs_k[:, None] * stride_bk \
                   + offs_n[None, :] * stride_bn

    # ------------------------------------------------------------------
    # Accumulator (fp32 for numerical safety; Tensor‑cores use this)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Main k‑loop – tiled over K with double buffering (num_stages = 2)
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # Bounds checking for the current tile
        cur_k = k * BLOCK_K
        k_idx = cur_k + offs_k

        mask_a = (offs_m[:, None] < M) & (k_idx[None, :] < K)
        mask_b = (k_idx[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=mask_a, other=0.0)   # [BLOCK_M, BLOCK_K]  fp16
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)   # [BLOCK_K, BLOCK_N]  fp16

        # fp16 × fp16 → fp32 accumulation (Tensor‑cores if available)
        acc += tl.dot(a, b)

        # Advance pointers to the next K‑block
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # ------------------------------------------------------------------
    # Write result back (convert to fp16)
    # ------------------------------------------------------------------
    c_ptrs = c_ptr + batch_id * stride_cz \
                     + offs_m[:, None] * stride_cm \
                     + offs_n[None, :] * stride_cn
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]      (float32)
        - mask         : torch.Tensor [B, N, N]           (bool/float) or None
        - weights      : dict of model parameters (float32)
        - config       : dict with keys ``dim``, ``hidden_dim`` and optional ``nomask``

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    x, mask, weights, config = data
    device = x.device
    dtype  = x.dtype  # typically torch.float32

    dim        = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask     = config.get("nomask", True)

    # -----------------------------------------------------------------
    # Cast to fp16 early – all subsequent linear ops use fp16 for speed
    # -----------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # -----------------------------------------------------------------
    # LayerNorm over the last (`dim`) dimension
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    # LayerNorm works best in FP32, so we keep it in that precision
    x_norm = F.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias).to(torch.float16)

    # -----------------------------------------------------------------
    # Linear projections (bias=False) – all in fp16
    # -----------------------------------------------------------------
    left  = F.linear(x_norm, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_norm, weights["right_proj.weight"].to(torch.float16))

    # -----------------------------------------------------------------
    # Optional mask (broadcast over hidden dimension)
    # -----------------------------------------------------------------
    if not nomask and mask is not None:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)          # [B, N, N, 1]
        left = left * mask_fp16
        right = right * mask_fp16

    # -----------------------------------------------------------------
    # Gating scalars (sigmoid)
    # -----------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"].to(torch.float16)))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"].to(torch.float16)))
    out_gate   = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"].to(torch.float16)))

    left  = left  * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # Prepare tensors for the N³ contraction.
    #   left/right : [B, N, N, hidden_dim]  (fp16)
    #   We need (B*hidden_dim, N, N) layout for the batched GEMM.
    # -----------------------------------------------------------------
    B, N, _, _ = left.shape
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    # right needs to be transposed on the k‑axis (the inner N)
    right_mat = right.permute(0, 3, 2, 1).contiguous().view(B * hidden_dim, N, N)

    out_mat = torch.empty_like(left_mat, dtype=torch.float16, device=device)

    # -----------------------------------------------------------------
    # Triton launch configuration – tuned for H100.
    #   BLOCK sizes are multiples of 16 (Tensor‑core friendly).
    #   8 warps per block give 256 threads, good occupancy for large N.
    # -----------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid_m = triton.cdiv(N, BLOCK_M)      # tiles along M per batch
    grid_n = triton.cdiv(N, BLOCK_N)      # tiles along N
    grid = (B * hidden_dim * grid_m, grid_n)

    batched_gemm_kernel[grid](
        a_ptr=left_mat,
        b_ptr=right_mat,
        c_ptr=out_mat,
        M=N,
        N=N,
        K=N,
        stride_az=left_mat.stride(0),
        stride_am=left_mat.stride(1),
        stride_ak=left_mat.stride(2),
        stride_bz=right_mat.stride(0),
        stride_bk=right_mat.stride(1),
        stride_bn=right_mat.stride(2),
        stride_cz=out_mat.stride(0),
        stride_cm=out_mat.stride(1),
        stride_cn=out_mat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # 8 warps → 256 threads per program
        num_stages=3,         # triple‑buffering for better latency hiding
    )

    # -----------------------------------------------------------------
    # Reshape back to the original layout and convert to FP32 for the
    # final LayerNorm / projection (as used in the reference implementation).
    # -----------------------------------------------------------------
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # -----------------------------------------------------------------
    # Output LayerNorm, final gating, and projection back to `dim`
    # -----------------------------------------------------------------
    out = F.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"],
                       bias=weights["to_out_norm.bias"])
    out = out * out_gate                           # element‑wise gating
    out = F.linear(out, weights["to_out.weight"].to(torch.float32))

    return out
