"""
TriMul (outgoing) – Triton‑accelerated forward pass.

The heavy N³ contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

is implemented as a batched half‑precision GEMM (A @ B) where
A = left[b,d,i,k] and B = right[b,d,k,j].
All other operations (LayerNorms, Linear projections, gating) are
performed with PyTorch.  The kernel works on a flattened batch
dimension (batch * hidden_dim) and accumulates in FP32 for numerical
stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (C = A @ B) with fp16 inputs, fp32 accu.
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr,                     # *fp16  A (batch*H, M, K)
    b_ptr,                     # *fp16  B (batch*H, K, N)
    c_ptr,                     # *fp16  C (batch*H, M, N)
    M, N, K,                   # int32 matrix dimensions (all == seq_len)
    stride_az, stride_am, stride_ak,   # strides for A
    stride_bz, stride_bk, stride_bn,   # strides for B
    stride_cz, stride_cm, stride_cn,   # strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM: C[b] = A[b] @ B[b]   (A: M×K, B: K×N)
    The first program‑id dimension packs (batch , tile_m) and the second
    dimension is tile_n.
    """
    pid0 = tl.program_id(0)     # combined (batch, tile_m)
    pid1 = tl.program_id(1)     # tile_n

    # ------------------------------------------------------------------
    # Decode batch index and tile coordinates
    # ------------------------------------------------------------------
    num_m_tiles = tl.cdiv(M, BLOCK_M)                # tiles along M per batch
    batch_id   = pid0 // num_m_tiles
    tile_m     = pid0 %  num_m_tiles
    tile_n     = pid1

    # ------------------------------------------------------------------
    # Offsets inside the current tile
    # ------------------------------------------------------------------
    offs_m = tile_m * BLOCK_M + tl.arange(0, BLOCK_M)   # [BLOCK_M]
    offs_n = tile_n * BLOCK_N + tl.arange(0, BLOCK_N)   # [BLOCK_N]
    offs_k = tl.arange(0, BLOCK_K)                     # [BLOCK_K]

    # ------------------------------------------------------------------
    # Pointers to the beginning of the tile (k‑loop offset will be added later)
    # ------------------------------------------------------------------
    a_ptrs = a_ptr + batch_id * stride_az \
                  + offs_m[:, None] * stride_am \
                  + offs_k[None, :] * stride_ak
    b_ptrs = b_ptr + batch_id * stride_bz \
                  + offs_k[:, None] * stride_bk \
                  + offs_n[None, :] * stride_bn

    # ------------------------------------------------------------------
    # Accumulator in FP32
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Main k‑loop
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # Mask for out‑of‑bounds loads
        mask_a = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        mask_b = (offs_k[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=mask_a, other=0.0)   # [BLOCK_M, BLOCK_K] fp16
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)   # [BLOCK_K, BLOCK_N] fp16

        # fp16 ⋅ fp16 → fp32 accumulate
        acc += tl.dot(a, b)

        # Advance pointers to next k‑block
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # ------------------------------------------------------------------
    # Store result (cast back to fp16)
    # ------------------------------------------------------------------
    c_ptrs = c_ptr + batch_id * stride_cz \
                    + offs_m[:, None] * stride_cm \
                    + offs_n[None, :] * stride_cn
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)

def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)

        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask         : torch.Tensor of shape [B, N, N] (bool/float) or None
        - weights      : dict of model parameters (float32)
        - config       : dict with keys
                         {"dim": int,
                          "hidden_dim": int,
                          "nomask": bool (optional)}

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype  = input_tensor.dtype

    dim        = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask     = config.get("nomask", True)

    # ------------------------------------------------------------------
    # LayerNorm over the `dim` channel
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # ------------------------------------------------------------------
    # Linear projections (bias‑free)
    # ------------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # Optional masking (broadcast on hidden dimension)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f = mask.to(dtype).unsqueeze(-1)      # [B, N, N, 1]
        left  = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # Gating scalars (sigmoid after linear)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # ------------------------------------------------------------------
    # Apply gates to the projected tensors
    # ------------------------------------------------------------------
    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Convert to half‑precision before the O(N³) contraction
    # ------------------------------------------------------------------
    left_fp16  = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    B, N, _, _ = left_fp16.shape               # B, seq_len, seq_len, hidden_dim

    # ------------------------------------------------------------------
    # Rearrange to (batch*head, M, K) layout required by the kernel.
    #   left : (B, H, M, K)   where M=i, K=k
    #   right: (B, H, K, N)   where K=k, N=j  (already transposed)
    # ------------------------------------------------------------------
    left_mat  = left_fp16.permute(0, 3, 1, 2).contiguous()          # (B, H, M, K)
    right_mat = right_fp16.permute(0, 3, 2, 1).contiguous()         # (B, H, K, N)

    # Flatten the head dimension into the batch dimension for the kernel
    left_mat  = left_mat.view(B * hidden_dim, N, N)
    right_mat = right_mat.view(B * hidden_dim, N, N)

    # Allocate output buffer (half‑precision, accumulation stays FP32)
    out_mat = torch.empty_like(left_mat, dtype=torch.float16, device=device)

    # ------------------------------------------------------------------
    # Triton launch configuration
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid_m = triton.cdiv(N, BLOCK_M)      # tiles along M per head
    grid_n = triton.cdiv(N, BLOCK_N)      # tiles along N
    # pid0 packs (batch*head) * grid_m  -> total grid dimension for program_id(0)
    grid = (B * hidden_dim * grid_m, grid_n)

    batched_gemm_kernel[grid](
        a_ptr=left_mat,
        b_ptr=right_mat,
        c_ptr=out_mat,
        M=N,
        N=N,
        K=N,
        stride_az=left_mat.stride(0),
        stride_am=left_mat.stride(1),
        stride_ak=left_mat.stride(2),
        stride_bz=right_mat.stride(0),
        stride_bk=right_mat.stride(1),
        stride_bn=right_mat.stride(2),
        stride_cz=out_mat.stride(0),
        stride_cm=out_mat.stride(1),
        stride_cn=out_mat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # tuned for H100
        num_stages=4,
    )

    # ------------------------------------------------------------------
    # Reshape back to (B, N, N, hidden_dim) and promote to float32
    # ------------------------------------------------------------------
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)

    # ------------------------------------------------------------------
    # Output LayerNorm, gating and final projection back to `dim`
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    out = F.layer_norm(out, (hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias)

    out = out * out_gate                     # element‑wise gating

    out = F.linear(out, weights["to_out.weight"])   # (dim, hidden_dim)

    return out
