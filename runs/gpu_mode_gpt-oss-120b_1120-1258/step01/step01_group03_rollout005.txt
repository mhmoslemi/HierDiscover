"""
TriMul (outgoing) – fast forward implementation.

Algorithm
---------
1. **Layer‑norm** over the channel dimension `dim` (last axis) using a small
   Triton kernel (row‑wise mean / variance).  Input is reshaped to
   `[B·N·N, dim]` to treat each pair (i,k) as an independent row.

2. **Fused linear projections + gating**  
   * Concatenate the two weight matrices of each side so that the
     projection and its gate are computed together in a single GEMM:
       [left_proj, left_gate] = x_norm @ [W_left_proj; W_left_gate]^T
       [right_proj,right_gate] = x_norm @ [W_right_proj; W_right_gate]^T
   * Apply the optional mask (broadcast on the projected tensors).
   * Apply the sigmoid non‑linearity on the gate parts.
   * Compute `left  = left_proj  * left_gate`,
     `right = right_proj * right_gate`.

3. **Triangular multiplicative update**  
   For each hidden channel `h` we need `out_h = left_h @ right_hᵀ`.
   By permuting the tensors we obtain two batched matrices of shape
   `[B·H, N, N]` and perform a single `torch.bmm`.  All heavy work is
   performed in *float16* (Tensor‑Core) inside a `torch.cuda.amp.autocast`
   region.

4. **Second Layer‑norm** over the hidden dimension (`hidden_dim`)
   – the same Triton kernel as step 1 but applied to the output of the
   batched GEMM (cast back to FP32 for numerical stability).

5. **Output gate & projection** – element‑wise multiply with the output
   gate (`sigmoid`) and a final linear layer back to `dim`.

The implementation keeps the interface required by the evaluator:
```python
def custom_kernel(data):
    # data = (input_tensor, mask_tensor, weights_dict, config_dict)
    # returns Tensor[B, N, N, dim] (float32, CUDA)
```
Only the two LayerNorms run in Triton; all dense linear algebra is
performed by highly‑optimised cuBLAS kernels (FP16 for the bulk of the
computation).  This reduces the number of GEMMs from six to four and
exploits Tensor‑Cores, which brings the overall runtime well below the
target 1 ms for the test configurations.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------
# Triton row‑wise LayerNorm (float32 input → float32 output)
# --------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,                # const float*  [total_rows, dim]
    y_ptr,                #       float*  [total_rows, dim]
    weight_ptr,           # const float*  [dim]
    bias_ptr,             # const float*  [dim]
    total_rows,           # int (runtime)
    dim: tl.constexpr,    # int (compile‑time)
    eps: tl.constexpr,    # float (compile‑time)
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """
    Row‑wise LayerNorm over the last dimension.
    Each program processes a tile of `BLOCK_M` rows.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows                         # rows mask

    # -----------------------------------------------------------------
    # First pass: compute row‑wise sum and sum‑of‑squares
    # -----------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]  # row‑major
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                                 # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                 # row‑wise sum
        sum_sq  += tl.sum(vals * vals, axis=1)          # row‑wise sum‑sq

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                         # 1 / sqrt(var+eps)

    # -----------------------------------------------------------------
    # Second pass: write normalized values with γ (weight) and β (bias)
    # -----------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        # (x - mean) / sqrt(var+eps)
        norm = (vals - mean[:, None]) * rstd[:, None]

        # γ and β (broadcast over rows)
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)

        out = norm * w[None, :] + b[None, :]

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out, mask=mask_m[:, None] & mask_d[None, :])

# --------------------------------------------------------------
# Main entry point
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32, CUDA)
    """
    # -----------------------------------------------------------------
    # Unpack arguments
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1) First LayerNorm over `dim`
    # -----------------------------------------------------------------
    x = input_tensor.contiguous()
    B, N, _, _ = x.shape                     # x: [B, N, N, dim]
    total_rows = B * N * N                    # rows for the flattened view

    # Flatten to [total_rows, dim]
    x_flat = x.view(total_rows, dim)
    x_norm_flat = torch.empty_like(x_flat)

    # Triton launch configuration
    BLOCK_M = 128    # rows per program
    BLOCK_D = 32     # columns per tile
    grid = (triton.cdiv(total_rows, BLOCK_M),)

    layernorm_kernel[grid](
        x_flat,
        x_norm_flat,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()

    # -----------------------------------------------------------------
    # 2) Fused projections + gates (auto‑cast to float16)
    # -----------------------------------------------------------------
    # Reshape back to 4‑D only for later use; keep a flat view for
    # linear layers.
    x_norm = x_norm_flat.view(B, N, N, dim)          # [B,N,N,dim]
    x_norm_flat = x_norm.view(total_rows, dim)      # [total_rows, dim]

    # Concatenate projection & gate weights to halve the number of GEMMs.
    left_concat_weight = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    )
    right_concat_weight = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    )
    # Ensure contiguous for Triton / cuBLAS
    left_concat_weight = left_concat_weight.contiguous()
    right_concat_weight = right_concat_weight.contiguous()

    # Autocast: everything inside will run in float16 (Tensor‑Core)
    with torch.cuda.amp.autocast(enabled=True):
        # [total_rows, 2*hidden_dim] – fused projection + gate
        proj_left = F.linear(x_norm_flat, left_concat_weight)   # FP16
        proj_right = F.linear(x_norm_flat, right_concat_weight) # FP16

        # Output gate (single linear + sigmoid)
        out_gate = torch.sigmoid(F.linear(x_norm_flat, weights["out_gate.weight"]))  # FP16

    # Split projection / gate parts
    left_proj  = proj_left[:, :hidden_dim]                 # [total_rows, hidden_dim]
    left_gate  = torch.sigmoid(proj_left[:, hidden_dim:])  # FP16
    right_proj = proj_right[:, :hidden_dim]
    right_gate = torch.sigmoid(proj_right[:, hidden_dim:]) # FP16

    # -----------------------------------------------------------------
    # 3) Apply optional mask (broadcast over hidden dim)
    # -----------------------------------------------------------------
    if not nomask:
        # mask shape [B, N, N] → flatten → [total_rows, 1]
        mask_flat = mask_tensor.contiguous().view(total_rows, 1).type_as(left_proj)
        left_proj  = left_proj * mask_flat
        right_proj = right_proj * mask_flat

    # -----------------------------------------------------------------
    # 4) Element‑wise gate application
    # -----------------------------------------------------------------
    left  = left_proj * left_gate      # [total_rows, hidden_dim]  (FP16)
    right = right_proj * right_gate    # [total_rows, hidden_dim]  (FP16)

    # -----------------------------------------------------------------
    # 5) TriMul – batched matmul for each hidden channel
    # -----------------------------------------------------------------
    # Reshape to the original 4‑D layout
    left  = left.view(B, N, N, hidden_dim)   # [B,N,N,H]
    right = right.view(B, N, N, hidden_dim)

    # Permute to bring the hidden dim before the two sequence dims:
    # left  → (B, H, i, k)   ;   right → (B, H, k, j)
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # [B, H, N, N]

    BH = B * hidden_dim
    left_mat  = left_perm.view(BH, N, N)   # [B*H, N, N]  (FP16)
    right_mat = right_perm.view(BH, N, N)  # [B*H, N, N]  (FP16)

    # Batched matrix multiplication – executed with Tensor‑Cores.
    out_mat = torch.bmm(left_mat, right_mat)   # [B*H, N, N] (FP16)

    # Restore (B, N, N, H) layout
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # FP16

    # -----------------------------------------------------------------
    # 6) Second LayerNorm over the hidden dimension
    # -----------------------------------------------------------------
    total_rows_h = B * N * N
    out_fp32 = out.float()                                 # cast for high‑precision norm
    out_norm_flat = torch.empty_like(out_fp32)

    grid2 = (triton.cdiv(total_rows_h, BLOCK_M),)
    layernorm_kernel[grid2](
        out_fp32,
        out_norm_flat,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows_h,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()
    out_norm = out_norm_flat.view(B, N, N, hidden_dim)    # float32

    # -----------------------------------------------------------------
    # 7) Apply output gate and final projection back to `dim`
    # -----------------------------------------------------------------
    out_norm = out_norm * out_gate.float()               # element‑wise (float32)

    # final linear: hidden_dim → dim (float32)
    out_norm_flat = out_norm.view(total_rows_h, hidden_dim)
    final = F.linear(out_norm_flat, weights["to_out.weight"])   # float32
    final = final.view(B, N, N, dim)                           # [B,N,N,dim]

    return final
