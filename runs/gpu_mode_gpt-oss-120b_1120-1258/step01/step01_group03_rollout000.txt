custom_kernel
=============
Implementation of the “outgoing” TriMul operator (AlphaFold‑3) in a
high‑performance fashion.

Algorithm
---------
1. **Layer‑norm over the channel dimension (`dim`).**  
   A small Triton kernel computes per‑row mean/variance in FP32

2. **Fused linear projections + gating.**

   (broadcast) to `left` and `right`.
3. **TriMul (the O(N³) kernel).**  
   For each hidden channel we need
        out_h[i, j] = Σ_k left_h[i, k] * right_h[j, k] .
   This is exactly a batched matrix‑multiply `L_h @ R_hᵀ`.  By
   permuting the tensors we collapse the batch and channel dimensions
   (`B*H` batch size) and call `torch.bmm`, which on an H100 runs
   as a fused Tensor‑Core GEMM in FP16.
4. **Second Layer‑norm over the hidden dimension (`hidden_dim`).**  
   The same Triton kernel is reused.
5. **Final gating + output projection.**  
   The hidden‑dim tensor is multiplied by the output gate, projected
   back to `dim` by a fused FP16 linear layer and cast to FP32.

The implementation uses FP16 for all heavy matrix‑multiply
operations while keeping LayerNorm in FP32 for numerical stability.
Only the LayerNorm kernels are written in Triton; the remaining
operations rely on highly‑optimised PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# -------------------------------------------------------------
# Triton row‑wise LayerNorm (float32 input / output)
# -------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,                # *const float*   [total_rows, dim]
    y_ptr,                # *mut   float*   [total_rows, dim]
    weight_ptr,           # *const float*   [dim]
    bias_ptr,             # *const float*   [dim]
    total_rows,           # int (runtime)
    dim: tl.constexpr,    # int (compile‑time)
    eps: tl.constexpr,    # float (compile‑time)
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """Row‑wise LayerNorm for a 2‑D tensor."""
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows                         # bool mask for rows

    # -----------------------------------------------------------------
    # First pass: compute sum and sum of squares (mean & variance)
    # -----------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                               # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                # row‑wise sum
        sum_sq  += tl.sum(vals * vals, axis=1)         # row‑wise sum of squares

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                          # 1 / sqrt(var+eps)

    # -----------------------------------------------------------------
    # Second pass: write normalized values with γ (weight) and β (bias)
    # -----------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        # (x - mean) / sqrt(var + eps)
        norm = (vals - mean[:, None]) * rstd[:, None]

        # γ and β (broadcasted over rows)
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)

        out = norm * w[None, :] + b[None, :]

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out, mask=mask_m[:, None] & mask_d[None, :])


# -------------------------------------------------------------
# Entry point
# -------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor],
                               Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module.

    Args:
        data: tuple containing
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N]   (bool/int)
            - weights_dict : mapping of weight names to tensors
            - config_dict  : dictionary with keys
                * "dim"        : int
                * "hidden_dim": int
                * "nomask"    : bool (True ⇒ no mask)

    Returns:
        Tensor of shape [B, N, N, dim] (float32) on CUDA.
    """
    # -----------------------------------------------------------------
    # Unpack arguments
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # Prepare shape information
    # -----------------------------------------------------------------
    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N                       # rows for a flattened view

    # -----------------------------------------------------------------
    # 1) LayerNorm over the channel dimension (dim) – Triton kernel
    # -----------------------------------------------------------------
    x = input_tensor.contiguous()
    x_flat = x.view(total_rows, dim)                     # [total_rows, dim]
    x_norm_flat = torch.empty_like(x_flat)               # FP32 output

    # launch kernel (tune block sizes if needed)
    BLOCK_M = 128
    BLOCK_D = 64
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm_flat,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()
    # reshape back to 4‑D tensor (still FP32)
    x_norm = x_norm_flat.view(B, N, N, dim)               # [B, N, N, dim]

    # -----------------------------------------------------------------
    # 2) Fused linear projections + gating (single GEMM, FP16)
    # -----------------------------------------------------------------
    # Cast to half‑precision for the heavy matrix‑multiplies.
    x_norm_fp16 = x_norm.view(total_rows, dim).to(torch.float16)

    # Concatenate all five weight matrices (out_features, in_features)
    # Shape after cat: (5*hidden_dim, dim)
    w_left_proj = weights["left_proj.weight"]
    w_right_proj = weights["right_proj.weight"]
    w_left_gate = weights["left_gate.weight"]
    w_right_gate = weights["right_gate.weight"]
    w_out_gate = weights["out_gate.weight"]
    combined_weight_fp16 = torch.cat(
        [w_left_proj, w_left_gate, w_right_proj, w_right_gate, w_out_gate],
        dim=0,
    ).to(torch.float16)                                    # (5*H, dim)

    # Single fused linear
    fused_out = F.linear(x_norm_fp16, combined_weight_fp16)  # [total_rows, 5*H]

    # Split the result
    left_proj, left_gate, right_proj, right_gate, out_gate = torch.split(
        fused_out, hidden_dim, dim=1
    )                                                       # each [total_rows, H]

    # Sigmoids for gates (still half‑precision)
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    # Apply element‑wise gating
    left = left_proj * left_gate          # [total_rows, H]
    right = right_proj * right_gate

    # -----------------------------------------------------------------
    # 3) Optional mask (broadcast over the hidden dimension)
    # -----------------------------------------------------------------
    if not nomask:
        # mask shape: [B, N, N] → [total_rows, 1]
        mask_flat = mask_tensor.contiguous().view(total_rows, 1).to(left.dtype)
        left = left * mask_flat
        right = right * mask_flat

    # -----------------------------------------------------------------
    # 4) TriMul: batched GEMM L @ Rᵀ (FP16)
    # -----------------------------------------------------------------
    # reshape to [B, N, N, H]
    left = left.view(B, N, N, hidden_dim)
    right = right.view(B, N, N, hidden_dim)

    # bring hidden dimension to the batch axis
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 2, 1).contiguous() # [B, H, N, N] (transpose k↔j)

    BH = B * hidden_dim
    left_mat = left_perm.view(BH, N, N)   # [B*H, N, N]
    right_mat = right_perm.view(BH, N, N) # [B*H, N, N]

    # Batched matrix multiplication (Tensor Core FP16)
    out_mat = torch.bmm(left_mat, right_mat)            # [B*H, N, N]

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # FP16

    # -----------------------------------------------------------------
    # 5) Second LayerNorm over hidden_dim (FP32)
    # -----------------------------------------------------------------
    # Convert to FP32 for the LayerNorm kernel
    out_fp32 = out.to(torch.float32)
    out_flat = out_fp32.view(total_rows, hidden_dim)       # [total_rows, H]
    out_norm_flat = torch.empty_like(out_flat)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm_flat,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()

    out_norm = out_norm_flat.view(B, N, N, hidden_dim)     # FP32

    # -----------------------------------------------------------------
    # 6) Apply final output gate and project back to `dim`
    # -----------------------------------------------------------------
    out_norm = out_norm * out_gate.to(torch.float32)       # element‑wise

    # Final linear projection (hidden_dim → dim) in FP16 for speed
    out_norm_fp16 = out_norm.view(total_rows, hidden_dim).to(torch.float16)
    w_to_out_fp16 = weights["to_out.weight"].to(torch.float16)   # shape (dim, hidden_dim)

    final_fp16 = F.linear(out_norm_fp16, w_to_out_fp16)          # [total_rows, dim]
    final = final_fp16.to(torch.float32).view(B, N, N, dim)      # FP32 output

    return final
```
