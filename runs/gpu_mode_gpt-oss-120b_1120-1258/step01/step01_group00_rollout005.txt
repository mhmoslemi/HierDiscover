"""
TriMul (outgoing) forward pass – Triton‑accelerated N³ reduction.

Algorithm
---------
1. Layer‑norm over the input feature dimension (dim).
2. Two linear projections (left / right) → shape [B, S, S, H].
3. (Optional) element‑wise mask, then gated element‑wise multiplication.
4. Cast left/right to FP16 and reshape to [B, H, S, S] (hidden dimension
   contiguous) so we can treat each hidden slice as an independent matrix.
5. Batched GEMM C = A @ Bᵀ for every (batch, hidden) pair using a custom
   Triton kernel.  The kernel tiles the multiplication with
   BLOCK_M×BLOCK_N×BLOCK_K tiles, accumulates in FP32 and writes the result
   in FP32.
6. View the result as [B, S, S, H] (no copy) and apply a second LayerNorm
   over the hidden dimension (implemented with mean/var to avoid an extra
   contiguous view).  Apply the out‑gate and a final linear projection back
   to the original feature dimension.
The heavy N³ work is entirely performed by the Triton kernel, while the
lightweight linear / norm / gate operations stay in PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------------------- #
# Triton kernel: batched GEMM for the TriMul reduction
# --------------------------------------------------------------------------- #
@triton.jit
def _trimul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # matrix dimensions (all == seq_len)
    stride_am, stride_ah, stride_ai, stride_ak,   # A strides: batch, hidden, i, k
    stride_ba, stride_bh, stride_bk, stride_bn,   # B strides: batch, hidden, k, j
    stride_cm, stride_ch, stride_ci, stride_cj,   # C strides: batch, hidden, i, j
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    hidden_dim: tl.constexpr,
):
    """
    Compute C = A @ Bᵀ for each (batch, hidden) slice.
    A : (batch, hidden, M, K) – fp16
    B : (batch, hidden, N, K) – fp16 (interpreted as K×N)
    C : (batch, hidden, M, N) – fp32
    """
    pid_m = tl.program_id(0)   # tile row
    pid_n = tl.program_id(1)   # tile column
    pid_z = tl.program_id(2)   # combined (batch * hidden)

    # ---- decode batch and hidden indices ----
    batch_idx = pid_z // hidden_dim
    hidden_idx = pid_z - batch_idx * hidden_dim

    # base pointers for this slice
    a_base = a_ptr + batch_idx * stride_am + hidden_idx * stride_ah
    b_base = b_ptr + batch_idx * stride_ba + hidden_idx * stride_bh
    c_base = c_ptr + batch_idx * stride_cm + hidden_idx * stride_ch

    # ---- thread‑local offsets ----
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---- tiled K loop ----
    for start_k in range(0, K, BLOCK_K):
        cur_k = start_k + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # Load A tile  (M × K)
        a_ptrs = a_base + (offs_m[:, None] * stride_ai
                           + cur_k[None, :] * stride_ak)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # Load B tile  (K × N)  – note B is accessed as (k, j)
        b_ptrs = b_base + (cur_k[:, None] * stride_bk
                           + offs_n[None, :] * stride_bn)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # Accumulate the dot product (uses Tensor‑Cores for FP16×FP16→FP32)
        acc += tl.dot(a, b)

    # ---- write back ----
    c_ptrs = c_base + (offs_m[:, None] * stride_ci
                       + offs_n[None, :] * stride_cj)
    tl.store(c_ptrs, acc,
             mask=mask_m[:, None] & mask_n[None, :])

# --------------------------------------------------------------------------- #
# Triton‑accelerated TriMul forward
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, S, S, C] (float32)
            - mask_tensor : torch.Tensor of shape [B, S, S]  (bool/float) or None
            - weights_dict : dict with all layer‑norm and linear weights (already on device)
            - config_dict  : {'dim': C, 'hidden_dim': H, 'nomask': bool, ...}

    Returns:
        torch.Tensor of shape [B, S, S, C] (float32)
    """
    # -------------------------------------------------- #
    # 0. unpack
    # -------------------------------------------------- #
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, S, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -------------------------------------------------- #
    # 1. input layer‑norm (dim is last dim → contiguous)
    # -------------------------------------------------- #
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )   # [B, S, S, dim]  (float32)

    # -------------------------------------------------- #
    # 2. linear projections (no bias) → left / right
    # -------------------------------------------------- #
    left  = F.linear(x, weights["left_proj.weight"])   # [B, S, S, H]
    right = F.linear(x, weights["right_proj.weight"]) # [B, S, S, H]

    # -------------------------------------------------- #
    # 3. optional mask (broadcast on hidden dim)
    # -------------------------------------------------- #
    if not nomask and mask_tensor is not None:
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)   # [B, S, S, 1]
        left  = left * mask
        right = right * mask

    # -------------------------------------------------- #
    # 4. gating (sigmoid) and gating multiplication
    # -------------------------------------------------- #
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B, S, S, H]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))  # [B, S, S, H]
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))    # [B, S, S, H]

    left  = left * left_gate
    right = right * right_gate

    # -------------------------------------------------- #
    # 5. cast to half and reshape for the Triton GEMM
    # -------------------------------------------------- #
    left_h  = left.to(torch.float16).contiguous()
    right_h = right.to(torch.float16).contiguous()

    # bring hidden dimension to the 2nd axis: [B, H, S, S]
    left_h  = left_h.permute(0, 3, 1, 2).contiguous()
    right_h = right_h.permute(0, 3, 1, 2).contiguous()

    # output buffer (FP32 accumulator)
    out_fp32 = torch.empty_like(left_h, dtype=torch.float32, device=device)

    # -------------------------------------------------- #
    # 6. launch the Triton kernel
    # -------------------------------------------------- #
    # Tuned block sizes for H100 (FP16×FP16 → FP32)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 128   # larger K‑tile reduces loop overhead

    grid = (
        triton.cdiv(S, BLOCK_M),          # rows  (i)
        triton.cdiv(S, BLOCK_N),          # cols  (j)
        B * hidden_dim                    # (batch * hidden)
    )

    _trimul_gemm_kernel[grid](
        a_ptr=left_h,
        b_ptr=right_h,
        c_ptr=out_fp32,
        M=S,
        N=S,
        K=S,
        stride_am=left_h.stride(0),
        stride_ah=left_h.stride(1),
        stride_ai=left_h.stride(2),
        stride_ak=left_h.stride(3),
        stride_ba=right_h.stride(0),
        stride_bh=right_h.stride(1),
        stride_bk=right_h.stride(3),   # k‑dimension (last)
        stride_bn=right_h.stride(2),   # j‑dimension (the i‑axis after permute)
        stride_cm=out_fp32.stride(0),
        stride_ch=out_fp32.stride(1),
        stride_ci=out_fp32.stride(2),
        stride_cj=out_fp32.stride(3),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        hidden_dim=hidden_dim,
        num_warps=8,   # leverage Tensor‑Core warps on H100
    )

    # -------------------------------------------------- #
    # 7. view output as [B, S, S, H] (no copy) and hidden‑norm
    # -------------------------------------------------- #
    # shape view (non‑contiguous)
    out_view = out_fp32.permute(0, 2, 3, 1)   # [B, S, S, H]

    # manual LayerNorm over the last dimension (H) – avoids an extra contiguous copy
    eps = 1e-5
    mean = out_view.mean(dim=-1, keepdim=True)
    var = out_view.var(dim=-1, unbiased=False, keepdim=True)
    inv_std = torch.rsqrt(var + eps)

    # reshape weight / bias for broadcasting
    w_out = weights["to_out_norm.weight"].view(1, 1, 1, -1)
    b_out = weights["to_out_norm.bias"].view(1, 1, 1, -1)

    out_norm = (out_view - mean) * inv_std
    out_norm = out_norm * w_out + b_out   # [B, S, S, H]

    # -------------------------------------------------- #
    # 8. apply out‑gate and final projection back to dim
    # -------------------------------------------------- #
    out = out_norm * out_gate                # element‑wise gate
    out = F.linear(out, weights["to_out.weight"])   # [B, S, S, dim]

    return out
```
