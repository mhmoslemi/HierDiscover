    M, N, K,
    stride_Ab, stride_Ar, stride_Ac,
    stride_Bb, stride_Br, stride_Bc,
    stride_Cb, stride_Cr, stride_Cc,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,

    pid_n = tl.program_id(1)
    pid_b = tl.program_id(2)
    # Compute offsets

    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_n = offs_n < N

def custom_kernel(data):
    import torch
    import triton
    import triton.language as tl

    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # 1. LayerNorm
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )
    # Cast to half for speed
    x_half = x.to(torch.float16)

    # 2. Linear projections
    left = torch.nn.functional.linear(x_half, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_half, weights["right_proj.weight"])

    # 3. Gates
    left_gate = torch.nn.functional.linear(x_half, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x_half, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x_half, weights["out_gate.weight"]).sigmoid()

    # 4. Optional mask preparation
    if not nomask and mask is not None:
        mask_f = mask.to(torch.float16).unsqueeze(-1)
    else:
        mask_f = torch.ones_like(left, dtype=torch.float16)

    # 5. Fuse mask and gates via Triton kernel
    # Ensure tensors are contiguous
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    mask_f = mask_f.contiguous()

    B, N, _, H = left.shape  # B,N,N,hidden

    total_elements = B * N * N * H

    BLOCK = 1024  # adjust maybe 2048
    num_blocks = (total_elements + BLOCK - 1) // BLOCK

    @triton.jit
    def _gate_mask_kernel(proj_ptr, gate_ptr, mask_ptr, out_ptr, N_ELEM: tl.constexpr):
        pid = tl.program_id(0)
        offs = pid * BLOCK + tl.arange(0, BLOCK)
        mask = offs < N_ELEM
        proj = tl.load(proj_ptr + offs, mask=mask)
        gate = tl.load(gate_ptr + offs, mask=mask)
        mask_val = tl.load(mask_ptr + offs, mask=mask)
        out = proj * gate * mask_val
        tl.store(out_ptr + offs, out, mask=mask)

    # Apply to left and right separately
    left_ptr = left.data_ptr()
    left_gate_ptr = left_gate.data_ptr()
    mask_ptr = mask_f.data_ptr()
    # Use same left buffer as output (in-place)
    _gate_mask_kernel[(num_blocks,)](
        left_ptr,
        left_gate_ptr,
        mask_ptr,
        left_ptr,
        total_elements,
        BLOCK=BLOCK,
    )

    # Right
    right_ptr = right.data_ptr()
    right_gate_ptr = right_gate.data_ptr()
    _gate_mask_kernel[(num_blocks,)](
        right_ptr,
        right_gate_ptr,
        mask_ptr,
        right_ptr,
        total_elements,
        BLOCK=BLOCK,
    )
    # At this point left & right are gated and masked (in-place)

    # 6. Heavy matmul via batched GEMM
    # Permute and reshape
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(-1, N, N)  # (B*H,N,N)
    right_T = right.permute(0, 3, 2, 1).contiguous().view(-1, N, N)   # transpose seq axes

    # Use torch.bmm
    out_flat = torch.bmm(left_flat, right_T)  # fp16 output

    # Reshape back
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # 7. Final layernorm, out gate, projection
    out = torch.nn.functional.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    # multiply by out_gate (cast to float32)
    out = out * out_gate.to(torch.float32)
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

def custom_kernel(data):
    """
    <docstring>
    """
    import torch
    import triton
    import triton.language as tl

    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # 1. Input LayerNorm
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )
    x = x.to(torch.float16)

    # 2. Linear projections
    left_proj = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right_proj = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # 3. Gates
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    # 4. Prepare mask (flattened)
    B, N, _, _ = left_proj.shape
    if not nomask and mask is not None:
        mask_fp = mask.to(torch.float16).contiguous()
    else:
        mask_fp = torch.ones((B, N, N), dtype=torch.float16, device=device)
    mask_flat = mask_fp.view(-1)  # shape (B*N*N,)

    # Ensure contiguous memory layout for all tensors
    left_proj = left_proj.contiguous()
    right_proj = right_proj.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()

    total_elements = B * N * N * hidden   # number of elements after broadcasting over hidden

    BLOCK = 1024  # threads per program; must <= 1024
    grid = ( (total_elements + BLOCK - 1) // BLOCK, )

    # Triton kernel: fuse masking and gating for left and right tensors
    @triton.jit
    def _gate_mask_fusion_kernel(
        left_proj_ptr, left_gate_ptr,
        right_proj_ptr, right_gate_ptr,
        mask_ptr,
        left_out_ptr, right_out_ptr,
        N_ELEM: tl.constexpr,
        HIDDEN: tl.constexpr,
        BLOCK: tl.constexpr,
    ):
        pid = tl.program_id(0)
        offs = pid * BLOCK + tl.arange(0, BLOCK)
        mask = offs < N_ELEM

        # Load values
        lp = tl.load(left_proj_ptr + offs, mask=mask)
        lg = tl.load(left_gate_ptr + offs, mask=mask)
        rp = tl.load(right_proj_ptr + offs, mask=mask)
        rg = tl.load(right_gate_ptr + offs, mask=mask)

        # Broadcast mask across hidden dimension
        mask_idx = offs // HIDDEN
        m = tl.load(mask_ptr + mask_idx, mask=mask)

        # Apply mask and gates
        l_out = lp * lg * m
        r_out = rp * rg * m

        tl.store(left_out_ptr + offs, l_out, mask=mask)
        tl.store(right_out_ptr + offs, r_out, mask=mask)

    _gate_mask_fusion_kernel[grid](
        left_proj.data_ptr(),
        left_gate.data_ptr(),
        right_proj.data_ptr(),
        right_gate.data_ptr(),
        mask_flat.data_ptr(),
        left_proj.data_ptr(),    # in-place write
        right_proj.data_ptr(),
        total_elements,
        hidden,
        BLOCK=BLOCK,
    )

    # 5. Batched matrix multiplication using torch.bmm (FP16)
    # reshape to (B*hidden, N, N) and compute out = left @ right^T
    left_mat = left_proj.permute(0, 3, 1, 2).contiguous().view(-1, N, N)   # (B*H, N, N)
    right_mat = right_proj.permute(0, 3, 2, 1).contiguous().view(-1, N, N)  # (B*H, N, N) with k axis first
    out_mat = torch.bmm(left_mat, right_mat)  # FP16 result

    # reshape back to (B, N, N, hidden)
    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # 6. Final LayerNorm, out‑gate, and linear projection
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate.to(torch.float32)
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

def custom_kernel(data):
    """
    Outgoing TriMul (AlphaFold3) forward pass – highly‑optimized implementation.

    Algorithm
    ---------
    1. Layer‑norm the 4‑D input on its channel dimension (float32).
    2. Cast to fp16 and apply the two linear projections (left/right).
    3. Compute three gating tensors (left‑gate, right‑gate, out‑gate) with a
       bias‑free linear + sigmoid.
    4. Fuse **mask**, left‑gate and right‑gate with a small Triton kernel:
          left  = left_proj  * left_gate  * mask
          right = right_proj * right_gate * mask
       The mask is broadcast over the hidden dimension without extra memory.
    5. Heavy pairwise multiplicative update:
          out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
       This is a batched matrix multiplication for each hidden slice.
       We reshape to (B·H, N, N) and use torch.bmm (fp16, Tensor‑cores) which
       is far faster than a naïve Triton matmul for the tested sizes.
    6. Reshape back to [B, N, N, H] and cast to float32.
    7. Layer‑norm on the hidden dimension, multiply by the out‑gate, and
       a final bias‑free linear projection back to the original channel size.

    All heavy element‑wise work (mask + gating) is fused in a Triton kernel,
    while the N³ compute is delegated to cuBLAS via torch.bmm for maximal
    throughput on H100 GPUs.
    """
    import torch
    import triton
    import triton.language as tl

    # ----------------------------------------------------------------------
    # Unpack inputs
    # ----------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ----------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ----------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )                               # [B, N, N, dim]  (float32)

    # ----------------------------------------------------------------------
    # Cast to half for all subsequent dense work
    # ----------------------------------------------------------------------
    x = x.to(torch.float16)

    # ----------------------------------------------------------------------
    # 2. Linear projections (no bias) – fp16
    # ----------------------------------------------------------------------
    left_proj = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right_proj = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ----------------------------------------------------------------------
    # 3. Gate tensors (sigmoid) – fp16
    # ----------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()  # [B,N,N,H]

    # ----------------------------------------------------------------------
    # 4. Broadcast mask (if any) and fuse mask+gate with a Triton kernel
    # ----------------------------------------------------------------------
    B, N, _, _ = left_proj.shape                         # B,N,N,H
    if (not nomask) and mask is not None:
        mask_fp = mask.to(torch.float16).contiguous()    # [B,N,N]
    else:
        mask_fp = torch.ones((B, N, N), dtype=torch.float16, device=device)

    mask_flat = mask_fp.view(-1)                         # [B*N*N]

    # Ensure contiguous layout for all tensors that will be flattened.
    left_proj = left_proj.contiguous()
    right_proj = right_proj.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()

    total_elems = B * N * N * hidden                     # elements after broadcasting over hidden
    BLOCK = 1024                                          # ≤ max threads per program (1024 on NVIDIA)

    grid = ((total_elems + BLOCK - 1) // BLOCK,)

    @triton.jit
    def _gate_mask_fusion_kernel(
        left_proj_ptr, left_gate_ptr,
        right_proj_ptr, right_gate_ptr,
        mask_ptr,
        left_out_ptr, right_out_ptr,
        N_ELEM: tl.constexpr,
        HIDDEN: tl.constexpr,
        BLOCK: tl.constexpr,
    ):
        pid = tl.program_id(0)
        offs = pid * BLOCK + tl.arange(0, BLOCK)
        mask = offs < N_ELEM

        # Load the four tensors
        lp = tl.load(left_proj_ptr + offs, mask=mask)
        lg = tl.load(left_gate_ptr + offs, mask=mask)
        rp = tl.load(right_proj_ptr + offs, mask=mask)
        rg = tl.load(right_gate_ptr + offs, mask=mask)

        # Broadcast mask across the hidden dimension
        mask_idx = offs // HIDDEN               # integer division
        m = tl.load(mask_ptr + mask_idx, mask=mask)

        # Apply mask and gates
        l_out = lp * lg * m
        r_out = rp * rg * m

        # Store results (in‑place)
        tl.store(left_out_ptr + offs, l_out, mask=mask)
        tl.store(right_out_ptr + offs, r_out, mask=mask)

    # In‑place fusion for left and right tensors
    _gate_mask_fusion_kernel[grid](
        left_proj.data_ptr(),
        left_gate.data_ptr(),
        right_proj.data_ptr(),
        right_gate.data_ptr(),
        mask_flat.data_ptr(),
        left_proj.data_ptr(),      # overwrite with gated result
        right_proj.data_ptr(),
        total_elems,
        hidden,
        BLOCK=BLOCK,
    )

    # ----------------------------------------------------------------------
    # 5. Heavy pairwise update – batched half‑precision matrix multiply
    # ----------------------------------------------------------------------
    # reshape to (B*H, N, N) and transpose the second operand on the two pairwise axes
    left_mat  = left_proj.permute(0, 3, 1, 2).contiguous().view(-1, N, N)   # (B*H, N, K)
    right_mat = right_proj.permute(0, 3, 2, 1).contiguous().view(-1, N, N)  # (B*H, K, N)

    # Tensor‑core batched GEMM (FP16 → FP32 accumulation)
    out_mat = torch.bmm(left_mat, right_mat)                               # (B*H, N, N) fp16

    # Reshape back to [B, N, N, hidden] and convert to float32 for the rest
    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # ----------------------------------------------------------------------
    # 6. Final LayerNorm, out‑gate, and projection back to `dim`
    # ----------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate.to(torch.float32)
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
