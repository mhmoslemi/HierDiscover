def elementwise_mul_kernel(a_ptr, b_ptr, c_ptr, out_ptr, n_elements, stride_a, stride_b, stride_c, stride_out, BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offsets = block_start + tl.arange(0, BLOCK_SIZE)
    mask = offsets < n_elements
    a = tl.load(a_ptr + offsets * stride_a, mask=mask)
    b = tl.load(b_ptr + offsets * stride_b, mask=mask)
    c = tl.load(c_ptr + offsets * stride_c, mask=mask)
    out = a * b * c
    tl.store(out_ptr + offsets * stride_out, out, mask=mask)

@triton.jit
def fused_mul_kernel(a_ptr, b_ptr, c_ptr, out_ptr, n_elements,
                     stride_a: tl.constexpr, stride_b: tl.constexpr, stride_c: tl.constexpr, stride_out: tl.constexpr,
                     BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offsets = block_start + tl.arange(0, BLOCK_SIZE)
    mask = offsets < n_elements

    a = tl.load(a_ptr + offsets * stride_a, mask=mask, other=0.0)
    b = tl.load(b_ptr + offsets * stride_b, mask=mask, other=0.0)
    c = tl.load(c_ptr + offsets * stride_c, mask=mask, other=0.0)

    out = a * b * c
    # store as float16
    tl.store(out_ptr + offsets * stride_out, out.to(tl.float16), mask=mask)

import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def fused_mul_fp32_to_fp16_kernel(a_ptr, b_ptr, c_ptr, out_ptr, N, BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < N
    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    c = tl.load(c_ptr + offs, mask=mask, other=0.0)
    out = a * b * c
    tl.store(out_ptr + offs, out.to(tl.float16), mask=mask)

@triton.jit
def elementwise_mul_fp32_kernel(a_ptr, b_ptr, out_ptr, N, BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < N
    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    tl.store(out_ptr + offs, a * b, mask=mask)

def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing Triangle‑Multiplicative‑Update.
    Steps:
    1) Layer‑norm the 4‑D pair tensor.
    2) Linear projections to hidden_dim (no bias) for left/right.
    3) Sigmoid gates (left_gate, right_gate, out_gate) computed from the same normalized input.
    4) Fuse mask, left‑gate and right‑gate with a tiny Triton kernel
       (produces half‑precision tensors ready for the massive batched GEMM).
    5) Batched matrix multiplication via a single torch.bmm call:
          out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h] .
    6) Second layer‑norm over the hidden dimension.
    7) Apply the output gate using another Triton element‑wise kernel.
    8) Final linear projection back to the original channel dimension.
    All heavy GEMM work stays in FP16 (tensor cores) while accumulation
    is kept in FP32 for numerical stability.
    """
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 0) move everything to the same device, no grads
    # ------------------------------------------------------------------
    device = input_tensor.device
    with torch.no_grad():
        # ------------------------------------------------------------------
        # 1) First LayerNorm over channel dim
        # ------------------------------------------------------------------
        x = F.layer_norm(
            input_tensor,
            normalized_shape=(dim,),
            weight=weights["norm.weight"],
            bias=weights["norm.bias"],
            eps=eps,
        )  # [B,N,N,dim]

        B, N, _, _ = x.shape

        # ------------------------------------------------------------------
        # 2) Linear projections (float32)
        # ------------------------------------------------------------------
        # flatten the pair axis for efficient gemm
        x_flat = x.view(B * N * N, dim)

        # left/right projections
        left_proj = F.linear(x_flat, weights["left_proj.weight"])   # (B*N*N, hidden_dim)
        right_proj = F.linear(x_flat, weights["right_proj.weight"])

        # gates
        left_gate = torch.sigmoid(F.linear(x_flat, weights["left_gate.weight"]))
        right_gate = torch.sigmoid(F.linear(x_flat, weights["right_gate.weight"]))
        out_gate = torch.sigmoid(F.linear(x_flat, weights["out_gate.weight"]))

        # reshape back to [B,N,N,hidden_dim]
        left_proj = left_proj.view(B, N, N, hidden_dim)
        right_proj = right_proj.view(B, N, N, hidden_dim)
        left_gate = left_gate.view(B, N, N, hidden_dim)
        right_gate = right_gate.view(B, N, N, hidden_dim)
        out_gate = out_gate.view(B, N, N, hidden_dim)

        # ------------------------------------------------------------------
        # 3) Broadcast mask (if any) to hidden_dim
        # ------------------------------------------------------------------
        if mask_tensor is None:
            mask = torch.ones((B, N, N), dtype=x.dtype, device=device)
        else:
            mask = mask_tensor.to(x.dtype)
        mask = mask.unsqueeze(-1).expand(-1, -1, -1, hidden_dim).contiguous()

        # ------------------------------------------------------------------
        # 4) Fuse mask + gates with a tiny Triton kernel (produces FP16 tensors)
        # ------------------------------------------------------------------
        left = torch.empty_like(left_proj, dtype=torch.float16)
        right = torch.empty_like(right_proj, dtype=torch.float16)

        total = left_proj.numel()
        BLOCK = 131072  # 128k elements per program (tune for occupancy)
        grid = (triton.cdiv(total, BLOCK),)

        # left = left_proj * left_gate * mask   (FP32 -> FP16)
        fused_mul_fp32_to_fp16_kernel[grid](
            left_proj.view(-1),
            left_gate.view(-1),
            mask.view(-1),
            left.view(-1),
            total,
            BLOCK_SIZE=BLOCK,
        )
        # right = right_proj * right_gate * mask
        fused_mul_fp32_to_fp16_kernel[grid](
            right_proj.view(-1),
            right_gate.view(-1),
            mask.view(-1),
            right.view(-1),
            total,
            BLOCK_SIZE=BLOCK,
        )

        # free memory that is no longer needed
        del left_proj, right_proj, left_gate, right_gate, mask, x, x_flat

        # ------------------------------------------------------------------
        # 5) Batched GEMM (left @ rightᵀ) using torch.bmm in FP16
        # ------------------------------------------------------------------
        # reshape to (B*H, N, N)
        left = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
        right = right.permute(0, 3, 1, 2).contiguous()

        batch = B * hidden_dim
        left_mat = left.view(batch, N, N)               # (batch, N, K)
        right_mat = right.view(batch, N, N)             # (batch, N, K)

        # matmul: (batch, N, K) @ (batch, K, N) -> (batch, N, N)
        out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

        # reshape back to [B,N,N,hidden_dim] and cast to FP32 for later ops
        out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous().float()

        # ------------------------------------------------------------------
        # 6) Second LayerNorm over hidden_dim
        # ------------------------------------------------------------------
        out = F.layer_norm(
            out,
            normalized_shape=(hidden_dim,),
            weight=weights["to_out_norm.weight"],
            bias=weights["to_out_norm.bias"],
            eps=eps,
        )

        # ------------------------------------------------------------------
        # 7) Apply output gate (FP32) with a tiny Triton kernel
        # ------------------------------------------------------------------
        total_out = out.numel()
        grid_out = (triton.cdiv(total_out, BLOCK),)
        elementwise_mul_fp32_kernel[grid_out](
            out.view(-1),
            out_gate.view(-1),
            out.view(-1),
            total_out,
            BLOCK_SIZE=BLOCK,
        )

        # ------------------------------------------------------------------
        # 8) Final linear projection back to original dim
        # ------------------------------------------------------------------
        # flatten for linear
        out_flat = out.view(B * N * N, hidden_dim)
        out = F.linear(out_flat, weights["to_out.weight"]).view(B, N, N, dim)

        return out

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def elementwise_mul_fp32_kernel(a_ptr, b_ptr, out_ptr, N, BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < N
    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    tl.store(out_ptr + offs, a * b, mask=mask)

    Triton‑accelerated forward pass of the TriMul (outgoing) module.
    The heavy O(N³) pairwise product is performed with a single torch.bmm that
    runs in FP16 on the Tensor‑Core‑enabled H100.  A tiny Triton kernel is
    used to apply the final output‑gate elementwise.  All linear projections and
    gates are executed in FP16 to exploit tensor cores, while the two LayerNorm
    steps run in FP32 for numerical stability.
    """
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    device = input_tensor.device

    # ------------------------------------------------------------------
    # 0) disable gradients for speed
    # ------------------------------------------------------------------
    with torch.no_grad():
        # ------------------------------------------------------------------
        # 1) First LayerNorm over the channel dimension (FP32)
        # ------------------------------------------------------------------
        x = F.layer_norm(
            input_tensor,
            normalized_shape=(dim,),
            weight=weights["norm.weight"],
            bias=weights["norm.bias"],
            eps=eps,
        )  # [B, N, N, dim]

        B, N, _, _ = x.shape

        # ------------------------------------------------------------------
        # 2) Cast to FP16 for all subsequent GEMMs
        # ------------------------------------------------------------------
        x_fp16 = x.to(torch.float16).contiguous()
        # Half‑precision weight copies
        left_proj_w = weights["left_proj.weight"].to(torch.float16)
        right_proj_w = weights["right_proj.weight"].to(torch.float16)
        left_gate_w = weights["left_gate.weight"].to(torch.float16)
        right_gate_w = weights["right_gate.weight"].to(torch.float16)
        out_gate_w = weights["out_gate.weight"].to(torch.float16)

        # flatten to (B*N*N, dim) for efficient Linear
        x_flat = x_fp16.view(B * N * N, dim)

        # ------------------------------------------------------------------

        # ------------------------------------------------------------------
        left_proj = F.linear(x_flat, left_proj_w)          # (B*N*N, hidden_dim)
        right_proj = F.linear(x_flat, right_proj_w)

        left_gate = torch.sigmoid(F.linear(x_flat, left_gate_w))
        right_gate = torch.sigmoid(F.linear(x_flat, right_gate_w))
        out_gate = torch.sigmoid(F.linear(x_flat, out_gate_w))

        # reshape back to [B, N, N, hidden_dim]
        new_shape = (B, N, N, hidden_dim)
        left_proj = left_proj.view(new_shape)
        right_proj = right_proj.view(new_shape)
        left_gate = left_gate.view(new_shape)
        right_gate = right_gate.view(new_shape)
        out_gate = out_gate.view(new_shape)   # keep in FP16 for now

        # ------------------------------------------------------------------
        # 4) Apply mask (if present) and fuse with gates (torch elementwise)
        # ------------------------------------------------------------------
        if mask_tensor is None:
            mask = torch.ones((B, N, N), dtype=torch.float16, device=device)
        else:
            mask = mask_tensor.to(dtype=torch.float16, device=device)
        mask = mask.unsqueeze(-1).expand(-1, -1, -1, hidden_dim).contiguous()

        left = left_proj * left_gate * mask   # FP16 result
        right = right_proj * right_gate * mask

        # free intermediate FP16 tensors that are no longer needed
        del left_proj, right_proj, left_gate, right_gate, mask, x_fp16, x_flat

        # ------------------------------------------------------------------

        # ------------------------------------------------------------------
        # bring hidden dimension to batch dimension: (B, H, N, N)
        left = left.permute(0, 3, 1, 2).contiguous()
        right = right.permute(0, 3, 1, 2).contiguous()
        batch = B * hidden_dim
        left_mat = left.view(batch, N, N)
        right_mat = right.view(batch, N, N)

        # (batch, N, N) = (batch, N, K) @ (batch, K, N)
        out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

        # reshape back to [B, N, N, hidden_dim] and promote to FP32
        out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous().float()

        # ------------------------------------------------------------------
        # 6) Second LayerNorm over hidden_dim (FP32)
        # ------------------------------------------------------------------

            out,
            normalized_shape=(hidden_dim,),
            weight=weights["to_out_norm.weight"],
            bias=weights["to_out_norm.bias"],
            eps=eps,

            total,
            BLOCK_SIZE=BLOCK,

        # ------------------------------------------------------------------
        out_flat = out.view(B * N * N, hidden_dim)

        return out

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def elementwise_mul_fp32_kernel(a_ptr, b_ptr, out_ptr, n_elem, BLOCK_SIZE: tl.constexpr):
    """out = a * b   (both FP32).  One‐dimensional launch."""
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < n_elem
    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    tl.store(out_ptr + offs, a * b, mask=mask)


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the **outgoing** TriMul operator.

    Algorithm
    ---------
    1) Layer‑norm the 4‑D input (FP32 for stability).
    2) Cast the normalized tensor to FP16 – all subsequent GEMMs run in FP16
       on the H100 Tensor Cores.
    3) Five bias‑free linear layers (left/right projections, three gates) are
       performed with half‑precision weights.
    4) An optional mask is broadcast to the hidden dimension and fused with the
       left/right projections using ordinary PyTorch element‑wise multiplication.
    5) The heavy O(N³) pairwise product is a *batched* matrix multiplication:
           out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
       Implemented as a single ``torch.bmm`` on half‑precision tensors.
    6) Convert the result back to FP32 and apply a second LayerNorm over the
       hidden dimension.
    7) Multiply by the output gate using a tiny Triton kernel (still FP32).
    8) Final linear projection back to the original channel dimension.

    All memory‑intensive work stays in FP16, while the two LayerNorms and the
    final gating are carried out in FP32 for numerical accuracy.
    """
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    device = input_tensor.device

    with torch.no_grad():
        # -----------------------------------------------------------------
        # 1) First LayerNorm (FP32)
        # -----------------------------------------------------------------
        x = F.layer_norm(
            input_tensor,
            normalized_shape=(dim,),
            weight=weights["norm.weight"],
            bias=weights["norm.bias"],
            eps=eps,
        )  # [B, N, N, dim]

        B, N, _, _ = x.shape

        # -----------------------------------------------------------------
        # 2) Cast to FP16 – the heavy work will use Tensor Cores
        # -----------------------------------------------------------------
        x_fp16 = x.to(torch.float16).contiguous()

        # half‑precision copies of the linear weights
        left_proj_w  = weights["left_proj.weight"].to(torch.float16)
        right_proj_w = weights["right_proj.weight"].to(torch.float16)
        left_gate_w  = weights["left_gate.weight"].to(torch.float16)
        right_gate_w = weights["right_gate.weight"].to(torch.float16)
        out_gate_w   = weights["out_gate.weight"].to(torch.float16)

        # flatten for the linear layers: (B·N·N, dim)
        x_flat = x_fp16.view(B * N * N, dim)

        # -----------------------------------------------------------------
        # 3) Projections & gates (all half‑precision)
        # -----------------------------------------------------------------
        left_proj = F.linear(x_flat, left_proj_w)          # (B·N·N, hidden_dim)
        right_proj = F.linear(x_flat, right_proj_w)

        left_gate = torch.sigmoid(F.linear(x_flat, left_gate_w))
        right_gate = torch.sigmoid(F.linear(x_flat, right_gate_w))
        out_gate = torch.sigmoid(F.linear(x_flat, out_gate_w))

        # reshape back to [B, N, N, hidden_dim]
        new_shape = (B, N, N, hidden_dim)
        left_proj = left_proj.view(new_shape)
        right_proj = right_proj.view(new_shape)
        left_gate = left_gate.view(new_shape)
        right_gate = right_gate.view(new_shape)
        out_gate = out_gate.view(new_shape)          # still FP16

        # -----------------------------------------------------------------
        # 4) Optional mask – broadcast & fuse with gates (still FP16)
        # -----------------------------------------------------------------
        if mask_tensor is None:
            mask = torch.ones((B, N, N), dtype=torch.float16, device=device)
        else:
            mask = mask_tensor.to(dtype=torch.float16, device=device)
        mask = mask.unsqueeze(-1).expand(-1, -1, -1, hidden_dim).contiguous()

        left  = left_proj * left_gate * mask   # FP16
        right = right_proj * right_gate * mask

        # free memory we no longer need
        del left_proj, right_proj, left_gate, right_gate, mask, x_fp16, x_flat

        # -----------------------------------------------------------------
        # 5) Batched GEMM (half‑precision) :  out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
        # -----------------------------------------------------------------
        # move hidden dimension into the batch axis: (B, H, N, N)
        left = left.permute(0, 3, 1, 2).contiguous()
        right = right.permute(0, 3, 1, 2).contiguous()
        batch = B * hidden_dim
        left_mat = left.view(batch, N, N)          # (batch, N, K)
        right_mat = right.view(batch, N, N)        # (batch, N, K)

        # (batch, N, N) = (batch, N, K) @ (batch, K, N)
        out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

        # reshape to [B, N, N, hidden_dim] and promote to FP32 for the next LN
        out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous().float()

        # -----------------------------------------------------------------
        # 6) Second LayerNorm over the hidden dimension (FP32)
        # -----------------------------------------------------------------
        out = F.layer_norm(
            out,
            normalized_shape=(hidden_dim,),
            weight=weights["to_out_norm.weight"],
            bias=weights["to_out_norm.bias"],
            eps=eps,
        )

        # -----------------------------------------------------------------
        # 7) Apply the output gate (FP32) with a tiny Triton kernel
        # -----------------------------------------------------------------
        out_gate_fp32 = out_gate.float()
        total = out.numel()
        BLOCK = 131072               # ~128 k elements per program
        grid = (triton.cdiv(total, BLOCK),)
        elementwise_mul_fp32_kernel[grid](
            out.view(-1),
            out_gate_fp32.view(-1),
            out.view(-1),
            total,
            BLOCK_SIZE=BLOCK,
        )

        # -----------------------------------------------------------------
        # 8) Final projection back to the original channel dimension (FP32)
        # -----------------------------------------------------------------
        out_flat = out.view(B * N * N, hidden_dim)
        out = F.linear(out_flat, weights["to_out.weight"]).view(B, N, N, dim)

        return out
