@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    M, N, K,
    stride_la, stride_li, stride_lk,
    stride_rb, stride_ri, stride_rk,
    stride_ob, stride_oi, stride_oj,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    PREFETCH_K: tl.constexpr,
):
    pid = tl.program_id(0) # row block
    pid_n = tl.program_id(1) # col block
    pid_b = tl.program_id(2) # batch*hidden

    # Compute offsets
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    a_ptrs = left_ptr + pid_b * stride_la + offs_m[:, None] * stride_li
    b_ptrs = right_ptr + pid_b * stride_rb + offs_n[None, :] * stride_ri

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Prefetch first K tile
    offs_k = tl.arange(0, BLOCK_K)
    a = tl.load(a_ptrs + offs_k[None, :] * stride_lk,
                mask=mask_m[:, None] & (offs_k[None,:] < K),
                other=0.0)
    b = tl.load(b_ptrs + offs_k[:, None] * stride_rk,
                mask=(offs_k[:, None] < K) & mask_n[None,:],
                other=0.0)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # compute accumulation
        acc += tl.dot(a, b)
        # Prefetch next tile unless at last
        if k < tl.cdiv(K, BLOCK_K) - 1:
            next_offs_k = (k+1) * BLOCK_K + tl.arange(0, BLOCK_K)
            a = tl.load(a_ptrs + next_offs_k[None, :] * stride_lk,
                        mask=mask_m[:, None] & (next_offs_k[None, :] < K),
                        other=0.0)
            b = tl.load(b_ptrs + next_offs_k[:, None] * stride_rk,
                        mask=(next_offs_k[:, None] < K) & mask_n[None,:],
                        other=0.0)

    c_ptr = out_ptr + pid_b * stride_ob + offs_m[:, None] * stride_oi + offs_n[None, :] * stride_oj
    tl.store(c_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data

    device = input_tensor.device
    eps = 1e-5
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # LayerNorm
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps)

    # Linear projections
    left = F.linear(x, weights["left_proj.weight"]).to(torch.float16)
    right = F.linear(x, weights["right_proj.weight"]).to(torch.float16)

    # Mask
    if not config.get("nomask", False):
        # mask shape [B,N,N] -> [B,N,N,1]
        mask = mask_tensor.to(left.dtype).unsqueeze(-1)
        left = left * mask
        right = right * mask

    # Gating
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"])).to(torch.float16)
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])).to(torch.float16)

    left = left * left_gate
    right = right * right_gate

    # Out gate (float32)
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # Reshape for kernel
    B, N, _, H = left.shape  # H = hidden_dim

    # Permute to [B, H, N, N] and view as [B*H, N, N]; ensure contiguous.
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    out_flat = torch.empty((B * H, N, N), dtype=torch.float32, device=device)

    # Kernel config
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * H,
    )

    tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        M=N,
        N=N,
        K=N,
        stride_fa=left_flat.stride(0),
        stride_fi=left_flat.stride(1),
        stride_fk=left_flat.stride(2),
        stride_rb=right_flat.stride(0),
        stride_ri=right_flat.stride(1),
        stride_rk=right_flat.stride(2),
        stride_ob=out_flat.stride(0),
        stride_oi=out_flat.stride(1),
        stride_oj=out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,

"""
TriMul (outgoing) – fast Triton implementation.

The operator receives a 4‑D tensor X∈ℝ^{B×L×L×C} and a mask M∈ℝ^{B×L×L}.
After a LayerNorm and two linear projections (left/right), the core
computation is

    out[b,i,j,h] = Σ_{k=0}^{L-1} left[b,i,k,h] * right[b,j,k,h]

which is a batched matrix multiplication L @ Rᵀ for each hidden unit h.
We fuse this heavy O(B·H·L³) work in a Triton kernel that works on
FP16 inputs and accumulates in FP32.  The surrounding Tensor‑Flow
operations (LayerNorm, gating, final projection) are performed with
standard PyTorch calls.

Key optimisations:
* Larger 128×128 tiles and 8 warps per block to better utilise the H100.
* FP16 loads and FP32 accumulation (tl.dot does the conversion internally).
* Minimal branching – the kernel is a pure tiled matmul.
* All tensor reshapes are performed with a single contiguous view, avoiding
  extra copies inside the kernel.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    left_ptr,          # *fp16  [B·H, L, L]
    right_ptr,         # *fp16  [B·H, L, L]
    out_ptr,           # *fp32  [B·H, L, L]
    M, N, K,           # all equal to L (seq_len)
    stride_fa, stride_fi, stride_fk,      # left strides  (batch·H, i, k)
    stride_rb, stride_ri, stride_rk,      # right strides (batch·H, j, k)
    stride_ob, stride_oi, stride_oj,      # out strides   (batch·H, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched matrix multiplication C = A @ Bᵀ  (fp16→fp32)."""
    pid_m = tl.program_id(0)   # row tile index
    pid_n = tl.program_id(1)   # col tile index
    pid_b = tl.program_id(2)   # batch·hidden index

    # -----------------------------------------------------------------
    # Tile offsets
    # -----------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # -----------------------------------------------------------------
    # Pointers to the start of the current tile
    # -----------------------------------------------------------------
    a_ptr = left_ptr + pid_b * stride_fa + offs_m[:, None] * stride_fi
    b_ptr = right_ptr + pid_b * stride_rb + offs_n[None, :] * stride_ri

    # -----------------------------------------------------------------
    # Accumulator in FP32
    # -----------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -----------------------------------------------------------------
    # Loop over K dimension (tiles of size BLOCK_K)
    # -----------------------------------------------------------------
    num_tiles_k = tl.cdiv(K, BLOCK_K)
    for k in range(num_tiles_k):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        a = tl.load(
            a_ptr + offs_k[None, :] * stride_fk,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        b = tl.load(
            b_ptr + offs_k[:, None] * stride_rk,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # tl.dot returns FP32 when inputs are FP16
        acc += tl.dot(a, b)

    # -----------------------------------------------------------------
    # Write result
    # -----------------------------------------------------------------
    c_ptr = out_ptr + pid_b * stride_ob + offs_m[:, None] * stride_oi + offs_n[None, :] * stride_oj
    tl.store(c_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    Input:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    dim = config["dim"]               # input channel dimension C
    hidden_dim = config["hidden_dim"] # projection dimension H (normally 128)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, L, L, C]  (float32)

    # -----------------------------------------------------------------
    # 2. Linear projections (fp16)
    # -----------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"]).to(torch.float16)
    right = F.linear(x, weights["right_proj.weight"]).to(torch.float16)

    # -----------------------------------------------------------------
    # 3. Optional mask (fp16)
    # -----------------------------------------------------------------
    if not config.get("nomask", False):
        mask = mask_tensor.to(left.dtype).unsqueeze(-1)  # [B, L, L, 1]
        left = left * mask
        right = right * mask

    # -----------------------------------------------------------------
    # 4. Gating (fp16)
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"])).to(torch.float16)
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])).to(torch.float16)

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Output gate (float32) – applied after the Triton matmul
    # -----------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))  # [B, L, L, H]

    # -----------------------------------------------------------------
    # 6. Prepare tensors for the Triton kernel
    #    Shape after permute: [B, H, L, L] -> view -> [B·H, L, L]
    # -----------------------------------------------------------------
    B, L, _, H = left.shape  # H == hidden_dim
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, L, L)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, L, L)

    out_flat = torch.empty((B * H, L, L), dtype=torch.float32, device=device)

    # -----------------------------------------------------------------
    # 7. Triton kernel launch
    # -----------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        (L + BLOCK_M - 1) // BLOCK_M,   # rows
        (L + BLOCK_N - 1) // BLOCK_N,   # cols
        B * H,                          # batch·hidden
    )

    tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        M=L,
        N=L,
        K=L,
        stride_fa=left_flat.stride(0),
        stride_fi=left_flat.stride(1),
        stride_fk=left_flat.stride(2),
        stride_rb=right_flat.stride(0),
        stride_ri=right_flat.stride(1),
        stride_rk=right_flat.stride(2),
        stride_ob=out_flat.stride(0),
        stride_oi=out_flat.stride(1),
        stride_oj=out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,   # better utilisation on H100
    )

    # -----------------------------------------------------------------
    # 8. Reshape back to [B, L, L, H]
    # -----------------------------------------------------------------
    out = out_flat.view(B, H, L, L).permute(0, 2, 3, 1).contiguous()  # float32

    # -----------------------------------------------------------------
    # 9. Final LayerNorm, out-gate, and projection to dim
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    output = F.linear(out, weights["to_out.weight"])

    return output
```

**Explanation of speed‑up**

* The heavy O(B·H·L³) work is entirely inside a single Triton kernel that
  treats each hidden channel as an independent matrix‑multiply.
* Tile sizes are enlarged to 128 × 128, and 8 warps are used, fully
  exploiting the H100’s tensor‑core bandwidth.
* All inputs to the kernel are FP16; the accumulator is FP32, avoiding a
  separate conversion step.
* Only a single permutation+contiguous view is required to feed the kernel,
  and all remaining element‑wise operations (masking, gating, LayerNorm,
  final projection) stay in PyTorch, which is already highly optimised.

The function `custom_kernel` follows the exact forward semantics of the
reference `TriMul` module while providing a dramatically faster core
multiplicative update.
