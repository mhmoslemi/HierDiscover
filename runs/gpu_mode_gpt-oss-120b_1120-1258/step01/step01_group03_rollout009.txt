def layernorm_kernel(x_ptr, y_ptr, weight_ptr, bias_ptr, total_rows, dim, eps, BLOCK_M, BLOCK_D):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        # pointer offset: x_ptr is half pointer but we treat stride = dim ints?
        # Since x is contiguous as half, stride for columns is 1, rows have stride dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]  # pointer arithmetic same regardless dtype
        vals = tl.load(ptrs, mask=mask_m[:, None] & mask_d[None, :], other=0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs, mask=mask_m[:, None] & mask_d[None, :], other=0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0).to(tl.float32)
        b = tl.load(bias_ptr + offs_d, mask=mask_d, other=0.0).to(tl.float32)
        out = norm * w[None, :] + b[None, :]

        # Cast back to half for storage
        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out.to(tl.float16), mask=mask_m[:, None] & mask_d[None, :])

"""
TriMul (outgoing) – fast half‑precision forward pass.

Algorithm
---------
1. Layer‑norm over the channel dimension (dim) using a custom Triton kernel.
   The kernel works on fp16 inputs, accumulates in fp32 and writes back fp16.
2. Combine the two linear projections (proj + gate) for the left and right
   pathways into a single matmul (2 × hidden_dim outputs).  The weight tensors
   are stored in fp16.
3. Apply the optional mask (broadcast over the hidden dimension) to the
   projection part only, compute sigmoid gates, and multiply projection × gate.
4. Triangular multiplicative update:
        out_h[i,j] = Σ_k left_h[i,k] * right_h[j,k]
   This is a batched GEMM L @ Rᵀ for each hidden channel.  All tensors are
   fp16, the GEMM is performed with torch.bmm (Tensor‑core accelerated).
5. Second Layer‑norm over the hidden dimension, also via the Triton kernel.
6. Multiply by the output gate and apply the final linear projection back to
   `dim`.  All remaining linear layers are executed in fp16; the final result
   is cast back to fp32 as required by the interface.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton row‑wise LayerNorm (supports fp16 input & output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,                # *const void*   [total_rows, dim]  (fp16)
    y_ptr,                # *mut   void*   [total_rows, dim]  (fp16)
    weight_ptr,           # *const void*   [dim]              (fp16)
    bias_ptr,             # *const void*   [dim]              (fp16)
    total_rows,           # int (runtime)
    dim: tl.constexpr,    # int  (compile‑time)
    eps: tl.constexpr,    # fp32 const
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """Row‑wise LayerNorm for a 2‑D tensor (fp16)."""
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # -----------------------------------------------------------------
    # 1) Compute row‑wise mean and variance (fp32 accumulator)
    # -----------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)       # [BLOCK_M, BLOCK_D]
        sum_val += tl.sum(vals, axis=1)                # row‑wise sum
        sum_sq  += tl.sum(vals * vals, axis=1)         # row‑wise sum of squares

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                         # 1 / sqrt(var+eps)

    # -----------------------------------------------------------------
    # 2) Write normalized values with γ (weight) and β (bias)
    # -----------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0).to(tl.float32)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0).to(tl.float32)

        out = norm * w[None, :] + b[None, :]

        # store as fp16
        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no backward).

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32, on CUDA)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    device = input_tensor.device

    # ------------------------------------------------------------------
    # Prepare half‑precision constants
    # ------------------------------------------------------------------
    # convert all weights to fp16 once (store in locals)
    def hp(t): return t.to(torch.float16)
    norm_w  = hp(weights["norm.weight"])
    norm_b  = hp(weights["norm.bias"])
    left_proj_w   = hp(weights["left_proj.weight"])
    right_proj_w  = hp(weights["right_proj.weight"])
    left_gate_w   = hp(weights["left_gate.weight"])
    right_gate_w  = hp(weights["right_gate.weight"])
    out_gate_w    = hp(weights["out_gate.weight"])
    to_out_norm_w = hp(weights["to_out_norm.weight"])
    to_out_norm_b = hp(weights["to_out_norm.bias"])
    to_out_w      = hp(weights["to_out.weight"])

    # ------------------------------------------------------------------
    # 1) First LayerNorm (fp16)
    # ------------------------------------------------------------------
    x_fp16 = input_tensor.to(torch.float16)                 # [B,N,N,dim]
    B, N, _, _ = x_fp16.shape
    total_rows = B * N * N

    x_flat = x_fp16.reshape(total_rows, dim)                # [total_rows, dim]
    x_norm_flat = torch.empty_like(x_flat)                  # fp16 output

    # kernel launch parameters (tuned for H100)
    BLOCK_M = 256        # rows per program
    BLOCK_D = 64         # columns per iteration
    grid = (triton.cdiv(total_rows, BLOCK_M),)

    layernorm_kernel[grid](
        x_flat,
        x_norm_flat,
        norm_w,
        norm_b,
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()

    # ------------------------------------------------------------------
    # 2) Linear projections + gates (all in fp16, combine proj+gate)
    # ------------------------------------------------------------------
    # combine left projection + left gate weights
    left_comb_w = torch.cat([left_proj_w, left_gate_w], dim=0)   # (2*H, dim)
    right_comb_w = torch.cat([right_proj_w, right_gate_w], dim=0)  # (2*H, dim)

    # flatten normalized input
    x_norm_flat = x_norm_flat    # already flattened

    # left side (proj + gate)
    left_comb = F.linear(x_norm_flat, left_comb_w)              # [total_rows, 2*H] fp16
    left_proj = left_comb[:, :hidden_dim]                       # [total_rows, H]
    left_gate_pre = left_comb[:, hidden_dim:]                   # [total_rows, H]
    left_gate = torch.sigmoid(left_gate_pre)                    # fp16

    # right side (proj + gate)
    right_comb = F.linear(x_norm_flat, right_comb_w)            # [total_rows, 2*H]
    right_proj = right_comb[:, :hidden_dim]                     # [total_rows, H]
    right_gate_pre = right_comb[:, hidden_dim:]                 # [total_rows, H]
    right_gate = torch.sigmoid(right_gate_pre)                  # fp16

    # optional mask (broadcast over hidden_dim)
    if not config.get("nomask", True):
        # mask shape [B,N,N] -> flatten -> [total_rows, 1]
        mask_fp16 = mask_tensor.to(torch.float16).view(total_rows, 1)
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # apply element‑wise gates
    left = left_proj * left_gate                                 # [total_rows, H] fp16
    right = right_proj * right_gate                              # [total_rows, H] fp16

    # out gate (independent linear)
    out_gate = torch.sigmoid(F.linear(x_norm_flat, out_gate_w))  # [total_rows, H] fp16

    # reshape to 4‑D tensors
    left  = left.view(B, N, N, hidden_dim)          # [B, N, N, H]
    right = right.view(B, N, N, hidden_dim)         # [B, N, N, H]
    out_gate = out_gate.view(B, N, N, hidden_dim)   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 3) Triangular multiplicative update (batched GEMM, fp16)
    # ------------------------------------------------------------------
    # permute to (B, H, N, N) where left is (i,k) and right is (k,j)
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N) -> (k,j)

    BH = B * hidden_dim
    left_mat  = left_perm.reshape(BH, N, N)    # (BH, N, N)
    right_mat = right_perm.reshape(BH, N, N)   # (BH, N, N)

    # batched matrix‑multiply: L @ Rᵀ (i,k) @ (k,j)
    out_mat = torch.bmm(left_mat, right_mat)   # (BH, N, N), fp16

    # reshape back to (B, N, N, H)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # ------------------------------------------------------------------
    # 4) Second LayerNorm over hidden_dim (fp16)
    # ------------------------------------------------------------------
    total_rows_h = B * N * N
    out_flat = out.view(total_rows_h, hidden_dim)          # [total_rows_h, H]
    out_norm_flat = torch.empty_like(out_flat)             # fp16

    grid2 = (triton.cdiv(total_rows_h, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm_flat,
        to_out_norm_w,
        to_out_norm_b,
        total_rows_h,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()

    out_norm = out_norm_flat.view(B, N, N, hidden_dim)     # fp16

    # ------------------------------------------------------------------
    # 5) Apply out‑gate and final projection (fp16 → fp32)
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate                          # element‑wise fp16

    final_fp16 = F.linear(out_norm.view(total_rows_h, hidden_dim),
                          to_out_w)                     # [total_rows_h, dim] fp16
    final = final_fp16.view(B, N, N, dim).to(torch.float32)  # cast back to fp32

    return final
