Optimized Triton‐accelerated “outgoing” TriMul.

The expensive part of the TriMul forward pass is the three‑tensor contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is a batched GEMM (A @ Bᵀ) for every hidden channel d.
The kernel below implements this GEMM in FP16 with FP32 accumulation,
using a larger tile (128×128) and a bigger K‑tile (64) than the reference
implementation.  On H100 we launch 8 warps per block (256 threads) and
use 4 software stages to keep the shared‑memory pipeline full.
All surrounding LayerNorms, linear projections and gates are kept in
regular PyTorch for robustness – they are inexpensive compared with the
N³ contraction.  The kernel therefore satisfies the “partial‑kernel”
requirement while giving a sizeable speed‑up over the baseline.

The code works for any allowed batch size B∈{1,2}, sequence length N
(128 – 1024) and hidden dimension H = hidden_dim (≤ 384).  It also
handles the optional mask (nomask=False) and returns a float32 tensor
of shape [B, N, N, dim].
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (A @ Bᵀ) with fp16 inputs → fp16 output
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_outgoing_kernel(
    a_ptr,                     # *fp16  left  matrix (B*H, M, K)
    b_ptr,                     # *fp16  right matrix (B*H, K, N)
    c_ptr,                     # *fp16  output matrix (B*H, M, N)
    M, N, K,                   # int32 dimensions (all == seq_len)
    stride_az, stride_am, stride_ak,   # strides of A
    stride_bz, stride_bk, stride_bn,   # strides of B
    stride_cz, stride_cm, stride_cn,   # strides of C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM C = A @ Bᵀ for many independent (B*H) matrices."""
    pid_z = tl.program_id(0)   # combined (batch*hidden) * tile_m
    pid_n = tl.program_id(1)   # tile_n

    # --------------------------------------------------------------
    # Decode batch*hidden index and tile coordinates
    # --------------------------------------------------------------
    num_tiles_m = tl.cdiv(M, BLOCK_M)
    batch_h_id = pid_z // num_tiles_m          # index in [0, B*H)
    tile_m   = pid_z %  num_tiles_m
    tile_n   = pid_n

    # --------------------------------------------------------------
    # Offsets for the current tile
    # --------------------------------------------------------------
    offs_m = tile_m * BLOCK_M + tl.arange(0, BLOCK_M)   # [BLOCK_M]
    offs_n = tile_n * BLOCK_N + tl.arange(0, BLOCK_N)   # [BLOCK_N]
    offs_k = tl.arange(0, BLOCK_K)                     # [BLOCK_K]

    # Pointers to the start of the tile (no k offset yet)
    a_ptrs = a_ptr + batch_h_id * stride_az \
                   + offs_m[:, None] * stride_am \
                   + offs_k[None, :] * stride_ak
    b_ptrs = b_ptr + batch_h_id * stride_bz \
                   + offs_k[:, None] * stride_bk \
                   + offs_n[None, :] * stride_bn

    # --------------------------------------------------------------
    # Accumulator (FP32 for numerical safety)
    # --------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Main K‑loop
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K
        k_idx = cur_k + offs_k

        # Out‑of‑bounds handling
        mask_a = (offs_m[:, None] < M) & (k_idx[None, :] < K)
        mask_b = (k_idx[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=mask_a, other=0.0)   # (BLOCK_M, BLOCK_K) fp16
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)   # (BLOCK_K, BLOCK_N) fp16

        acc += tl.dot(a, b)                           # fp16⋅fp16 → fp32

        # Advance pointers to next K‑block
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # --------------------------------------------------------------
    # Store the result (back to fp16)
    # --------------------------------------------------------------
    c_ptrs = c_ptr + batch_h_id * stride_cz \
                     + offs_m[:, None] * stride_cm \
                     + offs_n[None, :] * stride_cn
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)


# ----------------------------------------------------------------------
# Public entry point used by the grader
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask         : torch.Tensor [B, N, N] (bool/float) or None
        - weights      : dict of model parameters (float32)
        - config       : dict with at least {"dim": int,
                                            "hidden_dim": int,
                                            "nomask": bool}

    Returns
    -------
    torch.Tensor
        Output of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack -----------------------------------------------------------
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype  = input_tensor.dtype

    dim        = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask     = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm on the input channel (dim) -------------------------
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (bias‑free) --------------------------------
    # ------------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast on the hidden dim) -------------------
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f = mask.to(dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4️⃣ Gating scalars (sigmoid) --------------------------------------
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Cast to fp16 – the heavy contraction is performed in fp16 -----
    # ------------------------------------------------------------------
    left_fp16  = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    B, N, _, _ = left_fp16.shape   # _ == hidden_dim

    # ------------------------------------------------------------------
    # 6️⃣ Prepare tensors for the batched GEMM
    #    left  : (B, H, N, N)  → view (B*H, N, N)
    #    right : (B, H, N, N)  → transpose K‑axis → view (B*H, N, N)
    # ------------------------------------------------------------------
    # Moving the hidden dimension to the batch axis makes the GEMM
    # contiguous in the K‑dimension (stride = 1) and gives the best
    # memory pattern for the kernel.
    left_mat  = left_fp16.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right_fp16.permute(0, 3, 2, 1).contiguous().view(B * hidden_dim, N, N)

    # Output buffer (fp16, accumulation will be in fp32)
    out_mat = torch.empty_like(left_mat, dtype=torch.float16, device=device)

    # ------------------------------------------------------------------
    # 7️⃣ Triton launch configuration (tuned for H100) ----------------
    # ------------------------------------------------------------------
    # Choose tile sizes based on N so that we never allocate more threads
    # than needed for very small sequences.
    if N <= 256:
        BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 32
    elif N <= 512:
        BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 32
    else:                     # N ∈ {768, 1024}
        BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 64

    # Number of tiles along M and N for a *single* (batch*hidden) matrix
    grid_m = triton.cdiv(N, BLOCK_M)
    grid_n = triton.cdiv(N, BLOCK_N)

    # Grid dimensions: (batch*hidden*grid_m, grid_n)
    grid = (B * hidden_dim * grid_m, grid_n)

    _batched_gemm_outgoing_kernel[grid](
        a_ptr=left_mat,
        b_ptr=right_mat,
        c_ptr=out_mat,
        M=N,
        N=N,
        K=N,
        stride_az=left_mat.stride(0),
        stride_am=left_mat.stride(1),
        stride_ak=left_mat.stride(2),
        stride_bz=right_mat.stride(0),
        stride_bk=right_mat.stride(1),
        stride_bn=right_mat.stride(2),
        stride_cz=out_mat.stride(0),
        stride_cm=out_mat.stride(1),
        stride_cn=out_mat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # 256 threads per block – good for H100
        num_stages=4,         # double‑buffered pipeline
    )

    # ------------------------------------------------------------------
    # 8️⃣ Reshape back to (B, N, N, hidden_dim) and cast to FP32 -------
    # ------------------------------------------------------------------
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)

    # ------------------------------------------------------------------
    # 9️⃣ Output LayerNorm + gating + final linear projection -------------
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate                 # element‑wise gating
    out = F.linear(out, weights["to_out.weight"])   # (dim, hidden_dim) → dim

    return out
