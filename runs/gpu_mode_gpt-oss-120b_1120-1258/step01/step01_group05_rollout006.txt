"""
TriMul (outgoing) forward pass – Triton‐accelerated N³ update.

Algorithm
---------
1. Layer‑norm the input tensor X ∈ ℝ^{B×N×N×C}.
2. Linear projections (no bias) to the hidden dimension H:
       L = X·W_left          (B,N,N,H)
       R = X·W_right         (B,N,N,H)
3. Optional pair‑wise mask M (broadcasted on H) is applied to L,R.
4. Three sigmoid gates are computed from the normalized input:
       GL = σ(X·W_left_gate)   (B,N,N,H)
       GR = σ(X·W_right_gate)  (B,N,N,H)
       GO = σ(X·W_out_gate)    (B,N,N,H)
   L ← L·GL ,   R ← R·GR .
5. The heavy N³ work: for every hidden slice h,
       Out[...,i,j,h] = Σ_k L[...,i,k,h] * R[...,j,k,h] .
   This is a batch of matrix multiplications
       Out_h = L_h @ R_hᵀ .
   We fuse this with a custom Triton kernel that processes a
   batch size of B·H matrices simultaneously (half‑precision I/O,
   fp32 accumulation, tensor‑core friendly tiling).
6. Second LayerNorm over the hidden dimension.
7. Multiply by the output gate GO.
8. Final linear projection back to the original channel dimension C.

The kernel follows the “classic” Triton GEMM implementation
(tiles of 128×128×64, three‑dimensional grid: (row‑block, col‑block,
batch*hidden)).  All heavy arithmetic is performed in fp16 with
fp32 accumulation, which matches the precision of the reference
implementation while dramatically reducing memory traffic.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,          # pointers
    BATCH, M, N, K,               # matrix sizes
    stride_Ab, stride_Am, stride_Ak,   # A strides (batch, M, K)
    stride_Bb, stride_Bk, stride_Bn,   # B strides (batch, K, N) – B is already transposed
    stride_Cb, stride_Cm, stride_Cn,   # C strides (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C = A @ B   (A: (batch,M,K), B: (batch,K,N))"""
    pid_batch = tl.program_id(2)          # batch*hidden index
    pid_m = tl.program_id(0)              # row‑block
    pid_n = tl.program_id(1)              # col‑block

    # tile start indices
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out‑of‑bounds rows / cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -----------------------------------------------------------------
    # Loop over K dimension (inner product)
    # -----------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # pointers to the current tile of A and B
        a_ptrs = (A_ptr + pid_batch * stride_Ab
                  + offs_m[:, None] * stride_Am
                  + offs_k[None, :] * stride_Ak)

        b_ptrs = (B_ptr + pid_batch * stride_Bb
                  + offs_k[:, None] * stride_Bk
                  + offs_n[None, :] * stride_Bn)

        # masked loads (fp16 → fp32 for the dot)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # fused multiply‑accumulate (tensor‑core friendly)
        acc += tl.dot(a, b)          # (BLOCK_M, BLOCK_N)

    # -----------------------------------------------------------------
    # Write output tile
    # -----------------------------------------------------------------
    c_ptrs = (C_ptr + pid_batch * stride_Cb
              + offs_m[:, None] * stride_Cm
              + offs_n[None, :] * stride_Cn)

    tl.store(c_ptrs, acc,
             mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Triton‑backed forward pass for the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, C]  (float32)
        - mask_tensor  : torch.Tensor [B, N, N]  (float32 / bool) or None
        - weights_dict : dict with the following keys
            * "norm.weight", "norm.bias"
            * "left_proj.weight", "right_proj.weight"
            * "left_gate.weight", "right_gate.weight", "out_gate.weight"
            * "to_out_norm.weight", "to_out_norm.bias"
            * "to_out.weight"
        - config_dict  : must contain "dim" (C) and "hidden_dim" (H)
    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, C] (float32)
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]          # C
    hidden_dim = config["hidden_dim"]   # H
    eps = 1e-5

    # -----------------------------------------------------------------
    # 1. First LayerNorm over the channel dimension
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )   # shape [B, N, N, C]

    # -----------------------------------------------------------------
    # 2. Linear projections (no bias) to hidden dimension H
    # -----------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])    # [B,N,N,H]
    right = F.linear(x, weights["right_proj.weight"]) # [B,N,N,H]

    # -----------------------------------------------------------------
    # 3. Optional mask (broadcasted on the hidden axis)
    # -----------------------------------------------------------------
    if mask_tensor is None:
        mask = torch.ones(x.shape[:3], dtype=x.dtype, device=x.device)
    else:
        mask = mask_tensor.to(x.dtype)
    mask = mask.unsqueeze(-1)            # [B, N, N, 1]
    left = left * mask
    right = right * mask

    # -----------------------------------------------------------------
    # 4. Gating (sigmoid of linear projections of the normalized input)
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B,N,N,H]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])) # [B,N,N,H]
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))    # [B,N,N,H]

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Batched GEMM (outgoing update) – fused Triton kernel
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implements:  C = L @ Rᵀ  for each hidden slice h.
    # -----------------------------------------------------------------
    # Use half precision for the two operand tensors (bandwidth‑limited)
    left_h  = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, N, _, H = left_h.shape          # B, N, N, H
    batch = B * H                       # effective batch size for the GEMM

    # Rearrange to (batch, N, N) where batch = B*H
    left_perm  = left_h.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right_h.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    left_reshaped  = left_perm.view(batch, N, N)          # [B*H, N, N]
    right_reshaped = right_perm.view(batch, N, N)         # [B*H, N, N]

    # We need Rᵀ for the matmul: (B*H, K, N) where K == N
    right_t = right_reshaped.permute(0, 2, 1).contiguous()  # [B*H, N, N]

    # Output buffer (fp32 accumulation, fp32 final result)
    out_reshaped = torch.empty(
        (batch, N, N), dtype=torch.float32, device=left_h.device
    )

    # -------------------------------------------------------------
    # Strides (as Python ints)
    # -------------------------------------------------------------
    stride_Ab = left_reshaped.stride(0)
    stride_Am = left_reshaped.stride(1)
    stride_Ak = left_reshaped.stride(2)

    stride_Bb = right_t.stride(0)
    stride_Bk = right_t.stride(1)   # stride over K (the summed dimension)
    stride_Bn = right_t.stride(2)   # stride over N (output column)

    stride_Cb = out_reshaped.stride(0)
    stride_Cm = out_reshaped.stride(1)
    stride_Cn = out_reshaped.stride(2)

    # -------------------------------------------------------------
    # Kernel launch configuration (tuned for H100)
    # -------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),   # rows
        triton.cdiv(N, BLOCK_N),   # cols
        batch                      # batch*hidden
    )

    # -------------------------------------------------------------
    # Launch the Triton kernel
    # -------------------------------------------------------------
    _batched_gemm_kernel[grid](
        left_reshaped,
        right_t,
        out_reshaped,
        batch, N, N, N,
        stride_Ab, stride_Am, stride_Ak,
        stride_Bb, stride_Bk, stride_Bn,
        stride_Cb, stride_Cm, stride_Cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -------------------------------------------------------------
    # Reshape back to the original 4‑D layout: [B, N, N, H]
    # -------------------------------------------------------------
    out = out_reshaped.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------

    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )   # [B, N, N, H]

    # -----------------------------------------------------------------
    # 7. Apply output gate
    # -----------------------------------------------------------------
    out = out * out_gate

    # -----------------------------------------------------------------
    # 8. Final linear projection back to the original channel dimension C
    # -----------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])   # [B, N, N, C]

    return out
```
