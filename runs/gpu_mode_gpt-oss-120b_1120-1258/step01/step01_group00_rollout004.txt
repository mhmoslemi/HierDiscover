"""
TriMul (outgoing) forward pass – highly‑fused implementation.

Algorithm
---------
1. Layer‑norm on the input tensor (last dimension = dim).
2. Fuse the five linear projections (left, right, left_gate, right_gate,
   out_gate) into a **single** GEMM (`F.linear`) to amortise the cost of
   loading the massive (B, N, N, dim) tensor.
3. Apply the optional mask to the left/right tensors.
4. Apply sigmoid gates to the three gate tensors and multiply them into
   left/right (and later into the output).
5. Compute the N³ reduction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   which is a batched matrix‑multiply (left @ rightᵀ) per hidden slice.
   This is performed by a custom Triton kernel that tiles the GEMM
   (BLOCK_M × BLOCK_N × BLOCK_K) and accumulates in FP32 while the
   operands are stored in FP16.
6. Layer‑norm on the hidden dimension, multiply by the out‑gate and finish
   with a linear projection back to the original dim.

All heavy work (the N³ reduction) runs on the GPU in the Triton kernel,
while the cheap element‑wise ops stay in PyTorch.  The implementation
handles the presence/absence of masks, arbitrary batch size (1‑2), and
different channel sizes (dim ∈ {128,384}, hidden_dim ∈ {128,384,768}).
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------------------- #
# Triton kernel: batched GEMM for the TriMul reduction.
# --------------------------------------------------------------------------- #
@triton.jit
def _trimul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,                     # pointers
    M, N, K,                                 # matrix shapes (all = seq_len)
    stride_am, stride_ah, stride_ai, stride_ak,   # A strides: batch, hidden, i, k
    stride_ba, stride_bh, stride_bk, stride_bn,   # B strides: batch, hidden, k, j
    stride_cm, stride_ch, stride_ci, stride_cj,   # C strides: batch, hidden, i, j
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    hidden_dim: tl.constexpr,
):
    """
    Compute C = A @ Bᵀ  for each (batch, hidden) slice.
    A : (batch, hidden, M, K) – fp16
    B : (batch, hidden, N, K) – fp16  (read as (K, N) inside the kernel)
    C : (batch, hidden, M, N) – fp32
    """
    pid_m = tl.program_id(0)   # tile row
    pid_n = tl.program_id(1)   # tile column
    pid_z = tl.program_id(2)   # flattened (batch * hidden)

    # ------------------------------------------------------------------- #
    # Decode batch and hidden indices from pid_z
    # ------------------------------------------------------------------- #
    batch_idx = pid_z // hidden_dim
    hidden_idx = pid_z - batch_idx * hidden_dim

    # Base pointers for the current (batch, hidden) slice
    a_base = a_ptr + batch_idx * stride_am + hidden_idx * stride_ah
    b_base = b_ptr + batch_idx * stride_ba + hidden_idx * stride_bh
    c_base = c_ptr + batch_idx * stride_cm + hidden_idx * stride_ch

    # ------------------------------------------------------------------- #
    # Thread‑local offsets for the M × N tile
    # ------------------------------------------------------------------- #
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------- #
    # K‑loop: load tiles of A (M×K) and B (K×N) and accumulate with mma
    # ------------------------------------------------------------------- #
    for start_k in range(0, K, BLOCK_K):
        cur_k = start_k + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # A tile (M × K)
        a_ptrs = a_base + (offs_m[:, None] * stride_ai
                           + cur_k[None, :] * stride_ak)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # B tile (K × N) – we treat B as (k, j)
        b_ptrs = b_base + (cur_k[:, None] * stride_bk
                           + offs_n[None, :] * stride_bn)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # Accumulate with Tensor‑core mma (fp16 → fp32)
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------- #
    # Write the result tile back to global memory
    # ------------------------------------------------------------------- #
    c_ptrs = c_base + (offs_m[:, None] * stride_ci
                       + offs_n[None, :] * stride_cj)
    tl.store(c_ptrs, acc,
             mask=mask_m[:, None] & mask_n[None, :])


# --------------------------------------------------------------------------- #
# Entry point expected by the evaluation harness
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: Tensor of shape [B, S, S, dim]  (float32)
            - mask_tensor:  Tensor of shape [B, S, S] or None
            - weights_dict: model weights (all tensors are on the correct device)
            - config_dict:  {'dim': int,
                             'hidden_dim': int,
                             'nomask': bool (optional)}

    Returns:
        Tensor of shape [B, S, S, dim] (float32)
    """
    # --------------------------- unpack ----------------------------------- #
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, S, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------- weight shortcuts -------------------------- #
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]

    left_proj_weight  = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]

    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]

    # ----------------------- 1️⃣ Input LayerNorm ------------------------- #
    # LayerNorm over last dimension (dim)
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )   # [B, S, S, dim]  (float32)

    # ----------------------- 2️⃣ Fuse all linear projections ------------ #
    # Concatenate all five projection matrices -> shape (5*hidden, dim)
    fused_weight = torch.cat(
        [
            left_proj_weight,
            right_proj_weight,
            left_gate_weight,
            right_gate_weight,
            out_gate_weight,
        ],
        dim=0,
    )   # (5*hidden, dim)

    # One big GEMM:   x @ fused_weight.T   →   [B, S, S, 5*hidden]
    proj = F.linear(x, fused_weight)                     # fp32
    # Split back into the five tensors
    left, right, left_gate, right_gate, out_gate = torch.split(
        proj, hidden_dim, dim=-1
    )   # each: [B, S, S, hidden]

    # ----------------------- 3️⃣ Optional masking ------------------------ #
    if not nomask and mask_tensor is not None:
        # mask shape -> [B, S, S, 1] and broadcast over hidden dim
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)   # fp32
        left = left * mask
        right = right * mask

    # ----------------------- 4️⃣ Gates (sigmoid) ------------------------ #
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    left = left * left_gate
    right = right * right_gate

    # ----------------------- 5️⃣ Batched GEMM via Triton -------------- #
    # Cast to half‑precision for the heavy N³ reduction
    left_h  = left.to(torch.float16).contiguous()
    right_h = right.to(torch.float16).contiguous()

    # Rearrange to [B, hidden, S, S] so that K dimension is contiguous
    left_h  = left_h.permute(0, 3, 1, 2).contiguous()
    right_h = right_h.permute(0, 3, 1, 2).contiguous()

    # Output buffer (FP32 accumulation)
    out_fp32 = torch.empty(
        (B, hidden_dim, S, S), dtype=torch.float32, device=device
    )

    # Triton launch parameters – tuned for H100
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        triton.cdiv(S, BLOCK_M),          # rows (i)
        triton.cdiv(S, BLOCK_N),          # cols (j)
        B * hidden_dim,                   # (batch * hidden)
    )

    _trimul_gemm_kernel[grid](
        a_ptr=left_h,
        b_ptr=right_h,
        c_ptr=out_fp32,
        M=S,
        N=S,
        K=S,
        stride_am=left_h.stride(0),
        stride_ah=left_h.stride(1),
        stride_ai=left_h.stride(2),
        stride_ak=left_h.stride(3),
        stride_ba=right_h.stride(0),
        stride_bh=right_h.stride(1),
        stride_bk=right_h.stride(3),   # (k) is the innermost dimension after contig.
        stride_bn=right_h.stride(2),   # (j) – second‑innermost
        stride_cm=out_fp32.stride(0),
        stride_ch=out_fp32.stride(1),
        stride_ci=out_fp32.stride(2),
        stride_cj=out_fp32.stride(3),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        hidden_dim=hidden_dim,
    )

    # ----------------------- 6️⃣ Restore layout -------------------------- #
    # out_fp32: [B, hidden, S, S] → [B, S, S, hidden]
    out = out_fp32.permute(0, 2, 3, 1).contiguous()

    # ----------------------- 7️⃣ Output LayerNorm ----------------------- #
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )   # [B, S, S, hidden]

    # ----------------------- 8️⃣ Final gating & projection ------------- #
    out = out * out_gate
    out = F.linear(out, to_out_weight)   # back to [B, S, S, dim]

    return out
