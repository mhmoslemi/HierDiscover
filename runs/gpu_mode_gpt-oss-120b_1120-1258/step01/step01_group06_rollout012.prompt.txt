import torch
from torch import nn, einsum
import math

# Reference code in PyTorch
class TriMul(nn.Module):
    def __init__(
        self,
        dim: int,
        hidden_dim: int,
    ):
        super().__init__()

        self.norm = nn.LayerNorm(dim)

        self.left_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.right_proj = nn.Linear(dim, hidden_dim, bias=False)

        self.left_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.right_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.out_gate = nn.Linear(dim, hidden_dim, bias=False)

        self.to_out_norm = nn.LayerNorm(hidden_dim)
        self.to_out = nn.Linear(hidden_dim, dim, bias=False)

    def forward(self, x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
        """
        x: [bs, seq_len, seq_len, dim]
        mask: [bs, seq_len, seq_len]

        Returns:
            output: [bs, seq_len, seq_len, dim]
        """
        batch_size, seq_len, _, dim = x.shape

        x = self.norm(x)

        left = self.left_proj(x)
        right = self.right_proj(x)

        mask = mask.unsqueeze(-1)
        left = left * mask
        right = right * mask

        left_gate = self.left_gate(x).sigmoid()
        right_gate = self.right_gate(x).sigmoid()
        out_gate = self.out_gate(x).sigmoid()

        left = left * left_gate
        right = right * right_gate

        out = einsum('... i k d, ... j k d -> ... i j d', left, right)
        # This einsum is the same as the following:
        # out = torch.zeros(batch_size, seq_len, seq_len, dim, device=x.device)
        
        # # Compute using nested loops
        # for b in range(batch_size):
        #     for i in range(seq_len):
        #         for j in range(seq_len):
        #             # Compute each output element
        #             for k in range(seq_len):
        #                 out[b, i, j] += left[b, i, k, :] * right[b, j, k, :]

        out = self.to_out_norm(out)
        out = out * out_gate
        return self.to_out(out)

"""
TriMul (outgoing) implementation with a custom Triton kernel.

The kernel computes, for every batch and every hidden channel h,
    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]

This is a batched matrix multiplication L @ Rᵀ where L and R have shape
[B*H, N, N].  The rest of the TriMul pipeline (LayerNorm, linear
projections, gating, mask handling and final linear projection) is
performed with standard PyTorch ops.  The kernel operates on fp16
inputs with fp32 accumulation for numerical stability.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F


@triton.jit
def tri_mul_kernel(
    left_ptr,               # *fp16  [B*H, N, N]
    right_ptr,              # *fp16  [B*H, N, N]
    out_ptr,                # *fp32  [B*H, N, N]
    M, N, K,                # matrix dims (all equal to seq_len)
    stride_fa, stride_fi, stride_fk,   # strides for left  (batch*H, i, k)
    stride_rb, stride_ri, stride_rk,   # strides for right (batch*H, j, k)
    stride_ob, stride_oi, stride_oj,   # strides for out   (batch*H, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched matrix multiplication C = A @ Bᵀ  (A,B are fp16)."""
    pid_m = tl.program_id(0)          # block row
    pid_n = tl.program_id(1)          # block col
    pid_b = tl.program_id(2)          # which (batch*H) matrix

    # ------------------------------------------------------------------
    # Tile offsets
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Boundary masks
    mask_m = offs_m < M
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # Pointers to the beginning of the current tile
    # ------------------------------------------------------------------
    a_ptr = left_ptr + pid_b * stride_fa + offs_m[:, None] * stride_fi  # [M, K] base
    b_ptr = right_ptr + pid_b * stride_rb + offs_n[None, :] * stride_ri  # [K, N] base (transposed view)

    # ------------------------------------------------------------------
    # Accumulator (fp32)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension
    # ------------------------------------------------------------------
    num_tiles_k = tl.cdiv(K, BLOCK_K)
    for k in range(0, num_tiles_k):
        # K‑tile indices
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile: (BLOCK_M, BLOCK_K)  – fp16
        a = tl.load(
            a_ptr + offs_k[None, :] * stride_fk,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load B tile: (BLOCK_K, BLOCK_N)  – fp16.
        # Note: we read B as (k, n) which is equivalent to Bᵀ for the matmul.
        b = tl.load(
            b_ptr + offs_k[:, None] * stride_rk,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )

        # Compute partial dot product and accumulate in fp32
        acc += tl.dot(a, b).to(tl.float32)

    # ------------------------------------------------------------------
    # Write the result
    # ------------------------------------------------------------------
    c_ptr = out_ptr + pid_b * stride_ob + offs_m[:, None] * stride_oi + offs_n[None, :] * stride_oj
    tl.store(c_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    Arguments:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim]
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    dim = config["dim"]               # input channel dimension
    hidden_dim = config["hidden_dim"] # projection dimension (usually 128)

    # ------------------------------------------------------------------
    # 1) LayerNorm on the input
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (fp16 for the heavy part)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"]).to(torch.float16)
    right = F.linear(x, weights["right_proj.weight"]).to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Optional mask handling
    # ------------------------------------------------------------------
    if not config.get("nomask", False):
        # mask shape [B, N, N] -> [B, N, N, 1] and cast to fp16
        mask = mask_tensor.to(left.dtype).unsqueeze(-1)
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4) Gating (left/right) – keep gates in fp16 to stay inside the kernel
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"])).to(torch.float16)
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])).to(torch.float16)

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Out gate (kept in fp32 because it is applied after the kernel)
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 6) Prepare tensors for the Triton kernel
    #    Shape after permute: [B, hidden_dim, N, N] -> view -> [B*hidden_dim, N, N]
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    out_flat = torch.empty((B * H, N, N), dtype=torch.float32, device=device)

    # ------------------------------------------------------------------
    # 7) Launch Triton kernel
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # rows
        (N + BLOCK_N - 1) // BLOCK_N,   # cols
        B * H,                          # batch*hidden
    )

    tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        M=N,
        N=N,
        K=N,
        stride_fa=left_flat.stride(0),
        stride_fi=left_flat.stride(1),
        stride_fk=left_flat.stride(2),
        stride_rb=right_flat.stride(0),
        stride_ri=right_flat.stride(1),
        stride_rk=right_flat.stride(2),
        stride_ob=out_flat.stride(0),
        stride_oi=out_flat.stride(1),
        stride_oj=out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,   # reasonable default for fp16 matmul
    )

    # ------------------------------------------------------------------
    # 8) Reshape back to [B, N, N, hidden_dim]
    # ------------------------------------------------------------------
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 9) Final LayerNorm, out‑gate, and projection to the original dim
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    output = F.linear(out, weights["to_out.weight"])  # bias is False in the model

    return output
