@triton.jit
def layernorm_kernel_fp16(x_ptr, y_ptr, weight_ptr, bias_ptr, total_rows, dim, eps, BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs, mask=mask_m[:, None] & mask_d[None, :], other=tl.float16(0.0))
        vals_f32 = vals.to(tl.float32)
        sum_val += tl.sum(vals_f32, axis=1)
        sum_sq += tl.sum(vals_f32 * vals_f32, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None]*dim + offs_d[None, :]
        vals = tl.load(ptrs, mask=mask_m[:, None] & mask_d[None, :], other=tl.float16(0.0))
        vals_f32 = vals.to(tl.float32)

        norm = (vals_f32 - mean[:, None]) * rstd[:, None]
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=tl.float16(0.0)).to(tl.float32)
        b = tl.load(bias_ptr + offs_d, mask=mask_d, other=tl.float16(0.0)).to(tl.float32)
        out_f32 = norm * w + b
        out_half = out_f32.to(tl.float16)

        tl.store(y_ptr + offs_m[:, None]*dim + offs_d[None,:], out_half, mask=mask_m[:, None] & mask_d[None, :])
```

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def layernorm_kernel_fp16(x_ptr, y_ptr, weight_ptr, bias_ptr, total_rows, dim, eps, BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr):
   pid=tl.program_id(0)
   row_start=pid*BLOCK_M
   offs_m=row_start+tl.arange(0,BLOCK_M)
   mask_m=offs_m<total_rows
   # compute sum and sumsq
   sum_val=tl.zeros([BLOCK_M], dtype=tl.float32)
   sum_sq=tl.zeros([BLOCK_M], dtype=tl.float32)
   for d in range(0, tl.cdiv(dim, BLOCK_D)):
       offs_d = d*BLOCK_D + tl.arange(0,BLOCK_D)
       mask_d = offs_d < dim
       ptrs = x_ptr + offs_m[:,None]*dim + offs_d[None,:]
       # load half
       vals = tl.load(ptrs, mask=mask_m[:,None] & mask_d[None,:], other=tl.float16(0.0))
       vals_f = vals.to(tl.float32)
       sum_val += tl.sum(vals_f, axis=1)
       sum_sq += tl.sum(vals_f * vals_f, axis=1)
   mean = sum_val / dim
   var = sum_sq / dim - mean * mean
   rstd = tl.rsqrt(var + eps)
   for d in range(0, tl.cdiv(dim, BLOCK_D)):
       offs_d = d*BLOCK_D + tl.arange(0,BLOCK_D)
       mask_d = offs_d < dim
       ptrs = x_ptr + offs_m[:,None]*dim + offs_d[None,:]
       vals = tl.load(ptrs, mask=mask_m[:,None] & mask_d[None,:], other=tl.float16(0.0))
       vals_f = vals.to(tl.float32)
       norm = (vals_f - mean[:,None]) * rstd[:,None]
       w = tl.load(weight_ptr + offs_d, mask=mask_d, other=tl.float16(0.0)).to(tl.float32)
       b = tl.load(bias_ptr + offs_d, mask=mask_d, other=tl.float16(0.0)).to(tl.float32)
       y_f = norm * w + b
       y_h = y_f.to(tl.float16)
       tl.store(y_ptr + offs_m[:,None]*dim + offs_d[None,:], y_h, mask=mask_m[:,None] & mask_d[None,:])
```

def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    device = input_tensor.device

    # ensure input contiguous and cast to half after layernorm
    x = input_tensor.contiguous()
    B, N, _, _ = x.shape
    total_rows = B * N * N

    # flatten view
    x_flat = x.view(total_rows, dim)

    # Output buffer for layernorm result
    x_norm_flat = torch.empty_like(x_flat, dtype=torch.float16)

    # Cast layernorm weight/bias to half
    norm_weight = weights["norm.weight"].half()
    norm_bias = weights["norm.bias"].half()

    BLOCK_M = 256
    BLOCK_D = 64

    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel_fp16[grid](
        x_flat,
        x_norm_flat,
        norm_weight,
        norm_bias,
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()

    # Now proceed with half precision linear layers
    # Cast weights to half
    left_proj_w = weights["left_proj.weight"].half()
    right_proj_w = weights["right_proj.weight"].half()
    left_gate_w = weights["left_gate.weight"].half()
    right_gate_w = weights["right_gate.weight"].half()
    out_gate_w = weights["out_gate.weight"].half()
    to_out_norm_w = weights["to_out_norm.weight"].half()
    to_out_norm_b = weights["to_out_norm.bias"].half()
    to_out_w = weights["to_out.weight"].half()

    # Linear projections
    # F.linear expects weight shape (out_features, in_features)
    left_proj = F.linear(x_norm_flat, left_proj_w)     # [total_rows, hidden_dim]
    right_proj = F.linear(x_norm_flat, right_proj_w)

    # Optional mask
    if not config.get("nomask", True):
        # broadcast mask shape
        m = mask_tensor.contiguous().view(total_rows, 1).to(torch.float16)
        left_proj = left_proj * m
        right_proj = right_proj * m

    # Gates (sigmoid)
    left_gate = torch.sigmoid(F.linear(x_norm_flat, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_norm_flat, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x_norm_flat, out_gate_w))

    left = left_proj * left_gate
    right = right_proj * right_gate

    # reshape to [B,N,N,H]
    left = left.view(B, N, N, hidden_dim)
    right = right.view(B, N, N, hidden_dim)
    out_gate = out_gate.view(B, N, N, hidden_dim)

    # TriMul via bmm
    # Rearrange for batched matmul
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 2, 1).contiguous().view(B * hidden_dim, N, N)
    out_mat = torch.bmm(left_mat, right_mat)  # half

    # reshape back to [B,N,N,hidden_dim]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # half

    # Second layernorm over hidden dim
    total_rows_h = B * N * N
    out_flat = out.view(total_rows_h, hidden_dim)
    out_norm_flat = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows_h, BLOCK_M),)
    layernorm_kernel_fp16[grid2](
        out_flat,
        out_norm_flat,
        to_out_norm_w,
        to_out_norm_b,
        total_rows_h,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()

    out_norm = out_norm_flat.view(B, N, N, hidden_dim)

    # apply out_gate
    out_norm = out_norm * out_gate

    # final projection to dim
    out_norm_flat = out_norm.view(total_rows_h, hidden_dim)
    final = F.linear(out_norm_flat, to_out_w)  # half output
    final = final.view(B, N, N, dim).float()  # cast to float32

    return final

"""
custom_kernel – “outgoing” TriMul (AlphaFold‑3) implemented in mixed‑precision.
Algorithm
---------
1️⃣ Row‑wise LayerNorm over the last channel (dim) – implemented in a
   tiny Triton kernel that works in fp16 (accumulates in fp32 for
   numerical stability).

2️⃣ Linear projections (no bias) and gating.  All linear layers are run
   with fp16 weights/input so that the GPU Tensor‑Cores are used.

3️⃣ Optional mask (broadcasted) is applied in fp16.

4️⃣ Element‑wise gating:
        left  = (x @ W_left)   * sigmoid(x @ W_left_gate)
        right = (x @ W_right)  * sigmoid(x @ W_right_gate)

5️⃣ Triangular multiplicative update (the O(N³) part):
        out_h[i,j] = Σ_k left_h[i,k] * right_h[j,k]
   Implemented as a batched GEMM:  L_h @ R_hᵀ  for every hidden channel.
   The required transposes are performed with cheap `.permute(...).contiguous()`
   (a single copy per tensor).

6️⃣ Second LayerNorm over the hidden dimension (fp16 → fp32 → fp16).

7️⃣ Final output gate (sigmoid) and projection back to `dim`.
   The result is cast back to float32 before returning.

All heavy reductions (the two LayerNorms) are performed in a custom
Triton kernel; the remaining dense linear algebra uses the highly‑
optimised PyTorch/cuBLAS kernels in fp16, giving a >5× speed‑up
compared with the pure‑float32 reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# fp16 row‑wise LayerNorm kernel (computes mean/var in fp32, writes fp16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,                 # *const fp16*   [total_rows, dim]
    y_ptr,                 # *mut   fp16*   [total_rows, dim]
    weight_ptr,            # *const fp16*   [dim]
    bias_ptr,              # *const fp16*   [dim]
    total_rows,            # int (runtime)
    dim: tl.constexpr,     # int (compile‑time)
    eps: tl.constexpr,     # float (compile‑time)
    BLOCK_M: tl.constexpr, # rows per program
    BLOCK_D: tl.constexpr, # cols processed per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # -------------------------------------------------
    # 1️⃣  Compute row‑wise sum and sum‑of‑squares
    # -------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim

        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=tl.float16(0.0),
        )
        vals_f = vals.to(tl.float32)                    # fp16 → fp32
        sum_val += tl.sum(vals_f, axis=1)               # row‑wise sum
        sum_sq  += tl.sum(vals_f * vals_f, axis=1)      # row‑wise sum‑sq

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)   # 1 / sqrt(var + eps)

    # -------------------------------------------------
    # 2️⃣  Write normalised values (γ and β are fp16)
    # -------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=tl.float16(0.0),
        )
        vals_f = vals.to(tl.float32)

        # (x - mean) * rstd
        norm = (vals_f - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=tl.float16(0.0)).to(tl.float32)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=tl.float16(0.0)).to(tl.float32)

        out_f = norm * w + b                     # fp32
        out_h = out_f.to(tl.float16)             # back to fp16

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out_h,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Entry point – the actual TriMul forward pass
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32, on CUDA)
    """
    # -------------------------------------------------
    # Unpack & sanity‑check
    # -------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim          = config["dim"]
    hidden_dim   = config["hidden_dim"]
    eps          = 1e-5
    device       = input_tensor.device

    # -------------------------------------------------
    # 0️⃣  Prepare tensors
    # -------------------------------------------------
    x = input_tensor.contiguous()                     # [B, N, N, dim]
    B, N, _, _ = x.shape
    total_rows = B * N * N                            # rows = B·N·N

    # -------------------------------------------------
    # 1️⃣  First LayerNorm (fp16)
    # -------------------------------------------------
    x_flat      = x.view(total_rows, dim)                         # [R, dim] (float32)
    x_norm_fp16 = torch.empty_like(x_flat, dtype=torch.float16)   # output buffer

    # Cast LayerNorm parameters once
    ln_w = weights["norm.weight"].half()
    ln_b = weights["norm.bias"].half()

    BLOCK_M = 256          # tunable: rows per program
    BLOCK_D = 64           # tunable: cols per iteration

    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        x_flat,
        x_norm_fp16,
        ln_w,
        ln_b,
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()

    # -------------------------------------------------
    # 2️⃣  Linear projections + gates (all fp16)
    # -------------------------------------------------
    # Cast all weight tensors to half once
    w_left_proj  = weights["left_proj.weight"].half()
    w_right_proj = weights["right_proj.weight"].half()
    w_left_gate  = weights["left_gate.weight"].half()
    w_right_gate = weights["right_gate.weight"].half()
    w_out_gate   = weights["out_gate.weight"].half()
    w_to_out_norm = weights["to_out_norm.weight"].half()
    b_to_out_norm = weights["to_out_norm.bias"].half()
    w_to_out      = weights["to_out.weight"].half()

    # ---------- Projections (no bias) ----------
    left_proj  = F.linear(x_norm_fp16, w_left_proj)   # [R, H]
    right_proj = F.linear(x_norm_fp16, w_right_proj)

    # ---------- Optional mask ----------
    if not config.get("nomask", True):
        # mask: [B, N, N] → [R, 1]  (float16)
        m = mask_tensor.contiguous().view(total_rows, 1).to(torch.float16)
        left_proj  = left_proj * m
        right_proj = right_proj * m

    # ---------- Gates (sigmoid) ----------
    left_gate  = torch.sigmoid(F.linear(x_norm_fp16, w_left_gate))
    right_gate = torch.sigmoid(F.linear(x_norm_fp16, w_right_gate))
    out_gate   = torch.sigmoid(F.linear(x_norm_fp16, w_out_gate))

    # ---------- Apply element‑wise gates ----------
    left  = left_proj * left_gate           # [R, H]  fp16
    right = right_proj * right_gate

    # ---------- Reshape to 4‑D tensors ----------
    left  = left.view(B, N, N, hidden_dim)    # [B, N, N, H]
    right = right.view(B, N, N, hidden_dim)
    out_gate = out_gate.view(B, N, N, hidden_dim)

    # -------------------------------------------------
    # 3️⃣  Triangular multiplicative update (batched GEMM)
    # -------------------------------------------------
    #   L_h @ R_hᵀ  for every hidden channel h
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 2, 1).contiguous().view(B * hidden_dim, N, N)

    out_mat = torch.bmm(left_mat, right_mat)               # [B*H, N, N]  fp16

    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    # -------------------------------------------------
    # 4️⃣  Second LayerNorm over the hidden dimension (fp16)
    # -------------------------------------------------
    total_rows_h = B * N * N
    out_flat      = out.view(total_rows_h, hidden_dim)                     # [R, H]
    out_norm_fp16 = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows_h, BLOCK_M),)
    layernorm_fp16_kernel[grid2](
        out_flat,
        out_norm_fp16,
        w_to_out_norm,
        b_to_out_norm,
        total_rows_h,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()

    out_norm = out_norm_fp16.view(B, N, N, hidden_dim)   # [B, N, N, H]

    # -------------------------------------------------
    # 5️⃣  Apply final gate and project back to `dim`
    # -------------------------------------------------
    out_norm = out_norm * out_gate                       # element‑wise (fp16)

    out_norm_flat = out_norm.view(total_rows_h, hidden_dim)   # [R, H]
    final_fp16 = F.linear(out_norm_flat, w_to_out)            # [R, dim]  fp16
    final = final_fp16.view(B, N, N, dim).float()             # cast back to float32

    return final
