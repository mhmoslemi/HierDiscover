update is performed:

    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

This is a batched matrix multiplication L @ Rᵀ where each
matrix has shape (N,N) and we have B·H such matrices (H = hidden_dim).
The heavy reduction is off‑loaded to a custom Triton kernel that works in
float32 (fp16 would be possible but we keep fp32 to retain numerical
exactness with the reference implementation).  The rest of the
TriMul pipeline – a second LayerNorm, an output‑gate, and a final linear
projection – stays in PyTorch.

The Triton kernel uses a 64×64 tile for the output and a 32‑wide tile
for the reduction dimension, 8 warps per block, and 4 pipeline stages.
These parameters give high occupancy on an H100 while keeping the
register pressure modest.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    left_ptr,               # *fp32  [B*H, N, N]   (L matrix)
    right_ptr,              # *fp32  [B*H, N, N]   (R matrix)
    out_ptr,                # *fp32  [B*H, N, N]   (output)
    M, N, K,                # all equal to seq_len
    stride_fa, stride_fi, stride_fk,   # strides for left  (batch*H, i, k)
    stride_rb, stride_ri, stride_rk,   # strides for right (batch*H, j, k)
    stride_ob, stride_oi, stride_oj,   # strides for out   (batch*H, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM   out = left @ rightᵀ   (fp32 accumulation)."""
    pid_m = tl.program_id(0)          # output‑row block
    pid_n = tl.program_id(1)          # output‑col block
    pid_b = tl.program_id(2)          # which (batch*H) matrix

    # ------------------------------------------------------------------
    #  Tile offsets
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    #  Pointers to the start of the current tile
    # ------------------------------------------------------------------
    a_ptr = left_ptr + pid_b * stride_fa + offs_m[:, None] * stride_fi
    b_ptr = right_ptr + pid_b * stride_rb + offs_n[None, :] * stride_ri

    # ------------------------------------------------------------------
    #  Accumulator (fp32)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    #  Loop over the reduction dimension K
    # ------------------------------------------------------------------
    num_tiles_k = tl.cdiv(K, BLOCK_K)
    for k in range(0, num_tiles_k):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile: (BLOCK_M, BLOCK_K) – fp32
        a = tl.load(
            a_ptr + offs_k[None, :] * stride_fk,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # B tile: (BLOCK_K, BLOCK_N) – fp32
        b = tl.load(
            b_ptr + offs_k[:, None] * stride_rk,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )

        # Partial product (still fp32)
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    #  Write the result
    # ------------------------------------------------------------------
    c_ptr = out_ptr + pid_b * stride_ob + offs_m[:, None] * stride_oi + offs_n[None, :] * stride_oj
    tl.store(c_ptr, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator using a Triton kernel.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, C] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (float32) – may be ignored
        - weights       : dict of model weights (float32 tensors)
        - config        : dict containing at least
            * "dim"         – input channel dimension C
            * "hidden_dim" – projection dimension H
            * "nomask"     – bool, True → no mask is applied

    Returns
    -------
    torch.Tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    dim = config["dim"]                     # input channels (C)
    hidden_dim = config["hidden_dim"]       # projection size (H)

    # ------------------------------------------------------------------
    # 1) First LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # shape [B, N, N, dim], dtype float32

    # ------------------------------------------------------------------
    # 2) Linear projections (float32)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])       # [B, N, N, H]
    right = F.linear(x, weights["right_proj.weight"])    # [B, N, N, H]

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if not config.get("nomask", False):
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)    # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4) Gating (float32) – element‑wise sigmoid
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Output gate (float32, applied after the kernel)
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 6) Prepare tensors for the Triton kernel
    #    Convert from [B, N, N, H] → [B*H, N, N] (contiguous)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape   # H == hidden_dim
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    out_flat = torch.empty((B * H, N, N), dtype=torch.float32, device=device)

    # ------------------------------------------------------------------
    # 7) Triton kernel launch
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # rows
        (N + BLOCK_N - 1) // BLOCK_N,   # cols
        B * H,                          # batch * hidden
    )

    tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        M=N,
        N=N,
        K=N,
        stride_fa=left_flat.stride(0),
        stride_fi=left_flat.stride(1),
        stride_fk=left_flat.stride(2),
        stride_rb=right_flat.stride(0),
        stride_ri=right_flat.stride(1),
        stride_rk=right_flat.stride(2),
        stride_ob=out_flat.stride(0),
        stride_oi=out_flat.stride(1),
        stride_oj=out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # more warps → higher occupancy on H100
        num_stages=4,         # double‑buffering for better latency hiding
    )

    # ------------------------------------------------------------------
    # 8) Reshape back to [B, N, N, hidden_dim]
    # ------------------------------------------------------------------
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 9) Second LayerNorm, output‑gate and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    output = F.linear(out, weights["to_out.weight"])

    return output
