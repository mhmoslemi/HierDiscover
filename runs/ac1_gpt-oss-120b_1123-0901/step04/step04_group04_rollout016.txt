```
def local_search(a, target_sum, deadline):
    best_a = a.copy()
    best_score = evaluate_sequence(best_a.tolist())
    n = len(a)
    while time.time() < deadline:
        # pick two random indices i, j with i != j
        i, j = np.random.randint(0, n, size=2)
        if i == j: continue
        # choose transfer amount delta = factor * best_a[i]; factor random in [0,0.1]
        max_delta = min(best_a[i], best_a[i]*0.2) # can't exceed small portion
        delta = np.random.rand() * max_delta
        a_new = best_a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # optionally renormalize to target_sum (sum may be unchanged)
        # sum unchanged (transfer)
        # Clip to >=0 (though we ensured delta <= a_i)
        score = evaluate_sequence(a_new.tolist())
        if score < best_score:
            best_score = score
            best_a = a_new
            # maybe print improvement
    return best_a, best_score
```

"""
Propose a step function that minimises evaluate_sequence.

Algorithm:
- Convert the problem to a probability vector p (sum 1) and enforce
  the L1‐norm constraint Σa = sqrt(2n) by scaling.
- Optimise the smooth approximation τ·log Σ exp((p∗p)_k/τ) using
  limited‑memory BFGS.  The smooth max is gradually cooled,
  random jitter is added and several diverse starts (uniform,
  Dirichlet, sparse Dirichlet, U‑shapes, alternating) are tried.
- After the BFGS stage a cheap local hill‑climbing phase
  (random mass transfers) refines the best solution.
- The whole search respects the 1000 s budget and returns the
  best sequence found.
"""

import time
import math
import numpy as np
from typing import List, Tuple

# Try to import scipy.optimize; fall back to a simple gradient descent if unavailable.
try:
    from scipy.optimize import minimize
    _SCIPY_AVAILABLE = True
except Exception:  # pragma: no cover
    _SCIPY_AVAILABLE = False

# ----------------------------------------------------------------------
# Helper functions (no closures)

def _softmax_weights(conv: np.ndarray, tau: float) -> Tuple[np.ndarray, float]:
    """Stable soft‑max of convolution values.
    Returns (weights,w) where weights sum to 1 and w = smooth max."""
    maxc = conv.max()
    if tau <= 0.0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w, maxc
    ex = np.exp((conv - maxc) / tau)
    sum_ex = ex.sum()
    w = ex / sum_ex
    Ltau = maxc + tau * math.log(sum_ex)
    return w, Ltau


def _conv_full(p: np.ndarray, Nfft: int) -> np.ndarray:
    """Full convolution p∗p using real FFT, returns length 2·n‑1."""
    A_fft = np.fft.rfft(p, n=Nfft)
    conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * p.shape[0] - 1]
    return conv


def _weighted_rev(a: np.ndarray, w: np.ndarray, Nfft: int) -> np.ndarray:
    """
    Compute rev_i = Σ_k w_k a_{k‑i} using FFTs.
    (indices out of the support are ignored).
    """
    # FFT of a (zero‑padded)
    A_fft = np.fft.rfft(a, n=Nfft)
    # FFT of reversed weight vector
    W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)
    # Convolution
    conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
    # Valid part of length len(w)+len(a)-1
    conv_rev = conv_rev[: len(w) + a.shape[0] - 1]
    # The rev_i we need corresponds to an offset
    idx_start = 2 * a.shape[0] - 2
    rev = conv_rev[idx_start - np.arange(a.shape[0])]
    return rev


def _objective_and_grad(y: np.ndarray, tau: float, Nfft: int) -> Tuple[float, np.ndarray]:
    """Smooth max L_τ(p) and its gradient w.r.t. the log‑probability vector y."""
    # p = softmax(y)
    y_max = np.max(y)
    exp_y = np.exp(y - y_max)               # stability
    p = exp_y / exp_y.sum()

    # convolution p∗p
    conv = _conv_full(p, Nfft)

    # smooth max and weights
    w, Ltau = _softmax_weights(conv, tau)

    # gradient w.r.t. p
    rev = _weighted_rev(p, w, Nfft)
    grad_p = 2.0 * rev

    # transform to gradient w.r.t. y (softmax Jacobian)
    dot = np.dot(p, grad_p)
    grad_y = p * (grad_p - dot)
    return Ltau, grad_y


def _run_l_bfgs_extended(
    y0: np.ndarray,
    init_tau: float,
    target_sum: float,
    Nfft: int,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
) -> Tuple[np.ndarray, float]:
    """
    Perform a multi‑stage L‑BFGS optimisation of the smooth max.
    Returns the best raw sequence (scaled to target_sum) and its exact score.
    """
    y = y0.copy()
    tau = max(init_tau, 1e-12)
    tau_decay = 0.80          # slower cooling than the baseline
    min_tau = 1e-12
    # L‑BFGS options – allow a reasonable number of evaluations per stage
    options = {"maxfun": 400, "maxiter": 400, "ftol": 1e-12, "gtol": 1e-12}
    while time.time() < deadline and tau > min_tau:
        # ----- L‑BFGS on current temperature -----
        def fun(vec):
            return _objective_and_grad(vec, tau, Nfft)

        if _SCIPY_AVAILABLE:
            res = minimize(fun, y, method="L-BFGS-B", jac=True, options=options)
            y = res.x
        else:
            # Simple fallback: a few steps of gradient descent with momentum
            grad = np.zeros_like(y)
            for _ in range(30):
                L, g = _objective_and_grad(y, tau, Nfft)
                grad = 0.9 * grad + 0.1 * g
                y -= 0.1 * grad
        # ----- evaluate exact objective -----
        y_max = np.max(y)
        exp_y = np.exp(y - y_max)
        p = exp_y / exp_y.sum()
        a = p * target_sum
        cur_score = evaluate_sequence(a.tolist())
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[improve] τ={tau:.2e}  score={best_score:.6f}")

        # ----- cooling schedule & jitter -----
        tau *= tau_decay
        # occasional random jitter to escape plateaus
        if np.random.rand() < 0.12:
            y += np.random.randn(y.shape[0]) * 0.01

        if time.time() >= deadline:
            break
    return best_seq, best_score


def _local_search(a: np.ndarray, target_sum: float, deadline: float) -> Tuple[np.ndarray, float]:
    """
    Simple stochastic hill climbing: repeatedly transfer a tiny amount of mass
    from a random index i to a random index j (keeping the total sum constant).
    Accept only improving moves.  Stops when time runs out.
    """
    best_a = a.copy()
    best_score = evaluate_sequence(best_a.tolist())
    n = len(a)
    # scale for typical step size – a fraction of the current average entry
    avg = target_sum / n
    step_scale = avg * 0.1
    it = 0
    while time.time() < deadline:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # random transfer amount, limited by available mass at i
        max_delta = min(best_a[i], step_scale)
        if max_delta <= 0:
            continue
        delta = np.random.rand() * max_delta
        a_new = best_a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # evaluate
        score = evaluate_sequence(a_new.tolist())
        if score < best_score:
            best_score = score
            best_a = a_new
            # gradually shrink step size after an improvement
            step_scale = max(step_scale * 0.98, avg * 1e-4)
            print(f"[local] improve it={it} score={best_score:.6f}")
        it += 1
        # occasional reset of step scale to avoid getting stuck
        if it % 5000 == 0:
            step_scale = avg * 0.1
    return best_a, best_score


def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` within the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # keep a small safety margin

    # --- configuration -------------------------------------------------
    # We will work with a single length (2100) that proved useful before,
    # but the code can be easily extended to other lengths.
    n = 2100
    target_sum = math.sqrt(2.0 * n)
    Nfft = 1 << ((2 * n - 1).bit_length())

    # --- initialise a few diverse starting vectors --------------------
    eps_log = 1e-12
    starts_y = []

    # 1. Uniform
    uniform = np.ones(n) * (target_sum / n)
    p_uni = uniform / target_sum
    starts_y.append(np.log(p_uni + eps_log))

    # 2. Random Dirichlet (alpha=1)
    rand_exp = np.random.exponential(scale=1.0, size=n)
    p_rand = rand_exp / rand_exp.sum()
    starts_y.append(np.log(p_rand + eps_log))

    # 3. Sparse Dirichlet (alpha = 0.3)
    try:
        gamma_small = np.random.gamma(shape=0.3, scale=1.0, size=n)
    except Exception:
        gamma_small = np.random.exponential(scale=1.0, size=n) ** 4
    p_sparse = gamma_small / gamma_small.sum()
    starts_y.append(np.log(p_sparse + eps_log))

    # 4. U‑shape families (distance from centre raised to a power)
    for p_exp in (0.5, 1.0, 2.0, 3.0):
        d = np.abs(np.arange(n) - (n - 1) / 2.0)
        raw = (d + 1e-4) ** p_exp
        p_u = raw / raw.sum()
        starts_y.append(np.log(p_u + eps_log))

    # 5. Alternating high/low pattern
    alt = np.empty(n)
    alt[::2] = 2.0
    alt[1::2] = 0.5
    p_alt = alt / alt.sum()
    starts_y.append(np.log(p_alt + eps_log))

    # --- baseline best (uniform) --------------------------------------
    best_seq = uniform.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # --- run L‑BFGS from each start ------------------------------------
    for idx, y0 in enumerate(starts_y):
        if time.time() >= deadline:
            break
        print(f"[run] L‑BFGS start {idx + 1}/{len(starts_y)}")
        # initialise temperature from the current smooth max
        p0 = np.exp(y0 - np.max(y0))
        p0 /= p0.sum()
        conv0 = _conv_full(p0, Nfft)
        init_tau = max(conv0.max() * 2.0, 1e-12)

        best_seq, best_score = _run_l_bfgs_extended(
            y0=y0,
            init_tau=init_tau,
            target_sum=target_sum,
            Nfft=Nfft,
            deadline=deadline,
            best_seq=best_seq,
            best_score=best_score,
        )

    # --- local stochastic refinement ------------------------------------
    if time.time() < deadline:
        print("[local] starting hill‑climbing refinement")
        a_current = np.array(best_seq)
        a_refined, score_refined = _local_search(a_current, target_sum, deadline)

"""
Propose a step function that minimises ``evaluate_sequence``.

Algorithm
---------
* Work with a probability vector p = a / Σa (so Σp = 1).  By scaling
  Σa to √(2·n) the objective becomes exactly the maximal entry of
  the full autocorrelation of p, i.e. 2·n·max(p∗p).
* Optimise the smooth approximation τ·log ∑exp((p∗p)_k/τ) using
  limited‑memory BFGS (L‑BFGS‑B) on the unconstrained log‑softmax
  parameters y (p_i = exp(y_i)/∑exp(y)).  The temperature τ is
  gradually decreased (cooling) and small random jitter is added
  to escape plateaus.
* Several diverse starting points are tried:
  – uniform,
  – random Dirichlet,
  – sparse Dirichlet (γ‑shape < 1),
  – “U‑shape’’ families a_i∝dist(i,centre)^p,
  – alternating high/low pattern.
* After the BFGS phase a cheap stochastic hill‑climbing refinement
  repeatedly transfers a tiny amount of mass between random indices,
  accepting only improvements.
* If time permits a final short BFGS run from the refined solution
  is performed.
* The whole search respects the 1000 s budget; the best sequence
  found is returned as a plain Python list.
"""

import time
import math
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# optional SciPy import
try:
    from scipy.optimize import minimize
    _SCIPY_AVAILABLE = True
except Exception:  # pragma: no cover
    _SCIPY_AVAILABLE = False


def _softmax_weights(conv: np.ndarray, tau: float) -> Tuple[np.ndarray, float]:
    """Stable soft‑max of convolution values.
    Returns (weights, smooth_max)."""
    maxc = conv.max()
    if tau <= 0.0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w, maxc
    ex = np.exp((conv - maxc) / tau)
    sum_ex = ex.sum()
    w = ex / sum_ex
    Ltau = maxc + tau * math.log(sum_ex)
    return w, Ltau


def _conv_full(p: np.ndarray, Nfft: int) -> np.ndarray:
    """Full convolution p∗p using a real FFT.  Result length = 2·n‑1."""
    A_fft = np.fft.rfft(p, n=Nfft)
    conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * p.shape[0] - 1]
    return conv


def _weighted_rev(a: np.ndarray, w: np.ndarray, Nfft: int) -> np.ndarray:
    """
    Compute rev_i = Σ_k w_k a_{k‑i} using FFTs.
    (indices outside the support are ignored).
    """
    A_fft = np.fft.rfft(a, n=Nfft)
    W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)
    conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
    conv_rev = conv_rev[: len(w) + a.shape[0] - 1]
    idx_start = 2 * a.shape[0] - 2
    rev = conv_rev[idx_start - np.arange(a.shape[0])]
    return rev


def _objective_and_grad(y: np.ndarray, tau: float, Nfft: int) -> Tuple[float, np.ndarray]:
    """
    Smooth‑max L_τ(p) and its gradient w.r.t. the log‑softmax parameters y.
    """
    y_max = np.max(y)
    exp_y = np.exp(y - y_max)               # numerical stability
    p = exp_y / exp_y.sum()

    conv = _conv_full(p, Nfft)               # p∗p
    w, Ltau = _softmax_weights(conv, tau)    # smooth max

    rev = _weighted_rev(p, w, Nfft)          # Σ_k w_k p_{k‑i}
    grad_p = 2.0 * rev

    dot = np.dot(p, grad_p)
    grad_y = p * (grad_p - dot)              # chain rule through softmax
    return Ltau, grad_y


def _run_l_bfgs_extended(
    y0: np.ndarray,
    init_tau: float,
    target_sum: float,
    Nfft: int,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
) -> Tuple[np.ndarray, float]:
    """
    Multi‑stage L‑BFGS optimisation of the smooth max.
    Returns the best raw sequence (scaled to target_sum) and its exact score.
    """
    y = y0.copy()
    tau = max(init_tau, 1e-12)
    tau_decay = 0.80          # slower cooling than the baseline
    min_tau = 1e-12
    # allow a moderate number of function evaluations per stage
    options = {"maxfun": 400, "maxiter": 400, "ftol": 1e-12, "gtol": 1e-12}

    while time.time() < deadline and tau > min_tau:
        # ----- L‑BFGS on the current temperature -----
        def fun(vec):
            return _objective_and_grad(vec, tau, Nfft)

        if _SCIPY_AVAILABLE:
            res = minimize(fun, y, method="L-BFGS-B", jac=True, options=options)
            y = res.x
        else:
            # Very rough fallback gradient descent with momentum
            mom = np.zeros_like(y)
            for _ in range(30):
                L, g = _objective_and_grad(y, tau, Nfft)
                mom = 0.9 * mom + 0.1 * g
                y -= 0.1 * mom

        # ----- evaluate the exact objective -----
        y_max = np.max(y)
        exp_y = np.exp(y - y_max)
        p = exp_y / exp_y.sum()
        a = p * target_sum
        cur_score = evaluate_sequence(a.tolist())

        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[improve] τ={tau:.2e}  score={best_score:.6f}")

        # ----- cooling & jitter -----
        tau *= tau_decay
        if np.random.rand() < 0.12:          # occasional perturbation
            y += np.random.randn(y.shape[0]) * 0.01

        if time.time() >= deadline:
            break

    return best_seq, best_score


def _local_search(a: np.ndarray, target_sum: float, deadline: float) -> Tuple[np.ndarray, float]:
    """
    Stochastic hill‑climbing: repeatedly move a tiny amount of mass
    from a random index i to a random index j.  Only improving moves
    are kept.  Stops when the time budget expires.
    """
    best_a = a.copy()
    best_score = evaluate_sequence(best_a.tolist())
    n = len(a)
    avg = target_sum / n
    step_scale = avg * 0.1          # initial move magnitude
    it = 0

    while time.time() < deadline:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        max_delta = min(best_a[i], step_scale)
        if max_delta <= 0:
            continue
        delta = np.random.rand() * max_delta
        a_new = best_a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        score = evaluate_sequence(a_new.tolist())
        if score < best_score:
            best_score = score
            best_a = a_new
            step_scale = max(step_scale * 0.98, avg * 1e-4)   # slowly shrink step
            print(f"[local] it={it}  score={best_score:.6f}")
        it += 1
        if it % 5000 == 0:      # occasional reset to avoid stagnation
            step_scale = avg * 0.1

    return best_a, best_score


def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` within the given time budget.
    Returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # small safety margin

    # ------------------------------------------------------------------
    # Problem size – 2100 has proven a good trade‑off of flexibility vs.
    # runtime, but the code can be changed easily.
    n = 2100
    target_sum = math.sqrt(2.0 * n)
    Nfft = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------------
    # Build a collection of diverse initial points (log‑softmax vectors)
    eps_log = 1e-12
    starts_y = []

    # 1) uniform
    uniform = np.ones(n) * (target_sum / n)
    p_uni = uniform / target_sum
    starts_y.append(np.log(p_uni + eps_log))

    # 2) random Dirichlet (α=1)
    rand_exp = np.random.exponential(scale=1.0, size=n)
    p_rand = rand_exp / rand_exp.sum()
    starts_y.append(np.log(p_rand + eps_log))

    # 3) sparse Dirichlet (α≈0.3)
    try:
        gamma_small = np.random.gamma(shape=0.3, scale=1.0, size=n)
    except Exception:
        gamma_small = np.random.exponential(scale=1.0, size=n) ** 4
    p_sparse = gamma_small / gamma_small.sum()
    starts_y.append(np.log(p_sparse + eps_log))

    # 4) U‑shape families a_i ∝ dist(i,centre)^p
    for p_exp in (0.5, 1.0, 2.0, 3.0):
        d = np.abs(np.arange(n) - (n - 1) / 2.0)
        raw = (d + 1e-4) ** p_exp
        p_u = raw / raw.sum()
        starts_y.append(np.log(p_u + eps_log))

    # 5) alternating high/low pattern
    alt = np.empty(n)
    alt[::2] = 2.0
    alt[1::2] = 0.5
    p_alt = alt / alt.sum()
    starts_y.append(np.log(p_alt + eps_log))

    # ------------------------------------------------------------------
    # Initialise the “best so far’’ record with the uniform solution.
    best_seq = uniform.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Run the multi‑start L‑BFGS optimisation.
    for idx, y0 in enumerate(starts_y):
        if time.time() >= deadline:
            break
        print(f"[run] L‑BFGS start {idx + 1}/{len(starts_y)}")
        # initial temperature – a few times the current smooth max
        p0 = np.exp(y0 - np.max(y0))
        p0 /= p0.sum()
        conv0 = _conv_full(p0, Nfft)
        init_tau = max(conv0.max() * 2.0, 1e-12)

        best_seq, best_score = _run_l_bfgs_extended(
            y0=y0,
            init_tau=init_tau,
            target_sum=target_sum,
            Nfft=Nfft,
            deadline=deadline,
            best_seq=best_seq,
            best_score=best_score,
        )

    # ------------------------------------------------------------------
    # Stochastic hill‑climbing refinement (if time permits)
    if time.time() < deadline:
        print("[local] starting hill‑climbing refinement")
        a_curr = np.array(best_seq)
        a_refined, score_refined = _local_search(a_curr, target_sum, deadline)
        if score_refined < best_score:
            best_score = score_refined
            best_seq = a_refined

    # ------------------------------------------------------------------
    # One final short L‑BFGS run from the refined point (optional)
    if time.time() < deadline - 5.0:
        p_ref = best_seq / target_sum
        y_ref = np.log(p_ref + eps_log)
        conv_ref = _conv_full(p_ref, Nfft)
        init_tau_ref = max(conv_ref.max() * 2.0, 1e-12)
        best_seq, best_score = _run_l_bfgs_extended(
            y0=y_ref,
            init_tau=init_tau_ref,
            target_sum=target_sum,
            Nfft=Nfft,
            deadline=deadline,
            best_seq=best_seq,
            best_score=best_score,
        )

    # ------------------------------------------------------------------
    # Final sanity check using the official evaluator.
    final_score = evaluate_sequence(best_seq.tolist())
    if final_score < best_score:
        best_score = final_score
        best_seq = best_seq.copy()

    print(f"[done] best score = {best_score:.6f}, length = {len(best_seq)}")
    return best_seq.tolist()
