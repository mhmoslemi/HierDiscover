"""
Hybrid gradient + local‐search optimizer for the step‑function problem.

The objective after the internal normalisation performed by
``evaluate_sequence`` equals the maximal value of the linear
convolution a∗a (because Σ a = √(2·n) ⇒ 2·n/Σ² = 1).  Hence we minimise

    score(a) = max_k (a∗a)[k]    with   a_i ≥ 0 ,  Σ a_i = √(2·n).

The optimiser works in three stages:

1. **Diverse initialisation** – for several lengths (≈2000,…,8000) we
   create a small pool of vectors (uniform, Dirichlet, exponential,
   Hann, Blackman, perturbed uniform, and an interpolated version of
   the best known construction, if available).  All vectors are
   ℓ₁‑normalised to √(2·n).

2. **Adam optimisation with a soft‑max surrogate** – for each start we
   run a few hundred‑thousand Adam steps.  The gradient of the
   smoothed max‑convolution is obtained via a weighted reverse
   convolution; the temperature τ decays exponentially, so the landscape
   becomes gradually sharper.  After every step we project onto the
   feasible set (non‑negative, correct ℓ₁‑norm) and add a tiny jitter
   every few thousand steps to avoid plateaus.

3. **Stochastic hill‑climbing refinement** – the best vector found is
   further improved by repeatedly transferring a small amount of mass
   from a random source index to a random destination index.  A move is
   kept if it lowers the exact max‑convolution; otherwise it is undone.
   The transferred amount slowly shrinks, allowing both coarse and fine
   adjustments.

The whole search respects the wall‑clock limit (budget_s seconds) and
returns the best sequence as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities for the convolution‑based gradient
# ----------------------------------------------------------------------
def _softmax_weights(conv: np.ndarray, tau: float):
    """Stable soft‑max of the convolution values.
    Returns weight vector w (summing to 1) and the smoothed maximum Lτ."""
    maxc = conv.max()
    if tau <= 0.0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w, maxc
    ex = np.exp((conv - maxc) / tau)
    sum_ex = ex.sum()
    w = ex / sum_ex
    Ltau = maxc + tau * np.log(sum_ex)
    return w, Ltau


def _weighted_rev(a: np.ndarray, w: np.ndarray, Nfft: int) -> np.ndarray:
    """Compute rev_i = Σ_k w_k * a_{k‑i} using FFT (sub‑gradient of the
    soft‑max max‑convolution)."""
    A_fft = np.fft.rfft(a, n=Nfft)
    # reverse the weight vector before FFT
    W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)
    conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
    conv_rev = conv_rev[: len(w) + a.shape[0] - 1]
    # slice that corresponds to rev_i for i=0..n‑1
    idx_start = 2 * a.shape[0] - 2
    rev = conv_rev[idx_start - np.arange(a.shape[0])]
    return rev


def _max_convolution(a: np.ndarray, Nfft: int) -> float:
    """Exact max of the linear convolution a∗a (full length)."""
    A_fft = np.fft.rfft(a, n=Nfft)
    conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * a.shape[0] - 1]
    return float(conv.max())


def _next_pow2(x: int) -> int:
    """Small utility – next power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
# Adam optimiser (soft‑max + exact phase)
# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int,
    best_seq: np.ndarray,
    best_score: float,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser for the max‑convolution objective.
    Returns the best sequence seen in this run and its exact score
    (i.e. the maximal convolution value after ℓ₁‑normalisation).
    """
    target_sum = math.sqrt(2.0 * n)

    # Normalise the initial vector
    a = a0 * (target_sum / a0.sum())

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr0 = 0.07                # slightly larger than the previous 0.05
    jitter_interval = 4000    # more frequent jitter

    # Temperature schedule parameters
    tau_factor = 0.25        # start with a larger fraction of the current max
    decay_rate = max_iters / 12.0

    Nfft = _next_pow2(2 * n - 1)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ----- exact convolution -----
        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        M = float(conv.max())

        # ----- temperature & soft‑max -----
        tau = tau_factor * M * math.exp(-it / decay_rate)
        if tau < 5e-10:
            # exact sub‑gradient (all maximal indices equally weighted)
            mask = conv >= M - 1e-12
            cnt = mask.sum()
            w = mask.astype(float) / max(cnt, 1)
            Ltau = M
        else:
            w, Ltau = _softmax_weights(conv, tau)

        # ----- weighted reverse convolution (gradient of smoothed max) -----
        rev = _weighted_rev(a, w, Nfft)

        # ----- gradient of the *real* objective (after projection) -----
        # The objective for a fixed ℓ₁‑norm equals max(conv); the gradient
        # of max ≈ rev, but we also need the term that keeps Σa fixed.
        coeff = 4.0 * n / (target_sum * target_sum)          # =2 after normalisation
        const = 4.0 * n * Ltau / (target_sum ** 3)           # enforces Σa = const
        grad = coeff * rev - const

        # Adam updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / (1.0 + 1e-4 * it)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # ----- projection onto feasible set -----
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a = a * (target_sum / s)
        else:
            a = np.full_like(a, target_sum / n)

        # occasional jitter
        if it % jitter_interval == 0:
            jitter = np.random.randn(n) * (1e-3 * target_sum / n)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        # ----- exact score (after projection it equals M) -----
        cur_score = M
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[len {n:5d} iter {it:8d}] new best = {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Simple stochastic hill‑climber (mass‑transfer moves)
# ----------------------------------------------------------------------
def _local_refine(
    a: np.ndarray,
    n: int,
    target_sum: float,
    best_seq: np.ndarray,
    best_score: float,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Perform a large number of cheap mass‑transfer moves.
    Each move transfers a small amount of mass from a random source
    index to a random destination index and keeps the move if the
    maximal convolution value decreases.
    """
    Nfft = _next_pow2(2 * n - 1)
    cur_max = _max_convolution(a, Nfft)

    # initial step size – about 1 % of the total ℓ₁‑norm
    step = max(target_sum * 0.01, 1e-4)

    while time.time() < deadline:
        # pick source with some mass
        i = rng.integers(n)
        if a[i] <= 1e-12:
            continue
        j = rng.integers(n)
        if i == j:
            continue

        amt = step * rng.random()
        amt = min(amt, a[i])       # never go negative

        # apply move in‑place
        old_i, old_j = a[i], a[j]
        a[i] = old_i - amt
        a[j] = old_j + amt

        new_max = _max_convolution(a, Nfft)

        if new_max < cur_max - 1e-12:      # accept improvement
            cur_max = new_max
            if cur_max < best_score:
                best_score = cur_max
                best_seq = a.copy()
                print(f"[refine n={n}] improved to {best_score:.6f}")
        else:
            # revert
            a[i] = old_i
            a[j] = old_j

        # slowly shrink the step size – allows fine tuning later
        step = max(step * 0.9999, target_sum * 1e-6)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Initial vector factory
# ----------------------------------------------------------------------
def _make_initial_vectors(n: int, target_sum: float) -> list[np.ndarray]:
    """Generate a small, diverse pool of start vectors for length n."""
    rng = np.random.default_rng()
    init = []

    # 1) uniform
    init.append(np.full(n, target_sum / n))

    # 2) random Dirichlet (Gamma(1,1) i.i.d.)
    g = rng.gamma(1.0, 1.0, size=n)
    init.append(g / g.sum() * target_sum)

    # 3) exponential decay (random rate)
    rate = rng.uniform(0.0005, 0.01)
    x = np.exp(-rate * np.arange(n))
    init.append(x / x.sum() * target_sum)

    # 4) reversed exponential
    init.append(x[::-1] / x[::-1].sum() * target_sum)

    # 5) Hann window
    hann = 0.5 - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1))
    init.append(hann / hann.sum() * target_sum)

    # 6) Blackman window
    a0, a1, a2 = 0.42, 0.5, 0.08
    n_arr = np.arange(n)
    black = (a0
             - a1 * np.cos(2.0 * np.pi * n_arr / (n - 1))
             + a2 * np.cos(4.0 * np.pi * n_arr / (n - 1)))
    init.append(black / black.sum() * target_sum)

    # 7) small random perturbation of the uniform vector
    pert = np.full(n, target_sum / n)
    pert += rng.normal(scale=0.01 * target_sum / n, size=n)
    pert = np.maximum(pert, 0.0)
    pert = pert / pert.sum() * target_sum
    init.append(pert)

    return init
