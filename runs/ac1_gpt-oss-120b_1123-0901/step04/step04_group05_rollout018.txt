def propose_candidate(seed=123, budget_s=1000):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.1

    best_seq = None
    best_val = float('inf')

    # If global best has variable name height_sequence_1, try to use
    if 'height_sequence_1' in globals():
        try:
            seq0 = np.array(globals()['height_sequence_1'], dtype=np.float64)
            # Normalize
            seq0 = np.maximum(seq0, 0.0)
            total = seq0.sum()
            if total < 1e-8:
                seq0 = np.random.dirichlet(np.ones(len(seq0)))
            else:
                seq0 = seq0 / total
            best_seq = seq0
            best_val = compute_obj(seq0)
            print(f"Initialized from existing sequence length {len(seq0)} with objective {best_val:.6f}")
        except Exception as e:
            print("Failed to use height_sequence_1:", e)
            best_seq = None

    # Define lengths to try
    lengths = [2000,2500,3000,3500,4000]  # sorted

    # Phase 1: multiple restarts
    restart = 0
    while time.time() < deadline - 30:  # keep 30s for later phases
        # Choose length
        n = lengths[restart % len(lengths)]

        # init a0
        if best_seq is not None and len(best_seq) == n and np.random.rand() < 0.2:
            a0 = best_seq.copy()
        else:
            mode = restart % 7
            if mode == 0:
                a0 = np.random.dirichlet(np.ones(n))
            elif mode == 1:
                r = np.random.uniform(0.2, 0.9)
                a0 = r ** np.arange(n, dtype=np.float64)
                a0 /= a0.sum()
            elif mode == 2:
                i = np.arange(1, n+1, dtype=np.float64)
                a0 = np.sin(np.pi * i / (n + 1))
                a0 = np.maximum(a0, 0.0)
                a0 /= a0.sum()
            elif mode == 3:
                p = np.random.uniform(0.5, 2.0)
                a0 = (np.sin(np.pi * np.arange(1, n + 1) / (n + 1))) ** p
                a0 = np.maximum(a0, 0.0)
                a0 /= a0.sum()
            elif mode == 4:
                r = np.random.uniform(0.3, 0.85)
                a0 = r ** np.arange(n, dtype=np.float64)
                a0 += 0.01 * np.random.rand(n)
                a0 = np.maximum(a0, 0.0)
                a0 /= a0.sum()
            elif mode == 5:
                x = np.linspace(-1.0, 1.0, n)
                a0 = 1.0 - x ** 2
                a0 = np.maximum(a0, 0.0)
                a0 /= a0.sum()
            else:
                half = n // 2
                left = np.sin(np.pi * (np.arange(half) + 1) / (half + 1))
                right = np.sin(np.pi * (np.arange(n - half) + 1) / (n - half + 1))
                a0 = np.concatenate([left, right])
                a0 = np.maximum(a0, 0.0)
                a0 /= a0.sum()

        # Parameterize via logits
        u = np.log(a0 + 1e-30)
        m = np.zeros_like(u)
        v = np.zeros_like(u)

        # Precompute FFT sizes
        conv_fft = 1 << int(np.ceil(np.log2(2 * n - 1)))
        # For correlation of w and a maybe need size_corr up to 3n
        size_corr_max = 1 << int(np.ceil(np.log2(3 * n)))

        # Hyperparameters
        lam = 0.5
        lam_factor = 1.008   # slower increase to give more smooth steps
        base_lr = 0.05
        decay = 0.9996

        # Core loop
        it = 0
        max_iter_smooth = 12000
        max_iter_exact = 8000
        last_improve = time.time()

        # --- Smooth phase ---
        while it < max_iter_smooth and time.time() < deadline - 30:
            it += 1
            # compute a from logits
            u_shift = u - np.max(u)
            exp_u = np.exp(u_shift)
            a = exp_u / exp_u.sum()
            # convolution via FFT
            fa = np.fft.rfft(a, conv_fft)
            conv = np.fft.irfft(fa * fa, conv_fft)[:2*n-1]
            # softmax weights
            w = _softmax_weights(conv, lam)
            # gradient wrt a
            grad_a = _softmax_gradient(a, w, size_corr=size_corr_max)
            # gradient in logit space
            dot = np.dot(a, grad_a)
            grad_u = a * (grad_a - dot)
            # Adam update
            m = 0.9 * m + 0.1 * grad_u
            v = 0.999 * v + 0.001 * (grad_u ** 2)
            it_f = float(it)
            m_hat = m / (1.0 - 0.9 ** it_f)
            v_hat = v / (1.0 - 0.999 ** it_f)
            lr = base_lr * (decay ** it)
            u -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)
            # raise lambda

"""
Hybrid optimizer for the self‑convolution bound.

The search proceeds in three stages:

1. *Smooth Adam* – a standard Adam optimiser on a soft‑max
   surrogate of  max (a * a).  This provides a gentle landscape
   that steers the iterate towards promising regions.

2. *Exact max Adam* – after the surrogate has become sharp
   the algorithm switches to the true sub‑gradient of the max
   (the gradient of the most violated convolution entry).  The
   parameters are still updated in the log‑space (soft‑max)
   representation, guaranteeing non‑negativity and the unit sum.

3. *Local hill‑climbing* – the best sequence discovered so far is
   refined by tiny mass transfers between random coordinates and,
   occasionally, a small global perturbation.  This stage can
   escape shallow local minima that Adam may miss.

The routine respects a user supplied time budget (default 1000 s)
and returns the best sequence (plain list of floats) found.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray, size: int = None) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``size``."""
    n = a.shape[0]
    if size is None:
        size = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return conv


def _softmax_weights(conv: np.ndarray, lam: float) -> np.ndarray:
    """Smooth‑max weights wₖ = exp(lam·convₖ) / Σⱼ exp(lam·convⱼ)."""
    m = conv.max()
    w = np.exp(lam * (conv - m))
    s = w.sum()
    if s == 0.0:                     # safety – should not happen
        w.fill(1.0 / w.shape[0])
    else:
        w /= s
    return w


def _softmax_gradient(a: np.ndarray, w: np.ndarray, size_corr: int = None) -> np.ndarray:
    """
    Gradient of the smooth‑max surrogate with respect to the original
    probabilities ``a``.  The closed form is  ∇ = 2·(w ★ a) .
    """
    n = a.shape[0]
    rev_a = a[::-1]
    if size_corr is None:
        size_corr = 1 << int(np.ceil(np.log2(len(w) + n - 1)))
    fw = np.fft.rfft(w, size_corr)
    frev = np.fft.rfft(rev_a, size_corr)
    corr = np.fft.irfft(fw * frev, size_corr)
    grad = 2.0 * corr[n - 1 : n - 1 + n]
    return grad


def _grad_max_convolution(a: np.ndarray, idx: int) -> np.ndarray:
    """
    Exact gradient of the max‑entry of a*a at position ``idx``.
    For each i the gradient is 2·a[idx‑i] whenever the index lies in [0,n).
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    lo = max(0, idx - (n - 1))
    hi = min(n - 1, idx)
    i = np.arange(lo, hi + 1, dtype=np.int64)
    j = idx - i
    grad[i] = 2.0 * a[j]
    return grad

    """Exact value of evaluate_sequence up to the constant factor."""
    n = a.shape[0]
    conv = np.convolve(a, a)
    return 2.0 * n * conv.max() / (a.sum() ** 2)
