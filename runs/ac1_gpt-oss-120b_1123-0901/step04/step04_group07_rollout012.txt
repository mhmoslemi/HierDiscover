def _exp_gradient_optimize(a0, n, deadline, max_iters, best_seq, best_score):
    target_sum = sqrt(2*n)

    # Set logits y0 = log(a0 + eps)
    eps = 1e-12
    y = np.log(a0 + eps)

    # Adam parameters: m, v arrays same shape as y.
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1, beta2, epsadam = 0.9, 0.999, 1e-8
    lr0 = 0.1   # maybe larger due to scale.
    jitter_interval = 2000
    tau_factor = 0.5   # initial tau factor relative to M.
    decay_rate = max_iters / 8.0  # moderate

    Nfft = 1 << ((2*n - 1).bit_length())  # or maybe larger for weighted rev: 1<<((3*n-2).bit_length())

    for iteration in 1..max_iters:
        if time.time() >= deadline: break

        # compute a = target_sum * softmax(y)
        logits = y - np.max(y)   # shift for stability
        exp_logits = np.exp(logits)
        p = exp_logits / exp_logits.sum()
        a = target_sum * p

        # compute convolution: A_fft = rfft(a,Nfft); conv = irfft(...); M = conv.max()
        # compute weights w via softmax with tau schedule.
        tau = tau_factor * M * math.exp(-iteration / decay_rate)
        if tau < 1e-9:
            mask = conv >= M - 1e-12
            cnt = np.sum(mask)
            w = mask.astype(float) / max(cnt, 1)
            Ltau = M
        else:
            w, Ltau = _softmax_weights(conv, tau)

        # compute weighted reverse conv rev = _weighted_rev(a, w, Nfft)
        rev = _weighted_rev(a, w, Nfft)

        # gradient wrt a: g_a = 2 * rev
        g_a = 2.0 * rev

        # compute avg = np.dot(p, g_a)
        avg = np.dot(p, g_a)

        # gradient wrt logits: g_y = target_sum * p * (g_a - avg)
        # note p = a / target_sum
        g_y = target_sum * p * (g_a - avg)

        # Adam update on y
        m = beta1 * m + (1.0 - beta1) * g_y
        v = beta2 * v + (1.0 - beta2) * (g_y * g_y)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        lr = lr0 / (1.0 + 1e-4 * iteration)

"""
Propose a non‑negative step function (a vector) that minimises
``evaluate_sequence``.

Algorithm
---------
* Initialise a diverse pool of start vectors (uniform, Dirichlet,
  exponential, windows, small random perturbations) for several lengths.
* Optimise each start with a *soft‑max Adam* on the **log‑its**
  (exponentiated‑gradient parametrisation) which automatically enforces
  a≥0 and Σa = √(2 n).  A temperature τ controls a smooth approximation of
  max (a∗a); τ is annealed exponentially.
* Gradient of the smooth max is obtained via a weighted reverse convolution
  (FFT‑based, O(n log n)).  Adam updates are applied to the logits.
* The best vector of every run is kept globally.
* After the time budget for the Adam phase expires, a cheap stochastic
  refinement (pairwise mass transfers with simulated‑annealing acceptance)
  is performed on the current global best to squeeze out the last bits.
* The final vector is returned as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – all are top‑level (no closures)

def _softmax_weights(conv: np.ndarray, tau: float):
    """Stable soft‑max of the convolution values.
    Returns (weights, smoothed max).
    """
    maxc = conv.max()
    if tau <= 0.0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w, maxc
    # subtract max for numerical stability
    ex = np.exp((conv - maxc) / tau)
    sum_ex = ex.sum()
    w = ex / sum_ex
    Ltau = maxc + tau * np.log(sum_ex)
    return w, Ltau


def _weighted_rev(a: np.ndarray, w: np.ndarray, Nfft: int) -> np.ndarray:
    """
    Compute rev_i = Σ_k w_k * a_{k-i} (out‑of‑range indices ignored)
    using FFTs.  This is the gradient of the (soft‑max) max‑convolution.
    """
    A_fft = np.fft.rfft(a, n=Nfft)
    W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)          # reverse w before FFT
    conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
    # keep the linear part of the convolution
    conv_rev = conv_rev[: len(w) + a.shape[0] - 1]    # length = 3 n‑2
    idx_start = 2 * a.shape[0] - 2
    rev = conv_rev[idx_start - np.arange(a.shape[0])]
    return rev


def _max_convolution(a: np.ndarray, Nfft: int) -> float:
    """Return max_k (a∗a)[k] using an FFT of length Nfft."""
    A_fft = np.fft.rfft(a, n=Nfft)
    conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * a.shape[0] - 1]
    return float(conv.max())


def _make_initial_vectors(n: int, target_sum: float, rng: np.random.Generator):
    """Create a small diverse set of normalised start vectors."""
    init = []

    # 1) uniform
    init.append(np.full(n, target_sum / n))

    # 2) Dirichlet (Gamma(1,1))
    g = rng.gamma(shape=1.0, scale=1.0, size=n)
    init.append(g / g.sum() * target_sum)

    # 3) exponential decay (random rate)
    rate = rng.uniform(0.0005, 0.01)
    x = np.exp(-rate * np.arange(n))
    init.append(x / x.sum() * target_sum)

    # 4) reversed exponential
    init.append(x[::-1] / x[::-1].sum() * target_sum)

    # 5) Hann window
    hann = 0.5 - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1))
    init.append(hann / hann.sum() * target_sum)

    # 6) Blackman window
    a0, a1, a2 = 0.42, 0.5, 0.08
    n_arr = np.arange(n)
    black = (a0
             - a1 * np.cos(2.0 * np.pi * n_arr / (n - 1))
             + a2 * np.cos(4.0 * np.pi * n_arr / (n - 1)))
    init.append(black / black.sum() * target_sum)

    # 7) small random perturbation of the uniform vector
    pert = np.full(n, target_sum / n) + rng.normal(scale=0.01 * target_sum / n, size=n)
    pert = np.maximum(pert, 0.0)
    init.append(pert / pert.sum() * target_sum)

    # 8) triangular shape (peak in the centre)
    tri = np.minimum(np.arange(n), n - 1 - np.arange(n))
    init.append(tri / tri.sum() * target_sum)

    return init


def _exp_gradient_optimize(
    a0: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑its (exponentiated‑gradient) using a soft‑max
    approximation of the max‑convolution.  Returns the best vector seen in
    this run together with its exact objective value (i.e. max(a∗a) after
    normalisation).
    """
    target_sum = math.sqrt(2.0 * n)

    # initialise logits from the start vector (add tiny epsilon to avoid log(0))
    eps = 1e-16
    y = np.log(a0 + eps)

    # Adam state
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1, beta2, epsadam = 0.9, 0.999, 1e-8
    lr0 = 0.08                      # base learning rate for logits
    jitter_interval = 3000

    # temperature (soft‑max) schedule
    tau_factor = 0.30               # τ₀ = τ_factor·M
    decay_rate = max_iters / 8.0    # exponential decay factor

    # FFT size – must be large enough for linear conv of length 3 n‑2
    Nfft = 1 << ((3 * n - 2).bit_length())

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ----- current vector (softmax of logits) -----
        logits = y - np.max(y)               # shift for numerical stability
        exp_logits = np.exp(logits)
        p = exp_logits / exp_logits.sum()    # probability simplex
        a = target_sum * p                   # guaranteed Σa = target_sum, a_i≥0

        # ----- exact convolution -----
        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        M = float(conv.max())

        # ----- soft‑max weights for the max -----
        tau = tau_factor * M * math.exp(-it / decay_rate)
        if tau < 1e-9:
            mask = conv >= M - 1e-12
            cnt = mask.sum()
            w = mask.astype(float) / max(cnt, 1)
            Ltau = M
        else:
            w, Ltau = _softmax_weights(conv, tau)

        # ----- weighted reverse convolution (gradient of smoothed max) -----
        rev = _weighted_rev(a, w, Nfft)

        # ----- gradient wrt a (2·rev, see derivation) -----
        g_a = 2.0 * rev

        # ----- chain rule to logits (mirror‑descent on the simplex) -----
        avg = np.dot(p, g_a)                 # ⟨p, g_a⟩
        g_y = target_sum * p * (g_a - avg)   # ∂f/∂y_i

        # ----- Adam update on logits -----
        m = beta1 * m + (1.0 - beta1) * g_y
        v = beta2 * v + (1.0 - beta2) * (g_y * g_y)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / (1.0 + 1e-4 * it)
        y = y - lr * m_hat / (np.sqrt(v_hat) + epsadam)

        # occasional jitter to escape plateaux
        if it % jitter_interval == 0:
            jitter = rng.normal(scale=0.01, size=n)   # additive jitter on logits
            y += jitter

        # ----- keep best -----
        if M < best_score:
            best_score = M
            best_seq = a.copy()
            print(f"[len {n:5d} iter {it:8d}] new best = {best_score:.6f}")

    return best_seq, best_score


def _local_refine(
    a: np.ndarray,
    n: int,
    deadline: float,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Simple stochastic hill‑climbing on the simplex:
    repeatedly move a small fraction of mass from a random index i to j.
    Accept improvements always; accept worsening moves with a decaying
    simulated‑annealing probability.  Returns an improved vector (or the
    original one if no improvement is found before the deadline).
    """
    target_sum = a.sum()                     # already √(2 n)
    cur_seq = a.copy()
    cur_score = best_score

    # FFT size for fast max‑convolution
    Nfft = 1 << ((2 * n - 1).bit_length())

    # annealing schedule
    max_steps = 200_000
    step = 0
    while time.time() < deadline and step < max_steps:
        i, j = rng.integers(0, n, size=2, endpoint=False)
        if i == j or cur_seq[i] < 1e-12:
            step += 1
            continue

        # transfer a random fraction (1 % – 5 %) of the source mass
        frac = rng.uniform(0.01, 0.05)
        delta = cur_seq[i] * frac
        if delta <= 0.0:
            step += 1
            continue

        new_seq = cur_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        # compute new max‑convolution
        A_fft = np.fft.rfft(new_seq, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        new_score = float(conv.max())

        # deterministic improvement
        if new_score < cur_score - 1e-12:
            cur_seq = new_seq
            cur_score = new_score
            print(f"[refine] improvement → {cur_score:.6f} (step {step})")
        else:
            # simulated‑annealing acceptance
            T = max(0.01 * cur_score, 1e-5) * (1.0 - step / max_steps)
            prob = math.exp(-(new_score - cur_score) / max(T, 1e-12))
            if rng.random() < prob:
                cur_seq = new_seq
                cur_score = new_score
        step += 1

    return cur_seq, cur_score
