def proj_simplex(v):
    "Project onto simplex: nonnegative and sum = 1."
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
A fast projected sub‑gradient + stochastic hill‑climbing optimiser.
The search works on a probability vector p (∑p=1, p_i≥0).  The true
sequence is a = p·√(2·n) which satisfies the normalisation used by the
official evaluator (sum(a)=√(2·n)).  The objective for a is
    F(a) = 2·n·max(a∗a)/(sum(a)²) = 2·n·max(p∗p)
so minimising max(p∗p) is sufficient.

The optimiser consists of two stages:

1. **Projected sub‑gradient descent** – the exact sub‑gradient of the
   current maximal convolution entry is taken, a step is performed,
   and the result is projected back onto the simplex.  A simple back‑
   tracking line‑search is used, together with a periodic “shake”
   to escape plateaus.

2. **Stochastic hill‑climbing** – after the first stage we repeatedly
   transfer a small fraction of probability mass from a “bad’’ index
   (large gradient) to a random other index, keeping the move only if
   it reduces the maximal convolution value.  The step size is
   annealed.

Both stages are run several times with different random starts and
different lengths (1024, 2048, 4096).  The best sequence encountered
before the time budget expires is returned as a plain Python list.

The code respects the 1000 s budget, prints progress, and uses only
NumPy (no external optimisation libraries).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
#  Helper utilities
# ----------------------------------------------------------------------
def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of v onto the probability simplex {x≥0, Σx = 1}."""
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # find the rho index
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(p: np.ndarray) -> np.ndarray:
    """Linear convolution p*p using an FFT (real output)."""
    n = p.shape[0]
    L = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fp = np.fft.rfft(p, L)
    conv = np.fft.irfft(fp * fp, L)[: 2 * n - 1]
    return conv.real


def conv_max(p: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of p."""
    return full_convolution(p).max()


def subgradient_max(p: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of max(p∗p) at the index k where the maximum occurs.
    Returns g where g_j = 2·p_{k‑j} for all valid j, else 0.
    """
    n = p.shape[0]
    g = np.zeros_like(p)
    start = max(0, k - (n - 1))
    end = min(k, n - 1)
    i = np.arange(start, end + 1, dtype=np.int64)
    j = k - i
    g[i] = 2.0 * p[j]
    return g


def compute_objective_from_p(p: np.ndarray, n: int) -> float:
    """Exact objective for the scaled sequence a = p·√(2·n)."""
    # Since sum(a)=√(2·n), the objective reduces to 2·n·max(p∗p)
    return 2.0 * n * conv_max(p)


# ----------------------------------------------------------------------
#  Optimisation phases
# ----------------------------------------------------------------------
def run_projected_subgrad(
    n: int,
    init_p: np.ndarray,
    deadline: float,
    step0: float = 0.12,
    max_iters: int = 200_000,
):
    """
    Projected sub‑gradient descent on max(p∗p).  Returns the best
    probability vector found and its maximal convolution value.
    """
    p = init_p.copy()
    best_p = p.copy()
    best_val = conv_max(p)
    step = step0
    it = 0
    last_improve = time.time()

    while time.time() < deadline and it < max_iters:
        conv = full_convolution(p)
        k = int(conv.argmax())
        cur_val = conv[k]

        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_p = p.copy()
            last_improve = time.time()
            print(f"[SubGrad n={n} it={it:05d}] new best max={best_val:.6e}")

        # gradient of the *maximum* entry
        g = subgradient_max(p, k)

        # line‑search (simple back‑tracking)
        improved = False
        trial_step = step
        while trial_step > 1e-8:
            p_trial = proj_simplex(p - trial_step * g)
            new_val = conv_max(p_trial)
            if new_val < cur_val - 1e-14:  # strict improvement
                p = p_trial
                step = trial_step * 1.05   # slightly increase after success
                improved = True
                break
            trial_step *= 0.5

        if not improved:
            step *= 0.5  # shrink step if we are stuck

        # occasional random shake to avoid stagnation
        if time.time() - last_improve > 5.0:
            p = proj_simplex(p + 0.001 * np.random.randn(n))
            last_improve = time.time()
            print(f"[SubGrad n={n} it={it}] shake")

        it += 1

    return best_p, best_val


def run_hill_climbing(
    n: int,
    init_p: np.ndarray,
    deadline: float,
    start_eps: float = 0.03,
):
    """
    Simple (1+1) stochastic hill‑climber.  A small amount of mass is
    moved from an index with large gradient to a random other index.
    Only uphill moves are accepted (no temperature – pure greedy).
    """
    p = init_p.copy()
    best_p = p.copy()
    best_val = conv_max(p)

    eps = start_eps
    it = 0
    last_improve = time.time()

    while time.time() < deadline:
        # recompute gradient once per iteration
        conv = full_convolution(p)
        k = int(conv.argmax())
        g = subgradient_max(p, k)

        # pick a source index proportionally to its gradient magnitude
        if g.sum() == 0:
            src = np.random.randint(n)
        else:
            prob = g / g.sum()
            src = np.random.choice(np.arange(n), p=prob)

        # pick a different destination uniformly
        dst = np.random.randint(n - 1)
        if dst >= src:
            dst += 1

        delta = eps * p[src]          # move a fraction of the source mass
        if delta <= 0.0:
            eps *= 0.5
            continue

        p_new = p.copy()
        p_new[src] -= delta
        p_new[dst] += delta

        # project not needed – the move preserves the simplex exactly
        new_val = conv_max(p_new)

        if new_val < best_val - 1e-15:
            best_val = new_val
            best_p = p_new.copy()
            p = p_new
            last_improve = time.time()
            # slowly enlarge step size after a success
            eps = min(eps * 1.08, 0.2)
            if it % 500 == 0:
                print(f"[Hill n={n} it={it}] improved max={best_val:.6e}")
        else:
            # no improvement – shrink the step size
            eps *= 0.96

        # restart the step size if we have been stuck for a while
        if time.time() - last_improve > 6.0:
            eps = start_eps
            last_improve = time.time()

        it += 1

    return best_p, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the supplied
    evaluation function.  The routine respects the total time budget
    (default 1000 s) and returns the best sequence found as a plain
    Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # tiny safety margin

    best_seq = None
    best_obj = np.inf

    # try a few different lengths – larger n gives more freedom
    lengths = [1024, 2048, 4096]

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")

        sqrt_2n = np.sqrt(2.0 * n)

        # --------------------------------------------------------------
        # Build a small collection of sensible starts (probability vectors)
        # --------------------------------------------------------------
        starts = []

        # warm‑start from the globally best sequence if the size matches
        if best_seq is not None and len(best_seq) == n:
            p = np.asarray(best_seq, dtype=np.float64) / sqrt_2n
            starts.append(p)

        # use the historic construction if it exists
        if "height_sequence_1" in globals():
            hs = np.asarray(height_sequence_1, dtype=np.float64)
            if hs.size >= n:
                hs = hs[:n]
            else:
                hs = np.pad(hs, (0, n - hs.size))
            hs = np.maximum(hs, 0.0)
            s = hs.sum()
            if s > 0.0:
                starts.append(hs / s)

        # simple uniform start
        starts.append(np.full(n, 1.0 / n, dtype=np.float64))

        # a few random Dirichlet draws
        for _ in range(2):
            starts.append(np.random.dirichlet(np.ones(n)))

        # ------------------------------------------------------------------
        # Run the optimisation pipeline for each start (time slice allocated)
        # ------------------------------------------------------------------
        for start_id, p0 in enumerate(starts):
            if time.time() > deadline:
                break

            # ---- Projected sub‑gradient descent ----
            sub_deadline = time.time() + min((deadline - time.time()) * 0.4, 90.0)
            p_opt, max_opt = run_projected_subgrad(
                n, p0, sub_deadline, step0=0.12, max_iters=200_000
            )
            obj_opt = 2.0 * n * max_opt
            if obj_opt < best_obj:
                best_obj = obj_opt
                best_seq = (p_opt * sqrt_2n).tolist()
                print(
                    f"[n={n} start={start_id}] SUB‑GRAD GLOBAL best {best_obj:.6f}"
                )

            # ---- Stochastic hill‑climbing fine‑tuning ----
            if time.time() < deadline:
                hill_deadline = time.time() + min((deadline - time.time()) * 0.3, 60.0)
                p_hc, max_hc = run_hill_climbing(n, p_opt, hill_deadline, start_eps=0.03)
                obj_hc = 2.0 * n * max_hc
                if obj_hc < best_obj:
                    best_obj = obj_hc
                    best_seq = (p_hc * sqrt_2n).tolist()
                    print(
                        f"[n={n} start={start_id}] HILL‑CLIMB GLOBAL best {best_obj:.6f}"
                    )

        # ------------------------------------------------------------------
        # Try a brief population‑based search (very small diff‑evolution)
        # ------------------------------------------------------------------
        if time.time() < deadline:
            popsize = 6
            pop = []
            # seed the population with the best vectors we have so far
            if best_seq is not None and len(best_seq) == n:
                p_best = np.asarray(best_seq, dtype=np.float64) / sqrt_2n
                pop.append(p_best)
            while len(pop) < popsize:
                pop.append(np.random.dirichlet(np.ones(n)))
            # simple DE loop (no fancy logging)
            de_deadline = time.time() + min((deadline - time.time()) * 0.25, 80.0)
            F, CR = 0.8, 0.9
            obj_pop = [conv_max(p) for p in pop]
            best_pop_idx = int(np.argmin(obj_pop))
            best_pop = pop[best_pop_idx].copy()
            best_pop_val = obj_pop[best_pop_idx]

            while time.time() < de_deadline:
                for i in range(popsize):
                    idxs = list(range(popsize))
                    idxs.remove(i)
                    r1, r2, r3 = np.random.choice(idxs, 3, replace=False)
                    mutant = pop[r1] + F * (pop[r2] - pop[r3])
                    mutant = np.maximum(mutant, 0.0)
                    cross = np.random.rand(n) < CR
                    cross[np.random.randint(n)] = True
                    trial = np.where(cross, mutant, pop[i])
                    trial = proj_simplex(trial)
                    val = conv_max(trial)
                    if val < obj_pop[i]:
                        pop[i] = trial
                        obj_pop[i] = val
                        if val < best_pop_val:
                            best_pop_val = val
                            best_pop = trial.copy()
                # early exit if population stopped improving
                if time.time() - de_deadline > 10.0:
                    break

            obj_de = 2.0 * n * best_pop_val
            if obj_de < best_obj:
                best_obj = obj_de
                best_seq = (best_pop * sqrt_2n).tolist()
                print(f"[n={n}] DE GLOBAL best {best_obj:.6f}")

    # ----------------------------------------------------------------------
    # Fallback – uniform distribution if nothing better was found.
    # ----------------------------------------------------------------------
    if best_seq is None:
        n_fallback = 1024
        p_fallback = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        best_seq = (p_fallback * np.sqrt(2.0 * n_fallback)).tolist()
        best_obj = compute_objective_from_p(p_fallback, n_fallback)

    print(f"\nFinished optimisation. Best objective = {best_obj:.6f}")
    # convert to plain Python list as required by the external evaluator
    return best_seq
