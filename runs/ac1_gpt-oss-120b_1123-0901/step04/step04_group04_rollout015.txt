def project_simplex(v, z=1.0):
    # v is array, we want nonnegative entries sum to z.
    # Return projection.
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero?
        theta = cssv[-1] / n
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a step‑function (non‑negative vector) that minimises the
``evaluate_sequence`` objective.

Algorithm
---------
1.  Work in the simplex:  p = a / Σa,  Σp = 1  (Σa = √(2·n)).
    Parameterise p by log‑weights y (softmax), guaranteeing feasibility.
2.  Smooth the max‑convolution with a log‑sum‑exp:
        Lτ(p) = τ·log Σk exp( (p∗p)k / τ )
    Its gradient can be written using a weighted reverse convolution.
3.  For several random starts run a continuation of L‑BFGS‑B:
    – start with τ ≈ 2·max(p∗p) (very smooth) and multiply τ by 0.7
      after each L‑BFGS stage,
    – use a fairly generous budget (≈500 function calls) per stage,
    – after each stage evaluate the exact objective and keep the best
      sequence.
4.  After the smooth optimisation perform a cheap hill‑climbing pass:
    repeatedly move a small amount of mass from one index to another,
    keep the move only if the exact max‑convolution improves.
5.  The best sequence found (exactly evaluated once more) is returned.

The code is deliberately written without closures or lambdas so it can be
executed in the constrained evaluation environment.
"""

import time
import math
from typing import List

import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy – if unavailable we fall back to a tiny gradient loop.
try:
    from scipy.optimize import minimize

    _SCIPY_AVAILABLE = True
except Exception:  # pragma: no cover
    _SCIPY_AVAILABLE = False

# ----------------------------------------------------------------------
# Helper: project onto the simplex Σx = total, x ≥ 0 (not used for the
# soft‑max parametrisation but handy for the local hill‑climbing step).
def _project_simplex(v: np.ndarray, total: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=total}."""
    if total < 0:
        raise ValueError("total must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - total
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = cssv[-1] / n
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities based on FFT (fast, used in gradients).
def _conv_full(p: np.ndarray, Nfft: int) -> np.ndarray:
    """Full discrete convolution p∗p (length 2·len(p)‑1) using FFT."""
    A = np.fft.rfft(p, n=Nfft)
    conv = np.fft.irfft(A * A, n=Nfft)[: 2 * p.shape[0] - 1]
    return conv


def _softmax_weights(conv: np.ndarray, tau: float):
    """Soft‑max of the convolution values.  Returns weights w (∑w=1) and Lτ."""
    maxc = conv.max()
    if tau <= 0.0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w, maxc
    # stable soft‑max
    ex = np.exp((conv - maxc) / tau)
    s = ex.sum()
    w = ex / s
    Ltau = maxc + tau * math.log(s)
    return w, Ltau


def _weighted_rev(a: np.ndarray, w: np.ndarray, Nfft: int) -> np.ndarray:
    """
    Compute rev_i = Σ_k w_k a_{k‑i} efficiently with FFTs.
    The result has the same length as ``a``.
    """
    A_fft = np.fft.rfft(a, n=Nfft)
    W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)
    conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
    conv_rev = conv_rev[: len(w) + a.shape[0] - 1]          # keep valid part
    # pick the slice that corresponds to the reversed convolution
    start = 2 * a.shape[0] - 2
    rev = conv_rev[start - np.arange(a.shape[0])]
    return rev


def _objective_and_grad(y: np.ndarray, tau: float, Nfft: int):
    """
    Smoothed objective Lτ and its gradient w.r.t. the log‑weights y.
    Returns (Lτ, ∇_y Lτ).
    """
    # soft‑max to obtain a probability vector p
    y_max = np.max(y)
    exp_y = np.exp(y - y_max)          # stability
    p = exp_y / exp_y.sum()

    # convolution p∗p
    conv = _conv_full(p, Nfft)

    # smooth maximum
    w, Ltau = _softmax_weights(conv, tau)

    # gradient w.r.t. p: 2·rev where rev_i = Σ_k w_k p_{k‑i}
    rev = _weighted_rev(p, w, Nfft)
    grad_p = 2.0 * rev

    # back‑prop through soft‑max (Jacobian of soft‑max)
    dot = np.dot(p, grad_p)            # scalar = Σ_i p_i·grad_p_i
    grad_y = p * (grad_p - dot)

    return Ltau, grad_y


# ----------------------------------------------------------------------
# Core optimisation routine (L‑BFGS with temperature continuation).
def _run_l_bfgs(
    y0: np.ndarray,
    init_tau: float,
    target_sum: float,
    Nfft: int,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
) -> tuple[np.ndarray, float]:
    """Optimise the smoothed max‑convolution starting from y0."""
    y = y0.copy()
    tau = max(init_tau, 1e-12)
    tau_decay = 0.7          # how fast the temperature shrinks
    min_tau = 1e-12

    while time.time() < deadline and tau > min_tau:
        # ------------------------------------------------------------------
        # L‑BFGS on the current temperature
        def fun(yy):
            L, g = _objective_and_grad(yy, tau, Nfft)
            return L, g

        if _SCIPY_AVAILABLE:
            opts = {
                "maxfun": 500,
                "maxiter": 500,
                "ftol": 1e-12,
                "gtol": 1e-12,
                "disp": False,
            }
            res = minimize(fun, y, method="L-BFGS-B", jac=True, options=opts)
            y = res.x
        else:
            # very simple gradient descent fallback (rarely used)
            for _ in range(200):
                L, g = _objective_and_grad(y, tau, Nfft)
                y -= 0.1 * g

        # ------------------------------------------------------------------
        # Exact evaluation of the current candidate
        y_max = np.max(y)
        p = np.exp(y - y_max)
        p /= p.sum()
        a = p * target_sum                     # real step‑function
        conv_p = _conv_full(p, Nfft)
        max_conv_p = conv_p.max()
        cur_score = (target_sum ** 2) * max_conv_p   # = evaluate_sequence value

        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[improve] τ={tau:.2e}  score={best_score:.6f}")

        # ------------------------------------------------------------------
        # Temperature schedule and occasional jitter
        tau *= tau_decay
        if np.random.rand() < 0.15:
            y += np.random.randn(y.shape[0]) * 0.02

        if time.time() >= deadline:
            break

    return best_seq, best_score


# ----------------------------------------------------------------------
# Simple hill‑climbing refinement on the raw sequence.
def _local_search(
    a: np.ndarray,
    target_sum: float,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
) -> tuple[np.ndarray, float]:
    """Move small amounts of mass between coordinates, keeping improvements."""
    n = a.shape[0]
    Nfft = 1 << ((2 * n - 1).bit_length())
    cur = a.copy()

    while time.time() < deadline:
        i, j = np.random.randint(0, n, size=2)
        if i == j or cur[i] <= 0.0:
            continue
        # move a random fraction (up to 50 %) of the mass at i to j
        delta = cur[i] * np.random.rand() * 0.5
        new = cur.copy()
        new[i] -= delta
        new[j] += delta

        p = new / target_sum
        conv_p = _conv_full(p, Nfft)
        cur_score = (target_sum ** 2) * conv_p.max()

        if cur_score + 1e-12 < best_score:
            best_score = cur_score
            best_seq = new.copy()
            cur = new.copy()
            print(f"[local improve] score={best_score:.6f}")

    return best_seq, best_score
