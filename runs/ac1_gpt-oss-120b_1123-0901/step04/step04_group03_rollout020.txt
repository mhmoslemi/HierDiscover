import time
import numpy as np
from scipy.optimize import linprog

def softmax(x):
    mx = np.max(x)
    e = np.exp(x - mx)
    return e / np.sum(e)

def full_convolution(a):
    n = a.shape[0]
    L = 1 << int(np.ceil(np.log2(2*n-1)))
    fa = np.fft.rfft(a, L)
    conv = np.fft.irfft(fa * fa, L)[:2*n-1]
    return conv.real

def conv_max(a):
    return full_convolution(a).max()

def compute_objective(a):
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return 2.0 * a.shape[0] * conv_max(a) / (s * s)

def grad_submax(a):
    n = a.shape[0]
    conv = full_convolution(a)
    max_c = conv.max()
    idxs = np.flatnonzero(np.isclose(conv, max_c, atol=1e-14))
    k = np.random.choice(idxs)
    g = np.zeros_like(a)
    start = max(0, k - (n - 1))
    end = min(k, n - 1)
    i = np.arange(start, end + 1)
    j = k - i
    g[i] = 2.0 * a[j]
    return g, max_c

    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.05
    best_val = np.inf
    best_a = None
    t = 0
    last_improve = time.time()
    while time.time() < deadline and t < max_iters:
        a = softmax(x)
        grad_a, max_c = grad_submax(a)
        val = 2.0 * n * max_c   # sum(a)=1
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[Adam n={n} iter={t:05d}] new best: {val:.6f}")
        # back‑prop through softmax
        avg = np.dot(a, grad_a)
        grad_x = a * (grad_a - avg)
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)
        x = np.clip(x, -50.0, 50.0)
        if time.time() - last_improve > 5.0:
            x += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()
            print(f"[Adam n={n} iter={t}] shake")
    if best_a is None:
        best_a = np.full(n, 1.0 / n)
        best_val = compute_objective(best_a)
    return best_a, best_val

def run_de_opt(n, init_pop, deadline, popsize=10, max_gens=20_000):

def build_conv_matrix(f):
    """Return dense matrix A where (A @ b)[k] = (f * b)[k] for linear convolution."""
    n = f.shape[0]
    N = 2 * n - 1
    A = np.zeros((N, n), dtype=np.float64)
    # each column j receives f shifted down by j
    for j in range(n):
        A[j:j + n, j] = f
    return A

def solve_lp_max_sum(f, M, max_val=1000.0):
    """Solve  max  sum(b)  s.t.  f * b <= M,  b >=0,  b <= max_val."""
    n = f.shape[0]
    A = build_conv_matrix(f)
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)   # maximize sum(b) -> minimize -sum(b)
    bounds = [(0.0, max_val) for _ in range(n)]
    # Using HiGHS solver (default in recent SciPy)
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        return None
    return res.x

def lp_refine(a, deadline, mix=0.3, max_steps=10):

        b = solve_lp_max_sum(best, M, max_val=1000.0)

            break

        b_norm = b / (b.sum() + 1e-12)

            best = a_new
            print(f"[LP] improvement to {best_val:.6f}")
        else:
            # no improvement – shrink mixing factor
            mix *= 0.5
            if mix < 1e-4:
                break
    return best, best_val

"""
A multi‑stage optimiser for the step‑function problem.

The pipeline (all stages respect an overall time budget):
1. Adam optimiser on the exact sub‑gradient of max(a∗a) while keeping a on the simplex.
2. Small differential‑evolution (DE) population exploring the simplex globally.
3. (1+1) evolution‑strategy (ES) fine‑tuning with Gaussian perturbations.
4. Deterministic linear‑programming refinement: 
   maximise ∑b_j under the current convolution bound ‖f∗b‖_∞≤M.
   This is the LP‑step described in the statement.  A convex combination of
   the current best and the normalised LP‑solution is taken repeatedly.

The code returns the best non‑negative sequence found as a plain list.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Basic convolution utilities
# ----------------------------------------------------------------------
def softmax(x: np.ndarray) -> np.ndarray:
    mx = np.max(x)
    e = np.exp(x - mx)
    return e / e.sum()


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using FFT (real‑valued result)."""
    n = a.shape[0]
    L = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, L)
    conv = np.fft.irfft(fa * fa, L)[: 2 * n - 1]
    return conv.real


def conv_max(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of a."""
    return full_convolution(a).max()


def compute_objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(a∗a)/(sum(a)²)."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return 2.0 * a.shape[0] * conv_max(a) / (s * s)


def grad_submax(a: np.ndarray):
    """Sub‑gradient of max(a∗a) w.r.t. a."""
    n = a.shape[0]
    conv = full_convolution(a)
    max_c = conv.max()
    idxs = np.flatnonzero(np.isclose(conv, max_c, atol=1e-14))
    k = np.random.choice(idxs)
    g = np.zeros_like(a)
    start = max(0, k - (n - 1))
    end = min(k, n - 1)
    i = np.arange(start, end + 1)
    j = k - i
    g[i] = 2.0 * a[j]
    return g, max_c


# ----------------------------------------------------------------------
# Optimisation phases already used in the previous baseline
# ----------------------------------------------------------------------
def run_adam_opt(
    n: int,
    init_a: np.ndarray,
    deadline: float,
    max_iters: int = 200_000,
):
    """Adam optimiser on the sub‑gradient of max(a∗a)."""
    x = np.log(init_a + 1e-15)

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.05

    best_val = np.inf
    best_a = None
    t = 0
    last_improve = time.time()

    while time.time() < deadline and t < max_iters:
        a = softmax(x)                     # a ≥0, Σa = 1
        grad_a, max_c = grad_submax(a)     # exact sub‑gradient
        val = 2.0 * n * max_c               # exact objective (sum(a)=1)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[Adam n={n} iter={t:05d}] new best: {val:.6f}")

        # back‑propagation through softmax
        avg = np.dot(a, grad_a)
        grad_x = a * (grad_a - avg)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

        # keep logits in a reasonable range to avoid overflow
        x = np.clip(x, -50.0, 50.0)

        # shake when stagnating
        if time.time() - last_improve > 5.0:
            x += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()
            print(f"[Adam n={n} iter={t}] shake (stagnation)")

    if best_a is None:
        best_a = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = compute_objective(best_a)
    return best_a, best_val

def build_conv_matrix(f: np.ndarray) -> np.ndarray:

    Return dense matrix A (size (2n‑1)×n) such that
        (A @ b)[k] = (f ∗ b)[k]   (linear convolution).
    """
    n = f.shape[0]
    N = 2 * n - 1
    A = np.zeros((N, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f

def solve_lp_max_sum(f: np.ndarray, M: float, max_val: float = 1000.0):

    if not res.success:

    return res.x

            best_val = val_new
            best = new_a
            print(f"[LP] improvement to {best_val:.6f}")
        else:
            mix *= 0.5
            if mix < 1e-4:
                break
    return best, best_val

# Main entry point required by the grader
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    the supplied evaluation function.

    The search pipeline consists of Adam, DE, ES, and a deterministic
    LP‑refinement step.  It respects the overall time budget and returns the
    best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_val = np.inf

    # Lengths to try – medium sizes give a good balance between expressivity
    # and LP‑solve cost.  The LP step is only executed for n ≤ 2048.
    lengths = [1024, 2048, 4096]

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")

        # ------------------------------------------------------------------
        # Build a few sensible initial candidates
        # ------------------------------------------------------------------
        init_candidates = []

        if best_seq is not None and len(best_seq) == n:
            init_candidates.append(np.array(best_seq, dtype=np.float64))

        if "height_sequence_1" in globals():
            hs = np.asarray(height_sequence_1, dtype=np.float64)
            if hs.size >= n:
                hs = hs[:n]
            else:
                hs = np.pad(hs, (0, n - hs.size))
            hs = np.maximum(hs, 0.0)
            s = hs.sum()
            if s > 0.01:
                hs = hs / s
            else:
                hs = np.random.dirichlet(np.ones(n))
            init_candidates.append(hs)

        dec = np.arange(n, 0, -1, dtype=np.float64)
        dec = dec / dec.sum()
        init_candidates.append(dec)

        init_candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

        # ------------------------------------------------------------------
        # Phase 1 – Adam (few restarts)
        # ------------------------------------------------------------------
        for r in range(3):
            if time.time() > deadline:
                break
            a0 = init_candidates[r] if r < len(init_candidates) else np.random.dirichlet(np.ones(n))
            slice_time = min((deadline - time.time()) * 0.15, 25.0)
            sub_deadline = time.time() + slice_time
            a_adam, val_adam = run_adam_opt(n, a0, sub_deadline)
            if val_adam < best_val:
                best_val = val_adam
                best_seq = a_adam.copy()
                print(f"[n={n} Adam restart={r}] global best {best_val:.6f}")

        # ------------------------------------------------------------------
        # Phase 2 – Differential Evolution
        # ------------------------------------------------------------------
        if time.time() < deadline:
            de_time = min((deadline - time.time()) * 0.30, 120.0)
            sub_deadline = time.time() + de_time
            de_pop = init_candidates[:3]               # warm start
            a_de, val_de = run_de_opt(n, de_pop, sub_deadline,
                                      popsize=12, max_gens=15_000)
            if val_de < best_val:
                best_val = val_de
                best_seq = a_de.copy()
                print(f"[n={n} DE] new global best {best_val:.6f}")

        # ------------------------------------------------------------------
        # Phase 3 – (1+1) ES fine‑tuning
        # ------------------------------------------------------------------
        if time.time() < deadline and best_seq is not None and len(best_seq) == n:
            es_time = min((deadline - time.time()) * 0.25, 60.0)
            sub_deadline = time.time() + es_time
            a_es, val_es = run_es_opt(n, best_seq, sub_deadline)
            if val_es < best_val:
                best_val = val_es
                best_seq = a_es.copy()
                print(f"[n={n} ES] new global best {best_val:.6f}")

        # ------------------------------------------------------------------
        # Phase 4 – LP refinement (only for moderate dimensions)
        # ------------------------------------------------------------------
        if (
            time.time() < deadline
            and best_seq is not None
            and len(best_seq) == n
            and n <= 2048          # keep memory footprint reasonable
        ):
            lp_time = min((deadline - time.time()) * 0.20, 80.0)
            sub_deadline = time.time() + lp_time
            a_lp, val_lp = lp_refine(best_seq, sub_deadline,
                                     mix=0.35, max_steps=12)
            if val_lp < best_val:
                best_val = val_lp
                best_seq = a_lp.copy()
                print(f"[n={n} LP] global best improved to {best_val:.6f}")

    # ----------------------------------------------------------------------
    # Fallback – return a plain uniform vector if nothing succeeded.
    # ----------------------------------------------------------------------
    if best_seq is None:
        fallback_n = 1024
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)
        best_val = compute_objective(best_seq)

    print(f"\nBest value discovered: {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
