    size = 1 << int(np.ceil(np.log2(len(u) + len(v) - 1)))
    fu = np.fft.rfft(u, size)
    fv = np.fft.rfft(v, size)

1. Gather a starting point (global variable `height_sequence_1` if present,
   otherwise a random uniform vector).
2. Perform a cheap parametric sweep over several simple families
   (sine^p, polynomial^p, exponential) for a few lengths.
   For each candidate we compute the exact value
        val = 2·n·max(conv(a,a)),
   which coincides with `evaluate_sequence` when ∑a = 1.
   The best candidate from this sweep becomes the seed for the next stage.
3. Sub‑gradient descent on the true non‑smooth objective:
   •  a is kept on the probability simplex (projection after every step).
   •  The sub‑gradient of  max_k (a*a)[k]  is
        g = 2·(mask ★ rev(a)),
      where `mask` is the indicator of the active (max‑tight) convolution
      entries and `★` denotes linear convolution (computed by FFT).
   •  We use a simple diminishing step size and shake the vector if the
      objective has not improved for a few seconds.
      The best vector seen is stored globally.
4. A short simulated‑annealing polishing phase:
   Randomly move a tiny amount of mass between two coordinates;
   accept the move if it improves the objective or with a temperature‑dependent
   probability.  The temperature decays slowly.
   This phase can escape tiny local traps left by the sub‑gradient steps.
5. Return the best sequence as a plain Python list.

All heavy work (convolutions) is performed with FFTs; the whole routine
stops before the supplied time budget (default 1000 s) and prints progress.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def convolve_fft(u: np.ndarray, v: np.ndarray) -> np.ndarray:
    """
    Linear convolution using FFT.  Returns the full convolution of length
    len(u)+len(v)-1.
    """
    size = 1 << int(np.ceil(np.log2(len(u) + len(v) - 1)))
    fu = np.fft.rfft(u, size)
    fv = np.fft.rfft(v, size)
    w = np.fft.irfft(fu * fv, size)[: len(u) + len(v) - 1]
    return w


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i >= 0, Σ x_i = 1}.
    """
    if v.ndim != 1:
        raise ValueError("Simplex projection expects a 1‑D array")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # find the rho where the condition holds
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform simplex point
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the evaluation
    function.  The routine respects ``budget_s`` seconds and returns the best
    sequence found as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.3          # safety margin

    best_val = float("inf")
    best_seq = None

    # ------------------------------------------------------------------
    # 0.  Use the globally supplied construction if it exists.
    init = globals().get("height_sequence_1")
    if init is not None:
        a0 = np.array(init, dtype=np.float64)
        a0 = np.maximum(a0, 0.0)
        if a0.sum() == 0:
            a0 = np.full_like(a0, 1.0 / a0.size)
        else:
            a0 = a0 / a0.sum()
        conv0 = convolve_fft(a0, a0)
        val0 = 2.0 * a0.size * conv0.max()
        best_val, best_seq = val0, a0.copy()
        print(f"[Init] using height_sequence_1 → value {val0:.6f}")

    # ------------------------------------------------------------------
    # 1.  Cheap parametric sweep over a few families and lengths
    lengths = [2000, 2500, 3000, 3500, 4000]
    # parameter grids
    p_vals = np.linspace(0.1, 3.0, 20)      # for power families
    r_vals = np.linspace(0.2, 0.9, 15)      # for exponential decay

    for n in lengths:
        if time.time() >= deadline:
            break
        # ----- sine^p --------------------------------------------------
        i = np.arange(1, n + 1, dtype=np.float64)
        base = np.sin(np.pi * i / (n + 1))
        for p in p_vals:
            a = base ** p
            a_sum = a.sum()
            if a_sum == 0:
                continue
            a = a / a_sum
            conv = convolve_fft(a, a)
            val = 2.0 * n * conv.max()
            if val < best_val:
                best_val, best_seq = val, a.copy()
                print(f"[Sweep] n={n:4d} sine^p={p:5.2f} → new best {val:.6f}")

        # ----- polynomial (i·(n+1‑i))^p ---------------------------------
        for p in p_vals:
            a = (i * (n + 1 - i)) ** p
            a_sum = a.sum()
            if a_sum == 0:
                continue
            a = a / a_sum
            conv = convolve_fft(a, a)
            val = 2.0 * n * conv.max()
            if val < best_val:
                best_val, best_seq = val, a.copy()
                print(f"[Sweep] n={n:4d} poly^p={p:5.2f} → new best {val:.6f}")

        # ----- exponential decay ----------------------------------------
        for r in r_vals:
            a = r ** np.arange(n, dtype=np.float64)
            a_sum = a.sum()
            if a_sum == 0:
                continue
            a = a / a_sum
            conv = convolve_fft(a, a)
            val = 2.0 * n * conv.max()
            if val < best_val:
                best_val, best_seq = val, a.copy()
                print(f"[Sweep] n={n:4d} exp r={r:5.3f} → new best {val:.6f}")

    # ------------------------------------------------------------------
    # 2.  Sub‑gradient descent from the best candidate found so far.
    if best_seq is not None:
        # we will try a few restarts with a tiny jitter
        for restart in range(3):
            if time.time() >= deadline:
                break
            a = best_seq.copy()
            a += np.random.randn(a.size) * 1e-4      # small random shake
            a = np.maximum(a, 0.0)
            a = project_simplex(a)

            n = a.size
            max_iter = 8000
            step = 0.5
            last_improve = time.time()

            for it in range(max_iter):
                if time.time() >= deadline:
                    break

                # self‑convolution and active mask
                conv = convolve_fft(a, a)
                maxc = conv.max()
                mask = (conv >= maxc - 1e-12).astype(np.float64)

                # sub‑gradient of the max‑convolution term
                grad = 2.0 * convolve_fft(mask, a[::-1])
                grad = grad[n - 1:n - 1 + n]            # slice back to length n

                # simple projected gradient step
                a -= step * grad
                a = project_simplex(a)

                # decay step size gently
                step *= 0.9995

                # periodic exact evaluation
                if it % 200 == 0:
                    val = 2.0 * n * maxc
                    if val < best_val - 1e-7:
                        best_val, best_seq = val, a.copy()
                        last_improve = time.time()
                        print(f"[SubG] restart {restart} iter {it:5d} → new best {val:.6f}")

                # if we stall for a few seconds, inject a small random shake
                if time.time() - last_improve > 5.0:
                    a += (np.random.rand(n) - 0.5) * 1e-3
                    a = project_simplex(a)
                    step = 0.5
                    last_improve = time.time()

    # ------------------------------------------------------------------
    # 3.  Simulated‑annealing polishing (tiny mass swaps)
    if best_seq is not None:
        a = best_seq.copy()
        n = a.size
        cur_val = 2.0 * n * convolve_fft(a, a).max()
        temp = 0.01
        temp_decay = 0.9999
        step_max = 0.02
        it = 0
        while time.time() < deadline and it < 20000:
            i, j = np.random.choice(n, 2, replace=False)
            delta = (np.random.rand() * 2.0 - 1.0) * step_max

            # keep non‑negativity
            delta = max(-a[i], min(delta, a[j]))

            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta

            conv_new = convolve_fft(a_new, a_new)
            new_val = 2.0 * n * conv_new.max()

            # Metropolis acceptance
            if new_val < cur_val or np.random.rand() < np.exp(-(new_val - cur_val) / temp):
                a, cur_val = a_new, new_val
                if new_val < best_val:
                    best_val, best_seq = new_val, a.copy()
                    print(f"[Anneal] iter {it:5d} → new best {new_val:.6f}")

            temp *= temp_decay
            it += 1
            if it % 5000 == 0:
                temp = max(temp, 1e-4)      # ensure temperature never vanishes

    # ------------------------------------------------------------------
    # 4.  Fallback (should never trigger)
    if best_seq is None:
        n = 2500
        best_seq = np.full(n, 1.0 / n)
        best_val = 2.0 * n * convolve_fft(best_seq, best_seq).max()

    print(f"Final best value = {best_val:.6f} (length {len(best_seq)})")
    # convert to a plain Python list as expected by the evaluator
    return best_seq.tolist()
```
