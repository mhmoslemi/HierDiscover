def solve_lp_max_sum(f, M):
    n = len(f)
    # Build A matrix
    A = np.zeros((2*n-1, n), dtype=float)
    for i in range(n):
        A[i:i+n, i] = f
    # b = M for each row
    b = np.full(2*n-1, M, dtype=float)
    c = -np.ones(n)  # minimize -sum(g) => maximize sum(g)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs', options={'presolve': True, 'time_limit': 20})
    if res.success:
        return res.x
    else:
        return None

"""
Propose a non‑negative step function minimizing 2·n·max(conv)/sum^2.
Algorithm:
  1. Random‑restart Adam optimisation on the soft‑max parametrisation
     (fast, finds good baselines).
  2. If time permits, apply a linear‑programming refinement inspired
     by the iterative “f –> g” scheme: maximise sum(g) under the
     current max‑convolution bound, scale to the required sum and mix
     with the current sequence using the largest step size that does
     not increase the bound.  This usually decreases the peak of the
     autocorrelation further.
Both stages respect the 1000 s budget and keep the best sequence seen.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.linalg import toeplitz

def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using FFT (real)."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    # small negative values can appear due to floating point errors.
    np.maximum(conv, 0.0, out=conv)
    return conv

def _optimise_adam_softmax(
    n: int,
    x0: np.ndarray,
    target_sum: float,
    run_deadline: float,
) -> tuple[np.ndarray, float]:
    """Adam optimiser on the log‑it (soft‑max) representation.

    Returns the best sequence encountered (as a NumPy array) and its
    max‑convolution value (the objective, lower is better).
    """
    # Adam hyper‑parameters – choices that work well for many lengths.
    alpha = 0.06           # base learning rate
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # smoothing temperature schedule for the max‑surrogate
    init_tau_factor = 0.15
    min_tau_factor = 0.001

    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    best_seq = None
    best_val = float("inf")
    iteration = 0

    while time.time() < run_deadline:
        # 1. map logits to the simplex (a = target_sum·softmax(x))
        max_x = np.max(x)
        exp_shift = np.exp(x - max_x)          # shift for stability
        p = exp_shift / np.sum(exp_shift)      # probability vector, sum=1
        a = target_sum * p                     # current candidate

        # 2. compute convolution and current max
        conv = _full_convolution_fft(a, a)
        cur_max = conv.max()
        if cur_max < best_val:
            best_val = cur_max
            best_seq = a.copy()
            # we keep a compact progress line – prints are cheap
            print(f"[{time.time():.2f}] n={n}  adam new local best = {cur_max:.6f}")

        # 3. smooth‑max weights (soft‑max over the convolution values)
        tau = max(init_tau_factor * cur_max * (0.95 ** iteration),
                  min_tau_factor * cur_max)
        scaled = (conv - cur_max) / tau
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # 4. gradient w.r.t. a :  grad_a = 2·(a ⊛ w_rev)
        w_rev = w[::-1]
        conv2 = _full_convolution_fft(a, w_rev)        # length = 3n‑2
        # we need exactly n terms: they correspond to positions
        # (2n-2)‑i for i=0…n‑1
        idx = (2 * n - 2) - np.arange(n)
        grad_a = 2.0 * conv2[idx]

        # 5. chain rule to obtain gradient w.r.t. logits x
        # d a_i / d x_j = target_sum·p_i·(δ_{ij} – p_j)
        dotp = np.dot(p, grad_a)
        grad_x = target_sum * p * (grad_a - dotp)

        # 6. Adam update
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # occasional tiny random jitter to help escape plateaus
        if iteration % 250 == 0 and np.random.rand() < 0.2:
            x += np.random.randn(n) * 0.001

        # slowly decay the learning rate (optional)
        alpha *= 0.9995

    # final evaluation using the supplied evaluate_sequence to be safe
    final_score = evaluate_sequence(best_seq.tolist())
    return best_seq, final_score

def _solve_lp_max_sum(f: np.ndarray, M: float) -> np.ndarray | None:
    """Solve   max 1ᵀ g   subject to   f ∗ g ≤ M (componentwise), g ≥ 0.

    Returns the optimal g as a dense NumPy array (or None on failure).
    """
    n = f.shape[0]
    # Build convolution matrix A of shape (2n‑1, n)
    # A[i, j] = f[i‑j] if 0 ≤ i‑j < n else 0
    # Using toeplitz is the simplest way.
    col = np.concatenate([f, np.zeros(n - 1, dtype=f.dtype)])
    row = np.zeros(n, dtype=f.dtype)
    row[0] = f[0]
    A = toeplitz(col, row)          # dense (2n‑1) × n

    # Upper bound for each row is M.
    b = np.full(2 * n - 1, M, dtype=float)

    # Linear programme: minimise -1ᵀ g   (equivalently maximise 1ᵀ g)
    c = -np.ones(n, dtype=float)

    bounds = [(0.0, None)] * n

    # Use the 'highs' interior‑point method; it is fast for moderate sizes.
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if res.success:
        return res.x
    # If the problem is infeasible we return None – the caller will stop.
    return None

def _lp_refine_step(f: np.ndarray) -> tuple[np.ndarray, float] | None:
    """One iteration of the LP‑based refinement.

    Returns (new_f, new_max) if a strict improvement is obtained,
    otherwise None.
    """
    n = f.shape[0]
    target_sum = np.sqrt(2.0 * n)

    # current bound
    conv_ff = _full_convolution_fft(f, f)
    M = conv_ff.max()

    # Solve LP for g
    g0 = _solve_lp_max_sum(f, M)
    if g0 is None:
        return None

    sum_g = g0.sum()
    if sum_g <= 1e-12:
        return None

    # Scale g to satisfy the sum constraint
    g = g0 * (target_sum / sum_g)

    # Pre‑compute needed convolutions
    conv_fg = _full_convolution_fft(f, g)
    conv_gg = _full_convolution_fft(g, g)

    # Binary search the largest step size t ∈ (0,1] such that
    #   max( (1‑t)²·conv_ff + 2t(1‑t)·conv_fg + t²·conv_gg ) ≤ M
    lo, hi = 0.0, 1.0
    for _ in range(30):
        t = (lo + hi) * 0.5
        # quadratic combination
        combo = (
            (1.0 - t) ** 2 * conv_ff
            + 2.0 * t * (1.0 - t) * conv_fg
            + t ** 2 * conv_gg
        )
        if combo.max() <= M + 1e-12:
            lo = t
        else:
            hi = t
    t_opt = lo

    if t_opt < 1e-6:
        # No meaningful step can be taken.
        return None

    # Form the new sequence
    new_f = (1.0 - t_opt) * f + t_opt * g
    # Renormalise to eliminate tiny drift
    new_f = new_f * (target_sum / new_f.sum())
    new_max = np.max(_full_convolution_fft(new_f, new_f))
    # accept only if we actually improved the bound
    if new_max < M - 1e-9:
        return new_f, new_max
    return None

def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the given evaluation
    function.  The routine uses a two‑phase approach: many short Adam runs for
    exploration, followed by a few LP‑based refinement steps that exploit the
    known optimality condition of the original alternating algorithm.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.2   # keep a tiny safety margin

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # Warm‑start from a previously stored construction if available.
    # --------------------------------------------------------------
    try:
        hs = globals()["height_sequence_1"]
        hs_arr = np.array(hs, dtype=float)
        hs_arr = np.clip(hs_arr, 0.0, 1000.0)
        score = evaluate_sequence(hs_arr.tolist())
        if score < best_score:
            best_score = score
            best_seq = hs_arr.copy()
            print(f"[{time.time():.2f}] warm‑start global best = {best_score:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # Phase 1 – many random restarts with Adam.
    # --------------------------------------------------------------
    # We protect a final window (≈ 200 s) for the LP refinement.
    while time.time() < deadline - 200.0:
        remaining = deadline - time.time()
        # Pick a problem size. Larger n gives more expressive power but
        # also slower iterations. We let the size shrink as time runs out.
        if remaining > 600:
            n = np.random.randint(2000, 3500 + 1)
        elif remaining > 300:
            n = np.random.randint(1500, 3000 + 1)
        else:
            n = np.random.randint(800, 2500 + 1)

        # ---- initialise a plausible shape ----
        # Power‑law base
        p_exp = np.random.uniform(0.25, 0.75)
        idx = np.arange(1, n + 1, dtype=float)
        init = idx ** (-p_exp)

        # Occasionally taper with an exponential factor
        if np.random.rand() < 0.35:
            tau_exp = np.random.uniform(0.5, 2.0) * n
            init *= np.exp(-idx / tau_exp)

        # Occasionally smooth a little
        if np.random.rand() < 0.20:
            klen = np.random.randint(3, 9)
            kernel = np.ones(klen) / klen
            init = np.convolve(init, kernel, mode='same')

        # Random jitter – breaks exact symmetries
        init += np.random.rand(n) * 1e-5

        # Occasionally borrow part of the current global best
        if best_seq is not None and np.random.rand() < 0.45:
            if best_seq.size >= n:
                init = best_seq[:n].copy()
            else:
                pad = np.random.rand(n - best_seq.size) * 1e-5
                init = np.concatenate([best_seq, pad])

        # Ensure positivity before log‑softmax
        init = np.maximum(init, 1e-12)

        # Convert to logits for the Adam optimiser
        target_sum = np.sqrt(2.0 * n)
        init = init / init.sum() * target_sum
        x0 = np.log(init + 1e-12)

        # Allocate a time slice for this run – about 8‑12 s, but never
        # exceed the remaining budget.
        slice_time = min(12.0, remaining * 0.15)
        run_deadline = min(deadline, time.time() + slice_time)

        seq_local, score_local = _optimise_adam_softmax(
            n, x0, target_sum, run_deadline
        )
        if score_local < best_score:
            best_score = score_local
            best_seq = seq_local.copy()
            print(f"[{time.time():.2f}] GLOBAL new best (Adam) = {best_score:.6f} (n={n})")
            if best_score < 1.503:
                break

    # --------------------------------------------------------------
    # Phase 2 – LP‑based refinement of the best sequence found so far.
    # --------------------------------------------------------------
    while time.time() < deadline - 0.5:
        if best_seq is None:
            break
        # Run a limited number of refinement steps; we stop as soon as
        # no improvement is observed.
        result = _lp_refine_step(best_seq)
        if result is None:
            # No further improvement possible.
            break
        new_seq, new_val = result
        if new_val < best_score - 1e-9:
            best_score = new_val
            best_seq = new_seq.copy()
            print(f"[{time.time():.2f}] LP refinement improved bound to {best_score:.6f}")
            if best_score < 1.503:
                break
        else:
            # Should not happen because the step only returns improvement,
            # but we keep the guard.
            break

    # --------------------------------------------------------------
    # Fallback – in the unlikely event that no candidate was produced.
    # --------------------------------------------------------------
    if best_seq is None:
        n = 1000
        target_sum = np.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.2f}] fallback uniform sequence score {best_score:.6f}")

    # Ensure final output respects the evaluator’s expectations.
    return best_seq.tolist()

"""
Propose a non‑negative step function that minimises

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The evaluation simplifies (when Σa = √(2 n)) to `max(conv(a,a))`.
The search proceeds in two stages:

1️⃣  Many short Adam runs on a soft‑max parametrisation.  This explores a
    wide range of lengths and initial shapes (power‑law, exponential
    taper, slight smoothing).  The best sequence found is kept globally.

2️⃣  If time permits, a few iterations of the LP‑based refinement described
    in the problem statement are applied to the current best sequence.
    For a given sequence `f` we solve

        maximise   Σ g_j
        subject to  f ∗ g ≤ M  (componentwise) ,   g ≥ 0

    where `M = max(f∗f)`.  The optimal `g` is scaled to the required sum
    √(2 n) and mixed with `f` using the largest step size that does not
    increase the bound.  Repeating this step monotonically reduces the
    peak of the self‑convolution.

Both phases respect the 1000 s budget; the algorithm prints progress
messages and finally returns the best sequence as a plain Python list.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.linalg import toeplitz


# ----------------------------------------------------------------------
# Helper – full linear convolution using real FFT (fast for any length)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of real vectors x and y via FFT."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    np.maximum(conv, 0.0, out=conv)          # eliminate tiny negatives
    return conv


# ----------------------------------------------------------------------
# Adam optimiser on the soft‑max parametrisation
# ----------------------------------------------------------------------
def _optimise_adam_softmax(
    n: int,
    x0: np.ndarray,
    target_sum: float,
    run_deadline: float,
) -> tuple[np.ndarray, float]:
    """Adam optimisation on the log‑it (soft‑max) representation.

    Returns the best sequence encountered (as a NumPy array) and its
    max‑convolution value as reported by ``evaluate_sequence``.
    """
    # Adam hyper‑parameters (tuned for this problem)
    alpha = 0.06
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # Smoothing‑max schedule
    init_tau_factor = 0.15
    min_tau_factor = 0.001

    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    best_seq = None
    best_val = float("inf")
    iteration = 0

    while time.time() < run_deadline:
        # ----- 1. map logits → simplex (a = target_sum·softmax(x)) -----
        max_x = np.max(x)
        exp_shift = np.exp(x - max_x)          # improve numerical stability
        p = exp_shift / np.sum(exp_shift)      # probabilities, sum = 1
        a = target_sum * p                     # candidate sequence

        # ----- 2. evaluate current max‑convolution -----
        conv = _full_convolution_fft(a, a)
        cur_max = conv.max()
        if cur_max < best_val:
            best_val = cur_max
            best_seq = a.copy()
            print(f"[{time.time():.2f}] n={n}  adam new local best = {cur_max:.6f}")

        # ----- 3. smooth‑max weights (soft‑max over conv values) -----
        tau = max(init_tau_factor * cur_max * (0.95 ** iteration),
                  min_tau_factor * cur_max)
        scaled = (conv - cur_max) / tau
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- 4. gradient w.r.t. a  (grad_a = 2·(a ⊛ w_rev)) -----
        w_rev = w[::-1]
        conv2 = _full_convolution_fft(a, w_rev)      # length = 3n‑2
        idx = (2 * n - 2) - np.arange(n)              # mapping to gradient entries
        grad_a = 2.0 * conv2[idx]

        # ----- 5. gradient w.r.t. logits x (chain rule) -----
        dotp = np.dot(p, grad_a)
        grad_x = target_sum * p * (grad_a - dotp)

        # ----- 6. Adam update -----
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # occasional jitter to escape plateaus
        if iteration % 250 == 0 and np.random.rand() < 0.2:
            x += np.random.randn(n) * 0.001

        # decay the base learning‑rate slightly
        alpha *= 0.9995

    final_score = evaluate_sequence(best_seq.tolist())
    return best_seq, final_score


# ----------------------------------------------------------------------
# Linear‑programming sub‑routine used by the refinement step
# ----------------------------------------------------------------------
def _solve_lp_max_sum(f: np.ndarray, M: float) -> np.ndarray | None:
    """Solve   max 1ᵀ g   s.t.   f ∗ g ≤ M (componentwise), g ≥ 0.

    Returns the optimal g (dense NumPy array) or None on failure.
    """
    n = f.shape[0]

    # Construct the convolution matrix A (size (2n‑1) × n) via Toeplitz.
    # Row i, column j contains f[i‑j] if 0 ≤ i‑j < n, else 0.
    col = np.concatenate([f, np.zeros(n - 1, dtype=f.dtype)])
    row = np.zeros(n, dtype=f.dtype)
    row[0] = f[0]
    A = toeplitz(col, row)                     # dense matrix

    b = np.full(2 * n - 1, M, dtype=float)    # upper bound for each convolution entry
    c = -np.ones(n, dtype=float)              # minimise –sum(g)

    bounds = [(0.0, None)] * n

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


# ----------------------------------------------------------------------
# One LP‑based refinement iteration
# ----------------------------------------------------------------------
def _lp_refine_step(f: np.ndarray) -> tuple[np.ndarray, float] | None:
    """Perform a single LP refinement step.

    Returns (new_f, new_max) if an improvement is obtained, else None.
    """
    n = f.shape[0]
    target_sum = np.sqrt(2.0 * n)

    # Current bound
    conv_ff = _full_convolution_fft(f, f)
    M = conv_ff.max()

    # Solve LP for the auxiliary sequence g
    g0 = _solve_lp_max_sum(f, M)
    if g0 is None:
        return None

    sum_g = g0.sum()
    if sum_g <= 1e-12:
        return None

    # Scale g to meet the sum constraint Σg = √(2 n)
    g = g0 * (target_sum / sum_g)

    # Pre‑compute convolutions needed for the quadratic combination
    conv_fg = _full_convolution_fft(f, g)
    conv_gg = _full_convolution_fft(g, g)

    # Binary search the largest step size t ∈ (0,1] that keeps the bound ≤ M
    lo, hi = 0.0, 1.0
    for _ in range(30):
        t = (lo + hi) * 0.5
        combo = (
            (1.0 - t) ** 2 * conv_ff
            + 2.0 * t * (1.0 - t) * conv_fg
            + t ** 2 * conv_gg
        )
        if combo.max() <= M + 1e-12:
            lo = t
        else:
            hi = t
    t_opt = lo

    if t_opt < 1e-6:      # no meaningful move
        return None

    new_f = (1.0 - t_opt) * f + t_opt * g
    # Renormalise (tiny drift may appear)
    new_f = new_f * (target_sum / new_f.sum())
    new_max = np.max(_full_convolution_fft(new_f, new_f))

    if new_max < M - 1e-9:   # accept only a strict improvement
        return new_f, new_max
    return None

    # --------------------------------------------------------------
    if best_seq is not None:
        n = best_seq.shape[0]
        target_sum = np.sqrt(2.0 * n)
        # Convert the current sequence back to log‑it variables.
        p = best_seq / target_sum
        p = np.clip(p, 1e-12, None)
        x0 = np.log(p + 1e-12)

        remaining = deadline - time.time()
        if remaining > 5.0:
            slice_time = min(8.0, remaining * 0.5)
            run_deadline = time.time() + slice_time
            seq_local, score_local = _optimise_adam_softmax(
                n, x0, target_sum, run_deadline
            )
            if score_local < best_score:
                best_score = score_local
                best_seq = seq_local.copy()
                print(f"[{time.time():.2f}] final Adam polish improved to {best_score:.6f}")

    # --------------------------------------------------------------
    # Fallback – if nothing succeeded, return a uniform sequence.
    # --------------------------------------------------------------
    if best_seq is None:
        n = 1000
        target_sum = np.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.2f}] fallback uniform sequence score {best_score:.6f}")

    # Return as plain Python list (what the evaluator expects)
    return best_seq.tolist()
