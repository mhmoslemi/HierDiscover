"""
Hybrid random‑restart optimisation for the step‑function bound.

Algorithm
---------
1. **Random restarts** – choose a length `n` (500‑5000) and initialise a
   non‑negative vector using a power‑law (exponent in a wide range) with
   optional exponential taper and jitter.  The vector is normalised so
   that its ℓ₁‑norm equals √(2 n).

2. **Soft‑max Adam** – the variables are parametrised as a soft‑max
   (`a = S·softmax(x)`).  The objective `max(conv(a,a))` is smoothed
   with a temperature `τ` (log‑sum‑exp surrogate).  Adam updates are
   applied to `x`.  The temperature decays geometrically and a
   near‑hard‑max is used once it becomes very small.

3. **Local hill‑climbing** – after each Adam run the best sequence is
   refined by transferring a small amount of mass between two random
   positions (keeping the total sum fixed).  The convolution max is
   recomputed via FFT; moves that improve the objective are kept.
   This simple stochastic search often pushes the bound a little further.

4. **Best‑so‑far warm‑starts** – the global best sequence is occasionally
   injected as a warm‑start for new runs, and the final refinement runs
   on it until the time budget expires.

The routine respects a 1000 s budget, prints progress, and finally
returns the best sequence as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper – full linear convolution using real FFT (fast for any length)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of real vectors x and y via FFT."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    # eliminate tiny negative values caused by numerical noise
    np.maximum(conv, 0.0, out=conv)
    return conv

    *,
    init_tau_factor: float = 0.15,
    min_tau_factor: float = 1e-4,
    alpha: float = 0.04,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Perform Adam optimisation on a soft‑max parametrisation of a length‑n
    sequence.  Returns the best sequence (as a NumPy array) encountered
    and its evaluation (max(conv(a,a))) computed with the provided
    `evaluate_sequence` function.
    """
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    best_seq = None
    best_score = float("inf")
    iteration = 0

    while time.time() < run_deadline:
        # --------------------------------------------------------------
        # 1) map logits to the simplex (a = S·softmax(x))
        # --------------------------------------------------------------
        max_x = np.max(x)
        exp_shift = np.exp(x - max_x)               # stability
        p = exp_shift / np.sum(exp_shift)           # probabilities
        a = target_sum * p

        # --------------------------------------------------------------
        # 2) convolution and current (smooth) max
        # --------------------------------------------------------------
        conv = _full_convolution_fft(a, a)
        max_val = conv.max()
        if max_val < best_score:
            best_score = max_val
            best_seq = a.copy()
            print(f"[{time.time():.2f}] n={n} new local best = {max_val:.6f}")

        # --------------------------------------------------------------
        # 3) smooth‑max weights
        # --------------------------------------------------------------
        tau = max(init_tau_factor * max_val * (0.95 ** iteration),
                  min_tau_factor * max_val)
        # when tau becomes extremely tiny we fall back to a hard‑max mask
        if tau <= 1e-12 * max_val:
            w = np.zeros_like(conv)
            w[conv == max_val] = 1.0
            w /= w.sum()
        else:
            scaled = (conv - max_val) / tau
            w = np.exp(scaled)
            w_sum = w.sum()
            if w_sum == 0.0:
                w.fill(1.0 / w.size)
            else:
                w /= w_sum

        # --------------------------------------------------------------
        # 4) gradient w.r.t. a
        # --------------------------------------------------------------
        w_rev = w[::-1]
        conv2 = _full_convolution_fft(a, w_rev)          # length = 3n‑2
        idx = (2 * n - 2) - np.arange(n)                  # slice giving Σ_j a_j w_{i+j}
        grad_a = 2.0 * conv2[idx]

        # --------------------------------------------------------------
        # 5) gradient w.r.t. logits x (chain rule)
        # --------------------------------------------------------------
        dotp = np.dot(p, grad_a)
        grad_x = target_sum * p * (grad_a - dotp)

        # --------------------------------------------------------------
        # 6) Adam step
        # --------------------------------------------------------------
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # occasional tiny random perturbation to escape plateaux
        if iteration % 250 == 0 and np.random.rand() < 0.2:
            x += np.random.randn(n) * 0.001

        # very slowly decay the base learning rate
        alpha *= 0.9995

    # final objective (the same as max(conv) because we keep the sum fixed)
    final_score = max(_full_convolution_fft(best_seq, best_seq))
    return best_seq, final_score


# ----------------------------------------------------------------------
# Simple stochastic hill‑climbing refinement (mass transfer)
# ----------------------------------------------------------------------
def _hill_climb(seq: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Starting from `seq` (already normalised to the required sum),
    repeatedly move a small amount of mass from a random index i to a
    random index j.  Accept moves that reduce max(conv(seq,seq)).  Stop
    when the deadline is reached.
    """
    n = seq.shape[0]
    current = seq.copy()
    best = current.copy()
    # objective = max(conv)
    best_score = _full_convolution_fft(best, best).max()

    while time.time() < deadline:
        i, j = np.random.randint(0, n, size=2)
        if i == j or current[i] <= 0.0:
            continue
        # transfer a small fraction of the mass at i to j
        frac = np.random.uniform(0.001, 0.01)
        delta = current[i] * frac
        new = current.copy()
        new[i] -= delta
        new[j] += delta

        # compute objective cheaply
        score = _full_convolution_fft(new, new).max()
        if score < best_score:
            best = new
            best_score = score
            current = new
            print(f"[{time.time():.2f}] hill‑climb new best = {best_score:.6f}")
        else:
            # optionally keep the move with a tiny probability (not needed)
            pass

    return best, best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    2 * n * max(conv(a,a)) / (sum(a)²).

    The search combines random restarts, Adam optimisation on a smooth
    max surrogate and a final stochastic hill‑climbing phase.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin
    deadline = start_time + budget_s - 0.15

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # Warm‑start from a previously known construction, if present.
    # --------------------------------------------------------------
    try:
        hs = globals()["height_sequence_1"]
        hs_arr = np.array(hs, dtype=float)
        hs_arr = np.clip(hs_arr, 0.0, 1000.0)
        # ensure the sum constraint is satisfied
        n_hs = hs_arr.shape[0]
        target_sum_hs = np.sqrt(2.0 * n_hs)
        hs_arr = hs_arr / hs_arr.sum() * target_sum_hs
        score = _full_convolution_fft(hs_arr, hs_arr).max()
        best_seq = hs_arr
        best_score = score
        print(f"[{time.time():.2f}] Warm‑start from global seed = {score:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # Main restart loop – each restart receives a time slice.
    # --------------------------------------------------------------
    while time.time() < deadline:
        remaining = deadline - time.time()
        if remaining < 5.0:          # not enough time for another full pass
            break

        # ----------- pick a problem size --------------------------------
        if remaining > 300:
            n = np.random.randint(2000, 5000 + 1)
        elif remaining > 120:
            n = np.random.randint(1500, 3500 + 1)
        else:
            n = np.random.randint(1000, 2500 + 1)

        # ----------- initialise a candidate shape -----------------------
        p_exp = np.random.uniform(0.3, 2.0)                # power‑law exponent
        idx = np.arange(1, n + 1, dtype=float)
        init = idx ** (-p_exp)

        # optional exponential taper
        if np.random.rand() < 0.35:
            tau_exp = np.random.uniform(0.5, 2.5) * n
            init *= np.exp(-idx / tau_exp)

        # random reversal to explore asymmetric shapes
        if np.random.rand() < 0.25:
            init = init[::-1]

        # tiny random jitter for symmetry breaking
        init += np.random.rand(n) * 1e-5

        # --------------------------------------------------------------
        # Occasionally inject the current global best as a warm‑start.
        # --------------------------------------------------------------
        if best_seq is not None and np.random.rand() < 0.45:
            if best_seq.size >= n:
                init = best_seq[:n].copy()
            else:
                pad = np.random.rand(n - best_seq.size) * 1e-5
                init = np.concatenate([best_seq, pad])

        # --------------------------------------------------------------
        # Normalise to the required ℓ₁‑norm ( √(2 n) )
        # --------------------------------------------------------------
        target_sum = np.sqrt(2.0 * n)
        init = init / init.sum() * target_sum
        init = np.maximum(init, 1e-12)          # avoid exact zeros for log‑it

        # logits for soft‑max parametrisation
        x0 = np.log(init + 1e-12)

        # --------------------------------------------------------------
        # Allocate a time slice for this Adam run.
        # --------------------------------------------------------------
        # early runs get shorter slices, later runs get longer ones.
        if remaining > 300:
            slice_time = 10.0
        elif remaining > 120:
            slice_time = 18.0
        else:
            slice_time = min(30.0, remaining * 0.5)
        run_deadline = min(deadline, time.time() + slice_time)

        # --------------------------------------------------------------
        # Run the Adam optimiser.
        # --------------------------------------------------------------
        seq_local, score_local = _optimise_adam_softmax(
            n,
            x0,
            target_sum,
            run_deadline,
            init_tau_factor=0.15,
            min_tau_factor=1e-4,
            alpha=0.04,
            beta1=0.9,
            beta2=0.999,
        )

        # --------------------------------------------------------------
        # Quick hill‑climb refinement on the locally‑optimised sequence.
        # --------------------------------------------------------------
        refine_deadline = min(deadline, time.time() + 5.0)
        seq_refined, score_refined = _hill_climb(seq_local, refine_deadline)

        # --------------------------------------------------------------
        # Update the global best if we improved.
        # --------------------------------------------------------------
        if score_refined < best_score:
            best_score = score_refined
            best_seq = seq_refined.copy()
            print(f"[{time.time():.2f}] GLOBAL new best = {best_score:.6f} (n={n})")
            if best_score < 1.503:          # early exit if target beaten
                break

    # ------------------------------------------------------------------
    # Final hill‑climbing phase on the overall best sequence (if any).
    # ------------------------------------------------------------------
    if best_seq is not None:
        final_deadline = deadline
        best_seq, best_score = _hill_climb(best_seq, final_deadline)
        print(f"[{time.time():.2f}] FINAL best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback in the unlikely event that no candidate was generated.
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1500
        target_sum = np.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)

    # Return a plain Python list as required by the external evaluator.
    return best_seq.tolist()
