"""
Propose a non‑negative step‑function (vector) that minimises
``evaluate_sequence``.

Algorithm
---------
1. Work with a probability vector `p` (Σp = 1).  The exact objective
   equals `2*n*max(p∗p)`, where `∗` is the linear convolution.
2. Optimise a smooth surrogate of the maximum:
        L_τ(p) = τ·log Σ exp((p∗p)_k / τ)
   The gradient of `L_τ` w.r.t. the log‑probability parameters
   `y_i = log(p_i)` can be written in closed form using FFT based
   convolutions.
3. A continuation scheme gradually reduces the temperature `τ`.
   At each temperature an L‑BFGS‑B optimisation (SciPy) is run with a
   fairly large budget of function evaluations.
4. The whole routine is repeated from many random starts (Dirichlet
   draws, uniform, and a symmetric perturbation of the previous best
   solution).  The best sequence found so far is kept.
5. Several deterministic window shapes (Hann, Hamming, Blackman) are
   also evaluated – they are cheap to generate and sometimes beat the
   stochastic runs.
6. The search runs until the time budget (≈1000 s) is exhausted and
   finally returns a list of floats whose L1‑norm is
   `sqrt(2·n)`.  The returned vector is guaranteed to satisfy the
   evaluator’s non‑negativity and size checks.
"""

import time
import math
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# optional SciPy for L‑BFGS‑B
try:
    from scipy.optimize import minimize
    _SCIPY = True
except Exception:   # pragma: no cover
    _SCIPY = False


def _next_pow2(x: int) -> int:
    """Next power of two >= x."""
    return 1 << (x - 1).bit_length()


def _softmax_weights(conv: np.ndarray, tau: float):
    """Stable soft‑max of convolution values.
    Returns (weights, smooth max value)."""
    if tau <= 0.0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w, conv.max()
    cmax = conv.max()
    ex = np.exp((conv - cmax) / tau)
    s = ex.sum()
    w = ex / s
    L = cmax + tau * math.log(s)
    return w, L


def _full_conv(p: np.ndarray, Nfft: int) -> np.ndarray:
    """Linear convolution p∗p (real FFT, length 2·len(p)-1)."""
    A = np.fft.rfft(p, n=Nfft)
    conv = np.fft.irfft(A * A, n=Nfft)[: 2 * p.shape[0] - 1]
    return conv


def _weighted_rev(a: np.ndarray, w: np.ndarray, Nfft: int) -> np.ndarray:
    """
    Compute rev_i = Σ_k w_k a_{k-i}
    (convolution of a with the reversed weight vector).
    """
    A = np.fft.rfft(a, n=Nfft)
    W_rev = np.fft.rfft(w[::-1], n=Nfft)
    rev_full = np.fft.irfft(A * W_rev, n=Nfft)
    rev_full = rev_full[: len(w) + a.shape[0] - 1]
    # pick the appropriate slice: index shift = len(a)-1
    start = len(a) - 1
    rev = rev_full[start:start + len(a)]
    return rev


def _obj_and_grad(y: np.ndarray, tau: float, Nfft: int):
    """
    Smooth‑max objective L_τ and its gradient w.r.t. the logits y.
    """
    # p = softmax(y)
    y_max = np.max(y)
    exp_y = np.exp(y - y_max)        # stabilised
    p = exp_y / exp_y.sum()

    # convolution p∗p
    conv = _full_conv(p, Nfft)

    # smooth max
    w, L = _softmax_weights(conv, tau)

    # gradient w.r.t. p : 2 * rev
    rev = _weighted_rev(p, w, Nfft)
    grad_p = 2.0 * rev

    # chain rule through softmax
    dot = np.dot(p, grad_p)
    grad_y = p * (grad_p - dot)
    return L, grad_y


def _run_l_bfgs(y0: np.ndarray,
                init_tau: float,
                target_sum: float,
                Nfft: int,
                deadline: float,
                best_seq: np.ndarray,
                best_score: float) -> (np.ndarray, float):
    """
    Run the temperature‑continuation + L‑BFGS optimizer starting from `y0`.
    Returns the best sequence (as a float array of length n) and the
    best achieved objective value.
    """
    y = y0.copy()
    tau = max(init_tau, 1e-12)
    tau_decay = 0.78              # slower cooling than the previous baseline
    min_tau = 1e-13

    n = y.shape[0]

    while time.time() < deadline and tau > min_tau:
        # ----- L‑BFGS on the smooth objective -------------------------
        if not _SCIPY:
            # fall back to a very small (ineffective) step if SciPy missing
            break

        def fun(y_vec):
            L, g = _obj_and_grad(y_vec, tau, Nfft)
            return L, g

        # give the inner optimiser a fairly generous budget
        opts = {"maxfun": 2500, "maxiter": 2500, "ftol": 1e-13, "gtol": 1e-13}
        res = minimize(fun, y, method="L-BFGS-B", jac=True, options=opts)

        y = res.x

        # ----- evaluate exact objective (no smooth approximation) -------
        y_max = np.max(y)
        exp_y = np.exp(y - y_max)
        p = exp_y / exp_y.sum()
        conv = _full_conv(p, Nfft)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv   # this is exactly evaluate_sequence's value

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = (p * target_sum).copy()
            print(f"[improve] τ={tau:.2e}  score={best_score:.6f}")

        # ----- temperature schedule ------------------------------------
        tau *= tau_decay

        # occasional random jitter to shake out of plateaux
        if np.random.rand() < 0.2:
            y += np.random.randn(n) * 0.03

        if time.time() >= deadline:
            break

    return best_seq, best_score


def _window_candidate(name: str, n: int) -> np.ndarray:
    """Generate a deterministic window (Hann, Hamming, Blackman)."""
    t = np.arange(n, dtype=float) / (n - 1)
    if name == "hann":
        w = 0.5 - 0.5 * np.cos(2 * np.pi * t)
    elif name == "hamming":
        w = 0.54 - 0.46 * np.cos(2 * np.pi * t)
    elif name == "blackman":
        w = (0.42 - 0.5 * np.cos(2 * np.pi * t) + 0.08 * np.cos(4 * np.pi * t))
    else:
        raise ValueError("unknown window")
    return w


def propose_candidate(seed: int = 42, budget_s: float = 1000) -> List[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` within the allocated time budget.

    Returns a plain Python list containing the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the surrounding framework
    deadline = start_time + budget_s - 5.0

    # --------------------------------------------------------------
    # Choose a problem size.
    # Larger n gives a better asymptotic lower bound (≈1) at the cost
    # of a little extra compute.  5000 is a comfortable compromise.
    n = 5000
    target_sum = math.sqrt(2.0 * n)           # the required L1‑norm

    # ---- baseline: uniform distribution --------------------------------
    p_uniform = np.full(n, 1.0 / n, dtype=float)
    y_uniform = np.log(p_uniform + 1e-12)

    # initialise best record (use uniform as a safe start)
    best_seq = (p_uniform * target_sum).copy()
    best_score = float("inf")
    # a quick exact evaluation protects against pathological numerics
    try:
        best_score = evaluate_sequence(best_seq.tolist())
    except Exception:   # pragma: no cover
        # fallback using our own cheap formula
        conv = _full_conv(p_uniform, _next_pow2(2 * n - 1))
        best_score = 2.0 * n * conv.max()
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # --------------------------------------------------------------
    # Prepare FFT size (power of two) – used by all convolutions.
    Nfft = _next_pow2(2 * n - 1)

    # ----- deterministic window candidates --------------------------------
    for wname in ("hann", "hamming", "blackman"):
        w = _window_candidate(wname, n)
        p = w / w.sum()
        seq = (p * target_sum).copy()
        sc = evaluate_sequence(seq.tolist())
        if sc < best_score:
            best_score = sc
            best_seq = seq
            print(f"[window] {wname} score={sc:.6f}")

    # ----- generate several stochastic starts ------------------------------
    starts = []

    # 1) uniform (already present as y_uniform)
    starts.append(y_uniform)

    # 2) a single random Dirichlet draw
    rand_exp = np.random.exponential(scale=1.0, size=n)
    starts.append(np.log(rand_exp + 1e-12))

    # 3) another Dirichlet with a different scale (more concentrated)
    rand_exp2 = np.random.exponential(scale=0.2, size=n)
    starts.append(np.log(rand_exp2 + 1e-12))

    # 4) a small perturbation of the uniform start
    starts.append(y_uniform + np.random.randn(n) * 0.05)

    # 5) a symmetric start – enforce p_i = p_{n-1-i}
    half = (n + 1) // 2
    rand_half = np.random.exponential(scale=1.0, size=half)
    full = np.concatenate([rand_half, rand_half[-2::-1]]) if n % 2 == 0 else np.concatenate([rand_half, rand_half[-2::-1]])
    starts.append(np.log(full + 1e-12))

    # ----- main optimisation loop -----------------------------------------
    for idx, y0 in enumerate(starts):
        if time.time() >= deadline:
            break
        print(f"[run] start {idx + 1}/{len(starts)}")

        # initialise temperature from the current smooth max
        p_start = np.exp(y0 - np.max(y0))
        p_start = p_start / p_start.sum()
        conv0 = _full_conv(p_start, Nfft)
        init_tau = max(conv0.max() * 2.0, 1e-12)

        best_seq, best_score = _run_l_bfgs(
            y0=y0,
            init_tau=init_tau,
            target_sum=target_sum,
            Nfft=Nfft,
            deadline=deadline,
            best_seq=best_seq,
            best_score=best_score,
        )

    # ----------------------------------------------------------------------
    # Final sanity check – one last exact evaluation of the best vector
    final_score = evaluate_sequence(best_seq.tolist())
    if final_score < best_score:
        best_score = final_score
        best_seq = best_seq.copy()
    print(f"[done] best score = {best_score:.6f}, length = {best_seq.shape[0]}")

    # --------------------------------------------------------------
    # Return a plain Python list as required by the outer harness.
    return best_seq.tolist()
