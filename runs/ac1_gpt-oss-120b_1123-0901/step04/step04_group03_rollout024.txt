"""
Propose a step‑function (non‑negative heights) that minimises
    F(a) = 2·n·max(a∗a) / (sum(a)²)

The algorithm is a *linear‑programming based flattening* procedure:

1. Start from a random Dirichlet vector (strictly positive, Σa = 1).
2. Repeatedly solve the LP
        minimise   M
        subject to A·b ≤ M·1 ,   Σb = 1 ,   b ≥ 0
   where A is the (2n‑1)×n convolution matrix of the current
   candidate a (i.e. (A·b)_k = Σ_i a_i b_{k-i}).
   The solution b is the distribution with the smallest possible
   maximal self‑convolution against the current a.
3. Replace a←b and iterate while the maximal convolution decreases.
   Each LP solves a global sub‑problem, so only a few iterations are
   required to drive the objective close to its theoretical lower
   bound 2n/(2n‑1) ≈ 1.
4. After the LP phase a short Adam refinement (sub‑gradient of max
   convolution) is applied to polish the final sequence.

The whole routine respects the 1000 s time budget, works for any
length up to a few thousand entries and returns the best sequence
found as a plain Python list.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper utilities (no closures)
# ----------------------------------------------------------------------
def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using FFT (real output)."""
    n = a.shape[0]
    L = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, L)
    conv = np.fft.irfft(fa * fa, L)[: 2 * n - 1]
    return conv.real


def conv_max(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution of a."""
    return full_convolution(a).max()


def compute_objective(a: np.ndarray) -> float:
    """Exact objective used by evaluate_sequence (sum(a) is assumed > 0)."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return 2.0 * a.shape[0] * conv_max(a) / (s * s)


def build_conv_matrix(a: np.ndarray) -> np.ndarray:
    """
    Build the (2n‑1)×n matrix A with A[k, j] = a_{k‑j} when the index
    is valid and 0 otherwise.
    """
    n = a.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    # each column j contains a shifted copy of a starting at row j
    for j in range(n):
        A[j:j + n, j] = a
    return A


# ----------------------------------------------------------------------
# LP based refinement
# ----------------------------------------------------------------------
def lp_refine_once(a: np.ndarray) -> tuple[np.ndarray, float]:
    """
    One refinement step: solve the LP
        minimise   M
        s.t.      A·b ≤ M·1
                  Σb = 1 ,  b ≥ 0
    Returns (b_opt, M_opt).  If the LP fails, returns (a, current_max).
    """
    n = a.shape[0]
    rows = 2 * n - 1

    # Build convolution matrix of the current a
    A = build_conv_matrix(a)                     # shape (rows, n)

    # Augment with the scalar variable M
    A_ub = np.hstack([A, -np.ones((rows, 1))])   # A·b - M ≤ 0
    b_ub = np.zeros(rows)

    # Equality Σb = 1
    A_eq = np.hstack([np.ones((1, n)), np.zeros((1, 1))])
    b_eq = np.array([1.0])

    # Objective: minimise M
    c = np.concatenate([np.zeros(n), [1.0]])

    # Bounds: b_i ≥ 0,   M ≥ 0
    bounds = [(0.0, None)] * n + [(0.0, None)]

    res = linprog(
        c,
        A_ub=A_ub,
        b_ub=b_ub,
        A_eq=A_eq,
        b_eq=b_eq,
        bounds=bounds,
        method="highs",
        options={"presolve": True, "time_limit": 30.0},
    )

    if not res.success:
        # fallback – keep the current vector
        return a.copy(), conv_max(a)

    # Extract solution
    b_opt = np.maximum(res.x[:n], 0.0)          # clip tiny negatives
    M_opt = float(res.x[-1])

    # Normalise numerically (sum may be a hair off 1)
    b_opt /= b_opt.sum()
    return b_opt, M_opt


def lp_flatten(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Repeatedly apply lp_refine_once until the maximal convolution stops
    improving.  Returns the best sequence discovered and its objective.
    """
    a = a0.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_val = compute_objective(a)

    last_improve = time.time()
    iteration = 0

    while time.time() < deadline:
        iteration += 1

        # Solve the LP for the current a
        b, M_lp = lp_refine_once(a)

        # Compute the exact max‑convolution of the new candidate
        M_exact = conv_max(b)

        # Occasionally the LP may return a value slightly smaller than the
        # true max; we use the exact value for the objective.
        val = 2.0 * n * M_exact

        if val < best_val:
            best_val = val
            best_a = b.copy()
            last_improve = time.time()
            print(f"[LP iter={iteration}] new best: {best_val:.6f}")

        # Stop if improvement is negligible
        if time.time() - last_improve > 7.0:
            break

        # Prepare next iteration
        a = b

    return best_a, best_val


# ----------------------------------------------------------------------
# Optional Adam fine‑tuning (same sub‑gradient as the original code)
# ----------------------------------------------------------------------
def softmax(x: np.ndarray) -> np.ndarray:
    mx = np.max(x)
    e = np.exp(x - mx)
    return e / e.sum()


def grad_submax(a: np.ndarray):
    """Exact sub‑gradient of max(a∗a) w.r.t. a."""
    n = a.shape[0]
    conv = full_convolution(a)
    max_c = conv.max()
    idxs = np.flatnonzero(np.isclose(conv, max_c, atol=1e-14))
    k = np.random.choice(idxs)

    g = np.zeros_like(a)
    start = max(0, k - (n - 1))
    end = min(k, n - 1)
    i = np.arange(start, end + 1)
    j = k - i
    g[i] = 2.0 * a[j]
    return g, max_c


def run_adam_opt(a_init: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """Adam optimiser on the sub‑gradient of max(a∗a) on the simplex."""
    n = a_init.shape[0]
    x = np.log(a_init + 1e-15)

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.05

    best_val = np.inf
    best_a = None
    t = 0
    last_improve = time.time()

    while time.time() < deadline and t < 200_000:
        a = softmax(x)                     # a ≥ 0, Σa = 1
        grad_a, max_c = grad_submax(a)
        val = 2.0 * n * max_c

        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[Adam iter={t}] new best: {val:.6f}")

        # back‑propagation through softmax
        avg = np.dot(a, grad_a)
        grad_x = a * (grad_a - avg)

        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)
        x = np.clip(x, -50.0, 50.0)

        # shake when stagnating
        if time.time() - last_improve > 5.0:
            x += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()
            print(f"[Adam iter={t}] shake (stagnation)")

    if best_a is None:
        best_a = np.full(n, 1.0 / n)
        best_val = compute_objective(best_a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    the supplied evaluation function.  The routine uses a linear‑programming
    flattening phase followed by a short Adam refinement.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5      # safety margin

    best_seq = None
    best_val = np.inf

    # Try a couple of lengths – larger n gives a tighter lower bound.
    # We start with 2048 (≈ 66 MiB dense matrix) which comfortably fits
    # into typical memory limits.
    lengths = [2048, 1024, 512]

    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length n={n} ===")

        # ------------------------------------------------------------------
        # 1) generate a few sensible initial candidates (all Σ=1)
        # ------------------------------------------------------------------
        init_candidates = []

        # warm‑start from a previously found best (if length matches)
        if best_seq is not None and len(best_seq) == n:
            init_candidates.append(np.asarray(best_seq, dtype=np.float64))

        # historic sequence (if provided)
        if "height_sequence_1" in globals():
            hs = np.asarray(height_sequence_1, dtype=np.float64)
            if hs.size >= n:
                hs = hs[:n]
            else:
                hs = np.pad(hs, (0, n - hs.size))
            hs = np.maximum(hs, 0.0)
            s = hs.sum()
            if s > 0.01:
                hs = hs / s
                init_candidates.append(hs)

        # a random Dirichlet vector – guarantees positivity
        init_candidates.append(np.random.dirichlet(np.ones(n)))

        # a mildly decreasing exponential shape
        exp_shape = np.exp(-np.linspace(0, 4, n))
        exp_shape /= exp_shape.sum()
        init_candidates.append(exp_shape)

        # ------------------------------------------------------------------
        # 2) LP flattening phase – we give each candidate a modest slice.
        # ------------------------------------------------------------------
        for idx, a0 in enumerate(init_candidates):
            if time.time() > deadline:
                break
            slice_time = min((deadline - time.time()) * 0.25, 120.0)
            sub_deadline = time.time() + slice_time
            print(f"[n={n}] LP start from candidate {idx}")

            a_lp, val_lp = lp_flatten(a0, sub_deadline)

            if val_lp < best_val:
                best_val = val_lp
                best_seq = a_lp.copy()
                print(f"[n={n}] LP best improved to {best_val:.6f}")

        # ------------------------------------------------------------------
        # 3) short Adam fine‑tuning on the current global best (if any)
        # ------------------------------------------------------------------
        if best_seq is not None and len(best_seq) == n and time.time() < deadline:
            adam_time = min((deadline - time.time()) * 0.30, 60.0)
            sub_deadline = time.time() + adam_time
            a_adam, val_adam = run_adam_opt(best_seq, sub_deadline)
            if val_adam < best_val:
                best_val = val_adam
                best_seq = a_adam.copy()
                print(f"[n={n}] Adam fine‑tune improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – if nothing succeeded return a simple uniform vector.
    # ------------------------------------------------------------------
    if best_seq is None:
        fallback_n = 1024
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)
        best_val = compute_objective(best_seq)

    print(f"\nBest value discovered: {best_val:.6f} (length {len(best_seq)})")
    # Convert to plain Python list as required by the external evaluator
    return best_seq.tolist()
