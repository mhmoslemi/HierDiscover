"""Step‑function optimiser for the `evaluate_sequence` objective.

The goal is to minimise 
    2·n·max(conv(a,a)) / (sum(a)**2)               (1)

which is invariant under scaling of `a`.  
If we normalise the vector to have ℓ₁‑norm √(2·n) the objective becomes simply
the maximal entry of the linear convolution `a*a`.  The problem is therefore
to find a non‑negative probability distribution whose aperiodic autocorrelation
is as flat as possible.

Strategy
--------
* **Exploration phase** – randomly generate a large variety of candidate
  distributions (uniform, Dirichlet, exponential, power‑law, beta‑shape,
  sinusoidal windows, mixtures).  Each candidate is normalised to ℓ₁ = √(2·n)
  and immediately evaluated; the best vector so far is kept.
* **Refinement phase** – run a customised Adam optimiser on the current best
  vector.  The optimiser works on the *soft‑max* approximation of the max‑
  convolution and switches to the exact sub‑gradient when the temperature becomes
  tiny.  Compared with the reference implementation we use a higher initial
  temperature, a slower decay schedule and more frequent jitter, which give a
  smoother landscape and help escape plateaus.
* **Multiple lengths** – the search is performed on a small set of lengths
  (the original size if known, and the maximal allowed size 8000).  Larger
  vectors have a lower theoretical lower bound (≈1) and therefore give more
  room for improvement.
* **Time‑budget aware** – the total wall‑clock budget (1000 s) is split
  roughly 30 % for pure random search and the remaining time for Adam
  refinement.  All loops stop cleanly before the deadline.

The function returns a plain Python list ready for `evaluate_sequence`.  """

import time
import math
import random
import numpy as np

# Helper functions for the Adam optimiser (soft‑max max‑convolution)
# ----------------------------------------------------------------------
def _softmax_weights(conv: np.ndarray, tau: float):
    """Stable soft‑max of the convolution values.
    Returns a weight vector w (summing to 1) and the smoothed maximum Lτ."""
    maxc = conv.max()
    if tau <= 0.0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w, maxc
    ex = np.exp((conv - maxc) / tau)
    sum_ex = ex.sum()
    w = ex / sum_ex
    Ltau = maxc + tau * np.log(sum_ex)
    return w, Ltau


def _weighted_rev(a: np.ndarray, w: np.ndarray, Nfft: int) -> np.ndarray:
    """Weighted reverse convolution needed for the (soft‑max) gradient."""
    A_fft = np.fft.rfft(a, n=Nfft)
    W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)
    conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
    conv_rev = conv_rev[: len(w) + a.shape[0] - 1]
    idx_start = 2 * a.shape[0] - 2
    rev = conv_rev[idx_start - np.arange(a.shape[0])]
    return rev


def _max_convolution(a: np.ndarray, Nfft: int) -> float:
    """Return max_{k} (a∗a)[k] using an FFT of length Nfft."""
    A_fft = np.fft.rfft(a, n=Nfft)
    conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * a.shape[0] - 1]
    return float(conv.max())


def _adam_optimize(
    a: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int,
    best_seq: np.ndarray,
    best_score: float,
) -> tuple[np.ndarray, float]:
    """Adam optimiser with a softened max‑convolution.

    Parameters
    ----------
    a : np.ndarray
        Starting vector (will be normalised to ℓ₁ = √(2·n)).
    n : int
        Length of the vector.
    deadline : float
        Absolute time (seconds since epoch) after which the loop stops.
    max_iters : int
        Upper bound on the number of Adam steps.
    best_seq, best_score
        Current global best – updated when an improvement is found.

    Returns
    -------
    best_seq, best_score
        Improved best vector and its exact objective value (the maximal
        entry of the linear convolution after normalisation).
    """
    target_sum = math.sqrt(2.0 * n)

    # normalise the initial vector
    if a.sum() == 0.0:
        a = np.full_like(a, target_sum / n)
    else:
        a = a * (target_sum / a.sum())

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr0 = 0.12                # a little larger than the reference
    jitter_interval = 2000    # more frequent jitter

    # temperature schedule (more generous than the reference)
    tau_factor = 0.45         # start with a higher temperature
    decay_rate = max_iters / 5.0   # slower decay → smoother landscape longer

    Nfft = 1 << ((2 * n - 1).bit_length())   # next power‑of‑two ≥ 2·n‑1

    for iteration in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ---------- exact convolution ----------
        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        M = float(conv.max())

        # ---------- temperature & soft‑max ----------
        tau = tau_factor * M * math.exp(-iteration / decay_rate)
        if tau < 1e-9:
            # exact sub‑gradient (active set)
            mask = conv >= M - 1e-12
            cnt = mask.sum()
            w = mask.astype(float) / max(cnt, 1)
            Ltau = M
        else:
            w, Ltau = _softmax_weights(conv, tau)

        # ---------- weighted reverse convolution ----------
        rev = _weighted_rev(a, w, Nfft)

        # ---------- Adam gradient step ----------
        # after normalisation the scaling factor in the gradient is 2
        grad = 2.0 * rev - (2.0 * n * Ltau) / (target_sum ** 3) * np.ones_like(a)

        # Adam updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        lr = lr0 / (1.0 + 1e-4 * iteration)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # ---------- projection onto feasible set ----------
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a = a * (target_sum / s)
        else:
            a = np.full_like(a, target_sum / n)

        # occasional jitter to help escape plateaus
        if iteration % jitter_interval == 0:
            jitter = np.random.randn(n) * (1e-3 * target_sum / n)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        # ---------- exact objective (after projection it equals M) ----------
        cur_score = M
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[adam] len {n:5d} iter {iteration:8d}  new best = {best_score:.6f}")

    return best_seq, best_score

    elif mode == 4:   # beta‑shape
        a_par = rng.uniform(0.2, 4.0)
        b_par = rng.uniform(0.2, 4.0)
        x = (np.arange(n) + 0.5) / n
        a = (x ** (a_par - 1)) * ((1 - x) ** (b_par - 1))
    else:            # sinusoidal window
        a = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** 2
    a = np.maximum(a, 0.0)
    a = a * (target_sum / a.sum())
    return a


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises `evaluate_sequence`.

    The routine spends ~30 % of the time doing a broad random search
    (many different analytic families), then refines the current best
    vector with a customised Adam optimiser.  The best sequence found
    (as a plain Python list) is returned.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # Determine which lengths to explore
    # ------------------------------------------------------------------
    lengths = [8000]                     # biggest allowed size – most freedom
    # also try the size of the previously known construction if it exists
    try:
        base = np.array(height_sequence_1, dtype=float)   # type: ignore
        n0 = base.shape[0]
        lengths.append(min(8000, max(300, n0)))
    except Exception:
        pass
    lengths = sorted(set(lengths))

    # ------------------------------------------------------------------
    # Global best tracking
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 1) Random exploration phase (≈30 % of the time budget)
    # ------------------------------------------------------------------
    explore_deadline = start_time + 0.30 * (budget_s - 5.0)
    while time.time() < explore_deadline:
        n = rng.choice(lengths)
        target_sum = math.sqrt(2.0 * n)

        # occasionally inject a candidate derived from the current best
        if best_seq is not None and rng.random() < 0.2:
            # perturb the current best slightly
            a = best_seq.copy()
            jitter = rng.normal(scale=0.005 * target_sum / n, size=n)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())
        else:
            a = _random_candidate(n, target_sum, rng)

        # evaluate (fast, no extra work)
        score = evaluate_sequence(a.tolist())
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[rand] length={n:5d} new best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Deterministic initial vectors (if we still have time)
    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() >= deadline:
            break
        target_sum = math.sqrt(2.0 * n)
        init_vecs = _make_initial_vectors(n, target_sum, rng)

        # add a linearly interpolated version of the previous best (if size differs)
        if best_seq is not None and best_seq.shape[0] != n:
            old_x = np.linspace(0.0, 1.0, best_seq.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            interp = np.interp(new_x, old_x, best_seq)
            interp = np.maximum(interp, 0.0)
            interp = interp / interp.sum() * target_sum
            init_vecs.append(interp)

        for v in init_vecs:
            if time.time() >= deadline:
                break
            score = evaluate_sequence(v.tolist())
            if score < best_score:
                best_score = score
                best_seq = v.copy()
                print(f"[init] length={n:5d} new best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Refinement with Adam optimisation on the best vector found
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback to plain uniform (should never happen)
        n = lengths[0]
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n)
        best_score = evaluate_sequence(best_seq.tolist())
        print("[fallback] uniform vector used")

    # allocate the rest of the budget to Adam
    remaining_iters = int(6e6)               # a generous upper bound
    best_seq, best_score = _adam_optimize(
        a=best_seq.copy(),
        n=len(best_seq),
        deadline=deadline,
        max_iters=remaining_iters,
        best_seq=best_seq,
        best_score=best_score,
    )

    # ------------------------------------------------------------------
    # Final report
    # ------------------------------------------------------------------
    print(f"[final] best score = {best_score:.6f}, length = {len(best_seq)}")
    return best_seq.tolist()
