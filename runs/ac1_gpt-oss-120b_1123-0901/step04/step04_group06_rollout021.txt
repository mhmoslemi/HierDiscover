def generate_initial_shape(n):
    choice = random.choice(['power', 'exp', 'tri', 'dirichlet'])
    if choice == 'power':
        p = random.uniform(0.2,0.9)
        idx = np.arange(1, n+1)
        shape = idx**(-p)
    elif choice == 'exp':
        lam = random.uniform(0.001,0.01) * n
        idx = np.arange(n)
        shape = np.exp(-lam*idx)
    elif choice == 'tri':
        # triangular shape: first half rising linearly, second half decreasing
        half = n//2
        inc = np.arange(1, half+1)
        dec = np.arange(n-half,0,-1)
        shape = np.concatenate([inc, dec]) if n%2==0 else np.concatenate([inc, np.arange(half,0,-1)])
        shape = shape.astype(float)
    elif choice == 'dirichlet':
        shape = np.random.gamma(shape=1.0, scale=1.0, size=n)
    # optional jitter, optional smoothing
    shape += np.random.rand(n)*1e-5
    # ensure positive
    shape = np.maximum(shape, 1e-12)
    # return
    return shape

def propose_candidate(seed=42, budget_s=1000.0, **kwargs):
    start = time.time()
    deadline = start + budget_s - 0.1

    np.random.seed(seed)
    random.seed(seed + 1) # different streams

    best_seq = None
    best_score = float('inf')

    # attempt warm start global variable
    try:
        global height_sequence_1
        hs = globals().get('height_sequence_1')
        if hs is not None:
            arr = np.array(hs, dtype=float)
            arr = np.clip(arr, 0.0, 1000.0)
            sc = evaluate_sequence(arr.tolist())
            if sc < best_score:
                best_score = sc
                best_seq = arr.copy()
                print(f"[{time.time():.2f}] Warm start global best {sc:.6f}")
    except Exception as e:
        pass

    # Also generate a few deterministic base candidates
    for n in [1500, 2000, 2500]:
        if time.time() >= deadline:
            break
        init = np.arange(1, n+1) ** -0.5  # power -0.5
        init = init / init.sum() * np.sqrt(2.0*n)
        seq = init.copy()
        sc = evaluate_sequence(seq.tolist())
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(f"[{time.time():.2f}] Base candidate n={n} score={sc:.6f}")

    # main restart loop
    while time.time() < deadline - 30.0:
        remaining = deadline - time.time()
        # choose n based on remaining time
        if remaining > 300:
            n = random.randint(2000, 3500)
        elif remaining > 150:
            n = random.randint(1500, 3000)
        else:
            n = random.randint(800, 2000)

        # generate initial shape
        shape = _generate_initial_shape(n)

        # Ensure positive & scale
        shape = np.maximum(shape, 1e-12)
        target_sum = math.sqrt(2.0 * n)
        shape = shape / shape.sum() * target_sum

        # compute logits
        x0 = np.log(shape)

        # allocate time slice for this run: 8-12 seconds but not exceed deadline-30
        slice_time = min(10.0, remaining * 0.5)
        run_deadline = min(deadline - 30.0, time.time() + slice_time)

        seq_local, sc_local = _optimise_adam_softmax(
            n,
            x0,
            target_sum,
            run_deadline,
            init_tau_factor=0.15,
            min_tau_factor=0.001,
            alpha=0.04,
            beta1=0.9,
            beta2=0.999,
        )
        if sc_local < best_score:
            best_score = sc_local
            best_seq = seq_local.copy()
            print(f"[{time.time():.2f}] Global new best {best_score:.6f} (n={n})")
            # early break if impressive bound
            if best_score < 1.503:
                break

    # final refinement phase
    if best_seq is not None:
        n = len(best_seq)
        target_sum = math.sqrt(2.0 * n)
        # initialize logits from best_seq
        x0 = np.log(best_seq + 1e-12)

        # longer run (remaining time)
        remaining = deadline - time.time()
        if remaining > 2.0:
            run_deadline = deadline
            seq_ref, sc_ref = _optimise_adam_softmax(
                n,
                x0,
                target_sum,
                run_deadline,
                init_tau_factor=0.08,
                min_tau_factor=5e-5,
                alpha=0.02,
                beta1=0.9,
                beta2=0.999,
            )
            if sc_ref < best_score:
                best_score = sc_ref
                best_seq = seq_ref.copy()
                print(f"[{time.time():.2f}] Final optimizer best {best_score:.6f}")

        # hill climbing / simulated annealing
        best_seq = _hill_climb(best_seq, deadline)
        final_sc = evaluate_sequence(best_seq.tolist())
        if final_sc < best_score:
            best_score = final_sc
            print(f"[{time.time():.2f}] After hill climb best {best_score:.6f}")

    # fallback
    if best_seq is None:
        n = 1000
        target_sum = math.sqrt(2.0*n)
        best_seq = np.full(n, target_sum/n, dtype=float)

    print(f"[{time.time():.2f}] Finished, best score = {best_score}")
    return best_seq.tolist()

"""
Propose a non‑negative step function that minimises

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The optimiser works in three stages

1. **Exploratory restarts** – many short Adam‑optimisations on a
   soft‑max parametrisation `a = S·softmax(x)` with random lengths
   (≈ 800‑3500) and a variety of initial shapes (power‑law,
   exponential, triangular, Dirichlet).  The best sequence found so far
   is kept globally.

2. **Intensive polishing** – the current global best is fed back as a
   warm start to a longer Adam run (the temperature of the smooth‑max
   surrogate is driven to very small values and the learning‑rate is
   reduced).

3. **Greedy local refinement** – a cheap projected‑gradient hill‑climb
   that repeatedly moves a small amount of mass from indices that
   contribute most to the maximal convolution towards indices that
   contribute least.  The update is projected back onto the non‑negative
   simplex (fixed sum).  The refinement runs until the time budget is
   exhausted.

All heavy lifting (full linear convolution) is performed with a real‑FFT
helper.  The routine respects the supplied `budget_s` (default 1000 s)
and prints progress timestamps.  The final result is returned as a
plain Python list suitable for `evaluate_sequence`.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Fast full linear convolution using real FFT (zero‑padding to a power of 2)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the linear convolution of real vectors x and y (full length)."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    # remove tiny negative artefacts caused by numerical noise
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Euclidean projection onto the non‑negative simplex {u ≥ 0, Σu = z}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray, z: float = 1.0) -> np.ndarray:
    """Project vector v onto the simplex of non‑negative vectors with sum z."""
    if z < 0:
        raise ValueError("z must be non‑negative")
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform distribution
        return np.full_like(v, z / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Adam optimisation of a smoothed‑max surrogate on the log‑it space
# ----------------------------------------------------------------------
def _optimise_adam_softmax(
    n: int,
    x0: np.ndarray,
    target_sum: float,
    run_deadline: float,
    *,
    init_tau_factor: float = 0.12,
    min_tau_factor: float = 0.001,
    alpha: float = 0.05,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Optimise a sequence of length ``n`` using Adam on the log‑it variables.
    Returns the best sequence encountered (as a NumPy array) and its exact
    score obtained from ``evaluate_sequence``.
    """
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    best_seq = None
    best_score = float("inf")
    iteration = 0
    # the soft‑max normalisation factor
    S = target_sum

    while time.time() < run_deadline:
        # ---- 1. map logits to the simplex (a = S·softmax(x)) ----
        max_x = np.max(x)
        exp_shift = np.exp(x - max_x)                # stability
        p = exp_shift / np.sum(exp_shift)            # probabilities (sum = 1)
        a = S * p                                    # actual coefficients

        # ---- 2. convolution and current (smooth) max ----
        conv = _full_convolution_fft(a, a)
        max_val = conv.max()
        if max_val < best_score:
            best_score = max_val
            best_seq = a.copy()
            print(f"[{time.time():.2f}] n={n}  new local best = {max_val:.6f}")

        # ---- 3. smooth‑max weights ---------------------------------
        tau = max(init_tau_factor * max_val * (0.95 ** iteration),
                  min_tau_factor * max_val)
        scaled = (conv - max_val) / tau
        w = np.exp(scaled)                           # soft‑max over conv entries
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---- 4. gradient wrt a ------------------------------------
        w_rev = w[::-1]
        conv2 = _full_convolution_fft(a, w_rev)      # length 3n‑2
        idx = (2 * n - 2) - np.arange(n)              # positions needed for Σ_j a_j w_{i+j}
        grad_a = 2.0 * conv2[idx]

        # ---- 5. gradient wrt logits x (chain rule) -------------
        dotp = np.dot(p, grad_a)
        grad_x = S * p * (grad_a - dotp)

        # ---- 6. Adam update ---------------------------------------
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # occasional tiny perturbation to escape plateaus
        if iteration % 250 == 0 and random.random() < 0.2:
            x += np.random.randn(n) * 0.001

        # slowly decay the base learning‑rate
        alpha *= 0.9995

    # final exact evaluation (in case the best sequence was never stored)
    if best_seq is None:
        best_seq = S * np.exp(x - np.max(x))
        best_seq = best_seq / best_seq.sum() * S
    final_score = evaluate_sequence(best_seq.tolist())
    return best_seq, final_score


# ----------------------------------------------------------------------
# Greedy hill‑climb that transfers a small amount of mass from heavy
# to light contributors of the current maximal convolution.
# ----------------------------------------------------------------------
def _hill_climb(a: np.ndarray, time_deadline: float) -> np.ndarray:
    """
    Perform a fast projected‑gradient hill‑climb on ``a`` until the
    deadline is reached.  The update moves a fraction (10 %) of the
    mass from the index with the largest gradient contribution to the
    one with the smallest.
    """
    n = a.shape[0]
    target_sum = a.sum()
    a = a.copy()

    while time.time() < time_deadline:
        conv = _full_convolution_fft(a, a)
        max_val = conv.max()
        # active set: indices where conv attains (or nearly attains) the max
        active = np.flatnonzero(conv >= max_val - 1e-12)
        if active.size == 0:
            break

        # build a smooth weight vector that is uniform on the active set
        w = np.zeros_like(conv)
        w[active] = 1.0 / active.size

        # gradient of the weighted sum w.r.t. a (same formula as in Adam)
        w_rev = w[::-1]
        conv2 = _full_convolution_fft(a, w_rev)
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * conv2[idx]

        hi = int(np.argmax(grad))
        lo = int(np.argmin(grad))
        if grad[hi] <= grad[lo] + 1e-12:
            break      # no useful direction

        # move 10 % of the mass from hi to lo (but never below zero)
        delta = 0.10 * a[hi]
        if delta <= 1e-12:
            break
        a[hi] -= delta
        a[lo] += delta

        # keep the vector on the simplex (numeric drift)
        a = _project_simplex(a, target_sum)

    return a

def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):

    np.random.seed(seed)
