"""
Propose a low‑value step‑function sequence.

The objective that is minimized by the external evaluator is

    F(a) = 2·n·max(a*a) / (sum(a)**2)

where ``a*a`` is the linear convolution of the non‑negative vector ``a``.
Because the expression is homogeneous, we may restrict ourselves to the
probability simplex (sum(a)=1).  The problem is to make the largest
convolution entry as small as possible.

The algorithm works in two phases for several candidate lengths *n*:

1. **Projected sub‑gradient descent** on the convex (max‑convolution)
   objective.  At each iteration the index of the current maximum is
   found, its exact sub‑gradient
   ``∂max / ∂a_i = 2·a_{k‑i}`` (with out‑of‑range entries ignored) is
   computed, and the vector is updated with a diminishing stepsize
   ``η_t = η0 / sqrt(t+1)``.  After the update the point is projected back
   onto the simplex using the O(n log n) Euclidean projection.

2. **Local random refinement (simulated annealing)**.  Starting from the
   best vector found by the descent, tiny pairwise mass transfers are
   proposed.  Moves that improve the objective are always accepted; moves
   that worsen it are accepted with a temperature‑controlled probability.
   The temperature decays exponentially.

The code explores a few lengths (1024 … 16384), several random restarts
per length and respects the global time budget (default 1000 s).  The
best sequence found is returned as a plain Python list.

All heavy work (convolution, projection) uses NumPy/FFT and runs on a
single CPU; the implementation stays well inside the time limit while
searching aggressively for values below the target 1.503.
"""

def _full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a*a (length 2·len(a)-1) computed with FFT.
    The result is a 1‑D array of length 2·n‑1 (full convolution).
    """
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


def _subgradient_max(a: np.ndarray, idx: int) -> np.ndarray:
    """
    Sub‑gradient of the max‑convolution at index ``idx``.
    For the chosen index the exact gradient of c_idx = (a*a)[idx] is
        g_i = 2·a_{idx‑i}   if 0 ≤ idx‑i < n, else 0.
    """
    n = a.shape[0]
    grad = np.zeros_like(a)

    i_start = max(0, idx - (n - 1))
    i_end = min(n - 1, idx)
    # i in [i_start, i_end] → idx‑i is a valid index
    i_range = np.arange(i_start, i_end + 1, dtype=np.int64)
    grad[i_range] = a[idx - i_range]
    grad *= 2.0
    return grad


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        {x | x_i ≥ 0, Σ_i x_i = 1}
    """
    if v.ndim != 1:
        raise ValueError("v must be a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # Should not happen for a non‑empty vector, but keep safe.
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative sequence with a low evaluation score.
    The function respects ``budget_s`` seconds and returns the best
    sequence (as a plain Python list) discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None          # best vector found (already on the simplex)
    best_val = float("inf")  # best objective value seen so far

    # ------------------------------------------------------------------
    # Problem sizes to explore – larger n gives more freedom but costs
    # more per iteration.
    lengths = [1024, 2048, 4096, 8192, 16384]

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")
        # number of independent restarts for this length
        restarts = 5 if n <= 2048 else 3

        for r in range(restarts):
            if time.time() > deadline:
                break

            # -------------------- initialise a (probability vector) ------------
            if r == 0 and best_seq is not None and best_seq.shape[0] == n:
                a = best_seq.copy()
                print("[restart 0] warm start from global best")
            elif 'height_sequence_1' in globals():
                cand = np.asarray(height_sequence_1, dtype=np.float64)
                if cand.shape[0] >= n:
                    cand = cand[:n]
                else:
                    cand = np.pad(cand, (0, n - cand.shape[0]))
                cand = np.maximum(cand, 0.0)
                s = cand.sum()
                if s < 0.01:
                    a = np.random.dirichlet(np.ones(n))
                else:
                    a = cand / s
                print("[restart 0] start from height_sequence_1")
            else:
                a = np.random.dirichlet(np.ones(n))
                print("[restart] random start")

            # -----------------------------------------------------------------
            # Phase 1 – projected sub‑gradient descent on the max‑convolution
            lr0 = 0.5                    # initial step‑size factor
            max_iters = 50000 if n <= 2048 else 25000
            last_improve = time.time()

            for t in range(max_iters):
                if time.time() > deadline:
                    break

                conv = _full_convolution(a)
                k = int(conv.argmax())        # index of the current maximum
                maxc = conv[k]
                obj = 2.0 * n * maxc          # exact objective for sum(a)=1

                if obj < best_val:
                    best_val = obj
                    best_seq = a.copy()
                    print(f"[n={n} r={r} it={t:05d}] new global best: {obj:.6f}")

                # sub‑gradient step
                grad = _subgradient_max(a, k)
                step = lr0 / np.sqrt(t + 1)   # diminishing stepsize
                a = a - step * grad
                a = _project_simplex(a)

                # occasional random shake if stagnating
                if time.time() - last_improve > 5.0:
                    noise = (np.random.rand(n) - 0.5) * 1e-3
                    a = _project_simplex(a + noise)
                    last_improve = time.time()
                    print(f"[n={n} r={r} it={t}] shake")

                # early exit if we already beat the target
                if best_val < 1.503:
                    print(f"Target reached (value={best_val:.6f}); terminating.")
                    deadline = time.time()   # stop any further work
                    break

            # -----------------------------------------------------------------
            # Phase 2 – local random refinement (simulated annealing)
            if best_seq is not None and time.time() < deadline:
                a_loc = best_seq.copy()
                cur_conv = _full_convolution(a_loc)
                cur_val = 2.0 * n * cur_conv.max()

                # annealing schedule
                temp = 0.05
                decay = 0.9995
                min_temp = 1e-5

                while time.time() < deadline:
                    i, j = np.random.choice(n, 2, replace=False)
                    # move a small fraction of mass from i to j
                    delta = 0.02 * a_loc[i] * np.random.rand()
                    if delta == 0.0:
                        continue
                    prop = a_loc.copy()
                    prop[i] -= delta
                    prop[j] += delta
                    prop = _project_simplex(prop)

                    prop_conv = _full_convolution(prop)
                    prop_val = 2.0 * n * prop_conv.max()
                    diff = prop_val - cur_val

                    # Metropolis acceptance
                    if diff < 0 or np.random.rand() < np.exp(-diff / max(temp, 1e-12)):
                        a_loc = prop
                        cur_val = prop_val
                        if cur_val < best_val:
                            best_val = cur_val
                            best_seq = a_loc.copy()
                            print(f"[refine n={n}] improved to {cur_val:.6f}")

                    temp = max(temp * decay, min_temp)

    # ------------------------------------------------------------------
    # Fallback – uniform distribution if nothing has been found (unlikely).
    if best_seq is None:
        n_fallback = 1024
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        best_val = 2.0 * n_fallback * (1.0 / n_fallback)   # = 2.0
        print(f"Fallback uniform sequence, value = {best_val:.6f}")

    # Return a plain Python list as required by the external evaluator.
    return best_seq.tolist()
