"""
propose_candidate
-----------------

Hybrid mirror‑descent / exponentiated‑gradient optimizer for the
non‑negative step‑function problem.

* The objective to minimise is  F(a)=2·n·max(a*a)/ (∑a)² .
  By normalising the vector to lie on the probability simplex
  (∑a=1) the denominator is constant and we only need to minimise
  max(a*a).

* A sub‑gradient of max(a*a) is obtained from the active (maximal)
  convolution entries.  Using an exponentiated‑gradient (mirror)
  update keeps the iterate non‑negative; a simple ℓ₁‑projection
  (re‑normalisation) restores the simplex constraint.

* Several problem sizes (1024 … 16384) are explored, each with a few
  random restarts.  The current best solution is kept globally.
  After the gradient phase a tiny random‑hill‑climbing refinement
  attempts to squeeze a little more value.

* The implementation relies only on NumPy and the standard library,
  respects the provided time budget (default 1000 s) and returns the
  best sequence as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2·len(a)-1) using FFT.
    ``fft_len`` must be a power of two ≥ 2·len(a)-1.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.shape[0] - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Compute (w ★ a)[i] = Σ_k w[k]·a[i+k-(n-1)],   i = 0 … n‑1.
    This is the gradient of Σ_k w_k·(a*a)_k divided by 2.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        {x | x_i ≥ 0, Σ_i x_i = 1}
    """
    if v.ndim != 1:
        raise ValueError("v must be a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _symmetrize(a: np.ndarray) -> np.ndarray:
    """
    Force mirror symmetry a_i = a_{n-1-i} and re‑normalise.
    """
    a = (a + a[::-1]) * 0.5
    s = a.sum()
    if s == 0.0:
        a = np.full_like(a, 1.0 / a.shape[0])
    else:
        a = a / s
    return a


def _initial_arcsine(n: int) -> np.ndarray:
    """
    A deterministic “arcsine”‑like start: a_i ∝ 1/√((i+0.5)*(n-i-0.5)).
    This distribution is known to have a relatively flat self‑convolution.
    """
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals = np.maximum(vals, 0.0)
    s = vals.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return vals / s

                print(f"[refine] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform distribution if nothing succeeded.
    if best_seq is None:
        n_fallback = 2048
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        best_val = _objective(best_seq)
        print(f"Fallback uniform sequence, value = {best_val:.6f}")

    # Return a plain list as required by the external evaluator.
    return best_seq.tolist()
