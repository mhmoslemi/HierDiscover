def run_smoothed_opt(n, init_a, deadline, tau_start=0.5, tau_end=0.001, tau_decay=0.99, lr=0.1):
    a = init_a.copy()
    best_a = a.copy()
    best_val = compute_objective(a)

    tau = tau_start
    iteration = 0
    while time.time() < deadline and tau > tau_end:
        # compute gradient
        conv = full_convolution(a)
        # compute softmax weights
        max_c = np.max(conv)
        exps = np.exp((conv - max_c) / tau)
        w = exps / np.sum(exps)   # shape 2n-1
        # gradient
        grad = 2 * np.convolve(w, a[::-1])[n-1:n-1+n]
        # multiplicative update
        a = a * np.exp(-lr * grad)
        # renormalize to sum=1
        s = a.sum()
        if s == 0:
            a = np.full(n, 1.0 / n)
        else:
            a /= s
        # compute actual objective occasionally
        if iteration % 10 == 0:
            val = compute_objective(a)
            if val < best_val:
                best_val = val
                best_a = a.copy()
                # maybe adjust lr or tau
        # schedule tau down slowly each iteration
        tau *= tau_decay
        iteration += 1

        if iteration % 1000 == 0:
            # maybe reset or shuffle
            pass
    return best_a, best_val

def run_smoothed_opt(n, init_a, deadline):
    a = init_a.copy()
    # ensure positive
    a = np.maximum(a, 1e-12)
    a = a / a.sum()
    best_a = a.copy()
    best_val = compute_objective(a)
    # hyperparameters
    tau = 1.0   # initial temperature
    tau_min = 1e-4
    tau_decay = 0.98  # per iteration
    lr = 0.2
    # loops
    it = 0
    while time.time() < deadline:
        # compute convolution via FFT
        conv = full_convolution(a)
        # compute softmax weights
        max_c = conv.max()
        # numeric stable
        exps = np.exp((conv - max_c) / tau)
        w = exps / exps.sum()
        # gradient via convolution of w and reversed a
        grad = 2.0 * np.convolve(w, a[::-1])[n-1 : n-1 + n]
        # apply multiplicative update
        # clip gradient magnitude to avoid huge steps
        grad = np.clip(grad, -1e3, 1e3)
        a = a * np.exp(-lr * grad)
        # renorm
        s = a.sum()
        if s == 0 or np.isnan(s):
            a = np.full(n, 1.0/n)
        else:
            a = a / s
        # optional small jitter to avoid stagnation

            a = np.maximum(a, 0)

            val = compute_objective(a)
            if val < best_val:
                best_val = val
                best_a = a.copy()
                # maybe reduce lr if improvement
        # schedule tau
        tau = max(tau * tau_decay, tau_min)
        it += 1
        # maybe break if tau near min and no improvement for many iterations
    return best_a, best_val

"""
Propose a step‑function (non‑negative vector) that minimises

        F(a) = 2·n·max(a∗a) / (sum(a)²)

The algorithm consists of three stages:

1. **Smooth exponential‑mirror descent** – a soft‑max approximation of the
   max‑convolution provides a differentiable objective.  The update is
   multiplicative (exponentiated gradient) and the vector is always kept on
   the simplex.

2. **Random‑restart refinement** – several random Dirichlet seeds are
   processed; the best result is retained.

3. **(1+1)‑ES fine‑tuning** – Gaussian perturbations of the current best are
   accepted when they improve the true (non‑smooth) objective.

The whole search respects a global time budget (default 1000 s) and works
for any length up to a few‑tens of thousands.  Larger lengths give more
flexibility, but the implementation adapts automatically if the budget runs
out.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------
def logsumexp(x: np.ndarray, axis: int = None) -> float:
    """Stable log‑sum‑exp."""
    m = np.max(x, axis=axis, keepdims=True)
    s = np.exp(x - m).sum(axis=axis, keepdims=True)
    return (m + np.log(s)).squeeze()


def fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution using a real FFT.
    Returns a real array of length len(a)+len(b)-1.
    """
    n = a.shape[0] + b.shape[0] - 1
    L = 1 << int(np.ceil(np.log2(n)))
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b, L)
    conv = np.fft.irfft(fa * fb, L)[:n]
    return conv.real


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Shortcut for a * a."""
    return fft_convolve(a, a)


def compute_objective(a: np.ndarray) -> float:
    """
    Exact objective used by evaluate_sequence (sum(a) may be any positive number).
    Equivalent to 2·n·max(a∗a) / (sum(a)²).
    """
    s = a.sum()
    if s < 1e-12:
        return np.inf
    conv = full_convolution(a)
    return 2.0 * a.shape[0] * conv.max() / (s * s)


# ----------------------------------------------------------------------
# 1. Smooth exponential‑mirror descent
# ----------------------------------------------------------------------
def run_smooth_opt(
    n: int,
    init_a: np.ndarray,
    deadline: float,
    tau_start: float = 1.0,
    tau_end: float = 1e-4,
    lr: float = 0.25,
) -> tuple[np.ndarray, float]:
    """
    Minimises the smoothed max‑convolution using multiplicative updates.
    Returns the best vector (on the simplex) and its exact objective.
    """
    # keep a strictly positive and normalised
    a = np.maximum(init_a, 1e-12)
    a = a / a.sum()

    best_a = a.copy()
    best_val = compute_objective(a)

    tau = tau_start
    decay = 0.995          # geometric decay of the temperature
    it = 0
    last_improve = time.time()

    while time.time() < deadline:
        # ---------- smoothed max ----------
        conv = full_convolution(a)                 # shape (2n-1,)
        max_c = conv.max()
        # stable soft‑max weights
        exp_scaled = np.exp((conv - max_c) / tau)
        w = exp_scaled / exp_scaled.sum()          # ∑ w = 1

        # ---------- gradient of smooth max ----------
        # grad_j = 2 * Σ_k w_k * a_{k‑j}
        grad = 2.0 * fft_convolve(w, a[::-1])[n - 1 : n - 1 + n]

        # multiplicative (exponentiated) update
        # clip gradient magnitude to avoid overflow
        grad = np.clip(grad, -1e3, 1e3)
        a = a * np.exp(-lr * grad)

        # keep on the simplex
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0.0 or np.isnan(s):
            a = np.full(n, 1.0 / n)
        else:
            a /= s

        # occasional tiny jitter to break possible stagnation
        if it % 500 == 0:
            a += (np.random.rand(n) - 0.5) * 1e-4
            a = np.maximum(a, 0.0)
            a /= a.sum()

        # evaluate the true objective every few steps
        if it % 25 == 0:
            val = compute_objective(a)
            if val < best_val:
                best_val = val
                best_a = a.copy()
                last_improve = time.time()
                print(f"[smooth n={n} it={it:06d}] new best: {val:.6f}")

        # temperature schedule
        tau = max(tau * decay, tau_end)

        # stop if we have not improved for a while
        if time.time() - last_improve > 8.0:
            # reset to the best seen and boost the learning rate a little
            a = best_a.copy()
            lr = min(lr * 1.1, 0.5)
            last_improve = time.time()

        it += 1
        if it > 200_000:               # safety guard
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# 2. (1+1) Evolution Strategy – fine tuning on the true objective
# ----------------------------------------------------------------------
def run_es_opt(
    n: int,
    init_a: np.ndarray,
    deadline: float,
    sigma0: float = 0.015,
    max_iter: int = 60_000,
) -> tuple[np.ndarray, float]:
    """
    Simple (1+1)‑ES that perturbs the current best with Gaussian noise.
    Operates directly on the simplex.
    """
    a = init_a.copy()
    best_val = compute_objective(a)
    best_a = a.copy()
    sigma = sigma0
    it = 0
    last_improve = time.time()

    while time.time() < deadline and it < max_iter:
        it += 1
        trial = a + sigma * np.random.randn(n)
        trial = np.maximum(trial, 0.0)
        s = trial.sum()
        if s == 0.0:
            trial = np.full(n, 1.0 / n)
        else:
            trial /= s

        val = compute_objective(trial)
        if val < best_val:
            best_val = val
            best_a = trial.copy()
            a = trial
            sigma = min(sigma * 1.07, 0.2)
            last_improve = time.time()
            print(f"[ES n={n} it={it}] new best: {best_val:.6f}")
        else:
            sigma *= 0.992

        if time.time() - last_improve > 6.0:
            a = best_a.copy()
            sigma = sigma0
            last_improve = time.time()

    return best_a, best_val

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    evaluate_sequence().  The function respects the supplied time budget and
    returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_val = np.inf

    # Try progressively smaller problem sizes if time runs out.
    # Larger n gives more degrees of freedom, but also costs more per iteration.
    lengths = [16384, 8192, 4096, 2048, 1024]

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")

        # ---------- build a few sensible initial candidates ----------
        init_candidates = []

        # warm‑start from the global best if we already have one with same size
        if best_seq is not None and len(best_seq) == n:
            init_candidates.append(np.array(best_seq, dtype=np.float64))

        # historic construction if available
        if "height_sequence_1" in globals():
            hs = np.asarray(height_sequence_1, dtype=np.float64)
            if hs.size >= n:
                hs = hs[:n]
            else:
                hs = np.pad(hs, (0, n - hs.size))
            hs = np.maximum(hs, 1e-12)
            hs = hs / hs.sum()
            init_candidates.append(hs)

        # simple linear decay (U‑shape) – puts more mass at the opposite ends
        dec = np.arange(n, 0, -1, dtype=np.float64)
        dec = dec / dec.sum()
        init_candidates.append(dec)

        # uniform distribution (the baseline)
        init_candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

        # a few random Dirichlet draws
        for _ in range(2):
            init_candidates.append(np.random.dirichlet(np.ones(n)))

        # ---------- smooth mirror‑descent from each seed ----------
        for idx, a0 in enumerate(init_candidates):
            if time.time() > deadline:
                break
            # allocate a modest time slice for each run
            slice_time = min((deadline - time.time()) * 0.20, 30.0)
            sub_deadline = time.time() + slice_time
            a_opt, val_opt = run_smooth_opt(
                n,
                a0,
                sub_deadline,
                tau_start=1.0,
                tau_end=1e-4,
                lr=0.25,
            )
            if val_opt < best_val:
                best_val = val_opt
                best_seq = a_opt.copy()
                print(f"[n={n} seed={idx}] GLOBAL best improved to {best_val:.6f}")

        # ---------- random restarts with longer budget ----------
        while time.time() < deadline - 5.0:
            a_rand = np.random.dirichlet(np.ones(n))
            slice_time = min((deadline - time.time()) * 0.15, 25.0)
            sub_deadline = time.time() + slice_time
            a_opt, val_opt = run_smooth_opt(
                n,
                a_rand,
                sub_deadline,
                tau_start=0.8,
                tau_end=5e-5,
                lr=0.20,
            )
            if val_opt < best_val:
                best_val = val_opt
                best_seq = a_opt.copy()
                print(f"[n={n} random] GLOBAL best now {best_val:.6f}")
            # stop a bit earlier if we are close to the final deadline
            if (deadline - time.time()) < 8.0:
                break

        # ---------- final (1+1) ES fine‑tuning ----------
        if best_seq is not None and len(best_seq) == n and time.time() < deadline:
            es_time = min((deadline - time.time()) * 0.30, 60.0)
            sub_deadline = time.time() + es_time
            a_es, val_es = run_es_opt(n, best_seq, sub_deadline)
            if val_es < best_val:
                best_val = val_es
                best_seq = a_es.copy()
                print(f"[n={n}] ES final best now {best_val:.6f}")

    # ------------------------------------------------------------------
    # fallback – if nothing succeeded, return a uniform vector of size 1024.
    # ------------------------------------------------------------------
    if best_seq is None:
        fallback_n = 1024
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)
        best_val = compute_objective(best_seq)

    print(f"\nBest value discovered: {best_val:.6f} (length {len(best_seq)})")
    # convert to plain Python list as required by the external evaluator
    return best_seq.tolist()
