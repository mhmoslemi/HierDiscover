"""
Propose a non‑negative step‑function (probability vector) that minimises

    F(a) = 2·n·max(a*a) / (sum(a)^2)

The algorithm works on the probability simplex via a soft‑max parametrisation.
Instead of the non‑smooth max we use a temperature‑controlled
log‑sum‑exp (smooth‑max) which yields a true gradient.
The temperature τ is annealed from a relatively large value (≈0.1) down to
≈1e‑6, so early iterations see a smooth landscape and later ones focus on
the true peak.

Optimization is done with Adam on the logits and a few classic tricks:
* warm‑starting from the best known sequence,
* occasional random “shakes’’ when progress stalls,
* a final cheap local refinement that moves tiny amounts of mass between
  coordinates.

The whole search respects the 1000 s wall‑time budget and works for a range
of lengths (default 1024, 2048 and 4096).  The best sequence found is returned
as a plain Python list.
"""

import time
import numpy as np


def full_convolution(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) using FFT."""
    n = a.shape[0]
    if fft_len is None:
        fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


def cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) equal to
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def smooth_max_and_weights(conv: np.ndarray, tau: float):
    """
    Compute a temperature‑smoothed max of `conv` and the corresponding
    softmax weights (∂smooth_max/∂conv).
    """
    # Numerical stability: shift by the max
    c_max = conv.max()
    shifted = (conv - c_max) / tau
    # exp may underflow for very negative values – that's fine
    exp_shifted = np.exp(shifted)
    sum_exp = exp_shifted.sum()
    smooth = c_max + tau * np.log(sum_exp)
    weights = exp_shifted / sum_exp
    return smooth, weights


def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs) -> list:
    """
    Time‑bounded optimisation that returns the best non‑negative sequence
    found so far.  The function follows the strategy described in the
    module docstring.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # safety margin

    best_seq = None          # best sequence (numpy array, sum = 1)
    best_val = float("inf")  # best objective 2·n·max(conv)

    # Problem sizes – larger n gives a lower theoretical bound (≈1)
    lengths = [1024, 2048, 4096]

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")

        # FFT lengths reused many times
        fft_len_conv = 1 << int(np.ceil(np.log2(2 * n - 1)))
        fft_len_corr = 1 << int(np.ceil(np.log2(3 * n)))  # ≥ (2n‑1)+n‑1

        # Number of restarts – more for the smaller problems
        if n <= 1024:
            restarts = 8
        elif n <= 2048:
            restarts = 5
        else:
            restarts = 3

        for r in range(restarts):
            if time.time() > deadline:
                break

            # ----- initialise logits ------------------------------------------------
            if best_seq is not None and best_seq.shape[0] == n:
                a0 = best_seq.copy()
                a0 /= a0.sum()
                x = np.log(a0 + 1e-15)
                print(f"[n={n} restart={r}] warm start from previous best")
            elif 'height_sequence_1' in globals():
                cand = np.array(height_sequence_1, dtype=np.float64)
                if cand.shape[0] >= n:
                    cand = cand[:n]
                else:
                    cand = np.pad(cand, (0, n - cand.shape[0]))
                cand = np.maximum(cand, 0.0)
                if cand.sum() < 0.01:
                    cand = np.random.dirichlet(np.ones(n))
                else:
                    cand /= cand.sum()
                x = np.log(cand + 1e-15)
                print(f"[n={n} restart={r}] start from height_sequence_1")
            else:
                a0 = np.random.dirichlet(np.ones(n))
                x = np.log(a0 + 1e-15)
                print(f"[n={n} restart={r}] random start")

            # ----- Adam state -------------------------------------------------------
            m = np.zeros(n, dtype=np.float64)
            v = np.zeros(n, dtype=np.float64)
            beta1, beta2, eps = 0.9, 0.999, 1e-8
            lr = 0.1                     # initial learning rate
            decay_every = 5000           # decay interval (iterations)

            # Total iterations – we allocate a generous budget per restart
            # but always respect the global deadline.
            max_iters = 120000 if n <= 1024 else 80000 if n <= 2048 else 60000
            t = 0
            last_improve = time.time()

            # Temperature schedule (linear annealing)
            tau_initial = 0.12
            tau_final = 5e-7

            while time.time() < deadline and t < max_iters:
                # ----- soft‑max to obtain a (probability vector) -----
                max_x = np.max(x)
                exp_x = np.exp(x - max_x)
                a = exp_x / exp_x.sum()               # a_i ≥ 0, Σa=1

                # ----- convolution and smooth‑max -----
                conv = full_convolution(a, fft_len=fft_len_conv)
                # anneal temperature
                tau = max(tau_final,
                          tau_initial * (1.0 - t / max_iters))
                smooth, weights = smooth_max_and_weights(conv, tau)

                # true value (for bookkeeping) – the exact max, not the smoothed one
                cur_val = 2.0 * n * conv.max()

                if cur_val < best_val - 1e-12:
                    best_val = cur_val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} restart={r} iter={t:06d}] new best {cur_val:.6f}")

                # ----- gradient of smooth‑max w.r.t. a -----
                # grad_a = ∂smooth/∂a = 2·(weights ★ a)
                grad_a = 2.0 * cross_correlation(weights, a, fft_len=fft_len_corr)

                # back‑propagation through the soft‑max
                inner = np.dot(a, grad_a)
                grad_x = a * (grad_a - inner)

                # ----- Adam update -----
                t += 1
                m = beta1 * m + (1.0 - beta1) * grad_x
                v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
                m_hat = m / (1.0 - beta1 ** t)
                v_hat = v / (1.0 - beta2 ** t)
                x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

                # learning‑rate decay (geometric)
                if t % decay_every == 0:
                    lr *= 0.7

                # ----- stagnation handling -----
                if time.time() - last_improve > 35.0:
                    # Small random jitter in log‑space
                    x += (np.random.rand(n) - 0.5) * 6e-3
                    last_improve = time.time()
                    print(f"[n={n} restart={r} iter={t}] shake (stagnation)")

            # ----- final fine‑tuning at very low temperature -----
            if best_seq is not None and best_seq.shape[0] == n:
                # re‑initialise Adam with a much smaller learning rate
                a = best_seq.copy()
                max_x = np.max(np.log(a + 1e-15))
                x = np.log(a + 1e-15)
                m = np.zeros(n)
                v = np.zeros(n)
                lr = 0.02
                fine_iters = 12000
                for ft in range(fine_iters):
                    if time.time() > deadline:
                        break
                    # very low temperature → true max
                    tau = tau_final
                    max_x = np.max(x)
                    exp_x = np.exp(x - max_x)
                    a = exp_x / exp_x.sum()
                    conv = full_convolution(a, fft_len=fft_len_conv)
                    smooth, weights = smooth_max_and_weights(conv, tau)

                    cur_val = 2.0 * n * conv.max()
                    if cur_val < best_val - 1e-12:
                        best_val = cur_val
                        best_seq = a.copy()
                        print(f"[fine tune] improved to {cur_val:.6f}")

                    grad_a = 2.0 * cross_correlation(weights, a,
                                                    fft_len=fft_len_corr)
                    inner = np.dot(a, grad_a)
                    grad_x = a * (grad_a - inner)

                    # Adam step
                    m = beta1 * m + (1.0 - beta1) * grad_x
                    v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
                    m_hat = m / (1.0 - beta1 ** (ft + 1))
                    v_hat = v / (1.0 - beta2 ** (ft + 1))
                    x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

            # ----- cheap greedy refinement (mass shuffling) -----
            if best_seq is not None and best_seq.shape[0] == n:
                a_local = best_seq.copy()
                local_iters = 3000
                for it in range(local_iters):
                    if time.time() > deadline:
                        break
                    i, j = np.random.choice(n, 2, replace=False)
                    # sample a fraction of a_local[i] to move
                    eps_move = min(a_local[i], 0.008) * (np.random.rand() * 0.6 + 0.2)
                    if eps_move <= 0:

                    a_new = a_local.copy()
                    a_new[i] -= eps_move
                    a_new[j] += eps_move
                    conv_new = full_convolution(a_new, fft_len=fft_len_conv)
                    val_new = 2.0 * n * conv_new.max()
                    if val_new < best_val - 1e-12:
                        best_val = val_new
                        best_seq = a_new.copy()
                        a_local = a_new
                        print(f"[local refine] improved to {val_new:.6f}")

    # -------------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform vector (objective = 2)
        n_fallback = 1024
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)

    print(f"\nFinal best value: {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
