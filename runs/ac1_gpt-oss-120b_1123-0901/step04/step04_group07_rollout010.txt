   max‑convolution.  This local search is fast because the max‑convolution
   is evaluated with an FFT.
6. Return the final best sequence as a plain Python list.

The whole search respects a 1000 s wall‑clock budget, adapts the number of
iterations to the sequence length and makes extensive use of NumPy’s FFT
routines for speed.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper routines for the smooth max and its (sub)‑gradient
# ----------------------------------------------------------------------
def _softmax_weights(conv: np.ndarray, tau: float):
    """Stable soft‑max of the convolution values.
    Returns weight vector w (summing to 1) and the smoothed maximum Lτ."""
    maxc = conv.max()
    if tau <= 0.0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w, maxc
    ex = np.exp((conv - maxc) / tau)
    sum_ex = ex.sum()
    w = ex / sum_ex
    Ltau = maxc + tau * math.log(sum_ex)
    return w, Ltau


def _weighted_rev(a: np.ndarray, w: np.ndarray, Nfft: int) -> np.ndarray:
    """
    Compute   rev_i = Σ_k w_k * a_{k‑i}   (indices outside range ignored)
    using FFTs.  This is the (sub‑)gradient of the (soft‑max) max‑convolution.
    """
    # FFT of a (reused for both forward and reverse convolutions)
    A_fft = np.fft.rfft(a, n=Nfft)
    # reverse the weight vector before FFT
    W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)
    conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
    conv_rev = conv_rev[: len(w) + a.shape[0] - 1]

    # The slice that corresponds to rev_i for i = 0 .. n‑1
    idx_start = 2 * a.shape[0] - 2
    rev = conv_rev[idx_start - np.arange(a.shape[0])]
    return rev


def _max_convolution(a: np.ndarray, Nfft: int = None) -> float:
    """Return max_k (a∗a)[k] using an FFT of length Nfft."""
    n = a.shape[0]
    if Nfft is None:
        Nfft = 1 << ((2 * n - 1).bit_length())
    A_fft = np.fft.rfft(a, n=Nfft)
    conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
    return float(conv.max())


# ----------------------------------------------------------------------
# Adam optimiser with a smooth–then–exact max‑convolution phase
# ----------------------------------------------------------------------
def _adam_optimize(a: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int,
                   best_seq: np.ndarray,
                   best_score: float) -> tuple[np.ndarray, float]:
    """
    Adam optimiser for the max‑convolution objective.
    Returns the best sequence seen in this run together with its exact
    objective value (which equals the max‑convolution after normalisation).
    """
    target_sum = math.sqrt(2.0 * n)

    # Normalise the initial vector exactly once
    a = a * (target_sum / a.sum())

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr0 = 0.06                      # a slightly larger start‑learning‑rate
    jitter_interval = 4000          # inject noise less frequently

    # Temperature schedule parameters
    tau_factor = 0.18               # initial τ = τ_factor·M
    decay_rate = max_iters / 12.0   # slower exponential decay

    Nfft = 1 << ((2 * n - 1).bit_length())   # next power‑of‑two ≥ 2·n‑1

    for iteration in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ------- exact convolution -------
        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        M = float(conv.max())

        # ------- temperature & soft‑max -------
        tau = tau_factor * M * math.exp(-iteration / decay_rate)
        if tau < 1e-9:
            # Exact sub‑gradient: active set of indices attaining the max
            mask = conv >= M - 1e-12
            cnt = mask.sum()
            w = mask.astype(float) / max(cnt, 1)
            Ltau = M
        else:
            w, Ltau = _softmax_weights(conv, tau)

        # ------- weighted reverse convolution (gradient of smoothed max) -------
        rev = _weighted_rev(a, w, Nfft)

        # ------- Adam gradient step -------
        # The objective after normalisation is simply M,
        # its gradient is 4·n·rev / target_sum²  – 4·n·Lτ·1/target_sum³
        coeff = 4.0 * n / (target_sum * target_sum)          # =2 after normalisation
        const = 4.0 * n * Ltau / (target_sum ** 3)
        grad = coeff * rev - const

        # Adam updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        lr = lr0 / (1.0 + 1e-4 * iteration)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # ------- projection onto feasible set -------
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a = a * (target_sum / s)
        else:
            a = np.full_like(a, target_sum / n)

        # occasional jitter to escape shallow plateaus
        if iteration % jitter_interval == 0:
            jitter = np.random.randn(n) * (1e-3 * target_sum / n)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        # ------- best‑ever update -------
        cur_score = M
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[len {n:5d} iter {iteration:8d}] new best = {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Rich set of initial vectors for a given length
# ----------------------------------------------------------------------
def _make_initial_vectors(n: int, target_sum: float) -> list[np.ndarray]:
    """
    Produce a diverse set of initial vectors for length n.
    All vectors are already normalised to have ℓ₁‑norm = target_sum.
    """
    rng = np.random.default_rng()
    init = []

    # 1) uniform
    init.append(np.full(n, target_sum / n))

    # 2) Dirichlet (Gamma(1,1))
    g = rng.gamma(1.0, 1.0, size=n)
    init.append(g / g.sum() * target_sum)

    # 3) exponential decay (random rate)
    rate = rng.uniform(0.0005, 0.01)
    x = np.exp(-rate * np.arange(n))
    init.append(x / x.sum() * target_sum)

    # 4) reversed exponential
    init.append(x[::-1] / x[::-1].sum() * target_sum)

    # 5) Hann window
    hann = 0.5 - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1))
    init.append(hann / hann.sum() * target_sum)

    # 6) Blackman window
    a0, a1, a2 = 0.42, 0.5, 0.08
    n_arr = np.arange(n)
    black = (a0
             - a1 * np.cos(2.0 * np.pi * n_arr / (n - 1))
             + a2 * np.cos(4.0 * np.pi * n_arr / (n - 1)))
    init.append(black / black.sum() * target_sum)

    # 7) small random perturbation of the uniform vector
    pert = np.full(n, target_sum / n)
    pert += rng.normal(scale=0.01 * target_sum / n, size=n)
    pert = np.maximum(pert, 0.0)
    init.append(pert / pert.sum() * target_sum)

    # 8) two‑block (mass in the first half)
    half = (n + 1) // 2
    block = np.zeros(n)
    block[:half] = target_sum / half
    init.append(block)

    # 9) spikes (few random locations receive equal mass)
    k = max(2, n // 200)            # roughly 0.5 % of the positions
    spikes = np.zeros(n)
    idx = rng.choice(n, size=k, replace=False)
    spikes[idx] = target_sum / k
    init.append(spikes)

    # 10) mirrored random
    rnd = rng.random(size=n)
    mir = (rnd + rnd[::-1]) / 2.0
    init.append(mir / mir.sum() * target_sum)

    # 11) Gaussian bump centred in the middle
    mu = (n - 1) / 2.0
    sigma = n / 5.0
    xv = np.arange(n)
    gauss = np.exp(-0.5 * ((xv - mu) / sigma) ** 2)
    init.append(gauss / gauss.sum() * target_sum)

    return init


# ----------------------------------------------------------------------
# Stochastic refinement (simple hill‑climbing on the exact objective)
# ----------------------------------------------------------------------
def _refine_sequence(seq: np.ndarray,
                     deadline: float,
                     max_moves: int = 60_000) -> tuple[np.ndarray, float]:
    """Try to improve the sequence by moving tiny amounts of mass between entries."""
    n = seq.shape[0]
    target_sum = seq.sum()
    best_seq = seq.copy()
    best_val = _max_convolution(best_seq)
    rng = np.random.default_rng()
    move_counter = 0

    while time.time() < deadline and move_counter < max_moves:
        # pick two distinct indices
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        if best_seq[i] <= 0.0:
            continue

        # propose moving a random fraction of a_i to a_j
        eps = best_seq[i] * rng.random() * 0.4   # up to 40 % of the source mass
        if eps <= 0.0:
            continue

        cand = best_seq.copy()
        cand[i] -= eps
        cand[j] += eps

        # keep exact ℓ₁‑norm (numerical drift is negligible)
        val = _max_convolution(cand)
        if val < best_val - 1e-9:
            best_seq = cand
            best_val = val
            # occasional progress report
            if move_counter % 5000 == 0:
                print(f"[refine] move {move_counter:6d} – new best = {best_val:.6f}")

        move_counter += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence found (as a plain Python list) before the
    time budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a safety margin for the harness (the call itself may add a few ms)
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Load the previously discovered construction if it exists.
    base = None
    try:
        # ``height_sequence_1`` is injected by the surrounding harness, if present.
        base = np.array(height_sequence_1, dtype=float)  # type: ignore
        print(f"[info] loaded base sequence of length {base.shape[0]}")
    except Exception:
        base = None

    # ------------------------------------------------------------------
    # Determine which lengths to explore.
    if base is not None:
        n0 = base.shape[0]
    else:
        n0 = 2100          # historic fallback
    length_factors = [1.0, 1.5, 2.0]    # keep the classic schedule
    lengths = sorted({int(n0 * f) for f in length_factors})
    lengths = [max(300, min(l, 8000)) for l in lengths]

    print(f"[info] exploring lengths: {lengths}")

    # ------------------------------------------------------------------
    # Global best tracking
    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Main optimisation loop over the selected lengths
    for n in lengths:
        if time.time() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)
        init_vectors = _make_initial_vectors(n, target_sum)

        # If we have a base vector of a different size, linearly interpolate it
        if base is not None and base.shape[0] != n:
            old_x = np.linspace(0.0, 1.0, base.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            interp = np.interp(new_x, old_x, base)
            interp = np.maximum(interp, 0.0)
            interp = interp / interp.sum() * target_sum
            init_vectors.append(interp)

        # Rough iteration budget: more iterations for smaller n (cheaper per step)
        max_iters = int(250_000 * 2100 / n)          # scale inversely with length
        max_iters = max(120_000, max_iters)          # enforce a floor
        print(f"[run] length={n:5d}, max_iters≈{max_iters}, target_sum={target_sum:.3f}")

        for idx, a0 in enumerate(init_vectors):
            if time.time() >= deadline:
                break
            print(f"  start {idx+1}/{len(init_vectors)} for length {n}")

            # Run Adam optimisation (smooth → exact max)
            best_seq, best_score = _adam_optimize(
                a=a0.copy(),
                n=n,
                deadline=deadline,
                max_iters=max_iters,
                best_seq=best_seq if best_seq is not None else a0,
                best_score=best_score,
            )

    # ------------------------------------------------------------------
    # Fine‑grained stochastic refinement on the overall best sequence
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print(f"[info] starting local refinement (len={best_seq.shape[0]})")
        # Give the refinement routine a generous slice of the remaining budget
        refine_deadline = deadline
        best_seq, best_score = _refine_sequence(best_seq, refine_deadline)

    # ------------------------------------------------------------------
    # Final fallback – if nothing was found, use a uniform vector.
    # ------------------------------------------------------------------
    if best_seq is None:
        n = n0
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n)
        best_score = _max_convolution(best_seq)
        print("[fallback] using uniform sequence")

    print(f"[done] best score = {best_score:.6f}, length = {best_seq.shape[0]}")
    # Return a plain Python list as required by the harness
    return best_seq.tolist()
