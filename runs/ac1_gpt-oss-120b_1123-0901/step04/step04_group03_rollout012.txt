   works directly on the simplex and refines the best Adam result.

3. **(1+1) Evolution Strategy (ES)** – a long fine‑tuning phase with an
   adaptive mutation strength explores the neighbourhood of the best
   solution found so far.

All stages respect a global time budget (default 1000 s).  The returned
value is a plain Python list ready for `evaluate_sequence`.  The
routine uses the previously discovered `height_sequence_1` when
available and falls back to a uniform vector if nothing else is found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities -------------------------------------------------------
# ----------------------------------------------------------------------
def softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(x)
    e = np.exp(x - m)
    return e / e.sum()


def fft_convolve(u: np.ndarray, v: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using FFT."""
    n = u.shape[0] + v.shape[0] - 1
    L = 1 << int(np.ceil(np.log2(n)))
    fu = np.fft.rfft(u, L)
    fv = np.fft.rfft(v, L)
    conv = np.fft.irfft(fu * fv, L)[:n]
    return conv.real


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a*a (length 2·n‑1) via FFT."""
    return fft_convolve(a, a)


def compute_objective(a: np.ndarray) -> float:
    """
    Exact objective used by evaluate_sequence.
    The vector a is assumed non‑negative; its sum is not required to be 1
    because the expression is homogeneous.
    """
    s = a.sum()
    if s < 1e-12:
        return np.inf
    n = a.shape[0]
    max_conv = full_convolution(a).max()
    return 2.0 * n * max_conv / (s * s)


# ----------------------------------------------------------------------
# Gradient of a smoothed max -------------------------------------------
# ----------------------------------------------------------------------
def grad_smoothed_max(a: np.ndarray, beta: float):
    """
    Returns (∇φβ(a), current max convolution value) where

        φβ(a) = (1/β)·log Σ_i exp(β·(a*a)_i)

    is a smooth approximation of max(a*a).  The gradient is computed
    analytically in O(n log n) using FFTs.
    """
    n = a.shape[0]
    conv = full_convolution(a)               # length 2n‑1
    max_c = conv.max()

    # soft‑max weights (numerically stable)
    shift = max_c
    w = np.exp(beta * (conv - shift))
    w /= w.sum()

    # gradient = 2·(w * a) restricted to the central n entries
    full = fft_convolve(w, a)                # length 3n‑2
    start = n - 1
    grad = 2.0 * full[start:start + n]       # shape (n,)

    return grad, max_c


# ----------------------------------------------------------------------
# Smoothed Adam optimiser -----------------------------------------------
# ----------------------------------------------------------------------
def run_smooth_adam_opt(
    n: int,
    init_a: np.ndarray,
    deadline: float,
    max_iters: int = 200_000,
    beta_start: float = 0.5,
    beta_max: float = 300.0,
    beta_factor: float = 1.001,
):
    """
    Adam on the smoothed objective.  The temperature β starts at
    *beta_start* and grows geometrically (factor *beta_factor*) until
    *beta_max* is reached.
    """
    # Initialise logits (mirrored gradient on the simplex)
    x = np.log(init_a + 1e-15)

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.07

    best_val = np.inf
    best_a = None
    t = 0
    last_improve = time.time()
    beta = beta_start

    while time.time() < deadline and t < max_iters:
        a = softmax(x)                     # a ≥ 0, Σa = 1
        grad_a, max_c = grad_smoothed_max(a, beta)
        val = 2.0 * n * max_c               # exact objective (sum(a)=1)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[SmoothAdam n={n} iter={t:05d}] new best: {val:.6f}")

        # back‑propagation through the softmax
        avg = np.dot(a, grad_a)
        grad_x = a * (grad_a - avg)

        # Adam update on logits
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

        # keep logits bounded
        x = np.clip(x, -50.0, 50.0)

        # temperature schedule
        beta = min(beta_max, beta * beta_factor)

        # occasional shake to get out of stagnation
        if time.time() - last_improve > 5.0:
            x += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()
            print(f"[SmoothAdam n={n} iter={t}] shake (stagnation)")

    # Fallback – uniform distribution (should never be needed)
    if best_a is None:
        best_a = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = compute_objective(best_a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Differential Evolution ------------------------------------------------
# ----------------------------------------------------------------------
def run_de_opt(
    n: int,
    init_pop: list,
    deadline: float,
    popsize: int = 25,
    max_gens: int = 30_000,
):
    """
    Classic DE on the simplex.  The population is kept normalised after
    crossover; individuals that improve the exact objective replace their
    parents.
    """
    pop = []
    for cand in init_pop:
        a = np.asarray(cand, dtype=np.float64)
        s = a.sum()
        if s > 0:
            pop.append(a / s)
        else:
            pop.append(np.full(n, 1.0 / n))
    while len(pop) < popsize:
        pop.append(np.random.dirichlet(np.ones(n)))

    obj_vals = [compute_objective(p) for p in pop]
    best_idx = int(np.argmin(obj_vals))
    best_a = pop[best_idx].copy()
    best_val = obj_vals[best_idx]

    F = 0.8
    CR = 0.9
    last_improve = time.time()
    gen = 0

    while time.time() < deadline and gen < max_gens:
        gen += 1
        for i in range(popsize):
            # pick three distinct donors
            idxs = list(range(popsize))
            idxs.remove(i)
            r1, r2, r3 = np.random.choice(idxs, 3, replace=False)
            mutant = pop[r1] + F * (pop[r2] - pop[r3])
            mutant = np.maximum(mutant, 0.0)

            # binomial crossover
            cross = np.random.rand(n) < CR
            cross[np.random.randint(n)] = True
            trial = np.where(cross, mutant, pop[i])

            # normalise
            s = trial.sum()
            if s == 0.0:
                trial = np.full(n, 1.0 / n)
            else:
                trial = trial / s

            val = compute_objective(trial)
            if val < obj_vals[i]:
                pop[i] = trial
                obj_vals[i] = val
                if val < best_val:
                    best_val = val
                    best_a = trial.copy()
                    last_improve = time.time()
                    print(f"[DE n={n} gen={gen:04d} i={i}] new best: {val:.6f}")

        # early stop if nothing improved for a while
        if time.time() - last_improve > 12.0:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# (1+1) Evolution Strategy ----------------------------------------------
# ----------------------------------------------------------------------
def run_es_opt(
    n: int,
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 200_000,
):
    """
    Simple (1+1) ES with 1/5th success rule for the mutation strength.
    Operates directly on the simplex.
    """
    a = init_a.copy()
    val = compute_objective(a)
    best_a, best_val = a.copy(), val
    sigma = 0.02
    last_improve = time.time()
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        trial = a + sigma * np.random.randn(n)
        trial = np.maximum(trial, 0.0)
        s = trial.sum()
        if s == 0.0:
            trial = np.full(n, 1.0 / n)
        else:
            trial = trial / s

        val_trial = compute_objective(trial)
        if val_trial < best_val:
            best_val, best_a = val_trial, trial.copy()
            a = trial
            sigma = min(sigma * 1.05, 0.25)
            last_improve = time.time()
            print(f"[ES n={n} it={it}] new best: {best_val:.6f}")
        else:
            sigma *= 0.99

        # reset if stuck
        if time.time() - last_improve > 6.0:
            a = best_a.copy()
            sigma = 0.02
            last_improve = time.time()

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point ------------------------------------------------------
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the supplied
    evaluation function.  The routine respects the given time budget
    (default 1000 s) and returns the best sequence found as a plain
    Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # a small safety margin

    best_seq = None
    best_val = np.inf

    # Fixed length – 1024 gives a good trade‑off between quality and speed
    n = 1024
    print(f"\n=== Optimising length n={n} ===")

    # ------------------------------------------------------------------
    # Build a small set of sensible initial candidates
    # ------------------------------------------------------------------
    init_candidates = []

    # warm‑start from the globally best sequence (if it matches the size)
    if "height_sequence_1" in globals():
        hs = np.asarray(height_sequence_1, dtype=np.float64)
        if hs.size >= n:
            hs = hs[:n]
        else:
            hs = np.pad(hs, (0, n - hs.size))
        hs = np.maximum(hs, 0.0)
        s = hs.sum()
        if s > 0.01:
            hs /= s
        else:
            hs = np.random.dirichlet(np.ones(n))
        init_candidates.append(hs)

    # uniform distribution
    init_candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

    # decreasing linear shape
    dec = np.arange(n, 0, -1, dtype=np.float64)
    dec /= dec.sum()
    init_candidates.append(dec)

    # a few random Dirichlet draws
    for _ in range(2):
        init_candidates.append(np.random.dirichlet(np.ones(n)))

    # ------------------------------------------------------------------
    # 1️⃣ Smoothed Adam phase (multiple restarts)
    # ------------------------------------------------------------------
    restarts = 6
    for r in range(restarts):
        if time.time() >= deadline:
            break
        a0 = init_candidates[r] if r < len(init_candidates) else np.random.dirichlet(np.ones(n))
        # allocate a time slice (≈15 % of the remaining budget)
        slice_time = min((deadline - time.time()) * 0.15, 40.0)
        sub_deadline = time.time() + slice_time
        a_opt, val_opt = run_smooth_adam_opt(
            n, a0, sub_deadline,
            beta_start=0.5, beta_max=400.0, beta_factor=1.001,
        )
        if val_opt < best_val:
            best_val, best_seq = val_opt, a_opt.copy()
            print(f"[Adam restart={r}] GLOBAL best improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣ Differential Evolution (global exploration)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        de_time = min((deadline - time.time()) * 0.35, 200.0)
        sub_deadline = time.time() + de_time
        # keep a small but diverse population
        de_pop = init_candidates[:4]
        a_de, val_de = run_de_opt(n, de_pop, sub_deadline,
                                  popsize=30, max_gens=30_000)
        if val_de < best_val:
            best_val, best_seq = val_de, a_de.copy()
            print(f"[DE] GLOBAL best now {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣ (1+1) Evolution Strategy (fine‑tuning)
    # ------------------------------------------------------------------
    if time.time() < deadline and best_seq is not None:
        es_time = min((deadline - time.time()) * 0.40, 300.0)
        sub_deadline = time.time() + es_time
        a_es, val_es = run_es_opt(n, best_seq, sub_deadline,
                                  max_iter=500_000)
        if val_es < best_val:
            best_val, best_seq = val_es, a_es.copy()
            print(f"[ES] GLOBAL best now {best_val:.6f}")

    # ------------------------------------------------------------------
    # Safety fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = compute_objective(best_seq)

    print(f"\nBest value discovered: {best_val:.6f} (length {len(best_seq)})")
    # Return a plain Python list as required by the external evaluator
    return best_seq.tolist()
