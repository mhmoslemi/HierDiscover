"""
Hybrid search for a non‑negative step sequence minimizing

    F(a) = 2·n·max(a*a) / (sum a)² .

The algorithm mixes three ideas:

1. **Annealed smooth max** –  instead of a sub‑gradient of the hard max,
   we use a softmax (log‑sum‑exp) with a temperature β that is slowly
   increased.  This provides smoother gradients early on and focuses on
   the true peak later.

2. **Adam on the simplex** –  we work in the unconstrained log‑it (log‑softmax)
   space and use Adam with momentum.  This mirrors the successful approach
   used previously but benefits from the smoother objective.

3. **Multi‑scale warm‑starts** –  after optimising a shorter length we linearly
   interpolate the best vector to the next larger length, providing a good
   initial shape for higher dimensions.  Random Dirichlet and a discrete
   Cantor‑set initialisation are also tried.

A simple local greedy refinement (mass transfer between two coordinates)
follows each optimisation run.  The whole search respects a 1000 s wall‑time
budget.  The best sequence found (as a plain Python list) is returned.
"""

import time
import numpy as np


def full_convolution(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) using FFT."""
    n = a.shape[0]
    if fft_len is None:
        fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


def cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) equal to
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i=0…n‑1.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def _cand_from_cantor(n: int) -> np.ndarray:
    """
    Produce a rough Cantor–set distribution on {0,…,n‑1}.
    The Cantor measure (self‑convolution close to uniform) is approximated
    by using points whose ternary digits are only 0 or 2.
    """
    # choose L such that 3**L ≈ n
    L = max(1, int(np.floor(np.log(n) / np.log(3))))
    K = 2 ** L                       # number of points
    max_idx = 3 ** L - 1
    indices = np.empty(K, dtype=int)
    for i in range(K):
        idx = 0
        # build a number with ternary digits 0/2 according to binary of i
        for k in range(L):
            bit = (i >> k) & 1
            idx += bit * 2 * (3 ** k)
        # scale to the interval [0, n‑1]
        indices[i] = int(round(idx * (n - 1) / max_idx))
    a = np.zeros(n, dtype=np.float64)
    a[indices] = 1.0 / K
    return a


def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs) -> list:
    """
    Time‑bounded optimisation returning the best non‑negative step sequence.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # safety margin

    best_seq = None           # best array found so far
    best_val = float("inf")   # its objective value

    # problem sizes – we start small and grow, reusing the best shape.
    lengths = [256, 512, 1024, 2048, 4096, 8192, 16384]

    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length n={n} ===")

        # FFT lengths that are reused many times
        fft_len_conv = 1 << int(np.ceil(np.log2(2 * n - 1)))
        fft_len_corr = 1 << int(np.ceil(np.log2(3 * n)))   # ≥ (2n‑1)+n‑1

        # number of independent restarts (more for smaller problems)
        if n <= 1024:
            restarts = 10
        elif n <= 4096:
            restarts = 6
        elif n <= 8192:
            restarts = 4
        else:
            restarts = 2

        for r in range(restarts):
            if time.time() > deadline:
                break

            # ---- initialise logits -------------------------------------------------
            if best_seq is not None:
                # warm‑start by interpolating the previously best vector
                if len(best_seq) == n:
                    init_a = best_seq.copy()
                else:
                    # linear interpolation to the new size
                    old_x = np.arange(len(best_seq), dtype=np.float64)
                    new_x = np.linspace(0, len(best_seq) - 1, n)
                    init_a = np.interp(new_x, old_x, best_seq)
                init_a = np.maximum(init_a, 0.0)
                if init_a.sum() < 0.01:
                    init_a = np.random.dirichlet(np.ones(n))
                else:
                    init_a /= init_a.sum()
                x = np.log(init_a + 1e-15)
                print(f"[n={n} restart={r}] warm start (interpolated) ")
            elif r == 0:
                # first restart – use a discrete Cantor set as a heuristic seed
                a0 = _cand_from_cantor(n)
                x = np.log(a0 + 1e-15)
                print(f"[n={n} restart={r}] Cantor‑set start")
            else:
                # pure random Dirichlet
                a0 = np.random.dirichlet(np.ones(n))
                x = np.log(a0 + 1e-15)
                print(f"[n={n} restart={r}] random Dirichlet start")

            # ---- Adam state ---------------------------------------------------------
            m = np.zeros(n, dtype=np.float64)
            v = np.zeros(n, dtype=np.float64)
            beta1, beta2, eps = 0.9, 0.999, 1e-8
            lr = 0.2                     # initial learning rate
            decay_every = 3000          # learning‑rate decay interval

            # schedule for the smooth‑max temperature β
            beta = 1.0
            beta_target = 150.0
            max_iters = 50000 if n <= 1024 else 35000 if n <= 4096 else 25000
            it = 0
            last_improve = time.time()

            while time.time() < deadline and it < max_iters:
                # logits → probabilities (soft‑max)
                max_x = np.max(x)
                exp_x = np.exp(x - max_x)          # numerical stability
                a = exp_x / exp_x.sum()            # point on the simplex

                # linear self‑convolution
                conv = full_convolution(a, fft_len=fft_len_conv)

                # exact objective (used for bookkeeping)
                max_c = conv.max()
                val = 2.0 * n * max_c
                if val < best_val - 1e-12:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} r={r} it={it:05d}] NEW BEST {val:.6f}")

                # ----- smooth max (log‑sum‑exp) --------------------------------------
                # w is the softmax weight over all lags
                beta_conv = beta * conv
                # numerical stability: subtract the max before exponentiating
                shift = np.max(beta_conv)
                w = np.exp(beta_conv - shift)
                w /= w.sum()                # now Σ w = 1

                # gradient of the smooth objective w.r.t. a
                grad_a = 2.0 * cross_correlation(w, a, fft_len=fft_len_corr)

                # back‑propagation through the soft‑max
                inner = np.dot(a, grad_a)
                grad_x = a * (grad_a - inner)

                # Adam update
                it += 1
                m = beta1 * m + (1.0 - beta1) * grad_x
                v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
                m_hat = m / (1.0 - beta1 ** it)
                v_hat = v / (1.0 - beta2 ** it)
                x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

                # learning‑rate decay
                if it % decay_every == 0:
                    lr *= 0.7

                # linear increase of β (smooth‑max becomes a true max)
                if beta < beta_target:
                    beta += (beta_target - 1.0) / max_iters

                # shake if we have been stagnating for a while
                if time.time() - last_improve > 30.0:
                    x += (np.random.rand(n) - 0.5) * 1e-2
                    last_improve = time.time()
                    print(f"[n={n} r={r} it={it}] shake (stagnation)")

                # occasional exact check (reduces pressure on the scheduler)
                if it % 5000 == 0:
                    conv_check = full_convolution(a, fft_len=fft_len_conv)
                    val_check = 2.0 * n * conv_check.max()
                    if val_check < best_val - 1e-12:
                        best_val = val_check
                        best_seq = a.copy()
                        print(f"[n={n} r={r} it={it}] exact check improvement {val_check:.6f}")

            # ----- cheap local greedy refinement ------------------------------------
            if best_seq is not None and len(best_seq) == n:
                a_local = best_seq.copy()
                local_iters = 4000
                for _ in range(local_iters):
                    if time.time() > deadline:
                        break
                    i, j = np.random.choice(n, 2, replace=False)
                    # move a small random amount from i to j
                    eps_move = min(a_local[i], 0.02) * (np.random.rand() * 0.5 + 0.5)
                    if eps_move <= 0:
                        continue
                    a_new = a_local.copy()
                    a_new[i] -= eps_move
                    a_new[j] += eps_move
                    conv_new = full_convolution(a_new, fft_len=fft_len_conv)
                    val_new = 2.0 * n * conv_new.max()
                    if val_new < best_val - 1e-12:
                        best_val = val_new
                        best_seq = a_new.copy()
                        a_local = a_new
                        print(f"[local refine] improved to {val_new:.6f}")

    # -------------------------------------------------------------------------
    if best_seq is None:
        # very unlikely fallback – uniform vector (objective = 2)
        n_fallback = 2000
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)

    print(f"\nFinal best value: {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
