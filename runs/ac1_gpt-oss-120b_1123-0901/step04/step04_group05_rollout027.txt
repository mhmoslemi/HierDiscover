"""
Hybrid search for a low‑max‑convolution step function.

The routine mixes three cheap but complementary ideas:

1. **Monte‑Carlo sampling** – a few thousand random Dirichlet / exponential /
   half‑sine initialisations and direct evaluation of the true objective.
2. **Smooth‑max Adam optimisation** – a continuation method that minimises a
   differentiable surrogate of   F(a)=2·n·max(a*a)/(∑a)².
   The surrogate uses a temperature‑like λ, which is slowly increased.
   Several restarts with different random seeds and vector lengths are performed.
3. **Sparse Sidon‑type support search** – we build a binary indicator of a small
   support set (≈√n … 2·√n indices) and try to keep the integer self‑convolution
   (the pair‑sum count) as flat as possible.  Uniform weights on the support are
   then evaluated; the best of those candidates is merged with the best found by
   the Adam stage.

The whole routine obeys a user supplied time budget (default 1000 s) and
returns a plain Python list of non‑negative heights.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based full linear convolution (same as np.convolve but faster for large n)
def _full_convolution(a: np.ndarray, size: int = None) -> np.ndarray:
    n = a.shape[0]
    if size is None:
        size = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return conv


# soft‑max over the convolution entries (smooth upper bound for the max)
def _softmax_weights(conv: np.ndarray, lam: float) -> np.ndarray:
    m = conv.max()
    w = np.exp(lam * (conv - m))
    s = w.sum()
    if s == 0.0:                     # safety – should never happen
        w.fill(1.0 / w.shape[0])
    else:
        w /= s
    return w


# gradient of the smooth‑max surrogate w.r.t. the probability vector a
def _softmax_gradient(a: np.ndarray, w: np.ndarray, size_corr: int = None) -> np.ndarray:
    n = a.shape[0]
    rev_a = a[::-1]
    if size_corr is None:
        size_corr = 1 << int(np.ceil(np.log2(len(w) + n - 1)))
    fw = np.fft.rfft(w, size_corr)
    frev = np.fft.rfft(rev_a, size_corr)
    corr = np.fft.irfft(fw * frev, size_corr)
    grad = 2.0 * corr[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
def _random_sequence(n: int, mode: int, rng: np.random.Generator) -> np.ndarray:
    """Return a non‑negative vector of length n with sum 1."""
    if mode == 0:                     # Dirichlet (almost uniform)
        a = rng.dirichlet(np.ones(n))
    elif mode == 1:                   # exponential decay
        r = rng.uniform(0.2, 0.9)
        a = r ** np.arange(n, dtype=np.float64)
        a /= a.sum()
    elif mode == 2:                   # half‑sine
        i = np.arange(1, n + 1, dtype=np.float64)
        a = np.sin(np.pi * i / (n + 1))
        a = np.maximum(a, 0.0)
        a /= a.sum()
    elif mode == 3:                   # powered sine
        p = rng.uniform(0.5, 2.0)
        a = np.sin(np.pi * np.arange(1, n + 1) / (n + 1)) ** p
        a = np.maximum(a, 0.0)
        a /= a.sum()
    elif mode == 4:                   # decaying with tiny jitter
        r = rng.uniform(0.3, 0.85)
        a = r ** np.arange(n, dtype=np.float64)
        a += 0.01 * rng.random(n)
        a = np.maximum(a, 0.0)
        a /= a.sum()
    elif mode == 5:                   # quadratic bump
        x = np.linspace(-1.0, 1.0, n)
        a = 1.0 - x ** 2
        a = np.maximum(a, 0.0)
        a /= a.sum()
    else:                             # bimodal half‑sine
        half = n // 2
        left = np.sin(np.pi * (np.arange(half) + 1) / (half + 1))
        right = np.sin(np.pi * (np.arange(n - half) + 1) / (n - half + 1))
        a = np.concatenate([left, right])
        a = np.maximum(a, 0.0)
        a /= a.sum()
    return a


# ----------------------------------------------------------------------
def _adam_optimize(start_a: np.ndarray,
                   max_iter: int,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max surrogate.
    Returns the best vector (sum = 1) encountered and its exact objective.
    """
    n = start_a.shape[0]
    # FFT sizes are fixed for the whole inner run
    conv_fft_size = 1 << int(np.ceil(np.log2(2 * n - 1)))
    size_corr = 1 << int(np.ceil(np.log2(3 * n)))   # safe bound for gradients

    # log‑softmax parametrisation
    u = np.log(start_a + 1e-30)

    m = np.zeros_like(u)
    v = np.zeros_like(u)

    lam = 0.2                # start rather smooth
    lam_factor = 1.018       # raise λ gradually – fast enough to focus soon
    base_lr = 0.06

    best_val = float("inf")
    best_a = start_a.copy()
    last_improve = time.time()

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1

        # probabilities (guaranteed sum = 1)
        u_shift = u - np.max(u)
        a = np.exp(u_shift)
        a /= a.sum()

        # convolution and smooth‑max weights
        conv = _full_convolution(a, size=conv_fft_size)
        w = _softmax_weights(conv, lam)

        # gradient wrt a
        grad_a = _softmax_gradient(a, w, size_corr=size_corr)

        # map to log‑space (soft‑max Jacobian)
        dot = np.dot(a, grad_a)
        grad_u = a * (grad_a - dot)

        # Adam update
        m = 0.9 * m + 0.1 * grad_u
        v = 0.999 * v + 0.001 * (grad_u ** 2)

        t = float(it)
        m_hat = m / (1.0 - 0.9 ** t)
        v_hat = v / (1.0 - 0.999 ** t)

        lr = base_lr * (0.9996 ** it)   # gentle decay
        u -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # increase λ (sharpen the surrogate)
        lam *= lam_factor

        # exact evaluation every few steps
        if it % 12 == 0:
            val = 2.0 * n * conv.max()     # exact objective because Σa = 1
            if val < best_val:
                best_val = val
                best_a = a.copy()
                last_improve = time.time()
                # small progress report – can be silenced if desired
                print(f"[Adam] iter {it:05d}  new best = {val:.6f}")

        # shake if stagnating
        if time.time() - last_improve > 1.8:
            u += (rng.random(n) - 0.5) * 1e-3
            m.fill(0.0)
            v.fill(0.0)
            last_improve = time.time()

    return best_a, best_val


# ----------------------------------------------------------------------
def _support_search(n: int,
                    k: int,
                    deadline: float,
                    rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Search for a binary support of size k with a low integer self‑convolution.
    Uniform weights (1/k) are used on the support.
    Returns the best vector (sum = 1) and its evaluated objective.
    """
    # start from a random k‑subset
    support = set(rng.choice(n, size=k, replace=False))
    indicator = np.zeros(n, dtype=np.int8)
    indicator[list(support)] = 1
    conv = np.convolve(indicator, indicator)
    maxcnt = conv.max()

    # cooling schedule for simulated‑annealing
    T0 = 2.0
    T_end = 0.02
    total_swaps = 0
    max_swaps = int(4e5)   # safety limit; will be cut by time budget

    start_time = time.time()
    while time.time() < deadline and total_swaps < max_swaps:
        total_swaps += 1
        # pick a random element to remove and a random element to insert
        out_idx = rng.choice(list(support))
        in_idx = rng.integers(n)
        while in_idx in support:
            in_idx = rng.integers(n)

        # trial update
        indicator[out_idx] = 0
        indicator[in_idx] = 1
        new_conv = np.convolve(indicator, indicator)
        new_max = new_conv.max()

        # acceptance test
        accept = False
        if new_max <= maxcnt:
            accept = True
        else:
            # temperature linearly decreases with the number of swaps
            T = T0 + (T_end - T0) * (total_swaps / max_swaps)
            prob = np.exp(-(new_max - maxcnt) / max(T, 1e-8))
            if rng.random() < prob:
                accept = True

        if accept:
            support.remove(out_idx)
            support.add(in_idx)
            conv = new_conv
            maxcnt = new_max
        else:
            # revert
            indicator[out_idx] = 1
            indicator[in_idx] = 0

        # occasional exact evaluation (cheap because we already have conv)
        if total_swaps % 3000 == 0:
            # uniform weight on the support → each entry = maxcnt / k**2
            val = 2.0 * n * (maxcnt / (k * k))
            # store the best seen so far
            if val < _support_search.best_val:
                _support_search.best_val = val
                _support_search.best_vec = indicator.astype(float) / k
                print(f"[Support] k={k} swaps={total_swaps}  new best = {val:.6f}")

    # return the best vector found (maybe the final one if never improved)
    if _support_search.best_vec is not None:
        return _support_search.best_vec, _support_search.best_val
    else:
        # fallback – uniform vector of length n
        a = np.full(n, 1.0 / n)
        return a, 2.0 * n * (1.0 / n)   # = 2


# initialise the static attributes used by the support search
_support_search.best_val = float("inf")
_support_search.best_vec = None

    """
    Search for a non‑negative step function minimizing ``evaluate_sequence``.
    The routine combines Monte‑Carlo sampling, Adam‑based smooth‑max optimisation
    and a dedicated sparse‑support (Sidon‑type) search.  It respects the supplied
    time budget and returns the best sequence found as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.15   # a tiny safety margin

    best_val = float("inf")
    best_seq = None

    # ------------------------------------------------------------------
    # Phase 1 – cheap random sampling
    lengths = [1200, 1800, 2500, 3200, 4000]
    for _ in range(2200):

            break
        n = rng.choice(lengths)
        mode = rng.integers(7)
        a = _random_sequence(n, mode, rng)
        val = evaluate_sequence(a.tolist())
        if val < best_val:
            best_val, best_seq = val, a.copy()
            print(f"[MonteCarlo] n={n} mode={mode}  new best = {val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Adam optimiser with multiple restarts
    restarts = 0
    while time.time() < deadline and restarts < 150:
        # pick a length (cycle through a modest list)
        n = lengths[restarts % len(lengths)]

        # initialise – sometimes from the current best (if length matches)
        if best_seq is not None and best_seq.shape[0] == n and rng.random() < 0.12:
            a0 = best_seq.copy()
        else:
            mode = restarts % 7
            a0 = _random_sequence(n, mode, rng)

        # run Adam optimiser
        a_opt, val_opt = _adam_optimize(a0, max_iter=12000,
                                        deadline=deadline, rng=rng)

        # store the best result
        if val_opt < best_val:
            best_val, best_seq = val_opt, a_opt.copy()
            print(f"[Adam restart {restarts:03d}] n={n}  new best = {val_opt:.6f}")

        restarts += 1

    # ------------------------------------------------------------------
    # Phase 3 – sparse‑support (Sidon‑type) search
    # try a few support sizes around sqrt(2n)
    for n in (2500, 3000, 3500):
        k_start = int(np.sqrt(2 * n))
        for k in range(k_start, k_start + 60, 12):
            if time.time() >= deadline:
                break
            vec, val = _support_search(n, k, deadline, rng)
            if val < best_val:
                best_val, best_seq = val, vec.copy()
                print(f"[Support] n={n} k={k}  new best = {val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform vector (should never be reached)
    if best_seq is None:
        n = 2500
        best_seq = np.full(n, 1.0 / n)

    print(f"=== Final best objective = {best_val:.6f} (length {len(best_seq)})")
    # return a plain list as required by the evaluation harness
    return best_seq.tolist()
