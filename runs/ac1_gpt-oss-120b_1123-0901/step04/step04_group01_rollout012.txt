def run_optimize_length(n, deadline, init_a=None):
    # Set up FFT lengths
    fft_len_conv = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fft_len_corr = 1 << int(np.ceil(np.log2(3 * n)))  # >= (2n-1)+n-1
    
    # Initialize logits x
    if init_a is not None:
        a0 = init_a
        a0 = np.maximum(a0, 0.0)
        if a0.sum() < 1e-6:
            a0 = np.random.dirichlet(np.ones(n))
        a0 = a0 / a0.sum()
        x = np.log(a0 + 1e-15)
    else:
        a0 = np.random.dirichlet(np.ones(n))
        x = np.log(a0 + 1e-15)
    
    # Adam parameters
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.15   # start
    # schedule for beta (smoothness)
    beta_temp = 0.5      # start temperature for smooth max (inverse of beta)
    # We will use continuation: increase beta (tightness) gradually over iterations
    beta_factor = 1.02
    
    t = 0
    max_iters = 200000   # high limit but break on time
    best_local_val = float('inf')
    best_local_a = None
    
    while time.time() < deadline and t < max_iters:
        # softmax to get a
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x / exp_x.sum()
        
        # compute convolution via FFT
        fa = np.fft.rfft(a, fft_len_conv)
        conv = np.fft.irfft(fa * fa, fft_len_conv)[:2*n-1]
        max_c = conv.max()
        # smooth max via log-sum-exp
        # compute weights w = exp(beta * conv) / sum exp...
        # Choose beta gradually: we define inv_temp = beta (higher = sharper)
        beta_val = beta_temp
        # compute w
        # To avoid overflow, subtract max
        scaled = beta_val * conv
        scaled_max = np.max(scaled)
        exp_scaled = np.exp(scaled - scaled_max)
        w = exp_scaled / exp_scaled.sum()
        
        # compute grad_a = 2 * cross correlation of w and a
        grad_a = 2.0 * cross_correlation(w, a, fft_len_corr)
        
        # backprop through softmax
        inner = np.dot(a, grad_a)
        grad_x = a * (grad_a - inner)
        
        # Adam update
        t += 1
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * (grad_x * grad_x)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # update beta (sharpen)
        beta_temp *= beta_factor
        # clamp beta to avoid overflow
        if beta_temp > 1000:
            beta_temp = 1000
        
        # learning rate decay schedule
        # alternative: we can decay after certain steps
        if t % 5000 == 0:
            lr *= 0.9
        
        # evaluate objective using exact max (not smooth)
        obj_val = 2.0 * n * max_c   # sum a =1
        if obj_val < best_local_val:
            best_local_val = obj_val
            best_local_a = a.copy()
    return best_local_a

Propose a step‑function (non‑negative real vector) that minimizes

    F(a) = 2·n·max(a∗a) / (sum a)^2

The optimiser works in three stages:

1️⃣  *Multiscale Adam* – start on a modest length (≈ 512).  The vector is
    parametrised by logits x (soft‑max → probabilities).  A smooth
    approximation of the max (log‑sum‑exp) is used, with a continuation
    schedule that gradually tightens the approximation (β grows from 0.5
    to ~8).  Adam with an exponential learning‑rate decay optimises the
    logits.

2️⃣  *Upsampling & refinement* – the best vector from the coarse level is
    interpolated to larger sizes (1024 → 2048 → 4096) and each size is
    refined again with the same Adam routine.  This gives a good
    “global shape’’ while keeping the dimensionality low in the early
    stages.

3️⃣  *Local simulated‑annealing* – starting from the best vector
    obtained above, small mass transfers between random coordinates are
    tried.  Moves that improve the exact objective are always accepted;
    otherwise they are accepted with probability exp(−Δ / T) where the
    temperature T cools linearly.  This phase usually squeezes a few
    extra digits out of the value found by gradient descent.

All FFT work uses powers of two for speed; the code respects the 1000 s
time budget and prints progress information.  The returned list is a
plain Python list of non‑negative floats suitable for the supplied
`evaluate_sequence` routine.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper routines --------------------------------------------------------
# ----------------------------------------------------------------------


def full_convolution(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """
    Linear convolution a*a (length 2·len(a)-1) using an FFT.
    """
    n = a.shape[0]
    if fft_len is None:
        fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


def cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) equal to
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def exact_objective(a: np.ndarray) -> float:
    """
    Compute the exact objective 2·n·max(a*a)  (a is already normalised
    so that sum(a) = 1).  This is the same quantity that
    `evaluate_sequence` returns up to the scaling by the sum.
    """
    n = a.shape[0]
    conv = full_convolution(a)
    return 2.0 * n * conv.max()


def upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from len(a) to new_n, renormalised to sum 1."""
    old_x = np.linspace(0.0, 1.0, a.shape[0])
    new_x = np.linspace(0.0, 1.0, new_n)
    a_new = np.interp(new_x, old_x, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-12:
        a_new = np.full(new_n, 1.0 / new_n)
    else:
        a_new /= s
    return a_new


# ----------------------------------------------------------------------
# Adam optimiser on a given length -------------------------------------
# ----------------------------------------------------------------------


def adam_optimize(
    n: int,
    deadline: float,
    init_a: np.ndarray | None = None,
    max_iters: int = 80_000,
) -> np.ndarray:
    """
    Run a time‑bounded Adam optimisation on a vector of length n.
    Returns the best probability vector (sum = 1) found before the deadline.
    """
    # FFT lengths (re‑used many times)
    fft_len_conv = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fft_len_corr = 1 << int(np.ceil(np.log2(3 * n)))  # ≥ (2n‑1)+n‑1

    # --- initialise logits -------------------------------------------------
    if init_a is not None:
        a0 = np.maximum(init_a, 0.0)
        if a0.sum() < 1e-12:
            a0 = np.random.dirichlet(np.ones(n))
        else:
            a0 = a0 / a0.sum()
    else:
        a0 = np.random.dirichlet(np.ones(n))

    x = np.log(a0 + 1e-15)               # logits, unrestricted

    # --- Adam state --------------------------------------------------------
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.15                            # initial learning rate

    # continuation schedule for the smooth‑max temperature
    beta_min = 0.5
    beta_max = 8.0

    best_val = float("inf")
    best_a = None

    t = 0
    while time.time() < deadline and t < max_iters:
        # ----- soft‑max -----------------------------------------------------
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x / exp_x.sum()          # probabilities, Σa = 1

        # ----- exact convolution (only for monitoring) --------------------
        conv = full_convolution(a, fft_len=fft_len_conv)
        max_c = conv.max()
        exact_val = 2.0 * n * max_c
        if exact_val < best_val - 1e-12:
            best_val = exact_val
            best_a = a.copy()
            # small print – keep it short
            print(f"[n={n} iter={t:05d}] new best {exact_val:.6f}")

        # ----- smooth max (log‑sum‑exp) ------------------------------------
        # continuation: β grows linearly from beta_min to beta_max
        beta = beta_min + (beta_max - beta_min) * (t / max_iters)

        # w_i = exp(β·conv_i) / Σ exp(β·conv)
        scaled = beta * conv
        scaled_max = np.max(scaled)
        exp_scaled = np.exp(scaled - scaled_max)
        w = exp_scaled / exp_scaled.sum()

        # ----- gradient w.r.t. a -------------------------------------------
        grad_a = 2.0 * cross_correlation(w, a, fft_len=fft_len_corr)

        # ----- back‑prop through soft‑max -----------------------------------
        inner = np.dot(a, grad_a)
        grad_x = a * (grad_a - inner)

        # ----- Adam update -------------------------------------------------
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # learning‑rate decay (exponential)
        lr_t = lr * (0.95 ** (t // 4000))
        x -= lr_t * m_hat / (np.sqrt(v_hat) + eps)

        # occasional shake if we have been stagnant for a while
        if t % 10000 == 0 and time.time() < deadline:
            # add a tiny zero‑mean perturbation to the logits
            pert = (np.random.rand(n) - 0.5) * 1e-2
            x += pert

    # return the best vector found (or the last if none)
    if best_a is None:
        # fallback to uniform
        best_a = np.full(n, 1.0 / n, dtype=np.float64)
    return best_a


# ----------------------------------------------------------------------
# Simulated‑annealing local refinement ----------------------------------
# ----------------------------------------------------------------------


def annealing_refine(a: np.ndarray, deadline: float) -> np.ndarray:
    """
    Small stochastic hill‑climb that moves tiny amounts of probability
    between random coordinates.  Accepts worsening moves with a
    temperature that cools linearly.
    """
    n = a.shape[0]
    fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
    cur_a = a.copy()
    cur_val = exact_objective(cur_a)

    start_temp = cur_val * 0.1   # start with a modest temperature
    end_temp = cur_val * 1e-4   # final temperature
    total_iters = 5_000
    it = 0

    while time.time() < deadline and it < total_iters:
        it += 1
        # linear cooling schedule
        T = start_temp + (end_temp - start_temp) * (it / total_iters)

        i, j = np.random.randint(n, size=2)
        if i == j or cur_a[i] < 1e-14:
            continue

        # move a random fraction (≈ 10–30 %) of a[i] to a[j]
        eps_move = cur_a[i] * (0.1 + 0.2 * np.random.rand())
        a_new = cur_a.copy()
        a_new[i] -= eps_move
        a_new[j] += eps_move

        # compute exact objective for the trial
        conv = full_convolution(a_new, fft_len=fft_len)
        new_val = 2.0 * n * conv.max()

        delta = new_val - cur_val
        if delta < -1e-12 or np.random.rand() < np.exp(-delta / T):
            cur_a = a_new
            cur_val = new_val
            # occasional log
            if it % 1000 == 0:
                print(f"[anneal iter={it}] val={cur_val:.6f}")

    return cur_a


# ----------------------------------------------------------------------
# Main entry point ------------------------------------------------------
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs) -> list:
    """
    Search for a non‑negative step‑function that minimises the evaluation
    function.  The routine spends most of its time on a multiscale
    Adam optimisation (512 → 1024 → 2048 → 4096) and then finishes with a
    short simulated‑annealing local search.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # 1. Starting point – use the global `height_sequence_1` if present,
    #    otherwise a random Dirichlet vector.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        base = np.array(height_sequence_1, dtype=np.float64)
        base = np.maximum(base, 0.0)
        if base.sum() < 1e-12:
            base = np.random.dirichlet(np.ones(512))
        else:
            base = base / base.sum()
    else:
        base = np.random.dirichlet(np.ones(512))

    # ------------------------------------------------------------------
    # 2. Multiscale optimisation
    # ------------------------------------------------------------------
    lengths = [512, 1024, 2048, 4096]   # keep within the time budget
    best_seq = None
    best_val = float("inf")

    for n in lengths:
        if time.time() > deadline:
            break
        # Interpolate the current best (or the global base) to the new size
        if best_seq is None:
            a0 = upsample(base, n)
        else:
            a0 = upsample(best_seq, n)

        print(f"\n=== Optimising length n={n} ===")
        a_opt = adam_optimize(n, deadline, init_a=a0, max_iters=80_000 if n <= 1024 else 40_000)

        val_opt = exact_objective(a_opt)          # exact (sum = 1)
        print(f"Finished length {n}, exact value = {val_opt:.6f}")

        if val_opt < best_val:
            best_val = val_opt
            best_seq = a_opt.copy()

        # tiny local greedy “pair‑move’’ refinement (optional but cheap)
        # – 1000 cheap moves
        for _ in range(500):
            if time.time() > deadline:
                break
            i, j = np.random.randint(n, size=2)
            eps_move = min(best_seq[i], 0.01) * (0.5 + 0.5 * np.random.rand())
            cand = best_seq.copy()
            cand[i] -= eps_move
            cand[j] += eps_move
            cand_val = exact_objective(cand)
            if cand_val < best_val - 1e-12:
                best_seq, best_val = cand, cand_val

    # ------------------------------------------------------------------
    # 3. Final annealing refinement on the best length found
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n=== Simulated‑annealing refinement ===")
        refined = annealing_refine(best_seq, deadline)
        refined_val = exact_objective(refined)
        print(f"Refinement value = {refined_val:.6f}")
        if refined_val < best_val:
            best_seq, best_val = refined, refined_val

    # ------------------------------------------------------------------
    # 4. Return the best sequence (as plain Python list)
    # ------------------------------------------------------------------
    if best_seq is None:  # extremely unlikely fallback
        n_fallback = 1024
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)

    print(f"\nFinal best value (exact): {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
