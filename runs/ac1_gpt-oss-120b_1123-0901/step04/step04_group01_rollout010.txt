"""
Propose a step‑function (non‑negative vector) that minimizes

    F(a) = 2·n·max(a*a) / (sum(a)²)

The implementation combines three ideas:

1. **Smoothed max gradient** – Instead of a hard active‑set sub‑gradient,
   a softmax over the convolution values (with a temperature that
   decays during the run) yields a smooth approximation of the max.
   The gradient of this smooth objective is 2·(w ★ a) where w are the
   softmax weights.  The gradient is back‑propagated through a softmax
   parametrisation (logits) and optimised with Adam.

2. **Multi‑restart, multi‑size search** – Several problem sizes are tried
   (1024, 2048, 4096, 8192).  For each size a number of restarts is
   performed, using either the previous best vector (cropped / padded),
   the historic `height_sequence_1` vector, or a fresh Dirichlet sample.

3. **Local simulated‑annealing refinement** – After gradient descent,
   a cheap pair‑wise mass‑transfer Metropolis step is applied for a few
   thousand iterations to escape possible neighbourhood minima.

The total wall‑clock time is bounded by the supplied ``budget_s`` (default
1000 s).  The best sequence found (any length) is returned as a plain
Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – all top‑level, no closures.
# ----------------------------------------------------------------------


def full_convolution(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """
    Linear convolution a*a (length 2·len(a)-1) using the FFT.
    """
    n = a.shape[0]
    if fft_len is None:
        fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


def cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) equal to
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (returns a probability vector)."""
    mx = np.max(x)
    e = np.exp(x - mx)
    return e / e.sum()


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:

    Search for a non‑negative step‑function that minimises the evaluation
    function.  The search runs for at most ``budget_s`` seconds and returns
    the best sequence (as a plain Python list) found.

    # Leave a small safety margin for the interpreter shutdown.
    deadline = time.time() + budget_s - 0.5

    best_seq = None          # best sequence (numpy array)
    best_val = float("inf")  # best objective value seen

    # harder to optimise.  We start with 1024 (fast) and then try bigger
    # sizes if time permits.
    lengths = [1024, 2048, 4096, 8192]

    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length n={n} ===")

        # FFT lengths – keep them powers of two for speed.
        fft_len_conv = 1 << int(np.ceil(np.log2(2 * n - 1)))
        fft_len_corr = 1 << int(np.ceil(np.log2(3 * n)))  # ≥ (2n‑1)+n‑1

        # Number of independent restarts – more restarts for the cheaper
        # problem sizes.
        if n <= 2048:

        elif n <= 4096:
            restarts = 3
        else:

        for r in range(restarts):
            if time.time() > deadline:

                x = np.log(init_a + 1e-15)
                print(f"[n={n} restart={r}] warm start from previous best")

                x = np.log(init_a + 1e-15)

                tau = max(tau_min, tau0 * np.exp(-t / (max_iters / 3.0)))

            # --------------------------------------------------------------

                best_seq = a_local.copy()
                print(f"[n={n} restart={r}] post‑gd improvement {val_local:.6f}")

            # Parameters for the annealing phase – scaled with current value.
            anneal_iters = 20000 if n <= 2048 else 12000
            T_start = 0.02 * val_local + 1e-12
            for it in range(anneal_iters):
                if time.time() > deadline:
                    break
                # propose moving a small chunk from i to j
                i, j = np.random.randint(n, size=2)
                if a_local[i] < 1e-12:
                    continue
                delta = min(a_local[i], np.random.rand() * 0.3 * a_local[i])
                a_new = a_local.copy()
                a_new[i] -= delta
                a_new[j] += delta

                conv_new = full_convolution(a_new, fft_len=fft_len_conv)
                val_new = 2.0 * n * conv_new.max()

                # Accept if better, otherwise Metropolis criterion
                if val_new < val_local:
                    a_local, conv_local, val_local = a_new, conv_new, val_new
                    if val_new < best_val - 1e-12:
                        best_val = val_new
                        best_seq = a_new.copy()
                        print(f"[anneal n={n}] improved to {val_new:.6f}")
                else:
                    # linear temperature schedule
                    T = T_start * (1.0 - it / anneal_iters)
                    if np.random.rand() < np.exp(-(val_new - val_local) / (T + 1e-12)):
                        a_local, conv_local, val_local = a_new, conv_new, val_new

    # ----------------------------------------------------------------------
    # Fallback in the impossible case that nothing was found.
    # ----------------------------------------------------------------------
    if best_seq is None:
        n_fallback = 1024
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        best_val = 2.0
        print("Fallback to uniform distribution (objective = 2).")

    print(f"\nFinal best value: {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
