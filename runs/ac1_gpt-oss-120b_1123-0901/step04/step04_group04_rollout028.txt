def project_simplex(v):
    # ensures non-negative and sum=1
    # v can be any real
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

def project_simplex(v):
    # v: 1D np.ndarray
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    # Find rho where u_i > cssv_i / i
    cond = u - cssv / ind > 0
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def lp_step(f, M, target_sum, t=0.2):
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    # create convolution constraints
    # For each k: sum_i f_i * g_{k-i} <= M
    # Use Toeplitz matrix
    constraints = []
    for k in range(2*n-1):
        # compute sum_i f_i * g_{k-i}
        # Build a list of indices i where 0 <= k-i < n => i in [max(0, k-(n-1)), min(k, n-1)]
        i_start = max(0, k-(n-1))
        i_end = min(k, n-1)
        # expression = sum_{i=i_start}^{i_end} f_i * g_{k-i}
        expr = cp.sum(cp.multiply(f[i_start:i_end+1], g[k - np.arange(i_start, i_end+1)]))
        constraints.append(expr <= M)
    # Also bound g_i <= 1000 perhaps
    constraints.append(g <= 1000.0)
    
    obj = cp.Maximize(cp.sum(g))
    prob = cp.Problem(obj, constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, verbose=False, max_iters=5000)
    if g.value is None:
        return f  # fallback
    g0 = np.array(g.value, dtype=float)
    # rescale to target sum
    sum_g0 = g0.sum()
    if sum_g0 <= 0:
        return f
    g_scaled = g0 * (target_sum / sum_g0)
    # update
    f_new = (1 - t) * f + t * g_scaled
    return f_new

    """Projection of v onto the probability simplex {x >= 0, sum x = 1}."""
    n = v.shape[0]

Search for a step‑function (non‑negative vector) that minimises
``evaluate_sequence``.

Algorithm
---------
1.  Load the best known construction (``height_sequence_1``) if it exists,
    otherwise start from a uniform vector.
2.  Normalise the vector so that Σa = √(2 n) (the scaling makes the
    objective equal to the maximum convolution coefficient).
3.  Run several short L‑BFGS‑B optimisations on a *smooth* maximum
    (τ·log∑exp(conv/τ)).  The optimiser works on the log‑softmax
    parameters, guaranteeing non‑negativity and Σp = 1.
    A slower temperature schedule and a few random restarts are used
    to explore the landscape more thoroughly than the baseline code.
4.  Take the best result and polish it with a *projected sub‑gradient*
    descent on the *exact* objective  maxₖ(p∗p)ₖ.
    The gradient at the current worst convolution index is
    ``grad_i = 2·p_{k_max−i}`` (zero outside the overlap).
    After each step we project back onto the probability simplex.
    A simple back‑tracking line‑search guarantees monotone decrease,
    and occasional random jitter helps to escape plateaus.
5.  Return the best sequence found (scaled back to the required L1‑norm).

The whole routine respects a user‑provided time budget (default 1000 s)
and prints progress information.
"""

# Helper: projection onto the probability simplex {x ≥ 0, Σx = 1}
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1.0
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:               # all entries are non‑positive
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# Convolution utilities (FFT based)
def _conv_full(p: np.ndarray, Nfft: int) -> np.ndarray:
    """Full linear convolution p∗p (real FFT, length 2·n‑1)."""
    P = np.fft.rfft(p, n=Nfft)
    c = np.fft.irfft(P * P, n=Nfft)
    return c[: 2 * p.shape[0] - 1]


# ----------------------------------------------------------------------
# Smooth‑max utilities (used by the L‑BFGS stage)
def _softmax_weights(conv: np.ndarray, tau: float):
    """Stable soft‑max of convolution values.
    Returns the weight vector w (summing to 1) and the smoothed max Lτ."""
    maxc = conv.max()
    if tau <= 0.0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w, maxc
    ex = np.exp((conv - maxc) / tau)
    sum_ex = ex.sum()
    w = ex / sum_ex
    Ltau = maxc + tau * math.log(sum_ex)
    return w, Ltau


def _weighted_rev(a: np.ndarray, w: np.ndarray, Nfft: int) -> np.ndarray:
    """
    Compute rev_i = Σ_k w_k a_{k‑i}  (used in the gradient of Lτ).
    Implemented with FFT for speed.
    """
    A_fft = np.fft.rfft(a, n=Nfft)
    W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)
    conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
    conv_rev = conv_rev[: len(w) + a.shape[0] - 1]
    idx_start = 2 * a.shape[0] - 2
    rev = conv_rev[idx_start - np.arange(a.shape[0])]
    return rev


def _objective_and_grad(y: np.ndarray, tau: float, Nfft: int):
    """
    Smooth‑max objective Lτ and its gradient w.r.t. the log‑softmax variables y.
    """
    y_max = np.max(y)
    exp_y = np.exp(y - y_max)               # numerical stability
    p = exp_y / exp_y.sum()                 # soft‑max → probability vector

    conv = _conv_full(p, Nfft)               # p∗p
    w, Ltau = _softmax_weights(conv, tau)    # smoothed maximum

    rev = _weighted_rev(p, w, Nfft)          # gradient wrt p
    grad_p = 2.0 * rev

    dot = np.dot(p, grad_p)
    grad_y = p * (grad_p - dot)              # chain rule (soft‑max Jacobian)
    return Ltau, grad_y

# L‑BFGS optimisation (smooth‑max)
def _run_l_bfgs(y0: np.ndarray, init_tau: float,
                target_sum: float, Nfft: int,
                deadline: float,
                best_seq: np.ndarray,
                best_score: float) -> (np.ndarray, float):
    """
    Repeatedly minimise the smooth maximum with decreasing temperature.
    Returns the best (sequence, score) discovered.
    """
    if not _SCIPY_AVAILABLE:                # pragma: no cover
        return best_seq, best_score

    y = y0.copy()
    tau = max(init_tau, 1e-12)
    tau_decay = 0.70                         # a bit slower cooling than baseline
    min_tau = 1e-12

    while time.time() < deadline and tau > min_tau:
        # ----- L‑BFGS on the current temperature -----------------------
        def fun(y_vec):
            L, g = _objective_and_grad(y_vec, tau, Nfft)
            return L, g

        options = {'maxfun': 500, 'maxiter': 500, 'ftol': 1e-12, 'gtol': 1e-12}
        res = minimize(fun, y, method='L-BFGS-B', jac=True, options=options)

        y = res.x

        # ----- exact objective (no smoothing) -------------------------
        y_max = np.max(y)
        exp_y = np.exp(y - y_max)
        p = exp_y / exp_y.sum()
        conv = _conv_full(p, Nfft)
        max_conv = conv.max()
        cur_score = (target_sum ** 2) * max_conv   # == evaluate_sequence

        if cur_score < best_score:
            best_score = cur_score
            best_seq = (p * target_sum).copy()
            print(f"[improve L‑BFGS] τ={tau:.2e}  score={best_score:.6f}")

        # ----- schedule ------------------------------------------------
        tau *= tau_decay

        # occasional jitter to leave plateaus
        if np.random.rand() < 0.2:
            y += np.random.randn(y.shape[0]) * 0.01

        if time.time() >= deadline:
            break

    return best_seq, best_score

# Projected sub‑gradient optimisation on the exact max‑convolution

    # normalise to the required L1‑norm
    base = base * (target_sum / base.sum())
    p0 = base / target_sum                 # probability vector (Σp = 1)

    # initialise best record
    best_seq = base.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # FFT size (next power‑of‑two ≥ 2·n‑1)
    Nfft = 1 << ((2 * n - 1).bit_length())

    # --------------------------------------------------------------
    # ----- 1) L‑BFGS phase (smooth max) ----------------------------
    starts = []

    # a) known solution
    starts.append(np.log(p0 + 1e-12))

    # b) uniform
    starts.append(np.zeros(n, dtype=float))

    # c) random Dirichlet (exponential)
    rand_exp = np.random.exponential(scale=1.0, size=n)
    starts.append(np.log(rand_exp + 1e-12))

    # d) exponentially decreasing shape
    exp_shape = np.exp(-np.arange(n) / (n / 5.0))
    starts.append(np.log(exp_shape + 1e-12))

    # e) symmetric tent shape
    tent = np.concatenate([np.arange(1, n // 2 + 1),
                           np.arange(n // 2, 0, -1)])[:n]
    starts.append(np.log(tent + 1e-12))

    # f) a slightly perturbed version of the known solution
    starts.append(np.log(p0 + 1e-12) + np.random.randn(n) * 0.04)

    for idx_start, y_start in enumerate(starts):
        if time.time() >= deadline:
            break
        print(f"[run L‑BFGS] start {idx_start + 1}/{len(starts)}")
        # initial temperature roughly twice the current smooth max
        p_tmp = np.exp(y_start - np.max(y_start))
        p_tmp /= p_tmp.sum()
        conv0 = _conv_full(p_tmp, Nfft)
        init_tau = max(conv0.max() * 2.0, 1e-12)

        best_seq, best_score = _run_l_bfgs(
            y0=y_start,
            init_tau=init_tau,
            target_sum=target_sum,
            Nfft=Nfft,
            deadline=deadline,
            best_seq=best_seq,
            best_score=best_score,
        )

    # --------------------------------------------------------------
    # ----- 2) Projected sub‑gradient polishing ----------------------
    # Use the best probability vector obtained so far
    p_best = best_seq / target_sum
    best_seq, best_score = _subgradient_opt(
        p_start=p_best,
        Nfft=Nfft,
        deadline=deadline,
        best_seq=best_seq,
        best_score=best_score,
        target_sum=target_sum,
    )

    # --------------------------------------------------------------
    # Final sanity check (exact evaluation)
    final_score = evaluate_sequence(best_seq.tolist())
    if final_score < best_score:
        best_score = final_score
        best_seq = best_seq.copy()
    print(f"[done] best score = {best_score:.6f}, length = {best_seq.shape[0]}")
    return best_seq.tolist()
