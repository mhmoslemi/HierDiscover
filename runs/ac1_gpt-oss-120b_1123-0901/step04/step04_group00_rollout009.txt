        exp_logits = np.exp(logits - mx)
        a = exp_logits / exp_logits.sum()
        # conv and smooth max
        conv = _full_convolution(a, fft_len_conv)
        max_c = conv.max()
        # compute smooth max weights

        w_sum = w.sum()

            w = np.full_like(conv, 1.0/conv.size)
        else:
            w /= w_sum
        # gradient of smooth max
        grad_a = 2.0 * _cross_correlation(w, a, fft_len_corr)
        # backprop through softmax
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)
        # Adam update
        m = beta1*m + (1-beta1)*grad_log
        v = beta2*v + (1-beta2)*(grad_log**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)
        # update lambda
        lam = min(lam * lam_factor, 1e6)
        # evaluate exact objective
        val = 2.0 * n * max_c / (a.sum()**2)
        if val < best_val:
            best_val=val
            best_a=a.copy()
            last_improve = time.time()
            # optionally print
        # stagnation check
        if time.time() - last_improve > 3.0:
            # random shake
            logits += (np.random.rand(n)-0.5) * 1e-2
            last_improve = time.time()
    return best_a, best_val, a

"""
Algorithm
---------
1.   Work on a set of lengths (default 512, 1024, 2048, 4096,
     8192, 16384).  The longest lengths have the biggest
     potential to lower the bound but are also the most costly.
2.   For each length generate a small, diverse pool of *seed*
     sequences:
         • warm‑start from the current best (or an interpolated
           version of it);
         • random Dirichlet draws;
         • power‑law edge‑concentrated families;
         • two‑parameter “Beta‑like’’ families a_i ∝ (i+1)^α·(n‑i)^β;
         • symmetric double‑Gaussian tails.
3.   Evaluate every seed once (full convolution) and keep only the
     best few (default 5) for further optimisation.
4.   Optimise each selected seed in two phases:
         a) Adam on a smooth‑max surrogate of the max‑convolution.
            The surrogate uses a soft‑max weighting whose temperature
            (λ) grows geometrically, driving the surrogate toward the
            true max.  The iterate is kept on the probability simplex
            via the soft‑max (logits) parametrisation.
         b) A short exponentiated‑gradient sub‑gradient stage on the
            exact max (the active set of maximal convolution entries
            yields the sub‑gradient).
     After each exact evaluation we update the global best
     sequence and its value.
5.   All loops respect a global deadline (budget_s seconds).  If
     time runs out the best sequence seen so far is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.shape[0] - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Compute the linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) equal to
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact evaluation of 2·n·max(a*a) / (sum(a)²)."""
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_c / (s * s)


# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   fft_len_conv: int,
                   fft_len_corr: int,
                   deadline: float,
                   best_global_val: float) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on a smooth‑max surrogate.
    Returns the best sequence seen during this run and its exact value.
    """
    a = a0.copy()
    logits = np.log(a + 1e-15)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.03
    lam = 0.5
    lam_factor = 1.02

    # iteration budget scales roughly like O(1/n)
    max_iters = int(12000 * 1024 // n) + 2000
    best_a = a.copy()
    best_val = float("inf")
    last_improve = time.time()

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # ------ simplex via soft‑max ------
        mx = logits.max()
        exp_logits = np.exp(logits - mx)
        a = exp_logits / exp_logits.sum()

        # ------ smooth‑max of the convolution ------
        conv = _full_convolution(a, fft_len_conv)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or not np.isfinite(w_sum):
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w /= w_sum

        # ------ gradient of the surrogate ------
        grad_a = 2.0 * _cross_correlation(w, a, fft_len_corr)

        # back‑prop through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ------ Adam update ------
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # increase temperature
        lam = min(lam * lam_factor, 1e6)

        # exact objective
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            if val < best_global_val:
                print(f"[n={n} Adam] new global best {val:.6f}")

        # stagnation => tiny random shake
        if time.time() - last_improve > 3.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

    # return the best sequence we have seen and its exact value
    return best_a, best_val


# ----------------------------------------------------------------------
def _subgradient_refine(a0: np.ndarray,
                        n: int,
                        fft_len_conv: int,
                        fft_len_corr: int,
                        deadline: float,
                        best_global_val: float) -> tuple[np.ndarray, float]:
    """
    A few exponentiated‑gradient sub‑gradient steps on the exact max.
    Returns the best sequence observed and its exact value.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = float("inf")
    last_improve = time.time()
    eta = 0.6

    for it in range(2000):
        if time.time() > deadline:
            break

        conv = _full_convolution(a, fft_len_conv)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)

        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, fft_len_corr)

        # exponentiated‑gradient step (keeps a ≥ 0)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        # exact objective
        conv = _full_convolution(a, fft_len_conv)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            if val < best_global_val:
                print(f"[n={n} Subgrad] new global best {val:.6f}")

        if time.time() - last_improve > 3.0:
            break

    return best_a, best_val

def _generate_candidates(n: int, best_seq: np.ndarray | None) -> list[np.ndarray]:
    """
    Produce a modest pool of diverse starting points for length ``n``.
    """
    candidates = []

    # warm‑start from exact same length
    if best_seq is not None and len(best_seq) == n:
        candidates.append(np.array(best_seq, dtype=np.float64))

    # interpolate the current best to the new size
    if best_seq is not None:
        x_old = np.linspace(0.0, 1.0, len(best_seq))
        x_new = np.linspace(0.0, 1.0, n)
        cand = np.interp(x_new, x_old, best_seq)
        cand = np.maximum(cand, 0.0)
        if cand.sum() < 1e-12:
            cand = np.random.dirichlet(np.ones(n))
        else:
            cand = cand / cand.sum()
        candidates.append(cand)

    # pure random Dirichlet
    candidates.append(np.random.dirichlet(np.ones(n)))

    # edge‑concentrated power‑law ((i+1)*(n‑i))**α
    idx = np.arange(n, dtype=np.float64) + 1.0
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.5, -0.2, 9):
        shape = np.power(prod, alpha, where=prod > 0, out=np.zeros_like(prod))
        if shape.sum() > 0:
            shape = shape / shape.sum()
            candidates.append(shape)

    # two‑parameter beta‑like family (i+1)^α·(n‑i)^β
    alphas = np.linspace(-2.5, -0.2, 6)
    betas = np.linspace(-2.5, -0.2, 6)
    for a_pow in alphas:
        for b_pow in betas:
            left = np.power(idx, a_pow, where=idx > 0, out=np.zeros_like(idx))
            right = np.power(n - idx + 1.0, b_pow, where=(n - idx + 1.0) > 0,
                             out=np.ones_like(idx))
            shape = left * right
            if shape.sum() > 0:
                shape = shape / shape.sum()
                candidates.append(shape)

    # symmetric double‑Gaussian tails
    for gamma in [0.2, 0.5, 1.0, 2.0, 5.0]:
        i = np.arange(n, dtype=np.float64)
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        shape = shape / shape.sum()
        candidates.append(shape)

    return candidates

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function sequence that minimises

        F(a) = 2·n·max(a*a) / (sum(a)²)

    under a_i ≥ 0.  The routine respects the supplied time budget
    (default 1000 s) and returns a plain Python list of floats.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Length schedule – start small, finish large.
    lengths = [512, 1024, 2048, 4096, 8192, 16384]

    # Pre‑compute FFT lengths for all considered n
    conv_fft = {}
    corr_fft = {}
    for n in lengths:
        conv_fft[n] = _next_pow2(2 * n - 1)
        corr_fft[n] = _next_pow2(2 * n - 1 + n - 1)   # 3·n‑2 at most

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n--- Optimising length n={n} ---")
        fft_len_conv = conv_fft[n]
        fft_len_corr = corr_fft[n]

        # generate a pool of seeds
        seeds = _generate_candidates(n, best_seq)

        # quick evaluation of every seed (single conv) and keep the best few
        scored = []
        for cand in seeds:
            val = _objective(cand, n, fft_len_conv)
            scored.append((val, cand))
        scored.sort(key=lambda x: x[0])
        top_k = 5 if n <= 2048 else 3
        selected = [c for _, c in scored[:top_k]]

        # now run the two‑phase optimiser on the selected seeds
        for cid, a0 in enumerate(selected):
            if time.time() > deadline:
                break

            # ---- Phase A : Adam on smooth‑max ----
            a_adam, val_adam = _adam_optimize(
                a0, n, fft_len_conv, fft_len_corr, deadline, best_val
            )
            if val_adam < best_val:
                best_val = val_adam
                best_seq = a_adam.copy()
                print(f"[n={n} cand={cid} Adam] NEW BEST {best_val:.6f}")

            # ---- Phase B : exact sub‑gradient refinement ----
            a_sub, val_sub = _subgradient_refine(
                a_adam, n, fft_len_conv, fft_len_corr, deadline, best_val
            )
            if val_sub < best_val:
                best_val = val_sub
                best_seq = a_sub.copy()
                print(f"[n={n} cand={cid} Subgrad] NEW BEST {best_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform if nothing was found (should never happen)
    if best_seq is None:
        fallback_n = 2048
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)
        print("[fallback] using uniform sequence")

    # Return a plain Python list as required by the external evaluator
    return best_seq.tolist()
