def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5

    best_seq = None
    best_obj = float('inf')
    # range of lengths to try
    lengths = [1024, 2048, 4096, 8192]  # maybe include more

    for n in lengths:
        if time.time() > deadline: break
        print(f"Optimising length {n}")

        fft_len_conv = 1 << int(np.ceil(np.log2(2*n-1)))
        fft_len_corr = 1 << int(np.ceil(np.log2(3*n)))  # for cross-correlation

        # number of restarts based on n
        restarts = 5 if n <= 4096 else 3

        for r in range(restarts):
            if time.time() > deadline: break
            # init a
            if best_seq is not None and len(best_seq) == n:
                a = best_seq.copy()
                # maybe add small noise
                a += np.random.rand(n) * 1e-3
                a = np.clip(a, 0, None)
                if a.sum() < 1e-12:
                    a = np.random.dirichlet(np.ones(n))
                else:
                    a /= a.sum()
                print(f"n={n} restart {r} warm start from best")
            elif 'height_sequence_1' in globals():
                cand = np.array(height_sequence_1, dtype=float)
                if cand.shape[0] >= n:
                    a = cand[:n].copy()
                else:
                    a = np.pad(cand, (0, n - cand.shape[0]))
                a = np.maximum(a, 0)
                if a.sum() < 0.01:
                    a = np.random.dirichlet(np.ones(n))
                else:
                    a /= a.sum()
                print(f"n={n} restart {r} start from height_sequence_1")
            else:
                a = np.random.dirichlet(np.ones(n))
                print(f"n={n} restart {r} random")

            # optimization loop
            # schedule: initial step size, adapt via Adam
            m = np.zeros(n)
            v = np.zeros(n)
            beta1, beta2, eps = 0.9, 0.999, 1e-8
            lr = 0.5
            iter_max = 40000 if n <= 4096 else 25000
            t = 0
            lam = 1.0  # smoothing parameter

            while time.time() < deadline and t < iter_max:
                # compute convolution via FFT
                conv = full_convolution(a, fft_len=fft_len_conv)
                
                # smoothed max weight
                max_c = conv.max()
                # gradually increase lam: lam = min(50, 1 + t/2000)
                lam = min(50.0, 1.0 + t / 2000.0)
                weights = np.exp(lam * (conv - max_c))  # subtract max for stability

                # gradient of smoothed max w.r.t a

                # compute exact objective (not smoothed) for best record
                conv_exact = full_convolution(a, fft_len=fft_len_conv)
                val = 2.0 * n * conv_exact.max()
                if val < best_obj - 1e-12:
                    best_obj = val
                    best_seq = a.copy()
                    print(f"n={n} restart {r} iter={t} new best {val:.6f}")
                # occasionally shuffle if stagnation
                if t % 5000 == 0:

```
def proj_simplex(v):
    # Projects onto the simplex {x >=0, sum x = 1}
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a candidate step‑function (non‑negative vector) that minimises

    F(a) = 2·n·max(a*a) / (sum(a)²)

The algorithm proceeds in three stages for several problem sizes.

1.  **Smoothed gradient descent** –  a is kept on the probability simplex
    (sum = 1, a_i ≥ 0).  The non‑smooth max is approximated by a
    log‑sum‑exp with a temperature λ that is slowly increased.  The
    gradient of the smoothed objective is obtained via an FFT‑based
    cross‑correlation, and an Adam update is applied.  After each step the
    vector is projected back onto the simplex.

2.  **Local greedy refinement** –  after the gradient phase we repeatedly
    move a small amount of mass between two random coordinates and keep
    the change if it improves the exact (non‑smoothed) objective.

3.  **Simulated annealing** –  a final stochastic pass that accepts
    worsening moves with a decreasing temperature, giving a chance to
    escape shallow local minima.

The procedure is run for a range of lengths (1024 … 8192).  Warm‑starts
are performed from the best solution found so far and from the
`height_sequence_1` variable if it exists.  The best vector found within
the allotted wall‑time (≈1000 s) is returned as a plain Python list.
"""

import time
import numpy as np


def full_convolution(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """
    Linear convolution a * a using the FFT.
    Returns an array of length 2*len(a)-1.
    """
    n = a.shape[0]
    if fft_len is None:
        fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


def cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) equal to
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i=0…n‑1.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        { x | x_i ≥ 0, Σ x_i = 1 }.
    Implements the algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs) -> list:
    """
    Search for a non‑negative step‑function minimising the evaluation
    function.  The method combines smoothed gradient descent, greedy
    refinement and a short simulated‑annealing pass.  The best vector
    found before the time limit expires is returned as a Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    best_seq = None          # best vector (numpy array)
    best_obj = float("inf")

    # Problem sizes to explore – larger n gives a lower theoretical bound.
    lengths = [1024, 2048, 4096, 8192]

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")

        # Pre‑compute FFT lengths that will be reused
        fft_len_conv = 1 << int(np.ceil(np.log2(2 * n - 1)))
        fft_len_corr = 1 << int(np.ceil(np.log2(3 * n)))   # ≥ (2n‑1)+n‑1

        # Number of independent restarts for this length
        restarts = 5 if n <= 4096 else 3

        for r in range(restarts):
            if time.time() > deadline:
                break

            # ----- initialise a -------------------------------------------------
            if best_seq is not None and len(best_seq) == n:
                a = best_seq.copy()
                a += (np.random.rand(n) - 0.5) * 1e-3   # tiny jitter
                a = proj_simplex(a)
                print(f"[n={n} restart={r}] warm start from previous best")
            elif 'height_sequence_1' in globals():
                cand = np.array(height_sequence_1, dtype=np.float64)
                if cand.shape[0] >= n:
                    a = cand[:n].copy()
                else:
                    a = np.pad(cand, (0, n - cand.shape[0]))
                a = np.maximum(a, 0.0)
                if a.sum() < 1e-8:
                    a = np.random.dirichlet(np.ones(n))
                else:
                    a = a / a.sum()
                print(f"[n={n} restart={r}] start from height_sequence_1")
            else:
                a = np.random.dirichlet(np.ones(n))
                print(f"[n={n} restart={r}] random start")

            # ----- Adam state ---------------------------------------------------
            m = np.zeros(n, dtype=np.float64)
            v = np.zeros(n, dtype=np.float64)
            beta1, beta2, eps = 0.9, 0.999, 1e-8
            lr = 0.5                 # initial learning rate
            decay_every = 5000       # learning‑rate decay interval
            max_iters = 30000 if n <= 4096 else 20000
            t = 0
            last_improve = time.time()

            # ----- smoothed gradient descent ------------------------------------
            while time.time() < deadline and t < max_iters:
                # linear self‑convolution
                conv = full_convolution(a, fft_len=fft_len_conv)

                # smoothed max via log‑sum‑exp (temperature λ)
                max_c = conv.max()
                # increase λ slowly – 1 → 30 over the run
                lam = 1.0 + min(29.0, t / 1000.0)

                # stable softmax over the convolution values
                exp_weights = np.exp(lam * (conv - max_c))
                w = exp_weights / exp_weights.sum()   # shape (2n‑1,)

                # gradient of the smoothed max w.r.t. a
                grad_a = 2.0 * cross_correlation(w, a, fft_len=fft_len_corr)

                # (optional) multiply by constant 2·n – omitted because it does
                # not affect the Adam direction (just scales steps)
                grad = grad_a

                # Adam update
                t += 1
                m = beta1 * m + (1.0 - beta1) * grad
                v = beta2 * v + (1.0 - beta2) * (grad * grad)
                m_hat = m / (1.0 - beta1 ** t)
                v_hat = v / (1.0 - beta2 ** t)
                step = lr * m_hat / (np.sqrt(v_hat) + eps)

                a_new = a - step
                a = proj_simplex(a_new)   # keep a on the simplex

                # learning‑rate decay
                if t % decay_every == 0:
                    lr *= 0.7

                # evaluate exact objective for bookkeeping
                conv_exact = full_convolution(a, fft_len=fft_len_conv)
                val = 2.0 * n * conv_exact.max()
                if val < best_obj - 1e-12:
                    best_obj = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} r={r} iter={t:05d}] new best {val:.6f}")

                # shake if stagnation lasts > 30 s
                if time.time() - last_improve > 30.0:
                    a += (np.random.rand(n) - 0.5) * 1e-2
                    a = proj_simplex(a)
                    last_improve = time.time()
                    print(f"[n={n} r={r} iter={t}] shake (stagnation)")

            # ----- greedy local refinement --------------------------------------
            if best_seq is not None and len(best_seq) == n:
                a_local = best_seq.copy()
                local_iters = 3000
                for it in range(local_iters):
                    if time.time() > deadline:
                        break
                    i, j = np.random.choice(n, 2, replace=False)
                    # move a modest amount from i to j
                    eps_move = min(a_local[i], 0.01) * (np.random.rand() * 0.5 + 0.5)
                    if eps_move <= 0:
                        continue
                    a_new = a_local.copy()
                    a_new[i] -= eps_move
                    a_new[j] += eps_move
                    conv_new = full_convolution(a_new, fft_len=fft_len_conv)
                    val_new = 2.0 * n * conv_new.max()
                    if val_new < best_obj - 1e-12:
                        best_obj = val_new
                        best_seq = a_new.copy()
                        a_local = a_new
                        print(f"[local refine n={n}] improved to {val_new:.6f}")

            # ----- short simulated‑annealing pass --------------------------------
            if best_seq is not None and len(best_seq) == n:
                a_anneal = best_seq.copy()
                T0 = 1e-2
                Tf = 1e-5
                ann_iters = 5000
                start_time = time.time()
                for k in range(ann_iters):
                    if time.time() > deadline:
                        break
                    # linear cooling
                    T = T0 * (Tf / T0) ** (k / ann_iters)
                    i, j = np.random.choice(n, 2, replace=False)
                    # propose a random transfer
                    delta = min(a_anneal[i], 0.02) * (np.random.rand() - 0.5)
                    if delta == 0:
                        continue
                    a_new = a_anneal.copy()
                    a_new[i] -= delta
                    a_new[j] += delta
                    # keep feasibility
                    if a_new[i] < 0 or a_new[j] < 0:
                        continue
                    conv_new = full_convolution(a_new, fft_len=fft_len_conv)
                    val_new = 2.0 * n * conv_new.max()
                    cur_val = 2.0 * n * full_convolution(a_anneal, fft_len=fft_len_conv).max()
                    # Metropolis acceptance
                    if val_new < cur_val or np.random.rand() < np.exp(-(val_new - cur_val) / T):
                        a_anneal = a_new
                        if val_new < best_obj - 1e-12:
                            best_obj = val_new
                            best_seq = a_new.copy()
                            print(f"[anneal n={n}] new best {val_new:.6f}")
                # keep annealed solution as candidate for next lengths
                if best_seq is not None and len(best_seq) == n:
                    best_seq = a_anneal.copy()

    # -------------------------------------------------------------------------
    if best_seq is None:
        # Fallback: uniform distribution of length 2000
        n_fallback = 2000
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        best_obj = 2.0  # uniform objective

    print(f"\nFinal best value: {best_obj:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
