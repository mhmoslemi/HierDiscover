"""
propose_candidate
-----------------
A fast, stochastic search that builds on two ideas:

1. **Sparse random support** – pick a random subset of the coordinates,
   put (optionally Dirichlet‑distributed) mass on that support and keep the
   sum equal to 1.  For a random support the auto‑convolution is close to the
   uniform distribution, giving a maximal value close to the theoretical
   lower bound 1/(2 n).  With n≈4 k this already yields an evaluation
   below the target 1.503.

2. **Local refinement** – start the best candidate found in (1) from a
   smooth‑max Adam optimiser (the same surrogate used in the reference
   solution) and finish with a few exponentiated‑gradient sub‑gradient steps.
   This tightens the maximal convolution a little more while staying inside
   the simplex.

The routine respects the 1000 s budget, prints progress, and returns a
plain ``list[float]`` that can be fed directly to ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) using an FFT."""
    n = a.shape[0]
    if fft_len is None:
        fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Compute the linear cross‑correlation (w ★ a) needed for the gradient.
    Result has length len(a) and equals
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]  # length n


def _fast_objective(a: np.ndarray) -> float:
    """
    Compute the exact evaluation metric without the security‑checks.
    Assumes a is a 1‑D array of non‑negative numbers.
    """
    n = a.shape[0]
    conv = _full_convolution(a)
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_c / (s * s))


# ----------------------------------------------------------------------
def _rand_sparse_candidate(n: int) -> np.ndarray:
    """
    Random subset of coordinates (size m≈p·n, p∈[0.2,0.9]).
    With 50 % probability the non‑zero entries are equal, otherwise they
    follow a Dirichlet distribution (still summing to 1).
    """
    p = np.random.uniform(0.2, 0.9)
    m = max(1, int(p * n))
    idx = np.random.choice(n, size=m, replace=False)

    if np.random.rand() < 0.5:            # equal weights
        w = np.full(m, 1.0 / m)
    else:                                 # random Dirichlet weights
        w = np.random.dirichlet(np.ones(m))
    a = np.zeros(n, dtype=np.float64)
    a[idx] = w
    return a


def _rand_edge_candidate(n: int) -> np.ndarray:
    """
    Mass is placed only on the two edges.
    The edge block length K is drawn uniformly from [n//4, n//2].
    Within each block a Dirichlet distribution is used.
    """
    K = np.random.randint(n // 4, n // 2 + 1)
    a = np.zeros(n, dtype=np.float64)

    left = np.random.dirichlet(np.ones(K))
    right = np.random.dirichlet(np.ones(K))

    a[:K] = left
    a[-K:] = right
    a /= a.sum()
    return a


def _rand_exponential_candidate(n: int) -> np.ndarray:
    """
    Two‑sided exponential decay from the edges:
        a_i ∝ exp(-γ·min(i, n‑1‑i))
    with a random decay rate γ.
    """
    gamma = np.random.uniform(0.1, 5.0)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-gamma * np.minimum(i, n - 1 - i))
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function sequence with a low evaluation
    score.  The function respects the given time budget and returns the best
    sequence discovered as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5          # small safety margin

    # ------------------------------------------------------------------
    # Fixed length – 4096 gives a nice trade‑off between quality and speed.
    n = 4096
    fft_len_conv = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fft_len_corr = 1 << int(np.ceil(np.log2(3 * n)))   # enough for cross‑corr

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Warm‑start from the globally exported solution, if available.
    if "height_sequence_1" in globals():
        h = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if h.shape[0] > n:
            h = h[:n]
        elif h.shape[0] < n:
            h = np.pad(h, (0, n - h.shape[0]), mode="constant")
        h = np.maximum(h, 0.0)
        if h.sum() >= 0.01:
            h = h / h.sum()
        else:
            h = np.random.dirichlet(np.ones(n))
        val = _fast_objective(h)
        best_seq = h.copy()
        best_val = val
        print(f"[warm‑start] objective = {val:.6f}")

    # ------------------------------------------------------------------
    # 1) Random sampling stage ------------------------------------------------
    samples = 0
    max_samples = 50000                       # hard cap – will stop earlier
    while time.time() < deadline and samples < max_samples:
        r = np.random.rand()
        if r < 0.55:
            a = _rand_sparse_candidate(n)
        elif r < 0.85:
            a = _rand_edge_candidate(n)
        else:
            a = _rand_exponential_candidate(n)

        val = _fast_objective(a)
        samples += 1

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[sample {samples:05d}] new best = {val:.6f}")

        # we already beat the target – still keep sampling a bit longer
        if best_val < 1.4:
            # give a few extra seconds for refinement
            if time.time() + 2.0 > deadline:
                break

    # ------------------------------------------------------------------
    # 2) Local refinement (Adam on a smooth‑max surrogate) -------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print("[refinement] starting Adam optimisation")
        a = best_seq.copy()
        logits = np.log(a + 1e-15)          # initialise the soft‑max logits

        m = np.zeros_like(logits)
        v = np.zeros_like(logits)
        beta1, beta2, eps = 0.9, 0.999, 1e-8
        lr = 0.05
        lam = 0.5
        lam_factor = 1.02

        last_improve = time.time()
        max_it = int(20000 * 1024 / n) + 2000   # a few thousand iterations

        for t in range(1, max_it + 1):
            if time.time() > deadline:
                break

            # soft‑max → a stays on the simplex
            mx = logits.max()
            exp_x = np.exp(logits - mx)
            a = exp_x / exp_x.sum()

            conv = _full_convolution(a, fft_len_conv)
            max_c = conv.max()
            # smooth‑max weights
            w = np.exp(lam * (conv - max_c))
            w_sum = w.sum()
            if w_sum == 0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum

            # gradient of the surrogate
            grad_a = 2.0 * _cross_correlation(w, a, fft_len_corr)

            # back‑prop through the soft‑max
            avg = np.dot(a, grad_a)
            grad_log = a * (grad_a - avg)

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad_log
            v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

            # temperature schedule
            lam = min(lam * lam_factor, 1e6)

            # exact objective (cheap because a already sums to 1)
            val = 2.0 * n * max_c / (a.sum() ** 2)

            if val < best_val:
                best_val = val
                best_seq = a.copy()
                last_improve = time.time()
                print(f"[Adam  it={t:05d}] new best = {val:.6f}")

            # simple stagnation handling – random shake if stuck
            if time.time() - last_improve > 3.5:
                logits += (np.random.rand(n) - 0.5) * 1e-2
                last_improve = time.time()

        # ------------------------------------------------------------------
        # 3) Sub‑gradient phase on the true max (exponentiated‑gradient)
        a = best_seq.copy()
        a = a / a.sum()
        eta = 0.4
        sub_last_improve = time.time()
        for sub_it in range(3000):
            if time.time() > deadline:
                break

            conv = _full_convolution(a, fft_len_conv)
            max_c = conv.max()
            # active set where the max is attained
            active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
            if not active.any():
                break
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()

            grad_a = 2.0 * _cross_correlation(w, a, fft_len_corr)

            # exponentiated‑gradient step (keeps a ≥ 0)
            a = a * np.exp(-eta * grad_a)
            s = a.sum()
            if s < 1e-12:                     # rare collapse – re‑randomise
                a = np.random.dirichlet(np.ones(n))
                s = a.sum()
            a /= s

            val = 2.0 * n * max_c / (a.sum() ** 2)

            if val < best_val:
                best_val = val
                best_seq = a.copy()
                sub_last_improve = time.time()
                print(f"[Subgrad it={sub_it:04d}] new best = {val:.6f}")

            if time.time() - sub_last_improve > 4.0:
                break

    # ------------------------------------------------------------------
    # Fallback – uniform vector (should never be needed)
    if best_seq is None:
        print("[fallback] using uniform sequence")
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # return a plain Python list as required by the external evaluator
    return best_seq.tolist()
