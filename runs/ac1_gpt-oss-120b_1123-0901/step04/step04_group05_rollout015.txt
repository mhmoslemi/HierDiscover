"""
Propose a step‑function sequence that minimizes ``evaluate_sequence``.

The optimizer works in three stages:

1. **Initialisation** – if a previously discovered sequence
   ``height_sequence_1`` is available it is used (normalised to sum = 1);
   otherwise a random Dirichlet vector of length 2500 is created.

2. **Gradient‑based fine‑tuning** – the problem
   ``min 2·n·max(conv(a))`` (with ``conv`` the linear self‑convolution)
   is tackled directly on the simplex.
   The vector is parametrised by logits ``u`` (soft‑max) so that
   positivity and the Σa = 1 constraint are always satisfied.
   A sub‑gradient of the objective is obtained from the index where the
   convolution attains its maximum.  Adam updates are applied to ``u``.
   The learning‑rate decays gently, and a small random “shake’’ is
   performed when progress stalls.

3. **Local random walk** – after the Adam run, a cheap random‑swap
   hill‑climber performs a few thousand pairwise mass transfers;
   any improvement is kept.

The whole routine respects the user supplied ``budget_s`` (default
1000 s).  The best sequence found is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, size: int = None) -> np.ndarray:
    """
    Linear self‑convolution a*a using a real FFT.
    Returns a vector of length 2·len(a) − 1.
    """
    n = a.shape[0]
    if size is None:
        size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return conv


def _max_conv_gradient(a: np.ndarray, conv: np.ndarray) -> tuple[np.ndarray, float, int]:
    """
    Return a sub‑gradient of  max(conv)  w.r.t. ``a`` (the gradient of the
    convolution entry that attains the maximum), together with the max
    value and the index where it occurs.
    """
    k = int(np.argmax(conv))                # index of the peak
    n = a.shape[0]
    idx = k - np.arange(n)                  # candidate source indices
    valid = (idx >= 0) & (idx < n)          # keep only those inside [0,n)
    grad = np.zeros_like(a)
    grad[valid] = 2.0 * a[idx[valid]]       # d conv[k] / d a = 2·a_{k‑j}
    return grad, float(conv[k]), k


def _softmax(u: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (produces a probability vector)."""
    shift = u - np.max(u)
    e = np.exp(shift)
    return e / e.sum()


def _projected_adam(u: np.ndarray,
                    a: np.ndarray,
                    grad_u: np.ndarray,
                    m: np.ndarray,
                    v: np.ndarray,
                    it: int,
                    base_lr: float) -> tuple[np.ndarray, np.ndarray, np.ndarray]:
    """
    Perform one Adam update on the logits ``u``.
    """
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = beta1 * m + (1 - beta1) * grad_u
    v = beta2 * v + (1 - beta2) * (grad_u ** 2)

    it_f = float(it)
    m_hat = m / (1.0 - beta1 ** it_f)
    v_hat = v / (1.0 - beta2 ** it_f)

    lr = base_lr * (0.9995 ** it)          # slow exponential decay
    u = u - lr * m_hat / (np.sqrt(v_hat) + eps)
    return u, m, v

def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The function respects the time budget and returns the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.15   # tiny safety margin

    # ------------------------------------------------------------------
    # 1) initialise ``a`` (probability vector, sum = 1)
    if 'height_sequence_1' in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64)
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        if s == 0.0:

        else:
            a0 = a0 / s
    else:
        # fall back to a Dirichlet draw of moderate dimension
        n0 = 2500
        a0 = np.random.dirichlet(np.ones(n0))

    n = a0.shape[0]
    conv_fft_size = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # 2) Adam on logits, using the true max‑convolution sub‑gradient
    u = np.log(a0 + 1e-30)          # logits
    m = np.zeros_like(u)
    v = np.zeros_like(u)

    best_a = a0.copy()
    best_val = 2.0 * n * np.max(_full_convolution(best_a, conv_fft_size))
    print(f"Initial objective = {best_val:.6f}")

    base_lr = 0.12                 # a bit larger than the previous run
    it = 0

    max_iters = 200_000            # hard cap; real stop is time‑budget

    while time.time() < deadline and it < max_iters:
        it += 1

        # ----- probability vector from logits -----
        a = _softmax(u)                     # guaranteed a_i ≥ 0, Σa = 1

        # ----- convolution and max‑gradient -----
        conv = _full_convolution(a, conv_fft_size)
        grad_a_raw, conv_max, kmax = _max_conv_gradient(a, conv)

        grad_a = grad_a_raw * (2.0 * n)      # ∇(2·n·max) = 2·n·∇max

        if it % 10 == 0:
