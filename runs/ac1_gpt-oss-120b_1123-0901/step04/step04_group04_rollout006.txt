``evaluate_sequence``.

Algorithm
---------
* Work with the probability vector p = a / Σa ( Σp = 1 ).
  The objective equals 2·n·max(p∗p); the scale of a is irrelevant.

* The whole search respects a wall‑time budget (default 1000 s) and

def _softmax_weights(conv: np.ndarray, tau: float):
    """Stable soft‑max of the convolution values.
    Returns the weight vector (summing to 1) and the smooth max Lτ."""
    maxc = conv.max()
    if tau <= 0.0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w, maxc
    ex = np.exp((conv - maxc) / tau)
    sum_ex = ex.sum()
    w = ex / sum_ex
    Ltau = maxc + tau * math.log(sum_ex)
    return w, Ltau


def _weighted_rev(a: np.ndarray, w: np.ndarray, Nfft: int):
    """Compute rev_i = Σ_k w_k a_{k‑i} using FFTs."""
    A_fft = np.fft.rfft(a, n=Nfft)
    W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)
    conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
    conv_rev = conv_rev[: len(w) + a.shape[0] - 1]        # keep the valid part
    idx_start = 2 * a.shape[0] - 2
    rev = conv_rev[idx_start - np.arange(a.shape[0])]
    return rev


def _conv_full(p: np.ndarray, Nfft: int):
    """Full convolution p∗p (real FFT based). Length = 2·len(p)‑1."""
    A_fft = np.fft.rfft(p, n=Nfft)
    conv = np.fft.irfft(A_fft * A_fft, n=Nfft)
    return conv[: 2 * p.shape[0] - 1]


def _objective_and_grad(y: np.ndarray, tau: float, Nfft: int):
    """Smooth max Lτ and its gradient w.r.t. y (log‑softmax variables)."""
    y_max = np.max(y)
    exp_y = np.exp(y - y_max)               # numeric stability
    p = exp_y / exp_y.sum()

    conv = _conv_full(p, Nfft)               # p∗p
    w, Ltau = _softmax_weights(conv, tau)    # smooth max

    # gradient of Lτ w.r.t. p (2·rev)
    rev = _weighted_rev(p, w, Nfft)
    grad_p = 2.0 * rev

    # convert to gradient w.r.t. y (soft‑max Jacobian)
    dot = np.dot(p, grad_p)                  # scalar Σ p_i·grad_p_i
    grad_y = p * (grad_p - dot)
    return Ltau, grad_y


def _run_l_bfgs(y0: np.ndarray,
                init_tau: float,
                target_sum: float,
                Nfft: int,
                deadline: float,
                best_seq: np.ndarray,
                best_score: float) -> (np.ndarray, float):
    """
    Repeatedly minimise the smooth maximum with a temperature schedule.
    Returns the best (sequence, score) discovered during this run.
    """
    y = y0.copy()
    tau = max(init_tau, 1e-12)
    tau_decay = 0.85          # slower cooling than the reference code
    min_tau = 1e-12

    while time.time() < deadline and tau > min_tau:
        # ----- L‑BFGS on the current temperature -----------------------
        def fun(y_vec):
            L, g = _objective_and_grad(y_vec, tau, Nfft)
            return L, g

        # give the optimiser a decent budget (more than the reference)
        opts = {'maxfun': 800, 'maxiter': 800, 'ftol': 1e-12, 'gtol': 1e-12}
        if _SCIPY_AVAILABLE:
            res = minimize(fun, y, method='L-BFGS-B', jac=True, options=opts)
            y = res.x
        else:
            # fall back to a very simple gradient step if SciPy is missing
            L, g = fun(y)
            y -= 0.1 * g / (np.linalg.norm(g) + 1e-12)

        # ----- exact objective for the current y -----------------------
        y_max = np.max(y)
        exp_y = np.exp(y - y_max)
        p = exp_y / exp_y.sum()
        conv = _conv_full(p, Nfft)
        cur_score = (target_sum ** 2) * conv.max()   # = evaluate_sequence

        if cur_score < best_score:
            best_score = cur_score
            best_seq = (p * target_sum).copy()
            print(f"[improve] τ={tau:.2e}  score={best_score:.6f}")

        # ----- temperature update & jitter -----------------------------
        tau *= tau_decay

        # occasional jitter to leave shallow plateaus
        if np.random.rand() < 0.2:
            y += np.random.randn(y.shape[0]) * 0.02

        # break if out of time
        if time.time() >= deadline:
            break

    return best_seq, best_score


def propose_candidate(seed: int = 42,
                     budget_s: float = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` within the allotted time budget.

    Returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin for final checks

    # ------------------------------------------------------------------
    # Obtain a base construction (if one exists) or fall back to uniform.
    try:
        base = np.array(height_sequence_1, dtype=float)  # type: ignore
    except Exception:
        base_len = 2100
        base = np.ones(base_len, dtype=float)

    n = base.shape[0]
    target_sum = math.sqrt(2.0 * n)

    # Normalise to the required L1‑norm (the scaling is irrelevant for the
    # objective, but we keep it for a clean probability vector).
    base = base * (target_sum / base.sum())
    p0 = base / target_sum
    eps_log = 1e-12
    y0 = np.log(p0 + eps_log)

    # ------------------------------------------------------------------
    # Initial best record (evaluate once with the provided evaluator).
    best_seq = base.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # FFT size (next power‑of‑two ≥ 2·n‑1)
    Nfft = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------------
    # Build a collection of diverse starting points for the L‑BFGS stage.
    starts = []

    # 0) the known (or uniform) construction
    starts.append(y0)

    # 1) random Dirichlet via exponential sampling
    rand_exp = np.random.exponential(scale=1.0, size=n)
    starts.append(np.log(rand_exp + eps_log))

    # 2) pure uniform (all entries equal)
    starts.append(np.zeros(n, dtype=float))

    # 3) a perturbed version of the known solution
    starts.append(y0 + np.random.randn(n) * 0.05)

    # 4) sparse spike pattern (few equal masses)
    k_spikes = max(2, n // 200)               # ≈10 spikes for n≈2000
    spikes = np.zeros(n)
    idxs = np.random.choice(n, size=k_spikes, replace=False)
    spikes[idxs] = 1.0 / k_spikes
    starts.append(np.log(spikes + eps_log))

    # 5) mixture of uniform and known solution (more diversity)
    starts.append(0.5 * y0 + 0.5 * np.zeros(n))

    # ------------------------------------------------------------------
    # Run the continuation / L‑BFGS optimisation from each start.
    for s_idx, y_start in enumerate(starts):
        if time.time() >= deadline:
            break
        print(f"[run] start {s_idx + 1}/{len(starts)}")

        # a reasonable initial temperature – a few times the current max‑conv
        p_start = np.exp(y_start - np.max(y_start))
        p_start /= p_start.sum()
        conv0 = _conv_full(p_start, Nfft)
        init_tau = max(conv0.max() * 2.0, 1e-12)

        best_seq, best_score = _run_l_bfgs(
            y0=y_start,
            init_tau=init_tau,
            target_sum=target_sum,
            Nfft=Nfft,
            deadline=deadline,
            best_seq=best_seq,
            best_score=best_score
        )

    # ------------------------------------------------------------------
    # ----- Local stochastic refinement (simulated annealing) -----
    # Work with the probability vector p (sum = 1).
    p_cur = best_seq / target_sum
    cur_score = best_score

    time_left = deadline - time.time()
    if time_left > 15.0:                     # only attempt if we have a bit of time
        # Step size limits (relative to the total mass)
        max_step = target_sum * 0.01
        min_step = target_sum * 1e-5

        # Temperature schedule for the annealer
        T = cur_score * 0.02
        T_min = cur_score * 1e-6
        decay = 0.9993

        it = 0
        while time.time() < deadline:
            it += 1
            # draw a random step size
            step = np.random.uniform(min_step, max_step)

            # choose a source index that has enough mass
            src_candidates = np.where(p_cur > step)[0]
            if src_candidates.size == 0:
                continue
            i = np.random.choice(src_candidates)
            j = np.random.randint(n)
            if i == j:
                continue

            p_new = p_cur.copy()
            p_new[i] -= step
            p_new[j] += step

            # evaluate the exact max‑convolution via FFT (very cheap)
            conv = np.fft.irfft(np.fft.rfft(p_new, n=Nfft) ** 2, n=Nfft)
            new_max = conv[: 2 * n - 1].max()
            new_score = (target_sum ** 2) * new_max

            # Metropolis acceptance
            if new_score < cur_score or np.random.rand() < math.exp(-(new_score - cur_score) / T):
                p_cur = p_new
                cur_score = new_score
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = (p_cur * target_sum).copy()
                    print(f"[local improve] iter={it}  score={best_score:.6f}")

            # cool down
            T = max(T * decay, T_min)

    # ------------------------------------------------------------------
    # Final sanity check (evaluate once more with the official routine)
    final_score = evaluate_sequence(best_seq.tolist())
    if final_score < best_score:
        best_score = final_score
        best_seq = best_seq.copy()

    print(f"[done] best score = {best_score:.6f}, length = {len(best_seq)}")
    return best_seq.tolist()
