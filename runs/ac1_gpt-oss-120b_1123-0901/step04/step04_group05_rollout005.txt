def simplex_projection(v):
    # ensure all entries >=0 and sum to 1
    # Use algorithm by Duchi et al 2008.
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

def local_search(a, n_iters, eps_factor=0.1):
    a_cur = a.copy()
    best_val = compute_val(a_cur)
    for it in range(n_iters):
        conv = _full_convolution(a_cur)
        k = conv.argmax()
        # compute contributions
        # compute indices of possible i
        start = max(0, k-(n-1))
        end = min(n-1, k)
        # contributions array length maybe n
        # we can compute contributions quickly using vectorized slicing
        idx_i = np.arange(start, end+1)
        contrib = a_cur[idx_i] * a_cur[k - idx_i]
        # pick i weighted by contrib
        if contrib.sum() == 0:
            continue
        probs = contrib / contrib.sum()
        i = np.random.choice(idx_i, p=probs)
        # choose j from low a values
        low_idx = np.where(a_cur <= np.percentile(a_cur, 20))[0]  # bottom 20%
        if low_idx.size == 0:
            j = np.random.randint(n)

            j = np.random.choice(low_idx)
        # compute delta
        delta = min(a_cur[i], eps_factor * a_cur[i])
        if delta == 0:

        a_cur[i] -= delta
        a_cur[j] += delta
        # project if needed: guarantee nonnegative sum unchanged.
        # Evaluate
        val = compute_val(a_cur)

            best_val = val
            # keep changes
        else:
            # revert

def _softmax_weights(conv, lam):
    m = conv.max()
    w = np.exp(lam*(conv - m))
    s = w.sum()

"""
Hybrid optimizer for the self‑convolution bound.

The search consists of three stages:

1. **Adam on a smooth‑max surrogate** – a soft‑max weighted sum of the
   convolution values is minimised while the vector stays on the simplex.
   Several random initialisations and a slowly increasing temperature λ
   are used to explore the landscape.

2. **Targeted local search** – from the best vectors found by Adam we
   repeatedly move a small amount of mass from indices that contribute most
   to the maximal convolution entry to indices that are currently small.
   This is a cheap hill‑climbing step that directly attacks the max entry.

3. **LP refinement (optional)** – using CVXPY we solve a linear program:
   maximise Σ g_i subject to  g≥0 and (f∗g)_k ≤ M (M = max(f∗f)).
   The optimiser moves a small step towards the LP solution, guaranteeing
   a non‑increasing bound.

The whole routine respects a user supplied time budget (default 1000 s)
and returns the best sequence (plain Python list) discovered.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray, size: int = None) -> np.ndarray:
    """Linear self‑convolution a∗a using an FFT of length ``size``."""
    n = a.shape[0]
    if size is None:
        size = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return conv


def _softmax_weights(conv: np.ndarray, lam: float) -> np.ndarray:
    """Soft‑max weighting: w_k = exp(lam·(conv_k- max(conv))) / Σ."""
    m = conv.max()
    # shift to avoid overflow
    w = np.exp(lam * (conv - m))
    s = w.sum()
    if s == 0.0:
        w.fill(1.0 / w.size)
    else:
        w /= s
    return w


def _softmax_gradient(a: np.ndarray, w: np.ndarray,
                      size_corr: int = None) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a∗a)[k] with respect to a.
    Closed form:  ∇ = 2·(w ★ a) .
    """
    n = a.shape[0]
    rev_a = a[::-1]
    if size_corr is None:
        size_corr = 1 << int(np.ceil(np.log2(len(w) + n - 1)))
    fw = np.fft.rfft(w, size_corr)
    frev = np.fft.rfft(rev_a, size_corr)
    corr = np.fft.irfft(fw * frev, size_corr)
    grad = 2.0 * corr[n - 1 : n - 1 + n]   # slice length n
    return grad


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries zero – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(a∗a) for a that sums to 1."""
    n = a.shape[0]
    conv = _full_convolution(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
def _local_search(a: np.ndarray, max_iters: int, deadline: float) -> tuple:
    """
    Hill‑climbing that moves a small amount of mass from indices
    contributing heavily to the current maximal convolution entry to
    low‑mass indices. Returns the best vector found and its objective.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_val = _objective(best_a)

    for _ in range(max_iters):
        if time.time() >= deadline:
            break
        conv = _full_convolution(best_a)
        k = int(conv.argmax())

        # indices that take part in conv[k]
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1)
        contrib = best_a[idx] * best_a[k - idx]

        # pick a source index proportional to its contribution
        if contrib.sum() == 0:
            src = np.random.randint(n)
        else:
            probs = contrib / contrib.sum()
            src = np.random.choice(idx, p=probs)

        # pick a destination among the 20 % lightest entries
        low_thr = np.percentile(best_a, 20)
        low_candidates = np.where(best_a <= low_thr)[0]
        if low_candidates.size == 0:
            dst = np.random.randint(n)
        else:
            dst = np.random.choice(low_candidates)

        # move a modest fraction (≤5 %) of the source mass
        delta = min(best_a[src], 0.05 * best_a[src])
        if delta <= 0:

        candidate = best_a.copy()
        candidate[src] -= delta
        candidate[dst] += delta

def _lp_refine(a: np.ndarray, max_val: float, deadline: float) -> np.ndarray:
    """
    Solve the LP  maximise Σ g_i  s.t.  g≥0  and  (a∗g)_k ≤ max_val.
    Returns the optimiser g normalised to sum = 1.
    If CVXPY is unavailable or the solver fails, returns the original a.
    """
    try:
        import cvxpy as cp
    except Exception:
        return a

    n = a.shape[0]
    g = cp.Variable(n, nonneg=True)

    # Build convolution matrix A (2n‑1)×n  such that (a∗g) = A·g
    rows = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        idx = np.arange(i_start, i_end + 1)
        row[idx] = a[k - idx]
        rows.append(row)
    A = np.vstack(rows)   # shape (2n‑1, n)

    constraints = [A @ g <= max_val]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000, verbose=False)

        return a

    g_opt = np.maximum(g.value, 0.0)
    if g_opt.sum() == 0.0:
        return a
    g_opt /= g_opt.sum()
    return g_opt

            elif mode == 1:
                r = np.random.uniform(0.2, 0.9)
                a0 = r ** np.arange(n, dtype=np.float64)
                a0 /= a0.sum()
            elif mode == 2:
                i = np.arange(1, n + 1, dtype=np.float64)
                a0 = np.sin(np.pi * i / (n + 1))
                a0 = np.maximum(a0, 0.0)
                a0 /= a0.sum()
            elif mode == 3:
                p = np.random.uniform(0.5, 2.0)
                a0 = (np.sin(np.pi * np.arange(1, n + 1) / (n + 1))) ** p
                a0 = np.maximum(a0, 0.0)
                a0 /= a0.sum()
            elif mode == 4:
                r = np.random.uniform(0.3, 0.85)
                a0 = r ** np.arange(n, dtype=np.float64)
                a0 += 0.01 * np.random.rand(n)
                a0 = np.maximum(a0, 0.0)
                a0 /= a0.sum()
            elif mode == 5:
                x = np.linspace(-1.0, 1.0, n)
                a0 = 1.0 - x ** 2
                a0 = np.maximum(a0, 0.0)
                a0 /= a0.sum()
            else:
                # bimodal half‑sine
                half = n // 2
                left = np.sin(np.pi * (np.arange(half) + 1) / (half + 1))
                right = np.sin(np.pi * (np.arange(n - half) + 1) / (n - half + 1))
                a0 = np.concatenate([left, right])
                a0 = np.maximum(a0, 0.0)
                a0 /= a0.sum()

        # --------------------------------------------------------------
        # Adam optimisation on the smooth‑max surrogate
        u = np.log(a0 + 1e-30)          # logits, guarantees a ≥ 0 & Σa = 1
        m = np.zeros_like(u)
        v = np.zeros_like(u)

        lam = 0.5                       # start smooth
        lam_factor = 1.012
        base_lr = 0.05
        max_iter = 30000                # generous upper bound
        fft_size = 1 << int(np.ceil(np.log2(2 * n - 1)))
        corr_size = 1 << int(np.ceil(np.log2(3 * n)))

        last_improve = time.time()
        it = 0
        while time.time() < deadline and it < max_iter:
            it += 1

            # logits → probability simplex
            u_shift = u - np.max(u)
            a = np.exp(u_shift)
            a /= a.sum()

            # convolution & smooth‑max
            conv = _full_convolution(a, size=fft_size)
            w = _softmax_weights(conv, lam)

            # gradient w.r.t. a (smooth‑max)
            grad_a = _softmax_gradient(a, w, size_corr=corr_size)

            # map to logits gradient (soft‑max Jacobian)
            dot = np.dot(a, grad_a)
            grad_u = a * (grad_a - dot)

            # Adam update
            m = 0.9 * m + 0.1 * grad_u
            v = 0.999 * v + 0.001 * (grad_u ** 2)

            it_f = float(it)
            m_hat = m / (1.0 - 0.9 ** it_f)
            v_hat = v / (1.0 - 0.999 ** it_f)

            lr = base_lr * (0.9997 ** it)
            u -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

            # increase temperature → tighter approximation of the max
            lam *= lam_factor

            # exact evaluation every few steps
            if it % 15 == 0:
                val = 2.0 * n * conv.max()
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[restart {restart:03d}] iter {it:05d}  new best = {val:.6f}")

            # shake if we have stalled for ~2 s
            if time.time() - last_improve > 2.0:
                u += (np.random.rand(n) - 0.5) * 1e-4
                m.fill(0.0)
                v.fill(0.0)
                last_improve = time.time()

        # --------------------------------------------------------------
        # Targeted local search starting from the best vector of this restart
        if best_seq is not None and len(best_seq) == n:
            remaining = deadline - time.time()
            if remaining > 5.0:
                # allocate up to ~200 ms per 1000 local steps
                max_local = int(min(25000, remaining * 2000))
                best_seq, best_val = _local_search(best_seq, max_local, deadline)

        restart += 1
        print(f"[restart {restart:03d}] finished it={it}, best so far = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final fine‑tuning (high λ, smaller step) + optional LP refinement
    if best_seq is not None:
        n = len(best_seq)
        u = np.log(best_seq + 1e-30)
        m = np.zeros_like(u)
        v = np.zeros_like(u)

        lam = 5.0
        lam_factor = 1.001
        base_lr = 0.02
        fft_size = 1 << int(np.ceil(np.log2(2 * n - 1)))
        corr_size = 1 << int(np.ceil(np.log2(3 * n)))

        it = 0
        max_iter = 40000
        last_improve = time.time()
        while time.time() < deadline and it < max_iter:
            it += 1
            u_shift = u - np.max(u)
            a = np.exp(u_shift)
            a /= a.sum()
            conv = _full_convolution(a, size=fft_size)
            w = _softmax_weights(conv, lam)
            grad_a = _softmax_gradient(a, w, size_corr=corr_size)

            dot = np.dot(a, grad_a)
            grad_u = a * (grad_a - dot)

            m = 0.9 * m + 0.1 * grad_u
            v = 0.999 * v + 0.001 * (grad_u ** 2)

            it_f = float(it)
            m_hat = m / (1.0 - 0.9 ** it_f)
            v_hat = v / (1.0 - 0.999 ** it_f)

            lr = base_lr * (0.9999 ** it)
            u -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

            lam *= lam_factor

            if it % 20 == 0:
                val = 2.0 * n * conv.max()
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    print(f"[fine‑tune] iter {it:05d}  new best = {val:.6f}")

        # ------------------------------------------------------------------
        # LP refinement – a few cheap iterations if time permits
        while time.time() < deadline:
            conv = _full_convolution(best_seq)
            M = conv.max()
            g = _lp_refine(best_seq, M, deadline)
            # moving average update (step size t)
            t = 0.15
            new_a = (1 - t) * best_seq + t * g
            new_a = _simplex_projection(new_a)   # keep on simplex
            val = _objective(new_a)
            if val < best_val:
                best_val = val
                best_seq = new_a
                print(f"[LP refine] new best = {val:.6f}")
            else:
                break   # no progress → stop LP phase

    # ------------------------------------------------------------------
    # Return plain Python list (values already satisfy the evaluator's clamps)
    if best_seq is None:
        # fallback: uniform distribution of length 2000
        n = 2000
        best_seq = np.full(n, 1.0 / n)

    best_seq = np.clip(best_seq, 0.0, 1000.0)
    return best_seq.tolist()
