"""
Search for a non‑negative step‑function that minimises the provided
``evaluate_sequence`` objective.

Strategy
--------
1. **Parametric warm start** – the optimal shape is close to a
   truncated exponential distribution.  For a given length *n* we
   enumerate many decay rates  (r = 1‑t) and keep the one with the
   smallest exact objective (computed with an FFT‑based convolution).

2. **Local refinement** – the best exponential is polished with a
   projected Adam optimiser that works on the true non‑smooth objective.
   The optimiser keeps the ℓ₁‑norm fixed ( Σa = √(2 n) ) and guarantees
   a ≥ 0 after each step.

3. **Multiple lengths** – we try a few lengths (8000, 4000, 2000).  The
   objective is scale‑invariant, but larger *n* gives a lower theoretical
   bound (average convolution ≈ 1).  The best construction over all
   lengths is returned.

All heavy work (convolution, gradients) uses FFTs, so a single
iteration is cheap and the whole search comfortably fits into the
1000‑second wall‑clock limit.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper FFT utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << ((x - 1).bit_length())


def _max_convolution(a: np.ndarray, Nfft: int) -> float:
    """Return max_{k} (a∗a)[k] using an FFT of length Nfft."""
    A = np.fft.rfft(a, n=Nfft)
    conv = np.fft.irfft(A * A, n=Nfft)[: 2 * a.shape[0] - 1]
    return float(conv.max())


def _softmax_weights(conv: np.ndarray, tau: float):
    """Stable soft‑max over the convolution values.
    Returns weight vector w (summing to 1) and the smoothed maximum Lτ."""
    maxc = conv.max()
    if tau <= 0.0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w, maxc
    ex = np.exp((conv - maxc) / tau)
    sum_ex = ex.sum()
    w = ex / sum_ex
    Ltau = maxc + tau * np.log(sum_ex)
    return w, Ltau


def _weighted_rev(a: np.ndarray, w: np.ndarray, Nfft: int) -> np.ndarray:
    """
    Compute rev_i = Σ_k w_k * a_{k‑i}  (indices outside range ignored)
    using FFTs.  This is the (sub‑)gradient of the (soft‑max) max‑convolution.
    """
    A = np.fft.rfft(a, n=Nfft)
    W_rev = np.fft.rfft(w[::-1], n=Nfft)
    conv_rev = np.fft.irfft(W_rev * A, n=Nfft)
    conv_rev = conv_rev[: len(w) + a.shape[0] - 1]
    # slice that corresponds to rev_i for i=0..n‑1
    start = 2 * a.shape[0] - 2
    rev = conv_rev[start - np.arange(a.shape[0])]
    return rev


# ----------------------------------------------------------------------
# Adam optimiser (identical in spirit to the reference implementation)
# ----------------------------------------------------------------------
def _adam_optimize(
    a: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int,
    best_seq: np.ndarray,
    best_score: float,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser with a soft‑max smoothing of the max‑convolution.
    Returns the best sequence seen in this run together with its exact
    objective value (which equals the maximum convolution after the
    ℓ₁‑normalisation).
    """
    target_sum = math.sqrt(2.0 * n)

    # normalise the initial vector
    a = a * (target_sum / a.sum())

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr0 = 0.06                # a touch larger than the reference
    jitter_interval = 4000    # a bit more frequent

    # temperature schedule
    tau_factor = 0.20
    decay_rate = max_iters / 12.0

    Nfft = _next_pow2(2 * n - 1)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ----- exact convolution -----
        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        M = float(conv.max())

        # ----- soft‑max temperature -----
        tau = tau_factor * M * math.exp(-it / decay_rate)
        if tau < 1e-9:
            # active set (all indices within tolerance of the max)
            mask = conv >= M - 1e-12
            cnt = mask.sum()
            w = mask.astype(float) / max(cnt, 1)
            Ltau = M
        else:
            w, Ltau = _softmax_weights(conv, tau)

        # ----- gradient (weighted reverse convolution) -----
        rev = _weighted_rev(a, w, Nfft)

        # ----- Adam update (projected gradient of the exact objective) -----
        coeff = 4.0 * n / (target_sum * target_sum)          # =2 after normalisation
        const = 4.0 * n * Ltau / (target_sum ** 3)
        grad = coeff * rev - const

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / (1.0 + 1e-4 * it)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # ----- projection onto feasible set -----
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a = a * (target_sum / s)
        else:
            a = np.full_like(a, target_sum / n)

        # occasional jitter to help escape plateaus
        if it % jitter_interval == 0:
            jitter = np.random.randn(n) * (1e-3 * target_sum / n)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        cur_score = M          # after projection M equals the true objective
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[len {n:5d} iter {it:8d}] new best = {best_score:.6f}")

    return best_seq, best_score

# Initial shape generation (exponential families + a few generic seeds)
# ----------------------------------------------------------------------
def _exponential_candidate(n: int, target_sum: float, t: float) -> np.ndarray:
    """
    Return a normalised exponential vector.
    t = 1‑r  (positive, small).  The raw distribution is (1‑r)·r**i,
    i=0..n‑1 (sum = 1‑r**n).  It is then scaled to have ℓ₁‑norm = target_sum.
    """
    r = 1.0 - t
    idx = np.arange(n, dtype=np.float64)
    raw = t * np.power(r, idx)                # (1‑r)·r**i
    sum_raw = 1.0 - np.power(r, n)            # geometric series sum
    scaling = target_sum / sum_raw
    return raw * scaling


def _make_generic_initials(n: int, target_sum: float) -> list[np.ndarray]:
    """A small diverse set of generic starting points (uniform, Dirichlet, etc.)."""
    rng = np.random.default_rng()
    init = []

    # uniform
    init.append(np.full(n, target_sum / n))

    # Dirichlet (shape=1)
    g = rng.gamma(1.0, 1.0, size=n)
    init.append(g / g.sum() * target_sum)

    # random exponential (different rates)
    for _ in range(2):
        rate = rng.uniform(0.0005, 0.01)
        x = np.exp(-rate * np.arange(n))
        init.append(x / x.sum() * target_sum)

    # reversed exponential
    rev = init[-1][::-1]
    init.append(rev)

    # Hann window
    hann = 0.5 - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1))
    init.append(hann / hann.sum() * target_sum)

    return init

# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  Returns the best sequence found as a plain
    Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # small safety margin

    # try to reuse a previously discovered construction, if present
    try:
        base = np.array(height_sequence_1, dtype=float)   # type: ignore
    except Exception:
        base = None

    # ------------------------------------------------------------------
    # lengths to explore (large sizes give the strongest theoretical bound)
    lengths = [8000, 4000, 2000]
    lengths = [max(300, min(l, 8000)) for l in lengths]

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Main optimisation loop over the selected lengths
    for n in lengths:
        if time.time() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)
        Nfft = _next_pow2(2 * n - 1)
        print(f"[run] length={n}, target_sum={target_sum:.3f}")

        # --------------------------------------------------------------
        # 1) coarse exponential sweep (fast, cheap, often already good)
        # --------------------------------------------------------------
        # t values (1‑r) explored on a log scale; the optimum is expected to be ~e/(2n)
        t_vals = np.logspace(np.log10(1e-5), np.log10(5e-3), 250)
        for t in t_vals:
            if time.time() >= deadline:
                break
            a = _exponential_candidate(n, target_sum, t)

            # fast objective (max convolution after ℓ₁‑normalisation = max_b)
            M = _max_convolution(a, Nfft)
            score = M                     # because sum = target_sum ⇒ obj = M
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"[len {n}] exp(t={t:.2e}) new best = {best_score:.6f}")

        # also try the reversed exponential (mass concentrated at the end)
        for t in t_vals:
            if time.time() >= deadline:
                break
            a = _exponential_candidate(n, target_sum, t)[::-1]
            M = _max_convolution(a, Nfft)
            score = M
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"[len {n}] rev‑exp(t={t:.2e}) new best = {best_score:.6f}")

        # --------------------------------------------------------------
        # 2) add a few diverse seeds (including the previous best)
        # --------------------------------------------------------------
        init_vectors = _make_generic_initials(n, target_sum)

        # include the best found so far as an extra seed
        if best_seq is not None:
            init_vectors.append(best_seq.copy())

        # if we have a base vector of different size, interpolate it
        if base is not None and base.shape[0] != n:
            old_x = np.linspace(0.0, 1.0, base.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            interp = np.interp(new_x, old_x, base)
            interp = np.maximum(interp, 0.0)
            interp = interp / interp.sum() * target_sum
            init_vectors.append(interp)

        # --------------------------------------------------------------
        # 3) local refinement with Adam for each starter
        # --------------------------------------------------------------
        # allocate a generous but bounded iteration budget per start
        max_total_iters = 1_200_000                     # hard cap per length
        iters_per_start = max_total_iters // max(1, len(init_vectors))

        print(f"   refining {len(init_vectors)} starts, ≤{iters_per_start} iters each")
        for idx, a0 in enumerate(init_vectors):
            if time.time() >= deadline:
                break
            print(f"   start {idx+1}/{len(init_vectors)} for length {n}")
            best_seq, best_score = _adam_optimize(
                a=a0.copy(),
                n=n,
                deadline=deadline,
                max_iters=iters_per_start,
                best_seq=best_seq if best_seq is not None else a0,
                best_score=best_score,
            )

    # ------------------------------------------------------------------
    # final safeguard
    if best_seq is None:
        n = 2000
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n)
        best_score = 2.0

    print(f"[done] best score = {best_score:.6f}, length = {best_seq.shape[0]}")
    return best_seq.tolist()
