def subgradient_descent(a, n, max_iters, deadline):
    a = a.copy()
    best_val = float('inf')
    best_a = a.copy()
    step0 = 0.5   # initial step
    for it in range(1, max_iters+1):
        max_c, idx = compute_max_and_idx(a)
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
        # compute subgradient
        grad = np.zeros_like(a)
        start = max(0, idx - (n-1))

"""Propose a low‑score step‑function.

The algorithm works in three stages for a sequence length n:

1. **Adam on a smooth‑max surrogate** – a fast gradient method that
   minimises a soft maximum of the convolution.  The temperature
   (smoothness) is increased slowly so the surrogate approaches the real
   max.

2. **Sub‑gradient fine‑tuning** – after the surrogate has converged we
   switch to the true max.  The sub‑gradient of the max‑convolution is
   cheap to compute (a shifted copy of the current vector).  A
   diminishing stepsize on the probability simplex drives the peak
   further down.

3. **Local random hill‑climbing** – tiny mass transfers between two
   coordinates are accepted only if they improve the exact objective.
   This cleans up any remaining discretisation errors.

The procedure is run on a schedule of increasing lengths
[256, 512, 1024, 2048, 4096, 8192]; each longer length is warm‑started
by linear interpolation of the best sequence found so far.  The search
stops as soon as the time budget (default 1000 s) is exhausted or a
value below the target 1.503 is achieved.

All helper functions are defined at module level (no closures) and the
returned sequence is a plain Python list suitable for ``evaluate_sequence``.
"""

import time
import numpy as np
import scipy.sparse  # only needed for optional up‑sampling utilities

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.shape[0] - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Compute the cross‑correlation (w ★ a) needed for the smooth‑max gradient.
    Returns a vector of length n (= len(a)).
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        {x | x_i ≥ 0, Σ_i x_i = 1}
    """
    if v.ndim != 1:
        raise ValueError("v must be a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _smooth_max_grad(a: np.ndarray, lam: float, fft_len: int):
    """
    Gradient of the smooth‑max surrogate of max(a*a) with respect to a.
    Returns (grad, max_val) where max_val = max(a*a).
    """
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    # soft‑max weights (they sum to 1 and concentrate on the peak)
    w = np.exp(lam * (conv - max_c))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(conv, 1.0 / conv.size)
    else:
        w /= w_sum
    # gradient = 2 * (w ★ a)
    grad = 2.0 * _cross_correlation(w, a, fft_len)
    return grad, max_c


def _subgradient_step(a: np.ndarray, step: float):
    """
    One sub‑gradient step for the true max‑convolution.
    The sub‑gradient at the current peak is 2·a shifted by the peak index.
    Returns (a_new, max_val) after the step and simplex projection.
    """
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    k = int(np.argmax(conv))  # index of a current peak
    idx = k - np.arange(n)
    mask = (idx >= 0) & (idx < n)
    grad = np.zeros_like(a)
    grad[mask] = 2.0 * a[idx[mask]]
    a_new = _project_simplex(a - step * grad)
    return a_new, max_c


def _interpolate_to(a: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation of a probability vector to a new length.
    The result is renormalised to sum to 1.
    """
    old_n = a.shape[0]
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-12:
        a_new = np.full(new_n, 1.0 / new_n)
    else:
        a_new = a_new / s
    return a_new


def _exact_value(a: np.ndarray) -> float:
    """Exact evaluation 2·n·max(a*a) (sum of a is 1)."""
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    conv = _full_convolution(a, fft_len)
    return float(2.0 * n * conv.max())

        lr = 0.12
        lam = 0.2
        lam_factor = 1.05          # slower temperature growth than before
        max_iters = 20000
        fft_len = _next_pow2(2 * n - 1)
        last_improve = time.time()

        for it in range(1, max_iters + 1):
            if time.time() > deadline:
                break

            grad, max_c = _smooth_max_grad(a, lam, fft_len)

            # Adam update (pre‑projection)
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)
            a_new = a - step
            a = _project_simplex(a_new)

            # temperature schedule
            lam = min(lam * lam_factor, 1e5)

            # exact objective (cheap because we already have max_c)
            val = 2.0 * n * max_c
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                last_improve = time.time()
                print(f"[Adam n={n} it={it}] new best: {val:.6f}")
                if best_val < 1.503:
                    print("Target reached – terminating early.")
                    deadline = time.time()   # force outer loops to stop
                    break

            # stagnation handling
            if time.time() - last_improve > 8.0:
                noise = (np.random.rand(n) - 0.5) * 1e-3
                a = _project_simplex(a + noise)
                last_improve = time.time()
                print(f"[Adam n={n} it={it}] shake")

        # ------------------------------------------------------------------
        # Phase 2 – Sub‑gradient fine‑tuning on the true max
        step0 = 0.5
        sub_iters = 5000
        for it in range(1, sub_iters + 1):
            if time.time() > deadline:
                break
            step = step0 / np.sqrt(it)
            a, max_c = _subgradient_step(a, step)
            val = 2.0 * n * max_c
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[Subgrad n={n} it={it}] new best: {val:.6f}")
                if best_val < 1.503:
                    deadline = time.time()
                    break
            # occasional small shake to escape plateaus
            if it % 500 == 0 and np.random.rand() < 0.3:
                a = _project_simplex(a + (np.random.rand(n) - 0.5) * 5e-4)

        # ------------------------------------------------------------------
        # Phase 3 – Random local hill‑climbing (mass transfers)
        if time.time() > deadline:
            break
        print(f"[n={n}] Local random refinement")
        refine_iters = 200000
        it = 0
        while time.time() < deadline and it < refine_iters:
            i, j = np.random.choice(n, 2, replace=False)
            delta = a[i] * 0.005 * np.random.rand()
            if delta <= 0.0:
                continue
            a_prop = a.copy()
            a_prop[i] -= delta
            a_prop[j] += delta
            a_prop = _project_simplex(a_prop)
            val = _exact_value(a_prop)
            if val < best_val:
                best_val = val
                best_seq = a_prop.copy()
                a = a_prop
                print(f"[refine n={n} it={it}] improved to {val:.6f}")
            it += 1

    # ------------------------------------------------------------------
    # Fallback – uniform vector if nothing succeeded.
    if best_seq is None:
        n_fallback = 1024
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        best_val = _exact_value(best_seq)
        print(f"Fallback uniform sequence, value = {best_val:.6f}")

    print(f"\nBest value discovered: {best_val:.6f} (length {len(best_seq)})")
    # Return plain Python list as required by the external evaluator.
    return best_seq.tolist()
