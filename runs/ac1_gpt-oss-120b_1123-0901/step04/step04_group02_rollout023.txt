def lp_iter(a, max_iters, time_left):
    n = len(a)
    best_a = a.copy()
    best_val = _exact_obj(a,n)
    t = 0
    mix = 0.5  # can be constant
    while t < max_iters and time.time() < time_left:
        conv = np.convolve(a, a)
        M = conv.max()
        # Build A
        A = np.zeros((2*n-1, n))
        for i in range(n):
            A[i:i+n, i] = a[i]
        # Solve LP
        c = -np.ones(n)
        b = np.full(2*n-1, M)
        # use high performance method
        res = linprog(c, A_ub=A, b_ub=b, bounds=(0, None), method='highs')
        if not res.success:
            break
        g = res.x
        sum_g = g.sum()
        if sum_g == 0:
            break
        # Normalize g to sum 1
        g = g / sum_g
        # Ensure positivity, add eps
        # Update a via mixing
        a = (1 - mix) * a + mix * g
        # Normalize a (numerical)
        a = np.maximum(a, 0)
        a_sum = a.sum()
        if a_sum > 0:
            a = a / a_sum
        # compute objective
        val = _exact_obj(a,n)
        if val < best_val:
            best_val = val
            best_a = a.copy()
        t += 1
        # maybe reduce mix as t grows
        mix = max(0.01, mix * 0.95)
    return best_a, best_val

"""
propose_candidate
-----------------

Search for a non‑negative step function (vector on the simplex) that
minimises

    F(a) = 2·n·max(a*a) / (∑a)²  = 2·n·max(conv(a,a))      (since we keep ∑a = 1)

The algorithm combines two ingredients:

1. **Linear‑programming boosting** (the method described in the
   statement).  Starting from a feasible vector `a` we compute
   M = max(conv(a,a)) and then solve

        maximise   ∑ g_i
        subject to  a * g ≤ M          (convolution, component‑wise)
                    g_i ≥ 0

   (implemented with `scipy.optimize.linprog`).  The solution `g` is
   normalised to a probability vector and mixed with the current `a`
   (a ← (1‑t)a + t g).  Repeating this step monotonically decreases
   the objective.

2. **Finishing with a few Adam steps** on the smooth‑max surrogate
   (as in the previous best solver).  This cleans up any remaining
   irregularities and yields a slightly better local optimum.

Several lengths are tried (256 … 2048) with a few random restarts.
The best sequence found before the time budget expires is returned
as a plain Python list.

The whole routine respects a global time budget (default 1000 s) and
never returns an empty or invalid list.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.shape[0] - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Gradient of Σ w_k·(a*a)_k w.r.t. a (half the sub‑gradient)."""
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("v must be 1‑D")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _exact_obj(a: np.ndarray) -> float:
    """Exact value of the objective for a probability vector a (∑a = 1)."""
    n = a.shape[0]
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    return 2.0 * n * conv.max()


def _build_conv_matrix(a: np.ndarray) -> np.ndarray:
    """
    Build the (2n‑1)×n matrix A such that (A @ g) = a * g (linear convolution).
    Entry A[k,i] = a[i] if 0 ≤ k‑i < n else 0.
    """
    n = a.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = a[i]
    return A


def _solve_lp_max_sum(a: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve   maximise   ∑ g_i
            subject to a * g ≤ M    (entry‑wise)
                       g_i ≥ 0

    Returns the optimal g (may be all zeros if the LP is infeasible).
    """
    n = a.shape[0]
    A = _build_conv_matrix(a)
    b = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)            # maximise sum ⇒ minimise –sum
    res = linprog(c, A_ub=A, b_ub=b, bounds=(0, None), method="highs")
    if not res.success:
        return None
    return res.x


def _lp_boost_step(a: np.ndarray, mix: float) -> np.ndarray:
    """
    One LP‑boosting iteration:
        – compute M = max(conv(a,a))
        – solve the LP for g
        – normalise g to sum 1
        – return the convex combination (1‑mix)·a + mix·g
    If the LP fails, the original a is returned unchanged.
    """
    n = a.shape[0]
    conv = np.convolve(a, a)
    M = conv.max()
    g = _solve_lp_max_sum(a, M)
    if g is None or g.sum() <= 0.0:
        return a
    g = g / g.sum()
    # keep the vector on the simplex (convex combo preserves sum=1)
    a_new = (1.0 - mix) * a + mix * g
    # avoid numerical negatives
    a_new = np.maximum(a_new, 0.0)
    a_new /= a_new.sum() + 1e-30
    return a_new


def _adam_fine_tune(a: np.ndarray, max_iters: int, deadline: float) -> np.ndarray:
    """
    A short Adam optimisation on the smooth‑max surrogate.
    Starts from the current a (already on the simplex) and performs
    at most max_iters updates or until the deadline is reached.
    """
    n = a.shape[0]
    fft_len_conv = _next_pow2(2 * n - 1)
    fft_len_corr = _next_pow2(3 * n)

    # Adam state
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.12
    lam = 0.6          # start temperature for smooth‑max
    lam_factor = 1.25

    t = 0
    while time.time() < deadline and t < max_iters:
        conv = _full_convolution(a, fft_len_conv)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.shape[0])
        else:
            w /= w_sum

        grad = 2.0 * _cross_correlation(w, a, fft_len_corr)

        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_simplex(a - step)

        lam = min(lam * lam_factor, 1e6)

    return a


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Return a list of non‑negative numbers (a step function) that
    minimises the provided evaluation function.
    The routine respects the global time budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Helper to evaluate the exact objective (no clipping, sum = 1)
    # ------------------------------------------------------------------
    def _update_best(a_vec):
        nonlocal best_seq, best_val
        val = _exact_obj(a_vec)
        if val < best_val:
            best_val = val
            best_seq = a_vec.copy()
            print(f"[new best] n={a_vec.shape[0]}, value={val:.6f}")

    # ------------------------------------------------------------------
    # Problem sizes to explore (moderate sizes keep the LP tractable)
    # ------------------------------------------------------------------
    lengths = [256, 512, 1024, 2048]

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")
        fft_len_conv = _next_pow2(2 * n - 1)

        # number of independent restarts per length
        restarts = 4

        for r in range(restarts):
            if time.time() > deadline:
                break

            # ---------- initialise a ----------
            if r == 0:
                # warm‑start from the global best if it has the same length
                if best_seq is not None and best_seq.shape[0] == n:
                    a = best_seq.copy()
                    print("[restart 0] warm start from global best")
                else:
                    a = np.full(n, 1.0 / n, dtype=np.float64)   # uniform
                    print("[restart 0] uniform start")
            else:
                # random Dirichlet start – ensure strict positivity
                a = np.random.dirichlet(np.ones(n) * 0.5)
                print(f"[restart {r}] random Dirichlet start")

            # add a tiny epsilon to avoid zero entries (LP would become unbounded)
            eps = 1e-12
            a = np.maximum(a, eps)
            a /= a.sum()

            # ------------------------------------------------------------------
            # Phase 1 – LP boosting
            # ------------------------------------------------------------------
            mix = 0.5               # initial mixing coefficient
            for outer in range(30):   # at most 30 LP updates
                if time.time() > deadline:
                    break
                a = _lp_boost_step(a, mix)
                # slowly reduce mixing – we want finer adjustments later
                mix = max(0.02, mix * 0.93)
                _update_best(a)

            # ------------------------------------------------------------------
            # Phase 2 – short Adam fine‑tuning
            # ------------------------------------------------------------------
            if time.time() < deadline:
                a = _adam_fine_tune(a, max_iters=8000, deadline=deadline)
                _update_best(a)

            # ------------------------------------------------------------------
            # Phase 3 – very cheap random local search (swap tiny mass)
            # ------------------------------------------------------------------
            if time.time() < deadline:
                local_a = a.copy()
                for _ in range(2000):
                    if time.time() > deadline:
                        break
                    i, j = np.random.choice(n, 2, replace=False)
                    delta = 0.005 * local_a[i] * np.random.rand()
                    if delta <= 0.0:
                        continue
                    prop = local_a.copy()
                    prop[i] -= delta
                    prop[j] += delta
                    prop = _project_simplex(prop)
                    val = _exact_obj(prop)
                    if val < best_val:
                        best_val = val
                        best_seq = prop.copy()
                        local_a = prop
                        print(f"[local search] improved to {val:.6f}")

    # ------------------------------------------------------------------
    # Final fallback – uniform vector (should never happen, but safe)
    # ------------------------------------------------------------------
    if best_seq is None:
        n_fallback = 1024
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        best_val = _exact_obj(best_seq)
        print(f"Fallback uniform sequence, value = {best_val:.6f}")

    # Return plain Python list as required by the external evaluator.
    return best_seq.tolist()
