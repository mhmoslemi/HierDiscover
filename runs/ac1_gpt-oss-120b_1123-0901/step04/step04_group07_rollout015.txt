def proj_simplex(v, z=target_sum):
    # sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0

def proj_simplex(v, z):
    """Project onto the simplex {x >= 0, sum x = z}."""
    if z < 0:
        raise ValueError
    # shift v so that sum > z? Use algorithm
    # Following Duchi 2008
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n+1)

    if not np.any(cond):

        return np.zeros_like(v)

"""
Simple projected sub‑gradient optimizer for the “max‑convolution” problem.

The code starts from a few diverse initial vectors (including the previously
found construction ``height_sequence_1`` if it exists), interpolates them to a
set of lengths (the original length, ≈1.5× and 2×, never exceeding 8000) and
optimises each candidate by a projected sub‑gradient descent on the exact
objective

    f(a) = max_k (a * a)[k]

under the simplex constraint Σ a_i = √(2·n), a_i ≥ 0.  The gradient of the
max‑convolution is the (scaled) weighted reverse convolution which can be
computed efficiently with FFTs.  A diminishing step‑size together with a tiny
random jitter avoids stagnation.  The best sequence seen over all runs is
returned.

The implementation is fully self‑contained, uses only NumPy and runs well
within the 1000 s wall‑clock budget.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _proj_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Simple projection onto the non‑negative simplex {x ≥ 0, Σ x = target_sum}.
    This is the “truncate‑and‑scale” method used in many related solvers.
    """
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s > 0.0:
        v = v * (target_sum / s)
    else:
        # All entries vanished – fall back to a uniform vector
        v = np.full_like(v, target_sum / v.shape[0])
    return v


def _softmax_weights(conv: np.ndarray, tau: float):
    """Stable soft‑max of convolution values – returns the weight vector."""
    if tau <= 0.0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w, conv.max()
    maxc = conv.max()
    ex = np.exp((conv - maxc) / tau)
    sum_ex = ex.sum()
    w = ex / sum_ex
    Ltau = maxc + tau * math.log(sum_ex)
    return w, Ltau


def _weighted_rev(a: np.ndarray, w: np.ndarray, Nfft: int) -> np.ndarray:
    """
    Weighted reverse convolution.
    Computes  rev_i = Σ_k w_k * a_{k-i}  (indices out of bounds are ignored)
    using FFTs.  This is exactly the sub‑gradient of the (soft‑max) max‑convolution.
    """
    A_fft = np.fft.rfft(a, n=Nfft)
    W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)
    conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
    # keep the part that corresponds to the linear convolution length
    conv_rev = conv_rev[: len(w) + a.shape[0] - 1]
    # extract the slice giving rev_i for i = 0 … n‑1
    idx_start = 2 * a.shape[0] - 2
    rev = conv_rev[idx_start - np.arange(a.shape[0])]
    return rev


def _max_convolution(a: np.ndarray, Nfft: int) -> float:
    """Full linear convolution a * a and return its maximum value."""
    A_fft = np.fft.rfft(a, n=Nfft)
    conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * a.shape[0] - 1]
    return float(conv.max()), conv


# ----------------------------------------------------------------------
# Core optimiser
# ----------------------------------------------------------------------
def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    target_sum: float,
    deadline: float,
    max_iters: int,
    best_seq: np.ndarray,
    best_score: float,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑convolution objective.
    Returns the best vector seen and its (exact) objective value.
    """
    a = _proj_simplex(a0.copy(), target_sum)

    # FFT length – next power of two ≥ 2·n‑1
    Nfft = 1 << ((2 * n - 1).bit_length())

    # Initial step‑size scaling factor (chosen empirically)
    step0 = 0.8 * target_sum

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ----- current max‑convolution -----
        M, conv = _max_convolution(a, Nfft)
        cur_score = M                      # after normalisation the objective equals M

        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[n {n:5d} it {it:7d}] new best = {best_score:.6f}")

        # ----- sub‑gradient of the max (exact) -----
        # active set – all indices within a tiny tolerance of the maximum
        mask = conv >= M - 1e-12
        cnt = int(mask.sum())
        if cnt == 0:
            w = np.zeros_like(conv)
        else:
            w = mask.astype(float) / cnt

        rev = _weighted_rev(a, w, Nfft)
        grad = 2.0 * rev                    # ∂ max/∂a

        # ----- step‑size (diminishing) -----
        # Use a normalized direction to keep the update magnitude roughly
        # proportional to step0 / sqrt(iteration).
        gnorm = np.linalg.norm(grad) + 1e-12
        direction = grad / gnorm
        step = step0 / math.sqrt(it)        # classic 1/√t schedule

        # ----- try a descent step (simple back‑tracking) -----
        for _ in range(5):
            a_candidate = a - step * direction
            a_candidate = _proj_simplex(a_candidate, target_sum)
            M_cand, _ = _max_convolution(a_candidate, Nfft)
            if M_cand <= cur_score:
                a = a_candidate
                break
            step *= 0.5                      # reduce step and retry

        # ----- occasional jitter to shake out of plateaus -----
        if it % 3000 == 0:
            jitter = np.random.randn(n) * (1e-4 * target_sum / n)
            a = _proj_simplex(a + jitter, target_sum)

    return best_seq, best_score


def _make_initial_vectors(n: int, target_sum: float) -> list[np.ndarray]:
    """
    Produce a small, diverse collection of starting vectors of length ``n``.
    All vectors are already normalised to have ℓ₁‑norm = target_sum.
    """
    rng = np.random.default_rng()
    init = []

    # 1) uniform
    init.append(np.full(n, target_sum / n))

    # 2) random Dirichlet (Gamma(1,1))
    g = rng.gamma(1.0, 1.0, size=n)
    init.append(g / g.sum() * target_sum)

    # 3) Hann window (smooth cosine)
    hann = 0.5 - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1))
    init.append(hann / hann.sum() * target_sum)

    # 4) exponential decay (random rate)
    rate = rng.uniform(0.0005, 0.01)
    exp_vec = np.exp(-rate * np.arange(n))
    init.append(exp_vec / exp_vec.sum() * target_sum)

    # 5) reversed exponential
    rev = exp_vec[::-1]
    init.append(rev / rev.sum() * target_sum)

    return init
