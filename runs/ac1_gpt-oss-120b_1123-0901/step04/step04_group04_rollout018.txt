"""
Propose a step‑function (non‑negative vector) that minimises the
``evaluate_sequence`` objective.

The algorithm consists of three stages:

1. **Smooth‑max optimisation** – minimise a temperature‑smoothed
   version of  max(p∗p) using L‑BFGS‑B on the soft‑max parametrisation
   p_i = exp(y_i)/∑exp(y).  Several random starts (uniform, Dirichlet,
   perturbed known solution, etc.) are tried and the temperature is
   reduced gradually (τ‑annealing).  The best exact score seen is kept.

2. **Refinement** – after the L‑BFGS phase we run a short hill‑climbing
   loop that repeatedly moves a tiny amount of mass from one index to
   another and accepts the move if the exact ``evaluate_sequence`` value
   improves.  This helps to escape the last plateaus of the smoothed
   optimisation.

3. **Final sanity check** – evaluate the best candidate once more and
   return it as a plain Python list.

The code respects the 1000 s time budget, uses only NumPy (and SciPy if
available) and prints occasional progress messages.
"""

import time
import math
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# optional SciPy import
try:
    from scipy.optimize import minimize
    _SCIPY_AVAILABLE = True
except Exception:                       # pragma: no cover
    _SCIPY_AVAILABLE = False


def _softmax_weights(conv: np.ndarray, tau: float) -> Tuple[np.ndarray, float]:
    """
    Stable soft‑max of the convolution values.
    Returns the probability vector w (summing to 1) and the smooth max Lτ.
    """
    maxc = conv.max()
    if tau <= 0.0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w, maxc
    # stability trick
    ex = np.exp((conv - maxc) / tau)
    sum_ex = ex.sum()
    w = ex / sum_ex
    Ltau = maxc + tau * math.log(sum_ex)
    return w, Ltau


def _weighted_rev(a: np.ndarray, w: np.ndarray, Nfft: int) -> np.ndarray:
    """
    Compute   rev_i = Σ_k w_k a_{k‑i}
    using FFTs (indices outside the support are ignored).
    """
    A_fft = np.fft.rfft(a, n=Nfft)
    W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)
    conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
    # keep the valid part of the linear convolution
    conv_rev = conv_rev[: len(w) + a.shape[0] - 1]
    # the reverse‑lookup indices (see the paper on smooth max)
    idx_start = 2 * a.shape[0] - 2
    rev = conv_rev[idx_start - np.arange(a.shape[0])]
    return rev


def _conv_full(p: np.ndarray, Nfft: int) -> np.ndarray:
    """
    Full convolution p∗p (real FFT based).  Length = 2·len(p)‑1.
    """
    A_fft = np.fft.rfft(p, n=Nfft)
    conv = np.fft.irfft(A_fft * A_fft, n=Nfft)
    return conv[: 2 * p.shape[0] - 1]


def _objective_and_grad(y: np.ndarray, tau: float, Nfft: int) -> Tuple[float, np.ndarray]:
    """
    Smooth‑max objective Lτ(y) and its gradient w.r.t. y.
    The parametrisation is p_i = softmax(y)_i.
    """
    # soft‑max (stable)
    y_max = np.max(y)
    exp_y = np.exp(y - y_max)
    p = exp_y / exp_y.sum()

    # convolution p∗p
    conv = _conv_full(p, Nfft)

    # smooth max
    w, Ltau = _softmax_weights(conv, tau)

    # gradient w.r.t. p (see derivation in the description)
    rev = _weighted_rev(p, w, Nfft)
    grad_p = 2.0 * rev

    # back‑prop through the soft‑max
    dot = np.dot(p, grad_p)                    # Σ_i p_i·grad_p_i
    grad_y = p * (grad_p - dot)                # Jacobian of softmax
    return Ltau, grad_y


def _run_l_bfgs(
    y0: np.ndarray,
    init_tau: float,
    target_sum: float,
    Nfft: int,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
) -> Tuple[np.ndarray, float]:
    """
    Repeatedly minimise the smooth maximum with a decreasing temperature τ.
    Returns the best (sequence, score) found (score = exact evaluate_sequence).
    """
    y = y0.copy()
    tau = max(init_tau, 1e-12)
    tau_decay = 0.80            # slower decay -> finer annealing
    min_tau = 1e-12

    while time.time() < deadline and tau > min_tau:
        # ----------- L‑BFGS on the smooth problem -----------------------
        def fun(y_vec):
            L, g = _objective_and_grad(y_vec, tau, Nfft)
            return L, g

        # allow a reasonably large number of internal calls
        options = {"maxfun": 500, "maxiter": 500, "ftol": 1e-12, "gtol": 1e-12}
        if _SCIPY_AVAILABLE:
            res = minimize(fun, y, method="L-BFGS-B", jac=True, options=options)
            y = res.x
        else:
            # very simple fallback: a few gradient steps
            step = 0.1
            for _ in range(100):
                L, g = fun(y)
                y -= step * g
                # re‑normalise to avoid overflow
                y -= np.max(y)

        # ----------- evaluate exact objective ---------------------------
        y_max = np.max(y)
        exp_y = np.exp(y - y_max)
        p = exp_y / exp_y.sum()
        conv = _conv_full(p, Nfft)
        max_conv = conv.max()
        cur_score = (target_sum ** 2) * max_conv   # exact evaluate_sequence value

        if cur_score < best_score:
            best_score = cur_score
            best_seq = (p * target_sum).copy()
            print(f"[improve] τ={tau:.2e}  score={best_score:.6f}")

        # ----------- schedule ------------------------------------------
        tau *= tau_decay

        # occasional jitter to escape shallow basins
        if np.random.rand() < 0.15:
            y += np.random.randn(y.shape[0]) * 0.02

        # break if we run out of time
        if time.time() >= deadline:
            break

    return best_seq, best_score


def _hill_climb(
    seq: np.ndarray,
    cur_score: float,
    deadline: float,
    step_fraction: float = 0.01,
    max_trials: int = 20000,
) -> Tuple[np.ndarray, float]:
    """
    Very cheap stochastic hill‑climbing: repeatedly move a small amount of
    mass from a random index i to a random index j and keep the move if
    it improves the exact ``evaluate_sequence`` score.
    """
    n = seq.shape[0]
    trials = 0
    while time.time() < deadline and trials < max_trials:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        delta = seq[i] * step_fraction
        if delta <= 0.0:
            continue
        new_seq = seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        # evaluate exact objective (fast enough for a modest number of trials)
        new_score = evaluate_sequence(new_seq.tolist())
        if new_score < cur_score:
            seq = new_seq
            cur_score = new_score
            print(f"[hill] improved to {cur_score:.6f}")
        trials += 1
    return seq, cur_score


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` within the given time budget.

    The function returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # small safety margin

    # ---------------------------------------------------------------
    # Base length – use the length of the known construction if present,
    # otherwise fall back to 2100 (the length that gave the previous best).
    try:
        base = np.array(height_sequence_1, dtype=float)      # type: ignore
        n = base.shape[0]
    except Exception:
        n = 2100
        base = np.ones(n, dtype=float)

    # Normalise to the required ℓ₁‑norm  (the objective is scale‑invariant,
    # but keeping the canonical normalisation makes the maths tidy).
    target_sum = math.sqrt(2.0 * n)
    base = base * (target_sum / base.sum())

    # probability vector (Σp = 1)
    p0 = base / target_sum
    eps_log = 1e-12
    y0 = np.log(p0 + eps_log)

    # initialise best record
    best_seq = base.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] n={n}  initial score={best_score:.6f}")

    # FFT size (next power‑of‑two ≥ 2·n‑1)
    Nfft = 1 << ((2 * n - 1).bit_length())

    # ---------------------------------------------------------------
    # generate a variety of starting points
    starts = []

    # 1) the known solution (if any)
    starts.append(y0)

    # 2) uniform start
    starts.append(np.zeros(n, dtype=float))

    # 3) random Dirichlet (exponential) – fairly dense
    rand_exp = np.random.exponential(scale=1.0, size=n)
    starts.append(np.log(rand_exp + eps_log))

    # 4) sparse Dirichlet (Gamma with shape <1) – encourages many near‑zeros
    rand_gamma = np.random.gamma(shape=0.3, scale=1.0, size=n)
    starts.append(np.log(rand_gamma + eps_log))

    # 5) perturbed known solution
    starts.append(y0 + np.random.randn(n) * 0.05)

    # ---------------------------------------------------------------
    # Run optimisation from each start while respecting the deadline.
    for idx, y_start in enumerate(starts):
        if time.time() >= deadline:
            break
        print(f"[run] start {idx + 1}/{len(starts)}")

        # initial temperature – a few times the smooth max at the start
        p_start = np.exp(y_start - np.max(y_start))
        p_start /= p_start.sum()
        conv0 = _conv_full(p_start, Nfft)
        init_tau = max(conv0.max() * 3.0, 1e-12)

        best_seq, best_score = _run_l_bfgs(
            y0=y_start,
            init_tau=init_tau,
            target_sum=target_sum,
            Nfft=Nfft,
            deadline=deadline,
            best_seq=best_seq,
            best_score=best_score,
        )

    # ---------------------------------------------------------------
    # Final local refinement with a cheap hill‑climbing loop
    if time.time() < deadline - 2.0:
        print("[refine] starting hill‑climbing")
        best_seq, best_score = _hill_climb(best_seq, best_score, deadline)

    # ---------------------------------------------------------------
    # One final sanity check (exact evaluation)
    final_score = evaluate_sequence(best_seq.tolist())
    if final_score < best_score:
        best_score = final_score
        best_seq = best_seq.copy()

    print(f"[done] best score = {best_score:.6f}, length = {best_seq.shape[0]}")
    return best_seq.tolist()
