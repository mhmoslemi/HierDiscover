Projection onto simplex algorithm:

def proj_simplex(v):
    u = np.sort(v)[::-1]
    sv = np.cumsum(u)
    rho = np.where(u > (sv - 1) / np.arange(1, len(v)+1))[0][-1]
    theta = (sv[rho] - 1) / (rho+1)
    w = np.maximum(v - theta, 0)
    return w
```

def _project_simplex(v):
    v = np.asarray(v)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond][-1] - 1
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
propose_candidate
-----------------

Search for a non‑negative step‑function (vector) that minimises the
``evaluate_sequence`` objective.

The algorithm works in three phases:

1. **Warm‑start generation** – several initial probability vectors are
   built (the provided best solution, a uniform vector, Dirichlet samples
   and small perturbations).

2. **Smooth‑max optimisation** – each start is refined with a
   continuation of L‑BFGS on a soft‑max approximation of the max‑convolution.
   This quickly reaches a good local optimum.

3. **Exact max‑convolution sub‑gradient descent** – starting from the best
   solution found above a projected sub‑gradient method directly minimises
   the true objective `2·n·max(p∗p)`.  The gradient of the max‑convolution
   is obtained analytically via a weighted reverse convolution; after each
   gradient step the vector is projected onto the probability simplex.
   Simple back‑tracking line‑search controls the step size.

The best vector is finally scaled to meet the required ℓ₁‑norm
`sum(a)=sqrt(2·n)` and returned as a plain list.
"""

def _softmax_weights(conv: np.ndarray, tau: float):
    """Stable soft‑max of convolution values.
    Returns weight vector w (summing to 1) and the smooth max L_τ."""
    maxc = conv.max()
    if tau <= 0.0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w, maxc
    ex = np.exp((conv - maxc) / tau)
    sum_ex = ex.sum()
    w = ex / sum_ex
    Ltau = maxc + tau * math.log(sum_ex)
    return w, Ltau


def _weighted_rev(a: np.ndarray, w: np.ndarray, Nfft: int) -> np.ndarray:
    """
    Compute rev_i = Σ_k w_k a_{k‑i}
    using FFTs (indices outside the support are ignored).
    """
    # FFT of a (zero‑padded)
    A_fft = np.fft.rfft(a, n=Nfft)
    W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)
    conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
    # keep the valid part (length len(w)+len(a)-1)
    conv_rev = conv_rev[: w.shape[0] + a.shape[0] - 1]
    # rev_i = conv_rev[ 2*len(a)-2 - i ] for i = 0..len(a)-1
    idx_start = 2 * a.shape[0] - 2
    rev = conv_rev[idx_start - np.arange(a.shape[0])]
    return rev


def _conv_full(p: np.ndarray, Nfft: int) -> np.ndarray:
    """Full convolution p∗p (real FFT based). Returns array of length 2·n‑1."""
    A_fft = np.fft.rfft(p, n=Nfft)
    conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * p.shape[0] - 1]
    return conv


def _objective_and_grad(y: np.ndarray, tau: float, Nfft: int):
    """
    Return (L_τ, ∇_y L_τ) for the log‑probability variables y.
    The soft‑max over the convolution implements a smooth approximation of the
    max‑convolution.
    """
    # soft‑max to obtain probabilities
    y_max = np.max(y)
    exp_y = np.exp(y - y_max)
    p = exp_y / exp_y.sum()

    conv = _conv_full(p, Nfft)
    w, Ltau = _softmax_weights(conv, tau)

    rev = _weighted_rev(p, w, Nfft)               # dL/dp = 2·rev
    grad_p = 2.0 * rev

    # chain‑rule for y → p (softmax Jacobian)
    dot = np.dot(p, grad_p)
    grad_y = p * (grad_p - dot)
    return Ltau, grad_y


def _run_l_bfgs(y0: np.ndarray,
                init_tau: float,
                target_sum: float,
                Nfft: int,
                deadline: float,
                best_seq: np.ndarray,
                best_score: float):
    """
    Run a temperature‑annealing L‑BFGS optimisation of the smoothed
    max‑convolution for a single start.  Updates the global best sequence
    whenever an improvement is found.
    """
    y = y0.copy()
    tau = max(init_tau, 1e-12)
    tau_decay = 0.65
    min_tau = 1e-12
    n = best_seq.shape[0]

    while time.time() < deadline and tau > min_tau:
        # ----- L‑BFGS on current temperature ---------------------------------
        def fun(y_vec):
            L, g = _objective_and_grad(y_vec, tau, Nfft)
            return L, g

        options = {'maxfun': 250, 'maxiter': 250, 'ftol': 1e-12, 'gtol': 1e-12}
        if _SCIPY_AVAILABLE:
            res = minimize(fun, y, method='L-BFGS-B', jac=True, options=options)
            y = res.x
        else:
            # fallback: a few Adam‑style steps (very crude)
            grad = np.zeros_like(y)
            for _ in range(30):
                L, grad = _objective_and_grad(y, tau, Nfft)
                y -= 0.01 * grad

        # ----- evaluate exact objective ---------------------------------------
        y_max = np.max(y)
        exp_y = np.exp(y - y_max)
        p = exp_y / exp_y.sum()
        conv = _conv_full(p, Nfft)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv

        if cur_score < best_score:
            best_score = cur_score
            best_seq = (p * target_sum).copy()
            print(f"[improve] τ={tau:.2e}  score={best_score:.6f}")

        # ----- temperature schedule --------------------------------------------
        tau *= tau_decay

        # occasional jitter to escape shallow plateaus
        if random.random() < 0.15:
            y += np.random.randn(y.shape[0]) * 0.02

        if time.time() >= deadline:
            break

    return best_seq, best_score


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # find the last index where u_i > cssv_i / i
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all non‑positive – return the uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _subgradient_refine(p0: np.ndarray,
                        Nfft: int,
                        deadline: float,
                        best_score: float,
                        best_p: np.ndarray) -> (np.ndarray, float):
    """
    Projected sub‑gradient descent on the exact objective
    f(p) = 2·n·max(p∗p).  The method works directly on the probability
    vector p (Σ p = 1, p_i ≥ 0).  It stops when the time budget is exhausted.
    """
    p = p0.copy()
    n = p.shape[0]
    step = 0.5                # initial step size
    min_step = 1e-8
    improvement = True

    while time.time() < deadline and step > min_step:
        conv = _conv_full(p, Nfft)
        max_val = conv.max()
        # mask of active constraints (allow a tiny tolerance)
        eps = max_val * 1e-14 + 1e-16
        active = conv >= max_val - eps
        num_active = active.sum()
        if num_active == 0:
            break
        w = active.astype(float) / float(num_active)

        # gradient of max_conv w.r.t. p  ( = 2·rev )
        rev = _weighted_rev(p, w, Nfft)
        grad = 2.0 * rev

        # ----- backtracking line search ------------------------------------
        best_local = p
        best_local_val = max_val
        step_tmp = step
        success = False
        for _ in range(12):   # limited inner line-search steps
            p_candidate = _project_simplex(p - step_tmp * grad)
            conv_cand = _conv_full(p_candidate, Nfft)
            max_cand = conv_cand.max()
            if max_cand < max_val - 1e-12:
                # improvement found
                best_local = p_candidate
                best_local_val = max_cand
                success = True
                break
            step_tmp *= 0.5
        if not success:
            # no improvement – shrink step and continue
            step *= 0.5
            continue

        # accept the improvement
        p = best_local
        max_val = best_local_val
        # slightly increase step size to accelerate when we are making progress
        step = min(step * 1.1, 1.0)

        cur_score = 2.0 * n * max_val
        if cur_score + 1e-12 < best_score:
            best_score = cur_score
            best_p = p.copy()
            print(f"[subgrad] improved score={best_score:.6f}")

        # early stop if improvement falls below a tiny threshold
        if step < min_step:
            break

    return best_p, best_score


def propose_candidate(seed: int = 42,
                     budget_s: float = 1000,
                     **kwargs) -> List[float]:
    """
    Main entry point.  Returns a list of non‑negative heights that (hopefully)
    yields a lower value of ``evaluate_sequence`` than the previous best
    1.513617.
    """
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # small safety margin

    # ----------------------------------------------------------------------
    # Load the historically best construction if present, otherwise use uniform.
    try:
        base = np.array(height_sequence_1, dtype=float)   # type: ignore
    except Exception:
        base = np.ones(2100, dtype=float)

    n = base.shape[0]
    target_sum = math.sqrt(2.0 * n)               # enforced ℓ₁‑norm

    # Normalise to the required L1‑norm (probability vector with Σ=1)
    base = base * (target_sum / base.sum())
    p0 = base / target_sum                         # probability vector

    # Initialise best record (score = 2·n·max(p∗p))
    best_seq = base.copy()
    best_score = 2.0 * n * (_conv_full(p0, 1 << ((2 * n - 1).bit_length()))).max()
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # FFT size (next power‑of‑two ≥ 2·n‑1)
    Nfft = 1 << ((2 * n - 1).bit_length())

    # ----------------------------------------------------------------------
    # Prepare several starting points.
    starts = []

    # 1) existing construction (probability log‑space)
    eps_log = 1e-12
    y_base = np.log(p0 + eps_log)
    starts.append(y_base)

    # 2) random Dirichlet (exponential samples)
    rand_exp = np.random.exponential(scale=1.0, size=n)
    starts.append(np.log(rand_exp + eps_log))

    # 3) uniform distribution
    starts.append(np.zeros(n, dtype=float))

    # 4) perturbed base
    starts.append(y_base + np.random.randn(n) * 0.05)

    # 5) a few extra Dirichlet draws with different concentrations
    for _ in range(2):
        conc = np.random.uniform(0.2, 5.0)
        gamma_draw = np.random.gamma(shape=conc, scale=1.0, size=n)
        starts.append(np.log(gamma_draw + eps_log))

    # ----------------------------------------------------------------------
    # Run L‑BFGS on each start, followed by sub‑gradient refinement.
    for idx, y_start in enumerate(starts):
        if time.time() >= deadline:
            break
        print(f"[run] start {idx + 1}/{len(starts)}")
        # initial temperature – a few times the current smooth max
        p_start = np.exp(y_start - np.max(y_start))
        p_start /= p_start.sum()
        conv0 = _conv_full(p_start, Nfft)
        init_tau = max(conv0.max() * 2.0, 1e-12)

        # ----- smooth‑max L‑BFGS -------------------------------------------------
        best_seq, best_score = _run_l_bfgs(
            y0=y_start,
            init_tau=init_tau,
            target_sum=target_sum,
            Nfft=Nfft,
            deadline=deadline,
            best_seq=best_seq,
            best_score=best_score,
        )

        # ----- sub‑gradient refinement on the probability vector ---------------
        p_current = best_seq / target_sum
        best_p = p_current.copy()
        best_score, best_p = _subgradient_refine(
            p0=p_current,
            Nfft=Nfft,
            deadline=deadline,
            best_score=best_score,
            best_p=best_p,
        )
        # translate refined probability back to a sequence
        refined_seq = best_p * target_sum
        refined_score = 2.0 * n * (_conv_full(best_p, Nfft)).max()
        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined_seq.copy()
            print(f"[refine] score after sub‑grad = {best_score:.6f}")

    # ----------------------------------------------------------------------
    # Final sanity check with the official evaluator (in case of numerical drift)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        if final_score < best_score:
            best_score = final_score
            best_seq = best_seq.copy()
    except Exception:
        pass

    print(f"[done] best score = {best_score:.6f}, length = {best_seq.shape[0]}")
    return best_seq.tolist()
