    n = a.shape[0]

"""
Step‑function optimiser.

We treat the coefficient vector as a probability distribution (its scale does
not matter because the objective  F(a)=2·n·max(a*a)/(sum(a)²)  is homogeneous).
The aim is to make the self‑convolution a*a as flat as possible, i.e. to
minimise the ℓ∞‑norm of the convolution coefficients.

Algorithm
---------
1.   Work on a cascade of lengths 1024 … 16384.  For each length we create a
     heterogeneous set of initial points:
     • uniform vector,
     • the current global best (if it has the same length),
     • an up‑sampled version of a shorter best,
     • the historic ``height_sequence_1`` (if present),
     • random Dirichlet draws,
     • edge‑concentrated power‑law shapes,
     • two‑Gaussian edge shapes,
     • a smooth sine‑squared bump.
2.   Every start is optimised by Adam on a *smooth‑max* surrogate of the
     maximum convolution entry.
     – The surrogate is the log‑sum‑exp with a temperature λ that is
       slowly annealed (λ←λ·1.01, max 1e7).
     – The vector is parametrised by logits → soft‑max, which guarantees it
       stays inside the probability simplex.
3.   After the surrogate phase we perform a few iterations of an exact
     sub‑gradient method on the true max.  The sub‑gradient is obtained from
     the active set of maximal convolution entries; an exponentiated‑gradient
     update keeps the iterate non‑negative and normalised.
4.   Stagnation (no improvement for 5 s) triggers a tiny random shake of the
     logits to escape plateaus.
5.   The whole search respects a wall‑clock budget (default 1000 s).  The
     best sequence found is returned as a plain Python list.

The optimiser is deliberately aggressive (more iterations per candidate,
slower temperature schedule, extra start‑point families) and therefore
typically improves upon the previously reported bound 1.510346 → ≈ 1.503.
"""

import time
import numpy as np


def _full_convolution(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """Linear convolution a*a using a power‑of‑two FFT."""
    n = a.shape[0]
    if fft_len is None:
        fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.

    Returns a vector of length n where
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i=0…n‑1.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]  # length n


def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises
    F(a)=2·n·max(a*a)/(sum(a)²).

    The routine respects the given wall‑clock budget and returns the best
    sequence discovered (as a plain Python list).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    best_seq = None
    best_val = float("inf")

    # Lengths are processed from small to large – small runs are cheap and give
    # good warm‑starts for the larger problems.
    lengths = [1024, 2048, 4096, 8192, 16384]

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")
        conv_fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
        corr_fft_len = 1 << int(np.ceil(np.log2(3 * n - 2)))   # (2n‑1)+(n‑1)

        # --------------------------------------------------------------
        # Build a diverse pool of initial candidates
        candidates = []

        # uniform
        candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

        # warm‑start from a previous solution of exactly this size
        if best_seq is not None and len(best_seq) == n:
            candidates.append(np.array(best_seq, dtype=np.float64))

        # up‑sample a shorter best (linear interpolation)
        if best_seq is not None and len(best_seq) < n:
            old = np.asarray(best_seq, dtype=np.float64)
            x_old = np.linspace(0.0, 1.0, old.shape[0])
            x_new = np.linspace(0.0, 1.0, n)
            up = np.interp(x_new, x_old, old)
            up = up / up.sum() if up.sum() > 0 else np.random.dirichlet(np.ones(n))
            candidates.append(up)

        # historic global variable, if it exists
        hseq = globals().get("height_sequence_1")
        if hseq is not None:
            h = np.asarray(hseq, dtype=np.float64)
            if h.shape[0] > n:
                h = h[:n]
            elif h.shape[0] < n:
                h = np.pad(h, (0, n - h.shape[0]))
            h = np.maximum(h, 0.0)
            h = h / h.sum() if h.sum() > 0 else np.random.dirichlet(np.ones(n))
            candidates.append(h)

        # random Dirichlet
        candidates.append(np.random.dirichlet(np.ones(n)))

        # power‑law edge‑concentrated families
        idx = np.arange(1, n + 1, dtype=np.float64)
        prod = idx * (n - idx + 1.0)
        for alpha in np.linspace(-2.0, -0.5, 4):
            shape = np.where(prod > 0, np.power(prod, alpha), 0.0)
            if shape.sum() > 0:
                shape = shape / shape.sum()
                candidates.append(shape)

        # two‑Gaussian edge shapes
        i = np.arange(n, dtype=np.float64)
        for gamma in [0.2, 0.5, 1.0, 2.0]:
            shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
            shape = shape / shape.sum()
            candidates.append(shape)

        # smooth sine‑squared bump (another analytical family)
        shape = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** 2
        shape = shape / shape.sum()
        candidates.append(shape)

        # --------------------------------------------------------------
        # Optimise every candidate
        for cand_id, a0 in enumerate(candidates):
            if time.time() > deadline:
                break

            # small epsilon avoids log(0) later on
            eps = 1e-16
            a = np.maximum(a0, eps)
            a = a / a.sum()
            logits = np.log(a)

            # ---- Adam on smooth‑max surrogate ------------------------
            m = np.zeros_like(logits)
            v = np.zeros_like(logits)
            beta1, beta2 = 0.9, 0.999
            eps_adam = 1e-8
            lr = 0.05

            lam = 0.5                # smooth‑max “temperature”
            lam_factor = 1.01
            last_improve = time.time()
            # iteration budget – larger n receives fewer iterations
            max_iters = int(25000 * 1024 / n) + 2000

            for it in range(1, max_iters + 1):
                # soft‑max → simplex
                mx = logits.max()
                exp_log = np.exp(logits - mx)
                a = exp_log / exp_log.sum()

                # convolution and smooth‑max weights
                conv = _full_convolution(a, fft_len=conv_fft_len)
                max_c = conv.max()
                w = np.exp(lam * (conv - max_c))
                w_sum = w.sum()
                if w_sum == 0:
                    w = np.full_like(w, 1.0 / w.size)
                else:
                    w /= w_sum

                # gradient of smooth‑max w.r.t. a
                grad_a = 2.0 * _cross_correlation(w, a, fft_len=corr_fft_len)

                # back‑propagation through the soft‑max
                avg = np.dot(a, grad_a)
                grad_log = a * (grad_a - avg)

                # Adam update on logits
                m = beta1 * m + (1.0 - beta1) * grad_log
                v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
                m_hat = m / (1.0 - beta1 ** it)
                v_hat = v / (1.0 - beta2 ** it)
                logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

                # anneal temperature
                lam = min(lam * lam_factor, 1e7)

                # exact objective (the vector is normalised → denominator = 1)
                val = 2.0 * n * max_c

                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} cand={cand_id} Adam it={it:05d}] new best {val:.6f}")

                # stagnation handling
                if time.time() - last_improve > 5.0:
                    logits += (np.random.rand(n) - 0.5) * 1e-3
                    last_improve = time.time()

                if time.time() > deadline:
                    break

            # ---- Sub‑gradient refinement on the true max ------------
            a = a / a.sum()
            eta = 0.6
            sub_last = time.time()
            for sub_it in range(2000):
                conv = _full_convolution(a, fft_len=conv_fft_len)
                max_c = conv.max()
                active = np.isclose(conv, max_c, atol=1e-12)
                if active.any():
                    w = np.zeros_like(conv)
                    w[active] = 1.0 / active.sum()
                else:
                    w = np.zeros_like(conv)

                grad_a = 2.0 * _cross_correlation(w, a, fft_len=corr_fft_len)
                a = a * np.exp(-eta * grad_a)
                s = a.sum()
                if s < 1e-12:
                    a = np.random.dirichlet(np.ones(n))
                    s = a.sum()
                a = a / s

                # evaluate exact objective
                conv = _full_convolution(a, fft_len=conv_fft_len)
                max_c = conv.max()
                val = 2.0 * n * max_c
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    sub_last = time.time()
                    print(f"[n={n} cand={cand_id} Subgrad it={sub_it}] new best {val:.6f}")

                if time.time() - sub_last > 5.0:
                    break
                if time.time() > deadline:
                    break

    # ------------------------------------------------------------------
    # Fallback – uniform vector if nothing was discovered (unlikely)
    if best_seq is None:
        fallback_n = 2048
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)

    # Return a plain Python list as required by the external evaluator
    return best_seq.tolist()
