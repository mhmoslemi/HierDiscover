def max_convolution(a):
    N = len(a)
    Nfft = 1 << ((2*N -1).bit_length())
    A = np.fft.rfft(a, n=Nfft)
    conv = np.fft.irfft(A * A, n=Nfft)[:2*N-1]
    return conv.max()

Algorithm
---------
* Work in the probability simplex: p_i = a_i / Σa (Σp = 1).
* Minimise  max_k (p∗p)_k  (the L∞ norm of the self‑convolution).

    – Optimise L_τ with L‑BFGS‑B on the unconstrained variables y where
      p = softmax(y).  τ is reduced geometrically (continuation).

def _softmax_weights(conv: np.ndarray, tau: float):
    """Stable soft‑max of convolution values.
    Returns (weights w summing to 1, smooth max Lτ)."""
    maxc = conv.max()
    if tau <= 0.0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w, maxc
    ex = np.exp((conv - maxc) / tau)
    sum_ex = ex.sum()
    w = ex / sum_ex
    Ltau = maxc + tau * math.log(sum_ex)
    return w, Ltau


def _conv_full(p: np.ndarray, Nfft: int):
    """Full convolution p∗p using real FFT. Length = 2·len(p)‑1."""
    A = np.fft.rfft(p, n=Nfft)
    conv = np.fft.irfft(A * A, n=Nfft)[: 2 * p.shape[0] - 1]
    return conv


def _weighted_rev(a: np.ndarray, w: np.ndarray, Nfft: int):
    """
    Compute rev_i = Σ_k w_k·a_{k‑i}  (indices outside the support are ignored)
    using FFTs.
    """
    A_fft = np.fft.rfft(a, n=Nfft)
    W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)
    conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
    conv_rev = conv_rev[: len(w) + a.shape[0] - 1]
    idx_start = 2 * a.shape[0] - 2
    rev = conv_rev[idx_start - np.arange(a.shape[0])]
    return rev


def _objective_and_grad(y: np.ndarray, tau: float, Nfft: int):
    """
    Smooth‑max objective L_τ and its gradient w.r.t. y (softmax parameters).
    Returns (Lτ, ∇_y Lτ).
    """
    # softmax to obtain probability vector p
    y_max = np.max(y)
    exp_y = np.exp(y - y_max)          # stability
    p = exp_y / exp_y.sum()

    # convolution p∗p
    conv = _conv_full(p, Nfft)

    # smooth max
    w, Ltau = _softmax_weights(conv, tau)

    # gradient of Lτ w.r.t. p (2·rev)
    rev = _weighted_rev(p, w, Nfft)
    grad_p = 2.0 * rev

    # transform gradient to y via softmax Jacobian
    dot = np.dot(p, grad_p)           # scalar Σ p_i·grad_p_i
    grad_y = p * (grad_p - dot)
    return Ltau, grad_y


def _run_l_bfgs(y0: np.ndarray, init_tau: float,
                target_sum: float, Nfft: int,
                deadline: float,
                best_seq: np.ndarray,
                best_score: float):

    Run continuation L‑BFGS on the smooth‑max surrogate.
    Returns the best (sequence, score) seen during the run.

    y = y0.copy()

    tau_decay = 0.65
    min_tau = 1e-12

        # ---- L‑BFGS for the current temperature -------------------------
        def fun(y_vec):
            L, g = _objective_and_grad(y_vec, tau, Nfft)
            return L, g

        if _SCIPY_AVAILABLE:
            res = minimize(fun, y, method='L-BFGS-B', jac=True, options=options)

    return best_seq, best_score


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries would become zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution(v: np.ndarray, Nfft: int) -> float:
    """Return max_{k} (v∗v)_k using FFT (no scaling)."""
    A = np.fft.rfft(v, n=Nfft)
    conv = np.fft.irfft(A * A, n=Nfft)[: 2 * v.shape[0] - 1]
    return float(conv.max())

    # 5) a highly sparse start – a few random spikes
    spikes = np.zeros(n)
    pos = np.random.choice(n, size=min(60, n), replace=False)
    spikes[pos] = np.random.rand(len(pos))
    spikes /= spikes.sum()
    starts.append(np.log(spikes + eps_log))

    # --------------------------------------------------------------
    for idx, y_start in enumerate(starts):
        if time.time() >= deadline:
            break
        print(f"[run] start {idx + 1}/{len(starts)}")

        # initial temperature based on current smooth max
        y_tmp = y_start - np.max(y_start)
        p_tmp = np.exp(y_tmp)
        p_tmp /= p_tmp.sum()
        conv0 = _conv_full(p_tmp, Nfft)
        init_tau = max(conv0.max() * 2.0, 1e-12)

        best_seq, best_score = _run_l_bfgs(
            y0=y_start,
            init_tau=init_tau,
            target_sum=target_sum,
            Nfft=Nfft,
            deadline=deadline,
            best_seq=best_seq,
            best_score=best_score,
        )

    # --------------------------------------------------------------
    # At this point best_seq is a vector a (Σa = target_sum).
    # Refine it with a direct projected gradient descent on the exact max.
    p_best = best_seq / target_sum
    max_p_val = _max_convolution(p_best, Nfft)        # max(p∗p)

    # Run the gradient‑descent phase on the remaining time budget
    if time.time() < deadline:
        p_refined, max_p_val_refined = _gradient_descent_max_conv(
            p0=p_best,
            Nfft=Nfft,
            deadline=deadline,
            init_max=max_p_val,
        )
        # Convert back to coefficients a = p·target_sum
        a_refined = p_refined * target_sum
        refined_score = evaluate_sequence(a_refined.tolist())
        if refined_score < best_score:
            best_score = refined_score
            best_seq = a_refined.copy()
            print(f"[refine] improved final score to {best_score:.6f}")

    # --------------------------------------------------------------
    # Final sanity evaluation
    final_score = evaluate_sequence(best_seq.tolist())
    if final_score < best_score:
        best_score = final_score
        best_seq = best_seq.copy()
    print(f"[done] best score = {best_score:.6f}, length = {best_seq.shape[0]}")

    return best_seq.tolist()
