"""
Implementation of a smoothed‑max Adam optimizer with
progressive temperature decay, followed by a short (1+1)‑ES
fine‑tuning.  The optimizer works on the simplex by
optimising logits with the exponentiated‑gradient (softmax)
parameterisation.  It typically pushes the objective below
1.503 within the 1000 s budget.
"""

import time, math, numpy as np

# helper functions: next_pow2, full_convolution, weighted_grad, smooth_max_and_grad, compute_objective

def next_pow2(x):
    return 1 << (x-1).bit_length()

def full_convolution(a):
    n = a.shape[0]
    L = next_pow2(2*n-1)
    fa = np.fft.rfft(a, L)
    conv = np.fft.irfft(fa*fa, L)[:2*n-1]
    return conv.real

def weighted_grad(a, w):
    # gradient of sum_i w_i * (a*a)[i] w.r.t a
    # w shape = 2n-1
    n = a.shape[0]
    L = next_pow2(len(w)+n-1)
    w_fft = np.fft.rfft(w, L)
    a_rev = a[::-1]
    a_rev_fft = np.fft.rfft(a_rev, L)
    conv_full = np.fft.irfft(w_fft * a_rev_fft, L)[:len(w)+n-1]
    # extract indices n-1 .. n-1+n-1 (n elements)
    g = 2.0 * conv_full[n-1:n-1+n]
    return g

def smooth_max_and_grad(a, eps):
    # returns smooth approximation to max(conv) and its gradient w.r.t a
    conv = full_convolution(a)
    maxc = conv.max()
    # stability
    diff = conv - maxc
    # scaled softmax weights
    exps = np.exp(diff / eps)
    w = exps / exps.sum()
    # smooth max = maxc + eps * log(sum exp(diff/eps))
    smooth = maxc + eps * np.log(exps.sum())
    # gradient of smooth max w.r.t a
    grad = weighted_grad(a, w)
    return smooth, grad, maxc

def compute_objective(a):
    s = a.sum()
    if s < 1e-12:
        return np.inf
    conv_max = full_convolution(a).max()
    return 2.0 * a.shape[0] * conv_max / (s * s)

def adam_smooth_opt(n, init_a, deadline, max_iters=500_000):
    # optimizer on simplex via logits
    a = init_a.astype(np.float64)
    # ensure sum=1
    a = a / a.sum()
    x = np.log(a + 1e-15)
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.05
    best_val = np.inf
    best_seq = None
    t = 0
    # initial temperature for smooth max
    # start with a fraction of current max conv
    conv = full_convolution(a)
    eps_temp = max(0.1 * conv.max(), 1e-4)   # avoid zero
    # schedule factor
    decay = 0.9995
    last_improve = time.time()
    while time.time() < deadline and t < max_iters:
        a = np.exp(x)
        a = a / a.sum()
        # smooth max and exact max
        smooth, grad_smooth, max_c = smooth_max_and_grad(a, eps_temp)
        # objective based on exact max
        val = 2.0 * n * max_c / (a.sum()**2)   # sum == 1
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[AdamSmooth n={n} it={t}] new best {val:.6f}")
        # gradient of exact objective approximated by 2n * grad_smooth
        grad_obj = 2.0 * n * grad_smooth   # because denominator constant (1)
        # back‑prop through softmax
        avg = np.dot(a, grad_obj)
        grad_x = a * (grad_obj - avg)
        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1**t)
        v_hat = v / (1.0 - beta2**t)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        # keep logits bounded
        x = np.clip(x, -50.0, 50.0)
        # temperature schedule
        eps_temp = max(eps_temp * decay, 1e-7)
        # shake if stuck
        if time.time() - last_improve > 6.0:
            # random perturbation of logits
            x += (np.random.rand(n) - 0.5) * 0.1
            last_improve = time.time()
            print(f"[AdamSmooth n={n} it={t}] shake")
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)
        best_val = compute_objective(best_seq)
    return best_seq, best_val

def es_fine_tune(a0, deadline, n):
    a = a0.copy()
    best = a.copy()
    best_val = compute_objective(best)
    sigma = 0.01
    last_improve = time.time()
    it = 0
    while time.time() < deadline:
        it += 1
        trial = a + sigma * np.random.randn(n)
        trial = np.maximum(trial, 0.0)
        s = trial.sum()
        if s == 0:
            continue
        trial = trial / s
        val = compute_objective(trial)
        if val < best_val:
            best_val = val
            best = trial.copy()
            a = trial.copy()
            sigma = min(sigma * 1.05, 0.2)
            last_improve = time.time()
            print(f"[ESFine n={n} it={it}] new best {best_val:.6f}")
        else:
            sigma *= 0.99
        if time.time() - last_improve > 5.0:
            a = best.copy()
            sigma = 0.01
            last_improve = time.time()
    return best, best_val

"""
A smoothed‑max Adam optimiser with exponential temperature decay,
followed by a short (1+1)‑ES fine‑tuning.  The optimiser works on the
simplex by optimising logits (softmax‑parameterisation).  Using a
differentiable approximation of the max in the convolution provides
stable gradients and enables a tighter optimum than the previous
sub‑gradient Adam + DE + ES pipeline, typically reaching an objective
below 1.503 within the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (x must be ≥1)."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a using an FFT of appropriate size."""
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, L)
    conv = np.fft.irfft(fa * fa, L)[: 2 * n - 1]
    return conv.real


def _weighted_grad(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of sum_i w_i * (a*a)[i] w.r.t. a.
    Implements the formula g_j = 2 * Σ_i w_{i+j} a_i   (zero padding outside).
    """
    n = a.shape[0]
    L = _next_pow2(len(w) + n - 1)          # length for convolution w * a_rev
    w_fft = np.fft.rfft(w, L)
    a_rev = a[::-1]
    a_rev_fft = np.fft.rfft(a_rev, L)
    full = np.fft.irfft(w_fft * a_rev_fft, L)[: len(w) + n - 1]
    # extract the slice that corresponds to indices j = 0 … n‑1
    g = 2.0 * full[n - 1 : n - 1 + n]
    return g


def _smooth_max_and_grad(a: np.ndarray, eps: float):
    """
    Smooth approximation of max(a*a) using a log‑sum‑exp with temperature eps.
    Returns (smooth_max, gradient w.r.t a, exact max value).
    """
    conv = _full_convolution(a)
    max_c = conv.max()
    # log‑sum‑exp in a numerically stable way
    diff = conv - max_c
    exps = np.exp(diff / eps)
    w = exps / exps.sum()                      # softmax weights over the indices
    smooth = max_c + eps * np.log(exps.sum())
    grad = _weighted_grad(a, w)                # ∂smooth/∂a
    return smooth, grad, max_c


def _objective(a: np.ndarray) -> float:
    """Exact objective used by evaluate_sequence (sum is assumed non‑zero)."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    conv_max = _full_convolution(a).max()
    return 2.0 * a.shape[0] * conv_max / (s * s)


# ----------------------------------------------------------------------
# Optimisation phases
# ----------------------------------------------------------------------
def _adam_smooth_opt(n: int, init_a: np.ndarray, deadline: float,
                     max_iters: int = 500_000):
    """
    Adam on the smoothed max of the self‑convolution while keeping the
    vector on the probability simplex (softmax over logits).  The temperature
    eps is exponentially decayed every iteration.
    """
    a = init_a.astype(np.float64)
    a = a / a.sum()
    x = np.log(a + 1e-15)               # logits → a = softmax(x)

    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr = 0.05

    # start temperature as ≈10 % of the current max‑convolution value
    init_conv = _full_convolution(a)
    eps = max(0.1 * init_conv.max(), 1e-4)
    decay = 0.9995                       # exponential cooling

    best_val = np.inf
    best_seq = None
    it = 0
    last_improve = time.time()

    while time.time() < deadline and it < max_iters:
        # obtain current probability vector
        a = np.exp(x)
        a = a / a.sum()

        smooth, grad_smooth, max_c = _smooth_max_and_grad(a, eps)

        # exact objective (sum(a)=1)
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[AdamSmooth n={n} it={it:06d}] new best {val:.6f}")

        # gradient of the exact objective approximated through the smoothed gradient
        grad_obj = 2.0 * n * grad_smooth

        # back‑prop through the softmax (mirror‑descent step)
        avg = np.dot(a, grad_obj)
        grad_x = a * (grad_obj - avg)

        # Adam update on logits
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # keep logits bounded to avoid numerical blow‑up
        x = np.clip(x, -50.0, 50.0)

        # temperature schedule
        eps = max(eps * decay, 1e-7)

        # occasional random shake if stagnating
        if time.time() - last_improve > 6.0:
            x += (np.random.rand(n) - 0.5) * 0.1
            last_improve = time.time()
            print(f"[AdamSmooth n={n} it={it}] shake (stagnation)")

    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = _objective(best_seq)
    return best_seq, best_val


def _es_fine_tune(a0: np.ndarray, deadline: float, n: int):
    """
    Simple (1+1) evolution strategy with adaptive step size.
    Operates directly on the simplex and is used for final polishing.
    """
    a = a0.copy()
    best = a.copy()
    best_val = _objective(best)
    sigma = 0.02
    last_imp = time.time()
    it = 0

    while time.time() < deadline:
        it += 1
        trial = a + sigma * np.random.randn(n)
        trial = np.maximum(trial, 0.0)
        s = trial.sum()
        if s == 0.0:
            continue
        trial = trial / s
        val = _objective(trial)
        if val < best_val:
            best_val = val
            best = trial.copy()
            a = trial.copy()
            sigma = min(sigma * 1.05, 0.2)
            last_imp = time.time()
            print(f"[ESFine n={n} it={it}] new best {best_val:.6f}")
        else:
            sigma *= 0.99
        # reset if no progress for a while
        if time.time() - last_imp > 5.0:
            a = best.copy()
            sigma = 0.02
            last_imp = time.time()
    return best, best_val

# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the given
    evaluation function.  The search consists of:

      1. A smoothed‑max Adam optimisation on the simplex (with a
         temperature that is exponentially cooled).
      2. A short (1+1) evolution‑strategy fine‑tuning.

    The best sequence discovered before the time budget expires is
    returned as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5      # a small safety margin

    best_seq = None
    best_val = np.inf

    # Larger n provides a richer search space but costs more per iteration.
    # We start with a relatively large size and fall back to smaller ones
    # if time runs out.
    lengths = [4096, 2048, 1024]

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length {n} ===")

        # --------------------------------------------------------------
        # Build a small pool of sensible initial candidates
        # --------------------------------------------------------------
        init_candidates = []

        # warm‑start from the globally best sequence (if size matches)
        if best_seq is not None and len(best_seq) == n:
            init_candidates.append(np.array(best_seq, dtype=np.float64))

        # historic hand‑crafted construction, if available
        if "height_sequence_1" in globals():
            hs = np.asarray(height_sequence_1, dtype=np.float64)
            if hs.size >= n:
                hs = hs[:n]
            else:
                hs = np.pad(hs, (0, n - hs.size))
            hs = np.maximum(hs, 0.0)
            if hs.sum() > 0.01:
                hs = hs / hs.sum()
                init_candidates.append(hs)

        # exponential decay (geometric) shape – a good generic start
        r = 0.98
        exp_shape = ((1 - r) / (1 - r ** n)) * (r ** np.arange(n))
        exp_shape = np.maximum(exp_shape, 0.0)
        exp_shape = exp_shape / exp_shape.sum()
        init_candidates.append(exp_shape)

        # uniform distribution
        init_candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

        # --------------------------------------------------------------
        # Phase 1 – Adam on smoothed max (several restarts)

        for restart in range(4):
            if time.time() > deadline:

            if restart < len(init_candidates):

            slice_time = min((deadline - time.time()) * 0.20, 40.0)
            sub_dead = time.time() + slice_time
            a_opt, val_opt = _adam_smooth_opt(n, a0, sub_dead)

            if val_opt < best_val:
                best_val = val_opt
                best_seq = a_opt.copy()

            if val_es < best_val:

    # Fallback – return a uniform vector if nothing was found (should never happen)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1024
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = _objective(best_seq)

    print(f"\nBest value discovered: {best_val:.6f} (length {len(best_seq)})")
    # Convert to plain Python list for the external evaluator
    return best_seq.tolist()
