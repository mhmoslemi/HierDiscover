def project_simplex(v, s):
    # ensure non-negative sum s
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a non‑negative step‑function that minimises

        2 * n * max(conv(a, a)) / (sum(a) ** 2)

The evaluator returns ``max(conv(a, a))`` when we keep the L1‑norm
fixed to √(2 n), therefore the problem is to flatten the self‑convolution
while staying on the simplex.

Algorithm
---------
* **Initialisation** – Several families of random shapes are tried
  (power‑law, exponential, block‑wise constant, pure random).  Every
  candidate is scaled to the required sum √(2 n).
* **Phase 1 – Smooth‑max Adam** –  The vector is represented as a
  soft‑max of log‑it variables.  A log‑sum‑exp surrogate of the max is
  minimised with Adam (adaptive moments).  The temperature of the
  surrogate is slowly reduced.
* **Phase 2 – Sub‑gradient projection** –  From the Adam result we run a
  few iterations of a pure sub‑gradient method on the true max:
  the index of the peak convolution is found, the exact gradient of that
  entry is computed, a step opposite to the gradient is taken and the new
  point is projected back onto the simplex (non‑negative L1‑norm fixed).
  Small random jitter is added periodically to escape plateaus.
* **Multi‑restart** –  The two‑phase optimiser is executed many times
  (different lengths, different seeds).  The best sequence seen over the
  whole budget is kept.  If a global variable ``height_sequence_1`` is
  present it is used as a warm‑start.
* **Budget handling** –  The whole routine stops as soon as the 1000 s
  budget is exhausted or when a candidate with objective < 1.503 is found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# 1.  Utility – full linear convolution via FFT (fast for all lengths)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of real vectors x and y using real FFT."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    # numerical noise may produce tiny negatives – clip them
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# 2.  Projection of a vector onto the simplex {z≥0, sum(z)=S}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex of size S."""
    if S < 0:
        raise ValueError("Simplex size must be non‑negative")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    ind = np.arange(n) + 1
    # find the largest rho where u_i > cssv_i / i
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero – this happens only for S≈0
        return np.zeros_like(v)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# 3.  Adam optimiser on the log‑it (soft‑max) representation
# ----------------------------------------------------------------------
def _optimise_adam_softmax(
    n: int,
    x0: np.ndarray,
    target_sum: float,
    run_deadline: float,
    *,
    init_tau_factor: float = 0.12,
    min_tau_factor: float = 0.001,
    alpha: float = 0.05,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑it variables.
    Returns the best sequence encountered (as a NumPy array) and its
    evaluate_sequence score.
    """
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    best_seq = None
    best_score = float("inf")
    iteration = 0

    while time.time() < run_deadline:
        # ---- 1. map logits to the simplex (a = target_sum·softmax(x)) ----
        max_x = np.max(x)
        exp_shift = np.exp(x - max_x)                 # stable soft‑max
        p = exp_shift / np.sum(exp_shift)             # probabilities
        a = target_sum * p

        # ---- 2. convolution and (smoothed) max ----
        conv = _full_convolution_fft(a, a)
        max_val = conv.max()
        if max_val < best_score:
            best_score = max_val
            best_seq = a.copy()
            print(f"[{time.time():.2f}] n={n}  new local best (Adam) = {max_val:.6f}")

        # ---- 3. smooth‑max weights (log‑sum‑exp) -----------------------
        tau = max(init_tau_factor * max_val * (0.95 ** iteration),
                  min_tau_factor * max_val)
        scaled = (conv - max_val) / tau
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum == 0.0:                # extreme numerical case – fall back to uniform
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---- 4. gradient w.r.t. a (cross‑correlation) -----------------
        w_rev = w[::-1]
        conv2 = _full_convolution_fft(a, w_rev)          # length = 3n‑2
        idx = (2 * n - 2) - np.arange(n)                 # positions giving Σ a_i w_{i+j}
        grad_a = 2.0 * conv2[idx]

        # ---- 5. gradient w.r.t. logits x (chain rule) ----------------
        dotp = np.dot(p, grad_a)                         # <p,∇a>
        grad_x = target_sum * p * (grad_a - dotp)

        # ---- 6. Adam update -------------------------------------------
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # occasional tiny random jitter to break plateaus
        if iteration % 250 == 0 and np.random.rand() < 0.2:
            x += np.random.randn(n) * 0.001

        # very slow decay of the base learning rate – helps with finetuning
        alpha *= 0.9995

    # ------------------------------------------------------------------
    # Final evaluation with the official evaluator (identical rounding)
    # ------------------------------------------------------------------
    final_score = float("inf")
    if best_seq is not None:
        final_score = evaluate_sequence(best_seq.tolist())
    return best_seq, final_score


# ----------------------------------------------------------------------
# 4.  Sub‑gradient optimiser on the true max (projected on the simplex)
# ----------------------------------------------------------------------
def _optimise_subgradient(
    a0: np.ndarray,
    target_sum: float,
    run_deadline: float,
    *,
    step0: float = 1.0,
    decay: float = 0.99,
) -> tuple[np.ndarray, float]:
    """
    Simple sub‑gradient descent on the exact max‑convolution.
    The iterate is always projected onto the simplex (non‑negative,
    L1‑norm = target_sum).
    """
    a = a0.copy()
    n = a.shape[0]

    best_seq = a.copy()
    best_score = evaluate_sequence(a.tolist())

    iteration = 0
    step = step0

    while time.time() < run_deadline:
        conv = _full_convolution_fft(a, a)
        max_val = conv.max()
        if max_val < best_score:
            best_score = max_val
            best_seq = a.copy()
            print(f"[{time.time():.2f}] n={n}  new local best (SubGrad) = {max_val:.6f}")

        # index of the current peak
        k = int(conv.argmax())

        # gradient of the peak entry w.r.t. a
        grad = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            j = k - i
            if i != j:
                grad[i] = 2.0 * a[j]
            else:
                grad[i] = a[i]          # diagonal term appears only once

        # step size schedule
        iteration += 1
        step = step0 * (decay ** iteration)

        # gradient step and projection onto the simplex
        a = a - step * grad
        a = _project_onto_simplex(a, target_sum)

        # occasional random jitter (helps to climb out of flat regions)
        if iteration % 200 == 0 and np.random.rand() < 0.3:
            jitter = np.random.randn(n) * 0.01 * target_sum
            a = a + jitter
            a = _project_onto_simplex(a, target_sum)

    final_score = evaluate_sequence(best_seq.tolist())
    return best_seq, final_score


# ----------------------------------------------------------------------
# 5.  Random initialiser – several families of shapes
# ----------------------------------------------------------------------
def _random_initial_sequence(n: int) -> np.ndarray:
    """
    Produce a random non‑negative vector of length ``n`` and scale it
    to the required L1‑norm √(2 n).
    """
    choice = np.random.rand()
    if choice < 0.35:                     # power‑law
        p = np.random.uniform(0.25, 0.8)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif choice < 0.60:                   # exponential‑like
        tau = np.random.uniform(0.5, 2.0) * n
        a = np.exp(-np.arange(n, dtype=float) / tau)
    elif choice < 0.80:                   # block‑wise constant
        blocks = np.random.randint(5, 15)
        # random partition of n into ``blocks`` parts
        sizes = np.random.multinomial(n, np.ones(blocks) / blocks)
        heights = np.random.rand(blocks) + 0.1
        a = np.repeat(heights, sizes)[:n]
    else:                                 # plain random
        a = np.random.rand(n) + 1e-5

    a = np.maximum(a, 1e-12)              # avoid exact zeros for log‑softmax
    target = np.sqrt(2.0 * n)
    a = a / a.sum() * target
    return a

    best_seq = None
    best_score = float("inf")

    try:

        if score < best_score:
            best_score = score
            best_seq = hs_arr.copy()
            print(f"[{time.time():.2f}] Warm‑start from global seed = {score:.6f}")
    except Exception:
        pass  # no warm‑start available

    # ------------------------------------------------------------------
    # Main multi‑restart loop.
    # ------------------------------------------------------------------
    while time.time() < deadline:
        remaining = deadline - time.time()

        # --------------------------------------------------------------

        if remaining > 300:

        elif remaining > 120:

        # --------------------------------------------------------------
        # Initialise a candidate.
        # --------------------------------------------------------------
        a_init = _random_initial_sequence(n)

        # Occasionally borrow the current global best (trim / pad)

            else:

        # --------------------------------------------------------------
