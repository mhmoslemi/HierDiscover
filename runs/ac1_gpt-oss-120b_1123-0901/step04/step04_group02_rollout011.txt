def _cross_correlation(w,a,fft_len):
    n = len(a)
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n-1
    return corr[start:start+n]

"""
Propose a step‑function (non‑negative sequence) that minimizes

    F(a) = 2·n·max(a*a) / (sum(a)^2)

where ``a*a`` is the linear convolution of ``a`` with itself.
The algorithm works on the probability simplex (sum(a)=1) because the
objective is scale‑invariant.  It combines:

* an Adam optimiser on a smooth approximation of the max‑convolution,
* a later exact sub‑gradient phase (weight = indicator of the maximal
  convolution entries),
* occasional random “shakes’’ to avoid stagnation,
* a fast hill‑climbing refinement that moves tiny amounts of mass
  between coordinates.

Multiple sequence lengths are tried (1024 … 16384) and the best
candidate found before the time budget expires is returned as a plain
Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.shape[0] - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Compute (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)],  i = 0 … n‑1.
    This is the gradient of Σ_k w_k·(a*a)_k divided by 2.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("v must be 1‑D")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):                     # all zero after projection
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _objective(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (sum(a)=1):
        2·n·max(conv) .
    """
    n = a.shape[0]
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    return float(2.0 * n * conv.max())

# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative sequence with a low evaluation score.
    The function respects ``budget_s`` seconds and returns the best
    sequence (plain Python list) discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.3          # small safety margin

    best_seq = None          # NumPy array, sum == 1
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Lengths to explore – larger n gives more freedom but costs more.
    length_schedule = [1024, 2048, 4096, 8192, 16384]

    for n in length_schedule:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")
        fft_len_conv = _next_pow2(2 * n - 1)
        fft_len_corr = _next_pow2(3 * n)           # safe for cross‑correlation

        # Number of independent random restarts for this length.
        restarts = 6 if n <= 4096 else 4

        for r in range(restarts):
            if time.time() > deadline:
                break

            # -------------------- initialise a (probability vector) --------------------
            if best_seq is not None and best_seq.shape[0] == n:
                a = best_seq.copy()
                print("[restart] warm start from global best")
            elif 'height_sequence_1' in globals():
                cand = np.asarray(globals()['height_sequence_1'], dtype=np.float64)
                if cand.shape[0] >= n:
                    a = cand[:n]
                else:
                    a = np.pad(cand, (0, n - cand.shape[0]))
                a = np.maximum(a, 0.0)
                s = a.sum()
                if s < 1e-9:
                    a = np.random.dirichlet(np.ones(n))
                else:
                    a = a / s
                print("[restart] start from height_sequence_1")
            else:
                a = np.random.dirichlet(np.ones(n))
                print("[restart] random initialisation")

            # -------------------- Adam state --------------------
            m = np.zeros(n, dtype=np.float64)
            v = np.zeros(n, dtype=np.float64)
            beta1, beta2, eps = 0.9, 0.999, 1e-8
            lr = 0.14                     # base learning rate
            lam = 0.5                     # temperature for smooth‑max
            lam_factor = 1.25
            max_iters = 25000
            last_improve = time.time()

            # --------------------------------------------------- Optimisation loop

                max_c = conv.max()

                lam = min(lam * lam_factor, 1e8)

                # ---- exact objective for monitoring (sum(a)=1) ----
                val = _objective(a)

                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} r={r} it={t:05d}] new best {val:.6f}")

                    if best_val < 1.503:
                        print("Target achieved – stopping early.")
                        deadline = time.time()    # force outer loops to finish
                        break

                # ---- stagnation handling (shake) ----
                if time.time() - last_improve > 6.0:
                    noise = (np.random.rand(n) - 0.5) * 1e-2
                    a = _project_simplex(a + noise)
                    last_improve = time.time()
                    print(f"[n={n} r={r} it={t}] shake (stagnation)")

            # end of one restart
        # end of length loop

    # ------------------------------------------------------------------
    # Local random hill‑climbing refinement (if time permits)
    # --------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        a = best_seq.copy()
        n = a.shape[0]
        fft_len_conv = _next_pow2(2 * n - 1)
        print("\n--- Local refinement (tiny mass transfers) ---")

            delta = a[i] * 0.004 * np.random.rand()

                continue
            a_prop = a.copy()
            a_prop[i] -= delta
            a_prop[j] += delta
            a_prop = _project_simplex(a_prop)   # safeguard against rounding
            val = _objective(a_prop)
            if val < best_val:
                best_val = val
                best_seq = a_prop.copy()
                a = a_prop
                print(f"[refine] improved to {val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform vector if nothing succeeded.
    if best_seq is None:
        n_fallback = 2048
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        best_val = _objective(best_seq)
        print(f"Fallback uniform sequence, value = {best_val:.6f}")

    # Return as plain Python list.
    return best_seq.tolist()
