"""
Propose a step‑function (non‑negative vector) that minimises
``evaluate_sequence``.

The search combines three ingredients

* **Smooth‑max minimisation** – we minimise a differentiable surrogate
  of ``max(p∗p)`` with respect to the log‑probabilities ``y``.
  A continuation on the temperature ``τ`` (soft‑max temperature) together
  with L‑BFGS‑B (via SciPy) drives the solution toward a true
  minimiser of the exact objective.

* **Diverse restarts** – uniform, random Dirichlet and a “spike” pattern
  are tried for several sequence lengths (the default length of
  ``height_sequence_1`` is used when available).  The best candidate across
  all restarts is kept.

* **Fast local refinement** – after the L‑BFGS phases a cheap
  coordinate‑wise hill‑climber (randomly transfers a small amount of mass
  between two positions) is run for a few hundred iterations.  Moves that
  improve the true ``evaluate_sequence`` score are accepted.

The routine is time‑bounded (default 1000 s).  All intermediate results are
kept in ``float`` lists, which satisfy the security checks of
``evaluate_sequence``.  The final return value is a plain ``list[float]``.
"""

import time
import math
import numpy as np
from typing import List, Tuple

# Import SciPy if it is present – it gives a much faster L‑BFGS implementation.
try:
    from scipy.optimize import minimize
    _SCIPY = True
except Exception:   # pragma: no cover
    _SCIPY = False


def _softmax(y: np.ndarray) -> np.ndarray:
    """Return soft‑max of ``y`` as a probability vector (sum = 1)."""
    y_max = np.max(y)
    e = np.exp(y - y_max)
    s = e.sum()
    if s == 0.0:
        # extremely unlikely – fall back to uniform
        return np.full_like(y, 1.0 / y.size)
    return e / s


def _next_pow2(x: int) -> int:
    """Small helper: next power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_full(p: np.ndarray, Nfft: int) -> np.ndarray:
    """
    Full linear convolution p * p using real FFT.
    Result length = 2*len(p) - 1.
    """
    A = np.fft.rfft(p, n=Nfft)
    conv = np.fft.irfft(A * A, n=Nfft)[: 2 * p.size - 1]
    return conv


def _smooth_max_and_grad(y: np.ndarray,
                         tau: float,
                         Nfft: int) -> Tuple[float, np.ndarray]:
    """
    Smooth approximation of the maximum of ``p∗p`` and its gradient w.r.t. y.

    Returns
    -------
    Lτ    – smooth maximum (scalar)
    grad  – gradient ∇_y Lτ (same shape as y)
    """
    p = _softmax(y)

    # convolution p * p
    conv = _conv_full(p, Nfft)

    # stable soft‑max of the convolution values
    maxc = conv.max()
    if tau <= 0.0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        L = maxc
    else:
        ex = np.exp((conv - maxc) / tau)
        sum_ex = ex.sum()
        w = ex / sum_ex
        L = maxc + tau * math.log(sum_ex)

    # gradient w.r.t. p : 2 * reverse‑weighted convolution
    # rev_i = Σ_k w_k * p_{k-i}
    # we compute it with one more FFT
    A = np.fft.rfft(p, n=Nfft)
    W_rev = np.fft.rfft(w[::-1], n=Nfft)
    rev = np.fft.irfft(W_rev * A, n=Nfft)[: p.size]
    grad_p = 2.0 * rev

    # chain rule for the soft‑max (Jacobian of the soft‑max)
    dot = np.dot(p, grad_p)
    grad_y = p * (grad_p - dot)
    return L, grad_y


def _lbfgs_continuation(y0: np.ndarray,
                        init_tau: float,
                        target_sum: float,
                        Nfft: int,
                        time_limit: float,
                        best_seq: np.ndarray,
                        best_score: float) -> Tuple[np.ndarray, float]:
    """
    Repeated L‑BFGS optimisations with a cooling schedule on τ.

    Parameters
    ----------
    y0         – initial log‑probabilities
    init_tau   – starting temperature
    target_sum – the L¹‑norm we finally want (√(2·n))
    Nfft       – FFT length (power of two)
    time_limit – absolute time (seconds) after which we stop
    best_seq   – current best sequence (as numpy array)
    best_score – current best exact score (float)

    Returns
    -------
    best_seq, best_score   – possibly improved best candidate.
    """
    y = y0.copy()
    tau = max(init_tau, 1e-12)
    tau_decay = 0.6          # a little more aggressive than the reference code
    min_tau = 1e-12

    while time.time() < time_limit and tau > min_tau:
        # ----------- L‑BFGS for the current τ -----------------
        def fun(vec):
            L, g = _smooth_max_and_grad(vec, tau, Nfft)
            return L, g

        if _SCIPY:
            # modest limits – we only need a few good-ish steps per τ
            opts = {"maxfun": 500, "maxiter": 500, "ftol": 1e-12, "gtol": 1e-12}
            res = minimize(fun, y, method="L-BFGS-B", jac=True, options=opts)
            y = res.x
        else:
            # fallback: a tiny hand‑rolled gradient step
            L, g = fun(y)
            y -= 0.1 * g   # very small step because we have no line‑search

        # ------------- evaluate exact objective -----------------
        p = _softmax(y)
        a = p * target_sum               # the actual candidate
        cur_score = evaluate_sequence(a.tolist())
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[improve] τ={tau:.2e}  score={best_score:.6f}")

        # ------------- temperature update & jitter -------------
        tau *= tau_decay
        if np.random.rand() < 0.12:      # occasional random push
            y += 0.01 * np.random.randn(y.size)

    return best_seq, best_score


def _local_hill_climb(p: np.ndarray,
                      target_sum: float,
                      time_limit: float,
                      best_seq: np.ndarray,
                      best_score: float) -> Tuple[np.ndarray, float]:
    """
    Very cheap coordinate‑wise hill‑climber.  Works directly on the
    probability vector (sum = 1).  At each iteration a random pair of
    indices (i, j) is chosen; a small chunk ``δ`` of mass is moved from i
    to j and the new exact score is evaluated.  Accept only improving
    moves.
    """
    n = p.size
    y = np.log(p + 1e-12)          # work in log‑space to avoid zeros
    while time.time() < time_limit:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # propose a transfer of a modest amount of probability
        delta = np.random.rand() * 0.03   # up to 3 % of the total mass
        pi = np.exp(y[i] - np.max(y))
        pj = np.exp(y[j] - np.max(y))
        # ensure we do not make a negative entry
        if pi < delta:
            delta = pi * 0.9
        y[i] = np.log(max(pi - delta, 1e-20))
        y[j] = np.log(pj + delta)

        # normalise via soft‑max
        p_new = _softmax(y)
        a_new = p_new * target_sum
        cur_score = evaluate_sequence(a_new.tolist())
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a_new.copy()
            print(f"[local‑improve] score={best_score:.6f}")
        else:
            # reject – roll back the change
            y[i] = np.log(pi)
            y[j] = np.log(pj)

    return best_seq, best_score
