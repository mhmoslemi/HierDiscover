from scipy.optimize import linprog
import numpy as np
import scipy.sparse as sp

def solve_lp(a, M):
    n = len(a)
    # Build sparse A of shape (2n-1, n)
    rows = []
    cols = []
    data = []
    for i in range(n):
        # rows where column i appears
        r = np.arange(i, i+n)
        rows.append(r)
        cols.append(np.full(n, i))
        data.append(np.full(n, a[i]))
    rows = np.concatenate(rows)
    cols = np.concatenate(cols)
    data = np.concatenate(data)
    A = sp.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    # Objective: maximize sum b => minimize -1^T b
    c = -np.ones(n)
    bounds = [(0, None)]*n
    res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=bounds, method='highs')
    if not res.success:
        raise RuntimeError("LP failed")
    b = res.x
    return b
```

def eval_and_grad(x, n, fft_len_conv, fft_len_corr, tau):
    max_x = np.max(x)
    exp_x = np.exp(x - max_x)
    a = exp_x / exp_x.sum()
    conv = full_convolution(a, fft_len=fft_len_conv)
    # smoothed max
    max_c = conv.max()
    # compute weights w: softmax over conv* tau
    w = np.exp(tau * (conv - max_c))  # to avoid overflow
    w_sum = np.sum(w)
    w = w / w_sum
    # gradient w.r.t a
    grad_a = 2.0 * cross_correlation(w, a, fft_len=fft_len_corr)

    inner = np.dot(a, grad_a)

"""
Optimise a non‑negative step function (a probability vector) for the
evaluation function

    F(a) = 2·n·max(conv(a,a)) / (sum(a)**2)

The search is based on two ideas:

1.  A fast Adam optimiser on the soft‑max logits (the code from the
    previous run).  This provides a good “continuous” descent.
2.  After a period of stagnation we solve a *linear programme* that,
    for the current vector a, finds a new vector b maximising its mass
    while keeping the convolution a★b bounded by the current maximal
    convolution M = max(conv(a,a)).  Mixing a with b (with a short line‑
    search on the mixing weight) strictly reduces M and therefore the
    objective.

The LP constraints are linear because a★b is linear in b.  The LP is
solved with ``scipy.optimize.linprog`` (method *highs*).  The algorithm
alternates the two phases until the 1000‑second budget expires.  The
best sequence encountered (as a flat Python list) is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions for convolution / cross‑correlation (FFT based)
# ----------------------------------------------------------------------
def full_convolution(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)−1) using the FFT."""
    n = a.shape[0]
    if fft_len is None:
        fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


def cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) equal to
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


# ----------------------------------------------------------------------
# Linear programme: maximise Σ b_j   subject to   a★b ≤ M ,  b ≥ 0
# ----------------------------------------------------------------------
def solve_lp(a: np.ndarray, M: float):
    """
    Solve the LP described above and return the optimal vector b.
    The constraint matrix is built as a sparse CSR matrix; the LP
    uses SciPy's HiGHS interior‑point solver.
    """
    from scipy.optimize import linprog
    import scipy.sparse as sp

    n = a.shape[0]
    rows = []
    cols = []
    data = []

    # each column i contributes a[i] to rows i … i+n‑1
    for i in range(n):
        rows.append(np.arange(i, i + n, dtype=np.int32))
        cols.append(np.full(n, i, dtype=np.int32))
        data.append(np.full(n, a[i], dtype=np.float64))

    rows = np.concatenate(rows)
    cols = np.concatenate(cols)
    data = np.concatenate(data)

    A = sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

    # maximise Σ b_j  →  minimise −Σ b_j
    c = -np.ones(n, dtype=np.float64)
    ub = np.full(2 * n - 1, M, dtype=np.float64)
    bounds = [(0, None)] * n

    res = linprog(
        c,
        A_ub=A,
        b_ub=ub,
        bounds=bounds,
        method="highs",
        options={"presolve": True},
    )
    if not res.success:
        # In the very unlikely event of failure fall back to a zero vector.
        return np.zeros_like(a)
    return res.x

        # ---------- initialise logits -----------------------------------
