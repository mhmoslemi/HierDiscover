"""
Optimisation by multi‑restart smooth‑max L‑BFGS followed by a fast
random‑sampling refinement.

* p = a / Σa lives on the probability simplex (Σp = 1, p ≥ 0).
* The true objective is 2·n·max(p∗p); minimising the smooth
  approximation Lτ(p)=τ·log∑exp((p∗p)/τ) gives a usable gradient.
* Several initial points (uniform, Dirichlet with different
  concentrations, a known construction) are optimised with
  L‑BFGS‑B; the schedule for τ is slower (τ←0.8·τ) and each inner
  optimisation is allowed many more calls.
* After the deterministic phase the remaining time budget is spent
  on cheap random Dirichlet samples; the best candidate is kept.
* The best vector is finally scaled back to the original L1‑norm and
  returned as a plain Python list.
"""

import time
import math
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Helper functions – all are top‑level, no closures.
# ----------------------------------------------------------------------
def _softmax_weights(conv: np.ndarray, tau: float):
    """Stable soft‑max of convolution values."""
    maxc = conv.max()
    if tau <= 0.0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w, maxc
    ex = np.exp((conv - maxc) / tau)
    sum_ex = ex.sum()
    w = ex / sum_ex
    Ltau = maxc + tau * math.log(sum_ex)
    return w, Ltau


def _conv_full(p: np.ndarray, Nfft: int):
    """Full (linear) convolution p∗p using FFTs."""
    A = np.fft.rfft(p, n=Nfft)
    conv = np.fft.irfft(A * A, n=Nfft)[: 2 * p.shape[0] - 1]
    return conv


def _weighted_rev(a: np.ndarray, w: np.ndarray, Nfft: int):
    """Compute rev_i = Σ_k w_k a_{k‑i} using a single FFT."""
    # FFT of a (zero padded)
    A = np.fft.rfft(a, n=Nfft)
    # FFT of reversed weight vector
    W_rev = np.fft.rfft(w[::-1], n=Nfft)
    conv_rev = np.fft.irfft(W_rev * A, n=Nfft)[: len(w) + a.shape[0] - 1]
    idx_start = 2 * a.shape[0] - 2
    rev = conv_rev[idx_start - np.arange(a.shape[0])]
    return rev


def _objective_and_grad(y: np.ndarray, tau: float, Nfft: int):
    """
    Returns (Lτ, ∇_y Lτ) for the soft‑max smooth maximum.
    y are the log‑probability parameters (softmax yields p).
    """
    y_max = np.max(y)
    exp_y = np.exp(y - y_max)          # stable soft‑max numerator
    p = exp_y / exp_y.sum()

    conv = _conv_full(p, Nfft)
    w, L = _softmax_weights(conv, tau)

    rev = _weighted_rev(p, w, Nfft)
    grad_p = 2.0 * rev
    dot = np.dot(p, grad_p)
    grad_y = p * (grad_p - dot)
    return L, grad_y


def _run_lbfgs(y0: np.ndarray, init_tau: float,
               target_sum: float, Nfft: int,
               deadline: float,
               best_seq: np.ndarray,
               best_score: float) -> (np.ndarray, float):
    """
    Optimise the smooth maximum for a single starting point.
    Returns the best sequence (as a NumPy array) and its objective.
    """
    y = y0.copy()
    tau = max(init_tau, 1e-12)
    tau_decay = 0.80                # slower decay – more fine tuning
    min_tau = 1e-12

    # import minimise once
    from scipy.optimize import minimize

    while time.time() < deadline and tau > min_tau:
        # --------------------------------------------------------------
        # L‑BFGS on the current temperature
        # --------------------------------------------------------------
        def fun(vec):
            L, g = _objective_and_grad(vec, tau, Nfft)
            return L, g

        # give the optimiser a fairly generous budget
        opts = {'maxfun': 1000, 'maxiter': 1000,
                'ftol': 1e-12, 'gtol': 1e-12}
        res = minimize(fun, y, method='L-BFGS-B', jac=True,
                       options=opts)
        y = res.x

        # --------------------------------------------------------------
        # Evaluate the *exact* objective (no smoothing)
        # --------------------------------------------------------------
        exp_y = np.exp(y - np.max(y))
        p = exp_y / exp_y.sum()
        conv = _conv_full(p, Nfft)
        max_conv = conv.max()
        cur_score = (target_sum ** 2) * max_conv   # identical to evaluate_sequence

        if cur_score < best_score:
            best_score = cur_score
            best_seq = (p * target_sum).copy()
            print(f"[improve] τ={tau:.2e}  score={best_score:.6f}")

        # --------------------------------------------------------------
        # Temperature schedule & tiny random perturbations
        # --------------------------------------------------------------
        tau *= tau_decay
        if np.random.rand() < 0.25:
            y += np.random.randn(y.shape[0]) * 0.02

        if time.time() >= deadline:
            break

    return best_seq, best_score

    """
    Compute rev_i = Σ_k w_k a_{k‑i} for all i using a single FFT.
    `w` is the weight vector from the soft‑max.
    """
    A = np.fft.rfft(a, n=Nfft)
    W_rev = np.fft.rfft(w[::-1], n=Nfft)
    conv_rev = np.fft.irfft(W_rev * A, n=Nfft)[: len(w) + a.shape[0] - 1]
    idx_start = 2 * a.shape[0] - 2
    rev = conv_rev[idx_start - np.arange(a.shape[0])]
    return rev


def _objective_and_grad(y: np.ndarray, tau: float, Nfft: int):
    """
    Smooth‑max value and its gradient w.r.t. log‑probability parameters y.
    Returns (Lτ, ∇_y Lτ).  The soft‑max turns y into a probability vector p.
    """
    y_max = np.max(y)
    exp_y = np.exp(y - y_max)               # stable exponentiation
    p = exp_y / exp_y.sum()

    conv = _conv_full(p, Nfft)
    w, L = _softmax_weights(conv, tau)

    rev = _weighted_rev(p, w, Nfft)
    grad_p = 2.0 * rev
    dot = np.dot(p, grad_p)
    grad_y = p * (grad_p - dot)
    return L, grad_y


def _run_lbfgs(y0: np.ndarray, init_tau: float,
               target_sum: float, Nfft: int,
               deadline: float,
               best_seq: np.ndarray,
               best_score: float) -> (np.ndarray, float):
    """
    Optimise the smooth‑max from a given start y0.
    Returns the best sequence (as a NumPy array) and its exact score.
    """
    from scipy.optimize import minimize

    y = y0.copy()
    tau = max(init_tau, 1e-12)
    tau_decay = 0.80          # slower decay for finer convergence
    min_tau = 1e-12

    while time.time() < deadline and tau > min_tau:
        # ---- L‑BFGS on the current temperature ---------------------------------
        def fun(vec):
            L, g = _objective_and_grad(vec, tau, Nfft)
            return L, g

        opts = {'maxfun': 1000, 'maxiter': 1000,
                'ftol': 1e-12, 'gtol': 1e-12}
        res = minimize(fun, y, method='L-BFGS-B', jac=True, options=opts)
        y = res.x

        # ---- Exact objective ----------------------------------------------------
        exp_y = np.exp(y - np.max(y))
        p = exp_y / exp_y.sum()
        max_conv = _conv_full(p, Nfft).max()
        cur_score = (target_sum ** 2) * max_conv   # identical to evaluate_sequence

        if cur_score < best_score:
            best_score = cur_score
            best_seq = (p * target_sum).copy()
            print(f"[improve] τ={tau:.2e}  score={best_score:.6f}")

        # ---- Temperature schedule & jitter --------------------------------------
        tau *= tau_decay
        if np.random.rand() < 0.25:
            y += np.random.randn(y.shape[0]) * 0.02

        if time.time() >= deadline:
            break

    return best_seq, best_score


# ----------------------------------------------------------------------
# Entry point required by the evaluator.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``
    within the allotted time budget.
    """
    np.random.seed(seed)
    start = time.time()
    # keep a few seconds as a safety margin
    deadline = start + budget_s - 5.0

    # ------------------------------------------------------------------
    # Load a known construction (if present); otherwise use a uniform vector.
    # ------------------------------------------------------------------
    try:
        base = np.array(height_sequence_1, dtype=float)     # type: ignore
    except Exception:
        base = np.ones(2100, dtype=float)

    n = base.shape[0]
    target_sum = math.sqrt(2.0 * n)

    # Normalise to the required L¹‑norm ( Σa = √(2 n) ).
    base = base * (target_sum / base.sum())

    # ------------------------------------------------------------------
    # Initialise the record of the best solution.
    # ------------------------------------------------------------------
    best_seq = base.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # FFT size: next power‑of‑two ≥ 2·n‑1 → exact linear convolution.
    Nfft = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------------
    # Prepare a collection of diverse starting points (log‑probability vectors).
    # ------------------------------------------------------------------
    eps_log = 1e-12
    starts = []

    # 1) the supplied construction
    p0 = base / target_sum
    starts.append(np.log(p0 + eps_log))

    # 2) pure uniform vector
    starts.append(np.zeros(n, dtype=float))

    # 3‑6) Dirichlet samples with different concentration parameters
    for alpha in (0.3, 0.7, 2.0, 5.0):
        gamma = np.random.gamma(shape=alpha, scale=1.0, size=n)
        starts.append(np.log(gamma + eps_log))

    # 7) slightly perturbed version of the known construction
    starts.append(np.log(p0 + eps_log) + np.random.randn(n) * 0.05)

    # ------------------------------------------------------------------
    # Deterministic optimisation phase (L‑BFGS from every start).
    # ------------------------------------------------------------------
    for idx, y_start in enumerate(starts):
        if time.time() >= deadline:
            break
        print(f"[run] start {idx + 1}/{len(starts)}")

        # Reasonable initial temperature for this start
        p_start = np.exp(y_start - np.max(y_start))
        p_start = p_start / p_start.sum()
        init_tau = max(_conv_full(p_start, Nfft).max() * 2.0, 1e-12)

        best_seq, best_score = _run_lbfgs(
            y0=y_start,
            init_tau=init_tau,
            target_sum=target_sum,
            Nfft=Nfft,
            deadline=deadline,
            best_seq=best_seq,
            best_score=best_score,
        )

    # ------------------------------------------------------------------
    # Random‑sampling refinement for the remaining wall‑clock time.
    # ------------------------------------------------------------------
    while time.time() + 0.05 < deadline:
        # Mix two Dirichlet concentration regimes to diversify the search.
        alpha = 0.3 if np.random.rand() < 0.5 else 2.0
        gamma = np.random.gamma(shape=alpha, scale=1.0, size=n)
        p = gamma / gamma.sum()
        cur_score = (target_sum ** 2) * _conv_full(p, Nfft).max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = (p * target_sum).copy()
            print(f"[rand] new best {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check with the official evaluator.
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq.tolist())
    if final_score < best_score:
        best_score = final_score
        best_seq = best_seq.copy()

    print(f"[done] best score = {best_score:.6f}, length = {best_seq.shape[0]}")
    # Convert to plain Python list as required by the evaluator.
    return best_seq.tolist()
