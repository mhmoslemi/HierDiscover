"""
Hybrid Linear‑Programming / Exponential‑Mirror optimizer.

The routine works on a single length (default 2500) but will try a few
different lengths while respecting the global time budget.  For each length
we start from a simplex‑point (sum = sqrt(2·n)) and iteratively improve the
upper bound M = max (f * f) by solving a small linear programme:

    maximise   Σ g_j                (the total mass)
    subject to  (f * g)_k ≤ M  for all k  (no convolution entry exceeds the
                current bound)
                g_j ≥ 0

The feasible g is then rescaled to the required total mass
S = sqrt(2·n) and mixed with the current f (convex combination).  This
procedure strictly non‑increases M.  After a handful of LP steps we
switch to a few hundred simple exponentiated‑gradient (mirror‑descent)
steps on the exact max‑convolution objective to obtain a fine‑tuned
solution.

All numbers are kept non‑negative, the sum is fixed to S, and the final
list of floats is returned.  The implementation uses only NumPy,
SciPy (linprog with the HiGHS solver) and the provided ``evaluate_sequence``.
"""

import time
import numpy as np
from scipy import sparse, optimize

# --------------------------------------------------------------
def _full_convolution(a: np.ndarray, size: int = None) -> np.ndarray:
    """Linear self‑convolution a*a via FFT (fast for n≈2500)."""
    n = a.shape[0]
    if size is None:
        size = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return conv


def _build_conv_matrix(f: np.ndarray) -> sparse.csc_matrix:
    """
    Build the (2n‑1) × n matrix A such that  A @ g = f * g  (linear convolution).
    Each column i contains the vector f shifted by i positions.
    """
    n = f.shape[0]
    # every diagonal d (0 ≤ d < n) is the constant vector f[d]
    diags = [np.full(n, f[d]) for d in range(n)]
    offsets = np.arange(n)           # d = row – col
    A = sparse.diags(diags, offsets, shape=(2 * n - 1, n), format="csc")
    return A


def _lp_step(f: np.ndarray, M: float, S: float) -> np.ndarray | None:
    """
    Solve the LP that maximises Σ g_j under (f * g) ≤ M.
    Returns the rescaled feasible vector g (sum = S) or None on failure.
    """
    n = f.shape[0]
    A = _build_conv_matrix(f)                 # (2n‑1) × n
    b = np.full(2 * n - 1, M)                # RHS for all convolution constraints
    c = -np.ones(n)                          # maximise Σg  ⇔ minimise −Σg
    bounds = [(0.0, None)] * n                # g_j ≥ 0

    # use HiGHS (fast, supports sparse A)
    res = optimize.linprog(
        c, A_ub=A, b_ub=b, bounds=bounds,
        method="highs", options={"presolve": True, "time_limit": 30}
    )
    if not res.success:
        return None
    g = res.x
    total = g.sum()
    if total == 0.0:
        return None
    # rescale to the required total mass S = √(2·n)
    g = g * (S / total)
    return g


def propose_candidate(seed: int = 42, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    Returns the best sequence found before the time budget expires.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.2    # tiny safety margin

    best_val = float("inf")
    best_seq = None

    # ------------------------------------------------------------------
    # lengths to explore – short runs for small n, longer runs for the target n
    lengths = [2000, 2500, 3000, 3500]

    for n in lengths:
        if time.time() >= deadline:
            break
        S = np.sqrt(2.0 * n)                    # required ℓ₁‑norm

        # number of independent restarts for this length
        restart = 0
        while time.time() < deadline and restart < 12:
            # ---------- initialise f ------------------------------------
            if restart == 0:
                # uniform start (the obvious feasible point)
                f = np.full(n, S / n)
            else:
                # Dirichlet random start, then scale to total mass S
                f = np.random.dirichlet(np.ones(n))
                f = f * S

            # ---------- inner LP‑based improvement loop ----------------
            for it_lp in range(30):
                if time.time() >= deadline:
                    break

                conv = _full_convolution(f)          # exact linear self‑conv
                M = conv.max()
                val = 2.0 * n * M / (S * S)           # same as evaluate_sequence()
                if val < best_val:
                    best_val = val
                    best_seq = f.copy()
                    print(f"[n={n} restart={restart} LP‑it={it_lp}] new best = {val:.6f}")

                # solve the LP, obtain a feasible g and blend it in
                g = _lp_step(f, M, S)
                if g is None:
                    # LP failed – abort this restart
                    break
                # convex combination; t controls step size (chosen empirically)
                t = 0.18
                f = (1.0 - t) * f + t * g

            # ---------- optional fine‑tuning with mirror‑descent --------
            # after the LP stage the bound usually stabilises; a few
            # exponentiated‑gradient steps can squeeze out a bit more.
            lr = 0.07
            for it_md in range(250):
                if time.time() >= deadline:
                    break
                conv = _full_convolution(f)
                k0 = int(conv.argmax())                # index where the max occurs
                # gradient of max‑convolution w.r.t. f (hard max)
                grad = np.zeros_like(f)
                lo = max(0, k0 - (n - 1))
                hi = min(k0, n - 1)
                i = np.arange(lo, hi + 1, dtype=int)
                grad[i] = 2.0 * f[k0 - i]             # 2·a_{k0‑i}
                # mirror‑descent (exponential weights) on the simplex
                f = f * np.exp(-lr * grad)
                f_sum = f.sum()
                if f_sum == 0.0:
                    f = np.full(n, S / n)
                else:
                    f = f * (S / f_sum)               # project back to ℓ₁‑norm = S

                # evaluate occasionally
                if it_md % 25 == 0:
                    conv = _full_convolution(f)
                    M = conv.max()
                    val = 2.0 * n * M / (S * S)
                    if val < best_val:
                        best_val = val
                        best_seq = f.copy()
                        print(f"[n={n} restart={restart} MD‑it={it_md}] new best = {val:.6f}")

            restart += 1
            print(f"[n={n}] completed restart {restart}, best so far = {best_val:.6f}")

    # ------------------------------------------------------------------
    # fall back to a simple uniform vector if nothing was found
    if best_seq is None:
        n = 2500
        best_seq = np.full(n, 1.0 / n)

    print(f"Final best objective = {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
