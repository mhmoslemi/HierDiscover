def solve_lp_for_b(a, M):
    n = len(a)
    # Build sparse matrix A shape (2n-1, n)
    rows = []
    cols = []
    data = []
    for k in range(2*n-1):
        # a_{k-j} non-zero for j s.t. 0 <= k-j < n => j in [k-n+1, k]
        j_start = max(0, k - (n-1))
        j_end = min(n-1, k)  # inclusive
        for j in range(j_start, j_end+1):
            i = k - j
            rows.append(k)
            cols.append(j)
            data.append(a[i])
    A = csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    # bound each constraint <= M
    ub = np.full(2*n-1, M)
    # Solve: max sum b => min -sum b
    c = -np.ones(n)
    # linprog expects A_ub x <= b_ub
    res = linprog(c, A_ub=A, b_ub=ub, bounds=[(0, None)]*n, method='highs', options={'presolve': True})
    if not res.success:
        return None
    return res.x

def alternating_optimize(a, target_sum, run_deadline, max_iter=20, t=0.2):
    best_seq = a.copy()
    best_score = evaluate_sequence(a.tolist())
    while time.time() < run_deadline:
        for i in range(max_iter):
            conv_a = conv(a,a)
            M = conv_a.max()
            # early exit if M already less than best_score, maybe continue to refine.
            # solve LP
            b = solve_lp_for_b(a, M)
            if b is None:
                break
            sum_b = b.sum()
            if sum_b <= 1e-12:
                break
            # scale to target sum
            scale = target_sum / sum_b
            b = b * scale
            # update
            a = (1 - t) * a + t * b
            # renormalize due to numeric
            a = np.maximum(a, 0)
            a_sum = a.sum()
            if a_sum != 0:
                a = a * (target_sum / a_sum)
            # evaluate
            score = evaluate_sequence(a.tolist())
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"[{time.time():.2f}] n={len(a)}  alt best = {best_score:.6f}")
            # if M already <= best_score maybe break.
        # maybe break entire if time near
        if time.time() >= run_deadline:
            break
    return best_seq, best_score

"""
Propose a non‑negative step‑function that minimises

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The evaluator clamps the sum to √(2 n) ⇒ the objective is simply
max(conv(a, a)).  The search combines three complementary ideas:

1. **Soft‑max Adam descent** –  a = S·softmax(x) enforces non‑negativity
   and the simplex constraint.  A smooth approximation of the max is
   optimised with Adam for a few seconds; this quickly yields a decent
   shape.

2. **Linear‑programming alternating improvement** –  given a, we solve

        max   Σ b_i
        s.t.  a * b ≤ M      (M = max(a*a))
              b_i ≥ 0

   (a * b is the linear convolution).  The LP finds a vector b that
   pushes the total mass as high as possible while respecting the same
   convolution bound M.  After scaling b to the required sum S we set

        a ← (1‑t)·a + t·b  (t≈0.2, gradually decayed)

   Repeating this step (a few LP solves) strictly reduces the max‑convolution.

3. **Fine‑tuned Adam** – a short Adam run after the LP phase refines the
   final shape.

The routine performs many independent restarts (different lengths,
initial power‑law shapes, occasional warm‑starts) within the 1000 s
budget and keeps the best sequence found.
"""

import time
import math
import random
import numpy as np
import scipy.sparse
import scipy.optimize

# ----------------------------------------------------------------------
# Fast full linear convolution (real FFT)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of real vectors x and y via FFT."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    # eliminate tiny negatives caused by numerical noise
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Adam optimiser on the soft‑max (log‑it) parametrisation
# ----------------------------------------------------------------------
def _optimise_adam_softmax(
    n: int,
    x0: np.ndarray,
    target_sum: float,
    run_deadline: float,
    *,
    init_tau_factor: float = 0.12,
    min_tau_factor: float = 0.001,
    alpha: float = 0.05,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smoothed‑max surrogate.
    Returns the best sequence encountered (as a NumPy array) and its
    evaluation score obtained from ``evaluate_sequence``.
    """
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    best_seq = None
    best_score = float("inf")
    iteration = 0

    while time.time() < run_deadline:
        # ---- logits → simplex (a = target_sum·softmax(x)) ----
        max_x = np.max(x)
        exp_shift = np.exp(x - max_x)            # stable softmax
        p = exp_shift / np.sum(exp_shift)        # probabilities, sum = 1
        a = target_sum * p                        # a respects Σa = target_sum

        # ---- current (exact) max of the self‑convolution ----
        conv = _full_convolution_fft(a, a)
        max_val = conv.max()                     # because Σa² = 2n, this equals eval.

        if max_val < best_score:
            best_score = max_val
            best_seq = a.copy()
            print(f"[{time.time():.2f}] n={n}  new local best (Adam) = {max_val:.6f}")

        # ---- smooth‑max weights (temperature τ) ----
        tau = max(init_tau_factor * max_val * (0.95 ** iteration),
                  min_tau_factor * max_val)
        scaled = (conv - max_val) / tau
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---- gradient w.r.t. a (cross‑correlation) ----
        w_rev = w[::-1]
        conv2 = _full_convolution_fft(a, w_rev)          # length = 3n‑2
        idx = (2 * n - 2) - np.arange(n)                # positions needed for ∂/∂a_i Σ a_j w_{i+j}
        grad_a = 2.0 * conv2[idx]

        # ---- chain rule: gradient w.r.t. logits x ----
        dotp = np.dot(p, grad_a)
        grad_x = target_sum * p * (grad_a - dotp)

        # ---- Adam update ----
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # occasional tiny random perturbation to escape plateaus
        if iteration % 250 == 0 and np.random.rand() < 0.2:
            x += np.random.randn(n) * 0.001

        # slow learning‑rate decay
        alpha *= 0.9995

    # Final evaluation using the provided evaluator (identical rounding)
    final_score = evaluate_sequence(best_seq.tolist())
    return best_seq, final_score


# ----------------------------------------------------------------------
# Solve the LP   max Σ b_i   s.t.  a * b ≤ M,  b ≥ 0
# ----------------------------------------------------------------------
def _solve_lp_for_b(a: np.ndarray, M: float) -> np.ndarray | None:
    """
    Given a fixed vector a and a bound M, solve the linear program

        maximize   Σ b_i
        subject to conv(a, b)_k ≤ M   for all k = 0 … 2n‑2
                    b_i ≥ 0

    Returns the optimal b (or None if the solver fails).
    """
    n = a.shape[0]
    # Build the convolution matrix (Toeplitz, lower‑triangular part)
    # A_{k,j} = a_{k-j} for 0 ≤ k-j < n, else 0.
    offsets = np.arange(n)                     # 0 … n‑1
    data = [np.full(n - d, a[d], dtype=float) for d in range(n)]
    A = scipy.sparse.diags(data, offsets, shape=(2 * n - 1, n), format="csr")

    # Linear programme: minimise −Σ b_i  (i.e. maximise Σ b_i)
    c = -np.ones(n, dtype=float)
    b_ub = np.full(2 * n - 1, M * (1.0 + 1e-12), dtype=float)   # tiny slack
    bounds = [(0.0, None) for _ in range(n)]

    res = scipy.optimize.linprog(
        c,
        A_ub=A,
        b_ub=b_ub,
        bounds=bounds,
        method="highs",
        options={"presolve": True},
    )
    if not res.success:
        return None
    return res.x


# ----------------------------------------------------------------------
# Alternating LP‑based improvement
# ----------------------------------------------------------------------
def _alternating_optimize(
    a: np.ndarray,
    target_sum: float,
    run_deadline: float,
    *,
    t: float = 0.2,
    max_iter: int = 15,
) -> tuple[np.ndarray, float]:
    """
    Starting from ``a`` (already normalised to target_sum), repeatedly
    solve the LP that pushes the total mass while respecting the current
    max‑convolution bound M.  The update a ← (1‑t)a + t b
    strictly reduces max(conv(a,a)) and converges within a few dozen
    iterations.
    Returns the best sequence encountered and its max‑convolution value.
    """
    best_seq = a.copy()
    conv = _full_convolution_fft(a, a)
    best_score = conv.max()
    iteration = 0

    while time.time() < run_deadline and iteration < max_iter:
        # current bound
        M = best_score

        # LP step
        b = _solve_lp_for_b(a, M)
        if b is None:
            break
        sum_b = b.sum()
        if sum_b <= 1e-12:
            break

        # scale to the required total mass
        b = b * (target_sum / sum_b)

        # convex combination
        a = (1.0 - t) * a + t * b

        # enforce non‑negativity, clip extreme values and renormalise
        a = np.clip(a, 0.0, 1000.0)
        a_sum = a.sum()
        if a_sum > 0:
            a = a * (target_sum / a_sum)

        # evaluate new max‑convolution (exact, no extra evaluation call)
        conv = _full_convolution_fft(a, a)
        M_new = conv.max()
        if M_new < best_score - 1e-12:
            best_score = M_new
            best_seq = a.copy()
            print(f"[{time.time():.2f}] n={len(a)}  alt new best = {best_score:.6f}")

        # decay the mixing factor to stabilise
        t *= 0.9
        iteration += 1

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative height sequence that minimises the given
    evaluation function.  The routine runs for at most ``budget_s`` seconds
    and returns the best sequence discovered.
    """
    np.random.seed(seed)
    random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2      # safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Warm‑start from a previously known construction, if present.
    # ------------------------------------------------------------------
    try:
        hs = globals().get("height_sequence_1")
        if hs is not None:
            hs_arr = np.array(hs, dtype=float)
            n0 = hs_arr.shape[0]
            target0 = math.sqrt(2.0 * n0)
            hs_arr = np.clip(hs_arr, 0.0, 1000.0)
            if hs_arr.sum() > 0:
                hs_arr = hs_arr * (target0 / hs_arr.sum())
            sc = evaluate_sequence(hs_arr.tolist())
            if sc < best_score:
                best_score = sc
                best_seq = hs_arr.copy()
                print(f"[{time.time():.2f}] Warm‑start score = {sc:.6f}")
    except Exception:
        pass  # silently ignore any problem with warm‑start

    # ------------------------------------------------------------------
    # Main restart loop – each restart receives a time slice.
    # ------------------------------------------------------------------
    while time.time() < deadline:
        remaining = deadline - time.time()

        # Choose a length (broader exploration early on)
        if remaining > 400:
            n = np.random.randint(1500, 3000 + 1)
        elif remaining > 200:
            n = np.random.randint(1000, 2500 + 1)
        else:
            n = np.random.randint(600, 1800 + 1)

        target_sum = math.sqrt(2.0 * n)

        # ----- Initial shape (power‑law + optional taper) -----
        p_exp = np.random.uniform(0.20, 0.80)
        idx = np.arange(1, n + 1, dtype=float)
        init = idx ** (-p_exp)

        if np.random.rand() < 0.30:                      # exponential taper
            tau = np.random.uniform(0.5, 2.0) * n
            init *= np.exp(-idx / tau)

        init += np.random.rand(n) * 1e-6                 # break symmetry
        init = np.maximum(init, 1e-12)                   # avoid exact zero

        # Normalise to the required sum
        init = init / init.sum() * target_sum

        # ----- Allocate a slice of the remaining budget -----
        slice_time = min(12.0, remaining * 0.5)

        # 1) Short Adam phase (≈35 % of the slice)
        adam_deadline = time.time() + slice_time * 0.35
        logits0 = np.log(init + 1e-12)
        seq_adam, _ = _optimise_adam_softmax(
            n,
            logits0,
            target_sum,
            adam_deadline,
            init_tau_factor=0.12,
            min_tau_factor=0.001,
            alpha=0.04,
            beta1=0.9,
            beta2=0.999,
        )
        seq_adam = np.array(seq_adam, dtype=float)
        seq_adam = np.clip(seq_adam, 0.0, 1000.0)
        seq_adam = seq_adam * (target_sum / max(seq_adam.sum(), 1e-12))

        # 2) Alternating LP refinement (≈55 % of the slice)
        alt_deadline = time.time() + slice_time * 0.55
        seq_alt, alt_score = _alternating_optimize(
            seq_adam,
            target_sum,
            alt_deadline,
            t=0.2,
            max_iter=12,
        )

        # 3) Final short Adam fine‑tune (≈10 % of the slice)
        fine_deadline = time.time() + slice_time * 0.10
        logits_fine = np.log(seq_alt + 1e-12)
        seq_final, final_score = _optimise_adam_softmax(
            n,
            logits_fine,
            target_sum,
            fine_deadline,
            init_tau_factor=0.08,
            min_tau_factor=0.0005,
            alpha=0.02,
            beta1=0.9,
            beta2=0.999,
        )

        # Update the global best if improvement was found
        if final_score < best_score:
            best_score = final_score
            best_seq = np.array(seq_final, dtype=float)
            print(f"[{time.time():.2f}] GLOBAL new best = {best_score:.6f} (n={n})")
            if best_score < 1.503:          # target reached – early exit
                break

    # ------------------------------------------------------------------
    # Fallback in the unlikely event that no candidate was generated.
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)

    # The evaluator expects a plain Python list.
    return best_seq.tolist()
