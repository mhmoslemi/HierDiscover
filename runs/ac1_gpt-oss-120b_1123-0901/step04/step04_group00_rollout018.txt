def propose_candidate(seed=0, budget_s=1000):
    import time, numpy as np
    np.random.seed(seed)
    deadline = time.time() + budget_s - 1.0

    best_seq = None
    best_val = float('inf')
    
    lengths = [8192, 4096, 2048] # prioritized longest first
    for n in lengths:
        if time.time() > deadline: break
        print(f"Optimizing n={n}")
        fft_len = 1 << int(np.ceil(np.log2(2*n - 1)))
        # Candidate generator function yields arrays
        candidates = []

        # Dirichlet random few
        for _ in range(5):
            cand = np.random.dirichlet(np.ones(n))
            candidates.append(cand)

        # Power-law edge
        idx = np.arange(n) + 1.0
        prod = idx * (n - idx + 1.0)
        for alpha in np.linspace(-3.0, -0.05, 30):
            shape = np.where(prod>0, prod**alpha, 0.0)
            if np.sum(shape)>0:
                shape = shape/shape.sum()
                candidates.append(shape)
        # Exponential edge
        for beta in np.linspace(0.0005, 0.02, 30):
            i = np.arange(n)
            shape = np.exp(-beta * i) + np.exp(-beta * (n-1-i))
            shape = shape/shape.sum()
            candidates.append(shape)
        # Gaussian edge (gamma)
        for gamma in [0.2, 0.5, 1.0, 2.0, 5.0]:
            i = np.arange(n)
            shape = np.exp(-gamma * (i**2)) + np.exp(-gamma * ((n-1-i)**2))
            shape = shape/shape.sum()
            candidates.append(shape)
        # Mixed shape: (exp(-beta*i) * prod**alpha)
        for alpha in np.linspace(-2.0, -0.2, 10):
            for beta in np.linspace(0.001,0.01,5):
                shape = (np.exp(-beta * i) + np.exp(-beta * (n-1-i))) * np.where(prod>0, prod**alpha,0.0)
                shape = shape/shape.sum()
                candidates.append(shape)
        # Also add previous best if any (resized).
        if best_seq is not None:
            # Resize best_seq to length n (by interpolation or truncation/padding)
            old = np.array(best_seq)
            if len(old) == n:
                cand = old.copy()
            elif len(old) > n:
                # downsample by averaging groups
                factor = len(old) // n
                cand = old[:factor*n].reshape(n, factor).mean(axis=1)
            else:
                # upsample using linear interpolation
                x_old = np.linspace(0,1,len(old))
                x_new = np.linspace(0,1,n)
                cand = np.interp(x_new, x_old, old)
            cand = np.maximum(cand, 0.0)
            s = cand.sum()
            if s < 1e-12:
                cand = np.random.dirichlet(np.ones(n))
            else:
                cand = cand / s
            candidates.append(cand)

        # evaluate each candidate
        for idx_c, a in enumerate(candidates):
            if time.time()>deadline:
                break
            conv = np.fft.irfft(np.fft.rfft(a, fft_len) * np.fft.rfft(a, fft_len), fft_len)[:2*n-1]
            max_c = conv.max()
            val = 2.0 * n * max_c  # sum=1
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[n={n} cand={idx_c}] new best {val:.6f}")

        # After scanning, refine best for this n using local optimization (if time left)
        if time.time() > deadline:
            break
        # start local refinement from best_seq shortened/padded to length n
        if best_seq is not None:
            # ensure length n
            # if best_seq length != n, we need to adjust as earlier.
            a_local = np.array(best_seq)
            if len(a_local) != n:
                # do interpolation scaling
                x_old = np.linspace(0,1,len(a_local))
                x_new = np.linspace(0,1,n)
                a_local = np.interp(x_new, x_old, a_local)
                a_local = np.maximum(a_local, 0.0)
                a_local = a_local / a_local.sum()
            # Exponentiated gradient subgradient loops
            # Use w as indicator of max positions (may be multiple)
            step = 0.5
            last_improve = time.time()
            for sub_it in range(3000):
                # compute conv
                conv = np.fft.irfft(np.fft.rfft(a_local, fft_len) * np.fft.rfft(a_local, fft_len), fft_len)[:2*n-1]
                max_c = conv.max()
                val = 2.0 * n * max_c
                if val < best_val - 1e-9:  # improved
                    best_val = val
                    best_seq = a_local.copy()
                    last_improve = time.time()
                    print(f"[n={n} refine iter={sub_it}] new best {val:.6f}")
                # Determine active indices
                active = np.isclose(conv, max_c, atol=1e-12)
                if not active.any():
                    break
                w = np.zeros_like(conv)
                w[active] = 1.0 / active.sum()
                # gradient: 2 * (a conv w); use cross-correlation
                # cross-correlation via FFT: conv of w with reverse a
                rev_a = a_local[::-1]
                grad = 2.0 * np.fft.irfft(np.fft.rfft(w, fft_len) * np.fft.rfft(rev_a, fft_len), fft_len)[:n]
                # exponentiated update
                a_local = a_local * np.exp(-step * grad)
                s = a_local.sum()
                if s == 0 or np.isnan(s):
                    a_local = np.random.dirichlet(np.ones(n))
                    s = a_local.sum()
                a_local = a_local / s
                if time.time() > deadline:
                    break
                if time.time() - last_improve > 5.0:
                    # maybe random perturb to escape plateau
                    a_local = a_local * np.exp(0.01 * (np.random.rand(n) - 0.5))
                    a_local = a_local / a_local.sum()
                    last_improve = time.time()
        # continue to next length or break

    # if best_seq still None -> fallback uniform with moderate length
    if best_seq is None:
        best_seq = np.full(1024, 1.0/1024.0)
    # Return python list
    return best_seq.tolist()

def adam_optimize(a, n, fft_len, max_iters, lr=0.05):
        logits = np.log(a + 1e-15)
        m = np.zeros_like(logits)
        v = np.zeros_like(logits)
        beta1, beta2, eps = 0.9, 0.999, 1e-8
        lam = 0.5
        lam_factor = 1.02
        best_local_val = float('inf')
        best_local_a = a.copy()
        for t in range(1, max_iters+1):
            # compute a via softmax
            mx = logits.max()
            exp_x = np.exp(logits - mx)
            a = exp_x / exp_x.sum()
            conv = full_convolution(a, fft_len)
            max_c = conv.max()
            w = np.exp(lam * (conv - max_c))
            w_sum = w.sum()
            if w_sum == 0:

            else:

            grad_a = 2.0 * cross_correlation(w, a, fft_len_corr)  # need len for cross corr
            avg = np.dot(a, grad_a)
            grad_log = a * (grad_a - avg)

"""
A fast multi‑stage search for a step‑function with a small value of

        F(a) = 2·n·max(a*a) / (sum(a)²)  ,   a_i ≥ 0 .

The algorithm works in three phases

1. **Broad candidate generation** – for a few chosen lengths (8192, 4096,
   16384, …) we build many simple families (edge‑concentrated power‑law,
   exponential / Gaussian edges, random Dirichlet draws, mixed families).
   All candidates are normalised to a unit sum, their auto‑convolution is
   obtained with a single FFT and the exact objective is evaluated.
   The best candidate over all lengths is kept as a warm start.

2. **Exponent‑power fine‑tuning** – the best candidate is transformed by
   raising it to a power p ∈ [0.2, 5] (followed by re‑normalisation).
   This cheap sweep often moves the solution into a more favourable
   region of the search space.

3. **Local optimisation** – starting from the best candidate we run
   two complementary gradient‑based refinements:

   * an Adam optimiser on a *smooth‑max* surrogate (soft‑max over the
     convolution entries).  The temperature λ is increased geometrically
     so that the surrogate becomes tighter.
   * a sub‑gradient method on the exact max (exponentiated‑gradient
     updates keep the iterate on the probability simplex).

   Random “shakes’’ are injected when progress stalls.

The whole routine respects a 1000 s budget; if time runs out the best
sequence found so far is returned as a plain ``list`` of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power‑of‑two ≥ x."""
    return 1 << int(np.ceil(np.log2(x)))


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2·len(a)-1) using FFT.
    ``fft_len`` must be ≥ 2·len(a)-1 and is usually a power of two.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.shape[0] - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Result length = len(a); for i=0…n‑1

        (w ★ a)[i] = Σ_k w[k]·a[i + k − (n−1)]

    where out‑of‑range indices of ``a`` are treated as zero.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]  # length n


# ----------------------------------------------------------------------
def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact objective for a unit‑sum sequence (sum(a)=1)."""
    conv = _full_convolution(a, fft_len)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
def _generate_candidates(n: int) -> list[np.ndarray]:
    """Return a list of plausible initial shapes of length ``n``."""
    cand = []

    # 1) a few random Dirichlet draws
    for _ in range(8):
        cand.append(np.random.dirichlet(np.ones(n)))

    # 2) uniform vector (useful baseline)
    cand.append(np.full(n, 1.0 / n))

    # 3) edge‑concentrated power‑law shapes
    idx = np.arange(n, dtype=np.float64) + 1.0
    prod = idx * (n - idx + 1.0)               # (i+1)*(n-i)
    for alpha in np.linspace(-4.0, -0.05, 36):  # many negative exponents
        shape = np.where(prod > 0, prod**alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    # 4) simple exponential edge shapes
    for beta in np.linspace(0.0005, 0.025, 30):
        i = np.arange(n, dtype=np.float64)
        shape = np.exp(-beta * i) + np.exp(-beta * (n - 1 - i))
        shape /= shape.sum()
        cand.append(shape)

    # 5) Gaussian edge shapes (exp(‑γ·i²))
    for gamma in np.linspace(0.1, 5.0, 30):
        i = np.arange(n, dtype=np.float64)
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        shape /= shape.sum()
        cand.append(shape)

    # 6) mixed power‑law + exponential edge shapes
    for alpha in np.linspace(-2.5, -0.2, 12):
        for beta in np.linspace(0.001, 0.01, 6):
            shape = prod**alpha * (np.exp(-beta * np.arange(n)) +
                                  np.exp(-beta * (n - 1 - np.arange(n))))
            shape = np.where(shape > 0, shape, 0.0)
            if shape.sum() > 0:
                shape /= shape.sum()
                cand.append(shape)

    return cand


# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   fft_len_conv: int,
                   fft_len_corr: int,
                   max_iters: int,
                   lr: float = 0.05) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on a smooth‑max surrogate of max(conv).
    Returns the best sequence found and its exact objective value.
    """
    a = a0.copy()
    logits = np.log(a + 1e-15)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lam = 0.5
    lam_factor = 1.02

    best_a = a.copy()
    best_val = _objective(a, n, fft_len_conv)

    for t in range(1, max_iters + 1):
        # soft‑max → simplex
        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()

        # convolution and smooth‑max weights
        conv = _full_convolution(a, fft_len_conv)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0:
            w[:] = 1.0 / w.size
        else:
            w /= w_sum

        # gradient of the surrogate (2·(w ★ a))
        grad_a = 2.0 * _cross_correlation(w, a, fft_len_corr)

        # back‑prop through soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # increase temperature → tighter surrogate
        lam = min(lam * lam_factor, 1e6)

        # exact objective
        val = 2.0 * n * max_c  # sum(a)=1
        if val < best_val:
            best_val = val
            best_a = a.copy()

    return best_a, best_val


# ----------------------------------------------------------------------
def _subgradient_optimize(a0: np.ndarray,
                          n: int,
                          fft_len_conv: int,
                          fft_len_corr: int,
                          max_iters: int,
                          eta: float = 0.6) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the true max (exponentiated‑gradient update).
    Returns best sequence encountered and its exact objective value.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a, n, fft_len_conv)

    last_improve = time.time()
    for it in range(max_iters):
        conv = _full_convolution(a, fft_len_conv)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val - 1e-12:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()

        # active set (indices where conv ≈ max_c)
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if not active.any():
            break
        w = np.zeros_like(conv)
        w[active] = 1.0 / active.sum()

        # gradient of max(conv) w.r.t a
        grad_a = 2.0 * _cross_correlation(w, a, fft_len_corr)

        # exponentiated‑gradient step (keeps a ≥ 0)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s <= 0 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        # stagnation handling
        if time.time() - last_improve > 4.0:
            # small random shake
            a = a * np.exp(0.01 * (np.random.rand(n) - 0.5))
            a = a / a.sum()
            last_improve = time.time()

    return best_a, best_val


# ----------------------------------------------------------------------
def _power_transform(a: np.ndarray, p: float) -> np.ndarray:
    """Raise all components to power ``p`` and renormalise (p>0)."""
    if p <= 0:
        return a
    a2 = np.power(a, p)
    s = a2.sum()
    if s <= 0:
        return a
    return a2 / s


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function sequence that minimises
    ``evaluate_sequence``.  The function respects the supplied time budget
    (in seconds) and returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    best_seq = None
    best_val = float("inf")
    best_len = None

    # ------------------------------------------------------------------
    # 1) GLOBAL scan over a few lengths
    lengths = [8192, 16384, 4096, 1024]   # longer lengths first
    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Scanning length n={n} ===")
        fft_len_conv = _next_pow2(2 * n - 1)
        fft_len_corr = _next_pow2(3 * n - 2)

        candidates = _generate_candidates(n)

        # If we already have a good sequence of the same length, add it as a warm start
        if best_seq is not None and len(best_seq) == n:
            candidates.append(np.asarray(best_seq, dtype=np.float64))

        # Evaluate every candidate (early‑exit if time runs out)
        for cid, cand in enumerate(candidates):
            if time.time() > deadline:
                break
            # Force unit sum
            a = cand / cand.sum()
            conv = _full_convolution(a, fft_len_conv)
            val = 2.0 * n * conv.max()
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                best_len = n
                print(f"[n={n} cand={cid:03d}] new best {val:.6f}")

    # ------------------------------------------------------------------
    # If nothing was found, fall back to a modest uniform vector
    if best_seq is None:
        n = 1024
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_len = n
        best_val = _objective(best_seq, n, _next_pow2(2 * n - 1))
        print("Fallback to uniform sequence.")

    # ------------------------------------------------------------------
    # 2) Power‑transform sweep on the current best (cheap local search)
    n = best_len
    fft_len_conv = _next_pow2(2 * n - 1)
    fft_len_corr = _next_pow2(3 * n - 2)

    power_grid = np.concatenate((
        np.linspace(0.2, 1.0, 9, endpoint=False),
        np.linspace(1.0, 5.0, 9)[1:]))

    for p in power_grid:
        if time.time() > deadline:
            break
        a = _power_transform(best_seq, p)
        val = _objective(a, n, fft_len_conv)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[power={p:.2f}] improved to {val:.6f}")

    # ------------------------------------------------------------------
    # 3) Local optimisation – Adam followed by sub‑gradient (time‑aware)
    # Remaining budget determines how many iterations we can afford.
    time_left = deadline - time.time()
    # allocate ~60 % to Adam, ~30 % to sub‑gradient, ~10 % to random shakes
    adam_iters = max(200, int(0.55 * time_left * 1000 / (n * np.log2(n) + 1)))
    subgrad_iters = max(300, int(0.25 * time_left * 500 / (n * np.log2(n) + 1)))

    # Adam phase
    a_adam, val_adam = _adam_optimize(best_seq, n,
                                       fft_len_conv, fft_len_corr,
                                       max_iters=adam_iters,
                                       lr=0.04)
    if val_adam < best_val:
        best_val = val_adam
        best_seq = a_adam.copy()
        print(f"[Adam] improved to {val_adam:.6f}")

    # Sub‑gradient phase
    a_sub, val_sub = _subgradient_optimize(best_seq, n,
                                           fft_len_conv, fft_len_corr,
                                           max_iters=subgrad_iters,
                                           eta=0.5)
    if val_sub < best_val:
        best_val = val_sub
        best_seq = a_sub.copy()
        print(f"[Sub‑grad] improved to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # 4) Random shakes + short Adam passes (quick escape from local minima)
    shake_budget = deadline - time.time()
    shake_iters = 0
    while shake_budget > 2.0:   # at least a couple of seconds per shake
        sigma = 0.02 * (0.5 ** (shake_iters // 2))   # slowly shrink noise
        a_shake = best_seq * np.exp(sigma * (np.random.rand(n) - 0.5))
        a_shake = a_shake / a_shake.sum()

        # a very short Adam refinement (≈200 iters)
        a_tmp, val_tmp = _adam_optimize(a_shake, n,
                                        fft_len_conv, fft_len_corr,
                                        max_iters=200,
                                        lr=0.03)
        if val_tmp < best_val:
            best_val = val_tmp
            best_seq = a_tmp.copy()
            print(f"[Shake {shake_iters}] improved to {val_tmp:.6f}")

        shake_iters += 1
        shake_budget = deadline - time.time()

    # ------------------------------------------------------------------
    # Final sanity – ensure non‑negativity and unit sum (evaluate_sequence
    # does its own clipping, but keeping the vector well‑behaved is safer)
    best_seq = np.maximum(best_seq, 0.0)
    if best_seq.sum() == 0 or np.isnan(best_seq.sum()):
        best_seq = np.random.dirichlet(np.ones(best_seq.shape[0]))
    best_seq = best_seq / best_seq.sum()

    print(f"\nFinished. Best value discovered: {best_val:.6f}")
    return best_seq.tolist()
