        # transfer delta

def generate_candidates(n, best_seq):
    candidates = []
    # best_seq copy if available and length matches
    if best_seq is not None and len(best_seq) == n:
        candidates.append(np.asarray(best_seq, dtype=np.float64))
    # random Dirichlet
    candidates.append(np.random.dirichlet(np.ones(n)))
    # Mixture of best and random (if best_seq available)
    if best_seq is not None and len(best_seq) == n:
        for t in [0.25,0.5,0.75]:
            rnd = np.random.dirichlet(np.ones(n))
            cand = t * np.asarray(best_seq) + (1-t) * rnd
            candidates.append(cand / cand.sum())
    # windows
    i = np.arange(n)
    # Hann
    w_hann = 0.5 - 0.5 * np.cos(2 * np.pi * i / (n-1))
    candidates.append(w_hann / w_hann.sum())
    # Hamming
    w_ham = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n-1))
    candidates.append(w_ham / w_ham.sum())
    # Blackman
    w_black = 0.42 - 0.5 * np.cos(2 * np.pi * i / (n-1)) + 0.08 * np.cos(4 * np.pi * i / (n-1))
    candidates.append(w_black / w_black.sum())
    # Bartlett (triangular)
    w_tri = 1 - np.abs((i - (n-1)/2) / ((n-1)/2))
    candidates.append(w_tri / w_tri.sum())
    # Power-law (edge concentrated)
    prod = (i+1) * (n - i)
    for alpha in np.linspace(-3, -0.2, 10):
        shape = np.power(prod, alpha, where=prod>0, out=np.zeros_like(prod))
        if shape.sum() > 0:
            candidates.append(shape/shape.sum())
    # Power-law centre peaked (positive alpha)
    for alpha in np.linspace(0.2, 3, 10):
        shape = np.power(prod, alpha, where=prod>0, out=np.zeros_like(prod))
        if shape.sum() > 0:
            candidates.append(shape/shape.sum())
    # Double exponential edge
    for gamma in [0.5, 1, 2, 5]:
        shape = np.exp(-gamma * i) + np.exp(-gamma * (n-1-i))
        candidates.append(shape/shape.sum())
    # Gaussian centered
    for sigma in [0.1, 0.2, 0.4]:
        x = (i - (n-1)/2) / (sigma * n)
        shape = np.exp(-0.5 * x**2)
        candidates.append(shape/shape.sum())
    return candidates

"""
Propose a step‑function (non‑negative list) that minimises

        F(a) = 2·n·max(a⨂a) / (sum(a)²)

where ⨂ is the linear convolution.
The algorithm combines a wide variety of hand‑crafted initial families
(edge‑concentrated power laws, classic signal‑processing windows,
gaussian / double‑exponential shapes, random Dirichlet draws) and
refines the most promising candidates with three optimisation phases:

1. Adam on a smooth‑max (log‑sum‑exp) surrogate for the max‑convolution.
2. A sub‑gradient / exponentiated‑gradient phase on the exact max.
3. A cheap stochastic “mass‑transfer” hill‑climb that swaps a tiny
   amount of probability between two positions.

All phases respect the simplex constraint (a_i ≥ 0, Σa_i = 1) and the
global time budget (≈1000 s).  The best sequence found across several
lengths (1024 … 16384) is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Small helper – next power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2·len(a)-1) using FFT.
    The caller must provide a suitable FFT length (power‑of‑2 ≥ 2·n‑1).
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.shape[0] - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) equal to
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def _objective(a: np.ndarray, n: int, fft_len_conv: int) -> float:
    """Exact objective F(a).  a is assumed to be non‑negative."""
    conv = _full_convolution(a, fft_len_conv)
    max_c = conv.max()
    s = a.sum()
    # denominator very close to zero is guarded against in the external evaluator,
    # but we keep the same check here for safety.
    if s < 1e-12:
        return np.inf
    return 2.0 * n * max_c / (s * s)


# ----------------------------------------------------------------------
def _generate_candidates(n: int, best_seq_of_same_len: np.ndarray | None) -> list[np.ndarray]:
    """
    Produce a diverse set of starting vectors (always normalised to sum 1).
    """
    candidates = []

    # 1) warm start from a known good solution of the same length
    if best_seq_of_same_len is not None:
        candidates.append(best_seq_of_same_len.copy())

    # 2) mixture of best and a fresh Dirichlet draw (if we have a best)
    if best_seq_of_same_len is not None:
        for mix in (0.25, 0.5, 0.75):
            rnd = np.random.dirichlet(np.ones(n))
            cand = mix * best_seq_of_same_len + (1.0 - mix) * rnd
            candidates.append(cand / cand.sum())

    # 3) pure random Dirichlet
    candidates.append(np.random.dirichlet(np.ones(n)))

    # 4) classic signal‑processing windows (all normalised)
    i = np.arange(n, dtype=np.float64)
    # Hann
    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    candidates.append(w / w.sum())
    # Hamming
    w = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    candidates.append(w / w.sum())
    # Blackman
    w = (
        0.42
        - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
        + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    )
    candidates.append(w / w.sum())
    # Bartlett (triangular, zero at the ends)
    w = 1.0 - np.abs((i - (n - 1) / 2.0) / ((n - 1) / 2.0))
    candidates.append(w / w.sum())

    # 5) edge‑concentrated power‑law families
    prod = (i + 1.0) * (n - i)
    for alpha in np.linspace(-3.0, -0.2, 9):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            candidates.append(shape / shape.sum())
    # centre‑concentrated power‑law families
    for alpha in np.linspace(0.2, 3.0, 9):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            candidates.append(shape / shape.sum())

    # 6) double‑exponential edges
    for gamma in (0.2, 0.5, 1.0, 2.0, 5.0):
        shape = np.exp(-gamma * i) + np.exp(-gamma * (n - 1 - i))
        candidates.append(shape / shape.sum())

    # 7) centred Gaussian‐like bumps
    centre = (n - 1) / 2.0
    for sigma in (0.05, 0.10, 0.20):
        x = (i - centre) / (sigma * n)
        shape = np.exp(-0.5 * x * x)
        candidates.append(shape / shape.sum())

    # 8) a few random Dirichlet draws for extra diversity
    for _ in range(3):
        candidates.append(np.random.dirichlet(np.ones(n)))

    return candidates


# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_corr: int,
    deadline: float,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max surrogate.
    Returns the best vector found (still on the simplex) and its exact objective.
    """
    a = a0.copy()
    # logits for the soft‑max trick
    logits = np.log(a + 1e-15)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.05

    lam = 0.5
    lam_factor = 1.02

    best_a = a.copy()
    best_val = _objective(a, n, fft_len_conv)

    last_improve = time.time()
    max_iters = int(15000 * 1024 / n) + 2000

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # ----- simplex via soft‑max -----
        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()

        # ----- smooth‑max weight -----
        conv = _full_convolution(a, fft_len_conv)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the surrogate -----
        grad_a = 2.0 * _cross_correlation(w, a, fft_len_corr)

        # back‑prop through soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # temperature schedule – approach the true max
        lam = min(lam * lam_factor, 1e6)

        # ----- exact evaluation -----
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} Adam iter={t:05d}] new local best {val:.6f}")

        # stagnation handling – gentle random shake
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        # early exit if we already beat the current global best by a comfortable margin
        if best_val < global_best_val * 0.99:
            # keep training a few more steps but do not waste too much time
            pass

    return best_a, best_val


# ----------------------------------------------------------------------
def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_corr: int,
    deadline: float,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient / exponentiated‑gradient optimisation on the exact max.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a, n, fft_len_conv)

    eta = 0.6
    last_improve = time.time()
    max_iters = int(4000 * 1024 / n) + 1000

    for it in range(max_iters):
        if time.time() > deadline:
            break

        conv = _full_convolution(a, fft_len_conv)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)

        if not active.any():
            w = np.zeros_like(conv)
        else:
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()

        grad_a = 2.0 * _cross_correlation(w, a, fft_len_corr)

        # exponentiated‑gradient update (keeps us in the simplex)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        val = _objective(a, n, fft_len_conv)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} Subgrad iter={it:05d}] new local best {val:.6f}")

        if time.time() - last_improve > 4.0:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
def _hill_climb_optimize(
    a0: np.ndarray,
    n: int,
    fft_len_conv: int,
    deadline: float,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Very cheap stochastic mass‑transfer hill‑climb.  Proposes tiny
    redistributions of probability mass between two positions.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a, n, fft_len_conv)

    # temperature for occasional uphill moves (simulated‑annealing flavour)
    temp = 1e-3
    start = time.time()

    while time.time() < deadline:
        # pick two distinct indices
        i = np.random.randint(0, n)
        j = np.random.randint(0, n - 1)
        if j >= i:
            j += 1

        # move a small fraction of the mass from i to j
        delta = np.random.rand() * min(a[i] * 0.5, 0.001)
        if delta <= 0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        val = _objective(a_new, n, fft_len_conv)

        # Metropolis acceptance – we accept better moves always,
        # and occasionally a worse move to escape flat plateaus.
        if val < best_val or np.random.rand() < np.exp(-(val - best_val) / temp):
            a = a_new
            if val < best_val:
                best_val = val
                best_a = a_new.copy()
                print(f"[n={n} HillClimb] new local best {val:.6f}")

        # slowly cool the temperature
        temp = max(1e-6, temp * 0.9999)

        # break if we have not improved for a while
        if time.time() - start > 20.0:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative sequence with a low evaluation score.
    The function respects the time budget and returns the best
    sequence found as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # lengths we will explore – longer ones give a better theoretical bound,
    # but they also cost more per iteration.
    lengths = [1024, 2048, 4096, 8192, 16384]

    # pre‑compute suitable FFT lengths
    fft_len_conv_cache = {}
    fft_len_corr_cache = {}
    for n in lengths:
        conv_len = 2 * n - 1
        fft_len_conv_cache[n] = _next_pow2(conv_len)
        fft_len_corr_cache[n] = _next_pow2(conv_len + n - 1)

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")
        fft_len_conv = fft_len_conv_cache[n]
        fft_len_corr = fft_len_corr_cache[n]

        # ------------------------------------------------------------------
        # 1) generate a rich pool of initial candidates
        cand_pool = _generate_candidates(n, best_seq if best_seq is not None and len(best_seq) == n else None)

        # 2) evaluate them quickly and keep the top‑k for refinement
        evaluated = []
        for idx, cand in enumerate(cand_pool):
            if time.time() > deadline:
                break
            # ensure exact simplex property
            cand = np.maximum(cand, 0.0)
            s = cand.sum()
            if s < 1e-12:
                cand = np.random.dirichlet(np.ones(n))
                s = cand.sum()
            cand = cand / s
            val = _objective(cand, n, fft_len_conv)
            evaluated.append((val, cand))
            if val < best_val:
                best_val = val
                best_seq = cand.copy()
                print(f"[n={n} init cand={idx}] new global best {val:.6f}")

        # keep only a few most promising starters
        evaluated.sort(key=lambda x: x[0])
        top_candidates = [c for (_, c) in evaluated[:min(5, len(evaluated))]]

        # ------------------------------------------------------------------
        # 3) refine each selected candidate
        for cid, cand in enumerate(top_candidates):
            if time.time() > deadline:
                break

            # ---- Adam phase ----
            a_adam, val_adam = _adam_optimize(
                cand, n, fft_len_conv, fft_len_corr, deadline, best_val
            )
            if val_adam < best_val:
                best_val = val_adam
                best_seq = a_adam.copy()
                print(f"[n={n} cand={cid} after Adam] new global best {val_adam:.6f}")

            # ---- Sub‑gradient phase ----
            a_sub, val_sub = _subgradient_optimize(
                a_adam, n, fft_len_conv, fft_len_corr, deadline, best_val
            )
            if val_sub < best_val:
                best_val = val_sub
                best_seq = a_sub.copy()
                print(f"[n={n} cand={cid} after Subgrad] new global best {val_sub:.6f}")

            # ---- Hill‑climb phase (lightweight) ----
            a_hc, val_hc = _hill_climb_optimize(
                a_sub, n, fft_len_conv, deadline, best_val
            )
            if val_hc < best_val:
                best_val = val_hc
                best_seq = a_hc.copy()
                print(f"[n={n} cand={cid} after HillClimb] new global best {val_hc:.6f}")

    # ----------------------------------------------------------------------
    # 4) fallback – uniform distribution if nothing succeeded (unlikely)
    if best_seq is None:
        fallback_n = 2048
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)
        print("[fallback] returning uniform vector")

    print(f"\n=== Finished. Best value found: {best_val:.6f} (length {len(best_seq)}) ===")
    return best_seq.tolist()
