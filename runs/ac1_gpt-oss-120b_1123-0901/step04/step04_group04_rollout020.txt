"""
Propose a step‑function (non‑negative vector) that minimises
``evaluate_sequence``.

Algorithm
---------
* The objective after fixing the L1‑norm ( Σa = √(2·n) ) is exactly the
  maximum value of the self‑convolution a*a.
* Starting from the known construction (global ``height_sequence_1``) we
  repeatedly solve the linear programme

        maximise    Σ g_i
        subject to  (a * g)_k ≤ M   for all k,   g_i ≥ 0

  where M = max (a*a).  For a given a the constraints are linear in g,
  therefore an LP (SciPy HiGHS) gives the best possible direction g that
  stays inside the old bound M.
* The feasible vector g is rescaled to the required L1‑norm and a
  Frank–Wolfe step  a←(1‑t)·a + t·g  is performed.  The step size t
  follows the standard schedule t = 2/(iteration+2) but is reduced if
  no improvement is observed.
* The procedure is repeated while time permits.  Small random jitter is
  added when the line‑search fails to make progress, which helps to get
  out of plateaus.
* The best vector found (according to the exact ``evaluate_sequence``
  call) is returned.
"""

import time
import math
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def conv_full(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a * a (real FFT implementation).
    Returns a 1‑D array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    Nfft = 1 << ((2 * n - 1).bit_length())
    A_fft = np.fft.rfft(a, n=Nfft)
    conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
    return conv


def max_convolution(a: np.ndarray) -> float:
    """
    Return max_k (a*a)[k] .  For vectors normalised to Σa = √(2·n)
    this value equals the exact ``evaluate_sequence`` output.
    """
    return conv_full(a).max()


def build_constraint_matrix(f: np.ndarray) -> sp.csr_matrix:
    """
    Build the (2·n‑1) × n matrix A such that (A @ g)[k] = (f * g)[k].
    For each offset i (0 ≤ i < n) the i‑th diagonal of A is constant f[i].
    """
    n = f.shape[0]
    # Create a list of diagonals, each of length n, filled with the same f[i].
    diags = [np.full(n, val, dtype=np.float64) for val in f]
    offsets = np.arange(n, dtype=int)          # 0,1,…,n‑1
    A = sp.diags(diags, offsets, shape=(2 * n - 1, n), format="csr")
    return A


def solve_lp_for_direction(f: np.ndarray, M: float):
    """
    Solve
        maximise    Σ g_i
        subject to  (f * g)_k ≤ M   for all k,
                    g_i ≥ 0 .
    Returns a feasible vector g (already rescaled to the required L1‑norm)
    and the objective value Σ g_i, or (None, None) on failure.
    """
    n = f.shape[0]
    A = build_constraint_matrix(f)
    b = np.full(2 * n - 1, M, dtype=np.float64)

    # maximise Σ g_i  <=> minimise -Σ g_i
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n

    res = opt.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method="highs")
    if not res.success:
        return None, None

    g_raw = res.x
    sum_raw = g_raw.sum()
    if sum_raw <= 0:
        return None, None
    return g_raw, sum_raw

# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.

    The function runs for at most ``budget_s`` seconds (default 1000) and
    returns the best sequence found as a plain Python list.
    """

    start_time = time.time()
    deadline = start_time + budget_s - 5.0    # small safety margin

    target_sum = math.sqrt(2.0 * n)          # Σa = √(2·n)

    # ------------------------------------------------------------------
    eps = 1e-8
    base = np.maximum(base, eps)
    base = base * (target_sum / base.sum())
    a = base.copy()                         # current iterate

    # ------------------------------------------------------------------
    # Best‑so‑far bookkeeping
    # ------------------------------------------------------------------
    best_a = a.copy()
    best_score = max_convolution(best_a)     # equals evaluate_sequence(best_a)
    print(f"[init] n={n}, initial score={best_score:.6f}")

        # --------------------------------------------------------------

        # --------------------------------------------------------------

    # Final sanity check with the exact evaluator (adds clipping, etc.)
    # ------------------------------------------------------------------
    final_score = float(np.inf)
    try:
        final_score = evaluate_sequence(best_a.tolist())
    except Exception:
        pass
    if final_score < best_score:
        best_score = final_score
        best_a = best_a.copy()
        print(f"[exact‑check] updated best score to {best_score:.6f}")

    print(f"[done] best score = {best_score:.6f}, length = {n}")
    return best_a.tolist()
