"""
Propose a low‑value step‑function for the evaluation metric

    F(a) = 2·n·max(a*a) / (sum(a)^2) ,

where a*a is the linear convolution of the sequence with itself.
The search consists of two stages:

1. **Parametric sweep** – families that are known to flatten the
   self‑convolution (power‑law and geometric decay).  For each length
   `n` we generate a candidate, normalise it to the simplex and
   evaluate the exact objective using an FFT based convolution.
   The best vector found in this cheap sweep becomes the starting point
   for the second stage.

2. **Local refinement** – a projected Adam optimiser that works on the
   probability simplex.  The gradient is the exact sub‑gradient of the
   max‑convolution (i.e. an average over all indices that attain the
   current maximum).  A few thousand Adam steps, occasional random
   “shakes’’ and a final hill‑climbing pass usually improve the result
   further.

The routine respects a global time budget (`budget_s`, default 1000 s)
and prints progress.  It returns a plain Python list that can be fed to
`evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.shape[0] - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)],   i = 0 … n‑1.
    This is the gradient of Σ_k w_k·(a*a)_k  (factor 2 included later).
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        {x | x_i ≥ 0, Σ_i x_i = 1}
    """
    if v.ndim != 1:
        raise ValueError("v must be a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list[float]:
    """
    Search for a non‑negative sequence that minimises the evaluation
    function.  The algorithm first explores a few parametric families,
    then refines the best candidate with a simplex‑projected Adam
    optimiser using the exact sub‑gradient of the max‑convolution.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # --------------------------------------------------------------
    # 1.  Parametric sweep (very cheap, explores many shapes)
    # --------------------------------------------------------------
    lengths = [256, 512, 1024, 2048, 4096]
    # powers p (negative => edge‑heavy).  Use a small offset (0.5) to avoid 0⁰.
    power_ps = np.linspace(-1.4, 0.0, 29)          # -1.4 … 0
    # geometric decay parameters (λ close to 1 gives almost uniform).
    lambda_vals = np.linspace(0.70, 0.99, 21)      # 0.70 … 0.99

    for n in lengths:
        if time.time() > deadline:
            break
        fft_len = _next_pow2(2 * n - 1)
        idx = np.arange(n, dtype=np.float64)

        # ----- power‑law family -----
        for p in power_ps:
            if time.time() > deadline:
                break
            # a_i ∝ (i+½)^p + (n‑1‑i+½)^p   (symmetrised)
            w = (idx + 0.5) ** p + (n - 1 - idx + 0.5) ** p
            a = np.maximum(w, 0.0)
            s = a.sum()
            if s < 1e-12:
                continue
            a /= s
            conv = _full_convolution(a, fft_len)
            val = 2.0 * n * conv.max()
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[power] n={n}, p={p: .3f}, val={val:.6f}")
                if best_val < 1.503:
                    return best_seq.tolist()

        # ----- geometric‑decay family -----
        for lam in lambda_vals:
            if time.time() > deadline:
                break
            w = lam ** idx + lam ** (n - 1 - idx)
            a = np.maximum(w, 0.0)
            a /= a.sum()
            conv = _full_convolution(a, fft_len)
            val = 2.0 * n * conv.max()
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[geom ] n={n}, λ={lam: .4f}, val={val:.6f}")
                if best_val < 1.503:
                    return best_seq.tolist()

    # --------------------------------------------------------------
    # 2.  Local refinement of the best candidate (if any time left)
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        a = best_seq.copy()
        n = a.shape[0]
        fft_len_conv = _next_pow2(2 * n - 1)
        fft_len_corr = _next_pow2(3 * n)

        # Adam state
        m = np.zeros(n, dtype=np.float64)
        v = np.zeros(n, dtype=np.float64)
        beta1, beta2, eps = 0.9, 0.999, 1e-8
        lr = 0.05                     # modest learning rate for stability
        max_iters = 10000
        t = 0
        last_improve = time.time()

        while time.time() < deadline and t < max_iters:
            conv = _full_convolution(a, fft_len_conv)
            cur_max = conv.max()
            # sub‑gradient: uniform weight on all indices attaining the maximum
            mask = conv >= cur_max - 1e-12
            w = mask.astype(np.float64)
            if w.sum() == 0:
                w[:] = 1.0 / w.size
            else:
                w /= w.sum()

            # gradient of the max‑convolution
            grad = 2.0 * _cross_correlation(w, a, fft_len_corr)

            t += 1
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)

            a = _project_simplex(a - step)

            val = 2.0 * n * cur_max
            if val < best_val - 1e-9:
                best_val = val
                best_seq = a.copy()
                last_improve = time.time()
                print(f"[Adam ] iter={t:04d}, val={val:.6f}")
                if best_val < 1.503:
                    break

            # shake if we are stuck for a few seconds
            if time.time() - last_improve > 4.0:
                noise = (np.random.rand(n) - 0.5) * 1e-3
                a = _project_simplex(a + noise)
                last_improve = time.time()
                print(f"[shake] iter={t}")

    # --------------------------------------------------------------
    #   Final cheap hill‑climbing (still respecting the time limit)
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        a = best_seq.copy()
        n = a.shape[0]
        fft_len = _next_pow2(2 * n - 1)
        while time.time() < deadline:
            i, j = np.random.choice(n, 2, replace=False)
            # move a tiny fraction of mass from i to j
            delta = 0.005 * a[i] * np.random.rand()
            if delta == 0.0:
                continue
            a_prop = a.copy()
            a_prop[i] -= delta
            a_prop[j] += delta
            a_prop = _project_simplex(a_prop)
            conv = _full_convolution(a_prop, fft_len)
            val = 2.0 * n * conv.max()
            if val < best_val:
                best_val = val
                best_seq = a_prop.copy()
                a = a_prop
                print(f"[hill ] improved to {val:.6f}")

    # --------------------------------------------------------------
    # Fallback – uniform distribution if nothing succeeded.
    # --------------------------------------------------------------
    if best_seq is None:
        n_fallback = 1024
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        best_val = 2.0 * n_fallback * (1.0 / n_fallback)   # = 2
        print(f"fallback uniform, value = {best_val:.6f}")

    # Return a plain Python list (required by the external evaluator)
    return best_seq.tolist()
