def initial_shape_random(n, rng):
    # Choose base shape type:
    typ = rng.choice(['power', 'exponential', 'mixed', 'uniform'])
    idx = np.arange(n, dtype=float) + 1
    if typ == 'power':
        p = rng.uniform(0.1, 1.2)  # exponent
        shape = idx ** (-p)
    elif typ == 'exponential':
        tau = rng.uniform(0.2, 0.8) * n
        shape = np.exp(-idx / tau)
    elif typ == 'mixed':
        # Combine decreasing and increasing piecewise
        split = rng.randint(1, n-1)
        left = idx[:split] ** (-rng.uniform(0.2, 0.8))
        right = (n - idx[split:] + 1) ** (-rng.uniform(0.2, 0.8))
        shape = np.concatenate([left, right])
    else:  # uniform
        shape = np.ones_like(idx)
    # Add noise
    shape += rng.random(n) * 1e-4
    shape = np.maximum(shape, 1e-12)
    shape = shape / shape.sum() * np.sqrt(2.0 * n)  # scale to target sum
    return shape

"""
Search for a non‑negative step‑function that minimises the evaluation
2·n·max(conv(a,a))/sum(a)^2.
The algorithm runs many Adam‑optimised restarts on the soft‑max
parameterisation a = S·softmax(x).  A smooth approximation of the max
is used (log‑sum‑exp) with an aggressive temperature schedule.
The best sequence found in the allotted time is returned.
"""

import time
import numpy as np

def _full_convolution_fft(x, y):
    """Full linear convolution of two real vectors using a FFT of power‑of‑2 size."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    # tiny negative numbers may appear because of rounding
    np.maximum(conv, 0.0, out=conv)
    return conv

def _initial_shape_random(n, rng):
    """Return a positive array of length n with the prescribed total sum sqrt(2n)."""
    typ = rng.choice(['power', 'exponential', 'mixed', 'uniform'])
    idx = np.arange(1, n + 1, dtype=float)
    if typ == 'power':
        p = rng.uniform(0.1, 2.0)          # allow steeper decays
        shape = idx ** (-p)
    elif typ == 'exponential':
        tau = rng.uniform(0.2, 0.9) * n
        shape = np.exp(-idx / tau)
    elif typ == 'mixed':
        split = rng.integers(1, n - 1)
        left = idx[:split] ** (-rng.uniform(0.2, 1.0))
        right = (n - idx[split:] + 1) ** (-rng.uniform(0.2, 1.0))
        shape = np.concatenate([left, right])
    else:  # uniform
        shape = np.ones_like(idx)
    # tiny jitter to break symmetry
    shape += rng.random(n) * 1e-6
    shape = np.maximum(shape, 1e-12)
    # scale to the target sum S = sqrt(2n)
    target = np.sqrt(2.0 * n)
    shape = shape / shape.sum() * target
    return shape

    n,
    x0,
    target_sum,
    run_deadline,
    rng,
    *,
    init_tau_factor=0.2,
    min_tau_factor=5e-4,
    alpha=0.07,

    beta2=0.999,
    eps=1e-8,
    tau_decay=0.96,
    max_iter=None,
):
    """
    Adam optimiser on the log‑its (soft‑max) representation.

    Returns the best sequence (as a NumPy array) encountered inside the
    time window and its evaluation score obtained from ``evaluate_sequence``.
    """
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    best_seq = None
    best_val = float('inf')
    iteration = 0
    last_improve = 0

    while time.time() < run_deadline:
        # 1. map logits to simplex
        max_x = np.max(x)
        exp_shift = np.exp(x - max_x)
        p = exp_shift / exp_shift.sum()
        a = target_sum * p

        # 2. convolution and current max
        conv = _full_convolution_fft(a, a)
        max_conv = conv.max()

        if max_conv < best_val:
            best_val = max_conv
            best_seq = a.copy()
            last_improve = iteration
            print(f"[{time.time():.2f}] n={n}  local best = {max_conv:.8f}")

        # 3. smooth‑max weights
        tau = max(init_tau_factor * max_conv * (tau_decay ** iteration),
                  min_tau_factor * max_conv)
        scaled = (conv - max_conv) / tau
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # 4. gradient w.r.t a
        w_rev = w[::-1]
        conv2 = _full_convolution_fft(a, w_rev)          # length = n + (2n-1) -1 = 3n-2
        idx = (2 * n - 2) - np.arange(n)
        grad_a = 2.0 * conv2[idx]

        # 5. gradient w.r.t. logits x
        dotp = np.dot(p, grad_a)
        grad_x = target_sum * p * (grad_a - dotp)

        norm = np.linalg.norm(grad_x)

        alpha *= 0.9997

        if max_iter is not None and iteration >= max_iter:

The core of the search is an Adam optimiser that works on the soft‑max

aggressively annealed so that the optimisation eventually behaves like a
sub‑gradient step on the true max.  Many independent restarts with diverse
initial shapes (power‑law, exponential, mixed, uniform) are performed.
The best construction seen so far is kept globally, and good candidates

def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:

    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power of two

    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    # eliminate tiny negative values caused by numerical noise
    np.maximum(conv, 0.0, out=conv)
    return conv
