Propose a non‑negative step‑function that (approximately) minimises
``evaluate_sequence``.

Algorithm
---------
1. **Initial vector** – use the supplied construction (``height_sequence_1``)
   if present, otherwise a uniform vector.

   The exact objective (using an FFT for the convolution) is evaluated
   after each move, keeping the best point seen.
4. Return the best sequence (as a plain list).  All numbers are non‑negative
   and the L‑1 norm equals ``sqrt(2·n)`` which is the normalisation used by
   ``evaluate_sequence``.
"""
import time
import math
import numpy as np
import warnings

# ----------------------------------------------------------------------
# Utility functions for the smooth‑max optimisation (L‑BFGS part)
# ----------------------------------------------------------------------
def _softmax_weights(conv: np.ndarray, tau: float):
    """Stable soft‑max of the convolution values.
    Returns (weights, smooth_max)."""
    maxc = conv.max()
    if tau <= 0.0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w, maxc
    ex = np.exp((conv - maxc) / tau)
    sum_ex = ex.sum()
    w = ex / sum_ex
    Ltau = maxc + tau * math.log(sum_ex)
    return w, Ltau


def _weighted_rev(a: np.ndarray, w: np.ndarray, Nfft: int):
    """
    Compute rev_i = Σ_k w_k a_{k‑i}
    using FFTs. ``a`` and ``w`` are 1‑D arrays.
    """
    A_fft = np.fft.rfft(a, n=Nfft)
    W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)
    conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
    conv_rev = conv_rev[: len(w) + a.shape[0] - 1]
    idx_start = 2 * a.shape[0] - 2
    rev = conv_rev[idx_start - np.arange(a.shape[0])]
    return rev


def _conv_full(p: np.ndarray, Nfft: int):
    """Full linear convolution p∗p (real FFT based)."""
    A_fft = np.fft.rfft(p, n=Nfft)
    conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * p.shape[0] - 1]
    return conv


def _objective_and_grad(y: np.ndarray, tau: float, Nfft: int):
    """
    Smooth‑max objective and its gradient w.r.t. the log‑softmax parameters y.
    Returns (Lτ, ∇_y Lτ).
    """
    # 1. soft‑max to obtain the probability vector p
    y_max = np.max(y)
    exp_y = np.exp(y - y_max)               # stabilised
    p = exp_y / exp_y.sum()

    # 2. convolution p∗p
    conv = _conv_full(p, Nfft)

    # 3. smooth max
    w, Ltau = _softmax_weights(conv, tau)

    # 4. gradient w.r.t. p (∂L/∂p) = 2·rev
    rev = _weighted_rev(p, w, Nfft)
    grad_p = 2.0 * rev

    # 5. back‑propagate through the soft‑max (Jacobian of soft‑max)
    dot = np.dot(p, grad_p)                 # scalar = Σ p_i·grad_p_i
    grad_y = p * (grad_p - dot)
    return Ltau, grad_y


# ----------------------------------------------------------------------
# Fast exact evaluation (used by simulated annealing and final check)
# ----------------------------------------------------------------------
def _fast_score(a: np.ndarray, target_sum: float, Nfft: int) -> float:
    """
    Compute the exact ``evaluate_sequence`` value for a vector a that already
    satisfies Σa = target_sum.  The convolution is done with a real FFT.
    """
    conv = np.fft.irfft(
        np.fft.rfft(a, n=Nfft) * np.fft.rfft(a, n=Nfft), n=Nfft
    )[: 2 * a.shape[0] - 1]
    max_conv = conv.max()
    return (target_sum ** 2) * max_conv


# ----------------------------------------------------------------------
# L‑BFGS optimiser (multiple starts, temperature schedule)
# ----------------------------------------------------------------------
def _run_lbfgs(
    y0: np.ndarray,
    target_sum: float,
    Nfft: int,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
) -> tuple[np.ndarray, float]:
    """
    Optimise the smooth‑max objective from the starting point ``y0``.
    Returns the (possibly) improved best sequence and its score.
    """
    y = y0.copy()
    # initial temperature – a few times the current smooth max
    p_start = np.exp(y - np.max(y))
    p_start /= p_start.sum()
    conv0 = _conv_full(p_start, Nfft)
    tau = max(conv0.max() * 5.0, 1e-6)

    tau_decay = 0.70               # geometrical decay
    min_tau = 1e-12

    while time.time() < deadline and tau > min_tau:
        # ----- L‑BFGS on the current temperature -----
        def fun(y_vec):
            L, g = _objective_and_grad(y_vec, tau, Nfft)
            return L, g

        # a modest budget per stage – enough for a descent but still cheap
        opts = {"maxfun": 500, "maxiter": 500, "ftol": 1e-12, "gtol": 1e-12}
        with warnings.catch_warnings():
            warnings.simplefilter("ignore")
            try:
                from scipy.optimize import minimize
                res = minimize(fun, y, method="L-BFGS-B", jac=True, options=opts)
                y = res.x
            except Exception:   # pragma: no cover
                # fallback: simple gradient step
                L, g = fun(y)
                y = y - 1e-2 * g

        # ----- exact evaluation of the current point -----
        p = np.exp(y - np.max(y))
        p /= p.sum()
        a = p * target_sum
        cur_score = _fast_score(a, target_sum, Nfft)

        if cur_score < best_score - 1e-9:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[lbfgs improve] τ={tau:.2e}  score={best_score:.6f}")

        # ----- temperature schedule & jitter -----
        tau *= tau_decay
        if np.random.rand() < 0.2:
            y += np.random.randn(y.shape[0]) * 0.01

        if time.time() >= deadline:
            break

    return best_seq, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing refinement
# ----------------------------------------------------------------------
def _simulated_annealing(
    a0: np.ndarray,
    target_sum: float,
    Nfft: int,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
) -> tuple[np.ndarray, float]:
    """
    Perform a cheap simulated‑annealing walk starting from ``a0``.
    Returns the best sequence seen during the walk.
    """
    cur_a = a0.copy()
    cur_score = _fast_score(cur_a, target_sum, Nfft)

    n = cur_a.shape[0]
    # a simple exponential cooling schedule
    T_start = 0.1 * cur_score
    T_end = 1e-4 * cur_score

    # number of iterations is not fixed – we stop when time is up
    it = 0
    while time.time() < deadline:
        # remaining fraction (for temperature)
        frac = (deadline - time.time()) / (deadline - (deadline - (deadline - time.time())))
        temp = T_end + (T_start - T_end) * frac

        # propose a tiny mass move
        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1
        if cur_a[i] <= 0.0:
            continue
        # epsilon is a small fraction of the source entry
        eps = cur_a[i] * (0.001 + 0.004 * np.random.rand())
        new_a = cur_a.copy()
        new_a[i] -= eps
        new_a[j] += eps

        # score of the proposal
        new_score = _fast_score(new_a, target_sum, Nfft)
        delta = new_score - cur_score

        if delta < 0 or np.random.rand() < math.exp(-delta / max(temp, 1e-12)):
            cur_a = new_a
            cur_score = new_score
            if cur_score < best_score - 1e-9:
                best_score = cur_score
                best_seq = cur_a.copy()
                print(f"[sa improve] it={it}  score={best_score:.6f}")

        it += 1
        if it % 5000 == 0 and time.time() >= deadline:
            break

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point required by the platform
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine combines a smooth‑max L‑BFGS optimisation with a short
    simulated‑annealing refinement.  It respects the 1000 s time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # small safety margin

    # ------------------------------------------------------------------
    # 1. Load or build an initial construction
    # ------------------------------------------------------------------
    try:
        base = np.array(height_sequence_1, dtype=float)  # type: ignore
    except Exception:
        # fallback: uniform of a moderate size
        base = np.ones(1900, dtype=float)

    n = base.shape[0]
    target_sum = math.sqrt(2.0 * n)

    # normalise to the required L1‑norm
    base = base * (target_sum / base.sum())
    base = np.clip(base, 0.0, 1000.0)

    # keep the best solution found so far
    best_seq = base.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] n={n}  initial score={best_score:.6f}")

    # FFT size – next power‑of‑two ≥ 2·n‑1
    Nfft = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------------
    # 2. L‑BFGS optimisation from several starts
    # ------------------------------------------------------------------
    starts_y = []

    # a) start from the known construction
    p0 = best_seq / target_sum
    starts_y.append(np.log(p0 + 1e-12))

    # b) random Dirichlet (exponential samples)
    rand_exp = np.random.exponential(scale=1.0, size=n)
    starts_y.append(np.log(rand_exp + 1e-12))

    # c) pure uniform
    starts_y.append(np.zeros(n, dtype=float))

    # d) perturbed known solution
    starts_y.append(starts_y[0] + np.random.randn(n) * 0.05)

    for idx, y0 in enumerate(starts_y):
        if time.time() >= deadline:
            break
        print(f"[lbfgs start] {idx + 1}/{len(starts_y)}")
        best_seq, best_score = _run_lbfgs(
            y0=y0,
            target_sum=target_sum,
            Nfft=Nfft,
            deadline=deadline,
            best_seq=best_seq,
            best_score=best_score,
        )

    # ------------------------------------------------------------------
    # 3. Simulated‑annealing refinement (use remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline - 1.0:
        print("[sa] starting refinement")
        best_seq, best_score = _simulated_annealing(
            a0=best_seq,
            target_sum=target_sum,
            Nfft=Nfft,
            deadline=deadline,
            best_seq=best_seq,
            best_score=best_score,
        )

    # ------------------------------------------------------------------
    # 4. Final exact evaluation (ensure we return the true objective)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq.tolist())
    if final_score < best_score:
        best_score = final_score
        best_seq = best_seq.copy()
        print("[final] exact evaluation improved the best score")

    print(f"[done] best score = {best_score:.6f}  length = {len(best_seq)}")
    return best_seq.tolist()
