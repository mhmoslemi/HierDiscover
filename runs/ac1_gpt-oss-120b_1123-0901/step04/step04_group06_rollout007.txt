def _smooth_weights(conv, tau):
    max_val = conv.max()
    scaled = (conv - max_val) / tau
    # To avoid overflow, subtract max(scaled) before exponentiate
    # Actually scaled <= 0
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum == 0:
        return np.full_like(w, 1.0 / w.size)
    return w / w_sum

"""
Search for a non‑negative step function that minimizes

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The evaluation function returns exactly max(conv(a, a)) when the sum of a is
fixed to √(2 n).  The algorithm works as follows:

* **Log‑it (soft‑max) parameterisation** – a is written as
  a = S·softmax(x) with S = √(2 n).  This guarantees a ≥ 0 and Σa = S
  without any projection step.

* **Smooth‑max surrogate** – the true non‑smooth max is approximated by a
  temperature‑controlled soft‑max weight vector w over the convolution
  entries.  The weight definition is
        w_i = exp((c_i – max(c)) / τ) / Σ_j exp((c_j – max(c)) / τ)
  where c = a * a (full linear convolution) and τ > 0 gradually decays.

* **Gradient** – the gradient of the weighted sum with respect to a is
        ∇_a = 2·(a ⊛ rev(w))
  which is computed efficiently with an FFT‑based convolution.  Chain rule
  gives the gradient w.r.t. the logits x.

* **Adam optimiser** – an Adam update on the logits performs adaptive
  steps.  A moderate learning‑rate schedule together with occasional tiny
  random jitter helps to escape plateaus.

* **Multi‑restart + diverse initialisation** – many independent runs are
  performed within the budget.  Lengths n range from 1 500 to 6 000 and
  initial shapes are drawn from several families (power‑law, exponential,
  linear, uniform, random).  After each Adam run a short *swap‑refinement*
  (random pairwise swaps) is applied to the best local sequence.

* **Global bookkeeping** – the best sequence seen so far (including an
  optional warm‑start from a globally supplied `height_sequence_1`) is
  stored and returned at the end.

The routine respects the 1000 s time limit, prints progress timestamps and
requires only NumPy and the supplied `evaluate_sequence` function.
"""

import time
import random
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper – full linear convolution via real FFT (fast for any length)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of real vectors x and y via FFT."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())            # next power of two
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    return conv


# ----------------------------------------------------------------------
# Compute max of full convolution (fast, no clipping)
# ----------------------------------------------------------------------
def _max_conv(a: np.ndarray) -> float:
    """Maximum value of the linear convolution a * a."""
    conv = _full_convolution_fft(a, a)
    return float(conv.max())


# ----------------------------------------------------------------------
# Soft‑max weights for the smoothed‑max surrogate
# ----------------------------------------------------------------------
def _smooth_weights(conv: np.ndarray, tau: float) -> np.ndarray:
    """Return a probability vector w approximating the arg‑max of conv."""
    max_val = conv.max()
    # shift by max to avoid overflow; values are ≤0 after division by τ
    scaled = (conv - max_val) / max(tau, 1e-12)
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum == 0.0:
        # fallback to uniform – should not happen
        return np.full_like(w, 1.0 / w.size)
    return w / w_sum


# ----------------------------------------------------------------------
# Gradient of the weighted sum with respect to a
# ----------------------------------------------------------------------
def _gradient_a(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_j w_j·(a*a)[j] w.r.t. a.
    Returns a length‑n vector.
    """
    n = a.shape[0]
    w_rev = w[::-1]
    conv2 = _full_convolution_fft(a, w_rev)          # length = 2n‑1
    idx = (2 * n - 2) - np.arange(n)                 # map to needed positions
    grad = 2.0 * conv2[idx]
    return grad


# ----------------------------------------------------------------------
# Adam optimisation on the log‑it representation
# ----------------------------------------------------------------------
def _adam_optimize(
    n: int,
    x0: np.ndarray,
    target_sum: float,
    run_deadline: float,
    *,
    init_tau_factor: float = 0.20,
    min_tau_factor: float = 0.001,
    alpha: float = 0.04,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Optimise a sequence of length ``n`` using Adam on the log‑it variables.
    Returns the best sequence (as a NumPy array) and its evaluation score.
    """
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    best_seq = None
    best_val = float("inf")
    iteration = 0

    while time.time() < run_deadline:
        # ---- 1. soft‑max mapping (enforces positivity & correct sum) ----
        max_x = np.max(x)
        exp_shift = np.exp(x - max_x)                 # stability
        p = exp_shift / np.sum(exp_shift)             # probabilities
        a = target_sum * p

        # ---- 2. current convolution and (smooth) max ----
        conv = _full_convolution_fft(a, a)
        cur_max = float(conv.max())

        if cur_max < best_val:
            best_val = cur_max
            best_seq = a.copy()
            print(f"[{time.time():.2f}] n={n}   new local best = {cur_max:.6f}")

        # ---- 3. compute smooth‑max weights ----
        # temperature decays exponentially with iterations, never below min factor
        tau = max(init_tau_factor * cur_max * (0.95 ** iteration),
                  min_tau_factor * cur_max)

        w = _smooth_weights(conv, tau)

        # ---- 4. gradient w.r.t. a ----
        grad_a = _gradient_a(a, w)

        # ---- 5. gradient w.r.t. logits x (chain rule) ----
        dotp = np.dot(p, grad_a)                      # scalar = ⟨p, grad_a⟩
        grad_x = target_sum * p * (grad_a - dotp)

        # ---- 6. Adam update ----
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # occasional tiny random perturbation to encourage exploration
        if iteration % 300 == 0 and random.random() < 0.25:
            x += np.random.randn(n) * 0.001

        # slowly decay the base learning‑rate (helps convergence)
        alpha *= 0.9996

    # Final evaluation using the provided evaluator (identical rounding)
    final_score = evaluate_sequence(best_seq.tolist())
    return best_seq, final_score


# ----------------------------------------------------------------------
# Light‑weight local refinement by random pairwise swaps
# ----------------------------------------------------------------------
def _local_swap_refine(a: np.ndarray, time_budget: float) -> tuple[np.ndarray, float]:
    """
    Starting from ``a``, perform random swaps for a short time and keep any
    improvement in max‑convolution.  Returns the best sequence encountered
    and its corresponding max‑conv value.
    """
    start = time.time()
    best_seq = a.copy()
    best_val = _max_conv(best_seq)
    n = a.shape[0]

    while time.time() - start < time_budget:
        i, j = random.sample(range(n), 2)
        # swap positions i and j
        cand = best_seq.copy()
        cand[i], cand[j] = cand[j], cand[i]
        val = _max_conv(cand)
        if val < best_val:
            best_val = val
            best_seq = cand
    return best_seq, best_val

        # Normalise to the required sum Σa = √(2 n)
        # --------------------------------------------------------------
        target_sum = math.sqrt(2.0 * n)
        init = init / init.sum() * target_sum

        # --------------------------------------------------------------
        # Transform to log‑it representation for the Adam optimiser.
        # --------------------------------------------------------------
        x0 = np.log(init + 1e-12)

        # --------------------------------------------------------------
        # Allocate a time slice for this Adam run.
        # --------------------------------------------------------------
        slice_time = min(10.0, remaining * 0.5)          # never exceed half of the budget left
        run_deadline = min(deadline, time.time() + slice_time)

        # --------------------------------------------------------------
        # Run the Adam optimiser.
        # --------------------------------------------------------------
        seq_local, score_local = _adam_optimize(
            n,
            x0,
            target_sum,
            run_deadline,
            init_tau_factor=0.20,
            min_tau_factor=0.001,
            alpha=0.04,
            beta1=0.9,
            beta2=0.999,
        )

        # --------------------------------------------------------------
        # Record an improvement if we have one.
        # --------------------------------------------------------------
        if score_local < best_score:
            best_score = score_local
            best_seq = seq_local.copy()
            print(f"[{time.time():.2f}] GLOBAL new best = {best_score:.6f} (n={n})")
            if best_score < 1.503:          # target achieved – stop early
                break

        # --------------------------------------------------------------
        # Short local refinement via random swaps (helps escape local minima).
        # --------------------------------------------------------------
        # we give at most 1.2 s to this step, but never exceed the overall deadline
        refine_budget = min(1.2, deadline - time.time())
        if refine_budget > 0.1:
            seq_refined, val_ref = _local_swap_refine(seq_local, refine_budget)
            if val_ref < best_score:
                best_score = val_ref
                best_seq = seq_refined.copy()
                print(f"[{time.time():.2f}] GLOBAL best after swap = {best_score:.6f}")
                if best_score < 1.503:
                    break

    # ------------------------------------------------------------------
    # Fallback in the unlikely event that no candidate was generated.
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)

    # The evaluator expects a plain Python list.
    return best_seq.tolist()
