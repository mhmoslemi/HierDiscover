"""
Propose a step‑function (non‑negative vector) that minimises the
``evaluate_sequence`` objective.

Algorithm
---------
* A coarse optimisation is performed on a short vector (n≈500) with many
  Adam‑steps and several random initialisations (uniform, Dirichlet,
  exponential, triangular, Gaussian, Hann, etc.).  The short vector
  converges quickly because the FFT is cheap.
* The best short vector is linearly interpolated to the larger target
  lengths (≈2100, 3150, 4200, 6000; never >8000).  After interpolation the
  vectors are re‑normalised to the prescribed ℓ₁‑norm √(2 n).
* Each long vector is then fine‑tuned by a short Adam run.
* The Adam optimiser works on the smoothed max‑convolution
  (soft‑max with decreasing temperature τ).  The gradient of the
  smoothed maximum is obtained by a weighted reverse convolution computed
  with FFTs.  After each step the vector is projected onto the feasible
  set (a_i≥0, Σa_i = √(2 n)).  Small Gaussian jitter is added every few
  thousand steps to escape plateaus.
* A few different learning‑rate / temperature schedules are tried
  (different τ‑factors and initial step sizes).  The best sequence over
  all runs (measured by the exact objective ``max(a∗a)``) is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (all top‑level, no closures)
# ----------------------------------------------------------------------


def _softmax_weights(conv: np.ndarray, tau: float):
    """Stable soft‑max of the convolution values."""
    maxc = conv.max()
    if tau <= 0.0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w, maxc
    ex = np.exp((conv - maxc) / tau)
    sum_ex = ex.sum()
    w = ex / sum_ex
    Ltau = maxc + tau * math.log(sum_ex)
    return w, Ltau


def _weighted_rev(a: np.ndarray, w: np.ndarray, Nfft: int) -> np.ndarray:
    """
    Compute  rev_i = Σ_k w_k * a_{k‑i}   (indices outside range ignored)
    using FFTs.  This is the (sub‑)gradient of the (soft‑max) max‑convolution.
    """
    A_fft = np.fft.rfft(a, n=Nfft)
    W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)
    conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
    # keep only the part that matters
    conv_rev = conv_rev[: len(w) + a.shape[0] - 1]
    idx_start = 2 * a.shape[0] - 2
    rev = conv_rev[idx_start - np.arange(a.shape[0])]
    return rev


def _max_convolution(a: np.ndarray, Nfft: int) -> float:
    """Exact max of the linear convolution a∗a (real FFT implementation)."""
    A_fft = np.fft.rfft(a, n=Nfft)
    conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * a.shape[0] - 1]
    return float(conv.max())


def _interpolate(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a 1‑D array to a new length."""
    old_n = seq.shape[0]
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(new_x, old_x, seq)
    return new_seq


def _project_onto_simplex(a: np.ndarray, target_sum: float) -> np.ndarray:
    """Projection onto {a_i ≥ 0, Σ a_i = target_sum}."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 0.0:
        a = a * (target_sum / s)
    else:
        # all zero – fall back to uniform
        a = np.full_like(a, target_sum / a.shape[0])
    return a


def _make_initial_vectors(n: int, target_sum: float, rng: np.random.Generator):
    """Return a diverse set of start vectors (already normalised)."""
    init = []

    # 1) uniform
    init.append(np.full(n, target_sum / n))

    # 2) Dirichlet (Gamma(1,1))
    g = rng.gamma(1.0, 1.0, size=n)
    init.append(g / g.sum() * target_sum)

    # 3) exponential decay (random rate)
    rate = rng.uniform(0.0005, 0.01)
    x = np.exp(-rate * np.arange(n))
    init.append(x / x.sum() * target_sum)

    # 4) reversed exponential
    init.append(x[::-1] / x[::-1].sum() * target_sum)

    # 5) Hann window
    hann = 0.5 - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1))
    init.append(hann / hann.sum() * target_sum)

    # 6) Blackman window
    a0, a1, a2 = 0.42, 0.5, 0.08
    n_arr = np.arange(n)
    black = (
        a0
        - a1 * np.cos(2.0 * np.pi * n_arr / (n - 1))
        + a2 * np.cos(4.0 * np.pi * n_arr / (n - 1))
    )
    init.append(black / black.sum() * target_sum)

    # 7) Triangular (peak in the middle)
    tri = np.concatenate([np.arange(1, n // 2 + 1), np.arange(n // 2, 0, -1)])
    init.append(tri / tri.sum() * target_sum)

    # 8) Gaussian centred, random sigma
    sigma = rng.uniform(n * 0.1, n * 0.4)
    gauss = np.exp(-0.5 * ((np.arange(n) - n / 2) / sigma) ** 2)
    init.append(gauss / gauss.sum() * target_sum)

    # 9) Small random perturbation of uniform
    pert = np.full(n, target_sum / n)
    pert += rng.normal(scale=0.02 * target_sum / n, size=n)
    pert = np.maximum(pert, 0.0)
    pert = pert / pert.sum() * target_sum
    init.append(pert)

    return init


def _adam_optimize(
    a0: np.ndarray,
    n: int,
    target_sum: float,
    deadline: float,
    max_iters: int,
    lr0: float,
    tau_factor: float,
    jitter_interval: int,
    best_seq: np.ndarray,
    best_score: float,
):
    """
    Adam optimiser with soft‑max temperature annealing.
    Returns the best sequence (as np.ndarray) and its exact max‑convolution
    value observed during the run.
    """
    # Normalise initial vector
    a = a0 * (target_sum / a0.sum())

    # Adam parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    # FFT length (power of two ≥ 2·n‑1)
    Nfft = 1 << ((2 * n - 1).bit_length())

    # Pre‑compute constants appearing in the Lagrange multiplier term
    coeff = 4.0 * n / (target_sum * target_sum)  # equals 2 after normalisation

    # Decay schedule for temperature τ
    decay_rate = max_iters / 12.0  # slower decay than the old code

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ----- compute exact convolution & its max -----
        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        M = float(conv.max())

        # ----- temperature & soft‑max weights -----
        tau = tau_factor * M * math.exp(-it / decay_rate)
        if tau < 5e-10:
            # exact sub‑gradient – use uniform weight over active set
            mask = conv >= M - 1e-12
            cnt = mask.sum()
            w = mask.astype(float) / max(cnt, 1)
            Ltau = M
        else:
            w, Ltau = _softmax_weights(conv, tau)

        # ----- gradient (weighted reverse convolution) -----
        rev = _weighted_rev(a, w, Nfft)

        # Lagrange multiplier for the sum constraint (see original code)
        const = 4.0 * n * Ltau / (target_sum ** 3)

        grad = coeff * rev - const

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / (1.0 + 1e-4 * it)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # ----- projection onto feasible set -----
        a = _project_onto_simplex(a, target_sum)

        # ----- occasional jitter to escape plateaus -----
        if it % jitter_interval == 0:
            jitter = np.random.randn(n) * (1e-3 * target_sum / n)
            a = np.maximum(a + jitter, 0.0)
            a = _project_onto_simplex(a, target_sum)

        # ----- keep global best (exact value after projection = M) -----
        cur_score = M
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(
                f"[len {n:5d} it {it:8d}] new best = {best_score:.6f}"
            )

    return best_seq, best_score

            interp_base = interp_base / interp_base.sum() * target_sum
            extra_inits.append(interp_base)

        # Full candidate list for this length
        candidates = [a0] + extra_inits[:3]   # keep the set small to stay within budget

        # Hyper‑parameters for the fine‑tuning stage (more conservative)
        fine_params = [
            (0.04, 0.15, 4000),
            (0.06, 0.22, 5000),
        ]

        # Iteration budget – scaled roughly inversely with n
        max_iters = int(300_000 * short_n / n)
        max_iters = max(80_000, max_iters)

        print(
            f"[stage 2] length={n}, max_iters≈{max_iters}, target_sum={target_sum:.3f}"
        )

        for cand_idx, a_start in enumerate(candidates):
            if time.time() >= deadline:
                break
            for lr0, tau_factor, jitter_int in fine_params:
                if time.time() >= deadline:
                    break
                print(
                    f"  [len {n:5d} cand {cand_idx+1}/{len(candidates)}] "
                    f"lr0={lr0:.3f} τ={tau_factor:.2f}"
                )
                best_seq, best_score = _adam_optimize(
                    a0=a_start.copy(),
                    n=n,
                    target_sum=target_sum,
                    deadline=deadline,
                    max_iters=max_iters,
                    lr0=lr0,
                    tau_factor=tau_factor,
                    jitter_interval=jitter_int,
                    best_seq=best_seq if best_seq is not None else a_start,
                    best_score=best_score,
                )

    # ------------------------------------------------------------------
    # Fallback – if nothing succeeded (should never happen)
    if best_seq is None:
        n = short_n
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n)
        best_score = float("inf")

    print(f"[done] best score = {best_score:.6f}, length = {best_seq.shape[0]}")
    # Return plain Python list as required by the harness
    return best_seq.tolist()
```

This implementation follows the ideas outlined above:
* a fast coarse optimisation on a short vector,
* linear interpolation to larger lengths,
* several Adam runs with soft‑max smoothing, projection, jitter, and a small grid of learning‑rate / temperature parameters,
* diversification via many analytically‑motivated initial shapes (uniform, exponential, triangular, Gaussian, Hann, Blackman, random perturbations) and via any previously discovered construction.

The algorithm respects the 1000 s wall‑clock limit by checking the deadline each iteration and adapts the number of Adam steps to the vector length, allowing ample optimisation time while staying safely within the budget.
