"""
Propose a step‑function (non‑negative heights) that minimises

        F(a) = 2·n·max(a∗a) / (sum(a)²)

The implementation improves the previous approach by
* Optimising a *smooth* surrogate of the max‑convolution using
  Adam on the simplex (log‑softmax parametrisation).  The smooth
  loss is  L = (1/β)·log ∑ exp(β·conv)  whose gradient is simply the
  soft‑max of the convolution entries.  By annealing β from a small
  to a large value the optimiser first sees a smooth landscape and
  later focuses on the true maximum.
* Efficient FFT‑based convolutions (O(n log n)) for both the
  self‑convolution and the weighted convolution needed for the
  gradient.
* A modest Differential‑Evolution (DE) phase and a (1+1) Evolution
  Strategy (ES) refinement, both respecting the total time budget.

The algorithm works for any length up to a few thousand entries,
keeps the best vector found and returns it as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution helpers
# ----------------------------------------------------------------------
def fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution a*b using a real FFT (fast for large n)."""
    n = a.size + b.size - 1
    L = 1 << (n - 1).bit_length()          # next power of two
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b, L)
    conv = np.fft.irfft(fa * fb, L)[:n]
    return conv.real


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Return the linear self‑convolution a*a (length 2·len(a)‑1)."""
    return fft_convolve(a, a)


def conv_max(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of a."""
    return full_convolution(a).max()


# ----------------------------------------------------------------------
# Objective (exact) – scale invariant, sum is not forced to 1
# ----------------------------------------------------------------------
def compute_objective(a: np.ndarray) -> float:
    """Exact objective used by evaluate_sequence."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return 2.0 * a.size * conv_max(a) / (s * s)


# ----------------------------------------------------------------------
# Numerically stable softmax / log‑sum‑exp
# ----------------------------------------------------------------------
def softmax(x: np.ndarray) -> np.ndarray:
    """Stable softmax (maps logits to a probability vector)."""
    mx = np.max(x)
    e = np.exp(x - mx)
    return e / e.sum()


def logsumexp(z: np.ndarray) -> float:
    """Stable log‑sum‑exp."""
    m = np.max(z)
    return m + np.log(np.sum(np.exp(z - m)))


# ----------------------------------------------------------------------
# Adam optimiser on a smoothed max‑convolution (β‑annealed)
# ----------------------------------------------------------------------
def run_adam_smooth_opt(
    n: int,
    init_a: np.ndarray,
    deadline: float,
    max_iters: int = 200_000,
    beta_start: float = 1.0,
    beta_end: float = 200.0,
) -> tuple[np.ndarray, float]:
    """
    Adam on the simplex to minimise a smooth surrogate of max(a∗a).

    a is represented as a softmax of logits `x`.  The loss is
        Lβ(a) = (1/β)·log∑exp(β·conv(a)),
    whose gradient wrt a is the soft‑max weights over the convolution.
    The temperature β is linearly increased from beta_start to beta_end.
    """
    # initialise logits from a (already non‑negative, sum≈1)
    x = np.log(init_a + 1e-15)

    m = np.zeros(n)
    v = np.zeros(n)
    beta_range = beta_end - beta_start
    beta = beta_start

    # Adam hyper‑parameters
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.05

    best_val = np.inf
    best_a = None
    t = 0
    last_improve = time.time()

    while time.time() < deadline and t < max_iters:
        # ----- current probability vector -----
        a = softmax(x)                     # a ≥ 0, Σa = 1

        # ----- convolution of a with itself -----
        conv = full_convolution(a)         # length 2n‑1

        # ----- anneal β -----
        beta = beta_start + beta_range * (t / max_iters)

        # ----- smooth loss (log‑sum‑exp) & its softmax weights -----
        z = beta * conv
        max_z = np.max(z)
        exp_z = np.exp(z - max_z)          # stabilised
        sumexp = exp_z.sum()
        loss = (max_z + np.log(sumexp)) / beta
        w = exp_z / sumexp                 # w_i = ∂L/∂conv_i

        # ----- gradient of loss w.r.t. a (exact) -----
        # weighted convolution w * a_rev
        a_rev = a[::-1]
        conv_w = fft_convolve(w, a_rev)   # length 3n‑2
        slice_w = conv_w[n - 1 : n - 1 + n]   # the part corresponding to grad_a
        w_diag = w[::2]                    # entries w_{2k}, length n
        grad_a = 2.0 * slice_w - a * w_diag   # shape (n,)

        # ----- mirror‑descent back‑propagation through softmax -----
        avg = np.dot(a, grad_a)
        grad_x = a * (grad_a - avg)

        # ----- Adam update on logits -----
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

        # keep logits bounded to avoid overflow
        x = np.clip(x, -50.0, 50.0)

        # ----- evaluate the *exact* objective for bookkeeping -----
        val = compute_objective(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[Adamβ n={n} iter={t:05d}] new best: {val:.6f}")

        # ----- shake if stuck for a while -----
        if time.time() - last_improve > 5.0:
            x += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()
            print(f"[Adamβ n={n} iter={t}] shake (stagnation)")

    # fallback – uniform if nothing found (should not happen)
    if best_a is None:
        best_a = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = compute_objective(best_a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Differential Evolution (global search) – unchanged core logic
# ----------------------------------------------------------------------
def run_de_opt(
    n: int,
    init_pop: list,
    deadline: float,
    popsize: int = 20,
    max_gens: int = 30_000,
) -> tuple[np.ndarray, float]:
    """Simple DE on the simplex; individuals are kept normalised."""
    # initialise population
    pop = []
    for cand in init_pop:
        a = np.asarray(cand, dtype=np.float64)
        s = a.sum()
        if s > 0:
            pop.append(a / s)
        else:
            pop.append(np.full(n, 1.0 / n))
    while len(pop) < popsize:
        pop.append(np.random.dirichlet(np.ones(n)))

    obj_vals = [compute_objective(a) for a in pop]
    best_idx = int(np.argmin(obj_vals))
    best_a = pop[best_idx].copy()
    best_val = obj_vals[best_idx]

    F = 0.8   # differential weight
    CR = 0.9  # crossover probability
    last_improve = time.time()
    gen = 0

    while time.time() < deadline and gen < max_gens:
        gen += 1
        for i in range(popsize):
            # three distinct donors
            idxs = list(range(popsize))
            idxs.remove(i)
            r1, r2, r3 = np.random.choice(idxs, 3, replace=False)
            mutant = pop[r1] + F * (pop[r2] - pop[r3])
            mutant = np.maximum(mutant, 0.0)               # enforce non‑negativity

            # binomial crossover (guarantee at least one mutant component)
            cross = np.random.rand(n) < CR
            cross[np.random.randint(n)] = True
            trial = np.where(cross, mutant, pop[i])

            s = trial.sum()
            if s == 0.0:
                trial = np.full(n, 1.0 / n)
            else:
                trial = trial / s

            val = compute_objective(trial)
            if val < obj_vals[i]:
                pop[i] = trial
                obj_vals[i] = val
                if val < best_val:
                    best_val = val
                    best_a = trial.copy()
                    last_improve = time.time()
                    print(f"[DE n={n} gen={gen:04d} i={i}] new best: {val:.6f}")

        # early stop if no progress for a while
        if time.time() - last_improve > 10.0:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# (1+1) Evolution Strategy (fine‑tuning) – unchanged
# ----------------------------------------------------------------------
def run_es_opt(
    n: int,
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 30_000,
) -> tuple[np.ndarray, float]:
    """Gaussian mutation ES on the simplex."""
    a = init_a.copy()
    val = compute_objective(a)
    best_a, best_val = a.copy(), val
    sigma = 0.02                     # mutation strength
    last_improve = time.time()
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        trial = a + sigma * np.random.randn(n)
        trial = np.maximum(trial, 0.0)
        s = trial.sum()
        if s == 0.0:
            trial = np.full(n, 1.0 / n)
        else:
            trial = trial / s

        val_trial = compute_objective(trial)
        if val_trial < best_val:
            best_val = val_trial
            best_a = trial.copy()
            a = trial                               # success → keep
            sigma = min(sigma * 1.05, 0.2)           # increase step‑size a bit
            last_improve = time.time()
            print(f"[ES n={n} it={it}] new best: {best_val:.6f}")
        else:
            sigma *= 0.99                           # decay on failure

        if time.time() - last_improve > 5.0:
            a = best_a.copy()
            sigma = 0.02
            last_improve = time.time()

    return best_a, best_val
