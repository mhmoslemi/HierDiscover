    x = np.log(init_a + 1e-15)  # logits
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.1
    best_val = np.inf
    best_a = None
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        # current alpha schedule: linear increase
        alpha = alpha_start + (alpha_end - alpha_start) * min(t/max_iters, 1.0)
        # Compute a
        a = softmax(x)
        # compute convolution via FFT
        conv = full_convolution(a)   # length 2n-1
        # compute smooth max via log-sum-exp
        # compute softmax weights w = exp(alpha * conv) / sum(...)
        # to avoid overflow, subtract max
        maxc = conv.max()
        exp_term = np.exp(alpha * (conv - maxc))
        sum_exp = exp_term.sum()
        w = exp_term / sum_exp
        # compute smooth objective L = (log sum_exp)/alpha + maxc (since we used shift)
        smooth_val = (np.log(sum_exp) / alpha) + maxc
        # actual objective F = 2*n*conv_max (approx). But we can compute exact objective with actual max.
        # We'll compute exact objective to evaluate best.
        exact_val = compute_objective(a)
        if exact_val < best_val:
            best_val = exact_val
            best_a = a.copy()
            print(f"[Adam smooth n={n} iter={t}] exact best {best_val:.6f}")

        # w length = 2n-1
        # compute gradient via convolution: g = 2 * (a_rev * w) sliced
        a_rev = a[::-1]
        L = 1 << int(np.ceil(np.log2(3*n)))   # enough to hold conv length n + (2n-1) -1 = 3n-2
        Fa = np.fft.rfft(a_rev, L)
        Fw = np.fft.rfft(w, L)
        conv_rev = np.fft.irfft(Fa * Fw, L)[: (n + (2*n-1) -1)]  # length 3n-2
        # slice central n elements
        g = 2.0 * conv_rev[n-1 : n-1 + n]
        # Backprop through softmax
        avg = np.dot(a, g)
        grad_x = a * (g - avg)
        # Adam update
        m = beta1*m + (1-beta1)*grad_x
        v = beta2*v + (1-beta2)*(grad_x*grad_x)
        m_hat = m/(1-beta1**t)
        v_hat = v/(1-beta2**t)
        x = x - lr * m_hat / (np.sqrt(v_hat)+eps)
        # clip logits
        x = np.clip(x, -50, 50)
        # optionally add noise if stagnating
    return best_a, best_val

"""
Propose a non‑negative step‑function that (approximately) minimises

        F(a) = 2·n·max(a∗a) / (sum(a)²)

using a smooth‑max Adam optimiser and a short (1+1)‑ES finetuner.
The smooth‑max replaces the hard max by a log‑sum‑exp (temperature α)
so the gradient contains information from many convolution entries.
α is increased during optimisation, turning the objective gradually
into the true max‑norm.  The algorithm works on the probability simplex
by optimisation in log‑it space (softmax) and respects the total
time‑budget (default 1000 s).  Several random restarts and a fallback
to a uniform sequence guarantee a valid answer.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------
def softmax(x: np.ndarray) -> np.ndarray:
    """Stable softmax – maps logits to a probability vector (sum=1)."""
    mx = np.max(x)
    e = np.exp(x - mx)
    s = e.sum()
    if s == 0.0:
        # should never happen, but guard against division by zero
        return np.full_like(x, 1.0 / x.size)
    return e / s


def next_pow2(k: int) -> int:
    """Small helper – smallest power of two ≥ k."""
    return 1 << (k - 1).bit_length()


def full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a*a (length 2·len(a)‑1) using FFT.
    Returns a real array.
    """
    n = a.shape[0]
    L = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, L)
    conv = np.fft.irfft(fa * fa, L)[: 2 * n - 1]
    return conv.real


def conv_max(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution of a."""
    return full_convolution(a).max()


def compute_objective(a: np.ndarray) -> float:
    """
    Exact objective used by evaluate_sequence for an arbitrary vector.
    Equivalent to 2·n·max(a∗a) / (sum(a)²).
    """
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return 2.0 * a.shape[0] * conv_max(a) / (s * s)


# ----------------------------------------------------------------------
# Smooth‑max gradient (log‑sum‑exp) helpers
# ----------------------------------------------------------------------
def smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Return the soft‑max weights w_i = exp(alpha*conv_i)/Σ_j exp(alpha*conv_j).
    The conv vector is shifted by its maximum before exponentiation for
    numerical stability.
    """
    # classic log‑sum‑exp trick
    shift = conv.max()
    ex = np.exp(alpha * (conv - shift))
    total = ex.sum()
    if total == 0.0:
        # all entries underflowed – fall back to a hard max
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w
    return ex / total


def gradient_smooth_max(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of L(a)=∑_k w_k (a∗a)_k  w.r.t. a.
    The formula is  g_j = 2·∑_k w_k a_{k‑j}.
    It is implemented efficiently via FFT as a correlation of w with a.
    """
    n = a.shape[0]
    # a_rev = a reversed
    a_rev = a[::-1]

    # length needed for linear convolution of w (len 2n‑1) and a_rev (len n)
    L = next_pow2((2 * n - 1) + n - 1)
    Fw = np.fft.rfft(w, L)
    Fa = np.fft.rfft(a_rev, L)

    conv_rev = np.fft.irfft(Fw * Fa, L)[: (2 * n - 1) + n - 1]  # length 3n‑2
    # slice the central n entries: indices n‑1 … n‑1+n‑1
    g = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return g


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------
def run_adam_smooth_max(
    n: int,
    init_a: np.ndarray,
    deadline: float,
    max_iters: int = 4000,
    alpha_start: float = 8.0,
    alpha_end: float = 400.0,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max approximation of max(a∗a).
    Operates on the simplex via softmax(logits).
    Returns (best_a, best_exact_value) discovered before *deadline*.
    """
    # initialise logits from the probability vector (add tiny epsilon)
    x = np.log(init_a + 1e-15)

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.07

    best_val = np.inf
    best_a = None
    last_improve = time.time()

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # anneal temperature
        alpha = alpha_start + (alpha_end - alpha_start) * min(t / max_iters, 1.0)

        a = softmax(x)                      # a ≥ 0, Σa = 1
        conv = full_convolution(a)          # length 2n‑1
        w = smooth_weights(conv, alpha)     # soft‑max over convolution entries

        # exact objective (used for bookkeeping)
        exact_val = compute_objective(a)
        if exact_val < best_val:
            best_val = exact_val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[Adam n={n} iter={t:05d}]  exact best: {best_val:.6f}")

        # gradient of the smooth‑max surrogate
        g = gradient_smooth_max(a, w)

        # back‑propagation through softmax
        avg = np.dot(a, g)
        grad_x = a * (g - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

        # keep logits bounded
        x = np.clip(x, -50.0, 50.0)

        # shake the logits if stagnation > 5 s
        if time.time() - last_improve > 5.0:
            x += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()
            print(f"[Adam n={n} iter={t}]  shake (stagnation)")

    # fallback – uniform distribution if nothing was ever accepted
    if best_a is None:
        best_a = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = compute_objective(best_a)

    return best_a, best_val


def run_es_opt(
    n: int,
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 25000,
) -> tuple[np.ndarray, float]:
    """
    Simple (1+1) Evolution Strategy – Gaussian perturbations on the simplex.
    Returns best vector found before *deadline*.
    """
    a = init_a.copy()
    best_a = a.copy()
    best_val = compute_objective(a)

    sigma = 0.02
    last_improve = time.time()
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        # Gaussian mutation in log‑space would keep positivity automatically,
        # but we stay in the original space and clip.
        trial = a + sigma * np.random.randn(n)
        trial = np.maximum(trial, 0.0)
        s = trial.sum()
        if s == 0.0:
            trial = np.full(n, 1.0 / n)
        else:
            trial = trial / s

        val = compute_objective(trial)
        if val < best_val:
            best_val = val
            best_a = trial.copy()
            a = trial
            sigma = min(sigma * 1.05, 0.25)
            last_improve = time.time()
            print(f"[ES n={n} it={it}]  new best: {best_val:.6f}")
        else:
            sigma *= 0.99

        if time.time() - last_improve > 5.0:
            a = best_a.copy()
            sigma = 0.02
            last_improve = time.time()

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    the evaluation function.  The routine respects a time‑budget and returns
    the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_val = np.inf

    # Work on several lengths – larger n gives more degrees of freedom.
    # We start with the largest (8192) and fall back if time runs out.
    lengths = [8192, 4096, 2048, 1024, 512]

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")

        # ------------------------------------------------------------------
        # Build a small set of sensible initial candidates
        # ------------------------------------------------------------------
        init_candidates = []

        # warm‑start from the current global best (if its length matches)
        if best_seq is not None and len(best_seq) == n:
            init_candidates.append(np.asarray(best_seq, dtype=np.float64))

        # if we have a best of different length, interpolate it
        if best_seq is not None and len(best_seq) != n:
            old = np.asarray(best_seq, dtype=np.float64)
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            interpolated = np.interp(x_new, x_old, old)
            interpolated = np.maximum(interpolated, 0.0)
            s = interpolated.sum()
            if s > 1e-12:
                interpolated = interpolated / s
            else:
                interpolated = np.random.dirichlet(np.ones(n))
            init_candidates.append(interpolated)

        # monotone decreasing linear shape
        dec = np.arange(n, 0, -1, dtype=np.float64)
        dec = dec / dec.sum()
        init_candidates.append(dec)

        # uniform distribution
        init_candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

        # a few random Dirichlet draws
        for _ in range(2):
            init_candidates.append(np.random.dirichlet(np.ones(n)))

        # ------------------------------------------------------------------
        # Adam‑smooth‑max phase – several short restarts
        # ------------------------------------------------------------------
        restarts = max(1, int( (deadline - time.time()) * 0.3 // 30 ))
        for r in range(restarts):
            if time.time() > deadline:
                break
            a0 = init_candidates[r] if r < len(init_candidates) else np.random.dirichlet(np.ones(n))
            remaining = deadline - time.time()
            slice_time = min(remaining * 0.15, 30.0)   # each Adam run ≤30 s
            sub_dead = time.time() + slice_time
            a_opt, val_opt = run_adam_smooth_max(n, a0, sub_dead)
            if val_opt < best_val:
                best_val = val_opt
                best_seq = a_opt.copy()
                print(f"[n={n} restart={r}] GLOBAL best improved to {best_val:.6f}")

        # ------------------------------------------------------------------
        # (1+1)‑ES fine‑tuning of the current best for this length
        # ------------------------------------------------------------------
        if best_seq is not None and len(best_seq) == n and time.time() < deadline:
            remaining = deadline - time.time()
            es_time = min(remaining * 0.20, 60.0)
            sub_dead = time.time() + es_time
            a_es, val_es = run_es_opt(n, best_seq, sub_dead)
            if val_es < best_val:
                best_val = val_es
                best_seq = a_es.copy()
                print(f"[n={n} ES] GLOBAL best improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform distribution of a modest size if nothing was found
    # ------------------------------------------------------------------
    if best_seq is None:
        fallback_n = 1024
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)
        best_val = compute_objective(best_seq)

    print(f"\nBest value discovered: {best_val:.6f} (length {len(best_seq)})")
    # Convert to plain Python list as required by the external evaluator
    return best_seq.tolist()
