"""
Search for a non‑negative step‑function that minimizes

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The evaluation simplifies to max(conv(a, a)) when the sum is fixed to
S = √(2 n).  The algorithm proceeds as follows:

* Randomly generate an initial shape (power‑law with optional exponential
  taper and a tiny jitter).  The shape is scaled so that its sum equals S.
* Parameterise the sequence by logits x (a_i = S·softmax(x_i)) – this
  guarantees non‑negativity and the simplex constraint.
* Optimise x using Adam on a smooth approximation of the max‑operator:
  the smoothed max is the log‑sum‑exp with a temperature τ that decays
  exponentially during optimisation.  The gradient of the surrogate is
  obtained by a cross‑correlation (two FFT‑based convolutions per step).
* Every few hundred iterations a small random “mass‑transfer’’ move
  (δ from one index to another) is tried; if it lowers the true max we
  accept it and reset the optimiser state – this helps to escape plateaus.
* Many independent restarts with varying lengths are performed within the
  overall time budget (≈ 1000 s).  The best sequence found overall is
  returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper – full linear convolution using real FFT (fast for any length)
# ----------------------------------------------------------------------
def full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of real vectors x and y via FFT."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())      # next power of two
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    # Numerical noise can produce tiny negative values; clip them.
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Random initial shape (power‑law + optional exponential taper)
# ----------------------------------------------------------------------
def random_initial(n: int) -> np.ndarray:
    """Generate a positive initial array of length n."""
    p_exp = np.random.uniform(0.25, 0.75)                # power‑law exponent
    idx = np.arange(1, n + 1, dtype=float)
    init = idx ** (-p_exp)

    # optional exponential taper
    if np.random.rand() < 0.35:
        tau_exp = np.random.uniform(0.5, 2.0) * n
        init *= np.exp(-idx / tau_exp)

    # optional moving‑average smoothing
    if np.random.rand() < 0.20:
        klen = np.random.randint(3, 9)
        kernel = np.ones(klen) / klen
        init = np.convolve(init, kernel, mode='same')

    # tiny jitter to break exact symmetries
    init += np.random.rand(n) * 1e-5
    return init


# ----------------------------------------------------------------------
# Adam optimisation on the log‑it (soft‑max) parametrisation
# ----------------------------------------------------------------------
def optimise_sequence(
    n: int,
    init_a: np.ndarray,
    run_deadline: float,
    *,
    alpha0: float = 0.12,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    tau_factor_start: float = 0.18,
    tau_factor_min: float = 0.001,
) -> tuple[np.ndarray, float]:
    """
    Optimise a sequence of length ``n`` for a given time slice.
    Returns the best sequence encountered (as a NumPy array) and its
    evaluation score (max of the self‑convolution).
    """
    # target sum S = √(2 n)  (the evaluator normalises to this)
    target_sum = float(np.sqrt(2.0 * n))

    # initialise logits (log‑softmax) from the provided a
    a0 = np.maximum(init_a, 1e-12)
    a0 = a0 / a0.sum() * target_sum
    x = np.log(a0 + 1e-12)

    m = np.zeros_like(x)
    v = np.zeros_like(x)

    best_seq = None
    best_score = float("inf")
    iteration = 0
    alpha = alpha0

    while time.time() < run_deadline:
        # ---- 1. logits → probabilities → a (enforces sum = target_sum) ----
        max_x = np.max(x)
        exp_shift = np.exp(x - max_x)                # stable soft‑max
        p = exp_shift / exp_shift.sum()
        a = target_sum * p

        # ---- 2. convolution and current max ----
        conv = full_convolution_fft(a, a)
        max_val = float(conv.max())

        if max_val < best_score:
            best_score = max_val
            best_seq = a.copy()
            print(f"[{time.time():.2f}] n={n}  new local best = {max_val:.6f}")

        # ---- 3. smooth‑max weights (log‑sum‑exp) ----
        # exponentially decaying temperature τ
        tau = max(tau_factor_start * max_val * (0.95 ** iteration),
                  tau_factor_min * max_val)

        # w_k ∝ exp((conv_k - max_val)/τ)  → max weight = 1 at the current max
        scaled = (conv - max_val) / tau
        # scaled ≤ 0, safe to exponentiate
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---- 4. gradient w.r.t. a (cross‑correlation) ----
        w_rev = w[::-1]
        # convolution of a with reversed weights
        tmp = full_convolution_fft(a, w_rev)   # length = 3 n‑2
        idx = (2 * n - 2) - np.arange(n)       # picks Σ_j a_j w_{i+j}
        grad_a = 2.0 * tmp[idx]

        # ---- 5. gradient w.r.t. logits (chain rule) ----
        dotp = float(np.dot(p, grad_a))
        grad_x = target_sum * p * (grad_a - dotp)

        # ---- 6. Adam update ----
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # occasional small random perturbation to shake the optimiser
        if iteration % 250 == 0 and np.random.rand() < 0.2:
            x += np.random.randn(n) * 0.001

        # slowly decay learning‑rate
        alpha *= 0.9995

        # ---- 7. Simple local “mass‑transfer’’ move every 500 iterations ----
        if iteration % 500 == 0:
            i, j = np.random.randint(0, n, size=2)
            if i != j:
                # transfer a small fraction from the larger entry to the smaller one
                if a[i] > a[j]:
                    delta = np.random.rand() * a[i] * 0.03
                    a_new = a.copy()
                    a_new[i] -= delta
                    a_new[j] += delta
                else:
                    delta = np.random.rand() * a[j] * 0.03
                    a_new = a.copy()
                    a_new[j] -= delta
                    a_new[i] += delta

                conv_new = full_convolution_fft(a_new, a_new)
                if float(conv_new.max()) < max_val:
                    # accept the move – rebuild logits from the new a
                    x = np.log(a_new + 1e-12)
                    m = np.zeros_like(x)
                    v = np.zeros_like(x)
                    # we keep the current best_seq unchanged (it will be updated
                    # in the next outer loop pass when the new max_val is seen)

    # Final verification using the official evaluator (identical definition)
    final_score = evaluate_sequence(best_seq.tolist())
    return best_seq, final_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative height sequence that minimises the given
    evaluation function.  The routine runs for at most ``budget_s`` seconds
    and returns the best sequence discovered.

    Strategy
    --------
    * Perform many independent restarts with random lengths (≈ 500 – 3500).
    * Initialise each restart with a power‑law shape (optional exponential
      taper) or – with some probability – from the current global best.
    * Optimise the shape with Adam on a smoothed‑max surrogate; every few
      hundred iterations a tiny random mass‑transfer move is attempted.
    * Keep track of the best sequence across all runs; stop early if the
      bound falls below the target of 1.503.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.05  # tiny safety margin

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # Warm‑start from any previously discovered construction.
    # --------------------------------------------------------------
    try:
        hs = globals()["height_sequence_1"]
        hs_arr = np.array(hs, dtype=float)
        hs_arr = np.clip(hs_arr, 0.0, 1000.0)
        score = evaluate_sequence(hs_arr.tolist())
        if score < best_score:
            best_score = score
            best_seq = hs_arr.copy()
            print(f"[{time.time():.2f}] Warm‑start from global seed = {score:.6f}")
    except Exception:
        pass  # no warm‑start available

    # --------------------------------------------------------------
    # Main optimisation loop: many restarts until time runs out.
    # --------------------------------------------------------------
    while time.time() < deadline:
        remaining = deadline - time.time()

        # Choose a problem size. Early on we explore a broad band, later
        # we concentrate on moderate sizes to keep iterations cheap.
        if remaining > 300:
            n = np.random.randint(1200, 3500 + 1)
        elif remaining > 120:
            n = np.random.randint(800, 2500 + 1)
        else:
            n = np.random.randint(500, 2000 + 1)

        # ------------------- initialise candidate --------------------
        # With probability 0.45 start from the current global best (cropped/padded)
        if best_seq is not None and np.random.rand() < 0.45:
            bs = best_seq
            if bs.size >= n:
                init = bs[:n].copy()
            else:
                pad = np.random.rand(n - bs.size) * 1e-5
                init = np.concatenate([bs, pad])
        else:
            init = random_initial(n)

        # Ensure positivity for the log‑it conversion
        init = np.maximum(init, 1e-12)

        # Allocate a time slice for this run.
        # Early runs are short (≈ 5 s); later runs may consume up to 30 s.
        slice_time = min(30.0, remaining * 0.5)
        run_deadline = min(deadline, time.time() + slice_time)

        # Run the optimiser
        seq_local, score_local = optimise_sequence(
            n,
            init,
            run_deadline,
            alpha0=0.12,
            beta1=0.9,
            beta2=0.999,
            tau_factor_start=0.18,
            tau_factor_min=0.001,
        )

        # Update global best if improvement was found
        if score_local < best_score:
            best_score = score_local
            best_seq = seq_local.copy()
            print(
                f"[{time.time():.2f}] GLOBAL new best = {best_score:.6f} (n={n})"
            )
            # early exit if we comfortably beat the target bound
            if best_score < 1.503:
                break

    # ------------------------------------------------------------------
    # Fallback – if for some reason we never produced a candidate.
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        target_sum = np.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)

    # Return a plain Python list as required by the evaluator.
    return best_seq.tolist()
