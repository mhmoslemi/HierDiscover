"""
Propose a step‑function sequence that (approximately) minimises

        F(a) = 2·n·max(a*a) / (sum(a)²)

under a_i ≥ 0.

Algorithm
---------
1. Choose a target length (preferably the length of the best known
   sequence – 4096 in the current archive) and initialise a set of
   symmetric candidate distributions (random Dirichlet, power‑law,
   double‑Gaussian, raised‑cosine, up‑sampled historic best, …).
2. For every candidate run a two‑phase local optimiser
   * **Adam on a smooth‑max surrogate** – the max of the convolution
     is approximated by a soft‑max with a temperature λ that is
     increased slowly.  The parameters are re‑parameterised with a
     soft‑max (logits) so the simplex constraint (non‑negative,
     unit‑sum) is satisfied automatically.  After each Adam step the
     vector is symmetrised to keep the search in the subspace of
     palindromic step functions.
   * **Exact sub‑gradient descent** – after Adam the true max is
     identified, the sub‑gradient is obtained from the active set of
     maximal convolution entries and an exponentiated‑gradient update
     (mirror descent on the simplex) is applied, again keeping the
     vector symmetric.
3. The best sequence found so far is further refined with a cheap
   random‑perturbation hill‑climber that respects the simplex and
   symmetry constraints.  Small zero‑mean noise vectors are added,
   negative entries are clipped and the vector is re‑normalised.
4. The whole search is stopped as soon as the allocated time budget
   (default 1000 s) is exhausted; the best sequence discovered is
   returned as a plain Python list.

The implementation is completely self‑contained, uses only NumPy and
the standard library, and respects the 2‑CPU limit by performing only
FFT‑based convolutions (O(n log n)).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT helpers -----------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Return the next power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2·len(a)-1) using FFT of size ``fft_len``.
    """
    n = a.shape[0]
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Compute the linear cross‑correlation (w ★ a) needed for the gradient.
    Result length = len(a).  Implements the formula used in the original
    code: (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)] .
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


# ----------------------------------------------------------------------
# Optimisation primitives ----------------------------------------------

def _adam_optimize(
    a0: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_corr: int,
    deadline: float,
    best_val: float,
    best_seq: np.ndarray | None,
) -> tuple[np.ndarray, float, np.ndarray | None]:
    """
    Adam on the smooth‑max surrogate.
    Returns the best sequence seen during this run, its value and the
    (possibly updated) global best.
    """
    # initialise logits (soft‑max parametrisation)
    eps = 1e-15
    logits = np.log(a0 + eps)

    # Adam state
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05

    # temperature schedule
    lam = 0.5
    lam_factor = 1.05  # a bit faster than the original 1.02

    # iteration budget – a few thousand per length, scaled with n
    max_iters = int(25000 * 1024 / n) + 5000

    # time‑tracking
    last_improve = time.time()

    a = None  # will be filled each iteration

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # ----- simplex via soft‑max (and enforce symmetry) ----------
        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()
        # enforce symmetry (palindromic shape)
        a = (a + a[::-1]) * 0.5
        # keep logits symmetric as well
        logits = (logits + logits[::-1]) * 0.5

        # ----- convolution and smooth‑max weights --------------------
        conv = _full_convolution(a, fft_len_conv)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the surrogate ----------------------------
        grad_a = 2.0 * _cross_correlation(w, a, fft_len_corr)

        # back‑prop through soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ----- Adam update -----------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- temperature schedule --------------------------------
        lam = min(lam * lam_factor, 1e6)

        # ----- exact value (sum(a)=1) -----------------------------
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Adam n={n} iter={t:05d}] new best {val:.6f}")

        # ----- stagnation handling ---------------------------------
        if time.time() - last_improve > 4.0:
            # small random shake of logits to escape flat regions
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

    return a, best_val, best_seq


def _subgrad_optimize(
    a0: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_corr: int,
    deadline: float,
    best_val: float,
    best_seq: np.ndarray | None,
) -> tuple[np.ndarray, float, np.ndarray | None]:
    """
    Exact sub‑gradient descent on the true max‑convolution.
    Uses exponentiated‑gradient updates (mirror descent on the simplex).
    """
    a = a0 / a0.sum()
    eta = 0.6  # step size for exponentiated update
    sub_last_improve = time.time()

    for sub_it in range(4000):
        if time.time() > deadline:
            break

        conv = _full_convolution(a, fft_len_conv)
        max_c = conv.max()

        # active set (indices where conv is (almost) maximal)
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if not active.any():
            w = np.zeros_like(conv)
        else:
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()

        # sub‑gradient of max(conv) w.r.t. a
        grad_a = 2.0 * _cross_correlation(w, a, fft_len_corr)

        # exponentiated‑gradient step (keeps a ≥ 0)
        a = a * np.exp(-eta * grad_a)

        # re‑normalise and enforce symmetry
        a = (a + a[::-1]) * 0.5
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        # evaluate
        conv = _full_convolution(a, fft_len_conv)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            sub_last_improve = time.time()
            print(f"[Subgrad n={n} iter={sub_it:04d}] new best {val:.6f}")

        # early stop if no improvement for a few seconds
        if time.time() - sub_last_improve > 4.0:
            break

    return a, best_val, best_seq


def _hill_climb(
    a0: np.ndarray,
    n: int,
    fft_len_conv: int,
    deadline: float,
    best_val: float,
    best_seq: np.ndarray | None,
) -> tuple[np.ndarray, float, np.ndarray | None]:
    """
    Very cheap random‑perturbation local search.
    Generates zero‑mean Gaussian noise vectors of decreasing magnitude,
    projects onto the simplex (by clipping negatives and renormalising)
    and keeps any improvement.
    """
    a = a0.copy()
    # start with a modest sigma and reduce it slowly
    sigma = 2e-3
    sigma_decay = 0.9995

    while time.time() < deadline:
        # build a zero‑mean perturbation
        delta = np.random.randn(n) * sigma
        delta -= delta.mean()        # enforce zero sum

        a_new = a + delta
        # clip negatives, renormalise, enforce symmetry
        a_new = np.maximum(a_new, 0.0)
        a_new = (a_new + a_new[::-1]) * 0.5
        s = a_new.sum()
        if s < 1e-12:
            sigma *= sigma_decay
            continue
        a_new /= s

        conv = _full_convolution(a_new, fft_len_conv)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a_new.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a_new.copy()
            a = a_new
            print(f"[HillClimb n={n}] new best {val:.6f}")

        sigma *= sigma_decay

    return a, best_val, best_seq

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    the evaluation function.  The search respects the given time budget
    and returns the best sequence discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Determine the length to work on – use the length of the historic
    # best if it exists, otherwise fall back to 4096.
    hseq = globals().get("height_sequence_1")
    if hseq is not None:
        target_n = int(len(hseq))
    else:
        target_n = 4096

    # ------------------------------------------------------------------
    # Pre‑compute FFT lengths for the chosen size
    fft_len_conv = _next_pow2(2 * target_n - 1)
    fft_len_corr = _next_pow2(2 * target_n - 1 + target_n - 1)  # ≤ 3·n

    # ------------------------------------------------------------------
    # Helper to symmetrise and normalise a vector
    def _prepare(v: np.ndarray) -> np.ndarray:
        v = np.maximum(v, 0.0)
        v = (v + v[::-1]) * 0.5
        s = v.sum()
        if s < 1e-12:
            v = np.random.dirichlet(np.ones(target_n))
        else:
            v = v / s
        return v

    # ------------------------------------------------------------------
    # Build a list of diverse initial candidates (all symmetric)
    candidates = []

    # 1) warm start from historic best (if any)
    if hseq is not None:
        cand = np.asarray(hseq, dtype=np.float64)
        if cand.shape[0] > target_n:
            cand = cand[:target_n]
        elif cand.shape[0] < target_n:
            # simple linear up‑sampling to the new length
            x_old = np.linspace(0, 1, cand.shape[0])
            x_new = np.linspace(0, 1, target_n)
            cand = np.interp(x_new, x_old, cand)
        candidates.append(_prepare(cand))

    # 2) random Dirichlet vectors (3 variants)
    for _ in range(3):
        candidates.append(_prepare(np.random.dirichlet(np.ones(target_n))))

    # 3) power‑law edge‑concentrated shapes
    idx = np.arange(target_n, dtype=np.float64) + 1.0
    prod = idx * (target_n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.2, 7):
        shape = np.power(prod, alpha, where=prod > 0, out=np.zeros_like(prod))
        if shape.sum() > 0:
            candidates.append(_prepare(shape))

    # 4) two‑Gaussian edge shapes
    i = np.arange(target_n, dtype=np.float64)
    for gamma in [0.2, 0.5, 1.0, 2.0, 5.0]:
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((target_n - 1 - i) ** 2))
        candidates.append(_prepare(shape))

    # 5) raised‑cosine shape: a_i ∝ (1 - cos(2π i/(n‑1)))⁺
    shape = 1.0 - np.cos(2.0 * np.pi * i / (target_n - 1))
    candidates.append(_prepare(shape))

    # ------------------------------------------------------------------
    # Global trackers
    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Optimise each candidate in turn
    for cid, cand in enumerate(candidates):
        if time.time() > deadline:
            break

        # ----- Adam phase -------------------------------------------------
        a_after_adam, best_val, best_seq = _adam_optimize(
            cand,
            target_n,
            fft_len_conv,
            fft_len_corr,
            deadline,
            best_val,
            best_seq,
        )

        # ----- Sub‑gradient phase -----------------------------------------
        a_after_sub, best_val, best_seq = _subgrad_optimize(
            a_after_adam,
            target_n,
            fft_len_conv,
            fft_len_corr,
            deadline,
            best_val,
            best_seq,
        )

        # ----- Local random hill‑climb ------------------------------------
        _, best_val, best_seq = _hill_climb(
            a_after_sub,
            target_n,
            fft_len_conv,
            deadline,
            best_val,
            best_seq,
        )

        print(f"[Candidate {cid}] best value so far {best_val:.6f}")

    # ------------------------------------------------------------------
    # If nothing succeeded (unlikely) fall back to a uniform distribution.
    if best_seq is None:
        best_seq = np.full(target_n, 1.0 / target_n, dtype=np.float64)

    # Return a plain Python list as required by the external evaluator.
    return best_seq.tolist()
