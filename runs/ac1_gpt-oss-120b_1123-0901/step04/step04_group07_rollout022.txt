def conv_max(a):
    n = len(a)

    A = np.fft.rfft(a, n=Nfft)
    conv = np.fft.irfft(A*A, n=Nfft)[:2*n-1]
    return conv.max()

def optimize_one(a0, target_sum, max_iters, deadline):
    a = a0.copy()
    # ensure normalization
    a = np.maximum(a, 0.0)
    s = a.sum()
    a = a * (target_sum / s) if s>0 else np.full_like(a, target_sum/len(a))
    n = len(a)
    Nfft = 1 << ((2*n - 1).bit_length())
    best_local = a.copy()
    best_local_score = conv_max(a)
    for iter_ in range(max_iters):
        if time.time() >= deadline:
            break
        # compute convolution
        A = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A*A, n=Nfft)[:2*n-1]
        M = conv.max()
        # compute weight
        # use softmax weights with temperature tau

            style = rng.integers(0, 8)
            if style == 0:
                vec = np.full(n, target_sum / n)

                vec = g / g.sum() * target_sum

                # random exponential decay

                vec = x / x.sum() * target_sum

                # reversed exponential
                rate = rng.uniform(0.0005, 0.02)

                p = rng.uniform(0.2, 2.0)
                x = (np.arange(1, n+1) ** -p)
                vec = x / x.sum() * target_sum
            elif style == 5:

                p = rng.uniform(0.2, 2.0)
                x = (np.arange(1, n+1) ** -p)[::-1]
                vec = x / x.sum() * target_sum

                x = np.exp(-0.5 * ((np.arange(n) - mu) / sigma) ** 2)
                vec = x / x.sum() * target_sum

                    cur = b
                vec = arr / arr.sum() * target_sum
            # optional small jitter
            if rng.random() < 0.1:

                vec = np.maximum(vec + jitter, 0)
                vec = vec / vec.sum() * target_sum

            if score < best_score:

        )

    # Print final best score
    print(f"[final] best score = {best_score:.6f}, length={best_seq.shape[0]}")

    return best_seq.tolist()

def propose_candidate(seed=12345, budget_s=1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The algorithm combines:
    * a diverse set of random initial vectors,
    * a fast projected Adam optimiser (soft‑max approximation of the max‑convolution),
    * an alternating linear‑programming refinement (the “LP‑step” described in the
      paper) that provably never increases the objective,
    * a final short Adam fine‑tuning.
    The search is performed on several lengths (≤ 8000) while respecting the
    1000 s wall‑clock budget.
    """
    import time, math, numpy as np
    from scipy.optimize import linprog

    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 5.0   # safety margin

    # Helper: projection onto non‑negative vectors with a given ℓ₁‑norm
    def project(v, target):
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s > 0.0:
            v = v * (target / s)
        else:
            v = np.full_like(v, target / len(v))
        return v

    # Helper: max‑convolution using FFT
    def max_conv(v):
        n = len(v)
        Nfft = 1 << ((2 * n - 1).bit_length())
        A = np.fft.rfft(v, n=Nfft)
        conv = np.fft.irfft(A * A, n=Nfft)[: 2 * n - 1]
        return float(conv.max())

    # Soft‑max weights for a smooth approximation of the max
    def soft_weights(conv, tau):
        if tau <= 0:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
            return w, conv.max()
        maxc = conv.max()
        ex = np.exp((conv - maxc) / tau)
        sum_ex = ex.sum()
        w = ex / sum_ex
        Ltau = maxc + tau * math.log(sum_ex)
        return w, Ltau

    # Weighted reverse convolution (gradient of the smoothed max)
    def weighted_rev(a, w):
        n = len(a)
        Nfft = 1 << ((2 * n - 1).bit_length())
        A = np.fft.rfft(a, n=Nfft)
        Wrev = np.fft.rfft(w[::-1], n=Nfft)
        rev_full = np.fft.irfft(A * Wrev, n=Nfft)
        # rev_full length = 2*n-1+ n-1? Actually convolution length is n+len(w)-1 = 2n-1
        # Extract the relevant slice that corresponds to rev_i for i=0..n‑1
        # According to earlier code: idx_start = 2 * n - 2
        idx_start = 2 * n - 2
        rev = rev_full[idx_start - np.arange(n)]
        return rev

    # Adam optimiser (same flavour as the previous baseline but with a few
    # more iterations and a simplified learning‑rate schedule)
    def adam_optimize(a, n, target, max_iters, best_seq, best_score):
        a = project(a, target)
        # Adam state
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        beta1, beta2, eps = 0.9, 0.999, 1e-8
        lr0 = 0.08
        Nfft = 1 << ((2 * n - 1).bit_length())
        # temperature schedule
        tau_factor = 0.25

        for it in range(1, max_iters + 1):
            if time.time() >= deadline:
                break

            # exact convolution & max
            A = np.fft.rfft(a, n=Nfft)
            conv = np.fft.irfft(A * A, n=Nfft)[: 2 * n - 1]
            M = float(conv.max())

            # temperature
            tau = tau_factor * M * math.exp(-it / (max_iters / 8.0))

            # soft‑max weights
            if tau < 1e-9:
                mask = conv >= M - 1e-12
                cnt = mask.sum()
                w = mask.astype(float) / max(cnt, 1)
                Ltau = M
            else:
                w, Ltau = soft_weights(conv, tau)

            # gradient (2 × weighted reverse convolution)
            rev = weighted_rev(a, w)
            grad = 2.0 * rev

            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad * grad)
            m_hat = m / (1 - beta1 ** it)
            v_hat = v / (1 - beta2 ** it)
            lr = lr0 / (1 + 1e-4 * it)
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

            # projection onto feasible set
            a = project(a, target)

            # occasional jitter to shake out of plateaus
            if it % 4000 == 0:
                jitter = rng.normal(scale=0.001 * target / n, size=n)
                a = project(np.maximum(a + jitter, 0.0), target)

            cur_score = M
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[Adam] n={n:5d} it={it:7d} new best={best_score:.6f}")

        return best_seq, best_score

    # Linear‑programming refinement step
    def lp_refine(f, M, target, t):
        """Return a new vector after one LP‑step with step size t."""
        n = len(f)
        # Build the convolution matrix A such that A @ b = f * b
        rows = 2 * n - 1
        A = np.zeros((rows, n), dtype=np.float64)
        for i in range(n):
            A[i:i + n, i] = f[i]

        c = -np.ones(n)
        bounds = [(0, None)] * n

        res = linprog(c, A_ub=A, b_ub=ub, bounds=bounds, method="highs", options={"presolve": True})
        if not res.success:
            # fallback to the current vector (no improvement)
            return f
        b = res.x
        # normalise to the required ℓ₁‑norm
        b_sum = b.sum()
        if b_sum == 0:
            g = np.full_like(b, target / n)
        else:
            g = b * (target / b_sum)
        # convex combination
        f_new = (1.0 - t) * f + t * g
        # small numerical drift – re‑project
        f_new = project(f_new, target)
        return f_new

    # ----------------------------------------------------------------------
    # Load a previously discovered construction if it exists.
    base = None
    try:
        base = np.array(height_sequence_1, dtype=float)  # type: ignore
    except Exception:
        pass

    # ----------------------------------------------------------------------
    # Lengths we will explore (keep them moderate to let the LP‑step run fast).
    lengths = [400, 600, 800, 1200, 2000]

    lengths = sorted(set(int(l) for l in lengths if 200 <= l <= 8000))

    # Global best
    best_seq = None
    best_score = float("inf")

    # ----------------------------------------------------------------------
    # Main optimisation loop
    for n in lengths:
        if time.time() >= deadline:
            break
        target = math.sqrt(2.0 * n)

        # Prepare a collection of initial vectors for this length
        init_vecs = []

        # 1) uniform
        init_vecs.append(np.full(n, target / n))

        # 2) Dirichlet
        g = rng.gamma(1.0, 1.0, size=n)
        init_vecs.append(g / g.sum() * target)

        # 3) exponential decay
        rate = rng.uniform(0.0005, 0.015)
        x = np.exp(-rate * np.arange(n))
        init_vecs.append(x / x.sum() * target)

        # 4) reversed exponential
        init_vecs.append(x[::-1] / x.sum() * target)

        # 5) power‑law
        p = rng.uniform(0.2, 2.0)
        y = (np.arange(1, n + 1) ** -p)
        init_vecs.append(y / y.sum() * target)

        # 6) reversed power‑law
        init_vecs.append(y[::-1] / y.sum() * target)

        # 7) Gaussian window
        mu = rng.uniform(0.3, 0.7) * n
        sigma = rng.uniform(0.1, 0.3) * n
        z = np.exp(-0.5 * ((np.arange(n) - mu) / sigma) ** 2)
        init_vecs.append(z / z.sum() * target)

        # 8) random piecewise‑constant (3‑5 blocks)
        K = rng.integers(3, 6)
        cuts = np.sort(rng.choice(n - 1, K - 1, replace=False) + 1)
        heights = rng.random(K)
        piece = np.empty(n)
        start = 0
        for blk, end in enumerate(np.append(cuts, n)):
            piece[start:end] = heights[blk]
            start = end
        init_vecs.append(piece / piece.sum() * target)

        # if we have a base of a different size, interpolate it
        if base is not None and base.shape[0] != n:
            old_x = np.linspace(0.0, 1.0, base.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            interp = np.interp(new_x, old_x, base)
            interp = np.maximum(interp, 0.0)
            interp = interp / interp.sum() * target
            init_vecs.append(interp)

        # Random jittered copies of the above (add a few more candidates)
        extra = []
        for vec in init_vecs:
            jitter = rng.normal(scale=0.02 * target / n, size=n)
            v = np.maximum(vec + jitter, 0.0)
            v = v / v.sum() * target
            extra.append(v)
        init_vecs.extend(extra)

        # ------------------------------------------------------------------
        # Perform a short Adam optimisation from each start, then run the
        # LP‑refinement loop.
        for start_idx, a0 in enumerate(init_vecs):
            if time.time() >= deadline:
                break

            # --------------------------------------------------------------
            # Stage A – Adam (smooth) optimisation
            max_iters = 500_000
            best_seq, best_score = adam_optimize(
                a=a0.copy(),
                n=n,
                target=target,
                max_iters=max_iters,
                best_seq=best_seq if best_seq is not None else a0,
                best_score=best_score,
            )

            if time.time() >= deadline:
                break

            # --------------------------------------------------------------
            # Stage B – LP refinement (a few passes, decreasing step size)
            f = best_seq if best_seq.shape[0] == n else a0.copy()
            f = project(f, target)
            M = max_conv(f)
            t = 0.5
            for lp_iter in range(30):
                if time.time() >= deadline:
                    break
                f_new = lp_refine(f, M, target, t)
                M_new = max_conv(f_new)
                if M_new < best_score:
                    best_score = M_new
                    best_seq = f_new.copy()
                    print(f"[LP] n={n:5d} iter={lp_iter:2d} new best={best_score:.6f}")
                # accept the step if it does not increase the peak
                if M_new < M - 1e-6:
                    f = f_new
                    M = M_new
                    # slowly decay step size for stability
                    t = max(t * 0.9, 0.05)
                else:

                    t *= 0.5
                    if t < 1e-3:

            # --------------------------------------------------------------
            # Stage C – a final quick Adam fine‑tuning
            max_iters = 150_000
            best_seq, best_score = adam_optimize(
                a=f.copy(),
                n=n,
                target=target,
                max_iters=max_iters,
                best_seq=best_seq,
                best_score=best_score,

    # ----------------------------------------------------------------------

    if best_seq is None:

        target = math.sqrt(2.0 * n)

    print(f"[Done] best score = {best_score:.6f}, length = {best_seq.shape[0]}")
    # Return a plain Python list for the harness
    return best_seq.tolist()

def propose_candidate(seed: int = 12345, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.

    The search combines three ingredients:

    1. **Diversified initialisations** – uniform, Dirichlet, exponential,
       power‑law, Gaussian and piece‑wise constant shapes (plus jittered copies).

    2. **Projected Adam optimisation** – a smooth soft‑max approximation of the
       max‑convolution is used as surrogate; the iterates are constantly
       projected onto the feasible set a_i ≥ 0 and Σ a_i = √(2 n).

    3. **LP‑refinement** – after a short Adam run we apply the “alternating
       linear‑programming” step described in the statement.  For a fixed
       vector *f* we maximise Σ bₖ subject to (f * b)[ℓ] ≤ M (M = max (f*f))
       and bₖ ≥ 0.  The LP solution is rescaled to the required ℓ₁‑norm and
       blended with *f* (convex combination).  Repeating the LP step with a
       decreasing step size never increases the objective and drives the peak
       of the autocorrelation down.

    The algorithm explores several lengths (≤ 8000) while respecting the
    1000 s wall‑clock budget, keeps the best sequence seen and finally returns
    it as a plain Python list.
    """
    import time, math, numpy as np
    from scipy.optimize import linprog

    rng = np.random.default_rng(seed)
    start_time = time.time()
    # safety margin for the harness
    deadline = start_time + budget_s - 5.0

    # ----------------------------------------------------------------------
    # Helper utilities
    # ----------------------------------------------------------------------
    def project(v, target_sum):
        """Non‑negative projection with prescribed ℓ₁‑norm."""
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s > 0.0:
            v = v * (target_sum / s)
        else:
            v = np.full_like(v, target_sum / v.size)
        return v

    def max_conv(v):
        """Maximum entry of the linear convolution v*v (FFT implementation)."""
        n = v.size
        Nfft = 1 << ((2 * n - 1).bit_length())
        A = np.fft.rfft(v, n=Nfft)
        conv = np.fft.irfft(A * A, n=Nfft)[: 2 * n - 1]
        return float(conv.max())

    def soft_weights(conv, tau):
        """Stable soft‑max of convolution values."""
        if tau <= 0.0:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
            return w, conv.max()
        maxc = conv.max()
        ex = np.exp((conv - maxc) / tau)
        sum_ex = ex.sum()
        w = ex / sum_ex
        Ltau = maxc + tau * math.log(sum_ex)
        return w, Ltau

    def weighted_rev(a, w):
        """
        Compute rev_i = Σ_k w_k * a_{k-i} (indices outside range ignored)
        using an FFT.  This is the (sub‑)gradient of the (soft‑max) max‑convolution.
        """
        n = a.size
        Nfft = 1 << ((2 * n - 1).bit_length())
        A = np.fft.rfft(a, n=Nfft)
        Wrev = np.fft.rfft(w[::-1], n=Nfft)
        rev_full = np.fft.irfft(A * Wrev, n=Nfft)
        # extract the slice corresponding to rev_i for i=0..n‑1
        idx_start = 2 * n - 2
        rev = rev_full[idx_start - np.arange(n)]
        return rev

    # ----------------------------------------------------------------------
    # Adam optimiser (smooth phase → exact phase)
    # ----------------------------------------------------------------------
    def adam_optimize(a, n, target, max_iters, best_seq, best_score):
        """Projected Adam optimiser; returns the best (local) sequence encountered."""
        a = project(a, target)

        # Adam state
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        beta1, beta2, eps = 0.9, 0.999, 1e-8
        lr0 = 0.08
        tau_factor = 0.25
        Nfft = 1 << ((2 * n - 1).bit_length())

        for it in range(1, max_iters + 1):
            if time.time() >= deadline:
                break

            # exact convolution & peak
            A = np.fft.rfft(a, n=Nfft)
            conv = np.fft.irfft(A * A, n=Nfft)[: 2 * n - 1]
            M = float(conv.max())

            # temperature schedule
            tau = tau_factor * M * math.exp(-it / (max_iters / 8.0))

            # soft‑max weights (or exact active set when τ tiny)
            if tau < 1e-9:
                mask = conv >= M - 1e-12
                cnt = mask.sum()
                w = mask.astype(float) / max(cnt, 1)
                Ltau = M
            else:
                w, Ltau = soft_weights(conv, tau)

            # gradient of the (smoothed) max‑convolution
            rev = weighted_rev(a, w)
            grad = 2.0 * rev

            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad * grad)
            m_hat = m / (1 - beta1 ** it)
            v_hat = v / (1 - beta2 ** it)
            lr = lr0 / (1 + 1e-4 * it)
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

            # projection onto feasible set
            a = project(a, target)

            # occasional jitter to escape plateaus
            if it % 4000 == 0:
                jitter = rng.normal(scale=0.001 * target / n, size=n)
                a = project(np.maximum(a + jitter, 0.0), target)

            cur_score = M
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[Adam] n={n:5d} it={it:7d} new best={best_score:.6f}")

        return best_seq, best_score

    # ----------------------------------------------------------------------
    # LP refinement step – the “alternating linear programming” move
    # ----------------------------------------------------------------------
    def lp_refine(f, M, target, step):
        """One LP‑step with step size *step* (0 < step ≤ 1)."""
        n = f.size
        rows = 2 * n - 1
        A = np.zeros((rows, n), dtype=np.float64)
        for i in range(n):
            A[i:i + n, i] = f[i]               # each column is a shifted copy of f[i]

        # maximise Σ b  ⇔ minimise –Σ b
        c = -np.ones(n)
        bounds = [(0, None)] * n
        ub = M * np.ones(rows)

        res = linprog(c, A_ub=A, b_ub=ub, bounds=bounds,
                      method="highs", options={"presolve": True})
        if not res.success:
            # If the LP fails, fall back to the current vector.
            return f

        b = res.x
        s = b.sum()
        if s == 0.0:
            g = np.full_like(b, target / n)
        else:
            g = b * (target / s)                # renormalise to the required ℓ₁‑norm

        f_new = (1.0 - step) * f + step * g
        f_new = project(f_new, target)
        return f_new

    # ----------------------------------------------------------------------
    # Load previously discovered construction if it exists.
    # ----------------------------------------------------------------------
    base = None
    try:
        base = np.array(height_sequence_1, dtype=float)  # type: ignore
    except Exception:
        pass

    # ----------------------------------------------------------------------
    # Lengths to explore (keep them modest so the LP step stays cheap).
    # ----------------------------------------------------------------------
    lengths = [400, 600, 800, 1200, 2000]

    # ----------------------------------------------------------------------

    # ----------------------------------------------------------------------
    for n in lengths:

        # ---- generate a diverse pool of initial vectors -----------------
        init_vecs = []

        # 2) Dirichlet

        init_vecs.append(g / g.sum() * target_sum)

        # 3) exponential decay
        rate = rng.uniform(0.0005, 0.015)
        x = np.exp(-rate * np.arange(n))
        init_vecs.append(x / x.sum() * target_sum)

        # 4) reversed exponential
        init_vecs.append(x[::-1] / x.sum() * target_sum)

        # 5) power‑law
        p = rng.uniform(0.2, 2.0)
        y = (np.arange(1, n + 1) ** -p)
        init_vecs.append(y / y.sum() * target_sum)

        # 6) reversed power‑law
        init_vecs.append(y[::-1] / y.sum() * target_sum)

        # 7) Gaussian window
        mu = rng.uniform(0.3, 0.7) * n
        sigma = rng.uniform(0.1, 0.3) * n
        z = np.exp(-0.5 * ((np.arange(n) - mu) / sigma) ** 2)
        init_vecs.append(z / z.sum() * target_sum)

        # 8) random piecewise‑constant (3–5 blocks)
        K = rng.integers(3, 6)

        heights = rng.random(K)
        piece = np.empty(n)
        start = 0
        for blk, end in enumerate(np.append(cuts, n)):
            piece[start:end] = heights[blk]
            start = end
        init_vecs.append(piece / piece.sum() * target_sum)

        # interpolated base if it has a different size
        if base is not None and base.shape[0] != n:

            interp = np.interp(new_x, old_x, base)
            interp = np.maximum(interp, 0.0)

        init_vecs.extend(extra)

        # ------------------------------------------------------------------
        # Optimise each start – light Adam → LP refinement → short Adam fine‑tune
        # ------------------------------------------------------------------
        for start_id, a0 in enumerate(init_vecs):
