def propose_candidate(seed=0, budget_s=1000):
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5

    best_seq = None
    best_val = float('inf')

    def exact_f(a):
        conv = _full_convolution(a, _next_pow2(2*len(a)-1))
        return 2 * len(a) * conv.max()

    lengths = [1024, 2048, 4096, 8192, 16384]
    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n--- Optimizing length {n} ---")
        fft_len = _next_pow2(2*n-1)

        for restart in range(5):
            if time.time() > deadline:
                break
            # init
            if best_seq is not None and best_seq.shape[0] == n:
                a = best_seq.copy()
                print("[restart] warm start")
            else:
                # random initialization options
                init_type = rng.integers(0,3)
                if init_type == 0:
                    a = rng.dirichlet(np.ones(n))
                elif init_type == 1:
                    # sparse random: choose k positions
                    k = max(2, n // 30)  # about 3% non-zero
                    idx = rng.choice(n, size=k, replace=False)
                    vals = rng.random(k)
                    vals /= vals.sum()
                    a = np.zeros(n)
                    a[idx] = vals
                else:
                    # geometric decay
                    r = rng.uniform(0.2, 0.8)
                    weights = r**np.arange(n)
                    weights /= weights.sum()
                    a = weights

            # Adam or simple momentum:
            m = np.zeros_like(a)
            v = np.zeros_like(a)
            beta1, beta2, eps = 0.9, 0.999, 1e-8
            lr = 0.12
            t = 0

            # schedule for smoothing factor
            lam = 0.3
            lam_factor = 1.05

            last_improve = time.time()
            max_iters = 60000
            while time.time() < deadline and t < max_iters:
                conv = _full_convolution(a, fft_len)
                max_c = conv.max()
                # Determine indices within small gap
                thr = max_c - 1e-6  # small epsilon
                idxs = np.where(conv >= thr)[0]
                if len(idxs) > 1:
                    w = np.zeros_like(conv)
                    w[idxs] = 1.0 / len(idxs)
                else:
                    w = np.zeros_like(conv)
                    w[conv.argmax()] = 1.0

                # gradient (using cross-correlation)
                grad = 2.0 * _cross_correlation(w, a, _next_pow2( len(w)+len(a) ))
                # Adam updates
                t += 1
                m = beta1 * m + (1-beta1) * grad
                v = beta2 * v + (1-beta2) * (grad * grad)
                m_hat = m / (1 - beta1**t)
                v_hat = v / (1 - beta2**t)
                step = lr * m_hat / (np.sqrt(v_hat) + eps)
                a_new = a - step
                a = _project_simplex(a_new)

                # schedule
                lam = min(lam * lam_factor, 1e6)  # not used but keep for compatibility

                # evaluation
                val = exact_f(a)

                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} restart={restart} iter={t}] new best {val:.6f}")
                    if best_val < 1.503:
                        print("Target reached!")
                        deadline = time.time()

                if time.time() - last_improve > 6.0:
                    # shake
                    noise = (rng.random(n) - 0.5) * 1e-2
                    a = _project_simplex(a + noise)
                    last_improve = time.time()
                    print(f"[n={n} restart={restart} iter={t}] shake")
            # restart loop
    # final refinement
    if best_seq is not None and time.time() < deadline:
        a = best_seq.copy()
        n = a.shape[0]
        fft_len = _next_pow2(2*n-1)
        print("\n--- Final refinement ---")
        while time.time() < deadline:
            i, j = rng.choice(n, 2, replace=False)
            delta = 0.001 * a[i] * rng.random()
            if delta <= 0: continue
            a_prop = a.copy()
            a_prop[i] -= delta
            a_prop[j] += delta
            a_prop = _project_simplex(a_prop)
            val = exact_f(a_prop)
            if val < best_val:
                best_val = val
                best_seq = a_prop.copy()
                a = a_prop
                print(f"[refine] improved to {val:.6f}")

    # fallback uniform
    if best_seq is None:
        n = 2048
        best_seq = np.full(n, 1.0 / n)
        best_val = exact_f(best_seq)
        print(f"Fallback uniform: {best_val}")

    return best_seq.tolist()

"""
Hybrid optimizer for the max‑convolution objective

    F(a) = 2·n·max(a*a) / (sum(a)**2)

The algorithm works on the probability simplex (∑a = 1, a_i ≥ 0) and
combines three ingredients

1. **Projected Adam** on the simplex using the exact sub‑gradient of the
   current maximum convolution entry.  The gradient for a given shift k
   is simply a shifted copy of the vector a, which allows a very cheap
   O(n) computation.

2. **Stagnation handling** – after a few seconds without improvement a
   small random “shake” is added and the Adam moments are reset.  This
   helps to escape plateaus.

3. **Final local random refinement** – a fast hill‑climbing phase that
   moves a tiny amount of mass between two coordinates and accepts the
   move only if the exact objective improves.  It is run until the
   overall time budget is exhausted.

The code respects a global time budget (default 1000 s) and returns the
best sequence found as a plain Python list, ready to be passed to the
provided ``evaluate_sequence`` function.
"""

import time
import numpy as np


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*a (full length 2·len(a)-1) using FFT.
    ``fft_len`` must be ≥ 2·len(a)-1 and a power of two.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.shape[0] - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        {x | x_i ≥ 0, Σ_i x_i = 1}
    """
    if v.ndim != 1:
        raise ValueError("Input must be a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_grad(a: np.ndarray, fft_len: int):
    """
    Compute the full convolution, the index of the maximal entry,
    and the sub‑gradient of that entry with respect to ``a``.
    The gradient is 2 * a shifted by the maximising lag.
    """
    conv = _full_convolution(a, fft_len)
    kmax = int(conv.argmax())
    n = a.shape[0]
    idx = np.arange(n)
    j = kmax - idx
    mask = (j >= 0) & (j < n)
    grad = np.zeros_like(a)
    grad[mask] = 2.0 * a[j[mask]]
    return conv, kmax, grad


def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (vector) that minimises
    ``evaluate_sequence``.  The routine respects the given time budget
    and returns the best candidate as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None          # numpy array, ∑a = 1
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Exact objective for a probability vector (∑a = 1)
    def _exact_obj(a: np.ndarray) -> float:
        n = a.shape[0]
        conv = _full_convolution(a, _next_pow2(2 * n - 1))
        return float(2.0 * n * conv.max())

    # ------------------------------------------------------------------
    # Problem sizes to explore (larger n → more degrees of freedom)
    lengths = [1024, 2048, 4096, 8192, 16384]

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n = {n} ===")
        fft_len = _next_pow2(2 * n - 1)

        # number of independent restarts for this length
        restarts = 4

        for r in range(restarts):
            if time.time() > deadline:
                break

            # -------------------- initialise --------------------
            if r == 0 and best_seq is not None and best_seq.shape[0] == n:
                a = best_seq.copy()
                print("[restart 0] warm start from global best")
            elif 'height_sequence_1' in globals():
                cand = np.asarray(height_sequence_1, dtype=np.float64)
                if cand.shape[0] >= n:
                    cand = cand[:n]
                else:
                    cand = np.pad(cand, (0, n - cand.shape[0]))
                cand = np.maximum(cand, 0.0)
                s = cand.sum()
                a = cand / s if s >= 0.01 else rng.dirichlet(np.ones(n))
                print("[restart 0] start from height_sequence_1")
            else:
                # three simple random constructions, chosen uniformly
                mode = rng.integers(0, 3)
                if mode == 0:
                    a = rng.dirichlet(np.ones(n))
                elif mode == 1:
                    # sparse random support
                    k = max(2, n // 30)                 # ≈ 3 % non‑zero
                    idx = rng.choice(n, size=k, replace=False)
                    vals = rng.random(k)
                    vals /= vals.sum()
                    a = np.zeros(n)
                    a[idx] = vals
                else:
                    # geometric decay, then normalised
                    r = rng.uniform(0.2, 0.8)
                    a = r ** np.arange(n)
                    a /= a.sum()
                print("[restart] random start")

            # -------------------- Adam state --------------------
            m = np.zeros_like(a)
            v = np.zeros_like(a)
            beta1, beta2, eps = 0.9, 0.999, 1e-8
            lr = 0.08
            t = 0
            last_improve = time.time()
            max_iters = 50000

            # ------------------------------------------------------------------
            # Main optimisation loop (projected Adam with exact max‑gradient)
            # ------------------------------------------------------------------
            while time.time() < deadline and t < max_iters:
                conv, kmax, grad = _max_grad(a, fft_len)

                # Adam update
                t += 1
                m = beta1 * m + (1.0 - beta1) * grad
                v = beta2 * v + (1.0 - beta2) * (grad * grad)
                m_hat = m / (1.0 - beta1 ** t)
                v_hat = v / (1.0 - beta2 ** t)
                step = lr * m_hat / (np.sqrt(v_hat) + eps)

                a_new = a - step
                a = _project_simplex(a_new)

                # exact objective (no extra cost – we already have conv.max())
                val = float(2.0 * n * conv.max())

                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} r={r} it={t:05d}] new best: {val:.6f}")

                    if best_val < 1.503:
                        print("Target reached – terminating early.")
                        deadline = time.time()
                        break

                # ------------------------------------------------------------------
                # Stagnation handling – add a tiny perturbation and reset Adam
                # ------------------------------------------------------------------
                if time.time() - last_improve > 5.0:
                    noise = (rng.random(n) - 0.5) * 1e-2
                    a = _project_simplex(a + noise)
                    m.fill(0.0)
                    v.fill(0.0)
                    last_improve = time.time()
                    print(f"[n={n} r={r} it={t}] shake (stagnation)")

            # end of one restart
        # end of lengths loop

    # ------------------------------------------------------------------
    # Final local random refinement (hill climbing)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        a = best_seq.copy()
        n = a.shape[0]
        fft_len = _next_pow2(2 * n - 1)
        print("\n--- Local random refinement ---")
        stagnation_counter = 0
        while time.time() < deadline and stagnation_counter < 20000:
            i, j = rng.choice(n, 2, replace=False)
            if a[i] < 1e-12:
                continue
            # move up to 1 % of the mass of i towards j
            delta = a[i] * 0.01 * rng.random()
            a_prop = a.copy()
            a_prop[i] -= delta
            a_prop[j] += delta
            a_prop = _project_simplex(a_prop)
            conv = _full_convolution(a_prop, fft_len)
            val = float(2.0 * n * conv.max())
            if val < best_val:
                best_val = val
                best_seq = a_prop.copy()
                a = a_prop
                stagnation_counter = 0
                print(f"[refine] improved to {val:.6f}")
            else:
                stagnation_counter += 1

    # ------------------------------------------------------------------
    # Fallback – uniform vector if nothing succeeded
    # ------------------------------------------------------------------
    if best_seq is None:
        n_fallback = 2048
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        best_val = _exact_obj(best_seq)
        print(f"Fallback uniform sequence, value = {best_val:.6f}")

    # Return as plain Python list for the external evaluator
    return best_seq.tolist()
