def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))

def random_shape(rng, n):
    choice = rng.integers(0, 8)
    x = np.arange(n)
    mu = (n-1)/2
    if choice == 0:
        # uniform
        return np.ones(n)
    elif choice == 1:
        # Dirichlet
        g = rng.gamma(1.0, 1.0, size=n)
        return g
    elif choice == 2:
        # exponential
        rate = rng.uniform(0.0005, 0.02)
        return np.exp(-rate * np.arange(n))
    elif choice == 3:
        # reversed exponential
        rate = rng.uniform(0.0005, 0.02)
        return np.exp(-rate * np.arange(n)[::-1])
    elif choice == 4:
        # gaussian
        sigma = rng.uniform(0.1*n, 0.5*n)
        return np.exp(-((x - mu) ** 2) / (2 * sigma ** 2))
    elif choice == 5:
        # laplace
        scale = rng.uniform(0.1*n, 0.5*n)
        return np.exp(-np.abs(x - mu) / scale)
    elif choice == 6:
        # triangular width
        w = rng.uniform(0.1*n, 0.5*n)
        return np.maximum(0.0, 1 - np.abs(x - mu) / w)
    else:
        # raised cosine
        w = rng.uniform(0.1*n, 0.5*n)
        arr = np.where(np.abs(x - mu) <= w,
                       0.5 * (1 + np.cos(np.pi * (x - mu) / w)),
                       0.0)
        return arr

def compute_score(a):
    n = a.shape[0]
    Nfft = 1 << ((2*n - 2).bit_length())
    A = np.fft.rfft(a, n=Nfft)
    conv = np.fft.irfft(A * A)[:2*n-1]
    max_b = conv.max()
    sum_a = a.sum()
    return 2.0 * n * max_b / (sum_a * sum_a)

def compute_score(a):
    # a is numpy array
    n = a.shape[0]
    # Fast convolution via FFT
    Nfft = 1 << ((2 * n - 2).bit_length())
    A = np.fft.rfft(a, n=Nfft)
    conv = np.fft.irfft(A * A, n=Nfft)[: 2 * n - 1]
    max_b = float(conv.max())
    sum_a = float(a.sum())
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a * sum_a)

"""
Propose a step‑function (non‑negative vector) that minimises the
``evaluate_sequence`` objective.

Algorithm
---------
* Choose a few target lengths (original size, 1.5×, 2×, capped at 8000).
* For each length perform a massive random‑shape search:
    – draw vectors from many families (Dirichlet, exponential,
      reversed exponential, Gaussian, Laplace, triangular, raised‑cosine,
      power‑law, random sparse spikes) and normalise them to
      Σa = √(2 n).
    – Keep the best vector found for that length.
* Refine the best vectors with a projected Adam optimiser that
  minimises a smooth approximation of the max‑convolution.  The optimiser
  uses a soft‑max temperature that decays slowly, an occasional jitter,
  and projection onto the feasible set (a_i ≥ 0, Σa = √(2 n)).
* After the Adam phase run a very cheap simulated‑annealing style
  pair‑wise mass‑transfer local search while time permits.
* The globally best sequence over all lengths is returned.
* The whole routine respects the 1000 s wall‑clock budget.

The implementation relies only on NumPy and FFTs for fast convolution.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility functions ----------------------------------------------------
# ----------------------------------------------------------------------


def compute_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) / (sum(a)^2)."""
    n = a.shape[0]
    # next power‑of‑two ≥ 2·n‑1 for a full convolution via FFT
    Nfft = 1 << ((2 * n - 2).bit_length())
    A = np.fft.rfft(a, n=Nfft)
    conv = np.fft.irfft(A * A, n=Nfft)[: 2 * n - 1]
    max_b = float(conv.max())
    s = float(a.sum())
    if s < 0.01:
        return float("inf")
    return 2.0 * n * max_b / (s * s)


def _softmax_weights(conv: np.ndarray, tau: float):
    """Stable soft‑max of convolution values. Returns (weights, smoothed max)."""
    maxc = conv.max()
    if tau <= 0.0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w, maxc
    ex = np.exp((conv - maxc) / tau)
    sum_ex = ex.sum()
    w = ex / sum_ex
    Ltau = maxc + tau * np.log(sum_ex)
    return w, Ltau


def _weighted_rev(a: np.ndarray, w: np.ndarray, Nfft: int) -> np.ndarray:
    """
    Compute rev_i = Σ_k w_k * a_{k-i}   (indices outside range are ignored)
    using FFTs.
    """
    A_fft = np.fft.rfft(a, n=Nfft)
    W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)
    conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
    conv_rev = conv_rev[: len(w) + a.shape[0] - 1]
    idx_start = 2 * a.shape[0] - 2
    rev = conv_rev[idx_start - np.arange(a.shape[0])]
    return rev


# ----------------------------------------------------------------------
# Adam optimiser -------------------------------------------------------
# ----------------------------------------------------------------------


def _adam_optimize(
    a: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int,
    best_seq: np.ndarray,
    best_score: float,
) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimiser that minimises max(a*a).  Returns the best
    sequence discovered in this run together with its exact objective value.
    """
    target_sum = math.sqrt(2.0 * n)

    # Normalise the start vector
    a = a * (target_sum / a.sum())

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr0 = 0.05
    jitter_interval = 2000

    # Temperature schedule (soft‑max)
    tau_factor = 0.30                # start with a fairly warm temperature
    decay_rate = max_iters / 12.0    # slower decay than the reference code

    Nfft = 1 << ((2 * n - 2).bit_length())   # power‑of‑two ≥ 2·n‑1

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ----- exact convolution of current a -----
        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        M = float(conv.max())

        # ----- soft‑max temperature & weighting -----
        tau = tau_factor * M * math.exp(-it / decay_rate)
        if tau < 1e-9:
            # exact active set of the max;
            mask = conv >= M - 1e-12
            cnt = mask.sum()
            w = mask.astype(float) / max(cnt, 1)
            Ltau = M
        else:
            w, Ltau = _softmax_weights(conv, tau)

        # ----- sub‑gradient w.r.t. a (weighted reverse convolution) -----
        rev = _weighted_rev(a, w, Nfft)

        # ----- Adam update (projected) -----
        grad = rev                     # descent direction for minimising M

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / (1.0 + 1e-4 * it)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # ----- projection onto feasible set -----
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a = a * (target_sum / s)
        else:
            a = np.full_like(a, target_sum / n)

        # occasional jitter to escape plateaus
        if it % jitter_interval == 0:
            jitter = np.random.randn(n) * (1e-3 * target_sum / n)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        # ----- exact objective after projection -----
        cur_score = compute_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[len {n:5d} iter {it:8d}] new best = {best_score:.6f}")

    return best_seq, best_score

# Random shape generators -----------------------------------------------

def _random_vector(rng: np.random.Generator, n: int) -> np.ndarray:
    """
    Produce a random non‑negative vector of length n from a diverse family.
    The raw vector is NOT normalised; caller should normalise to the target sum.
    """
    x = np.arange(n, dtype=float)
    mu = (n - 1) / 2.0
    choice = rng.integers(0, 10)

    if choice == 0:
        # Dirichlet / Gamma(1,1)
        g = rng.gamma(1.0, 1.0, size=n)
        return g
    elif choice == 1:
        # Uniform
        return np.ones(n)
    elif choice == 2:
        # Exponential decay from the left
        rate = rng.uniform(0.0005, 0.02)
        return np.exp(-rate * x)
    elif choice == 3:
        # Reversed exponential (decay from the right)
        rate = rng.uniform(0.0005, 0.02)
        return np.exp(-rate * (n - 1 - x))
    elif choice == 4:
        # Gaussian centred in the middle
        sigma = rng.uniform(0.08 * n, 0.35 * n)
        return np.exp(-((x - mu) ** 2) / (2.0 * sigma * sigma))
    elif choice == 5:
        # Laplace (double‑exponential)
        scale = rng.uniform(0.08 * n, 0.35 * n)
        return np.exp(-np.abs(x - mu) / scale)
    elif choice == 6:
        # Triangular window (linear rise–fall)
        width = rng.uniform(0.3 * n, 0.9 * n)
        return np.maximum(0.0, 1.0 - np.abs(x - mu) / width)
    elif choice == 7:
        # Raised‑cosine (smooth bump)
        width = rng.uniform(0.2 * n, 0.6 * n)
        mask = np.abs(x - mu) <= width
        arr = np.zeros_like(x)
        arr[mask] = 0.5 * (1.0 + np.cos(np.pi * (x[mask] - mu) / width))
        return arr
    elif choice == 8:
        # Power‑law decay (i+1)^{-p}
        p = rng.uniform(0.3, 2.0)
        return (x + 1) ** (-p)
    else:
        # Sparse random spikes – useful to explore extreme shapes
        k = rng.integers(2, min(10, n // 10) + 1)
        idx = rng.choice(n, size=k, replace=False)
        a = np.zeros(n)
        a[idx] = 1.0
        return a


def _interpolate(old: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of an existing vector to a new length."""
    old_x = np.linspace(0.0, 1.0, old.shape[0])
    new_x = np.linspace(0.0, 1.0, new_len)
    interp = np.interp(new_x, old_x, old)
    interp = np.maximum(interp, 0.0)
    return interp

def propose_candidate(seed: int = 12345, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence as a plain Python list.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a few seconds safety margin
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # If a previous construction exists, load it.
    try:
        base = np.array(height_sequence_1, dtype=float)  # type: ignore
    except Exception:
        base = None

    # ------------------------------------------------------------------
    # Choose lengths to explore.
    if base is not None:
        n0 = base.shape[0]
    else:
        n0 = 2100                     # a reasonable default
    factors = [0.75, 1.0, 1.5, 2.0]
    lengths = sorted({max(300, min(int(n0 * f), 8000)) for f in factors})

    # ------------------------------------------------------------------
    # Global best tracking
    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Main loop over the selected lengths
    for n in lengths:
        if time.time() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)
        print(f"[run] length={n}, target Σ={target_sum:.3f}")

        # ---------- Phase 1 : massive random sampling ----------
        max_random = 5000 if n <= 4000 else 3000
        for i in range(max_random):
            if time.time() >= deadline:
                break
            a = _random_vector(rng, n)
            a = a * (target_sum / a.sum())
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
                print(f"[len {n}] new random best = {best_score:.6f}")

        # Incorporate the previous best (interpolated) if it has a different size
        init_vectors = []
        if best_seq is not None and best_seq.shape[0] == n:
            init_vectors.append(best_seq.copy())
        if base is not None and base.shape[0] != n:
            interp = _interpolate(base, n)
            interp = interp * (target_sum / interp.sum())
            init_vectors.append(interp)

        # add a few fresh random starts for the optimiser
        for _ in range(3):
            a = _random_vector(rng, n)
            a = a * (target_sum / a.sum())
            init_vectors.append(a)

        # ---------- Phase 2 : Adam refinement ----------
        # allocate a modest number of iterations – more iterations for smaller n
        max_iters = int(250_000 * 2100 / n)
        max_iters = max(150_000, max_iters)

        for idx, a0 in enumerate(init_vectors):
            if time.time() >= deadline:
                break
            print(f"  optimiser start {idx+1}/{len(init_vectors)} for length {n}")
            best_seq, best_score = _adam_optimize(
                a=a0,
                n=n,
                deadline=deadline,
                max_iters=max_iters,
                best_seq=best_seq if best_seq is not None else a0,
                best_score=best_score,
            )

        # ---------- Phase 3 : cheap local pair‑wise mass transfer ----------
        # Run while we still have time left.
        while time.time() < deadline:
            # pick two distinct indices
            i, j = rng.choice(n, size=2, replace=False)
            if best_seq[i] <= 0:
                continue
            # transfer a random fraction (up to 10 %) from i to j
            frac = rng.random() * 0.10
            delta = best_seq[i] * frac
            cand = best_seq.copy()
            cand[i] -= delta
            cand[j] += delta
            # sum is invariant; only need to enforce non‑negativity
            cand = np.maximum(cand, 0.0)
            # renormalise (protect against tiny round‑off loss)
            cand = cand * (target_sum / cand.sum())
            sc = compute_score(cand)
            if sc < best_score:
                best_score = sc
                best_seq = cand
                print(f"[local] improvement to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback if nothing succeeded (should never happen)
    if best_seq is None:
        n = n0
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n)
        best_score = compute_score(best_seq)
        print("[fallback] using uniform sequence")

    print(f"[done] best score = {best_score:.6f}, length = {best_seq.shape[0]}")
    return best_seq.tolist()
