def project_onto_simplex(v, s=1.0):
    # Duchi et al. 2008
    n = len(v)
    if s < 0:
        raise ValueError
    # Sort v decreasing
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Simulated‑annealing + projected gradient search for a non‑negative step
function minimizing

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm works on the simplex (a_i ≥ 0, Σ a_i = 1); the objective
is scale‑invariant, so fixing the sum to 1 does not affect the score.
Inside the search we

* initialise a variety of shapes (power‑law, “U”‑shaped, flat,
  random) for a random length n,
* repeatedly propose a tiny transfer of mass between two positions,
  accepting it with a Metropolis rule (simulated annealing),
* occasionally perform a projected‑gradient step based on the
  sub‑gradient of the current maximal convolution entry,
* keep the globally best sequence found (including a possible warm‑start
  from the global variable ``height_sequence_1``).

All heavy work (full linear convolution, sub‑gradient) is performed
with a real FFT, which is fast up to several thousand coefficients.
The routine runs for at most ``budget_s`` seconds, prints progress and
returns the best sequence as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT based full linear convolution (real input → real output)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of two real vectors using FFT."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    # remove possible tiny negatives caused by floating error
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Projection of a vector onto the probability simplex:
#    {z | z_i ≥ 0, Σ z_i = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex with total mass ``s``."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    n = v.shape[0]

    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    # find the last index where u_i > (cumulative_sum_i - s) / i
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero after projection
        return np.zeros_like(v)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Compute the exact evaluation score used by the official evaluator.
# ----------------------------------------------------------------------
def _score_of_sequence(a: np.ndarray) -> float:
    """
    Return the value of ``evaluate_sequence`` for a given array ``a``.
    The implementation mirrors the security checks of the official
    function (clipping to [0,1000] and the 0.01 sum guard).
    """
    a = np.clip(a, 0.0, 1000.0)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return float("inf")
    conv = _full_convolution_fft(a, a)
    max_b = conv.max()
    n = a.shape[0]
    return float(2.0 * n * max_b / (sum_a ** 2))


# ----------------------------------------------------------------------
# Sub‑gradient of the max‑convolution (sum of gradients of all active
# indices). Returns a vector of the same length as ``a``.
# ----------------------------------------------------------------------
def _subgradient_max_conv(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    conv = _full_convolution_fft(a, a)
    max_val = conv.max()
    # tolerance to decide which entries are “active”
    eps = max_val * 1e-12 + 1e-15
    active = conv >= max_val - eps
    # weight vector of length 2n‑1 (1 where active, 0 elsewhere)
    w = active.astype(float)
    # reverse w for the cross‑correlation that yields the gradient
    w_rev = w[::-1]
    conv2 = _full_convolution_fft(a, w_rev)           # length = n + (2n‑1) – 1 = 3n‑2
    idx = (2 * n - 2) - np.arange(n)                  # pick the correct slice
    grad = 2.0 * conv2[idx]                           # ∂/∂a of Σ_active conv_i
    return grad


# ----------------------------------------------------------------------
# Simulated‑annealing + occasional gradient step for a single run.
# ----------------------------------------------------------------------
def _anneal_run(
    a0: np.ndarray,
    run_deadline: float,
    *,
    init_temp_factor: float = 0.10,
    grad_interval: int = 50,
    grad_step: float = 0.30,
) -> tuple[np.ndarray, float]:
    """
    Perform a stochastic descent on a single problem size.

    Parameters
    ----------
    a0 : np.ndarray
        Starting point (already normalised to sum = 1).
    run_deadline : float
        Absolute time (seconds since epoch) at which to stop.
    init_temp_factor : float
        Initial temperature as a fraction of the current score.
    grad_interval : int
        How many annealing steps between two projected‑gradient updates.
    grad_step : float
        Step size multiplier for the gradient move.

    Returns
    -------
    best_seq, best_score
        Best sequence found during this run and its evaluation score.
    """
    a = a0.copy()
    n = a.shape[0]

    cur_score = _score_of_sequence(a)
    best_seq = a.copy()
    best_score = cur_score

    # initial temperature – a small fraction of the current objective
    T0 = max(cur_score * init_temp_factor, 1e-6)

    start = time.time()
    it = 0
    while time.time() < run_deadline:
        it += 1
        # ---------- temperature decay (linear) ----------
        elapsed = time.time() - start
        T = max(T0 * (1.0 - elapsed / (run_deadline - start)), 1e-9)

        # ---------- propose a random transfer ----------
        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        # move at most 30 % of the source mass (tunable)
        delta = np.random.rand() * 0.30 * a[i]
        a_prop = a.copy()
        a_prop[i] -= delta
        a_prop[j] += delta
        # (sum stays constant, non‑negativity guaranteed by construction)

        prop_score = _score_of_sequence(a_prop)

        # ---------- Metropolis acceptance ----------
        if prop_score < cur_score or np.random.rand() < np.exp(-(prop_score - cur_score) / T):
            a = a_prop
            cur_score = prop_score
            if prop_score < best_score:
                best_score = prop_score
                best_seq = a_prop.copy()
                print(f"[{time.time():.2f}] anneal new best = {best_score:.6f} (n={n})")

        # ---------- occasional projected gradient step ----------
        if it % grad_interval == 0:
            # gradient of the active max‑convolution entries
            g = _subgradient_max_conv(a)
            # move opposite to the gradient (descent)
            a_tmp = a - grad_step * g
            # re‑project onto the simplex (sum = 1, non‑negative)
            a_tmp = _project_onto_simplex(a_tmp, 1.0)
            g_score = _score_of_sequence(a_tmp)
            if g_score < cur_score:
                a = a_tmp
                cur_score = g_score
                if g_score < best_score:
                    best_score = g_score
                    best_seq = a_tmp.copy()
                    print(f"[{time.time():.2f}] grad  new best = {best_score:.6f} (n={n})")
            else:
                # occasional uphill acceptance (tiny probability) to escape plateaux
                if np.random.rand() < 0.001:
                    a = a_tmp
                    cur_score = g_score

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative height sequence (step function) that
    minimises the evaluation function.  The routine respects the
    ``budget_s`` time limit, prints progress information and finally
    returns the best sequence as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2  # tiny safety margin

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # Warm‑start from a previously known construction, if present.
    # --------------------------------------------------------------
    try:
        hs = globals()["height_sequence_1"]
        hs_arr = np.asarray(hs, dtype=float)
        hs_arr = np.clip(hs_arr, 0.0, 1000.0)
        sc = _score_of_sequence(hs_arr)
        if sc < best_score:
            best_score = sc
            best_seq = hs_arr.copy()
            print(f"[{time.time():.2f}] warm‑start score = {best_score:.6f}")
    except Exception:
        pass  # no warm‑start available

    # --------------------------------------------------------------
    # Main restart loop – each restart receives a time slice.
    # --------------------------------------------------------------
    while time.time() < deadline:
        remaining = deadline - time.time()

        # Choose a problem size. Early on we explore larger n,
        # later we focus on cheaper sizes.
        if remaining > 300:
            n = np.random.randint(2500, 3800 + 1)
        elif remaining > 120:
            n = np.random.randint(1500, 2600 + 1)
        else:
            n = np.random.randint(800, 1800 + 1)

        # -------------------- initialise a shape --------------------
        shape_kind = np.random.choice(["power", "U", "flat", "rand"])
        idx = np.arange(1, n + 1, dtype=float)

        if shape_kind == "power":
            p = np.random.uniform(0.2, 1.0)
            init = idx ** (-p)
        elif shape_kind == "U":
            p = np.random.uniform(0.2, 1.0)
            left = idx ** (-p)
            right = (n - idx + 1) ** (-p)
            init = left + right
        elif shape_kind == "flat":
            init = np.ones(n, dtype=float)
        else:  # "rand"
            init = np.random.rand(n)

        # small jitter to break exact symmetries
        init += np.random.rand(n) * 1e-5
        init = np.maximum(init, 1e-12)

        # optionally blend in the current global best (helps to explore neighbourhood)
        if best_seq is not None and np.random.rand() < 0.40:
            bs = best_seq
            if bs.size >= n:
                init = bs[:n].copy()
            else:
                pad = np.random.rand(n - bs.size) * 1e-5
                init = np.concatenate([bs, pad])
        # normalise to sum = 1 (scale‑invariant)
        init = init / np.sum(init)

        # ------------------------------------------------------------------
        # Allocate a time slice for this run: enough to perform many SA steps.
        # ------------------------------------------------------------------
        slice_time = min(30.0, remaining * 0.5)   # never exceed half the remaining budget
        run_deadline = time.time() + slice_time

        # Run the annealing optimiser for the selected length.
        seq_local, score_local = _anneal_run(init, run_deadline)

        # --------------------------------------------------------------
        # Record global best.
        # --------------------------------------------------------------
        if score_local < best_score:
            best_score = score_local
            best_seq = seq_local.copy()
            print(f"[{time.time():.2f}] GLOBAL new best = {best_score:.6f} (n={n})")
            # early exit if we beat the target
            if best_score < 1.503:
                break

    # ------------------------------------------------------------------
    # Fallback: uniform distribution (guaranteed to be valid)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        best_seq = np.full(n, 1.0 / n, dtype=float)
        best_score = _score_of_sequence(best_seq)

    # Return a plain Python list as required by the evaluation harness.
    return best_seq.tolist()
