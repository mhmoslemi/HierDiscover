def proj_simplex(v):
    # based on algorithm from Duchi et al. (2008)
    # v is numpy array
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u - (cssv - 1) / (np.arange(len(v)) + 1) > 0)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

def propose_candidate(seed=0, budget_s=1000):
    # set seed
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5

    # choose n based on remaining time
    # try largest n that fits 2*size memory. Use 8192 if budget > 300 secs else 4096.

    lengths = [8192, 4096, 2048, 1024]
    best_seq = None
    best_val = np.inf

    for n in lengths:
        if time.time() > deadline:
            break
        print(f'optimising length {n}')
        # precompute initial candidates
        init_cands = []
        if best_seq is not None and len(best_seq) == n:
            init_cands.append(np.array(best_seq, dtype=np.float64))
        if 'height_sequence_1' in globals():
            hs = np.asarray(height_sequence_1, dtype=np.float64)
            if hs.size >= n:
                hs = hs[:n]
            else:
                hs = np.pad(hs, (0, n - hs.size))
            hs = np.maximum(hs, 0)
            s = hs.sum()
            if s > 1e-6:
                hs = hs / s
            else:
                hs = np.random.dirichlet(np.ones(n))
            init_cands.append(hs)
        # linearly decreasing shape (v-shaped? Use decreasing then reflect)
        dec = np.arange(n, 0, -1, dtype=np.float64)
        dec = dec / dec.sum()
        init_cands.append(dec)  # also shape
        # uniform
        init_cands.append(np.full(n, 1.0 / n, dtype=np.float64))

        # Phase 1: Adam (exponentiated gradient)
        restarts = 8
        for r in range(restarts):
            if time.time() > deadline: break
            a0 = init_cands[r] if r < len(init_cands) else np.random.dirichlet(np.ones(n))
            # allocate slice time proportionally to remaining
            slice_time = min((deadline - time.time()) * 0.12, 40.0)  # up to 40 secs per run
            sub_deadline = time.time() + slice_time
            a_opt, val_opt = run_adam_opt(n, a0, sub_deadline, max_iters=300_000)
            if val_opt < best_val:
                best_val = val_opt
                best_seq = a_opt.copy()
                print(f'[Adam restart={r} n={n}] new best {val_opt:.6f}')

        # Phase 2: Differential Evolution, with larger popsize
        if time.time() < deadline:
            popsize = 30
            # limit DE runtime to about 40% remaining
            de_time = min((deadline - time.time()) * 0.4, 250.0)
            sub_deadline = time.time() + de_time
            # initialize pop with random Dirichlet and some candidates
            init_pop = init_cands[:5]

                best_val = val_de
                best_seq = a_de.copy()

            if val_es < best_val:
                best_val = val_es
                best_seq = a_es.copy()
                print(f'[ES n={n}] new best {val_es:.6f}')
        # small local pairwise search
        if time.time() < deadline and best_seq is not None:
            # simple greedy swaps
            a = best_seq.copy()
            val = compute_objective(a)
            while time.time() < deadline:
                i, j = np.random.randint(n, size=2)
                if i==j: continue
                # try moving epsilon proportion from i to j
                eps = 0.01 * a[i]  # 1% of amount at i
                if eps ==0: continue
                a_new = a.copy()
                a_new[i] -= eps

                    val = val_new
                    if val < best_val:

            # end while

    # fallback

        n = 1024
        best_seq = np.full(n, 1.0 / n)
        best_val = compute_objective(best_seq)

    print(f'final best value {best_val:.6f}')
    return best_seq.tolist()

def run_de_opt_custom(n, init_pop, popsize, deadline, max_gens=50000):
    # ensure each candidate normalized.
    pop = []
    for cand in init_pop:
        a = np.asarray(cand, dtype=np.float64)
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 1e-12:
            pop.append(a / s)
        else:
            pop.append(np.full(n, 1/n))

    while len(pop) < popsize:
        pop.append(np.random.dirichlet(np.ones(n)))

    # evaluate
    obj = np.array([compute_objective(p) for p in pop])
    best_idx = np.argmin(obj)
    best_a = pop[best_idx].copy()
    best_val = obj[best_idx]

    F = 0.8
    CR = 0.9

    while time.time() < deadline:

                    best_val = val

                    last_improve = time.time()

        if time.time() - last_improve > 10.0:

        if gen >= max_gens:

    return best_a, best_val

"""
Propose a step‑function (non‑negative vector that sums to 1) that minimizes

        F(a) = 2·n·max(a∗a) / (sum(a)²)

The optimiser works in three stages and respects a global time‑budget
(default 1000 s).  All stages operate on the simplex by
*softmax* (exponentiated‑gradient) or by explicit renormalisation,
therefore the scale of the vector never matters.

1. **Accelerated Adam on the exact sub‑gradient of max(a∗a).**
   Several restarts (8) are performed; each restart gets a short,
   adaptive time slice.  The Adam optimiser works on the log‑odds
   (logits) of the probabilities, which guarantees a≥0 and Σa=1.

2. **Large‑population Differential Evolution (DE).**
   A fairly big population (≈30) explores the simplex globally.
   Both “rand/1” and “best/1” mutation strategies are mixed.
   Every trial is clipped to ≥0 and re‑normalised; the best individual
   found is kept.

3. **(1+1) Evolution Strategy (ES) fine‑tuning.**
   Gaussian perturbations of the current best are accepted when they
   improve the exact objective; a simple step‑size control (1% decay/
   5% increase) is used.

If the global budget runs out before any of the stages finish,
the algorithm falls back to a uniform distribution.  The best
vector discovered (as a plain Python list) is returned – this is the
object that will be handed to ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers (no closures, no lambdas)
# ----------------------------------------------------------------------
def softmax(x: np.ndarray) -> np.ndarray:
    """Stable softmax – maps logits to a probability vector (sum = 1)."""
    mx = np.max(x)
    e = np.exp(x - mx)
    return e / e.sum()


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using an FFT of length power‑of‑two."""
    n = a.shape[0]
    L = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, L)
    conv = np.fft.irfft(fa * fa, L)[: 2 * n - 1]
    return conv.real


def conv_max(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution of a."""
    return full_convolution(a).max()


def compute_objective(a: np.ndarray) -> float:
    """Exact objective used by ``evaluate_sequence`` (sum(a) may differ from 1)."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return 2.0 * a.shape[0] * conv_max(a) / (s * s)


def grad_submax(a: np.ndarray):
    """
    Sub‑gradient of max(a∗a) w.r.t. a.
    Returns (gradient, current max value).
    The gradient follows ∂max/∂a_j = 2·a_{k*‑j} for a convolution index k* attaining the maximum.
    """
    n = a.shape[0]
    conv = full_convolution(a)
    max_c = conv.max()
    # pick a random index among all maxima (break ties randomly)
    idxs = np.flatnonzero(np.isclose(conv, max_c, atol=1e-14))
    k = np.random.choice(idxs)

    g = np.zeros_like(a)
    start = max(0, k - (n - 1))
    end = min(k, n - 1)
    i = np.arange(start, end + 1, dtype=int)
    j = k - i
    g[i] = 2.0 * a[j]
    return g, max_c


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------
def run_adam_opt(n: int,
                 init_a: np.ndarray,
                 deadline: float,
                 max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the sub‑gradient of max(a∗a) while staying on the simplex.
    Returns the best vector (sum = 1) found before *deadline* and its objective value.
    """
    # initialise logits from the probability vector
    x = np.log(init_a + 1e-15)

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.05

    best_val = np.inf
    best_a = None
    t = 0
    last_improve = time.time()

    while time.time() < deadline and t < max_iters:
        a = softmax(x)                     # a ≥0, Σa = 1
        grad_a, max_c = grad_submax(a)     # exact sub‑gradient of max convolution
        val = 2.0 * n * max_c               # exact objective (∑a=1)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[Adam n={n} iter={t:06d}] new best {val:.6f}")

        # back‑prop through the softmax
        avg = np.dot(a, grad_a)
        grad_x = a * (grad_a - avg)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

        # keep logits bounded
        x = np.clip(x, -50.0, 50.0)

        # occasional random “shake” to escape plateaus
        if time.time() - last_improve > 5.0:
            x += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()
            print(f"[Adam n={n} iter={t}] shake (stagnation)")

    if best_a is None:           # fallback: uniform
        best_a = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = compute_objective(best_a)
    return best_a, best_val


def run_de_opt_custom(n: int,
                      init_pop: list,
                      popsize: int,
                      deadline: float,
                      max_gens: int = 30_000):
    """
    Differential Evolution on the simplex.
    Uses a mixed mutation strategy (rand/1 and best/1) and keeps the best individual.
    """
    # ------------------------------------------------------------------
    # initialise population (each individual is already normalised)
    # ------------------------------------------------------------------
    pop = []
    for cand in init_pop:
        a = np.asarray(cand, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0:
            pop.append(a / s)
        else:
            pop.append(np.full(n, 1.0 / n))
    while len(pop) < popsize:
        pop.append(np.random.dirichlet(np.ones(n)))

    # evaluate
    obj_vals = np.array([compute_objective(p) for p in pop])
    best_idx = int(np.argmin(obj_vals))
    best_a = pop[best_idx].copy()
    best_val = obj_vals[best_idx]

    F = 0.8   # differential weight
    CR = 0.9  # crossover probability
    last_improve = time.time()
    gen = 0

    while time.time() < deadline and gen < max_gens:
        gen += 1
        for i in range(popsize):
            # three distinct donors
            idxs = list(range(popsize))
            idxs.remove(i)
            r1, r2, r3 = np.random.choice(idxs, 3, replace=False)

            # randomly pick mutation strategy
            if np.random.rand() < 0.5:
                # best/1
                mutant = best_a + F * (pop[r1] - pop[r2])
            else:
                # rand/1
                mutant = pop[r1] + F * (pop[r2] - pop[r3])

            mutant = np.maximum(mutant, 0.0)               # enforce non‑negativity

            # binomial crossover (ensure at least one component from mutant)
            cross = np.random.rand(n) < CR
            cross[np.random.randint(n)] = True
            trial = np.where(cross, mutant, pop[i])

            s = trial.sum()
            if s == 0.0:
                trial = np.full(n, 1.0 / n)
            else:
                trial = trial / s

            val = compute_objective(trial)
            if val < obj_vals[i]:
                pop[i] = trial
                obj_vals[i] = val
                if val < best_val:
                    best_val = val
                    best_a = trial.copy()
                    last_improve = time.time()
                    print(f"[DE n={n} gen={gen:04d} i={i}] new best {val:.6f}")

        # early stop if no improvement for a while
        if time.time() - last_improve > 12.0:
            break

    return best_a, best_val


def run_es_opt(n: int,
               init_a: np.ndarray,
               deadline: float,
               max_iter: int = 200_000):
    """
    (1+1) Evolution Strategy – Gaussian perturbations of the current best.
    Returns the best vector discovered before *deadline*.
    """
    a = init_a.copy()
    val = compute_objective(a)
    best_a, best_val = a.copy(), val
    sigma = 0.02
    last_improve = time.time()
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        trial = a + sigma * np.random.randn(n)
        trial = np.maximum(trial, 0.0)
        s = trial.sum()
        if s == 0.0:
            trial = np.full(n, 1.0 / n)
        else:
            trial = trial / s

        val_trial = compute_objective(trial)
        if val_trial < best_val:
            best_val = val_trial
            best_a = trial.copy()
            a = trial                               # success → keep
            sigma = min(sigma * 1.05, 0.2)           # increase step size a bit
            last_improve = time.time()
            print(f"[ES n={n} it={it}] new best {best_val:.6f}")
        else:
            sigma *= 0.99                           # decay on failure

        # occasional reset if stuck
        if time.time() - last_improve > 5.0:
            a = best_a.copy()
            sigma = 0.02
            last_improve = time.time()

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the provided evaluation
    function.  The routine respects the global time budget and returns a plain
    Python list (the “height sequence”).  It combines Adam, Differential Evolution
    and a (1+1) Evolution Strategy, all working on the probability simplex.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin for the host

    best_seq = None
    best_val = np.inf

    # Try first the length that historically gives the best trade‑off.
    # If time permits we also try a larger length.
    lengths = [4096, 8192, 2048, 1024]

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")

        # --------------------------------------------------------------
        # Build a small set of sensible initial candidates
        # --------------------------------------------------------------
        init_candidates = []

        # warm‑start from the current global best (if shape matches)
        if best_seq is not None and len(best_seq) == n:
            init_candidates.append(np.array(best_seq, dtype=np.float64))

        # historic construction, if available
        if "height_sequence_1" in globals():
            hs = np.asarray(height_sequence_1, dtype=np.float64)
            if hs.size >= n:
                hs = hs[:n]
            else:
                hs = np.pad(hs, (0, n - hs.size))
            hs = np.maximum(hs, 0.0)
            s = hs.sum()
            if s > 1e-6:
                hs = hs / s
            else:
                hs = np.random.dirichlet(np.ones(n))
            init_candidates.append(hs)

        # linearly decreasing shape (simple decreasing ramp)
        dec = np.arange(n, 0, -1, dtype=np.float64)
        dec = dec / dec.sum()
        init_candidates.append(dec)

        # uniform distribution
        init_candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

        # --------------------------------------------------------------
        # Phase 1 – Adam (multiple restarts)
        # --------------------------------------------------------------
        restarts = 8
        for r in range(restarts):
            if time.time() > deadline:
                break
            if r < len(init_candidates):
                a0 = init_candidates[r]
            else:
                a0 = np.random.dirichlet(np.ones(n))

            # allocate a modest slice of the remaining budget for this restart
            slice_time = min((deadline - time.time()) * 0.12, 35.0)
            sub_deadline = time.time() + slice_time
            a_opt, val_opt = run_adam_opt(n, a0, sub_deadline)

            if val_opt < best_val:
                best_val = val_opt
                best_seq = a_opt.copy()
                print(f"[n={n} Adam restart={r}] GLOBAL best improved to {best_val:.6f}")

        # --------------------------------------------------------------
        # Phase 2 – Differential Evolution (global search)
        # --------------------------------------------------------------
        if time.time() < deadline:
            # give DE roughly 35 % of the remaining time
            de_time = min((deadline - time.time()) * 0.35, 250.0)
            sub_deadline = time.time() + de_time
            popsize = 30
            de_pop = init_candidates[:5]   # a small warm‑start collection
            a_de, val_de = run_de_opt_custom(n, de_pop, popsize, sub_deadline)
            if val_de < best_val:
                best_val = val_de
                best_seq = a_de.copy()
                print(f"[n={n} DE] GLOBAL best now {best_val:.6f}")

        # --------------------------------------------------------------
        # Phase 3 – (1+1) ES fine‑tuning
        # --------------------------------------------------------------
        if time.time() < deadline and best_seq is not None and len(best_seq) == n:
            es_time = min((deadline - time.time()) * 0.30, 120.0)
            sub_deadline = time.time() + es_time
            a_es, val_es = run_es_opt(n, best_seq, sub_deadline)
            if val_es < best_val:
                best_val = val_es
                best_seq = a_es.copy()
                print(f"[n={n} ES] GLOBAL best now {best_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – if nothing was found (should never happen) return uniform.
    # ------------------------------------------------------------------
    if best_seq is None:
        fallback_n = 1024
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)
        best_val = compute_objective(best_seq)

    print(f"\nBest value discovered: {best_val:.6f} (length {len(best_seq)})")
    # return a plain Python list as required by the external evaluator
    return best_seq.tolist()
