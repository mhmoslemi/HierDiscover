def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5
    best_seq = None
    best_val = float('inf')
    lengths = [1024, 2048, 4096, 8192, 16384]

    # Precompute FFT lengths
    fft_cache = {}
    for n in lengths:
        conv_len = 2*n-1
        fft_cache[n] = 1 << int(np.ceil(np.log2(conv_len)))

    for n in lengths:
        if time.time() > deadline: break
        print(f"=== Optimising length {n} ===")
        fft_len = fft_cache[n]

        # Candidate generation
        candidates = []
        # warm start if best_seq matches length
        if best_seq is not None:
            if len(best_seq) == n:
                candidates.append(np.array(best_seq, dtype=float))
            else:
                # upsample or downsample
                best_arr = np.array(best_seq, dtype=float)
                # linear interpolation to new length
                x_old = np.linspace(0, 1, len(best_arr))
                x_new = np.linspace(0, 1, n)
                cand = np.interp(x_new, x_old, best_arr)
                candidates.append(cand)

        # historic height_sequence_1
        hseq = globals().get('height_sequence_1')
        if hseq is not None:
            arr = np.asarray(hseq, dtype=float)
            if arr.shape[0] != n:
                x_old = np.linspace(0, 1, arr.shape[0])
                x_new = np.linspace(0, 1, n)
                arr = np.interp(x_new, x_old, arr)
            candidates.append(arr)

        # random Dirichlet
        candidates.append(np.random.dirichlet(np.ones(n)))

        # power-law edge shapes (single exponent)
        idx = np.arange(n) + 1.0
        prod = idx * (n - idx + 1.0)
        for alpha in np.linspace(-2.0, -0.1, 7):
            shape = np.power(prod, alpha, where=prod>0, out=np.zeros_like(prod))
            if shape.sum()>0:
                shape /= shape.sum()
                candidates.append(shape)

        # beta-shaped (i+1)^alpha * (n-i)^beta
        for alpha in [-1.5,-0.5,0.5,1.5]:
            for beta in [-1.5,-0.5,0.5,1.5]:
                shape = np.power(idx, alpha, where=idx>0, out=np.zeros_like(idx)) * \
                        np.power(n-idx+1, beta, where=(n-idx+1)>0, out=np.zeros_like(idx))
                if shape.sum()>0:
                    shape /= shape.sum()
                    candidates.append(shape)

        # double Gaussian
        iarr = np.arange(n, dtype=float)
        for gamma in [0.2,0.5,1.0,2.0,5.0]:
            shape = np.exp(-gamma * (iarr**2)) + np.exp(-gamma * ((n-1-iarr)**2))
            shape /= shape.sum()
            candidates.append(shape)

        # For each candidate run two-phase optimization
        for cand_id, cand0 in enumerate(candidates):
            if time.time() > deadline:
                break
            a = np.maximum(cand0, 0.0)
            if a.sum() < 1e-12:
                a = np.random.dirichlet(np.ones(n))
            a = a / a.sum()
            # optimize via Adam+smoothmax, then subgradient
            # set hyperparameters based on n
            # Use more iterations for larger n
            max_iter = int(20000 * 1024 / n) + 5000  # scaling
            # Guarantee at least certain number
            max_iter = max(2000, max_iter)
            # Adam init
            logits = np.log(a + 1e-15)
            m = np.zeros_like(logits)
            v = np.zeros_like(logits)
            beta1 = 0.9
            beta2 = 0.999
            eps = 1e-8
            lr = 0.1  # maybe larger step
            lam = 0.5
            lam_factor = 1.05
            last_improve = time.time()
            for it in range(1, max_iter+1):
                # compute a from logits
                mx = logits.max()
                exp_x = np.exp(logits - mx)
                a = exp_x / exp_x.sum()
                # convolution
                conv = _full_convolution(a, fft_len)
                max_c = conv.max()
                # compute softmax weights
                w = np.exp(lam * (conv - max_c))
                w_sum = w.sum()
                if w_sum == 0:
                    w = np.full_like(w, 1.0 / w.size)
                else:
                    w /= w_sum
                # gradient of smoothmax
                grad_a = 2.0 * _cross_correlation(w, a, fft_len)
                # backprop through softmax: gradient wrt logits
                avg = np.dot(a, grad_a)
                grad_log = a * (grad_a - avg)

                # Adam update
                m = beta1 * m + (1 - beta1) * grad_log
                v = beta2 * v + (1 - beta2) * (grad_log ** 2)
                m_hat = m / (1 - beta1 ** it)
                v_hat = v / (1 - beta2 ** it)
                logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

                # increase temperature
                lam = min(lam * lam_factor, 1e6)

                # compute exact objective (fast version)
                val = 2.0 * n * max_c / (a.sum() ** 2)

                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} cand={cand_id} Adam iter={it}] new best {val:.6f}")

                # early exit if no improvement for 3 seconds
                if time.time() - last_improve > 3.0:
                    # apply small random perturbation to logits to escape
                    logits += (np.random.rand(n) - 0.5) * 1e-2
                    last_improve = time.time()
                if time.time() > deadline:
                    break

            # Subgradient phase
            # Ensure a on simplex
            a = a / a.sum()
            eta = 0.8
            sub_last_improve = time.time()
            for sub_it in range(3000):
                conv = _full_convolution(a, fft_len)
                max_c = conv.max()
                active = np.isclose(conv, max_c, atol=1e-12)
                if active.any():
                    w = np.zeros_like(conv)
                    w[active] = 1.0 / active.sum()
                else:
                    w = np.zeros_like(conv)
                grad_a = 2.0 * _cross_correlation(w, a, fft_len)

                # exponentiated gradient step
                a = a * np.exp(-eta * grad_a)
                s = a.sum()
                if s < 1e-12:
                    a = np.random.dirichlet(np.ones(n))
                    s = a.sum()
                a = a / s

                val = 2.0 * n * max_c / (a.sum() ** 2)
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    sub_last_improve = time.time()
                    print(f"[n={n} cand={cand_id} Subgrad iter={sub_it}] new best {val:.6f}")

                # step size decay
                eta = max(0.01, eta * 0.9995)

                if time.time() - sub_last_improve > 3.0:
                    break
                if time.time() > deadline:
                    break

    if best_seq is None:
        # fallback uniform of length 1024
        n0 = 1024
        best_seq = np.full(n0, 1.0 / n0)
        best_val = evaluate_sequence(best_seq.tolist())
        print("fallback uniform used, val:", best_val)

    print("Final best value:", best_val, "length:", len(best_seq))
    return best_seq.tolist()

"""
Propose a step‑function sequence that (approximately) minimises

        F(a) = 2·n·max(a*a) / (sum(a)²)

under a_i ≥ 0.

Algorithm
---------
* Work on several lengths (1024, 2048, 4096, 8192, 16384).
* For each length build a diverse set of start‑points:
    – warm‑start from the current global best (interpolated when necessary);
    – historic solution (height_sequence_1);
    – random Dirichlet draws;
    – power‑law edge shapes;
    – beta‑type shapes   a_i ∝ (i+1)^α (n‑i)^β;
    – two‑Gaussian edge shapes.
* Every start is refined by two phases
    1. Adam optimisation on a *smooth‑max* surrogate of the max‑convolution.
       The surrogate uses a softmax with a temperature λ that is
       increased geometrically.  Variables live in the probability simplex
       via the log‑softmax (logits) parametrisation.
    2. Exponentiated‑gradient (mirror‑descent) on the *exact* max‑convolution.
       The sub‑gradient is obtained from the active set of maximal
       convolution entries.
* The best sequence seen across all lengths is kept.
* If time permits, a short “shake‑and‑refine” loop is run on the
  current best (small Gaussian perturbations + a few Adam steps) to
  escape possible local minima.
* The routine respects a global 1000 s budget and returns a plain
  Python list of non‑negative floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.shape[0] - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) equal to
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]  # length n


def _objective_from_conv(conv: np.ndarray, n: int) -> float:
    """Exact objective 2·n·max(conv) / (sum(a)²)  (sum(a)≈1)."""
    return float(2.0 * n * conv.max())  # a is kept normalised to sum=1


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative sequence with a low evaluation score.
    Returns the best sequence found (list of floats).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None          # best sequence (numpy array, sum≈1)
    best_val = float("inf")  # best objective value seen so far

    # lengths to explore – larger sizes give more freedom but are slower
    lengths = [1024, 2048, 4096, 8192, 16384]

    # pre‑compute FFT lengths for each n (next power of two)
    fft_len_cache = {}
    for n in lengths:
        conv_len = 2 * n - 1
        fft_len_cache[n] = 1 << int(np.ceil(np.log2(conv_len)))

    # ------------------------------------------------------------------
    # Helper to evaluate a candidate (exact objective, assumes sum≈1)
    def _eval_exact(a: np.ndarray, n: int) -> float:
        conv = _full_convolution(a, fft_len_cache[n])
        return _objective_from_conv(conv, n)

    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length n={n} ===")
        fft_len = fft_len_cache[n]

        # --------------------- generate start points --------------------
        candidates = []

        # warm start from the globally best (interpolated if lengths differ)
        if best_seq is not None:
            if len(best_seq) == n:
                candidates.append(np.array(best_seq, copy=True))
            else:
                # linear interpolation to new grid
                old = np.array(best_seq, copy=True)
                x_old = np.linspace(0.0, 1.0, old.shape[0])
                x_new = np.linspace(0.0, 1.0, n)
                cand = np.interp(x_new, x_old, old)
                candidates.append(cand)

        # historic solution if supplied
        hseq = globals().get("height_sequence_1")
        if hseq is not None:
            arr = np.asarray(hseq, dtype=np.float64)
            if arr.shape[0] != n:
                x_old = np.linspace(0.0, 1.0, arr.shape[0])
                x_new = np.linspace(0.0, 1.0, n)
                arr = np.interp(x_new, x_old, arr)
            candidates.append(arr)

        # random Dirichlet start
        candidates.append(np.random.dirichlet(np.ones(n)))

        # edge‑concentrated power‑law shapes  ((i+1)*(n-i))**α , α<0
        idx = np.arange(n, dtype=np.float64) + 1.0
        prod = idx * (n - idx + 1.0)
        for alpha in np.linspace(-2.0, -0.1, 7):
            shape = np.power(prod, alpha, where=prod > 0, out=np.zeros_like(prod))
            if shape.sum() > 0:
                shape /= shape.sum()
                candidates.append(shape)

        # beta‑type shapes  (i+1)^α (n-i)^β
        for alpha in [-1.5, -0.5, 0.5, 1.5]:
            for beta in [-1.5, -0.5, 0.5, 1.5]:
                shape = np.power(idx, alpha, where=idx > 0, out=np.zeros_like(idx)) \
                        * np.power(n - idx + 1, beta, where=(n - idx + 1) > 0, out=np.zeros_like(idx))
                if shape.sum() > 0:
                    shape /= shape.sum()
                    candidates.append(shape)

        # two‑Gaussian edge shapes
        i_arr = np.arange(n, dtype=np.float64)
        for gamma in [0.2, 0.5, 1.0, 2.0, 5.0]:
            shape = np.exp(-gamma * (i_arr ** 2)) + np.exp(-gamma * ((n - 1 - i_arr) ** 2))
            shape /= shape.sum()
            candidates.append(shape)

        # ------------------------------------------------------------------
        # Optimise each candidate
        for cand_id, cand0 in enumerate(candidates):
            if time.time() > deadline:
                break

            a = np.maximum(cand0, 0.0)
            if a.sum() < 1e-12:
                a = np.random.dirichlet(np.ones(n))
            a = a / a.sum()

            # ------------------- Adam on smooth‑max --------------------
            # logits parameterisation (soft‑max → simplex)
            logits = np.log(a + 1e-15)
            m = np.zeros_like(logits)
            v = np.zeros_like(logits)
            beta1, beta2, eps = 0.9, 0.999, 1e-8
            lr = 0.1
            lam = 0.5               # temperature for smooth‑max
            lam_factor = 1.05

            # iteration budget scales inversely with n (more iterations for small n)
            max_iter = int(20000 * 1024 / n) + 5000
            max_iter = max(2000, max_iter)

            last_improve = time.time()
            for it in range(1, max_iter + 1):
                # a = softmax(logits)
                mx = logits.max()
                exp_x = np.exp(logits - mx)
                a = exp_x / exp_x.sum()

                # convolution and smooth‑max weights
                conv = _full_convolution(a, fft_len)
                max_c = conv.max()
                w = np.exp(lam * (conv - max_c))
                w_sum = w.sum()
                if w_sum == 0.0:
                    w = np.full_like(w, 1.0 / w.size)
                else:
                    w /= w_sum

                # gradient of smooth‑max w.r.t a
                grad_a = 2.0 * _cross_correlation(w, a, fft_len)

                # back‑prop through soft‑max
                avg = np.dot(a, grad_a)
                grad_log = a * (grad_a - avg)

                # Adam update on logits
                m = beta1 * m + (1.0 - beta1) * grad_log
                v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
                m_hat = m / (1.0 - beta1 ** it)
                v_hat = v / (1.0 - beta2 ** it)
                logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

                # temperature schedule
                lam = min(lam * lam_factor, 1e6)

                # exact objective (a is already normalised)
                val = 2.0 * n * max_c  # denominator = 1

                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} cand={cand_id} Adam it={it:05d}] new best {val:.6f}")

                # stagnation handling
                if time.time() - last_improve > 3.0:
                    logits += (np.random.rand(n) - 0.5) * 1e-2
                    last_improve = time.time()

                if time.time() > deadline:
                    break

            # ------------------- Sub‑gradient on exact max -------------------
            a = a / a.sum()
            eta = 0.8  # initial step size for exponentiated‑gradient
            sub_last_improve = time.time()
            for sub_it in range(3000):
                conv = _full_convolution(a, fft_len)
                max_c = conv.max()
                active = np.isclose(conv, max_c, atol=1e-12)
                if active.any():
                    w = np.zeros_like(conv)
                    w[active] = 1.0 / active.sum()
                else:
                    w = np.zeros_like(conv)

                grad_a = 2.0 * _cross_correlation(w, a, fft_len)

                s = a.sum()

                    s = a.sum()
                a = a / s

                conv_new = _full_convolution(a, fft_len)
                val = 2.0 * n * conv_new.max()
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    sub_last_improve = time.time()
                    print(f"[n={n} cand={cand_id} Subgrad it={sub_it}] new best {val:.6f}")

                # step‑size decay
                eta = max(0.01, eta * 0.9995)

                if time.time() - sub_last_improve > 3.0:
                    break
                if time.time() > deadline:
                    break

        # ------------------------------------------------------------------
        # Optional “shake‑and‑refine” on the current global best of this length
        if best_seq is not None and len(best_seq) == n and time.time() < deadline:
            a = best_seq.copy()
            for shake in range(3):
                if time.time() > deadline:
                    break
                # small Gaussian perturbation while staying non‑negative
                perturb = np.random.randn(n) * 0.02
                a = np.maximum(a + perturb, 0.0)
                a = a / a.sum()
                # a few Adam steps with a high temperature (tight surrogate)
                logits = np.log(a + 1e-15)
                m = np.zeros_like(logits)
                v = np.zeros_like(logits)
                beta1, beta2, eps = 0.9, 0.999, 1e-8
                lr = 0.05
                lam = 1.0
                lam_factor = 1.2
                for it in range(500):
                    mx = logits.max()
                    exp_x = np.exp(logits - mx)
                    a = exp_x / exp_x.sum()
                    conv = _full_convolution(a, fft_len)
                    max_c = conv.max()
                    w = np.exp(lam * (conv - max_c))
                    w_sum = w.sum()
                    if w_sum == 0.0:
                        w = np.full_like(w, 1.0 / w.size)
                    else:
                        w /= w_sum
                    grad_a = 2.0 * _cross_correlation(w, a, fft_len)
                    avg = np.dot(a, grad_a)
                    grad_log = a * (grad_a - avg)

                    m = beta1 * m + (1.0 - beta1) * grad_log
                    v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
                    m_hat = m / (1.0 - beta1 ** (it + 1))
                    v_hat = v / (1.0 - beta2 ** (it + 1))
                    logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

                    lam = min(lam * lam_factor, 1e6)

                    if time.time() > deadline:

                    print(f"[n={n} shake={shake}] new best after shake {val:.6f}")

    # Fallback – uniform vector if nothing was found (unlikely)
    if best_seq is None:

        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)

        print("Fallback uniform used, val =", best_val)

    print("\nFinal best value:", best_val, " (length =", len(best_seq), ")")
    return best_seq.tolist()
