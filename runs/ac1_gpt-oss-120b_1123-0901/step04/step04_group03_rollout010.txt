def project_simplex(v):
    """Project onto simplex {x>=0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    # find rho = max {j: u_j - cssv_j / j > 0}
    cond = u - cssv / ind > 0
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def subgradient_descent(a0, deadline, max_iters=200000):
    a = a0.copy()
    best_a = a.copy()
    best_val = compute_objective(a)
    n = a.shape[0]
    t = 0
    last_improve = time.time()
    while time.time() < deadline and t < max_iters:
        t += 1
        conv = full_convolution(a)
        max_c = conv.max()
        # choose random arg if multiple maxima
        idxs = np.flatnonzero(conv == max_c)
        k = np.random.choice(idxs)
        # gradient direction (subgradient) for max conv
        g = np.zeros_like(a)
        start = max(0, k - (n - 1))
        end = min(k, n - 1)
        i = np.arange(start, end + 1)
        j = k - i
        g[i] = 2.0 * a[j]
        # step size: decline
        step = 0.8 / np.sqrt(t)
        a_new = a - step * g
        a_new = project_simplex(a_new)
        val = compute_objective(a_new)
        if val < best_val:
            best_val = val
            best_a = a_new.copy()
            last_improve = time.time()
            print(f"[SubGrad n={n} iter={t}] new best {val:.6f}")

        a = a_new

        if time.time() - last_improve > 10.0:
            # random perturbation % of best
            a = best_a + 0.02 * np.random.rand(n)
            a = project_simplex(a)
            last_improve = time.time()
            print(f"[SubGrad n={n} iter={t}] shake")
    return best_a, best_val

def simulated_annealing(a0, deadline):
    a = a0.copy()
    n = a.shape[0]
    val = compute_objective(a)
    best_a = a.copy()
    best_val = val
    T0 = 0.05
    T = T0
    it = 0
    last_improve = time.time()
    while time.time() < deadline:
        it += 1
        # propose move by transferring mass
        i, j = np.random.choice(n, 2, replace=False)
        if a[i] <= 0:
            continue
        # delta fraction of a[i] up to maybe 10% of a[i] or 0.01
        max_delta = min(a[i], 0.05)
        delta = np.random.rand() * max_delta
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # keep sum unchanged, nonneg holds
        val_new = compute_objective(a_new)
        # accept if better or with probability
        if val_new < val or np.random.rand() < np.exp(-(val_new - val) / max(T, 1e-12)):
            a = a_new
            val = val_new
        if val_new < best_val:
            best_val = val_new
            best_a = a_new.copy()
            last_improve = time.time()
            print(f"[SA n={n} it={it}] new best {best_val:.6f}")
        # cooling schedule: reduce T slowly every 1000 iter
        if it % 1000 == 0:
            T *= 0.99
            if T < 1e-6:
                T = 1e-6
        # occasional reheat if stagnation
        if time.time() - last_improve > 8.0:
            a = best_a + 0.02 * np.random.rand(n)
            a = project_simplex(a)
            val = compute_objective(a)
            last_improve = time.time()
            print(f"[SA n={n} it={it}] reheat")
    return best_a, best_val

def propose_candidate(seed=0, budget_s=1000, **kwargs):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5
    best_seq = None
    best_val = np.inf
    lengths = [8192, 4096, 2048, 1024]  # possibly also 16384 if time
    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length {n} ===")
        # create initial candidates
        init_candidates = []
        # warm start
        if best_seq is not None and len(best_seq) == n:
            init_candidates.append(np.asarray(best_seq, dtype=np.float64))
        if "height_sequence_1" in globals():
            hs = np.asarray(height_sequence_1, dtype=np.float64)
            if hs.size >= n:
                hs = hs[:n]
            else:
                hs = np.pad(hs, (0, n - hs.size))
            hs = np.maximum(hs, 0.0)
            total = hs.sum()
            if total > 1e-6:
                hs = hs / total
            else:
                hs = np.random.dirichlet(np.ones(n))
            init_candidates.append(hs)
        # monotone decreasing
        dec = np.arange(n, 0, -1, dtype=np.float64)
        dec = dec / dec.sum()
        init_candidates.append(dec)
        # uniform

        # random
        init_candidates.append(np.random.dirichlet(np.ones(n)))

        # --- Subgradient descent phase ---
        # allocate about 30% of remaining time to subgradient per length
        if time.time() < deadline:
            sg_time = min((deadline - time.time()) * 0.3, 120.0)
            sg_deadline = time.time() + sg_time
            # try a few restarts
            for attempt, cand in enumerate(init_candidates[:3]):  # up to 3 attempts
                a0 = cand.copy()

                if val_opt < best_val:
                    best_val = val_opt
                    best_seq = a_opt.copy()
                    print(f"[Length {n} SubGrad attempt {attempt}] best improved to {best_val:.6f}")
                if time.time() >= sg_deadline:
                    break

        # --- Simulated Annealing phase ---
        if time.time() < deadline and best_seq is not None and len(best_seq) == n:
            sa_time = min((deadline - time.time()) * 0.3, 120.0)
            sa_deadline = time.time() + sa_time
            a_sa, val_sa = simulated_annealing(best_seq, sa_deadline)
            if val_sa < best_val:
                best_val = val_sa
                best_seq = a_sa.copy()

        # --- Evolution Strategy fine‑tuning ---
        if time.time() < deadline and best_seq is not None and len(best_seq) == n:
            es_time = min((deadline - time.time()) * 0.4, 180.0)
            es_deadline = time.time() + es_time

"""
Hybrid optimiser for the non‑negative step‑function problem.

The algorithm works on any length *n* that fits into the time budget.
It combines three complementary phases:

1. **Projected sub‑gradient descent** on the convex objective
   `F(a)=2·n·max(a∗a)/(sum(a)²)`.  The simplex constraint
   `a_i≥0, Σa_i=1` is handled by an explicit Euclidean projection.

2. **Simulated annealing** that moves a small amount of mass from a
   “high‑contributing’’ index to a “low‑contributing’’ index, with a
   temperature schedule to escape local minima.

3. **(1+1) Evolution Strategy** (Gaussian mutation, self‑adaptive σ)
   for fine‑tuning the best solution found so far.

All phases respect a global deadline (default 1000 s).  The routine
returns the best sequence found as a plain Python list, ready for
`evaluate_sequence`.  The code deliberately differs from the previous
baseline (which used Adam + DE) in order to explore new search dynamics.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    # find the largest rho satisfying the condition
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero – return uniform
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a * a using an FFT of the next power‑of‑two length.
    Returned array has length 2·len(a)‑1.
    """
    n = a.shape[0]
    L = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, L)
    conv = np.fft.irfft(fa * fa, L)[: 2 * n - 1]
    return conv.real


def compute_objective(a: np.ndarray) -> float:
    """
    Exact objective used by the external evaluator.
    The vector is assumed to be non‑negative; no additional scaling
    is required because the expression is homogeneous.
    """
    s = a.sum()
    if s < 1e-12:
        return np.inf
    max_c = full_convolution(a).max()
    return float(2.0 * a.shape[0] * max_c / (s * s))


# ----------------------------------------------------------------------
# 1. Projected sub‑gradient descent
# ----------------------------------------------------------------------


def subgradient_descent(
    a0: np.ndarray, deadline: float, max_iters: int = 200_000
) -> Tuple[np.ndarray, float]:
    """
    Simple sub‑gradient descent on the simplex.
    The step size follows η_t = η₀ / sqrt(t).
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = compute_objective(a)
    n = a.shape[0]
    t = 0
    last_improve = time.time()
    eta0 = 0.8

    while time.time() < deadline and t < max_iters:
        t += 1
        conv = full_convolution(a)
        max_c = conv.max()
        # pick a random maximal index to avoid bias
        idxs = np.flatnonzero(conv == max_c)
        k = np.random.choice(idxs)

        # sub‑gradient of the max‑convolution entry
        g = np.zeros_like(a)
        start = max(0, k - (n - 1))
        end = min(k, n - 1)
        i = np.arange(start, end + 1, dtype=int)
        j = k - i
        g[i] = 2.0 * a[j]

        # diminishing step size
        step = eta0 / np.sqrt(t)
        a_new = a - step * g
        a_new = project_simplex(a_new)

        val = compute_objective(a_new)
        if val < best_val:
            best_val = val
            best_a = a_new.copy()
            last_improve = time.time()
            print(f"[SGD n={n} iter={t:06d}] new best {val:.6f}")

        a = a_new

        # occasional random shake when stagnating
        if time.time() - last_improve > 10.0:
            a = best_a + 0.02 * np.random.rand(n)
            a = project_simplex(a)
            last_improve = time.time()
            print(f"[SGD n={n} iter={t}] shake (stagnation)")

    return best_a, best_val


# ----------------------------------------------------------------------
# 2. Simulated annealing on the simplex
# ----------------------------------------------------------------------


def simulated_annealing(
    a0: np.ndarray, deadline: float, init_T: float = 0.05, decay: float = 0.992
) -> Tuple[np.ndarray, float]:
    """
    Simple SA that repeatedly transfers a small mass δ from one index to another.
    Acceptance follows the Metropolis rule with a slowly cooling temperature.
    """
    a = a0.copy()
    n = a.shape[0]
    val = compute_objective(a)
    best_a = a.copy()
    best_val = val
    T = init_T
    it = 0
    last_improve = time.time()

    while time.time() < deadline:
        it += 1
        # pick a donor and a receiver
        i, j = np.random.choice(n, 2, replace=False)
        if a[i] <= 0.0:
            continue
        # choose a small transferable chunk
        max_delta = min(a[i], 0.03)  # at most 3 % of the donor mass
        delta = np.random.rand() * max_delta
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        val_new = compute_objective(a_new)

        # Metropolis acceptance
        if val_new < val or np.random.rand() < np.exp(-(val_new - val) / max(T, 1e-12)):
            a = a_new
            val = val_new

        if val_new < best_val:
            best_val = val_new
            best_a = a_new.copy()
            last_improve = time.time()
            print(f"[SA n={n} it={it}] new best {best_val:.6f}")

        # cooling schedule
        if it % 1000 == 0:
            T = max(T * decay, 1e-6)

        # a mild reheating if we have not improved for a while
        if time.time() - last_improve > 8.0:
            a = best_a + 0.02 * np.random.rand(n)
            a = project_simplex(a)
            val = compute_objective(a)
            last_improve = time.time()
            print(f"[SA n={n} it={it}] reheating")

    return best_a, best_val


# ----------------------------------------------------------------------
# 3. (1+1) Evolution Strategy – fine tuning
# ----------------------------------------------------------------------


def run_es_opt(
    n: int, init_a: np.ndarray, deadline: float, max_iter: int = 30_000
) -> Tuple[np.ndarray, float]:
    """
    Simple (1+1)‑ES with self‑adapting mutation strength σ.
    Operates directly on the simplex via a projection step.
    """
    a = init_a.copy()
    best_a = a.copy()
    best_val = compute_objective(a)
    sigma = 0.02
    last_improve = time.time()
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        trial = a + sigma * np.random.randn(n)
        trial = np.maximum(trial, 0.0)
        trial = project_simplex(trial)

        val_trial = compute_objective(trial)
        if val_trial < best_val:
            best_val = val_trial
            best_a = trial.copy()
            a = trial
            sigma = min(sigma * 1.05, 0.2)
            last_improve = time.time()
            print(f"[ES n={n} it={it}] new best {best_val:.6f}")
        else:
            sigma *= 0.99

        if time.time() - last_improve > 5.0:
            a = best_a.copy()
            sigma = 0.02
            last_improve = time.time()
            print(f"[ES n={n} it={it}] reset (stagnation)")

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function minimizing the evaluation metric.
    The routine respects the total time budget and returns the best
    sequence discovered (as a plain Python list).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # safety margin
    best_seq = None
    best_val = np.inf

    # try descending lengths – the largest gives most freedom
    lengths = [8192, 4096, 2048, 1024]

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")

        # --------------------------------------------------------------
        # Build a few sensible initial candidates
        # --------------------------------------------------------------
        init_candidates = []

        # warm‑start from the globally best sequence (if size matches)
        if best_seq is not None and len(best_seq) == n:
            init_candidates.append(np.asarray(best_seq, dtype=np.float64))

        # historic construction if present
        if "height_sequence_1" in globals():
            hs = np.asarray(height_sequence_1, dtype=np.float64)
            if hs.size >= n:
                hs = hs[:n]
            else:
                hs = np.pad(hs, (0, n - hs.size))
            hs = np.maximum(hs, 0.0)
            s = hs.sum()
            hs = hs / s if s > 1e-6 else np.random.dirichlet(np.ones(n))
            init_candidates.append(hs)

        # simple decreasing (linear) shape
        dec = np.arange(n, 0, -1, dtype=np.float64)
        dec = dec / dec.sum()
        init_candidates.append(dec)

        # uniform distribution
        init_candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

        # a random Dirichlet point
        init_candidates.append(np.random.dirichlet(np.ones(n)))

        # --------------------------------------------------------------
        # Phase 1 – Projected sub‑gradient descent
        # --------------------------------------------------------------
        if time.time() < deadline:
            sg_time = min((deadline - time.time()) * 0.30, 120.0)
            sg_deadline = time.time() + sg_time
            # run a handful of restarts
            for r, cand in enumerate(init_candidates[:3]):  # at most three restarts
                if time.time() >= sg_deadline:
                    break
                a_opt, val_opt = subgradient_descent(cand, sg_deadline)
                if val_opt < best_val:
                    best_val = val_opt
                    best_seq = a_opt.copy()
                    print(f"[n={n} SGD restart={r}] global best {best_val:.6f}")

        # --------------------------------------------------------------
        # Phase 2 – Simulated annealing (global exploration)
        # --------------------------------------------------------------
        if time.time() < deadline and best_seq is not None and len(best_seq) == n:
            sa_time = min((deadline - time.time()) * 0.30, 120.0)
            sa_deadline = time.time() + sa_time
            a_sa, val_sa = simulated_annealing(best_seq, sa_deadline)
            if val_sa < best_val:
                best_val = val_sa
                best_seq = a_sa.copy()
                print(f"[n={n} SA] global best {best_val:.6f}")

        # --------------------------------------------------------------
        # Phase 3 – (1+1)‑ES fine‑tuning

        if time.time() < deadline and best_seq is not None and len(best_seq) == n:
            es_time = min((deadline - time.time()) * 0.35, 180.0)
            es_deadline = time.time() + es_time
            a_es, val_es = run_es_opt(n, best_seq, es_deadline)
            if val_es < best_val:
                best_val = val_es
                best_seq = a_es.copy()
                print(f"[n={n} ES] global best {best_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – in the unlikely event that nothing was produced.
    # ------------------------------------------------------------------
    if best_seq is None:
        fallback_n = 1024
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)
        best_val = compute_objective(best_seq)
        print(f"[Fallback] uniform sequence, value {best_val:.6f}")

    print(f"\n*** Overall best value = {best_val:.6f} (length {len(best_seq)}) ***")
    # Return a plain Python list as required by the external evaluator
    return best_seq.tolist()
