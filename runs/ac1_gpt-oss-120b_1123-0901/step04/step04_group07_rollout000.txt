"""
Propose a non‑negative step function (list of heights) that minimises
``evaluate_sequence``.  

Algorithm
---------
* Explore several lengths (up to 8000) and many diverse initialisations:
  uniform, Dirichlet, exponential, reversed exponential, Hann/Blackman,
  triangular / power‑triangular, sinusoidal powers and interpolations of the
  current best solution.
* For each start run a two‑phase Adam optimiser:
  – A soft‑max smoothing of the max‑convolution is used initially,
    with an exponentially decaying temperature τ.
  – When τ becomes tiny we switch to the exact sub‑gradient of the
    max‑convolution.
  The Adam step is projected onto the feasible simplex (non‑negative,
  ℓ₁‑norm = √(2 n)).  Small Gaussian jitter is added periodically.
* After Adam finishes, a cheap stochastic “mass‑swap” local search
  (simulated‑annealing style) performs a few thousand random moves that
  keep the ℓ₁‑norm fixed and accept only improving (or occasionally
  slightly worsening) moves.
* The best vector over all runs and lengths is returned.
* All time‑critical loops check the global deadline (1000 s budget).

The code is pure NumPy+Python, respects the time limit and prints progress.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions for convolution and soft‑max smoothing
# ----------------------------------------------------------------------


def _softmax_weights(conv: np.ndarray, tau: float):
    """Soft‑max of convolution values.
    Returns the weight vector (summing to 1) and the smoothed maximum."""
    maxc = conv.max()
    if tau <= 0.0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w, maxc
    ex = np.exp((conv - maxc) / tau)
    sum_ex = ex.sum()
    w = ex / sum_ex
    Ltau = maxc + tau * np.log(sum_ex)
    return w, Ltau


def _weighted_rev(a: np.ndarray, w: np.ndarray, Nfft: int) -> np.ndarray:
    """Weighted reverse convolution needed for the (soft‑max) gradient.
    Computes rev_i = Σ_k w_k * a_{k‑i} (zero‑padding outside the range)."""
    A_fft = np.fft.rfft(a, n=Nfft)
    W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)
    conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
    conv_rev = conv_rev[: len(w) + a.shape[0] - 1]
    # Slice to obtain rev_i for i = 0 .. n‑1
    idx_start = 2 * a.shape[0] - 2
    rev = conv_rev[idx_start - np.arange(a.shape[0])]
    return rev


def _max_convolution(a: np.ndarray, Nfft: int) -> float:
    """Exact max‑convolution of a with itself (using FFT)."""
    A_fft = np.fft.rfft(a, n=Nfft)
    conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * a.shape[0] - 1]
    return float(conv.max())


# ----------------------------------------------------------------------
# Adam optimiser (soft‑max → exact)
# ----------------------------------------------------------------------


def _adam_optimize(
    a0: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int,
    init_lr: float = 0.10,
    tau_factor: float = 0.25,
) -> tuple[np.ndarray, float]:
    """
    Run Adam on a single starting vector.
    Returns the best vector seen and its exact objective value (max‑convolution).
    """
    target_sum = math.sqrt(2.0 * n)

    # Normalise the initial vector to the required ℓ₁‑norm.
    a = a0 * (target_sum / a0.sum())

    # Adam state.
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr0 = init_lr
    jitter_interval = 3000

    # Decay of the soft‑max temperature.
    decay_rate = max_iters / 12.0

    # FFT length (next power of two ≥ 2·n‑1).
    Nfft = 1 << ((2 * n - 1).bit_length())

    best_seq = a.copy()
    best_score = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ---- current exact convolution (needed for the objective) ----
        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        M = float(conv.max())

        # ---- temperature and (soft‑)max ----
        tau = tau_factor * M * math.exp(-it / decay_rate)
        if tau < 5e-10:
            # exact sub‑gradient: average over all active indices
            mask = conv >= M - 1e-12
            cnt = mask.sum()
            if cnt == 0:
                w = np.zeros_like(conv)
                w[0] = 1.0
            else:
                w = mask.astype(float) / cnt
            Ltau = M
        else:
            w, Ltau = _softmax_weights(conv, tau)

        # ---- weighted reverse convolution (gradient of the smoothed max) ----
        rev = _weighted_rev(a, w, Nfft)

        # ---- Adam step ----
        # The gradient of the *normalised* objective is 2·rev minus a
        # constant that enforces the ℓ₁‑constraint; we absorb the factor
        # into the learning‑rate.
        grad = 2.0 * rev - (4.0 * n * Ltau) / (target_sum ** 3)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / (1.0 + 1e-4 * it)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # ---- projection onto feasible set (non‑negative, fixed ℓ₁‑norm) ----
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a = a * (target_sum / s)
        else:
            a = np.full_like(a, target_sum / n)

        # ---- occasional jitter to escape plateaus ----
        if it % jitter_interval == 0:
            jitter = np.random.randn(n) * (1e-3 * target_sum / n)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        # ---- keep the best exact objective seen so far ----
        cur_score = M
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[len {n:5d} iter {it:7d}] new best = {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Simple stochastic local search (mass‑swap moves)
# ----------------------------------------------------------------------


def _local_search(a: np.ndarray, deadline: float, max_moves: int = 8000) -> tuple[np.ndarray, float]:
    """
    Perform a cheap random‑move hill‑climbing / simulated‑annealing phase.
    Moves a small amount of mass from one index to another while keeping the
    ℓ₁‑norm fixed.  Returns the best vector found and its exact objective.
    """
    n = a.shape[0]
    target_sum = a.sum()
    Nfft = 1 << ((2 * n - 1).bit_length())
    best_a = a.copy()
    best_score = _max_convolution(best_a, Nfft)

    # Annealing temperature schedule (linear decay)
    T0 = 0.001 * best_score
    for move in range(1, max_moves + 1):
        if time.time() >= deadline:
            break
        # linearly shrinking temperature
        T = T0 * (1.0 - move / max_moves)

        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # amount to transfer (relative to average height)
        delta = np.random.rand() * 0.02 * target_sum / n
        if best_a[i] < delta:
            continue

        a_new = best_a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        score = _max_convolution(a_new, Nfft)
        delta_score = score - best_score

        if delta_score < -1e-12:
            best_a, best_score = a_new, score
        else:
            # Metropolis acceptance
            if T > 0 and np.random.rand() < math.exp(-delta_score / T):
                best_a, best_score = a_new, score

    return best_a, best_score


# ----------------------------------------------------------------------
# Construction of diverse initial vectors
# ----------------------------------------------------------------------


def _make_initial_vectors(n: int, target_sum: float) -> list[np.ndarray]:
    """Return a list of diversified starting vectors (already ℓ₁‑normalised)."""
    rng = np.random.default_rng()
    init = []

    # 1) Uniform
    init.append(np.full(n, target_sum / n))

    # 2) Random Dirichlet
    g = rng.gamma(1.0, 1.0, size=n)
    init.append(g / g.sum() * target_sum)

    # 3) Exponential decay (random rate)
    rate = rng.uniform(0.0005, 0.02)
    x = np.exp(-rate * np.arange(n))
    init.append(x / x.sum() * target_sum)

    # 4) Reversed exponential
    init.append(x[::-1] / x.sum() * target_sum)

    # 5) Hann window
    hann = 0.5 - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1))
    init.append(hann / hann.sum() * target_sum)

    # 6) Blackman window
    a0, a1, a2 = 0.42, 0.5, 0.08
    idx = np.arange(n)
    black = a0 - a1 * np.cos(2.0 * np.pi * idx / (n - 1)) + a2 * np.cos(4.0 * np.pi * idx / (n - 1))
    init.append(black / black.sum() * target_sum)

    # 7) Power‑triangular shapes
    x_norm = np.linspace(0, 1, n)
    tri = 1.0 - np.abs(x_norm - 0.5) * 2.0
    for p in [0.5, 1, 2, 4, 8]:
        shape = np.maximum(tri, 0) ** p
        init.append(shape / shape.sum() * target_sum)

    # 8) Sinusoidal powers
    sin_part = np.sin(np.pi * x_norm)  # full sine bump
    for p in [0.5, 1, 2, 3]:
        shape = sin_part ** p
        init.append(shape / shape.sum() * target_sum)

    # 9) Small random perturbations of uniform
    pert = np.full(n, target_sum / n)
    pert += rng.normal(scale=0.02 * target_sum / n, size=n)
    pert = np.maximum(pert, 0.0)
    pert = pert / pert.sum() * target_sum
    init.append(pert)

    return init
