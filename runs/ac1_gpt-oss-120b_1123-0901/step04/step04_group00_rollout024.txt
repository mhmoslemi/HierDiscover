def full_convolution(a):
    n = a.shape[0]
    fft_len = 1 << int(np.ceil(np.log2(2*n-1)))
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[:2*n-1]
    return conv

def cross_correlation(w, a):
    n = a.shape[0]
    fft_len = 1 << int(np.ceil(np.log2(len(w)+n-1)))
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(a[::-1], fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n-1
    return corr[start:start+n]

    n = a.shape[0]

"""
Propose a step‑function sequence that (approximately) minimises

        F(a) = 2·n·max(a*a) / (sum(a)²)

under a_i ≥ 0.

Algorithm
---------
1. Multi‑start optimisation for several lengths (1024 … 16384).  For each
   length a handful of diverse initial vectors are created
   (warm‑start, historic, random Dirichlet, edge‑concentrated power‑law
   and two‑Gaussian shapes).

2. Every candidate is refined by two phases
   * Adam optimisation of a smooth‑max surrogate of the max‑convolution.
   * Exponentiated‑gradient sub‑gradient descent on the exact max.
   After each exact evaluation the global best sequence is updated.

3. When the global time budget is almost exhausted a cheap local‑search
   phase runs on the current best sequence.  It repeatedly transfers a
   small amount of mass from the index that contributes most to the
   current maximal lag (obtained from the sub‑gradient) to the index
   that contributes least, updating the convolution incrementally.
   Accept only improving moves – occasional random “shakes’’ avoid
   stagnation.

All helper functions are top‑level, no closures or lambdas are used.
The function respects the given time budget and returns the best
sequence found as a plain Python list of non‑negative floats.

import time
import numpy as np

# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    if fft_len is None:
        fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Compute the linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) equal to
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]

                shape /= shape.sum()
                candidates.append(shape)

        # two‑Gaussian edge shapes
        i = np.arange(n, dtype=np.float64)
        for gamma in [0.2, 0.5, 1.0, 2.0]:
            shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
            shape /= shape.sum()
            candidates.append(shape)

        # ------------------------------------------------------------------
        # optimise every candidate
        for cand_id, a0 in enumerate(candidates):
            if time.time() > deadline:
                break

            a = a0 / a0.sum()                     # probability simplex
            logits = np.log(a + 1e-15)            # parametrise the simplex

            # -------------------- Adam on smooth‑max surrogate ----------
            m = np.zeros_like(logits)
            v = np.zeros_like(logits)
            beta1, beta2, eps = 0.9, 0.999, 1e-8
            lr = 0.05
            lam = 0.5
            lam_factor = 1.02
            last_improve = time.time()
            # iteration budget scales roughly with 1/n
            max_iters = int(20000 * 1024 / n) + 3000

            for t in range(1, max_iters + 1):
                # soft‑max → simplex
                mx = logits.max()
                exp_x = np.exp(logits - mx)
                a = exp_x / exp_x.sum()

                # convolution and smooth‑max weights
                conv = _full_convolution(a, fft_len_conv)
                max_c = conv.max()
                w = np.exp(lam * (conv - max_c))
                w_sum = w.sum()
                if w_sum == 0:
                    w = np.full_like(w, 1.0 / w.size)
                else:
                    w /= w_sum

                # gradient of the surrogate
                grad_a = 2.0 * _cross_correlation(w, a, fft_len_corr)

                # back‑prop through soft‑max
                avg = np.dot(a, grad_a)
                grad_log = a * (grad_a - avg)

                # Adam update
                m = beta1 * m + (1.0 - beta1) * grad_log
                v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
                m_hat = m / (1.0 - beta1 ** t)
                v_hat = v / (1.0 - beta2 ** t)
                logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

                # temperature schedule
                lam = min(lam * lam_factor, 1e6)

                # exact objective (sum(a) ≈ 1)
                val = 2.0 * n * max_c / (a.sum() ** 2)

                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} cand={cand_id} Adam iter={t}] new best {val:.6f}")

                # stagnation handling – add a tiny random shake
                if time.time() - last_improve > 4.0:
                    logits += (np.random.rand(n) - 0.5) * 1e-2
                    last_improve = time.time()

                if time.time() > deadline:
                    break

            # -------------------- Sub‑gradient phase on the true max -----
            a = a / a.sum()
            eta = 0.6
            sub_last_improve = time.time()
            for sub_it in range(3000):
                conv = _full_convolution(a, fft_len_conv)
                max_c = conv.max()
                active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
                if not active.any():
                    w = np.zeros_like(conv)
                else:
                    w = np.zeros_like(conv)
                    w[active] = 1.0 / active.sum()

                grad_a = 2.0 * _cross_correlation(w, a, fft_len_corr)

                # exponentiated‑gradient step (keeps a ≥ 0)
                a = a * np.exp(-eta * grad_a)
                s = a.sum()
                if s < 1e-12:                     # avoid collapse
                    a = np.random.dirichlet(np.ones(n))
                    s = a.sum()
                a = a / s

                val = 2.0 * n * max_c / (a.sum() ** 2)

                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    sub_last_improve = time.time()
                    print(f"[n={n} cand={cand_id} Subgrad iter={sub_it}] new best {val:.6f}")

                # early stop if stuck for several seconds
                if time.time() - sub_last_improve > 4.0:
                    break
                if time.time() > deadline:
                    break

    # ------------------------------------------------------------------
    # Final local‑search refinement (mass‑transfer “hill‑climbing”)
    if best_seq is not None and time.time() + 30.0 < deadline:
        n = len(best_seq)
        print("\n=== Local refinement stage ===")
        a = np.array(best_seq, dtype=np.float64)
        conv = _full_convolution(a)               # exact convolution a*a
        best_val_local = 2.0 * n * conv.max()
        best_a_local = a.copy()
        no_improve = 0
        fft_len_corr = fft_len_corr_cache[n]

        while time.time() < deadline:
            max_c = conv.max()
            active = np.isclose(conv, max_c, atol=1e-12)
            w = np.zeros_like(conv)
            if active.any():
                w[active] = 1.0 / active.sum()
            grad_a = 2.0 * _cross_correlation(w, a, fft_len_corr)

            # source = index with largest gradient (contributes most to max)
            i = int(np.argmax(grad_a))
            # destination = index with smallest gradient
            j = int(np.argmin(grad_a))
            if i == j:
                j = (i + 1) % n

            # amount to move – at most half of the source mass
            max_delta = a[i] * 0.5
            if max_delta < 1e-14:
                # source exhausted – perform a random shake and continue
                a = np.random.dirichlet(np.ones(n))
                conv = _full_convolution(a)
                no_improve = 0
                continue
            delta = max_delta * np.random.rand()

            # ----- incremental update of the convolution ---------------
            conv_new = conv.copy()
            conv_new[i:i + n] -= 2.0 * delta * a
            conv_new[j:j + n] += 2.0 * delta * a
            ii = i + i
            jj = j + j
            ij = i + j
            conv_new[ii] += delta * delta
            conv_new[jj] += delta * delta
            conv_new[ij] -= 2.0 * delta * delta

            new_max = conv_new.max()
            new_val = 2.0 * n * new_max

            if new_val < best_val_local - 1e-12:
                # accept the move
                a[i] -= delta
                a[j] += delta
                conv = conv_new
                best_val_local = new_val
                best_a_local = a.copy()
                # also update the global record if it improved overall
                if new_val < best_val:
                    best_val = new_val
                    best_seq = a.copy()
                print(f"Local improvement: {new_val:.6f}")
                no_improve = 0
            else:
                no_improve += 1
                # occasional random shake to escape flat plateaus
                if no_improve > 2000:
                    a = np.random.dirichlet(np.ones(n))
                    conv = _full_convolution(a)
                    no_improve = 0
                    print("Random shake")
            # stay within the overall deadline
            if time.time() > deadline:
                break

        best_seq = best_a_local

    # ------------------------------------------------------------------
    # Fallback – uniform vector if nothing was found (very unlikely)
    if best_seq is None:
        fallback_n = 2048
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)

    # Return a plain Python list as required by the external evaluator
    return best_seq.tolist()
