The function `propose_candidate` searches for a non‑negative step function (a vector `a`) that minimises  

\[
F(a)=\frac{2\,n\;\max (a*a)}{(\sum a)^2},
\qquad a*a =\text{linear convolution of }a\text{ with itself}.
\]

Because the objective is scale‑invariant we may restrict `a` to the probability simplex  
`∑ a_i = 1`.  The problem then becomes

* minimise `max(conv)`  (where `conv = a*a`)  
* subject to `a_i ≥ 0 ,  ∑a_i = 1`.

The algorithm combines three ideas

1. **Projected Adam on a smooth‑max surrogate** –  
   `max(conv)` is approximated by a soft‑max with temperature `λ`.  
   Adam updates are performed on the simplex (Euclidean projection).

2. **Multi‑scale warm‑starts** –  
   We first optimise a short vector (256 entries), then repeatedly
   upscale the current best solution (simple repetition) and continue the
   optimisation at larger sizes (512, 1024, …).  This prevents the optimiser
   from getting trapped in a poor local minimum.

3. **Final random‑pair hill‑climbing** –  
   After the gradient phase a cheap local search moves a tiny amount of
   probability mass between two coordinates and accepts the move iff the
   exact objective improves.

The whole routine respects a global time budget (`budget_s`, default 1000 s)
and prints progress.  All helper functions are defined at top‑level, no
closures or file I/O are used.

```python
"""
propose_candidate – gradient + multi‑scale optimisation for the
max‑convolution objective.

The method:
    * Optimise on the probability simplex (sum = 1) using Adam.
    * Replace the true max by a soft‑max with temperature λ that is
      increased slowly (λ_start → λ_max).  When λ_max is reached a
      hard‑max (uniform weight on all near‑max indices) is used.
    * Work in a multi‑scale fashion: start with n=256, then 512, 1024,
      2048, 4096, 8192.  Each larger problem is warm‑started by repeating
      the best vector of the previous size.
    * After gradient optimisation a cheap random‑pair hill‑climb
      (transfer a tiny fraction of mass between two positions) refines
      the best solution.
The routine stops after the allotted time and returns a plain Python
list of non‑negative numbers.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions (no closures)

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using an FFT of length fft_len."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.shape[0] - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Compute (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)],   i = 0 … n‑1.
    This is the gradient of Σ_k w_k·(a*a)_k divided by 2.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("v must be 1‑D")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero -> uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None          # numpy array, ∑ = 1
    best_val = float("inf")

    # --------- exact objective for a simplex vector --------------------
    def _exact_obj(a: np.ndarray) -> float:
        n = a.shape[0]
        conv = _full_convolution(a, _next_pow2(2 * n - 1))
        return 2.0 * n * conv.max()

    # ------------------------------------------------------------------
    # Multi‑scale lengths.  The smallest size gives a cheap exploration,
    # the largest size (8 k) allows a much flatter convolution.
    lengths = [256, 512, 1024, 2048, 4096, 8192]

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")

        fft_len_conv = _next_pow2(2 * n - 1)
        fft_len_corr = _next_pow2(3 * n)          # enough for (2n‑1)+n

        # --------------------------------------------------------------
        # initialise a (probability vector) using the best solution of the
        # previous, smaller size if it exists.
        if best_seq is not None and best_seq.shape[0] < n:
            # simple up‑sampling by repetition
            repeat = n // best_seq.shape[0]
            a0 = np.repeat(best_seq, repeat)
            # pad if n is not a multiple of prev length
            if a0.shape[0] < n:
                a0 = np.pad(a0, (0, n - a0.shape[0]), constant_values=0.0)
            a0 = a0 / a0.sum()
            print("   warm‑start by up‑sampling previous best")
        else:
            a0 = None

        # --------------------------------------------------------------
        # Run several independent restarts for the current size.
        # The number of restarts is chosen adaptively – we spend more
        # restarts on the cheaper sizes.
        if n <= 1024:
            restarts = 12
            max_iters = 40000
        elif n <= 4096:
            restarts = 8
            max_iters = 25000
        else:
            restarts = 4
            max_iters = 15000

        for r in range(restarts):
            if time.time() > deadline:
                break

            # ---- initialise ----
            if a0 is not None and r == 0:
                a = a0.copy()
                print(f"[restart {r}] warm start from up‑sampled best")
            else:
                a = np.random.dirichlet(np.ones(n))
                print(f"[restart {r}] random start")

            # Adam state
            m = np.zeros(n, dtype=np.float64)
            v = np.zeros(n, dtype=np.float64)
            beta1, beta2, eps = 0.9, 0.999, 1e-8
            lr = 0.08                     # moderate learning rate

            # soft‑max temperature (smooth approximation of max)
            lam = 0.1                     # start very smooth
            lam_max = 50.0                # never let λ explode
            lam_factor = 1.03

            last_improve = time.time()
            it = 0

            while time.time() < deadline and it < max_iters:
                # ----- convolution and weights for the smooth max -----
                conv = _full_convolution(a, fft_len_conv)
                max_c = conv.max()

                if lam < lam_max:
                    # soft‑max weights, numerically stable
                    w = np.exp(lam * (conv - max_c))
                    w_sum = w.sum()
                    if w_sum == 0.0:
                        w = np.full_like(conv, 1.0 / conv.shape[0])
                    else:
                        w /= w_sum
                else:
                    # hard max – uniform weight on all “near‑max’’ indices
                    tol = 1e-12 + 1e-9 * max_c
                    mask = conv >= max_c - tol
                    cnt = mask.sum()
                    if cnt == 0:
                        w = np.full_like(conv, 1.0 / conv.shape[0])
                    else:
                        w = np.where(mask, 1.0 / cnt, 0.0)

                # ----- gradient of the surrogate -----
                grad_a = 2.0 * _cross_correlation(w, a, fft_len_corr)

                # ----- Adam update (before projection) -----
                it += 1
                m = beta1 * m + (1.0 - beta1) * grad_a
                v = beta2 * v + (1.0 - beta2) * (grad_a * grad_a)
                m_hat = m / (1.0 - beta1 ** it)
                v_hat = v / (1.0 - beta2 ** it)
                step = lr * m_hat / (np.sqrt(v_hat) + eps)
                a_new = a - step

                # ----- projection onto the simplex -----
                a = _project_simplex(a_new)

                # ----- temperature schedule -----
                lam = min(lam * lam_factor, lam_max)

                # ----- exact objective for monitoring -----
                val = _exact_obj(a)

                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} r={r} it={it:05d}] new best: {val:.6f}")

                    # early exit if we already beat the target
                    if best_val < 1.503:
                        print("Target reached – terminating early.")
                        deadline = time.time()   # break outer loops
                        break

                # ----- stagnation handling -----
                if time.time() - last_improve > 5.0:
                    # small random shake to escape flat regions
                    noise = (np.random.rand(n) - 0.5) * 1e-2
                    a = _project_simplex(a + noise)
                    last_improve = time.time()
                    print(f"[n={n} r={r} it={it}] shake (stagnation)")

            # end of one restart
        # end of length loop

    # ------------------------------------------------------------------
    # Final local random refinement (pairwise mass transfer)
    # --------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        a = best_seq.copy()
        n = a.shape[0]
        fft_len_conv = _next_pow2(2 * n - 1)
        print("\n--- Local random refinement (pairwise moves) ---")
        last_improve = time.time()
        while time.time() < deadline:
            i, j = np.random.choice(n, 2, replace=False)
            # move at most 0.5 % of the total mass from i to j
            delta = 0.005 * a[i] * np.random.rand()
            if delta <= 0.0:
                continue
            a_prop = a.copy()
            a_prop[i] -= delta
            a_prop[j] += delta
            a_prop = _project_simplex(a_prop)   # safeguard against rounding
            val = _exact_obj(a_prop)
            if val < best_val:
                best_val = val
                best_seq = a_prop.copy()
                a = a_prop
                last_improve = time.time()
                print(f"[refine] improved to {val:.6f}")

            # occasional reset to the global best if we have been idle
            if time.time() - last_improve > 30.0:
                a = best_seq.copy()
                last_improve = time.time()

    # ------------------------------------------------------------------
    # Fallback (should never be needed)
    if best_seq is None:
        n_fallback = 1024
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        best_val = _exact_obj(best_seq)
        print(f"Fallback uniform sequence, value = {best_val:.6f}")

    print(f"\nBest value found: {best_val:.6f}")
    return best_seq.tolist()
```
