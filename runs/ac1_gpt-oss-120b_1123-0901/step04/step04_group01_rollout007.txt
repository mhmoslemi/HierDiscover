"""
Propose a candidate step‑function (non‑negative vector) that minimizes

        F(a) = 2·n·max(a*a) / (sum(a)^2)

The algorithm works on the probability simplex (sum = 1) and uses
a smooth approximation of the max (log‑sum‑exp) with an annealed
temperature.  The gradient of the smooth max is obtained via a
convolution – the same FFT kernels as the reference implementation.
An Adam optimiser updates unconstrained logits, the logits are kept
symmetric (a_i = a_{n‑1‑i}) which reduces the effective dimension
and empirically guides the search to flatter autocorrelations.
After the main descent a cheap greedy “mass‑transfer” local search
tries to improve the best solution found.

The routine respects the 1000 s time budget, tries several problem
sizes (1024 … 16384) and several random restarts, and keeps the best
sequence overall.  The returned list can be fed directly to
`evaluate_sequence`.
"""

import time
import numpy as np


def full_convolution(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """
    Linear (full) convolution a * a using FFT.
    Returns an array of length 2*len(a)-1.
    """
    n = a.shape[0]
    if fft_len is None:
        fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


def cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) where
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i = 0…n‑1.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]

    The function runs for at most ``budget_s`` seconds and returns the best
    sequence (as a plain Python list) that was found.
    """

    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_val = float("inf")

    # problem sizes – larger n gives a lower theoretical bound ≈1
    lengths = [1024, 2048, 4096, 8192, 16384]

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")

        # FFT lengths that will be reused many times
        fft_len_conv = 1 << int(np.ceil(np.log2(2 * n - 1)))
        fft_len_corr = 1 << int(np.ceil(np.log2(3 * n)))  # ≥ (2n‑1)+n‑1

        # number of independent restarts for this length
        restarts = 5 if n <= 4096 else 3 if n <= 8192 else 2

        for r in range(restarts):
            if time.time() > deadline:
                break

            if best_seq is not None and len(best_seq) == n:
                a0 = best_seq.copy()

                x = np.log(a0 + 1e-15)
