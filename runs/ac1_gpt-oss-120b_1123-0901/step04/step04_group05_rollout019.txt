"""
Hybrid random‑search / evolutionary / LP‑boosted optimizer for the
self‑convolution bound.

The algorithm works in three stages within the given time budget:

1. **Random families** – generate many candidates from simple parametric
   families (exponential, polynomial, half‑sine, Gaussian, Dirichlet,
   bimodal).  Keep a small population of the best ones.

2. **LP boost** – for the current best sequence f, solve a linear program
   that maximises the total mass of a new sequence g under the
   constraint (f⋆g)_i ≤ M (where M = max(f⋆f)).  After rescaling g to the
   required ℓ₁‑norm we move a fraction t towards g.  Repeating this a few
   times quickly lowers the bound.

3. **Adam fine‑tuning** – run the smooth‑max Adam optimiser (soft‑max
   parametrisation) starting from the best sequence found so far.  The
   temperature λ is increased slowly (continuation) and the learning
   rate decays, providing a precise final refinement.

All steps respect the global deadline (default 1000 s).  The routine
returns a plain Python list of non‑negative heights.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions for convolutions and the smooth‑max surrogate
def _full_convolution(a: np.ndarray, size: int = None) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``size``."""
    n = a.shape[0]
    if size is None:
        size = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return conv


def _softmax_weights(conv: np.ndarray, lam: float) -> np.ndarray:
    """Soft‑max weights wₖ = exp(lam·convₖ) / Σⱼ exp(lam·convⱼ)."""
    m = conv.max()
    w = np.exp(lam * (conv - m))
    s = w.sum()
    if s == 0.0:
        w.fill(1.0 / w.shape[0])
    else:
        w /= s
    return w


def _softmax_gradient(a: np.ndarray, w: np.ndarray, size_corr: int = None) -> np.ndarray:
    """
    Gradient of the smooth‑max surrogate with respect to the original
    probabilities ``a``.  The closed form is  ∇ = 2·(w ★ a).
    """
    n = a.shape[0]
    rev_a = a[::-1]
    if size_corr is None:
        size_corr = 1 << int(np.ceil(np.log2(len(w) + n - 1)))
    fw = np.fft.rfft(w, size_corr)
    frev = np.fft.rfft(rev_a, size_corr)
    corr = np.fft.irfft(fw * frev, size_corr)
    grad = 2.0 * corr[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Simple candidate families
def _sample_exponential(n):
    r = np.random.uniform(0.85, 0.999)
    a = r ** np.arange(n, dtype=np.float64)
    return a


def _sample_exponential_rev(n):
    a = _sample_exponential(n)[::-1]
    return a


def _sample_polynomial_decay(n):
    p = np.random.uniform(0.5, 5.0)
    a = (np.arange(1, n + 1, dtype=np.float64)) ** (-p)
    return a


def _sample_polynomial_decay_rev(n):
    a = _sample_polynomial_decay(n)[::-1]
    return a


def _sample_sine_power(n):
    p = np.random.uniform(0.5, 4.0)
    a = np.sin(np.pi * (np.arange(1, n + 1, dtype=np.float64)) / (n + 1))
    a = np.maximum(a, 0.0) ** p
    return a


def _sample_sine_power_rev(n):
    a = _sample_sine_power(n)[::-1]
    return a


def _sample_gaussian_left(n):
    mu = np.random.uniform(0.0, 0.2 * n)
    sigma = np.random.uniform(0.05 * n, 0.3 * n)
    x = np.arange(n, dtype=np.float64)
    a = np.exp(-((x - mu) ** 2) / (2.0 * sigma ** 2))
    return a


def _sample_gaussian_right(n):
    mu = np.random.uniform(0.8 * n, n)
    sigma = np.random.uniform(0.05 * n, 0.3 * n)
    x = np.arange(n, dtype=np.float64)
    a = np.exp(-((x - mu) ** 2) / (2.0 * sigma ** 2))
    return a


def _sample_bimodal_exp(n):
    split = np.random.randint(1, n - 1)
    r1 = np.random.uniform(0.85, 0.999)
    r2 = np.random.uniform(0.85, 0.999)
    left = r1 ** np.arange(split, dtype=np.float64)
    right = r2 ** np.arange(n - split, dtype=np.float64)[::-1]
    a = np.concatenate([left, right])
    return a


def _sample_dirichlet(n):
    a = np.random.dirichlet(np.ones(n))
    return a


# ----------------------------------------------------------------------
# Linear‑program (LP) helper – builds the convolution matrix for a given vector
def _conv_matrix(a: np.ndarray) -> np.ndarray:
    """Build matrix A where (A @ g) = a ⋆ g (full linear convolution)."""
    n = a.shape[0]
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    # each column j contains a shifted copy of `a` starting at row j
    for j in range(n):
        A[j : j + n, j] = a
    return A

def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step function (list of heights) that minimises
    ``evaluate_sequence``.  The routine runs for at most ``budget_s`` seconds
    and returns the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.2    # safety margin

    # ------------------------------------------------------------------
    # Determine problem size – reuse the historic solution if available
    hs1 = globals().get('height_sequence_1')
    if hs1 is not None:
        n = len(hs1)
        init_seq = np.array(hs1, dtype=np.float64)
        # normalise to the common ℓ₁‑norm √(2 n)
        init_seq = init_seq * (np.sqrt(2.0 * n) / init_seq.sum())
        best_seq = init_seq.copy()
        best_val = float('inf')
    else:
        n = 3000                     # a comfortable size for the budget
        best_seq = None
        best_val = float('inf')
        init_seq = None

    # Pre‑compute FFT sizes (used many times)
    conv_fft_size = 1 << int(np.ceil(np.log2(2 * n - 1)))
    size_corr = 1 << int(np.ceil(np.log2(3 * n)))

    # ------------------------------------------------------------------
    # Helper to evaluate a raw ndarray (fast version, but we also call
    # the official ``evaluate_sequence`` for a final, safe check).
    def _objective(a):
        conv = _full_convolution(a, size=conv_fft_size)
        return 2.0 * n * conv.max() / (a.sum() ** 2)

    # ------------------------------------------------------------------
    # ------------------------------------------------------------------
    # 1) RANDOM / EVOLUTIONARY SEARCH
    # ------------------------------------------------------------------
    pop_size = 20          # keep a small elite population
    population = []        # list of (seq, val)
    families = [
        _sample_exponential,
        _sample_exponential_rev,
        _sample_polynomial_decay,
        _sample_polynomial_decay_rev,
        _sample_sine_power,
        _sample_sine_power_rev,
        _sample_gaussian_left,
        _sample_gaussian_right,
        _sample_bimodal_exp,
        _sample_dirichlet,
    ]

    iter_counter = 0
    while time.time() < deadline:
        iter_counter += 1

        # --------------------------------------------------------------
        # 50% chance we generate a brand‑new candidate, otherwise we
        # crossover two elite members (if we already have a population).
        if len(population) < pop_size or np.random.rand() < 0.5:
            # random draw from a family
            fam = np.random.choice(families)
            cand = fam(n)
        else:
            # crossover + mutation
            p1, p2 = np.random.choice(population, 2, replace=False)
            a1, a2 = p1[0], p2[0]
            w = np.random.rand()
            cand = w * a1 + (1 - w) * a2
            # small Gaussian shake
            if np.random.rand() < 0.3:
                cand += np.random.normal(scale=0.01, size=n)
            cand = np.maximum(cand, 0.0)

        # Normalise (scale invariance – we use √(2 n) for convenience)
        if cand.sum() == 0.0:
            continue
        cand = cand * (np.sqrt(2.0 * n) / cand.sum())

        # Evaluate via the official function (includes clipping)
        try:
            val = evaluate_sequence(cand.tolist())
        except Exception:            # safety – fall back to the fast version
            val = _objective(cand)

        # Update global best
        if val < best_val:
            best_val = val
            best_seq = cand.copy()
            print(f"[Random] iter {iter_counter:06d}  new global best = {val:.6f}")

        # Maintain the elite population
        if len(population) < pop_size:
            population.append((cand, val))
        else:
            # replace the worst if this candidate is better
            worst_idx = max(range(pop_size), key=lambda i: population[i][1])
            if val < population[worst_idx][1]:
                population[worst_idx] = (cand, val)

        # Periodically prune to keep the population size exactly pop_size
        if len(population) > pop_size:
            population.sort(key=lambda x: x[1])
            population = population[:pop_size]

        # To avoid spending all the budget on random search,
        # we break after a reasonable number of iterations.
        if iter_counter >= 2000:
            break

    # ------------------------------------------------------------------
    # 2) LP BOOST – try to improve the currently best sequence using the
    #    linear‑program described in the paper.
    # ------------------------------------------------------------------
    try:
        from scipy.optimize import linprog
        have_lp = True
    except Exception:
        have_lp = False
        print("[LP] SciPy linprog not available – skipping LP boost.")

    if have_lp and best_seq is not None:
        # we will perform a few LP steps (tuned by remaining time)
        t_lp = 0.45              # convex combination weight toward the LP solution
        lp_iters = 0
        max_lp_iters = 12        # a small fixed budget – each LP is expensive

        while time.time() < deadline and lp_iters < max_lp_iters:
            lp_iters += 1

            # current max of self‑convolution
            conv_f = _full_convolution(best_seq, size=conv_fft_size)
            M = conv_f.max()

            # build the constraint matrix A (f ⋆ b) <= M
            A = _conv_matrix(best_seq)          # shape (2n‑1, n)
            b_ub = np.full(A.shape[0], M, dtype=np.float64)

            # objective: maximise sum(b)  → minimise -sum(b)
            c = -np.ones(n, dtype=np.float64)

            # solve LP
            res = linprog(c, A_ub=A, b_ub=b_ub,
                          bounds=[(0, None)] * n,
                          method='highs',
                          options={'presolve': True, 'time_limit': max(10.0, (deadline - time.time()))})

            if not res.success:
                print(f"[LP] iteration {lp_iters} failed ({res.message})")
                break

            b0 = res.x
            sum_b0 = b0.sum()
            if sum_b0 == 0.0:
                break

            # Rescale to the required ℓ₁‑norm √(2 n)
            scale = np.sqrt(2.0 * n) / sum_b0
            g = b0 * scale

            # Convex combination step

            # Clean up numerical noise and renormalise (still ℓ₁‑norm √(2 n))
            best_seq = np.maximum(best_seq, 0.0)
            best_seq = best_seq * (np.sqrt(2.0 * n) / best_seq.sum())

            # Evaluate new bound

                val = evaluate_sequence(best_seq.tolist())
            except Exception:
                val = _objective(best_seq)

            else:
                # if no improvement, shrink the step size a little
                t_lp = max(0.1, t_lp * 0.7)

    # ------------------------------------------------------------------
    # 3) ADAM REFINEMENT – smooth‑max Adam optimiser, seeded from the
    #    best sequence discovered so far.
    # ------------------------------------------------------------------
    if best_seq is not None:
        # log‑softmax parametrisation (guarantees non‑negativity and Σa = 1)
        eps = 1e-30
        u = np.log(best_seq + eps)

        # Adam state
        m = np.zeros_like(u)
        v = np.zeros_like(u)

        # continuation parameters
        lam = 0.6                     # start smooth, increase slowly
        lam_factor = 1.015
        base_lr = 0.045
        t = 0
        max_iter = 30000
        last_improve = time.time()
        while time.time() < deadline and t < max_iter:
            t += 1

            # probabilities a (simple soft‑max, Σa = 1)
            u_shift = u - np.max(u)
            a = np.exp(u_shift)
            a /= a.sum()

            # convolution and soft‑max weights
            conv = _full_convolution(a, size=conv_fft_size)
            w = _softmax_weights(conv, lam)

            # gradient wrt a (smooth‑max surrogate)
            grad_a = _softmax_gradient(a, w, size_corr=size_corr)

            # map gradient to log‑space (soft‑max Jacobian)
            dot = np.dot(a, grad_a)
            grad_u = a * (grad_a - dot)

            # Adam update
            m = 0.9 * m + 0.1 * grad_u
            v = 0.999 * v + 0.001 * (grad_u ** 2)

            it_f = float(t)
            m_hat = m / (1.0 - 0.9 ** it_f)
            v_hat = v / (1.0 - 0.999 ** it_f)

            lr = base_lr * (0.9997 ** t)
            u -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

            # raise smoothness

                except Exception:
                    val = 2.0 * n * conv.max() / (a.sum() ** 2)

                    last_improve = time.time()
                    print(f"[Adam] iter {t:05d}  new best = {val:.6f}")

                m.fill(0.0)
                v.fill(0.0)
                last_improve = time.time()

    # evaluate once more for safety
    try:
        _ = evaluate_sequence(best_seq.tolist())
    except Exception:
        pass

    return best_seq.tolist()
