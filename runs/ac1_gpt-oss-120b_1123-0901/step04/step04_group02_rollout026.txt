"""
propose_candidate
-----------------

Search for a non‑negative step‑function (a probability vector) that
minimises

       F(a) = 2·n·max(a*a) / (sum(a)**2)

where ``a*a`` is the linear convolution of ``a`` with itself.
The routine mixes three ideas:

1. **Smooth‑max Adam optimisation** – the max‑convolution is replaced
   by a differentiable soft‑max, gradients are taken w.r.t. the simplex
   (with Euclidean projection) and Adam updates are performed.

2. **Fast hill‑climbing** – after the Adam phase a large number of tiny
   mass‑transfer moves are tried.  The convolution is updated incrementally
   (O(n) per move) using the analytical formula for a rank‑1 update.
   Only moves that strictly improve the objective are kept (greedy
   descent).

3. **Final local refinement** – a short random “shake‑and‑accept”
   phase copies the best sequence found and tries tiny random transfers
   between two coordinates.

The whole search respects a global time budget (default 1000 s).  Several
different lengths are tried (smaller lengths allow many more iterations
while larger ones give more freedom).  If a previous construction is
available through the global name ``height_sequence_1`` it is used as a
warm‑start, otherwise random Dirichlet initialisations are employed.

The function returns a plain Python list – the external evaluator will
call ``evaluate_sequence`` on this list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.shape[0] - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Compute the gradient of the smooth‑max surrogate.
    (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)], i = 0 … n‑1.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex:
        {x | x_i ≥ 0, Σ_i x_i = 1}
    """
    if v.ndim != 1:
        raise ValueError("v must be a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _smooth_max_adam(a: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 30000) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the simplex using a smooth‑max surrogate of
    max(conv).  Returns the (possibly) improved vector and its exact
    objective value.
    """
    fft_len_conv = _next_pow2(2 * n - 1)
    fft_len_corr = _next_pow2(3 * n)          # enough for (2n‑1)+n

    # Adam state
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.12                                  # a bit more aggressive
    lam = 0.4                                   # temperature for soft‑max
    lam_factor = 1.3

    best_val = np.inf
    best_a = a.copy()
    t = 0
    while time.time() < deadline and t < max_iters:
        # ----- smooth‑max weights -----
        conv = _full_convolution(a, fft_len_conv)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w /= w_sum

        # ----- gradient w.r.t. a -----
        grad = 2.0 * _cross_correlation(w, a, fft_len_corr)

        # ----- Adam update -----
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step
        a = _project_simplex(a_new)

        # ----- temperature schedule -----
        lam = min(lam * lam_factor, 5e5)

        # ----- exact objective for monitoring -----
        exact_val = 2.0 * n * _full_convolution(a, fft_len_conv).max()
        if exact_val < best_val:
            best_val = exact_val
            best_a = a.copy()
        # small stagnation handling – a random “shake”
        if t % 3000 == 0:
            noise = (np.random.rand(n) - 0.5) * 2e-3
            a = _project_simplex(a + noise)

    return best_a, best_val


def _incremental_move(a: np.ndarray,
                      conv: np.ndarray,
                      i: int,
                      j: int,
                      delta: float) -> np.ndarray:
    """
    Return a new convolution after moving ``delta`` mass from index i
    to index j (the vector ``a`` is *not* modified).
    """
    n = a.shape[0]
    new_conv = conv.copy()
    # linear terms 2*delta*(a_{k-j} - a_{k-i})
    new_conv[i : i + n] -= 2.0 * delta * a
    new_conv[j : j + n] += 2.0 * delta * a
    # quadratic delta^2 terms
    new_conv[2 * i] += delta * delta
    new_conv[2 * j] += delta * delta
    new_conv[i + j] -= 2.0 * delta * delta
    return new_conv


def _hill_climb(a: np.ndarray,
                conv: np.ndarray,
                n: int,
                sub_deadline: float) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climbing: repeatedly transfer a small fraction of mass
    from a random coordinate to another one if the move reduces the
    exact objective.  Operates in O(n) per move by updating the
    convolution incrementally.
    """
    best_a = a.copy()
    best_val = 2.0 * n * conv.max()
    cur_a = a.copy()
    cur_conv = conv.copy()
    cur_val = best_val

    while time.time() < sub_deadline:
        i = np.random.randint(n)
        if cur_a[i] < 1e-12:
            continue
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # move at most 30 % of the mass at i
        max_delta = 0.30 * cur_a[i]
        if max_delta <= 0.0:
            continue
        delta = max_delta * np.random.rand()

        cand_conv = _incremental_move(cur_a, cur_conv, i, j, delta)
        cand_val = 2.0 * n * cand_conv.max()
        if cand_val < cur_val:
            # accept move
            cur_a[i] -= delta
            cur_a[j] += delta
            cur_conv = cand_conv
            cur_val = cand_val
            if cand_val < best_val:
                best_val = cand_val
                best_a = cur_a.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) beats the
    current best upper bound.  The algorithm mixes Adam optimisation on a
    smooth‑max surrogate with a fast hill‑climbing phase.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # helper to evaluate the exact objective for a probability vector
    def _exact_val(a: np.ndarray) -> float:
        n = a.shape[0]
        conv = _full_convolution(a, _next_pow2(2 * n - 1))
        return 2.0 * n * conv.max()

    # ------------------------------------------------------------------
    # Lengths to try – smaller lengths give many cheap iterations,
    # larger lengths provide more freedom.
    lengths = [256, 512, 1024, 2048]

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")
        for restart in range(3):
            if time.time() > deadline:
                break

            # ---- initialise a (probability vector) --------------------
            if restart == 0 and best_seq is not None and best_seq.shape[0] == n:
                a = best_seq.copy()
                print("[restart 0] warm start from global best")
            elif 'height_sequence_1' in globals():
                cand = np.asarray(height_sequence_1, dtype=np.float64)
                if cand.shape[0] >= n:
                    cand = cand[:n]
                else:
                    cand = np.pad(cand, (0, n - cand.shape[0]))
                cand = np.maximum(cand, 0.0)
                s = cand.sum()
                if s < 0.01:
                    a = np.random.dirichlet(np.ones(n))
                else:
                    a = cand / s
                print("[restart] start from height_sequence_1")
            else:
                a = np.random.dirichlet(np.ones(n))
                print("[restart] random start")

            # ---- Adam phase -----------------------------------------
            sub_deadline = min(deadline, time.time() + 0.45 * (deadline - time.time()))
            a_adam, val_adam = _smooth_max_adam(a, n, sub_deadline)
            if val_adam < best_val:
                best_val = val_adam
                best_seq = a_adam.copy()
                print(f"[n={n} restart={restart}] Adam improved to {val_adam:.6f}")

            # ---- hill‑climbing phase --------------------------------
            # recompute the exact convolution for the Adam result
            conv = _full_convolution(a_adam, _next_pow2(2 * n - 1))
            hc_deadline = min(deadline, time.time() + 0.40 * (deadline - time.time()))
            a_hc, val_hc = _hill_climb(a_adam, conv, n, hc_deadline)
            if val_hc < best_val:
                best_val = val_hc
                best_seq = a_hc.copy()
                print(f"[n={n} restart={restart}] hill‑climb improved to {val_hc:.6f}")

            # ---- tiny random refinement (optional) ------------------
            # only if we still have time
            if time.time() < deadline and best_seq is not None:
                a_ref = best_seq.copy()
                n_ref = a_ref.shape[0]
                fft_len_conv = _next_pow2(2 * n_ref - 1)
                # simple shake: move up to 0.1 % of total mass between two slots
                while time.time() < deadline:
                    i, j = np.random.choice(n_ref, 2, replace=False)
                    delta = 0.001 * a_ref[i] * np.random.rand()
                    if delta <= 0.0:
                        continue
                    a_prop = a_ref.copy()
                    a_prop[i] -= delta
                    a_prop[j] += delta
                    val = _exact_val(a_prop)
                    if val < best_val:
                        best_val = val
                        best_seq = a_prop.copy()
                        a_ref = a_prop
                        print(f"[refine] improved to {val:.6f}")

    # ------------------------------------------------------------------
    # fallback – uniform vector if nothing succeeded
    if best_seq is None:
        n_fallback = 1024
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        best_val = _exact_val(best_seq)
        print(f"Fallback uniform sequence, value = {best_val:.6f}")

    # Return plain Python list as required by the external evaluator.
    return best_seq.tolist()
