# ----------------------------------------------------------------------
# Helper – full linear convolution using real FFT (fast for any length)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of real vectors x and y via FFT."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    # eliminate tiny negative contributions caused by numerical noise
    np.maximum(conv, 0.0, out=conv)
    return conv

        idx = (2 * n - 2) - np.arange(n)
        grad_a = 2.0 * conv2[idx]

        # ------------------------------------------------------------------
        # 5. Chain rule back to logits x (soft‑max Jacobian)
        # ------------------------------------------------------------------
        dotp = np.dot(p, grad_a)
        grad_x = target_sum * p * (grad_a - dotp)

        # ------------------------------------------------------------------
        # 6. Optional gradient clipping (stabilises large updates)
        # ------------------------------------------------------------------
        norm = np.linalg.norm(grad_x)
        if norm > grad_clip_norm:
            grad_x = grad_x * (grad_clip_norm / norm)

        # ------------------------------------------------------------------
        # 7. Adam update
        # ------------------------------------------------------------------
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # ------------------------------------------------------------------
        # 8. Occasional tiny random “jitter” to escape plateaus
        # ------------------------------------------------------------------
        if iteration % 300 == 0 and np.random.rand() < 0.25:
            x += np.random.randn(n) * 0.0005

        # ------------------------------------------------------------------
        # 9. Decay the base learning‑rate slowly
        # ------------------------------------------------------------------
        alpha *= alpha_decay

    # final evaluation of the best sequence found during this run
    final_score = evaluate_sequence(best_seq.tolist())
    return best_seq, final_score


# ----------------------------------------------------------------------
# Main entry point required by the external harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative height sequence that minimises the given
    evaluation function.  The routine respects the supplied time budget
    and returns the best sequence discovered (as a plain Python list).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # small safety margin

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # 1. Warm‑start from any previously known construction
    # --------------------------------------------------------------
    try:
        hs = globals()["height_sequence_1"]
        hs_arr = np.array(hs, dtype=float)
        hs_arr = np.clip(hs_arr, 0.0, 1000.0)
        score = evaluate_sequence(hs_arr.tolist())
        if score < best_score:
            best_score = score
            best_seq = hs_arr.copy()
            print(f"[{time.time():.2f}] Warm‑start from global seed = {score:.6f}")
    except Exception:
        pass   # no warm‑start available

    # --------------------------------------------------------------
    # 2. Main multi‑restart search loop
    # --------------------------------------------------------------
    while time.time() < deadline:
        remaining = deadline - time.time()

        # ----------------------------------------------------------
        # Choose a length.  Early on we favour larger n (up to 12 000);
        # later we reduce the range to keep iterations cheap.
        # ----------------------------------------------------------
        if remaining > 400:
            n = np.random.randint(6000, 12000 + 1)
        elif remaining > 180:
            n = np.random.randint(4000, 9000 + 1)
        else:
            n = np.random.randint(2000, 5000 + 1)

        # ----------------------------------------------------------
        # Build a diverse initial shape.
        # ----------------------------------------------------------
        # primary power‑law component
        p1 = np.random.uniform(0.05, 1.2)
        idx = np.arange(1, n + 1, dtype=float)
        init = idx ** (-p1)

        # optional second component (adds richness)
        if np.random.rand() < 0.38:
            p2 = np.random.uniform(0.05, 1.2)
            w2 = np.random.rand() * 0.6 + 0.2        # weight in [0.2,0.8]
            init = w2 * (idx ** (-p2)) + (1 - w2) * init

        # optional exponential taper (pushes mass towards the start)
        if np.random.rand() < 0.45:
            tau_exp = np.random.uniform(0.4, 2.5) * n
            init *= np.exp(-idx / tau_exp)

        # optional short moving‑average smoothing
        if np.random.rand() < 0.25:
            klen = np.random.randint(3, 11)
            kernel = np.ones(klen) / klen
            init = np.convolve(init, kernel, mode='same')

        # tiny jitter to break symmetry
        init += np.random.rand(n) * 1e-6

        # ----------------------------------------------------------
        # Occasionally inject the current global best (truncated or padded)
        # ----------------------------------------------------------
        if best_seq is not None and np.random.rand() < 0.5:
            bs = best_seq
            if bs.size >= n:
                init = bs[:n].copy()
            else:
                pad = np.random.rand(n - bs.size) * 1e-6
                init = np.concatenate([bs, pad])

        # ----------------------------------------------------------
        # Ensure strict positivity before the log‑it transform
        # ----------------------------------------------------------
        init = np.maximum(init, 1e-12)

        # ----------------------------------------------------------
        # Normalise to the required sum Σa = √(2 n)
        # ----------------------------------------------------------
        target_sum = math.sqrt(2.0 * n)
        init = init / init.sum() * target_sum

        # Log‑it representation (log of a positive vector)
        x0 = np.log(init + 1e-12)

        # ----------------------------------------------------------
        # Allocate a time slice for this run (max 20 s, never exceed deadline)
        # ----------------------------------------------------------
        slice_time = min(20.0, remaining * 0.5)
        run_deadline = min(deadline, time.time() + slice_time)

        # ----------------------------------------------------------
        # Run the Adam optimiser on the soft‑max parametrisation
        # ----------------------------------------------------------
        seq_local, score_local = _optimise_adam_softmax(
            n=n,
            x0=x0,
            target_sum=target_sum,
            run_deadline=run_deadline,
            init_alpha=0.2,
            alpha_decay=0.9997,
            init_tau_factor=0.2,
            min_tau_factor=1e-4,
            grad_clip_norm=12.0,
        )

        # ----------------------------------------------------------
        # Update the global best if we have found an improvement
        # ----------------------------------------------------------
        if score_local < best_score:
            best_score = score_local
            best_seq = seq_local.copy()
            print(
                f"[{time.time():.2f}] GLOBAL new best = {best_score:.6f} (n={n})"
            )
            # early exit if we comfortably beat the target bound
            if best_score < 1.503:
                break

    # ------------------------------------------------------------------
    # 3. Fallback – generate a simple uniform sequence if nothing else.
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1500
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)

    # Return a plain Python list – the evaluator expects that format.
    return best_seq.tolist()
