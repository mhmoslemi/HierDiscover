def _adam_optimize(a0, n, fft_len, fft_len_corr, max_iters=20000, lr=0.05, lam_start=0.5, lam_factor=1.02, time_deadline=None):
    logits = np.log(a0 + 1e-15)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lam = lam_start
    best_a = a0.copy()
    best_val = float('inf')
    last_improve = time.time()
    for t in range(1, max_iters+1):
        if time_deadline is not None and time.time() > time_deadline:
            break
        # softmax
        mx = logits.max()
        exp_logits = np.exp(logits - mx)
        a = exp_logits / exp_logits.sum()
        # convolution
        conv = _full_convolution(a, fft_len)
        max_c = conv.max()
        # smooth max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w = w / w_sum
        # gradient wrt a
        grad_a = 2.0 * _cross_correlation(w, a, fft_len_corr)
        # gradient wrt logits via softmax jacobian
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)
        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)
        # increase lam
        lam = min(lam * lam_factor, 1e6)
        # compute objective
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
        # stagnation termination
        if time.time() - last_improve > 4.0:
            # small perturbation to logits
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()
    return best_a, best_val

def _subgradient_optimize(a0, n, fft_len, fft_len_corr, max_iters=2000, eta=0.6, time_deadline=None):
    a = a0.copy()
    # ensure sum normalized
    a = a / a.sum()
    best_a = a.copy()
    # compute initial conv and val
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    best_val = 2.0 * n * max_c / (a.sum() ** 2)
    last_improve = time.time()
    for it in range(max_iters):
        if time_deadline is not None and time.time() > time_deadline:
            break
        conv = _full_convolution(a, fft_len)
        max_c = conv.max()
        # find active indices
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)
        grad_a = 2.0 * _cross_correlation(w, a, fft_len_corr)
        # Exponentiated gradient step
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s
        # evaluate
        conv = _full_convolution(a, fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
        # stagnation break
        if time.time() - last_improve > 4.0:
            break
    return best_a, best_val

def _interpolate_sequence(seq, target_n):
    if len(seq) == target_n:
        return seq.copy()
    x_src = np.linspace(0, 1, len(seq))
    x_tgt = np.linspace(0, 1, target_n)
    seq_interp = np.interp(x_tgt, x_src, seq)
    # ensure non-negative
    seq_interp = np.maximum(seq_interp, 0.0)
    s = seq_interp.sum()
    if s < 1e-12:
        seq_interp = np.random.dirichlet(np.ones(target_n))
    else:
        seq_interp = seq_interp / s
    return seq_interp

"""
Propose a step‑function sequence (non‑negative, sum‐normalised) that
minimises

        F(a) = 2·n·max(a*a) / (sum(a)²)

using a three‑phase heuristic:

1. Adam optimisation on a smooth‑max surrogate of the convolution.
   The surrogate temperature λ is increased geometrically; the
   variables are kept on the probability simplex via the soft‑max
   (logits) trick.

2. Exponentiated‑gradient sub‑gradient descent on the exact max‑convolution.

3. Short simulated‑annealing fine‑tuning (small random mass transfers).

The routine works for several lengths (1024 … 16384), builds a diverse
set of initial candidates (warm‑starts, random Dirichlet draws,
edge‑concentrated power‑laws, bi‑Gaussian edges, interpolated historic
solutions) and runs the three phases for each candidate while respecting
the global time budget (default 1000 s).  The best sequence found
anywhere is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """
    Linear convolution a*a (length 2·len(a)-1) using FFT.
    """
    n = a.shape[0]
    if fft_len is None:
        fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation needed for the gradient.
    Returns a vector of length n where
        (w ★ a)[i] = Σ_k w[k]·a[i+k−(n−1)]   for i=0…n−1.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]  # length n


def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact evaluation of F(a) (ignoring the caps that evaluate_sequence later adds)."""
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    return 2.0 * n * max_c / (a.sum() ** 2)


def _interpolate_sequence(seq: np.ndarray, target_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a new length."""
    if seq.shape[0] == target_n:
        return seq.copy()
    x_src = np.linspace(0.0, 1.0, seq.shape[0])
    x_tgt = np.linspace(0.0, 1.0, target_n)
    interp = np.interp(x_tgt, x_src, seq)
    interp = np.maximum(interp, 0.0)
    s = interp.sum()
    if s < 1e-12:
        interp = np.random.dirichlet(np.ones(target_n))
    else:
        interp = interp / s
    return interp

def _adam_optimize(a0: np.ndarray, n: int, fft_len: int, fft_len_corr: int,
                   max_iters: int = 20000, lr: float = 0.05,
                   lam_start: float = 0.5, lam_factor: float = 1.02,
                   time_deadline: float = None) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth‑max surrogate.
    Returns the best vector seen and its objective value.
    """
    logits = np.log(a0 + 1e-15)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lam = lam_start
    best_a = a0.copy()
    best_val = float("inf")
    last_improve = time.time()

    for t in range(1, max_iters + 1):
        if time_deadline is not None and time.time() > time_deadline:
            break

        # soft‑max → simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # convolution and smooth‑max weights
        conv = _full_convolution(a, fft_len)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w = w / w_sum

        # gradient of surrogate w.r.t. a
        grad_a = 2.0 * _cross_correlation(w, a, fft_len_corr)

        # back‑prop through soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # temperature schedule
        lam = min(lam * lam_factor, 1e6)

        # exact objective (sum(a) = 1 by construction)
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()

        # stagnation handling – small random “shake”
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

    return best_a, best_val


def _subgradient_optimize(a0: np.ndarray, n: int, fft_len: int,
                          fft_len_corr: int, max_iters: int = 2000,
                          eta: float = 0.6, time_deadline: float = None) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient descent on the true max‑convolution.
    """
    a = a0.copy()
    a = np.maximum(a, 0.0)
    a = a / a.sum()
    best_a = a.copy()
    conv = _full_convolution(a, fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    last_improve = time.time()

    for it in range(max_iters):
        if time_deadline is not None and time.time() > time_deadline:
            break

        conv = _full_convolution(a, fft_len)
        max_c = conv.max()
        # active set (indices where the max is attained)
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, fft_len_corr)

        # exponentiated‑gradient step (keeps a ≥ 0)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        conv = _full_convolution(a, fft_len)
        val = 2.0 * n * conv.max() / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()

        if time.time() - last_improve > 4.0:
            break

    return best_a, best_val


def _simulated_annealing_optimize(a0: np.ndarray, n: int, fft_len: int,
                                  max_steps: int = 2000, init_temp: float = 0.01,
                                  final_temp: float = 1e-6, max_move: float = 0.005,
                                  time_deadline: float = None) -> tuple[np.ndarray, float]:
    """
    Tiny random‑mass‑transfer SA to escape shallow local minima.
    """
    a = a0.copy()
    a = a / a.sum()
    conv = _full_convolution(a, fft_len)
    cur_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()
    best_val = cur_val

    for step in range(max_steps):
        if time_deadline is not None and time.time() > time_deadline:
            break

        # geometric temperature schedule
        temp = init_temp * ((final_temp / init_temp) ** (step / (max_steps - 1)))

        # propose a small transfer of mass
        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1
        eps_move = max_move * np.random.rand()
        eps_move = min(eps_move, a[i])            # keep non‑negative
        a_prime = a.copy()
        a_prime[i] -= eps_move
        a_prime[j] += eps_move

        conv = _full_convolution(a_prime, fft_len)
        val = 2.0 * n * conv.max() / (a_prime.sum() ** 2)
        delta = val - cur_val

        if delta < 0 or np.exp(-delta / max(temp, 1e-12)) > np.random.rand():
            a = a_prime
            cur_val = val
            if val < best_val:
                best_val = val
                best_a = a.copy()

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative sequence that minimises evaluate_sequence().
    The function respects the supplied time budget and returns the best
    sequence found as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin for the host

    best_seq = None
    best_val = float("inf")

    # lengths to explore (larger n give lower theoretical bound)
    lengths = [1024, 2048, 4096, 8192, 16384]

    # pre‑compute FFT lengths
    fft_len_cache = {}
    fft_len_corr_cache = {}
    for n in lengths:
        conv_len = 2 * n - 1
        fft_len_cache[n] = 1 << int(np.ceil(np.log2(conv_len)))
        fft_len_corr_cache[n] = 1 << int(np.ceil(np.log2(conv_len + n - 1)))

    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")
        fft_len = fft_len_cache[n]
        fft_len_corr = fft_len_corr_cache[n]

        # -------------------------- build initial candidates --------------
        candidates = []

        # warm start from global best (exact length)
        if best_seq is not None and len(best_seq) == n:
            candidates.append(np.array(best_seq, dtype=np.float64))

        # warm start from global best interpolated to current length
        if best_seq is not None and len(best_seq) != n:
            candidates.append(_interpolate_sequence(np.array(best_seq, dtype=np.float64), n))

        # historic solution if the runner provided one
        hseq = globals().get("height_sequence_1")
        if hseq is not None:
            h = np.asarray(hseq, dtype=np.float64)
            if h.shape[0] != n:
                h = _interpolate_sequence(h, n)
            h = np.maximum(h, 0.0)
            h = h / max(h.sum(), 1e-12)
            candidates.append(h)

        # pure random Dirichlet
        candidates.append(np.random.dirichlet(np.ones(n)))

        # edge‑concentrated power‑law shapes
        idx = np.arange(n, dtype=np.float64) + 1.0
        prod = idx * (n - idx + 1.0)
        for alpha in np.linspace(-2.0, -0.5, 5):
            shape = np.where(prod > 0, np.power(prod, alpha), 0.0)
            if shape.sum() > 0:
                shape = shape / shape.sum()
                candidates.append(shape)

        # two‑Gaussian edges
        i_arr = np.arange(n, dtype=np.float64)
        for gamma in [0.1, 0.3, 0.7, 1.5]:
            shape = np.exp(-gamma * (i_arr ** 2)) + np.exp(-gamma * ((n - 1 - i_arr) ** 2))
            shape = shape / shape.sum()
            candidates.append(shape)

        # mixture of best (if any) and a fresh random draw
        if best_seq is not None:
            base = np.array(best_seq, dtype=np.float64)
            if base.shape[0] != n:
                base = _interpolate_sequence(base, n)
            mix = 0.5 * base + 0.5 * np.random.dirichlet(np.ones(n))
            mix = mix / mix.sum()
            candidates.append(mix)

        # limit the candidate list to keep the run inside the budget
        if len(candidates) > 8:
            candidates = candidates[:8]

        # -------------------------- optimise each candidate -------------
        for cand_id, a0 in enumerate(candidates):
            if time.time() > deadline:
                break

            # normalise safely
            a0 = np.maximum(a0, 0.0)
            s = a0.sum()
            if s < 1e-12:
                a0 = np.random.dirichlet(np.ones(n))
                s = a0.sum()
            a0 = a0 / s

            # --- Adam phase (smooth‑max) ---
            remaining = max(0.0, deadline - time.time())
            # a few ten‑thousands of iterations are cheap; we cap it dynamically
            max_iters = int(min(25000, max(5000, remaining * 30)))
            a_adam, val_adam = _adam_optimize(
                a0, n, fft_len, fft_len_corr,
                max_iters=max_iters,
                lr=0.05,
                lam_start=0.5,
                lam_factor=1.02,
                time_deadline=deadline,
            )
            if val_adam < best_val:
                best_val = val_adam
                best_seq = a_adam.copy()
                print(f"[n={n} cand={cand_id} Adam] new best {val_adam:.6f}")

            # --- Sub‑gradient phase (exact max) ---
            a_sub, val_sub = _subgradient_optimize(
                a_adam, n, fft_len, fft_len_corr,
                max_iters=2000,
                eta=0.6,
                time_deadline=deadline,
            )
            if val_sub < best_val:
                best_val = val_sub
                best_seq = a_sub.copy()
                print(f"[n={n} cand={cand_id} Subgrad] new best {val_sub:.6f}")

            # --- Simulated‑annealing fine‑tuning ---
            a_sa, val_sa = _simulated_annealing_optimize(
                a_sub, n, fft_len,
                max_steps=2000,
                init_temp=0.01,
                final_temp=1e-6,
                max_move=0.005,
                time_deadline=deadline,
            )
            if val_sa < best_val:
                best_val = val_sa
                best_seq = a_sa.copy()
                print(f"[n={n} cand={cand_id} SA] new best {val_sa:.6f}")

    # ------------------------------------------------------------------
    # fallback in the improbable case that nothing was found
    if best_seq is None:
        fallback_n = 2048
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)
        best_val = _objective(best_seq, fallback_n, fft_len_cache[fallback_n])

    print(f"\nBest value found: {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
