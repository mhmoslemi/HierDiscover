"""
Propose a non‑negative step‑function that (approximately) minimises

        F(a) = 2·n·max(a*a) / (sum(a)²)

where a*a is the linear convolution of a with itself.  

Algorithm
---------
* Work on several lengths (1024 … 16384).  For each length we spend a
  fraction of the total time budget on two phases.
* **Phase 1 – fast parametric search**  
  – generate candidates from several simple families:
        • uniform,
        • Dirichlet draws with various concentrations,
        • two‑parameter power‑law shapes  a[i] ∝ (i+1)^α (n‑i)^β,
        • symmetric exponential edge shapes  a[i] ∝ e^{-c i}+e^{-c (n‑1‑i)},
        • random mixtures of the above.
  – evaluate the exact objective by an FFT‑based convolution
    (O(n log n)) and keep the best candidate for the current length.
* **Phase 2 – local refinement**  
  – initialise Adam on a smooth‑max surrogate of the max‑convolution.
    The surrogate weight vector is w_i ∝ exp(λ·conv_i) with λ
    increasing geometrically.
  – the iterate lives on the probability simplex via the
    soft‑max (log‑its) parametrisation.
  – after the Adam stage, a few exponentiated‑gradient sub‑gradient
    steps are performed on the true max‑convolution.
* The global best sequence seen (across all lengths) is returned.
* Time‑budget handling guarantees termination well before the 1000 s
  limit.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT helpers -----------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Small helper – next power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.size - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) equal to
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    n = a.size
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]  # length n


def _objective(a: np.ndarray, n: int, fft_len_conv: int) -> float:
    """Exact objective 2·n·max(conv) / (sum(a)²); a is assumed non‑negative."""
    conv = _full_convolution(a, fft_len_conv)
    max_c = conv.max()
    s = a.sum()
    return 2.0 * n * max_c / (s * s)


# ----------------------------------------------------------------------
# Phase 1 – fast parametric search --------------------------------------
def _parametric_search(n: int, time_budget: float, fft_len_conv: int) -> tuple:
    """
    Generate many cheap candidates, keep the best.
    Returns (best_vector, best_value).  If time runs out returns the best
    found so far (or None, +inf).
    """
    start = time.time()
    best_seq = None
    best_val = float("inf")

    # uniform baseline – cheap to evaluate
    a0 = np.full(n, 1.0 / n, dtype=np.float64)
    val0 = _objective(a0, n, fft_len_conv)
    best_seq, best_val = a0, val0
    # print baseline
    print(f"[n={n}] uniform baseline {val0:.6f}")

    # pre‑allocate index array for power‑law shapes
    idx = np.arange(1, n + 1, dtype=np.float64)          # i+1
    rev_idx = np.arange(n, 0, -1, dtype=np.float64)    # n-i+1

    # loop until budget exhausted
    while time.time() - start < time_budget:
        r = np.random.rand()

        # 1) Dirichlet draw -------------------------------------------------
        if r < 0.20:
            conc = np.random.choice([0.2, 0.5, 1.0, 2.0, 5.0])
            a = np.random.dirichlet(np.full(n, conc))

        # 2) Two‑parameter power‑law shape ----------------------------------
        elif r < 0.40:
            alpha = np.random.uniform(-5.0, -0.05)
            beta = np.random.uniform(-5.0, -0.05)
            # log‑space to avoid overflow
            log_shape = alpha * np.log(idx) + beta * np.log(rev_idx)
            log_shape -= np.max(log_shape)          # stabilise
            shape = np.exp(log_shape)
            if shape.sum() == 0:
                continue
            a = shape / shape.sum()

        # 3) Symmetric exponential edge shape -------------------------------
        elif r < 0.60:
            c = np.random.uniform(0.01, 6.0)
            i = np.arange(n, dtype=np.float64)
            shape = np.exp(-c * i) + np.exp(-c * (n - 1 - i))
            a = shape / shape.sum()

        # 4) Mixture of two random Dirichlet draws -------------------------
        elif r < 0.80:
            w = np.random.rand()
            a1 = np.random.dirichlet(np.ones(n))
            a2 = np.random.dirichlet(np.ones(n))
            a = w * a1 + (1.0 - w) * a2
            a = a / a.sum()

        # 5) Slight random perturbation of the current best -----------------
        else:
            if best_seq is None:
                a = np.random.dirichlet(np.ones(n))
            else:
                perturb = (np.random.rand(n) - 0.5) * 1e-2
                logit = np.log(best_seq + 1e-15) + perturb
                exp_log = np.exp(logit - np.max(logit))
                a = exp_log / exp_log.sum()

        # evaluate
        val = _objective(a, n, fft_len_conv)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[n={n}] new parametric best {val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Phase 2 – Adam + sub‑gradient refinement -------------------------------
def _refine_sequence(a0: np.ndarray,
                    n: int,
                    time_budget: float,
                    fft_len_conv: int,
                    fft_len_corr: int) -> tuple:
    """
    Starting from a0 (already normalised), run Adam on a smooth‑max
    surrogate followed by a few exponentiated‑gradient sub‑gradient steps.
    Returns (refined_sequence, refined_value) – the best seen inside
    the allocated time budget.
    """
    if a0 is None:
        return None, float("inf")

    # initialise logits for the soft‑max parametrisation
    logits = np.log(a0 + 1e-15)

    # Adam hyper‑parameters
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.5
    lam_factor = 1.02

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    best_seq = None
    best_val = float("inf")
    start = time.time()
    it = 0

    # ---- Adam on smooth‑max ------------------------------------------------
    while time.time() - start < time_budget * 0.65:
        it += 1
        # soft‑max → probability vector on the simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # convolution
        conv = _full_convolution(a, fft_len_conv)
        max_c = conv.max()

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        w = w / w_sum if w_sum != 0 else np.full_like(w, 1.0 / w.size)

        # gradient of the surrogate (2·(w ★ a))
        grad_a = 2.0 * _cross_correlation(w, a, fft_len_corr)

        # back‑prop through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # increase the temperature λ
        lam = min(lam * lam_factor, 1e6)

        # exact objective (no smoothing)
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            # occasional progress report
            if it % 500 == 0:
                print(f"[n={n}] Adam iter={it:05d}  val={val:.6f}")

    # ---- Sub‑gradient phase on the true max -------------------------------
    # Use whatever time is left
    sub_start = time.time()
    a = best_seq if best_seq is not None else a0
    a = a / a.sum()
    eta = 0.6                       # step size for exponentiated update
    sub_it = 0
    sub_best_val = best_val
    sub_best_seq = best_seq if best_seq is not None else a0

    while time.time() - sub_start < time_budget * 0.35:
        sub_it += 1
        conv = _full_convolution(a, fft_len_conv)
        max_c = conv.max()
        # active set (indices attaining the max, within a tiny tolerance)
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if not active.any():
            w = np.zeros_like(conv)
        else:
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()

        # sub‑gradient of the exact max
        grad_a = 2.0 * _cross_correlation(w, a, fft_len_corr)

        # exponentiated‑gradient step (stays in the simplex)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:                     # avoid collapse to zero
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < sub_best_val:
            sub_best_val = val
            sub_best_seq = a.copy()
            # occasional progress report
            if sub_it % 200 == 0:
                print(f"[n={n}] Sub‑grad iter={sub_it:04d}  val={val:.6f}")

    # return the best sequence seen during the whole refinement
    final_seq = sub_best_seq if sub_best_seq is not None else best_seq
    final_val = sub_best_val if sub_best_seq is not None else best_val
    return final_seq, final_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with low evaluation score.
    The routine respects the given time budget and returns the best
    sequence found as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # lengths that we will try – larger lengths give more freedom
    lengths = [16384, 8192, 4096, 2048, 1024]

    best_seq = None
    best_val = float("inf")

    # pre‑compute FFT lengths for all n
    fft_len_conv_cache = {}
    fft_len_corr_cache = {}
    for n in lengths:
        fft_len_conv_cache[n] = _next_pow2(2 * n - 1)
        fft_len_corr_cache[n] = _next_pow2(3 * n - 2)   # 2n-1 + (n-1)

    # iterate over lengths, allocating time proportionally to n
    total_n = sum(lengths)
    for idx, n in enumerate(lengths):
        now = time.time()
        if now >= deadline:
            break

        remaining = deadline - now
        # allocate a share proportional to the size of n relative to the
        # total size of the *remaining* lengths
        remaining_n = sum(lengths[idx:])
        alloc = remaining * (n / remaining_n)
        # never exceed a reasonable per‑length cap
        alloc = min(alloc, 300.0)

        print(f"\n=== Optimising length n={n} (time budget ≈{alloc:.1f}s) ===")
        fft_len_conv = fft_len_conv_cache[n]
        fft_len_corr = fft_len_corr_cache[n]

        # --------------------------------------------------------------
        # Phase 1 – fast parametric search
        param_time = alloc * 0.60
        cand_seq, cand_val = _parametric_search(n, param_time, fft_len_conv)
        if cand_seq is not None and cand_val < best_val:
            best_seq, best_val = cand_seq.copy(), cand_val
            print(f"[n={n}] after parametric search best {cand_val:.6f}")

        # --------------------------------------------------------------
        # Phase 2 – local refinement (Adam + sub‑gradient)
        refine_time = alloc - param_time
        refined_seq, refined_val = _refine_sequence(cand_seq,
                                                    n,
                                                    refine_time,
                                                    fft_len_conv,
                                                    fft_len_corr)
        if refined_seq is not None and refined_val < best_val:
            best_seq, best_val = refined_seq.copy(), refined_val
            print(f"[n={n}] after refinement best {refined_val:.6f}")

        # --------------------------------------------------------------
        # Try a few tiny random “shakes” around the current global best
        # (only if we still have some time left)
        shake_budget = deadline - time.time()
        if shake_budget > 5.0 and best_seq is not None and len(best_seq) == n:
            shakes = int(min(5, shake_budget // 1.0))
            for s in range(shakes):
                # tiny random perturbation in log‑space
                pert = (np.random.rand(n) - 0.5) * 5e-3
                logits = np.log(best_seq + 1e-15) + pert
                exp_log = np.exp(logits - logits.max())
                a_shake = exp_log / exp_log.sum()
                # short refinement (just a few Adam steps)
                short_seq, short_val = _refine_sequence(a_shake,
                                                        n,
                                                        time_budget=1.0,
                                                        fft_len_conv=fft_len_conv,
                                                        fft_len_corr=fft_len_corr)
                if short_seq is not None and short_val < best_val:
                    best_seq, best_val = short_seq.copy(), short_val
                    print(f"[n={n}] shake {s+1}/{shakes} improved to {short_val:.6f}")

        print(f"[n={n}] best value so far {best_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform vector if nothing succeeded (very unlikely)
    if best_seq is None:
        fallback_n = 2048
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)

    # return a plain Python list as required by the external evaluator
    return best_seq.tolist()
