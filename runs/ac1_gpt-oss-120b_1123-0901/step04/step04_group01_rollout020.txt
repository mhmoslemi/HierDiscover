def project_simplex(v):
    """Project onto probability simplex (nonnegative, sum=1)."""
    # v is numpy array
    n = v.shape[0]
    # sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    # find rho = max { j | u_j - cssv_j / j > 0 }
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
Search for a non‑negative step‑function (probability vector) that minimises

    F(a) = 2·n·max(a*a) / (sum(a)²)

The algorithm combines several ideas that improve on the previous Adam‑softmax
implementation:

*   The “max” is replaced by a smooth soft‑max (log‑sum‑exp) with a temperature
    parameter α that is slowly increased.  This gives a well‑behaved gradient
    everywhere while still driving the true maximum down.

*   Probabilities are kept on the simplex by an explicit Euclidean projection
    (no soft‑max parametrisation).  This avoids the saturation problems of the
    exponentials for very small or very large logits.

*   Adam (first‑ and second‑moment adaptation) is used for the updates; the
    learning‑rate is decayed and α is increased during the run.

*   Several independent restarts are performed for a range of problem sizes.
    Warm‑starts are taken from the best solution found so far or from the
    supplied ``height_sequence_1`` variable.

*   When progress stalls a small random “shake’’ is added to the current vector.

*   After the main optimisation a cheap greedy pairwise refinement is applied
    (move a tiny amount of mass from one index to another).

The whole routine respects the 1000 s budget, prints intermediate progress
and finally returns the best list of heights discovered.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level – no closures)
# ----------------------------------------------------------------------


def full_convolution(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """Linear convolution a*a using an FFT of length ``fft_len``."""
    n = a.shape[0]
    if fft_len is None:
        fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


def cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) equal to
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i = 0 … n‑1.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        {x | x_i ≥ 0, Σ_i x_i = 1}
    (Duchi et al. 2008)
    """
    if v.ndim != 1:
        raise ValueError("input must be a 1‑D array")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    # find the largest index where the condition holds
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return the uniform simplex point
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Main entry point required by the framework
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs) -> list:
    """
    Run a time‑bounded optimisation and return the best sequence found.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Time budget in seconds (the function stops a little earlier to
        finish cleanly).

    Returns
    -------
    list[float]
        The best probability vector discovered (the sum is 1, all entries
        are non‑negative).  The global evaluator will use the same vector.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # tiny safety margin

    best_seq = None      # numpy array – current overall best
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Problem sizes – larger n gives a lower theoretical bound (≈1).
    # The previous best was obtained for n = 1024; we try larger sizes as
    # well, keeping a modest number of restarts for each.
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 4096, 8192, 16384]

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")

        # Pre‑compute FFT lengths that will be reused
        fft_len_conv = 1 << int(np.ceil(np.log2(2 * n - 1)))
        fft_len_corr = 1 << int(np.ceil(np.log2(3 * n)))  # ≥ (2n‑1)+n‑1

        # Number of independent restarts for this length
        if n <= 2048:
            restarts = 7
        elif n <= 8192:
            restarts = 4
        else:
            restarts = 2

        for r in range(restarts):
            if time.time() > deadline:
                break

            # --------------------------------------------------------------
            # Initialise a probability vector
            # --------------------------------------------------------------
            if best_seq is not None and len(best_seq) == n:
                a = best_seq.copy()
                a += np.random.normal(scale=0.02, size=n)   # tiny warm‑start noise
                a = np.maximum(a, 0.0)
                a /= a.sum()
                init_desc = "warm start"
            elif "height_sequence_1" in globals():
                base = np.array(height_sequence_1, dtype=np.float64)
                if base.shape[0] > n:
                    base = base[:n]
                else:
                    base = np.pad(base, (0, n - base.shape[0]))
                a = np.maximum(base, 1e-12)
                a /= a.sum()
                init_desc = "height_sequence_1"
            else:
                a = np.random.dirichlet(np.ones(n))
                init_desc = "random Dirichlet"

            print(f"[n={n} restart={r}] init: {init_desc}")

            # --------------------------------------------------------------
            # Adam optimiser state
            # --------------------------------------------------------------
            m = np.zeros(n, dtype=np.float64)   # first moment
            v = np.zeros(n, dtype=np.float64)   # second moment
            beta1, beta2, eps = 0.9, 0.999, 1e-8

            lr = 0.5                # initial learning rate
            lr_decay = 0.7
            decay_every = 3000      # iterations

            # soft‑max temperature (α) – start smooth, end sharp
            alpha = 5.0
            alpha_mul = 1.025
            max_alpha = 200.0

            # stagnation handling
            last_improve = time.time()
            max_iters = 50000 if n <= 4096 else 30000 if n <= 8192 else 20000
            t = 0

            while time.time() < deadline and t < max_iters:
                # ---------- convolution and exact objective ----------
                conv = full_convolution(a, fft_len=fft_len_conv)
                max_c = conv.max()
                # (a.sum() is ≈1 after projection)
                val = 2.0 * n * max_c / (a.sum() ** 2)

                if val < best_val - 1e-12:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} r={r} it={t:05d}] NEW BEST {val:.6f}")

                # ---------- smooth max (soft‑max) ----------
                # shift by max_c to avoid overflow, then normalise
                w_raw = np.exp(alpha * (conv - max_c))
                w_sum = w_raw.sum()
                if w_sum == 0.0:
                    w = np.full_like(conv, 1.0 / conv.size)
                else:
                    w = w_raw / w_sum

                # ---------- gradient of the smooth objective ----------
                grad_a = 2.0 * cross_correlation(w, a, fft_len=fft_len_corr)

                # ---------- Adam update ----------
                t += 1
                m = beta1 * m + (1.0 - beta1) * grad_a
                v = beta2 * v + (1.0 - beta2) * (grad_a * grad_a)
                m_hat = m / (1.0 - beta1 ** t)
                v_hat = v / (1.0 - beta2 ** t)
                a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

                # ---------- projection onto the simplex ----------
                a = project_onto_simplex(a)

                # ---------- learning‑rate & temperature schedules ----------
                if t % decay_every == 0:
                    lr *= lr_decay
                if alpha < max_alpha:
                    alpha *= alpha_mul

                # ---------- stagnation shake ----------
                if time.time() - last_improve > 30.0:
                    a += (np.random.rand(n) - 0.5) * 2e-2
                    a = np.maximum(a, 0.0)
                    a = project_onto_simplex(a)
                    last_improve = time.time()
                    print(f"[n={n} r={r} it={t}] shake (stagnation)")

            # ------------------------------------------------------------------
            # Optional greedy local refinement (tiny pairwise moves)
            # ------------------------------------------------------------------
            if best_seq is not None and len(best_seq) == n:
                a_local = best_seq.copy()
                local_iters = 3000
                for it in range(local_iters):
                    if time.time() > deadline:
                        break
                    i, j = np.random.choice(n, 2, replace=False)
                    # move a random fraction of mass from i to j
                    eps_move = min(a_local[i], 0.01) * (0.5 + np.random.rand() * 0.5)
                    if eps_move <= 0:
                        continue
                    cand = a_local.copy()
                    cand[i] -= eps_move
                    cand[j] += eps_move
                    # evaluate exact objective quickly
                    conv_new = full_convolution(cand, fft_len=fft_len_conv)
                    val_new = 2.0 * n * conv_new.max() / (cand.sum() ** 2)
                    if val_new < best_val - 1e-12:
                        best_val = val_new
                        best_seq = cand.copy()
                        a_local = cand
                        print(f"[n={n} local] improved to {val_new:.6f}")

    # ----------------------------------------------------------------------
    # Fallback (should never be needed)
    # ----------------------------------------------------------------------
    if best_seq is None:
        n_fallback = 2000
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        best_val = 2.0  # uniform value

    print(f"\nFinal best value: {best_val:.6f} (length {len(best_seq)})")
    # Return a plain Python list as required by the outer harness
    return best_seq.tolist()
