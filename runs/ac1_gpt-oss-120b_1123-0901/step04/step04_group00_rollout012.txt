def deep_optimize(init_a, n, conv_fft_len, corr_fft_len, deadline, best_seq, best_val):

    a = init_a.copy()
    logits = np.log(a + 1e-15)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1,beta2,eps = 0.9,0.999,1e-8
    lr = 0.05
    lam = 0.5
    lam_factor = 1.02

    # track improvement time
    improve_time = time.time()
    # max iterations dynamic based on time
    iter = 0
    while time.time() < deadline and iter < 50000:
        # compute a via softmax
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # conv & max
        conv = full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        # update global best
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            improve_time = time.time()
            print(f"[n={n} Adam iter={iter}] new best {val:.6f}")

        # compute w
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w = w / w_sum

        # gradient
        grad_a = 2.0 * cross_correlation(w, a, corr_fft_len)
        # backprop through softmax
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_log
        v = beta2 * v + (1 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (iter+1))
        v_hat = v / (1.0 - beta2 ** (iter+1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # schedule lam and maybe lr decay
        lam = min(lam * lam_factor, 1e6)

        # occasional random shake if no improvement for some time
        if time.time() - improve_time > 3.0:
            # small perturbation:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            improve_time = time.time()

        iter += 1

    # Subgradient phase
    a = best_seq.copy()
    # ensure sum=1
    a = a / a.sum()
    eta = 0.6
    sub_improve = time.time()
    for sub_it in range(8000):
        if time.time() > deadline:
            break
        conv = full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            sub_improve = time.time()
            print(f"[n={n} Subgrad iter={sub_it}] new best {val:.6f}")

        # active set
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * cross_correlation(w, a, corr_fft_len)

        # exponentiated gradient
        a = a * np.exp(-eta * grad_a)
        # renormalize
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        # early stop if no improve for some time
        if time.time() - sub_improve > 5.0:
            break

    return best_seq, best_val

def generate_initial_candidates(n, best_seq):

    candidates = []

    if best_seq is not None:
        # maybe use as is
        candidates.append(best_seq.copy())
        # rotated versions
        for shift in [n//4, n//2, 3*n//4]:
            cand = np.roll(best_seq, shift)
            candidates.append(cand)

    # random
    candidates.append(np.random.dirichlet(np.ones(n)))

    # power law
    idx = np.arange(n, dtype=np.float64) + 1.0
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.5, 5):
        shape = np.where(prod > 0, prod**alpha, 0.0)
        if shape.sum() > 0:
            shape = shape / shape.sum()
            candidates.append(shape)

    # Gaussian edges
    i = np.arange(n, dtype=np.float64)
    for gamma in [0.2, 0.5, 1.0, 2.0]:
        shape = np.exp(-gamma * (i**2)) + np.exp(-gamma * ((n-1-i)**2))
        shape = shape / shape.sum()
        candidates.append(shape)

    # Sinusoidal positive
    shape = np.sin(np.pi * (np.arange(n)+0.5) / n)  # positive
    shape = shape / shape.sum()
    candidates.append(shape)

    # Uniform
    shape = np.full(n, 1.0/n)
    candidates.append(shape)

    return candidates

propose_candidate
-----------------

Search for a non‑negative step‑function (probability vector) that minimises

        F(a) = 2·n·max(a∗a) / (sum(a)²)

using a continuation method on a smooth‑max surrogate followed by a
sub‑gradient “exact‑max” refinement.  The optimisation runs on the simplex
via the entropy mirror map (soft‑max parametrisation) and uses the Adam
optimizer.  Several diverse initialisations (previous best, random Dirichlet,
power‑law edges, double‑Gaussian edges, sinusoidal shape, uniform) are tried.
The routine respects the given time budget (default 1000 s) and returns the
best sequence found as a plain Python list.

import numpy as np

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2·len(a)-1) using FFT.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) equal to
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """
    Exact objective value (sum(a) is assumed ≈ 1).
    """
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c / (a.sum() ** 2))


# ----------------------------------------------------------------------
def _deep_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Run Adam on a smooth‑max surrogate (log‑sum‑exp) followed by a
    sub‑gradient phase on the exact max.  The routine updates and returns the
    best sequence/value found (globally across all candidates).
    """
    # initialise logits for the soft‑max parametrisation
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05

    lam = 0.5                  # start temperature for smooth‑max
    lam_factor = 1.02

    # track improvement times for stagnation handling
    last_improve = time.time()
    it = 0

    while time.time() < deadline:
        # -------- soft‑max → simplex ----------
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # -------- convolution & smooth‑max weights ----------
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        # exact objective (useful for tracking)
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < global_best_val:
            global_best_val = val
            global_best_seq = a.copy()
            last_improve = time.time()
            print(f"[n={n} Adam iter={it:05d}] new best {val:.6f}")

        # smooth‑max (log‑sum‑exp) weighting
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate (chain‑rule through soft‑max)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature schedule (smooth‑max becomes true max)
        lam = min(lam * lam_factor, 1e6)

        # occasional random “shake” when stagnating
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1
        # safety break – we don’t want an endless loop if the deadline is far
        if it > 200_000:
            break

    # ------------------------------------------------------------------
    # Sub‑gradient phase on the true max (exponentiated‑gradient updates)
    a = global_best_seq if global_best_seq is not None else a0
    a = np.maximum(a, 0.0)
    a /= a.sum()

    eta = 0.6
    sub_last_improve = time.time()
    for sub_it in range(8_000):
        if time.time() >= deadline:
            break

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < global_best_val:
            global_best_val = val
            global_best_seq = a.copy()
            sub_last_improve = time.time()
            print(f"[n={n} Subgrad iter={sub_it:04d}] new best {val:.6f}")

        # active set of maximal convolution entries
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # exponentiated‑gradient step (keeps a ≥ 0)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:                     # avoid collapse to zero
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        if time.time() - sub_last_improve > 5.0:
            break

    return global_best_seq, global_best_val


# ----------------------------------------------------------------------
def _initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """
    Build a modest pool of diverse start vectors for the optimiser.
    """
    cand = []

    # warm start (previous best) and a few rotated copies
    if warm is not None:
        cand.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    # pure random Dirichlet
    cand.append(np.random.dirichlet(np.ones(n)))

    # power‑law edge‑concentrated shapes
    idx = np.arange(n, dtype=np.float64) + 1.0
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.5, 5):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    # two‑Gaussian edge shapes
    i = np.arange(n, dtype=np.float64)
    for gamma in (0.2, 0.5, 1.0, 2.0):
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        shape /= shape.sum()
        cand.append(shape)

    # simple sinusoidal shape (positive half‑wave)
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand.append(shape)

    # uniform vector (baseline)
    cand.append(np.full(n, 1.0 / n))

    return cand


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The function respects the given time budget and returns the best sequence.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Primary optimisation length (the current best known bound was at 4096)
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # Assemble initial candidates
    warm_start = globals().get("height_sequence_1")
    if warm_start is not None:
        warm_start = np.asarray(warm_start, dtype=np.float64)
        if warm_start.shape[0] > n:
            warm_start = warm_start[:n]
        elif warm_start.shape[0] < n:
            warm_start = np.pad(warm_start, (0, n - warm_start.shape[0]))
        warm_start = np.maximum(warm_start, 0.0)
        if warm_start.sum() < 0.01:
            warm_start = None
        else:
            warm_start = warm_start / warm_start.sum()
    candidates = _initial_candidates(n, warm_start)

    # ------------------------------------------------------------------
    # Optimise each candidate (more iterations for the first few)
    for cid, cand in enumerate(candidates):
        if time.time() >= deadline:
            break

        a0 = np.maximum(cand, 0.0)
        s = a0.sum()
        if s < 0.01:
            a0 = np.random.dirichlet(np.ones(n))
        else:
            a0 = a0 / s

        # the deep optimiser respects the global deadline and updates the
        # overall best sequence/value
        best_seq, best_val = _deep_optimize(
            a0, n, conv_fft_len, corr_fft_len, deadline, best_seq, best_val
        )

    # ------------------------------------------------------------------
    # If time permits, try a larger length (8192) by interpolating the best
    # found distribution.  Larger n can sometimes lower the objective further.
    if time.time() + 10 < deadline:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        if best_seq is not None:
            # simple linear interpolation to the larger grid
            x_old = np.linspace(0, 1, n, endpoint=False)
            x_new = np.linspace(0, 1, n_big, endpoint=False)
            a_big = np.interp(x_new, x_old, best_seq)
            a_big = np.maximum(a_big, 0.0)
            a_big /= a_big.sum()
        else:
            a_big = np.random.dirichlet(np.ones(n_big))

        best_seq, best_val = _deep_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            best_seq,
            best_val,
        )

    # ------------------------------------------------------------------
    # Fallback – if nothing succeeded produce a uniform vector.
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Clip to the evaluator’s allowed range (non‑negative, ≤ 1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    return best_seq.tolist()
