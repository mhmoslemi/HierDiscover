```python
def _full_convolution(a, fft_len=None):
    n = a.shape[0]
    if fft_len is None:
        fft_len = 1 << int(np.ceil(np.log2(2*n - 1)))
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[:2*n-1]
    return conv

def _cross_correlation(w, a, fft_len):
    # Compute (w ★ a) length n as per earlier.
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start+n]
    
def _subgradient_optimize(a0, n, max_iters, deadline, fft_len_conv, fft_len_corr):
    import numpy as np, time
    a = a0.copy()
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        conv = _full_convolution(a, fft_len=fft_len_conv)
        max_c = conv.max()
        # active set
        active = np.isclose(conv, max_c, atol=1e-12)
        if not active.any():
            w = np.zeros_like(conv)
        else:
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        # gradient
        grad = 2.0 * _cross_correlation(w, a, fft_len=fft_len_corr)
        # exponentiated gradient step
        eta = 0.6  # step size
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s <= 0:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s
    return a
```

```
def _adam_optimize(p0, n, max_iters, deadline, fft_len_conv, fft_len_corr):
    a = p0.copy()
    # logits:
    logits = np.log(a + 1e-15)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.5
    lam_factor = 1.02
    for t in range(1, max_iters+1):
        if time.time() >= deadline:
            break
        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()
        conv = _full_convolution(a, fft_len=fft_len_conv)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum
        grad_a = 2.0 * _cross_correlation(w, a, fft_len=fft_len_corr)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)
        lam = min(lam * lam_factor, 1e6)
    # final projection to simplex
    mx = logits.max()
    exp_x = np.exp(logits - mx)
    a_opt = exp_x / exp_x.sum()
    return a_opt
```

import time
import numpy as np

def _full_convolution(a, fft_len=None):
    n = a.shape[0]
    if fft_len is None:
        fft_len = 1 << int(np.ceil(np.log2(2*n - 1)))
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[:2*n - 1]
    return conv

def _cross_correlation(w, a, fft_len):
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]

def _objective_fast(a, n, fft_len):
    conv = _full_convolution(a, fft_len=fft_len)
    return 2.0 * n * conv.max() / (a.sum() ** 2)

def _adam_optimize(p0, n, max_iters, deadline, fft_len_conv, fft_len_corr):
    a = p0.copy()
    logits = np.log(a + 1e-15)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.5
    lam_factor = 1.02
    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break
        # softmax to simplex
        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()
        # convolution and smooth‑max weights
        conv = _full_convolution(a, fft_len=fft_len_conv)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w = w / w_sum
        # gradient of surrogate
        grad_a = 2.0 * _cross_correlation(w, a, fft_len=fft_len_corr)
        # back‑prop through soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)
        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)
        # increase temperature
        lam = min(lam * lam_factor, 1e6)
    # final projection back to simplex
    mx = logits.max()
    exp_x = np.exp(logits - mx)
    a_opt = exp_x / exp_x.sum()
    return a_opt

    length (default 4096) but can fall back to other lengths if time permits.
    It maintains a population, performs crossover, log‑space mutation and
    occasional Adam‑based local refinement.
    """
    import time, numpy as np
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # slight safety margin

    # You may change this length; 4096 was the best found so far.
    lengths = [4096, 8192]
    best_seq = None
    best_val = float("inf")

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"\n=== Searching length {n} ===")
        # Pre‑compute FFT sizes needed for convolutions.
        conv_len = 2 * n - 1
        fft_len_conv = 1 << int(np.ceil(np.log2(conv_len)))
        fft_len_corr = 1 << int(np.ceil(np.log2(conv_len + n - 1)))

        # ---------- initialise population ----------
        pop_size = 80
        elite_frac = 0.2
        elite_size = max(2, int(pop_size * elite_frac))
        population = []

        # uniform distribution – baseline
        population.append(np.full(n, 1.0 / n, dtype=np.float64))

        # warm‑start from a previously‑found global best (if compatible)
        if best_seq is not None and len(best_seq) == n:

        # historic solution if present
        hseq = globals().get("height_sequence_1")
        if hseq is not None:

            if cand.shape[0] > n:

            else:

            cand = np.maximum(cand, 0.0)
            if cand.sum() < 1e-12:
                cand = np.random.dirichlet(np.ones(n))
            else:
                cand = cand / cand.sum()
            population.append(cand)

        for _ in range(3):
            population.append(np.random.dirichlet(np.ones(n)))

        # edge‑concentrated power‑law shapes

        for alpha in np.linspace(-2.0, -0.2, 5):
            shape = np.power(prod, alpha, where=prod > 0, out=np.zeros_like(prod))
            if shape.sum() > 0:
                shape = shape / shape.sum()
                population.append(shape)

        i = np.arange(n, dtype=np.float64)

        generation = 0
        sigma = 0.2                     # mutation magnitude in log‑space
        last_improve_time = time.time()

        while time.time() < deadline:
            generation += 1

            # sort by score (lower is better)
            order = np.argsort(scores)
            population = [population[i] for i in order]
            scores = [scores[i] for i in order]
            elite = population[:elite_size]

            # decay mutation strength slowly
            if generation % 30 == 0:
                sigma = max(0.02, sigma * 0.9)

            # produce offspring
            offspring = []
            while len(offspring) < pop_size - elite_size:
                p1, p2 = elite[np.random.randint(elite_size)], elite[np.random.randint(elite_size)]
                alpha = np.random.rand()
                child = alpha * p1 + (1.0 - alpha) * p2
                # log‑space mutation
                log_child = np.log(child + 1e-15)
                log_child += sigma * np.random.randn(n)

                else:
                    child = child / s
                offspring.append(child)

            population = elite + offspring
            scores = [_objective_fast(p, n, fft_len_conv) for p in population]

            for p, s in zip(population, scores):

                    best_val = s
                    best_seq = p.copy()
                    last_improve_time = time.time()
                    print(f"[n={n} gen={generation}] new best {s:.6f}")

                for _ in range(5):

                    population[-1] = new_rand  # replace worst entry
                    scores[-1] = _objective_fast(new_rand, n, fft_len_conv)
                last_improve_time = time.time()

            if best_val < 1.503:
                print(f"Target achieved for n={n} with value {best_val:.6f}")
                break

            remaining = deadline - time.time()
            if remaining > 5.0:
                refined = _adam_optimize(best_seq, n,

                                         fft_len_conv=fft_len_conv,
                                         fft_len_corr=fft_len_corr)

                    best_seq = refined

        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)

    # Convert to plain Python list as required by the outer evaluator
    return best_seq.tolist()

# ----------------------------------------------------------------------
# Core linear‑algebra helpers (FFT based, fast for the lengths we use)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray, fft_len: int | None = None) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) using an FFT."""
    n = a.shape[0]
    if fft_len is None:
        fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the surrogate gradient.
    Returns a vector of length len(a) equal to
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def _objective_fast(a: np.ndarray, n: int, fft_len_conv: int) -> float:
    """Exact objective 2·n·max(a*a)/(sum(a)²) – fast because sum(a)=1."""
    conv = _full_convolution(a, fft_len=fft_len_conv)
    return 2.0 * n * conv.max() / (a.sum() ** 2)


# ----------------------------------------------------------------------
# Adam optimiser on the smooth‑max surrogate (used for a final polish)
# ----------------------------------------------------------------------
def _adam_optimize(p0: np.ndarray,
                   n: int,
                   max_iters: int,
                   deadline: float,
                   fft_len_conv: int,
                   fft_len_corr: int) -> np.ndarray:
    """Runs a few Adam steps on a soft‑max surrogate of max(conv)."""
    a = p0.copy()
    logits = np.log(a + 1e-15)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.5
    lam_factor = 1.02

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # soft‑max → simplex
        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()

        # convolution and smooth‑max weights
        conv = _full_convolution(a, fft_len=fft_len_conv)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate
        grad_a = 2.0 * _cross_correlation(w, a, fft_len=fft_len_corr)

        # back‑prop through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)

    mx = logits.max()
    exp_x = np.exp(logits - mx)
    a_opt = exp_x / exp_x.sum()
    return a_opt
