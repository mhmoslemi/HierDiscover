def lp_step(a, M, max_iter=1000, solver_time_limit=30):
    # solve LP: max sum(b) s.t. conv(a,b) <= M, b>=0.
    n = len(a)
    b = cp.Variable(n, nonneg=True)
    constraints = []
    # Build constraints for each lag
    for k in range(2*n - 1):
        # compute convolution term linear in b
        # indices where a_i multiplies b_{k-i}
        start_i = max(0, k - (n - 1))
        end_i = min(k, n - 1)
        if start_i > end_i:
            continue
        # the linear expression = sum a[i] * b[k-i]
        expr = cp.sum(cp.multiply(a[start_i:end_i+1], b[k - np.arange(start_i, end_i+1)]))
        constraints.append(expr <= M * (1 + 1e-12))  # tolerance
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=max_iter, verbose=False)
    if b.value is None:
        return None
    g = np.maximum(b.value, 0.0)
    return g

"""
Propose a candidate step‑function (non‑negative sequence) that minimises

        F(a) = 2·n·max(a*a) / (sum(a)²)

The algorithm combines three ideas:

1.  Stochastic sub‑gradient optimisation on the simplex via a soft‑max
    parametrisation (Adam optimiser).  This is the same convex‑objective
    optimiser used previously but with a larger iteration budget,
    adaptive learning‑rate decay and a more aggressive “shake”
    when stagnation is detected.

2.  A linear‑programming refinement (when the dimension is modest, n ≤ 2048).
    For a fixed sequence a we solve

        maximise   Σ b_j
        subject to   (a * b)_k ≤ M   for all lags k,
                     b_j ≥ 0,

    where M = max(a*a).  The linear program finds a direction b that
    increases the L¹‑mass without increasing the maximal convolution.
    Mixing a and the normalised optimiser b (convex combination) strictly
    reduces the objective.

3.  A cheap local greedy “mass‑transfer” step that moves a tiny amount of
    probability from the index contributing most to the active lag set
    to a random index.  This helps to escape shallow plateaux after the
    LP step.

All stages respect the 1000 s time budget, switch to larger problem sizes
only when time permits and fall back to a uniform vector if nothing is
found.  The function returns the best sequence (as a plain Python list)
found within the allowed time.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Try to import cvxpy – if unavailable we simply skip the LP refinement.
try:
    import cvxpy as cp
    _CVXPY_AVAILABLE = True
except Exception:                     # pragma: no cover
    _CVXPY_AVAILABLE = False

# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """Linear convolution a*a using FFT (real‑to‑real)."""
    n = a.shape[0]
    if fft_len is None:
        fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the sub‑gradient.
    Returns a vector of length len(a) equal to
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(a*a) / (sum(a)²).  Sum is assumed ≠0."""
    n = a.shape[0]
    conv = _full_convolution(a)
    return 2.0 * n * conv.max() / (np.sum(a) ** 2)


# ----------------------------------------------------------------------
def _lp_improvement(a: np.ndarray, max_val: float) -> np.ndarray | None:
    """
    Solve the linear programme
        maximise Σ b_j
        subject to  (a*b)_k ≤ max_val  (∀k),  b ≥ 0.
    Returns the optimiser b (not normalised) or None if the solve fails.
    The LP is built only for moderate dimensions (n ≤ 2048) because the
    constraint matrix grows as O(n²).
    """
    if not _CVXPY_AVAILABLE:
        return None

    n = a.shape[0]
    if n > 2048:
        return None

    b = cp.Variable(n, nonneg=True)
    constraints = []

    # each lag k gives a linear inequality Σ_i a_i · b_{k-i} ≤ max_val
    for k in range(2 * n - 1):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        if i_start > i_end:
            continue
        idx_i = np.arange(i_start, i_end + 1)
        idx_b = k - idx_i
        coeff = a[idx_i]
        expr = coeff @ b[idx_b]
        constraints.append(expr <= max_val * (1.0 + 1e-12))

    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=400, verbose=False)
    except Exception:                     # pragma: no cover
        return None

    if b.value is None:
        return None
    return np.maximum(np.asarray(b.value, dtype=np.float64), 0.0)


def _mix_and_evaluate(a: np.ndarray, g: np.ndarray,
                      mix_factors: tuple[float, ...]) -> tuple[np.ndarray, float]:
    """
    Mix a (already normalised to sum=1) with a normalised g
    using several mixing coefficients and keep the best improvement.
    Returns the new sequence and its objective value.
    """
    sum_g = np.sum(g)
    if sum_g <= 0.0:
        return a, _objective(a)

    g_norm = g / sum_g
    best_seq, best_val = a, _objective(a)
    for alpha in mix_factors:
        cand = (1.0 - alpha) * a + alpha * g_norm
        val = _objective(cand)
        if val < best_val:
            best_seq, best_val = cand, val
    return best_seq, best_val


def _local_transfer(a: np.ndarray, n_iters: int = 2000) -> np.ndarray:
    """
    Very cheap greedy refinement: repeatedly move a tiny amount of mass
    from the index that currently contributes the most to the active
    lag set to a random other index.
    """
    n = a.shape[0]
    for _ in range(n_iters):
        conv = _full_convolution(a)
        max_conv = conv.max()
        # indices where the max is attained (active set)
        active = np.where(conv >= max_conv * (1 - 1e-12))[0]
        # choose a random active lag and a random source index that belongs to it
        lag = np.random.choice(active)
        i_start = max(0, lag - (n - 1))
        i_end = min(lag, n - 1)
        src_candidates = np.arange(i_start, i_end + 1)
        src = np.random.choice(src_candidates)
        if a[src] <= 0.0:
            continue
        dst = np.random.randint(0, n)
        if dst == src:
            continue
        # transfer a tiny fraction (10‑30 % of the source mass)
        delta = a[src] * (0.1 + 0.2 * np.random.rand())
        a[src] -= delta
        a[dst] += delta
    return a


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs) -> list:
    """
    Search for a non‑negative step‑function that minimises the evaluation
    function.  The routine runs several Adam‑based sub‑gradient runs,
    optional LP‑based refinement (for n ≤ 2048) and a cheap local
    greedy transfer.  It respects the time budget and returns the best
    sequence found as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Problem sizes – larger n gives a lower theoretical bound but needs
    # more work.  We start with 1024 (the size that produced the current
    # best) and enlarge only if time permits.
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 4096, 8192, 16384]

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")

        # ---- Warm‑start vector -------------------------------------------------
        if "height_sequence_1" in globals():
            base = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if base.shape[0] >= n:
                start_vec = base[:n]
            else:
                start_vec = np.pad(base, (0, n - base.shape[0]))
            start_vec = np.maximum(start_vec, 0.0)
            if start_vec.sum() < 0.01:
                start_vec = np.random.dirichlet(np.ones(n))
            else:
                start_vec /= start_vec.sum()
        else:
            start_vec = np.random.dirichlet(np.ones(n))

        # ---- Optimisation parameters -------------------------------------------
        restarts = 4 if n <= 2048 else 2
        max_iters = 50000 if n <= 1024 else 30000 if n <= 2048 else 20000

        # Pre‑compute FFT lengths (re‑used many times)
        fft_len_conv = 1 << int(np.ceil(np.log2(2 * n - 1)))
        fft_len_corr = 1 << int(np.ceil(np.log2(3 * n)))

        for r in range(restarts):
            if time.time() > deadline:
                break

            # Initialise logits (soft‑max parameters)
            if r == 0:
                a0 = start_vec.copy()
                a0 = np.maximum(a0, 0.0)
                a0 /= a0.sum()
                x = np.log(a0 + 1e-15)
                print(f"[n={n} restart={r}] warm start")
            else:
                a0 = np.random.dirichlet(np.ones(n))
                x = np.log(a0 + 1e-15)
                print(f"[n={n} restart={r}] random start")

            # Adam state
            m = np.zeros(n)
            v = np.zeros(n)
            beta1, beta2, eps = 0.9, 0.999, 1e-8
            lr = 0.07
            decay_every = 4000

            best_local = None
            best_local_val = float("inf")
            last_improve = time.time()
            t = 0

            while time.time() < deadline and t < max_iters:
                # ----- soft‑max (keeps Σa = 1) -----
                mx = np.max(x)
                exp_x = np.exp(x - mx)
                a = exp_x / exp_x.sum()

                # ----- objective and sub‑gradient -----
                conv = _full_convolution(a, fft_len=fft_len_conv)
                max_c = conv.max()
                val = 2.0 * n * max_c            # denominator = 1
                if val < best_local_val - 1e-12:
                    best_local_val = val
                    best_local = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} r={r} iter={t:05d}] new best {val:.6f}")

                # Build sub‑gradient of the max‑function
                tol = max_c * 1e-12 + 1e-15
                active = np.where(conv >= max_c - tol)[0]
                w = np.zeros_like(conv)
                w[active] = 1.0 / active.size
                grad_a = 2.0 * _cross_correlation(w, a, fft_len=fft_len_corr)

                # Back‑prop through the soft‑max
                inner = np.dot(a, grad_a)
                grad_x = a * (grad_a - inner)

                # Adam update
                t += 1

                v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
                m_hat = m / (1.0 - beta1 ** t)
