def _adam_optimize_v2(a, n, deadline, max_iters, best_seq, best_score):
    target_sum = math.sqrt(2.0 * n)
    # normalize a
    a = a * (target_sum / a.sum())
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr0 = 0.1
    jitter_interval = 2000
    # temperature schedule
    tau_factor = 0.5
    decay_rate = max_iters / 5.0
    Nfft = 1 << ((2 * n - 1).bit_length())

    # keep track of iterations since last improvement
    no_imp = 0

    for iteration in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # compute convolution via FFT
        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        M = float(conv.max())

        # temperature
        tau = tau_factor * M * math.exp(-iteration / decay_rate)
        if tau < 1e-9:
            # active set for max
            mask = conv >= M - 1e-12
            cnt = mask.sum()
            w = mask.astype(float) / max(cnt, 1)
            Ltau = M
        else:
            w, Ltau = _softmax_weights(conv, tau)

        # weighted reverse convolution
        rev = _weighted_rev(a, w, Nfft)

        # gradient: we follow earlier expression
        coeff = 4.0 * n / (target_sum * target_sum)  # = 2 after normalization
        const = 4.0 * n * Ltau / (target_sum ** 3)
        grad = coeff * rev - const

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        lr = lr0 / (1.0 + 1e-4 * iteration)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # project onto feasible set
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0:
            a = a * (target_sum / s)
        else:
            a = np.full_like(a, target_sum / n)

        # jitter
        if iteration % jitter_interval == 0:
            jitter = np.random.randn(n) * (1e-3 * target_sum / n)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        cur_score = M  # objective after projection equals M
        if cur_score < best_score - 1e-9:
            best_score = cur_score
            best_seq = a.copy()

            a = np.maximum(a, 0.0)
            a = a * (target_sum / a.sum())
            no_imp = 0

    return best_seq, best_score

"""
search for a non‑negative step function that minimises
evaluate_sequence(sequence) = 2·n·max(a∗a) / (Σa)².

Approach
---------
* Explore several lengths (up to 8000).  Larger n gives a lower
  theoretical bound (≈2·n/(2·n‑1) → 1) and more “room” to flatten the
  self‑convolution.
* For each length we generate a diverse set of initial vectors
  (uniform, Dirichlet with different concentration, exponential,
  random Gaussian bumps, sparse spikes, etc.) and also interpolate the
  previously known best construction if it exists.
* Every start is refined by a two‑phase Adam optimiser:
  – Phase 1 uses a soft‑max (log‑sum‑exp) approximation of the max‑
    convolution with a temperature τ that decays slowly.
  – Phase 2 switches automatically to the exact sub‑gradient when τ
    becomes tiny.
* The Adam step is projected onto the feasible set
  (a_i ≥ 0 and Σa_i = √(2 n)).  Small Gaussian jitter is added every
  few thousand steps and a larger random perturbation is triggered
  after long stagnation.
* The best sequence over **all** runs (measured with the exact
  objective) is kept and returned as a plain Python list.

The implementation is pure NumPy, uses FFTs for fast convolutions,
and respects the 1000 s wall‑clock budget.
"""

import time
import math
import numpy as np

def _softmax_weights(conv: np.ndarray, tau: float):
    """Stable soft‑max of the convolution values.
    Returns weights w (∑w=1) and the smoothed maximum L_τ."""
    maxc = conv.max()
    if tau <= 0.0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w, maxc
    ex = np.exp((conv - maxc) / tau)
    sum_ex = ex.sum()
    w = ex / sum_ex
    Ltau = maxc + tau * np.log(sum_ex)
    return w, Ltau


def _weighted_rev(a: np.ndarray, w: np.ndarray, Nfft: int) -> np.ndarray:
    """Gradient of the (soft‑max) max‑convolution.
    Computes rev_i = Σ_k w_k · a_{k‑i} using FFTs.
    """
    A_fft = np.fft.rfft(a, n=Nfft)
    # reverse the weight vector before FFT
    W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)
    conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
    conv_rev = conv_rev[: len(w) + a.shape[0] - 1]
    idx_start = 2 * a.shape[0] - 2
    rev = conv_rev[idx_start - np.arange(a.shape[0])]
    return rev


def _normalize(a: np.ndarray, target_sum: float) -> np.ndarray:
    """Scale a to have ℓ₁‑norm = target_sum (preserving non‑negativity)."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 0:
        a = a * (target_sum / s)
    else:
        a = np.full_like(a, target_sum / a.size)
    return a


def _make_initial_vectors_v2(n: int, target_sum: float) -> list[np.ndarray]:
    """Generate a diverse pool of start vectors for a given length."""
    rng = np.random.default_rng()
    init = []

    # 1) uniform
    init.append(np.full(n, target_sum / n))

    # 2) Dirichlet with several concentration parameters
    for alpha in (0.5, 1.0, 2.0, 5.0):
        g = rng.gamma(alpha, 1.0, size=n)
        init.append(g / g.sum() * target_sum)

    # 3) exponential decay (random rate)
    rate = rng.uniform(0.0005, 0.01)
    x = np.exp(-rate * np.arange(n))
    init.append(x / x.sum() * target_sum)

    # 4) reversed exponential
    init.append(x[::-1] / x.sum() * target_sum)

    # 5) single Gaussian bump (random centre / width)
    sigma = rng.uniform(0.05, 0.20) * n
    centre = rng.integers(0, n)
    idx = np.arange(n)
    g = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
    init.append(g / g.sum() * target_sum)

    # 6) sparse random spikes (≈√n non‑zero entries)
    k = max(2, int(np.sqrt(n)))
    spikes = np.zeros(n)
    positions = rng.choice(n, size=k, replace=False)
    spikes[positions] = rng.uniform(0.1, 1.0, size=k)
    init.append(spikes / spikes.sum() * target_sum)

    # 7) perturbed uniform vector
    pert = np.full(n, target_sum / n)
    pert += rng.normal(scale=0.01 * target_sum / n, size=n)
    init.append(_normalize(pert, target_sum))

    return init


def _adam_optimize_v2(
    a: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int,
    best_seq: np.ndarray,
    best_score: float,

    """
    Adam optimiser with a soft‑max approximation of the max‑convolution.
    Returns the best sequence found in this run together with its exact
    (max‑convolution) score.
    """
    target_sum = math.sqrt(2.0 * n)

    # normalise the initial vector
    a = _normalize(a, target_sum)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr0 = 0.10                     # a bit larger than the original code
    jitter_interval = 2000        # more frequent to escape plateaus

    # temperature schedule
    tau_factor = 0.5               # initial τ = τ_factor·M
    decay_rate = max_iters / 5.0   # slower exponential decay

    Nfft = 1 << ((2 * n - 1).bit_length())   # next power‑of‑two ≥ 2·n‑1

    # stagnation bookkeeping
    iters_since_improve = 0

    for iteration in range(1, max_iters + 1):
        if time.time() >= deadline:

        # ----- exact convolution (FFT) -----
        A_fft = np.fft.rfft(a, n=Nfft)

            cnt = mask.sum()

            Ltau = M
        else:
            w, Ltau = _softmax_weights(conv, tau)

        # ----- weighted reverse convolution (gradient of smoothed max) -----
        rev = _weighted_rev(a, w, Nfft)

        # ----- Adam gradient step (projected gradient of the exact objective) -----
        coeff = 4.0 * n / (target_sum * target_sum)          # =2 after normalisation
        const = 4.0 * n * Ltau / (target_sum ** 3)           # enforces Σa = target_sum
        grad = coeff * rev - const

        # Adam updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

            a = np.maximum(a + jitter, 0.0)

        if cur_score + 1e-12 < best_score:

            a = _normalize(a + 0.2 * rnd, target_sum)
            iters_since_improve = 0

    return best_seq, best_score
