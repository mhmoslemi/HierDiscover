"""
Propose candidate step function using hybrid Adam‑optimizer with exact max‑gradient.
The search runs multiple restarts with diverse random initialisations,
smooth‑max surrogate early on and a switch to the true max‑gradient later.
At the end a short simulated‑annealing refinement is applied.
"""
import time
import numpy as np

# ---------------------------- helper functions ----------------------------

def _next_pow2(x: int) -> int:
    return 1 << int(np.ceil(np.log2(x)))

def _full_convolution(a: np.ndarray, size: int = None) -> np.ndarray:
    n = a.shape[0]
    if size is None:
        size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return conv

def _softmax_weights(conv: np.ndarray, lam: float) -> np.ndarray:
    m = conv.max()
    w = np.exp(lam * (conv - m))
    s = w.sum()
    # avoid division by zero
    if s == 0.0:
        w.fill(1.0 / w.size)
    else:
        w /= s
    return w

def _softmax_gradient(a: np.ndarray, w: np.ndarray, size_corr: int = None) -> np.ndarray:
    n = a.shape[0]
    rev_a = a[::-1]
    if size_corr is None:
        size_corr = _next_pow2(len(w) + n - 1)
    fw = np.fft.rfft(w, size_corr)
    frev = np.fft.rfft(rev_a, size_corr)
    corr = np.fft.irfft(fw * frev, size_corr)
    # slice that corresponds to indices 0 … n‑1 of the original vector
    grad = 2.0 * corr[n - 1 : n - 1 + n]
    return grad

def _exact_max_and_grad(a: np.ndarray, conv: np.ndarray = None):
    """Return max(conv), gradient of max(conv) w.r.t a, and the convolution."""
    n = a.shape[0]
    if conv is None:
        conv = _full_convolution(a)
    max_val = conv.max()
    # indices attaining the maximum
    max_idxs = np.where(conv == max_val)[0]
    grad = np.zeros_like(a)
    for k in max_idxs:
        start = max(0, k - (n - 1))
        end = min(n - 1, k)
        i_range = np.arange(start, end + 1)
        grad[i_range] += 2.0 * a[k - i_range]
    return max_val, grad, conv

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Only 1‑D arrays are supported")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # corner case, all entries will be zero
        return np.zeros_like(v)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _random_initial(n: int, mode: int) -> np.ndarray:
    """Generate a non‑negative vector of length n according to a chosen family."""
    rng = np.random
    if mode == 0:
        a = rng.dirichlet(np.ones(n))
    elif mode == 1:
        r = rng.uniform(0.2, 0.9)
        a = r ** np.arange(n, dtype=np.float64)
        a /= a.sum()
    elif mode == 2:
        i = np.arange(1, n + 1, dtype=np.float64)
        a = np.sin(np.pi * i / (n + 1))
        a = np.maximum(a, 0.0)
        a /= a.sum()
    elif mode == 3:
        p = rng.uniform(0.5, 2.0)
        a = (np.sin(np.pi * np.arange(1, n + 1) / (n + 1))) ** p
        a = np.maximum(a, 0.0)
        a /= a.sum()
    elif mode == 4:
        r = rng.uniform(0.3, 0.85)
        a = r ** np.arange(n, dtype=np.float64)
        a += 0.01 * rng.rand(n)
        a = np.maximum(a, 0.0)
        a /= a.sum()
    elif mode == 5:
        x = np.linspace(-1.0, 1.0, n)
        a = 1.0 - x ** 2
        a = np.maximum(a, 0.0)
        a /= a.sum()
    elif mode == 6:
        half = n // 2
        left = np.sin(np.pi * (np.arange(half) + 1) / (half + 1))
        right = np.sin(np.pi * (np.arange(n - half) + 1) / (n - half + 1))
        a = np.concatenate([left, right])
        a = np.maximum(a, 0.0)
        a /= a.sum()
    elif mode == 7:
        # Gaussian centred at the middle
        sigma = rng.uniform(0.1, 0.4) * n
        i = np.arange(n, dtype=np.float64)

            it_f = float(it)
            m_hat = m / (1.0 - beta1 ** it_f)
            v_hat = v / (1.0 - beta2 ** it_f)

            lr = base_lr * (lr_decay ** it)
            u -= lr * m_hat / (np.sqrt(v_hat) + eps)

            if it % 20 == 0:
                val = 2.0 * n * max_val / (a.sum() ** 2)
                if val < best_val:
                    best_val = float(val)
                    best_seq = a.copy()
                    print(f"[Fine‑tune] iter {it:05d}  new best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Short simulated‑annealing / coordinate‑descent polishing
    if best_seq is not None and time.time() < deadline - 5.0:
        n = best_seq.size
        a = best_seq.copy()
        cur_val = 2.0 * n * _full_convolution(a).max() / (a.sum() ** 2)

        total_steps = 20000
        start_temp = 1e-3
        end_temp = 1e-6

        for step in range(total_steps):
            if time.time() >= deadline - 1.0:
                break
            i, j = np.random.randint(0, n, size=2)
            if i == j:
                continue
            # linear temperature schedule
            temp = start_temp + (end_temp - start_temp) * (step / total_steps)
            delta = (np.random.rand() - 0.5) * temp

            # keep feasibility (non‑negative, sum unchanged)
            if a[i] + delta < 0.0:
                delta = -a[i]
            if a[j] - delta < 0.0:
                delta = a[j]
            if delta == 0.0:
                continue

            a[i] += delta
            a[j] -= delta

            new_val = 2.0 * n * _full_convolution(a).max() / (a.sum() ** 2)

            if new_val < cur_val or np.random.rand() < np.exp(-(new_val - cur_val) / max(temp, 1e-12)):
                cur_val = new_val
                if cur_val < best_val:
                    best_val = float(cur_val)
                    best_seq = a.copy()
            else:
                # revert move
                a[i] -= delta
                a[j] += delta

        print(f"[SA] final simulated‑annealing value = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback (should never happen)
    if best_seq is None:
        n = 3000
        best_seq = np.full(n, 1.0 / n)

    print(f"Final best objective = {best_val:.6f} (length {len(best_seq)})")
    # ``evaluate_sequence`` expects a plain Python list.
    return best_seq.tolist()
