"""
Propose a step‑function sequence that minimises
F(a)=2·n·max(a*a)/(sum a)^2.
The optimiser combines:
1. Gradient descent on the variance of the self‑convolution (smooth
   proxy for flattening the whole convolution).
2. Adam on the exact max (sub‑gradient) for final polishing.
3. Differential‑Evolution and (1+1)‑ES for global exploration.
It respects the given time‑budget and works for any length up to
several thousands of entries.
"""
import time
import numpy as np

def softmax(x):
    mx = np.max(x)
    e = np.exp(x - mx)
    return e / e.sum()

def convolve_fft(x, y):
    """Linear convolution of two real vectors via FFT."""
    n = len(x) + len(y) - 1
    L = 1 << int(np.ceil(np.log2(n)))
    fx = np.fft.rfft(x, L)
    fy = np.fft.rfft(y, L)
    conv = np.fft.irfft(fx * fy, L)[:n]
    return conv.real

def full_convolution(a):
    """Linear self‑convolution a*a (length 2·n‑1)."""
    n = len(a)
    L = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, L)
    conv = np.fft.irfft(fa * fa, L)[: 2 * n - 1]
    return conv.real

def compute_objective(a):
    """Exact objective used by evaluate_sequence."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    conv = full_convolution(a)
    return 2.0 * a.shape[0] * conv.max() / (s * s)

def variance_loss_and_grad(a):
    """
    Returns (loss, grad) where loss = Σ (conv - μ)^2
    and grad = ∂loss/∂a.
    """
    n = len(a)
    conv = full_convolution(a)
    mu = 1.0 / (2 * n - 1)
    diff = conv - mu
    loss = np.sum(diff * diff)
    # gradient wrt a: 4 * (diff ⋆ a_rev)
    a_rev = a[::-1]
    conv_rev = convolve_fft(diff, a_rev)
    # extract the n entries that correspond to indices 0..n‑1
    grad = 4.0 * conv_rev[n - 1 : n - 1 + n]
    return loss, grad

def run_variance_adam_opt(n, init_a, deadline, max_iters=200_000):
    """
    Adam optimiser on the variance‑of‑convolution loss.
    Returns the best (a, value) seen before *deadline*.
    """
    # initialise logits (softmax parametrisation)
    x = np.log(init_a + 1e-15)

    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.05

    best_val = np.inf
    best_a = None
    t = 0
    last_improve = time.time()

    while time.time() < deadline and t < max_iters:
        a = softmax(x)
        loss, grad_a = variance_loss_and_grad(a)

        # evaluate true objective
        val = compute_objective(a)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[VarAdam n={n} iter={t:05d}] new best: {val:.6f}")

        # back‑propagation through softmax
        avg = np.dot(a, grad_a)
        grad_x = a * (grad_a - avg)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)
        x = np.clip(x, -50.0, 50.0)

        # occasional shake when stagnating
        if time.time() - last_improve > 5.0:
            x += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()
            print(f"[VarAdam n={n} iter={t}] shake (stagnation)")

    if best_a is None:
        best_a = np.full(n, 1.0 / n)
        best_val = compute_objective(best_a)

    return best_a, best_val

def run_subgrad_adam_opt(n, init_a, deadline, max_iters=150_000):
    """
    Adam optimiser on the exact sub‑gradient of max(a*a).
    """
    x = np.log(init_a + 1e-15)

    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.05

    best_val = np.inf
    best_a = None
    t = 0
    last_improve = time.time()

    while time.time() < deadline and t < max_iters:
        a = softmax(x)
        conv = full_convolution(a)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[SubgradAdam n={n} iter={t:05d}] new best: {val:.6f}")

        # sub‑gradient of max convolution (see earlier derivation)
        idxs = np.flatnonzero(np.isclose(conv, max_c, atol=1e-14))
        k = np.random.choice(idxs)
        g = np.zeros_like(a)
        start = max(0, k - (n - 1))
        end = min(k, n - 1)
        i = np.arange(start, end + 1)
        j = k - i
        g[i] = 2.0 * a[j]

        # back‑prop through softmax
        avg = np.dot(a, g)
        grad_x = a * (g - avg)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)
        x = np.clip(x, -50.0, 50.0)

        if time.time() - last_improve > 5.0:
            x += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()
            print(f"[SubgradAdam n={n} iter={t}] shake (stagnation)")

    if best_a is None:
        best_a = np.full(n, 1.0 / n)
        best_val = compute_objective(best_a)

    return best_a, best_val

def run_de_opt(n, init_pop, deadline, popsize=12, max_gens=20000):
    """Differential evolution on the simplex (same as before)."""
    pop = []
    for cand in init_pop:
        a = np.asarray(cand, dtype=np.float64)
        s = a.sum()
        if s > 0:
            pop.append(a / s)
        else:
            pop.append(np.full(n, 1.0 / n))
    while len(pop) < popsize:
        pop.append(np.random.dirichlet(np.ones(n)))

    obj_vals = [compute_objective(a) for a in pop]
    best_idx = int(np.argmin(obj_vals))
    best_a = pop[best_idx].copy()
    best_val = obj_vals[best_idx]

    F = 0.8
    CR = 0.9
    last_improve = time.time()
    gen = 0

    while time.time() < deadline and gen < max_gens:
        gen += 1
        for i in range(popsize):
            idxs = list(range(popsize))
            idxs.remove(i)
            r1, r2, r3 = np.random.choice(idxs, 3, replace=False)
            mutant = pop[r1] + F * (pop[r2] - pop[r3])
            mutant = np.maximum(mutant, 0.0)

            cross = np.random.rand(n) < CR
            cross[np.random.randint(n)] = True
            trial = np.where(cross, mutant, pop[i])

            s = trial.sum()
            if s == 0.0:
                trial = np.full(n, 1.0 / n)
            else:
                trial = trial / s

            val = compute_objective(trial)
            if val < obj_vals[i]:
                pop[i] = trial
                obj_vals[i] = val
                if val < best_val:
                    best_val = val
                    best_a = trial.copy()
                    last_improve = time.time()
                    print(f"[DE n={n} gen={gen:04d} i={i}] new best: {val:.6f}")

        if time.time() - last_improve > 10.0:
            break

    return best_a, best_val

def run_es_opt(n, init_a, deadline, max_iter=30000):
    """(1+1) ES fine‑tuning."""
    a = init_a.copy()
    val = compute_objective(a)
    best_a, best_val = a.copy(), val
    sigma = 0.02
    last_improve = time.time()
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        trial = a + sigma * np.random.randn(n)
        trial = np.maximum(trial, 0.0)
        s = trial.sum()
        if s == 0.0:
            trial = np.full(n, 1.0 / n)
        else:
            trial = trial / s

        val_trial = compute_objective(trial)
        if val_trial < best_val:
            best_val = val_trial
            best_a = trial.copy()
            a = trial
            sigma = min(sigma * 1.05, 0.2)
            last_improve = time.time()
            print(f"[ES n={n} it={it}] new best: {best_val:.6f}")
        else:
            sigma *= 0.99

        if time.time() - last_improve > 5.0:
            a = best_a.copy()
            sigma = 0.02
            last_improve = time.time()

    return best_a, best_val

def resize_sequence(seq, new_n):
    """Linear interpolation of a probability vector to a new length."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    # ensure non‑negativity and renormalise
    new_seq = np.maximum(new_seq, 0.0)
    s = new_seq.sum()
    if s < 1e-12:
        new_seq = np.full(new_n, 1.0 / new_n)
    else:
        new_seq = new_seq / s
    return new_seq

def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    the evaluation function.  The method uses a variance‑flattening Adam
    phase followed by a max‑subgradient Adam polish, a brief DE global
    search and a (1+1) ES fine‑tuning.  It respects the total time budget
    and returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5

    best_seq = None
    best_val = np.inf

    # Try a range of dimensions – larger n gives more flexibility.
    lengths = [8192, 4096, 2048, 1024]

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")

        # ---------- build a few sensible initial candidates ----------
        init_candidates = []

        # warm‑start from the globally best sequence (if length matches)
        if best_seq is not None and len(best_seq) == n:
            init_candidates.append(np.array(best_seq, dtype=np.float64))

        # warm‑start from previously discovered sequence (any length)
        if best_seq is not None:
            init_candidates.append(resize_sequence(best_seq, n))

        # use historic construction if available
        if "height_sequence_1" in globals():
            hs = np.asarray(height_sequence_1, dtype=np.float64)
            if hs.size >= n:
                hs = hs[:n]
            else:
                hs = np.pad(hs, (0, n - hs.size))
            hs = np.maximum(hs, 0.0)
            s = hs.sum()
            if s > 0.01:
                hs = hs / s
            else:
                hs = np.random.dirichlet(np.ones(n))
            init_candidates.append(hs)

        # uniform distribution
        init_candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

        # ---------- Variance‑Adam phase (global flattening) ----------
        # give each restart ~10 % of the remaining budget (but at least a few seconds)
        n_restarts = 4
        for r in range(n_restarts):
            if time.time() > deadline:
                break
            if r < len(init_candidates):
                a0 = init_candidates[r]
            else:
                a0 = np.random.dirichlet(np.ones(n))

            slice_time = min((deadline - time.time()) * 0.12, 30.0)
            sub_deadline = time.time() + slice_time
            a_opt, val_opt = run_variance_adam_opt(n, a0, sub_deadline)

            if val_opt < best_val:
                best_val = val_opt
                best_seq = a_opt.copy()
                print(f"[n={n} var‑adam restart={r}] GLOBAL best improved to {best_val:.6f}")

        # ---------- Sub‑gradient Adam polish ----------
        if time.time() < deadline and best_seq is not None and len(best_seq) == n:
            slice_time = min((deadline - time.time()) * 0.10, 20.0)
            sub_deadline = time.time() + slice_time
            a_opt, val_opt = run_subgrad_adam_opt(n, best_seq, sub_deadline)
            if val_opt < best_val:
                best_val = val_opt
                best_seq = a_opt.copy()
                print(f"[n={n}] sub‑grad Adam improved best to {best_val:.6f}")

        # ---------- Differential Evolution (global search) ----------
        if time.time() < deadline:
            de_time = min((deadline - time.time()) * 0.35, 120.0)
            sub_deadline = time.time() + de_time
            de_pop = init_candidates[:3]   # warm start set
            a_de, val_de = run_de_opt(n, de_pop, sub_deadline)
            if val_de < best_val:

                print(f"[n={n}] DE GLOBAL best now {best_val:.6f}")

        best_val = compute_objective(best_seq)

    print(f"\nBest value discovered: {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()

"""
Propose a step‑function sequence that minimises

        F(a) = 2·n·max(a∗a) / (sum(a)²)

under a_i ≥ 0.  The optimiser consists of four stages:

1. **Variance‑Adam** – gradient descent (Adam) on the squared‑error
   between the self‑convolution and its uniform average μ=1/(2n‑1).
   Flattening the whole convolution drives the peak towards the
   theoretical lower bound 2·n·μ≈1.

2. **Sub‑gradient Adam** – a short Adam run on the exact sub‑gradient
   of the max‑convolution to polish the final peak.

3. **Differential Evolution (DE)** – a small‑population global search
   on the simplex for further improvement.

4. **(1+1) Evolution Strategy (ES)** – fine‑tuning with Gaussian
   perturbations.

The routine respects a total time budget (default 1000 s) and works
for any length up to several thousand entries.  The best sequence found
is returned as a plain Python list.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
def softmax(x: np.ndarray) -> np.ndarray:
    """Stable softmax – maps logits to a probability vector (sum=1)."""
    mx = np.max(x)
    e = np.exp(x - mx)
    return e / e.sum()

    fy = np.fft.rfft(y, L)
    conv = np.fft.irfft(fx * fy, L)[:n]
    return conv.real

    n = a.shape[0]
    L = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, L)
    conv = np.fft.irfft(fa * fa, L)[: 2 * n - 1]
    return conv.real


def compute_objective(a: np.ndarray) -> float:

    s = a.sum()
    if s < 1e-12:
        return np.inf
    return 2.0 * a.shape[0] * full_convolution(a).max() / (s * s)


# ----------------------------------------------------------------------
# Variant 1 – variance of the convolution (smooth proxy for flattening)
# ----------------------------------------------------------------------
def variance_loss_and_grad(a: np.ndarray):
    """
    Returns (loss, grad) where loss = Σ (conv - μ)² and grad = ∂loss/∂a.
    """
    n = len(a)
    conv = full_convolution(a)                     # length 2n‑1
    mu = 1.0 / (2 * n - 1)
    diff = conv - mu
    loss = np.sum(diff * diff)

    # gradient wrt a: 4 * (diff ⋆ a_rev)
    a_rev = a[::-1]
    conv_rev = convolve_fft(diff, a_rev)          # length 3n‑2
    grad = 4.0 * conv_rev[n - 1 : n - 1 + n]       # keep the n relevant entries
    return loss, grad


def run_variance_adam_opt(
    n: int,
    init_a: np.ndarray,
    deadline: float,
    max_iters: int = 200_000,
):
    """
    Adam optimiser on the variance‑of‑convolution loss.
    Returns the best (a, value) discovered before *deadline*.
    """
    # logits → softmax parametrisation (guarantees positivity & Σa=1)
    x = np.log(init_a + 1e-15)

    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.05

    best_val = np.inf
    best_a = None
    t = 0
    last_improve = time.time()

    while time.time() < deadline and t < max_iters:
        a = softmax(x)
        loss, grad_a = variance_loss_and_grad(a)

        # true objective (the metric we finally care about)
        val = compute_objective(a)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[VarAdam n={n} iter={t:05d}] new best: {val:.6f}")

        # back‑propagation through softmax
        avg = np.dot(a, grad_a)
        grad_x = a * (grad_a - avg)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

        # keep logits bounded to avoid overflow
        x = np.clip(x, -50.0, 50.0)

        # shake if stagnating
        if time.time() - last_improve > 5.0:
            x += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()
            print(f"[VarAdam n={n} iter={t}] shake (stagnation)")

    if best_a is None:
        best_a = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = compute_objective(best_a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Variant 2 – direct sub‑gradient of max(a∗a) (as in the reference code)
# ----------------------------------------------------------------------
def run_subgrad_adam_opt(
    n: int,
    init_a: np.ndarray,
    deadline: float,
    max_iters: int = 150_000,
):
    """
    Adam optimiser on the exact sub‑gradient of max(a*a).
    """
    x = np.log(init_a + 1e-15)

    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.05

    best_val = np.inf
    best_a = None
    t = 0
    last_improve = time.time()

    while time.time() < deadline and t < max_iters:
        a = softmax(x)
        conv = full_convolution(a)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[SubGradAdam n={n} iter={t:05d}] new best: {val:.6f}")

        # sub‑gradient of the max entry
        idxs = np.flatnonzero(np.isclose(conv, max_c, atol=1e-14))
        k = np.random.choice(idxs)
        g = np.zeros_like(a)
        start = max(0, k - (n - 1))
        end = min(k, n - 1)
        i = np.arange(start, end + 1)
        j = k - i
        g[i] = 2.0 * a[j]

        # back‑prop through softmax
        avg = np.dot(a, g)
        grad_x = a * (g - avg)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)
        x = np.clip(x, -50.0, 50.0)

        if time.time() - last_improve > 5.0:
            x += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()
            print(f"[SubGradAdam n={n} iter={t}] shake (stagnation)")

    if best_a is None:
        best_a = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = compute_objective(best_a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Global search – Differential Evolution
# ----------------------------------------------------------------------
def run_de_opt(
    n: int,
    init_pop: list,
    deadline: float,
    popsize: int = 12,
    max_gens: int = 20000,
):
    """Simple DE on the simplex."""
    pop = []
    for cand in init_pop:
        a = np.asarray(cand, dtype=np.float64)
        s = a.sum()
        pop.append(a / s if s > 0 else np.full(n, 1.0 / n))
    while len(pop) < popsize:
        pop.append(np.random.dirichlet(np.ones(n)))

    obj_vals = [compute_objective(a) for a in pop]
    best_idx = int(np.argmin(obj_vals))
    best_a = pop[best_idx].copy()
    best_val = obj_vals[best_idx]

    F = 0.8
    CR = 0.9
    last_improve = time.time()
    gen = 0

    while time.time() < deadline and gen < max_gens:
        gen += 1
        for i in range(popsize):
            idxs = list(range(popsize))
            idxs.remove(i)
            r1, r2, r3 = np.random.choice(idxs, 3, replace=False)
            mutant = pop[r1] + F * (pop[r2] - pop[r3])
            mutant = np.maximum(mutant, 0.0)

            cross = np.random.rand(n) < CR
            cross[np.random.randint(n)] = True
            trial = np.where(cross, mutant, pop[i])

            s = trial.sum()
            trial = trial / s if s > 0 else np.full(n, 1.0 / n)

            val = compute_objective(trial)
            if val < obj_vals[i]:
                pop[i] = trial
                obj_vals[i] = val
                if val < best_val:
                    best_val = val
                    best_a = trial.copy()
                    last_improve = time.time()
                    print(f"[DE n={n} gen={gen:04d} i={i}] new best: {val:.6f}")

        if time.time() - last_improve > 10.0:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# Local search – (1+1) Evolution Strategy
# ----------------------------------------------------------------------
def run_es_opt(
    n: int,
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 30000,
):
    """Gaussian ES with 1+1‑selection on the simplex."""
    a = init_a.copy()
    val = compute_objective(a)
    best_a, best_val = a.copy(), val
    sigma = 0.02
    last_improve = time.time()
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        trial = a + sigma * np.random.randn(n)
        trial = np.maximum(trial, 0.0)
        s = trial.sum()
        trial = trial / s if s > 0 else np.full(n, 1.0 / n)

        val_trial = compute_objective(trial)
        if val_trial < best_val:
            best_val, best_a = val_trial, trial.copy()
            a = trial
            sigma = min(sigma * 1.05, 0.2)
            last_improve = time.time()
            print(f"[ES n={n} it={it}] new best: {best_val:.6f}")
        else:
            sigma *= 0.99

        if time.time() - last_improve > 5.0:
            a = best_a.copy()
            sigma = 0.02
            last_improve = time.time()

    return best_a, best_val


# ----------------------------------------------------------------------
# Helper – resizing a probability vector to a different length
# ----------------------------------------------------------------------
def resize_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a new length."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    s = new_seq.sum()
    return new_seq / s if s > 1e-12 else np.full(new_n, 1.0 / new_n)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    the evaluation function.  The routine respects a total time budget
    and returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # safety margin

    best_seq = None
    best_val = np.inf

    # Try a hierarchy of lengths – larger n gives more design freedom.
    lengths = [8192, 4096, 2048, 1024]

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")

        # ----------------- initialise a few candidates -----------------
        init_candidates = []

        # warm‑start from the current global best (if same length)
        if best_seq is not None and len(best_seq) == n:
            init_candidates.append(np.array(best_seq, dtype=np.float64))

        # warm‑start from the current global best resized
        if best_seq is not None:
            init_candidates.append(resize_sequence(best_seq, n))

        # historic construction, if provided
        if "height_sequence_1" in globals():
            hs = np.asarray(height_sequence_1, dtype=np.float64)
            hs = hs[:n] if hs.size >= n else np.pad(hs, (0, n - hs.size))
            hs = np.maximum(hs, 0.0)
            s = hs.sum()
            hs = hs / s if s > 0.01 else np.random.dirichlet(np.ones(n))
            init_candidates.append(hs)

        # simple uniform start
        init_candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

        # ----------------- 1️⃣ Variance‑Adam (global flattening) -----------------
        n_restarts = 4
        for r in range(n_restarts):
            if time.time() > deadline:
                break
            a0 = init_candidates[r] if r < len(init_candidates) else np.random.dirichlet(np.ones(n))
            slice_time = min((deadline - time.time()) * 0.12, 30.0)
            sub_deadline = time.time() + slice_time
            a_opt, val_opt = run_variance_adam_opt(n, a0, sub_deadline)
            if val_opt < best_val:
                best_val, best_seq = val_opt, a_opt.copy()
                print(f"[n={n} var‑adam restart={r}] GLOBAL best improved to {best_val:.6f}")

        # ----------------- 2️⃣ Sub‑gradient Adam (polish) -----------------
        if time.time() < deadline and best_seq is not None and len(best_seq) == n:
            slice_time = min((deadline - time.time()) * 0.10, 20.0)
            sub_deadline = time.time() + slice_time
            a_opt, val_opt = run_subgrad_adam_opt(n, best_seq, sub_deadline)
            if val_opt < best_val:
                best_val, best_seq = val_opt, a_opt.copy()
                print(f"[n={n}] sub‑grad Adam improved best to {best_val:.6f}")

        # ----------------- 3️⃣ Differential Evolution -----------------
        if time.time() < deadline:
            de_time = min((deadline - time.time()) * 0.35, 120.0)
            sub_deadline = time.time() + de_time
            de_pop = init_candidates[:3]  # warm‑start set
            a_de, val_de = run_de_opt(n, de_pop, sub_deadline)
            if val_de < best_val:
                best_val, best_seq = val_de, a_de.copy()
                print(f"[n={n}] DE GLOBAL best now {best_val:.6f}")

        # ----------------- 4️⃣ (1+1) ES fine‑tuning -----------------
        if time.time() < deadline and best_seq is not None and len(best_seq) == n:
            es_time = min((deadline - time.time()) * 0.30, 60.0)
            sub_deadline = time.time() + es_time
            a_es, val_es = run_es_opt(n, best_seq, sub_deadline)
            if val_es < best_val:
                best_val, best_seq = val_es, a_es.copy()
                print(f"[n={n}] ES GLOBAL best now {best_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – return a uniform vector if nothing was discovered.
    # ------------------------------------------------------------------
    if best_seq is None:
        fallback_n = 1024
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)
        best_val = compute_objective(best_seq)

    print(f"\nBest value discovered: {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
