        print(f"\n=== Optimising length {n} ===")
        init_candidates = []
        # warm start
        if best_seq is not None and len(best_seq) == n:

        if "height_sequence_1" in globals():

def run_annealed_adam_opt(n, init_a, deadline, max_iters=5000):
    # ensure init_a normalized sum 1
    a0 = np.asarray(init_a, dtype=np.float64)
    if a0.sum() == 0:
        a0 = np.full(n, 1.0/n)
    else:
        a0 = a0 / a0.sum()
    x = np.log(a0 + 1e-15)

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.05

    best_val = np.inf
    best_a = None
    t = 0
    last_improve = time.time()
    # initial temperature based on current conv
    a = softmax(x)
    conv = full_convolution(a)
    init_T = max(conv.max() * 0.5, 1e-8)

    while time.time() < deadline and t < max_iters:
        a = softmax(x)
        conv = full_convolution(a)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[Annealed Adam n={n} iter={t}] new best: {val:.6f}")

        # temperature schedule
        T = max(init_T * (0.98 ** t), 1e-12)

        # compute gradient of smooth max
        grad_smooth, _ = grad_smooth_max(a, T)
        grad_f = 2.0 * n * grad_smooth # because obj = 2n * smooth_max

        # backprop through softmax
        avg = np.dot(a, grad_f)
        grad_x = a * (grad_f - avg)

        # Adam update
        t += 1
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * (grad_x * grad_x)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

        # clip logits
        x = np.clip(x, -50, 50)

        # shake if stagnation
        if time.time() - last_improve > 5.0:
            x += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()
            print(f"[Annealed Adam n={n} iter={t}] shake due to stagnation")

    if best_a is None:
        best_a = np.full(n, 1.0/n)
        best_val = compute_objective(best_a)
    return best_a, best_val

"""
Annealed‑softmax Adam + (1+1)‑ES optimizer for the step‑function
minimisation problem.

The algorithm works on the simplex by representing the heights
as a softmax of unconstrained logits.  Instead of using the exact
sub‑gradient of the non‑smooth max‑convolution, we minimise a
smooth approximation (log‑sum‑exp) whose temperature is annealed
to zero.  This yields a stable gradient that drives the optimiser
away from plateaus.  Adam updates keep the iterates on the simplex.
After the global phase a short (1+1) Evolution‑Strategy fine‑tunes
the best candidate found.

The routine respects the supplied time budget and returns the
best sequence (list of floats) discovered.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities (all pure, top‑level)
# ----------------------------------------------------------------------
def softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (maps logits to a probability vector)."""
    mx = np.max(x)
    e = np.exp(x - mx)
    s = e.sum()
    if s == 0.0:
        # fallback to uniform distribution
        return np.full_like(x, 1.0 / x.size)
    return e / s


def full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a * a (length 2·len(a)‑1) using FFT.
    Returns a real array.
    """
    n = a.shape[0]
    L = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, L)
    conv = np.fft.irfft(fa * fa, L)[: 2 * n - 1]
    return conv.real


def conv_max(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of a."""
    return full_convolution(a).max()


def compute_objective(a: np.ndarray) -> float:
    """
    Exact objective used by evaluate_sequence:
        2·n·max(a∗a) / (sum(a)²)
    """
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return 2.0 * a.shape[0] * conv_max(a) / (s * s)


def cross_corr(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Linear cross‑correlation of w (length 2·n‑1) with a (length n).
    Returns a vector of length n where entry i = Σ_k w_{i+k}·a_k.
    Implemented via FFT for O(n log n) cost.
    """
    n = a.shape[0]
    m = w.shape[0]                # = 2·n‑1
    L = 1 << int(np.ceil(np.log2(m + n - 1)))
    W = np.fft.rfft(w, L)
    A = np.fft.rfft(a, L)
    corr_full = np.fft.irfft(W * np.conj(A), L)[: m + n - 1]
    start = n - 1
    end = start + n               # slice of length n
    return corr_full[start:end]


def grad_smooth_max(a: np.ndarray, T: float):
    """
    Gradient of the temperature‑smoothed maximum of the self‑convolution.
    Returns (grad, max_conv) where grad_i = ∂/∂a_i  (smoothed max).
    """
    conv = full_convolution(a)                # length 2·n‑1
    # stable softmax over conv/T
    scaled = conv / max(T, 1e-12)
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    w = exp_scaled / exp_scaled.sum()         # weights sum to 1

    # each conv entry contributes 2·a_{k‑i}
    corr = cross_corr(w, a)                   # length n, = Σ_k w_k·a_{k‑i}
    grad = 2.0 * corr                         # ∂ smoothed max / ∂a_i
    return grad, conv.max()


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------
def run_annealed_adam_opt(
    n: int,
    init_a: np.ndarray,
    deadline: float,
    max_iters: int = 5000,
):
    """
    Adam optimiser on the softmax‑parameterised simplex using the
    smoothed max‑convolution.  Temperature T is exponentially annealed.
    Returns the best vector (sum = 1) and its objective value.
    """
    a0 = np.asarray(init_a, dtype=np.float64)
    if a0.sum() <= 0.0:
        a0 = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a0 = a0 / a0.sum()

    x = np.log(a0 + 1e-15)                # logits
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.05

    best_val = np.inf
    best_a = None
    t = 0
    last_improve = time.time()

    # initialise temperature from the first convolution
    a = softmax(x)
    init_T = max(conv_max(a) * 0.5, 1e-9)

    while time.time() < deadline and t < max_iters:
        a = softmax(x)                     # stays on simplex
        conv = full_convolution(a)
        max_c = conv.max()
        val = 2.0 * n * max_c               # denominator = 1 (sum a = 1)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[Annealed‑Adam n={n} iter={t:05d}] new best: {val:.6f}")

        # temperature schedule
        T = max(init_T * (0.98 ** t), 1e-12)

        # gradient of the smoothed max (vector of length n)
        grad_smooth, _ = grad_smooth_max(a, T)
        # full objective gradient (since obj = 2·n·smooth_max, sum(a)=1)
        grad_f = 2.0 * n * grad_smooth

        # back‑prop through the softmax
        avg = np.dot(a, grad_f)
        grad_x = a * (grad_f - avg)

        # Adam update on logits
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

        # keep logits bounded to avoid overflow
        x = np.clip(x, -50.0, 50.0)

        # occasional random shake when stagnating
        if time.time() - last_improve > 5.0:
            x += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()
            print(f"[Annealed‑Adam n={n} iter={t}] shake (stagnation)")

    if best_a is None:          # fallback – should never happen
        best_a = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = compute_objective(best_a)

    return best_a, best_val


def run_es_opt(
    n: int,
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 30000,
):
    """
    Simple (1+1) Evolution Strategy on the simplex.
    Gaussian perturbations are accepted only when they improve the exact score.
    """
    a = np.asarray(init_a, dtype=np.float64)
    if a.sum() <= 0.0:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = a / a.sum()
    val = compute_objective(a)

    best_a, best_val = a.copy(), val
    sigma = 0.02                     # initial mutation strength
    last_improve = time.time()
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        trial = a + sigma * np.random.randn(n)
        trial = np.maximum(trial, 0.0)
        s = trial.sum()
        if s == 0.0:
            trial = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            trial = trial / s

        val_trial = compute_objective(trial)
        if val_trial < best_val:
            best_val = val_trial
            best_a = trial.copy()
            a = trial                               # success → keep
            sigma = min(sigma * 1.05, 0.2)           # increase step size a bit
            last_improve = time.time()
            print(f"[ES n={n} it={it}] new best: {best_val:.6f}")
        else:
            sigma *= 0.99                           # decay on failure

        # reset if stuck for a while
        if time.time() - last_improve > 5.0:
            a = best_a.copy()
            sigma = 0.02
            last_improve = time.time()

    return best_a, best_val
