"""
Propose a non‑negative step‑function that minimises

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm proceeds in two stages:

1. **Random parametric exploration** –  Choose a length `n` and a simple
   parametric family (geometric decay, power‑law decay, or a combination)
   with random parameters.  The raw vector is scaled so that
   `sum(a)=sqrt(2*n)`.  The convolution is computed with a real FFT and the
   maximum is recorded.

2. **Local refinement** –  The best candidates are refined with a short
   Adam optimiser working in log‑softmax space (the sequence is represented
   as `a = S·softmax(x)` with `S = sqrt(2*n)`).  The objective is a
   temperature‑smoothed approximation of `max(conv)`.  The optimiser runs
   only for a few seconds per restart, keeping the global best vector.

The search runs for the whole time budget (≈ 1000 s) with many independent
restarts, occasionally seeding a restart from the current global best.
The final best sequence (as a plain Python list) is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper – full linear convolution using real FFT (fast for any length)
# ----------------------------------------------------------------------
def full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of real vectors x and y via FFT."""
    n = x.shape[0] + y.shape[0] - 1
    # next power of two for efficient FFT
    size = 1 << ((n - 1).bit_length())
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    # eliminate numerical noise that can produce tiny negative values
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Adam optimisation of the smoothed‑max surrogate on the log‑it space
# ----------------------------------------------------------------------
def optimise_adam_softmax(
    n: int,
    x0: np.ndarray,
    target_sum: float,
    run_deadline: float,
    *,
    init_tau_factor: float = 0.12,
    min_tau_factor: float = 0.001,
    alpha: float = 0.06,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of a length‑`n` sequence in log‑softmax space.

    Parameters
    ----------
    n : int
        Length of the sequence.
    x0 : np.ndarray
        Initial logits (`log` of the unnormalised sequence).
    target_sum : float
        Desired sum of the coefficients (sqrt(2*n)).
    run_deadline : float
        UNIX time after which the optimisation must stop.
    init_tau_factor, min_tau_factor : float
        Controls the temperature for the smoothed max.
    alpha, beta1, beta2, eps : float
        Adam hyper‑parameters.

    Returns
    -------
    best_seq : np.ndarray
        The best sequence (already scaled to the target sum) encountered.
    best_score : float
        The true evaluation returned by ``evaluate_sequence``.
    """
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    best_seq = None
    best_score = float("inf")
    iteration = 0

    # Main optimisation loop – stops when the allocated time expires
    while time.time() < run_deadline:
        # ----- 1. map logits to the simplex (a = S·softmax(x)) -----
        max_x = np.max(x)
        exp_shift = np.exp(x - max_x)          # improve numerical stability
        p = exp_shift / np.sum(exp_shift)      # probabilities, sum = 1
        a = target_sum * p                     # enforce the sum constraint

        # ----- 2. full self‑convolution and current (smooth) max -----
        conv = full_convolution_fft(a, a)
        max_val = conv.max()
        # The true evaluation equals max_val because sum(a)=√(2n)
        if max_val < best_score:
            best_score = max_val
            best_seq = a.copy()
            print(f"[{time.time():.2f}] n={n}  new local best = {max_val:.6f}")

        # ----- 3. smoothed‑max weights ---------------------------------
        tau = max(init_tau_factor * max_val * (0.95 ** iteration),
                  min_tau_factor * max_val)
        # weights proportional to exp((c_i - max)/tau)
        scaled = (conv - max_val) / tau
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        # ----- 4. gradient w.r.t. a ------------------------------------
        # derivative of Σ_k w_k·(a*a)[k] w.r.t. a is 2·(a ⊛ w_rev)
        w_rev = w[::-1]
        conv2 = full_convolution_fft(a, w_rev)      # length = 3n‑2
        idx = (2 * n - 2) - np.arange(n)            # positions that give Σ_j a_j w_{i+j}
        grad_a = 2.0 * conv2[idx]

        # ----- 5. gradient w.r.t. logits x (chain rule) ---------------
        dotp = np.dot(p, grad_a)
        grad_x = target_sum * p * (grad_a - dotp)

        # ----- 6. Adam update -----------------------------------------
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # occasional tiny random jitter to help escape plateaus
        if iteration % 300 == 0 and np.random.rand() < 0.25:
            x += np.random.randn(n) * 0.001

        # slowly decay the base learning‑rate for stability
        alpha *= 0.9994

    # Final evaluation using the provided evaluator (identical rounding)
    final_score = evaluate_sequence(best_seq.tolist())
    return best_seq, final_score


# ----------------------------------------------------------------------
# Sequence generators for the random‑parameter stage
# ----------------------------------------------------------------------
def generate_geometric(n: int, r: float) -> np.ndarray:
    """Geometric decay: a_i ∝ r**i, i=0..n‑1."""
    idx = np.arange(n, dtype=float)
    raw = np.power(r, idx)
    return raw


def generate_powerlaw(n: int, p: float) -> np.ndarray:
    """Power‑law decay: a_i ∝ (i+1)^{-p}."""
    idx = np.arange(1, n + 1, dtype=float)
    raw = idx ** (-p)
    return raw


def generate_combo(n: int, p: float, r: float) -> np.ndarray:
    """Combined power‑law and geometric decay."""
    idx = np.arange(1, n + 1, dtype=float)
    raw = (idx ** (-p)) * (r ** (np.arange(n, dtype=float)))
    return raw


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative height sequence that minimises the given
    evaluation function.  The routine runs for at most ``budget_s`` seconds
    and returns the best sequence discovered.

    Strategy
    --------
    * Randomly pick a length ``n`` (2000 – 12000) and a simple parametric
      family (geometric, power‑law, or a combination).  Random parameters
      (decay rate, exponent) are drawn uniformly from sensible ranges.
    * Scale the raw vector so that its sum equals sqrt(2 n) – this enforces
      the exact normalisation required by the objective.
    * Keep the best candidate seen.  Whenever a candidate improves on the
      current best, run a short Adam‑based local refinement (see
      ``optimise_adam_softmax``) to push the objective further down.
    * Occasionally seed a new run with the current global best to exploit
      promising regions of the search space.
    * After the time budget expires, return the best sequence (as a plain
      Python list).
    """
    np.random.seed(seed)
    start_time = time.time()
    # small safety margin so that we never exceed the official deadline
    deadline = start_time + budget_s - 0.3

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # Warm‑start with a previously known construction if present.
    # --------------------------------------------------------------
    try:
        hs = globals()["height_sequence_1"]
        hs_arr = np.array(hs, dtype=float)
        hs_arr = np.clip(hs_arr, 0.0, 1000.0)
        # enforce the correct normalisation (the stored sequence may already satisfy it)
        score = evaluate_sequence(hs_arr.tolist())
        if score < best_score:
            best_score = score
            best_seq = hs_arr.copy()
            print(f"[{time.time():.2f}] Warm‑start from global seed = {score:.6f}")
    except Exception:
        pass  # no warm‑start available

    # --------------------------------------------------------------
    # Main search loop – each iteration consumes a small time slice.
    # --------------------------------------------------------------
    while time.time() < deadline:
        remaining = deadline - time.time()
        if remaining < 2.0:
            break  # not enough time for another meaningful run

        # ----- 1. Choose problem size -------------------------------------------------
        # Early on we explore a wider range; later we focus on cheaper lengths.
        if remaining > 300:
            n = np.random.randint(3000, 12000 + 1)
        elif remaining > 150:
            n = np.random.randint(2000, 10000 + 1)
        else:
            n = np.random.randint(1500, 8000 + 1)

        # ----- 2. Choose a parametric family and random parameters --------------------
        family = np.random.choice([0, 1, 2])  # 0: geometric, 1: power‑law, 2: combo
        if family == 0:
            # geometric decay, r in (0.90, 0.9997)
            r = np.random.uniform(0.90, 0.9997)
            raw = generate_geometric(n, r)
        elif family == 1:
            # power‑law, exponent p in (0.1, 2.5)
            p = np.random.uniform(0.1, 2.5)
            raw = generate_powerlaw(n, p)
        else:
            # combo: power‑law * geometric
            p = np.random.uniform(0.1, 2.0)
            r = np.random.uniform(0.90, 0.9995)
            raw = generate_combo(n, p, r)

        # ----- 3. Tiny random jitter to break perfect symmetries ---------------------
        raw += np.random.rand(n) * 1e-7

        # ----- 4. Normalise to the required sum ------------------------------------
        target_sum = np.sqrt(2.0 * n)
        raw_sum = np.sum(raw)
        if raw_sum == 0.0:
            continue  # should never happen, but be safe
        a = raw * (target_sum / raw_sum)

        # ----- 5. Quick evaluation of the raw candidate -----------------------------
        conv = full_convolution_fft(a, a)
        max_val = conv.max()

        # ----- 6. Decide whether to invest extra time for refinement -----------------
        # We only launch an Adam refinement if the raw candidate is promising.
        # Some slack is added to avoid missing near‑optimal points.
        promising = max_val < best_score * 1.005

        # Occasionally force a refinement even if not immediately better,
        # to keep diversity.
        if not promising and np.random.rand() < 0.02:
            promising = True

        if promising:
            # Prepare logits for the optimiser – add a tiny epsilon to avoid log(0)
            x0 = np.log(a + 1e-12)

            # Allocate a modest time slice (3‑9 s) – never exceed the global deadline
            slice_time = min(9.0, remaining * 0.4)
            run_deadline = min(deadline, time.time() + slice_time)

            seq_local, score_local = optimise_adam_softmax(
                n,
                x0,
                target_sum,
                run_deadline,
                init_tau_factor=0.15,
                min_tau_factor=0.0008,
                alpha=0.07,
                beta1=0.9,
                beta2=0.999,
            )

            if score_local < best_score:
                best_score = score_local
                best_seq = seq_local.copy()
                print(
                    f"[{time.time():.2f}] GLOBAL new best = {best_score:.6f} (n={n})"
                )
                # early termination if we already beat the target comfortably
                if best_score < 1.503:
                    break

        else:
            # Even if we skip refinement we still keep the raw candidate if it is the best seen.
            if max_val < best_score:
                best_score = max_val
                best_seq = a.copy()
                print(
                    f"[{time.time():.2f}] GLOBAL new best (raw) = {best_score:.6f} (n={n})"
                )
                if best_score < 1.503:
                    break

        # ----- 7. Occasionally reseed from the current global best ------------------
        # This gives the optimiser a better starting point in future runs.
        if best_seq is not None and np.random.rand() < 0.30:
            # Build a fresh logits array from the best known sequence
            # (might be shorter or longer than the upcoming n)
            src = best_seq
            if src.size >= n:
                a_src = src[:n]
            else:
                # pad with a tiny amount of random noise to keep the length
                pad = np.random.rand(n - src.size) * 1e-6
                a_src = np.concatenate([src, pad])
            # normalise again (the sum after truncation may have changed)
            a_src = a_src * (target_sum / a_src.sum())
            x0 = np.log(a_src + 1e-12)
            # Run a short, cheap refinement (≈ 2 s) – can occasionally improve.
            slice_time = min(2.5, remaining * 0.2)
            run_deadline = min(deadline, time.time() + slice_time)
            seq_local, score_local = optimise_adam_softmax(
                n,
                x0,
                target_sum,
                run_deadline,
                init_tau_factor=0.10,
                min_tau_factor=0.0010,
                alpha=0.05,
                beta1=0.9,
                beta2=0.999,
            )
            if score_local < best_score:
                best_score = score_local
                best_seq = seq_local.copy()
                print(
                    f"[{time.time():.2f}] GLOBAL new best (warm) = {best_score:.6f} (n={n})"
                )
                if best_score < 1.503:
                    break

    # ------------------------------------------------------------------
    # Fallback in the unlikely event that no candidate was generated.
    # ------------------------------------------------------------------
    if best_seq is None:
        # Uniform distribution is a safe fallback.
        n = 1000
        target_sum = np.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.2f}] Fallback uniform sequence, score = {best_score:.6f}")

    # The evaluator expects a plain Python list.
    return best_seq.tolist()
