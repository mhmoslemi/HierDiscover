"""
Propose a step‑function (non‑negative vector) that minimises the
``evaluate_sequence`` objective.

Algorithm
---------
1. Load the best known construction (global ``height_sequence_1``) if it
   exists; otherwise fall back to a uniform vector.
2. For several lengths (original, 1.5× and 2×, capped at 4000) run an
   *alternating linear‑programming / Adam* optimisation.
   * **LP phase** – given a current vector f, solve
         maximise   Σ g_i
         subject to  g_i ≥ 0  and  (f * g)_k ≤ M  ∀k,
     where M = max(f∗f).  The LP is tiny (≤ 8000 constraints) and is
     solved with SciPy's HiGHS optimiser.  g is scaled to the required
     ℓ₁‑norm and mixed into f (f←(1‑t)f+t g) with t≈0.3.  Repeating this
     phase monotonically reduces M.
   * **Adam phase** – after a few LP steps the vector is refined with a
     projected Adam optimiser that minimises the exact max‑convolution.
     The optimiser uses a soft‑max approximation of the max (temperature
     τ) that is exponentially cooled, and projects onto the feasible set
     (a_i≥0, Σa_i=√(2 n)) after every step.
3. Track the best vector (lowest exact max‑convolution) over all runs
   and return it as a plain Python list.

The two phases complement each other: the LP quickly discovers a good
global shape, while Adam performs fine‑grained local optimisation.
The whole search respects the 1000 s wall‑clock limit.

def _softmax_weights(conv: np.ndarray, tau: float):
    """Stable soft‑max of the convolution values.
    Returns weights w (summing to 1) and the smoothed maximum Lτ."""
    maxc = conv.max()
    if tau <= 0.0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w, maxc
    ex = np.exp((conv - maxc) / tau)
    sum_ex = ex.sum()
    w = ex / sum_ex
    Ltau = maxc + tau * np.log(sum_ex)
    return w, Ltau


def _weighted_rev(a: np.ndarray, w: np.ndarray, Nfft: int) -> np.ndarray:
    """Weighted reverse convolution needed for the (soft‑max) gradient.
    Computes rev_i = Σ_k w_k * a_{k‑i}."""
    A_fft = np.fft.rfft(a, n=Nfft)
    W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)
    conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
    conv_rev = conv_rev[: len(w) + a.shape[0] - 1]
    idx_start = 2 * a.shape[0] - 2
    rev = conv_rev[idx_start - np.arange(a.shape[0])]
    return rev


def _max_convolution(a: np.ndarray, Nfft: int) -> float:
    """Maximum entry of the self‑convolution a∗a."""
    A_fft = np.fft.rfft(a, n=Nfft)
    conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * a.shape[0] - 1]
    return float(conv.max())


def _adam_optimize(a: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int,
                   best_seq: np.ndarray,
                   best_score: float) -> tuple[np.ndarray, float]:
    """Projected Adam optimiser (soft‑max → exact) for a fixed length."""
    target_sum = math.sqrt(2.0 * n)

    # normalise the initial vector
    a = a * (target_sum / a.sum())

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr0 = 0.05
    jitter_interval = 5000

    # temperature schedule
    tau_factor = 0.20
    decay_rate = max_iters / 10.0

    Nfft = 1 << ((2 * n - 1).bit_length())

    for iteration in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # exact convolution (used as the current objective)
        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        M = float(conv.max())

        # temperature & soft‑max
        tau = tau_factor * M * math.exp(-iteration / decay_rate)
        if tau < 1e-9:
            mask = conv >= M - 1e-12
            cnt = mask.sum()
            w = mask.astype(float) / max(cnt, 1)
            Ltau = M
        else:
            w, Ltau = _softmax_weights(conv, tau)

        # gradient of the smoothed max (weighted reverse convolution)
        rev = _weighted_rev(a, w, Nfft)

        # Adam gradient step (projected onto Σa = target_sum later)
        coeff = 4.0 * n / (target_sum * target_sum)          # =2 after normalisation
        const = 4.0 * n * Ltau / (target_sum ** 3)
        grad = coeff * rev - const

        # Adam updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        lr = lr0 / (1.0 + 1e-4 * iteration)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto feasible set
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a = a * (target_sum / s)
        else:
            a = np.full_like(a, target_sum / n)

        # occasional jitter to escape plateaus
        if iteration % jitter_interval == 0:
            jitter = np.random.randn(n) * (1e-3 * target_sum / n)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        # exact objective after projection equals M
        cur_score = M
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[Adam len {n:5d} iter {iteration:8d}] new best = {best_score:.6f}")

    return best_seq, best_score


def _lp_refine(f: np.ndarray,
               n: int,
               deadline: float,
               max_iters: int = 30,
               mix_factor: float = 0.30) -> tuple[np.ndarray, float]:
    """Alternating LP refinement (see main docstring)."""
    if not _SCIPY_OK:          # pragma: no cover
        return f, _max_convolution(f, 1 << ((2 * n - 1).bit_length()))

    target_sum = math.sqrt(2.0 * n)
    Nfft = 1 << ((2 * n - 1).bit_length())
    M = _max_convolution(f, Nfft)

    best_f = f.copy()
    best_M = M

    offsets = np.arange(n)                # diagonal offsets (0 … n‑1)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # build sparse convolution matrix T_f (size (2n‑1)×n)
        # diagonal d contains the scalar f[d] repeated (n‑d) times
        data = [np.full(n - d, f[d], dtype=float) for d in range(n)]
        A = _sp.diags(data, offsets, shape=(2 * n - 1, n), format='csc')

        # linear programme: maximise Σ g_i   ⇔   minimise −Σ g_i
        c = -np.ones(n, dtype=float)
        b_ub = np.full(2 * n - 1, M, dtype=float)

        res = _opt.linprog(c,
                           A_ub=A,
                           b_ub=b_ub,
                           bounds=(0, None),
                           method='highs',
                           options={'presolve': True})

        if not res.success:
            # if the LP fails we abort the refinement
            break

        g0 = res.x
        sum_g0 = g0.sum()
        if sum_g0 <= 0.0:
            break

        # scale g0 to the required ℓ₁‑norm and mix it into f
        g = g0 * (target_sum / sum_g0)
        f = (1.0 - mix_factor) * f + mix_factor * g

        # re‑project just in case numerical drift occurs
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s > 0.0:
            f = f * (target_sum / s)
        else:
            f = np.full_like(f, target_sum / n)

        # new max‑convolution
        M = _max_convolution(f, Nfft)

        if M < best_M - 1e-9:
            best_M = M
            best_f = f.copy()
            print(f"[LP len {n:5d} iter {it:3d}] new best M = {best_M:.6f}")

        # tiny improvement → stop early
        if best_M / M < 1.000001:
            # no significant relative gain
            pass

    return best_f, best_M


def _make_initial_vectors(n: int, target_sum: float, rng: np.random.Generator):
    """A small set of diverse starts (uniform, Dirichlet, perturbed)."""
    init = []

    # uniform
    init.append(np.full(n, target_sum / n))

    # Dirichlet (Gamma(1,1) i.i.d.)
    g = rng.gamma(shape=1.0, scale=1.0, size=n)
    init.append(g / g.sum() * target_sum)

    # small random perturbation of uniform
    pert = np.full(n, target_sum / n)
    pert += rng.normal(scale=0.02 * target_sum / n, size=n)
    pert = np.maximum(pert, 0.0)
    pert = pert / pert.sum() * target_sum
    init.append(pert)

    return init


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000,
                     **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence as a plain Python list.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # Load previously discovered construction if it exists.
    try:
        base = np.array(height_sequence_1, dtype=float)  # type: ignore
    except Exception:  # pragma: no cover
        base = None

    # ------------------------------------------------------------------
    # Lengths to explore (original size + scaled versions, capped at 4000)
    if base is not None:
        n0 = base.shape[0]
    else:
        n0 = 2100               # safe default used previously

    length_factors = [1.0, 1.5, 2.0]
    lengths = sorted({int(n0 * f) for f in length_factors})
    lengths = [max(300, min(l, 4000)) for l in lengths]

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)
        print(f"[Run] length={n}, target_sum={target_sum:.3f}")

        # --------------------------------------------------------------
        # Assemble initial vectors for this length.
        init_vectors = []

        # 1) Interpolated base (if we have one of a different size)
        if base is not None:
            if base.shape[0] != n:
                old_x = np.linspace(0.0, 1.0, base.shape[0])
                new_x = np.linspace(0.0, 1.0, n)
                interp = np.interp(new_x, old_x, base)
                interp = np.maximum(interp, 0.0)
                interp = interp / interp.sum() * target_sum
                init_vectors.append(interp)
            else:
                init_vectors.append(base.copy())

        # 2) Small set of generic starts
        init_vectors.extend(_make_initial_vectors(n, target_sum, rng))

        # --------------------------------------------------------------
        # Run optimisation from each start.
        for start_id, a0 in enumerate(init_vectors, 1):
            if time.time() >= deadline:
                break
            a = a0.copy()
            print(f"  [Start {start_id}/{len(init_vectors)}]")

            # ----- LP refinement (quick global shape) -----
            a_lp, lp_score = _lp_refine(a, n, deadline,
                                        max_iters=30, mix_factor=0.30)

            # ----- Adam fine‑tuning (local improvement) -----
            # allocate remaining budget roughly proportionally to length
            remaining = deadline - time.time()
            # we keep a modest number of Adam steps per run (≈150 k)
            max_adam_iters = int(150_000 * (2100 / n))
            max_adam_iters = max(80_000, max_adam_iters)

            a_final, final_score = _adam_optimize(
                a=a_lp.copy(),
                n=n,
                deadline=deadline,
                max_iters=max_adam_iters,
                best_seq=best_seq if best_seq is not None else a_lp,
                best_score=best_score,
            )

            # ----- Global update -----
            if final_score < best_score:
                best_score = final_score
                best_seq = a_final.copy()
                print(f"[Global] new best score = {best_score:.6f} (len={n})")

    # ------------------------------------------------------------------
    # Fallback – should never be needed, but ensures a valid return.
    if best_seq is None:
        n = n0
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n)
        best_score = evaluate_sequence(best_seq.tolist())
        print("[Fallback] using uniform sequence")

    print(f"[Done] best score = {best_score:.6f}, length = {best_seq.shape[0]}")
    return best_seq.tolist()
