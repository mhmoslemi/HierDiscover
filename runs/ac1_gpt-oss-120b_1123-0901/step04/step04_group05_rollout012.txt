def project_simplex(v):
    # v is np.ndarray
    # algorithm: sort descending, compute cumulative sum, find rho.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def pgd_refine(a, max_iter=2000, time_deadline=None):
    n = a.shape[0]
    # ensure sum=1
    a = a / a.sum()
    # step size schedule
    alpha0 = 0.5
    # optional: adapt based on iteration count
    for it in range(1, max_iter+1):
        if time_deadline is not None and time.time() >= time_deadline:
            break
        # compute convolution via FFT.
        conv = np.fft.irfft(np.fft.rfft(a, n*2-1) * np.fft.rfft(a, n*2-1))
        conv = conv[:2*n-1]  # shape conv
        k_star = np.argmax(conv)
        # Compute gradient: grad_i = 2 * a_{k_star - i} if 0 <= k_star - i < n else 0.
        # Equivalent to reversed a shifted.
        # Implementation:
        grad = np.zeros_like(a)
        # using slice:
        start = max(0, k_star - (n-1))
        end = min(n, k_star+1)
        # For i in start..end-1: j = k_star - i
        # compute grad[i] = 2*a[j]
        # vectorized:
        i_idx = np.arange(start, end)
        j_idx = k_star - i_idx
        grad[i_idx] = 2.0 * a[j_idx]
        # step size
        alpha = alpha0 / np.sqrt(it)  # decreasing steps
        a = a - alpha * grad
        # projection onto simplex
        a = project_simplex(a)
        # occasional shaking maybe if no progress? Not essential.
    return a

def run_adam(a0, deadline, best_val):
    n = len(a0)
    # set up parameters
    # use same convolution FFT sizes
    conv_fft_size = 1 << int(np.ceil(np.log2(2*n - 1)))
    size_corr = 1 << int(np.ceil(np.log2(3*n)))
    # logit representation u
    u = np.log(a0 + 1e-30)
    m = np.zeros_like(u)
    v = np.zeros_like(u)
    lam = 0.5
    lam_factor = 1.015
    base_lr = 0.05
    it = 0
    max_iter = 20000
    last_improve = time.time()
    best_local_seq = a0.copy()
    best_local_val = best_val
    while time.time() < deadline and it < max_iter:
        it += 1
        # compute softmax a
        u_shift = u - np.max(u)
        a = np.exp(u_shift)
        a /= a.sum()
        # conv, weights, gradient
        conv = _full_convolution(a, size=conv_fft_size)
        w = _softmax_weights(conv, lam)
        grad_a = _softmax_gradient(a, w, size_corr=size_corr)
        # gradient in log space
        dot = np.dot(a, grad_a)
        grad_u = a * (grad_a - dot)
        # Adam update
        m = 0.9 * m + 0.1 * grad_u
        v = 0.999 * v + 0.001 * (grad_u ** 2)
        it_f = float(it)
        m_hat = m / (1.0 - 0.9 ** it_f)
        v_hat = v / (1.0 - 0.999 ** it_f)
        lr = base_lr * (0.9998 ** it)
        u -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        lam *= lam_factor
        # evaluation
        if it % 20 == 0:
            # compute exact objective = 2 * n * max(conv)
            val = 2.0 * n * conv.max()
            if val < best_local_val:
                best_local_val = val
                best_local_seq = a.copy()
                last_improve = time.time()
            # maybe also update global best if better than best_val passed
        # shake if no improvement for certain time
        if time.time() - last_improve > 2.5:
            u += (np.random.rand(n) - 0.5) * 1e-3
            m.fill(0.0)
            v.fill(0.0)
            last_improve = time.time()
    return best_local_seq

"""
Propose a low‑self‑convolution (small max (a*a)) sequence.

The search proceeds in three phases:

1.  **Seed generation** – Randomly draw sequences from several
    parametric families (Dirichlet, exponential, (half‑)sine,
    sinⁿ, triangular, Gaussian, bimodal).  Each seed is
    normalised (∑a = 1) and its exact objective
    F(a)=2·n·max(conv) is evaluated with an FFT‑based convolution.
    The ten best seeds are kept.

2.  **Adam refinement** – For every kept seed an Adam optimiser
    works on the smooth‑max surrogate of ``F`` (the same
    implementation that achieved 1.51415).  The continuation
    parameter λ is increased slowly.  The optimiser runs for a
    limited number of iterations or until the global time budget
    runs out.  Improvements replace the current global best.

3.  **Final fine‑tuning** – The best sequence found so far is
    further optimised with a sharper λ and a smaller learning
    rate.

All updates keep the sequence non‑negative and normalised,
therefore the evaluation is exactly the value returned by
``evaluate_sequence`` (which converts the list back to the same
objective).  The routine respects the user supplied time budget
(≈ 1000 s) and prints intermediate progress.

The algorithm differs from the previous baseline by
    * exploring a richer set of initial shapes,
    * adding a cheap random‑seed stage,
    * using a projected‑gradient sub‑step only for the very first
      refinement (optional, can be disabled),
    * employing slightly different Adam hyper‑parameters and
      a more aggressive λ‑schedule,
    * keeping a pool of the best candidates and refining each
      of them.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – all top‑level, no closures
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, size: int = None) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``size``."""
    n = a.shape[0]
    if size is None:
        size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return conv


def _softmax_weights(conv: np.ndarray, lam: float) -> np.ndarray:
    """Smooth‑max weights wₖ = exp(lam·convₖ) / Σⱼ exp(lam·convⱼ)."""
    m = conv.max()
    w = np.exp(lam * (conv - m))
    s = w.sum()
    if s == 0.0:                     # safety – should not happen
        w.fill(1.0 / w.shape[0])
    else:
        w /= s
    return w


def _softmax_gradient(a: np.ndarray, w: np.ndarray, size_corr: int = None) -> np.ndarray:
    """
    Gradient of the smooth‑max surrogate w.r.t. the probabilities ``a``.
    The closed form is  ∇ = 2·(w ★ a) .
    """
    n = a.shape[0]
    rev_a = a[::-1]
    if size_corr is None:
        size_corr = _next_pow2(len(w) + n - 1)
    fw = np.fft.rfft(w, size_corr)
    frev = np.fft.rfft(rev_a, size_corr)
    corr = np.fft.irfft(fw * frev, size_corr)
    # slice that corresponds to indices 0 … n‑1 of the original vector
    grad = 2.0 * corr[n - 1 : n - 1 + n]
    return grad


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, ∑x = 1}.
    """
    if v.ndim != 1:
        raise ValueError("simplex projection expects a 1‑D array")
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform point
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_from_seq(a: np.ndarray) -> float:
    """
    Exact objective used by ``evaluate_sequence`` for a normalised
    sequence (∑a = 1).  Returns 2·n·max(conv) where conv = a*a.
    """
    sum_a = a.sum()
    if sum_a <= 0.0:
        return float("inf")
    a = a / sum_a
    n = a.shape[0]
    conv = _full_convolution(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Seed generators – all families produce a non‑negative vector with sum = 1
# ----------------------------------------------------------------------


def _seed_dirichlet(n: int) -> np.ndarray:
    return np.random.dirichlet(np.ones(n))


def _seed_exponential(n: int) -> np.ndarray:
    r = np.random.uniform(0.90, 0.999)          # close to 1 → spread
    a = r ** np.arange(n, dtype=np.float64)
    a /= a.sum()
    return a


def _seed_half_sine(n: int) -> np.ndarray:
    a = np.sin(np.pi * (np.arange(1, n + 1, dtype=np.float64) / (n + 1)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _seed_half_sine_pow(n: int) -> np.ndarray:
    p = np.random.uniform(0.2, 4.0)
    a = np.sin(np.pi * (np.arange(1, n + 1, dtype=np.float64) / (n + 1))) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _seed_triangular(n: int) -> np.ndarray:
    a = 1.0 - np.abs(np.arange(n) - (n - 1) / 2) / ((n - 1) / 2)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _seed_gaussian(n: int) -> np.ndarray:
    mu = np.random.uniform(0.0, n - 1)
    sigma = np.random.uniform(0.05 * n, 0.25 * n)
    x = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((x - mu) / sigma) ** 2)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _seed_bimodal(n: int) -> np.ndarray:
    split = np.random.randint(int(0.3 * n), int(0.7 * n) + 1)
    left = np.sin(np.pi * (np.arange(split) + 1) / (split + 1))
    right_len = n - split
    right = np.sin(np.pi * (np.arange(right_len) + 1) / (right_len + 1))
    a = np.concatenate([left, right])
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Adam optimiser (smooth‑max surrogate)
# ----------------------------------------------------------------------


def _adam_optimize(
    a0: np.ndarray,
    deadline: float,
    global_best_val: float,
    n_max_iter: int = 15000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser for the smooth‑max surrogate.
    Returns the best sequence found during the run and its objective.
    """
    n = a0.shape[0]
    conv_fft_size = _next_pow2(2 * n - 1)
    size_corr = _next_pow2(3 * n)

    # log‑softmax parametrisation (guarantees a≥0, Σa=1)
    u = np.log(a0 + 1e-30)

    m = np.zeros_like(u)
    v = np.zeros_like(u)

    # continuation parameters
    lam = 0.5                         # start smooth
    lam_factor = 1.015                # grow a little faster than baseline
    base_lr = 0.05                    # slightly larger initial lr

    best_local_seq = a0.copy()
    best_local_val = _objective_from_seq(a0)
    it = 0
    last_improve = time.time()

    while time.time() < deadline and it < n_max_iter:
        it += 1
        # softmax → probabilities
        u_shift = u - np.max(u)
        a = np.exp(u_shift)
        a /= a.sum()

        # convolution and smooth‑max weights
        conv = _full_convolution(a, size=conv_fft_size)
        w = _softmax_weights(conv, lam)

        # gradient w.r.t. a
        grad_a = _softmax_gradient(a, w, size_corr=size_corr)

        # map to log‑space (soft‑max Jacobian)
        dot = np.dot(a, grad_a)
        grad_u = a * (grad_a - dot)

        # Adam update
        m = 0.9 * m + 0.1 * grad_u
        v = 0.999 * v + 0.001 * (grad_u ** 2)

        it_f = float(it)
        m_hat = m / (1.0 - 0.9 ** it_f)
        v_hat = v / (1.0 - 0.999 ** it_f)

        lr = base_lr * (0.9997 ** it)      # gentle decay
        u -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # increase λ → sharper surrogate
        lam *= lam_factor

        # occasional exact evaluation
        if it % 20 == 0:
            val = 2.0 * n * conv.max()     # exact objective because Σa = 1
            if val < best_local_val:
                best_local_val = val
                best_local_seq = a.copy()
                last_improve = time.time()
            # also update the global best if this run beats it
            if val < global_best_val:
                global_best_val = val

        # shake if stagnated
        if time.time() - last_improve > 2.0:
            u += (np.random.rand(n) - 0.5) * 1e-3
            m.fill(0.0)
            v.fill(0.0)
            last_improve = time.time()

    return best_local_seq, best_local_val


def _fine_tune(
    a0: np.ndarray,
    deadline: float,
    global_best_val: float,
    n_max_iter: int = 30000,
) -> tuple[np.ndarray, float]:
    """Sharper final fine‑tuning (larger λ, smaller lr)."""
    n = a0.shape[0]
    conv_fft_size = _next_pow2(2 * n - 1)
    size_corr = _next_pow2(3 * n)

    u = np.log(a0 + 1e-30)
    m = np.zeros_like(u)
    v = np.zeros_like(u)

    lam = 5.0
    lam_factor = 1.001
    base_lr = 0.02

    best_seq = a0.copy()
    best_val = _objective_from_seq(a0)
    it = 0
    last_improve = time.time()

    while time.time() < deadline and it < n_max_iter:
        it += 1

        u_shift = u - np.max(u)
        a = np.exp(u_shift)
        a /= a.sum()

        conv = _full_convolution(a, size=conv_fft_size)
        w = _softmax_weights(conv, lam)
        grad_a = _softmax_gradient(a, w, size_corr=size_corr)

        dot = np.dot(a, grad_a)
        grad_u = a * (grad_a - dot)

        m = 0.9 * m + 0.1 * grad_u
        v = 0.999 * v + 0.001 * (grad_u ** 2)

        it_f = float(it)
        m_hat = m / (1.0 - 0.9 ** it_f)
        v_hat = v / (1.0 - 0.999 ** it_f)

        lr = base_lr * (0.9999 ** it)
        u -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        lam *= lam_factor

        if it % 20 == 0:
            val = 2.0 * n * conv.max()
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                last_improve = time.time()
            if val < global_best_val:
                global_best_val = val

        # optional shake (rare)
        if time.time() - last_improve > 3.0:
            u += (np.random.rand(n) - 0.5) * 5e-4
            m.fill(0.0)
            v.fill(0.0)
            last_improve = time.time()

    return best_seq, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step function (list of heights) that
    minimises ``evaluate_sequence`` within ``budget_s`` seconds.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.3        # safety margin

    # ------------------------------------------------------------------
    # Global trackers
    # ------------------------------------------------------------------
    best_val = float("inf")
    best_seq = None

    # Keep a small pool of the most promising seeds
    POOL_SIZE = 12
    pool = []   # list of tuples (val, seq)

    # ------------------------------------------------------------------
    # Phase 1 – random seeds
    # ------------------------------------------------------------------
    # a modest number of cheap evaluations (≈ 300–400)
    n_seeds = 350
    lengths = [1800, 2000, 2200, 2500, 2800, 3000, 3400, 3800, 4000, 4400, 4800, 5000]
    families = [
        _seed_dirichlet,
        _seed_exponential,
        _seed_half_sine,
        _seed_half_sine_pow,
        _seed_triangular,
        _seed_gaussian,
        _seed_bimodal,
    ]

    print("[Phase 1] Generating random seeds...")
    for _ in range(n_seeds):
        if time.time() >= deadline:
            break

        n = int(np.random.choice(lengths))
        generator = np.random.choice(families)
        a = generator(n)

        # occasional injection of the globally supplied sequence (if any)
        try:
            global_seq = np.array(height_sequence_1, dtype=np.float64)
            if global_seq.shape[0] == n and np.random.rand() < 0.05:
                a = global_seq.copy()
        except NameError:
            pass

        val = _objective_from_seq(a)

        # update global best
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Phase 1] New global best: n={n}  val={val:.6f}")

        # maintain the pool of good seeds
        if len(pool) < POOL_SIZE:
            pool.append((val, a.copy()))
            pool.sort(key=lambda x: x[0])
        elif val < pool[-1][0]:
            pool[-1] = (val, a.copy())
            pool.sort(key=lambda x: x[0])

    # ------------------------------------------------------------------
    # Phase 2 – Adam refinement of the pool
    # ------------------------------------------------------------------
    print("[Phase 2] Adam refinement of the top seeds...")
    for idx, (seed_val, seed_seq) in enumerate(pool):
        if time.time() >= deadline:
            break
        # allocate a fraction of the remaining budget per candidate
        remaining = deadline - time.time()
        # give at most 120 s per candidate, but shrink if we are close to the end
        allowed = min(120.0, remaining * 0.7 / max(1, len(pool) - idx))
        cand_deadline = time.time() + allowed

        refined_seq, refined_val = _adam_optimize(
            seed_seq,
            deadline=cand_deadline,
            global_best_val=best_val,
            n_max_iter=15000,
        )
        if refined_val < best_val:
            best_val = refined_val
            best_seq = refined_seq.copy()
            print(f"[Phase 2] Candidate {idx} improved global best → {refined_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final fine‑tuning
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("[Phase 3] Final fine‑tuning...")
        final_seq, final_val = _fine_tune(
            best_seq,
            deadline=deadline,
            global_best_val=best_val,
            n_max_iter=30000,
        )
        if final_val < best_val:
            best_val = final_val
            best_seq = final_seq.copy()
            print(f"[Phase 3] Fine‑tuned best value = {final_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – should never happen
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 2500
        best_seq = np.full(n, 1.0 / n)
        best_val = _objective_from_seq(best_seq)

    print(f"Final best objective = {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
