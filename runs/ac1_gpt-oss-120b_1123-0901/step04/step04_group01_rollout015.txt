def anneal(a, time_budget, start_temp=1e-2, end_temp=1e-5):
    start_time = time.time()
    cur_a = a.copy()
    cur_val = compute_val(cur_a)
    while time.time() < start_time + time_budget:
        # compute temp based on elapsed proportion
        elapsed = time.time() - start_time
        frac = elapsed / time_budget
        temp = start_temp * (1 - frac) + end_temp * frac
        i, j = np.random.choice(n, 2, replace=False)
        max_move = min(cur_a[i], 0.1) # bound
        delta = max_move * np.random.rand()
        if delta == 0: continue
        a_new = cur_a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_val = compute_val(a_new)
        diff = new_val - cur_val
        if diff < 0 or np.random.rand() < np.exp(-diff / temp):
            cur_a = a_new
            cur_val = new_val
    return cur_a, cur_val

def simplex_projection(v):
    """Project vector v onto simplex {x >=0, sum=1}."""
    if v.sum() == 1 and np.all(v >= 0):
        return v
    # sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

The evaluation function  

\[
F(a)=\frac{2\,n\max (a*a)}{(\sum a)^2}
\]

is scale‑invariant, therefore we may restrict ourselves to **probability vectors**
\(\sum a_i = 1,\;a_i\ge 0\).  After this normalisation  

\[
F(a)=2\,n\; \max (a*a) .
\]

The problem is non‑convex because the convolution is quadratic, so we use a
stochastic optimisation that mixes three complementary ideas:

1. **Projected Adam on the simplex** – we optimise a directly (not the logits)
   and after each Adam‑step we project back onto the probability simplex.
2. **Active‑set sub‑gradient** – the gradient of `max(conv)` is obtained by
   averaging the gradients of all active lags (the same idea as the reference
   code, but we work directly on the simplex).
3. **Simulated‑annealing refinement** – after the Adam phase we perform a
   temperature‑controlled random walk that moves a small amount of mass from
   one index to another.  This helps escape the narrow basins that Adam can
   get stuck in.

The whole routine is time‑bounded (default 1000 s).  It tries several problem
sizes (1024 … 8192), several random restarts and, when a previous solution
(`height_sequence_1`) is available, warms‑starts from it.  The best sequence
found during the run is returned.

```python
"""
Propose a non‑negative step function (list of floats) that minimises

    F(a) = 2·n·max(a*a) / (sum(a)²)

The algorithm consists of two phases:

1. Projected Adam optimisation on the simplex, using a sub‑gradient of the
   max‑convolution (active‑set averaging).  The optimisation is performed for
   several problem sizes and random restarts.

2. Simulated‑annealing refinement that randomly transfers a small amount of
   mass between two coordinates; uphill moves are accepted with a temperature
   schedule.

Both phases respect the time limit (default 1000 s).  The best sequence seen
is returned as a plain Python list.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
# Basic convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------
def full_convolution(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) using the FFT."""
    n = a.shape[0]
    if fft_len is None:
        fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


def cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the sub‑gradient.
    Returns a vector of length len(a) equal to
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


# ----------------------------------------------------------------------
# Projection onto the probability simplex
# ----------------------------------------------------------------------
def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Projects a vector onto the simplex {x | x >= 0, sum(x) = 1}.
    Implementation follows the O(n) algorithm by Wang & Carreira‑Perpiñán.
    """
    if v.ndim != 1:
        raise ValueError("simplex_projection expects a 1‑D array")
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform vector
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    # numerical safety: re‑normalise
    w_sum = w.sum()
    if w_sum == 0:
        return np.full_like(v, 1.0 / n)
    return w / w_sum


# ----------------------------------------------------------------------
# Simulated annealing refinement
# ----------------------------------------------------------------------
def anneal_refine(
    a0: np.ndarray,
    n: int,
    time_budget: float,
    cur_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Given a starting probability vector a0, perform a temperature‑controlled
    random walk for at most `time_budget` seconds.  Returns the best vector
    found and its objective value (2·n·max(conv)).
    """
    start = time.time()
    a = a0.copy()
    best_a = a.copy()
    best_val = cur_best_val

    # pre‑compute FFT length for the convolution
    fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))

    while time.time() - start < time_budget:
        # linear temperature decay
        elapsed = time.time() - start
        frac = elapsed / time_budget
        # start relatively warm, finish almost frozen
        temp = 0.03 * (1.0 - frac) + 0.0005 * frac

        i, j = np.random.choice(n, 2, replace=False)
        max_move = a[i] * 0.5  # move at most half of the source mass
        if max_move <= 1e-12:
            continue
        delta = max_move * np.random.rand()

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        conv = full_convolution(a_new, fft_len=fft_len)
        val = 2.0 * n * conv.max()
        diff = val - best_val

        accept = False
        if diff < 0:
            accept = True
        else:
            # Metropolis rule
            if np.random.rand() < np.exp(-diff / temp):
                accept = True

        if accept:
            a = a_new
            if val < best_val:
                best_val = val
                best_a = a_new.copy()

    return best_a, best_val


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs) -> list:
    """
    Time‑bounded search for a step‑function that minimises `evaluate_sequence`.
    Returns the best sequence (list of floats) found before the deadline.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None       # numpy array
    best_val = float("inf")

    # problem sizes – larger n gives a lower theoretical bound, but is slower.
    lengths = [1024, 2048, 4096, 8192]

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")

        # FFT lengths that will be reused many times
        fft_len_conv = 1 << int(np.ceil(np.log2(2 * n - 1)))
        fft_len_corr = 1 << int(np.ceil(np.log2(3 * n)))  # ≥ (2n‑1)+n‑1

        # number of independent restarts – more for the cheap sizes
        if n <= 2048:
            restarts = 10
            max_iters = 120_000
        elif n <= 4096:
            restarts = 6
            max_iters = 80_000
        else:
            restarts = 4
            max_iters = 60_000

        for r in range(restarts):
            if time.time() > deadline:
                break

            # ------------------------------------------------------------------
            # initialise a (probability vector)
            # ------------------------------------------------------------------
            if best_seq is not None and len(best_seq) == n:
                a = best_seq.copy()
                a = a / a.sum()
                print(f"[n={n} restart={r}] warm start from previous best")
            elif 'height_sequence_1' in globals():
                # use the provided historic sequence (trim / pad)
                cand = np.array(height_sequence_1, dtype=np.float64)
                if cand.shape[0] >= n:
                    cand = cand[:n]
                else:
                    cand = np.pad(cand, (0, n - cand.shape[0]))
                cand = np.maximum(cand, 0.0)
                if cand.sum() < 0.01:
                    a = np.random.dirichlet(np.ones(n))
                else:
                    a = cand / cand.sum()
                print(f"[n={n} restart={r}] start from height_sequence_1")
            else:
                # random Dirichlet with random concentration (to span sparse ↔ dense)
                alpha = np.random.uniform(0.3, 5.0)
                a = np.random.dirichlet(alpha * np.ones(n))
                print(f"[n={n} restart={r}] random Dirichlet (α={alpha:.2f})")

            # ------------------------------------------------------------------
            # Projected Adam optimiser on the simplex
            # ------------------------------------------------------------------
            m = np.zeros(n, dtype=np.float64)
            v = np.zeros(n, dtype=np.float64)
            beta1, beta2, eps = 0.9, 0.999, 1e-8
            lr = 0.2          # initial learning rate
            decay_every = 5000
            last_improve = time.time()

            best_local_seq = a.copy()
            best_local_val = float("inf")
            t = 0

            while time.time() < deadline and t < max_iters:
                # ---- convolution and objective ----
                conv = full_convolution(a, fft_len=fft_len_conv)
                max_c = conv.max()
                cur_val = 2.0 * n * max_c

                # keep the best local solution
                if cur_val < best_local_val - 1e-12:
                    best_local_val = cur_val
                    best_local_seq = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} r={r} it={t:05d}] new local best {cur_val:.6f}")

                # ---- sub‑gradient of max(conv) ----
                tol = max_c * 1e-12 + 1e-15
                active = np.where(conv >= max_c - tol)[0]
                w = np.zeros_like(conv)
                w[active] = 1.0 / active.size
                grad_a = 2.0 * cross_correlation(w, a, fft_len=fft_len_corr)

                # ---- Adam update ----
                t += 1
                m = beta1 * m + (1.0 - beta1) * grad_a
                v = beta2 * v + (1.0 - beta2) * (grad_a * grad_a)
                m_hat = m / (1.0 - beta1 ** t)
                v_hat = v / (1.0 - beta2 ** t)
                step = lr * m_hat / (np.sqrt(v_hat) + eps)

                a = a - step
                # projection onto the simplex (keeps non‑negativity and sum=1)
                a = simplex_projection(a)

                # learning‑rate decay
                if t % decay_every == 0:
                    lr *= 0.7

                # occasional random shake if we are stuck
                if time.time() - last_improve > 25.0:
                    a += (np.random.rand(n) - 0.5) * 1e-2
                    a = simplex_projection(a)
                    last_improve = time.time()
                    print(f"[n={n} r={r} it={t}] shake due to stagnation")

            # ------------------------------------------------------------------
            # Update global best with the best local Adam result
            # ------------------------------------------------------------------
            if best_local_val < best_val:
                best_val = best_local_val
                best_seq = best_local_seq.copy()
                print(f"*** Global best improved to {best_val:.6f} (n={n}) ***")

            # ------------------------------------------------------------------
            # Simulated‑annealing refinement (short, cheap)
            # ------------------------------------------------------------------
            remaining = deadline - time.time()
            # allocate at most 30 s for annealing, but never exceed the deadline
            anneal_time = min(30.0, remaining * 0.3)
            if anneal_time > 2.0:
                a_ref, val_ref = anneal_refine(
                    best_local_seq,
                    n,
                    anneal_time,
                    best_local_val,
                )
                if val_ref < best_val:
                    best_val = val_ref
                    best_seq = a_ref.copy()
                    print(f"*** After annealing: new global best {best_val:.6f} (n={n}) ***")

    # ----------------------------------------------------------------------
    # Fallback – should never happen, but keep it safe
    # ----------------------------------------------------------------------
    if best_seq is None:
        n_fallback = 1024
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        best_val = 2.0 * n_fallback * (1.0 / n_fallback)   # =2.0

    print(f"\nFinal best value: {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
```
