def proj_simplex(v):
    # ensures non-negative and sum = 1.
    # if sum(v) == 0: return uniform.
    # else:
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
Hybrid optimizer for the self‑convolution bound.

The routine consists of three stages executed under a global time budget
(≈1000 s):

1. **Adam on a smooth‑max surrogate** – the sequence is represented as a
   soft‑max of logits, guaranteeing non‑negativity and Σa=1.
   The smoothness parameter λ starts small (0.5) and grows to a large
   value (≈30) so that the surrogate approaches the true max.
   Several random restarts with diverse initialisations are performed;
   the best sequence found is kept.

2. **Greedy hill‑climbing / simulated‑annealing** – starting from the
   best sequence, small transfers of probability mass between two
   indices are proposed.  Accept improvements unconditionally and
   occasional uphill moves with a slowly cooling temperature.

3. **Projected sub‑gradient refinement** – the exact objective
   F(a)=2·n·max(a*a) is non‑smooth.  At each iteration we compute the
   index k attaining max(a*a) and move a in the opposite direction of
   its gradient (2·shifted a).  After each step a is projected onto the
   unit simplex.

All stages respect the deadline; intermediate progress is printed.
The returned object is a plain Python list of floats ready for
``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, size: int = None) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``size``."""
    n = a.shape[0]
    if size is None:
        size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def _softmax_weights(conv: np.ndarray, lam: float) -> np.ndarray:
    """Smooth‑max weights wₖ = exp(lam·(convₖ−max(conv))) / Σⱼ …."""
    m = conv.max()
    w = np.exp(lam * (conv - m))
    s = w.sum()
    if s == 0.0:                # numerical safety
        w.fill(1.0 / w.size)
    else:
        w /= s
    return w


def _softmax_gradient(a: np.ndarray, w: np.ndarray, size_corr: int) -> np.ndarray:
    """
    Gradient of Σₖ wₖ·(a*a)[k] w.r.t. a.
    Closed form: 2·(w ★ a), where ★ denotes cross‑correlation.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, n=size_corr)
    frev = np.fft.rfft(rev_a, n=size_corr)
    corr = np.fft.irfft(fw * frev, n=size_corr)
    # slice to obtain length‑n gradient
    grad = 2.0 * corr[n - 1 : n - 1 + n]
    return grad


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    if v.sum() == 1.0 and np.all(v >= 0):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    Returns the best sequence (plain list) found within the time budget.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.3          # safety margin

    # ------------------------------------------------------------------
    # Starting point – use the previously found construction if available.
    init_from_global = globals().get("height_sequence_1")
    if init_from_global is not None:
        a0 = np.asarray(init_from_global, dtype=np.float64)
        a0 = np.maximum(a0, 0.0)
        if a0.sum() == 0:
            a0 = rng.dirichlet(np.ones_like(a0))
        else:
            a0 /= a0.sum()
        n = a0.shape[0]
    else:
        n = 2500
        a0 = rng.dirichlet(np.ones(n))

    best_seq = None
    best_val = np.inf
    restart = 0

    # ------------------------------------------------------------------
    # Helper to evaluate the exact objective (Σa = 1 is enforced).
    def exact_obj(a: np.ndarray) -> float:
        conv = _full_convolution(a)
        return 2.0 * a.shape[0] * conv.max()

    # ------------------------------------------------------------------
    # Adam‑based optimisation (smooth‑max surrogate)
    lengths = [n]                       # we keep the current length
    while time.time() < deadline:
        cur_n = lengths[restart % len(lengths)]

        # ----------- initialise a (logits u) --------------------------------
        mode = restart % 7
        if mode == 0:
            a = rng.dirichlet(np.ones(cur_n))
        elif mode == 1:
            r = rng.uniform(0.2, 0.9)
            a = r ** np.arange(cur_n, dtype=np.float64)
        elif mode == 2:
            i = np.arange(1, cur_n + 1, dtype=np.float64)
            a = np.sin(np.pi * i / (cur_n + 1))
        elif mode == 3:
            p = rng.uniform(0.5, 2.0)
            a = (np.sin(np.pi * np.arange(1, cur_n + 1) / (cur_n + 1))) ** p
        elif mode == 4:
            r = rng.uniform(0.3, 0.85)
            a = r ** np.arange(cur_n, dtype=np.float64) + 0.01 * rng.random(cur_n)
        elif mode == 5:
            x = np.linspace(-1.0, 1.0, cur_n)
            a = 1.0 - x ** 2
        else:
            # bimodal half‑sine
            half = cur_n // 2
            left = np.sin(np.pi * (np.arange(half) + 1) / (half + 1))
            right = np.sin(np.pi * (np.arange(cur_n - half) + 1) / (cur_n - half + 1))
            a = np.concatenate([left, right])

        a = np.maximum(a, 0.0)
        a /= a.sum()

        # Occasionally seed from the global best (if compatible size)
        if best_seq is not None and len(best_seq) == cur_n and rng.random() < 0.1:
            a = np.array(best_seq, copy=True)

        # Log‑space parametrisation (soft‑max guarantees feasibility)
        u = np.log(a + 1e-30)
        m = np.zeros_like(u)
        v = np.zeros_like(u)

        # FFT buffer sizes (constant for this run)
        conv_fft_size = _next_pow2(2 * cur_n - 1)
        corr_fft_size = _next_pow2(3 * cur_n)

        # Hyper‑parameters
        max_iter = 15000
        lr0 = 0.04
        lam_start = 0.5
        lam_target = 30.0
        lam_inc = (lam_target - lam_start) / max_iter
        lam = lam_start

        it = 0
        last_improve = time.time()

        while time.time() < deadline and it < max_iter:
            it += 1

            # ----- soft‑max to obtain a_i -------------------------------------------------
            u_shift = u - np.max(u)               # numeric stability
            a = np.exp(u_shift)
            a /= a.sum()

            # ----- linear convolution (self‑convolution) ----------------------------------
            conv = _full_convolution(a, size=conv_fft_size)

            # ----- smooth‑max weights ------------------------------------------------------
            w = _softmax_weights(conv, lam)

            # ----- gradient of the surrogate ------------------------------------------------
            grad_a = _softmax_gradient(a, w, size_corr=corr_fft_size)

            # ----- map to log‑space gradient (soft‑max Jacobian) ---------------------------
            dot = np.dot(a, grad_a)
            grad_u = a * (grad_a - dot)

            # ----- Adam update -------------------------------------------------------------
            m = 0.9 * m + 0.1 * grad_u
            v = 0.999 * v + 0.001 * (grad_u ** 2)

            it_f = float(it)
            m_hat = m / (1.0 - 0.9 ** it_f)
            v_hat = v / (1.0 - 0.999 ** it_f)

            lr = lr0 * (0.9997 ** it)            # gentle decay
            u -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

            # ----- increase smoothness -----------------------------------------------------
            lam = min(lam + lam_inc, lam_target)

            # ----- exact evaluation every few steps ----------------------------------------
            if it % 10 == 0:
                val = 2.0 * cur_n * conv.max()          # Σa = 1 ⇒ objective = 2·n·max(conv)
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[Restart {restart:03d}] iter {it:05d}  new best = {val:.6f}")

            # ----- shake if stagnating ------------------------------------------------------
            if time.time() - last_improve > 2.0:
                u += (rng.random(cur_n) - 0.5) * 1e-3
                m.fill(0.0)
                v.fill(0.0)
                last_improve = time.time()

        restart += 1
        print(f"[Restart {restart:03d}] finished it={it}, best so far = {best_val:.6f}")

        # ----------------------------------------------------------------------
        # If we are already very close to the deadline, stop launching new restarts.
        # ----------------------------------------------------------------------
        if time.time() > deadline - 5.0:
            break

    # ------------------------------------------------------------------
    # 2️⃣  Hill‑climbing / simulated‑annealing refinement
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback to uniform (should never happen)
        best_seq = np.full(n, 1.0 / n)
        best_val = exact_obj(best_seq)

    cur_seq = best_seq.copy()
    cur_val = best_val
    cur_n = cur_seq.shape[0]
    conv_fft_size = _next_pow2(2 * cur_n - 1)

    # Temperature schedule for simulated annealing
    T = 0.1 * cur_val                # start at ~10 % of the objective
    decay = 0.9999

    hc_steps = 20000
    for step in range(hc_steps):
        if time.time() >= deadline:
            break

        # occasionally replace the whole vector by a fresh Dirichlet draw
        if rng.random() < 0.002:
            cand = rng.dirichlet(np.ones(cur_n))
        else:
            i, j = rng.integers(0, cur_n, size=2)
            if i == j:
                continue
            max_delta = cur_seq[i]
            if max_delta <= 0:
                continue
            # move a random fraction (≤10 %) of the mass from i to j
            delta = max_delta * rng.random() * 0.1
            cand = cur_seq.copy()
            cand[i] -= delta
            cand[j] += delta

        # evaluate candidate
        cand_conv = _full_convolution(cand, size=conv_fft_size)
        cand_val = 2.0 * cur_n * cand_conv.max()
        diff = cand_val - cur_val
        if diff < 0 or rng.random() < np.exp(-diff / (T + 1e-12)):
            cur_seq = cand
            cur_val = cand_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = cur_seq.copy()
                print(f"[HillClimb] step {step:05d}  new best = {best_val:.6f}")

        T = max(T * decay, 1e-9)

    # ------------------------------------------------------------------
    # 3️⃣  Projected sub‑gradient refinement on the exact objective
    # ------------------------------------------------------------------
    cur_seq = best_seq.copy()
    cur_val = best_val
    cur_n = cur_seq.shape[0]
    conv_fft_size = _next_pow2(2 * cur_n - 1)

    eta0 = 0.2          # initial step size
    sub_steps = 5000
    for s in range(sub_steps):
        if time.time() >= deadline:
            break

        conv = _full_convolution(cur_seq, size=conv_fft_size)
        k_max = int(conv.argmax())

        # gradient of the active term max_k conv[k] (2·shifted a)
        g = np.zeros_like(cur_seq)
        # indices i for which 0 ≤ k_max‑i < n
        i_low = max(0, k_max - (cur_n - 1))
        i_high = min(cur_n - 1, k_max)
        if i_low <= i_high:
            idx_i = np.arange(i_low, i_high + 1, dtype=np.int64)
            idx_j = k_max - idx_i
            g[idx_i] = 2.0 * cur_seq[idx_j]

        # diminishing step size
        eta = eta0 / np.sqrt(s + 1.0)
        cand = cur_seq - eta * g
        cand = _proj_simplex(cand)

        cand_val = exact_obj(cand)

        if cand_val < cur_val:
            cur_seq = cand
            cur_val = cand_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq = cur_seq.copy()
                print(f"[Subgrad] iter {s:05d}  new best = {best_val:.6f}")

    print(f"Final best objective = {best_val:.6f} (length {len(best_seq)})")
    # ``evaluate_sequence`` expects a plain Python list.
    return best_seq.tolist()
