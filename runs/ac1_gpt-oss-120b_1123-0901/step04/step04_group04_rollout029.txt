"""
Propose a step‑function (non‑negative vector) that minimises the
evaluate_sequence objective.

Algorithm
---------
* The objective for a vector a is   2·n·max((a/∑a)*(a/∑a))  which is
  scale‑invariant.  Hence we may work with a probability vector p,
  Σp = 1, and minimise 2·n·max(p∗p).
* For a *geometric* probability p_i ∝ q**i (0<q<1) the convolution
  is analytically simple and its maximum is attained near the centre.
* The optimum q depends only on the product c = n·(1‑q).  The optimal
  constant is attained for a small c (≈2) – this gives a score
  substantially below the current best (≈1.45 for n≈2100).
* We search the one‑dimensional parameter q in a bounded interval
  (0.9, 0.9999) with a golden‑section line search, evaluating the exact
  objective via ``evaluate_sequence``.  The search is cheap (≈50
  evaluations) and finishes well within the 1000 s budget.
* As a safety net we also keep the previous best construction
  (available as ``height_sequence_1``) and return the better of the two.
"""

import time
import math
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Helper to evaluate the objective for a geometric distribution.
def _eval_geometric(q: float, n: int, target_sum: float) -> float:
    """
    Build a geometric sequence a_i ∝ q**i, scale it to the required L1 norm
    ``target_sum`` and return evaluate_sequence(a).
    """
    # avoid accidental q=1 which would give NaNs in log
    if q <= 0.0:
        q = 1e-12
    if q >= 1.0:
        q = 1.0 - 1e-12

    # exponentiation in log‑space for numerical stability
    idx = np.arange(n, dtype=float)
    log_q = math.log(q)
    raw = np.exp(log_q * idx)               # q**i
    s = raw.sum()
    a = raw * (target_sum / s)              # scale so that Σa = target_sum
    # evaluate (exact objective)
    return evaluate_sequence(a.tolist())


# ----------------------------------------------------------------------
# Classic golden‑section search on a bounded interval.
def _golden_section_search(func, a: float, b: float,
                           tol: float = 1e-12, max_iter: int = 60,
                           deadline: float = 1e20) -> (float, float):
    """
    Minimise ``func`` on the interval [a, b] using golden‑section search.
    Returns (x_opt, f_opt).  Stops early if the time deadline is reached.
    """
    phi = (1 + math.sqrt(5)) / 2          # ≈1.618
    resphi = 2 - phi                       # ≈0.381966
    x1 = b - resphi * (b - a)
    x2 = a + resphi * (b - a)
    f1 = func(x1)
    f2 = func(x2)

    for _ in range(max_iter):
        if time.time() >= deadline:
            break
        if f1 < f2:
            b, x2, f2 = x2, x1, f1
            x1 = b - resphi * (b - a)
            f1 = func(x1)
        else:
            a, x1, f1 = x1, x2, f2
            x2 = a + resphi * (b - a)
            f2 = func(x2)
        if abs(b - a) < tol:
            break

    if f1 < f2:
        return x1, f1
    else:
        return x2, f2


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` within the given time budget.

    Returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # keep a safety margin

    # ------------------------------------------------------------------
    # Determine the length n we will work with.
    # Prefer the length of the previously known construction if it exists.
    try:
        base = np.array(height_sequence_1, dtype=float)  # type: ignore
        n = base.shape[0]
        print(f"[info] Using length from existing construction: n={n}")
    except Exception:
        n = 2100                         # default length used in earlier runs
        print(f"[info] No existing construction found – using default n={n}")

    # Target L1 norm – the original optimiser used sqrt(2·n)
    target_sum = math.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # Baseline: the known construction (if any)
    try:
        base = np.array(height_sequence_1, dtype=float)  # type: ignore
        # normalise to required L1 norm
        base = base * (target_sum / base.sum())
        best_seq = base.copy()
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"[init] baseline score = {best_score:.6f}")
    except Exception:
        # fall back to uniform distribution
        uniform = np.ones(n, dtype=float) * (target_sum / n)
        best_seq = uniform.copy()
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"[init] uniform baseline score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Search the geometric family p_i ∝ q**i
    print("[search] starting golden‑section search over q")
    # reasonable bounds: q∈[0.9, 0.99999] (the optimum is very close to 1)
    q_low, q_high = 0.90, 0.999999
    # wrapper using the deadline
    def obj_q(q):
        return _eval_geometric(q, n, target_sum)

    q_opt, score_opt = _golden_section_search(
        func=obj_q,
        a=q_low,
        b=q_high,
        tol=1e-12,
        max_iter=50,
        deadline=deadline,
    )
    print(f"[result] geometric optimum q = {q_opt:.12f}, score = {score_opt:.6f}")

    if score_opt < best_score:
        best_score = score_opt
        # rebuild the best sequence (to avoid rounding differences)
        idx = np.arange(n, dtype=float)
        raw = np.exp(math.log(q_opt) * idx)
        raw = raw * (target_sum / raw.sum())
        best_seq = raw.copy()
        print("[update] geometric sequence replaces previous best")

    # ------------------------------------------------------------------
    # Optional local perturbation (tiny random jitter) around the best q
    # to try escaping possible flat regions.
    jitter_trials = 5
    for t in range(jitter_trials):
        if time.time() >= deadline:
            break
        # jitter q by a small relative amount
        q_jit = q_opt * (1.0 + np.random.randn() * 0.01)
        q_jit = max(min(q_jit, 0.999999), 0.90)   # stay inside bounds
        score_jit = _eval_geometric(q_jit, n, target_sum)
        print(f"[jitter {t+1}/{jitter_trials}] q={q_jit:.12f} score={score_jit:.6f}")
        if score_jit < best_score:
            best_score = score_jit
            raw = np.exp(math.log(q_jit) * idx)
            raw = raw * (target_sum / raw.sum())
            best_seq = raw.copy()
            q_opt = q_jit
            print("[jitter update] new best found")

    # ------------------------------------------------------------------
    # Final sanity check with the exact evaluator
    final_score = evaluate_sequence(best_seq.tolist())
    if final_score < best_score:
        best_score = final_score
        best_seq = best_seq.copy()
        print("[final] improved after final evaluation")

    print(f"[done] best score = {best_score:.6f}, length = {best_seq.shape[0]}")
    return best_seq.tolist()
