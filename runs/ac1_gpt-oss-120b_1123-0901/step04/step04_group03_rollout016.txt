def smooth_gradient(a, epsilon):
    conv = full_convolution(a)
    # compute w relative pointing to high values; using stable exponent
    v = (conv - conv.max())/epsilon
    w = np.exp(v)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w = w / w_sum
    # compute grad_a = 2 * cross-correlation of w and a (as described)
    a_rev = a[::-1]
    # compute convolution via FFT
    L = 1 << int(np.ceil(np.log2(len(a_rev) + len(w) - 1)))
    fa_rev = np.fft.rfft(a_rev, L)
    fw = np.fft.rfft(w, L)
    conv_rev = np.fft.irfft(fa_rev * fw, L)[: len(a_rev) + len(w) - 1]
    grad = 2.0 * conv_rev[(len(a_rev)-1) : (len(a_rev)-1 + len(a))]
    return grad

"""
Search for a non‑negative step‑function minimizing 
    F(a)=2·n·max(a∗a)/(sum(a)²)

The algorithm:
* Work on the probability simplex (sum=1) using a soft‑max parametrisation.
* Minimise a smooth surrogate of the max‑convolution:
      Sε(a)=ε·log Σk exp( (a∗a)_k / ε )
  whose gradient can be computed analytically using FFTs.
* Perform Adam optimisation on the surrogate while gradually
  decreasing ε (annealing).  The surrogate converges to the true max,
  therefore the final a is a strong candidate for the original objective.
* Several simple initialisations (uniform, decreasing linear, exponential,
  warm‑start from the historic best) are tried; the best result among
  them is kept.  A final (1+1) ES fine‑tuning step is applied.

The routine respects a global time budget (default 1000 s) and returns
the best sequence found as a plain Python list.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def softmax(x: np.ndarray) -> np.ndarray:
    """Stable softmax – maps logits to a probability vector (sum=1)."""
    m = np.max(x)
    e = np.exp(x - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(x, 1.0 / x.size)
    return e / s


def full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a*a (length 2·len(a)‑1) using FFT.
    Returns a real array.
    """
    n = a.shape[0]
    L = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, L)
    conv = np.fft.irfft(fa * fa, L)[: 2 * n - 1]
    return conv.real


def compute_objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(a∗a)/(sum(a)²) – assumes a≥0."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    n = a.shape[0]
    max_c = full_convolution(a).max()
    return 2.0 * n * max_c / (s * s)


# ----------------------------------------------------------------------
# Gradient of the smoothed surrogate
# ----------------------------------------------------------------------
def smooth_gradient(a: np.ndarray, epsilon: float) -> np.ndarray:
    """
    Gradient of Sε(a) = ε·log Σ_k exp((a*a)_k / ε) wrt a.
    Returns a vector of length len(a).
    The computation uses two FFT‑based convolutions.
    """
    n = a.shape[0]
    # linear self‑convolution
    conv = full_convolution(a)            # length 2n‑1
    # softmax weights over convolution entries
    max_c = conv.max()
    # stable exponentiation
    w = np.exp((conv - max_c) / epsilon)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # grad_i = 2 Σ_k w_k a_{k‑i}
    # this is a cross‑correlation of a and w.
    a_rev = a[::-1]
    L = 1 << int(np.ceil(np.log2(len(a_rev) + len(w) - 1)))
    fa_rev = np.fft.rfft(a_rev, L)
    fw = np.fft.rfft(w, L)
    # full convolution of a_rev and w
    conv_rev = np.fft.irfft(fa_rev * fw, L)[: len(a_rev) + len(w) - 1]
    start = len(a_rev) - 1
    grad = 2.0 * conv_rev[start:start + n]
    return grad


# ----------------------------------------------------------------------
# Optimisation with Adam on the smoothed surrogate
# ----------------------------------------------------------------------
def run_smooth_adam(n: int, init_a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth surrogate Sε with an annealing schedule.
    Returns (best_a, best_value) discovered before *deadline*.
    """
    # initialise logits so that softmax(logits) = init_a
    x = np.log(init_a + 1e-15)

    # Adam parameters
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.12

    t = 0
    best_val = np.inf
    best_a = None
    last_improve = time.time()

    # annealing schedule – (epsilon, max_iters) pairs
    schedule = [
        (0.02, 2000),
        (0.01, 3000),
        (0.005, 5000),
        (0.002, 8000),
        (0.001, 12000),
        (0.0005, 20000),
        (0.0002, 30000),
    ]

    for eps_sched, max_iters in schedule:
        if time.time() > deadline:
            break
        for _ in range(max_iters):
            if time.time() > deadline:
                break
            a = softmax(x)                     # a ≥0, Σa = 1
            # current convolution and surrogate gradient
            grad_a = smooth_gradient(a, eps_sched)

            # back‑propagate through softmax
            avg = np.dot(a, grad_a)
            grad_x = a * (grad_a - avg)

            # Adam update
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_x
            v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

            # keep logits bounded – avoids overflow in the exponent
            x = np.clip(x, -70.0, 70.0)

            # occasional exact evaluation
            if t % 15 == 0:
                a_cur = softmax(x)
                val = compute_objective(a_cur)
                if val < best_val:
                    best_val = val
                    best_a = a_cur.copy()
                    last_improve = time.time()
                    print(f"[SmoothAdam n={n} ε={eps_sched:.5f} iter={t}] New best: {val:.6f}")

            # shaking to escape stagnation
            if time.time() - last_improve > 6.0:
                x += (np.random.rand(n) - 0.5) * 1e-2
                last_improve = time.time()
                # optional debug
                # print(f"[SmoothAdam n={n}] shake after stagnation")

    # fallback in the unlikely case no improvement was recorded
    if best_a is None:
        best_a = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = compute_objective(best_a)

    return best_a, best_val

# (1+1) Evolution Strategy – fine‑tuning (re‑used from previous approach)
# ----------------------------------------------------------------------
def run_es_opt(n: int, init_a: np.ndarray, deadline: float, max_iter: int = 30000):
    """
    Simple (1+1) ES that perturbs the current best with Gaussian noise,
    keeping improvements. Operates directly on the simplex.
    """
    a = init_a.copy()
    val = compute_objective(a)
    best_a, best_val = a.copy(), val
    sigma = 0.02
    last_improve = time.time()
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        trial = a + sigma * np.random.randn(n)
        trial = np.maximum(trial, 0.0)
        s = trial.sum()
        if s == 0.0:
            trial = np.full(n, 1.0 / n)
        else:
            trial = trial / s

        val_trial = compute_objective(trial)
        if val_trial < best_val:
            best_val = val_trial
            best_a = trial.copy()
            a = trial
            sigma = min(sigma * 1.05, 0.2)
            last_improve = time.time()
            print(f"[ES n={n} it={it}] new best: {best_val:.6f}")
        else:
            sigma *= 0.99

        if time.time() - last_improve > 5.0:
            a = best_a.copy()
            sigma = 0.02
            last_improve = time.time()

    return best_a, best_val

"""
Search for a non‑negative step‑function (height sequence) that minimises

        F(a) = 2·n·max(a∗a) / (sum(a)²)

The search works on the probability simplex (sum = 1) and consists of
two stages:

1. **Smooth Adam** – minimise the smoothed surrogate
       Sε(a) = ε·log Σk exp((a∗a)k / ε)
   using Adam on the logits of a soft‑max parametrisation.
   The gradient of Sε is obtained analytically via two FFT‑based
   convolutions.  ε is annealed from 2·10⁻² down to 2·10⁻⁴,
   so the optimiser gradually focuses on the true max‑convolution.

2. **(1+1) Evolution Strategy** – a cheap Gaussian mutation step that
   refines the best vector found by Adam.

Several simple initialisations (uniform, decreasing linear, a few
exponential decays and the historic `height_sequence_1` if present) are
evaluated, the most promising one is fed to the Adam optimiser, and the
best result is kept.  The routine respects a total time budget (default
1000 s) and returns the best sequence as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities (soft‑max, exact convolution, exact objective)
# ----------------------------------------------------------------------
def softmax(x: np.ndarray) -> np.ndarray:
    """Stable soft‑max – maps logits to a probability vector (sum = 1)."""
    m = np.max(x)
    e = np.exp(x - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(x, 1.0 / x.size)
    return e / s


def full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Linear self‑convolution a*a (length 2·len(a)‑1) via FFT.
    Returns a real NumPy array.
    """
    n = a.shape[0]
    L = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, L)
    conv = np.fft.irfft(fa * fa, L)[: 2 * n - 1]
    return conv.real


def compute_objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(a∗a) / (sum(a)²).  Assumes a ≥ 0.
    """
    s = a.sum()
    if s < 1e-12:
        return np.inf
    n = a.shape[0]
    max_c = full_convolution(a).max()
    return 2.0 * n * max_c / (s * s)


# ----------------------------------------------------------------------
# Gradient of the smoothed surrogate Sε
# ----------------------------------------------------------------------
def smooth_gradient(a: np.ndarray, epsilon: float) -> np.ndarray:
    """
    Gradient of
        Sε(a) = ε·log Σk exp((a∗a)k / ε)
    with respect to a (∂Sε/∂a).  Uses two FFT convolutions.
    """
    n = a.shape[0]

    # 1️⃣ linear self‑convolution
    conv = full_convolution(a)                     # length 2n‑1

    # 2️⃣ soft‑max weights over the convolution entries
    max_c = conv.max()
    w = np.exp((conv - max_c) / epsilon)           # stability by shifting
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # 3️⃣ grad_i = 2 Σk w_k a_{k‑i}   (cross‑correlation)
    a_rev = a[::-1]                                # reversed a
    L = 1 << int(np.ceil(np.log2(len(a_rev) + len(w) - 1)))
    fa_rev = np.fft.rfft(a_rev, L)
    fw = np.fft.rfft(w, L)
    conv_rev = np.fft.irfft(fa_rev * fw, L)[: len(a_rev) + len(w) - 1]
    start = len(a_rev) - 1
    grad = 2.0 * conv_rev[start:start + n]
    return grad


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed surrogate (with annealing)
# ----------------------------------------------------------------------
def run_smooth_adam(n: int, init_a: np.ndarray, deadline: float):
    """
    Adam optimisation of the smoothed surrogate Sε with a decreasing
    ε schedule.  Returns the best vector discovered and its exact
    objective value before *deadline*.
    """
    # logits such that softmax(logits) = init_a
    x = np.log(init_a + 1e-15)

    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.12

    t = 0
    best_val = np.inf
    best_a = None
    last_improve = time.time()

    # ε‑annealing schedule (ε, max inner iterations)
    schedule = [
        (0.020, 2000),
        (0.010, 3000),
        (0.005, 5000),
        (0.002, 8000),
        (0.001, 12000),
        (0.0005, 20000),
        (0.0002, 30000),
    ]

    for eps_sched, max_iters in schedule:
        if time.time() > deadline:
            break
        for _ in range(max_iters):
            if time.time() > deadline:
                break
            a = softmax(x)                     # simplex point
            # gradient of the surrogate
            grad_a = smooth_gradient(a, eps_sched)

            # back‑propagation through soft‑max
            avg = np.dot(a, grad_a)
            grad_x = a * (grad_a - avg)

            # Adam update
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_x
            v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

            # keep logits bounded – avoids overflow in exp
            x = np.clip(x, -70.0, 70.0)

            # exact evaluation every few iterations
            if t % 15 == 0:
                a_cur = softmax(x)
                val = compute_objective(a_cur)
                if val < best_val:
                    best_val = val
                    best_a = a_cur.copy()
                    last_improve = time.time()
                    print(f"[SmoothAdam n={n} ε={eps_sched:.5f} iter={t}] "
                          f"new best {val:.6f}")

            # shake if no improvement for a while
            if time.time() - last_improve > 6.0:
                x += (np.random.rand(n) - 0.5) * 1e-2
                last_improve = time.time()

    # fallback – should never happen, but keep a sane vector
    if best_a is None:
        best_a = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = compute_objective(best_a)

    return best_a, best_val


# ----------------------------------------------------------------------
# (1+1) Evolution Strategy – fine‑tuning
# ----------------------------------------------------------------------
def run_es_opt(n: int, init_a: np.ndarray, deadline: float, max_iter: int = 30000):
    """
    Simple (1+1) ES that perturbs the current best with Gaussian noise,
    keeping improvements. Operates directly on the simplex.
    """
    a = init_a.copy()
    val = compute_objective(a)
    best_a, best_val = a.copy(), val
    sigma = 0.02
    last_improve = time.time()
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        trial = a + sigma * np.random.randn(n)
        trial = np.maximum(trial, 0.0)
        s = trial.sum()
        if s == 0.0:
            trial = np.full(n, 1.0 / n)
        else:
            trial = trial / s

        val_trial = compute_objective(trial)
        if val_trial < best_val:
            best_val = val_trial
            best_a = trial.copy()
            a = trial
            sigma = min(sigma * 1.05, 0.2)
            last_improve = time.time()
            print(f"[ES n={n} it={it}] new best {best_val:.6f}")
        else:
            sigma *= 0.99

        if time.time() - last_improve > 5.0:
            a = best_a.copy()
            sigma = 0.02
            last_improve = time.time()

    return best_a, best_val

    best_val = np.inf

    # more expensive.  The loop stops automatically when time runs out.
    for n in [8192, 4096, 2048, 1024]:

        candidates = []

            hs = np.maximum(hs, 0.0)
            s = hs.sum()
            if s > 0.0:

            else:

        candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

        for _ in range(2):
            candidates.append(np.random.dirichlet(np.ones(n)))

        # ------------------------------------------------------------------

        best_val = compute_objective(best_seq)

    print(f"\nBest value discovered: {best_val:.6f} (length {len(best_seq)})")
    # Convert to a plain Python list as required by the external evaluator
    return best_seq.tolist()
