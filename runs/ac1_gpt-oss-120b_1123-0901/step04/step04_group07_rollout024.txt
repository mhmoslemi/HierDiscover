def compute_objective(seq: np.ndarray) -> float:
    """Return the objective 2 n max(conv(seq,seq)) / (sum(seq)^2)."""
    n = seq.shape[0]
    sum_a = seq.sum()
    if sum_a < 0.01:
        return np.inf
    # Use FFT for speed:
    Nfft = 1 << ((2 * n - 1).bit_length())
    a_fft = np.fft.rfft(seq, n=Nfft)
    conv = np.fft.irfft(a_fft * a_fft, n=Nfft)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (sum_a * sum_a))

def normalize(arr: np.ndarray, target_sum: float) -> np.ndarray:
    """Clip negative, normalize sum to target_sum."""
    arr = np.maximum(arr, 0.0)
    s = arr.sum()
    if s > 0:
        arr = arr * (target_sum / s)
    else:
        arr = np.full_like(arr, target_sum / arr.shape[0])
    return arr

def generate_initial_shapes(n: int, target_sum: float, rng) -> list[np.ndarray]:
    """Create diverse initial vectors."""
    shapes = []
    # Uniform
    shapes.append(np.full(n, target_sum / n))
    # Random Dirichlet
    g = rng.gamma(1.0, 1.0, size=n)
    shapes.append(g / g.sum() * target_sum)
    # Exponential decays
    rates = [rng.uniform(0.0005, 0.02) for _ in range(2)]
    for rate in rates:
        x = np.exp(-rate * np.arange(n))
        shapes.append(x / x.sum() * target_sum)
        # reversed
        shapes.append(x[::-1] / x.sum() * target_sum)
    # Power law
    for p in [-2.0, -1.5, -1.0, -0.5, 0.5, 1.0, 1.5, 2.0]:
        x = (np.arange(1, n+1) ** p)
        shapes.append(x / x.sum() * target_sum)
        # reversed
        shapes.append(x[::-1] / x.sum() * target_sum)
    # Hann and Blackman windows
    hann = 0.5 - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1))
    shapes.append(hann / hann.sum() * target_sum)
    # Blackman
    a0, a1, a2 = 0.42, 0.5, 0.08
    n_arr = np.arange(n)
    black = a0 - a1 * np.cos(2.0 * np.pi * n_arr / (n - 1)) + a2 * np.cos(4.0 * np.pi * n_arr / (n - 1))

    pert = np.full(n, target_sum / n)
    pert += rng.normal(scale=0.01 * target_sum / n, size=n)
    pert = np.maximum(pert, 0.0)
    shapes.append(pert / pert.sum() * target_sum)
    # Optionally, shape from earlier known 'height_sequence_1' if provided later
    return shapes

def _lp_refine(f_seq: np.ndarray, n: int, target_sum: float, deadline: float, cur_best: float) -> tuple[np.ndarray, float]:
    """Iteratively improve sequence by solving linear program for direction g."""
    # limit number of LP steps
    max_steps = 20
    t = 0.3  # mixing factor
    while max_steps > 0 and time.time() < deadline:
        max_steps -= 1
        # compute M
        M = compute_max_convolution(f_seq)  # we need function
        # Build LP constraints: A g <= M
        # Build sparse matrix
        from scipy.sparse import lil_matrix
        A = lil_matrix((2*n-1, n))
        for i in range(n):
            A[i:i+n, i] = f_seq[i]
        A = A.tocsr()
        # Solve LP
        c = -np.ones(n)  # maximize sum(g)
        b_ub = np.full(2*n-1, M, dtype=float)
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
        if not res.success:
            break
        g0 = res.x
        sum_g = g0.sum()
        if sum_g <= 0:
            break
        # Scale to target sum
        g = g0 * (target_sum / sum_g)
        # Mix
        f_new = (1.0 - t) * f_seq + t * g
        # Project to feasible: nonnegative & renorm
        f_new = normalize(f_new, target_sum)
        # Evaluate
        new_score = compute_objective(f_new)
        if new_score < cur_best:
            cur_best = new_score
            f_seq = f_new
            print(f"[LP] improvement to {cur_best:.6f}")
        else:
            # if not improved, maybe reduce mixing factor or break
            t *= 0.6
    return f_seq, cur_best

def _softmax_weights(conv: np.ndarray, tau: float):
    maxc = conv.max()
    if tau <= 0.0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w, maxc
    # stable softmax
    ex = np.exp((conv - maxc) / tau)
    sum_ex = ex.sum()
    w = ex / sum_ex
    Ltau = maxc + tau * np.log(sum_ex)
    return w, Ltau

"""
Propose a non‑negative step function (a list of heights) that minimises
``evaluate_sequence``.  The algorithm works in three stages:

1. **Diverse initialisation** – for a chosen length (up to 8000) a few
   analytically‑motivated shapes (uniform, Dirichlet, exponential,
   power‑law, Hann, etc.) are generated and normalised to the optimal
   ℓ₁‑norm √(2 n).  The best of these is kept.

2. **Projected Adam optimisation** – the current best vector is refined
   using an Adam optimiser on a *soft‑max* smoothed version of the
   max‑convolution.  The optimiser works with exact sub‑gradients once
   the temperature becomes tiny.  After each step the vector is projected
   onto the feasible set (non‑negative, correct sum).  Jitter is added
   occasionally to escape plateaus.

3. **Optional LP fine‑tuning** – if ``scipy`` is available, a tiny linear‑
   programming loop (≤ 15 iterations) is performed on a modest length
   (≈2100).  The LP maximises the ℓ₁‑norm of a partner vector ``g`` under
   the constraint ``f*g ≤ M`` (where ``M`` is the current maximal
   convolution).  The partner is rescaled and mixed with the current
   iterate, yielding a further reduction of the peak.

The whole search stops after the allotted wall‑clock budget (default
1000 s) leaving the best sequence discovered.
"""

import time
import math
import numpy as np

# optional linear‑programming imports
try:
    from scipy.optimize import linprog
    from scipy.sparse import lil_matrix, csr_matrix
    _SCIPY = True
except Exception:               # pragma: no cover
    _SCIPY = False


def _next_pow2(x: int) -> int:
    """Small helper: next power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _softmax_weights(conv: np.ndarray, tau: float):
    """
    Stable soft‑max over the convolution values.
    Returns the weight vector (summing to 1) and the smoothed maximum L_τ.
    """
    maxc = conv.max()
    if tau <= 0.0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w, maxc
    ex = np.exp((conv - maxc) / tau)
    sum_ex = ex.sum()
    w = ex / sum_ex
    Ltau = maxc + tau * np.log(sum_ex)
    return w, Ltau


def _weighted_rev(a: np.ndarray, w: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Compute the (sub‑)gradient of the (soft‑max) max‑convolution:

        rev_i = Σ_k w_k * a_{k‑i}      (out‑of‑range indices ignored)

    using FFTs for O(n log n) speed.
    """
    A_fft = np.fft.rfft(a, n=fft_len)
    W_rev_fft = np.fft.rfft(w[::-1], n=fft_len)
    conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=fft_len)
    # linear convolution length = len(a) + len(w) - 1
    conv_full = conv_rev[: len(w) + a.shape[0] - 1]
    # index that aligns rev_i with the centre of the full convolution
    idx_start = 2 * a.shape[0] - 2
    rev = conv_full[idx_start - np.arange(a.shape[0])]
    return rev


def _max_convolution(a: np.ndarray, fft_len: int) -> float:
    """Maximum of the (linear) convolution a*a."""
    A_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A_fft * A_fft, n=fft_len)[: 2 * a.shape[0] - 1]
    return float(conv.max())


def _objective(a: np.ndarray) -> float:
    """
    Exact objective used by ``evaluate_sequence`` (but without the
    safety clipping – our vectors are already feasible).
    """
    n = a.shape[0]
    s = a.sum()
    if s < 0.01:
        return float("inf")
    fft_len = _next_pow2(2 * n - 1)
    max_b = _max_convolution(a, fft_len)
    return float(2.0 * n * max_b / (s * s))


def _project(a: np.ndarray, target_sum: float) -> np.ndarray:
    """Clip to non‑negative and renormalise ℓ₁‑norm to ``target_sum``."""
    a = np.maximum(a, 0.0)
    tot = a.sum()
    if tot > 0.0:
        a = a * (target_sum / tot)
    else:
        a = np.full_like(a, target_sum / a.shape[0])
    return a


def _adam_optimize(a: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int,
                   best_seq: np.ndarray,
                   best_score: float) -> tuple[np.ndarray, float]:
    """
    Adam optimiser with a soft‑max phase followed by an exact
    sub‑gradient phase.  Returns the best vector seen in this run.
    """
    target_sum = math.sqrt(2.0 * n)
    a = _project(a, target_sum)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr0 = 0.05
    jitter_interval = 4000

    # temperature schedule
    tau_factor = 0.25                     # a bit larger than before
    decay_rate = max_iters / 8.0          # slower decay → smoother optimisation

    fft_len = _next_pow2(2 * n - 1)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ---- exact convolution of current iterate ----
        A_fft = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(A_fft * A_fft, n=fft_len)[: 2 * n - 1]
        M = float(conv.max())

        # ---- temperature & soft‑max weights ----
        tau = tau_factor * M * math.exp(-it / decay_rate)
        if tau < 1e-9:                              # exact sub‑gradient
            mask = conv >= M - 1e-12
            cnt = mask.sum()
            w = mask.astype(float) / max(cnt, 1)
            Ltau = M
        else:
            w, Ltau = _softmax_weights(conv, tau)

        # ---- weighted reverse convolution (gradient of the smoothed max) ----
        rev = _weighted_rev(a, w, fft_len)

        # ---- gradient of the normalised objective (see comment in docstring) ----
        coeff = 4.0 * n / (target_sum * target_sum)          # =2 for our normalisation
        const = 4.0 * n * Ltau / (target_sum ** 3)           # = 2*M/target_sum
        grad = coeff * rev - const                            # direction that *increases* the peak

        # ---- Adam update (we move opposite to increase) ----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / (1.0 + 1e-4 * it)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)          # descend

        # ---- projection onto feasible set ----
        a = _project(a, target_sum)

        # ---- occasional jitter to shake out plateaus ----
        if it % jitter_interval == 0:
            jitter = np.random.randn(n) * (1e-3 * target_sum / n)
            a = _project(np.maximum(a + jitter, 0.0), target_sum)

        # ---- exact objective after projection (equals M) ----
        cur_score = M
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[len {n:5d} iter {it:8d}] new best = {best_score:.6f}")

    return best_seq, best_score


def _lp_refine(f_seq: np.ndarray,
               n: int,
               target_sum: float,
               deadline: float,
               cur_best: float) -> tuple[np.ndarray, float]:
    """
    Small linear‑programming based improvement loop (only if SciPy is
    available).  Works best on moderate lengths (≈2000) where the
    convolution matrix fits comfortably in memory.
    """
    if not _SCIPY:
        return f_seq, cur_best

    max_steps = 15
    mix = 0.35                     # mixing factor (new = (1‑mix)·old + mix·g)

    while max_steps and time.time() < deadline:
        max_steps -= 1

        M = _max_convolution(f_seq, _next_pow2(2 * n - 1))

        # Build sparse constraint matrix A (size (2n‑1) × n)
        A = lil_matrix((2 * n - 1, n), dtype=float)
        for i in range(n):
            A[i:i + n, i] = f_seq[i]
        A = csr_matrix(A)

        # Linear programme:
        #   maximise   Σ g_i    ⇔ minimise  -Σ g_i
        #   subject to  A·g ≤ M      (upper bound on the convolution)
        #               g_i ≥ 0
        c = -np.ones(n)                         # minimise –sum(g)
        b_ub = np.full(2 * n - 1, M, dtype=float)

        res = linprog(c, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if not res.success:
            break

        g0 = res.x
        if g0.sum() <= 0.0:
            break

        # Scale to the required ℓ₁‑norm and mix.
        g = g0 * (target_sum / g0.sum())
        f_new = (1.0 - mix) * f_seq + mix * g
        f_new = _project(f_new, target_sum)

        new_score = _objective(f_new)
        if new_score < cur_best:
            cur_best = new_score
            f_seq = f_new
            print(f"[LP] improvement → {cur_best:.6f}")
        else:
            # no improvement – shrink the mixing factor and try again
            mix *= 0.6

    return f_seq, cur_best


def _generate_initial_shapes(n: int,
                             target_sum: float,
                             rng: np.random.Generator) -> list[np.ndarray]:
    """Return a small, diverse pool of initial vectors (already normalised)."""
    out = []

    # 1) uniform
    out.append(np.full(n, target_sum / n))

    # 2) random Dirichlet
    g = rng.gamma(1.0, 1.0, size=n)
    out.append(g / g.sum() * target_sum)

    # 3) exponential decay (random rate)
    for _ in range(2):
        rate = rng.uniform(0.0005, 0.015)
        x = np.exp(-rate * np.arange(n))
        out.append(x / x.sum() * target_sum)
        out.append(x[::-1] / x.sum() * target_sum)   # reversed

    # 4) power‑law family
    for p in (-2.0, -1.5, -1.0, -0.5, 0.5, 1.0, 1.5, 2.0):
        x = (np.arange(1, n + 1, dtype=float) ** p)
        out.append(x / x.sum() * target_sum)
        out.append(x[::-1] / x.sum() * target_sum)

    # 5) Hann window
    hann = 0.5 - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1))
    out.append(hann / hann.sum() * target_sum)

    # 6) Blackman window
    a0, a1, a2 = 0.42, 0.5, 0.08
    n_arr = np.arange(n)
    black = a0 - a1 * np.cos(2.0 * np.pi * n_arr / (n - 1)) + a2 * np.cos(4.0 * np.pi * n_arr / (n - 1))
    out.append(black / black.sum() * target_sum)

    # 7) tiny random perturbation of the uniform vector
    pert = np.full(n, target_sum / n)
    pert += rng.normal(scale=0.01 * target_sum / n, size=n)
    pert = np.maximum(pert, 0.0)
    out.append(pert / pert.sum() * target_sum)

    return out
