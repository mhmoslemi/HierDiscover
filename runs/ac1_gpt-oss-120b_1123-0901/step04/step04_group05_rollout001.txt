import time, numpy as np

def _conv_fft(a, size=None):
    n = a.shape[0]
    if size is None:
        size = 1 << int(np.ceil(np.log2(2*n-1)))
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:2*n-1]
    return conv

def _softmax(conv, lam):
    m = conv.max()
    w = np.exp(lam * (conv - m))
    s = w.sum()
    if s == 0.0:
        w[:] = 1.0/w.shape[0]
    else:
        w /= s
    return w

def _grad_smoothmax(a, w, size_corr=None):
    n = a.shape[0]
    rev_a = a[::-1]
    if size_corr is None:
        size_corr = 1 << int(np.ceil(np.log2(len(w)+n-1)))
    fw = np.fft.rfft(w, size_corr)
    frev = np.fft.rfft(rev_a, size_corr)
    corr = np.fft.irfft(fw * frev, size_corr)
    grad = 2.0 * corr[n-1:n-1+n]
    return grad
```

def propose_candidate(seed=0,budget_s=1000):
    np.random.seed(seed)
    end_time = time.time()+budget_s-0.5
    best_val = float('inf')
    best_seq = None
    restart = 0
    lengths = [2500,4000,6000,8000,10000]
    while time.time() < end_time:
        n = lengths[restart % len(lengths)]
        conv_fft_size = 1 << int(np.ceil(np.log2(2*n-1)))
        corr_size = 1 << int(np.ceil(np.log2(3*n)))

        # initialization: use upsampled previous best if available
        if best_seq is not None and len(best_seq) == n:
            a0 = np.array(best_seq)
        elif best_seq is not None:
            # upsample to new n
            # using linear interpolation
            old = np.array(best_seq)
            old_n = len(old)
            # create new array via np.interp
            xs_old = np.linspace(0,1,old_n)
            xs_new = np.linspace(0,1,n)
            a0 = np.interp(xs_new, xs_old, old)
            a0 = np.maximum(a0, 0)
        else:
            # random init
            mode = restart % 6
            if mode == 0:
                a0 = np.random.dirichlet(np.ones(n))
            elif mode == 1:
                r = np.random.uniform(0.2,0.9)
                a0 = r**np.arange(n)
                a0 /= a0.sum()
            elif mode == 2:
                i = np.arange(1,n+1)
                a0 = np.sin(np.pi*i/(n+1))
                a0 = np.maximum(a0,0)
                a0 /= a0.sum()
            elif mode == 3:
                p = np.random.uniform(0.5,2.0)
                a0 = np.sin(np.pi*np.arange(1,n+1)/(n+1))**p
                a0 = np.maximum(a0,0)
                a0 /= a0.sum()
            elif mode == 4:
                a0 = np.exp(-np.linspace(0,5,n))
                a0 /= a0.sum()
            else:
                a0 = np.ones(n)/n

        # logistic parameterization
        u = np.log(a0+1e-30)

        # Adam accumulators
        m = np.zeros_like(u)
        v = np.zeros_like(u)

        lam = 0.2
        lam_factor = 1.015
        base_lr = 0.03
        max_iter = 20000 if n <= 4000 else 12000

        last_improve = time.time()

        for it in range(1, max_iter+1):

            a = np.exp(u_shift)
            a /= a.sum()

            grad_u = a * (grad_a - dot)

            # Adam update

            v = 0.999*v + 0.001*grad_u**2

            # evaluate every 20 steps

                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[Restart {restart}] n={n} it={it} new best {val:.6f}")

            # if stuck for >2s, apply shake

                u += (np.random.rand(n)-0.5)*1e-3
                m.fill(0)
                v.fill(0)
                last_improve = time.time()

            if time.time() > end_time:
                break

        restart += 1
        print(f"[Restart {restart}] finished n={n} best {best_val:.6f}")

    # final fine-tuning on best sequence
    if best_seq is not None:
        n = len(best_seq)
        conv_fft_size = 1 << int(np.ceil(np.log2(2*n-1)))
        corr_size = 1 << int(np.ceil(np.log2(3*n)))
        u = np.log(best_seq+1e-30)
        m = np.zeros_like(u)
        v = np.zeros_like(u)
        lam = 5.0
        lam_factor = 1.001
        base_lr = 0.01
        last_improve = time.time()
        for it in range(1, 40000+1):
            if time.time() > end_time:
                break
            u_shift = u - np.max(u)
            a = np.exp(u_shift)
            a /= a.sum()
            conv = _conv_fft(a, size=conv_fft_size)
            w = _softmax(conv, lam)
            grad_a = _grad_smoothmax(a, w, size_corr=corr_size)
            dot = np.dot(a, grad_a)
            grad_u = a * (grad_a - dot)

            m = 0.9*m + 0.1*grad_u
            v = 0.999*v + 0.001*grad_u**2

            it_f = float(it)
            m_hat = m/(1-0.9**it_f)

            lr = base_lr * (0.99995**it)

    print(f"Final best {best_val:.6f} length {len(best_seq)}")

"""
Hybrid continuation‑Adam optimiser for the self‑convolution bound.

* Variables are represented with logits u and turned into a probability
  vector a = softmax(u) → Σ a = 1, a_i ≥ 0 (scaling does not affect the
  objective 2·n·max(a*a)/(Σa)²).

* A smooth‑max surrogate  Σ_k w_k·(a*a)[k] with
  w_k = exp(λ·conv_k) / Σ_j exp(λ·conv_j) is minimised.
  λ starts small (very smooth) and is multiplied by a factor >1 each
  iteration (continuation), so the optimisation first sees a smooth
  landscape and later focuses on the true maximum.

* Adam with bias‑correction updates the logits.
  If no improvement is seen for a couple of seconds a tiny random shake
  of the logits is applied and the Adam moments are reset – this helps
  escaping plateaus.

* Several restarts are performed with very different initialisations
  (Dirichlet, exponential, half‑sine, power‑sine, exponential‑tail,
  uniform).  When a good sequence has been found it is up‑sampled and
  reused as a starting point for longer vectors.

* After the main phase a fine‑tuning run with a larger λ and a smaller
  learning‑rate further refines the best sequence.

* The sequence is forced to be symmetric (a_i = a_{n‑1‑i}) after each
  soft‑max step – the optimum is expected to be symmetric, and this
  reduces the effective dimension.

The function returns the best plain‑Python list found within the time
budget (default 1000 s).  It prints progress information for monitoring.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray, size: int = None) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``size``."""
    n = a.shape[0]
    if size is None:
        size = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return conv


def _softmax_weights(conv: np.ndarray, lam: float) -> np.ndarray:
    """Soft‑max weights w_k = exp(lam·conv_k) / Σ exp(lam·conv)."""
    m = conv.max()
    # shift by the maximum for numerical stability (exponents ≤ 0)
    w = np.exp(lam * (conv - m))
    s = w.sum()
    if s == 0.0:                     # safety – should not happen
        w.fill(1.0 / w.shape[0])
    else:
        w /= s
    return w


def _softmax_gradient(a: np.ndarray, w: np.ndarray, size_corr: int = None) -> np.ndarray:
    """
    Gradient of the smooth‑max surrogate with respect to a.
    Closed form: ∇_a = 2·(w ★ a) .
    """
    n = a.shape[0]
    rev_a = a[::-1]
    if size_corr is None:
        size_corr = 1 << int(np.ceil(np.log2(len(w) + n - 1)))
    fw = np.fft.rfft(w, size_corr)
    frev = np.fft.rfft(rev_a, size_corr)
    corr = np.fft.irfft(fw * frev, size_corr)
    # slice that corresponds to indices 0 … n‑1 of the original vector
    grad = 2.0 * corr[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step function (list of heights) that minimises
    ``evaluate_sequence``.  The routine runs for at most ``budget_s`` seconds
    and returns the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.25      # safety margin

    # ------------------------------------------------------------------
    # keep the best solution overall
    best_val = float("inf")
    best_seq = None

    # ------------------------------------------------------------------
    # candidate lengths – we let the search explore increasingly long vectors
    lengths = [2500, 4000, 6000, 8000, 10000]

    restart = 0
    while time.time() < deadline:
        # --------------------------------------------------------------
        # pick length for the current restart (cycle through the list)
        n = lengths[restart % len(lengths)]

        # pre‑compute FFT sizes that stay constant inside one inner run
        conv_fft_size = 1 << int(np.ceil(np.log2(2 * n - 1)))
        corr_fft_size = 1 << int(np.ceil(np.log2(3 * n)))   # safe upper bound

        # --------------------------------------------------------------
        # Build an initial vector.
        #   – if we already have a good construction, up‑sample it;
        #   – otherwise use a diverse random family.
        init_vec = None
        old = globals().get("height_sequence_1")
        if old is not None and isinstance(old, (list, tuple, np.ndarray)):
            old = np.asarray(old, dtype=float)
            if old.size == n:
                init_vec = old.copy()
            else:
                # linear interpolation to the new size
                xs_old = np.linspace(0.0, 1.0, old.size)
                xs_new = np.linspace(0.0, 1.0, n)
                init_vec = np.interp(xs_new, xs_old, old)
                init_vec = np.maximum(init_vec, 0.0)

        if init_vec is None:
            mode = restart % 7
            if mode == 0:            # Dirichlet (almost uniform)
                init_vec = np.random.dirichlet(np.ones(n))
            elif mode == 1:          # geometric decay
                r = np.random.uniform(0.2, 0.9)
                init_vec = r ** np.arange(n, dtype=float)
                init_vec /= init_vec.sum()
            elif mode == 2:          # half‑sine
                i = np.arange(1, n + 1, dtype=float)
                init_vec = np.sin(np.pi * i / (n + 1))
                init_vec = np.maximum(init_vec, 0.0)
                init_vec /= init_vec.sum()
            elif mode == 3:          # powered half‑sine
                p = np.random.uniform(0.5, 2.0)
                init_vec = np.sin(np.pi * np.arange(1, n + 1) / (n + 1)) ** p
                init_vec = np.maximum(init_vec, 0.0)
                init_vec /= init_vec.sum()
            elif mode == 4:          # exponential tail + tiny noise
                r = np.random.uniform(0.3, 0.85)
                init_vec = r ** np.arange(n, dtype=float)
                init_vec += 0.01 * np.random.rand(n)
                init_vec = np.maximum(init_vec, 0.0)
                init_vec /= init_vec.sum()
            elif mode == 5:          # quadratic “bump’’
                x = np.linspace(-1.0, 1.0, n)
                init_vec = 1.0 - x ** 2
                init_vec = np.maximum(init_vec, 0.0)
                init_vec /= init_vec.sum()
            else:                    # uniform
                init_vec = np.full(n, 1.0 / n)

        # --------------------------------------------------------------
        # Work in the log‑softmax (logits) domain – this guarantees Σa = 1.
        u = np.log(init_vec + 1e-30)

        # Adam state
        m = np.zeros_like(u)
        v = np.zeros_like(u)

        # Continuation parameters
        lam = 0.2               # start with a very smooth surrogate
        lam_factor = 1.015
        base_lr = 0.03

        # Iteration limits (shorter for the larger vectors)
        max_iter = 20000 if n <= 4000 else 12000

        last_improve = time.time()

        # --------------------------------------------------------------
        for it in range(1, max_iter + 1):
            # ----- logits → probability vector (soft‑max) -----------------
            u_shift = u - np.max(u)                # numerical stability
            a = np.exp(u_shift)
            a /= a.sum()

            # enforce symmetry – the optimum is expected to be symmetric
            a = 0.5 * (a + a[::-1])

            # ----- convolution & smooth‑max weights --------------------
            conv = _conv_fft(a, size=conv_fft_size)
            w = _softmax_weights(conv, lam)

            # ----- gradient of the surrogate w.r.t. a ------------------
            grad_a = _softmax_gradient(a, w, size_corr=corr_fft_size)

            # ----- map gradient to log‑space (soft‑max Jacobian) -------
            dot = np.dot(a, grad_a)
            grad_u = a * (grad_a - dot)

            # ----- Adam update -----------------------------------------
            m = 0.9 * m + 0.1 * grad_u
            v = 0.999 * v + 0.001 * (grad_u ** 2)

            # bias‑corrected estimates
            it_f = float(it)
            m_hat = m / (1.0 - 0.9 ** it_f)
            v_hat = v / (1.0 - 0.999 ** it_f)

            lr = base_lr * (0.9999 ** it)          # gentle decay
            u -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

            # ----- increase smoothness ---------------------------------
            lam *= lam_factor

            # ----- evaluate the true objective every few iterations -----
            if it % 20 == 0:
                # because Σa = 1, the exact value is 2·n·max(conv)
                val = 2.0 * n * conv.max()
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[Restart {restart:03d}] n={n:05d} it={it:05d}  new best = {val:.6f}")

            # ----- shake if we have stagnated for >2 s -----------------
            if time.time() - last_improve > 2.0:
                u += (np.random.rand(n) - 0.5) * 1e-3
                m.fill(0.0)
                v.fill(0.0)
                last_improve = time.time()

            # ----- respect the global time budget ----------------------
            if time.time() >= deadline:
                break

        # end of inner optimisation loop
        restart += 1
        print(f"[Restart {restart:03d}] finished length {n}, best so far = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final fine‑tuning on the best sequence discovered so far.
    # (higher λ, slower learning‑rate, many more steps)
    if best_seq is not None:
        n = len(best_seq)
        conv_fft_size = 1 << int(np.ceil(np.log2(2 * n - 1)))
        corr_fft_size = 1 << int(np.ceil(np.log2(3 * n)))

        u = np.log(best_seq + 1e-30)
        m = np.zeros_like(u)
        v = np.zeros_like(u)

        lam = 5.0
        lam_factor = 1.001
        base_lr = 0.01

        last_improve = time.time()
        fine_max_iter = 40000

        for it in range(1, fine_max_iter + 1):
            if time.time() >= deadline:
                break

            u_shift = u - np.max(u)
            a = np.exp(u_shift)
            a /= a.sum()
            a = 0.5 * (a + a[::-1])

            conv = _conv_fft(a, size=conv_fft_size)
            w = _softmax_weights(conv, lam)
            grad_a = _softmax_gradient(a, w, size_corr=corr_fft_size)

            dot = np.dot(a, grad_a)
            grad_u = a * (grad_a - dot)

            m = 0.9 * m + 0.1 * grad_u
            v = 0.999 * v + 0.001 * (grad_u ** 2)

            it_f = float(it)
            m_hat = m / (1.0 - 0.9 ** it_f)
            v_hat = v / (1.0 - 0.999 ** it_f)

            lr = base_lr * (0.99995 ** it)
            u -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

            lam *= lam_factor

            if it % 30 == 0:
                val = 2.0 * n * conv.max()
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    print(f"[Fine‑tune] it={it:05d}  new best = {val:.6f}")

    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform vector (should never happen)
        n = 3000
        best_seq = np.full(n, 1.0 / n)

    print(f"Final best objective = {best_val:.6f} (length {len(best_seq)})")
    # ``evaluate_sequence`` expects a plain Python list
    return best_seq.tolist()
