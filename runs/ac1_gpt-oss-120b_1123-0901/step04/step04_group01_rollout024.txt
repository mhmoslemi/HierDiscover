projected_subgradient_search.py

A simple projected sub‑gradient descent with occasional stochastic shakes
and a final local hill‑climb.  The algorithm works directly on the
probability simplex (∑a_i = 1, a_i ≥ 0) instead of the soft‑max
parameterisation used previously.  For each iteration we

* compute the linear self‑convolution a*a (FFT based, O(n log n)),
* identify the lag(s) where the maximum is attained,
* build a sub‑gradient of the max‑function by averaging the gradients of
  the active lags,
* take a step in the opposite direction,
* project the resulting vector back onto the simplex.

A small random perturbation is added when the optimisation has not made
progress for a while (stagnation handling).  After the main loop a very
cheap hill‑climbing refinement swaps small amounts of mass between two
coordinates.  Multiple problem sizes and restarts are tried; the best
sequence found (lowest value of 2·n·max(a*a)/(∑a)²) is returned.

The routine complies with the competition constraints:
* only numpy, time and built‑in modules are used,
* all helper functions are top‑level (no closures),

import time

def full_convolution(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """
    Linear (non‑circular) convolution a*a using the FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    if fft_len is None:
        fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


def cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) where
        (w ★ a)[i] = Σ_k w[k]·a[i + k – (n‑1)]   for i = 0 … n‑1 .
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ_i x_i = 1}.
    Implementation follows Duchi et al. (2008), O(n log n).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform vector
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w
