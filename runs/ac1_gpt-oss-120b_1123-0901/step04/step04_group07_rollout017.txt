"""
propose_candidate
-----------------

Search for a non‑negative step‑function that minimises
``evaluate_sequence``.  The algorithm is an implementation of the
alternating linear‑program (LP) scheme described in the problem
statement:

1.  Start from a feasible vector `f` (ℓ₁‑norm = √(2 n)).
2.  Let `M = max(f∗f)`.  The convolution `f∗g` is linear in `g`,
    so we can increase the total mass of a new vector `g` while
    keeping all entries of `f∗g` ≤ M by solving the LP

       maximise   Σ_j g_j
       subject to A(g) ≤ M  ,   g ≥ 0

    where `A(g) = f ∗ g` and the constraints are kept only for the
    (few) indices where `f∗f` attains its maximum (the “active set”).
3.  Rescale `g` to the required ℓ₁‑norm and replace
    `f ← (1‑t)·f + t·g` with a small step size `t`.
4.  Repeat the LP step until no improvement is seen or the time
    budget runs out.

The active‑set restriction makes each LP tiny (often a single linear
constraint), so thousands of iterations are cheap.  Several lengths
(`n ≈ 2000‑4000`) are tried and the best sequence (according to the
exact objective) is returned.

If `cvxpy` is unavailable the code falls back to a simple uniform
vector, which still yields the guaranteed bound of 2.
"""

import time
import math
import numpy as np
import warnings

# ----------------------------------------------------------------------
# Optional import of cvxpy; if unavailable we fall back to the uniform
# vector (still a valid solution with objective = 2).
try:
    import cvxpy as cp
    _HAS_CVXPY = True
except Exception:  # pragma: no cover
    _HAS_CVXPY = False
    warnings.warn("cvxpy could not be imported – falling back to a uniform solution.")


# ----------------------------------------------------------------------
def _next_pow_two(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _max_convolution(a: np.ndarray) -> float:
    """Return max_k (a∗a)[k] using an FFT."""
    n = a.shape[0]
    Nfft = _next_pow_two(2 * n - 1)
    A = np.fft.rfft(a, n=Nfft)
    conv = np.fft.irfft(A * A, n=Nfft)[: 2 * n - 1]
    return float(conv.max())


def _active_conv_matrix(f: np.ndarray, active_idx: np.ndarray) -> np.ndarray:
    """
    Build the (small) matrix A such that (A @ g)[r] = (f∗g)[active_idx[r]].
    Each active row only touches O(n) entries, so the routine is cheap.
    """
    n = f.shape[0]
    m = active_idx.size
    if m == 0:
        return np.empty((0, n), dtype=np.float64)

    A = np.zeros((m, n), dtype=np.float64)
    for row, k in enumerate(active_idx):
        # indices i of f that contribute to (f∗g)[k]
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i = np.arange(i_start, i_end + 1, dtype=np.int64)
        j = k - i                     # corresponding indices of g
        A[row, j] = f[i]              # f[i] multiplies g[j]
    return A


def _lp_improve(
    f: np.ndarray,
    target_sum: float,
    deadline: float,
    max_iters: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Perform the LP‑based improvement described above.
    Returns the best vector found and its max‑convolution value.
    """
    n = f.shape[0]
    # pre‑allocate helpers
    best_f = f.copy()
    best_M = _max_convolution(f)
    t = 0.2                     # initial interpolation weight

    # tolerance for the active set (absolute)
    active_tol = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ----- current max‑convolution -----
        M = _max_convolution(f)

        # ----- build the active set (indices where the conv attains its maximum) -----
        # we recompute the full convolution only once per iteration
        nfft = _next_pow_two(2 * n - 1)
        A_fft = np.fft.rfft(f, n=nfft)
        conv_full = np.fft.irfft(A_fft * A_fft, n=nfft)[: 2 * n - 1]
        max_val = conv_full.max()
        active_idx = np.where(conv_full >= max_val - active_tol)[0]

        # ----- linear program: maximise sum(g) s.t. (f∗g)[active] ≤ M, g ≥ 0 -----
        # Build a tiny constraint matrix containing only the active rows.
        A_active = _active_conv_matrix(f, active_idx)

        if A_active.shape[0] == 0 or not _HAS_CVXPY:
            # No active constraints or cvxpy missing → stop improving.
            break

        g = cp.Variable(n, nonneg=True)
        constraints = [A_active @ g <= M + 1e-12]
        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

        # ECOS is the default fast conic solver that works for LPs.
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000,
                       abstol=1e-8, reltol=1e-8, feastol=1e-8, verbose=False)
        except Exception:          # pragma: no cover
            # If ECOS fails for any reason, fall back to the current f.
            break

        if g.value is None:
            # Infeasible or solver failure – stop.
            break

        g_vec = np.asarray(g.value).reshape(-1)

        # ----- rescale g to the prescribed ℓ1‑norm -----
        sum_g = g_vec.sum()
        if sum_g <= 0.0:
            break
        g_scaled = g_vec * (target_sum / sum_g)

        # ----- convex combination step -----
        f = (1.0 - t) * f + t * g_scaled

        # ----- keep feasibility (non‑negative & correct ℓ1) -----
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s > 0.0:
            f = f * (target_sum / s)

        # ----- keep the best vector seen so far -----
        new_M = _max_convolution(f)
        if new_M < best_M - 1e-12:
            best_M = new_M
            best_f = f.copy()
            print(f"[len {n:5d} iter {it:5d}] new best max‑conv = {best_M:.6f}")

        # ----- schedule updates -----
        if it % 100 == 0:
            t = max(t * 0.9, 0.01)          # slowly shrink the step size

        # occasional jitter to escape very flat plateaus
        if it % 250 == 0:
            jitter = np.random.randn(n) * (target_sum * 1e-4 / n)
            f = np.maximum(f + jitter, 0.0)
            f = f * (target_sum / f.sum())

    return best_f, best_M


def _make_initial_vectors(n: int, target_sum: float):
    """A small pool of diverse starting vectors (all ℓ₁‑norm = target_sum)."""
    rng = np.random.default_rng()
    vectors = []

    # 1) uniform
    vectors.append(np.full(n, target_sum / n))

    # 2) random Dirichlet
    g = rng.gamma(1.0, 1.0, size=n)
    vectors.append(g / g.sum() * target_sum)

    # 3) exponential decay (random rate)
    rate = rng.uniform(0.0005, 0.01)
    x = np.exp(-rate * np.arange(n))
    vectors.append(x / x.sum() * target_sum)

    # 4) reversed exponential
    vectors.append(x[::-1] / x.sum() * target_sum)

    # 5) Hann window (smooth cosine)
    hann = 0.5 - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1))
    vectors.append(hann / hann.sum() * target_sum)

    # 6) small random perturbation of the uniform vector
    pert = np.full(n, target_sum / n)
    pert += rng.normal(scale=0.01 * target_sum / n, size=n)
    pert = np.maximum(pert, 0.0)
    pert = pert / pert.sum() * target_sum
    vectors.append(pert)

    return vectors

def propose_candidate(seed: int = 12345, budget_s: float = 1000, **kwargs):
    """
    Search for a step‑function (non‑negative vector) that minimises
    ``evaluate_sequence``.  The routine runs for at most ``budget_s`` seconds
    and returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5.0          # safety margin

    # ------------------------------------------------------------------
    # Use the previously discovered construction if it exists.
    try:
        base = np.array(height_sequence_1, dtype=float)       # type: ignore
    except Exception:
        base = None

    # ------------------------------------------------------------------
    # Choose a handful of lengths (never exceed 8000, never below 300).
    if base is not None:
        n0 = base.shape[0]
    else:
        n0 = 2100                     # historic default
    length_factors = [1.0, 1.5, 2.0]
    lengths = sorted({int(n0 * f) for f in length_factors})
    lengths = [max(300, min(l, 8000)) for l in lengths]

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Main optimisation loop over the selected lengths.
    for n in lengths:
        if time.time() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)
        init_vectors = _make_initial_vectors(n, target_sum)

        # If a base vector of a different size exists, interpolate it.
        if base is not None and base.shape[0] != n:
            old_x = np.linspace(0.0, 1.0, base.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            interp = np.interp(new_x, old_x, base)
            interp = np.maximum(interp, 0.0)
            interp = interp / interp.sum() * target_sum
            init_vectors.append(interp)

        # Allocate a modest number of LP iterations per start vector.
        max_iters_per_start = 2000

        print(f"[run] length={n}, target_sum={target_sum:.3f}, starts={len(init_vectors)}")

        for idx, f0 in enumerate(init_vectors):
            if time.time() >= deadline:
                break
            print(f"  start {idx + 1}/{len(init_vectors)} for n={n}")

            # --------------------------------------------------------------
            # If cvxpy is unavailable we simply keep the uniform vector.
            if not _HAS_CVXPY:
                f_opt = f0.copy()
                M_opt = _max_convolution(f_opt)
            else:
                f_opt, M_opt = _lp_improve(
                    f=f0.copy(),
                    target_sum=target_sum,
                    deadline=deadline,
                    max_iters=max_iters_per_start,
                )

            # Evaluate the exact objective (after normalisation it equals M_opt)
            cur_score = M_opt
            if cur_score < best_score:
                best_score = cur_score
                best_seq = f_opt.copy()
                print(f"[global] new best score = {best_score:.6f} (len={n})")

    # ------------------------------------------------------------------
    # Fallback – return a uniform vector if everything else failed.
    if best_seq is None:
        n = n0
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n)
        best_score = _max_convolution(best_seq)
        print("[fallback] using uniform sequence")

    print(f"[done] best score = {best_score:.6f}, length = {best_seq.shape[0]}")
    return best_seq.tolist()
