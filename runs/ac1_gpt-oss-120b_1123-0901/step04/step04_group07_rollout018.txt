"""
Propose a non‑negative step function (vector) that minimizes the
``evaluate_sequence`` objective.

Algorithm
---------
1. **Initialisation**
   * Choose a modest length (default 2100 – fits comfortably into memory).
   * Build a diverse pool of starting vectors (uniform, Dirichlet,
     exponential shapes, Hann/Blackman windows, an interpolated historic
     construction if available).
   * Normalise every start so that Σ a_i = sqrt(2·n) (the standard normalisation;
     after this the objective equals the maximal entry of the convolution).

2. **Continuous optimisation**
   * Run a projected Adam optimiser on each start.
   * A soft‑max approximation of the max‑convolution is used at the beginning
     (temperature τ proportional to the current max and slowly exponential‑ly
     decayed).  When τ becomes tiny the exact sub‑gradient of the max
     convolution is used.
   * After each step the vector is projected onto the feasible set:
        a_i ≥ 0 and Σ a_i = sqrt(2·n).
   * Occasional Gaussian jitter helps escaping plateaus.
   * The optimiser stops automatically when the global time budget is exhausted.

   *Because Σ a_i = sqrt(2·n) the objective equals the maximal convolution
   entry M, therefore we keep track of the smallest M seen.*

3. **Linear‑programming refinement**
   * After the Adam phase we perform up to three LP refinements.
   * For the current vector f compute M = max (f ∗ f).
   * Build the convolution matrix C (size (2n‑1)×n) such that C g = f ∗ g.
     The constraints C g ≤ M are linear in g.
   * Solve the LP  
        maximise 1ᵀ g subject to g ≥ 0, C g ≤ M
     (CVXPy + ECOS).  The solution g tries to increase the ℓ₁‑norm while
     keeping the convolution with f below the current peak.
   * Rescale g to the required ℓ₁‑norm and mix it with f:
        f ← (1‑t) f + t g   (t = 0.05).
   * Project again onto the feasible set and re‑evaluate M.
   * If this improves the best score we keep the new vector.

4. **Return**
   * The best vector found (as a plain Python list) is returned.
   * If nothing was found a simple uniform vector is used as fallback.
"""

def _softmax_weights(conv: np.ndarray, tau: float):
    """Soft‑max approximation of max(convolution).  Returns weights that sum to 1
    and the smoothed maximum L_τ."""
    maxc = conv.max()
    if tau <= 0.0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w, maxc
    ex = np.exp((conv - maxc) / tau)
    sum_ex = ex.sum()
    w = ex / sum_ex
    Ltau = maxc + tau * np.log(sum_ex)
    return w, Ltau


def _weighted_rev(a: np.ndarray, w: np.ndarray, Nfft: int) -> np.ndarray:
    """Weighted reverse convolution needed for the (soft) max‑gradient."""
    A_fft = np.fft.rfft(a, n=Nfft)
    W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)
    conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
    conv_rev = conv_rev[: len(w) + a.shape[0] - 1]
    idx_start = 2 * a.shape[0] - 2
    rev = conv_rev[idx_start - np.arange(a.shape[0])]
    return rev


def _max_convolution(a: np.ndarray, Nfft: int) -> float:
    """Maximum entry of the full convolution a∗a."""
    A_fft = np.fft.rfft(a, n=Nfft)
    conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * a.shape[0] - 1]
    return float(conv.max())


def _adam_optimize(a: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int,
                   best_seq: np.ndarray,
                   best_score: float) -> tuple[np.ndarray, float]:
    """Projected Adam optimiser with a soft‑max temperature schedule.
    Returns the best vector discovered in this run and its (exact) score.
    """
    target_sum = math.sqrt(2.0 * n)

    # normalise start
    a = a * (target_sum / a.sum())

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr0 = 0.05
    jitter_interval = 5000

    # temperature parameters
    tau_factor = 0.20
    decay_rate = max_iters / 10.0

    Nfft = 1 << ((2 * n - 1).bit_length())

    for iteration in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # exact convolution of a with itself
        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        M = float(conv.max())

        # temperature & soft‑max
        tau = tau_factor * M * math.exp(-iteration / decay_rate)
        if tau < 1e-9:
            mask = conv >= M - 1e-12
            cnt = mask.sum()
            w = mask.astype(float) / max(cnt, 1)
            Ltau = M
        else:
            w, Ltau = _softmax_weights(conv, tau)

        # weighted reverse convolution (gradient of smoothed max)
        rev = _weighted_rev(a, w, Nfft)

        # gradient of the exact objective after normalisation:
        coeff = 4.0 * n / (target_sum * target_sum)          # =2 after normalisation
        const = 4.0 * n * Ltau / (target_sum ** 3)           # projects onto Σa = target_sum
        grad = coeff * rev - const

        # Adam updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        lr = lr0 / (1.0 + 1e-4 * iteration)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto feasible set
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a = a * (target_sum / s)
        else:
            a = np.full_like(a, target_sum / n)

        # occasional jitter
        if iteration % jitter_interval == 0:
            jitter = np.random.randn(n) * (1e-3 * target_sum / n)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        # current exact objective (after normalisation it equals M)
        cur_score = M
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[len {n:5d} iter {iteration:8d}] new best = {best_score:.6f}")

    return best_seq, best_score


def _make_initial_vectors(n: int, target_sum: float) -> list[np.ndarray]:
    """Create a small, diverse set of normalised start vectors."""
    rng = np.random.default_rng()
    init_list = []

    # uniform
    init_list.append(np.full(n, target_sum / n))

    # Dirichlet
    g = rng.gamma(shape=1.0, scale=1.0, size=n)
    init_list.append(g / g.sum() * target_sum)

    # exponential decay
    rate = rng.uniform(0.0005, 0.01)
    x = np.exp(-rate * np.arange(n))
    init_list.append(x / x.sum() * target_sum)

    # reversed exponential
    init_list.append(x[::-1] / x.sum() * target_sum)

    # Hann window
    hann = 0.5 - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1))
    init_list.append(hann / hann.sum() * target_sum)

    # Blackman window
    a0, a1, a2 = 0.42, 0.5, 0.08
    n_arr = np.arange(n)
    black = (a0 - a1 * np.cos(2.0 * np.pi * n_arr / (n - 1))
             + a2 * np.cos(4.0 * np.pi * n_arr / (n - 1)))
    init_list.append(black / black.sum() * target_sum)

    # small random perturbation of uniform
    pert = np.full(n, target_sum / n)
    pert += rng.normal(scale=0.01 * target_sum / n, size=n)
    pert = np.maximum(pert, 0.0)
    pert = pert / pert.sum() * target_sum
    init_list.append(pert)

    return init_list


def _build_conv_matrix(f: np.ndarray) -> sp.csr_matrix:
    """Return the (2n‑1)×n convolution matrix C such that C @ g == f ∗ g."""
    n = f.shape[0]
    rows = 2 * n - 1
    cols = n
    # build by stacking shifted diagonals f[i] on diagonal offset i
    data = []
    offsets = []
    for i, fi in enumerate(f):
        # each diagonal has n entries (all equal to fi)
        data.append(np.full(cols, fi, dtype=float))
        offsets.append(i)
    C = sp.diags(data, offsets, shape=(rows, cols), format="csr")
    return C


def _solve_lp_max_sum(f: np.ndarray,
                      M: float,
                      conv_mat: sp.csr_matrix,
                      deadline: float) -> np.ndarray | None:
    """Solve  max Σ g_i   s.t. g ≥ 0 , conv_mat @ g ≤ M .
    Returns the optimal g (numpy array) or None if the solver fails."""
    n = f.shape[0]
    g = cp.Variable(n, nonneg=True)

    constraints = [conv_mat @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    # very short time limit per LP – we rely on the global deadline
    try:
        prob.solve(solver=cp.ECOS, warm_start=True,
                   max_iters=2000, abstol=1e-6, reltol=1e-6)
    except (cp.error.SolverError, KeyboardInterrupt):
        return None

    if g.value is None:
        return None
    return np.array(g.value, dtype=float)

def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """Search for a non‑negative step‑function that minimises evaluate_sequence.

    Returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin

    # ------------------------------------------------------------------
    # Try to load the historic construction (if the global variable exists)

    except Exception:
        base = None

    # ------------------------------------------------------------------
    # Fixed length (chosen to keep the LP tractable while still powerful)
    n = 2100
    target_sum = math.sqrt(2.0 * n)

    init_vectors = _make_initial_vectors(n, target_sum)

    # add interpolated historic vector if it has a different size
    if base is not None and base.shape[0] != n:
        old_x = np.linspace(0.0, 1.0, base.shape[0])
        new_x = np.linspace(0.0, 1.0, n)
        interp = np.interp(new_x, old_x, base)
        interp = np.maximum(interp, 0.0)
        interp = interp / interp.sum() * target_sum
        init_vectors.append(interp)

    best_seq: np.ndarray | None = None
    best_score = float("inf")      # best (max convolution) seen so far

    # ------------------------------------------------------------------
    # Main optimisation loop over the initial vectors
    for idx, a0 in enumerate(init_vectors):
        if time.time() >= deadline:
            break
        print(f"[run] start {idx + 1}/{len(init_vectors)} length={n}")

        # --- Adam phase -------------------------------------------------
        a, score = _adam_optimize(
            a=a0.copy(),
            n=n,
            deadline=deadline,
            max_iters=5_000_000,          # generous; will stop when time runs out
            best_seq=best_seq if best_seq is not None else a0,
            best_score=best_score,
        )
        # record improvement from Adam
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[after Adam] new global best = {best_score:.6f}")

        # --- Linear‑programming refinement ------------------------------
        for lp_step in range(3):
            if time.time() >= deadline:
                break

            # current max convolution (equals the objective because Σa = sqrt(2n))
            Nfft = 1 << ((2 * n - 1).bit_length())
            M = _max_convolution(a, Nfft)

            # build convolution matrix for the LP
            conv_mat = _build_conv_matrix(a)

            # solve LP that maximises ℓ₁‑norm while keeping f∗g ≤ M
            g = _solve_lp_max_sum(a, M, conv_mat, deadline)
            if g is None:
                print("[LP] solver failed – skipping further refinements")
                break

            # rescale g to the required ℓ₁‑norm
            sum_g = g.sum()
            if sum_g > 0:
                g = g * (target_sum / sum_g)
            else:
                g = np.full_like(a, target_sum / n)

            # gentle mixing (convex combination) and projection
            t = 0.05
            a = (1.0 - t) * a + t * g
            a = np.maximum(a, 0.0)
            a = a * (target_sum / a.sum())

            # evaluate new max after mixing
            M_new = _max_convolution(a, Nfft)
            print(f"[LP step {lp_step}] max conv = {M_new:.6f}")

            if M_new < best_score:
                best_score = M_new
                best_seq = a.copy()
                print(f"[LP] new global best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback in the unlikely case that nothing was found
    if best_seq is None:
        best_seq = np.full(n, target_sum / n)
        best_score = _max_convolution(best_seq, 1 << ((2 * n - 1).bit_length()))
        print("[fallback] using uniform sequence")

    print(f"[done] best score = {best_score:.6f}, length = {best_seq.shape[0]}")
    return best_seq.tolist()
