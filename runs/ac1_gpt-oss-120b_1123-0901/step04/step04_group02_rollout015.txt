def random_distribution(n):
    choice = np.random.choice(['dirichlet', 'geom', 'power'])
    if choice == 'dirichlet':
        return np.random.dirichlet(np.ones(n))
    elif choice == 'geom':
        r = np.random.rand() * 0.9 + 0.05  # r between 0.05 and 0.95
        a = (1 - r) * (r ** np.arange(n))
        a = a / a.sum()
        return a
    else:
        p = np.random.rand() * 3 + 0.5  # p between 0.5 and 3.5
        a = np.arange(1, n+1) ** (-p)
        a = a / a.sum()
        return a

"""
propose_candidate
-----------------
Hybrid optimiser for the “max‑convolution” objective.
The routine first explores the search space with a fast
simulated‑annealing (SA) that updates the full convolution
incrementally (O(n) per move).  After a good point is found,
a short Adam‐based smooth‑max gradient descent is applied
to polish the solution, followed by a tiny random hill‑climb.
All steps respect the global time budget and keep the
vector on the probability simplex (non‑negative, sum = 1).
The code works for any length; the default is n=1024
which is fast enough to perform many millions of SA moves
inside the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures, no lambdas)

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.shape[0] - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Compute (w ★ a)[i] = Σ_k w[k]·a[i+k-(n-1)], i = 0 … n‑1.
    This is the gradient of Σ_k w_k·(a*a)_k divided by 2.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("v must be 1‑D")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero -> uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that
    minimises 2·n·max(a*a)/(sum(a)²).  The function respects the time
    budget and returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Problem size – we concentrate on n=1024 (fast SA) but allow a
    # fallback to a larger length if time permits.
    n = 1024
    best_seq = None          # numpy array, sum ≈ 1
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Initialise a starting point (warm‑start if a historic solution exists)

    def _initial_sequence():
        """Return a random or warm‑started probability vector of length n."""
        # 1) historic sequence if present
        hist = globals().get('height_sequence_1')
        if hist is not None:
            cand = np.asarray(hist, dtype=np.float64)
            if cand.shape[0] >= n:
                cand = cand[:n]
            else:
                cand = np.pad(cand, (0, n - cand.shape[0]))
            cand = np.maximum(cand, 0.0)
            s = cand.sum()
            if s >= 1e-6:
                return cand / s

        # 2) choose a random family
        typ = np.random.choice(['dirichlet', 'geom', 'power'])
        if typ == 'dirichlet':
            return np.random.dirichlet(np.ones(n))
        if typ == 'geom':
            # geometric decay a[i] ∝ r^i, r∈(0.05,0.95)
            r = np.random.rand() * 0.90 + 0.05
            a = (1 - r) * (r ** np.arange(n))
            a = a / a.sum()
            return a
        # power‑law
        p = np.random.rand() * 3.0 + 0.5          # p ∈ (0.5, 3.5)
        a = (np.arange(1, n + 1) ** (-p))
        a = a / a.sum()
        return a

    # ------------------------------------------------------------------
    # Incremental convolution update (used by SA)

    def _apply_move(a_vec, conv_vec, i, j, delta):
        """
        Move 'delta' mass from index i to index j.
        Returns new vectors (a_new, conv_new).  All operations are O(n).
        """
        a_new = a_vec.copy()
        conv_new = conv_vec.copy()

        # update the probability vector
        a_new[i] -= delta
        a_new[j] += delta

        # update convolution entries
        # shift of the whole a vector by i and by j
        conv_new[i:i + n] -= 2.0 * delta * a_vec
        conv_new[j:j + n] += 2.0 * delta * a_vec

        # self‑convolution of the delta spikes
        conv_new[2 * i] += delta * delta
        conv_new[2 * j] += delta * delta
        conv_new[i + j] -= 2.0 * delta * delta

        # numerical safety – enforce non‑negative entries
        np.maximum(conv_new, 0.0, out=conv_new)
        return a_new, conv_new

    # ------------------------------------------------------------------
    # Simulated annealing phase ------------------------------------------------

    # Choose a starting point
    a = _initial_sequence()
    conv = np.convolve(a, a)                    # exact linear convolution
    cur_val = 2.0 * n * conv.max()

    # init global best
    best_seq = a.copy()
    best_val = cur_val

    # temperature schedule
    T = max(1e-4, cur_val * 0.05)               # start a bit above zero
    cooling = 0.99997                           # exponential cooling factor

    # move size – fraction of the source mass
    max_frac = 0.2

    iteration = 0
    last_report = time.time()
    while time.time() < deadline:
        # pick two distinct indices
        i, j = np.random.choice(n, 2, replace=False)
        # choose a mass to transfer (≤ a[i])
        if a[i] <= 0.0:
            continue
        delta = a[i] * max_frac * np.random.rand()
        if delta <= 0.0:
            continue

        # apply move tentatively
        a_new, conv_new = _apply_move(a, conv, i, j, delta)
        new_val = 2.0 * n * conv_new.max()

        # Metropolis acceptance
        if new_val < cur_val:
            accept = True
        else:
            prob = np.exp(-(new_val - cur_val) / max(T, 1e-12))
            accept = np.random.rand() < prob

        if accept:
            a, conv, cur_val = a_new, conv_new, new_val
            # keep best ever seen
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                print(f"[SA iter={iteration}] new best: {best_val:.6f}")

        # temperature decay
        T = max(1e-12, T * cooling)

        # occasional status update
        iteration += 1
        if iteration % 10000 == 0:
            now = time.time()
            print(f"[SA] iter={iteration:,}  cur={cur_val:.6f}  best={best_val:.6f}  T={T:.2e}  time={now - (deadline - budget_s):.1f}s")
            if now > deadline:
                break

    # ------------------------------------------------------------------
    # Adam‑based smooth‑max gradient descent (fine tuning)

    # reuse the best SA point
    a = best_seq.copy()
    # preparation for Adam
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.10                      # slightly larger than in the reference code
    lam = 0.5
    lam_factor = 1.1

    fft_len_conv = _next_pow2(2 * n - 1)
    fft_len_corr = _next_pow2(3 * n)   # safe length for cross‑correlation

    # function that returns the exact objective (no scaling needed)
    def _exact_obj(a_vec):
        conv_full = _full_convolution(a_vec, fft_len_conv)
        return 2.0 * n * conv_full.max()

    max_adam_iters = 15000
    t = 0
    last_improve = time.time()

    while time.time() < deadline and t < max_adam_iters:
        # smooth‑max weights for the current convolution
        conv_full = _full_convolution(a, fft_len_conv)
        max_c = conv_full.max()
        w = np.exp(lam * (conv_full - max_c))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(conv_full, 1.0 / conv_full.shape[0])
        else:
            w /= w_sum

        # gradient of the smooth surrogate
        grad = 2.0 * _cross_correlation(w, a, fft_len_corr)

        # Adam update (before projection)
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_candidate = a - step
        a_candidate = _project_simplex(a_candidate)

        cand_val = _exact_obj(a_candidate)

        # accept if better (always keep the better point)
        if cand_val < best_val:
            best_val = cand_val
            best_seq = a_candidate.copy()
            print(f"[Adam iter={t}] improved best to {best_val:.6f}")
            last_improve = time.time()

        # update current point regardless of improvement (standard Adam)
        a = a_candidate

        # temperature schedule for the smooth‑max (approaches true max)
        lam = min(lam * lam_factor, 1e6)

        # stagnation handling – small random shake
        if time.time() - last_improve > 4.0:
            noise = (np.random.rand(n) - 0.5) * 5e-3
            a = _project_simplex(a + noise)
            last_improve = time.time()
            print(f"[Adam] shake at iter={t}")

    # ------------------------------------------------------------------
    # Final tiny local random refinement (hill climbing)

    a = best_seq.copy()
    conv = np.convolve(a, a)
    cur_val = 2.0 * n * conv.max()
    refine_deadline = deadline
    while time.time() < refine_deadline:
        i, j = np.random.choice(n, 2, replace=False)
        # move a tiny fraction (≤0.5 % of the donor)
        delta = a[i] * 5e-3 * np.random.rand()
        if delta <= 0.0:
            continue
        a_new, conv_new = _apply_move(a, conv, i, j, delta)
        new_val = 2.0 * n * conv_new.max()
        if new_val < cur_val:
            a, conv, cur_val = a_new, conv_new, new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = a.copy()
                print(f"[refine] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Safety net – if nothing was found, fall back to uniform
    if best_seq is None or best_val == float("inf"):
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = _exact_obj(best_seq)
        print(f"Fallback uniform sequence, value = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Report the final objective as computed by the external evaluator
    # (evaluate_sequence is assumed to be already imported)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Final reported score by evaluate_sequence: {final_score:.6f}")
    except Exception:
        # If the external function is unavailable we simply report our own value
        print(f"Final internal score (2·n·maxconv): {best_val:.6f}")

    # Return a plain Python list as required by the evaluation harness
    return best_seq.tolist()
