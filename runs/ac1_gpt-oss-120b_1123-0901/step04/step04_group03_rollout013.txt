def solve_lp_max_sum(f, M, init_active, max_cuts=50):
    n = len(f)
    # Ensure f positive
    eps = 1e-12
    f = np.maximum(f, eps)
    active = set(int(k) for k in init_active)
    # we will solve iteratively
    while True:
        if len(active) == 0:
            # add one arbitrary constraint
            active.add(0)
        # Build A matrix
        rows = []
        for k in sorted(active):
            start = max(0, k - (n - 1))
            end = min(k, n - 1)
            row = np.zeros(n, dtype=np.float64)
            if start <= end:
                row[start:end+1] = f[start:end+1]
            rows.append(row)
        A = np.vstack(rows) if rows else np.empty((0, n))
        b = np.full(A.shape[0], M, dtype=np.float64)

        c = -np.ones(n, dtype=np.float64)   # maximize sum

        bounds = [(0, 10.0) for _ in range(n)]  # upper bound to keep problem bounded

        # call LINPROG
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs', options={'presolve': True})
        if not res.success:
            # fallback: uniform 
            return np.full(n, 1.0 / n, dtype=np.float64)
        g = res.x

        # compute f * g
        conv_fg = linear_convolution(f, g)  # using FFT

        # find violations > M + tolerance
        tol = 1e-12
        viol = np.where(conv_fg > M + tol)[0]
        # filter new
        new = [int(k) for k in viol if k not in active]
        if not new or len(active) >= max_cuts:
            # done
            return g
        else:
            active.update(new)

"""
Improved step‑function optimizer.

The algorithm alternates two complementary phases:

1. **Linear‑programming equalisation** –  given the current
   distribution `f` we compute M = max(f*f).  Keeping this bound fixed,
   we solve a small linear program (LP) that maximises the total mass
   Σ gᵢ under the linear constraints (f * g)[k] ≤ M.  The LP is solved
   by a cutting‑plane scheme that starts from the active indices where
   the maximum is attained and adds any newly violated constraints.
   The resulting `g` is renormalised to unit sum and mixed with `f`.

   This step moves probability mass from high‑density regions of `f`
   to low‑density ones and provably never increases the current bound
   M.  Repeating it quickly drives the self‑convolution towards a
   flatter shape.

2. **Simple (1+1) Evolution Strategy fine‑tuning** – after the LP
   phase a few hundred Gaussian mutations are applied to the best
   sequence found so far.  Mutations are accepted only when they
   improve the exact objective.  This catches local refinements that
   the LP‑only process cannot see.

The whole search respects a total time budget (default 1000 s) and
works for any length up to a few × 10⁴.  It warm‑starts from the
global variable ``height_sequence_1`` if available and keeps the best
sequence discovered across all tested lengths.
"""

import time
import numpy as np

# SciPy is optional – the LP phase is only used when it is present.
try:
    from scipy.optimize import linprog
    _scipy_available = True
except Exception:               # pragma: no cover
    _scipy_available = False


# ----------------------------------------------------------------------
# Convolution utilities (FFT based, real‑valued)
# ----------------------------------------------------------------------
def _next_pow_two(x: int) -> int:
    """Small helper – next power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def linear_convolution(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear (non‑circular) convolution a * b using real FFT.
    Both inputs must have the same length n.  Returns an array of length
    2·n‑1.
    """
    n = a.shape[0]
    L = _next_pow_two(2 * n - 1)
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b, L)
    conv = np.fft.irfft(fa * fb, L)[: 2 * n - 1]
    return conv.real


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Convenient wrapper for a*a."""
    return linear_convolution(a, a)


def conv_max(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of a."""
    return full_convolution(a).max()


def compute_objective(a: np.ndarray) -> float:
    """
    Exact objective used by `evaluate_sequence` – 2·n·max(a*a) / (sum(a)²).
    The function works for any non‑negative vector; the caller must ensure
    sum(a) > 0.
    """
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return 2.0 * a.shape[0] * conv_max(a) / (s * s)


# ----------------------------------------------------------------------
# LP sub‑routine – maximise Σ gᵢ under (f * g) ≤ M
# ----------------------------------------------------------------------
def _build_constraint_matrix(f: np.ndarray, indices: list[int]) -> np.ndarray:
    """
    Build the matrix A (|indices| × n) of the linear constraints
    (f * g)[k] ≤ M for the selected convolution indices `indices`.
    Row k contains f on the interval where it participates in the sum.
    """
    n = f.shape[0]
    rows = []
    for k in indices:
        start = max(0, k - (n - 1))
        end = min(k, n - 1)
        row = np.zeros(n, dtype=np.float64)
        if start <= end:
            row[start:end + 1] = f[start:end + 1]
        rows.append(row)
    if rows:
        return np.vstack(rows)
    else:                         # empty – should never happen
        return np.empty((0, n), dtype=np.float64)


def solve_lp_max_sum(f: np.ndarray, M: float,
                     init_active: np.ndarray,
                     max_cuts: int = 50) -> np.ndarray:
    """
    Linear programme:
        maximise   Σ gᵢ
        subject to (f * g)[k] ≤ M   for all k,
                   gᵢ ≥ 0, gᵢ ≤ 10   (upper bound keeps the LP bounded).

    The programme is solved by a cutting‑plane loop that starts from the
    indices `init_active` (where f*f attains its maximum) and adds any
    newly violated constraints until none remain or `max_cuts` constraints
    have been added.
    """
    n = f.shape[0]
    eps = 1e-12
    f = np.maximum(f, eps)               # avoid exact zeros – keeps LP bounded

    active = set(int(k) for k in init_active)
    # safety – at least one constraint
    if not active:
        active.add(0)

    # variable bounds: 0 ≤ gᵢ ≤ 10 (large enough for all practical purposes)
    var_bounds = [(0.0, 10.0) for _ in range(n)]
    c = -np.ones(n, dtype=np.float64)   # maximise Σ gᵢ  → minimise -Σ gᵢ

    while True:
        # --------------------------------------------------------------
        # Build the linear constraints for the currently active set
        # --------------------------------------------------------------
        A = _build_constraint_matrix(f, sorted(active))
        b = np.full(A.shape[0], M, dtype=np.float64)

        # --------------------------------------------------------------
        # Solve the LP
        # --------------------------------------------------------------
        res = linprog(c, A_ub=A, b_ub=b,
                      bounds=var_bounds,
                      method='highs',
                      options={'presolve': True, 'disp': False})
        if not res.success:                           # pragma: no cover
            # fallback: return a uniform distribution
            return np.full(n, 1.0 / n, dtype=np.float64)

        g = res.x

        # --------------------------------------------------------------
        # Check for violated constraints (f * g) ≤ M
        # --------------------------------------------------------------
        conv_fg = linear_convolution(f, g)
        viol = np.where(conv_fg > M + 1e-12)[0]
        new = [int(k) for k in viol if k not in active]

        if not new or len(active) >= max_cuts:
            return g
        active.update(new)


# ----------------------------------------------------------------------
# (1+1) Evolution Strategy – fine‑tuning
# ----------------------------------------------------------------------
def evolve_es(a0: np.ndarray,
              deadline: float,
              max_iter: int = 20000,
              init_sigma: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Simple (1+1) ES that perturbs `a0` with isotropic Gaussian noise.
    The best solution encountered before `deadline` is returned.
    """
    n = a0.shape[0]
    eps = 1e-12
    a = np.maximum(a0, eps)
    a /= a.sum()
    val = compute_objective(a)
    best_a, best_val = a.copy(), val
    sigma = init_sigma
    last_improve = time.time()
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        trial = a + sigma * np.random.randn(n)
        trial = np.maximum(trial, 0.0)
        s = trial.sum()
        if s <= 0.0:
            continue
        trial /= s
        val_trial = compute_objective(trial)

        if val_trial < best_val:
            best_val = val_trial
            best_a = trial.copy()
            a = trial            # keep the successful mutant
            sigma = min(sigma * 1.05, 0.2)
            last_improve = time.time()
            print(f"[ES it={it}] new best: {best_val:.6f}")
        else:
            sigma *= 0.99

        # occasional reset if stuck
        if time.time() - last_improve > 5.0:
            a = best_a.copy()
            sigma = init_sigma
            last_improve = time.time()

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the scorer
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises the evaluation function within the given time budget.
    The routine combines an LP‑based equalisation phase with a short
    ES fine‑tuning phase.  It tries several lengths (larger lengths give
    more freedom) and returns the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    best_seq = None
    best_val = np.inf

    # ------------------------------------------------------------------
    # Helper to obtain a reasonable start vector for a given length
    # ------------------------------------------------------------------
    def initial_candidate(n: int) -> np.ndarray:
        # use the historic construction if present
        if "height_sequence_1" in globals():
            hs = np.asarray(height_sequence_1, dtype=np.float64)
            if hs.size >= n:
                hs = hs[:n]
            else:
                hs = np.pad(hs, (0, n - hs.size))
            hs = np.maximum(hs, 0.0)
            s = hs.sum()
            if s > 0.01:
                hs = hs / s
                return hs
        # fall back to a simple decreasing ramp
        dec = np.arange(n, 0, -1, dtype=np.float64)
        dec = dec / dec.sum()
        return dec

    # ------------------------------------------------------------------
    # Lengths – start with the largest that comfortably fits in memory.
    # We try them sequentially (larger first) and stop when time runs out.
    # ------------------------------------------------------------------
    lengths = [16384, 8192, 4096, 2048, 1024]

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")
        f = initial_candidate(n)

        # ensure strict positivity (required for the LP to stay bounded)
        eps = 1e-12
        f = np.maximum(f, eps)
        f /= f.sum()

        cur_val = compute_objective(f)
        if cur_val < best_val:
            best_val, best_seq = cur_val, f.copy()
            print(f"[init] n={n} value={cur_val:.6f}")

        # --------------------------------------------------------------
        # 1) LP‑based equalisation (only if SciPy is available)
        # --------------------------------------------------------------
        if _scipy_available:
            outer_iter = 0
            while time.time() < deadline and outer_iter < 5000:
                outer_iter += 1
                conv_f = full_convolution(f)
                M = conv_f.max()
                # start from all indices attaining the current maximum
                init_active = np.flatnonzero(np.isclose(conv_f, M, atol=1e-14))
                # solve the cutting‑plane LP
                g = solve_lp_max_sum(f, M, init_active, max_cuts=80)
                if g.sum() == 0.0:
                    break
                g_norm = g / g.sum()

                # mixing coefficient – decays slowly with the iteration count
                t = 0.5 / np.sqrt(outer_iter + 1.0)
                f = (1.0 - t) * f + t * g_norm
                f = np.maximum(f, eps)
                f /= f.sum()

                cur_val = compute_objective(f)
                if cur_val < best_val:
                    best_val, best_seq = cur_val, f.copy()
                    print(f"[LP iter={outer_iter}] new best: {cur_val:.6f}")

                # early exit if progress stalls
                if t < 0.001:
                    break

        # --------------------------------------------------------------
        # 2) (1+1) ES fine‑tuning – a quick local search
        # --------------------------------------------------------------
        if time.time() < deadline:
            # allocate a modest slice of the remaining budget
            es_deadline = min(deadline, time.time() + 30.0)
            a_es, val_es = evolve_es(best_seq.copy(), es_deadline,
                                     max_iter=20000, init_sigma=0.015)
            if val_es < best_val:
                best_val, best_seq = val_es, a_es.copy()
                print(f"[ES final] new best: {val_es:.6f}")

    # ------------------------------------------------------------------
    # Fallback – return a uniform vector if nothing succeeded.
    # ------------------------------------------------------------------
    if best_seq is None:
        fallback_n = 1024
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)
        best_val = compute_objective(best_seq)

    print(f"\nBest value discovered: {best_val:.6f} (length {len(best_seq)})")
    # Convert to plain Python list as required by the external evaluator
    return best_seq.tolist()
