"""
Hybrid LP + Adam optimizer.

*Phase 1 – Linear‑programming step*  
  Starting from a feasible non‑negative sequence `a` (Σa=1) we compute
  M = max((a*a)[k]).  The LP  

        maximise   Σ g_j
        subject to  (a⋆g)[k] ≤ M   for all k,
                    g_j ≥ 0

  is linear because (a⋆g) is linear in `g`.  Solving it gives a vector `g`
  that pushes as much total mass as possible while keeping every
  convolution entry below the current worst value `M`.  Mixing the old
  sequence with the normalised solution, `a ← (1‑t)·a + t·g`,
  strictly reduces (or keeps) the maximum self‑convolution.
  We repeat the LP step many times – it is cheap (≈ 5 s for a few
  hundred iterations on n≈2500).

*Phase 2 – Adam fine‑tuning*  
  After the LP phase we run a few thousand Adam updates on the
  smooth‑max surrogate of `F(a)=2·n·max(a⋆a)` (the same surrogate used in
  the reference code) to polish the result.

The whole routine respects the supplied time budget and returns the
best sequence (plain Python list) found.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray, size: int = None) -> np.ndarray:
    """Linear convolution a⋆a using an FFT of length ``size``."""
    n = a.shape[0]
    if size is None:
        size = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return conv


def _softmax_weights(conv: np.ndarray, lam: float) -> np.ndarray:
    """Soft‑max weights wₖ = exp(lam·convₖ) / Σⱼ exp(lam·convⱼ)."""
    m = conv.max()
    w = np.exp(lam * (conv - m))
    s = w.sum()
    if s == 0.0:
        w.fill(1.0 / w.shape[0])
    else:
        w /= s
    return w


def _softmax_gradient(a: np.ndarray, w: np.ndarray, size_corr: int = None) -> np.ndarray:
    """
    Gradient of the smooth‑max surrogate with respect to the original
    probabilities ``a``.  Closed form: ∇ = 2·(w ★ a).
    """
    n = a.shape[0]
    rev_a = a[::-1]
    if size_corr is None:
        size_corr = 1 << int(np.ceil(np.log2(len(w) + n - 1)))
    fw = np.fft.rfft(w, size_corr)
    frev = np.fft.rfft(rev_a, size_corr)
    corr = np.fft.irfft(fw * frev, size_corr)
    grad = 2.0 * corr[n - 1 : n - 1 + n]
    return grad

    t_mix = 0.5          # convex combination weight
    iter_lp = 0

        g /= sum_g

        # Mix old and new, then re‑normalise (numeric safety)
        a = (1.0 - t_mix) * a + t_mix * g
        a = np.maximum(a, 0.0)
        a /= a.sum()

        # Evaluate the objective
        conv = np.convolve(a, a)
        val = 2.0 * n * conv.max()
        if val + 1e-12 < best_val:            # strict improvement
            best_val = val
            best_seq = a.copy()
            best_time = time.time()
            print(f"[LP {iter_lp:04d}] new best = {best_val:.6f}")

        # Adaptive mixing: shrink t if progress stalls
        if iter_lp % 20 == 0:
            if time.time() - best_time > 5.0:
                t_mix = max(0.1, t_mix * 0.7)
                # small random shake to escape possible plateaus
                a += (np.random.rand(n) - 0.5) * 1e-4
                a = np.maximum(a, 0.0)
                a /= a.sum()
                best_time = time.time()

        # Stop LP phase early if we are very close to the theoretical lower bound
        # (2·n/(2·n‑1) ≈ 1 for large n).  This guard prevents wasting time.
        if best_val < 1.05:
            print("[LP] near‑optimal bound reached, leaving LP phase")
            break

    # ---- Phase 2 : Adam fine‑tuning (smooth‑max surrogate) -----------------
    # Use the best sequence from the LP phase as the starting point.
    a = best_seq.copy()
    u = np.log(a + 1e-30)                      # logits (soft‑max parametrisation)
    m = np.zeros_like(u)
    v = np.zeros_like(u)

    conv_fft_size = 1 << int(np.ceil(np.log2(2 * n - 1)))
    size_corr = 1 << int(np.ceil(np.log2(3 * n)))

    lam = 5.0
    lam_factor = 1.001
    base_lr = 0.02
    it = 0
    max_it = 25000

    while time.time() < deadline and it < max_it:
        it += 1

        # soft‑max to obtain a probability vector (Σa = 1)
        u_shift = u - np.max(u)
        a = np.exp(u_shift)
        a /= a.sum()

        # convolution and smooth‑max weights
        conv = _full_convolution(a, size=conv_fft_size)
        w = _softmax_weights(conv, lam)

        # gradient of the surrogate w.r.t. a
        grad_a = _softmax_gradient(a, w, size_corr=size_corr)

        # map to log‑space (soft‑max Jacobian)
        dot = np.dot(a, grad_a)
        grad_u = a * (grad_a - dot)

        # Adam update
        m = 0.9 * m + 0.1 * grad_u
        v = 0.999 * v + 0.001 * (grad_u ** 2)

        it_f = float(it)
        m_hat = m / (1.0 - 0.9 ** it_f)
        v_hat = v / (1.0 - 0.999 ** it_f)

        lr = base_lr * (0.9999 ** it)
        u -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # increase smoothness
        lam *= lam_factor

        # occasional exact evaluation
        if it % 25 == 0:
            conv = np.convolve(a, a)
            val = 2.0 * n * conv.max()
            if val + 1e-12 < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[Adam {it:05d}] new best = {best_val:.6f}")

        # simple shake if we appear stuck
        if it % 500 == 0 and time.time() - best_time > 3.0:
            u += (np.random.rand(n) - 0.5) * 2e-4
            m.fill(0.0)
            v.fill(0.0)
            best_time = time.time()

    # ----------------------------------------------------------------------
    final_obj = best_val
    print(f"Final best objective = {final_obj:.6f} (length {len(best_seq)})")
    # Convert to plain Python list as required by the evaluator.
    return best_seq.tolist()
