The code below searches for a non‑negative step function (a probability vector) that minimises  

\[
F(a)=\frac{2\,n\;\max (a*a)}{(\sum a)^2},
\]

where `a*a` is the linear self‑convolution of `a`.  
The search is performed in three stages

1. **Coarse parametrised families** –  exponential and power‑law shapes are generated for a range of parameters and the best among them is kept.
2. **Adam‑based optimisation on the simplex** – the best vector from step 1 (or a warm‑start from the supplied global `height_sequence_1`) is refined by stochastic gradient descent using a sub‑gradient of the `max`‐operator.
3. **Local greedy refinement** – tiny mass transfers between two random coordinates are accepted whenever they improve the objective.

The algorithm works for several problem sizes (up to 16384 entries) and stops automatically after the allotted time (≈ 1000 s).  
All numbers are forced to be non‑negative; the simplex constraint is handled by a soft‑max representation.

```python
"""
Proposes a step‑function (non‑negative vector) that minimises
    F(a) = 2·n·max(a*a) / (sum(a)²)

The search proceeds in three phases:
  1. Fast scan of parametrised families (exponential, power‑law, sinusoid).
  2. Adam optimisation on the simplex (logits → soft‑max) with a sub‑gradient
     of the max‑convolution.
  3. Simple greedy refinement (move a small amount of mass between two
     coordinates).

The routine respects the 1000 s budget, prints progress and finally returns
the best sequence found as a plain Python list.
"""

import time
import numpy as np

# ------------------------------------------------------------
# Helper routines (top‑level – no closures)
# ------------------------------------------------------------

def full_convolution(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using the FFT."""
    n = a.shape[0]
    if fft_len is None:
        fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


def cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) equal to
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def objective_value(a: np.ndarray) -> float:
    """Exact objective 2·n·max(a*a) / (sum(a)²)."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max() / (a.sum() ** 2))


def upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Repeat the entries of *seq* to obtain a vector of length *new_n*."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    repeats = new_n // old_n
    remainder = new_n % old_n
    up = np.repeat(seq, repeats)
    if remainder:
        up = np.concatenate([up, seq[:remainder]])
    up = up / up.sum()          # re‑normalise to sum‑one
    return up


# ------------------------------------------------------------
# Main search routine required by the evaluator
# ------------------------------------------------------------

def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs) -> list:
    """
    Run a time‑bounded optimisation and return the best sequence found.
    The function respects the 1000 s limit, prints a short log and finally
    returns a plain Python list of non‑negative floats.
    """
    np.random.seed(seed)
    # safety margin to guarantee clean exit
    deadline = time.time() + budget_s - 0.5

    best_seq = None          # keep the best vector (as numpy array)
    best_val = float("inf")  # its objective value

    # --------------------------------------------------------------------
    # Candidate problem sizes – larger n gives a better asymptotic constant.
    # --------------------------------------------------------------------
    lengths = [1024, 2048, 4096, 8192, 16384]

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")

        # FFT lengths that will be reused many times
        fft_len_conv = 1 << int(np.ceil(np.log2(2 * n - 1)))
        fft_len_corr = 1 << int(np.ceil(np.log2(3 * n)))   # ≥ (2n‑1)+n‑1

        # ------------------------------------------------------------
        # 1) Scan easy parametrised families
        # ------------------------------------------------------------
        def try_candidate(vec):
            """Normalise, evaluate and update the global best."""
            nonlocal best_seq, best_val
            if vec.sum() < 1e-12:
                return
            vec = vec / vec.sum()
            val = objective_value(vec)
            if val < best_val - 1e-12:
                best_val = val
                best_seq = vec.copy()
                print(f"[n={n}] new best from family {val:.6f}")

        # a) exponential a_i ∝ r^i
        for r in np.linspace(0.90, 0.9999, 30):
            if time.time() > deadline:
                break
            expo = np.exp(np.arange(n, dtype=np.float64) * np.log(r))
            try_candidate(expo)

        # b) power‑law a_i ∝ (i+1)^{-p}
        for p in np.linspace(0.1, 2.0, 30):
            if time.time() > deadline:
                break
            power = 1.0 / np.power(np.arange(1, n + 1, dtype=np.float64), p)
            try_candidate(power)

        # c) half‑sinusoid (positive)
        x = np.arange(n, dtype=np.float64)
        sinus = np.sin(np.pi * (x + 0.5) / n)          # strictly > 0
        try_candidate(sinus)

        # d) uniform (baseline)
        try_candidate(np.full(n, 1.0, dtype=np.float64))

        # e) warm start from the global variable if it exists
        if 'height_sequence_1' in globals():
            cand = np.array(height_sequence_1, dtype=np.float64)
            if cand.shape[0] >= n:
                cand = cand[:n]
            else:
                cand = np.pad(cand, (0, n - cand.shape[0]))
            cand = np.maximum(cand, 0.0)
            if cand.sum() > 0.01:
                try_candidate(cand)

        # ------------------------------------------------------------
        # 2) Adam optimisation on the simplex
        # ------------------------------------------------------------
        if best_seq is not None and len(best_seq) == n:
            # initialise logits from the current best (soft‑max trick)
            x = np.log(best_seq + 1e-15)

            m = np.zeros_like(x)
            v = np.zeros_like(x)
            beta1, beta2, eps = 0.9, 0.999, 1e-8
            lr = 0.07                     # starting learning‑rate
            decay_every = 5000
            max_iters = 120000           # plenty – will be cut by the deadline
            last_improve = time.time()
            t = 0

            while time.time() < deadline and t < max_iters:
                # ---- simplex → probabilities (soft‑max) ----
                max_x = np.max(x)
                exp_x = np.exp(x - max_x)          # numerical stability
                a = exp_x / exp_x.sum()

                # ---- evaluate ----
                conv = full_convolution(a, fft_len=fft_len_conv)
                max_c = conv.max()
                val = 2.0 * n * max_c               # sum(a)=1 -> exact objective

                if val < best_val - 1e-12:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} iter={t:06d}]  new best {val:.6f}")

                # ---- sub‑gradient of the max (average over active lags) ----
                tol = max_c * 1e-12 + 1e-15
                active = np.where(conv >= max_c - tol)[0]
                w = np.zeros_like(conv)
                w[active] = 1.0 / active.size

                # gradient w.r.t. a : 2·(w ★ a)
                grad_a = 2.0 * cross_correlation(w, a, fft_len=fft_len_corr)

                # back‑propagation through the soft‑max
                inner = np.dot(a, grad_a)
                grad_x = a * (grad_a - inner)

                # ---- Adam update ----
                t += 1
                m = beta1 * m + (1.0 - beta1) * grad_x
                v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
                m_hat = m / (1.0 - beta1 ** t)
                v_hat = v / (1.0 - beta2 ** t)
                x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

                if t % decay_every == 0:
                    lr *= 0.7

                # ---- stagnation shake ----
                if time.time() - last_improve > 25.0:
                    x += (np.random.rand(n) - 0.5) * 1e-3
                    last_improve = time.time()
                    print(f"[n={n} iter={t}]  shake (stagnation)")

        # ------------------------------------------------------------
        # 3) Simple local greedy refinement
        # ------------------------------------------------------------
        if best_seq is not None and len(best_seq) == n:
            a_local = best_seq.copy()
            local_iters = 4000
            for it in range(local_iters):
                if time.time() > deadline:
                    break
                i, j = np.random.choice(n, 2, replace=False)
                # transfer a small amount from i to j
                eps_move = min(a_local[i], 0.01) * (0.5 + np.random.rand() * 0.5)
                if eps_move <= 0:
                    continue
                a_new = a_local.copy()
                a_new[i] -= eps_move
                a_new[j] += eps_move
                conv = full_convolution(a_new, fft_len=fft_len_conv)
                val = 2.0 * n * conv.max()
                if val < best_val - 1e-12:
                    best_val = val
                    best_seq = a_new.copy()
                    a_local = a_new
                    print(f"[n={n} local {it:04d}] improved to {val:.6f}")

        # ------------------------------------------------------------
        # Upsample the best vector to the next size (if any)
        # ------------------------------------------------------------
        if best_seq is not None and len(best_seq) < lengths[-1]:
            # The next length will be larger; keep the current best for warm‑starting.
            pass

    # --------------------------------------------------------------------
    # Fallback (should never be needed)
    # --------------------------------------------------------------------
    if best_seq is None:
        n_fallback = 1024
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)

    print(f"\nFinal best value: {best_val:.6f} (length {len(best_seq)})")
    # Return a plain Python list as required
    return best_seq.tolist()
```
