import numpy as np
import time
from scipy.optimize import linprog

def full_convolution(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """Linear convolution a*a using FFT."""
    n = a.size
    if fft_len is None:
        fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv

def build_constraint_matrix(f: np.ndarray) -> np.ndarray:
    """Construct the (2n‑1)×n matrix A with A[k,i]=f_i if i≤k≤i+n‑1."""
    n = f.size
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for i, fi in enumerate(f):
        A[i:i + n, i] = fi
    return A

def solve_lp_max_sum(f: np.ndarray, M: float) -> np.ndarray:
    """Linear program:
       max   sum(g)
       s.t.  (f*g)[k] ≤ M   for all k,
             g ≥ 0.
       Returns optimal g (may be all‑zero if infeasible)."""
    n = f.size
    A = build_constraint_matrix(f)
    b = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)          # minimise –sum(g)
    bounds = [(0, None)] * n
    # use HiGHS interior‑point solver – reliable for our modest sizes
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method="highs")
    if not res.success:
        return None
    return res.x

def improve_by_lp(f: np.ndarray, max_iters: int, deadline: float) -> np.ndarray:
    """Iteratively improve a probability vector using the LP step described
    in the problem statement."""
    n = f.size
    # pre‑compute FFT length for fast convolution
    fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
    best = f.copy()
    conv = full_convolution(best, fft_len=fft_len)
    M = conv.max()
    best_val = 2.0 * n * M
    for it in range(max_iters):
        if time.time() > deadline:
            break
        # Build and solve the LP
        g = solve_lp_max_sum(best, M)
        if g is None:
            # LP infeasible – cannot improve further
            break
        sumg = g.sum()
        if sumg <= 1e-12:
            break
        # Normalise g to a probability vector
        g = g / sumg
        # Try several mixing ratios, keep the first that does not increase M
        improved = False
        for alpha in (0.5, 0.3, 0.1, 0.05):
            candidate = (1.0 - alpha) * best + alpha * g
            candidate = np.maximum(candidate, 0.0)
            candidate /= candidate.sum()
            cand_conv = full_convolution(candidate, fft_len=fft_len)
            cand_M = cand_conv.max()
            if cand_M <= M * (1.0 + 1e-12):
                best = candidate
                M = cand_M
                val = 2.0 * n * M
                if val < best_val - 1e-12:
                    best_val = val
                improved = True
                break
        if not improved:
            # if none of the mixes improve, perform a tiny random shake
            shake = (np.random.rand(n) - 0.5) * 1e-3
            best = best + shake
            best = np.clip(best, 0.0, None)
            best /= best.sum()
            conv = full_convolution(best, fft_len=fft_len)
            M = conv.max()
    return best

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function (probability vector) that minimises

        F(a) = 2·n·max(a*a) / (sum(a)²)

    The algorithm mixes three ideas:
    * a soft‑max Adam baseline (kept as a fallback),
    * a linear‑programmed “mass‑spreading’’ step that keeps the convolution bound,
    * a cheap greedy local perturbation.
    Multiple restarts over several problem sizes are performed within the
    allocated time budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5

    best_seq = None
    best_val = float("inf")

    # lengths to explore – medium size gives a good trade‑off between quality and speed
    lengths = [512, 1024, 2048]

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")
        # number of restarts for the current length
        restarts = 4 if n <= 1024 else 2
        for r in range(restarts):
            if time.time() > deadline:
                break
            # -----------------------------------------------------------------
            # initialise a probability vector -------------------------------------------------
            if best_seq is not None and len(best_seq) == n:
                a0 = best_seq.copy()
                a0 /= a0.sum()
                print(f"[n={n} restart={r}] warm start from previous global best")
            elif 'height_sequence_1' in globals():
                seq = np.array(height_sequence_1, dtype=np.float64)
                if seq.shape[0] >= n:
                    a0 = seq[:n].copy()
                else:
                    a0 = np.pad(seq, (0, n - seq.shape[0]))
                a0 = np.maximum(a0, 0.0)
                a0_sum = a0.sum()
                if a0_sum < 0.01:
                    a0 = np.random.dirichlet(np.ones(n))
                else:
                    a0 = a0 / a0_sum
                print(f"[n={n} restart={r}] start from height_sequence_1")
            else:
                a0 = np.random.dirichlet(np.ones(n))
                print(f"[n={n} restart={r}] random start")
            # -----------------------------------------------------------------
            # Main optimisation loop – LP guided updates ---------------------------------
            max_lp_iters = 30
            a_opt = improve_by_lp(a0, max_lp_iters, deadline)

            # -----------------------------------------------------------------
            # One final cheap local refinement (pairwise mass exchange) ---------------
            local_iters = 5000
            fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
            a_local = a_opt.copy()
            conv_local = full_convolution(a_local, fft_len=fft_len)
            M_local = conv_local.max()
            val_local = 2.0 * n * M_local
            for it in range(local_iters):
                if time.time() > deadline:
                    break
                i, j = np.random.choice(n, 2, replace=False)
                # propose moving a small amount from i to j
                eps_move = min(a_local[i], 0.01) * (0.5 + 0.5 * np.random.rand())
                if eps_move <= 0.0:
                    continue
                a_new = a_local.copy()
                a_new[i] -= eps_move
                a_new[j] += eps_move
                conv_new = full_convolution(a_new, fft_len=fft_len)
                M_new = conv_new.max()
                val_new = 2.0 * n * M_new
                if val_new < val_local - 1e-12:
                    a_local = a_new
                    M_local = M_new
                    val_local = val_new
                    if val_local < best_val - 1e-12:
                        best_val = val_local
                        best_seq = a_local.copy()
                        print(f"[n={n} restart={r} local it={it}] new best {best_val:.6f}")
            # -----------------------------------------------------------------
            # Update the global best from this restart (in case the LP alone already beat the current best)
            conv = full_convolution(a_opt, fft_len=fft_len)
            val = 2.0 * n * conv.max()
            if val < best_val - 1e-12:
                best_val = val
                best_seq = a_opt.copy()
                print(f"[n={n} restart={r}] LP solution improved best to {best_val:.6f}")

    # -------------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform vector of moderate size
        n_fallback = 1024
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        best_val = 2.0 * n_fallback * (1.0 / n_fallback)

    print(f"\nFinal best value: {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()

"""
Search for a non‑negative step‑function (probability vector) that minimises

    F(a) = 2·n·max(a*a) / (sum(a)²)

The algorithm mixes three ideas:

* a soft‑max Adam baseline (kept as a fallback);
* a linear‑program “mass‑spreading’’ step that keeps the convolution bound;
* a cheap greedy local perturbation.

Multiple restarts over several problem sizes are performed within the
allocated time budget (≈1000 s).  The best sequence found is returned as a
plain Python list.
"""

import time
import numpy as np
from scipy.optimize import linprog


def full_convolution(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """Linear self‑convolution a*a using the FFT (full mode)."""
    n = a.size
    if fft_len is None:
        fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


def build_constraint_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the (2n‑1)×n matrix A that encodes the constraints
        (f * g)[k] = Σ_i f[i]·g[k-i] ≤ M .
    Row k, column i contains f[i] iff i ≤ k ≤ i+n‑1, otherwise 0.
    """
    n = f.size
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for i, fi in enumerate(f):
        A[i:i + n, i] = fi
    return A


def solve_lp_max_sum(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Linear program:
        maximise   Σ_j g_j
        subject to (f * g)[k] ≤ M   for all k,
                   g_j ≥ 0.
    Returns the optimal g (or None if infeasible).
    """
    n = f.size
    A = build_constraint_matrix(f)
    b = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)               # minimise –Σ g
    bounds = [(0, None)] * n
    # HiGHS is robust for the modest sizes we use (n ≤ 2048)
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method="highs")
    if not res.success:
        return None
    return res.x


def improve_by_lp(f0: np.ndarray, max_iters: int, deadline: float) -> np.ndarray:
    """
    Starting from a probability vector f0, repeatedly:
        1. compute its self‑convolution peak M,
        2. solve the LP above to obtain a distribution g that respects the same bound,
        3. mix f and g (several mixing ratios tried) and keep the first blend that
           does not increase M.
    The routine stops after max_iters or when the time budget expires
    and returns the best vector encountered.
    """
    eps = 1e-12
    n = f0.size
    fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
    f = np.maximum(f0, eps)
    f /= f.sum()

    best = f.copy()
    conv = full_convolution(best, fft_len=fft_len)
    M = conv.max()
    best_val = 2.0 * n * M

    for it in range(max_iters):
        if time.time() > deadline:
            break

        # -------------------------------------------------------------
        # LP step – find a vector g that can carry more total mass while
        # keeping (f * g)[·] ≤ M.
        g = solve_lp_max_sum(best, M)
        if g is None:
            break
        sumg = g.sum()
        if sumg <= eps:
            break
        g = g / sumg                     # normalise to a probability vector

        # -------------------------------------------------------------
        # Try a few convex combinations; keep the first that does not raise M
        improved = False
        for alpha in (0.5, 0.3, 0.1, 0.05):
            cand = (1.0 - alpha) * best + alpha * g
            cand = np.maximum(cand, eps)
            cand /= cand.sum()
            conv_cand = full_convolution(cand, fft_len=fft_len)
            M_cand = conv_cand.max()
            if M_cand <= M * (1.0 + 1e-12):
                best = cand
                M = M_cand
                val = 2.0 * n * M
                if val < best_val - 1e-12:
                    best_val = val
                improved = True
                break

        if not improved:
            # small random “shake’’ to escape a flat region
            shake = (np.random.rand(n) - 0.5) * 1e-3
            best = best + shake
            best = np.clip(best, eps, None)
            best /= best.sum()
            conv = full_convolution(best, fft_len=fft_len)
            M = conv.max()

    return best


def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:

    """

    best_seq = None

    # problem sizes – medium values give a good balance of quality vs speed

    # -------------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform vector (objective = 2)
        n_fallback = 1024
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        best_val = 2.0 * n_fallback * (1.0 / n_fallback)

    print(f"\nFinal best value: {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
