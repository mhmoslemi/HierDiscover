def _differential_evolution(a0, n, deadline, global_best_val):

    # initialize population
    pop = []
    for i in range(pop_size):
        if i == 0:
            pop.append(a0.copy())
        else:
            pop.append(np.random.dirichlet(np.ones(n)))
    # evaluate
    fft_len = _next_pow2(2*n-1)
    vals = [ _objective(p, n, fft_len) for p in pop ]
    best_idx = int(np.argmin(vals))
    best_a = pop[best_idx].copy()
    best_val = vals[best_idx]
    # DE parameters
    F = 0.7
    CR = 0.9
    generation = 0
    while time.time() < deadline and generation < 5000:

            idxs = list(range(pop_size))
            idxs.remove(i)

                else:
                    cross[j] = pop[i][j]
            # project onto simplex
            cross = _project_simplex(cross)
            # evaluate
            val = _objective(cross, n, fft_len)

"""
Propose a low‑value step‑function sequence.

The algorithm is a two‑stage hybrid:
1. **Projected Adam** on the smooth‑max surrogate of the ℓ∞‑norm
   of the self‑convolution (a*a).  Several restarts and a simple
   temperature schedule quickly find a good candidate for a given
   length `n`.
2. **Differential Evolution** (DE) refines the best candidate.
   DE works directly on the simplex (probability vectors) by
   mutating, crossing over and projecting each trial onto the simplex.
   The evolutionary search explores neighbourhoods that Adam may miss
   and typically yields a further reduction of the exact objective.

Both stages respect a global time budget (default 1000 s) and stop
as soon as the target value `1.503` is reached.  The final best
sequence (as a plain Python list) is returned.
"""

import time
import numpy as np
import random
import math

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using an FFT of length `fft_len`."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    # keep only the first 2·len(a)-1 elements (linear convolution)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Compute (w ★ a)[i] = Σ_k w[k]·a[i+k-(n-1)],  i = 0 … n‑1.
    This is the gradient of Σ_k w_k·(a*a)_k divided by 2.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of `v` onto the probability simplex
    {x | x_i ≥ 0, Σ_i x_i = 1}.
    """
    if v.ndim != 1:
        raise ValueError("v must be a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero after projection – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact objective for a probability vector `a` (sum = 1)."""
    conv = _full_convolution(a, fft_len)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray, n: int, deadline: float,
                   best_global_val: float) -> tuple[np.ndarray, float]:
    """
    Projected Adam on the smooth‑max surrogate.
    Returns (best_a, best_val) found before `deadline`.
    """
    fft_len_conv = _next_pow2(2 * n - 1)
    fft_len_corr = _next_pow2(3 * n)          # enough for (2n‑1)+n
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.15
    lam = 0.5
    lam_factor = 1.2
    t = 0
    max_iters = 25000
    best_a = a.copy()
    best_val = _objective(a, n, fft_len_conv)
    last_improve = time.time()

    while time.time() < deadline and t < max_iters:
        conv = _full_convolution(a, fft_len_conv)
        max_c = conv.max()
        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.shape[0])
        else:
            w = w / w_sum

        # gradient of the surrogate
        grad = 2.0 * _cross_correlation(w, a, fft_len_corr)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step
        a = _project_simplex(a_new)

        # temperature schedule (approach true max)
        lam = min(lam * lam_factor, 1e6)

        # exact objective for monitoring
        val = _objective(a, n, fft_len_conv)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[Adam n={n} iter={t:05d}] new best = {val:.6f}")

            if best_val < 1.503:
                return best_a, best_val

        # stagnation handling: gentle random shake
        if time.time() - last_improve > 6.0:
            noise = (np.random.rand(n) - 0.5) * 1e-2
            a = _project_simplex(a + noise)
            last_improve = time.time()
            print(f"[Adam n={n} iter={t}] shake (stagnation)")

        # early exit if we cannot beat current global best
        if best_val >= best_global_val and time.time() > deadline - 30:
            # give a short grace period, then stop this run
            break

    return best_a, best_val


# ----------------------------------------------------------------------
def _differential_evolution(a0: np.ndarray, n: int, deadline: float,
                            best_global_val: float) -> tuple[np.ndarray, float]:
    """
    Simple DE on the simplex.  Population size ≈ 10.
    Returns the best candidate found before `deadline`.
    """
    pop_size = 10
    fft_len = _next_pow2(2 * n - 1)

    # initialise population
    pop = []
    pop.append(a0.copy())               # keep the good seed
    for _ in range(pop_size - 1):
        pop.append(np.random.dirichlet(np.ones(n)))
    vals = [_objective(ind, n, fft_len) for ind in pop]

    best_idx = int(np.argmin(vals))
    best_a = pop[best_idx].copy()
    best_val = vals[best_idx]

    F = 0.7
    CR = 0.9
    generation = 0

    while time.time() < deadline and generation < 5000:
        generation += 1
        for i in range(pop_size):
            # mutation: select three distinct individuals ≠ i
            idxs = list(range(pop_size))
            idxs.remove(i)
            r1, r2, r3 = random.sample(idxs, 3)
            mutant = pop[r1] + F * (pop[r2] - pop[r3])
            mutant = _project_simplex(mutant)

            # crossover
            j_rand = np.random.randint(n)
            trial = np.empty(n, dtype=np.float64)
            for j in range(n):
                if np.random.rand() < CR or j == j_rand:
                    trial[j] = mutant[j]
                else:
                    trial[j] = pop[i][j]

            trial = _project_simplex(trial)
            val = _objective(trial, n, fft_len)

            if val < vals[i]:
                pop[i] = trial
                vals[i] = val
                if val < best_val:
                    best_val = val
                    best_a = trial.copy()
                    print(f"[DE gen={generation} i={i}] new best = {val:.6f}")
                    if best_val < 1.503:
                        return best_a, best_val

        # occasional restart of a random individual to maintain diversity
        if generation % 100 == 0 and time.time() < deadline:
            worst_idx = int(np.argmax(vals))
            pop[worst_idx] = np.random.dirichlet(np.ones(n))
            vals[worst_idx] = _objective(pop[worst_idx], n, fft_len)

    return best_a, best_val

    """

    """
    np.random.seed(seed)
    random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    best_seq = None          # numpy array, sum == 1
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Helper to obtain an initial vector of length n (probability simplex)
    def _initial_vector(n: int) -> np.ndarray:

            else:

            cand = np.maximum(cand, 0.0)
            s = cand.sum()
            if s < 0.01:
                raise ValueError
            return cand / s
        except Exception:
            # fallback: Dirichlet random vector
            return np.random.dirichlet(np.ones(n))

    # ------------------------------------------------------------------
    # Stage 1 – Projected Adam on several lengths
    lengths = [4096, 8192, 16384]   # try a few increasing sizes
    for n in lengths:

        fft_len_conv = _next_pow2(2 * n - 1)

        # number of restarts per length
        restarts = 3

            if time.time() >= deadline:

            if r == 0 and best_seq is not None and best_seq.shape[0] == n:

                print("[restart 0] warm start from global best")

                a0 = _initial_vector(n)

            a_opt, val_opt = _adam_optimize(

        print(f"Fallback uniform sequence, value = {best_val:.6f}")

    return best_seq.tolist()
