The objective to minimise is  

\[
F(a)=\frac{2\,n\;\max (a*a)}{(\sum a)^2},
\qquad a\ge 0,
\]

where “\(*\)” denotes the linear (aperiodic) convolution.
Because the objective is homogeneous of degree 0 we may restrict ourselves to the
probability simplex  

\[
\sum_i a_i = 1,\qquad a_i\ge 0 .
\]

Then `F(a) = 2·n·max(a*a)`.

The algorithm below searches this simplex with a *projected Adam* optimiser.
The gradient of the (non‑smooth) `max` is a **sub‑gradient**:
if  

\[
M = \max_k (a*a)_k,
\qquad\mathcal K = \{k : (a*a)_k = M\},
\]

then  

\[
\partial\,\max (a*a) = 2\; \sum_{k\in\mathcal K}
\bigl(e_k\star a\bigr) ,
\]

where `e_k★a` is a cross‑correlation.  
We compute this sub‑gradient efficiently with FFTs and apply Adam
updates, after which we project back onto the simplex (Euclidean projection).

To avoid stagnation we:

* use many diverse initialisations (Dirichlet, power‑law, the historic
  `height_sequence_1`, and a warm start from the current global best);
* after a period without improvement we add a small random “shake’’;
* run several restarts for each problem size;
* finish with a cheap random hill‑climbing phase that transfers a tiny
  amount of mass between two coordinates.

All heavy work (convolution and cross‑correlation) is performed with
FFT‑based O(n log n) code, so even lengths up to 32768 are affordable.
The routine respects the global time budget (`budget_s`), prints progress,
and finally returns the best sequence found as a plain Python list.

```python
"""
search for a non‑negative step function that minimises
    2·n·max(a*a) / (sum(a)^2)

The algorithm works on the probability simplex (sum(a)=1) and
optimises a *sub‑gradient* of the max‑convolution using Adam
updates together with a Euclidean projection onto the simplex.
Multiple restarts, diverse initialisations, occasional random
perturbations and a final random hill‑climbing refinement are
performed.  The whole search respects a global time budget
(default 1000 s) and returns the best sequence discovered.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2·len(a)-1) using FFT.
    fft_len must be a power of two ≥ 2·len(a)-1.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.shape[0] - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Compute (w ★ a)[i] = Σ_k w[k]·a[i+k-(n-1)],   i = 0 … n‑1.
    This is the gradient of Σ_k w_k·(a*a)_k divided by 2.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        {x | x_i ≥ 0, Σ_i x_i = 1}
    """
    if v.ndim != 1:
        raise ValueError("v must be a 1‑D array")
    n = v.shape[0]

    # Sort v in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries zero after projection – return uniform vector
        return np.full_like(v, 1.0 / n)

    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Find a non‑negative sequence with a low evaluation score.
    The search honour the `budget_s` time limit and prints progress.
    The best sequence (plain Python list) is returned.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None               # will always be normalised to sum‑1
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Helper: exact objective for a *probability* vector (sum = 1)
    def _exact_obj(a: np.ndarray, n: int) -> float:
        conv = _full_convolution(a, _next_pow2(2 * n - 1))
        return 2.0 * n * conv.max()

    # ------------------------------------------------------------------
    # Problem sizes we will try. Larger n gives more freedom.
    lengths = [1024, 2048, 4096, 8192, 16384, 32768]

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n = {n} ===")

        fft_len_conv = _next_pow2(2 * n - 1)
        fft_len_corr = _next_pow2(3 * n)

        # ------------------------------------------------------------------
        # Build a list of initial candidates.
        inits = []

        # 1) uniform
        inits.append(np.full(n, 1.0 / n))

        # 2) Dirichlet(1)
        inits.append(np.random.dirichlet(np.ones(n)))

        # 3) Dirichlet(0.5) – spikier
        inits.append(np.random.dirichlet(np.full(n, 0.5)))

        # 4) power‑law families
        idx = np.arange(1, n + 1, dtype=np.float64)
        for p in (0.5, 1.0, 1.5, 2.0):
            a = 1.0 / np.power(idx, p)
            a /= a.sum()
            inits.append(a)

        # 5) historic construction if available
        if 'height_sequence_1' in globals():
            cand = np.asarray(height_sequence_1, dtype=np.float64)
            if cand.shape[0] >= n:
                cand = cand[:n]
            else:
                cand = np.pad(cand, (0, n - cand.shape[0]))
            cand = np.maximum(cand, 0.0)
            s = cand.sum()
            if s < 0.01:
                cand = np.random.dirichlet(np.ones(n))
            else:
                cand = cand / s
            inits.append(cand)

        # 6) warm start from the global best (if same length)
        if best_seq is not None and best_seq.shape[0] == n:
            inits.append(best_seq.copy())

        # Limit the number of restarts to keep the overall budget reasonable
        max_restarts = min(12, len(inits))

        for restart_idx in range(max_restarts):
            if time.time() > deadline:
                break

            a = inits[restart_idx].copy()
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s < 0.01:
                a = np.random.dirichlet(np.ones(n))
            else:
                a = a / s

            # Adam state (all zeros at the beginning of each restart)
            m = np.zeros(n, dtype=np.float64)
            v = np.zeros(n, dtype=np.float64)
            beta1, beta2, eps = 0.9, 0.999, 1e-8
            lr = 0.25                     # relatively aggressive

            # Stagnation handling
            last_improve = time.time()
            max_iters = 30000

            it = 0
            while time.time() < deadline and it < max_iters:
                # ----- current convolution and active set -----
                conv = _full_convolution(a, fft_len_conv)
                max_c = conv.max()

                # ----- sub‑gradient of the max (average over all active indices) -----
                tol = max_c * 1e-12 + 1e-15
                active = np.where(conv >= max_c - tol)[0]
                w = np.zeros_like(conv)
                if active.size > 0:
                    w[active] = 1.0 / active.size
                grad = 2.0 * _cross_correlation(w, a, fft_len_corr)

                # ----- Adam update (before projection) -----
                it += 1
                m = beta1 * m + (1.0 - beta1) * grad
                v = beta2 * v + (1.0 - beta2) * (grad * grad)
                m_hat = m / (1.0 - beta1 ** it)
                v_hat = v / (1.0 - beta2 ** it)
                step = lr * m_hat / (np.sqrt(v_hat) + eps)
                a_new = a - step

                # ----- projection onto the simplex (keeps sum = 1) -----
                a = _project_simplex(a_new)

                # ----- evaluate exact objective (cheap, FFT based) -----
                conv_new = _full_convolution(a, fft_len_conv)
                val = 2.0 * n * conv_new.max()

                if val < best_val - 1e-10:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} restart={restart_idx} iter={it:05d}] new best: {val:.6f}")

                    # Early exit if we have already crossed the target.
                    if best_val < 1.503:
                        print("Target reached – terminating early.")
                        deadline = time.time()   # stop all loops
                        break

                # ----- stagnation detection -----
                if time.time() - last_improve > 6.0:
                    # small random perturbation + re‑projection
                    noise = (np.random.rand(n) - 0.5) * 1e-3
                    a = _project_simplex(a + noise)
                    last_improve = time.time()
                    print(f"[n={n} restart={restart_idx} iter={it}] shake (stagnation)")

                # optional tiny learning‑rate decay (helps later fine‑tuning)
                lr *= 0.9999

    # ------------------------------------------------------------------
    # Final cheap random hill‑climbing refinement (if time permits)
    if best_seq is not None and time.time() < deadline:
        a = best_seq.copy()
        n = a.shape[0]
        fft_len_conv = _next_pow2(2 * n - 1)
        print("\n--- Local random hill‑climbing refinement ---")
        while time.time() < deadline:
            i, j = np.random.choice(n, 2, replace=False)
            delta = a[i] * 0.005 * np.random.rand()   # move up to 0.5 % of a[i]
            if delta == 0.0:
                continue
            a_prop = a.copy()
            a_prop[i] -= delta
            a_prop[j] += delta
            a_prop = _project_simplex(a_prop)

            conv = _full_convolution(a_prop, fft_len_conv)
            val = 2.0 * n * conv.max()
            if val < best_val - 1e-12:
                best_val = val
                best_seq = a_prop.copy()
                a = a_prop
                print(f"[refine] improved to {val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform vector (should never be reached unless something went wrong)
    if best_seq is None:
        n_fallback = 2048
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        best_val = _exact_obj(best_seq, n_fallback)
        print(f"Fallback uniform sequence, value = {best_val:.6f}")

    print(f"\nBest value found: {best_val:.6f} (length = {best_seq.shape[0]})")
    # Return plain Python list as required by the external evaluator
    return best_seq.tolist()
```
