    rows = 2*n -1

"""
Propose a step‑function (non‑negative vector) that minimises the
``evaluate_sequence`` objective.

Algorithm
---------
* The objective depends only on the *shape* of the vector – it is invariant
  under global scaling.  Hence we optimise a probability vector ``p`` with
  ``∑ p = 1`` and later rescale it to the required L1‑norm ``sqrt(2·n)``.
* Two complementary optimisation stages are used.
  1. **Smooth‑max L‑BFGS** – minimise a smooth approximation of the infinity‑norm
     ``max_k (p∗p)_k`` using the log‑sum‑exp surrogate.  The variables are the
     unconstrained logits ``y`` (soft‑max parametrisation).  A continuation
     schedule reduces the temperature τ and several restarts explore the space.
  2. **Linear‑programming step (Frank–Wolfe style)** – for the current probability
     vector ``p`` we solve a small LP that finds a direction ``g`` with a larger
     L1‑mass while keeping the convolution ``p∗g`` bounded by the current maximum.
     The new vector is a convex combination ``p ← (1‑t)p + t·ĝ`` (``ĝ`` is the
     normalised LP solution).  Repeating this step monotonically reduces the
     maximal convolution value and often yields a modest but reliable improvement.
* The best sequence (scaled to the target L1‑norm) seen during the whole budget
  is returned.

The whole routine respects the 1000 s time budget, prints a few diagnostics
and uses only NumPy / SciPy (both available in the execution environment).
"""

import time
import math
import numpy as np
from typing import List

# --------------------------------------------------------------------------- #
# Optional SciPy imports (fallback to simple Adam if unavailable)
try:
    from scipy.optimize import minimize, linprog
    _SCIPY = True
except Exception:  # pragma: no cover
    _SCIPY = False

# --------------------------------------------------------------------------- #
# Helper functions for the smooth‑max L‑BFGS stage

def _softmax_weights(conv: np.ndarray, tau: float):
    """Soft‑max (stable) over the convolution values.
    Returns weights (sum 1) and the smooth max Lτ."""
    maxc = conv.max()
    if tau <= 0.0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w, maxc
    ex = np.exp((conv - maxc) / tau)
    sum_ex = ex.sum()
    w = ex / sum_ex
    Ltau = maxc + tau * math.log(sum_ex)
    return w, Ltau


def _weighted_rev(a: np.ndarray, w: np.ndarray, Nfft: int):
    """Weighted reverse convolution needed for the gradient.
    Computes rev_i = Σ_k w_k·a_{k‑i}."""
    A_fft = np.fft.rfft(a, n=Nfft)
    W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)
    conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
    conv_rev = conv_rev[:len(w) + a.shape[0] - 1]
    idx_start = 2 * a.shape[0] - 2
    rev = conv_rev[idx_start - np.arange(a.shape[0])]
    return rev


def _conv_full(p: np.ndarray, Nfft: int) -> np.ndarray:
    """Full linear convolution p∗p using FFTs."""
    A_fft = np.fft.rfft(p, n=Nfft)
    conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * p.shape[0] - 1]
    return conv


def _objective_and_grad(y: np.ndarray, tau: float, Nfft: int):
    """Smooth‑max value and its gradient w.r.t. logits y."""
    y_max = np.max(y)
    exp_y = np.exp(y - y_max)          # stable soft‑max numerator
    p = exp_y / exp_y.sum()

    conv = _conv_full(p, Nfft)         # (p∗p)
    w, Ltau = _softmax_weights(conv, tau)

    rev = _weighted_rev(p, w, Nfft)    # Σ_k w_k·p_{k‑i}
    grad_p = 2.0 * rev                 # ∂L / ∂p

    dot = np.dot(p, grad_p)            # Σ_i p_i·grad_p_i
    grad_y = p * (grad_p - dot)        # chain rule through soft‑max
    return Ltau, grad_y


def _run_l_bfgs(y0: np.ndarray, init_tau: float,
                target_sum: float, Nfft: int,
                deadline: float, best_seq: np.ndarray,
                best_score: float) -> (np.ndarray, float):
    """Repeated L‑BFGS optimisation with a temperature schedule."""
    y = y0.copy()
    tau = max(init_tau, 1e-12)
    tau_decay = 0.65
    min_tau = 1e-12

    while time.time() < deadline and tau > min_tau:
        def fun(y_vec):
            L, g = _objective_and_grad(y_vec, tau, Nfft)
            return L, g

        # a slightly larger budget than the original code – still cheap
        opts = {'maxfun': 500, 'maxiter': 500, 'ftol': 1e-12, 'gtol': 1e-12}
        res = minimize(fun, y, method='L-BFGS-B', jac=True, options=opts)
        y = res.x

        # exact objective for this candidate
        y_max = np.max(y)
        exp_y = np.exp(y - y_max)
        p = exp_y / exp_y.sum()
        conv = _conv_full(p, Nfft)
        cur_score = (target_sum ** 2) * conv.max()       # = evaluate_sequence
        if cur_score < best_score:
            best_score = cur_score
            best_seq = (p * target_sum).copy()
            print(f"[L‑BFGS improve] τ={tau:.2e}  score={best_score:.6f}")

        tau *= tau_decay
        # occasional random jitter to leave plateaus
        if np.random.rand() < 0.12:
            y += np.random.randn(y.shape[0]) * 0.02

        if time.time() >= deadline:
            break

    return best_seq, best_score


# --------------------------------------------------------------------------- #
# Linear‑programming improvement stage

def _build_conv_matrix(p: np.ndarray) -> np.ndarray:
    """
    Build the matrix A (size (2n‑1)×n) such that A @ g = p∗g (full linear convolution).
    Row k, column j contains p_{k‑j} when the index is valid, otherwise 0.
    """
    n = p.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = p
    return A


def _lp_direction(p: np.ndarray, M: float):
    """
    Solve   max  Σ g_j   s.t.   p∗g ≤ M,   g ≥ 0.
    Returns the normalised direction ĝ (sum(ĝ)=1) or None on failure.
    """
    n = p.shape[0]
    A = _build_conv_matrix(p)                 # (2n‑1) × n
    b = np.full(A.shape[0], M)

    # maximise Σ g ⇒ minimise –Σ g
    c = -np.ones(n)

    bounds = [(0, None)] * n
    # HiGHS is fast and handles moderate dense problems well.
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        return None
    g = res.x
    s = g.sum()
    if s <= 0:
        return None
    return g / s                               # normalised to a probability vector


def _lp_improve(p: np.ndarray, t0: float, max_iter: int,
                deadline: float) -> np.ndarray:
    """
    Perform a few Frank–Wolfe‑type LP steps.
    Each step solves the LP described above, then mixes the current
    probability vector with the LP direction using a decreasing step size.
    """
    p_cur = p.copy()
    for it in range(max_iter):
        if time.time() >= deadline:
            break
        # current bound
        M = np.convolve(p_cur, p_cur).max()
        g_hat = _lp_direction(p_cur, M)
        if g_hat is None:
            break

        # geometrically decreasing mixing factor
        t = t0 * (0.85 ** it)
        p_cur = (1.0 - t) * p_cur + t * g_hat

        # projection back onto the simplex
        p_cur = np.maximum(p_cur, 0.0)
        s = p_cur.sum()
        if s == 0:
            break
        p_cur /= s

        # occasional tiny jitter to avoid stagnation
        if np.random.rand() < 0.07:
            p_cur += np.random.randn(p_cur.shape[0]) * 1e-4
            p_cur = np.maximum(p_cur, 0.0)
            p_cur /= p_cur.sum()
    return p_cur

# Main entry point

def propose_candidate(seed: int = 42,
                     budget_s: float = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function with the smallest
    ``evaluate_sequence`` score within the given time budget.
    Returns the best sequence as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # small safety margin

    # ------------------------------------------------------------------- #
    # Load a known construction if available, otherwise start uniform.
    try:
        base = np.array(height_sequence_1, dtype=float)   # type: ignore
    except Exception:  # pragma: no cover
        base = np.ones(2100, dtype=float)                # default length

    n = base.shape[0]
    target_sum = math.sqrt(2.0 * n)

    # Normalise to the required L1‑norm (the objective is scale‑invariant)
    base = base * (target_sum / base.sum())

    # Work with probability vector (Σp = 1)
    p0 = base / target_sum
    eps = 1e-12
    y0 = np.log(p0 + eps)          # logits for the soft‑max parametrisation

    # ------------------------------------------------------------------- #
    # Record the best solution observed so far.
    best_seq = base.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] length={n}, initial score={best_score:.6f}")

    # FFT size for the smooth‑max stage
    Nfft = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------------- #
    # Define several diverse starting points.
    starts = [y0]                                          # known solution
    starts.append(np.log(np.random.exponential(scale=1.0, size=n) + eps))  # random Dirichlet
    starts.append(np.zeros(n, dtype=float))               # uniform logits
    starts.append(y0 + np.random.randn(n) * 0.05)         # perturbed known solution

    # ------------------------------------------------------------------- #
    # Run the two‑stage optimisation from each start.
    for idx, y_start in enumerate(starts):
        if time.time() >= deadline:
            break
        print(f"[run] start {idx + 1}/{len(starts)}")

        # --- L‑BFGS (smooth‑max) ------------------------------------
        p_start = np.exp(y_start - np.max(y_start))
        p_start /= p_start.sum()
        conv0 = _conv_full(p_start, Nfft)
        init_tau = max(conv0.max() * 2.0, 1e-12)

        best_seq, best_score = _run_l_bfgs(
            y0=y_start,
            init_tau=init_tau,
            target_sum=target_sum,
            Nfft=Nfft,
            deadline=deadline,
            best_seq=best_seq,
            best_score=best_score,
        )

        # --- LP improvement (Frank‑Wolfe style) ----------------------
        # Convert the best sequence found in this start back to probability.
        p_cur = best_seq / target_sum
        p_cur = np.maximum(p_cur, 0.0)
        p_cur /= p_cur.sum()

        p_cur = _lp_improve(p_cur, t0=0.25, max_iter=40, deadline=deadline)

        # Evaluate the exact objective after the LP phase.
        seq_candidate = (p_cur * target_sum).copy()
        cur_score = evaluate_sequence(seq_candidate.tolist())
        if cur_score < best_score:
            best_score = cur_score
            best_seq = seq_candidate
            print(f"[LP improve] new best score={best_score:.6f}")

    # ------------------------------------------------------------------- #
    # Final sanity check (one more exact evaluation)
    final_score = evaluate_sequence(best_seq.tolist())
    if final_score < best_score:
        best_score = final_score
        best_seq = best_seq.copy()

    print(f"[done] best score = {best_score:.6f}, length = {best_seq.shape[0]}")
    return best_seq.tolist()
