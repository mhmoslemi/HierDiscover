def _smooth_max_weights(conv, alpha):
    max_c = conv.max()
    # shift to avoid overflow
    shift = max_c
    exp_vals = np.exp(alpha * (conv - shift))
    exp_sum = exp_vals.sum()
    if exp_sum == 0:
        w = np.full_like(conv, 1.0 / conv.shape[0])
    else:
        w = exp_vals / exp_sum

"""
propose_candidate
-----------------

Gradient‑based optimizer for the step‑function problem.

The objective (up to an irrelevant scale) is
    F(a) = 2·n·max(a*a) / (∑a)² ,
with a ≥ 0.  Because the value is invariant under scaling,
we can restrict the search to the simplex Σa_i = 1.
The code works on this simplex directly.

*  A smooth approximation of the max is used:
      w_i ∝ exp(α·( (a*a)_i – max(a*a) ))
  When α → ∞ the weights become a one‑hot vector at the true
  maximizing index, so the gradient tends to the exact sub‑gradient.
*  The gradient of the smoothed max w.r.t. a is
      ∇_a  = 2·cross_correlation(w, a) .
  Both convolution and the cross‑correlation are computed with FFTs.
*  Adam updates are performed on the unconstrained vector and each
  iterate is projected back onto the probability simplex.
*  Several initial shapes (uniform, linear, exponential, random Dirichlet)
  and a warm‑start from the best solution found so far are tried for each
  length.  A small set of lengths up to 32768 is explored – larger lengths
  give more degrees of freedom but cost more time.
*  Stagnation is broken by adding a tiny random perturbation.
*  After the main optimisation a fast hill‑climbing “local refinement”
  moves a tiny fraction of mass between two coordinates.

The routine respects a global time budget (default 1000 s) and returns
the best sequence (as a plain Python list) discovered during the run.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
# Helper functions – all defined at top level (no closures, no lambdas)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2·len(a)-1) using FFT.
    ``fft_len`` must be a power of two ≥ 2·len(a)-1.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.shape[0] - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Compute (w ★ a)[i] = Σ_k w[k]·a[i+k-(n-1)], i = 0 … n‑1.
    This is the gradient of Σ_k w_k·(a*a)_k divided by 2.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        {x | x_i ≥ 0, Σ_i x_i = 1}
    """
    if v.ndim != 1:
        raise ValueError("v must be a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _smooth_max_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Returns the soft‑max weights w_i ∝ exp(alpha·(conv_i – max(conv))).
    The subtraction of max(conv) avoids overflow.
    """
    max_c = conv.max()
    shifted = conv - max_c          # now all ≤ 0
    exp_vals = np.exp(alpha * shifted)
    s = exp_vals.sum()
    if s == 0.0:
        # fallback – uniform weights (should never happen for finite alpha)
        return np.full_like(conv, 1.0 / conv.shape[0])
    return exp_vals / s


def exact_objective(a: np.ndarray) -> float:
    """
    Exact evaluation of the objective for a probability vector ``a``.
    Returns 2·n·max(conv) where conv = a*a (linear convolution).
    """
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    conv = _full_convolution(a, fft_len)
    return float(2.0 * n * conv.max())

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low evaluation
    score.  The routine runs for at most ``budget_s`` seconds and
    returns the best sequence found (as a plain Python list).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Problem sizes (larger n gives more freedom but costs more time)
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 4096, 8192, 16384, 32768]

    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length n={n} ===")
        fft_len_conv = _next_pow2(2 * n - 1)
        fft_len_corr = _next_pow2(3 * n)   # enough for (2n-1)+n

        # ------------------------------------------------------------------
        # Build a modest set of diverse initial vectors
        # ------------------------------------------------------------------
        inits: list[np.ndarray] = []

        # 1. Uniform
        inits.append(np.full(n, 1.0 / n))

        # 2. Linear decreasing (triangular)
        tri = np.arange(n, 0, -1, dtype=np.float64)
        tri = tri / tri.sum()
        inits.append(tri)

        # 3. Exponential decay
        exp_decay = np.exp(-np.arange(n, dtype=np.float64) / (n / 8.0))
        exp_decay = exp_decay / exp_decay.sum()
        inits.append(exp_decay)

        # 4-5. Random Dirichlet draws
        for _ in range(2):
            inits.append(np.random.dirichlet(np.ones(n)))

        # 6. Warm‑start from the global best (if same length)
        if best_seq is not None and best_seq.shape[0] == n:
            inits.append(best_seq.copy())

        # ------------------------------------------------------------------
        # Run Adam optimisation from each initializer
        # ------------------------------------------------------------------
        for init_id, a0 in enumerate(inits):
            if time.time() > deadline:
                break

            a = np.maximum(a0, 0.0)
            s = a.sum()
            if s < 1e-12:
                a = np.random.dirichlet(np.ones(n))
            else:
                a = a / s

            # Adam state
            m = np.zeros_like(a)
            v = np.zeros_like(a)
            beta1, beta2, eps = 0.9, 0.999, 1e-8
            lr = 0.2

            # Smooth‑max temperature (α)
            alpha = 0.5
            alpha_factor = 1.12   # geometric increase

            max_iters = 15000
            last_improve = time.time()

            for t in range(1, max_iters + 1):
                if time.time() > deadline:
                    break

                # ----- convolution and soft‑max weights -----
                conv = _full_convolution(a, fft_len_conv)
                w = _smooth_max_weights(conv, alpha)

                # ----- gradient of the smooth surrogate -----
                grad = 2.0 * _cross_correlation(w, a, fft_len_corr)

                # ----- Adam step (pre‑projection) -----
                m = beta1 * m + (1.0 - beta1) * grad
                v = beta2 * v + (1.0 - beta2) * (grad * grad)
                m_hat = m / (1.0 - beta1 ** t)
                v_hat = v / (1.0 - beta2 ** t)
                step = lr * m_hat / (np.sqrt(v_hat) + eps)
                a_new = a - step

                # ----- stay on the simplex -----
                a = _project_simplex(a_new)

                # ----- temperature schedule -----
                alpha = min(alpha * alpha_factor, 1e5)

                # ----- occasional exact evaluation for monitoring -----
                if t % 500 == 0 or t == max_iters:
                    val = exact_objective(a)
                    if val < best_val - 1e-9:
                        best_val = val
                        best_seq = a.copy()
                        last_improve = time.time()
                        print(f"[n={n} init={init_id} iter={t}] new best: {val:.6f}")

                        # early exit if target reached
                        if best_val < 1.503:
                            print("Target reached – terminating early.")
                            return best_seq.tolist()

                # ----- stagnation handling -----
                if time.time() - last_improve > 5.0:
                    # tiny random perturbation before projection
                    noise = (np.random.rand(n) - 0.5) * 1e-3
                    a = _project_simplex(a + noise)
                    last_improve = time.time()
                    print(f"[n={n} init={init_id} iter={t}] shake (stagnation)")

        # ------------------------------------------------------------------
        # End of all restarts for this length
        # ------------------------------------------------------------------

    # ----------------------------------------------------------------------
    # Final random local refinement (if any time left)
    # ----------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        a = best_seq.copy()
        n = a.shape[0]
        fft_len_conv = _next_pow2(2 * n - 1)
        print("\n--- Local random refinement ---")
        while time.time() < deadline:
            i, j = np.random.choice(n, 2, replace=False)
            # Transfer a tiny fraction of mass from i to j
            delta = 0.003 * a[i] * np.random.rand()
            if delta <= 0.0:
                continue
            a_prop = a.copy()
            a_prop[i] -= delta
            a_prop[j] += delta
            a_prop = _project_simplex(a_prop)
            val = exact_objective(a_prop)
            if val < best_val:
                best_val = val
                best_seq = a_prop.copy()
                a = a_prop
                print(f"[refine] improved to {val:.6f}")

    # ----------------------------------------------------------------------
    # Fallback – uniform vector if nothing succeeded.
    # ----------------------------------------------------------------------
    if best_seq is None:
        n_fallback = 2048
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        best_val = exact_objective(best_seq)
        print(f"Fallback uniform sequence, value = {best_val:.6f}")

    # Return the result as a plain Python list (required by the evaluator)
    return best_seq.tolist()
