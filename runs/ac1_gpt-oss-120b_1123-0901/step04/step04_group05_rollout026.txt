    # we can attempt limited number of tries within 0.1s maybe

        # compute sums with existing elements
        ok = True
        for s in S:
            sum_ = idx + s
            if used[sum_]:
                ok = False
                break

        if ok:
            # mark sums

                used[idx + s] = True
            used[2*idx] = True
            S.append(idx)

def generate_sidon_greedy(n):
    # Random greedy: shuffle indices, add if no collisions.
    order = np.random.permutation(n)
    used = np.zeros(2*n-1, dtype=np.bool_)
    S = []
    for idx in order:
        # compute sums with current set
        # quickly compute vector of sums: idx + np.array(S)
        # check used at those positions
        if len(S)==0:
            # check self sum
            if not used[2*idx]:
                used[2*idx] = True
                S.append(idx)
        else:
            sums = idx + np.array(S)
            if used[2*idx] or used[sums].any():
                continue
            # else add
            used[2*idx] = True
            used[sums] = True
            S.append(idx)

"""
Sparse‑Sidon + exact‑max Adam optimizer.

The search works in two phases.
1)  A greedy construction repeatedly builds large Sidon (B₂) sets.
    The set is used as a sparse initial vector (equal weight on the
    chosen indices).  A Sidon set of size m guarantees that every
    unordered pair (i,j) has a unique sum i+j, so for a uniform weight
    the self‑convolution peak is at most 1/m².  To beat the target
    1.503 we need m≈58 for n=2500.
2)  Starting from the best vector found so far we run an Adam optimizer
    on the *exact* max‑convolution objective (no smooth‑max surrogate).
    The gradient of max(conv) is simple: if k is the index where the
    convolution attains its maximum, then ∇ₐ max(conv) = 2·shift(a,k).
    Using logits u (soft‑max parametrisation) guarantees a≥0 and Σa=1.
    Several restarts with different random seeds, Dirichlet, half‑sine,
    exponential and Sidon initialisations are performed inside the given
    time budget (≤1000 s).  The best sequence is returned as a plain
    Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray, size: int = None) -> np.ndarray:
    """Linear self‑convolution a*a via FFT (real‑to‑real)."""
    n = a.shape[0]
    if size is None:
        size = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return conv


def _softmax_gradient(a: np.ndarray, w: np.ndarray, size_corr: int = None) -> np.ndarray:
    """
    Gradient of  Σₖ wₖ·(a*a)[k]  w.r.t. a.
    Closed form: ∇ = 2·(w ★ a) .
    The implementation uses FFT convolution.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    if size_corr is None:
        size_corr = 1 << int(np.ceil(np.log2(len(w) + n - 1)))
    fw = np.fft.rfft(w, size_corr)
    frev = np.fft.rfft(rev_a, size_corr)
    corr = np.fft.irfft(fw * frev, size_corr)
    # slice the part that corresponds to indices 0 … n‑1
    grad = 2.0 * corr[n - 1 : n - 1 + n]
    return grad


def _greedy_sidon_set(n: int) -> list[int]:
    """
    Build a Sidon (B₂) set by a simple greedy algorithm.
    Returns a list of indices 0 ≤ i < n such that all sums i+j are distinct.
    """
    order = np.random.permutation(n)
    used = np.zeros(2 * n - 1, dtype=np.bool_)
    S: list[int] = []

    for idx in order:
        # self‑sum collision?
        if used[2 * idx]:
            continue
        if S:
            sums = idx + np.array(S, dtype=np.int32)
            if used[sums].any():
                continue
        # no collision → accept
        used[2 * idx] = True
        if S:
            used[idx + np.array(S, dtype=np.int32)] = True
        S.append(int(idx))
    return S


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.  The routine respects the supplied time budget
    (default 1000 s) and returns the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.3   # safety margin

    best_val = float("inf")
    best_seq = None
    restart = 0

    # we keep the length fixed – 2500 is a good trade‑off
    lengths = [2500]

    # ------------------------------------------------------------------
    while time.time() < deadline:
        # --------------------------------------------------------------
        # pick length (only one here, but the list can be extended)
        n = lengths[restart % len(lengths)]

        # pre‑compute FFT sizes that stay constant during one inner run
        conv_fft_size = 1 << int(np.ceil(np.log2(2 * n - 1)))
        size_corr = 1 << int(np.ceil(np.log2(3 * n)))

        # ------------------------------------------------------------------
        # ----- random initialisation (several families) ----------
        mode = restart % 8
        if mode == 0:
            # Dirichlet (dense)
            a0 = np.random.dirichlet(np.ones(n))
        elif mode == 1:
            # exponential decay
            r = np.random.uniform(0.2, 0.9)
            a0 = r ** np.arange(n, dtype=np.float64)
            a0 /= a0.sum()
        elif mode == 2:
            # half‑sine (smooth, dense)
            i = np.arange(1, n + 1, dtype=np.float64)
            a0 = np.sin(np.pi * i / (n + 1))
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
        elif mode == 3:
            # ----------------- SIDON INITIALISATION -----------------
            S = _greedy_sidon_set(n)
            m = len(S)
            # fall back to Dirichlet if the set is too tiny
            if m < 20:
                a0 = np.random.dirichlet(np.ones(n))
            else:
                a0 = np.zeros(n)
                a0[S] = 1.0 / m
                print(f"[Restart {restart:03d}] Sidon set size = {m}")
        else:
            # plain uniform random (tiny values are later normalised)
            a0 = np.random.rand(n)
            a0 /= a0.sum()

        # Occasionally seed from the current best (if same length)
        if best_seq is not None and len(best_seq) == n and np.random.rand() < 0.07:
            a0 = np.array(best_seq)

        # --------------------------------------------------------------
        # Log‑space parametrisation guarantees positivity and Σa = 1
        u = np.log(a0 + 1e-30)

        # Adam moments
        m_adam = np.zeros_like(u)
        v_adam = np.zeros_like(u)

        # Adam hyper‑parameters (tuned for the exact‑max landscape)
        base_lr = 0.02
        it = 0
        max_iter = 12000
        last_improve = time.time()

        # --------------------------------------------------------------
        while time.time() < deadline and it < max_iter:
            it += 1

            # ----- probabilities from logits (soft‑max) -----
            u_shift = u - np.max(u)          # improve numerical stability
            a = np.exp(u_shift)
            a_sum = a.sum()
            if a_sum == 0.0:
                a = np.full(n, 1.0 / n)
            else:
                a /= a_sum                     # enforce Σa = 1

            # ----- convolution and location of the maximum -----
            conv = _full_convolution(a, size=conv_fft_size)
            k = int(conv.argmax())
            val = float(2.0 * n * conv[k])    # exact objective (sum=1)

            # ----- keep the best ever seen -----
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                last_improve = time.time()
                print(f"[Restart {restart:03d}] iter {it:05d}  new best = {val:.6f}")

            # ----- exact gradient of the max‑convolution -----
            w = np.zeros_like(conv)
            w[k] = 1.0                         # one‑hot on the active index
            grad_a = _softmax_gradient(a, w, size_corr=size_corr)   # ∇ₐ max(conv)

            # ----- chain rule to logits (soft‑max Jacobian) -----
            dot = np.dot(a, grad_a)
            grad_u = a * (grad_a - dot)

            # ----- Adam update -----
            m_adam = 0.9 * m_adam + 0.1 * grad_u
            v_adam = 0.999 * v_adam + 0.001 * (grad_u ** 2)

            it_f = float(it)
            m_hat = m_adam / (1.0 - 0.9 ** it_f)
            v_hat = v_adam / (1.0 - 0.999 ** it_f)

            lr = base_lr * (0.9998 ** it)     # gentle decay
            u -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

            # ----- stagnation handling -----
            if time.time() - last_improve > 2.0:
                # small random shake + reset moments
                u += (np.random.rand(n) - 0.5) * 1e-3
                m_adam.fill(0.0)
                v_adam.fill(0.0)
                last_improve = time.time()

        # end of inner loop for this restart
        restart += 1
        print(f"[Restart {restart:03d}] finished it={it}, best so far = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final fine‑tuning: run a longer Adam run starting from the best vector.
    if best_seq is not None:
        n = len(best_seq)
        conv_fft_size = 1 << int(np.ceil(np.log2(2 * n - 1)))
        size_corr = 1 << int(np.ceil(np.log2(3 * n)))
        u = np.log(best_seq + 1e-30)
        m_adam = np.zeros_like(u)
        v_adam = np.zeros_like(u)

        base_lr = 0.005
        it = 0
        max_iter = 30000
        last_improve = time.time()

        while time.time() < deadline and it < max_iter:
            it += 1

            u_shift = u - np.max(u)
            a = np.exp(u_shift)
            a /= a.sum()

            conv = _full_convolution(a, size=conv_fft_size)
            k = int(conv.argmax())
            val = float(2.0 * n * conv[k])

            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[Fine‑tune] iter {it:05d}  new best = {val:.6f}")
                last_improve = time.time()

            w = np.zeros_like(conv)
            w[k] = 1.0
            grad_a = _softmax_gradient(a, w, size_corr=size_corr)
            dot = np.dot(a, grad_a)
            grad_u = a * (grad_a - dot)

            m_adam = 0.9 * m_adam + 0.1 * grad_u
            v_adam = 0.999 * v_adam + 0.001 * (grad_u ** 2)

            it_f = float(it)

            v_hat = v_adam / (1.0 - 0.999 ** it_f)

            lr = base_lr * (0.9999 ** it)
            u -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

            # occasional shake if stuck
            if time.time() - last_improve > 1.5:
                u += (np.random.rand(n) - 0.5) * 5e-4
                m_adam.fill(0.0)
                v_adam.fill(0.0)
                last_improve = time.time()

    # ------------------------------------------------------------------
    # Fallback: a uniform vector (should never be needed)
    if best_seq is None:
        n = 2500
        best_seq = np.full(n, 1.0 / n)

    print(f"Final best objective = {best_val:.6f} (length {len(best_seq)})")
    # ``evaluate_sequence`` expects a plain Python list.
    return best_seq.tolist()
