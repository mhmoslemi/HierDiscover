import time
import numpy as np
from scipy.optimize import minimize

def _full_convolution(a, fft_len):
    n = a.shape[0]
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv

def _cross_correlation(w, a, fft_len):
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]

def _softmax(x):
    mx = x.max()
    e = np.exp(x - mx)
    return e / e.sum()

def _objective_and_grad(logits, beta, n, fft_len_conv, fft_len_corr):
    # a = softmax(logits)
    a = _softmax(logits)
    # convolution of a with itself
    conv = _full_convolution(a, fft_len_conv)   # length 2n-1
    # smoothed max weight
    max_c = conv.max()
    w_raw = np.exp(beta * (conv - max_c))
    w = w_raw / w_raw.sum()
    # smoothed max value = weighted sum
    smooth_max = np.dot(conv, w)
    # objective (scale invariant, sum==1)
    obj = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t a
    grad_a = 2.0 * _cross_correlation(w, a, fft_len_corr)   # shape (n,)
    # gradient of objective w.r.t a
    g_a = 2.0 * n * grad_a

    # chain rule through softmax: Jacobian J = diag(a) - a a^T
    avg = np.dot(a, g_a)
    grad_logits = a * (g_a - avg)
    return obj, grad_logits

def _exact_value(a):
    """Compute the exact evaluation function (no clipping)."""
    n = a.shape[0]
    conv = _full_convolution(a, 1 << int(np.ceil(np.log2(2 * n - 1))))
    max_c = conv.max()
    s = a.sum()
    if s < 0.01:
        return float('inf')
    return 2.0 * n * max_c / (s * s)

"""
Propose a step‑function sequence that (approximately) minimises

        F(a) = 2·n·max(a*a) / (sum(a)²)

under a_i ≥ 0.

Algorithm
---------
* Work on several lengths (1024 … 8192).  For each length a handful of
  initial candidates are generated (warm‑start, historic solution,
  random Dirichlet, power‑law edge shapes, double‑Gaussian edges).
* Each candidate is represented by free *logits*; the actual step heights
  are obtained with a soft‑max, therefore they are non‑negative and sum to
  one automatically.
* A sequence of increasing *temperature* parameters β is used.
  For a fixed β we minimise the smooth‑max surrogate

        φβ(a) = (1/β)·log Σ_k exp(β·(a*a)[k])

  using L‑BFGS‑B on the logits.  The gradient of φβ is obtained analytically
  via a cross‑correlation (FFT‑based, O(n log n)).
* After each temperature stage the exact objective (with the true
  max‑convolution) is evaluated; the best sequence is kept.
* Finally a short sub‑gradient refinement on the true max is performed.
* The routine respects the given time budget and returns the best
  sequence it has found.
"""

import time
import numpy as np
from scipy.optimize import minimize


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)-1)."""
    n = a.shape[0]
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for gradients.
    Returns a vector of length n:
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i=0…n‑1.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    mx = x.max()
    e = np.exp(x - mx)
    return e / e.sum()


def _objective_and_grad(
    logits: np.ndarray,
    beta: float,
    n: int,
    fft_len_conv: int,
    fft_len_corr: int,
) -> tuple[float, np.ndarray]:
    """
    Smooth‑max surrogate (temperature = β) and its gradient
    w.r.t. the free logits.
    The soft‑max guarantees a ≥ 0 and Σ a = 1, therefore the denominator
    of the original objective is constant and can be omitted.
    """
    # a = softmax(logits)
    a = _softmax(logits)

    # convolution a*a
    conv = _full_convolution(a, fft_len_conv)                 # length 2n‑1

    # weights for the smooth max: w_i ∝ exp(β·conv_i)
    max_c = conv.max()
    w_raw = np.exp(beta * (conv - max_c))
    w = w_raw / w_raw.sum()

    # smooth maximum value
    smooth_max = np.dot(conv, w)

    # objective (scale‑invariant because Σa = 1)
    obj = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. a  (see derivation in the docstring)
    grad_a = 2.0 * _cross_correlation(w, a, fft_len_corr)

    # chain rule through the soft‑max: J = diag(a) – a aᵀ
    g_a = 2.0 * n * grad_a
    avg = np.dot(a, g_a)
    grad_logits = a * (g_a - avg)

    return obj, grad_logits


def _exact_value(a: np.ndarray) -> float:
    """Exact evaluation 2·n·max(a*a) / (sum(a)²) (no clipping)."""
    n = a.shape[0]
    fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 0.01:
        return float("inf")
    return 2.0 * n * max_c / (s * s)
