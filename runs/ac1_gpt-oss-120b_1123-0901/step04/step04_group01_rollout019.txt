"""
Propose a candidate step‑function sequence that minimises

        F(a) = 2·n·max(a*a) / (sum(a)²)

The optimizer works on the probability simplex (∑a = 1) because the
objective is scale‑invariant.  It combines three complementary strategies:

1. **Global search** – a few Adam‑based gradient‑ascent runs on the
   soft‑max parametrisation for several problem sizes.  The exact linear
   convolution (via NumPy) is used to obtain the true objective value,
   avoiding the small FFT inaccuracies that the reference implementation
   suffered from.

2. **Random restarts & warm‑starts** – the best solution found so far is
   reused as a “warm start’’ for larger problems.  When a previous global
   optimum (`height_sequence_1`) is available it is also tried as a start.

3. **Local greedy refinement** – after the Adam phase a very cheap
   coordinate‑wise hill‑climber shuffles a small amount of mass between two
   random indices; any improvement is kept.

The routine respects the 1000‑second time budget, prints progress and
returns the best list of floats found.  The uniform distribution is a
theoretical lower bound (value = 2).  The optimiser therefore aims to
reach that bound; if a better value is discovered the algorithm will
report it.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (all top‑level – required by the problem statement)
# ----------------------------------------------------------------------


def full_linear_convolution(a: np.ndarray) -> np.ndarray:
    """
    Compute the exact linear convolution a*a (length 2·len(a)−1) using
    NumPy's direct convolution which is numerically exact for the
    moderate sizes we consider (≤ 32768).
    """
    return np.convolve(a, a)  # full convolution


def objective_value(a: np.ndarray) -> float:
    """
    Return the exact objective 2·n·max(a*a)/(sum(a)²) for a non‑negative
    vector a.  The vector is assumed to be on the simplex (sum≈1); the
    function works for any scale because the ratio is scale‑invariant.
    """
    n = a.shape[0]
    conv = full_linear_convolution(a)
    max_c = conv.max()
    s = a.sum()
    # guard against the pathological case s≈0 (should not happen)
    if s < 1e-12:
        return np.inf
    return 2.0 * n * max_c / (s * s)


def project_to_simplex(x: np.ndarray) -> np.ndarray:
    """
    Project arbitrary real vector x onto the probability simplex
    (non‑negative, sum = 1).  Implementation based on the classic O(n)
    algorithm of Duchi et al. (2008).
    """
    # shift so that the projection lies in the positive orthant
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(x)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(x - theta, 0.0)
    return w


def adam_step(logits: np.ndarray,
              grad_logits: np.ndarray,
              m: np.ndarray,
              v: np.ndarray,
              t: int,
              lr: float,
              beta1: float,
              beta2: float,
              eps: float):
    """
    Perform a single Adam update of the logits vector.
    Returns the updated logits, new moment estimates and updated t.
    """
    m = beta1 * m + (1.0 - beta1) * grad_logits
    v = beta2 * v + (1.0 - beta2) * (grad_logits * grad_logits)
    t += 1
    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)
    logits = logits - lr * m_hat / (np.sqrt(v_hat) + eps)
    return logits, m, v, t


def subgradient_max_convolution(a: np.ndarray) -> np.ndarray:
    """
    Compute a sub‑gradient of the map a ↦ max(a*a) with respect to a.
    For the L∞‑norm this is simply 2·(w ★ a) where w is a weight vector
    that is 1/|Active| on all active lags (those attaining the maximum)
    and 0 elsewhere.
    """
    n = a.shape[0]
    conv = full_linear_convolution(a)
    max_c = conv.max()
    # tolerance for active set (using a relative tolerance)
    tol = max_c * 1e-12 + 1e-15
    active = np.where(conv >= max_c - tol)[0]

    # build weight vector w (length 2n-1)
    w = np.zeros_like(conv)
    w[active] = 1.0 / active.size

    # cross‑correlation w ★ a (same length as a)
    # using FFT for speed (the cost is negligible compared to the Adam loop)
    fft_len = 1 << int(np.ceil(np.log2(2 * n - 1 + n - 1)))
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(a[::-1], fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    grad = 2.0 * corr[start:start + n]
    return grad


def local_refinement(a: np.ndarray,
                     budget_s: float,
                     deadline: float,
                     best_val: float) -> (np.ndarray, float):
    """
    Very cheap hill‑climber that repeatedly moves a small random amount of
    mass from one coordinate to another.
    """
    n = a.shape[0]
    cur_a = a.copy()
    cur_val = best_val
    while time.time() < deadline:
        i, j = np.random.choice(n, 2, replace=False)
        # move a random fraction (10‑30 %) of the mass at i to j
        eps_move = cur_a[i] * (0.1 + 0.2 * np.random.rand())
        if eps_move <= 0:
            continue
        new_a = cur_a.copy()
        new_a[i] -= eps_move
        new_a[j] += eps_move
        # re‑project to simplex to avoid rounding drift
        new_a = new_a / new_a.sum()
        new_val = objective_value(new_a)
        if new_val < cur_val - 1e-12:
            cur_a, cur_val = new_a, new_val
            # a tiny print helps to see progress
            print(f"[local] improved to {cur_val:.6f}")
    return cur_a, cur_val

def propose_candidate(seed: int = 0,

    """
    Search for a non‑negative step‑function (probability vector) that

    time budget (default 1000 s) and returns the best list found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # 1. Global optimisation via Adam on several lengths
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 4096, 8192, 16384]

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length {n} ===")

        # Number of restarts – more restarts for smaller problems
        restarts = 6 if n <= 4096 else 4

        for r in range(restarts):
            if time.time() > deadline:
                break

            # --------------------------------------------------------------
            # Initialise logits (soft‑max parameters)
            # --------------------------------------------------------------
            if best_seq is not None and len(best_seq) == n:
                init_a = best_seq.copy()
                # avoid division by zero if the vector is degenerate
                if init_a.sum() < 1e-12:

                else:

                logits = np.log(init_a + 1e-15)
                print(f"[n={n} r={r}] warm‑start from current best")
            elif 'height_sequence_1' in globals():

                else:

                logits = np.log(cand + 1e-15)
                print(f"[n={n} r={r}] start from height_sequence_1")
            else:

                logits = np.log(a0 + 1e-15)
                print(f"[n={n} r={r}] random start")

            # --------------------------------------------------------------
            # Adam optimiser state
            # --------------------------------------------------------------
            m = np.zeros(n, dtype=np.float64)
            v = np.zeros(n, dtype=np.float64)
            beta1, beta2, eps = 0.9, 0.999, 1e-8
            lr = 0.12
            decay_every = 2500
            t = 0

            last_improve = time.time()
            max_iters = 35000 if n <= 4096 else 25000

            while time.time() < deadline and t < max_iters:
                # logits → probabilities (soft‑max)
                max_logit = np.max(logits)
                exp_logit = np.exp(logits - max_logit)

                    best_seq = a.copy()
                    last_improve = time.time()

                grad_a = subgradient_max_convolution(a)

                # back‑propagate through the soft‑max
                inner = np.dot(a, grad_a)
                grad_logits = a * (grad_a - inner)

                t += 1
                logits, m, v, t = adam_step(logits, grad_logits, m, v, t,

                if t % decay_every == 0:
                    lr *= 0.7

                # shake if stagnating for >30 s
                if time.time() - last_improve > 30.0:
                    logits += (np.random.rand(n) - 0.5) * 1e-2
                    last_improve = time.time()
                    print(f"[n={n} r={r} iter={t}] shake (stagnation)")

            # --------------------------------------------------------------
            # 2. Local greedy refinement (cheap hill‑climber)
            # --------------------------------------------------------------

                                                            budget_s,

                if refined_val < best_val - 1e-12:
                    best_seq, best_val = refined_seq, refined_val
                    print(f"[n={n}] after local refinement best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform distribution (the provable optimum)
    # ------------------------------------------------------------------
    if best_seq is None:

        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)

        print(f"[fallback] using uniform of length {fallback_n}")

    print(f"\nFinal best value: {best_val:.6f} (length {len(best_seq)})")
    # Convert to plain Python list as required by the harness
    return best_seq.tolist()
