def _cross_correlation(w, a, fft_len):

"""
Search for a step‑function (non‑negative sequence) that minimises

        F(a) = 2·n·max(a*a) / (sum(a)²)

using a lightweight parametric family and a final local refinement
(Alexandrov‑style Adam + sub‑gradient).  The parametric family mixes an
edge‑concentrated power‑law, a pair of exponential edges and a uniform
background; differential evolution (or a random search fallback) is used
to find good parameters, after which a few thousand Adam iterations
followed by an exponentiated‑gradient sub‑gradient phase polish the
solution.  The routine respects a 1000 s budget and returns the best
sequence found as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑length caches (global, no closures)
_fft_len_conv_cache = {}
_fft_len_corr_cache = {}


def _fft_len_conv(n: int) -> int:
    """FFT length suitable for a linear convolution of length 2·n‑1."""
    if n not in _fft_len_conv_cache:
        conv_len = 2 * n - 1
        _fft_len_conv_cache[n] = 1 << int(np.ceil(np.log2(conv_len)))
    return _fft_len_conv_cache[n]


def _fft_len_corr(n: int) -> int:
    """FFT length suitable for a cross‑correlation used in the gradient."""
    if n not in _fft_len_corr_cache:
        # need length ≥ (2·n‑1) + n‑1 = 3·n‑2
        corr_len = 3 * n - 2
        _fft_len_corr_cache[n] = 1 << int(np.ceil(np.log2(corr_len)))
    return _fft_len_corr_cache[n]


def _full_convolution(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    if fft_len is None:
        fft_len = _fft_len_conv(n)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for gradients.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def _objective_value(a: np.ndarray) -> float:
    """
    Exact value of the evaluation function (without the safety clipping
    performed by evaluate_sequence – our vectors are already safe).
    """
    n = a.shape[0]
    conv = _full_convolution(a, _fft_len_conv(n))
    max_c = conv.max()
    s = a.sum()
    if s < 0.01:
        return float("inf")
    return 2.0 * n * max_c / (s * s)


# ----------------------------------------------------------------------
# Global variable used only by the DE objective (avoids closures)
_de_n = None


def _de_objective(params):
    """
    Objective used by differential evolution.
    The current length is stored in the global ``_de_n``.
    """
    global _de_n
    n = _de_n
    exp, gamma, w_pow, w_mix = params  # w_pow∈[0,1], w_mix∈[0,1]

    # ----- build the three elementary components -----
    i = np.arange(n, dtype=np.float64)
    # power‑law edge term  ((i+1)*(n-i))**exp  (exp is negative)
    prod = (i + 1.0) * (n - i)
    # safe log‑power to avoid under‑/overflow
    pow_raw = np.exp(exp * np.log(prod + 1e-12))
    sum_pow = pow_raw.sum()
    if sum_pow == 0.0:
        pow_norm = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        pow_norm = pow_raw / sum_pow

    # two‑exponential edge term
    gauss_raw = np.exp(-gamma * i) + np.exp(-gamma * (n - 1 - i))
    sum_gauss = gauss_raw.sum()
    if sum_gauss == 0.0:
        gauss_norm = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        gauss_norm = gauss_raw / sum_gauss

    # uniform background
    uniform = np.full(n, 1.0 / n, dtype=np.float64)

    # mixture:
    #   w_pow   = weight of the power‑law term
    #   w_mix   = relative weight of the exponential term within the
    #            remaining mass (1‑w_pow).  The rest is uniform.
    w_gauss = (1.0 - w_pow) * w_mix
    w_uni = 1.0 - w_pow - w_gauss
    seq = w_pow * pow_norm + w_gauss * gauss_norm + w_uni * uniform
    # Numerical safety (tiny round‑off may break sum‑to‑one)
    seq = np.maximum(seq, 0.0)
    seq = seq / seq.sum()
    return _objective_value(seq)


def _param_search(n: int, deadline: float):
    """
    Lightweight search over the 4‑dimensional parametric family.
    Uses SciPy's differential evolution if available, otherwise falls
    back to a simple random search.
    Returns the best sequence found and its objective value.
    """
    global _de_n
    _de_n = n
    bounds = [
        (-5.0, -0.01),   # exp (negative)
        (0.05, 5.0),     # gamma
        (0.0, 1.0),      # w_pow
        (0.0, 1.0)       # w_mix
    ]

    remaining = deadline - time.time()
    # Rough budget: we afford at most a few hundred evaluations.
    max_iter = 30
    popsize = 10

    # ------------------------------------------------------------------
    # Try SciPy's differential evolution
    try:
        from scipy.optimize import differential_evolution
        result = differential_evolution(
            _de_objective,
            bounds,
            maxiter=max_iter,
            popsize=popsize,
            polish=False,
            seed=0,
            updating='deferred',
            disp=False,
            workers=1,
        )
        best_params = result.x
        best_val = result.fun
    except Exception:          # pragma: no cover – SciPy may be missing
        # Simple random search as a safe fallback
        best_val = float("inf")
        best_params = None
        for _ in range(500):
            if time.time() > deadline:
                break
            cand = [
                np.random.uniform(-5.0, -0.01),   # exp
                np.random.uniform(0.05, 5.0),     # gamma
                np.random.rand(),                 # w_pow
                np.random.rand(),                 # w_mix
            ]
            val = _de_objective(cand)
            if val < best_val:
                best_val = val
                best_params = cand

    # ------------------------------------------------------------------
    # Re‑build the sequence from the best parameters
    exp, gamma, w_pow, w_mix = best_params
    i = np.arange(n, dtype=np.float64)
    prod = (i + 1.0) * (n - i)
    pow_raw = np.exp(exp * np.log(prod + 1e-12))
    sum_pow = pow_raw.sum()
    if sum_pow == 0.0:
        pow_norm = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        pow_norm = pow_raw / sum_pow

    gauss_raw = np.exp(-gamma * i) + np.exp(-gamma * (n - 1 - i))
    sum_gauss = gauss_raw.sum()
    if sum_gauss == 0.0:
        gauss_norm = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        gauss_norm = gauss_raw / sum_gauss

    uniform = np.full(n, 1.0 / n, dtype=np.float64)

    w_gauss = (1.0 - w_pow) * w_mix
    w_uni = 1.0 - w_pow - w_gauss
    seq = w_pow * pow_norm + w_gauss * gauss_norm + w_uni * uniform
    seq = np.maximum(seq, 0.0)
    seq = seq / seq.sum()
    return seq, best_val


# ----------------------------------------------------------------------
def _refine_sequence(a0: np.ndarray, deadline: float):
    """
    Local polishing of a sequence using:
      1. Adam on a smooth‑max surrogate of the max‑convolution.
      2. A short exponentiated‑gradient sub‑gradient phase on the exact max.
    Returns the refined sequence and its objective value.
    """
    a = a0.copy()
    n = a.shape[0]

    # Initialise logits for the simplex via soft‑max trick
    logits = np.log(a + 1e-15)

    # Adam state
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.5
    lam_factor = 1.02

    # bookkeeping
    best_a = a.copy()
    best_val = _objective_value(a)
    last_improve = time.time()

    max_iters = int(2e4 * 1024 / n) + 2000   # a few thousand per length

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # ---- soft‑max to stay on the simplex --------------------------------
        mx = logits.max()
        exp_logits = np.exp(logits - mx)
        a = exp_logits / exp_logits.sum()

        # ---- smooth‑max of the convolution ----------------------------------
        conv = _full_convolution(a, _fft_len_conv(n))
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum > 0.0:
            w /= w_sum
        else:
            w.fill(1.0 / w.size)

        # ---- gradient of the surrogate --------------------------------------
        grad_a = 2.0 * _cross_correlation(w, a, _fft_len_corr(n))

        # ---- back‑prop through soft‑max --------------------------------------
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ---- Adam update ----------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # ---- temperature schedule -------------------------------------------
        lam = min(lam * lam_factor, 1e6)

        # ---- exact objective ------------------------------------------------
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[Adam n={n} iter={t:05d}] new best {val:.6f}")

        # ---- stagnation handling --------------------------------------------
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

    # ----------------------------------------------------------------------
    # Sub‑gradient phase on the true max (exponentiated‑gradient update)
    a = best_a.copy()
    a = a / a.sum()
    eta = 0.6
    sub_last_improve = time.time()

    for sub_it in range(2000):
        if time.time() > deadline:
            break

        conv = _full_convolution(a, _fft_len_conv(n))
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-12)
        if not active.any():
            w = np.zeros_like(conv)
        else:
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()

        grad_a = 2.0 * _cross_correlation(w, a, _fft_len_corr(n))
        a = a * np.exp(-eta * grad_a)

        s = a.sum()
        if s < 1e-12:                # avoid collapse
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        conv = _full_convolution(a, _fft_len_conv(n))
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            sub_last_improve = time.time()
            print(f"[Subgrad n={n} iter={sub_it:04d}] new best {val:.6f}")

        if time.time() - sub_last_improve > 4.0:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Entry point for the competition.  Searches over a few lengths,
    first with a cheap parametric family (DE / random search) and then
    with a local Adam + sub‑gradient polish.  Returns the best sequence
    found as a plain list of non‑negative floats.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5    # spare margin for cleanup

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Warm‑start from the historic solution if available
    hseq = globals().get("height_sequence_1")
    if hseq is not None:
        cand = np.asarray(hseq, dtype=np.float64)
        cand = np.maximum(cand, 0.0)
        if cand.sum() > 0.0:
            cand = cand / cand.sum()
        else:
            cand = np.random.dirichlet(np.ones(cand.shape[0]))
        val = _objective_value(cand)
        print(f"[Init from height_sequence_1] obj={val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = cand.copy()

    # ------------------------------------------------------------------
    # Try a few lengths (larger lengths give a bigger search space but
    # still fit into the time budget).
    for n in (4096, 8192, 16384):
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")

        # --------- parametric family search ---------------------------------
        seq, val = _param_search(n, deadline)
        print(f"[Param‑search n={n}] best val {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = seq.copy()

        # --------- local refinement -----------------------------------------
        if time.time() < deadline:
            refined_seq, refined_val = _refine_sequence(seq, deadline)
            print(f"[Refinement n={n}] refined val {refined_val:.6f}")
            if refined_val < best_val:
                best_val = refined_val
                best_seq = refined_seq.copy()

    # ------------------------------------------------------------------
    # Fallback: uniform vector (should never happen)
    if best_seq is None:
        n = 2048
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = _objective_value(best_seq)

    print(f"\nFinal best objective {best_val:.6f} with length {len(best_seq)}")
    # Return a plain Python list
    return best_seq.tolist()
