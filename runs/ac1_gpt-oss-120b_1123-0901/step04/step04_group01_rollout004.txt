"""
Propose a candidate step‑function (non‑negative vector on the simplex) that
minimises the evaluation function

    F(a) = 2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm works on the probability simplex (sum = 1) and uses an
exponentiated‑gradient (mirror‑descent) scheme with a *smooth* approximation
of the max‑operator (soft‑max).  A small number of restarts with different
initialisations (Dirichlet, half‑wave sine, previous best) are performed for
several problem sizes.  Each iteration evaluates the convolution by FFT,
computes a soft‑max weighting of the active lags, obtains the sub‑gradient
∇a max(conv) = 2·(w ★ a) via another FFT, and updates the vector by
a·exp(−η·∇) followed by renormalisation.  The step size η and the smoothing
parameter β are annealed.  Periodic “shakes’’ (random jitter) are applied
when progress stalls, and a cheap greedy local refinement (mass transfer
between the most/least harmful coordinates) is interleaved.  The best
sequence found before the time budget expires is returned as a plain list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper FFT utilities
# ----------------------------------------------------------------------
def full_convolution(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """
    Linear convolution a*a (length 2·len(a)-1) using FFT.
    """
    n = a.shape[0]
    if fft_len is None:
        fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


def cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) equal to
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fa = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fa, fft_len)
    start = n - 1
    return corr[start:start + n]

    # theoretical bound (≈1) but each iteration is more expensive.
    lengths = [1024, 2048, 4096, 8192, 16384]

    for n in lengths:

        fft_len_conv = 1 << int(np.ceil(np.log2(2 * n - 1)))
        fft_len_corr = 1 << int(np.ceil(np.log2(3 * n)))  # ≥ (2n‑1)+(n‑1)

        # Number of independent restarts for this length.
        restarts = 4 if n <= 4096 else 3 if n <= 8192 else 2
