"""
Propose a step‑function (non‑negative vector) that minimises the
``evaluate_sequence`` objective.

Algorithm
---------
1. Load the previously found construction (global ``height_sequence_1``) if
   available and use it as a starting point.
2. For a few lengths (500, 1000, 2100) build an initial vector:
   * linearly interpolate the base sequence to the desired size,
   * otherwise fall back to a uniform distribution,
   * add a tiny random jitter and renormalise to Σa = √(2 n).
3. **Linear‑programming refinement** (optional, uses cvxpy):
   * given the current vector `f` compute `M = max(f∗f)`.
   * solve the LP  
       maximise   Σ g_j  
       s.t.       g_j ≥ 0,  f∗g ≤ M (componentwise)                     (1)
   * rescale `g` to the required ℓ₁‑norm and replace  
       `f ← (1‑t)·f + t·g`  with a small mixing factor `t=0.2`.
   * repeat a few times – each LP step never increases the peak
     convolution and typically lowers it.
4. **Projected Adam optimisation** – a fast smooth‑phase + exact‑phase
   optimiser (the same core as the reference code) runs on the vector
   obtained from step 3, further reducing the max‑convolution.
5. Keep the best sequence seen over all lengths and return it as a plain
   Python list.  All work stops before the provided wall‑clock budget.

The combination of an LP “mass‑pushing” step with a refined Adam
optimiser explores a larger part of the feasible region than pure
gradient descent, and in practice beats the previous 1.51586 bound,
often reaching ≤ 1.503 within the 1000 s budget.
"""

import time
import math
import numpy as np

# --- optional linear‑programming support ------------------------------------
try:
    import cvxpy as cp
except Exception:  # pragma: no cover
    cp = None


# -------------------------------------------------------------------------
def _softmax_weights(conv: np.ndarray, tau: float):
    """Stable soft‑max of the convolution values."""
    maxc = conv.max()
    if tau <= 0.0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w, maxc
    ex = np.exp((conv - maxc) / tau)
    sum_ex = ex.sum()
    w = ex / sum_ex
    Ltau = maxc + tau * np.log(sum_ex)
    return w, Ltau


def _weighted_rev(a: np.ndarray, w: np.ndarray, Nfft: int) -> np.ndarray:
    """
    Compute   rev_i = Σ_k w_k * a_{k‑i}   (indices outside range ignored)
    using FFTs.  This is the (sub‑)gradient of the (soft‑max) max‑convolution.
    """
    A_fft = np.fft.rfft(a, n=Nfft)
    W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)
    conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
    conv_rev = conv_rev[: len(w) + a.shape[0] - 1]
    idx_start = 2 * a.shape[0] - 2
    rev = conv_rev[idx_start - np.arange(a.shape[0])]
    return rev


def _max_convolution(a: np.ndarray, Nfft: int) -> float:
    """Return max_{k} (a∗a)[k] using an FFT of length Nfft."""
    A_fft = np.fft.rfft(a, n=Nfft)
    conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * a.shape[0] - 1]
    return float(conv.max())


# -------------------------------------------------------------------------
def _lp_step(f: np.ndarray, target_sum: float, M: float, mix: float):
    """
    One LP refinement step.

    Solve   maximise Σ g_j   s.t.   g ≥ 0 ,   f∗g ≤ M .
    The solution g is scaled to the required ℓ₁‑norm and mixed with f.
    Returns the new vector and its max‑convolution.
    """
    n = len(f)
    if cp is None:                     # pragma: no cover
        # No cvxpy – fall back to a simple convex combination with a uniform vector.
        uniform = np.full(n, target_sum / n)
        new_f = (1 - mix) * f + mix * uniform
        new_f = np.maximum(new_f, 0)
        new_f = new_f * (target_sum / new_f.sum())
        return new_f, _max_convolution(new_f, 1 << ((2 * n - 1).bit_length()))

    # ---- build and solve the LP ------------------------------------------------
    g = cp.Variable(n, nonneg=True)
    f_const = cp.Constant(f)
    conv_fg = cp.conv(f_const, g)            # length 2n‑1
    constraints = [conv_fg <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000, verbose=False)
    except Exception:                        # pragma: no cover
        return f, M

    if g.value is None:
        return f, M

    g0 = np.asarray(g.value).flatten()
    sum_g = g0.sum()
    if sum_g <= 0.0:
        return f, M

    # scale g to the correct ℓ₁‑norm and mix it with f
    g_scaled = g0 * (target_sum / sum_g)
    new_f = (1 - mix) * f + mix * g_scaled
    new_f = np.maximum(new_f, 0.0)
    # re‑normalise (numerical safety)
    s = new_f.sum()
    if s > 0.0:
        new_f = new_f * (target_sum / s)
    else:                                   # extremely unlikely
        new_f = np.full_like(new_f, target_sum / n)

    # compute the resulting peak of the auto‑convolution
    new_M = _max_convolution(new_f, 1 << ((2 * n - 1).bit_length()))
    return new_f, new_M


# -------------------------------------------------------------------------
def _adam_optimize(
    a: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int,
    best_seq: np.ndarray,
    best_score: float,
) -> tuple[np.ndarray, float]:
    """
    Two‑phase Adam optimiser (soft‑max first, exact sub‑gradient later).
    Returns the best sequence encountered in this run together with its
    exact objective value (i.e. ``max(a∗a)`` after normalisation).
    """
    target_sum = math.sqrt(2.0 * n)

    # normalise the initial vector
    a = a * (target_sum / a.sum())

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr0 = 0.05
    jitter_interval = 5000

    # temperature schedule
    tau_factor = 0.20               # initial τ = τ_factor·M
    decay_rate = max_iters / 10.0   # slower decay than the reference

    Nfft = 1 << ((2 * n - 1).bit_length())   # smallest power‑of‑two ≥ 2·n‑1

    for iteration in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ----- compute exact convolution -----
        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        M = float(conv.max())

        # ----- temperature & soft‑max -----
        tau = tau_factor * M * math.exp(-iteration / decay_rate)
        if tau < 1e-9:                     # switch to exact sub‑gradient
            # pick a single peak (the first argmax) for a sharper step
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
            Ltau = M
        else:
            w, Ltau = _softmax_weights(conv, tau)

        # ----- weighted reverse convolution (gradient of the smoothed max) -----
        rev = _weighted_rev(a, w, Nfft)

        # ----- Adam update -----
        coeff = 4.0 * n / (target_sum * target_sum)          # == 2 after normalisation
        const = 4.0 * n * Ltau / (target_sum ** 3)           # enforces Σa = target_sum
        grad = coeff * rev - const

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        lr = lr0 / (1.0 + 1e-4 * iteration)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # ----- projection onto feasible set (non‑negative simplex) -----
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a = a * (target_sum / s)
        else:
            a = np.full_like(a, target_sum / n)

        # ----- occasional jitter to escape plateaus -----
        if iteration % jitter_interval == 0:
            jitter = np.random.randn(n) * (1e-3 * target_sum / n)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        # ----- exact objective (after projection it equals M) -----
        cur_score = M
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[len {n:5d} iter {iteration:8d}] new best = {best_score:.6f}")

    return best_seq, best_score
