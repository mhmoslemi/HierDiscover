def lp_step(a):
    n = len(a)
    M = np.max(_full_convolution(a, _next_pow2(2*n-1)))
    g = cp.Variable(n)
    constraints = [g >= 0,
                   cp.conv(a, g) <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    prob.solve(solver=cp.ECOS, warm_start=True)
    if g.value is None:
        return a
    g_vec = np.maximum(g.value, 0)
    sum_g = np.sum(g_vec)
    if sum_g < 1e-9:
        return a
    g_norm = g_vec / sum_g
    return g_norm
```

def subgradient_descent(a, n, max_iters, deadline):
    best_local = a.copy()
    best_val = _exact_obj(a, n)
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr0 = 0.5
    t = 0
    last_improve = time.time()
    fft_len_conv = _next_pow2(2*n-1)
    fft_len_corr = _next_pow2(3*n)
    while t < max_iters and time.time() < deadline:
        conv = _full_convolution(a, fft_len_conv)
        M = conv.max()
        # indicator weight on max positions
        max_idx = np.where(conv >= M - 1e-12)[0]
        w = np.zeros_like(conv)
        w[max_idx] = 1.0 / len(max_idx)
        grad = 2.0 * _cross_correlation(w, a, fft_len_corr)
        # Adam update
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_simplex(a - step)
        val = 2.0 * n * M
        if val < best_val:
            best_val = val
            best_local = a.copy()
            last_improve = time.time()
        # stagnation shake
        if time.time() - last_improve > 5.0:
            noise = (np.random.rand(n) - 0.5) * 1e-2
            a = _project_simplex(a + noise)
            last_improve = time.time()
        # early exit if val extremely low
        # optional: if val < 1.5: break
    return best_local, best_val

def lp_optimize(a, n, M):
    import cvxpy as cp
    g = cp.Variable(n)
    constraints = [g >= 0, cp.conv(a, g) <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000) # maybe
    if g.value is None:
        return None
    gv = np.array(g.value).flatten()
    sum_gv = gv.sum()
    if sum_gv < 1e-9:
        return None
    return gv / sum_gv

def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Compute (w ★ a)[i] = Σ_k w[k]·a[i+k-(n-1)], i = 0 … n‑1.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]
```

"""
Propose a low‑value step‑function.

The algorithm works on the probability simplex (sum = 1) and
alternates two complementary optimisation phases:

1. **Projected sub‑gradient descent** on the exact max‑convolution
   (the gradient is an indicator on the current maximal convolution
   entries).  Adam updates with Euclidean projection onto the simplex
   keep the iterates feasible.

2. **Linear‑programming refinement**:  given the current vector `a`
   and its maximal convolution value `M`, we solve

        maximise   Σ_j g_j
        subject to  g ≥ 0,
                    conv(a, g) ≤ M

   (the convolution constraint is linear – expressed with cvxpy’s
   ``cp.conv``).  The LP pushes mass onto coordinates that are weakly
   represented in `a` without increasing the peak.  After normalising
   the LP solution we continue the alternating loop.

A few outer iterations of (1)+(2) are performed for several problem
sizes (starting from the historic `height_sequence_1` if available).
Finally a cheap random hill‑climbing refinement moves a tiny fraction
of probability mass between two coordinates and keeps any improvement.

All heavy operations (convolution, cross‑correlation) use FFTs, and
the routine respects a global time budget (default 1000 s).  The
returned sequence is a plain Python list of non‑negative numbers.
"""

import time
import numpy as np

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.shape[0] - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)],  i = 0 … n‑1.
    Used for the gradient of Σ_k w_k·(a*a)_k .
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto {x ≥ 0, Σx = 1}."""
    if v.ndim != 1:
        raise ValueError("v must be a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries zero – return uniform vector
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _exact_obj(a: np.ndarray, n: int) -> float:
    """Exact objective 2·n·max(a*a) (a already sums to 1)."""
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         n: int,
                         max_iters: int,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Projected Adam on the simplex using the exact sub‑gradient
    (indicator on the maximal convolution entries).
    Returns the best vector seen and its objective value.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _exact_obj(a, n)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr0 = 0.5

    fft_len_conv = _next_pow2(2 * n - 1)
    fft_len_corr = _next_pow2(3 * n)          # ≥ (2n‑1) + n
    last_improve = time.time()
    t = 0
    while t < max_iters and time.time() < deadline:
        conv = _full_convolution(a, fft_len_conv)
        M = conv.max()
        # indicator weight on the (possibly several) maximal entries
        idx = np.where(conv >= M - 1e-12)[0]
        w = np.zeros_like(conv)
        w[idx] = 1.0 / len(idx)

        grad = 2.0 * _cross_correlation(w, a, fft_len_corr)

        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)

        a = _project_simplex(a - step)

        val = 2.0 * n * M
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[GD n={n} iter={t:05d}] new best: {val:.6f}")

        # simple stagnation handling
        if time.time() - last_improve > 5.0:
            noise = (np.random.rand(n) - 0.5) * 1e-2
            a = _project_simplex(a + noise)
            last_improve = time.time()
    return best_a, best_val


# ----------------------------------------------------------------------
def _lp_step(a: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve   maximise Σ g_j   s.t.   g ≥ 0,
                                    conv(a, g) ≤ M .
    Returns the normalised solution (sum = 1) or None on failure.
    """
    try:
        import cvxpy as cp
    except Exception:
        return None

    n = a.shape[0]
    g = cp.Variable(n, nonneg=True)
    constraints = [cp.conv(a, g) <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    # ECOS is light‑weight and works for these modest sizes.
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, verbose=False)

    if g.value is None:
        return None
    gv = np.asarray(g.value).flatten()
    s = gv.sum()
    if s < 1e-12:
        return None
    return gv / s


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000):
    """
    Search for a non‑negative sequence that minimises the given evaluation
    function.  The routine respects the global time budget and returns the
    best sequence found (as a plain Python list).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5    # safety margin

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Determine a sensible starting length (use historic solution if any)
    start_len = 1024
    if "height_sequence_1" in globals():
        cand = np.asarray(height_sequence_1, dtype=float)
        if cand.ndim == 1 and cand.size > 0:
            start_len = int(cand.size)

    # Problem sizes – larger n gives more freedom but costs more time.
    lengths = [start_len, 2048, 4096, 8192, 16384]
    lengths = sorted(set(l for l in lengths if l > 0))

    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")
        fft_len_conv = _next_pow2(2 * n - 1)
        fft_len_corr = _next_pow2(3 * n)

        # ----------------------------------------------------------------
        # Initialise a probability vector of length n
        if "height_sequence_1" in globals():
            init = np.asarray(height_sequence_1, dtype=float)
            if init.shape[0] != n:
                if init.shape[0] < n:
                    init = np.pad(init, (0, n - init.shape[0]))
                else:
                    init = init[:n]
            init = np.maximum(init, 0.0)
            s = init.sum()
            a = init / s if s >= 0.01 else np.random.dirichlet(np.ones(n))
        else:
            a = np.random.dirichlet(np.ones(n))

        # ----------------------------------------------------------------
        # Alternating gradient / LP refinement
        outer_iters = 4
        for outer in range(outer_iters):
            if time.time() > deadline:
                break

            # ---- Gradient phase -------------------------------------------------
            # heuristic number of iterations (more for smaller n)
            max_iters = max(2000, int(15000 * 1024 / n))
            a_opt, val_opt = _subgradient_descent(a, n, max_iters, deadline)

            if val_opt < best_val:
                best_val = val_opt
                best_seq = a_opt.copy()
                print(f"[n={n} outer={outer}] new best via GD: {val_opt:.6f}")

            # ---- LP refinement --------------------------------------------------
            conv = _full_convolution(a_opt, fft_len_conv)
            M = conv.max()
            g = _lp_step(a_opt, M)

            if g is not None:
                val_g = _exact_obj(g, n)
                if val_g < best_val:
                    best_val = val_g
                    best_seq = g.copy()
                    print(f"[n={n} outer={outer}] new best via LP: {val_g:.6f}")
                a = g  # continue optimisation from LP solution
            else:
                a = a_opt  # fall back to gradient result

            # early stop if we already beat the target
            if best_val < 1.503:
                print("Target reached – terminating early.")
                deadline = time.time()   # force outer loops to stop
                break

    # ------------------------------------------------------------------
    # Final random hill‑climbing refinement (if time permits)
    if best_seq is not None and time.time() < deadline:
        a = best_seq.copy()
        n = a.shape[0]
        fft_len_conv = _next_pow2(2 * n - 1)
        print("\n--- Local random refinement ---")
        last_improve = time.time()
        while time.time() < deadline:
            i, j = np.random.choice(n, 2, replace=False)
            delta = 0.005 * a[i] * np.random.rand()   # up‑to 0.5 % transfer
            if delta <= 0.0:
                continue
            a_prop = a.copy()
            a_prop[i] -= delta
            a_prop[j] += delta
            a_prop = _project_simplex(a_prop)
            val = _exact_obj(a_prop, n)
            if val < best_val:
                best_val = val
                best_seq = a_prop.copy()
                a = a_prop
                last_improve = time.time()
                print(f"[refine] improved to {val:.6f}")
                if best_val < 1.503:
                    print("Target reached during refinement.")
                    break
            # random shake if stagnation lasts too long
            if time.time() - last_improve > 8.0:
                a = _project_simplex(a + (np.random.rand(n) - 0.5) * 1e-3)
                last_improve = time.time()

    # ------------------------------------------------------------------
    # Fallback – uniform sequence (should never happen in practice)
    if best_seq is None:
        n_fallback = 2048
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=float)
        best_val = _exact_obj(best_seq, n_fallback)
        print(f"Fallback uniform sequence, value = {best_val:.6f}")

    # Return a plain Python list as required by the external evaluator.
    return best_seq.tolist()
