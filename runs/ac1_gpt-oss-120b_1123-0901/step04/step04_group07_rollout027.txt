"""
Propose a step‑function (non‑negative vector) that minimises
``evaluate_sequence``.

Algorithm
---------
* Choose a few problem sizes (the original size and up to 8000 items).
* For every size generate a diverse pool of initial vectors:
    – uniform, random Dirichlet (α≈1 and α≈0.3), exponential,
      reversed exponential, Hann window and a few sparse “spike’’ vectors.
* Each start is refined with a two‑phase Adam optimiser.
    – Phase 1 uses a soft‑max approximation of the max‑convolution to give
      a smooth landscape; the temperature τ decays slowly.
    – Phase 2 switches to the exact sub‑gradient once τ is tiny.
* After every Adam step the vector is projected onto the feasible set
  (a_i≥0 and Σa_i = √(2 n)).  Small Gaussian jitter is injected every
  few thousand steps to escape plateaus.
* When the global time budget (≈ 1000 s) is nearly exhausted a cheap
  stochastic hill‑climb is performed starting from the current best
  candidate: a random amount of mass is moved from one index to another;
  the move is kept iff the exact objective improves.
* The best sequence found over all runs is returned as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities ------------------------------------------------------
# ----------------------------------------------------------------------


def _softmax_weights(conv: np.ndarray, tau: float):
    """Stable soft‑max of the convolution values.
    Returns a weight vector w (summing to 1) and the smoothed maximum Lτ.
    """
    maxc = conv.max()
    if tau <= 0.0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w, maxc
    ex = np.exp((conv - maxc) / tau)
    sum_ex = ex.sum()
    w = ex / sum_ex
    Ltau = maxc + tau * math.log(sum_ex)
    return w, Ltau


def _weighted_rev(a: np.ndarray, w: np.ndarray, Nfft: int) -> np.ndarray:
    """
    Compute  rev_i = Σ_k w_k * a_{k‑i}   (indices outside range ignored)
    using FFTs.  This is the (sub‑)gradient of the (soft‑max) max‑convolution.
    """
    A_fft = np.fft.rfft(a, n=Nfft)
    W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)
    conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
    conv_rev = conv_rev[: len(w) + a.shape[0] - 1]
    idx_start = 2 * a.shape[0] - 2
    rev = conv_rev[idx_start - np.arange(a.shape[0])]
    return rev


def _max_convolution(a: np.ndarray, Nfft: int) -> float:
    """Return max_{k} (a∗a)[k] using an FFT of length Nfft."""
    A_fft = np.fft.rfft(a, n=Nfft)
    conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * a.shape[0] - 1]
    return float(conv.max())


# ----------------------------------------------------------------------
# Optimiser ------------------------------------------------------------
# ----------------------------------------------------------------------


def _adam_optimize(
    a: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int,
    best_seq: np.ndarray,
    best_score: float,
    *,
    tau_factor: float = 0.6,
    decay_factor: float = 5.0,
    lr0: float = 0.07,
    jitter_interval: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser with a soft‑max phase followed by an exact sub‑gradient
    phase.  Returns the best sequence found in this run together with its
    (exact) objective value (i.e. ``max(a∗a)`` after normalisation).
    """
    target_sum = math.sqrt(2.0 * n)

    # normalise the initial vector
    a = a * (target_sum / a.sum())

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    Nfft = 1 << ((2 * n - 1).bit_length())   # next power‑of‑two ≥ 2·n‑1

    # pre‑compute a slowly decreasing schedule for τ
    decay_rate = max_iters / decay_factor

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ---- exact convolution (used for the exact objective) ----
        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        M = float(conv.max())

        # ---- temperature & soft‑max ----
        tau = tau_factor * M * math.exp(-it / decay_rate)
        if tau < 1e-9:
            # exact sub‑gradient: uniform distribution over the active set
            mask = conv >= M - 1e-12
            cnt = int(mask.sum())
            w = mask.astype(float) / max(cnt, 1)
            Ltau = M
        else:
            w, Ltau = _softmax_weights(conv, tau)

        # ---- weighted reverse convolution (gradient of the smoothed max) ----
        rev = _weighted_rev(a, w, Nfft)

        # ---- Adam gradient step ----
        # The exact gradient of 2·n·max/(Σa)² after the projection is
        #   g = 4·n·rev/target_sum² - 4·n·Lτ/target_sum³
        coeff = 4.0 * n / (target_sum * target_sum)          # = 2 after normalisation
        const = 4.0 * n * Ltau / (target_sum ** 3)
        grad = coeff * rev - const

        # Adam updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / (1.0 + 1e-4 * it)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # ---- projection onto the feasible set (non‑negative simplex) ----
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a = a * (target_sum / s)
        else:
            a = np.full_like(a, target_sum / n)

        # occasional jitter to escape plateaus
        if it % jitter_interval == 0:
            jitter = np.random.randn(n) * (1e-3 * target_sum / n)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        # ---- exact objective after projection (equals M) ----
        cur_score = M
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[len {n:5d} it {it:8d}] new best = {best_score:.6f}")

    return best_seq, best_score

# Initial vector generators ---------------------------------------------
# ----------------------------------------------------------------------


def _make_initial_vectors(n: int, target_sum: float) -> list[np.ndarray]:
    """Create a small, diverse pool of starting vectors for length ``n``."""
    rng = np.random.default_rng()
    out = []

    # 1) uniform
    out.append(np.full(n, target_sum / n))

    # 2) Dirichlet (α = 1.0 – fairly spread)
    g = rng.gamma(shape=1.0, scale=1.0, size=n)
    out.append(g / g.sum() * target_sum)

    # 3) Dirichlet (α = 0.3 – spikier)
    g = rng.gamma(shape=0.3, scale=1.0, size=n)
    out.append(g / g.sum() * target_sum)

    # 4) exponential decay (random rate)
    rate = rng.uniform(0.0005, 0.01)
    x = np.exp(-rate * np.arange(n))
    out.append(x / x.sum() * target_sum)

    # 5) reversed exponential
    out.append(x[::-1] / x.sum() * target_sum)

    # 6) Hann window
    hann = 0.5 - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1))
    out.append(hann / hann.sum() * target_sum)

    # 7) sparse spike vectors (pick K random positions)
    for K in (max(2, n // 200), max(2, n // 100)):
        idx = rng.choice(n, size=K, replace=False)
        vec = np.zeros(n)
        vec[idx] = target_sum / K
        out.append(vec)

    return out


# ----------------------------------------------------------------------
# Simple local hill‑climbing --------------------------------------------
# ----------------------------------------------------------------------


def _hill_climb(seq: np.ndarray, deadline: float, max_steps: int = 20000) -> np.ndarray:
    """
    Very cheap stochastic hill‑climber: move a tiny amount of mass from one
    index to another if the exact objective improves.
    """
    n = len(seq)
    target_sum = seq.sum()          # should already be ≈√(2n)
    best = seq.copy()
    best_score = evaluate_sequence(best.tolist())

    for step in range(max_steps):
        if time.time() >= deadline:
            break
        # pick two distinct indices
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # move a small epsilon from i to j
        eps = 1e-3 * target_sum / n
        if best[i] < eps:
            continue
        cand = best.copy()
        cand[i] -= eps
        cand[j] += eps
        # keep feasibility
        cand = np.maximum(cand, 0.0)
        # (sum unchanged by construction)
        score = evaluate_sequence(cand.tolist())
        if score < best_score:
            best = cand
            best_score = score
    return best


# ----------------------------------------------------------------------
# Main entry point ------------------------------------------------------
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5.0          # safety margin

    # ------------------------------------------------------------------
    # Base construction (if any) – use it as an additional start vector
    try:
        base = np.array(height_sequence_1, dtype=float)   # type: ignore
    except Exception:
        base = None

    # ------------------------------------------------------------------
    # Lengths to try – original size (or a sensible default) and larger ones.
    if base is not None:
        n0 = base.shape[0]
    else:
        n0 = 2100
    length_factors = [1.0, 1.5, 2.0]
    lengths = sorted({int(n0 * f) for f in length_factors})
    lengths = [max(300, min(l, 8000)) for l in lengths]

    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)

        # build initial pool
        init_vecs = _make_initial_vectors(n, target_sum)

        # possible interpolation of a previously found construction
        if base is not None and base.shape[0] != n:
            old_x = np.linspace(0.0, 1.0, base.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            interp = np.interp(new_x, old_x, base)
            interp = np.maximum(interp, 0.0)
            interp = interp / interp.sum() * target_sum
            init_vecs.append(interp)

        # give the longer problems a little more budget (they are more expensive)
        max_iters = int(2_000_000 * 2100 / n)
        max_iters = max(300_000, max_iters)     # at least a few hundred thousand steps

        print(f"[run] length={n}, max_iters≈{max_iters}, target_sum={target_sum:.3f}")

        for start_id, a0 in enumerate(init_vecs, 1):
            if time.time() >= deadline:
                break
            print(f"    start {start_id}/{len(init_vecs)} for length {n}")

            # run Adam optimisation
            best_seq, best_score = _adam_optimize(
                a=a0.copy(),
                n=n,
                deadline=deadline,
                max_iters=max_iters,
                best_seq=best_seq if best_seq is not None else a0,
                best_score=best_score,
                # hyper‑parameters – a bit more aggressive than the baseline
                tau_factor=0.6,
                decay_factor=5.0,
                lr0=0.07,
                jitter_interval=2000,
            )

    # ------------------------------------------------------------------
    # Final cheap hill‑climb to polish the champion (if any time left)
    if best_seq is not None and time.time() < deadline - 2.0:
        print("[post‑process] hill‑climbing refinement")
        best_seq = _hill_climb(best_seq, deadline, max_steps=15000)
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] refined score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform vector (should never be needed)
    if best_seq is None:
        n = n0
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n)
        best_score = evaluate_sequence(best_seq.tolist())
        print("[fallback] using uniform sequence")

    print(f"[done] best score = {best_score:.6f}, length = {best_seq.shape[0]}")
    return best_seq.tolist()
