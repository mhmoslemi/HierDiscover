def conv_max(a):
    n = a.shape[0]
    Nfft = 1 << ((2*n - 1).bit_length())
    A = np.fft.rfft(a, Nfft)
    conv = np.fft.irfft(A * A, Nfft)[:2*n-1]
    return conv.max()

def _conv_full(x: np.ndarray, Nfft: int) -> np.ndarray:
    """Full self‑convolution of a real vector x using FFT."""
    X = np.fft.rfft(x, Nfft)
    conv = np.fft.irfft(X * X, Nfft)[: 2 * x.shape[0] - 1]
    return conv


def _softmax_weights(conv: np.ndarray, tau: float):
    """stable soft‑max over the convolution values."""
    maxc = conv.max()
    if tau <= 0.0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w, maxc
    ex = np.exp((conv - maxc) / tau)
    sum_ex = ex.sum()
    w = ex / sum_ex
    Ltau = maxc + tau * math.log(sum_ex)
    return w, Ltau


def _weighted_rev(a: np.ndarray, w: np.ndarray, Nfft: int):
    """
    Compute rev_i = Σ_k w_k a_{k‑i}  (reverse weighted convolution).
    """
    A_fft = np.fft.rfft(a, Nfft)
    W_rev_fft = np.fft.rfft(w[::-1], Nfft)
    conv_rev = np.fft.irfft(W_rev_fft * A_fft, Nfft)
    conv_rev = conv_rev[: len(w) + a.shape[0] - 1]
    idx_start = 2 * a.shape[0] - 2
    rev = conv_rev[idx_start - np.arange(a.shape[0])]
    return rev


def _objective_and_grad(y: np.ndarray, tau: float, Nfft: int):
    """
    Return smooth objective L_τ and its gradient w.r.t. y.
    """
    # soft‑max to obtain probability vector
    y_max = np.max(y)
    exp_y = np.exp(y - y_max)
    p = exp_y / exp_y.sum()

    conv = _conv_full(p, Nfft)
    w, Ltau = _softmax_weights(conv, tau)

    # gradient w.r.t. p
    rev = _weighted_rev(p, w, Nfft)
    grad_p = 2.0 * rev

    # Jacobian of soft‑max
    dot = np.dot(p, grad_p)
    grad_y = p * (grad_p - dot)
    return Ltau, grad_y


# ----------------------------------------------------------------------
# L‑BFGS optimisation of the smooth max (continuation in τ)
# ----------------------------------------------------------------------
def _run_l_bfgs(y0: np.ndarray, init_tau: float,
                target_sum: float, Nfft: int,
                deadline: float,
                best_seq: np.ndarray,
                best_score: float) -> (np.ndarray, float):
    """Run the τ‑continuation L‑BFGS, keeping the best exact score."""
    y = y0.copy()
    tau = max(init_tau, 1e-12)
    tau_decay = 0.60          # slightly more aggressive than 0.65
    min_tau = 1e-12

    while time.time() < deadline and tau > min_tau:
        def fun(y_vec):
            L, g = _objective_and_grad(y_vec, tau, Nfft)
            return L, g

        # allow a reasonable number of function evaluations per stage
        options = {'maxfun': 400, 'maxiter': 400, 'ftol': 1e-12, 'gtol': 1e-12}
        res = minimize(fun, y, method='L-BFGS-B', jac=True, options=options)
        y = res.x

        # exact objective for the current point
        y_max = np.max(y)
        exp_y = np.exp(y - y_max)
        p = exp_y / exp_y.sum()
        conv = _conv_full(p, Nfft)
        cur_score = (target_sum ** 2) * conv.max()   # = max(a∗a)

        if cur_score < best_score:
            best_score = cur_score
            best_seq = (p * target_sum).copy()
            print(f"[improve] τ={tau:.2e}  score={best_score:.6f}")

        # shrink temperature
        tau *= tau_decay

        # occasional jitter to shake off plateaus
        if np.random.rand() < 0.20:
            y += np.random.randn(y.shape[0]) * 0.02

    return best_seq, best_score


# ----------------------------------------------------------------------
# Simple greedy local descent – move a tiny mass from the “worst” index
# ----------------------------------------------------------------------
def _greedy_descent(a: np.ndarray, target_sum: float,
                    Nfft: int, deadline: float,
                    max_iter: int = 20000) -> (np.ndarray, float):
    """
    Greedy mass‑transfer descent.
    Returns an improved vector and its max‑convolution value.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_val = _conv_full(best_a, Nfft).max()
    step_factor = 0.02
    it = 0

    while it < max_iter and time.time() < deadline:
        conv = _conv_full(a, Nfft)
        max_val = conv.max()
        max_idx = conv.argmax()

        # gradient of the active convolution entry (2·a shifted)
        grad = np.zeros_like(a)
        i_min = max(0, max_idx - (n - 1))
        i_max = min(n - 1, max_idx)
        idx_range = np.arange(i_min, i_max + 1)
        grad[idx_range] = 2.0 * a[max_idx - idx_range]

        # pick the index that contributes most (product of grad and current mass)
        scores = grad * a
        i = np.argmax(scores)

        # pick a different destination index uniformly at random
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = step_factor * a[i]
        if delta <= 0.0:
            # nothing to move – abort this iteration
            it += 1
            continue

        # perform tentative move
        a[i] -= delta
        a[j] += delta

        new_val = _conv_full(a, Nfft).max()
        if new_val < best_val:
            best_val = new_val
            best_a = a.copy()
        else:
            # revert
            a[i] += delta
            a[j] -= delta

        # slowly shrink step size
        if it % 2000 == 0 and step_factor > 1e-4:
            step_factor *= 0.85

        it += 1

    # final renormalisation (tiny rounding errors may accumulate)
    tot = best_a.sum()
    if abs(tot - target_sum) > 1e-9:
        best_a *= (target_sum / tot)

    return best_a, best_val


# ----------------------------------------------------------------------
# Linear‑programming refinement (requires SciPy + sparse)
# ----------------------------------------------------------------------
def _lp_improvement(a: np.ndarray, target_sum: float,
                    Nfft: int, deadline: float,
                    max_outer: int = 8, mix_step: float = 0.35) -> np.ndarray:
    """
    One‑step LP refinement as described in the paper:
    maximise Σ g_j  subject to  g ≥ 0 , (a∗g)_k ≤ M .
    The solution g is scaled back to the required L1‑norm and mixed
    into a with a small step size.
    """
    if not (_SCIPY_OPT and _SCIPY_SPARSE):
        return a  # silently skip if linprog not available

    n = a.shape[0]
    rows = 2 * n - 1
    for outer in range(max_outer):
        if time.time() >= deadline:
            break
        M = _conv_full(a, Nfft).max()

        # build sparse constraint matrix A_ub where (A_ub @ g)_k = (a∗g)_k
        # column i contains a_i on rows i … i+n‑1
        A = sp.lil_matrix((rows, n), dtype=np.float64)
        for i in range(n):
            A[i:i + n, i] = a[i]
        A = A.tocsr()

        b_ub = np.full(rows, M, dtype=np.float64)
        c = -np.ones(n, dtype=np.float64)               # maximise Σ g_j

        res = linprog(c, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'presolve': True, 'time_limit': max(1.0, (deadline - time.time()))})
        if not res.success:
            # LP failed – abort refinement
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= target_sum * (1.0 + 1e-12):
            # no increase possible – stop
            break

        # scale to correct L1‑norm and blend
        g_scaled = g * (target_sum / sum_g)
        a_new = (1.0 - mix_step) * a + mix_step * g_scaled

        # accept the update only if it improves the exact objective
        if _conv_full(a_new, Nfft).max() < M:
            a = a_new
        else:
            # if not improving, shrink the mixing step and retry
            mix_step *= 0.5
            if mix_step < 1e-4:
                break
    return a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` within the given time budget.

    The routine combines a smooth‑max L‑BFGS continuation,
    a greedy mass‑transfer refinement and an optional LP‑based
    improvement.  Several sequence lengths (including the known
    2100‑vector) are explored; the best solution encountered is
    returned as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # keep a small safety margin

    # ------------------------------------------------------------------
    # Determine a set of lengths to explore.
    # The known construction has length 2100; we also try a few larger
    # values that still fit comfortably in the time budget.
    # ------------------------------------------------------------------
    lengths = [2100, 2600, 3200, 4000]   # feel free to adjust if needed

    best_seq = None
    best_score = float('inf')

    for n in lengths:
        if time.time() >= deadline:
            break

        # ------------------------------------------------------------------
        # Initialise a baseline sequence for this length.
        # ------------------------------------------------------------------
        if 'height_sequence_1' in globals():
            base = np.array(height_sequence_1, dtype=float)
        else:
            base = np.ones(2100, dtype=float)

        if base.shape[0] > n:
            a0 = base[:n].copy()                       # crop if too long
        else:
            a0 = np.pad(base, (0, n - base.shape[0]), mode='constant')
        target_sum = math.sqrt(2.0 * n)
        a0 *= (target_sum / a0.sum())                # enforce Σa = √(2n)

        # probability vector for the smooth‑max optimisation
        p0 = a0 / target_sum
        eps_log = 1e-12
        y0 = np.log(p0 + eps_log)

        # FFT size (next power of two ≥ 2·n‑1)
        Nfft = 1 << ((2 * n - 1).bit_length())

        # ------------------------------------------------------------------
        # Prepare a handful of diverse starting points.
        # ------------------------------------------------------------------
        starts = [
            y0,
            np.log(np.random.exponential(scale=1.0, size=n) + eps_log),   # random Dirichlet
            np.zeros(n, dtype=float),                                       # uniform (log‑softmax of uniform)
            y0 + np.random.randn(n) * 0.04,                                # perturbed known solution
        ]

        # ------------------------------------------------------------------
        # Run the smooth‑max LBFGS from each start.
        # ------------------------------------------------------------------
        for start_id, y_start in enumerate(starts):
            if time.time() >= deadline:
                break
            # estimate a sensible initial temperature
            p_start = np.exp(y_start - np.max(y_start))
            p_start /= p_start.sum()
            conv0 = _conv_full(p_start, Nfft)
            init_tau = max(conv0.max() * 2.0, 1e-12)

            best_seq, best_score = _run_l_bfgs(
                y0=y_start,
                init_tau=init_tau,
                target_sum=target_sum,
                Nfft=Nfft,
                deadline=deadline,
                best_seq=best_seq if best_seq is not None else a0,
                best_score=best_score,
            )
            print(f"[run] length={n}, start={start_id+1}/{len(starts)} "
                  f"best_score={best_score:.6f}")

        # ------------------------------------------------------------------
        # Greedy mass‑transfer refinement.
        # ------------------------------------------------------------------
        a_refined, refined_val = _greedy_descent(
            best_seq.copy(), target_sum, Nfft, deadline)

        # ------------------------------------------------------------------
        # Optional LP‑based refinement (may be skipped if SciPy missing).
        # ------------------------------------------------------------------
        a_refined = _lp_improvement(a_refined, target_sum, Nfft, deadline)

        # ------------------------------------------------------------------
        # Final exact evaluation.
        # ------------------------------------------------------------------
        final_score = evaluate_sequence(a_refined.tolist())
        if final_score < best_score:
            best_score = final_score
            best_seq = a_refined.copy()
            print(f"[update] length={n} new best score={best_score:.6f}")

        # early exit if we already beat the target bound
        if best_score <= 1.503:
            print("[target reached] exiting early")
            break

    # ----------------------------------------------------------------------
    # Fallback – should never happen but provides a safe return value.
    # ----------------------------------------------------------------------
    if best_seq is None:
        n = 200
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)
        best_score = evaluate_sequence(best_seq.tolist())
        print("[fallback] using uniform construction")

    print(f"[done] best_score = {best_score:.6f}, length = {best_seq.shape[0]}")
    return best_seq.tolist()
