"""
Propose a non‑negative step‑function that minimizes

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The evaluator is homogeneous, so we keep the sum fixed to √(2 n) and
optimise the maximum of the self‑convolution.  The algorithm consists of
two stages:

1️⃣  **Soft‑max Adam** – a smooth surrogate of the max (log‑sum‑exp) is
    minimised with Adam on the log‑it (log‑softmax) variables.
    This quickly yields a low‑peak shape.

2️⃣  **Hard‑max Adam** – after the surrogate has converged the exact
    max‑index is used as a (sub‑)gradient.  A second Adam run on the
    same variables refines the solution further.

The whole search is performed many times with random lengths and
initialisations; the best sequence found before the time budget expires
is returned.  A warm‑start from a previously discovered construction is
automatically incorporated if the global variable ``height_sequence_1``
exists.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based linear convolution (real inputs)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of real vectors x and y via FFT."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    # eliminate tiny negative round‑off artifacts
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Soft‑max utilities
# ----------------------------------------------------------------------
def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (returns probabilities that sum to 1)."""
    m = np.max(x)
    e = np.exp(x - m)
    return e / np.sum(e)


def _logits_from_a(a: np.ndarray, eps: float = 1e-12) -> np.ndarray:
    """Return log‑it variables that reproduce a (up to a constant)."""
    return np.log(np.maximum(a, eps))


# ----------------------------------------------------------------------
# Gradient of a single convolution entry (sub‑gradient of max)
# ----------------------------------------------------------------------
def _grad_a_max_shift(a: np.ndarray, shift: int) -> np.ndarray:
    """
    Compute ∂(a*a)[shift]/∂a.  For shift = k the gradient is:
        grad_i = a_{i+k} + a_{i-k}   (if inside the array)
    with the special case k=0 giving 2·a.
    """
    n = a.size
    if shift == 0:
        return 2.0 * a
    grad = np.zeros_like(a)
    # contributions a_i * a_{i+shift}
    grad[: n - shift] += a[shift:]          # derivative w.r.t. the left term
    grad[shift:] += a[: n - shift]          # derivative w.r.t. the right term
    return grad


# ----------------------------------------------------------------------
# Adam optimiser – soft‑max surrogate (smooth max)
# ----------------------------------------------------------------------
def _adam_softmax(
    n: int,
    x0: np.ndarray,
    target_sum: float,
    deadline: float,
    *,
    init_tau_factor: float = 0.14,
    min_tau_factor: float = 1e-5,
    alpha: float = 0.05,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the log‑sum‑exp surrogate.
    Returns the best sequence (as a NumPy array) and its true evaluation.
    """
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    best_seq = None
    best_score = float("inf")
    it = 0

    while time.time() < deadline:
        # ----- map logits to the simplex ---------------------------------
        p = _softmax(x)                     # probabilities, sum = 1
        a = target_sum * p                  # actual heights (sum = target_sum)

        # ----- convolution & current max ---------------------------------
        conv = _full_convolution_fft(a, a)
        max_val = conv.max()
        if max_val < best_score:
            best_score = max_val
            best_seq = a.copy()
            print(f"[{time.time():.2f}] n={n}  softmax‑Adam new best = {max_val:.6f}")

        # ----- smooth‑max weights (log‑sum‑exp) -------------------------
        tau = max(init_tau_factor * max_val * (0.96 ** it),
                  min_tau_factor * max_val)
        # w ∝ exp((c - max)/τ)
        scaled = (conv - max_val) / tau
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient w.r.t. a (2·a ⊛ w_rev) ---------------------------
        w_rev = w[::-1]
        conv2 = _full_convolution_fft(a, w_rev)          # length = 3n‑2
        idx = (2 * n - 2) - np.arange(n)                 # slice that yields Σ a_j w_{i+j}
        grad_a = 2.0 * conv2[idx]

        # ----- chain rule to logits ------------------------------------
        dotp = np.dot(p, grad_a)
        grad_x = target_sum * p * (grad_a - dotp)

        # ----- Adam update ----------------------------------------------
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # occasional jitter to escape plateaus
        if it % 250 == 0 and np.random.rand() < 0.2:
            x += np.random.randn(n) * 0.001

        # slowly decay the base learning rate
        alpha *= 0.9995

    final_score = evaluate_sequence(best_seq.tolist())
    return best_seq, final_score


# ----------------------------------------------------------------------
# Adam optimiser – hard max (sub‑gradient of the true max)
# ----------------------------------------------------------------------
def _adam_hardmax(
    n: int,
    x0: np.ndarray,
    target_sum: float,
    deadline: float,
    *,
    alpha: float = 0.02,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation where the objective is the true max (sub‑gradient).
    """
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    best_seq = None
    best_score = float("inf")
    it = 0

    while time.time() < deadline:
        p = _softmax(x)
        a = target_sum * p

        conv = _full_convolution_fft(a, a)
        max_val = conv.max()
        if max_val < best_score:
            best_score = max_val
            best_seq = a.copy()
            print(f"[{time.time():.2f}] n={n}  hard‑max‑Adam new best = {max_val:.6f}")

        # index of the (first) maximal entry
        k = int(np.argmax(conv))

        # sub‑gradient of the max entry
        grad_a = _grad_a_max_shift(a, k)

        dotp = np.dot(p, grad_a)
        grad_x = target_sum * p * (grad_a - dotp)

        it += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # jitter every few hundred iterations
        if it % 250 == 0 and np.random.rand() < 0.1:
            x += np.random.randn(n) * 0.0005

        alpha *= 0.9995

    final_score = evaluate_sequence(best_seq.tolist())
    return best_seq, final_score

def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the given
    evaluation function.  The routine runs for at most ``budget_s``
    seconds and returns the best sequence discovered.

    Strategy
    --------
    * Randomly choose a length ``n`` (≈ 800 – 2500) and initialise a
      power‑law or Dirichlet distribution.
    * Phase‑1: Adam on a smooth surrogate (log‑sum‑exp) – fast convergence.
    * Phase‑2: Adam on the true max (sub‑gradient) – fine‑tuning.
    * Keep the globally best sequence across many independent restarts.
    * Warm‑start from a previously known construction if available.
    * Early exit if the target bound (≈ 1.503) is reached.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.2          # tiny safety margin

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # Warm‑start from a global construction if present
    # --------------------------------------------------------------
    try:
        hs = globals()["height_sequence_1"]
        hs_arr = np.array(hs, dtype=float)
        hs_arr = np.clip(hs_arr, 0.0, 1000.0)
        score = evaluate_sequence(hs_arr.tolist())
        if score < best_score:
            best_score = score
            best_seq = hs_arr.copy()
            print(f"[{time.time():.2f}] Warm‑start from global seed = {score:.6f}")
    except Exception:
        pass  # no warm‑start available

    # --------------------------------------------------------------
    # Main restart loop
    # --------------------------------------------------------------
    while time.time() < deadline:
        remaining = deadline - time.time()
        # pick problem size – explore widely early on
        if remaining > 500:
            n = np.random.randint(1800, 3500 + 1)
        elif remaining > 200:
            n = np.random.randint(1200, 2800 + 1)
        else:
            n = np.random.randint(800, 2000 + 1)

        target_sum = np.sqrt(2.0 * n)        # Σ a_i = √(2 n)

        # ---- initial shape -------------------------------------------------
        # mixture of power‑law & Dirichlet – gives diverse starts
        if np.random.rand() < 0.6:
            # power‑law tail
            p_exp = np.random.uniform(0.2, 0.8)
            idx = np.arange(1, n + 1, dtype=float)
            init = idx ** (-p_exp)
            if np.random.rand() < 0.4:
                tau_exp = np.random.uniform(0.5, 2.0) * n
                init *= np.exp(-idx / tau_exp)
        else:
            # flat Dirichlet sampling (α = 1 gives uniform‑random simplex point)
            alpha_vec = np.ones(n)
            init = np.random.dirichlet(alpha_vec)

        # small jitter and simple smoothing
        init += np.random.rand(n) * 1e-5
        if np.random.rand() < 0.15:
            klen = np.random.randint(3, 9)
            kern = np.ones(klen) / klen
            init = np.convolve(init, kern, mode='same')
            init = np.maximum(init, 1e-12)

        # optional warm‑start injection from current global best
        if best_seq is not None and np.random.rand() < 0.45:
            bs = best_seq
            if bs.size >= n:
                init = bs[:n].copy()
            else:
                pad = np.random.rand(n - bs.size) * 1e-5
                init = np.concatenate([bs, pad])

        # ---- map to logits -------------------------------------------------
        init = np.maximum(init, 1e-12)
        init = init / init.sum() * target_sum          # scale to correct sum
        x0 = _logits_from_a(init)                     # logits for soft‑max

        # ---- allocate time slice for this restart -------------------------
        # Phase‑1 gets the bulk of the slice, Phase‑2 the remainder.
        slice_time = min(20.0, remaining * 0.6)
        phase1_deadline = time.time() + slice_time
        seq1, score1 = _adam_softmax(
            n,
            x0,
            target_sum,
            phase1_deadline,
            init_tau_factor=0.14,
            min_tau_factor=1e-5,
            alpha=0.05,
        )

        # Prepare for Phase‑2 (use the logits that produced seq1)
        # Retrieve logits from seq1 (inverse soft‑max)
        p1 = seq1 / target_sum
        # Guard against zeros caused by clipping
        p1 = np.maximum(p1, 1e-12)
        x1 = np.log(p1)

        # Phase‑2 gets whatever time is left (but at least a couple seconds)
        remaining2 = deadline - time.time()
        if remaining2 > 2.0:
            phase2_deadline = time.time() + remaining2 * 0.9
            seq2, score2 = _adam_hardmax(
                n,
                x1,
                target_sum,
                phase2_deadline,
                alpha=0.02,
            )
        else:
            seq2, score2 = seq1, score1

        # ---- update global best -------------------------------------------
        if score2 < best_score:
            best_score = score2
            best_seq = seq2.copy()
            print(
                f"[{time.time():.2f}] GLOBAL new best = {best_score:.6f} (n={n})"

            if best_score < 1.503:               # target reached – stop early

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    if best_seq is None:

        target_sum = np.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)

    # The evaluator expects a plain Python list.
    return best_seq.tolist()
