def _sa_bias_optimize(a0, n, fft_len, deadline, cur_best_val):
    a = a0.copy()
    conv = _full_convolution(a, fft_len)   # conv length 2n-1
    cur_max = conv.max()
    cur_val = 2 * n * cur_max
    best_a = a.copy()
    best_val = cur_val

    # pre-alloc move counters
    it = 0
    start = time.time()
    while time.time() < deadline:
        # calculate fraction for cooling schedule (0..1)
        elapsed = time.time() - start
        fraction = elapsed / (deadline - start + 1e-12)
        T = max(1e-7, 0.2 * (1 - fraction) + 1e-7)   # start high 0.2 then down
        move_scale = max(0.01, 0.3 * (1 - fraction))

        # compute subgradient weight w for current max (active indices)
        max_c = cur_max
        active = np.isclose(conv, max_c, atol=1e-12)
        # gradient of max with respect to a
        w_active = np.zeros_like(conv)
        if active.any():
            w_active[active] = 1.0 / active.sum()
        grad = 2.0 * _cross_correlation(w_active, a, fft_len)  # shape n

        # choose source i weighted by positive gradient times a (or just a)
        pos_mask = grad > 0
        if np.any(pos_mask):
            probs_i = (a * grad) * pos_mask
            prob_sum = probs_i.sum()
            if prob_sum > 0:
                probs_i = probs_i / prob_sum
                i = np.random.choice(n, p=probs_i)
            else:
                i = np.random.choice(n, p=a / a.sum())
        else:
            i = np.random.choice(n, p=a / a.sum())
        if a[i] < 1e-14:
            continue
        # choose target j weighted by negative gradient
        neg_mask = grad < 0
        if np.any(neg_mask):
            probs_j = (-grad) * neg_mask
            prob_sum_j = probs_j.sum()
            if prob_sum_j > 0:
                probs_j = probs_j / prob_sum_j
                j = np.random.choice(n, p=probs_j)
            else:
                j = np.random.randint(n-1)
                if j >= i: j += 1
        else:
            # uniform target not equal to i
            j = np.random.randint(n-1)
            if j >= i: j += 1

        # move delta proportionate to a[i] and move_scale
        delta = a[i] * (np.random.rand() * move_scale)
        if delta <= 0:
            continue

        # apply move to a and conv (like original SA)
        a_new = a.copy()
        conv_new = conv.copy()
        a_new[i] -= delta
        a_new[j] += delta

        # update convolution increments
        conv_new[i : i + n] -= 2.0 * delta * a
        conv_new[j : j + n] += 2.0 * delta * a
        conv_new[2 * i] += delta * delta
        conv_new[2 * j] += delta * delta
        conv_new[i + j] -= 2.0 * delta * delta

        new_max = conv_new.max()
        new_val = 2.0 * n * new_max

        if new_val < cur_val or np.random.rand() < np.exp(-(new_val - cur_val) / T):
            a, conv = a_new, conv_new
            cur_val = new_val
            cur_max = new_max
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[SA bias it={it}] new best {best_val:.6f}")

        it += 1
        if it % 5000 == 0:
            print(f"[SA bias it={it}] best {best_val:.6f}")

    return best_a, best_val

"""
Hybrid search for a non‑negative step function that minimizes

    F(a) = 2·n·max(a∗a) / (sum(a)²)

The pipeline:

1.  Build a rich pool of start‑vectors (uniform, Dirichlet,
    power‑law, arcsine, and a sweep of symmetric Beta(p,p) shapes).
2.  Keep the best of those (fast convolution via FFT).
3.  Refine the survivor with Adam on a smooth‑max surrogate
    (log‑sum‑exp) – many iterations, temperature slowly increased.
4.  Run an exact‑max exponentiated‑gradient (sub‑gradient) phase.
5.  Perform a gradient‑biased simulated‑annealing (SA) that moves
    mass from indices with positive sub‑gradient to those with negative.
6.  If time permits, upscale to a double‑size grid (8192), repeat
    Adam+sub‑gradient+SA, and finally return the best sequence found.
All steps respect the global time‑budget (default 1000 s) and the
returned list is guaranteed to be non‑negative and finite.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for the gradient."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   conv_fft_len: int,
                   corr_fft_len: int,
                   deadline: float,
                   max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """Adam on the log‑sum‑exp (smooth max) surrogate."""
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.03                # learning rate
    lam = 0.3                # initial temperature (smoothness)
    lam_factor = 1.015       # anneal speed

    best_a = a0.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        # build probability vector from logits
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c               # objective without normalisation (sum=1)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:05d}] new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)                     # derivative of the normalisation term
        grad_log = a * (grad_a - avg)                # chain rule through soft‑max

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # increase smoothness
        lam = min(lam * lam_factor, 1e6)

        # occasional tiny jitter to shake possible plateaus
        if it % 4000 == 0 and time.time() > deadline - 5:
            logits += (np.random.rand(n) - 0.5) * 1e-3

        it += 1
    return best_a, best_val


def _subgradient_optimize(a0: np.ndarray,
                          n: int,
                          conv_fft_len: int,
                          corr_fft_len: int,
                          deadline: float,
                          max_iters: int = 200_000,
                          eta0: float = 0.6) -> tuple[np.ndarray, float]:
    """Exact‑max sub‑gradient descent with exponentiated‑gradient updates."""
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:                       # avoid degenerate start
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s

    best_a = a.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 2000 == 0:
                print(f"[Subgrad it={it:05d}] new best {val:.6f}")

        # sub‑gradient (uniform weight on all active max locations)
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # diminishing step size
        eta = eta0 / np.sqrt(it + 1)

        # multiplicative (exponentiated‑gradient) update
        a = a * np.exp(-eta * grad_a)

        # project back onto the simplex
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1
    return best_a, best_val


# ----------------------------------------------------------------------
def _sa_bias_optimize(a0: np.ndarray,
                      n: int,
                      fft_len: int,
                      deadline: float,
                      cur_best_val: float) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing with a gradient‑biased move selection.
    Moves mass from indices with positive sub‑gradient to those with
    negative sub‑gradient, which is a principled direction for reducing
    the current peak.
    """
    a = a0.copy()
    conv = _full_convolution(a, fft_len)
    cur_max = conv.max()
    cur_val = 2.0 * n * cur_max
    best_a = a.copy()
    best_val = cur_val

    start = time.time()
    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        fraction = elapsed / (deadline - start + 1e-12)

        # linear cooling (starts at ~0.2, ends at ~1e‑6)
        T = max(1e-6, 0.2 * (1.0 - fraction))
        # move magnitude shrinks as we cool
        move_scale = max(0.01, 0.3 * (1.0 - fraction))

        # --- sub‑gradient for the current max ---
        active = np.isclose(conv, cur_max, atol=1e-12)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()
        grad = 2.0 * _cross_correlation(w, a, fft_len)   # shape (n,)

        # source index: preferably where gradient > 0 (mass hurts the peak)
        pos_mask = grad > 0
        if np.any(pos_mask):
            prob_src = a * grad
            prob_src = prob_src * pos_mask
            total = prob_src.sum()
            if total > 0:
                prob_src = prob_src / total
                i = np.random.choice(n, p=prob_src)
            else:
                i = np.random.choice(n, p=a / a.sum())
        else:
            i = np.random.choice(n, p=a / a.sum())

        if a[i] < 1e-14:
            continue

        # target index: preferably where gradient < 0 (mass helps)
        neg_mask = grad < 0
        if np.any(neg_mask):
            prob_tgt = -grad
            prob_tgt = prob_tgt * neg_mask
            total = prob_tgt.sum()
            if total > 0:
                prob_tgt = prob_tgt / total
                j = np.random.choice(n, p=prob_tgt)
                if j == i:
                    # pick a different target uniformly
                    j = (i + np.random.randint(1, n)) % n
            else:
                # uniform different target
                j = np.random.randint(n - 1)
                if j >= i:
                    j += 1
        else:
            # uniform different target
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

        # amount to transfer
        delta = a[i] * (np.random.rand() * move_scale)
        if delta <= 0:
            continue

        # ----- apply the move (incremental convolution update) -----
        a_new = a.copy()
        conv_new = conv.copy()
        a_new[i] -= delta
        a_new[j] += delta

        # update convolution (linear update derived from the algebra of a*a)
        conv_new[i : i + n] -= 2.0 * delta * a
        conv_new[j : j + n] += 2.0 * delta * a
        conv_new[2 * i] += delta * delta
        conv_new[2 * j] += delta * delta
        conv_new[i + j] -= 2.0 * delta * delta

        new_max = conv_new.max()
        new_val = 2.0 * n * new_max

        # Metropolis acceptance
        if new_val < cur_val or np.random.rand() < np.exp(-(new_val - cur_val) / T):
            a, conv = a_new, conv_new
            cur_val = new_val
            cur_max = new_max
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[SA bias it={it:05d}] new best {best_val:.6f}")

        it += 1
        if it % 5000 == 0:
            print(f"[SA bias it={it:05d}] best {best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
def _interpolate_and_refine(a_small: np.ndarray,
                            n_big: int,
                            deadline: float) -> tuple[np.ndarray, float]:
    """Upscale to a larger grid, run a short Adam + sub‑gradient pass."""
    n_small = a_small.shape[0]
    x_small = np.linspace(0.0, 1.0, n_small, endpoint=False)
    x_big = np.linspace(0.0, 1.0, n_big, endpoint=False)
    a_big = np.interp(x_big, x_small, a_small)
    a_big = np.maximum(a_big, 0.0)
    a_big /= a_big.sum()

    conv_fft_len_big = _next_pow2(2 * n_big - 1)
    corr_fft_len_big = _next_pow2(3 * n_big - 2)

    # allocate modest time for each phase
    sub_deadline = deadline - 5.0
    adam_deadline = min(sub_deadline, time.time() + 80.0)

    a_adam, _ = _adam_optimize(a_big, n_big,
                               conv_fft_len_big,
                               corr_fft_len_big,
                               adam_deadline,
                               max_iters=80_000)

    a_ref, val_ref = _subgradient_optimize(a_adam, n_big,
                                           conv_fft_len_big,
                                           corr_fft_len_big,
                                           sub_deadline,
                                           max_iters=100_000)
    return a_ref, val_ref

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises
    `2·n·max(a∗a)/(sum a)^2`.  The routine respects a 1000 s time‑budget,
    combines diverse warm‑starts, Adam, exponentiated‑gradient,
    a gradient‑biased simulated annealing, and an optional final up‑scale.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # 1) generate a diverse pool of start vectors
    candidates = []

    # uniform
    candidates.append(np.full(n, 1.0 / n))

    # Dirichlet with several concentration levels
    for alpha in [0.05, 0.1, 0.2, 0.5, 1.0, 2.0]:
        cand = np.random.dirichlet(alpha * np.ones(n))
        candidates.append(cand)

    # power‑law edge‑concentrated families
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in (-2.0, -1.5, -1.0, -0.75, -0.5, -0.25):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            candidates.append(shape)

    # arcsine‑type (Beta(½,½) analogue)
    arcs = 1.0 / np.sqrt(np.arange(1, n + 1) * np.arange(n, 0, -1))
    arcs /= arcs.sum()
    candidates.append(arcs)

    # symmetric Beta(p,p) sweep (U‑shaped distributions)
    xs = (np.arange(n, dtype=np.float64) + 0.5) / n
    for p in [0.05, 0.10, 0.15, 0.20, 0.25, 0.30, 0.35, 0.40]:
        shape = (xs ** (p - 1)) * ((1 - xs) ** (p - 1))
        shape = np.maximum(shape, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            candidates.append(shape)

    # optional warm start from a previously found solution
    warm = globals().get("height_sequence_1")
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] != n:
            if w.shape[0] > n:
                w = w[:n]
            else:
                w = np.pad(w, (0, n - w.shape[0]))
        w = np.maximum(w, 0.0)
        if w.sum() >= 0.01:
            w = w / w.sum()
            candidates.append(w)

    # --------------------------------------------------------------
    # 2) keep the best of the raw candidates (fast evaluation)
    best_seq = None
    best_val = float("inf")
    for cid, cand in enumerate(candidates):
        if time.time() >= deadline:
            break
        a = np.maximum(cand, 0.0)
        s = a.sum()
        if s < 0.01:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s
        conv = _full_convolution(a, conv_fft_len)
        cur_val = 2.0 * n * conv.max()
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[init cand {cid}] new best {cur_val:.6f}")

    # safety net
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # --------------------------------------------------------------
    # 3) Adam (smooth‑max) refinement
    remaining = deadline - time.time()
    if remaining > 60:
        adam_deadline = time.time() + min(remaining * 0.4, 250.0)
        a_adam, val_adam = _adam_optimize(best_seq, n,
                                          conv_fft_len, corr_fft_len,
                                          adam_deadline,
                                          max_iters=300_000)
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Adam] improved best to {val_adam:.6f}")

    # --------------------------------------------------------------
    # 4) Exact‑max sub‑gradient (exponentiated‑gradient) refinement
    remaining = deadline - time.time()
    if remaining > 30:
        sub_deadline = time.time() + min(remaining * 0.4, 180.0)
        a_sub, val_sub = _subgradient_optimize(best_seq, n,
                                               conv_fft_len, corr_fft_len,
                                               sub_deadline,
                                               max_iters=200_000)
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Subgrad] improved best to {val_sub:.6f}")

    # --------------------------------------------------------------
    # 5) Gradient‑biased simulated annealing
    remaining = deadline - time.time()
    if remaining > 30:
        a_sa, val_sa = _sa_bias_optimize(best_seq, n,
                                         conv_fft_len,
                                         deadline,
                                         best_val)
        if val_sa < best_val:
            best_val = val_sa
            best_seq = a_sa.copy()
            print(f"[SA bias] improved best to {val_sa:.6f}")

    # --------------------------------------------------------------
    # 6) Optional upscale to 8192 and short refinement
    if time.time() + 20 < deadline:
        n_big = 8192
        a_big, val_big = _interpolate_and_refine(
            best_seq if best_seq is not None else np.random.dirichlet(np.ones(n_big)),
            n_big,
            deadline - 5.0)
        if val_big < best_val:
            best_val = val_big
            best_seq = a_big.copy()
            print(f"[Upscale] final best {val_big:.6f}")

    # --------------------------------------------------------------
    # Final checks, clipping, reporting
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # guarantee a valid sum (evaluate_sequence would normalise anyway)
    if best_seq.sum() < 1e-12:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.shape[0])

    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"Final reported value {final_val:.6f} (length {len(best_seq)})")
    except Exception:
        print(f"Final internal best value {best_val:.6f} (length {len(best_seq)})")

    # expose for possible warm‑starts in later runs
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
