"""
Hybrid search for the step‑function that minimizes

    F(a) = 2·n·max(a∗a) / (sum(a)^2)

The routine builds a large, diverse pool of analytical “shapes”
(Beta‑type, power‑law, edge‑Gaussian, sine‑power, etc.), evaluates them
very quickly with an FFT‑based convolution and keeps the best one.
The best candidate is then refined by three complementary local
optimisers:

* Adam on a smooth‐max (log‑sum‑exp) surrogate.
* Exact‑max sub‑gradient descent with exponentiated‑gradient updates.
* A fast simulated‑annealing move that updates the convolution in O(n).

If time permits, the solution is interpolated to a finer grid (8192
points) and the whole pipeline is repeated.  The whole search respects
the supplied time budget (default 1000 s) and always returns a
non‑negative list that can be fed to ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Cross‑correlation w★a needed for the gradient."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   conv_fft_len: int,
                   corr_fft_len: int,
                   deadline: float,
                   max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """Adam on the log‑sum‑exp surrogate of the maximum convolution."""
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.03                # modest learning rate
    lam = 0.3                # start temperature for the smooth‑max
    lam_factor = 1.015       # slow anneal

    best_a = a0.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max to obtain the probability vector a
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        lam = min(lam * lam_factor, 1e6)

        # occasional jitter when time is about to run out
        if it % 4000 == 0 and time.time() > deadline - 5:
            logits += (np.random.rand(n) - 0.5) * 1e-3

        it += 1

    return best_a, best_val


def _subgradient_optimize(a0: np.ndarray,
                          n: int,
                          conv_fft_len: int,
                          corr_fft_len: int,
                          deadline: float,
                          max_iters: int = 300_000,
                          eta0: float = 0.6) -> tuple[np.ndarray, float]:
    """Exact‑max sub‑gradient descent with exponentiated‑gradient updates."""
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a /= s

    best_a = a.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 2000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active set (indices attaining the maximum)
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # diminishing step‑size
        eta = eta0 / np.sqrt(it + 1)

        # exponentiated‑gradient update
        a = a * np.exp(-eta * grad_a)

        # renormalise onto the simplex
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray,
                         n: int,
                         fft_len: int,
                         deadline: float,
                         cur_best_val: float) -> tuple[np.ndarray, float]:
    """Fast SA that updates the convolution in O(n) per elementary move."""
    a = a0.copy()
    conv = _full_convolution(a, fft_len)
    cur_max = conv.max()
    cur_val = 2.0 * n * cur_max
    best_a = a.copy()
    best_val = cur_val

    start = time.time()
    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        fraction = elapsed / (deadline - start + 1e-12)

        # linear cooling
        T = max(1e-6, 0.1 * (1.0 - fraction))
        # move magnitude shrinks as we cool
        move_scale = max(0.01, 0.3 * (1.0 - fraction))

        # pick source proportional to its mass
        mass_sum = a.sum()
        if mass_sum == 0:
            break
        i = np.random.choice(n, p=a / mass_sum)
        if a[i] < 1e-12:
            continue
        # pick a different target uniformly
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * (np.random.rand() * move_scale)

        a_new = a.copy()
        conv_new = conv.copy()
        a_new[i] -= delta
        a_new[j] += delta

        # update convolution
        conv_new[i : i + n] -= 2.0 * delta * a
        conv_new[j : j + n] += 2.0 * delta * a
        conv_new[2 * i] += delta * delta
        conv_new[2 * j] += delta * delta
        conv_new[i + j] -= 2.0 * delta * delta

        new_max = conv_new.max()
        new_val = 2.0 * n * new_max

        # Metropolis acceptance
        if new_val < cur_val or np.random.rand() < np.exp(-(new_val - cur_val) / T):
            a, conv = a_new, conv_new
            cur_val = new_val
            cur_max = new_max
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[SA it={it:06d}] new best {best_val:.6f}")

        it += 1
        if it % 5000 == 0:
            print(f"[SA it={it:06d}] best {best_val:.6f}")

    return best_a, best_val


def _interpolate_and_refine(a_small: np.ndarray,
                            n_big: int,
                            deadline: float) -> tuple[np.ndarray, float]:
    """Interpolate to a larger grid and perform a short Adam+sub‑gradient pass."""
    n_small = a_small.shape[0]
    x_small = np.linspace(0.0, 1.0, n_small, endpoint=False)
    x_big = np.linspace(0.0, 1.0, n_big, endpoint=False)
    a_big = np.interp(x_big, x_small, a_small)
    a_big = np.maximum(a_big, 0.0)
    if a_big.sum() < 1e-12:
        a_big = np.full(n_big, 1.0 / n_big)
    else:
        a_big /= a_big.sum()

    conv_fft_len_big = _next_pow2(2 * n_big - 1)
    corr_fft_len_big = _next_pow2(3 * n_big - 2)

    a_adam, _ = _adam_optimize(a_big, n_big,
                               conv_fft_len_big,
                               corr_fft_len_big,
                               deadline,
                               max_iters=80_000)

    a_ref, val_ref = _subgradient_optimize(a_adam, n_big,
                                           conv_fft_len_big,
                                           corr_fft_len_big,
                                           deadline,
                                           max_iters=100_000)
    return a_ref, val_ref


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises
    `2·n·max(a∗a)/(sum a)^2`.  The routine builds a large set of analytical
    candidates, picks the best, refines it with Adam, sub‑gradient descent
    and simulated‑annealing, and finally tries a larger grid if time permits.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # leave a tiny safety margin

    # ------------------------------------------------------------------
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm start (if a previous run stored a global variable)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        else:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # ------------------------------------------------------------------
    # Build a very diverse pool of analytical shapes
    candidates = []

    # warm start & a few rotations
    if warm is not None:
        candidates.append(warm.copy())
        for shift in (n // 3, n // 2, 2 * n // 3):
            candidates.append(np.roll(warm, shift))

    # simple baselines
    candidates.append(np.full(n, 1.0 / n))
    candidates.append(np.random.dirichlet(np.ones(n)))

    # arcsine (Beta(½,½))
    arcs = 1.0 / np.sqrt(np.arange(1, n + 1) * np.arange(n, 0, -1))
    arcs /= arcs.sum()
    candidates.append(arcs)

    # edge‑Gaussian families
    i = np.arange(n, dtype=np.float64)
    for lam in np.linspace(0.05, 2.0, 15):
        shape = np.exp(-lam * (i ** 2)) + np.exp(-lam * ((n - 1 - i) ** 2))
        shape /= shape.sum()
        candidates.append(shape)

    # product‑power‑law families  ( (i·(n‑i+1))^γ )
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for gamma in np.linspace(-3.0, -0.25, 12):
        shape = np.where(prod > 0, prod ** gamma, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            candidates.append(shape)

    # Beta‑type shapes  a_i ∝ t^{α‑1}(1‑t)^{β‑1}
    t = (np.arange(n, dtype=np.float64) + 0.5) / n
    for alpha in np.linspace(0.2, 3.0, 9):
        for beta in np.linspace(0.2, 3.0, 9):
            shape = (t ** (alpha - 1)) * ((1 - t) ** (beta - 1))
            if shape.sum() > 0:
                shape /= shape.sum()
                candidates.append(shape)

    # sine‑power shapes  a_i ∝ sin(π·(i+½)/n)^p
    base = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    base = np.clip(base, 1e-12, None)          # avoid division by zero for p<0
    for p in np.linspace(-2.0, 2.0, 9):
        shape = base ** p
        if shape.sum() > 0:
            shape /= shape.sum()
            candidates.append(shape)

    # a handful of random Dirichlet draws with random concentration
    for _ in range(20):
        conc = np.random.uniform(0.05, 5.0)
        shape = np.random.dirichlet(np.full(n, conc))
        candidates.append(shape)

    # ------------------------------------------------------------------
    # Quick evaluation of the whole pool (scale‑invariant, sum is normalised to 1)
    best_seq = None
    best_val = float("inf")
    for cid, cand in enumerate(candidates):
        if time.time() >= deadline:
            break
        a = np.maximum(cand, 0.0)
        s = a.sum()
        if s < 0.01:
            continue
        a /= s
        conv = _full_convolution(a, conv_fft_len)
        cur_val = 2.0 * n * conv.max()
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[init cand {cid:03d}] new best {best_val:.6f}")

    # Fallback in the unlikely event that nothing survived
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = 2.0 * n * _full_convolution(best_seq, conv_fft_len).max()
        print("[fallback] using uniform vector")

    # ------------------------------------------------------------------
    # Deeper local optimisation – Adam (smooth‑max surrogate)
    if time.time() < deadline - 5:
        a_adam, val_adam = _adam_optimize(best_seq, n,
                                          conv_fft_len, corr_fft_len,
                                          deadline,
                                          max_iters=500_000)
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Adam] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Exact‑max sub‑gradient refinement
    if time.time() < deadline - 5:
        a_sub, val_sub = _subgradient_optimize(best_seq, n,
                                               conv_fft_len, corr_fft_len,
                                               deadline,
                                               max_iters=300_000)
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Subgrad] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Simulated‑annealing fine‑tuning
    if time.time() < deadline - 5:
        a_sa, val_sa = _simulated_annealing(best_seq, n,
                                            conv_fft_len,
                                            deadline,
                                            best_val)
        if val_sa < best_val:
            best_val = val_sa
            best_seq = a_sa.copy()
            print(f"[SA] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Upscale to a finer grid if time permits (8192 points)
    if time.time() + 30 < deadline:
        n_big = 8192
        a_big, val_big = _interpolate_and_refine(
            best_seq,
            n_big,
            deadline)
        if val_big < best_val:
            best_val = val_big
            best_seq = a_big.copy()
            print(f"[Big] final best {best_val:.6f}")

    # ------------------------------------------------------------------
    # Clip to the limits enforced by evaluate_sequence and store globally
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    globals()["height_sequence_1"] = best_seq.tolist()

    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"Final reported value {final_val:.6f} (length {len(best_seq)})")
    except Exception:
        print(f"Final internal best value {best_val:.6f} (length {len(best_seq)})")

    return best_seq.tolist()
