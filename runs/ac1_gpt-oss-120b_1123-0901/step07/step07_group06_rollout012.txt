def _l_bfgs_refine(a0, n, conv_fft_len, corr_fft_len, deadline, best_seq, best_val):
    # a0 is starting vector (np.ndarray) sum=1.
    # convert to logits
    eps = 1e-12
    logits = np.log(a0 + eps)

    alpha_vals = [5, 10, 20, 50, 100, 200, 500, 1000, 2000]
    for alpha in alpha_vals:
        if time.time() >= deadline - 2.0:
            break
        # define objective closure capturing alpha
        def fun(logits_vec):
            # compute a
            mx = logits_vec.max()
            exp = np.exp(logits_vec - mx)
            a = exp / exp.sum()
            conv = _full_convolution(a, conv_fft_len)
            # smooth max
            max_exp = np.max(alpha * conv)
            # stable compute weights
            # Actually compute w as softmax(alpha*conv)
            # w = np.exp(alpha*conv - max_exp) / sum(exp(...))
            w = np.exp(alpha * conv - max_exp)
            w_sum = w.sum()
            w = w / w_sum
            # compute smoothed max value
            smooth_max = (np.log(w_sum) + max_exp) / alpha  # because logsumexp = max_exp + log(sum(exp(...))) ; dividing by alpha.
            obj = 2.0 * n * smooth_max

            # gradient
            grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
            avg = np.dot(a, grad_a)
            grad_log = a * (grad_a - avg)
            return obj, grad_log

        # run minimize
        res = scipy.optimize.minimize(fun, logits, method='L-BFGS-B', jac=True,
                                      options={'maxiter': 200, 'disp': False})
        logits = res.x
        # compute actual a and objective
        a = np.exp(logits - np.max(logits))
        a = a / a.sum()
        val = evaluate_sequence(a.tolist())
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[L-BFGS α={alpha}] new best {val:.6f}")
    return best_seq, best_val

Step‑function search for the minimal value of

Algorithm
---------

   (n=8192) by linear interpolation and repeat step 3.
5. Return the best sequence found (as a plain list of floats).

The routine respects the 1000 s budget, prints progress and never exceeds
the allowed time limit.
"""

import time
import numpy as np
import scipy.optimize

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("v must be 1‑d")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _softmax(logits: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (returns a probability vector)."""
    mx = logits.max()
    e = np.exp(logits - mx)
    return e / e.sum()


# ----------------------------------------------------------------------
def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Fast evaluation assuming sum(a)≈1."""
    conv = _full_convolution(a, conv_fft_len)
    return float(2.0 * n * conv.max() / (a.sum() ** 2))


# ----------------------------------------------------------------------
def _adam_smooth_max(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """Adam optimiser on a log‑sum‑exp surrogate of the max‑convolution."""
    eps = 1e-15
    logits = np.log(a0 + eps)            # work in log‑space (soft‑max)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05

    lam = 0.5                # start temperature
    lam_factor = 1.03        # schedule speed
    lam_cap = 1e8

    last_improve = time.time()
    it = 0
    max_iter = 600_000

    while time.time() < deadline and it < max_iter:
        a = _softmax(logits)

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Adam n={n} it={it:06d}] new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        # gradient (chain‑rule through soft‑max)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature schedule
        lam = min(lam * lam_factor, lam_cap)

        # occasional noise if stuck
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
def _subgradient_refinement(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient steps on the true max‑convolution."""
    a = best_seq if best_seq is not None else a0.copy()
    a = np.maximum(a, 0.0)
    a /= a.sum()                     # stay on simplex

    eta = 0.6
    last_improve = time.time()
    max_iter = 250_000

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Subgrad n={n} it={it:06d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        if time.time() - last_improve > 5.0:
            break

        eta *= 0.9999

    return best_seq, best_val


# ----------------------------------------------------------------------
def _projected_gradient_refine(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """Projected gradient descent on the exact objective."""
    a = a0.copy()
    a = np.maximum(a, 0.0)
    a /= a.sum()

    lr = 0.2
    last_improve = time.time()
    max_iter = 180_000

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[ProjGrad n={n} it={it:06d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        step = -lr * grad_a
        a_candidate = _proj_simplex(a + step)

        conv_cand = _full_convolution(a_candidate, conv_fft_len)
        val_cand = 2.0 * n * conv_cand.max() / (a_candidate.sum() ** 2)

        if val_cand < val:
            a = a_candidate
            lr = min(lr * 1.02, 1.0)
        else:
            lr *= 0.7
            if lr < 1e-5:
                break

        if time.time() - last_improve > 5.0:
            break

        lr *= 0.99998

    return best_seq, best_val


# ----------------------------------------------------------------------
def _l_bfgs_objective(logits: np.ndarray,
                      n: int,
                      conv_fft_len: int,
                      corr_fft_len: int,
                      alpha: float):
    """
    Smooth‑max (log‑sum‑exp) objective and gradient w.r.t. logits.
    Returns (obj, grad) where obj = 2·n·smooth_max(conv) and grad has the same
    shape as logits.
    """
    # soft‑max to get a on the simplex
    a = _softmax(logits)

    conv = _full_convolution(a, conv_fft_len)

    # stable log‑sum‑exp of (alpha * conv)
    tmp = alpha * conv
    m = tmp.max()
    exp_tmp = np.exp(tmp - m)
    sum_exp = exp_tmp.sum()
    smooth_max = (m + np.log(sum_exp)) / alpha

    # soft‑max weights for the gradient
    w = exp_tmp / sum_exp

    # gradient wrt a
    grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

    # chain rule through the soft‑max
    avg = np.dot(a, grad_a)
    grad_logits = a * (grad_a - avg)

    obj = 2.0 * n * smooth_max
    return obj, grad_logits


def _l_bfgs_refine(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """
    L‑BFGS refinement using a continuation of smooth‑maxes
    (α increasing from 5 up to 2000).  The optimiser works on logits,
    guaranteeing the simplex constraints automatically.
    """
    eps = 1e-12
    logits = np.log(a0 + eps)                     # initialise logits
    # schedule of sharpening parameters
    alphas = [5.0, 10.0, 20.0, 50.0, 100.0, 200.0, 500.0, 1000.0, 2000.0]

    for alpha in alphas:
        if time.time() >= deadline - 2.0:
            break

        # minimise with L‑BFGS‑B; we pass extra arguments via `args`
        res = scipy.optimize.minimize(
            _l_bfgs_objective,
            logits,
            args=(n, conv_fft_len, corr_fft_len, alpha),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": 150, "disp": False},
        )
        logits = res.x

        # evaluate the true objective for tracking
        a = _softmax(logits)
        val = float(scipy.optimize._minimize._eval_funct(lambda x: evaluate_sequence(x.tolist()), a.tolist()))  # placeholder safety
        # Above line is just to obtain the value using evaluate_sequence.
        # We avoid the inline lambda closure by calling directly:
        val = evaluate_sequence(a.tolist())
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[L‑BFGS α={alpha:.0f}] new best {val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
def _deep_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Full optimisation pipeline for one start:
    Adam → sub‑gradient → projected‑gradient → L‑BFGS refinement.
    """
    best_seq, best_val = _adam_smooth_max(
        a0,
        n,
        conv_fft_len,
        corr_fft_len,
        deadline,
        global_best_seq,
        global_best_val,
    )
    best_seq, best_val = _subgradient_refinement(
        a0,
        n,
        conv_fft_len,
        corr_fft_len,
        deadline,
        best_seq,
        best_val,
    )
    best_seq, best_val = _projected_gradient_refine(
        a0,
        n,
        conv_fft_len,
        corr_fft_len,
        deadline,
        best_seq,
        best_val,
    )
    best_seq, best_val = _l_bfgs_refine(
        best_seq,
        n,
        conv_fft_len,
        corr_fft_len,
        deadline,
        best_seq,
        best_val,
    )
    return best_seq, best_val


# ----------------------------------------------------------------------
def _generate_parametric_candidates(n: int, warm_start: np.ndarray | None):
    """Create a diverse pool of normalized candidates."""
    cand_list = []

    # warm‑start and rotations
    if warm_start is not None:
        cand_list.append(warm_start.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand_list.append(np.roll(warm_start, shift))

    # uniform baseline
    cand_list.append(np.full(n, 1.0 / n))

    # Dirichlet draws – a broad range of concentrations
    for conc in (0.01, 0.02, 0.05, 0.1, 0.3, 0.5, 1.0, 2.0):
        cand_list.append(np.random.dirichlet(np.full(n, conc)))

    i = np.arange(n, dtype=np.float64)

    # edge‑exponential families (with optional uniform mixture)
    for alpha in np.logspace(-2, 1.0, num=14):
        edge = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
        edge /= edge.sum()
        for w in (0.0, 0.3, 0.6, 1.0):
            cand_list.append(w * edge + (1.0 - w) / n)

    # edge‑power‑law families
    for p in np.linspace(0.2, 5.0, num=14):
        edge = (i + 1.0) ** (-p) + (n - i) ** (-p)
        edge /= edge.sum()
        for w in (0.0, 0.4, 0.8, 1.0):
            cand_list.append(w * edge + (1.0 - w) / n)

    # two‑Gaussian edges
    for gamma in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        shape /= shape.sum()
        cand_list.append(shape)

    # sinusoidal half‑wave (positive part)
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand_list.append(shape)

    # beta‑shaped families
    for a in (0.2, 0.5, 1.0, 2.0, 5.0):
        for b in (0.2, 0.5, 1.0, 2.0, 5.0):
            edge = (i + 1) ** (a - 1) * (n - i) ** (b - 1)
            edge = np.maximum(edge, 0.0)
            edge /= edge.sum()
            cand_list.append(edge)

    # simple triangular / quadratic
    tri = np.minimum(i + 1, n - i)
    tri /= tri.sum()
    cand_list.append(tri)

    quad = (i + 1) * (n - i)
    quad /= quad.sum()
    cand_list.append(quad)

    return cand_list


# ----------------------------------------------------------------------
def _local_perturbations(
    base_seq: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """Stochastic jitter + short sub‑gradient refinement."""
    a0 = base_seq.copy()
    while time.time() < deadline:
        eps = 0.02
        a_pert = a0 + eps * (np.random.rand(n) - 0.5)
        a_pert = np.maximum(a_pert, 0.0)
        s = a_pert.sum()
        if s < 1e-8:
            continue
        a_pert /= s

        a_ref, val_ref = _subgradient_refinement(
            a_pert,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            best_seq,
            best_val,
        )
        if val_ref < best_val:
            best_val = val_ref
            best_seq = a_ref.copy()
            a0 = best_seq.copy()
            print(f"[Perturb] new best {best_val:.6f}")
        else:
            a0 = 0.9 * a0 + 0.1 * base_seq
    return best_seq, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Entry point – searches for a step‑function with minimal evaluation score.
    Returns the best sequence as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # --------------------------------------------------------------
    # Initial grid size (medium resolution)
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Optionally warm‑start from a previously found construction
    warm_start = globals().get("height_sequence_1")
    if warm_start is not None:
        warm_start = np.asarray(warm_start, dtype=np.float64)
        if warm_start.shape[0] > n:
            warm_start = warm_start[:n]
        elif warm_start.shape[0] < n:
            warm_start = np.pad(warm_start, (0, n - warm_start.shape[0]))
        warm_start = np.maximum(warm_start, 0.0)
        if warm_start.sum() >= 0.01:
            warm_start = warm_start / warm_start.sum()
        else:
            warm_start = None
    else:
        warm_start = None

    # --------------------------------------------------------------
    # Stage 1 – generate a large pool of parametric candidates
    all_candidates = _generate_parametric_candidates(n, warm_start)

    cand_vals = []
    for cand in all_candidates:
        if time.time() >= deadline:
            break
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 1e-8:
            cand = np.random.dirichlet(np.ones(n))
            s = cand.sum()
        cand /= s
        val = _objective(cand, n, conv_fft_len)
        cand_vals.append((val, cand))

    # keep the most promising starts
    cand_vals.sort(key=lambda x: x[0])
    top_k = 12
    top_candidates = [c for _, c in cand_vals[:top_k]]

    # --------------------------------------------------------------
    # Stage 2 – deep optimisation of each start
    for idx, start_vec in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        print(f"\n--- Deep optimisation on start #{idx + 1}/{len(top_candidates)} ---")
        best_seq, best_val = _deep_optimize(
            start_vec,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            best_seq,
            best_val,
        )

    # --------------------------------------------------------------
    # Stage 3 – stochastic local perturbations around the current best
    if best_seq is not None and time.time() < deadline - 5.0:
        print("\n--- Local stochastic perturbations ---")
        best_seq, best_val = _local_perturbations(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            best_seq,
            best_val,
        )

    # --------------------------------------------------------------
    # Stage 4 – upscale to a larger grid if time permits
    time_left = deadline - time.time()
    if time_left > 40.0:          # need head‑room for the bigger optimisation
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        if best_seq is not None:
            x_old = np.linspace(0, 1, n, endpoint=False)
            x_new = np.linspace(0, 1, n_big, endpoint=False)
            a_big = np.interp(x_new, x_old, best_seq)
            a_big = np.maximum(a_big, 0.0)
            a_big /= a_big.sum()
        else:
            a_big = np.random.dirichlet(np.ones(n_big))

        print("\n--- Optimising on larger grid (n=8192) ---")
        best_seq, best_val = _deep_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            best_seq,
            best_val,
        )
        if best_seq is not None:
            n = n_big

    # --------------------------------------------------------------
    # Fallback – uniform vector if everything failed
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Clip to the evaluator's allowed range and report
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    print(f"\n*** Finished search: best objective ≈ {best_val:.6f} (n={best_seq.shape[0]})")
    return best_seq.tolist()
