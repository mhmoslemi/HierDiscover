"""
Propose a low‑value step‑function.

Algorithm
---------
1. Fix a moderate grid size (n = 4096).  The objective
   F(a)=2·n·max(a∗a)/(sum a)² is invariant to scaling, so we
   always work with a normalised vector (sum = 1).

2. **Parametric grid search** – a very cheap but expressive family:
        a_i = c0                         (baseline, constant)
               + w·exp(-γ·i)             (left exponential)
               + w·exp(-γ·(n‑1‑i))        (right exponential)
   The three parameters (c0 , w , γ) are scanned on a coarse grid.
   For every candidate we compute the convolution by FFT,
   evaluate the exact objective and keep the best vector.

3. **Refinement** – the best candidate from the grid is fed to a
   pure sub‑gradient optimiser (exponentiated‑gradient on the simplex)
   with a decreasing step size η_t=η0/√t.  This phase runs as long as
   the time budget allows and always keeps the best point seen.

4. **Cheap hill‑climbing** – a final random‑move local search that
   swaps a random fraction of mass between two positions while only
   accepting improving moves.

5. **Optional up‑sampling** – if time is left, the current best is
   interpolated to a finer grid (n = 8192) and the refinement steps
   (3)–(4) are repeated.  The finer grid allows a slightly lower bound.

All steps respect the global deadline (default 1000 s) and the routine
always returns the best sequence found so far.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the sub‑gradient.
    Returns a length‑len(a) vector:
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i = 0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 0.85,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient phase (exponentiated‑gradient on the simplex).
    Returns the best vector seen and its exact objective.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            # occasional progress report
            if it % 20000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active set (indices attaining the maximum)
        active = np.isclose(conv, max_c, atol=1e-14)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()

        # sub‑gradient of the (exact) max‑convolution
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # decreasing step size η_t = η₀ / √t
        eta = eta0 / np.sqrt(it + 1)

        # mirror‑descent on the simplex (exponentiated‑gradient)
        a = a * np.exp(-eta * grad_a)

        # renormalise – guard against under‑flow
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 80_000,
) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climbing: move a random fraction of mass from one
    index to another, keep the move if it improves the objective.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()

    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        # up to 30 % of the source mass is transferred
        delta = a[i] * (np.random.rand() * 0.3)

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5000 == 0:
                print(f"[Local move {it:06d}] improved to {val2:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the given time budget and always returns the best
    sequence (as a Python list) found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # a tiny safety margin

    # ------------------------------------------------------------------
    # problem size (start with 4096 – cheap enough for many evaluations)
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Optional warm‑start from a previously saved solution
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm /= warm.sum()
        else:
            warm = None

    # ------------------------------------------------------------------
    # 1. PARAMETRIC GRID SEARCH (baseline + symmetric exponentials)
    best_seq = None
    best_val = float("inf")
    idx = np.arange(n, dtype=np.float64)

    # parameter ranges
    gammas = np.concatenate(
        (
            np.linspace(0.05, 3.0, 30),          # dense low‑γ region

    gammas = np.unique(np.clip(gammas, 1e-6, 10.0))

    weights = np.linspace(0.0, 1.0, 11)          # mixing factor for the two exponentials

    for gamma in gammas:

            n,

        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()
            print(f"[local] improved best {val_loc:.6f}")

    if time.time() + 40 < deadline:

            x_big = np.linspace(0, 1, n_big, endpoint=False)

            a_big,
            n_big,
