def _simulated_annealing(a, n, conv_fft_len, deadline, max_moves):
    a_current = a.copy()
    conv_current = _full_convolution(a_current, conv_fft_len)
    val_current = 2.0 * n * conv_current.max()
    best_a = a_current.copy()
    best_val = val_current
    # temperature schedule
    T0 = 0.01
    it = 0
    while time.time() < deadline and it < max_moves:
        # propose move: shift mass from i to j; choose i weighted by a_i * gradient?
        i = np.random.choice(n, p=a_current / a_current.sum())
        # avoid zero mass positions
        if a_current[i] < 1e-12:
            continue
        j = np.random.randint(n)
        if i == j: continue
        # choose delta fraction between 0 and max_move
        delta = a_current[i] * (np.random.rand() * 0.2) # up to 20% of i's mass
        a_new = a_current.copy()
        a_new[i] -= delta
        a_new[j] += delta
        conv_new = _full_convolution(a_new, conv_fft_len)
        val_new = 2.0 * n * conv_new.max()
        # acceptance
        if val_new < val_current or np.random.rand() < np.exp((val_current - val_new) / (T0 / (1 + it*1e-5))):
            a_current = a_new
            conv_current = conv_new
            val_current = val_new
            if val_new < best_val:
                best_val = val_new
                best_a = a_new.copy()
        it += 1
    return best_a, best_val

"""
Propose a step‑function (non‑negative vector) that minimizes

        F(a) = 2·n·max(a∗a) / (sum(a)^2)

Algorithm
---------
1. Load a warm start (`height_sequence_1`) if it exists and resize it to the
   working grid size.
2. Generate a diverse pool of candidates:
   - classic shapes (uniform, Dirichlet, double‑exponential, sinusoid,
     power‑law, spikes, etc.).
   - additional random parametric families (power‑law, double‑exponential,
     powered sine) sampled on the fly.
3. Evaluate every candidate quickly with `evaluate_sequence` and keep the
   best one.
4. Refine the current best with a few seconds of Adam optimisation on a
   smooth‑max (log‑sum‑exp) surrogate.  The optimisation runs until the
   allocated sub‑deadline is reached.
5. Perform a long exponentiated‑gradient sub‑gradient descent (the exact
   non‑smooth objective) for the remaining time.
6. Run a cheap simulated‑annealing hill‑climber that moves a random
   fraction of mass between two positions; occasional uphill moves are
   accepted with a linearly decreasing temperature.
7. If time permits, upscale the solution to 16384 points and repeat steps
   4‑6 on the finer grid.
8. Clip the final vector to the evaluator’s safety limits and return it.

All phases respect the global 1000 s budget; the routine always returns the
best sequence it has found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a):
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i = 0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
def _initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Generate a modest set of diverse start vectors."""
    cand = []

    # Warm start + a few rotations
    if warm is not None:
        cand.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    # Random Dirichlet
    cand.append(np.random.dirichlet(np.ones(n)))

    # Edge‑concentrated power‑law shapes
    idx = np.arange(n, dtype=np.float64) + 1.0
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-3.0, -0.2, 13):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    # Symmetric double‑exponential shapes
    i = np.arange(n, dtype=np.float64)
    for gamma in np.geomspace(0.2, 5.0, 8):
        shape = np.exp(-gamma * i) + np.exp(-gamma * (n - 1 - i))
        shape /= shape.sum()
        cand.append(shape)

    # Positive half‑wave sinusoid
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand.append(shape)

    # Uniform
    cand.append(np.full(n, 1.0 / n))

    return cand


def _spike_candidates(n: int) -> list[np.ndarray]:
    """A few “spike’’ candidates with k equally spaced non‑zero entries."""
    out = []
    for k in (32, 64, 128, 256, 512):
        a = np.zeros(n, dtype=np.float64)
        if k > 0:
            pos = np.round(np.linspace(0, n - 1, k)).astype(int)
            a[pos] = 1.0 / k
        out.append(a)
    return out

def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
) -> tuple[np.ndarray, float]:

    Adam on the smooth‑max (log‑sum‑exp) surrogate.
    Returns the best vector (sum≈1) encountered and its exact objective.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.08
    lam = 0.5                # temperature
    lam_factor = 1.03

    best_a = a0.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max → simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective (sum(a)=1)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam iter={it:06d}] new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate (chain‑rule through soft‑max)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature schedule → true max
        lam = min(lam * lam_factor, 1e6)

        # tiny random shake every few thousand iters
        if it % 7_000 == 0 and time.time() + 5 > deadline:
            logits += (np.random.rand(n) - 0.5) * 5e-3

        it += 1

    return best_a, best_val


def _subgradient_opt2(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 800_000,
    eta0: float = 0.7,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient phase with a decaying step size.
    Returns the best vector seen and its exact objective.
    """
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s

    best_a = a.copy()
    best_val = float("inf")
    stall = 0
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            stall = 0
            if it % 5_000 == 0:
                print(f"[Subgrad iter={it:07d}] new best {val:.6f}")
        else:
            stall += 1

        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # decaying step size
        eta = eta0 / np.sqrt(it + 1)

        # multiplicative update and renormalisation
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        # random perturbation if we have been stuck for a while
        if stall > 10_000:
            a = np.maximum(a + 0.02 * (np.random.rand(n) - 0.5), 0.0)
            a /= a.sum()
            stall = 0

        it += 1
    return best_a, best_val


def _simulated_annealing(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 20000,
) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing hill‑climber.
    Moves a random fraction of mass from one index to another.
    Accepts uphill moves with a linearly decreasing temperature.
    """
    a = np.maximum(a0, 0.0)
    a = a / a.sum()
    conv = _full_convolution(a, conv_fft_len)
    cur_val = 2.0 * n * conv.max()
    best_val = cur_val
    best_a = a.copy()

    T0 = 0.01  # initial temperature
    for it in range(max_moves):
        if time.time() >= deadline:
            break

        # source index – weighted by current mass
        i = np.random.choice(n, p=a)
        if a[i] < 1e-12:
            continue
        # destination index
        j = np.random.randint(n)
        if i == j:
            continue

        # move up to 20 % of the source mass
        delta = a[i] * (np.random.rand() * 0.2)

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        conv_new = _full_convolution(a_new, conv_fft_len)
        new_val = 2.0 * n * conv_new.max()

        # temperature schedule
        T = T0 * (1.0 - it / max_moves)
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            # Metropolis acceptance
            prob = math.exp((cur_val - new_val) / max(1e-12, T))
            if np.random.rand() < prob:
                accept = True

        if accept:
            a = a_new
            conv = conv_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a_new.copy()
                if it % 5000 == 0:
                    print(f"[SA move={it}] improved to {new_val:.6f}")

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The function respects the given time budget and returns the best
    sequence (as a Python list) found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Working resolution – a fairly large grid (8192) gives more degrees of freedom.
    n = 8192
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm start (previous best) -------------------------------------------------
    warm = None
    if "height_sequence_1" in globals():
        warm = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        # resize / interpolate to current size
        if warm.shape[0] != n:
            x_old = np.linspace(0, 1, warm.shape[0], endpoint=False)
            x_new = np.linspace(0, 1, n, endpoint=False)
            warm = np.interp(x_new, x_old, warm)
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None

    # ------------------------------------------------------------------
    # Phase 0 – generate diverse candidates (including random parametric families)
    candidates = _initial_candidates(n, warm) + _spike_candidates(n)

    # add a handful of random parametric families
    for _ in range(30):
        r = np.random.rand()
        if r < 0.33:
            # power‑law edge‑concentrated: (i·(n‑i))^α
            alpha = np.random.uniform(-3.0, -0.1)
            i = np.arange(1, n + 1, dtype=np.float64)
            shape = i * (n - i + 1)
            shape = np.where(shape > 0, shape ** alpha, 0.0)
        elif r < 0.66:
            # double‑exponential with random decay
            gamma = np.random.uniform(0.1, 6.0)
            i = np.arange(n, dtype=np.float64)
            shape = np.exp(-gamma * i) + np.exp(-gamma * (n - 1 - i))
        else:
            # powered sinusoid
            p = np.random.uniform(0.5, 4.0)
            shape = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** p

        if shape.sum() > 0:
            shape = shape / shape.sum()
            candidates.append(shape)

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Evaluate the pool (cheap evaluation via evaluate_sequence)
    for idx, cand in enumerate(candidates):
        if time.time() >= deadline:
            break
        a = np.maximum(cand, 0.0)
        s = a.sum()
        if s < 0.01:
            a = np.random.dirichlet(np.ones(n))
        else:
            a = a / s
        val = evaluate_sequence(a.tolist())
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[candidate {idx}] initial value {val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Adam optimisation on the current best (short, focused)
    if best_seq is not None and time.time() < deadline - 10:
        adam_deadline = min(deadline - 5.0, time.time() + 30.0)
        a_adam, val_adam = _adam_optimize(
            best_seq, n, conv_fft_len, corr_fft_len, adam_deadline, max_iters=500_000
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Adam] new best {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Long sub‑gradient descent for the remaining time
    if best_seq is not None and time.time() < deadline - 5:
        sub_deadline = deadline - 2.0
        a_sub, val_sub = _subgradient_opt2(
            best_seq, n, conv_fft_len, corr_fft_len, sub_deadline, max_iters=800_000
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Subgrad] new best {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Simulated‑annealing local search (cheap but can escape plateaus)
    if best_seq is not None and time.time() < deadline - 5:
        moves = int(min(30000, (deadline - time.time()) * 8))
        a_loc, val_loc = _simulated_annealing(
            best_seq, n, conv_fft_len, deadline, max_moves=moves
        )
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()
            print(f"[SA] improved best {val_loc:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – Upscale to a finer grid (16384) if time remains
    if best_seq is not None and time.time() + 30 < deadline:
        n_big = 16384
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        # linear interpolation to the finer grid
        x_old = np.linspace(0, 1, best_seq.shape[0], endpoint=False)
        x_new = np.linspace(0, 1, n_big, endpoint=False)
        a_big = np.interp(x_new, x_old, best_seq)

        a_big = np.maximum(a_big, 0.0)
        a_big = a_big / a_big.sum()

        # short Adam at the larger resolution
        big_deadline = min(deadline - 5.0, time.time() + 25.0)
        a_big_adam, val_big_adam = _adam_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            big_deadline,
            max_iters=300_000,
        )
        if val_big_adam < best_val:
            best_val = val_big_adam
            best_seq = a_big_adam.copy()
            print(f"[Big‑16384 Adam] new best {val_big_adam:.6f}")

        # final sub‑gradient refinement on the fine grid
        if time.time() < deadline - 5:
            a_big_sub, val_big_sub = _subgradient_opt2(
                best_seq,
                n_big,
                conv_fft_len_big,
                corr_fft_len_big,
                deadline,
                max_iters=500_000,
            )
            if val_big_sub < best_val:
                best_val = val_big_sub
                best_seq = a_big_sub.copy()
                print(f"[Big‑16384 Subgrad] new best {val_big_sub:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform vector if something went terribly wrong
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the evaluator's safety limits
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Store as a global for the next invocation (optional but convenient)
    globals()["height_sequence_1"] = best_seq.tolist()

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
