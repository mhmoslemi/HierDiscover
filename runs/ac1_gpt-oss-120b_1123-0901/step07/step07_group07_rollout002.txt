def propose_candidate(seed=0, budget_s=1000):
    import numpy as np, time
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5

    # n selection: choose large n if time allows
    n = 8192
    conv_fft_len = _next_pow2(2*n - 1)
    corr_fft_len = _next_pow2(3*n - 2)

    # Warm start
    warm_start = globals().get("height_sequence_1")
    if warm_start is not None:
        ws = np.array(warm_start, dtype=np.float64)
        if ws.shape[0] != n:
            # interpolate
            x_old = np.linspace(0, 1, ws.shape[0], endpoint=False)
            x_new = np.linspace(0, 1, n, endpoint=False)
            ws = np.interp(x_new, x_old, ws)
        ws = np.clip(ws, 0, None)
        if ws.sum() >= 0.01:
            ws = ws / ws.sum()
        else:
            ws = None
    else:
        ws = None

    # generate candidates
    cand_pool = _generate_candidates(n, ws)
    # evaluate
    evaluated = []
    for cand in cand_pool:
        if time.time() >= deadline:
            break
        # ensure positivity and sum normalization
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 0.01:
            cand = np.random.dirichlet(np.ones(n))
            s = cand.sum()
        cand = cand / s
        val = _objective(cand, n, conv_fft_len)
        evaluated.append((val, cand))

    if not evaluated:
        # fallback uniform
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = _objective(best_seq, n, conv_fft_len)
        return best_seq.tolist()

    evaluated.sort(key=lambda x: x[0])
    top_k = min(8, len(evaluated))
    top_candidates = [c for _, c in evaluated[:top_k]]
    best_seq, best_val = evaluated[0][1].copy(), evaluated[0][0]

    print(f"Initial best objective {best_val:.6f} with n={n}")

    # deep optimization
    for idx, start in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        print(f"Optimising start {idx+1}/{len(top_candidates)}")
        best_seq, best_val = _deep_optimize(start, n, conv_fft_len, corr_fft_len, deadline, best_seq, best_val)
        print(f"After start {idx+1}: best={best_val:.6f}")

    # final hill climbing (simple random transfers)
    if best_seq is not None and time.time() < deadline - 5.0:
        a = best_seq.copy()
        best_local_val = best_val
        it = 0
        max_local_iter = 20000
        while time.time() < deadline and it < max_local_iter:
            i, j = np.random.randint(n, size=2)
            # transfer small proportion of mass from i to j
            delta = (np.random.rand() - 0.5) * 0.1 * a[i]  # up to ±5% of a[i]
            # ensure a[i]-delta >=0, a[j]+delta >=0
            if delta > a[i]:
                delta = a[i]
            if -delta > a[j]:
                delta = -a[j]
            if abs(delta) < 1e-16:
                continue
            a[i] -= delta
            a[j] += delta
            # compute conv and objective
            conv = _full_convolution(a, conv_fft_len)
            max_c = conv.max()
            val = 2.0 * n * max_c  # sum=1
            if val < best_local_val:
                best_local_val = val
                best_seq = a.copy()
                # keep improvement
            else:
                # revert
                a[i] += delta
                a[j] -= delta
            it += 1
        best_val = best_local_val
        print(f"Final hill climbing done after {it} attempts, best={best_val:.6f}")

    # Clip for evaluator
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    print(f"*** Search complete: best objective ≈ {best_val:.6f} (n={n})")
    return best_seq.tolist()

def _subgradient_refinement(a0, n, conv_fft_len, corr_fft_len, deadline, init_best_seq, init_best_val):
    a = init_best_seq if init_best_seq is not None else a0.copy()
    # ensure positivity and sum=1
    a = np.maximum(a, 0.0)
    if a.sum() == 0:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_seq = a.copy()
    best_val = init_best_val
    eta0 = 0.8
    it = 0
    max_iter = 60_000
    last_improve = time.time()
    while time.time() < deadline and it < max_iter:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            # print optional
            if it % 5000 == 0:
                print(f"[Subgrad n={n} it={it}] new best {val:.6f}")

        # determine active set with tolerance
        tol = max(1e-9, 1e-8 * max_c)  # relative tolerance
        active = conv >= max_c - tol
        # compute w
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)
        # gradient
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        eta = eta0 / np.sqrt(it + 1)

        a = a * np.exp(-eta * grad_a)
        # renormalize
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        if time.time() - last_improve > 8.0:
            # break if no improvement for while
            break
        it += 1

    return best_seq, best_val

"""
Propose a low‑evaluation non‑negative step function.

Algorithm outline
-----------------
1. **Problem scaling** – we keep the total mass ‖a‖₁ = 1, therefore the
   objective reduces to `2·n·max(a*a)`.  Minimising the peak of the discrete
   autocorrelation is the sole goal.

2. **Candidate seeding** – a compact but diverse pool of symmetric shapes
   (uniform, beta‑shapes, edge‑exponential, power‑law, arcsine) together
   with a few random Dirichlet draws is generated for a large grid
   (`n = 8192`).  The best few candidates (by a fast FFT evaluation) are
   kept.

3. **Two‑phase continuous optimisation**
   * *Adam on a smooth‑max surrogate* – we optimise the log‑soft‑max
     parametrisation `a = softmax(logits)` using Adam.  A temperature
     `λ` turns the hard max into a log‑sum‑exp; it is increased
     exponentially during optimisation.
   * *Exact sub‑gradient refinement* – after Adam the true max‑convolution
     is used.  The active set (all lags within a tiny tolerance of the
     maximum) yields a sub‑gradient, and an exponentiated‑gradient step
     (mirror descent on the simplex) further reduces the peak.

   Both phases respect the time budget and keep track of the globally best
   vector.

4. **Local random hill‑climb** – a final short stochastic mass‑transfer
   loop (pairwise moves on the simplex) tries to squeeze a last bit of
   improvement.

All heavy work is performed with FFTs (`np.fft.rfft/irfft`), which makes
each iteration cheap even for `n = 8192`.  Hyper‑parameters (learning
rates, λ‑schedule, iteration caps, tolerances) were tuned to favour fast
convergence while staying within the 1000 s limit.

The function returns a plain Python list of non‑negative floats that can be
directly fed to `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*a using real FFT of length ``fft_len``.
    Returns the first ``2·len(a)-1`` entries.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for gradients.
    The result length equals ``len(a)``.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """
    Fast evaluation of the objective (sum(a) ≈ 1).
    Returns the exact value of ``evaluate_sequence`` (ignoring the
    evaluator’s clipping, which does not affect our tiny numbers).
    """
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s ** 2))

def _adam_smooth_max(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑sum‑exp (smooth‑max) surrogate.
    The simplex constraint is honoured by optimiser‑in‑the‑logits (soft‑max)
    parametrisation.
    """
    eps = 1e-12
    logits = np.log(a0 + eps)                # initialise in log‑softmax space
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.12                                 # relatively aggressive
    lam = 0.5                                 # start temperature
    lam_factor = 1.03
    lam_cap = 1e5

    best_seq = global_best_seq
    best_val = global_best_val
    last_improve = time.time()
    it = 0
    max_iter = 250_000

    while time.time() < deadline and it < max_iter:
        # ---- simplex via soft‑max ----
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # ---- objective (M = max conv) ----
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            if it % 10000 == 0:
                print(f"[Adam n={n} it={it:06d}] new best {val:.6f}")

        # ---- smooth‑max weighting (log‑sum‑exp) ----
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---- gradient of the surrogate ----
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)               # ⟨a,∇⟩ term from chain rule
        grad_log = a * (grad_a - avg)         # gradient w.r.t. logits (soft‑max)

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---- temperature schedule ----
        lam = min(lam * lam_factor, lam_cap)

        # ---- jitter if stuck ----
        if time.time() - last_improve > 6.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1

    return best_seq, best_val


def _subgradient_refinement(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    init_best_seq: np.ndarray | None,
    init_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient descent on the true max‑convolution.
    The active set is defined by a tiny relative tolerance to include
    near‑maximal lags, which yields a more informative sub‑gradient.
    """
    a = init_best_seq if init_best_seq is not None else a0.copy()
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_seq = a.copy()
    best_val = init_best_val
    eta0 = 0.8
    it = 0
    max_iter = 60_000
    last_improve = time.time()

    while time.time() < deadline and it < max_iter:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            if it % 5000 == 0:
                print(f"[Subgrad n={n} it={it:05d}] new best {val:.6f}")

        # ---- active set with relative tolerance ----
        tol = max(1e-12, 1e-8 * max_c)
        active = conv >= max_c - tol
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        # ---- sub‑gradient of the max‑convolution ----
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # ---- exponentiated‑gradient step (mirror descent on simplex) ----
        eta = eta0 / np.sqrt(it + 1)
        a = a * np.exp(-eta * grad_a)

        # ---- re‑normalise ----
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        if time.time() - last_improve > 8.0:
            break
        it += 1

    return best_seq, best_val


def _deep_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,

    global_best_val: float,

        a0,
        n,
        conv_fft_len,

        n,
        conv_fft_len,
        corr_fft_len,
        deadline,
        best_seq,
        best_val,
    )
    return best_seq, best_val

    # symmetric beta‑shapes

        candidates.append(shape)

        edge = (i + 1.0) ** (-p) + (n - i) ** (-p)

        for w in (0.0, 0.5, 1.0):

    shape /= shape.sum()

            x_old = np.linspace(0, 1, ws.shape[0], endpoint=False)
            x_new = np.linspace(0, 1, n, endpoint=False)
            ws = np.interp(x_new, x_old, ws)
        ws = np.clip(ws, 0.0, None)
        if ws.sum() >= 0.01:
            ws = ws / ws.sum()
        else:
            ws = None
    else:
        ws = None

    # --------------------------------------------------------------
    # Phase 1 – generate a moderate pool of seeds and keep the most promising.
    cand_pool = _generate_candidates(n, ws)

    evaluated = []
    for cand in cand_pool:
        if time.time() >= deadline:
            break
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 0.01:
            cand = np.random.dirichlet(np.ones(n))
            s = cand.sum()
        cand = cand / s
        val = _objective(cand, n, conv_fft_len)
        evaluated.append((val, cand))

    if not evaluated:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = _objective(best_seq, n, conv_fft_len)
        return best_seq.tolist()

    evaluated.sort(key=lambda x: x[0])
    top_k = min(8, len(evaluated))
    top_candidates = [c for _, c in evaluated[:top_k]]
    best_seq, best_val = evaluated[0][1].copy(), evaluated[0][0]

    print(f"Initial best objective {best_val:.6f} (n={n})")

    # --------------------------------------------------------------
    # Phase 2 – deep optimisation of the selected seeds.
    for idx, start in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        print(f"--- Optimising start {idx + 1}/{len(top_candidates)} ---")
        best_seq, best_val = _deep_optimize(
            start,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            best_seq,
            best_val,
        )
        print(f"After start {idx + 1}: best objective {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – final stochastic hill‑climb (tiny random mass transfers).
    if best_seq is not None and time.time() < deadline - 5.0:
        a = best_seq.copy()
        best_local_val = best_val
        it = 0
        max_local_iter = 20000
        while time.time() < deadline and it < max_local_iter:
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            # propose a transfer proportional to the smaller entry
            delta = (np.random.rand() - 0.5) * 0.1 * min(a[i], a[j])
            if delta > a[i]:
                delta = a[i]
            if -delta > a[j]:
                delta = -a[j]
            if abs(delta) < 1e-16:
                continue
            a[i] -= delta
            a[j] += delta

            conv = _full_convolution(a, conv_fft_len)
            max_c = conv.max()
            val = 2.0 * n * max_c / (a.sum() ** 2)

            if val < best_local_val:
                best_local_val = val
                best_seq = a.copy()
                # keep the improvement
            else:
                # revert the move
                a[i] += delta
                a[j] -= delta
            it += 1
        best_val = best_local_val
        print(f"Final local refinement finished after {it} attempts, best={best_val:.6f}")

    # --------------------------------------------------------------
    # Safety fallback – uniform vector if nothing was discovered.
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clamp to evaluator's allowed range.
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"*** Search complete: best objective ≈ {best_val:.6f} (n={n})")
    return best_seq.tolist()
