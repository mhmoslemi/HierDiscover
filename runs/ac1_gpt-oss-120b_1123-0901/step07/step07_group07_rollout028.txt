def anneal(a, best_val, deadline):
    a = a.copy()
    val = evaluate_sequence(a.tolist())
    best_seq = a.copy()
    best_val = val
    T0 = best_val * 0.01
    T = T0
    while time.time() < deadline:
        # propose small move
        i, j = random indices from 0..n-1
        if i==j: continue
        if a[i] < 1e-14 and a[j] < 1e-14: continue
        delta = (np.random.rand() - 0.5) * 0.2 * min(a[i], a[j])
        # ensure positivity after move
        new_i = a[i] + delta
        new_j = a[j] - delta
        if new_i < 0 or new_j < 0:
            continue
        a[i] = new_i
        a[j] = new_j
        val = evaluate_sequence(a.tolist())
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            T = max(T * 0.95, 1e-6)   # cool down
        else:
            accept = np.random.rand() < np.exp(-(val - best_val) / T)
            if accept:
                # accept move
                pass
            else:
                # revert
                a[i] = new_i - delta # actually revert to original
                a[j] = new_j + delta
        # optionally early break if time low
    return best_seq, best_val
```

"""
Propose a low‑evaluation step‑function.

The search works in three stages:

1. **Broad seeding** – generate a large, diverse pool of candidates
   (uniform, Dirichlet draws, exponential / power‑law families,
   beta‑shapes, sinusoidal, random permutations, warm‑start interpolation).
   All candidates are norm‑ed to ∑a_i = 1 and quickly evaluated with a
   FFT‑based convolution.  The best few (default 8) are kept.

2. **Deep optimisation** – each seed is refined by
   *Adam* on a log‑sum‑exp surrogate of the max‑convolution
   (smooth‑max temperature is increased steadily) and
   then by an *exponentiated‑gradient sub‑gradient* step that directly
   reduces the true max‑convolution.  Time budget is split among the
   seeds, allowing many thousands of iterations.

3. **Simulated‑annealing polishing** – the current global best is
   perturbed by tiny random mass transfers.  Moves are accepted with
   Metropolis probability, slowly cooling the temperature.
   This helps escape shallow local minima.

All FFT work uses real‑FFT for speed, everything stays on the probability
simplex (non‑negative, re‑normalised after each update).  The function
returns the best sequence found (as a plain Python list).  If nothing
works, the fallback is the uniform distribution (objective = 2.0).

The algorithm is fully self‑contained and respects the 1000 s deadline.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper FFT utilities
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via real FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """Fast evaluation of 2·n·max(a*a) / (sum(a)**2).  Assumes sum(a)>0."""
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c / (a.sum() ** 2))


# ----------------------------------------------------------------------
# Adam optimisation on a smooth‑max surrogate
def _adam_smooth_max(a0: np.ndarray,
                     n: int,
                     conv_fft_len: int,
                     corr_fft_len: int,
                     deadline: float,
                     global_best_seq: np.ndarray | None,
                     global_best_val: float) -> tuple[np.ndarray, float]:
    eps = 1e-15
    logits = np.log(a0 + eps)                 # soft‑max parameters
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.06                                 # a little smaller for stability
    lam = 0.3                                 # start temperature
    lam_factor = 1.03
    lam_cap = 1e6

    best_seq = global_best_seq
    best_val = global_best_val
    it = 0
    max_iter = 400_000
    last_improve = time.time()

    while time.time() < deadline and it < max_iter:
        # simplex via soft‑max
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # surrogate objective (smooth‑max)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Adam n={n}] it={it:06d} new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)                 # ⟨a,∇⟩ term from chain rule
        grad_log = a * (grad_a - avg)           # gradient wrt logits

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature schedule (smooth‑max becomes true max)
        lam = min(lam * lam_factor, lam_cap)

        # occasional jitter if stuck
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
# Sub‑gradient refinement on the true max‑convolution
def _subgradient_refinement(a0: np.ndarray,
                            n: int,
                            conv_fft_len: int,
                            corr_fft_len: int,
                            deadline: float,
                            init_best_seq: np.ndarray | None,
                            init_best_val: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    a = np.maximum(a, 0.0)
    a /= a.sum()

    best_seq = a.copy()
    best_val = init_best_val
    eta0 = 0.55
    it = 0
    max_iter = 60_000
    last_improve = time.time()

    while time.time() < deadline and it < max_iter:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Subgrad n={n}] it={it:06d} new best {val:.6f}")

        # active (max) index – use a single hard max for a clean sub‑gradient
        k_max = int(conv.argmax())
        w = np.zeros_like(conv)
        w[k_max] = 1.0

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # decaying step‑size
        eta = eta0 / np.sqrt(it + 1)

        # exponentiated‑gradient update (keeps a ≥ 0)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:                         # numerical safety
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        if time.time() - last_improve > 6.0:
            break
        it += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated annealing fine‑tuning
def _anneal_optimize(a0: np.ndarray,
                     n: int,
                     conv_fft_len: int,
                     deadline: float) -> tuple[np.ndarray, float]:
    cur = a0.copy()
    cur_val = _objective(cur, n, conv_fft_len)
    best = cur.copy()
    best_val = cur_val

    # initialise temperature proportional to current value
    T = max(1e-4, cur_val * 0.05)

    it = 0
    while time.time() < deadline:
        # propose a tiny mass transfer between two random positions
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        max_transfer = min(cur[i], cur[j])
        if max_transfer < 1e-12:
            continue
        # up to ±30 % of the smaller of the two
        delta = (np.random.rand() - 0.5) * 0.6 * max_transfer
        # enforce positivity
        if delta > cur[j]:
            delta = cur[j]
        if -delta > cur[i]:
            delta = -cur[i]
        if abs(delta) < 1e-14:
            continue

        # apply move
        cur[i] += delta
        cur[j] -= delta
        new_val = _objective(cur, n, conv_fft_len)

        dE = new_val - cur_val
        if dE < 0 or np.random.rand() < np.exp(-dE / max(T, 1e-12)):
            # accept
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = cur.copy()
                # a small drop in temperature after a genuine improvement
                T *= 0.97
        else:
            # reject – revert
            cur[i] -= delta
            cur[j] += delta

        # cooling schedule
        if it % 1000 == 0 and it > 0:
            T *= 0.95
        if T < 1e-12:
            break
        it += 1

    return best, best_val

# Deep optimisation pipeline (Adam → sub‑gradient → annealing)
def _deep_optimize(a0: np.ndarray,
                   n: int,
                   conv_fft_len: int,
                   corr_fft_len: int,
                   deadline: float,
                   global_best_seq: np.ndarray | None,
                   global_best_val: float) -> tuple[np.ndarray, float]:
    # Phase 1 – smooth‑max Adam
    best_seq, best_val = _adam_smooth_max(
        a0, n, conv_fft_len, corr_fft_len, deadline,
        global_best_seq, global_best_val,
    )
    # Phase 2 – exact sub‑gradient
    best_seq, best_val = _subgradient_refinement(
        a0, n, conv_fft_len, corr_fft_len, deadline,
        best_seq, best_val,
    )
    return best_seq, best_val

# Pool generation – many (mostly non‑symmetric) families
def _generate_parametric_candidates(n: int, warm_start: np.ndarray | None) -> list[np.ndarray]:
    i = np.arange(n, dtype=np.float64)
    candidates: list[np.ndarray] = []

    # 1. Warm‑start (if any) and a few cyclic shifts
    if warm_start is not None:
        ws = warm_start.copy()
        if ws.shape[0] != n:
            # linear interpolation to required size
            x_old = np.linspace(0, 1, ws.shape[0], endpoint=False)
            x_new = np.linspace(0, 1, n, endpoint=False)
            ws = np.interp(x_new, x_old, ws)
        ws = np.clip(ws, 0.0, None)
        if ws.sum() >= 0.01:
            ws = ws / ws.sum()
            candidates.append(ws)
            for shift in (n // 4, n // 2, 3 * n // 4):
                candidates.append(np.roll(ws, shift))

    # 2. Uniform baseline
    candidates.append(np.full(n, 1.0 / n))

    # 3. Random Dirichlet draws with various concentrations
    for conc in (0.02, 0.05, 0.1, 0.3, 0.6, 1.0, 2.0):
        cand = np.random.dirichlet(np.full(n, conc))
        candidates.append(cand)

    # 4. Random permutations of uniform and of a Dirichlet draw
    base = np.full(n, 1.0 / n)
    for _ in range(4):
        perm = np.random.permutation(n)
        candidates.append(base[perm])

    # 5. Edge‑exponential families (not forced symmetric – random shift)
    for alpha in np.logspace(-2, 1.0, num=12):
        edge = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
        edge = edge / edge.sum()
        candidates.append(edge)                                    # symmetric
        # same shape shifted by a random cyclic offset
        shift = np.random.randint(n)
        candidates.append(np.roll(edge, shift))

    # 6. Power‑law edge families
    for p in np.linspace(0.2, 4.5, num=12):
        edge = (i + 1.0) ** (-p) + (n - i) ** (-p)
        edge = edge / edge.sum()
        candidates.append(edge)

    # 7. Symmetric beta‑shapes   a_i ∝ (i+1)^{α‑1}(n‑i)^{α‑1}
    for alpha in (0.2, 0.4, 0.7, 1.0, 1.5, 2.5):
        shape = (i + 1.0) ** (alpha - 1.0) * (n - i) ** (alpha - 1.0)
        shape = shape / shape.sum()
        candidates.append(shape)

    # 8. Raised‑cosine / sine‑power families (non‑symmetric variants)
    for p in (1, 2, 3, 4, 6):
        shape = np.sin(np.pi * (i + 0.5) / n) ** p
        shape = shape / shape.sum()
        candidates.append(shape)
        # random cyclic shift
        candidates.append(np.roll(shape, np.random.randint(n)))

    # 9. Two‑Gaussian edge shapes (different σ)
    for gamma in (0.05, 0.15, 0.3, 0.6, 1.2):
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        shape = shape / shape.sum()
        candidates.append(shape)

    # 10. Arcsine‑type shape
    shape = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    shape = shape / shape.sum()
    candidates.append(shape)

    return candidates

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Main entry point – search for a non‑negative step function that minimizes
    ``evaluate_sequence``.  Returns the best sequence (plain Python list).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # 0.5 s safety margin

    # ------------------------------------------------------------------
    # Choose a problem size – larger size gives more freedom.
    # If we have very little time left we fall back to 4096.
    time_left = deadline - time.time()
    n = 8192 if time_left > 300 else 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm‑start (optional)
    warm_start = globals().get("height_sequence_1")
    if warm_start is not None:
        ws = np.asarray(warm_start, dtype=np.float64)
        if ws.shape[0] != n:
            # interpolate to the desired length
            x_old = np.linspace(0, 1, ws.shape[0], endpoint=False)
            x_new = np.linspace(0, 1, n, endpoint=False)
            ws = np.interp(x_new, x_old, ws)
        ws = np.clip(ws, 0.0, None)
        if ws.sum() >= 0.01:
            ws = ws / ws.sum()
        else:
            ws = None
    else:
        ws = None

    # ------------------------------------------------------------------
    # Stage 1 – broad candidate generation & quick pre‑selection
    cand_pool = _generate_parametric_candidates(n, ws)

    # a few extra random Dirichlets for extra diversity
    for _ in range(25):
        if time.time() >= deadline:
            break
        conc = np.random.choice([0.02, 0.05, 0.1, 0.3, 0.6, 1.0, 2.0])
        cand_pool.append(np.random.dirichlet(np.full(n, conc)))

    evaluated = []
    for cand in cand_pool:
        if time.time() >= deadline:
            break
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 0.01:               # re‑sample if too degenerate
            cand = np.random.dirichlet(np.ones(n))
            s = cand.sum()
        cand = cand / s
        val = _objective(cand, n, conv_fft_len)
        evaluated.append((val, cand))

    if not evaluated:               # safety fallback
        return np.full(n, 1.0 / n).tolist()

    evaluated.sort(key=lambda x: x[0])
    top_k = min(8, len(evaluated))
    top_candidates = [c for _, c in evaluated[:top_k]]
    best_seq: np.ndarray | None = top_candidates[0].copy()
    best_val = evaluated[0][0]
    print(f"[Stage 1] n={n} selected {top_k} seeds, best preliminary {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – deep optimisation on the selected seeds
    for idx, start_vec in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        remaining = deadline - time.time()
        # allocate a fair share of the remaining budget to this seed
        sub_deadline = time.time() + 0.85 * remaining / (len(top_candidates) - idx)
        print(f"[Deep opt] seed {idx + 1}/{len(top_candidates)} – deadline in {remaining:.1f}s")
        seq, val = _deep_optimize(start_vec, n, conv_fft_len, corr_fft_len,
                                   sub_deadline, best_seq, best_val)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[Update] new global best {best_val:.6f} from seed {idx + 1}")

    # ------------------------------------------------------------------
    # Stage 3 – final annealing / hill‑climbing if time permits
    if best_seq is not None and time.time() < deadline - 5.0:
        print("[Annealing] polishing the best solution")
        seq, val = _anneal_optimize(best_seq, n, conv_fft_len, deadline)
        if val < best_val:
            best_val, best_seq = val, seq
            print(f"[Anneal] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback guard
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the evaluator’s allowed range
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"*** Search complete – best objective ≈ {best_val:.6f} (n={best_seq.shape[0]})")
    return best_seq.tolist()
