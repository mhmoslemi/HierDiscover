"""
Hybrid optimizer with a linear‑programming (LP) “mass–inflation” phase.

*   Small problem (n≈1024) – repeatedly solve a linear program that
    maximises the total mass `∑g_j` under the constraint `f∗g ≤ M`
    (where `M = max(f∗f)`).  The LP produces a vector with larger total
    weight that respects the current worst‑case convolution bound.
    Mixing the current iterate `f` with the scaled LP solution strictly
    reduces the bound.

*   The LP stage is fast (dense matrix of size O(n²) with n≈1024) and
    converges in a few dozen iterations.

*   The resulting shape is up‑sampled to the target size (4096 or 8192)
    and refined using the Adam / exact‑sub‑gradient pipeline from the
    reference implementation.

The routine respects the supplied time budget and always returns a
non‑negative list with sum > 0.01 (required by `evaluate_sequence`).
"""

import time
import numpy as np

# SciPy is optional – if unavailable we simply skip the LP stage.
try:
    from scipy.optimize import linprog
    _SCIPY = True
except Exception:                     # pragma: no cover
    _SCIPY = False

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Cross‑correlation w ★ a (used for gradients)."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(a∗a) for a (sum is assumed to be 1)."""
    n = a.shape[0]
    conv = np.convolve(a, a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
def _lp_improvement_step(f: np.ndarray, t_mix: float) -> tuple[np.ndarray, bool]:
    """
    One LP‑based improvement of the current vector `f`.

    Returns the updated vector and a flag indicating whether the LP solved
    successfully.
    """
    n = f.shape[0]
    conv = np.convolve(f, f)
    M = conv.max()

    # Build the convolution matrix A such that (A @ g)[k] = (f∗g)[k].
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for j in range(n):
        A[j : j + n, j] = f

    if not _SCIPY:                     # pragma: no cover
        return f, False

    # maximise ∑g_j  ⇔ minimise −∑g_j
    res = linprog(
        c=-np.ones(n),
        A_ub=A,
        b_ub=np.full(rows, M),
        bounds=[(0, None)] * n,
        method="highs",
    )
    if not res.success:
        return f, False

    g0 = np.maximum(res.x, 0.0)        # numerical cleanup
    sum_g0 = g0.sum()
    if sum_g0 <= 1e-12:
        return f, False

    # Scale g to keep the total mass equal to 1 (same as f).
    g = g0 * (1.0 / sum_g0)

    # Convex mixing – the mixing factor slowly decays across iterations.
    f_new = (1.0 - t_mix) * f + t_mix * g
    f_new = np.maximum(f_new, 0.0)
    f_new /= f_new.sum()
    return f_new, True


# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """Adam optimisation on the smooth‑max (log‑sum‑exp) surrogate."""
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.03
    lam = 0.3
    lam_factor = 1.015

    best_a = a0.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution_fft(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum
        # gradient
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        lam = min(lam * lam_factor, 1e6)

        # tiny jitter near the end to escape plateaus
        if it % 4000 == 0 and time.time() > deadline - 5:
            logits += (np.random.rand(n) - 0.5) * 1e-3
        it += 1
    return best_a, best_val


def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 0.6,
) -> tuple[np.ndarray, float]:
    """Exact‑max sub‑gradient descent with exponentiated‑gradient updates."""
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s

    best_a = a.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution_fft(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
        # active set of max entries
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1)
        a = a * np.exp(-eta * grad_a)

        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s
        it += 1
    return best_a, best_val


# ----------------------------------------------------------------------
def _upsample(a_small: np.ndarray, n_big: int) -> np.ndarray:
    """Linear interpolation from a_small (size n_small) to n_big."""
    n_small = a_small.shape[0]
    x_small = np.linspace(0.0, 1.0, n_small, endpoint=False)
    x_big = np.linspace(0.0, 1.0, n_big, endpoint=False)
    a_big = np.interp(x_big, x_small, a_small)
    a_big = np.maximum(a_big, 0.0)
    a_big /= a_big.sum()
    return a_big

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises
    `2·n·max(a∗a)/(sum a)^2`.

    The algorithm consists of:
        1. A small‑scale LP “mass‑inflation” phase (n≈1024) that rapidly
           reduces the worst‑case convolution.
        2. Upsampling to the target size (4096 or 8192).
        3. Refinement with Adam on a smooth‑max surrogate and exact‑max
           sub‑gradient descent.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # 1️⃣  Small‑scale LP phase
    n_small = 1024
    # warm start from a previously‑found global (if any)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n_small:
            # simple down‑sampling by uniform stride
            idx = np.round(np.linspace(0, warm.shape[0] - 1, n_small)).astype(int)
            warm = warm[idx]
        elif warm.shape[0] < n_small:
            warm = np.pad(warm, (0, n_small - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm /= warm.sum()
        else:
            warm = None
    if warm is None:
        warm = np.random.dirichlet(np.ones(n_small))

    # a few random Dirichlet starts – keep the best as `f`
    candidates = [warm]
    for alpha in (0.2, 0.5, 1.0, 2.0):
        cand = np.random.dirichlet(alpha * np.ones(n_small))
        candidates.append(cand)

    best_f = None
    best_val = float("inf")
    for cand in candidates:
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()
        val = _objective(cand)
        if val < best_val:
            best_val = val
            best_f = cand.copy()
    print(f"[Init] best small‑scale objective: {best_val:.6f}")

    # Repeated LP improvement – the mixing factor slowly decays.
    t_mix = 0.35
    while time.time() < deadline - 2.0:
        new_f, success = _lp_improvement_step(best_f, t_mix)
        if not success:
            break
        new_val = _objective(new_f)
        if new_val < best_val:
            best_val = new_val
            best_f = new_f.copy()
            print(f"[LP] improved to {new_val:.6f}")
        # slowly decrease the mixing weight
        t_mix = max(0.05, t_mix * 0.93)

    # --------------------------------------------------------------
    # 2️⃣ Upsample to a larger grid
    # Decide target resolution based on remaining time.
    time_left = deadline - time.time()
    n_big = 8192 if (time_left > 200) else 4096
    a_big = _upsample(best_f, n_big)

    # --------------------------------------------------------------
    # 3️⃣ Refine with Adam + exact sub‑gradient
    conv_fft_len = _next_pow2(2 * n_big - 1)
    corr_fft_len = _next_pow2(3 * n_big - 2)

    # Adam phase – respect the global deadline.
    a_adam, val_adam = _adam_optimize(
        a_big,
        n_big,
        conv_fft_len,
        corr_fft_len,
        deadline,
        max_iters=180_000,
    )
    # Sub‑gradient refinement
    a_sub, val_sub = _subgradient_optimize(
        a_adam,
        n_big,
        conv_fft_len,
        corr_fft_len,
        deadline,
        max_iters=120_000,
    )

    # Choose the best among the three versions.
    candidates_big = [(a_big, _objective(a_big)),
                      (a_adam, val_adam),
                      (a_sub, val_sub)]
    best_seq, best_val_big = min(candidates_big, key=lambda kv: kv[1])

    # --------------------------------------------------------------
    # Final bookkeeping
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"Final reported value {final_val:.6f} (length {len(best_seq)})")
    except Exception as exc:            # pragma: no cover
        print(f"Could not call evaluate_sequence: {exc}")
        print(f"Best internal value {best_val_big:.6f}")

    return best_seq.tolist()
