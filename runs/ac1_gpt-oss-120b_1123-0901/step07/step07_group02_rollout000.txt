"""
Propose a non‑negative step‑function (vector) that minimizes

        F(a) = 2·n·max(a∗a) / (sum(a)^2)

Approach
--------
1. Build a diverse pool of start vectors (uniform, Dirichlet, classic
   window functions – Hann, Hamming, Blackman, Kaiser – and a few
   edge‑concentrated power‑law shapes).  If a previous solution
   ``height_sequence_1`` is present it is added as a warm start.

2. For each candidate run a two‑stage continuous optimiser:
   * Adam on a smooth‑max (log‑sum‑exp) surrogate of the convolution
     max.  The variables are the logits of a soft‑max, guaranteeing that
     the iterate stays on the probability simplex.
   * A long exponentiated‑gradient (multiplicative) sub‑gradient phase
     that directly minimizes the true max‑convolution objective.

3. Keep the best sequence across all candidates and refine it further
   with another long sub‑gradient run and a cheap random‑move hill‑climb.

4. If time permits upscale the solution to a finer grid
   (4096 → 8192 → 16384) by linear interpolation and repeat
   short Adam + sub‑gradient refinements on the larger problems.

All FFT based convolutions are performed with powers‑of‑two lengths for
speed.  The routine respects the wall‑clock budget (default 1000 s) and
always returns the best feasible vector found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a):
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i = 0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the smooth‑max surrogate.
    Returns the best vector (sum≈1) seen and its exact objective.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)          # work in log‑space → soft‑max
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.08                         # base learning rate
    lam = 0.5                         # initial “temperature’’
    lam_factor = 1.03                 # geometric increase

    best_a = a0.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        # ----- simplex projection via soft‑max -----
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # ----- exact objective (sum(a)==1) -----
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam iter={it:06d}] new best {val:.6f}")

        # ----- smooth‑max weights (log‑sum‑exp) -----
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the surrogate -----
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)       # chain‑rule through soft‑max

        # ----- Adam update on logits -----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- temperature schedule -----
        lam = min(lam * lam_factor, 1e6)

        # ----- occasional tiny random shake (helps escape tiny plateaus) -----
        if it % 7_000 == 0 and time.time() + 5 > deadline:
            logits += (np.random.rand(n) - 0.5) * 5e-3

        it += 1

    return best_a, best_val


def _subgradient_opt2(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
    eta0: float = 0.7,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient phase with a decaying step size.
    Returns the best vector seen and its exact objective.
    """
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s

    best_a = a.copy()
    best_val = float("inf")
    stall = 0
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            stall = 0
            if it % 5_000 == 0:
                print(f"[Subgrad iter={it:07d}] new best {val:.6f}")
        else:
            stall += 1

        # active set (indices where the max is attained)
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # decaying step size
        eta = eta0 / np.sqrt(it + 1)

        # multiplicative update + renorm
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        # random perturbation after long stagnation
        if stall > 10_000:
            a = np.maximum(a + 0.02 * (np.random.rand(n) - 0.5), 0.0)
            a /= a.sum()
            stall = 0

        it += 1
    return best_a, best_val


def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climbing: move a random fraction of mass from
    one index to another, keep the move if the objective improves.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    best_a = a.copy()
    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-12:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue
        # move up to 40 % of the source mass
        delta = a[i] * (np.random.rand() * 0.4)
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta
        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()
        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5_000 == 0:
                print(f"[Local move {it}] improved to {val2:.6f}")
        it += 1
    return best_a, best_val

            a0 = a0 / s

            a0, n, conv_fft_len, corr_fft_len, cand_deadline, max_iters=250_000

            best_seq = a_adam.copy()

        # ------- Sub‑gradient refinement -------
        if time.time() < deadline - 1.0:
            sub_deadline = min(deadline - 1.0, time.time() + per_cand_budget * 0.4)
            a_sub, val_sub = _subgradient_opt2(
                a_adam, n, conv_fft_len, corr_fft_len, sub_deadline, max_iters=150_000
            )
            if val_sub < best_val:
                best_val = val_sub
                best_seq = a_sub.copy()
                print(f"[cand {cid}] Subgrad best {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Global long sub‑gradient refinement (most remaining budget)
    if best_seq is not None and time.time() < deadline - 2.0:
        a_ref, val_ref = _subgradient_opt2(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline - 1.0,
            max_iters=400_000,
            eta0=0.9,
        )
        if val_ref < best_val:
            best_val = val_ref
            best_seq = a_ref.copy()
            print(f"[refine] Subgrad long best {val_ref:.6f}")

    # ------------------------------------------------------------------
    # Cheap random local search
    if best_seq is not None and time.time() < deadline - 2.0:
        a_loc, val_loc = _local_random_search(
            best_seq, n, conv_fft_len, deadline - 1.0, max_moves=250_000
        )
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()
            print(f"[local] improved best {val_loc:.6f}")

    # ------------------------------------------------------------------
    # Upscaling phases (if time permits)
    def _upscale_and_refine(a_small, n_small, n_big, time_allowed):
        """Interpolate to a larger grid and run a short Adam+sub‑grad."""
        x_s = np.linspace(0, 1, n_small, endpoint=False)
        x_b = np.linspace(0, 1, n_big, endpoint=False)
        a_big = np.interp(x_b, x_s, a_small)
        a_big = np.maximum(a_big, 0.0)
        a_big /= a_big.sum()

        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        sub_deadline = time.time() + time_allowed * 0.4
        a_adam, val_adam = _adam_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            sub_deadline,
            max_iters=150_000,
        )
        best_local = a_adam
        best_local_val = val_adam

        if time.time() + 0.5 < time.time() + time_allowed * 0.6:
            sub_deadline2 = time.time() + time_allowed * 0.6
            a_sub, val_sub = _subgradient_opt2(
                a_adam,
                n_big,
                conv_fft_len_big,
                corr_fft_len_big,
                sub_deadline2,
                max_iters=250_000,
            )
            if val_sub < best_local_val:
                best_local_val = val_sub
                best_local = a_sub

        return best_local, best_local_val

    # Try 8192 if enough time left
    if time.time() < deadline - 2.0:
        time_left = deadline - time.time()
        a_big, val_big = _upscale_and_refine(best_seq, n, 8192, time_left * 0.5)
        if val_big < best_val:
            best_val = val_big
            best_seq = a_big.copy()
            print(f"[8192] improved best {val_big:.6f}")

    # Try 16384 if still time left
    if time.time() < deadline - 2.0:
        time_left = deadline - time.time()
        a_big, val_big = _upscale_and_refine(best_seq, best_seq.shape[0], 16384, time_left * 0.5)
        if val_big < best_val:
            best_val = val_big
            best_seq = a_big.copy()
            print(f"[16384] improved best {val_big:.6f}")

    # ------------------------------------------------------------------
    # Final safety fallback
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best approx {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
