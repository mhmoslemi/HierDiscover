def subgradient_with_momentum(a0, n, conv_fft_len, corr_fft_len, deadline, max_iters, eta0):
    a = a0.copy()
    best_a, best_val = a.copy(), np.inf
    m = np.zeros_like(a)
    beta = 0.9
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum()**2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
        # find active
        active = np.isclose(conv, max_c, atol=1e-14)
        # compute w uniform over active
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)
        # momentum update on gradient in log-space
        eta = eta0 / np.sqrt(it + 1)
        m = beta*m + (1-beta)*grad
        a = a * np.exp(-eta * m)
        # renormalize
        s = a.sum()
        if s <= 0 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s
    return best_a, best_val

def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5

    # Phase 1: choose problem size
    n = 4096
    conv_fft_len = _next_pow2(2*n-1)
    corr_fft_len = _next_pow2(3*n-2)

    best_seq = None
    best_val = float('inf')
    
    # Phase A: parametric sweep
    # # family 1: edge-exponential
    for gamma in np.linspace(0.05, 4.0, 60):
        if time.time() >= deadline: break
        left = np.exp(-gamma * np.arange(n, dtype=np.float64))
        right = left[::-1]
        a = left + right
        a_sum = a.sum()
        if a_sum == 0: continue
        a /= a_sum
        val = _objective(a, n)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Sweep exp gamma={gamma:.3f}] new best {val:.6f}")
    
    # family 2: power law (beta-like)
    alphas = np.linspace(-3.0, -0.2, 30)
    betas = np.linspace(-3.0, -0.2, 30)
    for alpha in alphas:
        for beta in betas:
            if time.time() >= deadline: break
            idx = np.arange(1, n+1, dtype=np.float64)
            a = (idx ** alpha) * ((n - idx + 1) ** beta)
            total = a.sum()
            if total == 0: continue
            a /= total
            val = _objective(a, n)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[Sweep power alpha={alpha:.2f} beta={beta:.2f}] new best {val:.6f}")
    
    # family 3: double Gaussian
    sigma_vals = (0.05, 0.1, 0.2, 0.3, 0.5)
    for sigma1 in sigma_vals:
        for sigma2 in sigma_vals:
            if time.time() >= deadline: break
            i = np.arange(n, dtype=np.float64)
            gauss1 = np.exp(-0.5 * ((i) / (sigma1 * n))**2)
            gauss2 = np.exp(-0.5 * ((i - (n-1)) / (sigma2 * n))**2)
            a = gauss1 + gauss2
            a_sum = a.sum()
            if a_sum == 0: continue
            a /= a_sum
            val = _objective(a, n)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[Sweep gauss sigma1={sigma1:.2f} sigma2={sigma2:.2f}] new best {val:.6f}")
    
    # Phase B: refine best with Adam + subgradient
    if best_seq is not None and time.time() < deadline - 10:
        a_ref, val_ref = _adam_optimize(best_seq, n, conv_fft_len, corr_fft_len, deadline, max_iters=250_000)

    if best_seq is not None and time.time() < deadline - 10:
        a_ref, val_ref = _subgradient_momentum(best_seq, n, conv_fft_len, corr_fft_len, deadline, max_iters=500_000, eta0=0.9)
        if val_ref < best_val:
            best_val = val_ref
            best_seq = a_ref.copy()
            print(f"[Subgrad momentum] new best {val_ref:.6f}")

    # Phase C: random local moves
    if best_seq is not None and time.time() < deadline - 5:
        a_loc, val_loc = _local_random_search(best_seq, n, conv_fft_len, deadline, max_moves=100_000)
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()
            print(f"[Local search] new best {val_loc:.6f}")

    # Phase D: upscale to larger grid if time
    if time.time() + 30 < deadline:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2*n_big - 1)
        corr_fft_len_big = _next_pow2(3*n_big - 2)
        # linear interpolation

        x_new = np.linspace(0, 1, n_big, endpoint=False)

        a_big = np.maximum(a_big, 0.0)

        # refine big
        a_big_ref, val_big_ref = _adam_optimize(a_big, n_big, conv_fft_len_big, corr_fft_len_big, deadline, max_iters=200_000)
        if val_big_ref < best_val:
            best_val = val_big_ref
            best_seq = a_big_ref.copy()
            print(f"[Big Adam] new best {val_big_ref:.6f}")

        # final subgradient on big if time remains
        if time.time() < deadline - 5:
            a_big_ref2, val_big_ref2 = _subgradient_momentum(best_seq, len(best_seq), _next_pow2(2*len(best_seq)-1), _next_pow2(3*len(best_seq)-2), deadline, max_iters=250_000, eta0=0.8)
            if val_big_ref2 < best_val:
                best_val = val_big_ref2
                best_seq = a_big_ref2.copy()
                print(f"[Big Subgrad] final best {val_big_ref2:.6f}")

    # final clamp
    if best_seq is None:
        best_seq = np.full(4096, 1.0/4096)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()

"""
Propose a non‑negative step‑function (vector) that minimizes

        F(a) = 2·n·max(a∗a) / (sum(a)^2)

The search works in four stages:

1. **Fast parametric sweep** – explore three families of simple shapes
   (edge‑exponential, beta‑type power law, double‑Gaussian).  The sweep
   evaluates thousands of candidates with an FFT‑based surrogate of the
   true objective, keeping the best.

2. **Adam refinement** – start from the best candidate and run Adam on a
   smooth‑max (log‑sum‑exp) surrogate.  The temperature is increased
   slowly so the optimizer first sees a smooth landscape and then the
   true max.

3. **Momentum sub‑gradient phase** – an exponentiated‑gradient (mirror‑
   descent) update on the simplex with a momentum term.  The step size
   decays like η₀/√t, which works well for the non‑smooth objective.

4. **Optional up‑sampling** – if time permits, the current best shape is
   interpolated to a finer grid (8192 points) and the Adam+sub‑gradient
   phases are repeated.

All phases respect the global deadline (default 1000 s).  The routine
always returns the best sequence found so far.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a length‑len(a) vector:
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i = 0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Fast evaluation of 2·n·max(a∗a) / (sum(a)^2)."""
    conv = _full_convolution(a, conv_fft_len)
    return float(2.0 * n * conv.max() / (a.sum() ** 2))


# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """Adam on the smooth‑max (log‑sum‑exp) surrogate."""
    eps = 1e-15
    logits = np.log(a0 + eps)                # initialise on the simplex
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.07                                 # a bit larger than before
    lam = 0.05                                 # start temperature (soft)
    lam_factor = 1.05

    best_a = a0.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        # simplex projection (soft‑max)
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # mirror‑descent form (soft‑max Jacobian)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # temperature schedule (making the surrogate sharper)
        lam = min(lam * lam_factor, 1e7)

        # occasional tiny random shake near the deadline
        if it % 4000 == 0 and time.time() > deadline - 60:
            logits += (np.random.rand(n) - 0.5) * 2e-2

        # very gentle learning‑rate decay
        if it % 50_000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 0.8,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient phase using exponentiated‑gradient updates with
    a momentum term.  Step size ηₜ = η₀ / sqrt(t+1).
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0
    # momentum on the gradient (in log‑space)
    grad_mom = np.zeros_like(a)
    beta = 0.9

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-14)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()
        # gradient of the max (sub‑gradient)
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # momentum update
        grad_mom = beta * grad_mom + (1.0 - beta) * grad

        # decreasing step size
        eta = eta0 / np.sqrt(it + 1)

        # exponentiated‑gradient (mirror descent on the simplex)
        a = a * np.exp(-eta * grad_mom)

        # renormalise – protect against underflow
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 60_000,
) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climbing: move a random fraction of mass from one index
    to another, keep the move if the objective improves.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()

    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        # move up to 30 % of the source mass
        delta = a[i] * (np.random.rand() * 0.3)

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5000 == 0:
                print(f"[Local move {it:06d}] improved to {val2:.6f}")

        it += 1

    return best_a, best_val

#   Helper generators for the parametric sweep

def _gen_edge_exponential(n: int, gamma: float) -> np.ndarray:
    """Symmetric U‑shape: left‑exp + right‑exp, normalised."""
    left = np.exp(-gamma * np.arange(n, dtype=np.float64))
    right = left[::-1]
    a = left + right
    a_sum = a.sum()
    if a_sum == 0:
        return np.full(n, 1.0 / n)
    return a / a_sum


def _gen_beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Power‑law (beta) shape: (i+1)^α · (n‑i)^β, normalised.
    For negative exponents the distribution concentrates at the edges.
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** alpha) * ((n - i + 1) ** beta)
    total = a.sum()
    if total == 0:
        return np.full(n, 1.0 / n)
    return a / total


def _gen_double_gaussian(n: int, sigma1: float, sigma2: float) -> np.ndarray:
    """
    Two Gaussian bumps glued to the left and right borders.
    """
    i = np.arange(n, dtype=np.float64)
    g1 = np.exp(-0.5 * ((i) / (sigma1 * n)) ** 2)
    g2 = np.exp(-0.5 * ((i - (n - 1)) / (sigma2 * n)) ** 2)
    a = g1 + g2
    s = a.sum()
    if s == 0:
        return np.full(n, 1.0 / n)
    return a / s

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the given time budget and always returns the
    best sequence (as a Python list) found.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5   # tiny safety margin

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # 1.  PARAMETRIC SWEEP
    # --------------------------------------------------------------
    best_seq = None
    best_val = float("inf")

    # ---- edge‑exponential family ---------------------------------
    gammas = np.linspace(0.02, 4.0, 120)   # finer grid than before
    for gamma in gammas:
        if time.time() >= deadline:
            break
        a = _gen_edge_exponential(n, gamma)
        val = _objective(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Sweep exp γ={gamma:.3f}] new best {val:.6f}")

    # ---- beta‑type power‑law family -------------------------------
    alphas = np.linspace(-3.0, -0.1, 30)
    betas = np.linspace(-3.0, -0.1, 30)
    for alpha in alphas:
        for beta in betas:
            if time.time() >= deadline:
                break
            a = _gen_beta_shape(n, alpha, beta)
            val = _objective(a, n, conv_fft_len)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[Sweep beta α={alpha:.2f} β={beta:.2f}] new best {val:.6f}")

    # ---- double‑Gaussian family -----------------------------------
    sigmas = (0.04, 0.07, 0.10, 0.15, 0.22, 0.30, 0.45)
    for s1 in sigmas:
        for s2 in sigmas:
            if time.time() >= deadline:
                break
            a = _gen_double_gaussian(n, s1, s2)
            val = _objective(a, n, conv_fft_len)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[Sweep gauss σ₁={s1:.2f} σ₂={s2:.2f}] new best {val:.6f}")

    # ---- random Dirichlet draws (to seed diversity) ------------
    for _ in range(30):
        if time.time() >= deadline:

        a = np.random.dirichlet(np.ones(n))

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Random Dirichlet] new best {val:.6f}")

    # --------------------------------------------------------------
    # 2.  ADAM REFINEMENT (smooth‑max)
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 10:
        a_adam, val_adam = _adam_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=250_000,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Adam refine] new best {val_adam:.6f}")

    # --------------------------------------------------------------
    # 3.  MOMENTUM SUB‑GRADIENT REFINEMENT
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 10:
        a_sub, val_sub = _subgradient_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=600_000,
            eta0=0.9,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Subgrad momentum] new best {val_sub:.6f}")

    # --------------------------------------------------------------
    # 4.  CHEAP LOCAL RANDOM HILL‑CLIMBING
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5:
        a_loc, val_loc = _local_random_search(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=120_000,
        )
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()
            print(f"[Local search] new best {val_loc:.6f}")

    # --------------------------------------------------------------
    # 5.  OPTIONAL UPSAMPLING TO 8192 POINTS (if time permits)
    # --------------------------------------------------------------
    if time.time() + 30 < deadline:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        # linear interpolation to a finer grid
        if best_seq is not None:
            x_old = np.linspace(0, 1, n, endpoint=False)
            x_new = np.linspace(0, 1, n_big, endpoint=False)
            a_big = np.interp(x_new, x_old, best_seq)
        else:
            a_big = np.random.dirichlet(np.ones(n_big))

        a_big = np.maximum(a_big, 0.0)
        a_big = a_big / a_big.sum()

        # Adam on the larger grid
        a_big_adam, _ = _adam_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            max_iters=180_000,
        )

        # Sub‑gradient refinement on the large grid
        a_big_ref, val_big_ref = _subgradient_optimize(
            a_big_adam,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            max_iters=300_000,
            eta0=0.8,
        )
        if val_big_ref < best_val:
            best_val = val_big_ref
            best_seq = a_big_ref.copy()
            print(f"[Big grid] new best {val_big_ref:.6f}")

    # --------------------------------------------------------------
    # 6.  FALLBACK – ensure a valid return
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Clip to the evaluator’s accepted range (non‑negative, ≤1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
