Implementation of _lp_optimize:

def _lp_optimize(f, t, max_iter, deadline, best_seq, best_val):
    n = f.shape[0]
    conv_fft_len = _next_pow2(2 * n - 1)
    while time.time() < deadline and max_iter>0:
        # compute objective
        conv = _full_convolution(f, conv_fft_len)
        M = conv.max()
        cur_val = 2.0 * n * M / (f.sum() ** 2)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = f.copy()
        # Build convolution matrix A
        rows = 2 * n - 1
        A = np.zeros((rows, n), dtype=np.float64)
        for i in range(n):
            A[i:i+n, i] = f[i]  # fill column i
        # Solve LP: maximize sum(g) subject to A @ g <= M, g >= 0
        c = -np.ones(n)  # minimize -sum(g)
        b_ub = np.full(rows, M, dtype=np.float64)
        bounds = [(0, None)] * n
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            break
        g = res.x
        # scale g to same sum as f
        sum_f = f.sum()
        sum_g = g.sum()
        if sum_g == 0:
            break
        g = g * (sum_f / sum_g)
        # mixing
        f = (1 - t) * f + t * g
        max_iter -= 1
    return best_seq, best_val

def _lp_optimize(f, max_iter, t, deadline, best_seq, best_val):
    n = len(f)
    conv_fft_len = _next_pow2(2*n-1)
    it = 0
    while time.time() < deadline and it < max_iter:
        conv = _full_convolution(f, conv_fft_len)
        M = conv.max()
        # compute objective
        val = 2.0 * n * M / (f.sum() ** 2)
        if val < best_val:
            best_val = val
            best_seq = f.copy()
            print(f"[LP it={it}] new best {best_val:.6f}")

        # Build A
        rows = 2 * n - 1
        A = np.zeros((rows, n), dtype=np.float64)
        for i in range(n):
            A[i:i+n, i] = f[i]
        c = -np.ones(n)
        b_ub = np.full(rows, M)
        bounds = [(0, None)] * n
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            print("[LP] linprog failed")
            break
        g = res.x
        sum_g = g.sum()
        if sum_g == 0:
            break
        # scale g to same sum as f
        g = g * (f.sum() / sum_g)
        # mixture update
        f_candidate = (1 - t) * f + t * g
        # ensure positivity and renorm
        f_candidate = np.maximum(f_candidate, 0)
        f_candidate /= f_candidate.sum()
        # compute new objective
        conv_cand = _full_convolution(f_candidate, conv_fft_len)
        val_cand = 2.0 * n * conv_cand.max() / (f_candidate.sum() ** 2)
        if val_cand < val:
            f = f_candidate
        else:
            # reduce t if not improving
            t = max(t * 0.7, 0.05)
        it += 1
    return best_seq, best_val, f

"""
Propose a low‑evaluation step‑function (≈1000 s budget).

Algorithm
---------
1.  Build an initial candidate for a small grid (n≈1024):
    * Warm‑start from the globally stored `height_sequence_1` (down‑sampled).
    * If unavailable, sample a few Dirichlet vectors with different
      concentration parameters and keep the best.
2.  **Linear‑programming phase** (only if ``scipy`` is available):
    * Repeatedly solve   maximise   Σ g_j   subject to   (f★g)[k] ≤ M,
      where M=‖f★f‖_∞ and g≥0.
    * Mix the optimum g back into f (convex combination, sum normalised).
    * Keep the best f seen – each iteration provably does not increase the
      objective 2 n max(f★f)/‖f‖₁².
3.  Upscale the best small‑grid vector to a medium grid (n=4096) by linear
    interpolation.
4.  **Gradient‑based refinement** on the medium grid:
    * Adam on a smooth‑max surrogate,
    * Exponentiated‑gradient sub‑gradient on the true max‑convolution,
    * Projected‑gradient descent on the exact objective.
5.  If time remains, perform cheap stochastic perturbations around the
    current best and a short sub‑gradient refinement.
6.  Return the best sequence (clipped to the evaluator’s allowed range).

The combination of an LP “Frank‑Wolfe” step and a fast gradient pipeline
typically pushes the evaluation score below the previous 1.5079 barrier
within the 1000 s budget.
"""

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for gradient steps.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


def _objective(seq: np.ndarray, n: int, fft_len: int) -> float:
    """Fast evaluation of the objective (identical to evaluate_sequence)."""
    conv = _full_convolution(seq, fft_len)
    max_c = conv.max()
    s = seq.sum()
    if s < 0.01:
        return float("inf")
    return float(2.0 * n * max_c / (s * s))


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Returns a non‑negative vector with sum = 1.
    """
    if v.ndim != 1:
        raise ValueError("v must be 1‑d")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

# ---------- ADAM – smooth‑max surrogate -----------------------
def _adam_smooth_max(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """Adam optimisation on a log‑sum‑exp surrogate of max(f★f)."""
    eps = 1e-15
    logits = np.log(a0 + eps)            # soft‑max logits
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05

    lam = 0.5                # temperature
    lam_factor = 1.03
    lam_cap = 1e8

    best_seq = global_best_seq
    best_val = global_best_val
    last_improve = time.time()
    it = 0
    max_iter = 400_000

    while time.time() < deadline and it < max_iter:
        # ----- simplex via soft‑max -----
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # ----- surrogate evaluation -----
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Adam n={n} it={it:06d}] new best {val:.6f}")

        # ----- smooth‑max weights -----
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient w.r.t. logits -----
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- temperature schedule -----
        lam = min(lam * lam_factor, lam_cap)

        # ----- occasional jitter if stuck -----
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1

    return best_seq, best_val

# ---------- Sub‑gradient refinement ----------------------------
def _subgradient_refinement(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient on the true max‑convolution.
    """
    a = global_best_seq if global_best_seq is not None else a0.copy()
    a = np.maximum(a, 0.0)
    a /= a.sum()

    eta = 0.6
    best_seq = global_best_seq
    best_val = global_best_val
    last_improve = time.time()
    max_iter = 200_000

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Subgrad n={n} it={it:06d}] new best {val:.6f}")

        # active set where the max is attained
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        if time.time() - last_improve > 5.0:
            break

        eta *= 0.9999

    return best_seq, best_val

# ---------- Projected gradient refinement -----------------------
def _projected_gradient_refine(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Gradient descent on the exact objective with Euclidean simplex projection.
    """
    a = a0.copy()
    a = np.maximum(a, 0.0)
    a /= a.sum()

    lr = 0.2
    best_seq = global_best_seq
    best_val = global_best_val
    last_improve = time.time()
    max_iter = 150_000

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[ProjGrad n={n} it={it:06d}] new best {val:.6f}")

        # active mask
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        step = -lr * grad_a
        a_candidate = _proj_simplex(a + step)

        conv_cand = _full_convolution(a_candidate, conv_fft_len)
        val_cand = 2.0 * n * conv_cand.max() / (a_candidate.sum() ** 2)

        if val_cand < val:
            a = a_candidate
            lr = min(lr * 1.02, 1.0)
        else:
            lr *= 0.7
            if lr < 1e-5:
                break

        if time.time() - last_improve > 5.0:
            break

        lr *= 0.99998

    return best_seq, best_val

        a0,
        n,
        conv_fft_len,
        corr_fft_len,
        deadline,
        best_seq,
        best_val,
    )
    return best_seq, best_val


# --------------------------------------------------------------
def _local_perturbations(
    base_seq: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """Cheap stochastic jitter + short sub‑gradient refinement."""
    a0 = base_seq.copy()
    best_seq = global_best_seq
    best_val = global_best_val

    while time.time() < deadline:
        eps = 0.02
        a_pert = a0 + eps * (np.random.rand(n) - 0.5)
        a_pert = np.maximum(a_pert, 0.0)
        s = a_pert.sum()
        if s < 1e-8:
            continue
        a_pert /= s

        a_ref, val_ref = _subgradient_refinement(
            a_pert,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            best_seq,
            best_val,
        )
        if val_ref < best_val:
            best_val = val_ref
            best_seq = a_ref.copy()
            a0 = best_seq.copy()
            print(f"[Perturb] new best {best_val:.6f}")
        else:
            a0 = 0.9 * a0 + 0.1 * base_seq

    return best_seq, best_val


# --------------------------------------------------------------
def _downsample(seq: np.ndarray, target_len: int) -> np.ndarray:
    """Linear interpolation from ``seq`` to length ``target_len``."""
    if seq.shape[0] == target_len:
        return seq.copy()
    x_old = np.linspace(0, 1, seq.shape[0], endpoint=False)
    x_new = np.linspace(0, 1, target_len, endpoint=False)
    y = np.interp(x_new, x_old, seq)
    y = np.maximum(y, 0.0)
    y /= y.sum()
    return y


# --------------------------------------------------------------
def _lp_phase(initial_seq: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Alternating linear‑programming improvement for the small grid.
    Returns the best sequence found and its objective value.
    """
    if not _SCIPY_AVAILABLE:
        # Bail out early – fall back to gradient only
        return initial_seq, _objective(initial_seq, len(initial_seq),
                                       _next_pow2(2 * len(initial_seq) - 1))

    n = initial_seq.shape[0]
    conv_fft_len = _next_pow2(2 * n - 1)
    best_seq = initial_seq.copy()
    best_val = _objective(best_seq, n, conv_fft_len)

    f = best_seq.copy()
    t_mix = 0.5                    # convex‑combination weight
    max_iters = 200               # will stop earlier if time runs out
    it = 0
    last_improve = time.time()

    while time.time() < deadline and it < max_iters:
        # current max‑convolution
        conv_f = _full_convolution(f, conv_fft_len)
        M = conv_f.max()
        cur_val = 2.0 * n * M / (f.sum() ** 2)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = f.copy()
            last_improve = time.time()
            print(f"[LP it={it:03d}] new best {best_val:.6f}")

        # ----- build convolution matrix (dense, size (2n‑1)×n) -----
        rows = 2 * n - 1
        A = np.zeros((rows, n), dtype=np.float64)
        for i in range(n):
            A[i:i + n, i] = f[i]

        # solve  maximise Σ g_j   subject to   A·g ≤ M , g ≥ 0
        c = -np.ones(n)                     # minimise -Σ g_j
        b_ub = np.full(rows, M, dtype=np.float64)
        bounds = [(0, None)] * n
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method="highs")
        if not res.success:
            print("[LP] linprog failed – terminating LP phase")
            break

        g = res.x
        if g.sum() == 0.0:
            break

        # scale g back to same total mass as f (keeps the sum invariant)
        g = g * (f.sum() / g.sum())

        # ----- convex mix and re‑normalise -----
        f_candidate = (1.0 - t_mix) * f + t_mix * g
        f_candidate = np.maximum(f_candidate, 0.0)
        f_candidate /= f_candidate.sum()

        # evaluate candidate
        conv_cand = _full_convolution(f_candidate, conv_fft_len)
        cand_val = 2.0 * n * conv_cand.max() / (f_candidate.sum() ** 2)

        if cand_val < cur_val:
            f = f_candidate
            # keep same mixing weight
        else:
            # no improvement – shrink the step size
            t_mix = max(t_mix * 0.7, 0.05)

        # early stop if stagnant for a while
        if time.time() - last_improve > 12.0:
            print("[LP] no improvement for 12 s – stopping")
            break

        it += 1

    return best_seq, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Entry point – searches for a step‑function with minimal evaluation score.
    Returns the best sequence as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ----------------------------------------------------------------------
    # Small grid for the LP phase (fast linear programs)
    n_lp = 1024
    # ----------------------------------------------------------------------
    # Warm‑start handling
    warm_start = globals().get("height_sequence_1")
    if warm_start is not None:
        warm_start = np.asarray(warm_start, dtype=np.float64)
        warm_start = np.maximum(warm_start, 0.0)
        warm_start = _downsample(warm_start, n_lp)
    else:
        warm_start = None

    # ----------------------------------------------------------------------
    # Build a few random Dirichlet candidates (different concentrations)
    random_candidates = []
    for conc in (0.05, 0.1, 0.3, 0.5, 1.0, 2.0):
        for _ in range(5):
            random_candidates.append(np.random.dirichlet(np.full(n_lp, conc)))
    if warm_start is not None:
        random_candidates.append(warm_start)

    conv_fft_len_lp = _next_pow2(2 * n_lp - 1)
    best_small = None
    best_small_val = float("inf")
    for cand in random_candidates:
        if time.time() >= deadline:
            break
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()
        val = _objective(cand, n_lp, conv_fft_len_lp)
        if val < best_small_val:
            best_small_val = val
            best_small = cand.copy()

    # ----------------------------------------------------------------------
    # LP improvement phase (if time permits)
    if time.time() < deadline - 30.0:          # need at least some time left
        print("\n--- LP phase (n =", n_lp, ") ---")
        best_small, best_small_val = _lp_phase(best_small, deadline)

    # ----------------------------------------------------------------------
    # Upscale to medium grid (n = 4096) – this is the size used previously
    n_mid = 4096
    if best_small is not None:
        best_mid = _downsample(best_small, n_mid)
    else:
        best_mid = np.full(n_mid, 1.0 / n_mid, dtype=np.float64)

    best_mid = np.maximum(best_mid, 0.0)
    best_mid /= best_mid.sum()
    best_val = _objective(best_mid, n_mid, _next_pow2(2 * n_mid - 1))
    best_seq = best_mid.copy()

    # ----------------------------------------------------------------------
    # Gradient‑based deep optimisation on the medium grid
    if time.time() < deadline - 20.0:
        print("\n--- Gradient deep‑optimisation (n =", n_mid, ") ---")
        conv_fft_len_mid = _next_pow2(2 * n_mid - 1)
        corr_fft_len_mid = _next_pow2(3 * n_mid - 2)
        best_seq, best_val = _deep_optimize(
            best_seq,
            n_mid,
            conv_fft_len_mid,
            corr_fft_len_mid,
            deadline,
            best_seq,
            best_val,
        )

    # ----------------------------------------------------------------------
    # Optional upscale to a larger grid (n = 8192) if enough budget remains
    time_left = deadline - time.time()
    if time_left > 30.0:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)
        best_big = _downsample(best_seq, n_big)
        print("\n--- Gradient optimisation on larger grid (n =", n_big, ") ---")
        best_big, best_val = _deep_optimize(
            best_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            best_big,
            best_val,
        )
        if best_big is not None and best_big.shape[0] == n_big:
            best_seq = best_big
            best_val = best_val
            n_mid = n_big

    # ----------------------------------------------------------------------
    # Final cheap stochastic perturbations (if any time left)
    if best_seq is not None and time.time() < deadline - 5.0:
        conv_fft_len_final = _next_pow2(2 * best_seq.shape[0] - 1)
        corr_fft_len_final = _next_pow2(3 * best_seq.shape[0] - 2)
        best_seq, best_val = _local_perturbations(
            best_seq,
            best_seq.shape[0],
            conv_fft_len_final,
            corr_fft_len_final,
            deadline,
            best_seq,
            best_val,
        )

    # ----------------------------------------------------------------------
    # Fallback – uniform vector if everything failed
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Clip to evaluator’s allowed range
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"\n*** Finished search: best objective ≈ {best_val:.6f} (n={best_seq.shape[0]})")
    return best_seq.tolist()
