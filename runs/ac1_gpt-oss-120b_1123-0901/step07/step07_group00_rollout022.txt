"""
Propose a non‑negative step‑function (list of heights) that minimizes

        F(a) = 2·n·max(a∗a) / (sum(a)^2)

The search combines three ideas:

1. **Variance‑flattening phase** – an Adam optimizer that minimizes the
   variance of the full convolution (i.e. makes the convolution as flat
   as possible).  A flatter convolution necessarily lowers its maximum.

2. **Smooth‑max phase** – the classical Adam optimiser that uses a
   soft‑max (log‑sum‑exp) surrogate for the max‑operator.  This fine‑tunes
   the peak after the flattening step.

3. **Exact‑max sub‑gradient phase** – exponentiated‑gradient updates that
   directly minimise the true max‑objective, followed by a cheap random
   mass‑transfer hill‑climber.

Multiple cheap starts (uniform, Dirichlet, edge‑exponential, sparse
spikes) are tried; the best sequence is kept.  If time permits the final
solution is interpolated to a finer grid (8192 points) and refined again.

All phases respect the overall deadline (default 1000 s) and the routine
always returns the best sequence found so far.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a (length 2·len(a)-1) via FFT.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) required for the gradient.
    Returns a length‑len(a) vector:
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i = 0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
def _objective_val(a: np.ndarray, n: int) -> float:
    """Exact objective (no clipping) for a given vector."""
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    return 2.0 * n * conv.max() / (a.sum() ** 2)

def _adam_variance_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 150_000,
    lr: float = 0.05,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser that minimises the variance of the convolution.
    The surrogate gradient uses w = conv - mean(conv).
    Returns the best vector (sum≈1) and its exact objective.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)                # initialise on the simplex
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        # simplex projection via soft‑max
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # full convolution and centered weights
        conv = _full_convolution(a, conv_fft_len)
        avg = conv.mean()
        w = conv - avg                     # may be negative

        # gradient of the variance surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through the soft‑max (mirror‑descent form)
        avg_grad = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg_grad)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # track best exact objective
        val = _objective_val(a, n)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5_000 == 0:
                print(f"[VarAdam it={it:06d}] new best {val:.6f}")

        # mild learning‑rate decay
        if it % 50_000 == 0 and it > 0:
            lr *= 0.9

        it += 1

    return best_a, best_val


def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth‑max (log‑sum‑exp) surrogate.
    Returns the best vector encountered (sum≈1) and its exact objective.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)                # initialise on the simplex
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.07                                 # a bit larger than before
    lam = 0.05                                 # start temperature (soft)
    lam_factor = 1.05

    best_a = a0.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        # simplex projection (soft‑max)
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5_000 == 0:
                print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # temperature schedule
        lam = min(lam * lam_factor, 1e7)

        # occasional tiny random shake when close to the deadline
        if it % 4_000 == 0 and time.time() > deadline - 60:
            logits += (np.random.rand(n) - 0.5) * 2e-2

        # learning‑rate decay
        if it % 50_000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 0.8,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient phase using exponentiated‑gradient updates.
    Returns the best vector seen and its exact objective.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5_000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-14)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1)
        a = a * np.exp(-eta * grad_a)

        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 70_000,
) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climbing: move a random fraction of mass from one index
    to another, keeping the move if the objective improves.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()

    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        delta = a[i] * (np.random.rand() * 0.3)   # up to 30 % of source mass

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5_000 == 0:
                print(f"[Local move {it:06d}] improved to {val2:.6f}")

        it += 1

    return best_a, best_val

def _initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Create a modest but diverse pool of start vectors."""
    cand = []

    if warm is not None:

        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    # random Dirichlet (full support)
    cand.append(np.random.dirichlet(np.ones(n)))

    # edge‑exponential families
    for gamma in (0.2, 0.5, 1.0, 2.0):
        left = np.exp(-gamma * np.arange(n, dtype=np.float64))
        shape = left + left[::-1]
        shape /= shape.sum()
        cand.append(shape)

    # edge‑polynomial (power‑law) families
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.5, 5):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    # double‑Gaussian (different widths)
    i = np.arange(n, dtype=np.float64)
    for sigma in (0.1, 0.3, 0.6):
        shape = np.exp(-0.5 * ((i) / (sigma * n)) ** 2) \
                + np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
        shape /= shape.sum()
        cand.append(shape)

    # positive half‑wave sinusoid
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand.append(shape)

    # uniform
    cand.append(np.full(n, 1.0 / n))

    return cand


def _spike_candidates(n: int) -> list[np.ndarray]:
    """Sparse candidates with equally spaced spikes."""
    out = []
    for k in (32, 64, 128, 256, 512):
        a = np.zeros(n, dtype=np.float64)
        if k > 0:
            pos = np.round(np.linspace(0, n - 1, k)).astype(int)
            a[pos] = 1.0 / k
        out.append(a)
    return out

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the given time budget and always returns the best
    sequence (as a Python list) found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    # problem size
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # optional warm start
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None

    # build a diverse pool of starts
    pool = _initial_candidates(n, warm) + _spike_candidates(n)

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # --------------------------------------------------------------
    # Run a few starts – each gets a slice of the remaining budget
    per_start_budget = (deadline - time.time()) * 0.18
    for cid, cand in enumerate(pool):
        if time.time() >= deadline:
            break
        a0 = np.maximum(cand, 0.0)
        if a0.sum() < 0.01:
            a0 = np.random.dirichlet(np.ones(n))
        a0 = a0 / a0.sum()

        local_deadline = min(deadline - 2.0, time.time() + per_start_budget)

        # 1) variance‑flattening Adam
        a_flat, _ = _adam_variance_optimize(
            a0, n, conv_fft_len, corr_fft_len, local_deadline, max_iters=80_000, lr=0.05
        )

        # 2) smooth‑max Adam
        a_adam, val_adam = _adam_optimize(
            a_flat, n, conv_fft_len, corr_fft_len, local_deadline, max_iters=120_000
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[cand {cid}] Adam best {val_adam:.6f}")

        # 3) sub‑gradient refinement (short)
        if time.time() < deadline:
            a_sub, val_sub = _subgradient_optimize(
                best_seq, n, conv_fft_len, corr_fft_len, deadline, max_iters=30_000, eta0=0.9
            )
            if val_sub < best_val:
                best_val = val_sub
                best_seq = a_sub.copy()
                print(f"[cand {cid}] Subgrad best {val_sub:.6f}")

    # --------------------------------------------------------------
    # Deep refinement of the current best candidate
    if best_seq is not None and time.time() < deadline - 5:
        a_ref, val_ref = _subgradient_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=250_000,
            eta0=0.85,
        )
        if val_ref < best_val:
            best_val = val_ref
            best_seq = a_ref.copy()
            print(f"[Refine] Subgrad long best {val_ref:.6f}")

    # --------------------------------------------------------------
    # Cheap random local search
    if best_seq is not None and time.time() < deadline - 5:
        a_loc, val_loc = _local_random_search(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=80_000,
        )
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()
            print(f"[Local] improved best {val_loc:.6f}")

    # --------------------------------------------------------------
    # Optional up‑sample to a finer grid (8192) if time permits
    if time.time() + 30 < deadline:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        # linear interpolation to the finer grid
        if best_seq is not None:
            x_old = np.linspace(0, 1, n, endpoint=False)
            x_new = np.linspace(0, 1, n_big, endpoint=False)
            a_big = np.interp(x_new, x_old, best_seq)
        else:
            a_big = np.random.dirichlet(np.ones(n_big))

        a_big = np.maximum(a_big, 0.0)
        a_big = a_big / a_big.sum()

        # short Adam on the larger grid
        a_big_adam, _ = _adam_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            max_iters=150_000,
        )

        # longer sub‑gradient refinement on the large grid
        a_big_ref, val_big_ref = _subgradient_optimize(
            a_big_adam,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            max_iters=150_000,
            eta0=0.8,
        )
        if val_big_ref < best_val:
            best_val = val_big_ref
            best_seq = a_big_ref.copy()
            print(f"[Big] refined best {val_big_ref:.6f}")

    # --------------------------------------------------------------
    # Fallback – ensure a valid non‑empty return
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Clip to the evaluator’s accepted range (non‑negative, ≤1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
