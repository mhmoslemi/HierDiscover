    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    best_seq = best_seq; best_val = best_val
    it = 0
    lam = lam_start * (1 if lam_start else small)
    while time.time() < deadline:
        # compute a via softmax
        mx = logits.max()
        exp = np.exp(logits - mx)
        a = exp / exp.sum()
        # conv and objective
        conv = _full_convolution(a, conv_len)
        max_c = conv.max()
        val = 2 * n * max_c / (a.sum()**2)   # a.sum() approx 1
        if val < best_val:
            best_val = val; best_seq = a.copy()
        # compute smooth max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        w = w / w_sum if w_sum > 0 else np.full_like(w, 1.0 / w.size)
        # gradient
        grad_a = 2 * _cross_correlation(w, a, corr_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)
        # Adam update
        m = beta1 * m + (1-beta1)*grad_log
        v = beta2 * v + (1-beta2)*(grad_log**2)
        m_hat = m / (1-beta1**(it+1))
        v_hat = v / (1-beta2**(it+1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        # update lam: lam *= lam_factor; cap
        lam = min(lam * lam_factor, lam_cap)
        # jitter if no improvement for secs
        # omit for simplicity or include
        it += 1
    return best_seq, best_val

def _annealing_optimize(a, n, conv_len, deadline, best_seq, best_val):
    a_cur = a.copy()
    best_seq_local = best_seq
    best_val_local = best_val
    T0 = 0.1  # start temperature for objective difference
    T_end = 1e-6
    total_iters = 20000
    start_time = time.time()
    while time.time() < deadline:
        # progress fraction
        frac = (time.time() - start_time) / (deadline - start_time)
        T = T0 * (T_end / T0) ** frac
        # propose move:
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j:
            continue
        # mass to move
        move = min(a_cur[i], a_cur[j]) * (np.random.rand() * 0.5)  # up to 50% of smaller
        # random direction: move from i to j or opposite
        if np.random.rand() < 0.5:
            delta = move
            src, dst = i, j
        else:
            delta = -move
            src, dst = j, i
        # apply
        a_cur[src] -= delta
        a_cur[dst] += delta
        # ensure non-negative
        if a_cur.min() < -1e-12:
            # revert and continue
            a_cur[src] += delta
            a_cur[dst] -= delta
            continue
        # recompute objective
        conv = _full_convolution(a_cur, conv_len)
        val = 2 * n * conv.max() / (a_cur.sum()**2)
        delta_val = val - best_val_local
        if delta_val < 0 or np.exp(-delta_val / max(T, 1e-12)) > np.random.rand():
            if val < best_val_local:
                best_val_local = val
                best_seq_local = a_cur.copy()
        else:
            # revert move
            a_cur[src] += delta
            a_cur[dst] -= delta
    return best_seq_local, best_val_local

"""
Propose a non‑negative step function with low evaluation score.

The algorithm mixes a rich parametric seeding stage (including symmetric /
asymmetric beta, exponential, power‑law, Gaussian, arcsine and random
mixtures), a two‑phase optimisation (Adam on a smooth‑max surrogate then a
sub‑gradient/exponentiated‑gradient refinement) and, if time permits,
upscales the solution to a finer grid followed by a short simulated‑annealing
hill‑climb.  All stages respect the 1000 s deadline and keep the vector on
the probability simplex.
"""
import time
import math
import numpy as np

def _next_pow2(x: int) -> int:
    """Smallest power of two greater than or equal to x."""
    return 1 << (x - 1).bit_length()

def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT length ``fft_len``."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]

def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]

def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Fast evaluation of the objective (sum(a)≈1)."""
    conv = _full_convolution(a, conv_fft_len)
    return float(2.0 * n * conv.max() / (a.sum() ** 2))

def _generate_parametric_candidates(n: int, warm_start: np.ndarray | None) -> list[np.ndarray]:
    """Create a diverse pool of candidate shapes (all normalized to sum = 1)."""
    i = np.arange(n, dtype=np.float64)
    candidates = []

    # warm‑start and cyclic shifts
    if warm_start is not None:
        ws = warm_start.copy()
        ws = np.maximum(ws, 0.0)
        if ws.sum() < 0.01:
            ws = None
        else:
            ws = ws / ws.sum()
            candidates.append(ws)
            for shift in (n // 4, n // 2, 3 * n // 4):
                cand = np.roll(ws, shift)
                candidates.append(cand)

    # uniform baseline
    candidates.append(np.full(n, 1.0 / n))

    # random Dirichlet draws (various concentrations)
    for conc in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0):
        cand = np.random.dirichlet(np.full(n, conc))
        candidates.append(cand)

    # edge‑exponential families (with optional uniform mixing)
    for alpha in np.logspace(-2, 1.0, num=14):      # 0.01 … 10
        edge = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
        edge /= edge.sum()
        for w in (0.0, 0.25, 0.5, 0.75, 1.0):
            cand = w * edge + (1.0 - w) / n
            candidates.append(cand)

    # edge‑power‑law families
    for p in np.linspace(0.2, 5.0, num=14):
        edge = (i + 1.0) ** (-p) + (n - i) ** (-p)
        edge /= edge.sum()
        for w in (0.0, 0.4, 0.8, 1.0):
            cand = w * edge + (1.0 - w) / n
            candidates.append(cand)

    # symmetric beta‑shapes (a_i ∝ (i+1)^{α‑1}(n‑i)^{α‑1})
    for alpha in (0.2, 0.4, 0.5, 0.8, 1.2, 2.0, 3.0):
        shape = (i + 1.0) ** (alpha - 1.0) * (n - i) ** (alpha - 1.0)
        shape = np.maximum(shape, 0.0)
        shape /= shape.sum()
        candidates.append(shape)

    # asymmetric beta‑shapes (α,β independent)
    alphas = (0.2, 0.5, 1.0, 2.0, 3.0)
    betas  = (0.2, 0.5, 1.0, 2.0, 3.0)
    for a in alphas:
        for b in betas:
            shape = (i + 1.0) ** (a - 1.0) * (n - i) ** (b - 1.0)
            shape = np.maximum(shape, 0.0)
            shape /= shape.sum()
            candidates.append(shape)

    # raised‑cosine / sine‑power families
    for p in (1, 2, 4, 6):
        shape = np.sin(np.pi * (i + 0.5) / n) ** p
        shape = np.maximum(shape, 0.0)
        shape /= shape.sum()
        candidates.append(shape)

    # two‑Gaussian edge shapes
    for gamma in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        shape /= shape.sum()
        candidates.append(shape)

    # arcsine‑type shape   a_i ∝ 1 / sqrt((i+0.5)*(n-i-0.5))
    shape = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    candidates.append(shape)

    # random convex mixtures of a few base shapes
    base = list(candidates)  # snapshot of the shapes built so far
    for _ in range(30):
        k = np.random.randint(2, 5)          # mixture of 2‑4 components
        idx = np.random.choice(len(base), size=k, replace=False)
        weights = np.random.dirichlet(np.ones(k))
        mix = np.zeros_like(base[0])
        for w, ix in zip(weights, idx):
            mix += w * base[ix]
        mix /= mix.sum()
        candidates.append(mix)

    # final sanity – ensure every candidate is non‑negative & normalised
    cleaned = []
    for cand in candidates:
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 0.01:
            cand = np.random.dirichlet(np.ones(n))
            s = cand.sum()
        cand = cand / s
        cleaned.append(cand)
    return cleaned

def _adam_optimize(a0: np.ndarray,
                   n: int,
                   conv_fft_len: int,
                   corr_fft_len: int,
                   deadline: float,
                   global_best_seq: np.ndarray | None,
                   global_best_val: float) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on a log‑sum‑exp surrogate of the max‑convolution.
    Returns the best vector/value seen.
    """
    eps = 1e-15
    # initialise in soft‑max space (logits)
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.02                    # moderate learning‑rate
    lam = 0.01                   # start temperature (very smooth)
    lam_factor = 1.005           # slow increase
    lam_cap = 1e8

    best_seq = global_best_seq
    best_val = global_best_val
    last_improve = time.time()
    it = 0

    while time.time() < deadline:
        # --- simplex via soft‑max ---
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # --- evaluate surrogate objective ---
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[{n}] Adam it={it:06d} new best {val:.6f}")

        # --- smooth‑max weights (log‑sum‑exp) ---
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # --- gradient w.r.t a (using cross‑correlation) ---
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        # chain rule for logits
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # --- Adam update ---
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # --- temperature schedule ---
        lam = min(lam * lam_factor, lam_cap)

        # --- jitter if stuck for a while ---
        if time.time() - last_improve > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-3
            last_improve = time.time()

        it += 1

    return best_seq, best_val

def _subgradient_optimize(a0: np.ndarray,
                          n: int,
                          conv_fft_len: int,
                          corr_fft_len: int,
                          deadline: float,
                          init_best_seq: np.ndarray | None,
                          init_best_val: float) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient steps on the true max‑convolution.
    """
    a = a0.copy()
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum < 0.01:
        a = np.random.dirichlet(np.ones(n))
        a_sum = a.sum()
    a = a / a_sum

    best_seq = a.copy()
    best_val = init_best_val
    eta0 = 0.6
    it = 0
    last_improve = time.time()

    while time.time() < deadline:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[{n}] Subgrad it={it:05d} new best {val:.6f}")

        # active set where convolution attains its maximum
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # decaying step‑size
        eta = eta0 / math.sqrt(it + 1)

        # exponentiated‑gradient update (keeps a ≥ 0)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:         # avoid collapse
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        # early stop if nothing improves for a while
        if time.time() - last_improve > 7.0:
            break

        it += 1

    return best_seq, best_val

def _deep_optimize(a0: np.ndarray,
                   n: int,
                   conv_fft_len: int,
                   corr_fft_len: int,
                   deadline: float,
                   global_best_seq: np.ndarray | None,
                   global_best_val: float) -> tuple[np.ndarray, float]:
    """
    Run Adam (smooth‑max) followed by sub‑gradient refinement.
    Updates the global best.
    """
    # Phase‑1: Adam on smooth‑max
    best_seq, best_val = _adam_optimize(
        a0,
        n,
        conv_fft_len,
        corr_fft_len,
        deadline,
        global_best_seq,
        global_best_val,
    )
    # Phase‑2: exact sub‑gradient refinement (starting from best found)
    best_seq, best_val = _subgradient_optimize(
        best_seq,
        n,
        conv_fft_len,
        corr_fft_len,
        deadline,
        best_seq,
        best_val,
    )
    return best_seq, best_val

def _annealing_optimize(a0: np.ndarray,
                        n: int,
                        conv_fft_len: int,
                        deadline: float,
                        init_best_seq: np.ndarray | None,
                        init_best_val: float) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing hill‑climb that moves tiny amounts of mass
    between two random coordinates.  Accepts uphill moves with a temperature
    schedule that cools down exponentially.
    """
    a = a0.copy()
    best_seq = init_best_seq if init_best_seq is not None else a.copy()
    best_val = init_best_val
    start_time = time.time()
    total_time = deadline - start_time
    if total_time <= 0:
        return best_seq, best_val

    # temperature parameters (objective is around 1‑2, so these are tuned)
    T_start = 0.02
    T_end = 1e-6

    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / total_time
        T = T_start * (T_end / T_start) ** frac

        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # pick direction (i→j or j→i)
        if np.random.rand() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i

        # amount of mass to transfer – up to 5 % of the smaller bucket
        max_transfer = min(a[src], a[dst])
        if max_transfer < 1e-12:
            continue
        delta = (np.random.rand() * 0.05) * max_transfer  # up to 5%
        # apply move
        a[src] -= delta
        a[dst] += delta

        # evaluate
        conv = _full_convolution(a, conv_fft_len)
        val = 2.0 * n * conv.max() / (a.sum() ** 2)

        accept = False
        if val < best_val:
            accept = True
        else:
            # Metropolis criterion
            if np.exp(-(val - best_val) / max(T, 1e-12)) > np.random.rand():
                accept = True

        if accept:
            if val < best_val:
                best_val = val
                best_seq = a.copy()
        else:
            # revert
            a[src] += delta
            a[dst] -= delta

    return best_seq, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Entry point: search for a non‑negative step function that minimises
    ``evaluate_sequence``. Returns the best sequence (plain Python list).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Determine initial grid size
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm‑start from a previously discovered sequence, if available
    warm_start = globals().get("height_sequence_1")
    if warm_start is not None:
        ws = np.asarray(warm_start, dtype=np.float64)
        if ws.shape[0] > n:
            ws = ws[:n]
        elif ws.shape[0] < n:
            ws = np.pad(ws, (0, n - ws.shape[0]))
        ws = np.clip(ws, 0.0, None)
        if ws.sum() >= 0.01:
            ws = ws / ws.sum()
        else:
            ws = None
    else:
        ws = None

    # ------------------------------------------------------------------
    # Stage 1 – generate a rich pool of seeds and keep the most promising ones
    candidate_pool = _generate_parametric_candidates(n, ws)

    evaluated = []
    for idx, cand in enumerate(candidate_pool):
        if time.time() >= deadline:
            break
        val = _objective(cand, n, conv_fft_len)
        evaluated.append((val, cand))
        # optional progress line (commented to reduce output clutter)
        # print(f"[candidate {idx}] objective≈{val:.6f}")

    evaluated.sort(key=lambda x: x[0])
    top_k = min(12, len(evaluated))
    top_candidates = [c for _, c in evaluated[:top_k]]
    print(f"Stage 1: selected {top_k} promising starts (n={n})")

    # ------------------------------------------------------------------
    # Stage 2 – deep optimisation from each start
    for start_id, a0 in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        print(f"--- Deep optimisation start #{start_id + 1}/{len(top_candidates)} ---")
        best_seq, best_val = _deep_optimize(
            a0,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            best_seq,
            best_val,
        )

    # ------------------------------------------------------------------
    # Stage 3 – if we still have time, go to a finer grid and optimise again
    time_left = deadline - time.time()
    if time_left > 30.0:
        n_big = 8192
        conv_big = _next_pow2(2 * n_big - 1)
        corr_big = _next_pow2(3 * n_big - 2)

        if best_seq is not None:
            # Linear interpolation onto the finer grid
            x_old = np.linspace(0, 1, n, endpoint=False)
            x_new = np.linspace(0, 1, n_big, endpoint=False)
            a_big = np.interp(x_new, x_old, best_seq)
        else:
            a_big = np.random.dirichlet(np.ones(n_big))

        a_big = np.maximum(a_big, 0.0)

"""
Propose a non‑negative step function with a low evaluation score.

The pipeline consists of:

1. **Rich parametric seeding** – a large pool of analytic shapes
   (uniform, Dirichlet, edge‑exponential, power‑law, symmetric &
   asymmetric beta, raised‑cosine, Gaussian, arcsine) and random
   convex mixtures thereof is generated for a moderate grid
   (n = 4096).  The 10–15 best seeds (according to the exact
   objective) are retained.

2. **Two‑phase continuous optimisation** – each retained seed is first
   refined by Adam on a log‑sum‑exp (smooth‑max) surrogate of the
   max‑convolution, then by an exponentiated‑gradient sub‑gradient
   method that works directly on the true max.  Both stages respect the
   global deadline.

3. **Fine‑grained up‑scaling & polishing** – if time permits the current
   best vector is interpolated to a finer grid (n = 8192) and the two‑phase
   optimisation is repeated, followed by a short simulated‑annealing
   hill‑climb that accepts occasional uphill moves.

All vectors are kept on the probability simplex (sum = 1) and clipped
to the evaluator’s allowed range (0 … 1000).  No external solvers are
used; the implementation relies only on NumPy and FFTs and obeys the
1000 s time limit.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using a real FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact objective (sum(a) is ≈ 1)."""
    conv = _full_convolution(a, conv_fft_len)
    return float(2.0 * n * conv.max() / (a.sum() ** 2))

def _generate_parametric_candidates(
    n: int, warm_start: np.ndarray | None
) -> list[np.ndarray]:
    """Create a diverse pool of candidate shapes (all normalised to sum = 1)."""
    i = np.arange(n, dtype=np.float64)
    candidates = []

    # ------------------------------------------------------------------
    # Warm‑start and its cyclic shifts
    if warm_start is not None:
        ws = warm_start.copy()
        ws = np.maximum(ws, 0.0)
        if ws.sum() >= 0.01:
            ws = ws / ws.sum()
            candidates.append(ws)
            for shift in (n // 4, n // 2, 3 * n // 4):
                candidates.append(np.roll(ws, shift))

    # ------------------------------------------------------------------
    # Uniform baseline
    candidates.append(np.full(n, 1.0 / n))

    # ------------------------------------------------------------------
    # Random Dirichlet draws (various concentrations)
    for conc in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0):
        candidates.append(np.random.dirichlet(np.full(n, conc)))

    # ------------------------------------------------------------------
    # Edge‑exponential families (with optional uniform mixing)
    for alpha in np.logspace(-2, 1.0, num=14):  # 0.01 … 10
        edge = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
        edge /= edge.sum()
        for w in (0.0, 0.25, 0.5, 0.75, 1.0):
            candidates.append(w * edge + (1.0 - w) / n)

    # ------------------------------------------------------------------
    # Edge‑power‑law families
    for p in np.linspace(0.2, 5.0, num=14):
        edge = (i + 1.0) ** (-p) + (n - i) ** (-p)
        edge /= edge.sum()
        for w in (0.0, 0.4, 0.8, 1.0):
            candidates.append(w * edge + (1.0 - w) / n)

    # ------------------------------------------------------------------
    # Symmetric beta‑shapes
    for alpha in (0.2, 0.4, 0.5, 0.8, 1.2, 2.0, 3.0):
        shape = (i + 1.0) ** (alpha - 1.0) * (n - i) ** (alpha - 1.0)
        shape = np.maximum(shape, 0.0)
        shape /= shape.sum()
        candidates.append(shape)

    # ------------------------------------------------------------------
    # Asymmetric beta‑shapes (coarse grid)
    alphas = (0.2, 0.5, 1.0, 2.0, 3.0)
    betas = (0.2, 0.5, 1.0, 2.0, 3.0)
    for a in alphas:
        for b in betas:
            shape = (i + 1.0) ** (a - 1.0) * (n - i) ** (b - 1.0)
            shape = np.maximum(shape, 0.0)
            shape /= shape.sum()
            candidates.append(shape)

    # ------------------------------------------------------------------
    # Dense grid of asymmetric beta‑shapes (finer search)
    fine_alphas = np.linspace(0.1, 2.5, 25)
    fine_betas = np.linspace(0.1, 2.5, 25)
    for a in fine_alphas:
        for b in fine_betas:
            shape = (i + 1.0) ** (a - 1.0) * (n - i) ** (b - 1.0)
            shape = np.maximum(shape, 0.0)
            shape /= shape.sum()
            candidates.append(shape)

    # ------------------------------------------------------------------
    # Raised‑cosine / sine‑power families
    for p in (1, 2, 4, 6):
        shape = np.sin(np.pi * (i + 0.5) / n) ** p
        shape = np.maximum(shape, 0.0)
        shape /= shape.sum()
        candidates.append(shape)

    # ------------------------------------------------------------------
    # Two‑Gaussian edge shapes
    for gamma in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        shape /= shape.sum()
        candidates.append(shape)

    # ------------------------------------------------------------------
    # Arcsine‑type shape
    shape = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    candidates.append(shape)

    # ------------------------------------------------------------------
    # Random convex mixtures of a few base shapes
    base = list(candidates)  # snapshot
    for _ in range(30):
        k = np.random.randint(2, 5)  # 2‑4 components
        idx = np.random.choice(len(base), size=k, replace=False)
        weights = np.random.dirichlet(np.ones(k))
        mix = np.zeros_like(base[0])
        for w, ix in zip(weights, idx):
            mix += w * base[ix]
        mix /= mix.sum()
        candidates.append(mix)

    # ------------------------------------------------------------------
    # Normalise and prune tiny vectors
    cleaned = []
    for cand in candidates:
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 0.01:
            cand = np.random.dirichlet(np.ones(n))
            s = cand.sum()
        cleaned.append(cand / s)
    return cleaned

def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on a smooth‑max surrogate of the max‑convolution.
    Returns the best vector/value seen during the run.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)               # initialise in log‑soft‑max space
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.02                               # modest learning‑rate
    lam = 0.01                               # start temperature (very smooth)
    lam_factor = 1.005                       # very slow increase
    lam_cap = 1e8

    best_seq = global_best_seq
    best_val = global_best_val
    last_improve = time.time()
    it = 0

    while time.time() < deadline:
        # ----- probability simplex via soft‑max -----
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # ----- compute surrogate objective -----
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[{n}] Adam it={it:06d} new best {val:.6f}")

        # ----- smooth‑max weights (log‑sum‑exp) -----
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient w.r.t. a -----
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule for logits
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- temperature schedule -----
        lam = min(lam * lam_factor, lam_cap)

        # ----- occasional jitter if stuck -----
        if time.time() - last_improve > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-3
            last_improve = time.time()

        it += 1

    return best_seq, best_val


def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    init_best_seq: np.ndarray | None,
    init_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient refinement on the true max‑convolution.
    """
    a = a0.copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s

    best_seq = a.copy()
    best_val = init_best_val
    eta0 = 0.6
    it = 0
    last_improve = time.time()

    while time.time() < deadline:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[{n}] Subgrad it={it:05d} new best {val:.6f}")

        # active set where the convolution attains its maximum
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # decaying step size
        eta = eta0 / math.sqrt(it + 1)

        # exponentiated‑gradient update (keeps a ≥ 0)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:                     # avoid collapse to zero
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        # early exit if stagnation
        if time.time() - last_improve > 7.0:
            break

        it += 1

    return best_seq, best_val


def _deep_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Run Adam (smooth‑max) followed by a sub‑gradient refinement.
    Updates the globally best vector/value.
    """
    # Phase‑1 – Adam on the smooth surrogate
    best_seq, best_val = _adam_optimize(
        a0,
        n,
        conv_fft_len,
        corr_fft_len,
        deadline,
        global_best_seq,
        global_best_val,
    )
    # Phase‑2 – exact sub‑gradient refinement (starting from the Adam result)
    best_seq, best_val = _subgradient_optimize(
        best_seq,
        n,
        conv_fft_len,
        corr_fft_len,
        deadline,
        best_seq,
        best_val,
    )
    return best_seq, best_val


def _annealing_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    init_best_seq: np.ndarray | None,
    init_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Very short simulated‑annealing hill‑climb that moves small amounts of
    mass between two random coordinates.  Uphill moves are accepted with
    probability exp(−Δ/temperature), where the temperature cools down
    exponentially.
    """
    a = a0.copy()
    best_seq = init_best_seq if init_best_seq is not None else a.copy()
    best_val = init_best_val
    start_time = time.time()
    total_time = deadline - start_time
    if total_time <= 0:
        return best_seq, best_val

    T_start = 0.02
    T_end = 1e-6

    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / total_time
        T = T_start * (T_end / T_start) ** frac

        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # direction of transfer
        if np.random.rand() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i

        # amount to transfer – up to 5 % of the smaller bucket
        max_transfer = min(a[src], a[dst])
        if max_transfer < 1e-12:
            continue
        delta = (np.random.rand() * 0.05) * max_transfer

        a[src] -= delta
        a[dst] += delta

        conv = _full_convolution(a, conv_fft_len)
        val = 2.0 * n * conv.max() / (a.sum() ** 2)

        accept = False
        if val < best_val:
            accept = True
        else:
            if np.exp(-(val - best_val) / max(T, 1e-12)) > np.random.rand():
                accept = True

        if accept:
            if val < best_val:
                best_val = val
                best_seq = a.copy()
        else:
            # revert
            a[src] += delta
            a[dst] -= delta

    return best_seq, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    Returns the best sequence (plain Python list).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Base grid size
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm‑start (optional)
    warm_start = globals().get("height_sequence_1")
    if warm_start is not None:
        ws = np.asarray(warm_start, dtype=np.float64)
        if ws.shape[0] > n:
            ws = ws[:n]
        elif ws.shape[0] < n:
            ws = np.pad(ws, (0, n - ws.shape[0]))
        ws = np.clip(ws, 0.0, None)
        if ws.sum() >= 0.01:
            ws = ws / ws.sum()
        else:
            ws = None
    else:
        ws = None

    # ------------------------------------------------------------------
    # Stage 1 – generate a large pool of seeds and keep the most promising ones
    candidate_pool = _generate_parametric_candidates(n, ws)

    evaluated = []
    for idx, cand in enumerate(candidate_pool):
        if time.time() >= deadline:
            break
        val = _objective(cand, n, conv_fft_len)
        evaluated.append((val, cand))
        # uncomment for debugging:
        # print(f"[candidate {idx}] obj≈{val:.6f}")

    evaluated.sort(key=lambda x: x[0])
    top_k = min(10, len(evaluated))      # keep a modest number of seeds
    top_candidates = [c for _, c in evaluated[:top_k]]
    print(f"Stage 1: selected {top_k} promising seeds (n={n})")

    # ------------------------------------------------------------------
    # Stage 2 – deep optimisation from each seed
    for sid, a0 in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        print(f"--- Deep optimisation start #{sid + 1}/{len(top_candidates)} ---")
        best_seq, best_val = _deep_optimize(
            a0,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            best_seq,
            best_val,
        )

    # ------------------------------------------------------------------
    # Stage 3 – upscale to a finer grid if time permits
    time_left = deadline - time.time()
    if time_left > 30.0:
        n_big = 8192
        conv_big = _next_pow2(2 * n_big - 1)
        corr_big = _next_pow2(3 * n_big - 2)

        if best_seq is not None:
            # Linear interpolation onto the finer grid
            x_old = np.linspace(0, 1, n, endpoint=False)
            x_new = np.linspace(0, 1, n_big, endpoint=False)
            a_big = np.interp(x_new, x_old, best_seq)
        else:
            a_big = np.random.dirichlet(np.ones(n_big))

        a_big = np.maximum(a_big, 0.0)
        a_big = a_big / a_big.sum()

        print("--- Optimising on larger grid (n=8192) ---")
        best_seq, best_val = _deep_optimize(
            a_big,
            n_big,
            conv_big,
            corr_big,
            deadline,
            best_seq,
            best_val,
        )

        # short simulated‑annealing polishing
        if time.time() < deadline - 5.0:
            best_seq, best_val = _annealing_optimize(
                best_seq,
                n_big,
                conv_big,
                deadline,
                best_seq,
                best_val,
            )
        n = n_big
        conv_fft_len = conv_big
        corr_fft_len = corr_big

    # ------------------------------------------------------------------
    # Stage 4 – fallback to uniform if nothing succeeded
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Clip to the evaluator's allowed range
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"*** Search complete: best objective ≈ {best_val:.6f} (n={best_seq.shape[0]})")
    return best_seq.tolist()
