"""
Improved minimiser for the “max‑convolution” objective.

Approach
--------
1. Coarse search on a medium‑size grid (n≈2048) – cheap parametric families
   (uniform, Dirichlet, exponentials, power‑laws, triangular, sinusoid,
   sparse spikes).  The best few candidates are kept.

2. For each kept start we run a two‑phase optimiser:
   – Adam on a log‑sum‑exp surrogate (smooth‑max) with a slowly rising
     temperature.
   – Extended exponentiated‑gradient sub‑gradient descent on the *exact*
     max‑convolution, using a relaxed active set (all slices within 1 % of
     the peak).

   The optimiser works directly on the logits of a soft‑max representation,
   guaranteeing non‑negativity and the simplex constraint.

3. If time permits the best coarse solution is interpolated to a finer
   grid (n=4096) and refined again with the same optimiser.

4. Finally a cheap stochastic hill‑climber (Gaussian perturbations of the
   logits) and a simple simulated‑annealing loop are run to squeeze out
   the last few thousandths.

All heavy work is done with FFT‑based convolutions for speed.
The routine respects the 1000 s budget and returns a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (full length 2*len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    prod = fa * fa
    conv = np.fft.irfft(prod, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact objective: 2·n·max(conv)/(sum(a)²).  Here sum(a)=1 so denominator=1."""
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c)

def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    init_best_seq: np.ndarray | None,
    init_best_val: float,
    max_iter: int = 250_000,
) -> tuple[np.ndarray, float]:
    """Adam on a log‑sum‑exp surrogate of the max‑convolution."""
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.07                # a bit larger than before

    lam = 0.5                # start cooler, ramp up later
    lam_factor = 1.025
    lam_cap = 5e5

    best_seq = init_best_seq
    best_val = init_best_val
    last_improve = time.time()

    it = 0
    while time.time() < deadline and it < max_iter:
        # soft‑max to stay on simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # surrogate convolution
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c                    # sum(a)=1 by construction

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Adam] iter={it:06d} new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_log
        v = beta2 * v + (1 - beta2) * (grad_log ** 2)
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature schedule
        lam = min(lam * lam_factor, lam_cap)

        # occasional shake if stuck
        if time.time() - last_improve > 6.0:
            logits += (np.random.rand(n) - 0.5) * 3e-2
            last_improve = time.time()

        it += 1

    return best_seq, best_val


def _subgrad_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    init_best_seq: np.ndarray | None,
    init_best_val: float,
    max_iter: int = 120_000,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient descent on the *exact* max‑convolution.
    Uses a relaxed active set (all slices within 1 % of the peak).
    """
    a = init_best_seq if init_best_seq is not None else a0.copy()
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a /= a.sum()                     # simplex

    eta0 = 0.9
    decay = 0.99994
    eta = eta0

    best_seq = a.copy()
    best_val = init_best_val

    it = 0
    last_improve = time.time()

    while time.time() < deadline and it < max_iter:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Subgrad] iter={it:06d} new best {val:.6f}")

        # relaxed active set: allow 1 % slack
        slack = max_c * 1.01
        active = conv >= slack
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # exponentiated‑gradient update
        a = a * np.exp(-eta * grad_a)

        # renormalise, protect against collapse
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        eta *= decay
        it += 1

        # early stop if stalled
        if time.time() - last_improve > 8.0:
            break

    return best_seq, best_val


def _deep_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Run Adam‑smooth‑max followed by a long sub‑gradient phase,
    updating the global best.
    """
    best_seq, best_val = _adam_optimize(
        a0,
        n,
        conv_fft_len,
        corr_fft_len,
        deadline,
        global_best_seq,
        global_best_val,
        max_iter=200_000,
    )
    best_seq, best_val = _subgrad_optimize(
        a0,
        n,
        conv_fft_len,
        corr_fft_len,
        deadline,
        best_seq,
        best_val,
        max_iter=100_000,
    )
    return best_seq, best_val

def _generate_parametric_candidates(n: int, warm_start: np.ndarray | None):
    """
    Produce a diverse pool of candidate vectors.
    Families: uniform, Dirichlet, symmetric/exponential, asymmetric
    exponentials / power‑laws, linear / triangular, sinusoidal half‑wave,
    sparse spikes, random permutations.
    All are normalised to sum≈1.
    """
    candidates = []

    # warm‑start & its cyclic shifts
    if warm_start is not None:
        candidates.append(warm_start.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            candidates.append(np.roll(warm_start, shift))

    # uniform baseline
    candidates.append(np.full(n, 1.0 / n))

    # random Dirichlet draws with a wide range of concentrations
    for conc in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(8):
            cand = np.random.dirichlet(np.full(n, conc))
            candidates.append(cand)

    i = np.arange(n, dtype=np.float64)

    # symmetric edge‑exponential / power‑law families (as in the reference)
    for alpha in np.logspace(-2, 1.0, num=10):
        edge = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
        edge /= edge.sum()
        for w in (0.0, 0.3, 0.6, 1.0):
            shape = w * edge + (1.0 - w) / n
            candidates.append(shape)

    for p in np.linspace(0.3, 5.5, num=10):
        edge = (i + 1.0) ** (-p) + (n - i) ** (-p)
        edge /= edge.sum()
        for w in (0.0, 0.4, 0.8, 1.0):
            shape = w * edge + (1.0 - w) / n
            candidates.append(shape)

    # asymmetric two‑sided exponentials
    alphas = np.logspace(-2, 1.0, num=8)
    for wL in np.linspace(0.0, 0.6, 4):
        for wR in np.linspace(0.0, 0.6, 4):
            if wL + wR > 1.0:
                continue
            wU = 1.0 - wL - wR
            for aL in alphas:
                for aR in alphas:
                    left = np.exp(-aL * i)
                    right = np.exp(-aR * (n - 1 - i))
                    shape = wL * left + wR * right + wU / n
                    shape /= shape.sum()
                    candidates.append(shape)

    # asymmetric power‑law families
    p_vals = np.linspace(0.2, 5.0, num=8)
    for wL in np.linspace(0.0, 0.6, 4):
        for wR in np.linspace(0.0, 0.6, 4):
            if wL + wR > 1.0:
                continue
            wU = 1.0 - wL - wR
            for pL in p_vals:
                for pR in p_vals:
                    left = (i + 1.0) ** (-pL)
                    right = (n - i) ** (-pR)
                    shape = wL * left + wR * right + wU / n
                    shape /= shape.sum()
                    candidates.append(shape)

    # linear / triangular shapes
    lin = np.linspace(1.0, 0.0, n)
    tri = np.concatenate([np.linspace(0, 1, n // 2, endpoint=False),
                          np.linspace(1, 0, n - n // 2)])
    for base in (lin, tri):
        base /= base.sum()
        for w in (0.0, 0.3, 0.7):
            cand = w * base + (1.0 - w) / n
            cand /= cand.sum()
            candidates.append(cand)

    # sinusoidal half‑wave
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    candidates.append(shape)

    # sparse spike vectors – a few non‑zeros
    for spikes in (4, 6, 9, 12):
        for _ in range(15):
            idx = np.random.choice(n, size=spikes, replace=False)
            weights = np.random.dirichlet(np.ones(spikes))
            vec = np.zeros(n)
            vec[idx] = weights
            vec /= vec.sum()
            candidates.append(vec)

    # random permutations of a uniform vector (adds asymmetry)
    base = np.full(n, 1.0 / n)
    for _ in range(12):
        perm = np.random.permutation(n)
        candidates.append(base[perm])

    return candidates


# ----------------------------------------------------------------------
def _local_random_search(
    a_best: np.ndarray,
    n: int,
    conv_fft_len: int,
    best_val: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Cheap hill‑climbing: Gaussian noise on logits, accept if objective improves.
    """
    logits_best = np.log(a_best + 1e-15)

    while time.time() < deadline:
        noise = np.random.randn(n) * 0.04
        logits = logits_best + noise
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()
        val = _objective(a, n, conv_fft_len)

        if val < best_val:
            best_val = val
            a_best = a.copy()
            logits_best = np.log(a_best + 1e-15)
            print(f"[Local] improved to {val:.6f}")

    return a_best, best_val


def _simulated_annealing(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    start_temp: float = 0.05,
    cooling: float = 0.9995,
) -> tuple[np.ndarray, float]:
    """
    Very simple annealing on the simplex (log‑softmax parametrisation).
    """
    a_cur = a0.copy()
    val_cur = _objective(a_cur, n, conv_fft_len)
    best_seq = a_cur.copy()
    best_val = val_cur

    logits_cur = np.log(a_cur + 1e-15)
    temp = start_temp
    it = 0
    while time.time() < deadline:
        noise = np.random.randn(n) * (0.06 * temp)
        logits = logits_cur + noise
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a_new = exp_log / exp_log.sum()
        val_new = _objective(a_new, n, conv_fft_len)

        if val_new < val_cur or np.random.rand() < math.exp(-(val_new - val_cur) / max(temp, 1e-12)):
            a_cur = a_new
            val_cur = val_new
            logits_cur = logits
            if val_new < best_val:
                best_val = val_new
                best_seq = a_new.copy()
                print(f"[SA] iter={it:06d} new best {val_new:.6f}")

        temp *= cooling
        it += 1
        if temp < 1e-6:
            temp = 1e-6

    return best_seq, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Main entry point – searches for a non‑negative step function that
    minimises the supplied evaluation function.  Returns the best sequence
    (plain Python list) found within the time budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Coarse grid (n≈2048) – cheap enough to explore many restarts
    n = 2048
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # optional warm start from a previous run
    warm_start = globals().get("height_sequence_1")
    if warm_start is not None:
        warm_start = np.asarray(warm_start, dtype=np.float64)
        # down‑sample to n via simple averaging of neighbouring pairs
        if warm_start.shape[0] > n:
            factor = warm_start.shape[0] // n
            warm_start = warm_start[: factor * n].reshape(n, factor).mean(axis=1)
        elif warm_start.shape[0] < n:
            warm_start = np.pad(warm_start, (0, n - warm_start.shape[0]))
        warm_start = np.maximum(warm_start, 0.0)
        s = warm_start.sum()
        if s >= 0.01:
            warm_start = warm_start / s
        else:
            warm_start = None
    else:
        warm_start = None

    # ------------------------------------------------------------------
    # Stage 1 – generate a broad set of cheap candidates
    all_cands = _generate_parametric_candidates(n, warm_start)

    cand_vals = []
    for idx, cand in enumerate(all_cands):
        if time.time() >= deadline:
            break
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 0.01:
            cand = np.random.dirichlet(np.ones(n))
            s = cand.sum()
        cand = cand / s
        val = _objective(cand, n, conv_fft_len)
        cand_vals.append((val, cand))

    # keep a moderate‑size elite set
    cand_vals.sort(key=lambda x: x[0])
    top_k = 28
    elite = [c for _, c in cand_vals[:top_k]]
    print(f"[Stage 1] evaluated {len(cand_vals)} candidates, best={cand_vals[0][0]:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – deep optimisation of each elite start
    for start_id, a0 in enumerate(elite):
        if time.time() >= deadline:
            break
        print(f"--- Deep opt from elite #{start_id + 1}/{len(elite)} ---")
        best_seq, best_val = _deep_optimize(
            a0,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            best_seq,
            best_val,
        )

    # ------------------------------------------------------------------
    # Stage 3 – local stochastic refinements on the current global best
    if best_seq is not None:
        print("--- Local search (Gaussian) ---")
        best_seq, best_val = _local_random_search(
            best_seq, n, conv_fft_len, best_val, deadline
        )
        print("--- Simulated annealing ---")
        best_seq, best_val = _simulated_annealing(
            best_seq, n, conv_fft_len, deadline
        )

    # ------------------------------------------------------------------
    # Stage 4 – if we still have decent time, upscale to n=4096 and polish
    time_left = deadline - time.time()
    if time_left > 120.0:                     # need a non‑trivial budget
        n_big = 4096
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        # upscale by linear interpolation
        if best_seq is not None:
            x_old = np.linspace(0, 1, n, endpoint=False)
            x_new = np.linspace(0, 1, n_big, endpoint=False)
            a_big = np.interp(x_new, x_old, best_seq)
            a_big = np.maximum(a_big, 0.0)
            a_big /= a_big.sum()
        else:
            a_big = np.random.dirichlet(np.ones(n_big))

        print("--- Upscaled optimisation (n=4096) ---")
        best_seq, best_val = _deep_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            best_seq,
            best_val,
        )
        # final cheap refinements on the large grid
        if best_seq is not None:
            best_seq, best_val = _local_random_search(
                best_seq, n_big, conv_fft_len_big, best_val, deadline
            )
            best_seq, best_val = _simulated_annealing(
                best_seq, n_big, conv_fft_len_big, deadline
            )
        n = n_big   # remember final size

    # ------------------------------------------------------------------
    # Fallback – uniform vector if nothing was produced
    if best_seq is None:
        best_seq = np.full(2048, 1.0 / 2048, dtype=np.float64)

    # Clip to evaluator’s limits
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"*** Search finished: best objective ≈ {best_val:.6f} (n={best_seq.shape[0]})")
    return best_seq.tolist()
