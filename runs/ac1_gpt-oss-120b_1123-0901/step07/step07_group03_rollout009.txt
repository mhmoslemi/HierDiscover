"""
Multi‑restart optimiser for the step‑function bound.

The algorithm proceeds as follows:

1. **Initial pool** – a variety of seed vectors is generated:
   * the optional warm‑start `height_sequence_1`,
   * uniform mass,
   * Dirichlet samples (different concentrations),
   * anti‑pair reordered Dirichlet samples,
   * edge‑focused Gaussian shapes,
   * a simple triangular shape,
   * power‑law edge shapes.

2. **Per‑seed optimisation** – each seed is refined inside a strict
   time slice by three stages:
   * **Adam** on a smooth‑max (log‑sum‑exp) surrogate,
   * **exact‑max sub‑gradient** (exponentiated‑gradient) phase,
   * **local stochastic hill‑climbing** (swaps and fractional mass transfers).

   The three stages are run sequentially until the slice expires.
   The best vector seen during the stages is kept.

3. **Repeated restarts** – the above pipeline is executed for many
   independent seeds while the global time budget (≤ budget_s) allows
   it.  The overall best vector is retained.

4. **Optional upscale** – if time remains, the best vector is linearly
   interpolated to a finer grid (8192 points) and the three‑stage
   optimisation is run once more on this larger problem.

All returned numbers are non‑negative and clipped to the limits accepted
by `evaluate_sequence`.  The function `propose_candidate` conforms to
the required signature and returns a plain Python list.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
# FFT utilities
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for the gradient."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
# Exact objective (fast, no security checks)
def _objective(a: np.ndarray) -> float:
    """2·n·max(a*a) / (sum(a)**2) – assumes a ≥ 0."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    conv = _full_convolution(a, _next_pow2(2 * a.shape[0] - 1))
    return 2.0 * a.shape[0] * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Candidate generators
def _warm_start(n: int):
    """Return a normalised warm‑start if the global `height_sequence_1` exists."""
    warm = globals().get("height_sequence_1")
    if warm is None:
        return None
    warm = np.asarray(warm, dtype=np.float64)
    if warm.shape[0] > n:
        warm = warm[:n]
    elif warm.shape[0] < n:
        warm = np.pad(warm, (0, n - warm.shape[0]))
    warm = np.maximum(warm, 0.0)
    if warm.sum() < 0.01:
        return None
    return warm / warm.sum()


def _uniform_candidate(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _random_dirichlet_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    return np.random.dirichlet(alpha * np.ones(n)).astype(np.float64)


def _anti_pair_arrangement(vals: np.ndarray) -> np.ndarray:
    """
    Place the largest half of the values on the left side and the smallest
    half on the right side (reversed).  This is the classic “anti‑pair”
    ordering.
    """
    n = len(vals)
    sv = np.sort(vals)[::-1]          # descending
    out = np.empty(n, dtype=np.float64)
    half = n // 2
    out[:half] = sv[:half]           # biggest half
    out[-half:] = sv[-half:][::-1]   # smallest half, reversed
    if n % 2 == 1:
        out[half] = sv[half]
    return out


def _anti_pair_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    vals = np.random.dirichlet(alpha * np.ones(n))
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return _anti_pair_arrangement(vals)


def _gaussian_edges_candidate(n: int, gamma: float = 0.5) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
    shape /= shape.sum()
    return shape


def _triangle_candidate(n: int) -> np.ndarray:
    x = np.arange(n, dtype=np.float64)
    shape = np.minimum(x + 1, n - x)          # 1,2,…⌈n/2⌉,…,2,1
    shape /= shape.sum()
    return shape


def _edge_power_candidate(n: int, p: float = 1.0) -> np.ndarray:
    """U‑shaped power law: larger values at the ends, smaller in the centre."""
    i = np.arange(n, dtype=np.float64)
    denom = (i + 1) * (n - i)
    shape = 1.0 / (denom ** p)
    shape /= shape.sum()
    return shape


# ----------------------------------------------------------------------
# Optimisation primitives
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    lr: float = 0.07,
    lam_start: float = 1.0,
    lam_factor: float = 1.07,
) -> tuple[np.ndarray, float]:
    """Adam on the smooth‑max (log‑sum‑exp) surrogate."""
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lam = lam_start

    best_a = a0.copy()
    best_val = _objective(a0)

    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max → simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective (used for tracking)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val_exact = 2.0 * n * max_c
        if val_exact < best_val:
            best_val = val_exact
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best exact {best_val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient w.r.t. a
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature schedule → true max
        lam = min(lam * lam_factor, 1e6)

        # occasional jitter to escape plateaus
        if it % 5_000 == 0 and (deadline - time.time()) > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2

        it += 1

    return best_a, best_val


def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 0.7,
) -> tuple[np.ndarray, float]:
    """Exact‑max sub‑gradient phase (exponentiated‑gradient)."""
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s

    best_a = a.copy()
    best_val = _objective(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5_000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active set of indices attaining the maximum
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # step‑size schedule (decays like 1/√t)
        eta = eta0 / np.sqrt(it + 1)

        # multiplicative (exponentiated‑gradient) update
        a = a * np.exp(-eta * grad_a)

        # renormalise onto the simplex
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 120_000,
) -> tuple[np.ndarray, float]:
    """Simulated‑annealing style hill‑climbing (swap / fractional transfer)."""
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a)

    start_temp = 0.02  # slightly larger to allow occasional uphill moves
    it = 0
    while time.time() < deadline and it < max_moves:
        if np.random.rand() < 0.5:                # swap two positions
            i, j = np.random.randint(n, size=2)
            if i != j:
                a2 = a.copy()
                a2[i], a2[j] = a2[j], a2[i]
            else:
                continue
        else:                                     # transfer a fraction from i to j
            i = np.random.randint(n)
            if a[i] < 1e-14:
                continue
            j = np.random.randint(n)
            if i == j:
                continue
            delta = a[i] * (np.random.rand() * 0.4)   # move up to 40 %
            a2 = a.copy()
            a2[i] -= delta
            a2[j] += delta

        val2 = _objective(a2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
        else:
            # linear temperature decay
            temp = start_temp * (1.0 - (it / max_moves))
            if temp > 0.0 and np.random.rand() < np.exp(-(val2 - best_val) / temp):
                a = a2

        if it % 5_000 == 0:
            print(f"[Local move {it:06d}] best {best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Pipeline that bundles the three optimisation phases
def _optimize_pipeline(
    a_start: np.ndarray,
    n: int,
    time_budget: float,
    conv_fft_len: int,
    corr_fft_len: int,
) -> tuple[np.ndarray, float]:
    """
    Run Adam → sub‑gradient → local search inside `time_budget` seconds.
    Returns the best vector seen and its exact objective.
    """
    start = time.time()
    # allocate fractions (roughly 1/3 each)
    t_adam = start + time_budget * 0.33
    t_sub = t_adam + time_budget * 0.33
    t_local = t_sub + time_budget * 0.34

    # 1) Adam phase
    a_adam, val_adam = _adam_optimize(
        a_start,
        n,
        conv_fft_len,
        corr_fft_len,
        t_adam,
        max_iters=300_000,
        lr=0.07,
        lam_start=1.0,
        lam_factor=1.07,
    )
    # if time is already over, return immediately
    if time.time() >= t_sub:
        return a_adam, val_adam

    # 2) Exact sub‑gradient phase
    a_sub, val_sub = _subgradient_optimize(
        a_adam,
        n,
        conv_fft_len,
        corr_fft_len,
        t_sub,
        max_iters=300_000,
        eta0=0.7,
    )
    if time.time() >= t_local:
        # pick the best of the first two stages
        if val_sub < val_adam:
            return a_sub, val_sub
        return a_adam, val_adam

    # 3) Local stochastic search
    a_loc, val_loc = _local_random_search(
        a_sub,
        n,
        conv_fft_len,
        t_local,
        max_moves=120_000,
    )
    # choose the overall best
    best_a, best_val = a_adam, val_adam
    if val_sub < best_val:
        best_a, best_val = a_sub, val_sub
    if val_loc < best_val:
        best_a, best_val = a_loc, val_loc
    return best_a, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises
    `evaluate_sequence`.  The routine respects the given time budget and
    returns the best sequence found (as a plain Python list).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # problem size for the first stage
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Build a diverse list of candidate factories (no closures of mutable vars)
    factories = []

    # warm start (if it exists) and a few rolled copies
    warm = _warm_start(n)
    if warm is not None:
        factories.append(lambda w=warm: w.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            factories.append(lambda s=shift, w=warm: np.roll(w, s))

    # uniform
    factories.append(lambda: _uniform_candidate(n))

    # random Dirichlet samples with various concentrations
    for _ in range(4):
        alpha = np.random.uniform(0.2, 2.0)
        factories.append(lambda a=alpha: _random_dirichlet_candidate(n, a))

    # anti‑pair Dirichlet samples
    for _ in range(4):
        alpha = np.random.uniform(0.2, 2.0)
        factories.append(lambda a=alpha: _anti_pair_candidate(n, a))

    # Gaussian‑edge shapes
    for _ in range(6):
        gamma = np.random.uniform(0.1, 2.0)
        factories.append(lambda g=gamma: _gaussian_edges_candidate(n, g))

    # simple tent
    factories.append(lambda: _triangle_candidate(n))

    # power‑law edge shapes
    for _ in range(3):
        p = np.random.uniform(0.5, 3.0)
        factories.append(lambda e=p: _edge_power_candidate(n, e))

    # shuffle to increase diversity
    np.random.shuffle(factories)

    # ------------------------------------------------------------------
    # Global best tracker
    best_seq = None
    best_val = float("inf")
    pipeline_counter = 0

    # ------------------------------------------------------------------
    # Main multi‑restart loop
    min_pipeline_time = 30.0  # each restart gets at least this many seconds
    while factories and (deadline - time.time()) > (min_pipeline_time + 10.0):
        factory = factories.pop()
        cand = factory()
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 1e-12:
            cand = np.random.dirichlet(np.ones(n))
            s = cand.sum()
        cand = cand / s                                   # normalise to sum = 1

        init_val = _objective(cand)
        print(f"[Restart {pipeline_counter:02d}] initial value {init_val:.6f}")

        # allocate a slice of the remaining budget (keep ~10 s for final upscale)
        remaining = deadline - time.time()
        slice_time = min(remaining - 10.0, min_pipeline_time * 1.5)

        # run the three‑phase optimiser on this seed
        a_opt, val_opt = _optimize_pipeline(
            cand, n, slice_time, conv_fft_len, corr_fft_len
        )
        print(f"[Restart {pipeline_counter:02d}] best after optimisation {val_opt:.6f}")

        if val_opt < best_val:
            best_val = val_opt
            best_seq = a_opt.copy()
            print(f" *** New global best {best_val:.6f} ***")
        pipeline_counter += 1

    # ------------------------------------------------------------------
    # Final upscale & refinement if time permits
    time_left = deadline - time.time()
    if best_seq is not None and time_left > 20.0:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        # linear interpolation from the current best to the finer grid
        x_old = np.linspace(0.0, 1.0, n, endpoint=False)
        x_new = np.linspace(0.0, 1.0, n_big, endpoint=False)
        a_big = np.interp(x_new, x_old, best_seq)
        a_big = np.maximum(a_big, 0.0)
        a_big /= a_big.sum()

        print(f"[Upscale] refining on {n_big} points, time left {time_left:.1f}s")
        a_big_opt, val_big_opt = _optimize_pipeline(
            a_big,
            n_big,
            time_left,
            conv_fft_len_big,
            corr_fft_len_big,
        )
        if val_big_opt < best_val:
            best_val = val_big_opt
            best_seq = a_big_opt.copy()
            print(f"[Upscale] improved global best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Safeguard – ensure something valid is returned
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the range accepted by the evaluator
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Returning best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
