def beta_shape(n, alpha, beta):
    x = (np.arange(n) + 0.5)/n
    shape = np.power(x, alpha-1) * np.power(1-x, beta-1)
    shape[np.isnan(shape)] = 0.0
    shape = np.maximum(shape, 0.0)
    sum_ = shape.sum()
    if sum_ == 0:
        shape = np.full(n, 1/n)
    else:
        shape /= sum_
    return shape

def _beta_candidate(n, alpha, beta):
    x = (np.arange(n) + 0.5) / n
    # add epsilon to avoid 0**negative
    eps = 1e-12
    shape = np.power(x + eps, alpha - 1) * np.power(1 - x + eps, beta - 1)
    shape = np.where(np.isfinite(shape), shape, 0.0)

    shape = np.exp(-0.5 * ((i - n/2) / sigma) ** 2)
    shape /= shape.sum()
    return shape

                  max_iters=200_000, lr=0.02, lam_start=0.1, lam_factor=1.001):
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8

    best_a = a0.copy()
    best_val = np.inf
    it = 0
    lam = lam_start
    while time.time() < deadline and it < max_iters:
        # softmax to get a that sums to one
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w = w / w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)   # ∂f/∂a
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)   # chain‑rule to logits

        # Adam update

        v = beta2 * v + (1 - beta2) * (grad_log ** 2)

        v_hat = v / (1 - beta2 ** (it + 1))

        # occasional jitter near the deadline
        if it % 4000 == 0 and time.time() > deadline - 5:
            logits += (np.random.rand(n) - 0.5) * 1e-3

        it += 1

    return best_a, best_val

def subgradient_optimize(a0, n, conv_fft_len, corr_fft_len, deadline,

    s = a.sum()
    if s < 1e-12:

        s = a.sum()

        val = 2.0 * n * max_c

        # active set (indices achieving the max)
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)

        else:

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        a = a * np.exp(-eta * grad_a)

        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val

def simulated_annealing(a0, n, fft_len, deadline, cur_best_val):
    a = a0.copy()
    conv = _full_convolution(a, fft_len)
    cur_max = conv.max()
    cur_val = 2.0 * n * cur_max
    best_a = a.copy()
    best_val = cur_val

    start = time.time()
    it = 0

        fraction = elapsed / (deadline - start + 1e-12)

        # linear cooling from 0.2 to 1e‑6

        move_scale = max(0.01, 0.4 * (1.0 - fraction))

            continue
        j = np.random.randint(n - 1)

            j += 1

        conv_new = conv.copy()
        a_new[i] -= delta

        conv_new[2 * i] += delta * delta
        conv_new[2 * j] += delta * delta
        conv_new[i + j] -= 2.0 * delta * delta

        new_max = conv_new.max()
        new_val = 2.0 * n * new_max

Hybrid optimisation for the max‑auto‑convolution problem.

The code builds a diverse pool of warm‑starts (uniform, Dirichlet,
Beta‑law, Gaussian, arcsine, etc.), runs an Adam optimiser on a

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()

    """Linear convolution a*a via FFT (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]

    """Linear cross‑correlation (w ★ a) needed for the gradient."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1

    for k in (64, 128, 256, 512, 1024):
        a = np.zeros(n, dtype=np.float64)
        pos = np.round(np.linspace(0, n - 1, k)).astype(int)
        a[pos] = 1.0 / k
        cand.append(a)

    return cand


def _beta_candidate(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta(α,β) density sampled on the mid‑points of a uniform grid."""
    x = (np.arange(n) + 0.5) / n
    eps = 1e-12
    shape = np.power(x + eps, alpha - 1) * np.power(1 - x + eps, beta - 1)
    shape = np.where(np.isfinite(shape), shape, 0.0)
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s == 0:
        shape = np.full(n, 1.0 / n)
    else:
        shape /= s
    return shape


def _gaussian_candidate(n: int, sigma_factor: float) -> np.ndarray:
    """Discrete Gaussian centred on the grid; sigma = factor·n."""
    i = np.arange(n, dtype=np.float64)
    sigma = sigma_factor * n
    shape = np.exp(-0.5 * ((i - n / 2) / sigma) ** 2)
    shape /= shape.sum()
    return shape

# Optimisers ------------------------------------------------------------
def adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
    lr: float = 0.02,
    lam_start: float = 0.10,
    lam_factor: float = 1.001,
) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth‑max (log‑sum‑exp) surrogate.
    Returns the best vector seen and its value 2·n·max(conv) (sum is 1).
    """
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lam = lam_start

    best_a = a0.copy()
    best_val = np.inf
    it = 0

    while time.time() < deadline and it < max_iters:
        # softmax → a on the simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:05d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w = w / w_sum

        # gradient of the surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature schedule
        lam = min(lam * lam_factor, 1e6)

        # tiny jitter near the end to escape plateaus
        if it % 4000 == 0 and time.time() > deadline - 5:
            logits += (np.random.rand(n) - 0.5) * 1e-3

        it += 1

    return best_a, best_val


def subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
    eta0: float = 0.9,
) -> tuple[np.ndarray, float]:
    """Exact‑max sub‑gradient descent with exponentiated‑gradient updates."""
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s

    best_a = a.copy()
    best_val = np.inf
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 2000 == 0:
                print(f"[Subgrad it={it:05d}] new best {val:.6f}")

        # active set (indices achieving the current maximum)
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # diminishing learning‑rate
        eta = eta0 / np.sqrt(it + 1)

        # exponentiated‑gradient step and renormalisation
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def simulated_annealing(
    a0: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    cur_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Fast mass‑transfer simulated annealing.
    Updates the convolution in O(n) per elementary move.
    """
    a = a0.copy()
    conv = _full_convolution(a, fft_len)
    cur_val = 2.0 * n * conv.max()
    best_a = a.copy()
    best_val = cur_val

    start = time.time()
    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        fraction = elapsed / (deadline - start + 1e-12)

        # linear cooling from 0.2 down to 1e‑6
        T = max(1e-6, 0.2 * (1.0 - fraction))
        # move magnitude shrinks with the temperature
        move_scale = max(0.01, 0.4 * (1.0 - fraction))

        # source index – weighted by its current mass
        if a.sum() == 0:
            break
        i = np.random.choice(n, p=a / a.sum())
        if a[i] < 1e-12:
            continue

        # destination index – uniform among the others
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * (np.random.rand() * move_scale)

        # copy‑on‑write updates
        a_new = a.copy()
        conv_new = conv.copy()
        a_new[i] -= delta
        a_new[j] += delta

        # convolution update (exact)
        conv_new[i : i + n] -= 2.0 * delta * a
        conv_new[j : j + n] += 2.0 * delta * a
        conv_new[2 * i] += delta * delta
        conv_new[2 * j] += delta * delta
        conv_new[i + j] -= 2.0 * delta * delta

        new_val = 2.0 * n * conv_new.max()

        # Metropolis acceptance
        if new_val < cur_val or np.random.rand() < np.exp(-(new_val - cur_val) / T):
            a, conv = a_new, conv_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[SA it={it:05d}] new best {best_val:.6f}")

        it += 1
        if it % 5000 == 0:
            print(f"[SA it={it:05d}] best {best_val:.6f}")

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises
        2·n·max(a∗a) / (sum a)^2 .
    The algorithm follows these stages:
        1. construct a large pool of warm‑starts,
        2. optimise each start with Adam + sub‑gradient (short budget),
        3. perform a long simulated‑annealing run on the current best,
        4. upscale to 8192 points and repeat a short Adam + sub‑gradient pass,
    always respecting the overall time budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # Optional warm‑start from a previous run
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        small_n = 4096
        if warm.shape[0] > small_n:
            warm = warm[:small_n]
        elif warm.shape[0] < small_n:
            warm = np.pad(warm, (0, small_n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # ------------------------------------------------------------------
    # Small grid (fast exploration)
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # pool of candidates
    candidates = _initial_candidates(n, warm)

    # extra families (Beta, Gaussian, raised‑cosine)
    for a in (0.2, 0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0):
        candidates.append(_beta_candidate(n, a, a))
    for sf in (0.05, 0.10, 0.15, 0.20):
        candidates.append(_gaussian_candidate(n, sf))

    # raised‑cosine (positive half‑wave)
    i = np.arange(n, dtype=np.float64)
    rc = np.maximum(0.0, np.cos(np.pi * (i - n / 2) / n))
    rc /= rc.sum()
    candidates.append(rc)

    best_seq: np.ndarray | None = None
    best_val = np.inf

    # ------------------------------------------------------------------
    # 2.  Optimise each candidate while time permits
    per_cand_budget = 25.0          # seconds given to Adam + sub‑grad per candidate
    cand_idx = 0
    while time.time() < deadline - 10.0 and cand_idx < len(candidates):
        a0 = np.maximum(candidates[cand_idx], 0.0)
        if a0.sum() < 0.01:
            a0 = np.random.dirichlet(np.ones(n))
        else:
            a0 = a0 / a0.sum()

        local_deadline = min(deadline - 5.0, time.time() + per_cand_budget)

        # Adam (smooth‑max surrogate)
        a_adam, val_adam = adam_optimize(
            a0,
            n,
            conv_fft_len,
            corr_fft_len,
            local_deadline,
            max_iters=300_000,
            lr=0.02,
            lam_start=0.10,
            lam_factor=1.001,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[cand {cand_idx}] Adam improved best to {val_adam:.6f}")

        # Sub‑gradient refinement (exact max) – run on the current global best
        if time.time() < deadline - 5.0:
            sub_deadline = min(deadline - 5.0, time.time() + per_cand_budget * 0.6)
            a_sub, val_sub = subgradient_optimize(
                best_seq,
                n,
                conv_fft_len,
                corr_fft_len,
                sub_deadline,
                max_iters=250_000,
                eta0=0.9,
            )
            if val_sub < best_val:
                best_val = val_sub
                best_seq = a_sub.copy()
                print(f"[cand {cand_idx}] Sub‑grad improved best to {val_sub:.6f}")

        cand_idx += 1

    # ------------------------------------------------------------------
    # 3.  Long simulated‑annealing run on the best candidate found so far
    if best_seq is not None and time.time() < deadline - 5:
        a_sa, val_sa = simulated_annealing(
            best_seq,
            n,
            conv_fft_len,
            deadline - 2.0,
            best_val,
        )
        if val_sa < best_val:
            best_val = val_sa
            best_seq = a_sa.copy()
            print(f"[SA final] best improved to {val_sa:.6f}")

    # ------------------------------------------------------------------
    # 4.  Upscale to 8192 points and repeat a short Adam + sub‑gradient pass
    if best_seq is not None and time.time() + 30.0 < deadline:
        n_big = 8192
        # linear interpolation onto the larger grid
        x_small = np.linspace(0.0, 1.0, n, endpoint=False)
        x_big = np.linspace(0.0, 1.0, n_big, endpoint=False)
        a_big = np.interp(x_big, x_small, best_seq)
        a_big = np.maximum(a_big, 0.0)
        a_big /= a_big.sum()

        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        # brief Adam pass on the enlarged grid
        ad_deadline = min(deadline - 5.0, time.time() + 20.0)
        a_adam_big, _ = adam_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            ad_deadline,
            max_iters=300_000,
            lr=0.015,
            lam_start=0.08,
            lam_factor=1.001,
        )

        # final exact‑max sub‑gradient refinement
        if time.time() < deadline - 2.0:
            a_ref_big, val_ref_big = subgradient_optimize(
                a_adam_big,
                n_big,
                conv_fft_len_big,
                corr_fft_len_big,
                deadline,
                max_iters=200_000,
                eta0=0.8,
            )
            if val_ref_big < best_val:
                best_val = val_ref_big
                best_seq = a_ref_big.copy()
                print(f"[Big final] improved to {val_ref_big:.6f}")

    # ------------------------------------------------------------------
    # 5.  Fallback – guarantee a valid return
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Clamp to the limits enforced by evaluate_sequence
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Store for potential future warm‑starts
    globals()["height_sequence_1"] = best_seq.tolist()

    # Final reporting (optional)
    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"Final reported value {final_val:.6f} (length {len(best_seq)})")
    except Exception:
        print(f"Internal best value {best_val:.6f}")

    return best_seq.tolist()
