def _full_convolution(a, fft_len):
    fa = np.fft.rfft(a, fft_len)

"""
search for a step‑function that minimises the supplied evaluation.
-----------------------------------------------------------------
Idea
----
The objective is 2·n·max(conv)/sum(a)².  Since scaling does not change the
value we can fix sum(a)=1 and minimise max(conv), i.e. make the
distribution of the sum of two i.i.d. draws as flat as possible.
A dense uniform vector gives a triangular convolution (max≈1/n).  By
mixing a very shallow exponential (or its mirror) with a uniform part we
can tilt the mass to the ends while keeping the extremes small.
These parametric families are cheap to generate, so we exhaustively scan
them (including alphas as small as 1·10⁻⁶).  The best candidate is then
refined with a few thousand exponentiated‑gradient sub‑gradient steps
that directly minimise the exact max‑convolution, finally polished by a
tiny stochastic hill‑climb.

The whole search respects the 1000 s budget, stays on the simplex, and
returns a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation needed for the gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact objective 2·n·max(a*a)/(sum(a)²).  ``a`` is assumed non‑negative."""
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c / (a.sum() ** 2))


# ----------------------------------------------------------------------
def _candidate_exponential_mixtures(n: int):
    """
    Generate a compact pool of candidates:
      – pure left exponential,
      – pure right exponential,
      – mixtures of a (shallow) exponential and the uniform vector,
      – symmetric two‑sided mixtures.
    The decay parameters α span 1·10⁻⁶ … 0.3.
    """
    candidates = []

    i = np.arange(n, dtype=np.float64)

    # shallow alphas (log‑space)
    alphas = np.logspace(-6, -0.5, num=30)          # 1e‑6 … ≈0.3
    w_vals = np.linspace(0.0, 1.0, 11)               # weight of exponential

    # left‑only and right‑only mixtures
    for alpha in alphas:
        left = np.exp(-alpha * i)            # decreasing from the left edge
        left /= left.sum()
        right = np.exp(-alpha * (n - 1 - i)) # decreasing from the right edge
        right /= right.sum()

        for w in w_vals:
            # left + uniform
            shape = w * left + (1.0 - w) / n
            shape /= shape.sum()
            candidates.append(shape)

            # right + uniform
            shape = w * right + (1.0 - w) / n
            shape /= shape.sum()
            candidates.append(shape)

    # asymmetric two‑sided mixtures
    for alpha_l in alphas[::5]:
        for alpha_r in alphas[::5]:
            left = np.exp(-alpha_l * i)
            right = np.exp(-alpha_r * (n - 1 - i))
            for wL in (0.0, 0.2, 0.5, 0.8):
                for wR in (0.0, 0.2, 0.5, 0.8):
                    wU = 1.0 - wL - wR
                    if wU < -1e-12:
                        continue
                    shape = wL * left + wR * right + max(wU, 0.0) / n
                    shape /= shape.sum()
                    candidates.append(shape)

    # a few random Dirichlet draws for diversity
    for _ in range(20):
        cand = np.random.dirichlet(np.ones(n))
        candidates.append(cand)

    # final uniform fallback (always present)
    candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

    return candidates


# ----------------------------------------------------------------------
def _subgradient_refinement(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    init_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient optimisation on the *exact* max‑convolution.
    A single active shift (the current maximiser) is used, with a slowly decaying
    step size.  The routine stops when time runs out or no improvement for a few
    seconds.
    """
    a = a0.copy()
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a /= a.sum()

    best_seq = a.copy()
    best_val = init_best_val

    eta = 0.7                # start step‑size
    decay = 0.99995
    it = 0
    last_improve = time.time()

    while time.time() < deadline:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Sub] iter={it:06d} new best {val:.6f}")

        # index (or one of them) attaining the maximum
        k = int(conv.argmax())
        w = np.zeros_like(conv)
        w[k] = 1.0

        # gradient of the max‑convolution w.r.t. a
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # exponentiated‑gradient update (keeps non‑negativity)
        a = a * np.exp(-eta * grad)

        # renormalise – protect against collapse
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        eta *= decay
        it += 1

        # stop if stalled for a while
        if time.time() - last_improve > 6.0:
            break

    return best_seq, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Main entry point – searches for a non‑negative vector that minimises
    ``evaluate_sequence``.  The method builds a pool of shallow exponential
    mixtures (including very tiny decay rates), evaluates them, refines the
    best one with a short exponentiated‑gradient sub‑gradient phase and finally
    performs a cheap stochastic hill‑climb.  The best sequence seen is
    returned as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # problem size (the previous best used 4096)
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)      # length for a*a
    corr_fft_len = _next_pow2(3 * n - 2)      # length for cross‑correlation

    # --------------------------------------------------------------
    # initialise from a possible warm‑start
    warm = globals().get("height_sequence_1")
    if warm is not None:
        a0 = np.asarray(warm, dtype=np.float64)
        if a0.shape[0] != n:
            if a0.shape[0] > n:
                a0 = a0[:n]
            else:
                a0 = np.pad(a0, (0, n - a0.shape[0]))
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        if s < 0.01:
            a0 = np.random.dirichlet(np.ones(n))
        else:
            a0 = a0 / s
    else:
        a0 = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = a0.copy()
    best_val = _objective(best_seq, n, conv_fft_len)
    print(f"[Init] objective = {best_val:.6f}")

    # --------------------------------------------------------------
    # stage 1 – exhaustive sweep of exponential‑mixture families
    candidates = _candidate_exponential_mixtures(n)

    for cand in candidates:
        if time.time() >= deadline:
            break
        val = _objective(cand, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_seq = cand.copy()
            print(f"[Sweep] new best {val:.6f}")

    # --------------------------------------------------------------
    # stage 2 – exponentiated‑gradient sub‑gradient refinement
    if time.time() < deadline:
        best_seq, best_val = _subgradient_refinement(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            best_val,
        )

    # --------------------------------------------------------------
    # stage 3 – tiny stochastic hill‑climb around the current best
    logits = np.log(best_seq + 1e-15)
    while time.time() < deadline:
        # perturb logits a little
        noise = np.random.randn(n) * 0.02
        logp = logits + noise
        mx = logp.max()
        exp_log = np.exp(logp - mx)
        a = exp_log / exp_log.sum()
        val = _objective(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            logits = np.log(best_seq + 1e-15)
            print(f"[Hill] improved to {val:.6f}")

    # --------------------------------------------------------------
    # final safety – clip to evaluator limits and return
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"*** Finished search: best objective ≈ {best_val:.6f} (n={best_seq.shape[0]})")
    return best_seq.tolist()
