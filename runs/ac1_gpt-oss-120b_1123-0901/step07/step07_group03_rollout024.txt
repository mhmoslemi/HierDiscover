"""
Propose a low‑value step function.

Strategy
--------
1.  Build a pool of candidate multisets `v` (sorted descending values) using
    Dirichlet draws with many concentration parameters, exponential decay
    and power‑law shapes.
2.  Convert each multiset to a step function by the *anti‑pair* permutation,
    which is known to minimise the convolution maximum for a given set of
    masses.
3.  Keep the K best multisets (according to the exact objective).
4.  Refine each kept multiset with a projected exponentiated‑gradient loop:
        * a = anti_pair(v)
        * compute the exact sub‑gradient of max(a★a)
        * map the gradient back to v (inverse permutation)
        * multiplicative update v ← v·exp(−η·grad)  (η decays as 1/√t)
        * normalise to sum = 1 and re‑sort to enforce non‑increasing order
    The permutation guarantees that the ordering is always optimal,
    reducing the dimensionality of the problem.
5.  After the gradient phase, run a cheap stochastic hill‑climber
    (swap / mass‑transfer moves) on the best full‑size vector.
6.  Return the best sequence found (as a plain Python list).  All numbers
    are non‑negative and automatically satisfy the evaluator's clipping
    limits.

The whole routine respects a user‑provided time budget (default 1000 s)
and prints progress information.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT helpers -----------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the sub‑gradient.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
# Objective helpers -----------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact objective  2·n·max(a★a) / (sum(a)²).  a must be non‑negative.
    The code assumes sum(a) > 0 (otherwise returns +inf).
    """
    s = a.sum()
    if s < 1e-12:
        return np.inf
    conv = _full_convolution(a, _next_pow2(2 * a.shape[0] - 1))
    return 2.0 * a.shape[0] * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Anti‑pair permutation --------------------------------------------------
def _anti_pair_arrangement(vals: np.ndarray) -> np.ndarray:
    """
    Anti‑pair ordering: largest opposite the smallest,
    2nd‑largest opposite 2nd‑smallest, etc.
    The input is assumed to be a 1‑D array of non‑negative numbers
    (order does NOT matter – the function sorts it).
    """
    n = len(vals)
    sorted_vals = np.sort(vals)[::-1]          # descending
    a = np.empty(n, dtype=np.float64)
    half = n // 2
    a[:half] = sorted_vals[:half]             # biggest half on the left
    a[-half:] = sorted_vals[-half:][::-1]     # smallest half on the right
    if n % 2 == 1:
        a[half] = sorted_vals[half]
    return a


# ----------------------------------------------------------------------
# Projected exponentiated‑gradient optimisation on the multiset ----------
def _optimize_multiset(
    v0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
    eta0: float = 0.7,
) -> tuple[np.ndarray, float]:
    """
    Refine a sorted multiset `v` (descending) while keeping it sorted.
    The actual step‑function is a = anti_pair(v) and the objective
    is evaluated on `a`.  Multiplicative (exponentiated) updates with a
    decaying step size are used; after each update the vector is re‑sorted
    to stay in the monotone cone (the projection onto the cone of
    non‑increasing vectors is simply sorting).
    Returns the best multiset found and its objective value.
    """
    # initialise
    v = np.maximum(v0, 0.0)
    s = v.sum()
    if s < 1e-12:                     # degenerate case – fallback to uniform
        v = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        v = v / s

    # ensure exact descending order
    v = np.sort(v)[::-1]

    best_val = np.inf
    best_v = None

    it = 0
    half = n // 2
    while time.time() < deadline and it < max_iters:
        # form the step‑function from the multiset
        a = _anti_pair_arrangement(v)
        a = a / a.sum()               # keep sum = 1 (redundant but safe)

        # convolution and max‑value
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_v = v.copy()
            print(f"[Optimize it={it:06d}] new best {val:.6f}")

        # active set of indices attaining the maximum
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        # sub‑gradient w.r.t. a
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # map gradient to the multiset (inverse anti‑pair permutation)
        grad_v = np.empty_like(v)
        # first half of a corresponds directly to first half of v
        grad_v[:half] = grad_a[:half]
        # second half of a corresponds to reversed second half of v
        for i in range(half, n):
            idx = n - 1 - i
            grad_v[idx] = grad_a[i]

        # exponentiated‑gradient update
        eta = eta0 / np.sqrt(it + 1)
        v = v * np.exp(-eta * grad_v)

        # re‑normalise onto the simplex
        s = v.sum()
        if s < 1e-12:
            v = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            v = v / s

        # enforce monotone decreasing (projection = sorting)
        v = np.sort(v)[::-1]

        it += 1

    return best_v, best_val

# Simple stochastic hill‑climber on the full vector ----------------------
def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing style local search:
        * swap two entries OR transfer a fraction of mass between them
    Accept moves that improve the objective, otherwise accept with a
    small temperature that linearly decays.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a)

    start_temp = 0.02
    it = 0
    while time.time() < deadline and it < max_moves:
        if np.random.rand() < 0.5:                # swap
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            a2 = a.copy()
            a2[i], a2[j] = a2[j], a2[i]
        else:                                       # mass transfer
            i = np.random.randint(n)
            if a[i] < 1e-14:
                continue
            j = np.random.randint(n)
            if i == j:
                continue
            delta = a[i] * (np.random.rand() * 0.5)   # up to 50 % move
            a2 = a.copy()
            a2[i] -= delta
            a2[j] += delta

        val2 = _objective(a2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
        else:
            temp = start_temp * (1.0 - it / max_moves)
            if temp > 0.0 and np.random.rand() < np.exp(-(val2 - best_val) / temp):
                a = a2

        if it % 10_000 == 0:
            print(f"[Local move {it:06d}] best {best_val:.6f}")

        it += 1

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises
        F(a) = 2·n·max(a★a) / (sum(a)²)

    The routine respects the time budget and returns the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5    # safety margin

    # ------------------------------------------------------------------
    # parameters
    n = 4096                                 # base length
    conv_fft_len = _next_pow2(2 * n - 1)     # FFT lengths for convolutions
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # 1) generate a rich pool of candidate multisets (sorted descending)
    candidates_v = []

    # Dirichlet draws with many concentration values
    alphas = [0.01, 0.03, 0.07, 0.1, 0.2, 0.5, 1.0, 2.0]
    for alpha in alphas:
        for _ in range(4):                    # 4 draws per alpha
            v = np.random.dirichlet(alpha * np.ones(n))
            v = np.sort(v)[::-1]
            candidates_v.append(v)

    # Exponential decay shapes
    for lam in np.linspace(0.0, 5.0, 12):
        v = np.exp(-lam * np.arange(n, dtype=np.float64))
        v = v / v.sum()
        v = np.sort(v)[::-1]
        candidates_v.append(v)

    # Power‑law shapes
    for p in [0.2, 0.5, 1.0, 2.0]:
        v = 1.0 / (np.arange(1, n + 1, dtype=np.float64) ** p)
        v = v / v.sum()
        v = np.sort(v)[::-1]
        candidates_v.append(v)

    # ------------------------------------------------------------------
    # 2) evaluate each candidate (fast) and keep the best few
    evaluated = []
    for idx, v in enumerate(candidates_v):
        a = _anti_pair_arrangement(v)
        a = a / a.sum()
        val = _objective(a)
        evaluated.append((val, v))
        print(f"[Init #{idx:03d}] objective {val:.6f}")
    evaluated.sort(key=lambda x: x[0])
    top_k = 5
    top_candidates = evaluated[:top_k]

    # ------------------------------------------------------------------
    # 3) refined optimisation on the best multisets
    remaining = deadline - time.time()
    if remaining > 30.0:
        # split the remaining budget evenly among the top candidates
        per_candidate = remaining / len(top_candidates) - 5.0
        for val0, v0 in top_candidates:
            sub_deadline = min(deadline, time.time() + per_candidate)
            v_opt, val_opt = _optimize_multiset(
                v0,
                n,
                conv_fft_len,
                corr_fft_len,
                sub_deadline,
                max_iters=200_000,
                eta0=0.7,
            )
            if val_opt < val0:
                print(f"[Refine] improved from {val0:.6f} to {val_opt:.6f}")
                val0 = val_opt
                v0 = v_opt
            # keep the globally best
            if val_opt < float('inf'):
                if 'best_val' not in locals() or val_opt < best_val:
                    best_val = val_opt
                    best_v = v_opt.copy()
        # if optimisation never produced a result (unlikely), fall back:
        if 'best_v' not in locals():
            best_v = top_candidates[0][1].copy()
            best_val = top_candidates[0][0]
    else:
        # not enough time – just take the best evaluated one
        best_val, best_v = top_candidates[0]

    # ------------------------------------------------------------------
    # 4) final local stochastic improvement on the full vector
    best_a = _anti_pair_arrangement(best_v)
    best_a = best_a / best_a.sum()
    if time.time() < deadline - 5.0:
        a_local, val_local = _local_random_search(
            best_a,
            n,
            conv_fft_len,
            deadline - 2.0,
            max_moves=200_000,
        )
        if val_local < best_val:
            best_val = val_local
            best_a = a_local
            print(f"[Local] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5) safeguard: ensure a valid return
    if best_a is None or len(best_a) == 0:
        best_a = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to evaluator limits (0 ≤ x ≤ 1000)
    best_a = np.clip(best_a, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_a)})")
    return best_a.tolist()
