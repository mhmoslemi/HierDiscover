"""Multi‑restart optimizer for the step‑function bound.

The routine builds several diverse initial vectors (warm‑start, random
Dirichlet, anti‑pair permutations), then for each restart runs a short
Adam phase on a smooth‑max surrogate, a sub‑gradient (exponentiated‑gradient)
refinement, and a simulated‑annealing hill‑climber.  Hyper‑parameters are
randomised per restart to increase exploration.  The overall best sequence
found is returned; if time permits a final interpolation to a finer grid
(8192 points) is performed and refined once more.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]

# ----------------------------------------------------------------------
# Objective -------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(a*a)/(sum(a)²).  Assumes a ≥ 0."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    conv = _full_convolution(a, _next_pow2(2 * a.shape[0] - 1))
    return 2.0 * a.shape[0] * conv.max() / (s * s)

# ----------------------------------------------------------------------
# Warm‑start handling ---------------------------------------------------
def _warm_start(n: int):
    """Try to reuse the global height_sequence_1 if it exists."""
    warm = globals().get("height_sequence_1")
    if warm is None:
        return None
    warm = np.asarray(warm, dtype=np.float64)
    if warm.shape[0] > n:
        warm = warm[:n]
    elif warm.shape[0] < n:
        warm = np.pad(warm, (0, n - warm.shape[0]))
    warm = np.maximum(warm, 0.0)
    if warm.sum() >= 0.01:
        warm = warm / warm.sum()
        return warm
    return None

# ----------------------------------------------------------------------
# Candidate generators --------------------------------------------------
def _uniform_candidate(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _random_dirichlet_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    return np.random.dirichlet(alpha * np.ones(n)).astype(np.float64)


def _anti_pair_arrangement(vals: np.ndarray) -> np.ndarray:
    """
    Original anti‑pair layout: largest values on the left half,
    smallest on the right half (reversed).
    """
    n = len(vals)
    sorted_vals = np.sort(vals)[::-1]          # descending
    a = np.empty(n, dtype=np.float64)
    half = n // 2
    a[:half] = sorted_vals[:half]
    a[-half:] = sorted_vals[-half:][::-1]
    if n % 2 == 1:
        a[half] = sorted_vals[half]
    return a


def _balanced_anti_pair(vals: np.ndarray) -> np.ndarray:
    """
    Balanced anti‑pair: place the largest, then the second largest at opposite ends,
    third largest next to the first, etc.  This puts large masses on both extremes
    but never opposite each other.
    """
    n = len(vals)
    sorted_vals = np.sort(vals)[::-1]          # descending
    a = np.empty(n, dtype=np.float64)
    left, right = 0, n - 1
    for i, v in enumerate(sorted_vals):
        if i % 2 == 0:
            a[left] = v
            left += 1
        else:
            a[right] = v
            right -= 1
    return a


def _gaussian_edges_candidate(n: int, gamma: float = 0.5) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
    shape = shape / shape.sum()
    return shape


# ----------------------------------------------------------------------
# Optimisers ------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    lr: float = 0.07,
    lam_start: float = 1.0,
    lam_factor: float = 1.07,
) -> tuple[np.ndarray, float]:
    """
    Adam on a smooth‑max (log‑sum‑exp) surrogate.
    The vector is kept on the simplex via a softmax of logits.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lam = lam_start

    best_a = a0.copy()
    best_val = _objective(a0)

    it = 0
    while time.time() < deadline and it < max_iters:
        # projection onto simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective (used for tracking)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val_exact = 2.0 * n * max_c
        if val_exact < best_val:
            best_val = val_exact
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Adam it={it:06d}] new best exact {best_val:.6f}")

        # smooth‑max weights (softmax around the current maximum)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate wrt a
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature schedule → true max
        lam = min(lam * lam_factor, 1e6)

        # occasional jitter to escape plateaus
        if it % 5000 == 0 and (deadline - time.time()) > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2

        it += 1

    return best_a, best_val


def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 0.7,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient phase using exponentiated‑gradient updates.
    """
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s

    best_a = a.copy()
    best_val = _objective(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # indicator of active (max‑attaining) entries
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # step‑size schedule (decaying)
        eta = eta0 / np.sqrt(it + 1)

        # multiplicative (exponentiated‑gradient) update
        a = a * np.exp(-eta * grad_a)

        # re‑project onto simplex
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 120_000,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing style hill‑climbing:
    - swap two entries OR transfer a fraction of mass.
    - accept if improvement or with probability exp(-Δ/Temp).
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a)

    start_temp = 0.01
    it = 0
    while time.time() < deadline and it < max_moves:
        # choose move type
        if np.random.rand() < 0.5:          # swap two positions
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            a2 = a.copy()
            a2[i], a2[j] = a2[j], a2[i]
        else:                               # transfer a fraction from i to j
            i = np.random.randint(n)
            if a[i] < 1e-14:
                continue
            j = np.random.randint(n)
            if i == j:
                continue
            delta = a[i] * (np.random.rand() * 0.4)   # up to 40 % move
            a2 = a.copy()
            a2[i] -= delta
            a2[j] += delta

        val2 = _objective(a2)

        # annealing acceptance
        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
        else:
            temp = start_temp * (1.0 - (it / max_moves))
            if temp > 0.0 and np.random.rand() < np.exp(-(val2 - best_val) / temp):
                a = a2

        if it % 5000 == 0:
            print(f"[Local move {it:06d}] best {best_val:.6f}")

        it += 1

    return best_a, best_val

# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative height sequence that minimises
    `2·n·max(a*a)/(sum(a)²)`.  The algorithm runs several independent
    restarts (random Dirichlet + anti‑pair permutations) each consisting
    of an Adam phase, a sub‑gradient phase and a short stochastic local
    search.  The best solution across all restarts is returned.  If time
    permits the final vector is interpolated to a finer grid (8192) and
    refined once more.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # Fixed parameters
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # 0) Warm start (if available)
    warm = _warm_start(n)
    if warm is not None:
        best_seq = warm.copy()
        best_val = _objective(best_seq)
        print(f"[Warm] initial value {best_val:.6f}")
    else:
        best_seq = None
        best_val = float("inf")

    # ------------------------------------------------------------------
    # Determine how many independent restarts we can afford.
    # Rough budget: ~200 s per restart (Adam+Subgrad+Local) + final up‑scale.
    remaining = deadline - time.time()
    max_possible = int(remaining // 200)
    max_restarts = min(max_possible, 4)
    if max_restarts < 1:
        max_restarts = 1
    print(f"Planning {max_restarts} restarts (≈200 s each)")

    for r in range(max_restarts):
        if time.time() >= deadline - 30.0:
            break   # keep a small buffer for the final upscale

        # ----- 1) random start -------------------------------------------------
        alpha = np.random.choice([0.3, 0.7, 1.0, 2.0])
        cand = np.random.dirichlet(alpha * np.ones(n))
        arrangement = np.random.choice(["orig", "bal", "shuf"])
        if arrangement == "orig":
            cand = _anti_pair_arrangement(cand)
        elif arrangement == "bal":
            cand = _balanced_anti_pair(cand)
        else:  # shuf – keep the Dirichlet order (already random)
            np.random.shuffle(cand)   # extra shuffle for good measure

        cand = np.maximum(cand, 0.0)
        cand = cand / cand.sum()
        val0 = _objective(cand)
        print(f"[Restart {r}] init {arrangement} α={alpha:.2f} → {val0:.6f}")

        # ----- 2) allocate time for this restart --------------------------------
        # Ensure we leave enough for the final upscale (≈30 s).
        time_left = deadline - time.time() - 30.0
        if time_left <= 0:
            break
        # split the remaining time among the three phases
        adam_deadline = time.time() + min(120.0, 0.5 * time_left)
        sub_deadline = time.time() + min(60.0, 0.25 * time_left)
        local_deadline = time.time() + min(30.0, 0.15 * time_left)

        # ----- 3) Adam (smooth‑max) -------------------------------------------
        # Randomise a few hyper‑parameters to increase diversity.
        lr = np.random.uniform(0.04, 0.09)
        lam_start = np.random.uniform(0.5, 2.0)
        lam_factor = np.random.uniform(1.05, 1.12)

        a_adam, val_adam = _adam_optimize(
            cand,
            n,
            conv_fft_len,
            corr_fft_len,
            adam_deadline,
            max_iters=250_000,
            lr=lr,
            lam_start=lam_start,
            lam_factor=lam_factor,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Restart {r}] Adam improved to {val_adam:.6f}")

        # ----- 4) Sub‑gradient refinement ---------------------------------------
        eta0 = np.random.uniform(0.5, 0.9)
        a_sub, val_sub = _subgradient_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            sub_deadline,
            max_iters=200_000,
            eta0=eta0,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Restart {r}] Sub‑grad improved to {val_sub:.6f}")

        # ----- 5) Local stochastic search ---------------------------------------
        a_loc, val_loc = _local_random_search(
            best_seq,
            n,
            conv_fft_len,
            local_deadline,
            max_moves=120_000,
        )
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()
            print(f"[Restart {r}] Local search improved to {val_loc:.6f}")

        # ----- 6) Very short extra Adam (if any time left) --------------------
        if time.time() < deadline - 10.0:
            extra_deadline = min(deadline - 5.0, time.time() + 15.0)
            a_extra, val_extra = _adam_optimize(
                best_seq,
                n,
                conv_fft_len,
                corr_fft_len,
                extra_deadline,
                max_iters=50_000,
                lr=0.05,
                lam_start=5.0,
                lam_factor=1.05,
            )
            if val_extra < best_val:
                best_val = val_extra
                best_seq = a_extra.copy()
                print(f"[Restart {r}] Extra Adam improved to {val_extra:.6f}")

    # ------------------------------------------------------------------
    # Optional upscale to 8192 points if time permits
    if time.time() < deadline - 5.0:
        print("[Upscale] interpolating to 8192 points")
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        x_old = np.linspace(0.0, 1.0, n, endpoint=False)
        x_new = np.linspace(0.0, 1.0, n_big, endpoint=False)
        a_big = np.interp(x_new, x_old, best_seq)

        a_big = np.maximum(a_big, 0.0)
        a_big = a_big / a_big.sum()

        # brief Adam on the larger grid
        big_adam_deadline = min(deadline - 2.0, time.time() + 180.0)
        a_big_adam, val_big_adam = _adam_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            big_adam_deadline,
            max_iters=200_000,
            lr=0.07,
            lam_start=1.0,
            lam_factor=1.07,
        )
        # sub‑gradient refinement on the large grid
        big_sub_deadline = min(deadline - 0.5, time.time() + 200.0)
        a_big_sub, val_big_sub = _subgradient_optimize(
            a_big_adam,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            big_sub_deadline,
            max_iters=200_000,
            eta0=0.7,
        )
        if val_big_sub < best_val:
            best_val = val_big_sub
            best_seq = a_big_sub.copy()
            print(f"[Upscale] Improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Safeguard – guarantee a non‑empty valid return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the range accepted by evaluate_sequence
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
