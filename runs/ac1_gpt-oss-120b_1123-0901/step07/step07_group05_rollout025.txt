    deadline = time.time() + budget_s - 1.0

    # Determine n choose 4096 initially

    conv_fft_len = _next_pow2(2*n-1)

    # Warm start

    if warm_start is not None:
        ws = np.asarray(warm_start, dtype=np.float64)
        if ws.size != n:
            # Interpolate or trim
            if ws.size > n:
                ws = ws[:n]
            else:
                ws = np.pad(ws, (0, n-ws.size))
        ws = np.maximum(ws, 0.0)
        ws_sum = ws.sum()
        if ws_sum >= 0.01:
            ws = ws / ws_sum
        else:
            ws = None
    else:
        ws = None

    # Stage 1: generate parametric candidates
    candidates = []

    # add warm_start and its shifts
    if ws is not None:
        candidates.append(ws.copy())
        for shift in (n//4, n//2, 3*n//4):
            candidates.append(np.roll(ws, shift))

    # uniform baseline
    candidates.append(np.full(n, 1.0/n))

    # Dirichlet draws
    for conc in (0.001, 0.005, 0.01, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(6):
            cand = np.random.dirichlet(np.full(n, conc))
            candidates.append(cand)

    # Exponential shapes (left, right, symmetric)
    i = np.arange(n, dtype=np.float64)
    # left only
    for alpha in np.logspace(-2, 2, num=10): # up to 100
        left = np.exp(-alpha * i)
        left = left / left.sum()
        candidates.append(left)
        # right only
        right = np.exp(-alpha * (n-1-i))
        right = right / right.sum()
        candidates.append(right)
        # symmetric mix with uniform
        for w in (0.0, 0.2, 0.5, 0.8, 1.0):
            shape = w* (left + right) / 2 + (1-w)/n
            shape = shape / shape.sum()
            candidates.append(shape)

    # Power-law shapes
    for p in np.linspace(0.2, 8.0, num=12):
        left = (i+1.0) ** (-p)
        right = (n-i) ** (-p)
        base = left + right
        base = base / base.sum()
        # mix with uniform
        for w in (0.0, 0.3, 0.6, 1.0):
            shape = w*base + (1-w)/n
            shape = shape / shape.sum()
            candidates.append(shape)

    # Gaussian shapes
    for sigma_rel in (0.05, 0.1, 0.15, 0.2, 0.3):
        sigma = sigma_rel * n
        for mu_frac in (0.5, 0.25, 0.75):
            mu = mu_frac * (n-1)
            gauss = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
            gauss = gauss / gauss.sum()
            candidates.append(gauss)

    # Beta shapes (α,β)
    alphas = np.linspace(0.2, 5.0, num=6)
    betas = np.linspace(0.2, 5.0, num=6)
    for a in alphas:
        for b in betas:
            beta_shape = (i+1) ** (a-1) * (n-i) ** (b-1)
            beta_shape = beta_shape / beta_shape.sum()
            candidates.append(beta_shape)

    # Window functions using SciPy if available

        if s < 0.01:

        cand = cand / s

def _adam_smooth_max(a0, n, conv_fft_len, deadline, init_best_seq, init_best_val):
    corr_fft_len = _next_pow2(3*n-2)
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits); v = np.zeros_like(logits)
    beta1 = 0.9; beta2=0.999; eps_adam=1e-8
    lr = 0.05
    lam = 2.0; lam_factor = 1.012
    best_seq = init_best_seq
    best_val = init_best_val
    it=0
    last_improve = time.time()
    max_iter = 500_000
    while time.time() < deadline and it < max_iter:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()
        # compute conv and val
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Adam] iter {it} best {val:.6f}")
        # smooth max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum
        # gradient
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_log
        v = beta2 * v + (1 - beta2) * (grad_log ** 2)
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        # temperature schedule
        lam = min(lam * lam_factor, 1e6)
        # occasional shake if stuck
        if time.time() - last_improve > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()
        it += 1
    # return best_seq and best_val
    if best_seq is None:
        # fallback to a0
        return a0, 2.0 * n * _full_convolution(a0, conv_fft_len).max() / (a0.sum() ** 2)
    return best_seq, best_val

def _subgradient_refinement(a0, n, conv_fft_len, deadline, init_best_seq, init_best_val):
    corr_fft_len = _next_pow2(3*n-2)
    a = a0.copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s
    best_seq = a.copy()
    best_val = init_best_val
    eta = 0.8
    decay = 0.99995
    last_improve = time.time()
    it = 0
    max_iter = 100_000
    while time.time() < deadline and it < max_iter:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Subgrad] iter {it} best {val:.6f}")
        # active set
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        # exponentiated gradient
        a = a * np.exp(-eta * grad_a)
        # renormalize
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s
        eta *= decay
        it += 1
        if time.time() - last_improve > 6.0:
            break
    return best_seq, best_val

   cheap random moves:
   * additive Gaussian noise in log‑space,
   * a random “mass‑transfer’’ between two coordinates.
   Improvements are kept; the loop ends after a short idle period.

5. **Optional larger grid** – If a generous time margin remains,
   interpolate the best vector to `n = 8192` and repeat step 3.

All loops check the global deadline (`budget_s` seconds) and exit early.
The returned list contains only non‑negative values (≤ 1000) and respects the
evaluator’s clipping.  The whole routine fits into the 1000 s budget while
searching a much richer space than the reference implementation.
"""

import time
import numpy as np

# ------------------------------------------------------------
# Basic FFT utilities
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i = 0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact objective F = 2·n·max(a*a)/(sum(a)²)."""
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c / (a.sum() ** 2))


# ------------------------------------------------------------
# Adam optimisation on a smooth‑max surrogate
def _adam_smooth_max(a0: np.ndarray, n: int, conv_fft_len: int,
                     deadline: float,
                     init_best_seq: np.ndarray | None,
                     init_best_val: float) -> tuple[np.ndarray, float]:
    corr_fft_len = _next_pow2(3 * n - 2)
    eps = 1e-15
    logits = np.log(a0 + eps)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05                     # step‑size
    lam = 2.0                     # start temperature for soft‑max
    lam_factor = 1.012            # slower than before
    lam_cap = 1e6

    best_seq = init_best_seq
    best_val = init_best_val

    it = 0
    max_iter = 500_000
    last_improve = time.time()

    while time.time() < deadline and it < max_iter:
        # ---- soft‑max → probability vector a ----
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # ---- exact objective (used for bookkeeping) ----
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Adam] iter={it:06d} new best {val:.6f}")

        # ---- smooth‑max weights (log‑sum‑exp) ----
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---- gradient of the surrogate ----
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ---- Adam update on logits ----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---- temperature schedule ----
        lam = min(lam * lam_factor, lam_cap)

        # ---- occasional random shake if stuck ----
        if time.time() - last_improve > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1

    if best_seq is None:       # fallback (should never happen)
        best_seq = a0.copy()
        best_val = _objective(best_seq, n, conv_fft_len)
    return best_seq, best_val


# ------------------------------------------------------------
# Sub‑gradient refinement (exponentiated gradient)
def _subgradient_refinement(a0: np.ndarray, n: int, conv_fft_len: int,
                            deadline: float,
                            init_best_seq: np.ndarray | None,
                            init_best_val: float) -> tuple[np.ndarray, float]:
    corr_fft_len = _next_pow2(3 * n - 2)
    a = a0.copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a /= s                                      # stay on the simplex

    best_seq = a.copy()
    best_val = init_best_val
    eta = 0.8                                   # initial step‑size
    decay = 0.99995
    it = 0
    max_iter = 100_000
    last_improve = time.time()

    while time.time() < deadline and it < max_iter:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Subgrad] iter={it:06d} new best {val:.6f}")

        # ---- active set where the max is attained ----
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # exponentiated‑gradient update (keeps non‑negativity)
        a = a * np.exp(-eta * grad_a)

        # renormalise
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        eta *= decay
        it += 1

        if time.time() - last_improve > 6.0:
            break

    return best_seq, best_val


# ------------------------------------------------------------
def _deep_optimize(a0: np.ndarray, n: int, conv_fft_len: int,
                   deadline: float,
                   global_best_seq: np.ndarray | None,
                   global_best_val: float) -> tuple[np.ndarray, float]:
    """Run Adam → sub‑gradient on a single start."""
    best_seq, best_val = _adam_smooth_max(
        a0, n, conv_fft_len, deadline, global_best_seq, global_best_val
    )
    best_seq, best_val = _subgradient_refinement(
        best_seq, n, conv_fft_len, deadline, best_seq, best_val
    )
    return best_seq, best_val


# ------------------------------------------------------------
# Light stochastic refinement (Gaussian + mass‑transfer)
def _local_refinement(a_best: np.ndarray, n: int, conv_fft_len: int,
                      deadline: float, best_val: float) -> tuple[np.ndarray, float]:
    a = a_best.copy()
    best_seq = a.copy()
    best_obj = best_val
    last_improve = time.time()
    move_scale = 0.2                # relative amount moved between coordinates
    it = 0
    max_iter = 200_000

    while time.time() < deadline and it < max_iter:
        # 50 % chance for each perturbation type
        if np.random.rand() < 0.5:
            # Gaussian noise in log‑space
            logits = np.log(a + 1e-15)
            noise = np.random.randn(n) * 0.02
            logits += noise
            mx = logits.max()
            exp_log = np.exp(logits - mx)
            a_new = exp_log / exp_log.sum()
        else:
            # Transfer a random fraction of mass from i → j
            i, j = np.random.randint(0, n, size=2)
            if i == j or a[i] == 0.0:
                it += 1
                continue
            delta = np.random.rand() * move_scale * a[i]
            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta

        if (a_new < 0).any():
            it += 1
            continue

        val = _objective(a_new, n, conv_fft_len)
        if val < best_obj:
            best_obj = val
            best_seq = a_new.copy()
            a = a_new
            last_improve = time.time()
            print(f"[Local] iter={it:06d} improved to {val:.6f}")

        it += 1
        if time.time() - last_improve > 8.0:
            break

    return best_seq, best_obj


# ------------------------------------------------------------
def _generate_candidates(n: int, warm_start: np.ndarray | None):
    """Create a diverse pool of candidate vectors."""
    candidates = []

    # ---- warm‑start + circular shifts ----
    if warm_start is not None:
        candidates.append(warm_start.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            candidates.append(np.roll(warm_start, shift))

    # ---- uniform baseline ----
    candidates.append(np.full(n, 1.0 / n))

    i = np.arange(n, dtype=np.float64)

    # ---- Dirichlet draws (many concentrations) ----
    for conc in (0.001, 0.005, 0.01, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(6):
            cand = np.random.dirichlet(np.full(n, conc))
            candidates.append(cand)

    # ---- one‑sided exponentials (very steep) ----
    for alpha in np.logspace(-2, 2, num=12):          # 0.01 … 100
        left = np.exp(-alpha * i)
        left = left / left.sum()
        right = np.exp(-alpha * (n - 1 - i))
        right = right / right.sum()
        candidates.append(left)
        candidates.append(right)

        # symmetric mixtures with a uniform background
        for w in (0.0, 0.2, 0.5, 0.8, 1.0):
            shape = w * (left + right) / 2 + (1.0 - w) / n
            shape = shape / shape.sum()
            candidates.append(shape)

    # ---- power‑law families ----
    for p in np.linspace(0.2, 8.0, num=12):
        left = (i + 1.0) ** (-p)
        right = (n - i) ** (-p)
        base = left + right
        base /= base.sum()
        for w in (0.0, 0.3, 0.6, 1.0):
            shape = w * base + (1.0 - w) / n
            shape /= shape.sum()
            candidates.append(shape)

    # ---- Gaussian bumps (different centres & widths) ----
    for sigma_rel in (0.05, 0.1, 0.15, 0.2, 0.3):
        sigma = sigma_rel * n
        for mu_frac in (0.5, 0.25, 0.75):
            mu = mu_frac * (n - 1)
            gauss = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
            gauss = gauss / gauss.sum()
            candidates.append(gauss)

    # ---- Beta‑distribution shapes (α,β) ----
    alphas = np.linspace(0.2, 5.0, num=6)
    betas = np.linspace(0.2, 5.0, num=6)
    for a in alphas:
        for b in betas:
            beta_shape = (i + 1) ** (a - 1) * (n - i) ** (b - 1)
            beta_shape = beta_shape / beta_shape.sum()
            candidates.append(beta_shape)

    # ---- classic signal windows (requires SciPy) ----
    try:
        from scipy.signal import get_window
        win_defs = [
            ("hann", None),
            ("hamming", None),
            ("blackman", None),
            ("kaiser", [0.5, 2.0, 5.0, 8.0]),
        ]
        for name, params in win_defs:
            if params is None:
                w = get_window(name, n, fftbins=False).astype(np.float64)
                w = w / w.sum()
                candidates.append(w)
            else:
                for beta in params:
                    w = get_window((name, beta), n, fftbins=False).astype(np.float64)
                    w = w / w.sum()
                    candidates.append(w)
    except Exception:
        pass  # SciPy not available – ignore window families

    # ---- linear / triangular shapes ----
    lin = np.linspace(1.0, 0.0, n)
    lin = lin / lin.sum()
    tri = np.concatenate(
        [np.linspace(0, 1, n // 2, endpoint=False),
         np.linspace(1, 0, n - n // 2)]
    )
    tri = tri / tri.sum()
    candidates.extend([lin, tri])

    # ---- sparse spike vectors ----
    for spikes in (3, 5, 8, 12):
        for _ in range(10):
            idx = np.random.choice(n, size=spikes, replace=False)
            weights = np.random.dirichlet(np.ones(spikes))
            vec = np.zeros(n)
            vec[idx] = weights
            vec = vec / vec.sum()
            candidates.append(vec)

    return candidates


# ------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Entry point – searches for a non‑negative step function that minimises
    the supplied evaluation function.  Returns the best sequence found as
    a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 1.0          # safety margin

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # --------------------------------------------------------
    # problem size for the first stage
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)

    # --------------------------------------------------------
    # warm‑start (previous best) if it exists
    warm_start = globals().get("height_sequence_1")
    if warm_start is not None:
        ws = np.asarray(warm_start, dtype=np.float64)
        if ws.shape[0] != n:
            if ws.shape[0] > n:
                ws = ws[:n]
            else:
                ws = np.pad(ws, (0, n - ws.shape[0]))
        ws = np.maximum(ws, 0.0)
        s = ws.sum()
        if s >= 0.01:
            ws = ws / s
        else:
            ws = None
    else:
        ws = None

    # --------------------------------------------------------
    # Stage 1 – generate a large pool, keep the best few
    all_candidates = _generate_candidates(n, ws)

    top_k = 30
    best_candidates = []  # (value, vector)

    for cand in all_candidates:
        if time.time() >= deadline:
            break
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 0.01:
            continue
        cand = cand / s
        val = _objective(cand, n, conv_fft_len)
        best_candidates.append((val, cand))
        if len(best_candidates) > top_k:
            # keep only the top‑k
            best_candidates.sort(key=lambda x: x[0])
            best_candidates = best_candidates[:top_k]

    # final ordering
    best_candidates.sort(key=lambda x: x[0])
    top_candidates = [c for _, c in best_candidates[:top_k]]
    if best_candidates:
        print(
            f"Stage 1: evaluated {len(best_candidates)} candidates, "
            f"best so far = {best_candidates[0][0]:.6f}"
        )

    # --------------------------------------------------------
    # Stage 2 – deep optimisation from each promising start
    for idx, start_vec in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        print(f"--- Deep optimisation start #{idx + 1}/{len(top_candidates)} ---")
        seq_opt, val_opt = _deep_optimize(
            start_vec, n, conv_fft_len, deadline, best_seq, best_val
        )
        if val_opt < best_val:
            best_val = val_opt
            best_seq = seq_opt
            print(f"[Global] new best {best_val:.6f}")

    # --------------------------------------------------------
    # Stage 3 – cheap stochastic refinement from the overall best
    if best_seq is not None:
        print("--- Local stochastic refinement ---")
        best_seq, best_val = _local_refinement(
            best_seq, n, conv_fft_len, deadline, best_val
        )

    # --------------------------------------------------------
    # Stage 4 – if there is a comfortable time margin, go to a larger grid
    time_left = deadline - time.time()
    if time_left > 60.0:          # need a noticeable budget
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        if best_seq is not None:
            # linear interpolation onto the finer grid
            x_old = np.linspace(0, 1, n, endpoint=False)
            x_new = np.linspace(0, 1, n_big, endpoint=False)
            a_big = np.interp(x_new, x_old, best_seq)
            a_big = np.maximum(a_big, 0.0)
            a_big = a_big / a_big.sum()
        else:
            a_big = np.random.dirichlet(np.ones(n_big))
        print(f"--- Optimising on larger grid (n={n_big}) ---")
        seq_opt_big, val_opt_big = _deep_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            deadline,
            best_seq,
            best_val,
        )
        if val_opt_big < best_val:
            best_val = val_opt_big
            best_seq = seq_opt_big
            n = n_big
            print(f"[Big grid] improved best to {best_val:.6f}")

    # --------------------------------------------------------
    # Fallback – uniform vector if everything failed
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Clip to the evaluator’s allowed range and return
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    print(f"*** Finished search: best objective ≈ {best_val:.6f} (n={best_seq.shape[0]})")
    return best_seq.tolist()
