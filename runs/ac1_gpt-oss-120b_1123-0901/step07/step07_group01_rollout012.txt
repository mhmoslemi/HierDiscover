"""
Hybrid L‑BFGS + exact sub‑gradient optimizer for the step‑function problem.

The method works in three stages:

1. Initialise a log‑probability vector (logits) from a warm‑start or a random
   Gaussian.  The soft‑max of the logits yields a non‑negative vector with
   unit sum (the feasible set of the optimisation).

2. Minimise a smooth approximation of the max‑convolution using the
   log‑sum‑exp (smooth‑max) surrogate.  The surrogate is differentiable,
   and its gradient is obtained analytically via FFT‑based cross‑correlation.
   A continuation schedule slowly raises the temperature λ, turning the
   surrogate into the true max.  Each stage is solved with SciPy's L‑BFGS‑B
   (limited‑memory quasi‑Newton) which converges in a few hundred gradient
   evaluations.

3. Refine the best candidate with an exact max‑sub‑gradient (exponentiated‑
   gradient) descent for a short time budget.

All steps respect the given time limit and always return a valid sequence.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via FFT (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for the gradient."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _softmax(logits: np.ndarray) -> np.ndarray:
    """Numerically stable softmax, returns a probability vector."""
    m = np.max(logits)
    e = np.exp(logits - m)
    return e / e.sum()


def _smooth_max_objective(logits: np.ndarray,
                          lam: float,
                          n: int,
                          conv_len: int,
                          corr_len: int):
    """
    Returns (value, gradient) of the smooth‑max surrogate:

        f(logits) = (1/lam) * log Σ exp(lam * (a*a)),
        where a = softmax(logits).

    Gradient is computed analytically via FFT‑based cross‑correlation.
    """
    a = _softmax(logits)                     # a_i ≥ 0, Σ a_i = 1
    conv = _full_convolution(a, conv_len)    # a*a
    lc = lam * conv
    m = np.max(lc)
    exp_shift = np.exp(lc - m)               # stable
    Z = np.sum(exp_shift)
    lse = m + np.log(Z)                      # log‑sum‑exp
    value = lse / lam                        # smooth approximation of max(conv)

    # softmax weights for the surrogate gradient
    w = exp_shift / Z

    # gradient d f / d a
    grad_a = 2.0 * _cross_correlation(w, a, corr_len)

    # back‑prop through softmax (logits → a)
    avg = np.dot(a, grad_a)
    grad_log = a * (grad_a - avg)
    return value, grad_log


def _lbfgs_optimize(logits_init: np.ndarray,
                    lam: float,
                    n: int,
                    conv_len: int,
                    corr_len: int,
                    max_iter: int,
                    deadline: float):
    """
    L‑BFGS‑B minimisation of the smooth‑max surrogate for a fixed λ.
    Returns the final logits and the surrogate value.
    """
    def fun(x):
        val, grad = _smooth_max_objective(x, lam, n, conv_len, corr_len)
        return val, grad

    # SciPy can accept a function returning (value, grad) when jac=True.
    res = minimize(fun,
                   logits_init,
                   method='L-BFGS-B',
                   jac=True,
                   options={'maxiter': max_iter, 'disp': False})
    return res.x, res.fun


def _subgradient_optimize(a0: np.ndarray,
                          n: int,
                          conv_fft_len: int,
                          corr_fft_len: int,
                          deadline: float,
                          max_iters: int = 200_000,
                          eta0: float = 0.6) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient descent (exponentiated‑gradient updates).
    Returns the best vector found and its raw objective 2·n·max(a*a).
    """
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s

    best_a = a.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
        # active set – points that attain the max
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1)
        a = a * np.exp(-eta * grad_a)

        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s
        it += 1
    return best_a, best_val
