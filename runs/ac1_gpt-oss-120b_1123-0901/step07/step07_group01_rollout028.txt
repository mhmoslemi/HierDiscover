"""
Hybrid search for step‑function minimizing 2·n·max(a*a)/(sum a)^2.

The algorithm consists of:
  1. A rich parametric pool (Dirichlet, power‑laws, exponentials,
     Beta‑family, double‑exponential …).
  2. A fast “smooth‑max” Adam optimisation (log‑sum‑exp) with a
     slowly increasing temperature.
  3. Exact‑max sub‑gradient descent (multiplicative updates).
  4. A long simulated‑annealing phase that updates the convolution
     in O(n) per elementary mass‑transfer.
  5.  Optional up‑sampling to a larger grid and a short second pass.
All heavy convolution work is performed with FFTs and the routine
always respects the 1000 s time budget.
"""

import time, math, numpy as np

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()

def _fft_convolve(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via FFT (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]

def _fft_cross_corr(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Cross‑correlation w ★ a needed for the gradient."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]

def _make_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Create a diverse set of starting vectors."""
    cand = []

    # warm start and rotations
    if warm is not None:
        cand.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    # Dirichlet with various concentrations
    for a in [0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]:
        cand.append(np.random.dirichlet(np.full(n, a)))

    # Power‑law decreasing (i+1)^‑p
    for p in np.linspace(0.5, 3.0, 6):
        idx = np.arange(1, n + 1, dtype=np.float64)
        shape = idx ** (-p)
        shape /= shape.sum()
        cand.append(shape)

    # Exponential tails
    for gamma in np.linspace(1e-4, 1e-2, 6):
        shape = np.exp(-gamma * np.arange(n, dtype=np.float64))
        shape /= shape.sum()
        cand.append(shape)

    # Beta‑family a_i ∝ (i+1)^{α‑1} (n‑i)^{β‑1}
    alphas = [0.5, 0.8, 1.2, 2.0]
    betas  = [0.5, 0.8, 1.2, 2.0]
    for α in alphas:
        for β in betas:
            i = np.arange(n, dtype=np.float64)
            shape = (i + 1) ** (α - 1) * (n - i) ** (β - 1)
            shape = np.where(shape > 0, shape, 0.0)
            if shape.sum() > 0:
                shape /= shape.sum()
                cand.append(shape)

    # Double‑exponential (two edges)
    for γ in (0.2, 0.5, 1.0):
        i = np.arange(n, dtype=np.float64)
        shape = np.exp(-γ * i) + np.exp(-γ * (n - 1 - i))
        shape /= shape.sum()
        cand.append(shape)

    # Half‑sine
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand.append(shape)

    # Uniform
    cand.append(np.full(n, 1.0 / n))

    # Random spikes (k positions)
    for k in (64, 128, 256, 512, 1024):
        a = np.zeros(n, dtype=np.float64)
        pos = np.round(np.linspace(0, n - 1, k)).astype(int)
        a[pos] = 1.0 / k
        cand.append(a)

    # Miscellaneous mixed patterns
    # a slowly decreasing power‑law combined with a tiny Gaussian bump
    for p in [1.0, 1.5]:
        idx = np.arange(1, n + 1, dtype=np.float64)
        base = idx ** (-p)
        bump = np.exp(-((idx - n/8) ** 2) / (2 * (n/20) ** 2))
        shape = base + 0.05 * bump
        shape /= shape.sum()
        cand.append(shape)

    return cand

def _adam_optimize(a0: np.ndarray, n: int,
                   conv_fft_len: int, corr_fft_len: int,
                   deadline: float,
                   max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """Adam on the log‑sum‑exp (smooth‑max) surrogate."""
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.02                 # a bit more conservative
    lam = 0.1                 # start temperature (smooth‑max)
    lam_factor = 1.01         # very slow anneal

    best_a = a0.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        # softmax to recover a from logits
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _fft_convolve(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            # report only every few iterations to keep output short
            if it % 2000 == 0:
                print(f"[Adam it={it:06d}] best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of surrogate
        grad_a = 2.0 * _fft_cross_corr(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        lam = min(lam * lam_factor, 1e6)

        # occasional jitter
        if it % 5000 == 0:
            logits += (np.random.rand(n) - 0.5) * 1e-4

        it += 1

    return best_a, best_val

def _subgradient_optimize(a0: np.ndarray, n: int,
                          conv_fft_len: int, corr_fft_len: int,
                          deadline: float,
                          max_iters: int = 300_000,
                          eta0: float = 0.5) -> tuple[np.ndarray, float]:
    """Exact‑max sub‑gradient descent (multiplicative updates)."""
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s

    best_a = a.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _fft_convolve(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 2000 == 0:
                print(f"[Subgrad it={it:06d}] best {val:.6f}")

        # active set (those convolutions achieving the max)
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _fft_cross_corr(w, a, corr_fft_len)

        # diminishing step‑size
        eta = eta0 / np.sqrt(it + 1)

        # exponentiated‑gradient update
        a = a * np.exp(-eta * grad_a)

        # renormalise onto the simplex
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val

def _simulated_annealing(a0: np.ndarray, n: int,
                         fft_len: int,
                         deadline: float,
                         cur_best_val: float) -> tuple[np.ndarray, float]:
    """Fast SA that updates the convolution in O(n) per elementary move."""
    a = a0.copy()
    conv = _fft_convolve(a, fft_len)
    cur_max = conv.max()
    cur_val = 2.0 * n * cur_max
    best_a = a.copy()
    best_val = cur_val

    start = time.time()
    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / (deadline - start + 1e-12)

        # linear cooling (0.2 → 1e‑6)
        T = max(1e-6, 0.2 * (1.0 - frac))
        move_scale = max(0.005, 0.3 * (1.0 - frac))

        # source proportional to its mass
        if a.sum() == 0:
            break
        i = np.random.choice(n, p=a / a.sum())
        if a[i] < 1e-14:
            continue
        # target uniformly random distinct index
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * (np.random.rand() * move_scale)

        a_new = a.copy()
        conv_new = conv.copy()
        a_new[i] -= delta
        a_new[j] += delta

        # update convolution efficiently
        conv_new[i : i + n] -= 2.0 * delta * a
        conv_new[j : j + n] += 2.0 * delta * a
        conv_new[2 * i] += delta * delta
        conv_new[2 * j] += delta * delta
        conv_new[i + j] -= 2.0 * delta * delta

        new_max = conv_new.max()
        new_val = 2.0 * n * new_max

        # Metropolis acceptance
        if new_val < cur_val or np.random.rand() < np.exp(-(new_val - cur_val) / T):
            a, conv = a_new, conv_new
            cur_val = new_val
            cur_max = new_max
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[SA it={it:07d}] new best {best_val:.6f}")

        it += 1
        if it % 10000 == 0:
            print(f"[SA it={it:07d}] best {best_val:.6f}")

    return best_a, best_val

def _interpolate_and_refine(a_small: np.ndarray,
                            n_big: int,
                            deadline: float) -> tuple[np.ndarray, float]:
    """Upscale to a larger grid and run a brief Adam + sub‑gradient pass."""
    n_small = a_small.shape[0]
    x_small = np.linspace(0.0, 1.0, n_small, endpoint=False)
    x_big = np.linspace(0.0, 1.0, n_big, endpoint=False)
    a_big = np.interp(x_big, x_small, a_small)
    a_big = np.maximum(a_big, 0.0)
    a_big /= a_big.sum()

    conv_fft_len_big = _next_pow2(2 * n_big - 1)
    corr_fft_len_big = _next_pow2(3 * n_big - 2)

    a_adam, _ = _adam_optimize(a_big, n_big,
                               conv_fft_len_big, corr_fft_len_big,
                               deadline, max_iters=80_000)
    a_ref, val_ref = _subgradient_optimize(a_adam, n_big,
                                           conv_fft_len_big,
                                           corr_fft_len_big,
                                           deadline, max_iters=100_000)
    return a_ref, val_ref

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:

    annealing pass.  If time permits the best solution is up‑scaled

    """

"""
Hybrid search for a step‑function that minimises

    F(a) = 2·n·max(a∗a) / (sum a)^2

The routine works in three phases:

1️⃣  **Diverse warm‑starts** – Dirichlet draws, power‑law, exponential,
    Beta‑family, double‑edge Gaussians, half‑sine, uniform and a few
    spike patterns (including the previous best if available).

2️⃣  **Intensive optimisation** – each candidate receives a short
    Adam run on the smooth‑max (log‑sum‑exp) surrogate, while the
    current global best is refined with a long exact‑max sub‑gradient
    descent followed by a prolonged simulated‑annealing walk that
    updates the convolution in O(n) per elementary mass‑transfer.

3️⃣  **Optional upscale** – if time permits the best solution is
    interpolated to a longer grid (8192 points) and a brief Adam + sub‑gradient
    pass is performed again.

All heavy work uses FFT‑based convolutions for speed and the whole
search respects the 1000 s time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _fft_convolve(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via FFT (length 2·len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _fft_cross_corr(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation w ★ a needed for the gradient."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
def _make_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Create a rich pool of start vectors."""
    cand = []

    # warm start and several rotations
    if warm is not None:
        cand.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    # Dirichlet draws with several concentrations
    for a in [0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]:
        cand.append(np.random.dirichlet(np.full(n, a)))

    # Power‑law decreasing shapes (i+1)^‑p
    for p in np.linspace(0.5, 3.0, 6):
        idx = np.arange(1, n + 1, dtype=np.float64)
        shape = idx ** (-p)
        shape /= shape.sum()
        cand.append(shape)

    # Exponential tails
    for gamma in np.linspace(1e-4, 1e-2, 6):
        shape = np.exp(-gamma * np.arange(n, dtype=np.float64))
        shape /= shape.sum()
        cand.append(shape)

    # Beta‑family: a_i ∝ (i+1)^{α‑1} (n‑i)^{β‑1}
    alphas = [0.5, 0.8, 1.2, 2.0]
    betas  = [0.5, 0.8, 1.2, 2.0]
    for α in alphas:
        for β in betas:
            i = np.arange(n, dtype=np.float64)
            shape = (i + 1) ** (α - 1) * (n - i) ** (β - 1)
            shape = np.where(shape > 0, shape, 0.0)
            if shape.sum() > 0:
                shape /= shape.sum()
                cand.append(shape)

    # Two‑edge Gaussian‑like shapes
    for γ in (0.2, 0.5, 1.0):
        i = np.arange(n, dtype=np.float64)
        shape = np.exp(-γ * i) + np.exp(-γ * (n - 1 - i))
        shape /= shape.sum()
        cand.append(shape)

    # Half‑sine (positive half‑wave)
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand.append(shape)

    # Uniform
    cand.append(np.full(n, 1.0 / n))

    # Spike families (k equally spaced non‑zero entries)
    for k in (64, 128, 256, 512, 1024):
        a = np.zeros(n, dtype=np.float64)
        pos = np.round(np.linspace(0, n - 1, k)).astype(int)
        a[pos] = 1.0 / k
        cand.append(a)

    # Mixed power‑law + tiny Gaussian bump (adds a little local curvature)
    for p in [1.0, 1.5]:
        idx = np.arange(1, n + 1, dtype=np.float64)
        base = idx ** (-p)
        bump = np.exp(-((idx - n / 8) ** 2) / (2 * (n / 20) ** 2))
        shape = base + 0.05 * bump
        shape /= shape.sum()
        cand.append(shape)

    return cand


# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   conv_fft_len: int,
                   corr_fft_len: int,
                   deadline: float,
                   max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """Adam on the log‑sum‑exp (smooth‑max) surrogate."""
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.02                # modest learning rate
    lam = 0.1                # initial temperature for smooth‑max
    lam_factor = 1.01        # very slow anneal

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        # recover probabilities from logits
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _fft_convolve(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 2000 == 0:
                print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max (log‑sum‑exp) weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of surrogate (chain rule)
        grad_a = 2.0 * _fft_cross_corr(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        lam = min(lam * lam_factor, 1e6)

        # occasional tiny jitter to avoid stagnation
        if it % 5000 == 0:
            logits += (np.random.rand(n) - 0.5) * 1e-4

        it += 1

    return best_a, best_val


def _subgradient_optimize(a0: np.ndarray,
                          n: int,
                          conv_fft_len: int,
                          corr_fft_len: int,
                          deadline: float,
                          max_iters: int = 300_000,
                          eta0: float = 0.5) -> tuple[np.ndarray, float]:
    """Exact‑max sub‑gradient descent with exponentiated‑gradient updates."""
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _fft_convolve(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 2000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active set (indices attaining the maximum)
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _fft_cross_corr(w, a, corr_fft_len)

        # diminishing step size
        eta = eta0 / np.sqrt(it + 1)

        # multiplicative (exponentiated‑gradient) update
        a = a * np.exp(-eta * grad_a)

        # re‑normalise onto the simplex
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val

def _simulated_annealing(a0: np.ndarray,
                         n: int,
                         fft_len: int,
                         deadline: float,
                         cur_best_val: float) -> tuple[np.ndarray, float]:
    """Fast SA that updates the convolution in O(n) per elementary move."""
    a = a0.copy()
    conv = _fft_convolve(a, fft_len)
    cur_val = 2.0 * n * conv.max()
    best_a = a.copy()
    best_val = cur_val

    start = time.time()
    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / (deadline - start + 1e-12)

        # linear cooling schedule (0.2 → 1e‑6)
        T = max(1e-6, 0.2 * (1.0 - frac))
        move_scale = max(0.005, 0.3 * (1.0 - frac))

        # choose source proportional to its mass
        if a.sum() == 0:
            break
        i = np.random.choice(n, p=a / a.sum())
        if a[i] < 1e-14:
            continue
        # pick a different target uniformly
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * (np.random.rand() * move_scale)

        a_new = a.copy()
        conv_new = conv.copy()
        a_new[i] -= delta
        a_new[j] += delta

        # update convolution using the *old* a
        conv_new[i : i + n] -= 2.0 * delta * a
        conv_new[j : j + n] += 2.0 * delta * a
        conv_new[2 * i] += delta * delta
        conv_new[2 * j] += delta * delta
        conv_new[i + j] -= 2.0 * delta * delta

        new_val = 2.0 * n * conv_new.max()

        # Metropolis acceptance
        if new_val < cur_val or np.random.rand() < np.exp(-(new_val - cur_val) / T):
            a, conv = a_new, conv_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[SA it={it:07d}] new best {best_val:.6f}")

        it += 1
        if it % 10000 == 0:
            print(f"[SA it={it:07d}] best {best_val:.6f}")

    return best_a, best_val


def _interpolate_and_refine(a_small: np.ndarray,
                            n_big: int,
                            deadline: float) -> tuple[np.ndarray, float]:
    """Upscale to a larger grid and perform a short Adam + sub‑gradient pass."""
    n_small = a_small.shape[0]
    x_small = np.linspace(0.0, 1.0, n_small, endpoint=False)
    x_big = np.linspace(0.0, 1.0, n_big, endpoint=False)
    a_big = np.interp(x_big, x_small, a_small)
    a_big = np.maximum(a_big, 0.0)
    a_big /= a_big.sum()

    conv_fft_len_big = _next_pow2(2 * n_big - 1)
    corr_fft_len_big = _next_pow2(3 * n_big - 2)

    a_adam, _ = _adam_optimize(a_big, n_big,
                               conv_fft_len_big, corr_fft_len_big,
                               deadline, max_iters=80_000)

                                           conv_fft_len_big,
                                           corr_fft_len_big,
                                           deadline, max_iters=100_000)
    return a_ref, val_ref

        warm = np.maximum(warm, 0.0)

        else:

    else:

    best_seq = None

    for cid, cand in enumerate(candidates):

        else:

        # give each candidate a few seconds of Adam optimisation

                                          conv_fft_len, corr_fft_len,
