"""
Propose a step‑function (non‑negative vector) that minimises
    2·n·max(conv(p,p))     with   sum(p)=1 .
The code runs a multi‑stage global search:

1.  Warm‑start from any previously found sequence (global variable
    ``height_sequence_1``) and a small set of hand‑crafted parametric
    families (uniform, beta‑shapes, asymmetric exponentials, Dirichlet draws).

2.  For every start run a *deep optimisation* consisting of:
        – Adam optimisation on a smooth‑max surrogate of the maximal
          convolution value.
        – Exponentiated‑gradient sub‑gradient descent on the exact
          max‑convolution.
   Hyper‑parameters are deliberately generous; the optimiser stops
   early if the deadline is reached.

3.  Repeated short random restarts (random Dirichlet initialisations)
   with a lightweight optimisation budget.  This helps to escape
   local minima.

4.  A final long refinement on the current best vector.

All convolutions and cross‑correlations are performed by FFTs for speed.
The algorithm respects the evaluator’s clipping (non‑negative, ≤1000) and
returns the best sequence seen before the 1000 s budget expires.
"""

import time
import math
import numpy as np
from typing import Tuple, Optional

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


def _objective(p: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact objective = 2·n·max(conv(p,p)) (p already normalised)."""
    conv = _full_convolution(p, conv_fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c)

def _adam_phase(
    p0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iter: int,
    init_best_seq: Optional[np.ndarray],
    init_best_val: float,
) -> Tuple[np.ndarray, float]:
    """
    Adam on a log‑sum‑exp surrogate of the max‑convolution.
    Returns the best sequence seen in this phase and its exact objective.
    """
    eps = 1e-15
    logits = np.log(p0 + eps)                 # always stay on the simplex
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05

    lam = 2.0                # start temperature
    lam_factor = 1.012
    lam_cap = 1e6

    best_seq = init_best_seq
    best_val = init_best_val

    it = 0
    last_improve = time.time()

    while it < max_iter and time.time() < deadline:
        # ----- soft‑max to obtain a probability vector -----
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        p = exp_log / exp_log.sum()

        # ----- exact objective for bookkeeping -----
        val = _objective(p, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_seq = p.copy()
            last_improve = time.time()
            print(f"[Adam] iter={it:06d} new best {val:.6f}")

        # ----- smooth‑max weights (log‑sum‑exp) -----
        conv = _full_convolution(p, conv_fft_len)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the surrogate w.r.t. logits -----
        grad_a = 2.0 * _cross_correlation(w, p, corr_fft_len)
        avg = np.dot(p, grad_a)                # <p, ∇f(p)>
        grad_log = p * (grad_a - avg)          # chain rule for soft‑max

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- temperature schedule -----
        lam = min(lam * lam_factor, lam_cap)

        # occasional jitter if we seem stuck
        if time.time() - last_improve > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1

    return best_seq, best_val


def _subgrad_phase(
    p0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iter: int,
    init_best_seq: Optional[np.ndarray],
    init_best_val: float,
) -> Tuple[np.ndarray, float]:

    Exponentiated‑gradient sub‑gradient steps on the *exact* max‑convolution.
    Returns the best vector encountered.
    """
    p = init_best_seq if init_best_seq is not None else p0.copy()
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-12:
        p = np.random.dirichlet(np.ones(n))
        s = p.sum()
    p /= s

    eta0 = 0.8
    decay = 0.99995
    eta = eta0

    best_seq = p.copy()
    best_val = init_best_val
    it = 0
    last_improve = time.time()

    while it < max_iter and time.time() < deadline:
        conv = _full_convolution(p, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_seq = p.copy()
            last_improve = time.time()
            print(f"[Subgrad] iter={it:06d} new best {val:.6f}")

        # active set where the convolution attains its maximum
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad = 2.0 * _cross_correlation(w, p, corr_fft_len)

        # exponentiated‑gradient update (keeps p ≥ 0)
        p = p * np.exp(-eta * grad)

        # renormalise to stay on simplex
        s = p.sum()
        if s < 1e-12:
            p = np.random.dirichlet(np.ones(n))
            s = p.sum()
        p /= s

        eta *= decay
        it += 1

        if time.time() - last_improve > 6.0:
            break

    return best_seq, best_val


def _deep_optimize(
    p0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: Optional[np.ndarray],
    global_best_val: float,
    adam_iters: int = 300_000,
    subgrad_iters: int = 80_000,
) -> Tuple[np.ndarray, float]:
    """
    Run Adam → Sub‑gradient on a given start.  Updates the global best.
    """
    best_seq, best_val = _adam_phase(
        p0, n, conv_fft_len, corr_fft_len, deadline,
        adam_iters, global_best_seq, global_best_val,
    )
    best_seq, best_val = _subgrad_phase(
        p0, n, conv_fft_len, corr_fft_len, deadline,
        subgrad_iters, best_seq, best_val,
    )
    return best_seq, best_val

def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Symmetric beta‑like shape on the discrete grid."""
    i = np.arange(n, dtype=np.float64) + 0.5
    shape = (i ** (alpha - 1.0)) * ((n + 0.5 - i) ** (beta - 1.0))
    shape = np.maximum(shape, 0.0)
    total = shape.sum()
    if total < 1e-12:
        shape = np.full(n, 1.0 / n)
    else:
        shape /= total
    return shape


def _asym_exponential(n: int) -> np.ndarray:
    """Random asymmetric two‑sided exponential mixture."""
    i = np.arange(n, dtype=np.float64)
    aL = 10 ** np.random.uniform(-2, 1)
    aR = 10 ** np.random.uniform(-2, 1)
    left = np.exp(-aL * i)
    right = np.exp(-aR * (n - 1 - i))
    wL = np.random.rand()
    wR = np.random.rand() * (1.0 - wL)
    wU = 1.0 - wL - wR
    shape = wL * left + wR * right + wU / n
    shape /= shape.sum()
    return shape


def _random_dirichlet(n: int) -> np.ndarray:
    """Dirichlet draw with a random concentration."""
    conc = 10 ** np.random.uniform(-2, 1)   # between 0.01 and 10
    return np.random.dirichlet(np.full(n, conc))


def _generate_initial_seeds(n: int) -> list[np.ndarray]:
    """Collect a modest but diverse pool of starting vectors."""
    seeds: list[np.ndarray] = []

    # warm‑start from previous run if present
    warm = globals().get("height_sequence_1")
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] != n:
            if w.shape[0] > n:
                w = w[:n]
            else:
                w = np.pad(w, (0, n - w.shape[0]))
        w = np.maximum(w, 0.0)
        s = w.sum()
        if s >= 0.01:
            w = w / s
            seeds.append(w)

    # uniform baseline
    seeds.append(np.full(n, 1.0 / n))

    # a few handcrafted parametric families
    for _ in range(4):
        # random beta‑like shapes (α,β ∈ [0.2,6])
        α = np.random.uniform(0.2, 6.0)
        β = np.random.uniform(0.2, 6.0)
        seeds.append(_beta_shape(n, α, β))

    for _ in range(4):
        seeds.append(_asym_exponential(n))

    # random Dirichlet draws with a few fixed concentrations
    for conc in (0.02, 0.1, 0.5, 1.0, 2.0):
        for _ in range(2):
            seeds.append(np.random.dirichlet(np.full(n, conc)))

    # some random sparse spikes (3‑5 non‑zeros)
    for spikes in (3, 5):
        for _ in range(3):
            idx = np.random.choice(n, size=spikes, replace=False)
            weights = np.random.dirichlet(np.ones(spikes))
            vec = np.zeros(n)
            vec[idx] = weights
            vec /= vec.sum()
            seeds.append(vec)

    return seeds

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Entry point for the challenge.  Performs a time‑aware multi‑stage search
    and returns the best non‑negative step function (plain Python list).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # problem size – 4096 gave the previous best
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    best_seq: Optional[np.ndarray] = None
    best_val: float = float("inf")

    # ------------------------------------------------------------------
    # Stage 1 – deterministic seeds (warm start + a small parametric pool)
    seeds = _generate_initial_seeds(n)

    print(f"[Stage 1] evaluating {len(seeds)} deterministic seeds")
    for idx, start_vec in enumerate(seeds):
        if time.time() >= deadline:
            break
        print(f"--- Deep optimisation on seed #{idx + 1}/{len(seeds)} ---")
        best_seq, best_val = _deep_optimize(
            start_vec,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            best_seq,
            best_val,
            adam_iters=250_000,
            subgrad_iters=60_000,
        )

    # ------------------------------------------------------------------
    # Stage 2 – cheap random restarts (many short optimisations)
    restart_counter = 0
    while time.time() < deadline - 8.0:   # keep a few seconds for final polish
        restart_counter += 1
        cand = _random_dirichlet(n)
        # a lightweight run: 20 k Adam + 5 k sub‑grad
        cand_seq, cand_val = _deep_optimize(
            cand,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            best_seq,
            best_val,
            adam_iters=20_000,
            subgrad_iters=5_000,
        )
        if cand_val < best_val:
            best_val = cand_val
            best_seq = cand_seq
            print(f"[Restart #{restart_counter}] new global best {cand_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – final long refinement on the current best
    if best_seq is not None and time.time() < deadline - 2.0:
        print("--- Final long refinement on best vector ---")
        best_seq, best_val = _deep_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            best_seq,
            best_val,
            adam_iters=350_000,
            subgrad_iters=80_000,
        )

    # ------------------------------------------------------------------
    # Fallback – if nothing succeeded, return the uniform vector.
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to evaluator’s allowed range (non‑negative, ≤ 1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"*** Finished search: best objective ≈ {best_val:.6f} (n={best_seq.shape[0]})")
    return best_seq.tolist()
