def generate_candidates(n, warm_start=None):
    candidates = []
    # uniform
    candidates.append(np.full(n, 1.0/n))
    # Dirichlet draws
    for conc in [0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]:
        for _ in range(2): # two draws each
            candidates.append(np.random.dirichlet(np.full(n, conc)))
    # Edge-exponential + uniform mix
    i = np.arange(n)
    for alpha in np.logspace(-2, 1.0, 12):
        edge = np.exp(-alpha*i) + np.exp(-alpha*(n-1-i))
        edge = edge / edge.sum()
        for w in np.linspace(0, 1, 5):

"""
Multi‑stage optimisation for the “max‑auto‑correlation” step‑function problem.

The search proceeds as follows (all work respects a 1000 s wall‑clock limit).

1️⃣  **Parametric seeding (n = 8192).**  A diverse pool of symmetric
    analytic shapes (uniform, beta‑type, exponential‑edge,
    power‑law‑edge, raised‑cosine, arcsine, triangular) together with a few
    random Dirichlet draws and a possible warm‑start are generated,
    normalised to sum = 1 and evaluated with a fast FFT‑based convolution.
    The best few seeds are retained.

2️⃣  **Continuous refinement.**  Each retained seed is optimised by two
    phases:
        • *Adam on a smooth‑max surrogate* – the true max in the objective
          is replaced by a log‑sum‑exp with a temperature that grows
          exponentially (λ ← λ·1.05).  The sequence lives in the soft‑max
          (log‑probability) space, guaranteeing non‑negativity and the
          simplex constraint.
        • *Exact sub‑gradient refinement* – the true non‑smooth max is
          tackled with an exponentiated‑gradient step (multiplicative
          update) and a decaying step‑size.  Both phases stop as soon as
          the global deadline is reached.

3️⃣  **Local random hill‑climbing.**  If time permits the current best
    vector is further improved by many tiny random mass‑transfer moves
    (pick two indices, move a few percent of the smaller mass while keeping
    the sum constant).  Improvements are kept, otherwise the move is
    reverted.

All heavy operations use real FFTs (`np.fft.rfft`) and are implemented in
pure NumPy for speed.  The algorithm keeps track of the globally best
sequence/value and finally returns the best sequence found (clipped to the
evaluator’s allowed range).

The routine can be called repeatedly – it respects a provided random seed
and a per‑run time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a (length 2·len(a)‑1) using a real FFT of length
    ``fft_len`` (must be ≥ 2·len(a)‑1).  Returns only the valid part.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i = 0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """
    Fast evaluation of   2·n·max(a*a) / (sum(a)²)   for a already normalised.
    """
    conv = _full_convolution(a, conv_fft_len)
    return float(2.0 * n * conv.max() / (a.sum() ** 2))


# ----------------------------------------------------------------------
def _adam_smooth_max(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on a log‑sum‑exp surrogate of the max‑convolution.
    Operates in the soft‑max (log‑probability) space – this inherently keeps
    the iterates on the probability simplex and non‑negative.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)          # start in log‑soft‑max space
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr = 0.07                         # a bit aggressive for larger n
    lam = 0.4                         # initial temperature (smooth‑max)
    lam_factor = 1.05                 # slower growth – more fine‑grained steps
    lam_cap = 1e6

    best_seq = global_best_seq
    best_val = global_best_val
    last_improve = time.time()
    it = 0

    # Iteration bound is generous – the real stopper is the deadline.
    while time.time() < deadline:
        # ---- simplex via soft‑max (numerically stable) ----
        mx = logits.max()
        exp_shifted = np.exp(logits - mx)
        a = exp_shifted / exp_shifted.sum()

        # ---- true objective (for logging) ----
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[{n}] Adam it={it:07d}  new best {val:.6f}")

        # ---- smooth‑max weights (log‑sum‑exp) ----
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---- gradient of the surrogate (with respect to a) ----
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule for logits (soft‑max)
        avg = np.dot(a, grad_a)               # ⟨a,∇⟩ term
        grad_log = a * (grad_a - avg)

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---- temperature schedule ----
        lam = min(lam * lam_factor, lam_cap)

        # ---- jitter if stuck for a while ----
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1

    return best_seq, best_val


def _subgradient_refinement(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    init_best_seq: np.ndarray | None,
    init_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient descent on the true max‑convolution.
    Keeps the iterate on the simplex by construction.
    """
    a = init_best_seq if init_best_seq is not None else a0.copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    else:
        a /= s

    best_seq = a.copy()
    best_val = init_best_val
    eta0 = 0.6
    it = 0
    last_improve = time.time()

    while time.time() < deadline:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[{n}] Subgrad it={it:05d}  new best {val:.6f}")

        # active set (indices attaining the maximum)
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # decaying step‑size
        eta = eta0 / np.sqrt(it + 1)

        # multiplicative update (exponentiated gradient)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:                     # safety – re‑initialise if collapse
            a = np.random.dirichlet(np.ones(n))
        else:
            a /= s

        if time.time() - last_improve > 6.0:
            # no progress for a while → stop early
            break

        it += 1

    return best_seq, best_val


def _deep_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Run the Adam‑smooth‑max phase followed by the sub‑gradient phase.
    Returns the best sequence/value discovered in this call.
    """
    best_seq, best_val = _adam_smooth_max(
        a0,
        n,
        conv_fft_len,
        corr_fft_len,
        deadline,
        global_best_seq,
        global_best_val,
    )
    best_seq, best_val = _subgradient_refinement(
        a0,               # note: we start sub‑grad from the *original* seed,
        n,                # not from the Adam result – this mimics the earlier script.
        conv_fft_len,
        corr_fft_len,
        deadline,
        best_seq,
        best_val,
    )
    return best_seq, best_val

def _generate_candidates(n: int, warm_start: np.ndarray | None) -> list[np.ndarray]:
    """
    Produce a rich yet still manageable set of symmetric candidates.
    Every vector is normalised to have sum ≈ 1.
    """
    cand: list[np.ndarray] = []
    i = np.arange(n, dtype=np.float64)

    # ----- warm‑start and its circular shifts -----
    if warm_start is not None:
        ws = warm_start.copy()
        if ws.shape[0] != n:
            # linear interpolation onto the current grid
            x_old = np.linspace(0, 1, ws.shape[0], endpoint=False)
            x_new = np.linspace(0, 1, n, endpoint=False)
            ws = np.interp(x_new, x_old, ws)
        ws = np.clip(ws, 0.0, None)
        if ws.sum() >= 0.01:
            ws = ws / ws.sum()
            cand.append(ws)
            for shift in (n // 2, n // 3, 2 * n // 3):
                cand.append(np.roll(ws, shift))

    # ----- uniform baseline -----
    cand.append(np.full(n, 1.0 / n))

    # ----- random Dirichlet draws (various concentrations) -----
    for conc in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(2):
            cand.append(np.random.dirichlet(np.full(n, conc)))

    # ----- edge‑exponential families with uniform mixing -----
    for alpha in np.logspace(-2, 1.0, num=12):          # 0.01 … 10
        edge = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
        edge /= edge.sum()
        for w in np.linspace(0.0, 1.0, 5):
            cand.append(w * edge + (1.0 - w) / n)

    # ----- edge‑power‑law families -----
    for p in np.linspace(0.2, 5.0, num=12):
        edge = (i + 1.0) ** (-p) + (n - i) ** (-p)
        edge /= edge.sum()
        for w in (0.0, 0.3, 0.6, 1.0):
            cand.append(w * edge + (1.0 - w) / n)

    # ----- symmetric beta‑shapes (a_i ∝ (i+1)^{α‑1}(n‑i)^{α‑1}) -----
    for alpha in np.linspace(0.2, 2.5, num=12):
        shape = (i + 1.0) ** (alpha - 1.0) * (n - i) ** (alpha - 1.0)
        shape = np.maximum(shape, 0.0)
        shape /= shape.sum()
        cand.append(shape)

    # ----- raised‑cosine / sine‑power families -----
    for p in (1, 2, 4, 6, 8):
        shape = np.sin(np.pi * (i + 0.5) / n) ** p
        shape = np.maximum(shape, 0.0)
        shape /= shape.sum()
        cand.append(shape)

    # ----- two‑Gaussian edge shapes -----
    for gamma in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        shape /= shape.sum()
        cand.append(shape)

    # ----- arcsine‑type shape a_i ∝ 1 / sqrt((i+0.5)*(n‑i‑0.5)) -----
    shape = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    shape /= shape.sum()
    cand.append(shape)

    # ----- simple triangular shape -----
    tri = np.concatenate([np.linspace(0, 1, n // 2, endpoint=False),
                          np.linspace(1, 0, n - n // 2, endpoint=False)])
    tri = np.maximum(tri, 0.0)
    tri /= tri.sum()
    cand.append(tri)

    return cand


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Entry point for the competition.  Returns a list of non‑negative
    heights (floats) defining the step‑function.  The routine runs for at
    most ``budget_s`` seconds and prints occasional progress information.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # Choose a grid size that balances quality and speed.
    # 8192 works well on the available time budget.
    n = 8192
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm‑start: maybe a previously discovered sequence is stored.
    warm_start = globals().get("height_sequence_1")
    if isinstance(warm_start, (list, tuple, np.ndarray)):
        ws = np.asarray(warm_start, dtype=np.float64)
    else:
        ws = None

    # ------------------------------------------------------------------
    # ---- Stage 1: generate a pool of parametric seeds -----------------
    cand_pool = _generate_candidates(n, ws)

    # ---- Evaluate quickly and keep the top few ------------------------
    evaluated = []
    for idx, cand in enumerate(cand_pool):
        if time.time() >= deadline:
            break
        # guarantee feasibility
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 0.01:                     # degenerate – replace by a fresh Dirichlet draw
            cand = np.random.dirichlet(np.ones(n))
            s = cand.sum()
        cand = cand / s
        val = _objective(cand, n, conv_fft_len)
        evaluated.append((val, cand))
        # optional tiny progress print:
        # print(f"[seed {idx}]  val≈{val:.6f}")

    evaluated.sort(key=lambda x: x[0])
    top_k = min(6, len(evaluated))
    top_candidates = [c for _, c in evaluated[:top_k]]
    print(f"Stage 1 completed – selected {top_k} promising seeds (n={n})")

    # ------------------------------------------------------------------
    # ---- Stage 2: deep optimisation of the selected seeds -------------
    # allocate remaining time roughly equally among the seeds
    for start_id, a0 in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        # compute a dedicated sub‑deadline for this seed
        remaining = deadline - time.time()
        # give at most 1/3 of the remaining budget to a single seed
        sub_deadline = time.time() + min(remaining * 0.33, 250.0)
        print(f"--- Optimisation start #{start_id + 1}/{len(top_candidates)} (budget ≈ {sub_deadline - time.time():.1f}s) ---")
        best_seq, best_val = _deep_optimize(
            a0,
            n,
            conv_fft_len,
            corr_fft_len,
            sub_deadline,
            globals().get("height_sequence_1"),
            float("inf"),
        )
        # update the global best
        if best_seq is not None and best_val < globals().get("_global_best_val", float("inf")):
            globals()["_global_best_seq"] = best_seq
            globals()["_global_best_val"] = best_val
        print(f"   → best after this start: {best_val:.6f}")

    # retrieve the global best (fallback to the best among the seeds)
    best_seq = globals().get("_global_best_seq")
    best_val = globals().get("_global_best_val", float("inf"))
    if best_seq is None:
        # nothing succeeded – fall back to uniform
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = _objective(best_seq, n, conv_fft_len)

    # ------------------------------------------------------------------
    # ---- Stage 3: final local random mass‑transfer hill‑climb ----------
    if time.time() < deadline - 5.0:
        a = best_seq.copy()
        best_local_val = best_val
        it = 0
        max_iter = 12000
        print("Starting final local refinement (random mass transfers)...")
        while time.time() < deadline and it < max_iter:
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            # transfer a small fraction of the smaller bucket
            delta = (np.random.rand() - 0.5) * 0.4 * min(a[i], a[j])
            # clamp to keep positivity
            if delta > a[j]:
                delta = a[j]
            if -delta > a[i]:
                delta = -a[i]
            if abs(delta) < 1e-14:
                continue
            a[i] += delta
            a[j] -= delta

            conv = _full_convolution(a, conv_fft_len)
            val = 2.0 * n * conv.max() / (a.sum() ** 2)

            if val < best_local_val:
                best_local_val = val
                best_seq = a.copy()
                # keep the improvement
            else:
                # revert the move
                a[i] -= delta
                a[j] += delta
            it += 1
        best_val = best_local_val
        print(f"Local refinement finished after {it} attempts – best objective {best_val:.6f}")

    # ------------------------------------------------------------------
    # Ensure feasibility for the outer evaluator
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    total = best_seq.sum()
    if total < 0.01:                     # extremely unlikely, but guard against it
        best_seq = np.random.dirichlet(np.ones(n))
    else:
        best_seq = best_seq * (1.0 / total)   # normalise to sum = 1 (scale does not matter)

    print(f"*** Search finished – final objective ≈ {best_val:.6f} (n={n})")
    return best_seq.tolist()
