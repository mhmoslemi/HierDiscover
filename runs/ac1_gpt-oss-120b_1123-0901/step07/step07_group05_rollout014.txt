"""
Propose a step‑function (non‑negative vector) that minimises

    F(a) = 2·n·max(conv(a,a)) / (sum(a)²)

where n = len(a) and conv is the linear convolution.
The algorithm mixes a very diverse parametric pool, a fast
gradient‑based optimiser (Adam on a log‑sum‑exp surrogate of the
max‑convolution) and an exponentiated‑gradient sub‑gradient phase.
A cheap stochastic hill‑climber refines the final answer.
The whole search respects the 1000 s budget.
"""

import time
import math
import heapq
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (output length 2*len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact objective: 2·n·max(conv)/(sum)² (sum is 1 after normalisation)."""
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c / (a.sum() ** 2))

    for c in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(6):
            cand = np.random.dirichlet(np.full(n, c))
            yield cand

    # ---- symmetric edge exponential ------------------------------------
    alphas = np.logspace(-2, 2.0, 14)               # 0.01 … 100
    for alpha in alphas:
        edge = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
        edge /= edge.sum()
        for w in (0.0, 0.3, 0.6, 0.9, 1.0):
            shape = w * edge + (1.0 - w) / n
            shape /= shape.sum()
            yield shape

    # ---- symmetric power‑law -------------------------------------------
    for p in np.linspace(0.2, 8.0, 14):
        edge = (i + 1.0) ** (-p) + (n - i) ** (-p)
        edge /= edge.sum()
        for w in (0.0, 0.25, 0.5, 0.75, 1.0):
            shape = w * edge + (1.0 - w) / n
            shape /= shape.sum()
            yield shape

    # ---- asymmetric two‑sided exponential -------------------------------
    alphas = np.logspace(-2, 2.0, 12)
    for wL in (0.0, 0.2, 0.4, 0.6, 0.8):
        for wR in (0.0, 0.2, 0.4, 0.6, 0.8):
            if wL + wR > 1.0:
                continue
            wU = 1.0 - wL - wR
            for aL in alphas:
                for aR in alphas:
                    left = np.exp(-aL * i)
                    right = np.exp(-aR * (n - 1 - i))
                    shape = wL * left + wR * right + wU / n
                    shape /= shape.sum()
                    yield shape

    # ---- asymmetric power‑law -----------------------------------------
    p_vals = np.linspace(0.2, 8.0, 12)
    for wL in (0.0, 0.25, 0.5):
        for wR in (0.0, 0.25, 0.5):
            if wL + wR > 1.0:
                continue
            wU = 1.0 - wL - wR
            for pL in p_vals:
                for pR in p_vals:
                    left = (i + 1.0) ** (-pL)
                    right = (n - i) ** (-pR)
                    shape = wL * left + wR * right + wU / n
                    shape /= shape.sum()
                    yield shape

    # ---- linear / triangular -------------------------------------------
    lin = np.linspace(1.0, 0.0, n)
    tri = np.concatenate([np.linspace(0, 1, n // 2, endpoint=False),
                          np.linspace(1, 0, n - n // 2)])
    for base in (lin, tri):
        base /= base.sum()
        for w in (0.0, 0.4, 0.8):
            shape = w * base + (1.0 - w) / n
            shape /= shape.sum()
            yield shape

    # ---- sinusoidal half‑wave ------------------------------------------
    half = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    half = np.maximum(half, 0.0)
    half /= half.sum()
    yield half

    # ---- Beta‑family (i^p * (n‑i)^q) ----------------------------------
    for p in np.linspace(-0.8, 4.0, 10):
        for q in np.linspace(-0.8, 4.0, 10):
            shape = (i + 0.5) ** p * (n - 0.5 - i) ** q
            shape = np.maximum(shape, 0.0)
            if shape.sum() == 0:
                continue
            shape /= shape.sum()
            yield shape

    # ---- arcsine‑like (edge‑heavy) ------------------------------------
    for eps in (1e-6, 1e-4, 1e-2):
        shape = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
        shape /= shape.sum()
        yield shape

    # ---- sparse spikes -------------------------------------------------
    for spikes in (3, 5, 8):
        for _ in range(30):
            idx = np.random.choice(n, size=spikes, replace=False)
            weights = np.random.dirichlet(np.ones(spikes))
            vec = np.zeros(n)
            vec[idx] = weights
            vec /= vec.sum()
            yield vec

def _adam_optimize(a0: np.ndarray, n: int, conv_fft_len: int,
                    corr_fft_len: int, deadline: float,
                    start_best_seq: np.ndarray | None,
                    start_best_val: float) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on a log‑sum‑exp approximation of the max‑convolution.
    The temperature (lam) starts modestly and grows slowly.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.07                       # a tad larger than before
    lam = 0.8                       # start smoother
    lam_factor = 1.008
    lam_cap = 1e7

    best_seq = start_best_seq
    best_val = start_best_val

    it = 0
    max_iter = 400_000
    last_improve = time.time()

    while time.time() < deadline and it < max_iter:
        # soft‑max to stay on simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective (used for tracking)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Adam] iter={it:06d} new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient (chain rule)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam step
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature increase
        lam = min(lam * lam_factor, lam_cap)

        # occasional shake if stalled
        if time.time() - last_improve > 6.0:
            logits += (np.random.rand(n) - 0.5) * 5e-3
            last_improve = time.time()

        it += 1
    return best_seq, best_val


def _subgrad_optimize(a0: np.ndarray, n: int, conv_fft_len: int,
                      corr_fft_len: int, deadline: float,
                      start_best_seq: np.ndarray | None,
                      start_best_val: float) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient descent on the *exact* max‑convolution.
    Longer run‑time and a slower decay schedule than before.
    """
    a = start_best_seq if start_best_seq is not None else a0.copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-14:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a /= s

    eta = 0.7          # initial step‑size
    decay = 0.99994
    best_seq = a.copy()
    best_val = start_best_val
    it = 0
    max_iter = 120_000
    last_improve = time.time()

    while time.time() < deadline and it < max_iter:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Subgrad] iter={it:06d} new best {val:.6f}")

        # active anti‑diagonal set
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # exponentiated‑gradient update (keeps positivity)
        a = a * np.exp(-eta * grad_a)

        # renormalise
        s = a.sum()
        if s < 1e-14:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        eta *= decay
        it += 1

        if time.time() - last_improve > 7.0:
            break
    return best_seq, best_val


def _deep_optimize(a0: np.ndarray, n: int, conv_fft_len: int,
                   corr_fft_len: int, deadline: float,
                   global_best_seq: np.ndarray | None,
                   global_best_val: float) -> tuple[np.ndarray, float]:
    """
    Run Adam → Sub‑gradient on a single start and update the global best.
    """
    seq, val = _adam_optimize(a0, n, conv_fft_len, corr_fft_len,
                              deadline, global_best_seq, global_best_val)
    seq, val = _subgrad_optimize(seq, n, conv_fft_len, corr_fft_len,
                                 deadline, seq, val)
    return seq, val

def _local_random_search(a_best: np.ndarray, n: int,
                         conv_fft_len: int, deadline: float,
                         best_val: float) -> tuple[np.ndarray, float]:
    """
    Cheap hill‑climbing: perturb logits with small Gaussian noise and
    keep any improvement. Runs until time is exhausted.
    """
    logits_best = np.log(a_best + 1e-15)
    while time.time() < deadline:
        noise = np.random.randn(n) * 0.025
        logits = logits_best + noise
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()
        val = _objective(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            a_best = a.copy()
            logits_best = np.log(a_best + 1e-15)
            print(f"[Local] improved to {val:.6f}")
    return a_best, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Main entry point. Searches for a non‑negative step function that
    minimises the supplied evaluation function within ``budget_s`` seconds.
    Returns the best vector as a plain Python list.
    """
    np.random.seed(seed)
    safety = 0.5                      # keep a small safety margin
    deadline = time.time() + budget_s - safety

    # ------------------------------------------------------------------
    # problem size – 4096 gave the previous best; keep it for speed
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # warm‑start from a previously discovered construction if available
    warm_start = globals().get("height_sequence_1")
    if warm_start is not None:
        warm_start = np.asarray(warm_start, dtype=np.float64)
        if warm_start.shape[0] > n:
            warm_start = warm_start[:n]
        elif warm_start.shape[0] < n:
            warm_start = np.pad(warm_start, (0, n - warm_start.shape[0]))
        warm_start = np.maximum(warm_start, 0.0)
        s = warm_start.sum()
        if s >= 0.01:
            warm_start = warm_start / s
        else:
            warm_start = None
    else:
        warm_start = None

    # ------------------------------------------------------------------
    # Stage 1 – generate a broad pool and keep the best few
    top_k = 60
    heap = []                                   # stores (-val, array)
    cand_counter = 0

    for cand in _generate_candidates(n, warm_start):
        if time.time() >= deadline:
            break

        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 0.01:
            cand = np.random.dirichlet(np.ones(n))
            s = cand.sum()
        cand = cand / s

        val = _objective(cand, n, conv_fft_len)

        if len(heap) < top_k:
            heapq.heappush(heap, (-val, cand.copy()))
        else:
            # heap[0] holds the *worst* (largest) value among the kept items
            if val < -heap[0][0]:
                heapq.heapreplace(heap, (-val, cand.copy()))

        cand_counter += 1
        if cand_counter % 500 == 0:
            print(f"[Stage‑1] generated {cand_counter} candidates "
                  f"(heap best ≈ { -heap[0][0]:.6f})")

    # extract sorted list of top candidates (smallest value first)
    top_candidates = [arr for _, arr in sorted([(-v, a) for (v, a) in heap],
                                               key=lambda x: x[0])]

    if top_candidates:
        print(f"Stage 1 finished: evaluated {cand_counter} candidates, "
              f"best seen = { -heap[0][0]:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – deep optimisation from the selected starts
    best_seq = None
    best_val = float("inf")

    for idx, start_vec in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        print(f"--- Optimising start {idx + 1}/{len(top_candidates)} ---")
        seq, val = _deep_optimize(start_vec, n, conv_fft_len,
                                  corr_fft_len, deadline,
                                  best_seq, best_val)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[Global] new best after start {idx + 1}: {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – cheap stochastic refinement on the current global best
    if best_seq is not None:
        print("--- Local stochastic refinement ---")
        best_seq, best_val = _local_random_search(best_seq, n,
                                                  conv_fft_len,
                                                  deadline,
                                                  best_val)

    # ------------------------------------------------------------------
    # Stage 4 – if time permits, a longer sub‑gradient run on the best
    if best_seq is not None and time.time() < deadline - 30.0:
        print("--- Extended sub‑gradient refinement ---")
        best_seq, best_val = _subgrad_optimize(best_seq, n,
                                               conv_fft_len, corr_fft_len,
                                               deadline,
                                               best_seq, best_val)

    # ------------------------------------------------------------------
    # Stage 5 – optional upscale to 8192 points (more degrees of freedom)
    time_left = deadline - time.time()
    if time_left > 80.0:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        if best_seq is not None:
            # linear interpolation to the finer grid
            x_old = np.linspace(0, 1, n, endpoint=False)
            x_new = np.linspace(0, 1, n_big, endpoint=False)
            a_big = np.interp(x_new, x_old, best_seq)
            a_big = np.maximum(a_big, 0.0)
            a_big /= a_big.sum()
        else:
            a_big = np.random.dirichlet(np.ones(n_big))

        print("--- Optimising on larger grid (n=8192) ---")
        seq_big, val_big = _deep_optimize(a_big, n_big,
                                          conv_fft_len_big,
                                          corr_fft_len_big,
                                          deadline,
                                          best_seq, best_val)

        if val_big < best_val:
            best_seq = seq_big
            best_val = val_big
            n = n_big
            print(f"[Global] improved on larger grid: {best_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – if nothing succeeded, use uniform vector.
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Clip to the evaluator’s allowed range
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"*** Search finished: best objective ≈ {best_val:.6f} "
          f"(n={best_seq.shape[0]})")
    return best_seq.tolist()
