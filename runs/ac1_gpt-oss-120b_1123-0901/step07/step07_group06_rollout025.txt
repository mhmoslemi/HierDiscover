"""
Propose a step‑function (non‑negative vector) that minimizes

    2 * n * max(conv(a,a)) / (sum(a)^2)

within the 1000 s budget.

Algorithm
---------

1. **Parameterised library** – generate a rich pool of candidates:
   uniform, many Dirichlet draws, edge‑exponential / power‑law families,
   beta‑shaped, sinusoidal, triangular / quadratic and the previous best
   (if any).

2. **Quick screening** – evaluate every candidate with an FFT based
   convolution and keep the best `TOP_K` (≈30) for deep optimisation.

3. **Deep optimisation** – for each start run three phases
   * Adam on a log‑sum‑exp surrogate of the max‑convolution
     (slow temperature increase, many iterations).
   * Exponentiated‑gradient sub‑gradient steps on the true max‑convolution.
   * Projected‑gradient refinement (exact objective, Euclidean projection
     onto the simplex after each step).

   The phases stop once the global deadline is reached.

4. **Intensive local search** – starting from the current global best,
   repeatedly move a random small amount of mass from one index to another
   (pairwise exchange).  Accept only improving moves, recomputing the exact
   objective via a fast FFT convolution.  This simple stochastic hill‑climb
   often lowers the bound by a few · 10⁻⁴.

5. **Upscaling (optional)** – if there is >≈30 s left, interpolate the best
   vector to a finer grid (n = 8192) and repeat the deep optimisation pipeline.

All steps respect the non‑negativity and the automatic scaling of the
objective (the score is invariant to a global scaling of the vector).  The
function prints progress information and finally returns the best sequence
as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("v must be 1‑d")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact evaluation (same formula as evaluate_sequence)."""
    conv = _full_convolution(a, conv_fft_len)
    return float(2.0 * n * conv.max() / (a.sum() ** 2))


# ----------------------------------------------------------------------
def _adam_smooth_max(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑sum‑exp surrogate of max(conv).
    """
    eps = 1e-15
    logits = np.log(a0 + eps)            # work in log‑space (soft‑max)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05

    lam = 0.5                # start temperature (smoothness)
    lam_factor = 1.01        # slower schedule – more accurate gradients
    lam_cap = 1e12

    last_improve = time.time()
    it = 0
    max_iter = 1_200_000   # generous upper bound – actual stop is deadline

    while time.time() < deadline and it < max_iter:
        # simplex from logits (soft‑max)
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # surrogate evaluation (exact objective also cheap – we log it)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Adam n={n} it={it:07d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        # gradient of surrogate (chain‑rule through soft‑max)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature schedule
        lam = min(lam * lam_factor, lam_cap)

        # occasional jitter if stuck
        if time.time() - last_improve > 5.0:
            logits += (np.random.rand(n) - 0.5) * 5e-3
            last_improve = time.time()

        it += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
def _subgradient_refinement(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient steps on the true max‑convolution.
    """
    a = best_seq if best_seq is not None else a0.copy()
    a = np.maximum(a, 0.0)
    a /= a.sum()

    eta = 0.4
    last_improve = time.time()
    max_iter = 400_000

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Subgrad n={n} it={it:07d}] new best {val:.6f}")

        # active set (indices attaining the maximum)
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # exponentiated‑gradient update
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:                     # avoid collapse
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        # early stop if no improvement for a while
        if time.time() - last_improve > 6.0:
            break

        eta *= 0.9999

    return best_seq, best_val

def _projected_gradient_refine(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Gradient descent on the exact objective with explicit simplex projection.
    """
    a = a0.copy()
    a = np.maximum(a, 0.0)
    a /= a.sum()

    lr = 0.25
    last_improve = time.time()
    max_iter = 300_000

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[ProjGrad n={n} it={it:07d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)   # ∂M/∂a
        step = -lr * grad_a
        a_candidate = _proj_simplex(a + step)

        conv_cand = _full_convolution(a_candidate, conv_fft_len)
        val_cand = 2.0 * n * conv_cand.max() / (a_candidate.sum() ** 2)

        if val_cand < val:
            a = a_candidate
            lr = min(lr * 1.02, 1.0)
        else:
            lr *= 0.7
            if lr < 1e-6:
                break

        if time.time() - last_improve > 6.0:
            break

        lr *= 0.99998

    return best_seq, best_val

def _deep_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,

def _generate_parametric_candidates(n: int, warm_start: np.ndarray | None):
    """Create a diverse set of parametrised vectors (all sum≈1)."""
    cand_list = []

    # warm‑start and simple cyclic shifts
    if warm_start is not None:
        cand_list.append(warm_start.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand_list.append(np.roll(warm_start, shift))

    # uniform baseline
    cand_list.append(np.full(n, 1.0 / n))

    # many random Dirichlet draws (various concentrations)
    for conc in (0.02, 0.05, 0.1, 0.2, 0.4, 0.8, 1.0, 2.0, 4.0):
        for _ in range(12):
            cand_list.append(np.random.dirichlet(np.full(n, conc)))

    i = np.arange(n, dtype=np.float64)

    # edge‑exponential families (with optional uniform mix)
    for alpha in np.logspace(-2, 1.0, num=24):
        edge = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
        edge /= edge.sum()
        for w in (0.0, 0.25, 0.5, 0.75, 1.0):
            cand_list.append(w * edge + (1.0 - w) / n)

    # edge‑power‑law families
    for p in np.linspace(0.2, 5.0, num=24):
        edge = (i + 1) ** (-p) + (n - i) ** (-p)
        edge /= edge.sum()
        for w in (0.0, 0.33, 0.66, 1.0):
            cand_list.append(w * edge + (1.0 - w) / n)

    # two‑Gaussian edges (different widths)
    for gamma in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 4.0):
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        shape /= shape.sum()
        cand_list.append(shape)

    # sinusoidal half‑wave (positive part)
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand_list.append(shape)

    # beta‑shaped families (flexible edge/center bias)
    for alpha in (0.2, 0.5, 1.0, 2.0, 5.0):
        for beta in (0.2, 0.5, 1.0, 2.0, 5.0):
            edge = (i + 1) ** (alpha - 1) * (n - i) ** (beta - 1)
            edge = np.maximum(edge, 0.0)
            edge /= edge.sum()
            cand_list.append(edge)

    # simple triangular / quadratic shapes
    tri = np.minimum(i + 1, n - i)
    tri = tri.astype(np.float64)
    tri /= tri.sum()
    cand_list.append(tri)

    quad = (i + 1) * (n - i)
    quad = quad.astype(np.float64)
    quad /= quad.sum()
    cand_list.append(quad)

    return cand_list


# ----------------------------------------------------------------------
def _local_pairwise_search(
    a: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Simple stochastic hill‑climb: move a small amount of mass from one index
    to another and accept only if the true objective improves.
    """
    a_cur = a.copy()
    conv_cur = _full_convolution(a_cur, conv_fft_len)
    val_cur = 2.0 * n * conv_cur.max() / (a_cur.sum() ** 2)

    if best_seq is None or val_cur < best_val:
        best_seq = a_cur.copy()
        best_val = val_cur

    N = n
    while time.time() < deadline:
        # pick donor (must have some mass) and recipient
        i = np.random.randint(N)
        if a_cur[i] < 1e-8:
            continue
        j = np.random.randint(N - 1)
        if j >= i:
            j += 1

        # amount to transfer – a fraction of donor's mass
        delta = a_cur[i] * (0.01 + 0.09 * np.random.rand())  # 1‑10 % of donor

        a_new = a_cur.copy()
        a_new[i] -= delta
        a_new[j] += delta

        # quick feasibility – keep non‑negative
        if a_new[i] < 0:
            continue

        conv_new = _full_convolution(a_new, conv_fft_len)
        val_new = 2.0 * n * conv_new.max() / (a_new.sum() ** 2)

        if val_new < best_val:
            best_val = val_new
            best_seq = a_new.copy()
            a_cur = a_new
            conv_cur = conv_new
            # occasionally report progress
            print(f"[PairSwap] new best {best_val:.6f}")
        # otherwise reject

    return best_seq, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """Main entry point – returns the best sequence found within the budget."""
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # a small safety margin

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # --------------------------------------------------------------
    # Base grid size
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Warm‑start from a previously stored construction (if any)
    warm_start = globals().get("height_sequence_1")
    if warm_start is not None:
        warm_start = np.asarray(warm_start, dtype=np.float64)
        if warm_start.shape[0] > n:
            warm_start = warm_start[:n]
        elif warm_start.shape[0] < n:
            warm_start = np.pad(warm_start, (0, n - warm_start.shape[0]))
        warm_start = np.maximum(warm_start, 0.0)
        if warm_start.sum() >= 0.01:
            warm_start = warm_start / warm_start.sum()
        else:
            warm_start = None
    else:
        warm_start = None

    # --------------------------------------------------------------
    # Stage 1 – generate many parametric candidates
    all_candidates = _generate_parametric_candidates(n, warm_start)

    cand_vals = []
    for idx, cand in enumerate(all_candidates):
        if time.time() >= deadline:
            break
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 1e-8:
            cand = np.random.dirichlet(np.ones(n))
            s = cand.sum()
        cand /= s
        val = _objective(cand, n, conv_fft_len)
        cand_vals.append((val, cand))
        if idx % 200 == 0:
            print(f"[Screen] evaluated {idx+1}/{len(all_candidates)} candidates "
                  f"current best {best_val:.6f}")

    cand_vals.sort(key=lambda x: x[0])
    TOP_K = 30
    top_candidates = [c for _, c in cand_vals[:TOP_K]]
    print(f"[Screen] selected top {TOP_K} candidates for deep optimisation")

    # --------------------------------------------------------------
    # Stage 2 – deep optimisation of each selected start
    for start_idx, start_vec in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        print(f"\n=== Deep optimisation on start #{start_idx + 1}/{TOP_K} ===")
        best_seq, best_val = _deep_optimize(
            start_vec,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            best_seq,
            best_val,
        )

    # --------------------------------------------------------------
    # Stage 3 – intensive stochastic pairwise local search around current best
    if best_seq is not None and time.time() < deadline - 5.0:
        print("\n--- Intensive local pairwise search ---")
        best_seq, best_val = _local_pairwise_search(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            best_seq,
            best_val,
        )

    # --------------------------------------------------------------
    # Stage 4 – optional up‑sampling to a larger grid if time permits
    time_left = deadline - time.time()
    if time_left > 35.0:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        if best_seq is not None:
            x_old = np.linspace(0, 1, n, endpoint=False)
            x_new = np.linspace(0, 1, n_big, endpoint=False)
            a_big = np.interp(x_new, x_old, best_seq)
            a_big = np.maximum(a_big, 0.0)
            a_big /= a_big.sum()
        else:
            a_big = np.random.dirichlet(np.ones(n_big))

        print("\n--- Optimising on larger grid (n=8192) ---")
        best_seq, best_val = _deep_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            best_seq,
            best_val,
        )
        # final pairwise refinement on the big grid if we still have time
        if best_seq is not None and time.time() < deadline - 5.0:
            print("\n--- Final pairwise refinement on big grid ---")
            best_seq, best_val = _local_pairwise_search(
                best_seq,
                n_big,
                conv_fft_len_big,
                deadline,
                best_seq,
                best_val,
            )

    # --------------------------------------------------------------
    # Fallback in the extremely unlikely case we never found a candidate
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Clip to evaluator's allowed range (0 … 1000) and return
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    print(f"\n*** Search finished: best objective ≈ {best_val:.6f} (n={best_seq.shape[0]})")
    return best_seq.tolist()
