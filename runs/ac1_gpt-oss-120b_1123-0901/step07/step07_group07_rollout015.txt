def local_sa(a0, n, conv_len, deadline, start_val):
    a = a0.copy()
    cur_val = start_val
    best_seq = a.copy()
    best_val = cur_val
    start_time = time.time()
    while time.time() < deadline:
        # schedule temperature: T = T0 * (1 - (elapsed/total))  # linear decay
        # choose random indices i,j
        i, j = np.random.randint(n, size=2)
        if i==j: continue
        # compute maximum delta making both non-negative
        # choose delta proportion of min(a[i], a[j]) * random
        max_transfer = min(a[i], a[j])
        # propose delta extremely small? We can propose relative step
        delta = (np.random.rand() - 0.5) * 0.2 * max_transfer  # up to 10% of smaller
        a_new = a.copy()
        a_new[i] += delta
        a_new[j] -= delta
        if a_new[i] < 0 or a_new[j] < 0: continue
        # renorm? Mass transferred; sum remains same.
        conv = _full_convolution(a_new, conv_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a_new.sum() ** 2)
        # accept if better else prob
        elapsed = time.time() - start_time
        remaining = deadline - time.time()
        # compute temperature schedule decreasing from T_start to 1e-6:
        T = max(1e-6, (1.0 - elapsed / (deadline-start_time)) * 0.01)  # scale
        if val < cur_val or np.random.rand() < np.exp(-(val - cur_val) / T):
            a = a_new
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = a.copy()
        # maybe occasional reheat or random reset
    return best_seq, best_val

"""
Propose a low‑evaluation step‑function.

The search consists of three layers:

1. **Broad seeding** – many parametric families together with a large
   collection of random Dirichlet draws are generated, quickly evaluated
   (FFT‑based convolution) and the best ~15 are kept.

2. **Deep refinement** – each seed is optimized by
   *Adam* on a smooth‑max surrogate of the max‑convolution (log‑sum‑exp
   with a slowly increasing temperature) and then by an
   exponentiated‑gradient sub‑gradient scheme that works directly on the
   true max.  The total time budget is split evenly between the seeds.

3. **Final annealing** – the current global best is subjected to a short
   simulated‑annealing hill‑climber (random mass transfers with
   Metropolis acceptance) to escape any remaining local minima.

All vectors are kept on the probability simplex, non‑negativity is
enforced, and the routine respects the 1000 s deadline.
"""

import time
import numpy as np
from scipy import sparse
from scipy.optimize import linprog

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via real FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Cross‑correlation (w ★ a) of length len(a).
    (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Fast 2·n·max(a*a) / (sum(a)**2).  Here sum(a)≈1."""
    conv = _full_convolution(a, conv_fft_len)
    return float(2.0 * n * conv.max() / (a.sum() ** 2))


# ----------------------------------------------------------------------
def _adam_smooth_max(a0: np.ndarray,
                     n: int,
                     conv_fft_len: int,
                     corr_fft_len: int,
                     deadline: float,
                     global_best_seq: np.ndarray | None,
                     global_best_val: float) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on a log‑sum‑exp surrogate of the max‑convolution.
    Returns the best vector/value seen during the run.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)            # soft‑max parameters
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    # hyper‑parameters (tuned for this problem)
    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.07
    lam = 0.2               # start temperature (smoothness)
    lam_factor = 1.01
    lam_cap = 1e5

    best_seq = global_best_seq
    best_val = global_best_val
    last_improve = time.time()
    it = 0
    max_iter = 400_000

    while time.time() < deadline and it < max_iter:
        # ---------- simplex via soft‑max ----------
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # ---------- surrogate objective ----------
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[{n}] Adam it={it:06d}  new best {val:.6f}")

        # ---------- smooth‑max weights ----------
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---------- gradient ----------
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)                # ⟨a,∇⟩  (soft‑max chain rule)
        grad_log = a * (grad_a - avg)

        # ---------- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---------- temperature schedule ----------
        lam = min(lam * lam_factor, lam_cap)

        # ---------- occasional jitter ----------
        if time.time() - last_improve > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-3
            last_improve = time.time()

        it += 1

    return best_seq, best_val


def _subgradient_refinement(a0: np.ndarray,
                            n: int,
                            conv_fft_len: int,
                            corr_fft_len: int,
                            deadline: float,
                            init_best_seq: np.ndarray | None,
                            init_best_val: float) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient descent on the true max‑convolution.
    """
    a = init_best_seq if init_best_seq is not None else a0
    a = np.maximum(a, 0.0)
    a /= a.sum()

    best_seq = a.copy()
    best_val = init_best_val
    it = 0
    max_iter = 60_000
    last_improve = time.time()
    eta0 = 0.9

    while time.time() < deadline and it < max_iter:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[{n}] Subgrad it={it:06d}  new best {val:.6f}")

        # active set where the convolution attains its maximum
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # decaying step‑size
        eta = eta0 / np.sqrt(it + 1)

        # mirror‑descent (exponentiated‑gradient) update
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:            # numerical safety
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        # occasional random jitter to avoid stagnation
        if time.time() - last_improve > 8.0:
            jitter = (np.random.rand(n) - 0.5) * 1e-4
            a = np.maximum(a + jitter, 0.0)
            a /= a.sum()
            last_improve = time.time()

        it += 1

    return best_seq, best_val


def _simulated_annealing(a0: np.ndarray,
                         n: int,
                         conv_fft_len: int,
                         deadline: float,
                         start_val: float) -> tuple[np.ndarray, float]:
    """
    Simple SA on the simplex: random mass transfers between two bins.
    Accepts uphill moves with probability exp(-(Δ)/T).  Temperature
    decreases linearly to a tiny floor.
    """
    a = a0.copy()
    cur_val = start_val
    best_seq = a.copy()
    best_val = cur_val

    start_time = time.time()
    total_time = deadline - start_time
    if total_time <= 0:
        return best_seq, best_val

    it = 0
    while time.time() < deadline:
        # temperature linearly decays from 1e-2 to 1e-6
        elapsed = time.time() - start_time
        T = max(1e-6, (1.0 - elapsed / total_time) * 1e-2)

        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # mass that can be moved without violating non‑negativity
        max_transfer = min(a[i], a[j])
        if max_transfer < 1e-15:
            continue
        # propose a transfer up to 30 % of the smaller bucket
        delta = (np.random.rand() - 0.5) * 0.6 * max_transfer
        a_new = a.copy()
        a_new[i] += delta
        a_new[j] -= delta
        if a_new[i] < 0 or a_new[j] < 0:
            continue   # reject illegal move

        conv = _full_convolution(a_new, conv_fft_len)
        val_new = 2.0 * n * conv.max() / (a_new.sum() ** 2)

        if val_new < cur_val or np.random.rand() < np.exp(-(val_new - cur_val) / T):
            a = a_new
            cur_val = val_new
            if val_new < best_val:
                best_val = val_new
                best_seq = a.copy()
        it += 1
    print(f"[{n}] SA finished after {it} proposals, best {best_val:.6f}")
    return best_seq, best_val


def _generate_parametric_candidates(n: int, warm_start: np.ndarray | None) -> list[np.ndarray]:
    """Rich pool of analytic families (sym‑ and non‑sym‑shaped)."""
    candidates: list[np.ndarray] = []

    # warm‑start and its cyclic shifts
    if warm_start is not None:
        ws = warm_start.copy()
        candidates.append(ws)
        for shift in (n // 4, n // 2, 3 * n // 4):
            candidates.append(np.roll(ws, shift))

    # uniform
    candidates.append(np.full(n, 1.0 / n))

    # random Dirichlet with several concentrations
    for conc in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        candidates.append(np.random.dirichlet(np.full(n, conc)))

    # edge‑exponential family (and mixes with uniform)
    i = np.arange(n, dtype=np.float64)
    for alpha in np.logspace(-2, 1.0, num=12):
        edge = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
        edge /= edge.sum()
        for w in (0.0, 0.3, 0.6, 1.0):
            candidates.append(w * edge + (1.0 - w) / n)

    # edge‑power‑law family
    for p in np.linspace(0.2, 5.0, num=12):
        edge = (i + 1.0) ** (-p) + (n - i) ** (-p)
        edge /= edge.sum()
        for w in (0.0, 0.5, 1.0):
            candidates.append(w * edge + (1.0 - w) / n)

    # symmetric beta‑like shapes
    for alpha in (0.2, 0.4, 0.6, 0.9, 1.2, 1.6, 2.0):
        shape = (i + 1.0) ** (alpha - 1.0) * (n - i) ** (alpha - 1.0)
        shape = np.maximum(shape, 0.0)
        shape /= shape.sum()
        candidates.append(shape)

    # raised‑cosine / sine‑power families
    for p in (1, 2, 4, 6, 8):
        shape = np.sin(np.pi * (i + 0.5) / n) ** p
        shape = np.maximum(shape, 0.0)
        shape /= shape.sum()
        candidates.append(shape)

    # double‑Gaussian edge shapes
    for gamma in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        shape /= shape.sum()
        candidates.append(shape)

    # arcsine‑type shape
    shape = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    candidates.append(shape)

    return candidates

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Entry point – search for a non‑negative step function that minimises
    ``evaluate_sequence``.  Returns the best sequence (plain Python list).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Initial grid size
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm‑start (if a previous solution is available)
    warm_start = globals().get("height_sequence_1")
    if warm_start is not None:
        ws = np.asarray(warm_start, dtype=np.float64)
        if ws.shape[0] > n:
            ws = ws[:n]
        elif ws.shape[0] < n:
            ws = np.pad(ws, (0, n - ws.shape[0]))
        ws = np.clip(ws, 0.0, None)
        if ws.sum() >= 0.01:
            ws = ws / ws.sum()
        else:
            ws = None
    else:
        ws = None

    # ------------------------------------------------------------------
    # Stage 1 – generate a rich pool of candidates
    cand_pool = _generate_parametric_candidates(n, ws)

    # add a larger batch of random Dirichlet draws (different concentrations)
    for conc in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(12):
            cand_pool.append(np.random.dirichlet(np.full(n, conc)))

    # quick evaluation of everything
    evaluated: list[tuple[float, np.ndarray]] = []
    for cand in cand_pool:
        if time.time() >= deadline:
            break
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 0.01:
            cand = np.random.dirichlet(np.ones(n))
            s = cand.sum()
        cand = cand / s
        val = _objective(cand, n, conv_fft_len)
        evaluated.append((val, cand))

    evaluated.sort(key=lambda x: x[0])
    top_k = min(20, len(evaluated))
    top_candidates = [c for _, c in evaluated[:top_k]]
    print(f"Stage 1 – {top_k} seeds kept, best init value {evaluated[0][0]:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – blend a few seeds to enrich diversity
    mixed = []
    for _ in range(30):
        i, j = np.random.choice(len(top_candidates), 2, replace=False)
        α = np.random.rand()
        mixed_c = α * top_candidates[i] + (1 - α) * top_candidates[j]
        mixed_c = np.maximum(mixed_c, 0.0)
        mixed_c /= mixed_c.sum()
        mixed.append(mixed_c)

    for cand in mixed:
        if time.time() >= deadline:
            break
        val = _objective(cand, n, conv_fft_len)
        evaluated.append((val, cand))

    evaluated.sort(key=lambda x: x[0])
    top_k2 = min(15, len(evaluated))
    top_candidates = [c for _, c in evaluated[:top_k2]]
    print(f"Stage 2 – after mixing, best init value {evaluated[0][0]:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – deep optimisation of each seed (time split evenly)
    best_seq: np.ndarray | None = None
    best_val = float("inf")
    seeds_left = len(top_candidates)
    for idx, a0 in enumerate(top_candidates):
        # allocate a slice of the remaining budget to this seed
        time_now = time.time()
        time_left = deadline - time_now
        if time_left < 5.0:
            break
        # give each start roughly equal remaining time
        sub_deadline = time_now + time_left / (seeds_left - idx)
        print(f"--- Optimising seed {idx + 1}/{len(top_candidates)} (time slice {sub_deadline - time.time():.1f}s) ---")
        # Adam (smooth‑max)
        a_mid, val_mid = _adam_smooth_max(
            a0, n, conv_fft_len, corr_fft_len, sub_deadline, best_seq, best_val
        )
        # sub‑gradient refinement
        a_opt, val_opt = _subgradient_refinement(
            a_mid, n, conv_fft_len, corr_fft_len, sub_deadline, a_mid, val_mid
        )
        if val_opt < best_val:
            best_val, best_seq = val_opt, a_opt.copy()
            print(f">>> New global best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – upscale to a finer grid if time permits
    time_left = deadline - time.time()
    if time_left > 30.0:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        if best_seq is not None:
            # simple linear interpolation
            x_old = np.linspace(0, 1, n, endpoint=False)
            x_big = np.linspace(0, 1, n_big, endpoint=False)
            a_big = np.interp(x_big, x_old, best_seq)
            a_big = np.maximum(a_big, 0.0)
            a_big /= a_big.sum()
        else:
            a_big = np.random.dirichlet(np.ones(n_big))

        print("--- Optimising on larger grid (n=8192) ---")
        sub_deadline_big = time.time() + time_left - 5.0
        a_mid, val_mid = _adam_smooth_max(
            a_big, n_big, conv_fft_len_big, corr_fft_len_big, sub_deadline_big, best_seq, best_val
        )
        a_opt, val_opt = _subgradient_refinement(
            a_mid, n_big, conv_fft_len_big, corr_fft_len_big, sub_deadline_big, a_mid, val_mid
        )
        if val_opt < best_val:
            best_val, best_seq = val_opt, a_opt.copy()
            n, conv_fft_len, corr_fft_len = n_big, conv_fft_len_big, corr_fft_len_big
            print(f">>> New best after upscale = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 5 – final simulated‑annealing hill‑climb
    if best_seq is not None and time.time() < deadline - 5.0:
        print("--- Final simulated‑annealing refinement ---")
        cur_val = _objective(best_seq, n, conv_fft_len)
        a_final, val_final = _simulated_annealing(best_seq, n, conv_fft_len, deadline, cur_val)
        if val_final < best_val:
            best_val, best_seq = val_final, a_final.copy()
            print(f">>> Best after SA = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback (should never trigger)
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)
        best_val = _objective(best_seq, 4096, _next_pow2(2 * 4096 - 1))

    # Clip to evaluator limits and return
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    print(f"*** Search complete: best objective ≈ {best_val:.6f} (n={best_seq.shape[0]})")
    return best_seq.tolist()
