"""
Mixed‑strategy optimizer for the step‑function objective

F(a) = 2·n·max(a∗a) / (sum(a)²)

The search proceeds in three stages:

1. **Diverse warm‑starts** – a modest pool containing the previous
   solution, Dirichlet draws, edge‑exponential / polynomial shapes,
   double‑Gaussian bumps, sinusoid, uniform, and a few sparse “spike’’
   patterns (including random‑position spikes).

2. **Long Adam phase on a smooth‑max surrogate** – we optimise a
   soft‑max (log‑sum‑exp) approximation of the true max.  The temperature
   (λ) starts very small (≈0.01) and grows slowly (factor 1.02 per step)
   so the gradient gently focuses on the true peak.  Adam learning‑rate
   starts at 0.20 and decays gently; the best iterate is retained.

3. **Exact‑max exponentiated‑gradient sub‑gradient phase** – after the
   surrogate phase we switch to the true max, using a mirror‑descent
   update a←a·exp(‑η·g) with ηₜ=η₀/√(t+1).  This phase runs for the bulk
   of the remaining budget.

4. **Cheap random hill‑climbing** – mass is repeatedly moved between two
   random indices; moves are kept if they improve the objective.

5. **Optional up‑sampling** – if time permits the current best sequence
   is interpolated to a finer grid (n=8192) and the whole procedure is
   repeated on the larger problem, potentially achieving a lower bound.

All phases respect a global deadline (default 1000 s) and always return
the best sequence discovered.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a length‑len(a) vector.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
def _initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Create a modest but diverse pool of start vectors."""
    cand = []

    # warm start (if any) and a few cyclic shifts
    if warm is not None:
        cand.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    # random Dirichlet (full support)
    cand.append(np.random.dirichlet(np.ones(n)))

    # edge‑exponential families (mass concentrated at the left/right)
    for gamma in (0.2, 0.5, 1.0, 2.0):
        left = np.exp(-gamma * np.arange(n, dtype=np.float64))
        right = left[::-1]
        shape = left + right
        shape /= shape.sum()
        cand.append(shape)

    # edge‑polynomial (power‑law) families
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.5, 5):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    # double‑Gaussian (different widths)
    i = np.arange(n, dtype=np.float64)
    for sigma in (0.1, 0.3, 0.6):
        shape = np.exp(-0.5 * ((i) / (sigma * n)) ** 2) \
                + np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
        shape /= shape.sum()
        cand.append(shape)

    # positive half‑wave sinusoid
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand.append(shape)

    # uniform
    cand.append(np.full(n, 1.0 / n))

    return cand


def _spike_candidates(n: int) -> list[np.ndarray]:
    """Sparse candidates with a small number of equally spaced spikes."""
    out = []
    for k in (32, 64, 128, 256, 512):
        a = np.zeros(n, dtype=np.float64)
        if k > 0:
            pos = np.round(np.linspace(0, n - 1, k)).astype(int)
            a[pos] = 1.0 / k
        out.append(a)
    return out


def _random_spike_candidates(n: int, count: int = 8) -> list[np.ndarray]:
    """Randomly positioned spike patterns (few non‑zero entries)."""
    out = []
    rng = np.random.default_rng()
    for _ in range(count):
        k = rng.integers(8, 256)            # number of spikes
        a = np.zeros(n, dtype=np.float64)
        pos = rng.choice(n, size=k, replace=False)
        a[pos] = 1.0 / k
        out.append(a)
    return out


def _smooth_random_candidate(n: int, sigma_frac: float = 0.02) -> np.ndarray:
    """Generate a dense smooth vector by filtering a random Dirichlet."""
    a = np.random.rand(n)
    # Gaussian smoothing kernel
    x = np.arange(-3 * sigma_frac * n, 3 * sigma_frac * n + 1)
    sigma = sigma_frac * n
    kernel = np.exp(-0.5 * (x / sigma) ** 2)
    kernel /= kernel.sum()
    a = np.convolve(a, kernel, mode='same')
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a /= a.sum()
    return a

def _adam_optimize_custom(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
    lam_start: float = 0.01,
    lam_factor: float = 1.02,
    lam_max: float = 1e7,
    lr_start: float = 0.20,
    lr_decay: float = 0.9995,
    eps: float = 1e-15,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the smooth‑max surrogate.
    Returns the best vector seen (≈simplex) and its exact objective.
    """
    logits = np.log(a0 + eps)                # initialise on the simplex
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = lr_start
    lam = lam_start

    best_a = a0.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max projection onto the simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 2000 == 0:
                print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the smooth surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through soft‑max (mirror‑descent form)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # temperature schedule
        lam = min(lam * lam_factor, lam_max)

        # learning‑rate decay
        lr *= lr_decay

        # occasional tiny random shake near the end
        if it % 5000 == 0 and time.time() > deadline - 30:
            logits += (np.random.rand(n) - 0.5) * 2e-2

        it += 1

    return best_a, best_val


def _subgradient_optimize_custom(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
    eta0: float = 0.70,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient descent with exponentiated‑gradient updates.
    Returns the best vector seen and its exact objective.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # indices attaining the max
        active = np.isclose(conv, max_c, atol=1e-14)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # step size schedule ηₜ = η₀ / √(t+1)
        eta = eta0 / np.sqrt(it + 1)

        # exponentiated‑gradient mirror descent on the simplex
        a = a * np.exp(-eta * grad_a)

        # renormalise, guard against underflow
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 100_000,
) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climbing: move a random fraction of mass from one index
    to another, keep the move if the objective improves.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()
    it = 0
    rng = np.random.default_rng()
    while time.time() < deadline and it < max_moves:
        i = rng.integers(n)
        if a[i] < 1e-14:
            continue
        j = rng.integers(n)
        if i == j:
            continue

        # move up to 30 % of the source mass
        delta = a[i] * (rng.random() * 0.3)

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5000 == 0:
                print(f"[Local move {it:06d}] improved to {val2:.6f}")

        it += 1

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the given time budget and always returns the best
    sequence (as a Python list) found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # Problem size (start with 4096, later optionally upscale)
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm start from previous run (if any)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None

    # ------------------------------------------------------------------
    # Build a diverse pool of start vectors
    pool = (
        _initial_candidates(n, warm)
        + _spike_candidates(n)
        + _random_spike_candidates(n, count=6)
        + [_smooth_random_candidate(n)]
    )

    # Limit the pool size to keep the total runtime manageable
    if len(pool) > 20:
        pool = pool[:20]

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Phase 1 – explore a few candidates with a short Adam run
    per_candidate_time = 30.0   # seconds allotted to each candidate (approx.)
    for cid, cand in enumerate(pool):
        if time.time() >= deadline:
            break
        # Normalise the candidate onto the simplex
        a0 = np.maximum(cand, 0.0)
        s0 = a0.sum()
        if s0 < 1e-12:
            a0 = np.random.dirichlet(np.ones(n))
        else:
            a0 = a0 / s0

        # Give each candidate a modest time slice
        local_deadline = min(deadline - 5.0, time.time() + per_candidate_time)

        a_adam, val_adam = _adam_optimize_custom(
            a0,
            n,
            conv_fft_len,
            corr_fft_len,
            local_deadline,
            max_iters=400_000,
            lam_start=0.01,
            lam_factor=1.02,
            lr_start=0.20,
            lr_decay=0.9997,
        )

        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Candidate {cid}] Adam best {val_adam:.6f}")

        # Early bail‑out if we run out of time
        if time.time() >= deadline - 10:
            break

    # ------------------------------------------------------------------
    # Phase 2 – long exact‑max sub‑gradient refinement on the best candidate
    if best_seq is not None and time.time() < deadline - 8:
        sub_deadline = deadline - 5.0
        a_sub, val_sub = _subgradient_optimize_custom(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            sub_deadline,
            max_iters=400_000,
            eta0=0.70,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Refine] Sub‑gradient best {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – cheap random hill‑climbing
    if best_seq is not None and time.time() < deadline - 5:
        loc_deadline = deadline - 2.0
        a_loc, val_loc = _local_random_search(
            best_seq,
            n,
            conv_fft_len,
            loc_deadline,
            max_moves=120_000,
        )
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()
            print(f"[Local] improved best {val_loc:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – optional up‑sampling to a finer grid (8192) if time permits
    if time.time() + 40 < deadline:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        # Linear interpolation to the finer grid
        if best_seq is not None:
            x_old = np.linspace(0, 1, n, endpoint=False)
            x_new = np.linspace(0, 1, n_big, endpoint=False)
            a_big = np.interp(x_new, x_old, best_seq)
        else:
            a_big = np.random.dirichlet(np.ones(n_big))

        a_big = np.maximum(a_big, 0.0)
        a_big /= a_big.sum()

        # Short Adam on the larger grid
        big_deadline = min(deadline - 5.0, time.time() + 30.0)
        a_big_adam, _ = _adam_optimize_custom(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            big_deadline,
            max_iters=250_000,
            lam_start=0.01,
            lam_factor=1.025,
            lr_start=0.15,
            lr_decay=0.9995,
        )

        # Longer sub‑gradient refinement on the large grid
        if time.time() < deadline - 5:
            a_big_ref, val_big_ref = _subgradient_optimize_custom(
                a_big_adam,
                n_big,
                conv_fft_len_big,
                corr_fft_len_big,
                deadline,
                max_iters=300_000,
                eta0=0.65,
            )
            if val_big_ref < best_val:
                best_val = val_big_ref
                best_seq = a_big_ref.copy()
                n = n_big
                conv_fft_len = conv_fft_len_big
                corr_fft_len = corr_fft_len_big
                print(f"[Big] refined best {val_big_ref:.6f}")

    # ------------------------------------------------------------------
    # Fallback – ensure a valid non‑empty return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the allowed range (non‑negative, ≤1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Final sanity check (optional – uncomment if you want to see the exact score)
    # from evaluate_sequence import evaluate_sequence
    # final_score = evaluate_sequence(best_seq.tolist())
    # print(f"Final reported score {final_score:.6f} (length {len(best_seq)})")

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
