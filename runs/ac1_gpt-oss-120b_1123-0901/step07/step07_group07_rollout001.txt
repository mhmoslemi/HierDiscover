"""
Propose a low‑evaluation step‑function using a multi‑phase search.
-----------------------------------------------------------------
1. **Parametric seeding** – generate a diverse set of symmetric shapes
   (beta‑like, exponential, power‑law, sinusoidal, Gaussian, arcsine,
   random Dirichlet) on a medium grid (n≈4096).  Keep the best few.

2. **Symmetry‑preserving gradient descent** – each seed is refined by
   two complementary optimisers:
   * *Adam on a log‑sum‑exp (smooth‑max) surrogate* of the convolution.
     After each soft‑max conversion the vector is symmetrised
     (a ← ½(a+a[::-1])) and projected onto the probability simplex.
   * *Exponentiated‑gradient sub‑gradient descent* on the true
     max‑convolution (the exact sub‑gradient is a correlation with the
     active‑set weight).  Symmetrisation and simplex projection are
     applied after every update.

   The Adam temperature (λ) is increased very slowly (factor 1.01) to
   allow a gradual transition from a smooth to the true max.  The
   learning‑rate schedule and jitter are tuned to escape plateaus.
   Both phases run until the shared deadline.

3. **Upscaling** – if time permits, linearly interpolate the current
   best vector to a finer grid (n=8192) and repeat the two‑phase
   optimisation on the larger problem.

4. **Final random hill‑climb** – a short local search that moves a
   tiny amount of mass between two random positions while preserving
   symmetry.  The best improvement is kept.

All updates keep the coefficients non‑negative, enforce symmetry and
normalise to unit sum, which removes the scaling degree of freedom of
the original objective.  The algorithm prints intermediate objective
values and returns the best sequence found (clipped to the evaluator’s
allowed range).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using a real FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    # keep only the first 2·len(a)-1 entries
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Fast evaluation of the original objective."""
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c / (a.sum() ** 2))


def _symmetrize(a: np.ndarray) -> np.ndarray:
    """Enforce mirror symmetry a_i = a_{n-1-i}."""
    return (a + a[::-1]) * 0.5


def _project_simplex(a: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the probability simplex."""
    s = a.sum()
    if s < 1e-12:
        # fall back to uniform
        return np.full_like(a, 1.0 / a.shape[0])
    return a / s


# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Adam on a log‑sum‑exp surrogate of the max‑convolution.
    Enforces symmetry after each soft‑max conversion.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)               # start in log‑soft‑max space
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05                               # a bit smaller to stay stable
    lam = 0.2                               # initial temperature for smooth‑max
    lam_factor = 1.01                       # very slow increase
    lam_cap = 1e5

    best_seq = global_best_seq
    best_val = global_best_val
    last_improve = time.time()
    it = 0

    while time.time() < deadline:
        # ----- simplex via soft‑max -----
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()
        a = _symmetrize(a)
        a = _project_simplex(a)

        # ----- objective -----
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Adam] it={it:07d} new best {val:.6f}")

        # ----- smooth‑max weights (log‑sum‑exp) -----
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the surrogate -----
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)               # ⟨a,∇⟩ term from the chain rule
        grad_log = a * (grad_a - avg)         # gradient wrt logits

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- temperature schedule -----
        lam = min(lam * lam_factor, lam_cap)

        # ----- occasional jitter if stalled -----
        if time.time() - last_improve > 6.0:
            logits += (np.random.rand(n) - 0.5) * 1e-3
            last_improve = time.time()

        it += 1
        # safety break
        if it > 1_500_000:
            break

    return best_seq, best_val


def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    init_best_seq: np.ndarray | None,
    init_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient descent on the true max‑convolution.
    Symmetry and simplex projection are applied after each update.
    """
    a = init_best_seq if init_best_seq is not None else a0
    a = np.maximum(a, 0.0)
    a = _symmetrize(a)
    a = _project_simplex(a)

    best_seq = a.copy()
    best_val = init_best_val
    eta0 = 0.6
    it = 0
    last_improve = time.time()

    while time.time() < deadline:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Subgrad] it={it:06d} new best {val:.6f}")

        # active set where the convolution attains its maximum
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            # fallback: single maximal index
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # decaying step‑size
        eta = eta0 / np.sqrt(it + 1)

        # exponentiated‑gradient update (keeps a ≥ 0)
        a = a * np.exp(-eta * grad_a)

        # symmetry & projection
        a = _symmetrize(a)
        a = _project_simplex(a)

        if time.time() - last_improve > 12.0:
            # no progress for a while → stop this phase
            break

        it += 1
        # safety break
        if it > 300_000:
            break

    return best_seq, best_val


def _deep_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Run Adam (smooth‑max) followed by sub‑gradient refinement.
    Returns the best vector/value found (also updates the global best).
    """
    # Phase 1 – Adam on smooth‑max surrogate
    best_seq, best_val = _adam_optimize(
        a0,
        n,
        conv_fft_len,
        corr_fft_len,
        deadline,
        global_best_seq,
        global_best_val,
    )
    # Phase 2 – exact sub‑gradient descent
    best_seq, best_val = _subgradient_optimize(
        a0,
        n,
        conv_fft_len,
        corr_fft_len,
        deadline,
        best_seq,
        best_val,
    )
    # Phase 3 – a short second Adam for fine‑tuning (if time left)
    if time.time() < deadline - 5.0:
        best_seq, best_val = _adam_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            best_seq,
            best_val,
        )
    return best_seq, best_val

def _generate_parametric_candidates(n: int, warm_start: np.ndarray | None) -> list[np.ndarray]:
    """
    Assemble a rich pool of symmetric candidates, all normalised to unit sum.
    """
    candidates: list[np.ndarray] = []

    # Warm‑start & cyclic shifts
    if warm_start is not None:
        ws = warm_start.copy()
        candidates.append(ws)
        for shift in (n // 4, n // 2, 3 * n // 4):
            candidates.append(np.roll(ws, shift))

    # Uniform baseline
    candidates.append(np.full(n, 1.0 / n))

    # Random Dirichlets (different concentrations)
    for conc in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        candidates.append(np.random.dirichlet(np.full(n, conc)))

    i = np.arange(n, dtype=np.float64)

    # Edge‑exponential families (mixed with uniform)
    for alpha in np.logspace(-2, 1.0, num=12):
        edge = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
        edge /= edge.sum()
        for w in (0.0, 0.25, 0.5, 0.75, 1.0):
            candidates.append(w * edge + (1.0 - w) / n)

    # Edge‑power‑law families
    for p in np.linspace(0.2, 5.0, num=12):
        edge = (i + 1.0) ** (-p) + (n - i) ** (-p)
        edge /= edge.sum()
        for w in (0.0, 0.4, 0.8, 1.0):
            candidates.append(w * edge + (1.0 - w) / n)

    # Symmetric beta‑shapes
    for alpha in (0.2, 0.4, 0.5, 0.8, 1.2, 2.0):
        shape = (i + 1.0) ** (alpha - 1.0) * (n - i) ** (alpha - 1.0)
        shape = np.maximum(shape, 0.0)
        shape /= shape.sum()
        candidates.append(shape)

    # Raised‑cosine / sine‑power families
    for p in (1, 2, 4, 6):
        shape = np.sin(np.pi * (i + 0.5) / n) ** p
        shape = np.maximum(shape, 0.0)
        shape /= shape.sum()
        candidates.append(shape)

    # Two‑Gaussian edge shapes
    for gamma in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        shape /= shape.sum()
        candidates.append(shape)

    # Arcsine‑type shape
    shape = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    candidates.append(shape)

    return candidates

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:

    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    Returns the best sequence (plain Python list) found before the deadline.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5       # safety margin

    # ------------------------------------------------------------------
    # Primary grid size – start with a moderate resolution
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm‑start: use a previously discovered vector if available
    warm_start = globals().get("height_sequence_1")
    if warm_start is not None:
        ws = np.asarray(warm_start, dtype=np.float64)
        if ws.shape[0] > n:
            ws = ws[:n]
        elif ws.shape[0] < n:
            ws = np.pad(ws, (0, n - ws.shape[0]))
        ws = np.clip(ws, 0.0, None)
        if ws.sum() >= 0.01:
            ws = ws / ws.sum()
        else:
            ws = None
    else:
        ws = None

    # ------------------------------------------------------------------
    # ---- Stage 1: parametric seeding & quick evaluation ----------------
    candidates = _generate_parametric_candidates(n, ws)

    evaluated = []
    for idx, cand in enumerate(candidates):
        if time.time() >= deadline:
            break
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 0.01:
            cand = np.random.dirichlet(np.ones(n))
            s = cand.sum()
        cand = cand / s
        val = _objective(cand, n, conv_fft_len)
        evaluated.append((val, cand))

    evaluated.sort(key=lambda x: x[0])
    top_k = min(6, len(evaluated))
    top_starts = [c for _, c in evaluated[:top_k]]
    print(f"Stage 1: selected {top_k} promising seeds (n={n})")

    # ------------------------------------------------------------------
    # ---- Stage 2: deep optimisation on the selected seeds ---------------
    best_seq: np.ndarray | None = None
    best_val = float("inf")

    for start_id, a0 in enumerate(top_starts):
        if time.time() >= deadline:
            break
        print(f"--- Deep optimisation start #{start_id + 1}/{len(top_starts)} ---")
        # allocate a slice of the remaining time to each start proportionally
        remaining = deadline - time.time()
        per_start_deadline = time.time() + remaining / (len(top_starts) - start_id)
        best_seq, best_val = _deep_optimize(
            a0,
            n,
            conv_fft_len,
            corr_fft_len,
            per_start_deadline,
            best_seq,
            best_val,
        )
        print(f"  after start #{start_id + 1}: best_val={best_val:.6f}")

    # ------------------------------------------------------------------
    # ---- Stage 3: upscale to a finer grid if time permits ----------------
    time_left = deadline - time.time()
    if time_left > 40.0 and best_seq is not None:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        # linear interpolation to the finer grid
        x_old = np.linspace(0, 1, n, endpoint=False)
        x_new = np.linspace(0, 1, n_big, endpoint=False)
        a_big = np.interp(x_new, x_old, best_seq)
        a_big = np.maximum(a_big, 0.0)
        a_big = _project_simplex(a_big)

        print("--- Optimising on larger grid (n=8192) ---")
        best_seq, best_val = _deep_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            best_seq,
            best_val,
        )
        n = n_big
        conv_fft_len = conv_fft_len_big
        corr_fft_len = corr_fft_len_big

    # ------------------------------------------------------------------
    # ---- Stage 4: final random local hill‑climb -------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        a = best_seq.copy()
        best_local_val = best_val
        it = 0
        max_local_iter = 12000
        while time.time() < deadline and it < max_local_iter:
            # Propose a tiny symmetric mass transfer between two random indices
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            # amount transferred is a fraction of the smaller of the two masses
            max_transfer = min(a[i], a[j])
            if max_transfer < 1e-12:
                continue
            delta = (np.random.rand() - 0.5) * 0.5 * max_transfer   # ±25 % of the smaller mass
            if delta > a[i]:
                delta = a[i]
            if -delta > a[j]:
                delta = -a[j]
            a[i] -= delta
            a[j] += delta

            # re‑symmetrise and renormalise
            a = _symmetrize(a)
            a = _project_simplex(a)

            conv = _full_convolution(a, conv_fft_len)
            val = 2.0 * n * conv.max() / (a.sum() ** 2)

            if val < best_local_val:
                best_local_val = val
                best_seq = a.copy()
            else:
                # revert the move (by re‑normalising the previous state)
                a[i] += delta
                a[j] -= delta
                a = _symmetrize(a)
                a = _project_simplex(a)

            it += 1

        best_val = best_local_val
        print(f"Final local refinement finished after {it} attempts, best={best_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback in the extremely unlikely case that nothing was found
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Clip to the evaluator’s allowed range and return as plain list
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    print(f"*** Search complete: best objective ≈ {best_val:.6f} (n={best_seq.shape[0]})")
    return best_seq.tolist()
