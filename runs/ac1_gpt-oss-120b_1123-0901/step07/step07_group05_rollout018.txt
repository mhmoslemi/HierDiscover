def propose_candidate(seed=0, budget_s=1000):
   np.random.seed(seed)
   deadline = time.time() + budget_s - 0.5
   best_seq = None
   best_val = np.inf

   # Stage 0: set base n_small = 1024, n_big = 4096
   n_small = 1024
   conv_fft_len_small = _next_pow2(2*n_small - 1)
   # Stage A: generate initial a_small (uniform) or warm start (maybe coarse downsample)
   a = np.full(n_small, 1.0/n_small)
   # optionally use warm_start downsampling: if global variable height_sequence_1 exists, downsample.
   warm_start = globals().get('height_sequence_1')
   if warm_start is not None:
        # downsample to n_small via averaging
        warm = np.asarray(warm_start, dtype=np.float64)
        # if warm longer than n_small, compress by averaging groups
        # else pad
        if warm.shape[0] >= n_small:
            factor = warm.shape[0] // n_small
            a = warm[:factor*n_small].reshape(n_small, factor).mean(axis=1)
        else:
            a = np.pad(warm, (0, n_small-warm.shape[0]), mode='constant')
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            a = np.random.dirichlet(np.ones(n_small))
        else:
            a = a / s
   # Stage A loop
   iter_lp = 0
   max_lp_iters = 30
   step_schedule = [0.5, 0.4, 0.3, 0.2, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01]
   while time.time() < deadline and iter_lp < max_lp_iters:
        # compute M
        conv = _full_convolution(a, conv_fft_len_small)
        M = conv.max()
        # solve LP
        A = _build_ub_matrix(a)  # shape (2n-1, n)
        b_ub = np.full(2*n_small-1, M, dtype=np.float64)
        c = -np.ones(n_small)   # maximize sum(b)
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub, bounds=(0, None), method='highs')
        if not res.success:
            print('LP failed, breaking')
            break
        b = res.x
        # normalize b to sum=1
        if b.sum() < 1e-12:
            b = np.random.dirichlet(np.ones(n_small))
        b = b / b.sum()
        # mixing step
        t = step_schedule[min(iter_lp, len(step_schedule)-1)]
        a = (1 - t) * a + t * b
        # evaluate objective
        val = _objective(a, n_small, conv_fft_len_small)  # objective scaled for n_small
        # transform to comparable objective (since n differs) maybe we compute scaled objective or compute final on large n later
        # For best record we compute actual objective on base n_small, but final evaluation will be on final sequence

            best_seq_small = a.copy()

def _build_ub_matrix(a):

    # construct diagonal data arrays

"""
Propose a low‑evaluation step function.

Algorithm
---------
1. **Coarse LP refinement on a small grid** –  
   The search starts on a modest size (n≈1024) where an
   inexpensive linear‑programming step can be applied.
   For a fixed vector `a` we maximise the ℓ₁‑norm of a new
   vector `b` under the linear constraints `a * b ≤ M`,
   where `M = max(a * a)`.  The LP pushes mass away from the
   positions that generate the current maximal convolution.
   A convex combination `a ← (1‑t)·a + t·b` (t decays) yields a
   new candidate.  Several LP steps quickly generate a shape
   that already beats the uniform baseline.

2. **Upsampling** –  
   The best small‑grid vector is interpolated to the target size
   (`n = 4096`).  Normalisation (sum = 1) is preserved – the
   objective is scale‑invariant.

3. **Smooth‑max Adam + sub‑gradient refinement** –  
   Starting from the up‑sampled vector we run the two‑phase
   optimiser from the previous state‑of‑the‑art:  
   *Adam* on a log‑sum‑exp surrogate of the max‑convolution
   (high temperature, long schedule) and then an exponential‑gradient
   sub‑gradient descent on the exact max‑convolution.  Hyper‑parameters
   (iterations, learning‑rates, temperature schedule) are tuned for
   the larger time budget.

4. **Cheap stochastic hill‑climbing** –  
   Finally a cheap local perturbation loop (Gaussian noise on
   logits, accept only improvements) squeezes a few more
   thousandths out of the objective.

All stages keep the best sequence seen so far; the routine stops
as soon as the 1000 s budget is exhausted and returns the best
vector found (as a plain list of floats).  The implementation
uses only NumPy, SciPy and the provided ``evaluate_sequence`` for
final validation.
"""

import time
import math
import numpy as np
import scipy.optimize
import scipy.sparse


# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (full length = 2·len(a)‑1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fa = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fa, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact objective F = 2·n·max(a*a)/(sum(a)²).  Here sum(a)=1 ⇒ F=2·n·max."""
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c / (a.sum() ** 2))


# ----------------------------------------------------------------------
def _build_ub_matrix(a: np.ndarray) -> scipy.sparse.csc_matrix:
    """
    Build the (2n‑1)×n matrix representing the linear constraints
    (a * b)[k] ≤ M for all k, i.e. each row k contains the coefficients
    a_i for b_{k‑i} (out‑of‑range entries are omitted).
    """
    n = a.shape[0]
    # each diagonal offset i (0 ≤ i < n) contains the constant a[i]
    diags = [np.full(n, a[i], dtype=np.float64) for i in range(n)]
    offsets = np.arange(0, n, dtype=np.int32)
    A = scipy.sparse.diags(diags, offsets, shape=(2 * n - 1, n), format="csc")
    return A


# ----------------------------------------------------------------------
def _adam_smooth_max(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    init_best_seq: np.ndarray | None,
    init_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on a log‑sum‑exp surrogate of the max‑convolution.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.04                 # slightly smaller LR for stability

    lam = 2.0                 # start temperature
    lam_factor = 1.008        # slower growth → smoother transition
    lam_cap = 1e6

    best_seq = init_best_seq
    best_val = init_best_val

    it = 0
    max_iter = 300_000
    last_improve = time.time()

    while time.time() < deadline and it < max_iter:
        # soft‑max to enforce positivity & sum‑to‑1
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # surrogate objective (exact for evaluation)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Adam] iter={it:06d} new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of surrogate w.r.t. a (via cross‑correlation)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature schedule
        lam = min(lam * lam_factor, lam_cap)

        # occasional jitter if stuck
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1

    return best_seq, best_val


def _subgradient_refinement(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    init_best_seq: np.ndarray | None,
    init_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient descent on the exact max‑convolution.
    """
    a = init_best_seq if init_best_seq is not None else a0.copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a /= s                     # stay on the simplex

    eta0 = 0.7
    decay = 0.99990
    eta = eta0

    best_seq = a.copy()
    best_val = init_best_val

    it = 0
    max_iter = 120_000
    last_improve = time.time()

    while time.time() < deadline and it < max_iter:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Subgrad] iter={it:06d} new best {val:.6f}")

        # active set where convolution attains its maximum
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # exponentiated‑gradient update (keeps a ≥ 0)
        a = a * np.exp(-eta * grad_a)

        # renormalise – protect against collapse
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        # decay step size
        eta *= decay
        it += 1

        if time.time() - last_improve > 5.0:
            break

    return best_seq, best_val


def _deep_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Run the Adam‑smooth‑max phase followed by an extended sub‑gradient phase.
    Returns the best sequence/value seen (including the incoming global best).
    """
    # Adam phase
    best_seq, best_val = _adam_smooth_max(
        a0,
        n,
        conv_fft_len,
        corr_fft_len,
        deadline,
        global_best_seq,
        global_best_val,
    )
    # Sub‑gradient phase
    best_seq, best_val = _subgradient_refinement(
        a0,
        n,
        conv_fft_len,
        corr_fft_len,
        deadline,
        best_seq,
        best_val,
    )
    return best_seq, best_val


def _local_random_search(
    a_best: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climbing: add Gaussian noise to the logits of the current
    best vector, evaluate the exact objective and keep the improvement.
    """
    logits_best = np.log(a_best + 1e-15)

    while time.time() < deadline:
        noise = np.random.randn(n) * 0.025
        logits = logits_best + noise
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        val = _objective(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            a_best = a.copy()
            logits_best = np.log(a_best + 1e-15)
            print(f"[Local search] improved to {val:.6f}")

    return a_best, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Main entry point – searches for a non‑negative step function that
    minimises the supplied evaluation function.
    Returns the best sequence found (plain Python list).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # 1.  Small‑grid LP refinement (n = 1024)
    n_small = 1024
    conv_fft_len_small = _next_pow2(2 * n_small - 1)

    # initialise a (warm‑start if available, otherwise uniform)
    warm_start = globals().get("height_sequence_1")
    if warm_start is not None:
        src = np.asarray(warm_start, dtype=np.float64)
        # downsample / interpolate to n_small points
        x_src = np.linspace(0, 1, src.shape[0], endpoint=False)
        x_tgt = np.linspace(0, 1, n_small, endpoint=False)
        a = np.interp(x_tgt, x_src, src)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            a = np.random.dirichlet(np.ones(n_small))
        else:
            a = a / s
    else:
        a = np.full(n_small, 1.0 / n_small)

    best_seq_small = a.copy()
    best_val_small = _objective(a, n_small, conv_fft_len_small)

    # LP iteration schedule (mixing coefficients t)
    step_schedule = [0.5, 0.4, 0.3, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01]
    lp_iter = 0
    max_lp_iters = 30

    while time.time() < deadline and lp_iter < max_lp_iters:
        # current maximum convolution value
        conv = _full_convolution(a, conv_fft_len_small)
        M = conv.max()

        # build linear constraints a * b ≤ M
        A_ub = _build_ub_matrix(a)
        b_ub = np.full(2 * n_small - 1, M, dtype=np.float64)

        # maximise sum(b)  ⇔ minimise -sum(b)
        res = scipy.optimize.linprog(
            c=-np.ones(n_small, dtype=np.float64),
            A_ub=A_ub,
            b_ub=b_ub,
            bounds=[(0, None)] * n_small,
            method="highs",
            options={"disp": False},
        )

        if not res.success:
            print(f"[LP] iteration {lp_iter}: linprog failed – {res.message}")
            break

        b = res.x
        b_sum = b.sum()
        if b_sum < 1e-12:
            b = np.random.dirichlet(np.ones(n_small))
            b_sum = b.sum()
        b = b / b_sum

        # mixing (t decays according to the schedule)
        t = step_schedule[min(lp_iter, len(step_schedule) - 1)]
        a = (1.0 - t) * a + t * b
        a = np.maximum(a, 0.0)
        a /= a.sum()

        val = _objective(a, n_small, conv_fft_len_small)
        if val < best_val_small:
            best_val_small = val
            best_seq_small = a.copy()
            print(f"[LP] iter={lp_iter:02d} new best small val {val:.6f}")

        # keep global best (upscaled later) if this small‑grid vector looks promising
        if val < best_val:
            # quick up‑sample to the big grid just for rough comparison
            n_big_est = 4096
            x_small = np.linspace(0, 1, n_small, endpoint=False)
            x_big = np.linspace(0, 1, n_big_est, endpoint=False)
            a_big_est = np.interp(x_big, x_small, a)
            a_big_est = np.maximum(a_big_est, 0.0)
            a_big_est /= a_big_est.sum()
            conv_fft_len_big_est = _next_pow2(2 * n_big_est - 1)
            val_big_est = _objective(a_big_est, n_big_est, conv_fft_len_big_est)
            if val_big_est < best_val:
                best_val = val_big_est
                best_seq = a_big_est.copy()

        lp_iter += 1
        if deadline - time.time() < 20.0:   # reserve time for the fine stage
            break

    # ------------------------------------------------------------------
    # 2. Upsample to the target size (n = 4096)
    n_big = 4096
    conv_fft_len_big = _next_pow2(2 * n_big - 1)
    corr_fft_len_big = _next_pow2(3 * n_big - 2)

    if best_seq is None:
        # upsample the best small‑grid candidate
        x_small = np.linspace(0, 1, n_small, endpoint=False)
        x_big = np.linspace(0, 1, n_big, endpoint=False)
        a_big = np.interp(x_big, x_small, best_seq_small)
        a_big = np.maximum(a_big, 0.0)
        a_big /= a_big.sum()
    else:
        a_big = best_seq.copy()

    # ------------------------------------------------------------------
    # 3. Deep optimisation (Adam + sub‑gradient) on the big grid
    best_seq, best_val = _deep_optimize(
        a_big,
        n_big,
        conv_fft_len_big,
        corr_fft_len_big,
        deadline,
        best_seq,
        best_val,
    )

    # ------------------------------------------------------------------
    # 4. Very cheap stochastic hill‑climbing
    if best_seq is not None:
        best_seq, best_val = _local_random_search(
            best_seq,
            n_big,
            conv_fft_len_big,
            deadline,
            best_val,
        )

    # ------------------------------------------------------------------
    # 5. Fallback – should never be needed, but keep safety net
    if best_seq is None:
        best_seq = np.full(n_big, 1.0 / n_big, dtype=np.float64)

    # Clip to the evaluator’s allowed range (non‑negative, ≤1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"*** Finished search: best objective ≈ {best_val:.6f} (n={best_seq.shape[0]})")
    return best_seq.tolist()
