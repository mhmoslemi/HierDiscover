def _generate_parametric_candidates(n, warm_start):
    cand_list = []
    i = np.arange(n)
    # warm start
    if warm_start is not None:
        cand_list.append(warm_start.copy())
        for shift in (n//4, n//2, 3*n//4):
            cand_list.append(np.roll(warm_start, shift))

    # uniform
    cand_list.append(np.full(n, 1.0/n))

    # Dirichlet draws
    for conc in [0.05, 0.1, 0.2, 0.5, 1.0, 2.0]:
        cand_list.append(np.random.dirichlet(np.full(n, conc)))

    # exponential families
    for alpha in np.logspace(-2, 1.0, 12):
        edge = np.exp(-alpha * i) + np.exp(-alpha * (n-1-i))
        edge /= edge.sum()
        for w in [0.0, 0.3, 0.6, 0.9, 1.0]:
            cand_list.append(w*edge + (1-w)/n)

    # power law families
    for p in np.linspace(0.2, 5.0, 12):
        edge = (i+1)**(-p) + (n-i)**(-p)
        edge /= edge.sum()
        for w in [0.0, 0.5, 1.0]:
            cand_list.append(w*edge + (1-w)/n)

    # Gaussian edges
    for sigma in [0.02,0.05,0.1,0.2]:
        shape = np.exp(-0.5*((i-(n-1)/2)/ (sigma*n))**2)  # central
        # also edges: combination
        shape = np.exp(-0.5*((i)/ (sigma*n))**2) + np.exp(-0.5*((n-1-i)/(sigma*n))**2)
        shape /= shape.sum()
        cand_list.append(shape)

    # sine power
    for p in [1,2,3,4,5]:
        shape = np.sin(np.pi*(i+0.5)/n)**p
        shape = np.maximum(shape,0)
        shape /= shape.sum()
        cand_list.append(shape)

    # cosine shape (positive part)
    shape = 1 - np.cos(2*np.pi*i/(n-1))
    shape = np.maximum(shape,0)
    shape /= shape.sum()
    cand_list.append(shape)

    # triangular (already)
    tri = np.minimum(i+1, n-i)
    tri = tri.astype(float)
    tri /= tri.sum()
    cand_list.append(tri)

    # quadratic (already)
    quad = (i+1)*(n-i)
    quad = quad.astype(float)
    quad /= quad.sum()
    cand_list.append(quad)

    return cand_list

"""
Propose a low‑evaluation step‑function.

Algorithm
---------
1.  Build a diverse library of parametric vectors (uniform, Dirichlet,
    edge‑exponential / power‑law, Gaussian‑edges, sinusoidal powers,
    beta‑shaped, simple triangular / quadratic).  All are normalised
    to sum = 1.

2.  Evaluate every candidate with a fast FFT‑based convolution and keep the
    best K starts (K≈12).

3.  Deep optimisation for each start:
        * Adam on a log‑sum‑exp smooth‑max surrogate (very long run,
          temperature grows slowly).
        * Exponentiated‑gradient sub‑gradient steps on the true
          max‑convolution.
        * Projected‑gradient descent on the exact objective with
          back‑tracking line‑search.
    Every phase stops as soon as the global deadline is reached.

4.  Mixture refinement – mix the current best with a few simple shapes
    (uniform, triangular, sine‑power, …) and search the mixing weight
    on a fine grid.

5.  Stochastic local perturbations (tiny Gaussian jitter + projection)
    while time permits.

6.  If enough time remains, interpolate the best vector to a finer grid
    (n = 8192) and repeat steps 3‑5.

The routine never exceeds the allotted 1000 s, prints progress and
returns the best sequence (list of non‑negative floats) it discovered.
"""

import time
import numpy as np

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]                     # reverse for correlation
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1               # n‑1
    return corr[start:start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Fast evaluation of the objective (identical to evaluate_sequence)."""
    conv = _full_convolution(a, conv_fft_len)
    return float(2.0 * n * conv.max() / (a.sum() ** 2))


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection onto the probability simplex (non‑negative, sum=1). """
    if v.ndim != 1:                # defensive
        raise ValueError("v must be 1‑d")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

def _adam_smooth_max(a0, n, conv_fft_len, corr_fft_len, deadline,
                     best_seq, best_val):
    """Adam optimiser on a log‑sum‑exp surrogate of the max‑convolution."""
    eps = 1e-15
    logits = np.log(a0 + eps)                     # work in log‑space
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.04

    lam = 0.5
    lam_factor = 1.02
    lam_cap = 1e9

    last_improve = time.time()
    it = 0

    while time.time() < deadline:
        # ----- soft‑max to obtain a point on the simplex -----
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # ----- surrogate value (smooth max) -----
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Adam n={n} it={it:07d}] new best {val:.6f}")

        # ----- smooth‑max weights (log‑sum‑exp) -----
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the surrogate (chain‑rule through soft‑max) -----
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)   # d/da (smooth max)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)                         # gradient w.r.t. logits

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- temperature schedule -----
        lam = min(lam * lam_factor, lam_cap)

        # ----- occasional random noise if stuck -----
        if time.time() - last_improve > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1

    return best_seq, best_val


def _subgradient_refinement(a0, n, conv_fft_len, corr_fft_len,
                            deadline, best_seq, best_val):
    """Exponentiated‑gradient sub‑gradient steps on the true max‑convolution."""
    a = best_seq if best_seq is not None else a0.copy()
    a = np.maximum(a, 0.0)
    a /= a.sum()                         # stay on the simplex

    eta = 0.55
    last_improve = time.time()
    it = 0

    while time.time() < deadline:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Subgrad n={n} it={it:07d}] new best {val:.6f}")

        # ---- active set (where the max is attained) ----
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # ---- multiplicative / exponentiated‑gradient update ----
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:                     # avoid collapse to zero
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        # ---- decay / early stop ----
        it += 1
        eta *= 0.9999
        if time.time() - last_improve > 5.0:
            break

    return best_seq, best_val


def _projected_gradient_refine(a0, n, conv_fft_len, corr_fft_len,
                               deadline, best_seq, best_val):
    """Projected gradient descent on the exact objective."""
    a = a0.copy()
    a = np.maximum(a, 0.0)
    a /= a.sum()

    lr = 0.18
    last_improve = time.time()
    it = 0

    while time.time() < deadline:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[ProjGrad n={n} it={it:07d}] new best {val:.6f}")

        # active mask (tolerance to numerical errors)
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)   # ∂M/∂a
        step = -lr * grad_a
        a_candidate = _proj_simplex(a + step)

        # line‑search‑like acceptance
        conv_cand = _full_convolution(a_candidate, conv_fft_len)
        val_cand = 2.0 * n * conv_cand.max() / (a_candidate.sum() ** 2)

        if val_cand < val:
            a = a_candidate
            lr = min(lr * 1.02, 1.0)       # gently grow step when moving downhill
        else:
            lr *= 0.7
            if lr < 1e-5:
                break

        it += 1
        if time.time() - last_improve > 5.0:
            break
        lr *= 0.99998                     # slow decay overall

    return best_seq, best_val


def _deep_optimize(a0, n, conv_fft_len, corr_fft_len,
                   deadline, global_best_seq, global_best_val):
    """Run the three‑stage refinement pipeline for a single start."""
    best_seq, best_val = _adam_smooth_max(
        a0, n, conv_fft_len, corr_fft_len, deadline,
        global_best_seq, global_best_val
    )
    best_seq, best_val = _subgradient_refinement(
        a0, n, conv_fft_len, corr_fft_len, deadline,
        best_seq, best_val
    )
    best_seq, best_val = _projected_gradient_refine(
        a0, n, conv_fft_len, corr_fft_len, deadline,
        best_seq, best_val
    )
    return best_seq, best_val

def _generate_parametric_candidates(n, warm_start):
    """Create a heterogeneous pool of candidate vectors (all sum≈1, non‑negative)."""
    cand = []
    i = np.arange(n, dtype=np.float64)

    # warm‑start and simple rotations (if provided)
    if warm_start is not None:
        cand.append(warm_start.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm_start, shift))

    # uniform baseline
    cand.append(np.full(n, 1.0 / n))

    # random Dirichlet draws (different concentrations)
    for conc in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 4.0):
        cand.append(np.random.dirichlet(np.full(n, conc)))

    # edge‑exponential families (optional uniform mix)
    for alpha in np.logspace(-2, 1.0, num=14):
        edge = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
        edge /= edge.sum()
        for w in (0.0, 0.3, 0.6, 1.0):
            cand.append(w * edge + (1.0 - w) / n)

    # edge‑power‑law families
    for p in np.linspace(0.2, 5.0, num=14):
        edge = (i + 1.0) ** (-p) + (n - i) ** (-p)
        edge /= edge.sum()
        for w in (0.0, 0.5, 1.0):
            cand.append(w * edge + (1.0 - w) / n)

    # two‑Gaussian edges (different widths)
    for sigma in (0.02, 0.05, 0.1, 0.2, 0.4):
        shape = np.exp(-0.5 * ((i) / (sigma * n)) ** 2) + \
                np.exp(-0.5 * ((n - 1 - i) / (sigma * n)) ** 2)
        shape /= shape.sum()
        cand.append(shape)

    # sinusoidal half‑wave raised to a power
    for p in (1, 2, 3, 4, 5):
        shape = np.sin(np.pi * (i + 0.5) / n) ** p
        shape = np.maximum(shape, 0.0)
        shape /= shape.sum()
        cand.append(shape)

    # cosine‑positive part
    shape = 1.0 - np.cos(2.0 * np.pi * i / (n - 1))
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand.append(shape)

    # simple triangular / quadratic
    tri = np.minimum(i + 1, n - i)
    tri = tri.astype(np.float64)
    tri /= tri.sum()
    cand.append(tri)

    quad = (i + 1) * (n - i)
    quad = quad.astype(np.float64)
    quad /= quad.sum()
    cand.append(quad)

    return cand

def _mix_with_simple_shapes(best_seq, n, deadline, best_val):
    """
    Try convex combinations of the current best vector with a few
    elementary shapes.  Scan the mixing weight on a fine grid.
    """
    i = np.arange(n, dtype=np.float64)

    simple = []

    # uniform
    simple.append(np.full(n, 1.0 / n))

    # triangular
    tri = np.minimum(i + 1, n - i)
    tri = tri.astype(np.float64)
    tri /= tri.sum()
    simple.append(tri)

    # quadratic
    quad = (i + 1) * (n - i)
    quad = quad.astype(np.float64)
    quad /= quad.sum()
    simple.append(quad)

    # sine‑power (p=3)
    s3 = np.sin(np.pi * (i + 0.5) / n) ** 3
    s3 /= s3.sum()
    simple.append(s3)

    # edge‑exponential (α=0.1)
    alpha = 0.1
    edge = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
    edge /= edge.sum()
    simple.append(edge)

    # cosine‑positive part
    cos = 1.0 - np.cos(2.0 * np.pi * i / (n - 1))
    cos = np.maximum(cos, 0.0)
    cos /= cos.sum()
    simple.append(cos)

    lambdas = np.linspace(0.0, 1.0, 41)   # 0 % … 100 % in steps of 2.5 %

    for shape in simple:
        if time.time() >= deadline:
            break
        shape = shape / shape.sum()
        for lam in lambdas:
            if time.time() >= deadline:
                break
            cand = (1.0 - lam) * best_seq + lam * shape
            cand = np.maximum(cand, 0.0)
            s = cand.sum()
            if s < 1e-12:
                continue
            cand /= s
            val = evaluate_sequence(cand.tolist())
            if val < best_val:
                best_val = val
                best_seq = cand.copy()
                print(f"[Mix] λ={lam:.2f} improved to {val:.6f}")

    return best_seq, best_val


def _local_perturbations(base_seq, n, conv_fft_len, corr_fft_len,
                         deadline, best_seq, best_val):
    """Small Gaussian jitter + brief sub‑gradient refinement."""
    a0 = base_seq.copy()
    eps = 0.015  # jitter amplitude
    while time.time() < deadline - 1.0:
        a_pert = a0 + eps * (np.random.rand(n) - 0.5)
        a_pert = np.maximum(a_pert, 0.0)
        s = a_pert.sum()
        if s < 1e-12:
            continue
        a_pert /= s
        val = evaluate_sequence(a_pert.tolist())
        if val < best_val:
            best_val = val
            best_seq = a_pert.copy()
            a0 = best_seq.copy()   # centre next jitter around the new best
            print(f"[Perturb] improved to {val:.6f}")

    return best_seq, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Entry point – searches for a step‑function with minimal evaluation score.
    Returns the best sequence as a plain Python list of non‑negative floats.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.7   # tiny safety margin

    best_seq = None
    best_val = float("inf")

    # --------------------------------------------------------------
    # Base resolution
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Warm‑start from a previous construction, if available
    warm_start = globals().get("height_sequence_1")
    if warm_start is not None:
        warm_start = np.asarray(warm_start, dtype=np.float64)
        if warm_start.shape[0] > n:
            warm_start = warm_start[:n]
        elif warm_start.shape[0] < n:
            warm_start = np.pad(warm_start, (0, n - warm_start.shape[0]))
        warm_start = np.maximum(warm_start, 0.0)
        if warm_start.sum() >= 0.01:
            warm_start = warm_start / warm_start.sum()
        else:
            warm_start = None
    else:
        warm_start = None

    # --------------------------------------------------------------
    # Stage 1 – generate a large and diverse pool of parametric candidates
    all_cand = _generate_parametric_candidates(n, warm_start)

    cand_vals = []
    for cand in all_cand:
        if time.time() >= deadline:
            break
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 1e-12:
            cand = np.random.dirichlet(np.ones(n))
            s = cand.sum()
        cand /= s
        val = _objective(cand, n, conv_fft_len)
        cand_vals.append((val, cand))

    # keep the most promising starts
    cand_vals.sort(key=lambda x: x[0])
    top_k = 12
    top_cands = [c for _, c in cand_vals[:top_k]]

    # --------------------------------------------------------------
    # Stage 2 – deep optimisation of each promising start
    for idx, start_vec in enumerate(top_cands):
        if time.time() >= deadline:
            break
        print(f"\n--- Deep optimisation on start #{idx + 1}/{len(top_cands)} ---")
        seq, val = _deep_optimize(
            start_vec, n, conv_fft_len, corr_fft_len,
            deadline, best_seq, best_val,
        )
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"*** Global best now {best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 3 – convex‑combination refinement with simple shapes
    if best_seq is not None and time.time() < deadline - 5.0:
        print("\n--- Mixing with simple shapes ---")
        best_seq, best_val = _mix_with_simple_shapes(
            best_seq, n, deadline, best_val
        )

    # --------------------------------------------------------------
    # Stage 4 – stochastic local perturbations
    if best_seq is not None and time.time() < deadline - 8.0:
        print("\n--- Local stochastic perturbations ---")
        best_seq, best_val = _local_perturbations(
            best_seq, n, conv_fft_len, corr_fft_len,
            deadline, best_seq, best_val
        )

    # --------------------------------------------------------------
    # Stage 5 – upscale to a larger grid if time permits
    time_left = deadline - time.time()
    if time_left > 45.0:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        print("\n--- Upscaling to n=8192 and re‑optimising ---")
        # linear interpolation of the current best
        x_old = np.linspace(0.0, 1.0, n, endpoint=False)
        x_big = np.linspace(0.0, 1.0, n_big, endpoint=False)
        a_big = np.interp(x_big, x_old, best_seq)
        a_big = np.maximum(a_big, 0.0)
        a_big /= a_big.sum()

        seq_big, val_big = _deep_optimize(
            a_big, n_big, conv_fft_len_big, corr_fft_len_big,
            deadline, best_seq, best_val,
        )
        if val_big < best_val:
            best_val = val_big
            best_seq = seq_big.copy()
            n = n_big
            print(f"*** Improved after up‑scaling to {best_val:.6f}")

    # --------------------------------------------------------------
    # Fallback – uniform vector (should never be needed)
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Clip to the bounds enforced by evaluate_sequence
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"\n*** Finished search: best objective ≈ {best_val:.6f} (n={best_seq.shape[0]})")
    return best_seq.tolist()
