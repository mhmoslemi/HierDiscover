"""
Propose a step‑function with a very small max‑self‑convolution.

Algorithm
---------
1.  Build a diverse set of candidates for a modest size (n≈1024):
    * warm‑start (if a previous construction exists)
    * uniform, Dirichlet with many concentrations
    * several analytical families (exponential, power‑law, beta, triangular,…)
    * random “spectral” vectors obtained from random Fourier phases
2.  Evaluate every candidate with a fast FFT‑based convolution and keep the
    K≈12 best.
3.  For each start run a three‑phase refinement:
    * Adam optimiser on a smooth‑max surrogate of the max‑convolution
    * Sub‑gradient (exponentiated‑gradient) on the true max‑convolution
    * Projected gradient descent on the exact objective (simplex projection)
   All three phases stop automatically when the global deadline is reached.
4.  After the gradient based refinement perform a **linear‑programming boost**:
    maximise the ℓ₁‑norm of a new vector `g` under the linear constraints
    `f * g ≤ M` where `M` is the current maximal self‑convolution of `f`.
    The LP is solved with SciPy’s HiGHS interior‑point solver.
    The resulting `g` is renormalised and replaces the current best if it
    improves the objective.
5.  If enough time remains, interpolate the best small‑grid vector to a
    larger grid (n=8192) and run the same three‑phase refinement (without the
    LP step – the LP would be too large).  The larger grid gives a finer
    representation and usually yields a strictly smaller bound.
6.  Return the best vector found (clipped to the evaluator’s allowed range).

The whole routine respects the 1000 s budget, prints progress and finally
returns a plain ``list[float]``.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)−1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for gradients.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k−(n−1)]   for i=0…n−1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Fast evaluation of the benchmark objective (no safety checks)."""
    conv = _full_convolution(a, conv_fft_len)
    return float(2.0 * n * conv.max() / (a.sum() ** 2))


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    if v.ndim != 1:
        raise ValueError("v must be 1‑d")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

def _adam_smooth_max(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """Adam optimiser on a log‑sum‑exp surrogate of max(conv)."""
    eps = 1e-15
    logits = np.log(a0 + eps)            # work in log‑space (=soft‑max)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05

    lam = 0.5                # start temperature
    lam_factor = 1.03        # exponential schedule
    lam_cap = 1e8

    last_improve = time.time()
    it = 0
    max_iter = 300_000

    while time.time() < deadline and it < max_iter:
        # ----- simplex via soft‑max -----
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()          # guaranteed to sum to 1

        # ----- surrogate evaluation -----
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)   # a.sum() == 1 → 2·n·max_c

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Adam n={n} it={it:07d}] new best {val:.6f}")

        # ----- smooth‑max weights (log‑sum‑exp) -----
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the surrogate (chain rule through soft‑max) -----
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)        # ∂/∂logits (soft‑max chain)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- temperature schedule -----
        lam = min(lam * lam_factor, lam_cap)

        # ----- occasional noise if stuck -----
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1

    return best_seq, best_val


def _subgradient_refinement(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient sub‑gradient steps on the true max‑convolution."""
    a = best_seq if best_seq is not None else a0.copy()
    a = np.maximum(a, 0.0)
    a /= a.sum()                     # stay on the simplex

    eta = 0.6
    last_improve = time.time()
    max_iter = 200_000

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Subgrad n={n} it={it:07d}] new best {val:.6f}")

        # ----- active set (tight constraints) -----
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # ----- exponentiated‑gradient update (mirrored to simplex) -----
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:                     # avoid collapse
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        # ----- early stop if stalled -----
        if time.time() - last_improve > 5.0:
            break

        eta *= 0.9999                    # slow decay

    return best_seq, best_val


def _projected_gradient_refine(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """Projected gradient descent on the exact objective."""
    a = a0.copy()
    a = np.maximum(a, 0.0)
    a /= a.sum()

    lr = 0.2
    last_improve = time.time()
    max_iter = 150_000

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[ProjGrad n={n} it={it:07d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)   # dM/da
        step = -lr * grad_a
        a_candidate = _proj_simplex(a + step)

        conv_cand = _full_convolution(a_candidate, conv_fft_len)
        val_cand = 2.0 * n * conv_cand.max() / (a_candidate.sum() ** 2)

        if val_cand < val:
            a = a_candidate
            lr = min(lr * 1.02, 1.0)        # enlarge step when moving downhill
        else:
            lr *= 0.7
            if lr < 1e-5:
                break

        if time.time() - last_improve > 5.0:
            break

        lr *= 0.99998                     # very gentle decay

    return best_seq, best_val


def _deep_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """Full three‑phase optimisation pipeline for a single start."""
    best_seq, best_val = _adam_smooth_max(
        a0,
        n,
        conv_fft_len,
        corr_fft_len,
        deadline,
        global_best_seq,
        global_best_val,
    )
    best_seq, best_val = _subgradient_refinement(
        a0,
        n,
        conv_fft_len,
        corr_fft_len,
        deadline,
        best_seq,
        best_val,
    )
    best_seq, best_val = _projected_gradient_refine(
        a0,
        n,
        conv_fft_len,
        corr_fft_len,
        deadline,
        best_seq,
        best_val,
    )
    return best_seq, best_val

def _generate_parametric_candidates(n: int, warm_start: np.ndarray | None):
    """
    Produce a diverse set of non‑negative vectors (already normalised to sum≈1).
    """
    cand_list = []

    # warm‑start and simple cyclic shifts
    if warm_start is not None:
        cand_list.append(warm_start.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand_list.append(np.roll(warm_start, shift))

    # uniform baseline
    cand_list.append(np.full(n, 1.0 / n))

    # random Dirichlet draws (various concentrations)
    for conc in (0.02, 0.05, 0.1, 0.3, 0.5, 1.0, 2.0, 4.0):
        cand_list.append(np.random.dirichlet(np.full(n, conc)))

    i = np.arange(n, dtype=np.float64)

    # edge‑exponential families (optional uniform mix)
    for alpha in np.logspace(-2, 1.0, num=14):
        edge = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
        edge /= edge.sum()
        for w in (0.0, 0.3, 0.6, 1.0):
            shape = w * edge + (1.0 - w) / n
            cand_list.append(shape)

    # edge‑power‑law families
    for p in np.linspace(0.2, 4.0, num=14):
        edge = (i + 1.0) ** (-p) + (n - i) ** (-p)
        edge /= edge.sum()
        for w in (0.0, 0.5, 1.0):
            shape = w * edge + (1.0 - w) / n
            cand_list.append(shape)

    # two‑Gaussian edges (different widths)
    for gamma in (0.04, 0.1, 0.25, 0.5, 1.0, 2.0):
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        shape /= shape.sum()
        cand_list.append(shape)

    # sinusoidal half‑wave (positive part)
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand_list.append(shape)

    # beta‑shaped families
    for alpha in (0.2, 0.5, 1.0, 2.0, 5.0):
        for beta in (0.2, 0.5, 1.0, 2.0, 5.0):
            edge = (i + 1) ** (alpha - 1) * (n - i) ** (beta - 1)
            edge = np.maximum(edge, 0.0)
            edge /= edge.sum()
            cand_list.append(edge)

    # simple triangular / quadratic shapes
    tri = np.minimum(i + 1, n - i)
    tri = tri.astype(np.float64)
    tri /= tri.sum()
    cand_list.append(tri)

    quad = (i + 1) * (n - i)
    quad = quad.astype(np.float64)
    quad /= quad.sum()
    cand_list.append(quad)

    return cand_list


def _random_spectral_candidate(n: int) -> np.ndarray:
    """
    Generate a non‑negative vector by randomising Fourier phases,
    taking the absolute value of the inverse transform and normalising.
    """
    phi = np.random.rand(n) * 2 * np.pi
    freq = np.exp(1j * phi)
    # enforce Hermitian symmetry to obtain a real signal
    # keep first element (DC) real, mirror the rest
    half = n // 2
    freq[1:half] = np.conj(freq[-1:-(half):-1])
    if n % 2 == 0:
        freq[half] = np.real(freq[half])  # Nyquist term must be real
    a = np.abs(np.fft.ifft(freq)).real
    a += 1e-12                     # avoid exact zeros
    a /= a.sum()
    return a

def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the (2n‑1)×n matrix A s.t. A·g = f * g (linear convolution).
    Each row k contains the coefficients f_{k‑j} for the corresponding
    shift of g_j (zero‑padded outside the interval).
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for k in range(rows):
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        i_range = np.arange(i_min, i_max + 1, dtype=np.int64)
        j_range = k - i_range
        A[k, j_range] = f[i_range]
    return A


def _lp_boost(f: np.ndarray, M: float, deadline: float) -> np.ndarray | None:
    """
    Linear‑programming boost:
    maximise sum(g) subject to f*g ≤ M (entry‑wise).
    Returns a new vector g (scaled to have sum = sqrt(2·n)) or None if LP fails.
    """
    n = f.shape[0]
    # Normalise f to the standard sum √(2n) required for the LP constraints.
    target_sum = math.sqrt(2.0 * n)
    if f.sum() == 0:
        return None
    f_norm = f * (target_sum / f.sum())

    A = _build_conv_matrix(f_norm)
    b = np.full(2 * n - 1, M * (1.0 + 1e-12))   # tiny slack for numerical safety
    c = -np.ones(n)                             # maximise ∑g ⇒ minimise −∑g
    bounds = [(0, None) for _ in range(n)]

    # Solve the LP – HiGHS is fast and works well for medium‑sized problems.
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method='highs', options={'presolve': True, 'time_limit': max(0, deadline - time.time())})
    if not res.success or res.x is None:
        return None

    g = res.x
    g_sum = g.sum()
    if g_sum < 1e-12:
        return None
    # Rescale to the required sum √(2n)
    g = g * (target_sum / g_sum)
    return g

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Entry point – searches for a step‑function with minimal evaluation score.
    Returns the best sequence as a plain Python list of non‑negative floats.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # 1) Small‑grid stage (n≈1024) – rich candidate generation + LP boost
    # ------------------------------------------------------------------
    n_small = 1024
    conv_fft_len_s = _next_pow2(2 * n_small - 1)
    corr_fft_len_s = _next_pow2(3 * n_small - 2)

    # warm‑start from a previously stored construction, if available
    warm_start = globals().get("height_sequence_1")
    if warm_start is not None:
        warm_start = np.asarray(warm_start, dtype=np.float64)
        if warm_start.size != n_small:
            # interpolate / truncate to the required size
            x_old = np.linspace(0, 1, warm_start.size, endpoint=False)
            x_new = np.linspace(0, 1, n_small, endpoint=False)
            warm_start = np.interp(x_new, x_old, warm_start)
        warm_start = np.maximum(warm_start, 0.0)
        if warm_start.sum() < 1e-12:
            warm_start = None
        else:
            warm_start /= warm_start.sum()
    else:
        warm_start = None

    # generate a large pool of parametric candidates
    candidates = _generate_parametric_candidates(n_small, warm_start)

    # add a handful of random spectral candidates
    for _ in range(30):
        candidates.append(_random_spectral_candidate(n_small))

    # random Dirichlet draws with very low concentration (spiky) and high concentration
    for conc in (0.01, 0.03, 0.07, 0.2, 5.0, 10.0):
        candidates.append(np.random.dirichlet(np.full(n_small, conc)))

    # ------------------------------------------------------------------
    # Quick evaluation of every candidate (linear cost)
    # ------------------------------------------------------------------
    cand_vals = []
    for idx, cand in enumerate(candidates):
        if time.time() >= deadline:
            break
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 1e-12:
            cand = np.random.dirichlet(np.ones(n_small))
            s = cand.sum()
        cand = cand / s                         # normalise to sum 1 (probability simplex)
        val = _objective(cand, n_small, conv_fft_len_s)
        cand_vals.append((val, cand))
        if (idx + 1) % 100 == 0:
            print(f"[Candidate generation] evaluated {idx + 1}/{len(candidates)} candidates")

    # keep the most promising starts
    cand_vals.sort(key=lambda x: x[0])
    top_k = 12
    top_candidates = [c for _, c in cand_vals[:top_k]]
    print(f"[Selection] kept top {len(top_candidates)} candidates (out of {len(cand_vals)})")

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # 2) Deep optimisation of each start (gradient + LP boost)
    # ------------------------------------------------------------------
    for start_idx, start_vec in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        print(f"\n=== Optimising start #{start_idx + 1}/{len(top_candidates)} ===")
        # gradient‑based pipeline
        best_seq, best_val = _deep_optimize(
            start_vec,
            n_small,
            conv_fft_len_s,
            corr_fft_len_s,
            deadline,
            best_seq,
            best_val,
        )
        # ------------------------------------------------------------------
        # LP boost (only if some time left)
        # ------------------------------------------------------------------
        if time.time() < deadline - 5.0:
            conv = _full_convolution(best_seq, conv_fft_len_s)
            M = conv.max()
            g_lp = _lp_boost(best_seq, M, deadline)
            if g_lp is not None:
                # evaluate the LP‑improved vector
                val_lp = _objective(g_lp, n_small, conv_fft_len_s)
                print(f"[LP boost] candidate value {val_lp:.6f}")
                if val_lp < best_val:
                    best_seq, best_val = g_lp, val_lp
                    print(f"[LP boost] new global best {val_lp:.6f}")

    # ------------------------------------------------------------------
    # 3) Optional stochastic local perturbations (cheap cheap)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 10.0:
        print("\n--- Local stochastic perturbations ---")
        a0 = best_seq.copy()
        while time.time() < deadline:
            eps = 0.02
            a_pert = a0 + eps * (np.random.rand(n_small) - 0.5)
            a_pert = np.maximum(a_pert, 0.0)
            if a_pert.sum() < 1e-12:
                continue
            a_pert /= a_pert.sum()
            # short sub‑gradient run (few hundred iterations)
            a_ref, val_ref = _subgradient_refinement(
                a_pert,
                n_small,
                conv_fft_len_s,
                corr_fft_len_s,
                deadline,
                best_seq,
                best_val,
            )
            if val_ref < best_val:
                best_val = val_ref
                best_seq = a_ref.copy()
                a0 = best_seq.copy()
                print(f"[Perturb] new best {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4) Upscale to a larger grid (n=8192) if time allows
    # ------------------------------------------------------------------
    time_left = deadline - time.time()
    if time_left > 30.0:               # we need at least a little head‑room
        n_big = 8192
        conv_fft_len_b = _next_pow2(2 * n_big - 1)
        corr_fft_len_b = _next_pow2(3 * n_big - 2)

        # linear interpolation of the best small‑grid vector
        x_small = np.linspace(0, 1, n_small, endpoint=False)
        x_big = np.linspace(0, 1, n_big, endpoint=False)
        a_big = np.interp(x_big, x_small, best_seq)
        a_big = np.maximum(a_big, 0.0)
        if a_big.sum() < 1e-12:
            a_big = np.random.dirichlet(np.ones(n_big))
        a_big /= a_big.sum()           # probability simplex (sum=1)

        print("\n--- Optimising on larger grid (n=8192) ---")
        best_seq_big, best_val_big = _deep_optimize(
            a_big,
            n_big,
            conv_fft_len_b,
            corr_fft_len_b,
            deadline,
            best_seq,
            best_val,
        )
        # keep the larger‑grid result if it improved the bound
        if best_seq_big is not None and best_val_big < best_val:
            best_seq, best_val = best_seq_big, best_val_big
            print(f"[Upscale] improved to {best_val:.6f} on n={n_big}")

    # ------------------------------------------------------------------
    # 5) Final sanity – ensure non‑negativity and clip to allowed range
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback: uniform vector (should never happen)
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"\n*** Search finished – best objective ≈ {best_val:.6f} (length={best_seq.shape[0]}) ***")
    return best_seq.tolist()
