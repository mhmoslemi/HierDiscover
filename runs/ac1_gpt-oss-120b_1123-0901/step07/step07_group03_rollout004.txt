def _build_conv_matrix(a: np.ndarray) -> scipy.sparse.csr_matrix:
    n = a.shape[0]
    rows = 2*n - 1
    # We'll create a list of (data, offset) for diags
    diags_data = []
    offsets = []
    for i, coeff in enumerate(a):
        diags_data.append(np.full(n, coeff, dtype=np.float64))
        offsets.append(i)
    A = scipy.sparse.diags(diags_data, offsets, shape=(rows, n), format='csr')
    return A
```

"""
Mixed linear‑programming + gradient refinement search.

The core idea is an alternating scheme:
1. Start from a good “anti‑pair” ordering (or a warm start).
2. Given the current vector a (normalised to sum = 1) compute M = max(a∗a).
   Solve the linear programme
        maximise   1ᵀ b
        s.t.       a∗b ≤ M ,   b ≥ 0 .
   Scale b back to a probability vector g = b / Σb and replace a by a
   convex combination (1‑t)·a + t·g.  Repeating a few times dramatically
   reduces the maximal convolution value.
3. Refine the best candidate with a smooth‑max Adam phase, followed by an
   exact‑max exponentiated‑gradient sub‑gradient phase.
4. Finish with a cheap stochastic hill‑climbing (swap / mass transfer)
   to squeeze out any remaining improvement.

The routine respects the 1000 s budget, prints progress, and finally
returns a list of non‑negative floats ready for evaluate_sequence().
"""
import time
import numpy as np
import scipy.optimize
import scipy.sparse

# ----------------------------------------------------------------------
# Basic FFT utilities ---------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a). Returns a vector of length len(a)."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
# Exact objective (no safety belts) ------------------------------------
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(a*a) / (sum(a)^2).  Assumes a is non‑negative."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    conv = _full_convolution(a, _next_pow2(2 * a.shape[0] - 1))
    return 2.0 * a.shape[0] * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Candidate generators --------------------------------------------------
def _anti_pair_arrangement(vals: np.ndarray) -> np.ndarray:
    """Largest opposite smallest, second‑largest opposite second‑smallest, …"""
    n = len(vals)
    sorted_vals = np.sort(vals)[::-1]          # descending
    a = np.empty(n, dtype=np.float64)
    half = n // 2
    a[:half] = sorted_vals[:half]             # biggest half left side
    a[-half:] = sorted_vals[-half:][::-1]     # smallest half right side
    if n % 2 == 1:
        a[half] = sorted_vals[half]
    return a


def _anti_pair_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    vals = np.random.dirichlet(alpha * np.ones(n))
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return _anti_pair_arrangement(vals)


def _warm_start(n: int) -> np.ndarray | None:
    """If a previous solution (height_sequence_1) exists, adapt it to length n."""
    warm = globals().get("height_sequence_1")
    if warm is None:
        return None
    warm = np.asarray(warm, dtype=np.float64)
    if warm.shape[0] > n:
        warm = warm[:n]
    elif warm.shape[0] < n:
        warm = np.pad(warm, (0, n - warm.shape[0]))
    warm = np.maximum(warm, 0.0)
    if warm.sum() < 0.01:
        return None
    return warm / warm.sum()


# ----------------------------------------------------------------------
# Linear‑programming helpers --------------------------------------------
def _build_conv_matrix(a: np.ndarray) -> scipy.sparse.csr_matrix:
    """Build the (2n‑1)×n matrix A with (A·b)[k] = (a∗b)[k]."""
    n = a.shape[0]
    rows = 2 * n - 1
    # each offset i contributes a[i] on the diagonal i
    diags = [np.full(n, coeff, dtype=np.float64) for coeff in a]
    offsets = list(range(n))
    A = scipy.sparse.diags(diags, offsets, shape=(rows, n), format="csr")
    return A


def _lp_improvement(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Perform a few LP‑based improvement steps.
    a is assumed to be normalised to sum = 1.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_val = _objective(best_a)

    # mixing parameter – start aggressive, then decay
    t = 0.6
    max_iters = 12
    for it in range(max_iters):
        if time.time() >= deadline - 30.0:   # leave some slack for later phases
            break

        # current bound M
        conv = _full_convolution(best_a, _next_pow2(2 * n - 1))
        M = conv.max()

        # LP: maximise 1ᵀ b  subject to A·b ≤ M, b ≥ 0
        A = _build_conv_matrix(best_a)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)
        c = -np.ones(n, dtype=np.float64)   # maximise sum(b)

        res = scipy.optimize.linprog(
            c,
            A_ub=A,
            b_ub=b_ub,
            bounds=[(0, None)] * n,
            method="highs",
            options={"presolve": True},
        )
        if not res.success:
            print(f"[LP] iteration {it} failed: {res.message}")
            break

        b = res.x
        if b.sum() < 1e-12:
            print("[LP] trivial solution, aborting")
            break

        # scale to a probability vector
        g = b / b.sum()

        # convex mix
        new_a = (1.0 - t) * best_a + t * g
        new_a = np.maximum(new_a, 0.0)
        new_a /= new_a.sum()

        val = _objective(new_a)
        if val < best_val:
            best_val = val
            best_a = new_a
            print(f"[LP] iter {it:02d} improved value {best_val:.6f}")

        # shrink step size a little each round
        t = max(t * 0.85, 0.05)

    return best_a, best_val


# ----------------------------------------------------------------------
# Adam optimisation on a smooth‑max surrogate ---------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 250_000,
    lr: float = 0.07,
    lam_start: float = 1.0,
    lam_factor: float = 1.07,
) -> tuple[np.ndarray, float]:
    """
    Adam on the log‑sum‑exp smooth‑max surrogate.
    Returns the best exact (non‑smooth) objective encountered.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lam = lam_start

    best_a = a0.copy()
    best_val = _objective(a0)

    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max → simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # surrogate value and exact max
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val_exact = 2.0 * n * max_c / (a.sum() ** 2)

        if val_exact < best_val:
            best_val = val_exact
            best_a = a.copy()
            if it % 4000 == 0:
                print(f"[Adam it={it:06d}] new best exact {best_val:.6f}")

        # smooth‑max weights (log‑sum‑exp)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient wrt a
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # raise temperature → closer to true max
        lam = min(lam * lam_factor, 1e6)

        # occasional jitter to escape plateaus
        if it % 5_000 == 0 and (deadline - time.time()) > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact‑max exponentiated‑gradient sub‑gradient phase --------------------
def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 250_000,
    eta0: float = 0.7,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient descent using exponentiated updates.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a /= a.sum()

    best_a = a.copy()
    best_val = _objective(best_a)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # indices attaining the maximum (allow tiny tolerance)
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # step‑size decays like 1/√t
        eta = eta0 / np.sqrt(it + 1)

        # multiplicative (exponentiated‑gradient) update
        a = a * np.exp(-eta * grad_a)

        # renormalise
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val

# Simple stochastic local search (swap / transfer) ----------------------
def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 150_000,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing style hill climbing:
      – swap two entries OR transfer a fraction of mass.
    Accept if improvement or with probability exp(-Δ/Temp) where
    Temp linearly decays from 0.01 to 0.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a)

    start_temp = 0.01
    it = 0
    while time.time() < deadline and it < max_moves:
        if np.random.rand() < 0.5:          # swap two positions
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            a2 = a.copy()
            a2[i], a2[j] = a2[j], a2[i]
        else:                               # transfer a fraction from i to j
            i = np.random.randint(n)
            if a[i] < 1e-14:
                continue
            j = np.random.randint(n)
            if i == j:
                continue
            delta = a[i] * (np.random.rand() * 0.4)   # up to 40 % move
            a2 = a.copy()
            a2[i] -= delta
            a2[j] += delta

        val2 = _objective(a2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
        else:
            temp = start_temp * (1.0 - (it / max_moves))
            if temp > 0.0 and np.random.rand() < np.exp(-(val2 - best_val) / temp):
                a = a2

        if it % 5_000 == 0:
            print(f"[Local move {it:06d}] best {best_val:.6f}")

        it += 1

    return best_a, best_val

# Main entry point ------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function minimising
        F(a) = 2·n·max(a∗a) / (sum(a)²).

    The function runs a mixed LP/gradient/local‑search scheme within the
    allotted time budget and returns the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # keep a small safety margin

    # ------------------------------------------------------------------
    # Choose a moderate dimension (large enough for a good bound,
    # small enough for fast LP solves).  2048 works well in practice.
    n = 2048
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # 1) Initialise a (probability vector, i.e. sum = 1)
    a0 = _warm_start(n)
    if a0 is None:
        a0 = _anti_pair_candidate(n, alpha=0.7)

    a0 = np.maximum(a0, 0.0)
    if a0.sum() < 0.01:
        a0 = np.random.dirichlet(np.ones(n))
    a0 = a0 / a0.sum()

    best_a = a0.copy()
    best_val = _objective(best_a)
    print(f"[Init] objective = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2) LP‑based improvement (few aggressive iterations)
    lp_deadline = min(deadline - 200.0, time.time() + 30.0)  # reserve time for later phases
    best_a, best_val = _lp_improvement(best_a, lp_deadline)
    print(f"[After LP] best value = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3) Adam phase on smooth‑max surrogate
    adam_deadline = min(deadline - 120.0, time.time() + 250.0)
    if time.time() < adam_deadline:
        a_adam, val_adam = _adam_optimize(
            best_a,
            n,
            conv_fft_len,
            corr_fft_len,
            adam_deadline,
            max_iters=300_000,
            lr=0.07,
            lam_start=1.0,
            lam_factor=1.07,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_a = a_adam
            print(f"[Adam] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4) Exact‑max sub‑gradient refinement
    subgrad_deadline = min(deadline - 60.0, time.time() + 300.0)
    if time.time() < subgrad_deadline:
        a_sub, val_sub = _subgradient_optimize(
            best_a,
            n,
            conv_fft_len,
            corr_fft_len,
            subgrad_deadline,
            max_iters=300_000,
            eta0=0.7,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_a = a_sub
            print(f"[Subgrad] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5) Cheap stochastic local search (swap / transfer moves)
    if time.time() < deadline - 5.0:
        local_deadline = deadline - 2.0
        a_loc, val_loc = _local_random_search(
            best_a,
            n,
            conv_fft_len,
            local_deadline,
            max_moves=120_000,
        )
        if val_loc < best_val:
            best_val = val_loc
            best_a = a_loc
            print(f"[Local] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6) Optional up‑sampling to a finer grid if time permits
    if time.time() < deadline - 30.0:
        n_big = 4096
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        # linear interpolation from current best
        x_old = np.linspace(0.0, 1.0, n, endpoint=False)
        x_new = np.linspace(0.0, 1.0, n_big, endpoint=False)
        a_big = np.interp(x_new, x_old, best_a)

        a_big = np.maximum(a_big, 0.0)
        a_big /= a_big.sum()

        # brief Adam + sub‑gradient on the larger grid
        big_adam_deadline = min(deadline - 15.0, time.time() + 120.0)
        a_big_adam, val_big_adam = _adam_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            big_adam_deadline,
            max_iters=200_000,
            lr=0.07,
            lam_start=1.0,
            lam_factor=1.07,
        )
        big_sub_deadline = min(deadline, time.time() + 120.0)
        a_big_sub, val_big_sub = _subgradient_optimize(
            a_big_adam,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            big_sub_deadline,
            max_iters=200_000,
            eta0=0.7,
        )
        if val_big_sub < best_val:
            best_val = val_big_sub
            best_a = a_big_sub
            print(f"[Big grid] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 7) Final safety checks and return
    best_a = np.clip(best_a, 0.0, 1000.0)   # evaluator caps at 1000.0
    print(f"Final best value {best_val:.6f} (length {len(best_a)})")
    return best_a.tolist()
