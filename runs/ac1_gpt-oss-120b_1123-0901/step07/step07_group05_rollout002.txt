"""
Propose a low‑evaluation step function.

Algorithm
---------
1. **Parameter sweep** – generate a modest pool of diverse analytic
   shapes (uniform, Dirichlet draws, edge‑exponential/power‑law,
   asymmetric two‑sided exponentials/power‑laws, linear / triangular,
   sinusoidal, sparse spikes).  Evaluate them with the *exact*
   objective and keep the K best as warm‑starts.

2. **Smooth‑max L‑BFGS** – for each warm‑start optimise a
   log‑sum‑exp surrogate of ``max(a∗a)``.  The variables are the
   logits of a soft‑max (guaranteeing non‑negativity and unit sum).
   A schedule of increasing temperatures ``λ`` (2 → 1000) is used:
   at each temperature a limited L‑BFGS run (via ``scipy.optimize.fmin_l_bfgs_b``)
   drives the surrogate down while the exact objective is tracked.
   The gradient of the surrogate is obtained analytically using FFT‑based
   convolution and cross‑correlation, the same formulas as in the reference
   implementation.

3. **Exponential‑gradient refinement** – after the L‑BFGS phase a few
   exponentiated‑gradient sub‑gradient steps on the *exact* ``max`` are
   performed; this eliminates the remaining bias of the smooth surrogate.

4. **Cheap stochastic hill‑climbing** – a final random‑logit perturbation loop
   searches locally for any remaining improvement.

5. **Optional up‑sampling** – if time permits, linearly interpolate the best
   vector to a larger grid (n = 8192) and repeat steps 2‑4.

The whole routine respects the 1000 s budget, logs progress, and finally
returns the best sequence found (plain Python list).  All numbers are
non‑negative and ≤ 1000, satisfying the evaluator’s checks.
"""

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (length 2*len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) of lengths (2n‑1) and n.
    Returns a vector of length n with:
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)],  i = 0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


def _objective_exact(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact evaluation 2·n·max(a∗a) / (sum(a)²).  Assumes a ≥ 0."""
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    s = a.sum()
    return 2.0 * n * max_c / (s * s)

def _generate_candidates(n: int, warm_start: np.ndarray | None):
    """Return a list of diverse non‑negative vectors (unit sum)."""
    candidates = []

    # Warm‑start and its circular shifts
    if warm_start is not None:
        candidates.append(warm_start.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            candidates.append(np.roll(warm_start, shift))

    # Uniform baseline
    candidates.append(np.full(n, 1.0 / n))

    # Random Dirichlet draws with many concentrations
    for conc in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(8):
            cand = np.random.dirichlet(np.full(n, conc))
            candidates.append(cand)

    i = np.arange(n, dtype=np.float64)

    # Symmetric edge‑exponential family
    for alpha in np.logspace(-2, 1.0, num=10):
        edge = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
        edge /= edge.sum()
        for w in (0.0, 0.3, 0.6, 1.0):
            cand = w * edge + (1.0 - w) / n
            cand /= cand.sum()
            candidates.append(cand)

    # Symmetric power‑law family
    for p in np.linspace(0.2, 6.0, num=10):
        edge = (i + 1.0) ** (-p) + (n - i) ** (-p)
        edge /= edge.sum()
        for w in (0.0, 0.4, 0.8, 1.0):
            cand = w * edge + (1.0 - w) / n
            cand /= cand.sum()
            candidates.append(cand)

    # Asymmetric two‑sided exponentials
    alphas = np.logspace(-2, 1.0, num=6)
    w_vals = np.linspace(0.0, 1.0, 5)
    for wL in w_vals:
        for wR in w_vals:
            if wL + wR > 1.0:
                continue
            wU = 1.0 - wL - wR
            for aL in alphas:
                for aR in alphas:
                    left = np.exp(-aL * i)
                    right = np.exp(-aR * (n - 1 - i))
                    cand = wL * left + wR * right + wU / n
                    cand /= cand.sum()
                    candidates.append(cand)

    # Linear / triangular shapes
    lin = np.linspace(1.0, 0.0, n)
    tri = np.concatenate([np.linspace(0, 1, n // 2, endpoint=False),
                          np.linspace(1, 0, n - n // 2)])
    for base in (lin, tri):
        base /= base.sum()
        for w in (0.0, 0.4, 0.8):
            cand = w * base + (1.0 - w) / n
            cand /= cand.sum()
            candidates.append(cand)

    # Half‑wave sinusoid
    sin = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    sin = np.maximum(sin, 0.0)
    sin /= sin.sum()
    candidates.append(sin)

    # Sparse spike vectors
    for spikes in (3, 5, 8):
        for _ in range(12):
            idx = np.random.choice(n, size=spikes, replace=False)
            weights = np.random.dirichlet(np.ones(spikes))
            vec = np.zeros(n)
            vec[idx] = weights
            vec /= vec.sum()
            candidates.append(vec)

    return candidates

def _smooth_max_l_bfgs(logits0: np.ndarray,
                       lam: float,
                       n: int,
                       conv_fft_len: int,
                       corr_fft_len: int,
                       max_iter: int,
                       deadline: float,
                       best_val: float,
                       best_seq: np.ndarray):
    """
    Run a limited L‑BFGS minimise the log‑sum‑exp surrogate
    of max(a∗a).  Returns updated best_seq / best_val.
    """
    eps = 1e-15

    def fun_grad(y):
        """Return (value, gradient) for the smooth surrogate."""
        mx = y.max()
        exp_y = np.exp(y - mx)
        a = exp_y / exp_y.sum()                # soft‑max → a ≥0, Σa = 1

        conv = _full_convolution(a, conv_fft_len)   # length 2n‑1
        max_c = conv.max()
        # log‑sum‑exp (stable)
        exp_shift = np.exp(lam * (conv - max_c))
        sum_exp = exp_shift.sum()
        smooth_max = max_c + np.log(sum_exp) / lam
        val = 2.0 * n * smooth_max                # exact denominator = 1

        # soft‑max weights for gradient
        w = exp_shift / sum_exp
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through soft‑max
        avg = np.dot(a, grad_a)
        grad_y = a * (grad_a - avg)
        return val, grad_y.astype(np.float64)

    # fmin_l_bfgs_b expects a function returning (f, g) when fprime=None
    # Use a small maxfun / maxiter so we can respect the global deadline.
    # If we run out of time the outer loop will abort further calls.
    try:
        res, f_min, info = fmin_l_bfgs_b(func=fun_grad,
                                        x0=logits0,
                                        maxiter=max_iter,
                                        factr=1e7,
                                        pgtol=1e-12,
                                        iprint=-1)
    except Exception as e:
        # In rare cases L‑BFGS may fail; just skip this run.
        return best_seq, best_val

    # Extract final a
    mx = res.max()
    exp_res = np.exp(res - mx)
    a_opt = exp_res / exp_res.sum()
    exact_val = _objective_exact(a_opt, n, conv_fft_len)

    if exact_val < best_val:
        best_val = exact_val
        best_seq = a_opt.copy()
        print(f"[L‑BFGS λ={lam:.1f}] new best exact {exact_val:.6f}")

    return best_seq, best_val

def _subgradient_refine(a0: np.ndarray,
                        n: int,
                        conv_fft_len: int,
                        corr_fft_len: int,
                        deadline: float,
                        best_seq: np.ndarray,
                        best_val: float,
                        max_iter: int = 25000) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient steps on the *exact* max‑convolution.
    """
    a = a0.copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a /= s

    eta = 0.8
    decay = 0.99995
    last_improve = time.time()

    for it in range(max_iter):
        if time.time() >= deadline:
            break
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Sub‑grad] iter={it:06d} new best {val:.6f}")

        # active set (indices attaining the maximum)
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        a = a * np.exp(-eta * grad_a)

        # renormalise
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        eta *= decay

        if time.time() - last_improve > 6.0:
            break

    return best_seq, best_val

def _local_random_search(a_best: np.ndarray,
                         n: int,
                         conv_fft_len: int,
                         deadline: float,
                         best_val: float) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climbing: perturb logits with Gaussian noise,
    keep improvements on the exact objective.
    """
    logits_best = np.log(a_best + 1e-15)

    while time.time() < deadline:
        noise = np.random.randn(n) * 0.03
        logits = logits_best + noise
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        val = _objective_exact(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            a_best = a.copy()
            logits_best = np.log(a_best + 1e-15)
            print(f"[Local] improved to {val:.6f}")

    return a_best, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step function with minimal evaluation.
    Returns the best sequence found (plain Python list).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Choose problem size (larger n gives more degrees of freedom).
    # 8192 is a sweet spot for the remaining budget; fall back to 4096
    # if we are already close to the time limit.
    n = 8192 if (deadline - time.time()) > 300 else 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm‑start from a previously discovered construction, if any.
    warm_start = globals().get("height_sequence_1")
    if warm_start is not None:
        warm_start = np.asarray(warm_start, dtype=np.float64)
        if warm_start.shape[0] > n:
            warm_start = warm_start[:n]
        elif warm_start.shape[0] < n:
            warm_start = np.pad(warm_start, (0, n - warm_start.shape[0]))
        warm_start = np.maximum(warm_start, 0.0)
        s = warm_start.sum()
        if s >= 0.01:
            warm_start = warm_start / s
        else:
            warm_start = None
    else:
        warm_start = None

    # ------------------------------------------------------------------
    # Stage 1 – generate a pool, keep the top few.
    all_cands = _generate_candidates(n, warm_start)

    cand_vals = []
    for idx, cand in enumerate(all_cands):
        if time.time() >= deadline:
            break
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 0.01:
            cand = np.random.dirichlet(np.ones(n))
            s = cand.sum()
        cand = cand / s
        val = _objective_exact(cand, n, conv_fft_len)
        cand_vals.append((val, cand))

    cand_vals.sort(key=lambda x: x[0])
    top_k = 8
    top_candidates = [c for _, c in cand_vals[:top_k]]
    print(f"Stage 1: evaluated {len(cand_vals)} candidates, best = {cand_vals[0][0]:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – smooth‑max L‑BFGS on each selected start, multiple λ.
    lam_schedule = [2.0, 5.0, 10.0, 20.0, 50.0, 100.0, 200.0, 500.0, 1000.0]

    for start_id, a0 in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        print(f"--- L‑BFGS optimisation start #{start_id + 1}/{len(top_candidates)} ---")
        # initialise logits from the start vector
        logits = np.log(a0 + 1e-15)

        for lam in lam_schedule:
            if time.time() >= deadline:
                break
            # give each λ a modest iteration budget (max 120)
            max_iter = 120
            best_seq, best_val = _smooth_max_l_bfgs(logits,
                                                    lam,
                                                    n,
                                                    conv_fft_len,
                                                    corr_fft_len,
                                                    max_iter,
                                                    deadline,
                                                    best_val,
                                                    best_seq)
            # use the final logits of the previous λ as the start for the next
            if best_seq is not None:
                logits = np.log(best_seq + 1e-15)

        # sub‑gradient refinement on the best result from this start
        if best_seq is not None:
            best_seq, best_val = _subgradient_refine(best_seq,
                                                     n,
                                                     conv_fft_len,
                                                     corr_fft_len,
                                                     deadline,
                                                     best_seq,
                                                     best_val,
                                                     max_iter=30000)

    # ------------------------------------------------------------------
    # Stage 3 – cheap stochastic hill‑climbing from the overall best.
    if best_seq is not None:
        print("--- Local random refinement ---")
        best_seq, best_val = _local_random_search(best_seq,
                                                  n,
                                                  conv_fft_len,
                                                  deadline,
                                                  best_val)

    # ------------------------------------------------------------------
    # Stage 4 – optional up‑sampling to a larger grid if time permits.
    time_left = deadline - time.time()
    if time_left > 90.0 and n == 4096:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        # linear interpolation of the current best to the finer grid
        if best_seq is not None:
            x_old = np.linspace(0, 1, n, endpoint=False)
            x_new = np.linspace(0, 1, n_big, endpoint=False)
            a_big = np.interp(x_new, x_old, best_seq)
            a_big = np.maximum(a_big, 0.0)
            a_big /= a_big.sum()
        else:
            a_big = np.random.dirichlet(np.ones(n_big))

        print("--- Optimising on larger grid (n=8192) ---")
        # repeat the same pipeline on the bigger grid (but with fewer restarts)
        # Warm‑start L‑BFGS directly on the interpolated vector
        logits_big = np.log(a_big + 1e-15)
        for lam in lam_schedule:
            if time.time() >= deadline:
                break
            best_seq_big, best_val_big = _smooth_max_l_bfgs(logits_big,
                                                            lam,
                                                            n_big,
                                                            conv_fft_len_big,
                                                            corr_fft_len_big,
                                                            max_iter=100,
                                                            deadline=deadline,
                                                            best_val=best_val,
                                                            best_seq=best_seq if best_seq is not None else None)
            if best_seq_big is not None:
                logits_big = np.log(best_seq_big + 1e-15)
                best_seq = best_seq_big
                best_val = best_val_big

        if best_seq is not None:
            # final sub‑gradient refinement on the larger grid
            best_seq, best_val = _subgradient_refine(best_seq,
                                                     n_big,
                                                     conv_fft_len_big,
                                                     corr_fft_len_big,
                                                     deadline,
                                                     best_seq,
                                                     best_val,
                                                     max_iter=20000)
            n = n_big
            conv_fft_len = conv_fft_len_big
            corr_fft_len = corr_fft_len_big

    # ------------------------------------------------------------------
    # Fallback – if nothing succeeded, return a uniform vector.
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Clip to the evaluator’s admissible range.
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"*** Finished search: best objective ≈ {best_val:.6f} (n={best_seq.shape[0]})")
    return best_seq.tolist()
