def cross_corr_fft(w, a, fft_len):
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = len(a) - 1
    return corr[start:start+len(a)]
```

def adam_optimize(a0, n, conv_fft_len, corr_fft_len, deadline, best_seq, best_val):
    # a0 normalized
    eps = 1e-12
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1=0.9; beta2=0.999; eps_adam=1e-8
    lr=0.1
    lam=2.0
    lam_factor=1.04
    lam_cap=1e9
    it=0
    last_improve = time.time()
    max_iter = 1_000_000

    while time.time()<deadline and it<max_iter:
        # softmax to get a
        mx = logits.max()
        exp_shift = np.exp(logits - mx)
        a = exp_shift / exp_shift.sum()
        # compute conv
        conv = full_convolution_fft(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum()**2)  # but a sum =1

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Adam n={n} it={it:d}] new best {val:.6f}")

        # smooth max weights
        # shift conv so max is zero for numerical stability
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(conv, 1.0/conv.size)
        else:
            w /= w_sum

        # gradient for smooth max: grad_a = 2 * cross_correlation(w, a)
        grad_a = 2.0 * cross_corr_fft(w, a, corr_fft_len)
        # gradient w.r.t logits: chain rule for softmax: a * (grad_a - <grad_a, a>)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_log
        v = beta2 * v + (1 - beta2) * (grad_log ** 2)
        m_hat = m / (1 - beta1**(it+1))
        v_hat = v / (1 - beta2**(it+1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature schedule
        lam = min(lam * lam_factor, lam_cap)

        # occasional noise injection if stuck
        if time.time() - last_improve > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-3

        it += 1

    return best_seq, best_val
```

Use additive gradient step with step size adapt; project onto simplex each iteration.

We'll compute active set (indices where conv near max_c within tolerance 1e-12 perhaps). Compute w uniform over active (as used). Then grad = 2 * cross_corr_fft(w, a). Then step = -lr * grad. Update a = proj_simplex(a + step). Evaluate new metric and adapt lr accordingly. Use max_iters ~500k.

```
def proj_gradient_optimize(a0, n, conv_fft_len, corr_fft_len, deadline, best_seq, best_val):
    a = a0.copy()
    # ensure simplex
    a = np.maximum(a,0.0)
    s = a.sum()
    if s == 0:
        a = np.full(n, 1.0/n)
    else:
        a /= s

    lr = 0.3
    it = 0
    max_iter = 500_000
    last_improve = time.time()
    while time.time()<deadline and it<max_iter:
        conv = full_convolution_fft(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum()**2)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[ProjGrad n={n} it={it:d}] new best {val:.6f}")
        # active set
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()
        # gradient
        grad = 2.0 * cross_corr_fft(w, a, corr_fft_len)
        # descent step
        a_candidate = proj_simplex(a - lr * grad)
        # evaluate a_candidate
        conv_cand = full_convolution_fft(a_candidate, conv_fft_len)
        max_cand = conv_cand.max()
        val_cand = 2.0 * n * max_cand / (a_candidate.sum()**2)
        if val_cand < val:
            a = a_candidate
            lr = min(lr * 1.02, 1.0)
        else:
            lr *= 0.7
            if lr < 1e-6:
                break
        if time.time() - last_improve > 6.0:
            break
        it += 1
        lr *= 0.9999
    return best_seq, best_val
```

def generate_candidates(n, warm_start=None):
    cand = []
    # warm start and rotated versions
    if warm_start is not None:
        w = warm_start.copy()
        cand.append(w)
        for shift in (n//4, n//2, 3*n//4):
            cand.append(np.roll(w, shift))
    # uniform
    cand.append(np.full(n, 1.0/n))
    # random Dirichlet concentrations
    for conc in (0.01,0.05,0.1,0.2,0.3,0.5,1.0,2.0,5.0):
        cand.append(np.random.dirichlet(np.full(n, conc)))
    # edge-exponential
    i = np.arange(n, dtype=np.float64)
    for alpha in np.logspace(-2, 0.5, 12):
        edge = np.exp(-alpha * i) + np.exp(-alpha * (n-1-i))
        edge /= edge.sum()
        for wmix in (0.0, 0.2, 0.5, 0.8, 1.0):
            cand.append(wmix * edge + (1-wmix) * (1.0/n))
    # edge power-law
    for p in np.linspace(0.2, 5.0, 12):
        edge = (i+1)**(-p) + (n-i)**(-p)
        edge /= edge.sum()
        for wmix in (0.0, 0.3, 0.7, 1.0):
            cand.append(wmix * edge + (1-wmix) * (1.0/n))
    # beta shapes
    for a in (0.2,0.5,1.0,2.0,5.0,10.0):
        for b in (0.2,0.5,1.0,2.0,5.0,10.0):
            beta = (i+1)**(a-1) * (n-i)**(b-1)
            beta = np.maximum(beta, 0.0)
            beta /= beta.sum()
            cand.append(beta)
    # semi-circle shape
    semi = np.sqrt(i * (n-1-i))
    semi = np.maximum(semi, 0.0)
    semi /= semi.sum()
    cand.append(semi)
    # sin half-wave
    sine = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    sine = np.maximum(sine, 0.0)
    sine /= sine.sum()
    cand.append(sine)
    # triangular
    tri = np.minimum(i+1, n-i)
    tri = tri.astype(np.float64)
    tri /= tri.sum()
    cand.append(tri)
    # quadratic
    quad = (i+1)*(n-i)
    quad = quad.astype(np.float64)
    quad /= quad.sum()
    cand.append(quad)
    return cand
```

"""
Propose a low‑evaluation step‑function (≈1000 s budget).

Strategy
--------
1. **Coarse search on a tiny grid** (n≈256).  A very rich
   parametric library (uniform, Dirichlet, edge‑exponential,
   power‑law, beta‑shaped, semi‑circle, sinusoid, triangular,
   quadratic) is generated.  All candidates are normalized to
   sum = 1 and evaluated with a fast FFT‑based convolution.
   The 10 best shapes are kept.

2. **Intensive local optimisation on the tiny grid** – for every
   top‑candidate we run a long Adam optimiser on a
   log‑sum‑exp surrogate of the max‑convolution, followed by a
   projected‑gradient minimisation of the exact objective.
   Because n is tiny the optimiser can afford many
   iterations (hundreds of thousands).

3. **Upscale the best tiny shape** to a large grid (n=8192) by
   linear interpolation, add a few generic large‑grid starts
   (uniform, edge‑exponential, beta‑shapes) and keep the 3
   most promising.

4. **Deep optimisation on the large grid** – same Adam → projected
   gradient pipeline, but with iteration caps tuned to the larger
   dimension.  Most of the remaining time is spent here.

5. **Stochastic local perturbations** around the current best
   (Gaussian jitter + a short projected‑gradient refine) to escape
   possible local minima.

All steps respect the global deadline; the function finally
returns the best sequence (clipped to the evaluator’s allowed range).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_corr_fft(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex (sum = 1,
    all components ≥ 0).  Returns the projected vector.
    """
    if v.ndim != 1:
        raise ValueError("v must be 1‑d")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


def _metric(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact evaluation of the objective (sum(a) is assumed ≈1)."""
    conv = _full_convolution_fft(a, conv_fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_c / (s ** 2)


# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   conv_fft_len: int,
                   corr_fft_len: int,
                   deadline: float,
                   best_seq: np.ndarray | None,
                   best_val: float,
                   max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on a log‑sum‑exp surrogate of max(conv).
    """
    eps = 1e-12
    logits = np.log(a0 + eps)            # soft‑max parameters
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.12

    lam = 2.0                # starting temperature
    lam_factor = 1.04
    lam_cap = 1e9

    it = 0
    last_improve = time.time()

    while time.time() < deadline and it < max_iters:
        # soft‑max → a (automatically on simplex)
        mx = logits.max()
        exp_shift = np.exp(logits - mx)
        a = exp_shift / exp_shift.sum()

        # surrogate evaluation
        conv = _full_convolution_fft(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Adam n={n} it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w /= w_sum

        # gradient of smooth‑max (chain‑rule through soft‑max)
        grad_a = 2.0 * _cross_corr_fft(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature schedule
        lam = min(lam * lam_factor, lam_cap)

        # occasional jitter if stuck
        if time.time() - last_improve > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-3
            last_improve = time.time()

        it += 1

    return best_seq, best_val


def _proj_gradient_optimize(a0: np.ndarray,
                            n: int,
                            conv_fft_len: int,
                            corr_fft_len: int,
                            deadline: float,
                            best_seq: np.ndarray | None,
                            best_val: float,
                            max_iters: int = 150_000) -> tuple[np.ndarray, float]:
    """
    Projected‑gradient descent on the exact max‑convolution objective.
    """
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s

    lr = 0.25
    it = 0
    last_improve = time.time()

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution_fft(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[ProjGrad n={n} it={it:06d}] new best {val:.6f}")

        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-13, rtol=0.0)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()

        # gradient of the exact max (sub‑gradient)
        grad = 2.0 * _cross_corr_fft(w, a, corr_fft_len)

        # descent step + simplex projection
        a_candidate = _proj_simplex(a - lr * grad)

        # accept if improvement, otherwise shrink step size
        conv_c = _full_convolution_fft(a_candidate, conv_fft_len)
        val_c = 2.0 * n * conv_c.max() / (a_candidate.sum() ** 2)

        if val_c < val:
            a = a_candidate
            lr = min(lr * 1.02, 1.0)
        else:
            lr *= 0.68
            if lr < 1e-6:
                break

        # early‑stop if no progress for a while
        if time.time() - last_improve > 6.0:
            break

        it += 1
        lr *= 0.99995

    return best_seq, best_val


def _deep_optimize(start_vec: np.ndarray,
                   n: int,
                   conv_fft_len: int,
                   corr_fft_len: int,
                   deadline: float,
                   global_best_seq: np.ndarray | None,
                   global_best_val: float) -> tuple[np.ndarray, float]:
    """
    Full pipeline for a single start: Adam → projected‑gradient.
    Iteration caps are scaled with the problem size.
    """
    # iteration budgets (heuristic)
    if n <= 512:
        adam_iters, grad_iters = 400_000, 300_000
    else:
        adam_iters, grad_iters = 120_000, 120_000

    best_seq, best_val = _adam_optimize(
        start_vec,
        n,
        conv_fft_len,
        corr_fft_len,
        deadline,
        global_best_seq,
        global_best_val,
        max_iters=adam_iters,
    )
    best_seq, best_val = _proj_gradient_optimize(
        best_seq if best_seq is not None else start_vec,
        n,
        conv_fft_len,
        corr_fft_len,
        deadline,
        global_best_seq,
        best_val,
        max_iters=grad_iters,
    )
    return best_seq, best_val

def _generate_candidates(n: int, warm_start: np.ndarray | None = None) -> list[np.ndarray]:
    """
    Produce a diverse library of parametric vectors for a given size.
    All vectors are normalised to sum≈1 and contain only non‑negative entries.
    """
    cand = []

    # 0️⃣ Warm start + rotations
    if warm_start is not None:
        ws = warm_start.copy()
        cand.append(ws)
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(ws, shift))

    # 1️⃣ Uniform baseline
    cand.append(np.full(n, 1.0 / n))

    # 2️⃣ Random Dirichlet draws (various concentrations)
    for conc in (0.01, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0, 10.0):
        cand.append(np.random.dirichlet(np.full(n, conc)))

    i = np.arange(n, dtype=np.float64)

    # 3️⃣ Edge‑exponential families (wide range of decays)
    for alpha in np.logspace(-3, 2, 12):          # 0.001 … 100
        edge = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
        edge /= edge.sum()
        for mix in (0.0, 0.25, 0.5, 0.75, 1.0):
            cand.append(mix * edge + (1.0 - mix) / n)

    # 4️⃣ Edge‑power‑law families
    for p in np.linspace(0.2, 10.0, 12):
        edge = (i + 1.0) ** (-p) + (n - i) ** (-p)
        edge /= edge.sum()
        for mix in (0.0, 0.3, 0.6, 1.0):
            cand.append(mix * edge + (1.0 - mix) / n)

    # 5️⃣ Beta‑shaped families (very flexible)
    beta_vals = (0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0, 10.0, 20.0)
    for a in beta_vals:
        for b in beta_vals:
            shape = (i + 1) ** (a - 1) * (n - i) ** (b - 1)
            shape = np.maximum(shape, 0.0)
            shape_sum = shape.sum()
            if shape_sum > 0:
                shape /= shape_sum
                cand.append(shape)

    # 6️⃣ Semi‑circle (√(x·(1−x))) – often very flat convolution
    semi = np.sqrt(i * (n - 1 - i))
    semi = np.maximum(semi, 0.0)
    semi /= semi.sum()
    cand.append(semi)

    # 7️⃣ Positive half‑sine wave
    sine = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    sine = np.maximum(sine, 0.0)
    sine /= sine.sum()
    cand.append(sine)

    # 8️⃣ Simple triangular / quadratic shapes
    tri = np.minimum(i + 1, n - i)
    tri = tri.astype(np.float64)
    tri /= tri.sum()
    cand.append(tri)

    quad = (i + 1) * (n - i)
    quad = quad.astype(np.float64)
    quad /= quad.sum()
    cand.append(quad)

    return cand

def _local_perturbations(base_seq: np.ndarray,
                         n: int,
                         conv_fft_len: int,
                         corr_fft_len: int,
                         deadline: float,
                         best_seq: np.ndarray | None,
                         best_val: float) -> tuple[np.ndarray, float]:
    """
    Stochastic local search: jitter + a short projected‑gradient refine.
    """
    a_center = base_seq.copy()
    while time.time() < deadline:
        # small Gaussian jitter
        eps = 0.02
        a_pert = a_center + eps * (np.random.rand(n) - 0.5)
        a_pert = np.maximum(a_pert, 0.0)
        s = a_pert.sum()
        if s < 1e-12:
            continue
        a_pert /= s

        # brief refinement (few thousand steps)
        a_ref, val_ref = _proj_gradient_optimize(
            a_pert,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            best_seq,
            best_val,
            max_iters=6000,
        )
        if val_ref < best_val:
            best_val = val_ref
            best_seq = a_ref.copy()
            a_center = a_ref.copy()
            print(f"[Perturb] new best {best_val:.6f}")
        else:
            # slowly drift back towards the centre to avoid runaway
            a_center = 0.9 * a_center + 0.1 * a_pert

    return best_seq, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """Entry point – searches for a step‑function with minimal evaluation score."""
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.3  # small safety margin

    # ------------------------------------------------------------------
    # 1️⃣   Tiny grid search (n=256)
    n_small = 256
    conv_fft_small, corr_fft_small = _next_pow2(2 * n_small - 1), _next_pow2(3 * n_small - 2)

    warm_start = globals().get("height_sequence_1")
    if warm_start is not None:
        warm_start = np.asarray(warm_start, dtype=np.float64)
        if warm_start.shape[0] != n_small:
            if warm_start.shape[0] > n_small:
                warm_start = warm_start[:n_small]
            else:
                warm_start = np.pad(warm_start, (0, n_small - warm_start.shape[0]))
        warm_start = np.maximum(warm_start, 0.0)
        if warm_start.sum() > 1e-12:
            warm_start /= warm_start.sum()
        else:
            warm_start = None
    else:
        warm_start = None

    candidates_small = _generate_candidates(n_small, warm_start)

    # Quick evaluation of all tiny candidates
    scores_small = []
    for cand in candidates_small:
        if time.time() >= deadline:
            break
        c = np.maximum(cand, 0.0)
        s = c.sum()
        if s < 1e-12:
            continue
        c /= s
        val = _metric(c, n_small, conv_fft_small)
        scores_small.append((val, c))

    # Keep a handful of promising starts
    scores_small.sort(key=lambda x: x[0])
    top_small = [c for _, c in scores_small[:10]]

    # ------------------------------------------------------------------
    # 2️⃣   Intensive optimisation on the small grid
    best_seq_global = None
    best_val_global = float("inf")

    for idx, start_vec in enumerate(top_small):
        if time.time() >= deadline:
            break
        print(f"\n--- Small‑grid deep optimisation #{idx + 1}/{len(top_small)} ---")
        seq, val = _deep_optimize(
            start_vec,
            n_small,
            conv_fft_small,
            corr_fft_small,
            deadline,
            best_seq_global,
            best_val_global,
        )
        if seq is not None and val < best_val_global:
            best_val_global = val
            best_seq_global = seq

    # ------------------------------------------------------------------
    # 3️⃣   Upscale to the large grid (n=8192) if time permits
    time_left = deadline - time.time()
    if time_left < 30.0 or best_seq_global is None:
        # fallback – just return the best tiny vector interpolated to a reasonable size
        result = np.clip(best_seq_global, 0.0, 1000.0) if best_seq_global is not None else np.full(4096, 1.0 / 4096)
        print(f"\n*** Early exit: best metric ≈ {best_val_global:.6f} (n={result.shape[0]})")
        return result.tolist()

    n_big = 8192
    conv_fft_big, corr_fft_big = _next_pow2(2 * n_big - 1), _next_pow2(3 * n_big - 2)

    # Linear interpolation of the best tiny shape
    x_small = np.linspace(0, 1, n_small, endpoint=False)
    x_big = np.linspace(0, 1, n_big, endpoint=False)
    interp_big = np.interp(x_big, x_small, best_seq_global)
    interp_big = np.maximum(interp_big, 0.0)
    interp_big /= interp_big.sum()

    # Assemble a small portfolio of large‑grid starts
    big_starts = [interp_big, np.full(n_big, 1.0 / n_big)]

    i_big = np.arange(n_big, dtype=np.float64)
    for alpha in (0.01, 0.05, 0.1, 0.2, 0.5):
        edge = np.exp(-alpha * i_big) + np.exp(-alpha * (n_big - 1 - i_big))
        edge /= edge.sum()
        big_starts.append(edge)

    for a in (0.5, 1.0, 2.0, 5.0):
        for b in (0.5, 1.0, 2.0, 5.0):
            beta = (i_big + 1) ** (a - 1) * (n_big - i_big) ** (b - 1)
            beta = np.maximum(beta, 0.0)
            beta_sum = beta.sum()
            if beta_sum > 0:
                beta /= beta_sum
                big_starts.append(beta)

    # Quick evaluation on the large grid, keep the top‑3
    big_scores = []
    for cand in big_starts:
        if time.time() >= deadline:
            break
        c = np.maximum(cand, 0.0)
        s = c.sum()
        if s < 1e-12:
            continue
        c /= s
        val = _metric(c, n_big, conv_fft_big)
        big_scores.append((val, c))

    big_scores.sort(key=lambda x: x[0])
    top_big = [c for _, c in big_scores[:3]]

    # ------------------------------------------------------------------
    # 4️⃣   Deep optimisation on the large grid
    for idx, start_vec in enumerate(top_big):
        if time.time() >= deadline:
            break
        print(f"\n--- Large‑grid deep optimisation #{idx + 1}/{len(top_big)} ---")
        seq, val = _deep_optimize(
            start_vec,
            n_big,
            conv_fft_big,
            corr_fft_big,
            deadline,
            best_seq_global,
            best_val_global,
        )
        if seq is not None and val < best_val_global:
            best_val_global = val
            best_seq_global = seq

    # ------------------------------------------------------------------
    # 5️⃣   Final stochastic perturbations (if time remains)
    if best_seq_global is not None and time.time() < deadline - 5.0:
        print("\n--- Final stochastic perturbations ---")
        best_seq_global, best_val_global = _local_perturbations(
            best_seq_global,
            n_big,
            conv_fft_big,
            corr_fft_big,
            deadline,
            best_seq_global,
            best_val_global,
        )

    # ------------------------------------------------------------------
    # 6️⃣   Return result (clipped to evaluator limits)
    if best_seq_global is None:
        best_seq_global = np.full(4096, 1.0 / 4096)

    best_seq_global = np.clip(best_seq_global, 0.0, 1000.0)
    print(f"\n*** Finished search: best objective ≈ {best_val_global:.6f} (n={best_seq_global.shape[0]})")
    return best_seq_global.tolist()
