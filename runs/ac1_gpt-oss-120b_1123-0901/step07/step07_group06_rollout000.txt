"""
Propose a low‑evaluation step‑function (non‑negative heights).

Algorithm
---------
1.  Choose a grid size (default 8192 – large enough to give extra freedom).
2.  Build a modest library of parametric candidates (uniform, Dirichlet,
    exponential / power‑law edges, beta‑shaped, triangular, random Dirichlet).
    Quick evaluation via FFT keeps the top‑K (K≈7) promising starts.
3.  For each start perform a *continuation* optimisation:
        – Parameterise a on the probability simplex with a soft‑max:
              a_i = exp(x_i) / Σ_j exp(x_j) .
        – Minimise a smooth approximation of the true objective
              f(x) = 2·n·smooth_max_λ (a∗a) ,
          where smooth_max_λ uses a log‑sum‑exp with temperature λ.
        – λ is increased in a few stages (1 → ~100 → ~5 000) and each stage
          is solved with L‑BFGS‑B (exact gradient, no projection needed).
        – After each stage the exact objective 2·n·max(a∗a) is evaluated;
          the best vector is kept globally.
4.  If time remains, a short stochastic perturbation (Gaussian jitter) and
    a few exponentiated‑gradient steps are applied to the current best.
5.  The final vector is clipped to the evaluator’s allowed range and
    returned as a plain Python list.

The whole routine respects the 1000 s budget, prints progress messages and
always returns a feasible sequence (fallback is the uniform vector).
"""

import time
import heapq
import numpy as np
from scipy import optimize

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation needed for the gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact evaluation (2·n·max(conv)) – fast via FFT."""
    conv = _full_convolution(a, conv_fft_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
class _SmoothMaxObjGrad:
    """
    Callable returning (value, gradient) of the smoothed objective
    f(x) = 2·n·smooth_max_λ (softmax(x) * softmax(x)),
    where smooth_max_λ is a log‑sum‑exp with temperature λ.

    The gradient is computed analytically and returned in the same
    array shape as x.
    """

    def __init__(self, n: int, conv_fft_len: int, corr_fft_len: int, lam: float):
        self.n = n
        self.conv_fft_len = conv_fft_len
        self.corr_fft_len = corr_fft_len
        self.lam = lam
        self.eps = 1e-12

    def __call__(self, x: np.ndarray):
        # a = softmax(x)  (always on the simplex)
        ex = np.exp(x - np.max(x))          # stable softmax pre‑norm
        a = ex / ex.sum()

        # convolution a*a
        conv = _full_convolution(a, self.conv_fft_len)

        # smooth max via log‑sum‑exp (stable)
        max_c = conv.max()
        shift = conv - max_c               # max_c becomes zero exponent
        w_exp = np.exp(self.lam * shift)
        Z = w_exp.sum()
        w = w_exp / Z                      # softmax weights over all lags

        # smooth value = weighted average of conv = max_c + log(Z)/lam
        smooth = max_c + np.log(Z) / self.lam
        val = 2.0 * self.n * smooth

        # gradient w.r.t. a: d ( Σ w_i conv_i ) / da = 2 * (w ★ a)
        grad_a = 2.0 * _cross_correlation(w, a, self.corr_fft_len)

        # chain rule through soft‑max: da/dx = diag(a) - a a^T
        grad_val_a = 2.0 * self.n * grad_a
        dot = np.dot(a, grad_val_a)
        grad_x = a * (grad_val_a - dot)

        return val, grad_x

def _generate_parametric_candidates(n: int, warm_start: np.ndarray | None):
    """Create a diverse set of normalised non‑negative candidates."""
    cand = []

    # warm‑start rotations
    if warm_start is not None:
        cand.append(warm_start.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm_start, shift))

    # uniform baseline
    cand.append(np.full(n, 1.0 / n))

    # random Dirichlet draws (various concentrations)
    for conc in (0.05, 0.1, 0.3, 0.5, 1.0, 2.0, 4.0):
        cand.append(np.random.dirichlet(np.full(n, conc)))

    i = np.arange(n, dtype=np.float64)

    # edge‑exponential families (optionally mixed with uniform)
    for alpha in np.logspace(-2, 1.0, num=12):
        edge = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
        edge /= edge.sum()
        for w in (0.0, 0.3, 0.6, 1.0):
            shape = w * edge + (1.0 - w) / n
            cand.append(shape)

    # edge‑power‑law families
    for p in np.linspace(0.2, 5.0, num=12):
        edge = (i + 1.0) ** (-p) + (n - i) ** (-p)
        edge /= edge.sum()
        for w in (0.0, 0.5, 1.0):
            shape = w * edge + (1.0 - w) / n
            cand.append(shape)

    # two‑Gaussian edges (different widths)
    for gamma in (0.05, 0.2, 0.5, 1.0, 2.0):
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        shape /= shape.sum()
        cand.append(shape)

    # sinusoidal half‑wave (positive part)
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand.append(shape)

    # beta‑shaped families (flexible edge/center bias)
    for alpha in (0.2, 0.5, 1.0, 2.0, 5.0):
        for beta in (0.2, 0.5, 1.0, 2.0, 5.0):
            edge = (i + 1) ** (alpha - 1) * (n - i) ** (beta - 1)
            edge = np.maximum(edge, 0.0)
            edge /= edge.sum()
            cand.append(edge)

    # simple triangular / quadratic shapes
    tri = np.minimum(i + 1, n - i).astype(np.float64)
    tri /= tri.sum()
    cand.append(tri)

    quad = (i + 1) * (n - i)
    quad = quad.astype(np.float64)
    quad /= quad.sum()
    cand.append(quad)

    return cand

def _local_perturbations(base_seq: np.ndarray,
                         n: int,
                         conv_fft_len: int,
                         corr_fft_len: int,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_val: float) -> tuple[np.ndarray, float]:
    """Cheap stochastic jitter + a couple of exponentiated‑gradient steps."""
    a0 = base_seq.copy()
    while time.time() < deadline:
        eps = 0.015
        pert = a0 + eps * (np.random.rand(n) - 0.5)
        pert = np.maximum(pert, 0.0)
        s = pert.sum()
        if s < 1e-12:
            continue
        pert /= s

        # a very short sub‑gradient refinement (few hundred iters)
        a_ref, val_ref = _subgradient_refinement(
            pert,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            best_seq,
            best_val,
        )
        if val_ref < best_val:
            best_val = val_ref
            best_seq = a_ref.copy()
            a0 = best_seq.copy()
            print(f"[Perturb] new best {best_val:.6f}")
        else:
            # slowly drift back to the centre to avoid wandering too far
            a0 = 0.9 * a0 + 0.1 * pert
    return best_seq, best_val


# ----------------------------------------------------------------------
def _subgradient_refinement(a0: np.ndarray,
                            n: int,
                            conv_fft_len: int,
                            corr_fft_len: int,
                            deadline: float,
                            best_seq: np.ndarray | None,
                            best_val: float) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient steps on the true max‑convolution.
    Very short run (max 5000 iters) – used as a final polish after a jitter.
    """
    a = a0.copy()
    a = np.maximum(a, 0.0)
    a /= a.sum()
    eta = 0.8
    last_improve = time.time()
    max_iter = 5000

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Subgrad it={it:05d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        if time.time() - last_improve > 3.0:
            break

        eta *= 0.9995

    return best_seq, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Main entry point. Searches for a non‑negative step function with the
    smallest possible evaluation value within the given time budget.
    Returns the best sequence as a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Choose grid size – larger grids give more flexibility.
    n = 8192
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm‑start from a previously found construction if it exists.
    warm_start = globals().get("height_sequence_1")
    if warm_start is not None:
        ws = np.asarray(warm_start, dtype=np.float64)
        if ws.shape[0] != n:
            # linear interpolation to the desired size
            x_old = np.linspace(0, 1, ws.shape[0], endpoint=False)
            x_new = np.linspace(0, 1, n, endpoint=False)
            ws = np.interp(x_new, x_old, ws)
        ws = np.maximum(ws, 0.0)
        if ws.sum() >= 0.01:
            ws = ws / ws.sum()
        else:
            ws = None
        warm_start = ws
    else:
        warm_start = None

    # ------------------------------------------------------------------
    # Stage 1 – generate a large pool of parametric candidates and keep the best K.
    top_k = 7
    heap: list[tuple[float, np.ndarray]] = []  # max‑heap (negative val)

    candidates = _generate_parametric_candidates(n, warm_start)

    for cand in candidates:
        if time.time() >= deadline:
            break
        # normalise to a probability vector (sum≈1)
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 1e-12:
            cand = np.random.dirichlet(np.ones(n))
            s = cand.sum()
        cand /= s
        val = _objective(cand, n, conv_fft_len)
        # keep the K smallest values (heap stores negative)
        if len(heap) < top_k:
            heapq.heappush(heap, (-val, cand.copy()))
        else:
            if -val > heap[0][0]:
                heapq.heapreplace(heap, (-val, cand.copy()))

    # extract top candidates sorted by increasing objective
    top_candidates = [c for _, c in sorted(heap, key=lambda x: -x[0])]

    # ------------------------------------------------------------------
    # Stage 2 – deep optimisation of each top start via continuation L‑BFGS.
    best_seq: np.ndarray | None = None
    best_val = float("inf")
    lambda_schedule = [1.0, 100.0, 5000.0]          # smooth → near‑exact
    max_iter_per_stage = 250                       # generous but limited

    for idx, start_vec in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        print(f"\n=== Optimising start #{idx + 1}/{len(top_candidates)} (n={n}) ===")
        # initialise x = log(a)
        eps = 1e-12
        x = np.log(start_vec + eps)

        for lam in lambda_schedule:
            if time.time() >= deadline:
                break
            optimizer = _SmoothMaxObjGrad(n, conv_fft_len, corr_fft_len, lam)
            opts = {'maxiter': max_iter_per_stage, 'ftol': 1e-12, 'disp': False}
            try:
                res = optimize.minimize(optimizer,
                                        x,
                                        method='L-BFGS-B',
                                        jac=True,
                                        options=opts)
            except Exception as e:
                print(f"   L‑BFGS failed (lam={lam}): {e}")
                break
            x = res.x

            # evaluate exact objective
            a = np.exp(x - np.max(x))
            a /= a.sum()
            val = _objective(a, n, conv_fft_len)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"   λ={lam:.0f} → new best {best_val:.6f}")

            # early exit if we are out of time
            if time.time() >= deadline:
                break

    # ------------------------------------------------------------------
    # Stage 3 – cheap stochastic perturbations around the current best.
    if best_seq is not None and (time.time() < deadline - 2.0):
        print("\n--- Local stochastic perturbations ---")
        best_seq, best_val = _local_perturbations(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            best_seq,
            best_val,
        )

    # ------------------------------------------------------------------
    # Fallback – uniform vector if nothing succeeded.
    if best_seq is None:
        print("Fallback to uniform vector.")
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = _objective(best_seq, n, conv_fft_len)

    # Clip to the evaluator's allowed range.
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"\n*** Finished search: best objective ≈ {best_val:.6f} (n={best_seq.shape[0]})")
    return best_seq.tolist()
