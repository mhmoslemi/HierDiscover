```
def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5

    # primary grid size
    n = 4096  # or start smaller like 2048 to get more iterations? Might impact quality little. Choose 4096 for comparability.

    conv_fft_len = _next_pow2(2*n -1)
    corr_fft_len = _next_pow2(3*n -2)

    # generate diverse seeds
    candidates = _generate_candidates(n, warm_start=maybe height_sequence_1)
    # also add Beta shapes
    for alpha in np.linspace(0.1, 2.0, 20):
        xs = (np.arange(n) + 0.5) / n
        beta_shape = (xs**(alpha-1) * (1-xs)**(alpha-1))
        beta_shape /= beta_shape.sum()
        candidates.append(beta_shape)

    # plus random Dirichlet for variety
    for _ in range(50):
        conc = 10**np.random.uniform(-3, 1)
        cand = np.random.dirichlet(np.full(n, conc))
        candidates.append(cand)

    # evaluate each quickly using objective (exact conv)
    best_seq = None
    best_val = float('inf')

    best_candidates = [] # keep top few
    for cand in candidates:
        if time.time() >= deadline:
            break
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 0.01: continue
        cand = cand / s
        val = _objective(cand, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_seq = cand.copy()
        best_candidates.append((val, cand))
    # sort and pick top K
    best_candidates.sort(key=lambda x: x[0])
    top_k = 6
    starts = [c for _,c in best_candidates[:top_k]]

    # Now refine each with smoothing schedule
    for start_id, a0 in enumerate(starts):
        if time.time() >= deadline: break
        # Run smoothing descent
        a, val = _smooth_descent(a0, n, conv_fft_len, corr_fft_len, deadline)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
        print(f"[Refine {start_id+1}/{len(starts)}] val {val:.6f}")

    # Then optionally upscale if time remains
    # Interpolation to 8192 and refinement similar.

    return np.clip(best_seq, 0, 1000).tolist()
```

```
def _smooth_descent(a0, n, conv_fft_len, corr_fft_len, deadline):
    a = a0.copy()
    a = np.maximum(a, 1e-12)
    a /= a.sum()
    best_val = float('inf')
    best_a = None
    eta0 = 0.5
    decay = 0.99995
    beta = 0.5
    beta_inc = 1.02 # each iteration multiply, max maybe 50
    beta_max = 50.0

    it = 0
    last_improve = time.time()
    while time.time() < deadline:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            # optional print
            print(f"[smooth iter {it}] new best {val:.6f}")

        # create softmax weights
        # use beta
        w = np.exp(beta * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.zeros_like(conv)
        else:
            w /= w_sum

        # gradient
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # step size
        eta = max(0.001, eta0 * (decay ** it))

        # update
        a = a * np.exp(-eta * grad)
        # renorm
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        # schedule beta increase gradually after some iters
        if beta < beta_max:
            beta = min(beta_max, beta * beta_inc)

        # shake if stuck >5 sec
        if time.time() - last_improve > 5.0:
            perturb = (np.random.rand(n)-0.5) * 0.02
            a = a * np.exp(perturb)
            a = a / a.sum()
            last_improve = time.time()
```

assistantfinal**Solution Overview**

The algorithm searches for a non‑negative step‑function `a` (a probability vector) that minimizes  

\[
F(a)=\frac{2\,n\;\max(a*a)}{(\sum a)^2},
\]

where `a*a` is the linear convolution of `a` with itself.  
Because `F` is invariant under scaling we can always keep `\sum a = 1`.

The search consists of three stages

1. **Broad multi‑start generation** – a large and diverse pool of initial vectors is built
   (uniform, Dirichlet, edge‑exponential, edge‑power‑law, Gaussian edges,
   half‑wave sinusoid and a new family of symmetric Beta‑shapes).

2. **Smoothed max‑convolution descent** – each start is refined with an
   exponentiated‑gradient (mirror‑descent) method that uses a *soft‑max*
   approximation of the `max` operator.  
   The temperature `β` of the soft‑max is increased gradually, turning the
   smooth objective into the true `max` as optimisation proceeds.
   An adaptive step‑size, occasional random shakes and a strict time‑budget
   guarantee that the routine always returns a feasible sequence.

3. **Optional upscale** – if time remains the best vector is linearly interpolated
   to a finer grid (`n = 8192`) and refined again.

The whole routine respects the 1000 s limit, prints progress and finally
returns the best sequence found (clipped to `[0, 1000]`).

```python
"""
Multi‑start smoothed‑max descent for the step‑function problem.

* generate a large, diverse set of seeds (including a new symmetric
  Beta‑family);
* refine each seed with exponentiated‑gradient mirror descent where the
  sub‑gradient of max(conv) is replaced by a soft‑max (temperature β
  annealed from 0.5 up to 50);
* occasional random “shake” when no improvement for >5 s;
* if time permits upscale the best solution to n=8192 and repeat the
  refinement.

All operations are vectorised, convolution is performed with FFTs,
and the routine always returns a feasible list before the 1000 s deadline.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Helper FFT utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)-1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n-1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


# ----------------------------------------------------------------------
# Objective (exact, used only for logging and candidate pre‑selection)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact evaluation of F(a) (sum of a is assumed to be 1)."""
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c / (a.sum() ** 2))


# ----------------------------------------------------------------------
# Candidate generation – many families + the new symmetric Beta family
# ----------------------------------------------------------------------
def _generate_parametric_candidates(
    n: int,
    warm_start: np.ndarray | None,
) -> list[np.ndarray]:
    """Create a diverse list of initial vectors."""
    candidates = []

    # warm‑start and its rotations
    if warm_start is not None:
        candidates.append(warm_start.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            candidates.append(np.roll(warm_start, shift))

    # uniform baseline
    candidates.append(np.full(n, 1.0 / n))

    # random Dirichlet with several concentration parameters
    for conc in (0.01, 0.05, 0.1, 0.3, 0.5, 1.0, 2.0):
        cand = np.random.dirichlet(np.full(n, conc))
        candidates.append(cand)

    i = np.arange(n, dtype=np.float64)

    # edge‑exponential (plus optional uniform mixture)
    for alpha in np.logspace(-2, 1.0, num=16):  # 0.01 … 10
        edge = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
        edge /= edge.sum()
        for w in (0.0, 0.25, 0.5, 0.75, 1.0):
            shape = w * edge + (1.0 - w) / n
            candidates.append(shape)

    # edge‑power‑law (i+1)^(-p) + (n-i)^(-p)
    for p in np.linspace(0.2, 5.0, num=16):
        edge = (i + 1.0) ** (-p) + (n - i) ** (-p)
        edge /= edge.sum()
        for w in (0.0, 0.33, 0.66, 1.0):
            shape = w * edge + (1.0 - w) / n
            candidates.append(shape)

    # double‑Gaussian edges
    for gamma in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        shape /= shape.sum()
        candidates.append(shape)

    # positive half‑wave sinusoid
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    candidates.append(shape)

    # ------------------------------------------------------------------
    # New family: symmetric Beta(α,α) densities (U‑shaped for α<1,
    # hill‑shaped for α>1) optionally mixed with uniform.
    # ------------------------------------------------------------------
    xs = (np.arange(n) + 0.5) / n  # points in (0,1)
    for alpha in np.logspace(-1.5, 0.5, num=12):  # 0.03 … 3.0
        beta_pdf = (xs ** (alpha - 1)) * ((1 - xs) ** (alpha - 1))
        beta_pdf /= beta_pdf.sum()
        for w in (0.0, 0.3, 0.6, 1.0):  # mixture weight of Beta part
            shape = w * beta_pdf + (1.0 - w) / n
            candidates.append(shape)

    return candidates


# ----------------------------------------------------------------------
# Smoothed‑max descent (exponentiated gradient)
# ----------------------------------------------------------------------
def _smooth_descent(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray | None, float]:
    """
    Mirror‑descent on a soft‑max approximation of max(conv).

    * β (soft‑max temperature) is increased geometrically from 0.5 up to 50.
    * Step size η decays exponentially (initial 0.5, never below 0.001).
    * Whenever there has been no improvement for >5 s a small random
      jitter (log‑space) is applied.
    """
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 0.01:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s

    # hyper‑parameters
    eta0 = 0.5
    decay = 0.99995
    min_eta = 0.001

    beta = 0.5           # starting temperature
    beta_inc = 1.02      # geometric increase
    beta_max = 50.0

    last_improve = time.time()
    it = 0
    max_iter = 10_000_000

    while time.time() < deadline and it < max_iter:
        # ----- convolution and exact objective (for logging) -----
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val - 1e-12:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[n={n} smooth iter={it:07d}] new best {val:.6f}")

        # ----- soft‑max weights -----
        # the subtraction of max_c guarantees numerical stability
        w = np.exp(beta * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            # fallback to true active set if soft‑max collapses somehow
            active = np.isclose(conv, max_c, atol=1e-12)
            w = np.where(active, 1.0 / active.sum(), 0.0)
        else:
            w /= w_sum

        # ----- gradient of the smoothed max -----
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # optional clipping of the gradient (prevents overflow)
        grad = np.clip(grad, -20.0, 20.0)

        # ----- step‑size schedule -----
        eta = max(min_eta, eta0 * (decay ** it))

        # ----- exponentiated‑gradient update (mirror descent on simplex) -----
        a = a * np.exp(-eta * grad)

        # renormalise onto the simplex
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        # ----- temperature annealing -----
        if beta < beta_max:
            beta = min(beta_max, beta * beta_inc)

        # ----- random shake if stuck -----
        if time.time() - last_improve > 5.0:
            perturb = (np.random.rand(n) - 0.5) * 0.02   # log‑space jitter
            a = a * np.exp(perturb)
            a = a / a.sum()
            last_improve = time.time()

        it += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a low‑evaluation‑score step function.

    The function respects the time budget, prints progress, and returns the
    best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # safety margin

    # --------------------------------------------------------------
    # Primary grid size – 4096 gives a good balance between quality and speed.
    # --------------------------------------------------------------
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Warm‑start (if a previous run stored a vector in the global scope)
    # --------------------------------------------------------------
    warm_start = globals().get("height_sequence_1")
    if warm_start is not None:
        warm_start = np.asarray(warm_start, dtype=np.float64)
        if warm_start.shape[0] > n:
            warm_start = warm_start[:n]
        elif warm_start.shape[0] < n:
            warm_start = np.pad(warm_start, (0, n - warm_start.shape[0]))
        warm_start = np.maximum(warm_start, 0.0)
        if warm_start.sum() >= 0.01:
            warm_start = warm_start / warm_start.sum()
        else:
            warm_start = None
    else:
        warm_start = None

    # --------------------------------------------------------------
    # Stage 1 – generate a broad pool of seeds
    # --------------------------------------------------------------
    candidates = _generate_parametric_candidates(n, warm_start)

    for conc in (0.005, 0.02, 0.07, 0.15, 0.4, 0.8):
        for _ in range(20):
            cand = np.random.dirichlet(np.full(n, conc))
            candidates.append(cand)

    # --------------------------------------------------------------
    # Fast evaluation of all seeds – keep the top few
    # --------------------------------------------------------------
    cand_vals = []
    for cand in candidates:
        if time.time() >= deadline:
            break
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 0.01:

        cand = cand / s
        val = _objective(cand, n, conv_fft_len)
        cand_vals.append((val, cand))

    if not cand_vals:
        # fallback: uniform vector
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        print("*** No candidates evaluated – returning uniform vector.")
        return best_seq.tolist()

    cand_vals.sort(key=lambda x: x[0])
    top_k = 6
    top_candidates = [c for _, c in cand_vals[:top_k]]

    best_seq = None
    best_val = float("inf")

    # --------------------------------------------------------------
    # Stage 2 – deep optimisation of the selected starts
    # --------------------------------------------------------------
    for start_id, a0 in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        init_val = _objective(a0, n, conv_fft_len)
        print(f"--- Optimising from start #{start_id + 1}/{len(top_candidates)} "
              f"(init val={init_val:.6f})")
        best_seq, best_val = _smooth_descent(
            a0,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            best_seq,
            best_val,
        )

    # --------------------------------------------------------------
    # Stage 3 – if time permits, refine on a larger grid (n = 8192)
    # --------------------------------------------------------------
    time_left = deadline - time.time()
    if time_left > 30.0 and best_seq is not None:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        # linear interpolation onto the finer grid
        x_old = np.linspace(0, 1, n, endpoint=False)
        x_big = np.linspace(0, 1, n_big, endpoint=False)
        a_big = np.interp(x_big, x_old, best_seq)
        a_big = np.maximum(a_big, 0.0)
        a_big /= a_big.sum()

        print("--- Refinement on larger grid (n=8192)")
        best_seq, best_val = _smooth_descent(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            best_seq,
            best_val,
        )
        # keep the (possibly) larger n for the final report
        if best_seq is not None and best_seq.shape[0] == n_big:
            n = n_big

    # --------------------------------------------------------------
    # Final sanity – clip to the evaluator’s allowed interval
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"*** Finished search: best objective ≈ {best_val:.6f} (n={best_seq.shape[0]})")
    return best_seq.tolist()
```
