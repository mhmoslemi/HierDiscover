"""
Propose a step‑function that minimises

    F(a) = 2·n·max(conv(a,a)) / (sum(a)²)

where conv is the linear convolution of the non‑negative vector a with itself.
The optimisation runs inside a 1000 s budget and consists of three phases:

1. **Diverse warm‑starts** – a pool of candidates is built from
   *the previous best* (if present), uniform vectors, Dirichlet samples with
   several concentration parameters and a few analytic shapes.
   For every Dirichlet sample two permutations are tried:
   – the classic anti‑pair ordering (largest opposite smallest) and
   – an alternating‑extreme ordering (largest at the leftmost, second largest
     at the rightmost, third largest next to the leftmost, …).  These
   permutations are known to balance the anti‑diagonals of the outer product
   a aᵀ and therefore lower the convolution peak.

2. **Two‑stage continuous optimisation** – each of the best few pool members
   undergoes a short *Adam* optimisation on a smooth‑max surrogate (log‑sum‑exp)
   followed by a sub‑gradient (exponentiated‑gradient) refinement on the exact
   max‑convolution.  The number of iterations is adjusted to the remaining
   budget.

3. **Intensive final refinement** – the current best vector is refined with a
   long Adam → sub‑gradient → stochastic hill‑climbing (swap / mass‑transfer)
   pipeline that consumes the rest of the time.  If a few seconds remain the
   solution is interpolated to a finer grid (8192 entries) and refined once more.

All updates keep the vector on the probability simplex (non‑negative,
sum = 1).  The final vector is clipped to the range accepted by
`evaluate_sequence` and returned as a plain Python list.
"""

# FFT utilities
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]

def _objective(a: np.ndarray) -> float:

    s = a.sum()
    if s < 1e-12:
        return np.inf
    conv = _full_convolution(a, _next_pow2(2 * a.shape[0] - 1))
    return 2.0 * a.shape[0] * conv.max() / (s * s)

def _anti_pair_arrangement(vals: np.ndarray) -> np.ndarray:
    """
    Anti‑pair layout: largest half on the left, smallest half (reversed) on the right.
    """
    n = len(vals)
    sorted_vals = np.sort(vals)[::-1]          # descending
    a = np.empty(n, dtype=np.float64)
    half = n // 2
    a[:half] = sorted_vals[:half]             # biggest half on the left
    a[-half:] = sorted_vals[-half:][::-1]     # smallest half on the right
    if n % 2 == 1:
        a[half] = sorted_vals[half]           # centre element
    return a


def _alternating_extreme(vals: np.ndarray) -> np.ndarray:
    """
    Alternating‑extreme permutation:
    largest → leftmost, 2nd largest → rightmost, 3rd → next left, …
    """
    n = len(vals)
    sorted_vals = np.sort(vals)[::-1]          # descending
    a = np.empty(n, dtype=np.float64)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            a[left] = v
            left += 1
        else:
            a[right] = v
            right -= 1
        place_left = not place_left
    return a


def _anti_pair_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    """Dirichlet sample followed by anti‑pair ordering."""
    vals = np.random.dirichlet(alpha * np.ones(n))
    return _anti_pair_arrangement(vals)


def _alternating_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    """Dirichlet sample followed by alternating‑extreme ordering."""
    vals = np.random.dirichlet(alpha * np.ones(n))
    return _alternating_extreme(vals)


def _triangle_candidate(n: int) -> np.ndarray:
    """Tent shape (large at the edges, small in the centre)."""
    x = np.arange(n, dtype=np.float64)
    shape = np.minimum(x + 1, n - x)          # 1,2,…,⌈n/2⌉,…,2,1
    shape = shape / shape.sum()
    return shape


def _gaussian_edges_candidate(n: int, gamma: float = 0.5) -> np.ndarray:
    """Two symmetric exponentials concentrated at the two ends."""
    i = np.arange(n, dtype=np.float64)
    shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
    shape = shape / shape.sum()
    return shape

def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    lr: float = 0.05,
    lam_start: float = 1.0,
    lam_factor: float = 1.07,
) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth‑max (log‑sum‑exp) surrogate.
    Returns the best exact solution seen during the run.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lam = lam_start

    best_a = a0.copy()
    best_val = _objective(a0)

    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max → simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # current convolution and exact value
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val_exact = 2.0 * n * max_c
        if val_exact < best_val:
            best_val = val_exact
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best exact {best_val:.6f}")

        # smooth‑max weights (log‑sum‑exp temperature λ)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate w.r.t. a
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature schedule → true max
        lam = min(lam * lam_factor, 1e6)

        # occasional jitter to shake out of plateaus
        if it % 5000 == 0 and (deadline - time.time()) > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2

        it += 1

    return best_a, best_val


def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 0.7,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient phase using exponentiated‑gradient updates.
    """
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s

    best_a = a.copy()
    best_val = _objective(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active set (indices attaining the current max)
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # step‑size schedule (decays like 1/√t)
        eta = eta0 / np.sqrt(it + 1)

        # multiplicative update (mirror descent on the simplex)
        a = a * np.exp(-eta * grad_a)

        # renormalise onto the simplex
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 120_000,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing style hill‑climbing:
    - swap two entries OR transfer a fraction of mass.
    - accept if improvement or with probability exp(-Δ/Temp).
    Temperature decays linearly from 0.01 to 0.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a)

    start_temp = 0.01
    it = 0
    while time.time() < deadline and it < max_moves:
        if np.random.rand() < 0.5:          # swap move
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            a2 = a.copy()
            a2[i], a2[j] = a2[j], a2[i]
        else:                               # transfer move
            i = np.random.randint(n)
            if a[i] < 1e-14:
                continue
            j = np.random.randint(n)
            if i == j:
                continue
            delta = a[i] * (np.random.rand() * 0.4)   # up to 40 % of mass at i
            a2 = a.copy()
            a2[i] -= delta
            a2[j] += delta

        val2 = _objective(a2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
        else:
            # linear temperature schedule
            temp = start_temp * (1.0 - (it / max_moves))
            if temp > 0.0 and np.random.rand() < np.exp(-(val2 - best_val) / temp):
                a = a2

        if it % 5000 == 0:
            print(f"[Local move {it:06d}] best {best_val:.6f}")

        it += 1

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step function that minimises
    F(a)=2·n·max(a∗a)/(sum(a)²) under the given time budget.
    Returns the best vector found as a Python list.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5          # tiny safety margin

    # --------------------------------------------------------------
    # Primary problem size – 4096 gives a good trade‑off between flexibility
    # and per‑iteration cost.
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # 1) Build a rich pool of initial vectors
    pool = []

    # warm start (if available)
    warm = _warm_start(n)
    if warm is not None:
        pool.append(warm.copy())
        # rotate it a few times – rotation does not change the autocorrelation
        for shift in (n // 4, n // 2, 3 * n // 4):
            pool.append(np.roll(warm, shift))

    # deterministic baselines
    pool.append(_uniform_candidate(n))
    pool.append(_triangle_candidate(n))
    for gamma in (0.2, 0.5, 1.0):
        pool.append(_gaussian_edges_candidate(n, gamma))

    # a burst of Dirichlet samples with two permutations each
    alphas = [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0, 1.5]
    for alpha in alphas:
        vals = np.random.dirichlet(alpha * np.ones(n))
        pool.append(_anti_pair_arrangement(vals))
        pool.append(_alternating_extreme(vals))

    # --------------------------------------------------------------
    # 2) Quick evaluation of the pool, keep the 5 best raw candidates
    scored = []
    for idx, cand in enumerate(pool):
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()
        val = _objective(cand)
        scored.append((val, cand))
        print(f"[Init {idx:03d}] value {val:.6f}")

    scored.sort(key=lambda x: x[0])
    top_k = 5
    best_val, best_seq = scored[0]
    top_candidates = [c for _, c in scored[:top_k]]

    # --------------------------------------------------------------
    # 3) Short refinement on the top candidates (budget‑aware)
    time_left = deadline - time.time()
    if time_left > 80:
        per_candidate = max(1, int(time_left / (len(top_candidates) * 2)))
        refined = []
        for i, cand in enumerate(top_candidates):
            # Adam – modest budget (≈30 s each)
            adam_dead = min(deadline - 5,
                            time.time() + min(30, per_candidate))
            a_adam, val_adam = _adam_optimize(
                cand, n, conv_fft_len, corr_fft_len,
                adam_dead, max_iters=120_000,
                lr=0.08, lam_start=1.0, lam_factor=1.08
            )
            # Sub‑gradient – modest budget (≈30 s each)
            sub_dead = min(deadline - 5,
                           time.time() + min(30, per_candidate))
            a_sub, val_sub = _subgradient_optimize(
                a_adam, n, conv_fft_len, corr_fft_len,
                sub_dead, max_iters=120_000, eta0=0.8
            )
            cur_val = min(val_adam, val_sub)
            refined.append((cur_val, a_sub if val_sub < val_adam else a_adam))
            print(f"[Refine {i}] best {cur_val:.6f}")

        refined.sort(key=lambda x: x[0])
        best_val, best_seq = refined[0]
        # keep the two strongest for the final heavy refinement
        top_for_final = [seq for _, seq in refined[:2]]
    else:
        top_for_final = [best_seq]

    # --------------------------------------------------------------
    # 4) Heavy final refinement – consume the bulk of the remaining budget
    while time.time() < deadline - 30:
        # split the remaining time equally among the selected candidates
        remaining = deadline - time.time()
        per = remaining / len(top_for_final) - 5.0
        for idx, seq in enumerate(top_for_final):
            cur_deadline = time.time() + per
            # Long Adam
            a_adam, val_adam = _adam_optimize(
                seq, n, conv_fft_len, corr_fft_len,
                cur_deadline, max_iters=400_000,
                lr=0.07, lam_start=1.0, lam_factor=1.07
            )
            # Long sub‑gradient
            a_sub, val_sub = _subgradient_optimize(
                a_adam, n, conv_fft_len, corr_fft_len,
                cur_deadline, max_iters=400_000, eta0=0.7
            )
            # Local stochastic hill‑climbing
            a_loc, val_loc = _local_random_search(
                a_sub, n, conv_fft_len,
                cur_deadline, max_moves=300_000
            )
            if val_loc < best_val:
                best_val = val_loc
                best_seq = a_loc.copy()
                print(f"[Final improve] new best {best_val:.6f}")

        # Re‑evaluate the remaining time – stop when <15 s left
        if deadline - time.time() < 15:
            break

    # --------------------------------------------------------------
    # 5) Optional upscale to a finer grid (8192) if we still have a few seconds
    if time.time() < deadline - 10:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        # linear interpolation of best sequence onto the larger grid
        x_old = np.linspace(0.0, 1.0, n, endpoint=False)
        x_new = np.linspace(0.0, 1.0, n_big, endpoint=False)
        a_big = np.interp(x_new, x_old, best_seq)

        a_big = np.maximum(a_big, 0.0)
        a_big = a_big / a_big.sum()

        # brief Adam + sub‑gradient on the larger grid
        big_dead = min(deadline - 5, time.time() + 180.0)
        a_big_adam, _ = _adam_optimize(
            a_big, n_big, conv_fft_len_big, corr_fft_len_big,
            big_dead, max_iters=200_000,
            lr=0.07, lam_start=1.0, lam_factor=1.07
        )
        sub_big_dead = min(deadline, time.time() + 200.0)
        a_big_sub, val_big = _subgradient_optimize(
            a_big_adam, n_big, conv_fft_len_big, corr_fft_len_big,
            sub_big_dead, max_iters=200_000, eta0=0.7
        )
        if val_big < best_val:
            best_val = val_big
            best_seq = a_big_sub.copy()
            print(f"[Big grid] improved to {best_val:.6f}")

    # --------------------------------------------------------------
    # 6) Safeguard – ensure a valid return vector
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the range accepted by evaluate_sequence
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
