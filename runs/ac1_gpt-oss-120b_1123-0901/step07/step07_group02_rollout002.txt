def propose_candidate(seed=0, budget_s=1000):
    start = time.time()
    deadline = start + budget_s - 0.5

    np.random.seed(seed)

    # Determine base length n = 4096, may increase later.

    n = 4096
    conv_len = _next_pow2(2 * n - 1)
    corr_len = _next_pow2(3 * n - 2)

    # generate candidates
    candidates = _generate_candidates(n)
    # Evaluate each candidate quickly using evaluate_sequence
    best_seq = None
    best_val = float('inf')
    for idx, cand in enumerate(candidates):
        if time.time() >= deadline: break
        seq = cand.copy()
        # ensure nonnegative and sum>0
        seq = np.maximum(seq, 0.0)
        s = seq.sum()
        if s < 1e-12: continue
        seq = seq / s
        val = evaluate_sequence(seq.tolist())
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[Cand {idx}] init val {val:.6f}")
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)
        best_val = evaluate_sequence(best_seq.tolist())
        print(f"fallback seq val {best_val:.6f}")

    # Adam optimization on best_seq
    # allocate time = min(0.4 * (deadline - time.time()), 300.0) etc.
    remaining = deadline - time.time()
    adam_time = min(remaining * 0.4, 300.0)
    adam_deadline = time.time() + adam_time
    best_seq, val_adam = _adam_optimize(best_seq, n, conv_len, corr_len, adam_deadline, max_iters=500_000)
    if val_adam < best_val:
        best_val = val_adam
        print(f"[Adam] best {best_val:.6f}")

    # subgradient
    remaining = deadline - time.time()
    subgrad_time = min(remaining * 0.35, 400.0)
    subgrad_deadline = time.time() + subgrad_time
    best_seq, val_sub = _exponential_subgradient(best_seq, n, conv_len, corr_len, subgrad_deadline, max_iters=800_000)
    if val_sub < best_val:
        best_val = val_sub
        print(f"[Subgrad] best {best_val:.6f}")

    # Local search annealing
    remaining = deadline - time.time()
    if remaining > 10:
        best_seq, val_loc = _local_annealing(best_seq, n, conv_len, delta_time=remaining, max_iters=500_000)
        if val_loc < best_val:
            best_val = val_loc
            print(f"[Anneal] best {best_val:.6f}")

    # Upscale if time left > 120 secs
    remaining = deadline - time.time()
    if remaining > 150:
        # upscale to 8192
        n_big = 8192
        conv_len_big = _next_pow2(2 * n_big - 1)
        corr_len_big = _next_pow2(3 * n_big - 2)
        # interpolate
        x_old = np.linspace(0, 1, n, endpoint=False)
        x_new = np.linspace(0, 1, n_big, endpoint=False)
        a_big = np.interp(x_new, x_old, best_seq)
        a_big = np.maximum(a_big, 0.0)
        a_big /= a_big.sum()
        # run short Adam
        big_deadline = time.time() + min(remaining * 0.6, 120.0)
        a_big, val_big_adam = _adam_optimize(a_big, n_big, conv_len_big, corr_len_big, big_deadline, max_iters=400_000)
        # subgrad
        remaining_big = deadline - time.time()
        if remaining_big > 20:
            a_big, val_big_sub = _exponential_subgradient(a_big, n_big, conv_len_big, corr_len_big, deadline, max_iters=600_000)
        # local annealing if still time
        remaining_big = deadline - time.time()
        if remaining_big > 10:
            a_big, val_big_loc = _local_annealing(a_big, n_big, conv_len_big, delta_time=remaining_big, max_iters=400_000)
        # pick best among these
        val_big = evaluate_sequence(a_big.tolist())
        if val_big < best_val:
            best_val = val_big
            best_seq = a_big.copy()
            print(f"[Big] final best {best_val:.6f}")

    # ensure sum>0, clip values, convert to list
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    # ensure not all zero: if sum <1e-12 normalize
    s = best_seq.sum()
    if s < 1e-12:
        best_seq = np.full_like(best_seq, 1.0/ best_seq.size)
    else:
        best_seq = best_seq / s

    print(f"Final value {best_val:.6f} sequence length {len(best_seq)}")
    return best_seq.tolist()

"""
Propose a step‑function (non‑negative vector) that minimizes

        F(a) = 2·n·max(a∗a) / (sum(a)^2)

The algorithm works in several stages inside the given time budget:

1. **Candidate generation** – a diverse pool (uniform, Dirichlet,
   edge‑concentrated power‑law, mixtures of uniform + edge, sinusoidal,
   sparse spikes, and the previous best solution if any).

2. **Fast evaluation** – each candidate is normalised to sum = 1 and its
   exact objective is obtained with a single FFT‑based convolution.
   The best candidate becomes the start point.

3. **Adam on a smooth‑max surrogate** – we optimise the log‑sum‑exp
   approximation of the max‑convolution.  The parameters are kept on the
   simplex via a soft‑max; a temperature that multiplies the surrogate
   grows slowly until the true max is approached.  The learning rate
   decays with √t.

4. **Exponentiated‑gradient sub‑gradient phase** – a classic multiplicative
   update on the simplex with a step size η ∝ 1/√t, using the exact sub‑gradient
   of the max‑convolution (average of the active shifts).

5. **Simulated‑annealing local search** – cheap hill‑climbing moves that
   transfer a fraction of mass between two positions; moves are accepted
   deterministically if they improve the objective or probabilistically
   with a temperature schedule.

6. **Optional up‑scaling** – if time permits the solution is linearly
   interpolated to a finer grid (8192 and finally 16384 points) and the
   Adam + sub‑gradient + local‑search pipeline is repeated on the larger
   problem, exploiting the extra degrees of freedom.

All stages respect the global time budget (default 1000 s) and the
function always returns the best sequence found.  The code uses only
NumPy and standard library modules and works on a single CPU core.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a using FFT (result length 2·len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for gradients.
    Returns a vector of length len(a):
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i = 0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]

def _generate_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Create a modest yet diverse set of start vectors."""
    cand = []

    # warm‑start and a few rotations
    if warm is not None:
        cand.append(warm.copy())
        for shift in (n // 8, n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    # random Dirichlet (concentrated and flat)
    cand.append(np.random.dirichlet(np.ones(n)))
    cand.append(np.random.dirichlet(np.full(n, 0.2)))

    # uniform
    cand.append(np.full(n, 1.0 / n))

    # edge‑concentrated power‑law shapes
    idx = np.arange(n, dtype=np.float64) + 1.0
    prod = idx * (n - idx + 1.0)          # small at edges, large in middle
    for alpha in np.linspace(-6.0, -0.3, 20):
        shape = prod ** alpha
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    # mixtures of uniform + edge‑shape
    uniform = np.full(n, 1.0 / n)
    for alpha in np.linspace(-4.5, -0.8, 7):
        shape = prod ** alpha
        if shape.sum() == 0:
            continue
        shape /= shape.sum()
        for w in np.linspace(0.2, 0.8, 4):
            mix = w * uniform + (1.0 - w) * shape
            mix /= mix.sum()
            cand.append(mix)

    # positive half‑wave sinusoid and powered variants
    base = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    base = np.maximum(base, 0.0)
    base /= base.sum()
    cand.append(base)
    for beta in (0.5, 1.5, 2.0, 3.0):
        shape = base ** beta
        shape /= shape.sum()
        cand.append(shape)

    # spike patterns with equally spaced non‑zero entries
    for k in (4, 8, 16, 32, 64, 128):
        a = np.zeros(n, dtype=np.float64)
        if k > 0:
            pos = np.round(np.linspace(0, n - 1, k)).astype(int)
            a[pos] = 1.0 / k
        cand.append(a)

    return cand

def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the smooth‑max (log‑sum‑exp) surrogate.
    Returns the best vector (sum≈1) encountered and its exact objective.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)          # work in the log‑domain
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr0 = 0.05                         # initial learning rate
    lam = 0.5                          # temperature for smooth‑max
    lam_factor = 1.02                  # slowly increase temperature

    best_a = a0.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        # ---- projection onto the simplex via soft‑max ----
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective (sum(a)≈1)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 2000 == 0:
                print(f"[Adam] iter={it:06d} new best {val:.6f}")

        # ---- smooth‑max weights (log‑sum‑exp) ----
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---- gradient of the surrogate (through soft‑max) ----
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)          # chain‑rule for soft‑max

        # ---- Adam update on logits ----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        lr = lr0 / np.sqrt(it + 1.0)          # modest decay
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---- temperature schedule ----
        lam = min(lam * lam_factor, 1e6)

        # occasional tiny jitter to escape plateaus
        if it % 5000 == 0 and time.time() + 0.5 < deadline:
            logits += (np.random.rand(n) - 0.5) * 1e-3

        it += 1

    return best_a, best_val

def _exponential_subgradient(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
    eta0: float = 0.7,
) -> tuple[np.ndarray, float]:
    """
    Multiplicative (exponentiated‑gradient) sub‑gradient descent on
    the true max‑convolution.  Returns the best vector seen.
    """
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    else:
        a /= s

    best_a = a.copy()
    best_val = float("inf")
    stall = 0
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            stall = 0
            if it % 5000 == 0:
                print(f"[Subgrad] iter={it:07d} new best {val:.6f}")
        else:
            stall += 1

        # ---- active set (indices attaining the max) ----
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # ---- decaying step size ----
        eta = eta0 / np.sqrt(it + 1.0)

        # ---- multiplicative update + renormalisation ----
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        # ---- random shake if we have been stuck for a while ----
        if stall > 10_000:
            a = np.maximum(a + 0.02 * (np.random.rand(n) - 0.5), 0.0)
            a /= a.sum()
            stall = 0

        it += 1

    return best_a, best_val

def _local_annealing(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    delta_time: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing hill‑climbing:
    move a random fraction of mass from one index to another.
    Accept improvements deterministically, otherwise with probability
    exp(-Δ/T) where T cools down exponentially.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    best_a = a.copy()

    # initial temperature proportional to the current excess over the lower bound
    T = max((best_val - 1.0) * 0.1, 1e-4)
    start = time.time()
    it = 0
    while time.time() - start < delta_time and it < max_iters:
        i = np.random.randint(n)
        if a[i] < 1e-12:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        # move up to 30 % of the source mass
        delta = a[i] * (np.random.rand() * 0.3)
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()

        improve = val2 < best_val
        if improve or np.random.rand() < np.exp(-(val2 - best_val) / max(T, 1e-12)):
            a = a2
            conv = conv2
            if improve:
                best_val = val2
                best_a = a2.copy()
                if it % 2000 == 0:
                    print(f"[Anneal] iter={it:06d} new best {val2:.6f}")

        # exponential cooling
        if it % 500 == 0:
            T = max(T * 0.99, 1e-6)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the supplied time budget and always returns the
    best sequence found (as a Python list).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # a small safety margin

    # ------------------------------------------------------------------
    # Base resolution ---------------------------------------------------
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # Warm start: previous best if it exists
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None

    # ------------------------------------------------------------------
    # 1. generate and evaluate a modest pool of candidates ------------
    candidates = _generate_candidates(n, warm)

    best_seq = None
    best_val = float("inf")
    for idx, cand in enumerate(candidates):
        if time.time() >= deadline:
            break
        a = np.maximum(cand, 0.0)
        s = a.sum()
        if s < 1e-12:
            continue
        a = a / s
        # exact objective (sum≈1, so F = 2·n·max(conv))
        conv = _full_convolution(a, conv_fft_len)
        val = 2.0 * n * conv.max()
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Init cand {idx}] val={val:.6f}")

    # Fallback to uniform if everything failed
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        conv = _full_convolution(best_seq, conv_fft_len)
        best_val = 2.0 * n * conv.max()
        print(f"[Fallback] uniform val={best_val:.6f}")

    # ------------------------------------------------------------------
    # 2. Adam optimisation (smooth‑max) --------------------------------
    remaining = deadline - time.time()
    if remaining > 15.0:
        adam_time = min(remaining * 0.40, 300.0)      # allocate up to ~40 % time
        adam_deadline = time.time() + adam_time
        best_seq, val_adam = _adam_optimize(
            best_seq, n, conv_fft_len, corr_fft_len, adam_deadline, max_iters=500_000
        )
        if val_adam < best_val:
            best_val = val_adam
            print(f"[Adam] improved best to {val_adam:.6f}")

    # ------------------------------------------------------------------
    # 3. Long exponentiated‑gradient sub‑gradient ----------------------
    remaining = deadline - time.time()
    if remaining > 15.0:
        sub_time = min(remaining * 0.35, 400.0)
        sub_deadline = time.time() + sub_time
        best_seq, val_sub = _exponential_subgradient(
            best_seq, n, conv_fft_len, corr_fft_len, sub_deadline, max_iters=800_000
        )
        if val_sub < best_val:
            best_val = val_sub
            print(f"[Subgrad] improved best to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # 4. Cheap simulated‑annealing local search ------------------------
    remaining = deadline - time.time()
    if remaining > 10.0:
        best_seq, val_loc = _local_annealing(
            best_seq, n, conv_fft_len, delta_time=remaining, max_iters=300_000
        )
        if val_loc < best_val:
            best_val = val_loc
            print(f"[Anneal] improved best to {val_loc:.6f}")

    # ------------------------------------------------------------------
    # 5. Optional up‑scaling to a finer grid if time permits ------------
    if time.time() + 30.0 < deadline:
        # ------- upscale to 8192 -------------------------------------------------
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        # linear interpolation of the current best
        x_old = np.linspace(0, 1, n, endpoint=False)
        x_big = np.linspace(0, 1, n_big, endpoint=False)
        a_big = np.interp(x_big, x_old, best_seq)
        a_big = np.maximum(a_big, 0.0)
        a_big /= a_big.sum()

        # short Adam on the large grid
        big_adam_deadline = time.time() + min(120.0, (deadline - time.time()) * 0.25)
        a_big, val_big_adam = _adam_optimize(
            a_big, n_big, conv_fft_len_big, corr_fft_len_big, big_adam_deadline,
            max_iters=400_000
        )
        if val_big_adam < best_val:
            best_val = val_big_adam
            best_seq = a_big.copy()
            print(f"[Big‑8192 Adam] best {val_big_adam:.6f}")

        # sub‑gradient refinement on the large grid
        if time.time() + 20.0 < deadline:
            a_big, val_big_sub = _exponential_subgradient(
                best_seq, n_big, conv_fft_len_big, corr_fft_len_big,
                deadline, max_iters=600_000
            )
            if val_big_sub < best_val:
                best_val = val_big_sub
                best_seq = a_big.copy()
                print(f"[Big‑8192 Subgrad] best {val_big_sub:.6f}")

        # final cheap annealing on 8192 points
        remaining_big = deadline - time.time()
        if remaining_big > 10.0:
            a_big, val_big_loc = _local_annealing(
                best_seq, n_big, conv_fft_len_big, delta_time=remaining_big,
                max_iters=200_000
            )
            if val_big_loc < best_val:
                best_val = val_big_loc
                best_seq = a_big.copy()
                print(f"[Big‑8192 Anneal] best {val_big_loc:.6f}")

        # ------ optionally go to 16384 -----------------------------------------
        if time.time() + 25.0 < deadline:
            n_big2 = 16384
            conv_fft_len_big2 = _next_pow2(2 * n_big2 - 1)
            corr_fft_len_big2 = _next_pow2(3 * n_big2 - 2)

            x_old = np.linspace(0, 1, best_seq.shape[0], endpoint=False)
            x_new = np.linspace(0, 1, n_big2, endpoint=False)
            a_big2 = np.interp(x_new, x_old, best_seq)
            a_big2 = np.maximum(a_big2, 0.0)
            a_big2 /= a_big2.sum()

            # very short Adam
            big2_deadline = time.time() + min(50.0, deadline - time.time())
            a_big2, val_big2_adam = _adam_optimize(
                a_big2, n_big2, conv_fft_len_big2, corr_fft_len_big2,
                big2_deadline, max_iters=150_000
            )
            if val_big2_adam < best_val:
                best_val = val_big2_adam
                best_seq = a_big2.copy()
                print(f"[Big‑16384 Adam] best {val_big2_adam:.6f}")

            # sub‑gradient long run
            if time.time() + 20.0 < deadline:
                a_big2, val_big2_sub = _exponential_subgradient(
                    best_seq, n_big2, conv_fft_len_big2, corr_fft_len_big2,
                    deadline, max_iters=200_000
                )
                if val_big2_sub < best_val:
                    best_val = val_big2_sub
                    best_seq = a_big2.copy()
                    print(f"[Big‑16384 Subgrad] best {val_big2_sub:.6f}")

    # ------------------------------------------------------------------
    # Final sanity checks -------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, 1000.0)   # respect evaluator limits
    total = best_seq.sum()
    if total < 1e-12:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)
    else:
        best_seq = best_seq / total

    # Report final value via the official evaluator (just for logging)
    final_val = evaluate_sequence(best_seq.tolist())
    print(f"Final best value {final_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
