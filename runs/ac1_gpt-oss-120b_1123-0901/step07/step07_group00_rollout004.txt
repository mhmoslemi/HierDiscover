"""
propose_candidate
=================

Search for a non‑negative step‑function (vector) that minimizes

    F(a) = 2·n·max(a∗a) / (sum(a)^2)

The algorithm combines several ideas that proved useful in earlier
experiments and adds a few new ingredients:

1. **Rich warm‑start pool** – Dirichlet draws with many concentration
   parameters, edge‑exponential / polynomial families, double‑Gaussian,
   sinusoid, power‑law and logarithmic shapes, plus very sparse “spike”
   patterns.

2. **Adam on a smooth‑max surrogate** – we minimise a soft‑max
   (log‑sum‑exp) of the convolution using Adam, slowly sharpening the
   temperature so the surrogate converges to the true max.

3. **Exponentiated‑gradient sub‑gradient phase** – an exact‑max
   sub‑gradient (active‑set) descent on the original objective but
   using an entropic mirror‑descent (multiplicative) update, which is
   cheap and respects the simplex constraint.

4. **Additive projected‑gradient polishing** – a short phase of
   additive gradient steps with Euclidean projection onto the probability
   simplex to obtain a finer local optimum.

5. **Simulated‑annealing local moves** – random transfers of a small
   fraction of mass between two coordinates, accepted with a
   Metropolis rule; this helps escape shallow local minima.

6. **Optional up‑sampling** – if time remains, the best solution is
   interpolated to a finer grid (e.g. 8192 points) and the whole
   pipeline is repeated, giving the algorithm a chance to exploit the
   larger search space.

All phases respect the global deadline (default 1000 s) and the best
vector found so far is always returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) such that
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)].
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    # reference: Duchi et al. (2008)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries zero – return uniform simplex
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Create a diverse pool of start vectors."""
    cand = []

    # warm start and simple cyclic shifts
    if warm is not None:
        cand.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    # Dirichlet draws with a range of concentration parameters
    for alpha in (0.1, 0.3, 0.6, 1.0, 3.0, 10.0):
        cand.append(np.random.dirichlet(np.full(n, alpha)))

    # Edge‑exponential families
    for gamma in (0.2, 0.5, 1.0, 2.0):
        left = np.exp(-gamma * np.arange(n, dtype=np.float64))
        right = left[::-1]
        shape = left + right
        shape /= shape.sum()
        cand.append(shape)

    # Edge‑polynomial (power‑law) families
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.5, 5):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    # Double‑Gaussian families
    i = np.arange(n, dtype=np.float64)
    for sigma in (0.07, 0.15, 0.3):
        shape = (
            np.exp(-0.5 * ((i) / (sigma * n)) ** 2)
            + np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
        )
        shape /= shape.sum()
        cand.append(shape)

    # Power‑law tails (inverse‑distance)
    for beta in (0.5, 1.0, 1.5, 2.0):
        shape = (np.arange(1, n + 1, dtype=np.float64) ** (-beta)
                 + np.arange(n, 0, -1, dtype=np.float64) ** (-beta))
        shape /= shape.sum()
        cand.append(shape)

    # Logarithmic decay
    shape = 1.0 / (np.log(np.arange(1, n + 1) + 1) + np.log(np.arange(n, 0, -1) + 1))
    shape /= shape.sum()
    cand.append(shape)

    # Half‑wave sinusoid (positive part)
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand.append(shape)

    # Sinusoid squared (smooth bell)
    shape = np.sin(np.pi * np.arange(n) / (n - 1)) ** 2
    shape /= shape.sum()
    cand.append(shape)

    # Uniform
    cand.append(np.full(n, 1.0 / n))

    return cand


def _spike_candidates(n: int) -> list[np.ndarray]:
    """Sparse candidates with equally spaced spikes."""
    out = []
    for k in (16, 32, 64, 128, 256):
        a = np.zeros(n, dtype=np.float64)
        if k > 0:
            pos = np.round(np.linspace(0, n - 1, k)).astype(int)
            a[pos] = 1.0 / k
        out.append(a)
    return out

def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth‑max (log‑sum‑exp) surrogate.
    Returns best vector (sum≈1) encountered and its exact objective.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)                # initialise on the simplex
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.06
    lam = 0.02                # initial temperature (soft)
    lam_factor = 1.07

    best_a = a0.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max to stay on the simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective (for tracking)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights (soft‑max over convolution values)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate (w.r.t. a)
        grad_a = 2.0 * _cross_correlation(w, a, conv_fft_len)

        # chain rule through soft‑max (mirror‑descent form)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # temperature schedule – become sharper
        lam = min(lam * lam_factor, 1e7)

        # occasional tiny random shake near deadline
        if it % 4000 == 0 and time.time() > deadline - 60:
            logits += (np.random.rand(n) - 0.5) * 2e-2

        # mild learning‑rate decay
        if it % 50_000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


def _exp_subgrad_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_iters: int = 250_000,
    eta0: float = 0.85,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient descent using exponentiated‑gradient updates.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[ExpSubGrad it={it:06d}] new best {val:.6f}")

        # active set (indices attaining max)
        active = np.isclose(conv, max_c, atol=1e-14)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, conv_fft_len)

        # decreasing step size
        eta = eta0 / np.sqrt(it + 1)

        # exponentiated‑gradient update (entropic mirror descent)
        a = a * np.exp(-eta * grad_a)

        # renormalise – protect against underflow
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _pgd_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_iters: int = 12_000,
    eta0: float = 1.0,
) -> tuple[np.ndarray, float]:
    """
    Additive projected‑gradient descent (polishing phase).
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 2000 == 0:
                print(f"[PGD it={it:06d}] new best {val:.6f}")

        # active set – same as in the exponential phase
        active = np.isclose(conv, max_c, atol=1e-13)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad = 2.0 * _cross_correlation(w, a, conv_fft_len)

        eta = eta0 / np.sqrt(it + 1)
        a = a - eta * grad
        a = _project_simplex(a)          # enforces a ≥ 0, sum = 1

        it += 1

    return best_a, best_val


def _simulated_annealing(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 10_000,
) -> tuple[np.ndarray, float]:
    """
    Cheap Metropolis‑style local search: move a random fraction of
    mass from one index to another, accepting worsening moves with a
    temperature schedule.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()

    # initial temperature (chosen heuristically)
    T0 = 1e-3

    it = 0
    while time.time() < deadline and it < max_moves:
        # linear temperature decay
        time_left = max(deadline - time.time(), 0.0)
        T = T0 * (time_left / (deadline - (deadline - max_moves * 0)))  # crude

        i = np.random.randint(n)
        if a[i] < 1e-14:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        # propose moving a random fraction (up to 30 %) of a[i] to a[j]
        delta = a[i] * (np.random.rand() * 0.3)

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        # accept if improvement or with Metropolis probability
        accept = False
        if val2 < best_val:
            accept = True
        else:
            if T > 0:
                prob = np.exp(-(val2 - best_val) / T)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a = a2
            conv = conv2
            if val2 < best_val:
                best_val = val2
                best_a = a2.copy()
                if it % 2000 == 0:
                    print(f"[SA move {it:06d}] improved to {val2:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the given time budget and always returns the best
    sequence (as a Python list) found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # Problem size – we start with a moderate grid and upscale if possible
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Warm start handling
    # ------------------------------------------------------------------
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None

    # ------------------------------------------------------------------
    # Build candidate pool
    # ------------------------------------------------------------------
    pool = _initial_candidates(n, warm) + _spike_candidates(n)

    # ------------------------------------------------------------------
    # Exploration of the pool (quick Adam phases)
    # ------------------------------------------------------------------
    best_seq: np.ndarray | None = None
    best_val = float("inf")
    remaining = deadline - time.time()
    per_candidate_budget = remaining * 0.2 / max(1, len(pool))

    for cid, cand in enumerate(pool):
        if time.time() >= deadline:
            break
        a0 = np.maximum(cand, 0.0)
        if a0.sum() < 0.01:
            a0 = np.random.dirichlet(np.ones(n))
        a0 = a0 / a0.sum()

        loc_deadline = min(deadline - 2.0, time.time() + per_candidate_budget)

        # ---- short Adam ----
        a_adam, val_adam = _adam_optimize(
            a0,
            n,
            conv_fft_len,
            loc_deadline,
            max_iters=120_000,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[cand {cid}] Adam best {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Deep refinement of the current best candidate
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5:
        # Exponential‑gradient sub‑gradient phase
        a_sub, val_sub = _exp_subgrad_optimize(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_iters=250_000,
            eta0=0.9,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Refine] Exp‑subgrad best {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Polishing – additive PGD
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5:
        a_pgd, val_pgd = _pgd_optimize(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_iters=15_000,
            eta0=0.8,
        )
        if val_pgd < best_val:
            best_val = val_pgd
            best_seq = a_pgd.copy()
            print(f"[Polish] PGD best {val_pgd:.6f}")

    # ------------------------------------------------------------------
    # Simulated‑annealing local search
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5:
        a_sa, val_sa = _simulated_annealing(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=30_000,
        )
        if val_sa < best_val:
            best_val = val_sa
            best_seq = a_sa.copy()
            print(f"[SA] improved best {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Optional upscale to a finer grid (if time permits)
    # ------------------------------------------------------------------
    if time.time() + 30 < deadline:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        # linear interpolation of the current best
        if best_seq is not None:
            x_old = np.linspace(0, 1, n, endpoint=False)
            x_new = np.linspace(0, 1, n_big, endpoint=False)
            a_big = np.interp(x_new, x_old, best_seq)
        else:
            a_big = np.random.dirichlet(np.ones(n_big))

        a_big = np.maximum(a_big, 0.0)
        a_big = a_big / a_big.sum()

        # Adam on the larger grid (short)
        a_big_adam, _ = _adam_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            deadline,
            max_iters=100_000,
        )

        # Exponential‑gradient sub‑gradient (long) on large grid
        a_big_ref, val_big = _exp_subgrad_optimize(
            a_big_adam,
            n_big,
            conv_fft_len_big,
            deadline,
            max_iters=180_000,
            eta0=0.8,
        )
        if val_big < best_val:
            best_val = val_big
            best_seq = a_big_ref.copy()
            print(f"[Upscale] refined best {val_big:.6f}")

    # ------------------------------------------------------------------
    # Ensure a valid return
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
