def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2

    # Load warm if exists
    warm = globals().get('height_sequence_1')
    # Normalize warm if present
    ...

    # Choose n based on available time: if time > 100 secs, maybe use n=8192 else 4096.

    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # 1. generate many candidates
    cand_list = []
    # uniform
    cand_list.append(np.full(n, 1.0 / n))
    # warm rotated versions
    if warm is not None:
        warm_n = warm.copy()
        if warm_n.shape[0] != n:
            # reshape via interpolation
            warm_n = np.interp(np.linspace(0, 1, n, endpoint=False),
                               np.linspace(0, 1, warm_n.shape[0], endpoint=False),
                               warm_n)
        warm_n = np.maximum(warm_n, 0)
        warm_n /= warm_n.sum()
        cand_list.append(warm_n)
        # rotations
        for shift in (n//8, n//4, 3*n//8):
            cand_list.append(np.roll(warm_n, shift))

    # random Dirichlet with random concentration parameter
    for _ in range(30):
        alpha = np.random.lognormal(mean=0.0, sigma=1.0, size=n)
        a = np.random.dirichlet(alpha)
        cand_list.append(a)

    # power-law shapes with random exponent
    idx = np.arange(1, n+1)
    prod = idx * (n - idx + 1)
    for alpha in np.linspace(-4.5, -0.05, 30):
        shape = prod.astype(np.float64) ** alpha
        if shape.sum() > 0:
            shape /= shape.sum()
            cand_list.append(shape)
    # double exponential shapes
    i = np.arange(n, dtype=np.float64)
    for gamma in np.geomspace(0.2, 10.0, 20):
        shape = np.exp(-gamma*i) + np.exp(-gamma*(n-1-i))
        shape /= shape.sum()
        cand_list.append(shape)
    # sinusoid with random phase and frequency
    for freq in np.linspace(0.5, 3.0, 15):
        phase = np.random.rand() * 2*np.pi
        shape = np.sin(np.pi*freq*(np.arange(n)+0.5)/n + phase)
        shape = np.maximum(shape, 0)
        if shape.sum() > 0:
            shape /= shape.sum()
            cand_list.append(shape)

    # spike patterns: random positions
    for k in [32, 64, 128, 256]:
        a = np.zeros(n)
        pos = np.random.choice(n, size=k, replace=False)
        a[pos] = 1.0/k
        cand_list.append(a)

    # Evaluate all candidates and keep top K
    obj_vals = []
    for a in cand_list:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        obj = 2.0 * n * max_c
        obj_vals.append(obj)

    best_indices = np.argsort(obj_vals)[:20]
    best_seq = cand_list[best_indices[0]].copy()
    best_val = obj_vals[best_indices[0]]

    # 2. Subgradient optimization on top candidates
    # allocate time among top candidates equally
    remaining = deadline - time.time()
    if remaining > 5:
        per_candidate_time = remaining / len(best_indices)
        for idx in best_indices:
            if time.time() >= deadline:
                break
            a0 = cand_list[idx].copy()
            # ensure sum=1
            a0 = a0 / a0.sum()
            cand_deadline = min(deadline - 1.0, time.time() + per_candidate_time*0.8)
            a_opt, val_opt = _subgradient_opt2(
                a0, n, conv_fft_len, corr_fft_len, cand_deadline, max_iters=200_000
            )
            if val_opt < best_val:
                best_val = val_opt
                best_seq = a_opt.copy()
                print(f"[Stage2 refined] new best {best_val:.6f}")

    # 3. Greedy hill climbing moves (mass transfer) using incremental conv update
    # Precompute conv for best_seq
    conv = _full_convolution(best_seq, conv_fft_len)
    cur_val = 2.0 * n * conv.max()
    # time budget for local search: up to 20% of remaining time
    remaining = deadline - time.time()
    local_time_budget = remaining * 0.6  # allocate most of remaining for local search
    move_start = time.time()
    it = 0
    while time.time() - move_start < local_time_budget and it < 500_000:
        # pick source index with decent mass
        i = np.random.choice(np.where(best_seq > 1e-12)[0])
        j = np.random.randint(n)
        if i == j:
            continue
        delta = best_seq[i] * (np.random.rand() * 0.5)  # up to 50% move
        # compute new conv via incremental update
        # We'll compute conv_new = conv + 2*delta*(shift_j - shift_i)
        # compute shift vectors on the fly
        shift_i = np.zeros_like(conv)
        shift_j = np.zeros_like(conv)
        # shift_i: indices from i to i+n-1 contain a[0:n] shifted
        start_i = i
        end_i = i + n
        shift_i[start_i:end_i] = best_seq
        start_j = j
        end_j = j + n
        shift_j[start_j:end_j] = best_seq
        conv_new = conv + 2.0 * delta * (shift_j - shift_i)
        new_val = 2.0 * n * conv_new.max()
        if new_val < cur_val:
            # accept
            best_seq[i] -= delta
            best_seq[j] += delta
            conv = conv_new
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_seq_best = best_seq.copy()
                print(f"[Local move {it}] improved to {cur_val:.6f}")
        it += 1

    # Ensure normalisation (sum should still be 1; due to rounding may slight drift)
    best_seq = best_seq / best_seq.sum()

    # final: clip
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    print(f"Final best value {best_val:.6f} length {len(best_seq)}")
    return best_seq.tolist()

The whole routine respects the 1000 s time budget, prints progress
information and always returns a valid sequence.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the sub‑gradient.
    Returns a length‑len(a) vector:
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i = 0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


def _subgradient_opt2(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
    eta0: float = 0.7,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient optimisation on the simplex.
    Stops either when *deadline* is reached or *max_iters* iterations have
    elapsed.  Returns the best vector seen and its exact objective value.
    """
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s

    best_a = a.copy()
    best_val = float("inf")
    stall = 0
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            stall = 0
            if it % 5_000 == 0:
                print(f"[Subgrad iter={it:07d}] new best {val:.6f}")
        else:
            stall += 1

        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # decaying step size
        eta = eta0 / np.sqrt(it + 1)

        # multiplicative update and renormalisation
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        # occasional random shake when stuck
        if stall > 10_000:
            a = np.maximum(a + 0.02 * (np.random.rand(n) - 0.5), 0.0)
            a /= a.sum()
            stall = 0

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    Returns the best sequence (as a Python list) found within the time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Choose a grid size: larger grids give more freedom but cost more.
    # Use 8192 when we have enough time, otherwise fall back to 4096.
    if budget_s >= 400:
        n = 8192
    else:
        n = 4096

    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Warm start (previous best) if it exists.
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] != n:
            # linear interpolation / truncation to the current size
            x_old = np.linspace(0, 1, warm.shape[0], endpoint=False)
            x_new = np.linspace(0, 1, n, endpoint=False)
            warm = np.interp(x_new, x_old, warm)
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 1e-12:
            warm = warm / warm.sum()
        else:
            warm = None

    # --------------------------------------------------------------
    # 1) BUILD A LARGE DIVERSE POOL OF CANDIDATES
    candidates = []

    # warm start and a few rotations
    if warm is not None:
        candidates.append(warm.copy())
        for shift in (n // 8, n // 4, 3 * n // 8, n // 2):
            candidates.append(np.roll(warm, shift))

    # uniform
    candidates.append(np.full(n, 1.0 / n))

    # random Dirichlet draws with random concentration parameters
    for _ in range(30):
        alpha = np.random.lognormal(mean=0.0, sigma=1.0, size=n)
        a = np.random.dirichlet(alpha)
        candidates.append(a)

    # power‑law edge‑concentrated shapes  a[i] ∝ (i+1)*(n-i)^α
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-5.0, -0.1, 25):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape = shape / shape.sum()
            candidates.append(shape)

    # symmetric double‑exponential (Laplace‑like) tails
    i = np.arange(n, dtype=np.float64)
    for gamma in np.geomspace(0.2, 8.0, 20):
        shape = np.exp(-gamma * i) + np.exp(-gamma * (n - 1 - i))
        shape = shape / shape.sum()
        candidates.append(shape)

    # U‑shaped sinusoid families  a[i] ∝ sin(π·(i+½)/n)^β
    for beta in np.linspace(0.2, 2.0, 12):
        shape = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** beta
        shape = np.maximum(shape, 0.0)
        if shape.sum() > 0:
            shape = shape / shape.sum()
            candidates.append(shape)

    # sparse spike patterns with equally spaced non‑zero entries
    for k in (16, 32, 64, 128):
        a = np.zeros(n, dtype=np.float64)
        pos = np.random.choice(n, size=k, replace=False)
        a[pos] = 1.0 / k
        candidates.append(a)

    # --------------------------------------------------------------
    # 2) QUICK EVALUATION OF ALL CANDIDATES – keep the best few
    scores = []
    for a in candidates:
        conv = _full_convolution(a, conv_fft_len)
        val = 2.0 * n * conv.max()
        scores.append(val)

    scores = np.array(scores)
    top_k = min(20, len(candidates))
    best_indices = np.argsort(scores)[:top_k]

    best_seq = candidates[best_indices[0]].copy()
    best_val = scores[best_indices[0]]

    # --------------------------------------------------------------
    # 3) REFINEMENT OF ELITE CANDIDATES WITH SUB‑GRADIENT
    remaining = deadline - time.time()
    if remaining > 30.0:
        # Distribute remaining time uniformly among the elite candidates
        per_candidate = max(5.0, (remaining - 10.0) / top_k)
        for idx in best_indices:
            if time.time() >= deadline - 1.0:
                break
            a0 = candidates[idx].copy()
            a0 = a0 / a0.sum()
            cand_deadline = min(deadline - 1.0, time.time() + per_candidate)
            a_opt, val_opt = _subgradient_opt2(
                a0,
                n,
                conv_fft_len,
                corr_fft_len,
                cand_deadline,
                max_iters=400_000,
                eta0=0.7,
            )
            if val_opt < best_val:
                best_val = val_opt
                best_seq = a_opt.copy()
                print(f"[Elite refine] new best {best_val:.6f}")

    # --------------------------------------------------------------
    # 4) GREEDY LOCAL HILL‑CLIMBING (mass transfer) USING INCREMENTAL UPDATE
    conv = _full_convolution(best_seq, conv_fft_len)
    cur_val = 2.0 * n * conv.max()

    # allocate the rest of the budget to local moves
    remaining = deadline - time.time()
    if remaining > 5.0:
        move_deadline = time.time() + remaining
        it = 0
        while time.time() < move_deadline and it < 300_000:
            # source index drawn proportionally to current mass
            src = np.random.choice(n, p=best_seq)
            if best_seq[src] < 1e-12:
                it += 1
                continue
            dst = np.random.randint(n)
            if src == dst:
                it += 1
                continue

            # move up to 50 % of the source mass
            delta = best_seq[src] * (np.random.rand() * 0.5)

            # incremental convolution update (O(n))
            conv[src:src + n] -= 2.0 * delta * best_seq
            conv[dst:dst + n] += 2.0 * delta * best_seq

            new_val = 2.0 * n * conv.max()
            if new_val < cur_val - 1e-12:
                # accept move
                best_seq[src] -= delta
                best_seq[dst] += delta
                cur_val = new_val
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq_best = best_seq.copy()
                    print(f"[Local move {it}] improved best to {best_val:.6f}")
            else:
                # revert convolution change
                conv[src:src + n] += 2.0 * delta * best_seq
                conv[dst:dst + n] -= 2.0 * delta * best_seq
            it += 1

    # --------------------------------------------------------------
    # 5) FINAL NORMALISATION & SAFETY CLIP
    best_seq = best_seq / best_seq.sum()
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
