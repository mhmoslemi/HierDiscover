```python
def generate_candidates(n):
    cand = []
    # uniform
    cand.append(np.full(n, 1.0/n))
    # Dirichlet samples (5)
    for _ in range(5):
        cand.append(np.random.dirichlet(np.ones(n)))
    # edge power law shape: i*(n-i+1) ^ alpha for alphas
    idx = np.arange(n) + 1.0
    prod = idx*(n - idx + 1)
    alphas = np.concatenate([np.linspace(-6.0, -0.1, 30), np.linspace(-0.1, -0.01, 5)])
    for alpha in alphas:
        shape = np.where(prod>0, prod**alpha, 0.0)
        if shape.sum()>0:
            shape /= shape.sum()
            cand.append(shape)
    # edge distance power law shape: (i+1)**(-p) + (n-i)**(-p)
    p_vals = np.linspace(0.5, 5.0, 20)
    i = np.arange(n)
    for p in p_vals:
        shape = (i+1)**(-p) + (n-i)**(-p)
        shape /= shape.sum()
        cand.append(shape)
    # double exponential shape
    gamma_vals = np.geomspace(0.2, 10.0, 20)
    i = np.arange(n, dtype=np.float64)
    for gamma in gamma_vals:
        shape = np.exp(-gamma*i) + np.exp(-gamma*(n-1-i))
        shape /= shape.sum()
        cand.append(shape)
    # sin^q shape (half wave)
    for q in [1,2,3,4,5]:
        shape = np.sin(np.pi*(np.arange(n)+0.5)/n) ** q
        shape = np.maximum(shape,0)
        shape /= shape.sum()
        cand.append(shape)
    # triangular shape: linear up and down (peak in middle) -> maybe not helpful but include
    tri = np.concatenate([np.arange(1,n//2+1), np.arange(n//2,0,-1)])[:n]
    tri = tri.astype(float)
    tri /= tri.sum()
    cand.append(tri)
    # spikes: k equally spaced spikes
    for k in [8,16,32,64,128]:
        a = np.zeros(n)
        pos = np.linspace(0, n-1, k, dtype=int)
        a[pos] = 1.0/k
        cand.append(a)
    return cand
```

```python
def simulated_annealing(a, n, fft_len, deadline, max_iters=200_000):
    a = a.copy()
    conv = _full_convolution(a, fft_len)
    cur_val = 2.0*n*conv.max()
    best_a = a.copy()
    best_val = cur_val
    T0 = cur_val * 0.1  # start temperature as fraction of value
    Tend = 1e-5
    it = 0
    while time.time() < deadline and it < max_iters:
        # geometric temperature schedule
        T = T0 * ((Tend/T0) ** (it / max_iters))
        i = np.random.randint(n)
        if a[i] < 1e-14:
            continue
        j = np.random.randint(n)
        if i == j:
            continue
        delta = a[i] * (np.random.rand() * 0.2)  # move up to 20%
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta
        # ensure a2 nonnegative
        if a2[i] < 0:
            continue
        conv2 = _full_convolution(a2, fft_len)
        val2 = 2.0*n*conv2.max()
        diff = val2 - cur_val
        accept = False
        if diff < 0:
            accept = True
        else:
            # accept with probability exp(-diff/T)
            if np.random.rand() < np.exp(-diff / max(T, 1e-12)):
                accept = True
        if accept:
```

assistantfinal**Solution Overview**

The script below searches for a non‑negative step‑function (a vector `a`) that minimizes  

\[
F(a)=\frac{2\,n\;\max (a\star a)}{(\sum a)^2}\;,
\]

where `a⋆a` is the linear convolution of `a` with itself.  
The algorithm works inside the 1000 s budget and proceeds in the
following stages:

1. **Diverse warm‑starts** – a large pool of parametric shapes
   (uniform, Dirichlet, edge‑concentrated power‑laws, double‑exponential,
   inverse‑distance, sine‑powers, spikes) together with the previous best
   solution (`height_sequence_1`) is generated.  
   All candidates are normalised to $\sum a=1$ and their exact objective
   (via FFT) is evaluated. The ten best candidates are retained.

2. **Adam optimisation on a smooth‑max surrogate** – each retained candidate
   is refined with Adam on the log‑sum‑exp approximation of the maximal
   convolution.  The optimisation works in the simplex by using a soft‑max
   parametrisation (`logits`).  The temperature is annealed (multiplying by
   `1.03` each iteration) so that the surrogate converges to the true max.

3. **Exponentiated‑gradient sub‑gradient phase** – a long multiplicative
   sub‑gradient run (step size `η₀/√t`) further reduces the exact objective.
   Random “shakes’’ are injected when progress stalls.

4. **Simulated‑annealing local search** – a Metropolis move (transfer a
   random fraction of mass from one index to another) is accepted if it
   improves the objective, otherwise with probability `exp(-Δ/T)`.  The
   temperature is decreased geometrically; this lets the search escape
   shallow local minima.

5. **Upscaling** – if time permits the current best vector is interpolated
   to a finer grid (8192, then 16384 points) and the whole pipeline
   (Adam → sub‑gradient → simulated annealing) is repeated on the larger
   problem.  More degrees of freedom often allow a slightly lower bound.

All phases respect the global time budget; the routine always returns the
best vector found.  Only `numpy` and `scipy` (for the optional LP
fallback) are used, and a single CPU core is sufficient.

```python
#!/usr/bin/env python3
"""
search for a non‑negative step‑function that minimises
    F(a) = 2·n·max(a⋆a) / (sum(a)^2)

The algorithm combines:
  * a large, parameterised pool of warm‑starts,
  * Adam optimisation on a smooth‑max (log‑sum‑exp) surrogate,
  * a long exponentiated‑gradient sub‑gradient phase,
  * simulated‑annealing hill‑climbing,
  * optional up‑scaling to 8192 / 16384 points.

All phases are limited by a 1000 s wall‑time budget and the best
sequence found is returned as a plain Python list.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# basic FFT utilities ----------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a length‑len(a) vector:
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i = 0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
# exact objective (fast, scale‑invariant because we keep sum(a)=1) -----
def _objective(a: np.ndarray, fft_len: int) -> float:
    """Return 2·n·max(a⋆a) for a vector that already satisfies sum(a)=1."""
    n = a.shape[0]
    conv = _full_convolution(a, fft_len)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# candidate generator ----------------------------------------------------
def _candidate_pool(n: int) -> List[np.ndarray]:
    """Generate a diverse set of start vectors (all normalised to sum=1)."""
    cand: List[np.ndarray] = []

    # 1. warm start from the previously best solution, if present
    warm = globals().get("height_sequence_1")
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] > n:
            w = w[:n]
        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]))
        w = np.maximum(w, 0.0)
        if w.sum() >= 0.01:
            w = w / w.sum()
            cand.append(w.copy())
            # a few rotated copies
            for shift in (n // 4, n // 2, 3 * n // 4):
                cand.append(np.roll(w, shift))

    # 2. uniform
    cand.append(np.full(n, 1.0 / n))

    # 3. a few Dirichlet draws
    for _ in range(5):
        cand.append(np.random.dirichlet(np.ones(n)))

    # 4. edge‑concentrated power‑law: (i·(n‑i+1))^α
    idx = np.arange(n, dtype=np.float64) + 1.0
    prod = idx * (n - idx + 1.0)
    for α in np.linspace(-6.0, -0.05, 30):
        shape = np.where(prod > 0, prod ** α, 0.0)
        s = shape.sum()
        if s > 0:
            shape /= s
            cand.append(shape)

    # 5. inverse‑distance power‑law: (i+1)^‑p + (n‑i)^‑p
    i = np.arange(n, dtype=np.float64)
    for p in np.linspace(0.5, 5.0, 20):
        shape = (i + 1) ** (-p) + (n - i) ** (-p)
        shape /= shape.sum()
        cand.append(shape)

    # 6. double‑exponential (two-sided Laplace)
    for γ in np.geomspace(0.2, 10.0, 20):
        shape = np.exp(-γ * i) + np.exp(-γ * (n - 1 - i))
        shape /= shape.sum()
        cand.append(shape)

    # 7. half‑wave sinusoid with a power
    base = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    for q in [1, 2, 3, 4, 5]:
        shape = base ** q
        shape = np.maximum(shape, 0.0)
        shape /= shape.sum()
        cand.append(shape)

    # 8. triangular tent (peak in the centre) – sometimes useful
    tri = np.concatenate([np.arange(1, n // 2 + 1),
                          np.arange(n // 2, 0, -1)])[:n].astype(float)
    tri /= tri.sum()
    cand.append(tri)

    # 9. spike patterns (k equally spaced non‑zero entries)
    for k in (8, 16, 32, 64, 128, 256):
        a = np.zeros(n, dtype=np.float64)
        if k > 0:
            pos = np.round(np.linspace(0, n - 1, k)).astype(int)
            a[pos] = 1.0 / k
        cand.append(a)

    # final sanity: clip and renormalise (protect against accidental zeros)
    final = []
    for a in cand:
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            continue
        a = a / s
        final.append(a)
    return final


# ----------------------------------------------------------------------
# Adam optimiser (smooth‑max surrogate) ---------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
) -> Tuple[np.ndarray, float]:
    """
    Adam optimisation on the log‑sum‑exp surrogate of max(a*a).
    Returns the best vector found (still on the simplex) and its exact objective.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.08
    lam = 0.5            # start temperature for smooth‑max
    lam_factor = 1.03

    best_a = a0.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max → simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            # occasional progress report
            if it % 20_000 == 0:
                print(f"[Adam iter={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature schedule → true max
        lam = min(lam * lam_factor, 1e6)

        # tiny random shake every few thousand iterations (helps escape plateaus)
        if it % 7_000 == 0 and time.time() + 5 > deadline:
            logits += (np.random.rand(n) - 0.5) * 5e-3

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Exponentiated‑gradient sub‑gradient optimiser ---------------------------
def _subgradient_opt(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
    eta0: float = 0.7,
) -> Tuple[np.ndarray, float]:
    """
    Multiplicative sub‑gradient descent with a 1/√t step‑size.
    Returns the best vector seen and its exact objective.
    """
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s

    best_a = a.copy()
    best_val = float("inf")
    stall = 0
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            stall = 0
            if it % 5_000 == 0:
                print(f"[Subgrad iter={it:07d}] new best {val:.6f}")
        else:
            stall += 1

        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # decaying step size
        eta = eta0 / np.sqrt(it + 1)

        # multiplicative update and re‑normalisation
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:                     # re‑initialise if we get stuck at zero
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        # occasional random perturbation after long stalls
        if stall > 10_000:
            a = np.maximum(a + 0.02 * (np.random.rand(n) - 0.5), 0.0)
            a /= a.sum()
            stall = 0

        it += 1
    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing local search ----------------------------------------
def _simulated_annealing(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_iters: int = 250_000,
) -> Tuple[np.ndarray, float]:
    """
    Metropolis‑type hill climbing: move a random fraction of mass from i→j.
    Worse moves are accepted with probability exp(-Δ/T).  Temperature
    decreases geometrically.
    Returns the best vector encountered.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    cur_val = 2.0 * n * conv.max()
    best_a = a.copy()
    best_val = cur_val

    # temperature schedule (geometric)
    T0 = cur_val * 0.2
    Tend = 1e-5
    it = 0
    while time.time() < deadline and it < max_iters:
        T = T0 * ((Tend / T0) ** (it / max_iters))

        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue
        # move up to 20 % of the source mass
        delta = a[i] * (np.random.rand() * 0.2)
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()
        diff = val2 - cur_val
        accept = diff < 0.0 or np.random.rand() < np.exp(-diff / max(T, 1e-12))

        if accept:
            a = a2
            cur_val = val2
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
        it += 1
    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> List[float]:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The function respects a 1000 s wall‑time budget and returns the best
    sequence (as a Python list) found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # tiny safety margin

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Phase 0 – build a large pool of warm‑starts (n = 4096)
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    pool = _candidate_pool(n)

    # Quick evaluation of all candidates, keep the 10 best
    scored = []
    for idx, cand in enumerate(pool):
        if time.time() >= deadline:
            break
        val = _objective(cand, conv_fft_len)
        scored.append((val, cand))
        if idx % 100 == 0:
            print(f"[Pool] evaluated {idx+1}/{len(pool)} candidates")
    scored.sort(key=lambda x: x[0])
    top_candidates = [c for (_, c) in scored[:10]]
    print(f"[Pool] best pool objective = {scored[0][0]:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Adam refinement on the top candidates
    for cid, cand in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        cand_deadline = min(deadline - 5.0, time.time() + 12.0)
        a_adam, val_adam = _adam_optimize(
            cand, n, conv_fft_len, corr_fft_len, cand_deadline, max_iters=150_000
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Phase 1‑Adam cand={cid}] new best {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – long exponentiated‑gradient sub‑gradient on the current best
    if best_seq is not None and time.time() < deadline - 5:
        sub_deadline = min(deadline - 5.0, time.time() + 120.0)
        a_sub, val_sub = _subgradient_opt(
            best_seq, n, conv_fft_len, corr_fft_len, sub_deadline, max_iters=800_000
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Phase 2‑Subgrad] new best {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – simulated‑annealing local search
    if best_seq is not None and time.time() < deadline - 5:

        a_loc, val_loc = _simulated_annealing(
            best_seq, n, conv_fft_len, local_deadline, max_iters=300_000

        if val_loc < best_val:

            print(f"[Phase 3‑SA] improved best to {val_loc:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – upscale to 8192 if time permits
    if time.time() + 40 < deadline:

        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        # linear interpolation (preserves order)
        x_old = np.linspace(0.0, 1.0, best_seq.shape[0], endpoint=False)
        x_new = np.linspace(0.0, 1.0, n_big, endpoint=False)
        a_big = np.interp(x_new, x_old, best_seq)

        a_big = np.maximum(a_big, 0.0)

            best_seq = a_big_adam.copy()
            n = n_big
            conv_fft_len = conv_fft_len_big

            print(f"[Upscale 8192‑Adam] new best {val_big_adam:.6f}")

            if val_big_loc < best_val:

                best_seq = a_big_loc.copy()
                print(f"[Upscale 8192‑SA] improved best {val_big_loc:.6f}")

    # ------------------------------------------------------------------

        corr_fft_len_big2 = _next_pow2(3 * n_big2 - 2)

        x_old = np.linspace(0.0, 1.0, best_seq.shape[0], endpoint=False)
        x_new = np.linspace(0.0, 1.0, n_big2, endpoint=False)
        a_big2 = np.interp(x_new, x_old, best_seq)

        a_big2 = np.maximum(a_big2, 0.0)
        a_big2 /= a_big2.sum()

        # brief Adam on the very fine grid

        a_big2_adam, val_big2_adam = _adam_optimize(
            a_big2, n_big2, conv_fft_len_big2, corr_fft_len_big2,
            big_deadline, max_iters=80_000,
        )
        if val_big2_adam < best_val:
            best_val = val_big2_adam
            best_seq = a_big2_adam.copy()
            n = n_big2
            conv_fft_len = conv_fft_len_big2
            corr_fft_len = corr_fft_len_big2
            print(f"[Upscale 16384‑Adam] new best {val_big2_adam:.6f}")

        # final sub‑gradient on 16384

            a_big2_sub, val_big2_sub = _subgradient_opt(

    # Fallback – guarantee a valid return value
    if best_seq is None:
```
