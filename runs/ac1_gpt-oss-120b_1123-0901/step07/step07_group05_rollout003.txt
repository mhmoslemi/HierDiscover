"""
Propose a low‑evaluation step‑function.
-------------------------------------------------
The search is split into four stages:

1. **Broad parametric pool** – generate thousands of diversified
   candidates (uniform, Dirichlet draws, symmetric/asymmetric
   exponentials, power‑laws, linear/triangular, half‑wave sinusoid,
   sparse spikes).  All candidates are normalised to sum = 1 and
   evaluated with an exact FFT‑based convolution.  The best ~12 are
   retained.

2. **Deep optimisation** – each retained start is refined by
   *Adam* on a smooth‑max surrogate of the convolution (slowly
   increasing temperature) followed by an *exponentiated‑gradient*
   sub‑gradient phase with many more iterations than before.
   Both phases keep the best exact objective seen.

3. **Local stochastic refinement** – cheap hill‑climbing on the
   logits of the current best vector, accepting any improvement.

4. **Scale‑up** – if time permits, the current best is interpolated
   to a larger grid (n = 8192) and the deep optimiser is run again.
   This often yields a modest extra reduction because a finer
   discretisation can flatten the self‑convolution further.

All loops respect a global deadline (default 1000 s) and the
algorithm never returns an empty list – a uniform vector is used as
fallback.  The resulting sequence satisfies the evaluator’s clipping
rules (non‑negative, ≤ 1000) and is returned as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2*len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation w ★ a needed for the gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k−(n−1)]   for i=0…n−1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact objective F = 2·n·max(a*a) / (sum(a)²).  Assumes sum(a)>0."""
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c / (a.sum() ** 2))

def _adam_opt(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    init_best_seq: np.ndarray | None,
    init_best_val: float,
    max_iter: int = 600_000,
    lr: float = 0.04,
    lam_start: float = 0.5,
    lam_factor: float = 1.005,
    lam_cap: float = 1e4,
) -> tuple[np.ndarray, float]:
    """Adam on a log‑sum‑exp surrogate of the max‑convolution."""
    eps = 1e-15
    logits = np.log(a0 + eps)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8

    lam = lam_start
    best_seq = init_best_seq
    best_val = init_best_val

    last_improve = time.time()
    it = 0

    while time.time() < deadline and it < max_iter:
        # Soft‑max to stay on the simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # Exact objective (for tracking)
        val = _objective(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Adam] iter={it:07d} new best {val:.6f}")

        # Smooth‑max weights
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # Gradient of the smooth‑max surrogate w.r.t. a
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # Chain rule through soft‑max (grad w.r.t logits)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # Temperature schedule
        lam = min(lam * lam_factor, lam_cap)

        # Small random shake if stuck for >5 s
        if time.time() - last_improve > 5.0:
            logits += (np.random.rand(n) - 0.5) * 2e-2
            last_improve = time.time()

        it += 1

    return best_seq, best_val


def _subgradient_opt(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    init_best_seq: np.ndarray | None,
    init_best_val: float,
    max_iter: int = 150_000,
    eta0: float = 0.8,
    decay: float = 0.99998,
) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient sub‑gradient optimisation on the exact max‑conv."""
    a = init_best_seq if init_best_seq is not None else a0.copy()
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a /= a.sum()

    eta = eta0
    best_seq = a.copy()
    best_val = init_best_val

    last_improve = time.time()
    it = 0

    while time.time() < deadline and it < max_iter:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Subgrad] iter={it:07d} new best {val:.6f}")

        # Active set (indices attaining max)
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # Exponentiated‑gradient update (keeps a≥0)
        a = a * np.exp(-eta * grad_a)

        # Re‑normalise onto the simplex
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        eta *= decay
        it += 1

        # Stop early if no improvement for a while
        if time.time() - last_improve > 8.0:
            break

    return best_seq, best_val


def _local_random_search(
    a_best: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """Very cheap hill‑climbing in log‑space."""
    logits_best = np.log(a_best + 1e-15)

    while time.time() < deadline:
        noise = np.random.randn(n) * 0.03
        logits = logits_best + noise
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        val = _objective(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            a_best = a.copy()
            logits_best = np.log(a_best + 1e-15)
            print(f"[Local] improved to {val:.6f}")

    return a_best, best_val


def _simulated_annealing(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    init_val: float,
    max_steps: int = 250_000,
    temp_start: float = 0.02,
    temp_decay: float = 0.99995,
) -> tuple[np.ndarray, float]:
    """
    Simple SA on the simplex: move a small amount from i to j.
    Accepts worsening moves with probability exp(-Δ/ T).
    """
    a = a0.copy()
    val = init_val
    best_a = a.copy()
    best_val = val
    T = temp_start
    it = 0
    while time.time() < deadline and it < max_steps:
        i, j = np.random.choice(n, size=2, replace=False)
        # propose moving δ from i to j (δ positive if taken from i)
        max_move = min(a[i], T)  # scale move with temperature
        delta = np.random.uniform(0, max_move)
        a_prop = a.copy()
        a_prop[i] -= delta
        a_prop[j] += delta
        # no need to renormalise – sum is preserved
        val_prop = _objective(a_prop, n, conv_fft_len)
        if val_prop < best_val:
            best_val = val_prop
            best_a = a_prop.copy()
            print(f"[SA] step={it:07d} new best {val_prop:.6f}")

        # Metropolis accept
        if val_prop < val or np.random.rand() < math.exp(-(val_prop - val) / max(T, 1e-12)):
            a = a_prop
            val = val_prop

        T *= temp_decay
        it += 1

        # occasional restart when temperature becomes tiny
        if it % 50000 == 0 and T < 1e-6:
            T = temp_start

    return best_a, best_val


def _deep_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Run Adam → sub‑gradient → local search → SA.
    Update the globally best sequence/value.
    """
    # Adam phase
    best_seq, best_val = _adam_opt(
        a0,
        n,
        conv_fft_len,
        corr_fft_len,
        deadline,
        global_best_seq,
        global_best_val,
    )
    # Sub‑gradient phase
    best_seq, best_val = _subgradient_opt(
        a0,
        n,
        conv_fft_len,
        corr_fft_len,
        deadline,
        best_seq,
        best_val,
    )
    # Cheap local random search
    best_seq, best_val = _local_random_search(
        best_seq,
        n,
        conv_fft_len,
        deadline,
        best_val,
    )
    # Simulated annealing for a final push
    best_seq, best_val = _simulated_annealing(
        best_seq,
        n,
        conv_fft_len,
        deadline,
        best_val,
    )
    return best_seq, best_val

def _generate_candidates(n: int, warm_start: np.ndarray | None) -> list[np.ndarray]:
    """Create a diverse set of candidate vectors (all sum to 1)."""
    candidates = []

    # Warm‑start and its shifts
    if warm_start is not None:
        candidates.append(warm_start.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            candidates.append(np.roll(warm_start, shift))

    # Uniform baseline
    candidates.append(np.full(n, 1.0 / n))

    # Dirichlet draws (different concentrations)
    for conc in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(15):
            cand = np.random.dirichlet(np.full(n, conc))
            candidates.append(cand)

    i = np.arange(n, dtype=np.float64)
    # Symmetric exponentials
    for alpha in np.logspace(-2, 1.0, num=14):
        edge = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
        edge /= edge.sum()
        for w in (0.0, 0.3, 0.6, 0.9, 1.0):
            shape = w * edge + (1.0 - w) / n
            shape /= shape.sum()
            candidates.append(shape)

    # Symmetric power‑laws
    for p in np.linspace(0.2, 6.0, num=14):
        edge = (i + 1.0) ** (-p) + (n - i) ** (-p)
        edge /= edge.sum()
        for w in (0.0, 0.5, 1.0):
            shape = w * edge + (1.0 - w) / n
            shape /= shape.sum()
            candidates.append(shape)

    # Asymmetric two‑sided exponentials (coarse grid)
    alphas = np.logspace(-2, 1.0, num=8)
    for wL in (0.0, 0.3, 0.6):
        for wR in (0.0, 0.3, 0.6):
            if wL + wR > 1.0:
                continue
            wU = 1.0 - wL - wR
            for aL in alphas:
                for aR in alphas:
                    left = np.exp(-aL * i)
                    right = np.exp(-aR * (n - 1 - i))
                    shape = wL * left + wR * right + wU / n
                    shape /= shape.sum()
                    candidates.append(shape)

    # Linear and triangular shapes
    lin = np.linspace(1.0, 0.0, n)
    tri = np.concatenate([np.linspace(0, 1, n // 2, endpoint=False),
                          np.linspace(1, 0, n - n // 2)])
    for base in (lin, tri):
        base = np.maximum(base, 0.0)
        base /= base.sum()
        for w in (0.0, 0.5, 1.0):
            cand = w * base + (1.0 - w) / n
            cand /= cand.sum()
            candidates.append(cand)

    # Half‑wave sinusoid
    sinus = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    sinus = np.maximum(sinus, 0.0)
    sinus /= sinus.sum()
    candidates.append(sinus)

    # Sparse spike vectors (few non‑zeros)
    for spikes in (3, 5, 8):
        for _ in range(20):
            idx = np.random.choice(n, size=spikes, replace=False)
            weights = np.random.dirichlet(np.ones(spikes))
            vec = np.zeros(n)
            vec[idx] = weights
            vec /= vec.sum()
            candidates.append(vec)

    # Ensure non‑negativity and unit sum
    final = []
    for c in candidates:
        c = np.maximum(c, 0.0)
        s = c.sum()
        if s < 1e-12:
            continue
        final.append(c / s)
    return final


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Main entry point – searches for a non‑negative step function that
    minimises the supplied evaluation function within the given time
    budget.  Returns a plain Python list of heights.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # safety margin

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Primary grid size (was previously 4096)
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # Warm‑start from any previously discovered sequence
    warm_start = globals().get("height_sequence_1")
    if warm_start is not None:
        ws = np.asarray(warm_start, dtype=np.float64)
        if ws.shape[0] > n:
            ws = ws[:n]
        elif ws.shape[0] < n:
            ws = np.pad(ws, (0, n - ws.shape[0]))
        ws = np.maximum(ws, 0.0)
        s = ws.sum()
        if s >= 0.01:
            warm_start = ws / s
        else:
            warm_start = None
    else:
        warm_start = None

    # ------------------------------------------------------------------
    # Stage 1 – cheap exhaustive sweep over a large parametric pool
    all_candidates = _generate_candidates(n, warm_start)

    cand_vals: list[tuple[float, np.ndarray]] = []
    for idx, cand in enumerate(all_candidates):
        if time.time() >= deadline:
            break
        val = _objective(cand, n, conv_fft_len)
        cand_vals.append((val, cand))
        if (idx + 1) % 500 == 0:
            print(f"[Stage 1] evaluated {idx + 1}/{len(all_candidates)} candidates")

    # Keep the top few for expensive refinement
    cand_vals.sort(key=lambda x: x[0])
    top_k = 12
    top_candidates = [c for _, c in cand_vals[:top_k]]
    print(f"[Stage 1] completed {len(cand_vals)} evaluations, best = {cand_vals[0][0]:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – deep optimisation of the selected starts
    for start_id, a0 in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        print(f"--- Deep optimisation start #{start_id + 1}/{len(top_candidates)} ---")
        best_seq, best_val = _deep_optimize(
            a0,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            best_seq,
            best_val,
        )
        print(f"  after start #{start_id + 1}: best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – cheap stochastic hill‑climbing on the global best
    if best_seq is not None:
        print("--- Final local refinement ---")
        best_seq, best_val = _local_random_search(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            best_val,
        )

    # ------------------------------------------------------------------
    # Stage 4 – up‑sample to a larger grid if time permits
    time_left = deadline - time.time()
    if time_left > 60.0:                # generosity threshold
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        if best_seq is not None:
            # Linear interpolation to the finer grid
            x_old = np.linspace(0, 1, n, endpoint=False)
            x_new = np.linspace(0, 1, n_big, endpoint=False)
            a_big = np.interp(x_new, x_old, best_seq)
            a_big = np.maximum(a_big, 0.0)
            a_big /= a_big.sum()
        else:
            a_big = np.random.dirichlet(np.ones(n_big))

        print("--- Optimising on larger grid (n = %d) ---" % n_big)
        best_seq, best_val = _deep_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            best_seq,
            best_val,
        )
        n = n_big
        conv_fft_len = conv_fft_len_big
        corr_fft_len = corr_fft_len_big

    # ------------------------------------------------------------------
    # Fallback – uniform vector if nothing succeeded
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)
        best_val = _objective(best_seq, 4096, _next_pow2(2 * 4096 - 1))
        print("[Fallback] using uniform vector")

    # Apply evaluator's clipping (non‑negative, ≤ 1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"*** Search finished: best objective ≈ {best_val:.6f} (n={best_seq.shape[0]})")
    return best_seq.tolist()
