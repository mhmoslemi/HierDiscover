def adam_optimize(a0, deadline, max_iters=200_000, lr=0.05, lam_start=0.2, lam_factor=1.02, noise_sigma=0.02):

    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1=0.9; beta2=0.999; eps_adam=1e-8
    lam=lam_start
    best_local_a = a0.copy()
    best_local_val = float('inf')
    it=0
    while it<max_iters and time.time()<deadline:
        # compute a
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # conv and val
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum()**2)  # a.sum=1
        if val < best_local_val:
            best_local_val = val
            best_local_a = a.copy()
        # compute weight for smooth max
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0/w.size)
        else:
            w /= w_sum
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)  # shape same as logits

        # optionally add noise
        if noise_sigma>0:
            grad_log += np.random.normal(scale=noise_sigma, size=grad_log.shape)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it+1))
        v_hat = v / (1.0 - beta2 ** (it+1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e6)
        # degrade lr occasionally
        if it % 50000 == 0 and it>0:
            lr *= 0.9
        it+=1
    return best_local_a, best_local_val

"""
Propose a step‑function (non‑negative vector) that minimizes

        F(a) = 2·n·max(a∗a) / (sum(a)^2)

The algorithm is a hybrid of
* massive random‑spike sampling,
* a long Adam optimisation on a smooth‑max surrogate (log‑sum‑exp)
  with a very slow temperature schedule and injected Gaussian noise,
* a few noisy‑restart Adam runs,
* an exact‑max exponentiated‑gradient (sub‑gradient) refinement,
* a cheap final random hill‑climb.

All phases respect the time budget (default 1000 s) and the best
sequence found so far is always returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a (length 2·len(a)-1) via FFT.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a length‑len(a) vector:
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i = 0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _eval_objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact evaluation of the objective (no security checks)."""
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    # sum(a) is 1 for our normalised vectors, but keep the formula generic
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_c / (s * s))


# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    lr: float = 0.07,
    lam_start: float = 0.1,
    lam_factor: float = 1.02,
    noise_sigma: float = 0.02,
) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth‑max (log‑sum‑exp) surrogate.
    Returns the best vector encountered (sum≈1) and its exact objective.
    """
    eps = 1e-15
    a0 = np.clip(a0, eps, None)                     # avoid log(0)
    logits = np.log(a0)                             # initialise on the simplex
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8

    lam = lam_start
    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while it < max_iters and time.time() < deadline:
        # ---- simplex (soft‑max) projection -----------------
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # ---- exact objective --------------------------------
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # ---- smooth‑max weights -----------------------------
        # log‑sum‑exp with temperature 1/lam
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---- gradient of the surrogate -----------------------
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # optional stochastic perturbation
        if noise_sigma > 0:
            grad_log += np.random.normal(scale=noise_sigma, size=grad_log.shape)

        # ---- Adam update on logits ---------------------------
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # ---- temperature schedule ----------------------------
        lam = min(lam * lam_factor, 1e6)

        # gentle learning‑rate decay
        if it % 50_000 == 0 and it > 0:
            lr *= 0.9

        # occasional shake when we are close to the deadline
        if it % 4_000 == 0 and time.time() > deadline - 60:
            logits += (np.random.rand(n) - 0.5) * 2e-2

        it += 1

    return best_a, best_val


def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 0.8,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient phase using exponentiated‑gradient updates.
    Returns the best vector seen and its exact objective.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while it < max_iters and time.time() < deadline:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5_000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # ---- active set (indices attaining the max) ------------
        active = np.isclose(conv, max_c, atol=1e-14)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # decreasing step size η_t = η₀ / √(t+1)
        eta = eta0 / np.sqrt(it + 1)

        # exponentiated‑gradient update (mirror descent on the simplex)
        a = a * np.exp(-eta * grad_a)

        # renormalise – protect against under‑/overflow
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 70_000,
) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climbing: move a random fraction of mass from one
    coordinate to another, keep the move if the objective improves.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()

    it = 0
    while it < max_moves and time.time() < deadline:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        # move up to 30 % of the source mass
        delta = a[i] * (np.random.rand() * 0.3)

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5_000 == 0:
                print(f"[Local move {it:06d}] improved to {val2:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the given time budget and always returns the
    best sequence (as a Python list) found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # problem size
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # warm start (if a previous solution is available)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None

    # ------------------------------------------------------------------
    # --------------------------------------------------- fast random phase
    # generate many cheap random candidates (spike‑sets, Dirichlet,
    # two‑Gaussian mixtures) and keep the best.
    best_seq = None
    best_val = float("inf")

    # a tiny helper for evaluation (uses exact objective)
    def try_candidate(a):
        nonlocal best_seq, best_val
        val = _eval_objective(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Random] new best {val:.6f}")

    # include warm / uniform as baseline
    if warm is not None:
        try_candidate(warm)
    else:
        try_candidate(np.full(n, 1.0 / n))

    # random Dirichlet (full support)
    for _ in range(12):
        a = np.random.dirichlet(np.ones(n))
        try_candidate(a)

    # edge‑exponential families
    for gamma in (0.2, 0.5, 1.0, 2.0):
        left = np.exp(-gamma * np.arange(n, dtype=np.float64))
        right = left[::-1]
        a = left + right
        a /= a.sum()
        try_candidate(a)

    # half‑wave sinusoid
    a = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    try_candidate(a)

    # spike candidates – many random supports
    for k in (32, 64, 128, 256, 512):
        for _ in range(8):                               # 8 random draws per k
            idx = np.random.choice(n, size=k, replace=False)
            weights = np.random.dirichlet(np.ones(k))
            a = np.zeros(n, dtype=np.float64)
            a[idx] = weights
            try_candidate(a)

    # two‑Gaussian mixtures
    xs = np.arange(n, dtype=np.float64) / n
    for _ in range(10):
        mu1, mu2 = np.random.rand(2)
        sigma1, sigma2 = np.random.rand(2) * 0.15 + 0.02
        g1 = np.exp(-0.5 * ((xs - mu1) / sigma1) ** 2)
        g2 = np.exp(-0.5 * ((xs - mu2) / sigma2) ** 2)
        a = g1 + g2
        a /= a.sum()
        try_candidate(a)

    # ------------------------------------------------------------------
    # --------------------------------------------------- Adam refinement
    # allocate ~60 % of the remaining budget to a long Adam run
    remaining = deadline - time.time()
    adam_deadline = time.time() + 0.60 * remaining
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    a_adam, val_adam = _adam_optimize(
        best_seq,
        n,
        conv_fft_len,
        corr_fft_len,
        adam_deadline,
        max_iters=300_000,
        lr=0.07,
        lam_start=0.08,
        lam_factor=1.015,
        noise_sigma=0.025,
    )
    if val_adam < best_val:
        best_val = val_adam
        best_seq = a_adam.copy()
        print(f"[Adam main] improved best {val_adam:.6f}")

    # --------------------------------------------------- Noisy‑restart Adam
    # a few short Adam runs from noisy perturbations of the current best
    for restart in range(4):
        if time.time() >= deadline - 30:
            break
        # perturb logits with Gaussian noise
        eps = 1e-12
        logits = np.log(np.clip(best_seq, eps, None))
        logits += np.random.normal(scale=0.3, size=n)   # fairly large perturbation
        a0 = np.exp(logits - np.max(logits))
        a0 /= a0.sum()

        r_deadline = min(deadline - 5, time.time() + 0.12 * (deadline - time.time()))
        a_tmp, val_tmp = _adam_optimize(
            a0,
            n,
            conv_fft_len,
            corr_fft_len,
            r_deadline,
            max_iters=180_000,
            lr=0.06,
            lam_start=0.08,
            lam_factor=1.02,
            noise_sigma=0.02,
        )
        if val_tmp < best_val:
            best_val = val_tmp
            best_seq = a_tmp.copy()
            print(f"[Adam restart {restart}] new best {val_tmp:.6f}")

    # --------------------------------------------------- Sub‑gradient refinement
    if time.time() < deadline - 20:
        sg_deadline = deadline - 15
        a_sub, val_sub = _subgradient_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            sg_deadline,
            max_iters=250_000,
            eta0=0.85,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Subgrad] improved best {val_sub:.6f}")

    # --------------------------------------------------- Cheap local search
    if time.time() < deadline - 5:
        a_loc, val_loc = _local_random_search(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=80_000,
        )
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()
            print(f"[Local] improved best {val_loc:.6f}")

    # ------------------------------------------------------------------
    # fallback – ensure a valid non‑empty return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the evaluator’s accepted range (non‑negative, ≤1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
