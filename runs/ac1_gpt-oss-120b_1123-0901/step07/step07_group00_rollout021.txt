"""
Improved search for a non‑negative step‑function that minimizes

        F(a) = 2·n·max(a∗a) / (sum(a)^2)

The algorithm works in three stages.

Stage 1 – fast multi‑restart exploration on a medium sized grid (n≈2048):
    * Random start vectors (Dirichlet with random concentration, sparse spikes,
      exponential / power‑law tails and an arcsine‑like family) are generated.
    * Each start is refined by a short Adam optimisation on the smooth‑max
      (log‑sum‑exp) surrogate.  The best‑found vector is kept.

Stage 2 – up‑sample the current best to a fine grid (n≈8192) and perform a
    long Adam refinement (large number of iterations, slowly increasing
    temperature) followed by an exact‑max sub‑gradient phase (exponentiated‑
    gradient updates with a diminishing step‑size).

Stage 3 – cheap random local moves (mass transfer between two positions)
    that accept only improvements.  This often squeezes a few extra digits.

All heavy work is done with FFT‑based convolution / cross‑correlation,
so each iteration is O(n log n) and fits easily into the 1000 s budget.
The routine respects the deadline, always returns the best sequence found,
and prints progress information.

The code contains no closures or lambda expressions and works with up to two
CPU cores (the search is single‑process but can be parallelised by the caller
if desired).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2·len(a)-1) via FFT.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a length‑len(a) vector:
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i = 0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    lr0: float = 0.09,
    lam0: float = 0.01,
    lam_factor: float = 1.02,
) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth‑max (log‑sum‑exp) surrogate.
    Returns the best vector encountered (sum≈1) and its exact objective.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)            # initialise on the simplex
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = lr0
    lam = lam0

    best_a = a0.copy()
    # objective for the best_a
    conv = _full_convolution(a0, conv_fft_len)
    max_c = conv.max()
    best_val = 2.0 * n * max_c / (a0.sum() ** 2)

    it = 0
    while time.time() < deadline and it < max_iters:
        # ---------- simplex projection (soft‑max) ----------
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # ---------- exact objective ----------
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # ---------- smooth‑max weights ----------
        # weight each entry of the convolution according to its magnitude
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---------- gradient of the surrogate ----------
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # ---------- chain rule through soft‑max ----------
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ---------- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # ---------- temperature schedule ----------
        lam = min(lam * lam_factor, 1e7)

        # ---------- occasional shake near deadline ----------
        if it % 4000 == 0 and time.time() > deadline - 30:
            logits += (np.random.rand(n) - 0.5) * 1e-2

        # ---------- learning‑rate decay ----------
        if it % 50000 == 0 and it > 0:
            lr *= 0.93

        it += 1

    return best_a, best_val


def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 0.6,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient phase (exponentiated‑gradient updates).
    Returns the best vector seen and its exact objective.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active set (indices where the maximum is attained)
        active = np.isclose(conv, max_c, atol=1e-14)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # decreasing step size
        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update (mirror descent on the simplex)
        a = a * np.exp(-eta * grad_a)

        # renormalise – protect against underflow
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 80_000,
) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climbing: move a random fraction of mass from one index
    to another, keep the move if the objective improves.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()

    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        # move up to 30 % of the source mass
        delta = a[i] * (np.random.rand() * 0.3)

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5000 == 0:
                print(f"[Local move {it:06d}] improved to {val2:.6f}")

        it += 1

    return best_a, best_val

def _random_candidate(n: int) -> np.ndarray:
    """Draw a random feasible start vector for the coarse grid."""
    r = np.random.rand()
    if r < 0.4:
        # Dirichlet with random concentration parameter
        conc = np.random.rand() * 4.0 + 0.1   # between 0.1 and 4.1
        a = np.random.dirichlet(np.full(n, conc))
    elif r < 0.65:
        # sparse spikes, random number of spikes
        k = np.random.randint(2, max(2, n // 64))
        a = np.zeros(n, dtype=np.float64)
        pos = np.random.choice(n, k, replace=False)
        a[pos] = 1.0 / k
    elif r < 0.85:
        # exponential tails (edge‑concentrated)
        gamma = np.random.choice([0.2, 0.5, 1.0, 2.0])
        left = np.exp(-gamma * np.arange(n, dtype=np.float64))
        shape = left + left[::-1]
        shape /= shape.sum()
        a = shape
    else:
        # arcsine‑like shape: a_i ∝ 1/√(i·(n‑1‑i))
        idx = np.arange(1, n + 1, dtype=np.float64)
        prod = idx * (n - idx + 1.0)
        shape = np.where(prod > 0, 1.0 / np.sqrt(prod), 0.0)
        if shape.sum() == 0:
            shape = np.full(n, 1.0 / n)
        else:
            shape /= shape.sum()
        a = shape
    return a

    """

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # Stage 1 – coarse multi‑restart search on a medium grid
    n_coarse = 2048

    corr_fft_len_c = _next_pow2(3 * n_coarse - 2)

    best_seq = None
    best_val = float("inf")

    # reserve about 350 s for the fine stage (up‑sample + long optimisation)
    coarse_deadline = deadline - 350.0
    restart_counter = 0

    print("[Stage 1] coarse multi‑restart exploration")
    while time.time() < coarse_deadline:
        restart_counter += 1
        a0 = _random_candidate(n_coarse)

        # short Adam run – give at most ~15 s or 120 k iterations, whichever first
        sub_deadline = min(deadline - 5, time.time() + 15.0)
        a_opt, val_opt = _adam_optimize(
            a0,
            n_coarse,
            conv_fft_len_c,
            corr_fft_len_c,
            sub_deadline,
            max_iters=120_000,
            lr0=0.07,
            lam0=0.02,
            lam_factor=1.03,
        )
        if val_opt < best_val:
            best_val = val_opt
            best_seq = a_opt.copy()
            print(f"[Restart {restart_counter}] new coarse best {val_opt:.6f}")

        # quick check – if we are out of time, break
        if time.time() + 5.0 >= coarse_deadline:
            break

    # ------------------------------------------------------------------
    # If nothing was found (should not happen), fallback to uniform
    if best_seq is None:
        best_seq = np.full(n_coarse, 1.0 / n_coarse, dtype=np.float64)
        best_val = 2.0 * n_coarse * (1.0 / n_coarse)   # =2.0

    # ------------------------------------------------------------------
    # Stage 2 – up‑sample to a fine grid and perform deep optimisation
    n_fine = 8192
    conv_fft_len_f = _next_pow2(2 * n_fine - 1)
    corr_fft_len_f = _next_pow2(3 * n_fine - 2)

    # linear interpolation to the finer grid
    x_c = np.linspace(0, 1, n_coarse, endpoint=False)
    x_f = np.linspace(0, 1, n_fine, endpoint=False)
    a_fine = np.interp(x_f, x_c, best_seq)

    a_fine = np.maximum(a_fine, 0.0)
    a_fine = a_fine / a_fine.sum()

    print("[Stage 2] fine Adam optimisation (long run)")
    a_fine_adam, val_fine_adam = _adam_optimize(
        a_fine,
        n_fine,
        conv_fft_len_f,
        corr_fft_len_f,
        deadline,
        max_iters=500_000,
        lr0=0.09,
        lam0=0.01,
        lam_factor=1.02,
    )
    if val_fine_adam < best_val:
        best_val = val_fine_adam
        best_seq = a_fine_adam.copy()
        print(f"[Fine] Adam new best {val_fine_adam:.6f}")

    # ------------------------------------------------------------------
    # Exact‑max sub‑gradient refinement (long)
    print("[Stage 2] sub‑gradient refinement (long)")
    a_fine_sub, val_fine_sub = _subgradient_optimize(
        best_seq,
        n_fine,
        conv_fft_len_f,
        corr_fft_len_f,
        deadline,
        max_iters=600_000,
        eta0=0.55,
    )
    if val_fine_sub < best_val:
        best_val = val_fine_sub
        best_seq = a_fine_sub.copy()
        print(f"[Fine] Sub‑gradient new best {val_fine_sub:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – cheap random hill‑climbing on the fine grid
    if time.time() < deadline - 5:
        print("[Stage 3] local random search")
        a_fine_loc, val_fine_loc = _local_random_search(
            best_seq,
            n_fine,
            conv_fft_len_f,
            deadline,
            max_moves=100_000,
        )
        if val_fine_loc < best_val:
            best_val = val_fine_loc
            best_seq = a_fine_loc.copy()
            print(f"[Fine] Local search new best {val_fine_loc:.6f}")

    # ------------------------------------------------------------------
    # Ensure a valid return (non‑negative, bounded)
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
