def subgradient_exact(a, n, conv_fft_len, corr_fft_len, max_iters, deadline, best_seq, best_val):
   a = np.copy(a)
   a = np.maximum(a, 0)
   a /= a.sum()
   eta = 0.5
   decay = 0.9998
   for it in range(max_iters):
        if time.time() >= deadline: break
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_seq = a.copy()
        # compute gradient at active indices
        active = conv == max_c  # using tolerance
        w = np.zeros_like(conv)
        w[active] = 1.0 / np.sum(active)
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)
        # gradient descent with projection onto simplex:
        a = a - eta * grad
        # projection onto simplex:
        # Use algorithm from Michelot: sort descending.
        u = np.sort(a)[::-1]
        cssv = np.cumsum(u) - 1
        rho = np.where(u - cssv / np.arange(1, len(a)+1) > 0)[0][-1]
        theta = cssv[rho] / (rho+1)
        a = np.maximum(a - theta, 0)
        # step size decay
        eta *= decay
        # early break if no improvement for some time
   return best_seq, best_val

   uphill moves.

4. **Two‑grid strategy** – Optimisation starts on a medium size grid
   (n≈4096) to keep the per‑iteration cost low, then the best solution is
   interpolated to a finer grid (n≈8192) for a final refinement.  All loops
   respect a global deadline (budget_s = 1000 s).

The programme returns a plain Python list of non‑negative numbers.  The
returned sequence respects the evaluator’s clipping (≤ 1000) and is
normalised to have sum ≈ 1 (scaling does not affect the objective).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2*len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation needed for the gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact objective F = 2·n·max(a*a)/(sum(a)²).  Assumes sum(a)>0."""
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c / (a.sum() ** 2))


# ----------------------------------------------------------------------
def _adam_smooth_max(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
    max_iter: int = 600_000,
    lr: float = 0.045,
    lam_start: float = 2.0,
    lam_factor: float = 1.008,
) -> tuple[np.ndarray, float]:
    """Adam on a log‑sum‑exp surrogate of the max‑convolution."""
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lam = lam_start
    lam_cap = 1e6

    last_improve = time.time()
    it = 0

    while time.time() < deadline and it < max_iter:
        # soft‑max to stay on the simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective (used for bookkeeping)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Adam] iter={it:06d}  best={val:.6f}")

        # smooth‑max weight (log‑sum‑exp)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of surrogate (chain‑rule)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature schedule
        lam = min(lam * lam_factor, lam_cap)

        # occasional jitter if stalled
        if time.time() - last_improve > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1

    return best_seq, best_val


def _subgradient_refinement(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
    max_iter: int = 200_000,
    eta0: float = 0.85,
    decay: float = 0.99993,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient steps on the *exact* max‑convolution.
    """
    a = a0.copy()
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a /= a.sum()

    eta = eta0
    it = 0
    last_improve = time.time()

    while time.time() < deadline and it < max_iter:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Subgrad] iter={it:06d}  best={val:.6f}")

        # active set
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # multiplicative update (keeps positivity)
        a = a * np.exp(-eta * grad_a)

        # renormalise onto the simplex
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        eta *= decay
        it += 1

        # stop if no progress for a while
        if time.time() - last_improve > 7.0:
            break

    return best_seq, best_val


def _deep_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """Adam‑smooth‑max → sub‑gradient → local random perturbations."""
    # Adam phase
    best_seq, best_val = _adam_smooth_max(
        a0,
        n,
        conv_fft_len,
        corr_fft_len,
        deadline,
        global_best_seq,
        global_best_val,
    )
    # Sub‑gradient phase
    best_seq, best_val = _subgradient_refinement(
        a0,
        n,
        conv_fft_len,
        corr_fft_len,
        deadline,
        best_seq,
        best_val,
    )
    return best_seq, best_val


# ----------------------------------------------------------------------
def _beta_candidate(p: float, q: float, n: int) -> np.ndarray:
    """Beta(p,q) density on the discrete grid {0,…,n‑1}."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n  # centre points in (0,1)
    # avoid zero by tiny epsilon
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    shape = (x ** (p - 1)) * ((1 - x) ** (q - 1))
    shape = np.maximum(shape, 0.0)
    if shape.sum() == 0:
        shape = np.full_like(shape, 1.0 / n)
    else:
        shape /= shape.sum()
    return shape


def _generate_parametric_candidates(n: int, warm_start: np.ndarray | None):
    """Diverse family of candidate vectors."""
    candidates = []

    # Warm‑start and its shifts
    if warm_start is not None:
        candidates.append(warm_start.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            candidates.append(np.roll(warm_start, shift))

    # Uniform
    candidates.append(np.full(n, 1.0 / n))

    # Random Dirichlet draws
    for conc in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(9):
            cand = np.random.dirichlet(np.full(n, conc))
            candidates.append(cand)

    # Symmetric edge‑exponential / power‑law (already proved useful)
    i = np.arange(n, dtype=np.float64)

    for alpha in np.logspace(-2, 1.0, num=12):
        edge = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
        edge /= edge.sum()
        for w in (0.0, 0.25, 0.5, 0.75, 1.0):
            shape = w * edge + (1.0 - w) / n
            shape /= shape.sum()
            candidates.append(shape)

    for p in np.linspace(0.2, 6.0, num=12):
        edge = (i + 1.0) ** (-p) + (n - i) ** (-p)
        edge /= edge.sum()
        for w in (0.0, 0.3, 0.6, 1.0):
            shape = w * edge + (1.0 - w) / n
            shape /= shape.sum()
            candidates.append(shape)

    # Asymmetric two‑sided exponentials
    alphas = np.logspace(-2, 1.0, num=10)
    w_vals = np.linspace(0.0, 1.0, 7)
    for wL in w_vals:
        for wR in w_vals:
            if wL + wR > 1.0:
                continue
            wU = 1.0 - wL - wR
            for aL in alphas:
                for aR in alphas:
                    left = np.exp(-aL * i)
                    right = np.exp(-aR * (n - 1 - i))
                    shape = wL * left + wR * right + wU / n
                    shape /= shape.sum()
                    candidates.append(shape)

    # Asymmetric power‑law
    p_vals = np.linspace(0.2, 5.0, num=9)
    for wL in w_vals:
        for wR in w_vals:
            if wL + wR > 1.0:
                continue
            wU = 1.0 - wL - wR
            for pL in p_vals:
                for pR in p_vals:
                    left = (i + 1.0) ** (-pL)
                    right = (n - i) ** (-pR)
                    shape = wL * left + wR * right + wU / n
                    shape /= shape.sum()
                    candidates.append(shape)

    # Linear / triangular shapes
    lin = np.linspace(1.0, 0.0, n)
    tri = np.concatenate([np.linspace(0, 1, n // 2, endpoint=False),
                          np.linspace(1, 0, n - n // 2)])
    for base in (lin, tri):
        base = np.maximum(base, 0.0)
        base /= base.sum()
        for w in (0.0, 0.4, 0.8):
            shape = w * base + (1.0 - w) / n
            shape /= shape.sum()
            candidates.append(shape)

    # Sinusoidal half‑wave
    sin = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    sin = np.maximum(sin, 0.0)
    sin /= sin.sum()
    candidates.append(sin)

    # Beta family (including U‑shaped cases)
    beta_ps = [0.2, 0.3, 0.4, 0.5, 0.7, 1.0, 1.5, 2.0]
    beta_qs = beta_ps
    for p in beta_ps:
        for q in beta_qs:
            shape = _beta_candidate(p, q, n)
            for w in (0.0, 0.3, 0.6, 1.0):
                cand = w * shape + (1.0 - w) / n
                cand /= cand.sum()
                candidates.append(cand)

    # Sparse spike vectors
    for spikes in (3, 5, 8, 12):
        for _ in range(15):
            idx = np.random.choice(n, size=spikes, replace=False)
            weights = np.random.dirichlet(np.ones(spikes))
            vec = np.zeros(n)
            vec[idx] = weights
            vec /= vec.sum()
            candidates.append(vec)

    return candidates


# ----------------------------------------------------------------------
def _local_random_search(
    a_best: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """Cheap stochastic hill‑climbing around the current best."""
    logits_best = np.log(a_best + 1e-15)
    temperature = 0.02  # initial scale of perturbations

    while time.time() < deadline:
        noise = np.random.randn(n) * temperature
        logits = logits_best + noise
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        val = _objective(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            a_best = a.copy()
            logits_best = np.log(a_best + 1e-15)
            print(f"[Local] improved to {val:.6f}")

        # slowly cool
        temperature *= 0.9999
        if temperature < 1e-6:
            temperature = 1e-6

    return a_best, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Main entry point – searches for a non‑negative step function that
    minimises the supplied evaluation function.
    Returns the best sequence found (plain Python list).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # safety margin

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # --------------------------------------------------------------
    # Primary grid size (offers a good trade‑off between resolution and speed)
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Warm‑start from a previously found construction, if present
    warm_start = globals().get("height_sequence_1")
    if warm_start is not None:
        warm_start = np.asarray(warm_start, dtype=np.float64)
        if warm_start.shape[0] > n:
            warm_start = warm_start[:n]
        elif warm_start.shape[0] < n:
            warm_start = np.pad(warm_start, (0, n - warm_start.shape[0]))
        warm_start = np.maximum(warm_start, 0.0)
        if warm_start.sum() >= 0.01:
            warm_start /= warm_start.sum()
        else:
            warm_start = None
    else:
        warm_start = None

    # --------------------------------------------------------------
    # Stage 1 – cheap generation & quick filtering
    all_candidates = _generate_parametric_candidates(n, warm_start)

    cand_vals = []
    for idx, cand in enumerate(all_candidates):
        if time.time() >= deadline:
            break
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 1e-12:
            cand = np.random.dirichlet(np.ones(n))
            s = cand.sum()
        cand /= s
        val = _objective(cand, n, conv_fft_len)
        cand_vals.append((val, cand))

    cand_vals.sort(key=lambda x: x[0])
    top_k = 12
    top_candidates = [c for _, c in cand_vals[:top_k]]
    print(f"[Stage 1] evaluated {len(cand_vals)} candidates, best={cand_vals[0][0]:.6f}")

    # --------------------------------------------------------------
    # Stage 2 – deep optimisation on the elite starts
    for start_id, a0 in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        print(f"--- Deep optimisation start #{start_id + 1}/{len(top_candidates)} ---")
        best_seq, best_val = _deep_optimize(
            a0,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            best_seq,
            best_val,
        )

    # --------------------------------------------------------------
    # Stage 3 – stochastic hill‑climbing from the current best
    if best_seq is not None:
        print("--- Local stochastic refinement (grid n={}) ---".format(n))
        best_seq, best_val = _local_random_search(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            best_val,
        )

    # --------------------------------------------------------------
    # Stage 4 – if we still have time, move to a finer grid (n≈8192)
    time_left = deadline - time.time()
    if time_left > 45.0:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        if best_seq is not None:
            # linear interpolation onto the finer grid
            x_old = np.linspace(0, 1, n, endpoint=False)
            x_big = np.linspace(0, 1, n_big, endpoint=False)
            a_big = np.interp(x_big, x_old, best_seq)
            a_big = np.maximum(a_big, 0.0)
            a_big /= a_big.sum()
        else:
            a_big = np.random.dirichlet(np.ones(n_big))

        print("--- Optimising on larger grid (n={}) ---".format(n_big))
        best_seq, best_val = _deep_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            best_seq,
            best_val,
        )

        # final cheap local search on the big grid
        if best_seq is not None and time.time() < deadline:
            print("--- Final local refinement on big grid ---")
            best_seq, best_val = _local_random_search(
                best_seq,
                n_big,
                conv_fft_len_big,
                deadline,
                best_val,
            )
        # adopt the larger size if it proved beneficial
        if best_seq is not None and best_seq.shape[0] == n_big:
            n = n_big
            conv_fft_len = conv_fft_len_big
            corr_fft_len = corr_fft_len_big

    # --------------------------------------------------------------
    # Fallback – return a uniform vector if nothing was found
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Final clipping to satisfy evaluator’s bounds
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Print final statistics
    final_obj = _objective(best_seq, n, conv_fft_len)
    print(f"*** Search finished: best objective ≈ {final_obj:.6f} (n={n}) ***")
    return best_seq.tolist()
