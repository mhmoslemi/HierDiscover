```python
def propose_candidate(seed=0,budget_s=1000):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5

    # define n maybe 8192 (if time permits)
    n = 8192
    fft_len = _next_pow2(2*n-1)
    corr_fft_len = _next_pow2(3*n-2)

    # Warm start from global var if available and adapt to n
    warm = None
    if 'height_sequence_1' in globals():
        ws = np.asarray(height_sequence_1, dtype=np.float64)
        if ws.shape[0] != n:
            # interpolate to correct length
            x_old = np.linspace(0,1,ws.shape[0],endpoint=False)
            x_new = np.linspace(0,1,n,endpoint=False)
            ws = np.interp(x_new, x_old, ws)
        ws = np.clip(ws,0,1000)
        s = ws.sum()
        if s >= 0.01:
            ws = ws / s
            warm = ws

    # Generate candidates (list of np arrays)
    candidates = []
    # uniform
    candidates.append(np.full(n, 1.0/n))
    # warm start if any
    if warm is not None:
        candidates.append(warm.copy())
        # also some random rotations
        for shift in (n//5, n//3, n//2):
            candidates.append(np.roll(warm, shift))

    # Random Dirichlet with various alphas
    for alpha in [0.1, 0.3, 0.5, 1.0, 2.0, 5.0]:
        cand = np.random.dirichlet(np.full(n, alpha))
        candidates.append(cand)

    # Power-law edges
    idx = np.arange(1,n+1)
    prod = idx * (n - idx + 1)
    for exp in np.linspace(-2.5, -0.2, 9):
        shape = np.where(prod>0, prod**exp, 0.0)
        if shape.sum()>0:
            shape /= shape.sum()
            candidates.append(shape)

    # double exponential
    i = np.arange(n)
    for gamma in np.geomspace(0.1, 8.0, 10):
        shape = np.exp(-gamma*i) + np.exp(-gamma*(n-1-i))
        shape /= shape.sum()
        candidates.append(shape)

    # sinusoid (positive half wave)
    sin = np.sin(np.pi * (np.arange(n)+0.5)/n)
    sin = np.maximum(sin,0)
    sin /= sin.sum()
    candidates.append(sin)

    # spikes: few non-zero entries
    for k in [8,16,32,64,128]:
        a = np.zeros(n)
        pos = np.random.choice(n, k, replace=False)
        a[pos] = 1.0/k
        candidates.append(a)

    # deduplicate / ensure positive sum
    cleaned = []
    for cand in candidates:
        cand = np.maximum(cand,0.0)
        s = cand.sum()
        if s < 0.01:
            continue
        cand = cand / s
        cleaned.append(cand)
    candidates = cleaned

    # evaluate each candidate with short Adam
    best_seq = None
    best_val = float('inf')
    for idx,cand in enumerate(candidates):
        if time.time() >= deadline:
            break
        # allocate time per candidate: e.g., up to 8 sec for Adam
        cand_deadline = min(deadline-3.0, time.time() + 8.0)
        # initial Adam steps ~ 80k
        a_opt, val_opt = _adam_optimize(cand, n, fft_len, corr_fft_len, cand_deadline, max_iters=80_000)
        if val_opt < best_val:
            best_val = val_opt
            best_seq = a_opt
            print(f"[Initial {idx}] new best {val_opt:.6f}")

    # Global refinement on best_seq with longer Adam + subgrad
    if best_seq is not None and time.time() < deadline - 5:
        # longer Adam
        long_deadline = min(deadline-10.0, time.time() + 30.0)
        a_ref, val_ref = _adam_optimize(best_seq, n, fft_len, corr_fft_len, long_deadline, max_iters=200_000)
        if val_ref < best_val:
            best_val = val_ref
            best_seq = a_ref
            print(f"[Long Adam] new best {val_ref:.6f}")

        # subgradient refine
        sub_deadline = deadline - 5.0
        a_sub, val_sub = _subgradient_opt2(best_seq, n, fft_len, corr_fft_len, sub_deadline, max_iters=400_000)
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub
            print(f"[Subgrad final] new best {val_sub:.6f}")

    # Optional: random local search (swap mass) for remaining time.
    if best_seq is not None and time.time() < deadline - 2:
        a_local, val_local = _local_random_search(best_seq, n, fft_len, deadline, max_moves=300_000)
        if val_local < best_val:
            best_val = val_local
            best_seq = a_local
            print(f"[Local search] new best {val_local:.6f}")

    # ensure valid
    if best_seq is None:
        best_seq = np.full(n, 1.0/n)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final value {best_val:.6f} at length {len(best_seq)}")
    return best_seq.tolist()
```

```
def _adam_lowdim(theta0, B, n, conv_fft_len, corr_fft_len, deadline, max_iters=200_000):
    theta = theta0.copy()
    m = np.zeros_like(theta)
    v = np.zeros_like(theta)
    beta1=0.9; beta2=0.999; eps_adam=1e-8
    lr = 0.05
    lam = 0.5
    lam_factor = 1.02
    best_theta = theta.copy()
    best_val = float('inf')
    it = 0
    while time.time() < deadline and it < max_iters:
        logits = B @ theta
        exp_logits = np.exp(logits - np.max(logits))
        a = exp_logits / exp_logits.sum()
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c   # since sum(a)=1
        if val < best_val:
            best_val = val
            best_theta = theta.copy()
        # compute w
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0/w.size)
        else:
            w /= w_sum
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_logits = a * (grad_a - avg)
        grad_theta = B.T @ grad_logits
        # Adam update
        m = beta1 * m + (1-beta1) * grad_theta
        v = beta2 * v + (1-beta2) * (grad_theta**2)
        m_hat = m / (1-beta1**(it+1))
        v_hat = v / (1-beta2**(it+1))
        theta -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        # temperature schedule
        lam = min(lam * lam_factor, 1e6)
        # occasional random noise
        if it % 10000 == 0 and time.time() + 1 > deadline:
            theta += (np.random.rand(theta.size) - 0.5) * 0.01
        it += 1
    # after loop, compute final a from best_theta
    logits = B @ best_theta
    exp_logits = np.exp(logits - np.max(logits))
    a_best = exp_logits / exp_logits.sum()
    return a_best, best_val
```

"""
Optimisation of a non‑negative step function that minimizes
    F(a) = 2·n·max(a∗a) / (sum(a)^2).

The search proceeds in three increasing resolutions (4096 → 8192 → 16384). 
For each resolution we

* generate a small diverse pool of start vectors (uniform, Dirichlet,
  edge‑concentrated power‑laws, double‑exponential and a few “spike’’
  candidates);
* run a soft‑max (log‑sum‑exp) Adam optimiser that works on the logits
  of the simplex (soft‑max guarantees positivity and sum‑to‑one);
* continue with a multiplicative sub‑gradient phase (exponentiated
  gradient) that directly minimises the true max‑convolution; 
* finish the resolution with a cheap hill‑climbing random move phase.

The whole routine respects a 1000 s budget; it keeps the best sequence
found at any time and returns it as a plain Python list.
"""

import time
import numpy as np

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()

def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]

def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a length‑len(a) vector:
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i = 0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]

def _adam_optimize(a0: np.ndarray,
                   n: int,
                   conv_fft_len: int,
                   corr_fft_len: int,
                   deadline: float,
                   max_iters: int = 200_000,
                   lr: float = 0.07,
                   lam_start: float = 0.5,
                   lam_factor: float = 1.03) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth‑max (log‑sum‑exp) surrogate.
    Returns the best vector (sum≈1) encountered and its exact objective.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lam = lam_start

    best_a = a0.copy()
    best_val = float('inf')
    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max → simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective (sum(a)=1)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            # occasional debug output
            if it % 10_000 == 0:
                print(f"[Adam iter={it:06d}] new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate (chain‑rule through soft‑max)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature schedule → true max
        lam = min(lam * lam_factor, 1e6)

        # occasional tiny random shake if we seem to stall
        if it % 7_000 == 0 and time.time() + 5 > deadline:
            logits += (np.random.rand(n) - 0.5) * 5e-3

        it += 1

    return best_a, best_val

def _subgradient_opt2(a0: np.ndarray,
                      n: int,
                      conv_fft_len: int,
                      corr_fft_len: int,
                      deadline: float,
                      max_iters: int = 500_000,
                      eta0: float = 0.7) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient phase with a decaying step size.
    Returns the best vector seen and its exact objective.
    """
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s

    best_a = a.copy()
    best_val = float('inf')
    stall = 0
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            stall = 0
            if it % 5_000 == 0:
                print(f"[Subgrad iter={it:07d}] new best {val:.6f}")
        else:
            stall += 1

        # active‑set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # decaying step size
        eta = eta0 / np.sqrt(it + 1)

        # multiplicative update and renormalisation
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        # random perturbation if we have been stuck for a while
        if stall > 10_000:
            a = np.maximum(a + 0.02 * (np.random.rand(n) - 0.5), 0.0)
            a /= a.sum()
            stall = 0

        it += 1
    return best_a, best_val

def _local_random_search(a0: np.ndarray,
                         n: int,
                         conv_fft_len: int,
                         deadline: float,
                         max_moves: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climbing: move a random fraction of mass from one index
    to another, keep the move if the objective improves.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    best_a = a.copy()
    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-12:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue
        # move up to 35 % of the source mass
        delta = a[i] * (np.random.rand() * 0.35)
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta
        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()
        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5_000 == 0:
                print(f"[Local move {it}] improved to {val2:.6f}")
        it += 1
    return best_a, best_val

def _generate_initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """
    Produce a modest but diverse pool of start vectors.
    """
    cand = []

    # warm start and a few rotations
    if warm is not None:
        cand.append(warm.copy())
        for shift in (n // 5, n // 3, n // 2):
            cand.append(np.roll(warm, shift))

    # random Dirichlet with several concentrations
    for alpha in [0.3, 0.7, 1.0, 2.0, 5.0]:
        cand.append(np.random.dirichlet(np.full(n, alpha)))

    # edge‑concentrated power‑law shapes
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.5, -0.2, 11):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    # symmetric double‑exponential shapes
    i = np.arange(n, dtype=np.float64)
    for gamma in np.geomspace(0.15, 6.0, 9):
        shape = np.exp(-gamma * i) + np.exp(-gamma * (n - 1 - i))
        shape /= shape.sum()
        cand.append(shape)

    # positive half‑wave sinusoid
    s = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    s = np.maximum(s, 0.0)
    s /= s.sum()
    cand.append(s)

    # uniform
    cand.append(np.full(n, 1.0 / n))

    # sparse “spike’’ candidates
    for k in (8, 16, 32, 64, 128):
        a = np.zeros(n)
        pos = np.random.choice(n, k, replace=False)
        a[pos] = 1.0 / k
        cand.append(a)

    # ensure all vectors are proper probability vectors
    cleaned = []
    for c in cand:
        c = np.maximum(c, 0.0)
        s = c.sum()
        if s < 0.01:
            continue
        c = c / s
        cleaned.append(c)
    return cleaned

def _upscale_vector(v: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a vector to a new length."""
    x_old = np.linspace(0, 1, v.shape[0], endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    v_big = np.interp(x_new, x_old, v)
    v_big = np.maximum(v_big, 0.0)
    v_big /= v_big.sum()
    return v_big

"""
Optimisation of a non‑negative step function that minimises

    F(a) = 2·n·max(a∗a) / (sum(a)^2)

The routine works in three increasingly fine resolutions
(4096 → 8192 → 16384).  Each resolution performs

* generation of a modest but diverse pool of start vectors
  (uniform, Dirichlet, edge‑concentrated power‑laws, double‑exponential,
  half‑wave sinusoid, two‑block spikes, a triangular ramp and a few
  rotations of a warm start);
* a short Adam optimisation on the smooth (log‑sum‑exp) surrogate of the
  max‑convolution (the parameters are kept on the probability simplex
  via a soft‑max);
* a long multiplicative sub‑gradient (exponentiated‑gradient) phase that
  directly minimises the true max‑convolution;
* a cheap hill‑climbing random‑move phase.

When time permits a final polishing Adam phase with a very high
temperature is applied.  All phases respect the global 1000 s budget;
the function always returns the best sequence it has found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a length‑len(a) vector:
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i = 0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
    lr: float = 0.07,
    lam_start: float = 0.5,
    lam_factor: float = 1.03,
) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth‑max (log‑sum‑exp) surrogate.
    Returns the best vector (approximately on the simplex) seen and its exact objective.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lam = lam_start

    best_a = a0.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        # ---- simplex via soft‑max ----
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective (sum(a)=1)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 10_000 == 0:
                print(f"[Adam iter={it:06d}] new best {val:.6f}")

        # ---- smooth max weights (log‑sum‑exp) ----
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---- gradient of the surrogate ----
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ---- Adam update on logits ----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature schedule → true max
        lam = min(lam * lam_factor, 1e6)

        # occasional tiny random shake when we are close to the deadline
        if it % 7_000 == 0 and time.time() + 5 > deadline:
            logits += (np.random.rand(n) - 0.5) * 5e-3

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
def _subgradient_opt2(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
    eta0: float = 0.7,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient phase with a decaying step size.
    Returns the best vector seen and its exact objective.
    """
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s

    best_a = a.copy()
    best_val = float("inf")
    stall = 0
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            stall = 0
            if it % 5_000 == 0:
                print(f"[Subgrad iter={it:07d}] new best {val:.6f}")
        else:
            stall += 1

        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # decaying step size
        eta = eta0 / np.sqrt(it + 1)

        # multiplicative update + renormalisation
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        # random perturbation after a long stall
        if stall > 10_000:
            a = np.maximum(a + 0.02 * (np.random.rand(n) - 0.5), 0.0)
            a /= a.sum()
            stall = 0

        it += 1
    return best_a, best_val


# ----------------------------------------------------------------------
def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climbing: move a random fraction of mass from one index
    to another, keep the move if the objective improves.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    best_a = a.copy()
    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-12:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue
        delta = a[i] * (np.random.rand() * 0.35)  # up to 35 % of source mass
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta
        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()
        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5_000 == 0:
                print(f"[Local move {it}] improved to {val2:.6f}")
        it += 1
    return best_a, best_val


# ----------------------------------------------------------------------
def _upscale_vector(v: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a vector to a new length."""
    x_old = np.linspace(0, 1, v.shape[0], endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    v_big = np.interp(x_new, x_old, v)
    v_big = np.maximum(v_big, 0.0)
    v_big /= v_big.sum()
    return v_big


def _generate_initial_candidates(
    n: int,
    warm: np.ndarray | None,
) -> list[np.ndarray]:
    """Produce a modest, diverse pool of start vectors."""
    cand = []

    # warm start + a few rotations
    if warm is not None:
        cand.append(warm.copy())
        for shift in (n // 5, n // 3, n // 2):
            cand.append(np.roll(warm, shift))

    # Dirichlet draws with several concentrations
    for alpha in [0.3, 0.7, 1.0, 2.0, 5.0]:
        cand.append(np.random.dirichlet(np.full(n, alpha)))

    # Edge‑concentrated power‑law shapes
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.5, -0.2, 11):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    # Symmetric double‑exponential shapes
    i = np.arange(n, dtype=np.float64)
    for gamma in np.geomspace(0.15, 6.0, 9):
        shape = np.exp(-gamma * i) + np.exp(-gamma * (n - 1 - i))
        shape /= shape.sum()
        cand.append(shape)

    # Positive half‑wave sinusoid
    s = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    s = np.maximum(s, 0.0)
    s /= s.sum()
    cand.append(s)

    # Uniform
    cand.append(np.full(n, 1.0 / n))

    # Two‑block “spike’’ patterns (mass at both ends)
    for fraction in [0.05, 0.10, 0.20]:
        L = max(1, int(fraction * n))
        a = np.zeros(n)
        a[:L] = 0.5 / L
        a[-L:] = 0.5 / L
        cand.append(a)

    # Triangular ramp (increase then decrease)
    tri = np.arange(1, n // 2 + 1, dtype=np.float64)
    if n % 2 == 0:
        tri = np.concatenate([tri, tri[::-1]])
    else:
        tri = np.concatenate([tri, tri[-2::-1]])
    tri = np.maximum(tri, 0.0)
    tri /= tri.sum()
    cand.append(tri)

    # Sparse “spike’’ candidates with a few equally spaced non‑zeros
    for k in (8, 16, 32, 64, 128):
        a = np.zeros(n)
        pos = np.random.choice(n, k, replace=False)
        a[pos] = 1.0 / k
        cand.append(a)

    # final clean‑up – ensure proper simplex vectors
    cleaned = []
    for c in cand:
        c = np.maximum(c, 0.0)
        s = c.sum()
        if s < 0.01:
            continue
        c = c / s
        cleaned.append(c)
    return cleaned


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine runs a multi‑resolution optimisation pipeline and always
    returns the best sequence it has found before the time budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # Warm start: may be present from a previous run
    warm = None
    if "height_sequence_1" in globals():
        ws = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if ws.size > 0:
            warm = ws.copy()

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Resolutions we will visit (low → high).  Time is split roughly
    # equally among them, but the loop always respects the global deadline.
    for n in (4096, 8192, 16384):
        conv_fft_len = _next_pow2(2 * n - 1)
        corr_fft_len = _next_pow2(3 * n - 2)

        # adapt warm start to the current grid size
        if warm is not None:
            warm_n = _upscale_vector(warm, n) if warm.shape[0] != n else warm.copy()
        else:
            warm_n = None

        # ------------------------------------------------------------------
        # (1) generate and briefly optimise a handful of candidates
        candidates = _generate_initial_candidates(n, warm_n)
        for cid, cand in enumerate(candidates):
            if time.time() >= deadline:
                break
            cand_deadline = min(deadline - 5.0, time.time() + 8.0)
            a_adam, val_adam = _adam_optimize(
                cand,
                n,
                conv_fft_len,
                corr_fft_len,
                cand_deadline,
                max_iters=120_000,
                lr=0.07,
                lam_start=0.5,
                lam_factor=1.03,
            )
            if val_adam < best_val:
                best_val = val_adam
                best_seq = a_adam.copy()
                print(f"[Res {n} cand {cid}] Adam best {val_adam:.6f}")

        # ------------------------------------------------------------------
        # (2) long sub‑gradient phase on the best vector so far
        if best_seq is not None and time.time() < deadline - 5:
            sub_deadline = deadline - 5.0
            a_sub, val_sub = _subgradient_opt2(
                best_seq,
                n,
                conv_fft_len,
                corr_fft_len,
                sub_deadline,
                max_iters=400_000,
                eta0=0.7,
            )
            if val_sub < best_val:
                best_val = val_sub
                best_seq = a_sub.copy()
                print(f"[Res {n} Subgrad] new best {val_sub:.6f}")

        # ------------------------------------------------------------------
        # (3) cheap random local search
        if best_seq is not None and time.time() < deadline - 3:
            loc_deadline = deadline - 3.0
            a_loc, val_loc = _local_random_search(
                best_seq,
                n,
                conv_fft_len,
                loc_deadline,
                max_moves=150_000,
            )
            if val_loc < best_val:
                best_val = val_loc
                best_seq = a_loc.copy()
                print(f"[Res {n} Local] improved best {val_loc:.6f}")

        # ------------------------------------------------------------------
        # prepare warm start for the next, finer grid
        if best_seq is not None:
            warm = best_seq.copy()
        else:
            warm = None

        # if we are already out of time, abort the resolution loop
        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # Final polishing with a high‑temperature Adam (if any time is left)
    if best_seq is not None and time.time() < deadline - 2:
        n = best_seq.shape[0]
        conv_fft_len = _next_pow2(2 * n - 1)
        corr_fft_len = _next_pow2(3 * n - 2)
        final_deadline = deadline - 0.5
        a_fin, val_fin = _adam_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            final_deadline,
            max_iters=200_000,
            lr=0.04,
            lam_start=5.0,      # already a fairly sharp surrogate
            lam_factor=1.0,     # keep temperature fixed
        )
        if val_fin < best_val:
            best_val = val_fin
            best_seq = a_fin.copy()
            print(f"[Final polishing] new best {val_fin:.6f}")

    # ------------------------------------------------------------------
    # Fallback – guarantee a valid return value
    if best_seq is None:
        n = 4096
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    # store for a possible warm start in a subsequent run
    globals()["height_sequence_1"] = best_seq.copy()
    return best_seq.tolist()
