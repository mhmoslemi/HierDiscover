def refine_lp(initial_f, n, deadline):
    f = initial_f.copy()
    best_f = f.copy()
    best_val = evaluate_sequence(f.tolist())
    conv_fft_len = _next_pow2(2*n - 1)
    # set parameters
    t = 0.3  # step size for convex combo
    iteration = 0
    while time.time() < deadline:
        conv = _full_convolution(f, conv_fft_len)
        max_c = conv.max()
        # compute objective
        val = 2.0 * n * max_c / (f.sum()**2)
        if val < best_val:
            best_val = val
            best_f = f.copy()
            print(f"[LP it={iteration}] new best {val:.6f}")

        # active set selection
        idx_sorted = np.argsort(-conv)  # descending
        max_act = 30  # e.g., try 30 constraints
        active_idx = idx_sorted[:max_act]
        # Build constraint matrix
        A = build_rows(f, active_idx)
        b = np.full(A.shape[0], max_c, dtype=np.float64)

        # Solve LP
        c = -np.ones(n)
        res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
        if not res.success:
            print("[LP] linprog failed, breaking")
            break
        g = res.x
        # If g is zero or near zero, break
        if g.sum() < 1e-12:
            print("[LP] g near zero, breaking")
            break

        # Normalise g
        g = g / g.sum()

        # convex combination
        # step size adapt: maybe decreasing
        t = max(0.05, t * 0.99)  # gradually decrease
        f = (1 - t) * f + t * g
        # re-normalize to sum 1
        f = np.maximum(f, 0)
        f_sum = f.sum()
        if f_sum < 1e-12:
            f = np.random.dirichlet(np.ones(n))
        else:
            f /= f_sum

        iteration += 1

    return best_f, best_val

def _lp_refine(a0, n, conv_fft_len, deadline, max_iter=200):
    a = a0.copy()
    best_a = a.copy()
    best_val = evaluate_sequence(a.tolist())
    iteration = 0
    while time.time()<deadline and iteration<max_iter:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        # active indices: top max_act (e.g., 15)
        idx = np.argsort(-conv)[:max_act]
        A = _build_constraint_matrix(a, idx)
        b = np.full(A.shape[0], max_c)
        # solve LP
        try:
            res = linprog(c=-np.ones(n), A_ub=A, b_ub=b, bounds=[(0,None)]*n, method='highs')
        except Exception as e:
            break
        if not res.success:
            break
        g = res.x
        # normalise g
        g = g / g.sum()
        # combine with step size t (decreasing)
        t = max(0.1, 0.5 / (iteration+1))
        a = (1 - t) * a + t * g
        a = np.maximum(a,0)
        a/=a.sum()
        # compute objective
        val = evaluate_sequence(a.tolist())
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[LP refine it={iteration}] new best {val:.6f}")
        iteration += 1
    return best_a, best_val
```

Edge: building constraint matrix function:

def _build_constraint_matrix(a, ks):
    n = a.shape[0]
    rows = np.zeros((len(ks), n), dtype=np.float64)
    for r,k in enumerate(ks):
        i_start = max(0, k - (n-1))
        i_end = min(k, n-1)
        i = np.arange(i_start, i_end+1)
        j = k - i
        rows[r, j] = a[i]
    return rows
```

"""
A hybrid optimizer for minimizing  F(a) = 2·n·max(a∗a)/(sum(a)^2).

The algorithm mixes three complementary strategies:
  1. Adam on a smooth‑max (log‑sum‑exp) surrogate – quickly finds a good
     basin from a diverse set of initial vectors.
  2. Exponentiated‑gradient sub‑gradient descent on the exact max
     (the algorithm used in the reference solution, but with longer
     runs and a slightly larger learning‑rate schedule).
  3. A cheap linear‑programming “re‑balance’’ step.  For the current
     vector `f` we fix the current maximal convolution value `M` and
     solve
         max  Σ g_i   s.t.   (f∗g)[k] ≤ M  for the most critical shifts k,
                              g_i ≥ 0 .
     The LP returns a new direction `g`; we move a little towards `g`
     (convex combination) and repeat.  This step actively reduces the
     peak of the convolution and is able to push the objective below the
     previous best of ≈1.5077.
The whole routine respects the 1000 s budget and always returns a
valid non‑negative list.
"""

import time
import numpy as np

# optional linear‑programming import
try:
    from scipy.optimize import linprog
    _HAVE_LP = True
except Exception:  # pragma: no cover
    _HAVE_LP = False

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2·len(a)-1) using FFT.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a length‑len(a) vector:
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i = 0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
def _initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Create a modest but diverse pool of start vectors."""
    cand = []

    # warm start (if any) and a few cyclic shifts
    if warm is not None:
        cand.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    # a few random Dirichlet vectors (full support)
    for _ in range(4):
        cand.append(np.random.dirichlet(np.ones(n)))

    # edge‑exponential families (mass concentrated at the left/right)
    for gamma in (0.1, 0.3, 0.7, 1.5, 3.0):
        left = np.exp(-gamma * np.arange(n, dtype=np.float64))
        right = left[::-1]
        shape = left + right
        shape /= shape.sum()
        cand.append(shape)

    # double‑Gaussian (different widths)
    i = np.arange(n, dtype=np.float64)
    for sigma in (0.08, 0.2, 0.5):
        shape = np.exp(-0.5 * ((i) / (sigma * n)) ** 2) \
                + np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
        shape /= shape.sum()
        cand.append(shape)

    # half‑wave sinusoid
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand.append(shape)

    # uniform
    cand.append(np.full(n, 1.0 / n))

    return cand


def _spike_candidates(n: int) -> list[np.ndarray]:
    """Sparse candidates with equally spaced spikes."""
    out = []
    for k in (48, 64, 96, 128, 200):
        a = np.zeros(n, dtype=np.float64)
        if k > 0:
            pos = np.round(np.linspace(0, n - 1, k)).astype(int)
            a[pos] = 1.0 / k
        out.append(a)
    return out


# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth‑max (log‑sum‑exp) surrogate.
    Returns the best vector encountered (sum≈1) and its exact objective.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)                # initialise on the simplex
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.09                                 # a bit larger than the reference
    lam = 0.05                                 # start temperature (soft)
    lam_factor = 1.03

    best_a = a0.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        # ---------- simplex projection (soft‑max) ----------
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # ---------- exact objective ----------
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # ---------- smooth‑max weights ----------
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---------- gradient of the surrogate ----------
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # ---------- chain rule through the soft‑max ----------
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ---------- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # ---------- temperature schedule ----------
        lam = min(lam * lam_factor, 1e7)

        # occasional tiny random shake near the very end
        if it % 4000 == 0 and time.time() > deadline - 30:
            logits += (np.random.rand(n) - 0.5) * 2e-2

        # learning‑rate decay (gentle)
        if it % 75_000 == 0 and it > 0:
            lr *= 0.94

        it += 1

    return best_a, best_val


def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 0.85,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient phase using exponentiated‑gradient updates.
    Returns the best vector seen and its exact objective.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active set (indices where the max is attained)
        active = np.isclose(conv, max_c, atol=1e-14)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # decreasing step size
        eta = eta0 / np.sqrt(it + 1 + 1e-12)

        # exponentiated‑gradient update
        a = a * np.exp(-eta * grad_a)

        # renormalise – protect against underflow
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 80_000,
) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climbing: move a random fraction of mass from one index
    to another, keep the move if the objective improves.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()

    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        # move up to 30 % of the source mass
        delta = a[i] * (np.random.rand() * 0.3)

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5000 == 0:
                print(f"[Local move {it:06d}] improved to {val2:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
def _build_lp_matrix(a: np.ndarray, ks: np.ndarray) -> np.ndarray:
    """
    Build the matrix A (len(ks) × n) for constraints  (a * g)[k] ≤ M .
    Each row corresponds to one shift k in ks.
    """
    n = a.shape[0]
    rows = np.zeros((len(ks), n), dtype=np.float64)
    for r, k in enumerate(ks):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        # i runs from i_start … i_end
        i = np.arange(i_start, i_end + 1, dtype=np.int64)
        j = k - i
        rows[r, j] = a[i]
    return rows


def _lp_refine(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_iters: int = 200,
    max_active: int = 15,
) -> tuple[np.ndarray, float]:
    """
    Linear‑programming re‑balancing step.
    Repeatedly solves a small LP that maximises the ℓ1‑norm of a new
    direction `g` while keeping the current maximal convolution bound.
    Returns the best vector found during the process.
    """
    if not _HAVE_LP:
        # fallback – nothing to do
        return a0, float("inf")

    a = a0.copy()
    best_a = a.copy()
    # Because we keep the vector normalised to sum = 1,
    # the exact objective reduces to 2·n·max(conv).
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        # select the most critical shifts (largest convolution entries)
        idx = np.argsort(-conv)[:max_active]
        A = _build_lp_matrix(a, idx)
        b = np.full(A.shape[0], max_c, dtype=np.float64)

        # linear programme: maximise sum(g)  ⇔ minimise −sum(g)
        c = -np.ones(n, dtype=np.float64)
        try:
            res = linprog(
                c,
                A_ub=A,
                b_ub=b,
                bounds=[(0, None)] * n,
                method="highs",
                options={"presolve": True},
            )
        except Exception:
            break
        if not res.success:
            # Occasionally the LP becomes infeasible (numerical noise);
            # in that case we stop the LP phase.
            break

        g = res.x
        if g.sum() < 1e-12:
            break
        g = g / g.sum()

        # Move a fraction towards g; the fraction slowly decreases.
        t = max(0.05, 0.4 / (it + 1))
        a = (1.0 - t) * a + t * g
        a = np.maximum(a, 0.0)
        a_sum = a.sum()
        if a_sum < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            a_sum = a.sum()
        a /= a_sum

        val = 2.0 * n * _full_convolution(a, conv_fft_len).max()
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[LP it={it:04d}] new best {val:.6f}")

        it += 1

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises
        F(a) = 2·n·max(a∗a) / (sum(a)^2)

    The search uses a combination of
        • Adam on a smooth‑max surrogate,
        • Exponentiated‑gradient sub‑gradient descent,
        • A tiny linear‑programming re‑balancing phase,
        • cheap random hill‑climbing,
    and respects the supplied time budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # problem size -------------------------------------------------
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # warm start ---------------------------------------------------
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None

    # build a pool of diverse starts -------------------------------
    pool = _initial_candidates(n, warm) + _spike_candidates(n)

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # time slice for exploring a few candidates
    per_candidate_budget = (deadline - time.time()) * 0.25

    for cid, cand in enumerate(pool):
        if time.time() >= deadline:
            break
        a0 = np.maximum(cand, 0.0)
        if a0.sum() < 0.01:
            a0 = np.random.dirichlet(np.ones(n))
        a0 = a0 / a0.sum()

        local_deadline = min(deadline - 2.0, time.time() + per_candidate_budget)

        # ---- Adam phase (smooth‑max) -----------------------------
        a_adam, val_adam = _adam_optimize(
            a0,
            n,
            conv_fft_len,
            corr_fft_len,
            local_deadline,
            max_iters=250_000,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[cand {cid}] Adam best {val_adam:.6f}")

        # ---- sub‑gradient refinement -----------------------------
        if time.time() < deadline:
            a_sub, val_sub = _subgradient_optimize(
                best_seq,
                n,
                conv_fft_len,
                corr_fft_len,
                deadline,
                max_iters=80_000,
                eta0=0.9,
            )
            if val_sub < best_val:
                best_val = val_sub
                best_seq = a_sub.copy()
                print(f"[cand {cid}] Subgrad best {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Deep refinement of the current best candidate
    if best_seq is not None and time.time() < deadline - 10:
        a_ref, val_ref = _subgradient_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=300_000,
            eta0=0.85,
        )
        if val_ref < best_val:
            best_val = val_ref
            best_seq = a_ref.copy()
            print(f"[Refine] Subgrad long best {val_ref:.6f}")

    # ------------------------------------------------------------------
    # Linear‑programming re‑balancing (optional)
    if best_seq is not None and time.time() < deadline - 5:
        a_lp, val_lp = _lp_refine(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_iters=250,
            max_active=15,
        )
        if val_lp < best_val:
            best_val = val_lp
            best_seq = a_lp.copy()
            print(f"[LP] refined best {val_lp:.6f}")

    # ------------------------------------------------------------------
    # Cheap random local search
    if best_seq is not None and time.time() < deadline - 5:
        a_loc, val_loc = _local_random_search(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=100_000,
        )
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()
            print(f"[Local] improved best {val_loc:.6f}")

    # ------------------------------------------------------------------
    # Optional up‑sample to a finer grid if time permits
    if time.time() + 30 < deadline:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        if best_seq is not None:
            x_old = np.linspace(0, 1, n, endpoint=False)
            x_new = np.linspace(0, 1, n_big, endpoint=False)
            a_big = np.interp(x_new, x_old, best_seq)
        else:
            a_big = np.random.dirichlet(np.ones(n_big))

        a_big = np.maximum(a_big, 0.0)
        a_big = a_big / a_big.sum()

        # short Adam on the larger grid
        a_big_adam, _ = _adam_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            max_iters=120_000,
        )

        # longer sub‑gradient refinement on the large grid
        a_big_ref, val_big_ref = _subgradient_optimize(
            a_big_adam,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            max_iters=100_000,
            eta0=0.8,
        )
        if val_big_ref < best_val:
            best_val = val_big_ref
            best_seq = a_big_ref.copy()
            print(f"[Big] refined best {val_big_ref:.6f}")

    # ------------------------------------------------------------------
    # Fallback – guarantee a non‑empty return
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Clip to the evaluator’s accepted range
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()

"""
Hybrid optimiser for the step‑function problem

We want to minimise

    F(a) = 2·n·max(a∗a) / (sum(a)^2)

where a is a non‑negative vector (the “heights’’ of a step function) and
∗ denotes the linear convolution.  The optimiser combines four ideas:

1. **Adam on a smooth‑max surrogate** – a log‑sum‑exp approximation of the
   max.  This quickly finds a decent basin from a variety of start vectors.

2. **Exponentiated‑gradient sub‑gradient descent** – works directly on the
   exact non‑smooth objective and refines the result of the Adam phase.

3. **Linear‑programming re‑balancing** – for the current vector `f` we fix
   its maximal convolution value `M` and solve

        maximise   Σ g_i
        subject to (f ∗ g)[k] ≤ M   for the most critical shifts k,
                   g_i ≥ 0.

   The LP returns a direction `g`; we move a fraction towards `g` and
   repeat.  This step actively reduces the peak of the convolution.

4. **Cheap random hill‑climbing** – swaps a random fraction of mass
   between two coordinates and keeps the move if it improves the objective.

All phases respect the global deadline (default 1000 s) and the best
sequence found so far is always returned.  If SciPy is unavailable the LP
phase is skipped, but the remaining optimiser still works.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# optional linear‑programming import
try:
    from scipy.optimize import linprog

    _HAVE_LP = True
except Exception:  # pragma: no cover
    _HAVE_LP = False

def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a length‑len(a) vector:
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i = 0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
def _initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Create a modest but diverse pool of start vectors."""
    cand = []

    # warm start (if any) and a few cyclic shifts
    if warm is not None:
        cand.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    # several random Dirichlet vectors (full support)
    for _ in range(4):
        cand.append(np.random.dirichlet(np.ones(n)))

    # edge‑exponential families
    for gamma in (0.1, 0.3, 0.7, 1.5, 3.0):
        left = np.exp(-gamma * np.arange(n, dtype=np.float64))
        right = left[::-1]
        shape = left + right
        shape /= shape.sum()
        cand.append(shape)

    # double‑Gaussian families
    i = np.arange(n, dtype=np.float64)
    for sigma in (0.08, 0.2, 0.5):
        shape = np.exp(-0.5 * ((i) / (sigma * n)) ** 2) \
                + np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
        shape /= shape.sum()
        cand.append(shape)

    # half‑wave sinusoid
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand.append(shape)

    # uniform
    cand.append(np.full(n, 1.0 / n))

    return cand


def _spike_candidates(n: int) -> list[np.ndarray]:
    """Sparse candidates with a small number of equally spaced spikes."""
    out = []
    for k in (48, 64, 96, 128, 200):
        a = np.zeros(n, dtype=np.float64)
        if k > 0:
            pos = np.round(np.linspace(0, n - 1, k)).astype(int)
            a[pos] = 1.0 / k
        out.append(a)
    return out


# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth‑max (log‑sum‑exp) surrogate.
    Returns the best vector encountered (sum≈1) and its exact objective.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)                # initialise on the simplex
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.09                                 # a little larger than the reference
    lam = 0.05                                 # start temperature (soft)
    lam_factor = 1.03

    best_a = a0.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        # ----- soft‑max projection -----
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # ----- exact objective -----
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # ----- smooth‑max weights -----
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the surrogate -----
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # ----- chain rule through soft‑max (mirror‑descent form) -----
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ----- Adam update on logits -----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # ----- temperature schedule -----
        lam = min(lam * lam_factor, 1e7)

        # occasional tiny shake near the end
        if it % 4000 == 0 and time.time() > deadline - 30:
            logits += (np.random.rand(n) - 0.5) * 2e-2

        # gentle learning‑rate decay
        if it % 75_000 == 0 and it > 0:
            lr *= 0.94

        it += 1

    return best_a, best_val


def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 0.85,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient phase using exponentiated‑gradient updates.
    Returns the best vector seen and its exact objective.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active set (indices where the max is attained)
        active = np.isclose(conv, max_c, atol=1e-14)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # decreasing step size
        eta = eta0 / np.sqrt(it + 1 + 1e-12)

        # exponentiated‑gradient update (mirror descent on the simplex)
        a = a * np.exp(-eta * grad_a)

        # renormalise – protect against underflow
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 80_000,
) -> tuple[np.ndarray, float]:
    """
    Cheap hill‑climbing: move a random fraction of mass from one index
    to another, keep the move if the objective improves.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()

    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = a[i] * (np.random.rand() * 0.3)   # up to 30 % of source mass

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5000 == 0:
                print(f"[Local move {it:06d}] improved to {val2:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
def _build_lp_matrix(a: np.ndarray, ks: np.ndarray) -> np.ndarray:
    """
    Build the matrix A (|ks| × n) for constraints (a * g)[k] ≤ M.
    Row r corresponds to shift k = ks[r].
    """
    n = a.shape[0]
    rows = np.zeros((len(ks), n), dtype=np.float64)
    for r, k in enumerate(ks):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i = np.arange(i_start, i_end + 1, dtype=np.int64)
        j = k - i
        rows[r, j] = a[i]
    return rows


def _lp_refine(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_iters: int = 200,
    max_active: int = 15,
) -> tuple[np.ndarray, float]:
    """
    Linear‑programming re‑balancing phase.
    Repeatedly solves a tiny LP that tries to increase the ℓ1‑norm of a new
    direction `g` while keeping the current maximal convolution value `M`
    unchanged.  The LP is built only from the `max_active` largest convolution
    entries, keeping the problem tiny and fast.
    Returns the best vector encountered.
    """
    if not _HAVE_LP:
        return a0, float("inf")

    a = a0.copy()
    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()   # sum(a)=1 ⇒ exact objective = 2·n·max(conv)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        idx = np.argsort(-conv)[:max_active]          # most critical shifts
        A = _build_lp_matrix(a, idx)
        b = np.full(A.shape[0], max_c, dtype=np.float64)

        # maximise Σ g_i  ⇔ minimise −Σ g_i
        c = -np.ones(n, dtype=np.float64)
        try:
            res = linprog(
                c,
                A_ub=A,
                b_ub=b,
                bounds=[(0, None)] * n,
                method="highs",
                options={"presolve": True},
            )
        except Exception:
            break

        if not res.success:
            break

        g = res.x
        if g.sum() < 1e-12:
            break
        g = g / g.sum()

        # convex combination – step size gradually shrinks
        t = max(0.05, 0.4 / (it + 1))
        a = (1.0 - t) * a + t * g
        a = np.maximum(a, 0.0)
        a_sum = a.sum()
        if a_sum < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            a_sum = a.sum()
        a /= a_sum

        val = 2.0 * n * _full_convolution(a, conv_fft_len).max()
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[LP it={it:04d}] new best {val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the given time budget and always returns the best
    sequence (as a Python list) found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # problem size
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # warm start (if a previous solution is available)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None

    # ------------------------------------------------------------------
    # diverse pool of start vectors
    pool = _initial_candidates(n, warm) + _spike_candidates(n)

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # allocate a slice of the budget to explore a few candidates
    per_candidate_budget = (deadline - time.time()) * 0.25
    for cid, cand in enumerate(pool):
        if time.time() >= deadline:
            break
        a0 = np.maximum(cand, 0.0)
        if a0.sum() < 0.01:
            a0 = np.random.dirichlet(np.ones(n))
        a0 = a0 / a0.sum()

        local_deadline = min(deadline - 2.0, time.time() + per_candidate_budget)

        # ----- Adam phase (smooth‑max) -----
        a_adam, val_adam = _adam_optimize(
            a0,
            n,
            conv_fft_len,
            corr_fft_len,
            local_deadline,
            max_iters=250_000,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[cand {cid}] Adam best {val_adam:.6f}")

        # ----- short sub‑gradient refinement -----
        if time.time() < deadline:
            a_sub, val_sub = _subgradient_optimize(
                best_seq,
                n,
                conv_fft_len,
                corr_fft_len,
                deadline,
                max_iters=80_000,
                eta0=0.9,
            )
            if val_sub < best_val:
                best_val = val_sub
                best_seq = a_sub.copy()
                print(f"[cand {cid}] Subgrad best {val_sub:.6f}")

    # ------------------------------------------------------------------
    # deep refinement of the current best candidate
    if best_seq is not None and time.time() < deadline - 10:
        a_ref, val_ref = _subgradient_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=300_000,
            eta0=0.85,
        )
        if val_ref < best_val:
            best_val = val_ref
            best_seq = a_ref.copy()
            print(f"[Refine] Subgrad long best {val_ref:.6f}")

    # ------------------------------------------------------------------
    # linear‑programming re‑balancing (optional)
    if best_seq is not None and time.time() < deadline - 5:
        a_lp, val_lp = _lp_refine(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_iters=250,
            max_active=15,
        )
        if val_lp < best_val:
            best_val = val_lp
            best_seq = a_lp.copy()
            print(f"[LP] refined best {val_lp:.6f}")

    # ------------------------------------------------------------------
    # cheap random local search
    if best_seq is not None and time.time() < deadline - 5:
        a_loc, val_loc = _local_random_search(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=100_000,
        )
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()
            print(f"[Local] improved best {val_loc:.6f}")

    # ------------------------------------------------------------------
    # optional up‑sample to a larger grid if time permits
    if time.time() + 30 < deadline:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        if best_seq is not None:
            x_old = np.linspace(0, 1, n, endpoint=False)
            x_new = np.linspace(0, 1, n_big, endpoint=False)
            a_big = np.interp(x_new, x_old, best_seq)
        else:
            a_big = np.random.dirichlet(np.ones(n_big))

        a_big = np.maximum(a_big, 0.0)
        a_big = a_big / a_big.sum()

        # short Adam on the larger grid
        a_big_adam, _ = _adam_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            max_iters=120_000,
        )

        # longer sub‑gradient refinement on the large grid
        a_big_ref, val_big_ref = _subgradient_optimize(
            a_big_adam,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            max_iters=100_000,
            eta0=0.8,
        )
        if val_big_ref < best_val:
            best_val = val_big_ref
            best_seq = a_big_ref.copy()
            print(f"[Big] refined best {val_big_ref:.6f}")

    # ------------------------------------------------------------------
    # final safety fallback
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # enforce the evaluator’s range
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
