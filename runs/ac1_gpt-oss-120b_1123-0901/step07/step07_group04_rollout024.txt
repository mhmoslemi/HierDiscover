def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.1
    # initial n_small = 1024
    n = 1024
    conv_fft_len = _next_pow2(2*n - 1)
    corr_fft_len = _next_pow2(3*n - 2)

    # warm start: get global height_sequence_1 if present
    ws = None
    global height_sequence_1
    if 'height_sequence_1' in globals():
        ws_arr = np.asarray(height_sequence_1, dtype=np.float64)
        if ws_arr.shape[0] != n:
            # interpolate to size n
            x_old = np.linspace(0, 1, ws_arr.shape[0], endpoint=False)
            x_new = np.linspace(0, 1, n, endpoint=False)
            ws_arr = np.interp(x_new, x_old, ws_arr)
        ws_arr = np.maximum(ws_arr, 0.)
        if ws_arr.sum() < 0.01:
            ws = None
        else:
            ws_arr = ws_arr / ws_arr.sum()
            ws = ws_arr

    # generate candidates
    candidates = _generate_parametric_candidates(n, ws)

    # evaluate candidates
    cand_vals = []
    for cand in candidates:
        if time.time() >= deadline:
            break
        cand = np.maximum(cand, 0.)
        s = cand.sum()
        if s < 0.01:
            cand = np.random.dirichlet(np.ones(n))
            s = cand.sum()
        cand = cand / s
        val = _objective(cand, n, conv_fft_len)
        cand_vals.append((val, cand))

    if not cand_vals:
        best_seq = np.full(n, 1.0/n)
        best_val = _objective(best_seq, n, conv_fft_len)
        print("[fallback] Uniform vector.")
    else:
        cand_vals.sort(key=lambda x: x[0])
        best_val, best_seq = cand_vals[0]  # best after initial
        print(f"[init] Best initial objective = {best_val:.6f}")

    # gradient descent refinement
    # schedule
    beta = 0.5
    beta_max = 30.0
    beta_inc = (beta_max / beta) ** (1.0 / 500000)  # factor to reach max after many
    eta0 = 0.5
    min_eta = 0.001
    decay = 0.99995  # for step-size if using iteration based

    it = 0
    last_improve = time.time()
    while time.time() < deadline:
        # compute conv and weighted gradient
        conv = _full_convolution(best_seq, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (best_seq.sum()**2)
        # track best
        if val < best_val - 1e-12:
            best_val = val
            best_seq = best_seq.copy()
            last_improve = time.time()
            print(f"[iter={it:08d}] new best {val:.6f}")

        # compute softmax weights
        w = np.exp(beta * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.zeros_like(conv)
        else:
            w = w / w_sum

        # gradient
        grad = 2.0 * _cross_correlation(w, best_seq, corr_fft_len)

        # step size schedule
        eta = max(min_eta, eta0 * (decay ** it))

        # update using exponentiated gradient
        best_seq = best_seq * np.exp(-eta * grad)
        s = best_seq.sum()
        if s < 1e-12 or np.isnan(s):
            best_seq = np.random.dirichlet(np.ones(n))
            s = best_seq.sum()
        best_seq = best_seq / s

        # temperature increase
        beta = min(beta_max, beta * beta_inc)

        # jitter if stagnated
        if time.time() - last_improve > 5.0:
            perturb = (np.random.rand(n) - 0.5) * 0.02
            best_seq = best_seq * np.exp(perturb)
            best_seq = best_seq / best_seq.sum()
            last_improve = time.time()
            print("[jitter] injected random perturbation")

        it += 1

    # final upscale if time remains
    time_left = deadline - time.time()
    if time_left > 20.0:
        # upscale to larger n (8192)
        n_big = 8192
        conv_fft_len_big = _next_pow2(2*n_big - 1)
        corr_fft_len_big = _next_pow2(3*n_big - 2)
        # interpolate best_seq onto bigger grid
        x_small = np.linspace(0, 1, n, endpoint=False)
        x_big = np.linspace(0, 1, n_big, endpoint=False)
        a_big = np.interp(x_big, x_small, best_seq)
        a_big = np.maximum(a_big, 0.)
        a_big = a_big / a_big.sum()
        best_seq = a_big
        n = n_big
        conv_fft_len = conv_fft_len_big
        corr_fft_len = corr_fft_len_big
        print("[upscale] Upscaled to larger grid, continue refinement.")
        # continue refinement on bigger grid
        beta = 0.5
        it_big = 0
        last_improve = time.time()

            conv = _full_convolution(best_seq, conv_fft_len)
            max_c = conv.max()
            val = 2.0 * n * max_c / (best_seq.sum()**2)
            if val < best_val - 1e-12:
                best_val = val
                best_seq = best_seq.copy()
                last_improve = time.time()
                print(f"[big iter={it_big:08d}] new best {val:.6f}")

            w = np.exp(beta * (conv - max_c))
            w_sum = w.sum()
            if w_sum == 0 or np.isnan(w_sum):

"""
Improved search for a low‑evaluation step‑function.

The algorithm proceeds in three stages:

1. **Diverse seeds** – a large pool of parametric families (uniform,
   Dirichlet, edge‑exponential/power‑law, single‑sided exponential,
   Gaussian edges, sinusoid, linear ramps, etc.) is built for a modest
   size ``n_small`` (1024).  The best few seeds are kept.

2. **Smooth‑max mirror descent** – the objective
   ``F(a) = 2·n·max(a*a) / (sum(a)^2)`` is minimised using an
   exponentiated‑gradient (mirror‑descent) update.
   The non‑smooth “max” is replaced by a soft‑max with an annealed
   temperature β, giving a well‑behaved gradient:
       grad = 2·(w ★ a) ,   w_i ∝ exp(β·(conv_i−max(conv))).
   The step size decays slowly and a tiny random jitter is injected
   when progress stalls.

3. **Upscaling & fine‑tuning** – when time permits the best vector is
   interpolated to a larger grid (``n_big = 8192``) and the same
   smooth‑max optimisation continues, allowing the extra degrees of
   freedom to lower the peak of the convolution further.

All updates keep the vector non‑negative and on the probability simplex,
so scaling cancels out exactly as in the original problem.  The routine
respects the 1000 s budget, prints progress, and finally returns the
best sequence found as a plain Python list (clipped to the evaluator’s
range).
"""

import time
import math
import numpy as np
from typing import List, Tuple, Optional

# optional LP improvement – only used if SciPy is available
try:
    import scipy.sparse as sp
    from scipy.optimize import linprog
    _HAS_SCIPY = True
except Exception:  # pragma: no cover
    _HAS_SCIPY = False


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n-1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Fast evaluation of the objective for a (assumed normalised)."""
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c / (a.sum() ** 2))


def _generate_parametric_candidates(
    n: int,
    warm_start: Optional[np.ndarray],
) -> List[np.ndarray]:
    """
    Produce a diverse set of candidate vectors using several families:
    uniform, Dirichlet, edge‑exponential, single‑sided exponential,
    edge‑power‑law, double‑Gaussian edges, half‑wave sinusoid, linear ramp.
    All candidates are normalised to sum = 1.
    """
    candidates: List[np.ndarray] = []

    # ------------------------------------------------------------------
    # Warm‑start and simple rotations
    if warm_start is not None:
        candidates.append(warm_start.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            candidates.append(np.roll(warm_start, shift))

    # ------------------------------------------------------------------
    # Simple baselines
    candidates.append(np.full(n, 1.0 / n))                     # uniform
    candidates.append(np.linspace(1, 0, n, endpoint=False)   # linear decrease
                     / np.linspace(1, 0, n, endpoint=False).sum())

    # ------------------------------------------------------------------
    # Random Dirichlet draws with many concentration values
    for conc in (0.01, 0.05, 0.1, 0.3, 0.5, 1.0, 2.0):
        cand = np.random.dirichlet(np.full(n, conc))
        candidates.append(cand)

    # ------------------------------------------------------------------
    i = np.arange(n, dtype=np.float64)

    # Edge‑exponential (symmetric) + uniform mixtures
    for alpha in np.logspace(-2, 1.0, num=12):
        edge = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
        edge /= edge.sum()
        for w in (0.0, 0.25, 0.5, 0.75, 1.0):
            candidates.append(w * edge + (1.0 - w) / n)

    # Single‑sided exponential (left and right)
    for alpha in np.logspace(-2, 1.0, num=12):
        left = np.exp(-alpha * i)
        left /= left.sum()
        right = np.exp(-alpha * (n - 1 - i))
        right /= right.sum()
        candidates.append(left)
        candidates.append(right)

    # Edge‑power‑law (symmetric)
    for p in np.linspace(0.2, 5.0, num=12):
        edge = (i + 1.0) ** (-p) + (n - i) ** (-p)
        edge /= edge.sum()
        for w in (0.0, 0.33, 0.66, 1.0):
            candidates.append(w * edge + (1.0 - w) / n)

    # Double‑Gaussian edges
    for gamma in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        shape /= shape.sum()
        candidates.append(shape)

    # Positive half‑wave sinusoid
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    candidates.append(shape)

    return candidates


def _smooth_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Soft‑max weights for a given temperature β."""
    mx = conv.max()
    # subtract the maximum to avoid overflow; all entries ≤ 0
    exps = np.exp(beta * (conv - mx))
    s = exps.sum()
    if s == 0 or np.isnan(s):
        return np.zeros_like(conv)
    return exps / s


def _lp_step(f: np.ndarray, M: float, t: float,
             deadline: float) -> Tuple[np.ndarray, bool]:
    """
    One iteration of the classic alternating LP (if SciPy is available).

    Solve: maximise Σ g_i   s.t.   f * g ≤ M  (element‑wise) ,  g ≥ 0.
    The solution is then renormalised to sum = 1 and blended:
        f ← (1‑t)·f + t·ĝ .
    Returns the new f and a flag indicating success.
    """
    if not _HAS_SCIPY:
        return f, False

    n = f.shape[0]
    rows = []
    cols = []
    data = []

    # build sparse matrix A where (A·g)_k = Σ_i f_i·g_{k‑i}
    for k in range(2 * n - 1):
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(i)
            data.append(f[i])

    A = sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    b = np.full(2 * n - 1, M, dtype=np.float64)

    # maximise Σ g_i   ⇔   minimise −Σ g_i
    c = -np.ones(n, dtype=np.float64)

    # solve the linear program
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=(0, None),
                      method='highs', options={'presolve': True, 'time_limit': deadline - time.time()})
    except Exception:  # pragma: no cover
        return f, False

    if res.success:
        g = res.x
        s = g.sum()
        if s == 0 or np.isnan(s):
            return f, False
        g_norm = g / s
        new_f = (1 - t) * f + t * g_norm
        new_f = np.maximum(new_f, 0.0)
        new_f /= new_f.sum()
        return new_f, True
    return f, False
