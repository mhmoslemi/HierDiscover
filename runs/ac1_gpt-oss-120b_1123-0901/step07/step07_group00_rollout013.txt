def compute_conv(a, fft_len):
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[:2*a.shape[0]-1]

Define cross-correlation:

```
def cross_corr(w, a, fft_len):
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]
```

    a = a / a.sum()

        best_local_val = val

            if val_new < best_local_val:
                best_local_val = val_new
                best_local_a = a_new
        a = best_local_a
        it += 1
    return best_a, best_val

def _build_initial_pool(n):
    pool = []
    # warm possibilities
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm_arr = np.asarray(warm, dtype=np.float64)
        warm_resized = maybe_resize(warm_arr, n)
        pool.append(warm_resized)
        # rotate
        for shift in (n//4, n//2, 3*n//4):
            pool.append(np.roll(warm_resized, shift))
    # random Dirichlet
    pool.append(np.random.dirichlet(np.ones(n)))
    # edge exponential random
    for _ in range(3):
        beta = np.random.uniform(0.1, 3.0)
        left = np.exp(-beta * np.arange(n))
        shape = left + left[::-1]
        shape /= shape.sum()
        pool.append(shape)
    # polynomial power law
    for _ in range(3):
        alpha = np.random.uniform(-3.0, -0.3)
        idx = np.arange(1, n+1)
        prod = idx * (n - idx + 1)
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        shape_sum = shape.sum()
        if shape_sum > 0:
            shape /= shape_sum
            pool.append(shape)
    # two Gaussians
    for _ in range(3):
        sigma1 = np.random.uniform(0.05, 0.25)
        sigma2 = np.random.uniform(0.05, 0.25)
        i = np.arange(n)
        g1 = np.exp(-0.5 * ((i) / (sigma1 * n)) ** 2)
        g2 = np.exp(-0.5 * ((i - (n-1)) / (sigma2 * n)) ** 2)
        shape = g1 + g2
        shape /= shape.sum()
        pool.append(shape)
    # sinusoid with random phase
    phase = np.random.rand() * 2*np.pi
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n + phase)
    shape = np.maximum(shape, 0)
    shape /= shape.sum()
    pool.append(shape)
    # uniform
    pool.append(np.full(n, 1.0 / n))
    # sparse spikes
    for k in (8, 16, 32, 64):
        a = np.zeros(n)
        if k > 0:
            pos = np.round(np.linspace(0, n-1, k)).astype(int)
            a[pos] = 1.0 / k
        pool.append(a)
    return pool

def _directed_local_search(a, n, conv_fft_len, deadline, max_moves=50000):
    a_cur = a.copy()
    conv_cur = _full_convolution(a_cur, conv_fft_len)
    best_val = 2.0 * n * conv_cur.max() / (a_cur.sum()**2)
    best_a = a_cur.copy()
    it = 0
    while time.time() < deadline and it < max_moves:
        # compute gradient for current a (maybe could reuse)
        conv_cur = _full_convolution(a_cur, conv_fft_len)
        max_c = conv_cur.max()
        active = np.isclose(conv_cur, max_c, atol=1e-14)
        w = np.zeros_like(conv_cur)
        if active.any():
            w[active] = 1.0/active.sum()
        grad = 2.0 * _cross_correlation(w, a_cur, _next_pow2(3*n-2))
        # donors from high grad, receivers from low grad
        prob_donor = grad - grad.min() + 1e-12
        prob_donor /= prob_donor.sum()
        prob_recv = grad.max() - grad + 1e-12
        prob_recv /= prob_recv.sum()
        i = np.random.choice(n, p=prob_donor)
        j = np.random.choice(n, p=prob_recv)
        if i == j or a_cur[i] < 1e-14:
            it += 1
            continue
        delta = a_cur[i] * (np.random.rand() * 0.2)
        a_new = a_cur.copy()
        a_new[i] -= delta
        a_new[j] += delta
        conv_new = _full_convolution(a_new, conv_fft_len)
        val_new = 2.0 * n * conv_new.max() / (a_new.sum()**2)
        if val_new < best_val:

            best_a = a_new.copy()
            a_cur = a_new
            if it % 5000 == 0:
                print(f"[Directed move {it}] improved to {val_new:.6f}")

"""
Propose a step‑function (non‑negative vector) that minimises

        F(a) = 2·n·max(a∗a) / (sum(a)^2)

The strategy is a multi‑scale search:

1. **Small‑scale warm‑starts** – generate a diverse pool of candidates on a
   modest grid (n≈1024) and run a fast Adam optimisation on a smooth‑max
   surrogate (log‑sum‑exp).  The best small‑scale vector is kept.
2. **Upsampling** – linearly interpolate the best small‑scale vector to a
   larger grid (n≈4096 or 8192) and continue optimisation.
3. **Sub‑gradient phase with back‑tracking line‑search** – the exact max‑norm
   is differentiable only sub‑gradually; we apply exponentiated‑gradient
   updates and, for each iteration, try a few step‑sizes (η, η/2, η/5) and keep
   the one that yields the largest decrease.
4. **Directed random hill‑climbing** – cheap moves that transfer a fraction
   of mass from a “high‑gradient” index to a “low‑gradient” one, accepting
   only improving moves.
5. **Return the best sequence** (clipped to the evaluator’s allowed range).

All phases respect the supplied time budget; the current global best is
always returned, so the routine never times‑out with a negative score.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a (length 2·len(a)-1) via FFT.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a length‑len(a) vector:
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i = 0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]

def _build_initial_pool(n: int) -> list[np.ndarray]:
    """Create a diverse set of start vectors for a given size."""
    pool = []

    # Warm start (if any) and a few cyclic shifts
    warm = globals().get("height_sequence_1")
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] != n:
            x_old = np.linspace(0.0, 1.0, w.shape[0], endpoint=False)
            x_new = np.linspace(0.0, 1.0, n, endpoint=False)
            w = np.interp(x_new, x_old, w)
        w = np.maximum(w, 0.0)
        if w.sum() > 0:
            w = w / w.sum()
        pool.append(w.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            pool.append(np.roll(w, shift))

    # Random Dirichlet (full support)
    pool.append(np.random.dirichlet(np.ones(n)))

    # Edge‑exponential families (mass at the ends)
    for _ in range(3):
        beta = np.random.uniform(0.1, 3.0)
        left = np.exp(-beta * np.arange(n, dtype=np.float64))
        shape = left + left[::-1]
        shape /= shape.sum()
        pool.append(shape)

    # Edge‑polynomial (power‑law) families
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for _ in range(3):
        alpha = np.random.uniform(-3.0, -0.3)
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        s = shape.sum()
        if s > 0:
            shape /= s
            pool.append(shape)

    # Two‑Gaussian mixtures (different widths)
    i = np.arange(n, dtype=np.float64)
    for _ in range(3):
        sigma1 = np.random.uniform(0.05, 0.25)
        sigma2 = np.random.uniform(0.05, 0.25)
        g1 = np.exp(-0.5 * ((i) / (sigma1 * n)) ** 2)
        g2 = np.exp(-0.5 * ((i - (n - 1)) / (sigma2 * n)) ** 2)
        shape = g1 + g2
        shape /= shape.sum()
        pool.append(shape)

    # Positive half‑wave sinusoid with random phase
    phase = np.random.rand() * 2.0 * np.pi
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n + phase)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    pool.append(shape)

    # Uniform
    pool.append(np.full(n, 1.0 / n))

    # Sparse “spike’’ patterns
    for k in (8, 16, 32, 64, 128):
        a = np.zeros(n, dtype=np.float64)
        if k > 0:
            pos = np.round(np.linspace(0, n - 1, k)).astype(int)
            a[pos] = 1.0 / k
        pool.append(a)

    return pool


# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max surrogate.
    Returns the best vector encountered (normalised, sum≈1) and its
    exact objective value.
    """
    eps = 1e-15
    # initialise on the simplex via a log‑parameterisation
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05               # a bit more conservative than the previous run
    lam = 0.02              # start temperature
    lam_factor = 1.2        # accelerate temperature growth
    lam_max = 1e7

    best_a = a0.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        # ----- project logits onto the simplex (soft‑max) -----
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # ----- exact objective -----
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # ----- smooth‑max weights (soft‑max over convolution values) -----
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the surrogate -----
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # ----- mirror‑descent (soft‑max) update -----
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # ----- temperature schedule -----
        lam = min(lam * lam_factor, lam_max)

        # occasional tiny random perturbation near deadline
        if it % 4000 == 0 and time.time() > deadline - 30:
            logits += (np.random.rand(n) - 0.5) * 1e-2

        # gentle learning‑rate decay
        if it % 50_000 == 0 and it > 0:
            lr *= 0.9

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
def _subgrad_optimize_line_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
    eta0: float = 0.8,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient phase with a tiny back‑tracking line search.
    Uses exponentiated‑gradient updates (mirror descent on the simplex).
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active set (indices where the max is attained)
        active = np.isclose(conv, max_c, atol=1e-14)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # decreasing base step size
        eta = eta0 / np.sqrt(it + 1)

        # Try a few step‑sizes (full, half, one‑fifth) and keep the best.
        best_local_a = a
        best_local_val = val
        for factor in (1.0, 0.5, 0.2):
            step = eta * factor
            a_try = a * np.exp(-step * grad)
            s = a_try.sum()
            if s < 1e-12 or np.isnan(s):
                continue
            a_try = a_try / s
            conv_try = _full_convolution(a_try, conv_fft_len)
            val_try = 2.0 * n * conv_try.max() / (a_try.sum() ** 2)
            if val_try < best_local_val:
                best_local_val = val_try
                best_local_a = a_try

        a = best_local_a
        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 60_000,
) -> tuple[np.ndarray, float]:
    """
    Cheap hill‑climbing that randomly transfers a fraction of mass between
    two positions, keeping a move only if the objective improves.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()

    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        # move up to 30 % of the source mass
        delta = a[i] * (np.random.rand() * 0.3)

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5000 == 0:
                print(f"[Local move {it:06d}] improved to {val2:.6f}")

        it += 1

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the given time budget and always returns the best
    sequence (as a Python list) found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5      # tiny safety margin

    # --------------------------------------------------------------
    # 1️⃣  Small‑scale exploration (n≈1024)
    n_small = 1024
    conv_fft_small = _next_pow2(2 * n_small - 1)
    corr_fft_small = _next_pow2(3 * n_small - 2)

    small_pool = _build_initial_pool(n_small)

    best_seq_small = None
    best_val_small = float("inf")

    # allocate roughly one‑third of the total budget to the small stage
    small_deadline = time.time() + (deadline - time.time()) * 0.33
    per_cand_deadline = small_deadline - time.time()
    for idx, cand in enumerate(small_pool):
        if time.time() >= small_deadline:
            break
        a0 = np.maximum(cand, 0.0)
        if a0.sum() < 0.01:
            a0 = np.random.dirichlet(np.ones(n_small))
        a0 = a0 / a0.sum()

        # Adam on the smooth‑max surrogate
        local_deadline = min(small_deadline, time.time() + per_cand_deadline / max(1, len(small_pool) - idx))
        a_adam, _ = _adam_optimize(
            a0,
            n_small,
            conv_fft_small,
            corr_fft_small,
            local_deadline,
            max_iters=120_000,
        )

        # Sub‑gradient phase with line‑search
        if time.time() < small_deadline:
            a_sub, val_sub = _subgrad_optimize_line_search(
                a_adam,
                n_small,
                conv_fft_small,
                corr_fft_small,
                small_deadline,
                max_iters=30_000,
                eta0=0.9,
            )
        else:
            a_sub, val_sub = a_adam, float("inf")

        if val_sub < best_val_small:
            best_val_small = val_sub
            best_seq_small = a_sub.copy()
            print(f"[small cand {idx}] new best {val_sub:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  Upsample to a larger grid (preferably 4096 or 8192)
    if best_seq_small is None:
        # fallback: uniform distribution
        best_seq_small = np.full(n_small, 1.0 / n_small)
        best_val_small = 2.0 * n_small * (1.0 / n_small) / (1.0 ** 2)   # = 2

    # Choose a larger size based on remaining time
    remaining = deadline - time.time()
    if remaining > 200:
        n_large = 8192
    else:
        n_large = 4096

    # linear interpolation to the larger grid
    x_small = np.linspace(0.0, 1.0, n_small, endpoint=False)
    x_large = np.linspace(0.0, 1.0, n_large, endpoint=False)
    a_large = np.interp(x_large, x_small, best_seq_small)
    a_large = np.maximum(a_large, 0.0)
    a_large = a_large / a_large.sum()

    conv_fft_large = _next_pow2(2 * n_large - 1)
    corr_fft_large = _next_pow2(3 * n_large - 2)

    # --------------------------------------------------------------
    # 3️⃣  Adam on the large grid (short phase)
    if time.time() < deadline - 30:
        large_adam_deadline = min(deadline - 20, time.time() + (deadline - time.time()) * 0.2)
        a_large_adam, _ = _adam_optimize(
            a_large,
            n_large,
            conv_fft_large,
            corr_fft_large,
            large_adam_deadline,
            max_iters=200_000,
        )
    else:
        a_large_adam = a_large

    # --------------------------------------------------------------
    # 4️⃣  Long sub‑gradient refinement with line‑search
    if time.time() < deadline - 10:
        a_refined, val_refined = _subgrad_optimize_line_search(
            a_large_adam,
            n_large,
            conv_fft_large,
            corr_fft_large,
            deadline,
            max_iters=250_000,
            eta0=0.6,
        )
    else:
        # compute a quick objective if we ran out of time
        conv_tmp = _full_convolution(a_large_adam, conv_fft_large)
        val_refined = 2.0 * n_large * conv_tmp.max() / (a_large_adam.sum() ** 2)
        a_refined = a_large_adam

    best_seq = a_refined
    best_val = val_refined
    print(f"[Large] best after refinement {best_val:.6f}")

    # --------------------------------------------------------------
    # 5️⃣  Final cheap random local search
    if time.time() < deadline - 2:
        a_loc, val_loc = _local_random_search(
            best_seq,

    if best_seq is None:

    best_seq = np.clip(best_seq, 0.0, 1000.0)
    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
