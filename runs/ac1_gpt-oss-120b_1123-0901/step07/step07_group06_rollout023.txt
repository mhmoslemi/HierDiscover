def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5

    n = 4096
    conv_len = _next_pow2(2*n-1)
    corr_len = _next_pow2(3*n-2) # for cross correlation

    # warm-start
    warm_start = globals().get('height_sequence_1')
    if warm_start is not None:
        w = np.asarray(warm_start, dtype=float)
        if w.size != n:
            # interpolate to size n
            x_old = np.linspace(0, 1, w.size, endpoint=False)
            x_new = np.linspace(0, 1, n, endpoint=False)
            w = np.interp(x_new, x_old, w)
        w = np.clip(w, 0, None)
        if w.sum() < 0.01:
            w = None
        else:
            w = w / w.sum()

    # initial candidates list
    candidates = []

    # add warm start and rotations
    if warm_start is not None:
        cand = w.copy()
        candidates.append(cand)
        for shift in (n//4, n//2, 3*n//4):
            candidates.append(np.roll(cand, shift))

    # uniform
    candidates.append(np.full(n, 1.0/n))

    # random Dirichlet draws
    for conc in (0.01, 0.05, 0.1, 0.3, 0.5, 1.0, 2.0, 4.0):
        cand = np.random.dirichlet(np.full(n, conc))
        candidates.append(cand)

    # edge exponential family
    i = np.arange(n, dtype=float)
    for alpha in np.logspace(-2, 2, num=30):
        edge = np.exp(-alpha * i) + np.exp(-alpha * (n-1-i))
        edge = edge / edge.sum()
        for w in np.linspace(0,1,7):
            cand = w*edge + (1-w)/n
            candidates.append(cand)

    # stretched exponential (beta exponent)
    for beta in np.linspace(0.5, 2.0, num=10):
        for alpha in np.logspace(-2, 2, num=20):
            edge = np.exp(-alpha * (i**beta)) + np.exp(-alpha * ((n-1-i)**beta))
            edge = edge / edge.sum()
            for w in np.linspace(0,1,5):
                cand = w*edge + (1-w)/n
                candidates.append(cand)

    # power law
    for p in np.linspace(0.2, 5.0, num=30):
        edge = (i+1)**(-p) + (n-i)**(-p)
        edge = edge / edge.sum()
        for w in np.linspace(0,1,5):
            cand = w*edge + (1-w)/n
            candidates.append(cand)

    # beta-shape
    for a_param in (0.1,0.2,0.5,1.0,2.0,5.0):
        for b_param in (0.1,0.2,0.5,1.0,2.0,5.0):
            edge = (i+1)**(a_param-1) * (n-i)**(b_param-1)
            edge = edge / edge.sum()
            candidates.append(edge)

    # triangular and quadratic
    tri = np.minimum(i+1, n-i)
    tri = tri / tri.sum()
    candidates.append(tri)
    quad = (i+1)*(n-i)
    quad = quad / quad.sum()
    candidates.append(quad)

    # evaluate all candidates quickly
    pool = [] # list of (val, vector)
    for cand in candidates:
        if time.time() >= deadline:
            break
        cand = np.clip(cand, 0, None)
        s = cand.sum()
        if s < 1e-12:
            cand = np.random.dirichlet(np.ones(n))
            s = cand.sum()
        cand = cand / s
        val = _objective(cand, n, conv_len)
        pool.append((val, cand))

    # keep top pool_size best
    pool.sort(key=lambda x: x[0])
    pool_size = 30
    pool = pool[:pool_size]
    best_val, best_seq = pool[0][0], pool[0][1].copy()

    print(f"Initial best objective: {best_val:.6f}")

    # evolutionary loop
    while time.time() < deadline - 5.0:
        # choose operation
        if np.random.rand() < 0.6 and len(pool) >= 2:
            # blend two parents
            p1, p2 = np.random.choice(pool, size=2, replace=False)
            child = (p1[1] + p2[1]) / 2.0
        else:
            # mutate a parent
            parent = np.random.choice(pool)[1]
            child = parent.copy()

        # add gaussian jitter
        sigma = 0.02
        child += sigma * (np.random.rand(n) - 0.5)
        child = np.clip(child, 0, None)
        s = child.sum()
        if s < 1e-12:
            continue
        child /= s

        # evaluate
        val = _objective(child, n, conv_len)

        # update best
        if val < best_val:
            best_val = val
            best_seq = child.copy()
            print(f"[Evolution] new best {best_val:.6f}")

        # insert into pool if better than worst
        worst_val, _ = pool[-1]
        if val < worst_val:
            pool.append((val, child))
            pool.sort(key=lambda x: x[0])
            pool = pool[:pool_size]

    # final refinement using projected gradient descent
    # compute FFT lengths for gradient uses
    print(f"Starting final projected gradient refinement. Best so far {best_val:.6f}")
    a = best_seq.copy()
    # ensure positivity and normalization
    a = np.clip(a, 0, None)
    a_sum = a.sum()
    if a_sum < 1e-12:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum

    # use projected gradient refine
    a_refined, final_val = _projected_gradient_refine(
        a, n, conv_len, corr_len, deadline, None, best_val
    )
    if final_val < best_val:
        best_seq = a_refined
        best_val = final_val

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Finished search. Best objective ≈ {best_val:.6f}")
    return best_seq.tolist()

"""
Propose a low‑evaluation step‑function.

Algorithm (≈ 1000 s)
--------------------
1. Build a diverse library of parametric candidates (uniform,
   Dirichlet, edge‑exponential / stretched‑exponential,
   edge‑power‑law, Beta‑shaped, triangular / quadratic, and a
   warm‑start if available).  All candidates are normalised to sum=1.

2. Evaluate every candidate with a fast FFT‑based convolution and keep the
   best K (K≈30) for deep optimisation.

3. For each of the K starts run a three‑stage optimiser:
   * Adam on a smooth‑max surrogate (log‑sum‑exp) with an exponentially
     increasing temperature.
   * Exponentiated‑gradient sub‑gradient steps on the true max‑convolution.
   * Projected‑gradient descent on the exact objective (simplex projection).

   The time budget is split evenly over the K starts so that each start
   receives a substantial amount of iterations.

4. Perform cheap stochastic perturbations of the current global best and
   re‑run a short sub‑gradient refinement (simulated‑annealing style).

5. If enough time remains, upscale the best vector to n=8192 by linear
   interpolation and repeat the whole deep‑optimisation pipeline.

The routine obeys the time budget, prints progress and finally returns
the best sequence as a plain list of non‑negative floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Returns the projected vector (non‑negative, sum=1).
    """
    if v.ndim != 1:
        raise ValueError("v must be 1‑d")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact evaluation of the score (fast, assumes sum(a)≈1)."""
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c / (a.sum() ** 2))


# ----------------------------------------------------------------------
def _adam_smooth_max(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """Adam on a log‑sum‑exp surrogate of max(a*a)."""
    eps = 1e-15
    logits = np.log(a0 + eps)            # work in log‑space (soft‑max)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05

    lam = 0.5                # start temperature
    lam_factor = 1.03        # quick schedule
    lam_cap = 1e8

    last_improve = time.time()
    it = 0
    max_iter = 800_000       # generous upper bound – will break on deadline

    while time.time() < deadline and it < max_iter:
        # ---- simplex via soft‑max ----
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # ---- surrogate evaluation ----
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Adam n={n} it={it:06d}] new best {val:.6f}")

        # ---- smooth‑max weights (log‑sum‑exp) ----
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        # ---- gradient of the surrogate (chain‑rule through soft‑max) ----
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---- temperature schedule ----
        lam = min(lam * lam_factor, lam_cap)

        # ---- occasional noise if stuck ----
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1

    return best_seq, best_val


def _subgradient_refinement(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient sub‑gradient steps on the true max‑convolution."""
    a = best_seq if best_seq is not None else a0.copy()
    a = np.maximum(a, 0.0)
    a /= a.sum()                     # stay on the simplex

    eta = 0.6
    last_improve = time.time()
    max_iter = 300_000               # generous upper bound, will stop on deadline

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Subgrad n={n} it={it:06d}] new best {val:.6f}")

        # ---- active set (where max is attained) ----
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # ---- exponentiated‑gradient update (mirrored to the simplex) ----
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:                     # avoid collapse
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        # ---- early stop if stalled ----
        if time.time() - last_improve > 5.0:
            break

        # slowly decay eta to fine‑tune
        eta *= 0.9999

    return best_seq, best_val


def _projected_gradient_refine(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """Gradient descent on the exact objective with explicit simplex projection."""
    a = a0.copy()
    a = np.maximum(a, 0.0)
    a /= a.sum()

    lr = 0.2
    last_improve = time.time()
    max_iter = 300_000

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[ProjGrad n={n} it={it:06d}] new best {val:.6f}")

        # active mask (tolerance to numerical errors)
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)   # dM/da
        # objective gradient = 2·n·grad_a (sum fixed → no extra term)
        step = -lr * grad_a
        a_candidate = _proj_simplex(a + step)

        # accept if improvement, otherwise shrink step size
        conv_cand = _full_convolution(a_candidate, conv_fft_len)
        val_cand = 2.0 * n * conv_cand.max() / (a_candidate.sum() ** 2)

        if val_cand < val:
            a = a_candidate
            lr = min(lr * 1.02, 1.0)    # slowly enlarge step when moving downhill
        else:
            lr *= 0.7
            if lr < 1e-5:
                break

        if time.time() - last_improve > 5.0:
            break

        # gentle decay of learning rate as iterations progress
        lr *= 0.99998

    return best_seq, best_val


def _deep_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """Full optimisation pipeline for a single start."""
    best_seq, best_val = _adam_smooth_max(
        a0,
        n,
        conv_fft_len,
        corr_fft_len,
        deadline,
        global_best_seq,
        global_best_val,
    )
    best_seq, best_val = _subgradient_refinement(
        a0,
        n,
        conv_fft_len,
        corr_fft_len,

        best_seq,
        best_val,

    best_seq, best_val = _projected_gradient_refine(
        a0,

        conv_fft_len,
        corr_fft_len,

        best_seq,
        best_val,
    )
    return best_seq, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:

    Returns the best sequence as a plain Python list.

    deadline = time.time() + budget_s - 0.5   # safety margin

    n = 4096

    corr_fft_len = _next_pow2(3 * n - 2)

    warm_start = globals().get("height_sequence_1")
    if warm_start is not None:
        warm_start = np.asarray(warm_start, dtype=np.float64)
        if warm_start.shape[0] != n:
            # linear interpolation to required length
            x_old = np.linspace(0, 1, warm_start.shape[0], endpoint=False)
            x_new = np.linspace(0, 1, n, endpoint=False)
            warm_start = np.interp(x_new, x_old, warm_start)
        warm_start = np.maximum(warm_start, 0.0)
        if warm_start.sum() >= 0.01:
            warm_start = warm_start / warm_start.sum()
        else:
            warm_start = None
    else:
        warm_start = None

    cand_list = []

    # a) warm‑start and simple rotations
    if warm_start is not None:
        cand_list.append(warm_start.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand_list.append(np.roll(warm_start, shift))

    # b) uniform baseline
    cand_list.append(np.full(n, 1.0 / n))

    # c) random Dirichlet draws (various concentrations)
    for conc in (0.01, 0.05, 0.1, 0.3, 0.5, 1.0, 2.0, 4.0):
        cand_list.append(np.random.dirichlet(np.full(n, conc)))

    i = np.arange(n, dtype=np.float64)

    # d) edge‑exponential families (with optional uniform mix)
    for alpha in np.logspace(-2, 2, num=30):
        edge = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
        edge /= edge.sum()
        for w in np.linspace(0.0, 1.0, 7):
            shape = w * edge + (1.0 - w) / n
            cand_list.append(shape)

    # e) stretched‑exponential families (extra shape parameter beta)
    for beta in np.linspace(0.5, 2.0, num=10):
        for alpha in np.logspace(-2, 2, num=20):
            edge = np.exp(-alpha * (i ** beta)) + np.exp(-alpha * ((n - 1 - i) ** beta))
            edge /= edge.sum()
            for w in np.linspace(0.0, 1.0, 5):
                shape = w * edge + (1.0 - w) / n
                cand_list.append(shape)

    # f) edge‑power‑law families
    for p in np.linspace(0.2, 5.0, num=30):
        edge = (i + 1.0) ** (-p) + (n - i) ** (-p)
        edge /= edge.sum()
        for w in np.linspace(0.0, 1.0, 5):
            shape = w * edge + (1.0 - w) / n
            cand_list.append(shape)

    # g) Beta‑shaped families (flexible edge/centre bias)
    for a_param in (0.2, 0.5, 1.0, 2.0, 5.0):
        for b_param in (0.2, 0.5, 1.0, 2.0, 5.0):
            edge = (i + 1) ** (a_param - 1) * (n - i) ** (b_param - 1)
            edge = np.maximum(edge, 0.0)
            edge /= edge.sum()
            cand_list.append(edge)

    # h) simple triangular / quadratic shapes
    tri = np.minimum(i + 1, n - i)
    tri = tri.astype(np.float64)
    tri /= tri.sum()
    cand_list.append(tri)

    quad = (i + 1) * (n - i)
    quad = quad.astype(np.float64)
    quad /= quad.sum()
    cand_list.append(quad)

    # --------------------------------------------------------------
    # Quick evaluation of all candidates, keep the best K
    evaluated = []
    for cand in cand_list:
        if time.time() >= deadline:
            break
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 1e-12:
            cand = np.random.dirichlet(np.ones(n))
            s = cand.sum()
        cand = cand / s
        val = _objective(cand, n, conv_fft_len)
        evaluated.append((val, cand))

    evaluated.sort(key=lambda x: x[0])
    pool_size = 30
    pool = evaluated[:pool_size]
    best_val, best_seq = pool[0][0], pool[0][1].copy()
    print(f"Initial best objective after candidate sweep: {best_val:.6f}")

    # --------------------------------------------------------------
    # 2. Deep optimisation – split the remaining budget equally among the
    #    pool members so that each start receives a sizable amount of
    #    iterations.
    remaining = deadline - time.time()
    if remaining > 30.0:   # give at least a few seconds per start
        per_start = remaining / len(pool)
    else:
        per_start = remaining / max(1, len(pool))

    for idx, (_, start_vec) in enumerate(pool):
        if time.time() >= deadline:
            break
        sub_deadline = time.time() + per_start
        print(f"\n--- Deep optimisation on start #{idx + 1}/{len(pool)} ---")
        best_seq, best_val = _deep_optimize(
            start_vec,
            n,
            conv_fft_len,
            corr_fft_len,
            sub_deadline,
            best_seq,
            best_val,
        )

    # --------------------------------------------------------------
    # 3. Stochastic local perturbations around the current global best
    if best_seq is not None and time.time() < deadline - 5.0:
        print("\n--- Local stochastic perturbations ---")
        # a few cheap perturbation rounds
        while time.time() < deadline - 5.0:
            eps = 0.015
            pert = best_seq + eps * (np.random.rand(n) - 0.5)
            pert = np.maximum(pert, 0.0)
            s = pert.sum()
            if s < 1e-12:
                continue
            pert /= s
            val = _objective(pert, n, conv_fft_len)
            if val < best_val:
                best_val = val
                best_seq = pert
                print(f"[Perturb] new best {best_val:.6f}")

    # --------------------------------------------------------------
    # 4. Upscale to a larger grid if time permits
    time_left = deadline - time.time()
    if time_left > 40.0:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        # linear interpolation of the current best
        x_old = np.linspace(0, 1, n, endpoint=False)
        x_new = np.linspace(0, 1, n_big, endpoint=False)
        a_big = np.interp(x_new, x_old, best_seq)
        a_big = np.maximum(a_big, 0.0)
        a_big /= a_big.sum()

        print("\n--- Optimising on larger grid (n=8192) ---")
        sub_deadline = time.time() + time_left - 5.0
        best_seq, best_val = _deep_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            sub_deadline,
            best_seq,
            best_val,
        )
        if best_seq.shape[0] == n_big:
            n = n_big  # keep the larger size

    # --------------------------------------------------------------
    # 5. Fallback – uniform vector if nothing succeeded
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Clip to the evaluator's allowed range
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"\n*** Finished search: best objective ≈ {best_val:.6f} (n={best_seq.shape[0]})")
    return best_seq.tolist()
