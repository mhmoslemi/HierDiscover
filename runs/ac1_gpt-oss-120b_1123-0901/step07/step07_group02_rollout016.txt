def _adam_optimize_blocks(w0, n, blocksize, conv_fft_len, corr_fft_len, deadline, max_iters=20000):
    eps = 1e-12
    logits = np.log(w0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.08
    lam = 0.5
    lam_factor = 1.03

    best_w = w0.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        # softmax to get w
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        w = exp_log / exp_log.sum()
        # construct a
        a = np.repeat(w / blocksize, blocksize)

        # compute conv, max
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_w = w.copy()
            # optionally print
            if it % 1000 == 0:
                print(f"[AdamBlocks iter={it}] new best {val:.6f}")

        # compute smooth max weights
        w_s = np.exp(lam * (conv - max_c))
        w_s_sum = w_s.sum()
        if w_s_sum == 0 or np.isnan(w_s_sum):
            w_s = np.full_like(conv, 1.0 / conv.size)
        else:
            w_s /= w_s_sum

        grad_a = 2.0 * _cross_correlation(w_s, a, corr_fft_len)
        # block gradient
        # compute sum over each block
        # grad_w_j = sum_{i in block j} grad_a_i / blocksize
        block_sums = np.add.reduceat(grad_a, np.arange(0, n, blocksize))
        grad_w = block_sums / blocksize

        # gradient w.r.t logits
        avg = np.dot(w, grad_w)
        grad_log = w * (grad_w - avg)

        # Adam updates
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log * grad_log)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # update lam
        lam = min(lam * lam_factor, 1e6)

        # shake occasionally
        if it % 5000 == 0:
            logits += (np.random.rand(len(logits)) - 0.5) * 1e-3

        it += 1
    # after loop, return best_w and its objective
    return best_w, best_val

"""
Propose a step‑function (non‑negative vector) with a low evaluation score.

Algorithm
---------
1.  **Coarse block optimisation** (n≈1024, 64 blocks):
    * Represent the step‑function as a piecewise‑constant vector:
          a_i = w_{⌊i/bs⌋} / bs   (bs = block size, Σ w = 1, w ≥ 0).
    * Run many short Adam‑optimisations on the *smooth‑max* surrogate
      (log‑sum‑exp of the convolution).  The optimisation works on the
      block logits (soft‑max) which reduces the dimensionality dramatically.
    * Keep the best block vector `w`.

2.  **Upscale to the target resolution** (n = 4096, same block size 16):
    * Interpolate the best block weights to `K = n/bs` blocks and rebuild
      the full‑resolution vector `a`.

3.  **Full‑resolution refinement**:
    * A long sub‑gradient descent on the exact objective
      `2·n·max(a∗a)/(sum a)^2` (the same code as the reference solution).
    * A cheap hill‑climbing random‑move phase.

4.  **Final Adam polish** (optional, time permitting) on the full vector.

All phases respect the global time budget (default 1000 s) and the routine
always returns the best sequence found.  The code uses only NumPy and
SciPy (for a tiny `linspace` helper) and runs on a single CPU core.
"""

import time
import numpy as np
from scipy import interpolate  # only for linear interpolation, optional


# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation needed for the gradient.
    Returns a length‑len(a) vector:
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i = 0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
def _block_to_a(w: np.ndarray, blocksize: int) -> np.ndarray:
    """Expand block weights w (sum = 1) to a full vector a on the simplex."""
    return np.repeat(w / blocksize, blocksize)


def _block_grad_w(grad_a: np.ndarray, blocksize: int) -> np.ndarray:
    """Aggregate per‑element gradient to block‑wise gradient."""
    # grad_a length must be a multiple of blocksize
    sums = np.add.reduceat(grad_a, np.arange(0, grad_a.size, blocksize))
    return sums / blocksize


# ----------------------------------------------------------------------
def _adam_optimize_blocks(
    w0: np.ndarray,
    n: int,
    blocksize: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 30_000,
) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth‑max surrogate, operating on block‑logits.

    Returns the best block‑weight vector (sum = 1) and its exact objective.
    """
    eps = 1e-15
    logits = np.log(w0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.09
    lam = 0.5
    lam_factor = 1.02

    best_w = w0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        # soft‑max → simplex (block weights)
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        w = exp_log / exp_log.sum()

        # build full‑resolution vector
        a = np.repeat(w / blocksize, blocksize)

        # exact objective (sum(a)=1)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_w = w.copy()
            if it % 500 == 0:
                print(f"[AdamBlocks iter={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w_s = np.exp(lam * (conv - max_c))
        w_s_sum = w_s.sum()
        if w_s_sum == 0 or np.isnan(w_s_sum):
            w_s = np.full_like(conv, 1.0 / conv.size)
        else:
            w_s /= w_s_sum

        # gradient w.r.t a (smooth‑max surrogate)
        grad_a = 2.0 * _cross_correlation(w_s, a, corr_fft_len)

        # aggregate to block gradient
        grad_w = _block_grad_w(grad_a, blocksize)

        # soft‑max gradient chain rule
        avg = np.dot(w, grad_w)
        grad_log = w * (grad_w - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature ramp‑up
        lam = min(lam * lam_factor, 1e6)

        # occasional tiny random shake
        if it % 7_000 == 0:
            logits += (np.random.rand(logits.size) - 0.5) * 5e-3

        it += 1

    return best_w, best_val

def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation directly on the full‑resolution vector `a`
    (the implementation from the reference solution, kept unchanged).
    """
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.08
    lam = 0.5
    lam_factor = 1.03

    best_a = a0.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 10_000 == 0:
                print(f"[AdamFull iter={it:06d}] new best {val:.6f}")

        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        lam = min(lam * lam_factor, 1e6)

        if it % 7_000 == 0 and time.time() + 5 > deadline:
            logits += (np.random.rand(n) - 0.5) * 5e-3

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
def _subgradient_opt2(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
    eta0: float = 0.7,
) -> tuple[np.ndarray, float]:
    """
    Multiplicative (exponentiated‑gradient) sub‑gradient optimiser on the
    exact objective (no smoothing).  This is the same routine as in the
    reference implementation, kept verbatim.
    """
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s

    best_a = a.copy()
    best_val = float("inf")
    stall = 0
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            stall = 0
            if it % 5_000 == 0:
                print(f"[Subgrad iter={it:07d}] new best {val:.6f}")
        else:
            stall += 1

        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        eta = eta0 / np.sqrt(it + 1)

        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        if stall > 10_000:
            a = np.maximum(a + 0.02 * (np.random.rand(n) - 0.5), 0.0)
            a /= a.sum()
            stall = 0

        it += 1
    return best_a, best_val

def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Fast hill‑climbing: transfer a random fraction of mass from one index to
    another and keep the move if the exact objective improves.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    best_a = a.copy()
    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-12:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue
        delta = a[i] * (np.random.rand() * 0.4)   # move up to 40 % of source
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta
        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()
        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5_000 == 0:
                print(f"[Local move {it}] improved to {val2:.6f}")
        it += 1
    return best_a, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the given time budget and returns the best sequence
    (as a Python list) found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Phase A – coarse block optimisation (n0 = 1024, 64 blocks)
    n0 = 1024
    blocksize0 = 16                # n0 / 64
    K0 = n0 // blocksize0

    conv_fft_len0 = _next_pow2(2 * n0 - 1)
    corr_fft_len0 = _next_pow2(3 * n0 - 2)

    # initialise from a possible warm start (global variable `height_sequence_1`)
    warm = globals().get("height_sequence_1")
    warm_w = None
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n0:
            warm = warm[:n0]
        elif warm.shape[0] < n0:
            warm = np.pad(warm, (0, n0 - warm.shape[0]))
        # aggregate onto blocks
        block_sums = np.add.reduceat(warm, np.arange(0, n0, blocksize0))
        if block_sums.sum() > 0:
            warm_w = block_sums / block_sums.sum()

    best_w = None
    best_val = float("inf")

    # run many short restarts while time permits
    while time.time() < deadline - 30.0:  # keep ~30 s for later phases
        # random start (Dirichlet) – occasionally inject the warm start
        if warm_w is not None and np.random.rand() < 0.15:
            w0 = warm_w.copy()
        else:
            w0 = np.random.dirichlet(np.ones(K0))

        # occasional symmetrisation of the block vector
        if np.random.rand() < 0.2:
            w0 = (w0 + w0[::-1]) / 2.0
            w0 /= w0.sum()

        sub_deadline = min(deadline - 5.0, time.time() + 5.0)  # ~5 s per restart
        w_opt, _ = _adam_optimize_blocks(
            w0,
            n0,
            blocksize0,
            conv_fft_len0,
            corr_fft_len0,
            sub_deadline,
            max_iters=12_000,
        )
        a_opt = _block_to_a(w_opt, blocksize0)
        conv = _full_convolution(a_opt, conv_fft_len0)
        val = 2.0 * n0 * conv.max()
        if val < best_val:
            best_val = val
            best_w = w_opt.copy()
            print(f"[Coarse] new best {val:.6f}")

        # early stop if we already beat the target by a safe margin
        if best_val < 1.48:
            break

    if best_w is None:
        best_w = np.full(K0, 1.0 / K0)

    # ------------------------------------------------------------------
    # Phase B – upscale to the main resolution (n = 4096)
    n_big = 4096
    blocksize_big = 16               # fixed block size → K_big = 256
    K_big = n_big // blocksize_big

    # linear interpolation of block weights
    x_src = np.linspace(0, 1, best_w.size, endpoint=False)
    x_tgt = np.linspace(0, 1, K_big, endpoint=False)
    w_big = np.interp(x_tgt, x_src, best_w)
    w_big = np.maximum(w_big, 0.0)
    if w_big.sum() == 0:
        w_big = np.full(K_big, 1.0 / K_big)
    else:
        w_big /= w_big.sum()

    a_big = _block_to_a(w_big, blocksize_big)

    conv_fft_len_big = _next_pow2(2 * n_big - 1)
    corr_fft_len_big = _next_pow2(3 * n_big - 2)

    # ------------------------------------------------------------------
    # Phase C – long sub‑gradient refinement on the full vector
    sub_deadline = min(deadline - 10.0, time.time() + 45.0)
    a_ref, val_ref = _subgradient_opt2(
        a_big,
        n_big,
        conv_fft_len_big,
        corr_fft_len_big,
        sub_deadline,
        max_iters=400_000,
    )
    best_seq = a_ref.copy()
    best_val = val_ref
    print(f"[Full Subgrad] best {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase D – cheap local random moves
    if time.time() < deadline - 5:
        a_loc, val_loc = _local_random_search(
            best_seq,
            n_big,
            conv_fft_len_big,
            deadline,
            max_moves=250_000,
        )
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()
            print(f"[Local search] improved to {val_loc:.6f}")

    # ------------------------------------------------------------------
    # Phase E – final Adam polish if time is left
    if time.time() < deadline - 2:
        a_final, val_final = _adam_optimize(
            best_seq,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            max_iters=250_000,
        )
        if val_final < best_val:
            best_val = val_final
            best_seq = a_final.copy()
            print(f"[Final Adam] best {val_final:.6f}")

    # ------------------------------------------------------------------
    # Fallback – guarantee a valid return
    if best_seq is None:
        best_seq = np.full(n_big, 1.0 / n_big, dtype=np.float64)

    # Clip to the evaluator's safety limits
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
