def _project_simplex(v):
    # ensure sum(v) = 1 and v_i >=0.
    # Duchi et al. method:
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def _project_simplex(v):
    """Project onto the probability simplex {x >= 0, sum x = 1}."""
    # reference: Duchi et al. (2008)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all <=0: return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def _lbfgs_optimize(a0, n, conv_fft_len, corr_fft_len, deadline, max_iter=200):
    # operate on logits
    eps = 1e-12
    x0 = np.log(a0 + eps)

    def obj_and_grad(x):
        # a via softmax
        mx = np.max(x)
        exp_x = np.exp(x - mx)
        a = exp_x / exp_x.sum()
        # convolution
        conv = _full_convolution(a, conv_fft_len)
        # smoothed max with temperature tau
        tau = 50.0  # fixed temperature high enough to approximate max
        # compute log-sum-exp
        max_c = conv.max()
        # shift for numerical stability
        shifted = conv - max_c
        exp_vals = np.exp(tau * shifted)
        sum_exp = exp_vals.sum()
        smooth_max = max_c + np.log(sum_exp) / tau
        val = 2.0 * n * smooth_max
        # gradient
        w = exp_vals / sum_exp
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        # backprop through softmax: grad_x = a * (grad_a - np.dot(a, grad_a))
        grad_x = a * (grad_a - np.dot(a, grad_a))
        return val, grad_x.astype(np.float64)

    # Use L-BFGS-B with bounds (None) but we might set maxiter
    options = {'maxiter': max_iter, 'ftol': 1e-12}
    res = minimize(fun=lambda x: obj_and_grad(x)[0],
                   x0=x0,
                   jac=lambda x: obj_and_grad(x)[1],
                   method='L-BFGS-B',
                   options=options)
    # after optimization, compute final a and val precisely using exact max (no smoothing)
    x_opt = res.x
    mx = np.max(x_opt)
    exp_x = np.exp(x_opt - mx)
    a_opt = exp_x / exp_x.sum()
    conv_opt = _full_convolution(a_opt, conv_fft_len)
    val_opt = 2.0 * n * conv_opt.max()
    return a_opt, val_opt

"""
Propose a step‑function (non‑negative vector) with a low evaluation score.

Algorithm
---------
1. Generate a diverse pool of warm‑started candidates (uniform, Dirichlet,
   power‑law, double‑exponential, sinusoidal, sparse spikes with Gaussian
   smoothing).  If a previous solution exists it is used as a warm start.

2. Run a short Adam optimisation on a log‑sum‑exp surrogate of the
   convolution maximum for every candidate.  The parameters are kept on the
   simplex via a soft‑max.

3. Take the best candidate and run a long additive projected‑subgradient
   descent (exact sub‑gradient of the true max) with a decaying step size.
   After a long period of stagnation a tiny random perturbation is added.

4. Refine the result with a few L‑BFGS‑B iterations on the same smooth
   surrogate (optimising the logits of the soft‑max).  The final value is
   evaluated with the exact max‑convolution.

5. If time permits, interpolate the best sequence to a finer grid
   (8192 points) and repeat steps 3‑4 there.

All phases respect the global time budget (default 1000 s) and the routine
always returns the best sequence found.  No file I/O or lambda functions are
used.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a length‑len(a) vector:
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i = 0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of a vector onto the probability simplex {x≥0, Σx=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Generate a modest set of diverse start vectors."""
    cand = []

    # 1) Warm start and cyclic shifts
    if warm is not None:
        cand.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    # 2) Uniform
    cand.append(np.full(n, 1.0 / n))

    # 3) Random Dirichlet
    cand.append(np.random.dirichlet(np.ones(n)))

    # 4) Edge‑concentrated power‑law (1/(i·(n‑i+1)))
    i = np.arange(1, n + 1, dtype=np.float64)
    shape = 1.0 / (i * (n - i + 1))
    shape = np.where(shape > 0, shape, 0.0)
    shape /= shape.sum()
    cand.append(shape)

    # 5) Symmetric double‑exponential families
    idx = np.arange(n, dtype=np.float64)
    for gamma in (0.3, 0.7, 1.5, 3.0):
        shape = np.exp(-gamma * idx) + np.exp(-gamma * (n - 1 - idx))
        shape /= shape.sum()
        cand.append(shape)

    # 6) Half‑wave sinusoid
    s = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    s = np.maximum(s, 0.0)
    s /= s.sum()
    cand.append(s)

    # 7) Sparse spikes smoothed with a tiny Gaussian kernel
    sigma_factor = 0.02
    for k in (16, 32, 64):
        a = np.zeros(n, dtype=np.float64)
        pos = np.random.choice(n, size=k, replace=False)
        a[pos] = 1.0 / k
        sigma = max(1.0, sigma_factor * n / math.sqrt(k))
        t = np.arange(-int(3 * sigma), int(3 * sigma) + 1)
        g = np.exp(-0.5 * (t / sigma) ** 2)
        g /= g.sum()
        a = np.convolve(a, g, mode='same')
        a = np.maximum(a, 0.0)
        a /= a.sum()
        cand.append(a)

    return cand


# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max (log‑sum‑exp) surrogate.
    The vector is kept on the simplex via a soft‑max.
    Returns the best vector (sum≈1) seen and its exact objective.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.08
    lam = 0.5          # temperature for smooth‑max
    lam_factor = 1.03

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        # soft‑max → simplex
        mx = logits.max()
        exp_logits = np.exp(logits - mx)
        a = exp_logits / exp_logits.sum()

        # exact objective (sum(a)=1)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam iter={it:06d}] new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0.0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of surrogate (chain‑rule through soft‑max)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # increase temperature towards true max
        lam = min(lam * lam_factor, 1e6)

        # occasional tiny shake when very close to deadline
        if it % 7_000 == 0 and time.time() + 5.0 > deadline:
            logits += (np.random.rand(n) - 0.5) * 5e-3

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
def _additive_subgradient_opt(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 1_000_000,
    eta0: float = 0.5,
) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient descent on the true max‑convolution.
    Updates are additive and projected onto the probability simplex.
    """
    a = np.maximum(a0, 0.0)
    a = a / a.sum()

    best_a = a.copy()
    best_val = 2.0 * n * _full_convolution(a, conv_fft_len).max()
    stall = 0
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c

        if val < best_val:
            best_val = val
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # sub‑gradient of the max (uniform weight on active set)
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # decaying additive step size
        eta = eta0 / np.sqrt(it + 1.0)
        a = _project_simplex(a - eta * grad)

        # random shake if we have been stuck for a while
        if stall > 10_000:
            a = np.maximum(a + 0.02 * (np.random.rand(n) - 0.5), 0.0)
            a = _project_simplex(a)
            stall = 0

        it += 1

    # final exact evaluation of the best vector found
    final_val = 2.0 * n * _full_convolution(best_a, conv_fft_len).max()
    return best_a, final_val


# ----------------------------------------------------------------------
def _lbfgs_func(
    x: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
) -> tuple[float, np.ndarray]:
    """
    Objective + gradient for L‑BFGS‑B.
    The variables are logits; the soft‑max enforces a∈simplex.
    A smooth approximation of max with temperature τ=50 is used.
    """
    # soft‑max (stable)
    mx = np.max(x)
    exp_x = np.exp(x - mx)
    a = exp_x / exp_x.sum()

    # convolution
    conv = _full_convolution(a, conv_fft_len)

    # smooth max (log‑sum‑exp) with temperature τ
    tau = 50.0
    max_c = conv.max()
    shifted = conv - max_c
    exp_vals = np.exp(tau * shifted)
    sum_exp = exp_vals.sum()
    smooth_max = max_c + np.log(sum_exp) / tau

    val = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. a
    w = exp_vals / sum_exp
    grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

    # back‑prop through the soft‑max
    grad_x = a * (grad_a - np.dot(a, grad_a))

    return val, grad_x.astype(np.float64)


def _lbfgs_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iter: int = 200,
) -> tuple[np.ndarray, float]:
    """
    Refine a vector with L‑BFGS‑B on the smooth surrogate.
    Returns the refined vector and its exact objective (true max‑convolution).
    """
    eps = 1e-12
    x0 = np.log(a0 + eps)

    args = (n, conv_fft_len, corr_fft_len)
    res = minimize(
        _lbfgs_func,
        x0,
        args=args,
        method="L-BFGS-B",
        jac=True,
        options={"maxiter": max_iter, "ftol": 1e-12},
    )

    # extract soft‑max probabilities
    mx = np.max(res.x)
    exp_x = np.exp(res.x - mx)
    a_opt = exp_x / exp_x.sum()

    # exact evaluation
    conv_opt = _full_convolution(a_opt, conv_fft_len)
    val_opt = 2.0 * n * conv_opt.max()
    return a_opt, val_opt


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The function obeys the given time budget and returns the best sequence
    (as a Python list) found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin for the caller

    # ------------------------------------------------------------------
    # Base resolution (n = 4096)
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # Warm start from a previous solution, if present
    warm_raw = globals().get("height_sequence_1")
    warm = None
    if warm_raw is not None:
        warm = np.asarray(warm_raw, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None

    # ------------------------------------------------------------------
    # Phase 1 – generate candidates and run a short Adam on each
    candidates = _initial_candidates(n, warm)

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    for cid, cand in enumerate(candidates):
        if time.time() >= deadline:
            break
        a0 = np.maximum(cand.astype(np.float64), 0.0)
        if a0.sum() < 0.01:
            a0 = np.random.dirichlet(np.ones(n))
        else:
            a0 = a0 / a0.sum()

        # give each candidate a modest time slice (≈10 s)
        cand_deadline = min(deadline - 5.0, time.time() + 10.0)
        a_adam, val_adam = _adam_optimize(
            a0, n, conv_fft_len, corr_fft_len, cand_deadline, max_iters=120_000
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[cand {cid}] Adam best {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – longer Adam on the current best (if time permits)
    if best_seq is not None and time.time() < deadline - 10.0:
        sub_deadline = min(deadline - 5.0, time.time() + 25.0)
        a_adam, val_adam = _adam_optimize(
            best_seq, n, conv_fft_len, corr_fft_len, sub_deadline, max_iters=300_000
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Refine Adam] best {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – additive projected sub‑gradient (exact max) for a long run
    if best_seq is not None and time.time() < deadline - 5.0:
        a_sub, val_sub = _additive_subgradient_opt(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=800_000,
            eta0=0.5,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Additive Subgrad] best {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – L‑BFGS‑B refinement on the smooth surrogate
    if best_seq is not None and time.time() < deadline - 5.0:
        a_lb, val_lb = _lbfgs_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iter=200,
        )
        if val_lb < best_val:
            best_val = val_lb
            best_seq = a_lb.copy()
            print(f"[L‑BFGS] best {val_lb:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – upscale to 8192 points if we still have enough time
    if best_seq is not None and time.time() + 30.0 < deadline:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        # linear interpolation to the finer grid
        x_old = np.linspace(0.0, 1.0, best_seq.shape[0], endpoint=False)
        x_new = np.linspace(0.0, 1.0, n_big, endpoint=False)
        a_big = np.interp(x_new, x_old, best_seq)
        a_big = np.maximum(a_big, 0.0)
        a_big /= a_big.sum()

        # short Adam on the larger grid
        big_deadline = min(deadline - 5.0, time.time() + 20.0)
        a_big_adam, val_big_adam = _adam_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            big_deadline,
            max_iters=150_000,
        )
        if val_big_adam < best_val:
            best_val = val_big_adam
            best_seq = a_big_adam.copy()
            print(f"[Upscale Adam] best {val_big_adam:.6f}")

        # additive sub‑gradient on the larger grid
        if time.time() < deadline - 5.0:
            a_big_sub, val_big_sub = _additive_subgradient_opt(
                best_seq,
                n_big,
                conv_fft_len_big,
                corr_fft_len_big,
                deadline,
                max_iters=400_000,
                eta0=0.3,
            )
            if val_big_sub < best_val:
                best_val = val_big_sub
                best_seq = a_big_sub.copy()
                print(f"[Upscale Subgrad] best {val_big_sub:.6f}")

        # final L‑BFGS‑B on the larger grid
        if best_seq is not None and time.time() < deadline - 5.0:
            a_big_lb, val_big_lb = _lbfgs_optimize(
                best_seq,
                n_big,
                conv_fft_len_big,
                corr_fft_len_big,
                deadline,
                max_iter=150,
            )
            if val_big_lb < best_val:
                best_val = val_big_lb
                best_seq = a_big_lb.copy()
                print(f"[Upscale L‑BFGS] best {val_big_lb:.6f}")

    # ------------------------------------------------------------------
    # Fallback – ensure a valid return value
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Clip for the evaluator's safety limits and cast to python list
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
