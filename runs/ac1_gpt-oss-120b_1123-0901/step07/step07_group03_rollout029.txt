"""
Propose a low‑value step‐function for the given evaluation function.

Strategy
--------
1. Work first with a modest size (n_small = 1024) – cheap convolutions allow
   many independent random starts.
2. Generate a diverse pool of candidates:
   * uniform
   * Dirichlet samples (various concentrations)
   * two deterministic orderings of a Dirichlet draw:
        – classic anti‑pair (largest ↔ smallest)
        – alternating‑extremes (largest → leftmost, second largest → rightmost,
          third → next left, …)
3. Keep the K best (K≈5) after a quick evaluation.
4. Refine each of these K candidates with a three‑phase continuous optimiser
   (smooth‑max Adam → exact‑max exponentiated‑gradient → simulated‑annealing
   local moves).  The refinement runs on the small grid and consumes ~60 % of
   the total time budget.
5. Upsample the best refined small‑grid solution to the full size
   (n_big = 4096) by linear interpolation.
6. Perform a final optimisation pass on the full grid (again Adam → sub‑grad →
   local hill‑climbing) using the remaining time.
7. Return the best sequence found (clipped to the evaluator’s admissible range).

The code contains only top‑level functions (no closures) and respects the
1000 s total budget by checking deadlines after each major block.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Cross‑correlation w ★ a needed for sub‑gradient (returns length len(a))."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
# Objective (exact, without safety checks)
def _objective(a: np.ndarray) -> float:
    """Exact evaluation 2·n·max(a*a)/(sum(a)^2).  a is assumed non‑negative."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    n = a.shape[0]
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    return 2.0 * n * conv.max() / (s * s)

def _warm_start(n: int):
    """Return a warm‑start of length n taken from the global variable."""
    warm = globals().get("height_sequence_1")
    if warm is None:
        return None
    warm = np.asarray(warm, dtype=np.float64)
    if warm.size == 0:
        return None
    # Down‑sample / up‑sample via linear interpolation
    x_old = np.linspace(0.0, 1.0, warm.size, endpoint=False)
    x_new = np.linspace(0.0, 1.0, n, endpoint=False)
    warm = np.interp(x_new, x_old, warm)
    warm = np.maximum(warm, 0.0)
    if warm.sum() < 0.01:
        return None
    warm = warm / warm.sum()
    return warm


def _uniform_candidate(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _dirichlet_candidate(n: int, alpha: float) -> np.ndarray:
    """Sample Dirichlet(alpha·1) and normalise to L1 = 1."""
    a = np.random.dirichlet(alpha * np.ones(n))
    return a.astype(np.float64)


def _anti_pair_arrangement(vals: np.ndarray) -> np.ndarray:
    """Largest ↔ smallest (the classic anti‑pair ordering)."""
    n = len(vals)
    sorted_vals = np.sort(vals)[::-1]          # descending
    a = np.empty(n, dtype=np.float64)
    half = n // 2
    a[:half] = sorted_vals[:half]             # biggest half on the left
    a[-half:] = sorted_vals[-half:][::-1]     # smallest half on the right (reversed)
    if n % 2 == 1:
        a[half] = sorted_vals[half]
    return a


def _alternating_extreme_arrangement(vals: np.ndarray) -> np.ndarray:
    """
    Place values alternating from the two extremes:
    largest → leftmost, second largest → rightmost, third → next left, …
    """
    n = len(vals)
    sorted_vals = np.sort(vals)[::-1]          # descending
    a = np.empty(n, dtype=np.float64)
    left, right = 0, n - 1
    for i, v in enumerate(sorted_vals):
        if i % 2 == 0:
            a[left] = v
            left += 1
        else:
            a[right] = v
            right -= 1
    return a


def _anti_pair_candidate(n: int, alpha: float) -> np.ndarray:
    """Dirichlet sample + classic anti‑pair layout."""
    vals = np.random.dirichlet(alpha * np.ones(n))
    return _anti_pair_arrangement(vals)


def _alternating_candidate(n: int, alpha: float) -> np.ndarray:
    """Dirichlet sample + alternating‑extreme layout."""
    vals = np.random.dirichlet(alpha * np.ones(n))
    return _alternating_extreme_arrangement(vals)


def _geometric_candidate(n: int, decay: float = 0.999) -> np.ndarray:
    """Geometric (exponential) decay shape, normalised to L1 = 1."""
    i = np.arange(n, dtype=np.float64)
    shape = decay ** i
    shape = shape / shape.sum()
    return shape


def _beta_shape_candidate(n: int, a: float, b: float) -> np.ndarray:
    """U‑shaped / ∩‑shaped deterministic shape using the Beta(a,b) pdf."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n   # mid‑point rule
    # pdf ∝ x^{a-1} (1-x)^{b-1}
    shape = (x ** (a - 1.0)) * ((1.0 - x) ** (b - 1.0))
    shape = shape / shape.sum()
    return shape

# Optimisation primitives
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    lr: float = 0.07,
    lam_start: float = 0.2,
    lam_factor: float = 1.05,
    max_iters: int = 1_000_000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max surrogate (log‑sum‑exp).
    """
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lam = lam_start

    best_a = a0.copy()
    best_val = _objective(a0)

    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max → simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective (for tracking)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val_exact = 2.0 * n * max_c
        if val_exact < best_val:
            best_val = val_exact
            best_a = a.copy()
            if it % 10_000 == 0:
                print(f"[Adam it={it:06d}] new best {best_val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of surrogate (w.r.t. a)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature schedule – slowly concentrate on the true max
        lam = min(lam * lam_factor, 1e6)

        # small jitter every few thousand iterations to avoid stagnation
        if it % 5000 == 0 and (deadline - time.time()) > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-3

        it += 1

    return best_a, best_val


def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    eta0: float = 0.9,
    max_iters: int = 1_000_000,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient optimisation via exponentiated‑gradient updates.
    """
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s

    best_a = a.copy()
    best_val = _objective(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 10_000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        # sub‑gradient
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # step‑size schedule – decays like 1/√t
        eta = eta0 / np.sqrt(it + 1)

        # exponentiated‑gradient update (maintains positivity)
        a = a * np.exp(-eta * grad_a)

        # project onto simplex
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing hill‑climbing:
    – with 50 % probability swap two positions,
    – otherwise transfer a random fraction (≤40 %) of mass from i → j.
    Acceptance follows a linear cooling schedule.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a)

    start_temp = 0.02
    it = 0
    while time.time() < deadline and it < max_moves:
        # pick move type
        if np.random.rand() < 0.5:            # swap
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            a2 = a.copy()
            a2[i], a2[j] = a2[j], a2[i]
        else:                                 # transfer a fraction
            i = np.random.randint(n)
            if a[i] < 1e-14:
                continue
            j = np.random.randint(n)
            if i == j:
                continue
            delta = a[i] * (np.random.rand() * 0.4)   # up to 40 % of i mass
            a2 = a.copy()
            a2[i] -= delta
            a2[j] += delta

        val2 = _objective(a2)

        # accept if better or with temperature‑dependent probability
        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
        else:
            temp = start_temp * (1.0 - it / max_moves)
            if temp > 0.0 and np.random.rand() < np.exp(-(val2 - best_val) / temp):
                a = a2

        if it % 10_000 == 0:
            print(f"[Local move {it:06d}] best {best_val:.6f}")

        it += 1

    return best_a, best_val

        n_big,
        conv_fft_len_big,
        corr_fft_len_big,
        adam_big_deadline,
        lr=0.07,
        lam_start=0.2,
        lam_factor=1.04,
    )
    if val_adam_big < best_small_val:
        best_small_val, best_big = val_adam_big, a_adam_big.copy()
        print(f"[Big Adam] improved global best {best_small_val:.6f}")

    # ---- Sub‑gradient on the big grid
    sub_big_deadline = min(deadline - 2.0, time.time() + remaining * 0.35)
    a_sub_big, val_sub_big = _subgradient_optimize(
        best_big,
        n_big,
        conv_fft_len_big,
        corr_fft_len_big,
        sub_big_deadline,
        eta0=0.8,
    )
    if val_sub_big < best_small_val:
        best_small_val, best_big = val_sub_big, a_sub_big.copy()
        print(f"[Big Subgrad] improved global best {best_small_val:.6f}")

    # ---- Final local hill‑climbing
    if time.time() < deadline - 1.0:
        local_big_deadline = deadline - 0.5
        a_loc_big, val_loc_big = _local_random_search(
            best_big,
            n_big,
            conv_fft_len_big,
            local_big_deadline,
            max_moves=250_000,
        )
        if val_loc_big < best_small_val:
            best_small_val, best_big = val_loc_big, a_loc_big.copy()
            print(f"[Big Local] improved global best {best_small_val:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣  Final sanity – guarantee a valid, bounded return
    final_seq = best_big.clip(0.0, 1000.0)
    print(f"Final best value {best_small_val:.6f} (length {len(final_seq)})")
    return final_seq.tolist()
