def project_to_simplex(v):
    # ensures sum(v)=1 and v>=0
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a low‑value step function.

The algorithm follows a multi‑stage pipeline that is tuned for the
1000 s time limit:

1. **Diverse warm‑starts** – a pool of candidates is generated, among
   them warm‑starts (if any), uniform, Dirichlet samples, and several
   deterministic “anti‑pair” and “edge‑pair” orderings that are known
   to lower the auto‑convolution peak.

2. **Adam on a smooth‑max surrogate** – the best pool member is refined
   with Adam applied to a log‑sum‑exp approximation of the maximum
   convolution entry.  The temperature is increased exponentially,
   turning the surrogate into the true max.

3. **Projected sub‑gradient descent** – instead of the multiplicative
   update used previously, we employ a Euclidean projected sub‑gradient
   method on the exact objective.  After each gradient step the vector
   is projected onto the probability simplex (non‑negative, sum = 1).

4. **Simulated‑annealing hill climbing** – a cheap stochastic local
   search swaps entries or transfers a fraction of mass between two
   positions, accepting worsening moves with a linearly decreasing
   temperature.

5. **Optional upscale** – if time permits the solution is interpolated
   to a finer grid (8192 points) and refined once more.

All phases respect the global deadline; the best sequence found so
far is always kept and finally returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
# Objective helpers (exact, without the outer safety checks)
def _objective(a: np.ndarray) -> float:
    """Exact evaluation (no clipping) of 2·n·max(a*a)/(sum(a)^2)."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    n = a.shape[0]
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    max_c = conv.max()
    return 2.0 * n * max_c / (s * s)


# ----------------------------------------------------------------------
# Simple projection onto the probability simplex
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x>=0, sum(x)=1} (Duchi et al.)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _warm_start(n: int):
    """Extract a warm start from the global variable if present."""
    warm = globals().get("height_sequence_1")
    if warm is None:
        return None
    warm = np.asarray(warm, dtype=np.float64)
    if warm.shape[0] > n:
        warm = warm[:n]
    elif warm.shape[0] < n:
        warm = np.pad(warm, (0, n - warm.shape[0]))
    warm = np.maximum(warm, 0.0)
    if warm.sum() >= 0.01:
        warm = warm / warm.sum()
        return warm
    return None


def _uniform_candidate(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _random_dirichlet_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    a = np.random.dirichlet(alpha * np.ones(n))
    return a.astype(np.float64)


def _anti_pair_arrangement(vals: np.ndarray) -> np.ndarray:
    """
    Largest opposite smallest, second largest opposite second smallest, …
    """
    n = len(vals)
    sorted_vals = np.sort(vals)[::-1]          # descending
    a = np.empty(n, dtype=np.float64)
    half = n // 2
    a[:half] = sorted_vals[:half]             # biggest half on the left
    a[-half:] = sorted_vals[-half:][::-1]     # smallest half on the right
    if n % 2 == 1:
        a[half] = sorted_vals[half]           # median in the centre
    return a


def _anti_pair_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    """Dirichlet sample + anti‑pair ordering."""
    vals = np.random.dirichlet(alpha * np.ones(n))
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return _anti_pair_arrangement(vals)


def _edge_pair_arrangement(sorted_vals: np.ndarray) -> np.ndarray:
    """
    Place the largest values alternately at the leftmost and rightmost
    positions, making the edges heavy and the centre light.
    """
    n = len(sorted_vals)
    a = np.empty(n, dtype=np.float64)
    left, right = 0, n - 1
    place_left = True
    for v in sorted_vals:
        if place_left:
            a[left] = v
            left += 1
        else:
            a[right] = v
            right -= 1
        place_left = not place_left
    return a


def _edge_pair_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    """Dirichlet sample + edge‑pair arrangement."""
    vals = np.random.dirichlet(alpha * np.ones(n))
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    sorted_vals = np.sort(vals)[::-1]          # descending
    return _edge_pair_arrangement(sorted_vals)


def _tent_candidate(n: int) -> np.ndarray:
    """Tent shape (large at the ends, small in the centre)."""
    x = np.arange(n, dtype=np.float64)
    shape = np.minimum(x + 1, n - x)          # 1,2,…,⌈n/2⌉,…,2,1
    shape = shape / shape.sum()
    return shape


def _gaussian_edges_candidate(n: int, gamma: float = 0.5) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
    shape = shape / shape.sum()
    return shape

# Optimisation primitives
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    lr: float = 0.07,
    lam_start: float = 1.0,
    lam_factor: float = 1.07,
) -> tuple[np.ndarray, float]:
    """Adam on the log‑sum‑exp surrogate of max(conv)."""
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lam = lam_start

    best_a = a0.copy()
    best_val = _objective(a0)

    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max → simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective (sum = 1)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val_exact = 2.0 * n * max_c
        if val_exact < best_val:
            best_val = val_exact
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best exact {best_val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of surrogate w.r.t. a
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam step on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature schedule → true max
        lam = min(lam * lam_factor, 1e6)

        # occasional jitter to escape stagnation
        if it % 5_000 == 0 and (deadline - time.time()) > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2

        it += 1

    return best_a, best_val


def _subgrad_optimize_proj(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 0.7,
) -> tuple[np.ndarray, float]:
    """
    Euclidean projected sub‑gradient descent on the exact objective.
    """
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s

    best_a = a.copy()
    best_val = _objective(a)

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5_000 == 0:
                print(f"[ProjSubgrad it={it:06d}] new best {val:.6f}")

        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-12)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()

        # sub‑gradient of max(conv) w.r.t. a
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        # Euclidean step + projection onto simplex
        a = _project_to_simplex(a - eta * grad_a)

    return best_a, best_val


def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 120_000,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing style hill‑climbing:
    – swap two entries OR transfer a fraction of mass.
    – accept if improvement or with probability exp(-Δ/Temp).
    Temperature decays linearly from 0.01 to 0.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a)

    start_temp = 0.01
    it = 0
    while time.time() < deadline and it < max_moves:
        # choose move type
        if np.random.rand() < 0.5:          # swap
            i, j = np.random.randint(n, size=2)
            if i == j:
                it += 1
                continue
            a2 = a.copy()
            a2[i], a2[j] = a2[j], a2[i]
        else:                               # transfer a fraction from i to j
            i = np.random.randint(n)
            if a[i] < 1e-14:
                it += 1
                continue
            j = np.random.randint(n)
            if i == j:
                it += 1
                continue
            delta = a[i] * (np.random.rand() * 0.4)   # up to 40 % move
            a2 = a.copy()
            a2[i] -= delta
            a2[j] += delta

        val2 = _objective(a2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
        else:
            temp = start_temp * (1.0 - (it / max_moves))
            if temp > 0.0 and np.random.rand() < np.exp(-(val2 - best_val) / temp):
                a = a2

        if it % 5_000 == 0:
            print(f"[Local move {it:06d}] best {best_val:.6f}")

        it += 1

    return best_a, best_val

    # Gaussian‑edge shapes

        cand = cand / s                    # normalise to sum = 1
        val = _objective(cand)

        if val < best_val:

        best_seq,

        conv_fft_len,
        corr_fft_len,
        adam_deadline,
        max_iters=300_000,
        lr=0.07,
        lam_start=1.0,
        lam_factor=1.07,
    )
    if val_adam < best_val:
        best_val = val_adam
        best_seq = a_adam.copy()
        print(f"[Adam] improved to {val_adam:.6f}")

    # ------------------------------------------------------------------
    # 4) Projected sub‑gradient descent on the exact objective
    subgrad_deadline = min(deadline - 5.0, time.time() + 300.0)
    a_sub, val_sub = _subgrad_optimize_proj(
        best_seq,
        n,
        conv_fft_len,
        corr_fft_len,
        subgrad_deadline,
        max_iters=300_000,
        eta0=0.7,
    )
    if val_sub < best_val:
        best_val = val_sub
        best_seq = a_sub.copy()
        print(f"[ProjSubgrad] improved to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # 5) Local stochastic hill‑climbing (annealing)
    if best_seq is not None and time.time() < deadline - 5.0:
        local_deadline = deadline - 2.0
        a_loc, val_loc = _local_random_search(
            best_seq,
            n,
            conv_fft_len,
            local_deadline,
            max_moves=200_000,
        )
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()
            print(f"[Local] improved to {val_loc:.6f}")

    # ------------------------------------------------------------------
    # 6) Optional upscale to a finer grid (8192) if time permits
    if time.time() < final_upscale_deadline:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        # linear interpolation of the current best solution
        x_old = np.linspace(0.0, 1.0, n, endpoint=False)
        x_new = np.linspace(0.0, 1.0, n_big, endpoint=False)
        a_big = np.interp(x_new, x_old, best_seq)

        a_big = np.maximum(a_big, 0.0)
        a_big = a_big / a_big.sum()

        # brief Adam on larger grid
        ad_big_deadline = min(deadline - 5.0, time.time() + 180.0)
        a_big_adam, val_big_adam = _adam_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            ad_big_deadline,
            max_iters=200_000,
            lr=0.07,
            lam_start=1.0,
            lam_factor=1.07,
        )
        # projected sub‑gradient refinement on large grid
        sub_big_deadline = min(deadline, time.time() + 200.0)
        a_big_sub, val_big_sub = _subgrad_optimize_proj(
            a_big_adam,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            sub_big_deadline,
            max_iters=200_000,
            eta0=0.7,
        )
        if val_big_sub < best_val:
            best_val = val_big_sub
            best_seq = a_big_sub.copy()
            print(f"[Big] improved to {val_big_sub:.6f}")

    # ------------------------------------------------------------------
    # 7) Safeguard – guarantee a valid return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the range accepted by the evaluator
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
