def _adam_opt(log_a, n, fft_len, beta_init, beta_rate, eta, deadline):
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)
    t = 0
    best_val = float('inf')
    best_a = None

    while time.time() < deadline:
        t += 1
        a = np.exp(log_a)
        a_sum = a.sum()
        if a_sum == 0:
            a = np.random.dirichlet(np.ones(n))
            a_sum = a.sum()
        a /= a_sum

        conv = full_convolution_fft(a, fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c  # sum is 1
        if val < best_val - 1e-12:
            best_val = val
            best_a = a.copy()
            print(f"[Adam iter={t}] new best {val:.6f}")

        # compute weighted gradient
        beta = beta_init * (beta_rate ** t)
        # To avoid overflow compute w via exp(beta*(conv-max_c))
        # Doing stable softmax:
        shifted = conv - conv.max()
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.zeros_like(conv)
            w[0] = 1.0
        else:
            w /= w_sum

        grad_a = 2.0 * cross_correlation_fft(w, a, fft_len)  # same length as a

        # compute gradient in log-space for softmax a = softmax(log_a)
        # Actually we have a already normalized via exp and division; 
        # The gradient of objective w.r.t log_a is a * (grad_a - dot(grad_a, a))
        dot = np.dot(grad_a, a)
        grad_log = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_log
        v = beta2 * v + (1 - beta2) * (grad_log ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        log_a -= step

        # Optionally, re-center log_a to avoid overflow: subtract max
        log_a -= np.max(log_a)

        # If no improvement for a while, add random perturbation
        if t % 5000 == 0:
            # optionally small jitter
            jitter = np.random.normal(scale=0.01, size=n)
            log_a += jitter

    return best_a, best_val

"""
Propose a low‑objective step‑function using a multi‑start
Adam‑based mirror‑descent on a soft‑max approximation of the
convolution maximum.

Key ideas
---------
* Work in the log‑probability (log‑softmax) domain so updates stay
  on the simplex without explicit projection.
* Approximate  max(a∗a)  by  softmax_β(a∗a)  with a gradually
  increasing temperature β.  This yields a smooth gradient that
  converges to a true sub‑gradient when β becomes large.
* Adam’s adaptive moments give a more robust step‑size than the
  plain exponentiated‑gradient used previously.
* Start from a diverse pool of parametric seeds (uniform,
  Dirichlet, edge‑exponential, sinusoid, warm‑start) and keep the
  best few for a long‑run optimisation.
* If time permits, interpolate the current best vector onto a
  finer grid (n=8192) and perform a final refinement.
* All numbers are clipped to the range accepted by ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility FFT helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a (length 2·len(a)-1) using FFT.
    Returns a real array of length 2·len(a)-1.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n-1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """
    Fast evaluation of the exact objective for a (assumed normalised).
    """
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    # sum(a) == 1 after normalisation, so denominator is 1.
    return float(2.0 * n * max_c)


# ----------------------------------------------------------------------
# Candidate generation (light‑weight version of the previous pool)
# ----------------------------------------------------------------------
def _generate_parametric_candidates(
    n: int,
    warm_start: np.ndarray | None,
) -> list[np.ndarray]:
    """Create a modest yet diverse set of initial vectors."""
    candidates = []

    # warm start and a few cyclic shifts
    if warm_start is not None:
        candidates.append(warm_start.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            candidates.append(np.roll(warm_start, shift))

    # uniform baseline
    candidates.append(np.full(n, 1.0 / n))

    # random Dirichlet samples with varied concentration
    for conc in (0.02, 0.1, 0.5, 1.0, 2.0):
        cand = np.random.dirichlet(np.full(n, conc))
        candidates.append(cand)

    i = np.arange(n, dtype=np.float64)

    # edge‑exponential families
    for alpha in np.logspace(-2, 1.0, num=12):  # 0.01 … 10
        edge = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
        edge /= edge.sum()
        for w in (0.0, 0.5, 1.0):
            shape = w * edge + (1.0 - w) / n
            candidates.append(shape)

    # half‑wave sinusoid (positive part)
    sinusoid = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    sinusoid = np.maximum(sinusoid, 0.0)
    sinusoid /= sinusoid.sum()
    candidates.append(sinusoid)

    return candidates


# ----------------------------------------------------------------------
# Adam optimiser on the soft‑max surrogate of ``max(a*a)``.
# ----------------------------------------------------------------------
def _adam_opt(
    log_a0: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray | None, float]:
    """
    Adam optimisation in the log‑softmax space.
    Returns the best vector/value observed before the deadline.
    """
    # Adam hyper‑parameters
    eta = 0.25          # base learning rate
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # temperature schedule for the soft‑max (β ↑)
    beta0 = 0.5
    beta_factor = 1.001

    log_a = log_a0.copy()
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)

    t = 0
    last_improve = time.time()

    while time.time() < deadline:
        t += 1

        # ---- convert logits to a probability vector (simplex) ----
        a = np.exp(log_a)
        s = a.sum()
        if s == 0 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        # ---- current objective ----
        conv = _full_convolution(a, fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c       # sum(a) == 1
        if val < best_val - 1e-12:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Adam iter={t:07d}] new global best {val:.6f}")

        # ---- soft‑max weighting of the convolution entries ----
        beta = beta0 * (beta_factor ** t)   # gradually increase β
        shifted = conv - conv.max()         # for numerical stability
        exp_vals = np.exp(beta * shifted)
        w_sum = exp_vals.sum()
        if w_sum > 0.0:
            w = exp_vals / w_sum
        else:
            # fallback to a uniform distribution over the active set
            active = np.isclose(conv, conv.max(), atol=1e-12)
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()

        # ---- gradient of the smoothed max w.r.t. a ----
        grad_a = 2.0 * _cross_correlation(w, a, fft_len)

        # ---- transform gradient to the log‑softmax domain ----
        dot = np.dot(grad_a, a)                 # <grad_a , a>
        grad_log = a * (grad_a - dot)           # ∂F/∂log_a

        # ---- Adam moment updates ----
        m = beta1 * m + (1 - beta1) * grad_log
        v = beta2 * v + (1 - beta2) * (grad_log ** 2)

        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        step = eta * m_hat / (np.sqrt(v_hat) + eps)

        # ---- apply update in log‑space ----
        log_a = log_a - step

        # ---- recenter to avoid overflow/underflow ----
        log_a = log_a - np.max(log_a)

        # ---- occasional random jitter if we stagnate ----
        if time.time() - last_improve > 5.0:
            jitter = (np.random.rand(n) - 0.5) * 0.02
            log_a = log_a + jitter
            last_improve = time.time()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point used by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function (probability vector) minimizing

        F(a) = 2·n·max(a∗a) / (sum(a)²)

    The routine respects the 1000 s time budget, prints progress,
    and returns the best sequence (plain Python list) found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Problem size – start with a medium grid (n=4096) for speed.
    # ------------------------------------------------------------------
    n = 4096
    fft_len = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Warm‑start from a previously stored solution if available.
    # ------------------------------------------------------------------
    warm_start = globals().get("height_sequence_1")
    if warm_start is not None:
        ws = np.asarray(warm_start, dtype=np.float64)
        if ws.shape[0] > n:
            ws = ws[:n]
        elif ws.shape[0] < n:
            ws = np.pad(ws, (0, n - ws.shape[0]))
        ws = np.maximum(ws, 0.0)
        if ws.sum() >= 0.01:
            warm_start = ws / ws.sum()
        else:
            warm_start = None
    else:
        warm_start = None

    # ------------------------------------------------------------------
    # Stage 1 – generate a modest but diverse pool of seeds.
    # ------------------------------------------------------------------
    candidates = _generate_parametric_candidates(n, warm_start)

    # Add a handful of pure Dirichlet samples for extra diversity.
    for _ in range(20):
        candidates.append(np.random.dirichlet(np.ones(n)))

    # ------------------------------------------------------------------
    # Quick evaluation of all candidates (FFT‑based, inexpensive).
    # ------------------------------------------------------------------
    cand_vals = []
    for cand in candidates:
        if time.time() >= deadline:
            break
        cand = np.maximum(cand, 0.0)
        if cand.sum() < 0.01:
            cand = np.random.dirichlet(np.ones(n))
        cand = cand / cand.sum()
        val = _objective(cand, n, fft_len)
        cand_vals.append((val, cand))

    if not cand_vals:
        # Fallback – uniform vector
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        return best_seq.tolist()

    # Keep the few best start points.
    cand_vals.sort(key=lambda x: x[0])
    top_candidates = [c for _, c in cand_vals[:5]]

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Stage 2 – deep optimisation from the selected starts.
    # ------------------------------------------------------------------
    for idx, start_vec in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        # initialise log‑probabilities (add tiny epsilon to avoid log(0))
        log_a0 = np.log(start_vec + 1e-12)
        print(f"--- Optimising start #{idx + 1}/{len(top_candidates)}")
        best_seq, best_val = _adam_opt(
            log_a0,
            n,
            fft_len,
            deadline,
            best_seq,
            best_val,
        )

    # ------------------------------------------------------------------
    # Stage 3 – if time permits, upscale to n=8192 and refine again.
    # ------------------------------------------------------------------
    if best_seq is not None and (deadline - time.time()) > 30.0:
        n_big = 8192
        fft_len_big = _next_pow2(2 * n_big - 1)

        # Linear interpolation onto the finer grid
        x_old = np.linspace(0, 1, n, endpoint=False)
        x_big = np.linspace(0, 1, n_big, endpoint=False)
        a_big = np.interp(x_big, x_old, best_seq)
        a_big = np.maximum(a_big, 0.0)
        a_big /= a_big.sum()

        log_a_big = np.log(a_big + 1e-12)

        print("--- Refinement on larger grid (n=8192)")
        best_seq, best_val = _adam_opt(
            log_a_big,
            n_big,
            fft_len_big,
            deadline,
            best_seq,
            best_val,
        )
        if best_seq is not None:
            n = n_big

    # ------------------------------------------------------------------
    # Final sanity clip and report.
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Verify with the provided evaluator (optional, safe wrapper)
    try:
        verified = evaluate_sequence(best_seq.tolist())
        print(f"*** Verified objective via evaluate_sequence: {verified:.6f}")
    except Exception as exc:  # pragma: no cover
        print(f"Verification failed: {exc}")

    print(f"*** Finished search: best objective ≈ {best_val:.6f} (n={best_seq.shape[0]})")
    return best_seq.tolist()
