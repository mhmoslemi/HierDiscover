def _smooth_subgradient_opt(a0, n, conv_fft_len, corr_fft_len, deadline, best_seq, best_val):
    a = np.clip(a0, 1e-12, None)
    a = a / a.sum()
    eta0 = 0.5
    decay = 0.9999
    epsilon0 = 0.01
    epsilon_decay = 0.9999
    min_eta = 0.001
    eps_min = 1e-6
    it = 0
    last_improve = time.time()
    while time.time() < deadline:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val - 1e-12:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[n={n} Smooth iter={it}] new best {val}")
        # smoothing weight
        epsilon = max(eps_min, epsilon0 * (epsilon_decay ** it))
        w = np.exp((conv - max_c) / epsilon)
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.ones_like(conv) / conv.size
        else:
            w = w / w_sum
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)
        eta = max(min_eta, eta0 * (decay ** it))
        a = a * np.exp(-eta * grad)
        a = a / a.sum()
        if time.time() - last_improve > 5.0:
            # small random perturbation
            perturb = (np.random.rand(n) - 0.5) * 0.02
            a = a * np.exp(perturb)
            a = a / a.sum()
            last_improve = time.time()
        it += 1
    return best_seq, best_val

"""
Search for a non‑negative step function minimizing the evaluation metric.

The algorithm is a refined mirror‑descent (exponentiated‑gradient) optimiser.
Instead of using a hard active‑set (weights only on the maximal convolution
entries) we employ a *soft‑max* weighting with a temperature that is
annealed during the run.  This yields a smooth gradient even when the
maximum is attained at many positions and helps to escape plateaus.

Workflow
---------

1. **Seed generation** – a diverse pool of parametric families (uniform,
   Dirichlet, edge‑exponential, power‑law, double‑Gaussian, sinusoid) and
   a warm‑start (if ``height_sequence_1`` exists) is created.

2. **Quick screening** – each seed is normalised (sum = 1) and its true
   objective (computed via FFT) is evaluated; the best few are kept.

3. **Smooth mirror‑descent** – for every retained seed we run a long
   optimisation loop.
   * The current vector is stored in log‑space (`log_a`) to guarantee
     positivity.  At each iteration we
       - reconstruct `a` from `log_a`,
       - compute the linear convolution `c = a * a`,
       - form a soft‑max weight `w_i ∝ exp((c_i‑max)/ε)`,
       - obtain the gradient `∇_a = 2·corr(w, a)`,
       - translate it to the log‑space gradient `∇_log = ∇_a * a`,
       - take an exponentiated‑gradient step `log_a ← log_a – η·∇_log`,
       - re‑normalise implicitly by the usual log‑sum‑exp trick.
   * The step‑size `η` decays slowly, the temperature `ε` is annealed from
     0.1 down to `1e‑6`.  If no improvement is seen for several seconds a
     small random jitter is added in log‑space to re‑explore the neighbourhood.

4. **Refinement on a finer grid** – if time permits the best solution is
   interpolated to `n = 8192` and refined again with the same optimiser.

All values are finally clipped to the interval `[0, 1000]` (the evaluator’s
acceptable range).  The best sequence found is stored in the global variable
``height_sequence_1`` for possible warm‑starts in subsequent runs.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper FFT utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """
    Fast evaluation of the objective for a (assumed normalised, sum≈1).
    Returned value is exactly the one used by ``evaluate_sequence`` (ignoring
    clipping of tiny values that never occur here).
    """
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c / (a.sum() ** 2))

def _generate_parametric_candidates(
    n: int,
    warm_start: np.ndarray | None,
) -> list[np.ndarray]:
    """
    Produce a diverse set of candidate vectors using several families:
    uniform, Dirichlet, edge‑exponential, edge‑power‑law, double‑Gaussian,
    half‑wave sinusoid.  All candidates are normalised to sum≈1.
    """
    candidates = []

    # warm‑start and its rotations
    if warm_start is not None:
        candidates.append(warm_start.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            candidates.append(np.roll(warm_start, shift))

    # uniform baseline
    candidates.append(np.full(n, 1.0 / n))

    # random Dirichlet with a few concentration parameters
    for conc in (0.01, 0.05, 0.1, 0.3, 0.5, 1.0, 2.0):
        cand = np.random.dirichlet(np.full(n, conc))
        candidates.append(cand)

    i = np.arange(n, dtype=np.float64)

    # edge‑exponential (plus optional uniform mixture)
    for alpha in np.logspace(-2, 1.0, num=16):  # 0.01 … 10
        edge = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
        edge /= edge.sum()
        for w in (0.0, 0.25, 0.5, 0.75, 1.0):
            shape = w * edge + (1.0 - w) / n
            candidates.append(shape)

    # edge‑power‑law (i+1)^(-p) + (n-i)^(-p)
    for p in np.linspace(0.2, 5.0, num=16):
        edge = (i + 1.0) ** (-p) + (n - i) ** (-p)
        edge /= edge.sum()
        for w in (0.0, 0.33, 0.66, 1.0):
            shape = w * edge + (1.0 - w) / n
            candidates.append(shape)

    # double‑Gaussian edges
    for gamma in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        shape /= shape.sum()
        candidates.append(shape)

    # positive half‑wave sinusoid
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    candidates.append(shape)

    return candidates


# ----------------------------------------------------------------------
# Smoothed mirror‑descent optimiser
# ----------------------------------------------------------------------


def _smooth_mirror_descent_opt(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray | None, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on a *smooth* approximation of
    the max‑convolution.  The temperature ε is annealed from 0.1 down to 1e‑6.
    Returns the best vector and its objective observed during the run.
    """
    # initialise log‑probability vector (simplex)
    eps = 1e-15
    log_a = np.log(np.maximum(a0, eps))

    # hyper‑parameters
    eta0 = 0.8          # initial step size in log‑space
    eta_min = 0.001
    eta_decay = 0.99995

    eps0 = 0.1          # initial smoothing temperature
    eps_min = 1e-6
    eps_decay = 0.9999

    it = 0
    last_improve = time.time()

    while time.time() < deadline:
        # ----- reconstruct a from log_a (log‑sum‑exp ensures simplex) -----
        max_log = np.max(log_a)
        log_shift = log_a - max_log
        a = np.exp(log_shift)
        a_sum = np.sum(a)
        a = a / a_sum

        # ----- convolution & objective -----
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c  # sum(a) == 1, so denominator = 1

        if val < best_val - 1e-12:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[n={n} SmoothMD iter={it:07d}] new best {val:.6f}")

        # ----- soft‑max weight over convolution entries -----
        epsilon = max(eps_min, eps0 * (eps_decay ** it))
        # shift by max_c to keep values ≤ 1
        w = np.exp((conv - max_c) / epsilon)
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w = w / w_sum

        # ----- gradient w.r.t. a (still on the simplex) -----
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # ----- translate to gradient w.r.t. log_a (mirror descent) -----
        grad_log = grad_a * a

        # ----- step‑size -----
        eta = max(eta_min, eta0 * (eta_decay ** it))

        # ----- update log_a -----
        log_a = log_a - eta * grad_log

        # ----- occasional random perturbation to avoid stagnation -----
        if time.time() - last_improve > 5.0:
            perturb = (np.random.rand(n) - 0.5) * 0.01   # small log‑space jitter
            log_a = log_a + perturb
            last_improve = time.time()

        it += 1

    return best_seq, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a low‑evaluation‑score step function.
    Returns the best sequence found (plain Python list) before the time budget
    expires.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # Primary grid size – 4096 provides a good trade‑off.
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm start from a previously stored solution, if present.
    warm_start = globals().get("height_sequence_1")
    if warm_start is not None:
        ws = np.asarray(warm_start, dtype=np.float64)
        if ws.shape[0] > n:
            ws = ws[:n]
        elif ws.shape[0] < n:
            ws = np.pad(ws, (0, n - ws.shape[0]))
        ws = np.maximum(ws, 0.0)
        if ws.sum() >= 0.01:
            ws = ws / ws.sum()
        else:
            ws = None
        warm_start = ws
    else:
        warm_start = None

    # ------------------------------------------------------------------
    # Stage 1 – generate a wide set of seeds
    candidates = _generate_parametric_candidates(n, warm_start)

    # enrich with a few extra Dirichlet samples
    for conc in (0.005, 0.02, 0.07, 0.15, 0.4, 0.8):
        for _ in range(20):
            cand = np.random.dirichlet(np.full(n, conc))
            candidates.append(cand)

    # ------------------------------------------------------------------
    # Fast screening of all candidates (keep the best few)
    cand_vals = []
    for idx, cand in enumerate(candidates):
        if time.time() >= deadline:
            break
        # Normalise, protect against pathological sums
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 0.01:
            cand = np.random.dirichlet(np.ones(n))
            s = cand.sum()
        cand = cand / s
        val = _objective(cand, n, conv_fft_len)
        cand_vals.append((val, cand))

    if not cand_vals:
        # fallback – uniform vector
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        print("*** No candidates evaluated – returning uniform vector.")
        return best_seq.tolist()

    cand_vals.sort(key=lambda x: x[0])
    top_k = 5
    top_candidates = [c for _, c in cand_vals[:top_k]]

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Stage 2 – deep optimisation of the selected starts
    for start_id, seed_vec in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        init_val = _objective(seed_vec, n, conv_fft_len)
        print(
            f"--- Optimising start #{start_id + 1}/{len(top_candidates)} "
            f"(init val = {init_val:.6f})"
        )
        best_seq, best_val = _smooth_mirror_descent_opt(
            seed_vec,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            best_seq,
            best_val,
        )

    # ------------------------------------------------------------------
    # Stage 3 – optional refinement on a finer grid (n = 8192)
    time_left = deadline - time.time()
    if time_left > 30.0 and best_seq is not None:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        # linear interpolation of the best vector onto the finer grid
        x_old = np.linspace(0.0, 1.0, n, endpoint=False)
        x_big = np.linspace(0.0, 1.0, n_big, endpoint=False)
        a_big = np.interp(x_big, x_old, best_seq)
        a_big = np.maximum(a_big, 0.0)
        a_big = a_big / a_big.sum()

        print("--- Refinement on larger grid (n=8192)")
        best_seq, best_val = _smooth_mirror_descent_opt(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            best_seq,
            best_val,
        )
        if best_seq is not None and best_seq.shape[0] == n_big:
            n = n_big

    # ------------------------------------------------------------------
    # Final safety net – return uniform if nothing succeeded
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to evaluator’s accepted range and store for possible warm‑starts
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    globals()["height_sequence_1"] = best_seq.tolist()

    # One final evaluation with the provided evaluator (optional, cheap)
    try:
        final_obj = evaluate_sequence(best_seq.tolist())
        print(f"*** Finished search: best objective ≈ {final_obj:.6f} (n={best_seq.shape[0]})")
    except Exception:
        # If the evaluator is unavailable (unlikely), fall back to our own value
        print(f"*** Finished search: best objective ≈ {best_val:.6f} (n={best_seq.shape[0]})")

    return best_seq.tolist()
