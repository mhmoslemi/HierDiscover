def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5

    # set n
    n = 8192
    conv_fft_len = _next_pow2(2*n -1)
    corr_fft_len = _next_pow2(3*n -2)

    # generate initial candidates
    candidates = []

    # Warm-start
    warm_start = globals().get('height_sequence_1')
    if warm_start is not None:
        ws = np.asarray(warm_start, dtype=np.float64)
        if ws.shape[0] > n:
            ws = ws[:n]
        elif ws.shape[0] < n:
            ws = np.pad(ws, (0, n - ws.shape[0]))
        ws = np.clip(ws, 0, None)
        if ws.sum() > 0.01:
            ws = ws / ws.sum()
            candidates.append(ws)

    # Uniform baseline
    candidates.append(np.full(n, 1.0/n))

    # param families
    i = np.arange(n, dtype=np.float64)

    # Symmetric beta shapes
    for alpha in np.arange(0.1, 3.01, 0.1):
        shape = (i+1)**(alpha-1) * (n-i)**(alpha-1)
        shape_sum = shape.sum()
        if shape_sum > 0:
            cand = shape / shape_sum
            candidates.append(cand)

    # Edge exponential with mixture
    alphas = np.logspace(-2, 1.0, 30)  # 0.01..10
    for a in alphas:
        edge = np.exp(-a*i) + np.exp(-a*(n-1-i))
        edge_sum = edge.sum()
        edge = edge / edge_sum
        for w in np.linspace(0, 1, 21):
            cand = w * edge + (1-w) * (1.0/n)
            # ensure non-negative
            cand = np.maximum(cand, 0.0)
            cand = cand / cand.sum()
            candidates.append(cand)

    # Edge power law with mixture
    ps = np.linspace(0.2, 5.0, 30)
    for p in ps:
        edge = (i+1)**(-p) + (n-i)**(-p)
        edge = edge / edge.sum()
        for w in np.linspace(0, 1, 21):
            cand = w * edge + (1-w) * (1.0/n)
            cand = np.maximum(cand, 0.0)
            cand = cand / cand.sum()
            candidates.append(cand)

    # Two-Gaussian edge
    for gamma in np.logspace(-2, 1.0, 20):
        edge = np.exp(-gamma * (i**2)) + np.exp(-gamma * ((n-1-i)**2))
        edge = edge / edge.sum()
        for w in np.linspace(0, 1, 11):
            cand = w * edge + (1-w) * (1.0/n)
            cand = cand / cand.sum()
            candidates.append(cand)

    # arcsine shape combination
    arcs = 1.0 / np.sqrt((i+0.5)*(n-i-0.5))
    arcs = arcs / arcs.sum()
    for w in np.linspace(0, 1, 21):
        cand = w * arcs + (1-w) * (1.0/n)
        cand = cand / cand.sum()
        candidates.append(cand)

    # Random Dirichlet draws
    for conc in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        cand = np.random.dirichlet(np.full(n, conc))
        candidates.append(cand)

    # Evaluate all candidates and keep top
    evaluated = []
    for cand in candidates:
        if time.time() >= deadline:
            break
        # ensure positive and normalized
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 0.01:
            # reinitialize
            cand = np.random.dirichlet(np.ones(n))
            s = cand.sum()
        cand = cand / s
        val = _objective(cand, n, conv_fft_len)
        evaluated.append((val, cand))
    # Keep top 20
    evaluated.sort(key=lambda x: x[0])
    top_k = min(20, len(evaluated))
    top_candidates = [c for _, c in evaluated[:top_k]]
    print(f"Initial candidate sweep: best {evaluated[0][0]:.6f}, {top_k} kept.")

    # Global best
    best_seq = None
    best_val = float('inf')
    # Optimization per candidate
    for idx, init in enumerate(top_candidates):
        if time.time() >= deadline - 1.0:
            break
        print(f"Optimizing candidate {idx+1}/{top_k}")
        # Run Adam + subgradient
        seq, val = _deep_optimize(init, n, conv_fft_len, corr_fft_len, deadline, best_seq, best_val)
        if val < best_val:
            best_val = val
            best_seq = seq
            print(f"New global best {best_val:.6f}")

    # If time left, random hill climbing
    if best_seq is not None and time.time() < deadline - 5.0:
        a = best_seq.copy()
        # optional symmetrize
        a = (a + a[::-1]) / 2
        best_local_val = best_val
        it = 0
        max_iter = 200000  # many attempts
        while time.time() < deadline - 2.0 and it < max_iter:
            i_idx, j_idx = np.random.randint(n, size=2)
            if i_idx == j_idx:
                continue
            # small transfer
            total = a[i_idx] + a[j_idx]
            t = np.random.rand()
            new_i = t * total
            new_j = total - new_i
            # To enforce small changes, limit deviation:
            if abs(new_i - a[i_idx]) > 0.1 * a[i_idx] or abs(new_j - a[j_idx]) > 0.1 * a[j_idx]:
                # limit to within 10% change
                continue
            a_new = a.copy()
            a_new[i_idx] = new_i
            a_new[j_idx] = new_j
            # symmetrize
            a_new = (a_new + a_new[::-1]) / 2
            conv = _full_convolution(a_new, conv_fft_len)
            max_c = conv.max()
            val = 2.0 * n * max_c / (a_new.sum() ** 2)
            if val < best_local_val:
                best_local_val = val
                a = a_new
                best_seq = a_new.copy()
                # Also update best_val
                best_val = val
                # optionally print
                if it % 1000 == 0:
                    print(f"Hill climb iter {it}: val {val:.6f}")
            it += 1

    if best_seq is None:
        best_seq = np.full(n, 1.0/n)

    best_seq = np.clip(best_seq, 0.0, 1000.0)
    print(f"Search finished, best value ~{best_val:.6f}")
    return best_seq.tolist()

"""
Propose a low‑evaluation step‑function.

Algorithm
---------
1. **Dense parametric sweep** –  on a moderate grid (n0 = 4096) we generate
   many symmetric families (edge‑exponential, edge‑power‑law, symmetric
   beta‑shapes, arcsine) on a fine grid of their parameters.  Each candidate
   is evaluated with a fast FFT‑based convolution and the best few
   (≈20) are retained.

2. **Two‑stage local optimisation** – for every retained start we run
   (a) Adam on a log‑sum‑exp (smooth‑max) surrogate of the max‑convolution
   and (b) an exponentiated‑gradient sub‑gradient refinement on the true
   max‑convolution.  All updates are performed on the probability simplex
   and the vector is forced to stay symmetric (a = (a+a[::-1])/2).
   The optimizer stops early if the global deadline is near.

3. **Upscaling** – if time permits the current best vector is linearly
   interpolated to a finer grid (n = 8192) and the whole Stage‑2 optimisation
   is repeated, giving the algorithm a chance to exploit the higher resolution.

4. **Intensive random hill‑climb** – a final pure local search performs many
   tiny mass‑transfer moves (preserving the total sum) and accepts any move
   that reduces the true objective.  The symmetry is re‑enforced after each
   change.

The function returns the best sequence found (as a plain Python list) and
ensures all entries are non‑negative and ≤1000, which satisfies the
``evaluate_sequence`` security checks.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2·len(a)-1) using a real FFT of length
    ``fft_len`` (must be ≥ 2·len(a)-1).
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """
    Fast evaluation of the true objective (sum(a)≈1).
    Returns 2·n·max(a*a) / (sum(a)**2).
    """
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c / (a.sum() ** 2))


# ----------------------------------------------------------------------
def _adam_smooth_max(a0: np.ndarray,
                     n: int,
                     conv_fft_len: int,
                     corr_fft_len: int,
                     deadline: float,
                     global_best_seq: np.ndarray | None,
                     global_best_val: float) -> tuple[np.ndarray, float]:
    """
    Adam on a log‑sum‑exp surrogate of the max‑convolution.
    The iterate lives in the simplex via a soft‑max parametrisation.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)               # start in log‑softmax space
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.07
    lam = 0.3                # initial temperature
    lam_factor = 1.02
    lam_cap = 1e6

    best_seq = global_best_seq
    best_val = global_best_val
    last_improve = time.time()
    it = 0
    max_iter = 150_000

    while time.time() < deadline and it < max_iter:
        # --- soft‑max (simplex) ---
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()
        # enforce symmetry and renormalise
        a = (a + a[::-1]) * 0.5
        a /= a.sum()

        # --- surrogate objective (smooth max) ---
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[{n}] Adam it={it:06d} new best {val:.6f}")

        # ---- smooth‑max weights (log‑sum‑exp) ----
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---- gradient of surrogate (w.r.t. a) ----
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)                 # ⟨a,∇⟩ term
        grad_log = a * (grad_a - avg)           # chain rule to logits

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---- temperature schedule ----
        lam = min(lam * lam_factor, lam_cap)

        # ---- occasional jitter if stuck ----
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1

    return best_seq, best_val


def _subgradient_refinement(a0: np.ndarray,
                            n: int,
                            conv_fft_len: int,
                            corr_fft_len: int,
                            deadline: float,
                            init_best_seq: np.ndarray | None,
                            init_best_val: float) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient descent on the true max‑convolution.
    """
    a = init_best_seq if init_best_seq is not None else a0.copy()
    a = np.maximum(a, 0.0)
    a /= a.sum()

    best_seq = a.copy()
    best_val = init_best_val
    eta0 = 0.6
    it = 0
    max_iter = 40_000
    last_improve = time.time()

    while time.time() < deadline and it < max_iter:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[{n}] Subgrad it={it:05d} new best {val:.6f}")

        # active set where convolution attains its maximum
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # decaying step‑size
        eta = eta0 / np.sqrt(it + 1)

        # exponentiated gradient (keeps non‑negativity)
        a = a * np.exp(-eta * grad_a)
        # enforce symmetry and renormalise
        a = (a + a[::-1]) * 0.5
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        if time.time() - last_improve > 6.0:
            # no improvement for a while – stop early
            break
        it += 1

    return best_seq, best_val


def _deep_optimize(a0: np.ndarray,
                   n: int,
                   conv_fft_len: int,
                   corr_fft_len: int,
                   deadline: float,
                   global_best_seq: np.ndarray | None,
                   global_best_val: float) -> tuple[np.ndarray, float]:
    """
    Run Adam (smooth‑max) followed by a sub‑gradient refinement.
    Updates the globally best vector/value.
    """
    best_seq, best_val = _adam_smooth_max(
        a0,
        n,
        conv_fft_len,
        corr_fft_len,
        deadline,
        global_best_seq,
        global_best_val,
    )
    best_seq, best_val = _subgradient_refinement(
        best_seq,
        n,
        conv_fft_len,
        corr_fft_len,
        deadline,
        best_seq,
        best_val,
    )
    return best_seq, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Entry point – search for a non‑negative step function that minimises
    ``evaluate_sequence``.  Returns the best sequence (plain Python list).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Stage 0 – problem size
    n0 = 4096                     # fast base size for most of the work
    conv_fft_len0 = _next_pow2(2 * n0 - 1)
    corr_fft_len0 = _next_pow2(3 * n0 - 2)

    # ------------------------------------------------------------------
    # Warm‑start handling
    warm_start = globals().get("height_sequence_1")
    ws = None
    if warm_start is not None:
        ws = np.asarray(warm_start, dtype=np.float64)
        if ws.shape[0] > n0:
            ws = ws[:n0]
        elif ws.shape[0] < n0:
            ws = np.pad(ws, (0, n0 - ws.shape[0]))
        ws = np.clip(ws, 0.0, None)
        if ws.sum() >= 0.01:
            ws = ws / ws.sum()
        else:
            ws = None

    # ------------------------------------------------------------------
    # Helper to evaluate a candidate (normalised to sum=1)
    def eval_candidate(vec):
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s < 0.01:
            vec = np.random.dirichlet(np.ones(vec.shape[0]))
            s = vec.sum()
        vec = vec / s
        val = _objective(vec, n0, conv_fft_len0)
        return val, vec

    # ------------------------------------------------------------------
    # 1. Dense parametric sweep (many points, cheap evaluation)
    i = np.arange(n0, dtype=np.float64)
    uniform = np.full(n0, 1.0 / n0)

    candidates = []

    # warm‑start as a candidate
    if ws is not None:
        candidates.append(ws)

    # uniform baseline
    candidates.append(uniform)

    # -------- symmetric beta shapes ----------
    for alpha in np.arange(0.10, 3.001, 0.05):
        shape = (i + 1) ** (alpha - 1.0) * (n0 - i) ** (alpha - 1.0)
        if shape.sum() > 0:
            candidates.append(shape / shape.sum())

    # -------- edge‑exponential + uniform mixture ----------
    alpha_vals = np.logspace(-2, 1.0, 120)          # 0.01 … 10, dense
    w_vals = np.linspace(0.0, 1.0, 41)            # 0 … 1 step 0.025
    for a in alpha_vals:
        edge = np.exp(-a * i) + np.exp(-a * (n0 - 1 - i))
        edge = edge / edge.sum()
        for w in w_vals:
            cand = w * edge + (1.0 - w) * uniform
            candidates.append(cand)                # already normalised

    # -------- edge‑power‑law + uniform mixture ----------
    p_vals = np.linspace(0.2, 5.0, 120)
    for p in p_vals:
        edge = (i + 1) ** (-p) + (n0 - i) ** (-p)
        edge = edge / edge.sum()
        for w in w_vals:
            cand = w * edge + (1.0 - w) * uniform
            candidates.append(cand)

    # -------- arcsine shape ----------
    arcs = 1.0 / np.sqrt((i + 0.5) * (n0 - i - 0.5))
    arcs = arcs / arcs.sum()
    for w in w_vals:
        cand = w * arcs + (1.0 - w) * uniform
        candidates.append(cand)

    # -------- random Dirichlet draws ----------
    for conc in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        cand = np.random.dirichlet(np.full(n0, conc))
        candidates.append(cand)

    # ------------------------------------------------------------------
    # Quick evaluation of the whole pool
    evaluated = []
    for cand in candidates:
        if time.time() >= deadline:
            break
        val, vec = eval_candidate(cand)
        evaluated.append((val, vec))
    evaluated.sort(key=lambda x: x[0])

    top_k = min(20, len(evaluated))
    top_candidates = [c for _, c in evaluated[:top_k]]
    print(f"Parametric sweep finished – best value {evaluated[0][0]:.6f}; "
          f"keeping {top_k} seeds (n={n0})")

    # ------------------------------------------------------------------
    # 2. Local optimisation on the top seeds (still on n0)
    best_seq = None
    best_val = float("inf")
    for idx, start_vec in enumerate(top_candidates):
        if time.time() >= deadline - 2.0:
            break
        print(f"--- Optimising seed {idx+1}/{top_k} ---")
        best_seq, best_val = _deep_optimize(
            start_vec,
            n0,
            conv_fft_len0,
            corr_fft_len0,
            deadline,
            best_seq,
            best_val,
        )
        if best_val < best_val:
            print(f"New global best after seed {idx+1}: {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Upscaling to a finer grid if time permits
    time_left = deadline - time.time()
    if time_left > 40.0 and best_seq is not None:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        # linear interpolation of the best vector
        x_old = np.linspace(0, 1, n0, endpoint=False)
        x_new = np.linspace(0, 1, n_big, endpoint=False)
        a_big = np.interp(x_new, x_old, best_seq)
        a_big = np.maximum(a_big, 0.0)
        a_big /= a_big.sum()

        print("--- Upscaling to n=8192 and continuing optimisation ---")
        best_seq, best_val = _deep_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            best_seq,
            best_val,
        )
        n = n_big
        conv_fft_len = conv_fft_len_big
        corr_fft_len = corr_fft_len_big
        best_seq = np.maximum(best_seq, 0.0)
    else:
        # keep the n0 solution
        n = n0
        conv_fft_len = conv_fft_len0
        corr_fft_len = corr_fft_len0

    # ------------------------------------------------------------------
    # 4. Intense random hill‑climb (mass‑transfer moves) on the final grid
    if best_seq is not None and time.time() < deadline - 5.0:
        a = best_seq.copy()
        a = (a + a[::-1]) * 0.5          # enforce symmetry
        a /= a.sum()
        best_local_val = best_val
        it = 0
        max_iter = 300_000
        while time.time() < deadline - 2.0 and it < max_iter:
            i_idx, j_idx = np.random.randint(n, size=2)
            if i_idx == j_idx:
                continue
            tot = a[i_idx] + a[j_idx]
            # restrict the new split to be within ±20 % of the original masses
            t = np.random.rand()
            new_i = t * tot
            new_j = tot - new_i
            # reject too large jumps (helps stability)
            if (abs(new_i - a[i_idx]) > 0.2 * a[i_idx] or
                abs(new_j - a[j_idx]) > 0.2 * a[j_idx]):
                continue
            a_new = a.copy()
            a_new[i_idx] = new_i
            a_new[j_idx] = new_j
            # keep symmetry
            a_new = (a_new + a_new[::-1]) * 0.5
            # evaluate
            conv = _full_convolution(a_new, conv_fft_len)
            max_c = conv.max()
            val = 2.0 * n * max_c / (a_new.sum() ** 2)
            if val < best_local_val:
                best_local_val = val
                a = a_new
                best_seq = a_new.copy()
                best_val = val
                if it % 2000 == 0:
                    print(f"Hill‑climb iter {it:06d}: val {val:.6f}")
            it += 1
        print(f"Random hill‑climb finished after {it} attempts; best value {best_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback (should never happen)
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the evaluator's allowed numeric range
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"*** Search complete: best objective ≈ {best_val:.6f} (length={best_seq.shape[0]})")
    return best_seq.tolist()
