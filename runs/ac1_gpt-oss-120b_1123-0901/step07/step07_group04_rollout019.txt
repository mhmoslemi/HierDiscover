def full_convolution(a, fft_len):
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[:2*n - 1]

def cross_corr(w, a, fft_len):
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start+n]

def improve(a0, n, deadline, best_val=None, best_seq=None):
    a = np.clip(a0, 0, None)
    if a.sum() < 0.01:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()
    conv_fft_len = next_pow2(2*n -1)
    corr_fft_len = next_pow2(3*n -2)  # for cross correlation length maybe 2n-1 + n -1 =3n-2.

    gamma = 1.0
    gamma_max = 80.0
    gamma_inc = 1.02
    eta0 = 0.4
    it = 0

    last_improve = time.time()
    while time.time() < deadline:
        conv = full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)   # sum=1
        if val < best_val - 1e-12:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[n={n} iter={it}] new best {val:.6f}")

        # compute softmax weights
        # clip conv differences to avoid overflow
        # use gamma factor (increase gradually)
        # shift conv to avoid large exp
        max_conv = conv.max()
        exps = np.exp(gamma * (conv - max_conv))  # stable
        w = exps / exps.sum()

        grad = 2.0 * cross_corr(w, a, corr_fft_len)

        # step size schedule
        eta = eta0 / np.sqrt(it + 1)

        # exponentiated update
        a = a * np.exp(-eta * grad)

        # renorm
        s = a.sum()
        if s < 1e-12 or np.isnan(s) or np.isinf(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        # update gamma slowly
        if gamma < gamma_max:
            gamma = min(gamma_max, gamma * gamma_inc)

        # random perturb if idle
        if time.time() - last_improve > 8.0:
            perturb = (np.random.rand(n) - 0.5) * 0.05
            a = a * np.exp(perturb)
            a = a / a.sum()
            last_improve = time.time()

        it += 1

    return best_seq, best_val

def project_simplex(v):
    """Project vector v onto probability simplex (nonnegative sum=1)."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

import time
import numpy as np

def next_pow2(x: int) -> int:
    return 1 << (x - 1).bit_length()

def full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[:2*a.shape[0] - 1]

def cross_corr(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]

def objective(a: np.ndarray, conv_fft_len: int) -> float:
    conv = full_convolution(a, conv_fft_len)
    return float(2.0 * a.shape[0] * conv.max() / (a.sum() ** 2))

def softmax_weights(conv: np.ndarray, gamma: float) -> np.ndarray:
    # stable softmax
    max_c = conv.max()
    exps = np.exp(gamma * (conv - max_c))
    return exps / exps.sum()

def project_simplex(v: np.ndarray) -> np.ndarray:
    # Euclidean projection onto simplex
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback: uniform
        return np.full(n, 1.0 / n, dtype=v.dtype)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

def generate_seeds(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    seeds = []
    # warm start and rotations
    if warm is not None:
        seeds.append(warm.copy())
        # include reversed to allow monotone shape
        seeds.append(warm[::-1].copy())
    # uniform

    # Dirichlet many conc
    for conc in (0.001, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        seeds.append(np.random.dirichlet(np.full(n, conc)))
    # left‑exponential
    i = np.arange(n, dtype=np.float64)
    for alpha in np.logspace(-3, 0, num=12):  # 0.001 ... 1
        shape = np.exp(-alpha * i)
        shape = shape / shape.sum()
        seeds.append(shape)
    # left‑power‑law
    for p in np.linspace(0.1, 5.0, num=12):
        shape = (i + 1) ** (-p)
        shape = shape / shape.sum()
        seeds.append(shape)
    # half‑sine (positive part)
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape = shape / shape.sum()
    seeds.append(shape)
    # triangle decreasing (linear) a_i ∝ (n-i)
    shape = (n - i).astype(np.float64)
    shape = shape / shape.sum()
    seeds.append(shape)

                            eta_base: float = 0.5,
                            eps: float = 1e-12) -> tuple[np.ndarray | None, float]:

    if a.sum() < 0.01:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()
    gamma = gamma_start
    it = 0
    last_improve = time.time()

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    while time.time() < deadline:

        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val - 1e-12:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()

        # smoothed weights
        w = softmax_weights(conv, gamma)
        grad = 2.0 * cross_corr(w, a, corr_fft_len)
        # Adam update on log domain (mirror)
        # Equivalent to parameter x = log a; grad_x = grad
        # Update m,v
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))
        step = eta_base * m_hat / (np.sqrt(v_hat) + epsilon)
        # move in negative gradient direction in log-space
        log_a = np.log(a + eps) - step
        a = np.exp(log_a)
        # project onto simplex
        a_sum = a.sum()
        if a_sum == 0 or np.isnan(a_sum):
            a = np.random.dirichlet(np.ones(n))
        else:
            a = project_simplex(a)
        # increase gamma slowly

        # random perturb if idle

            perturb = (np.random.rand(n) - 0.5) * 0.02
            a = a * np.exp(perturb)
            a = project_simplex(a)
            last_improve = time.time()
        it += 1
    return best_seq, best_val

"""
Search for a non‑negative step‑function minimizing

        F(a) = 2·n·max(a∗a) / (sum(a)²)

The algorithm:

1. Build a diverse pool of parametric seeds (uniform, Dirichlet,
   monotone exponential / power‑law, half‑wave sinusoid, linear taper,
   optional warm‑start).
2. Evaluate all seeds quickly with an FFT‑based convolution and keep the
   best few.
3. Refine each promising seed with a *smooth‑max* gradient descent:
   – soft‑max weighting of the convolution entries (parameter γ grows
     so the method gradually focuses on the true maximum);
   – Adam‑style adaptive step on the log‑domain (mirrors the
     exponentiated‑gradient update);
   – Euclidean projection onto the probability simplex after each step.
   – occasional random jitter if progress stalls.
4. If time remains, linearly interpolate the current best vector onto a
   finer grid (n=8192) and run the same refinement again.
5. Return the best vector found (clipped to the evaluator’s range).

All operations respect the 1000 s budget, print progress and never
use closures or file/network I/O.
"""

import time
import numpy as np


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def cross_corr(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


def objective(a: np.ndarray, conv_fft_len: int) -> float:
    """Fast evaluation of the objective (sum of a is assumed ≈1)."""
    conv = full_convolution(a, conv_fft_len)
    max_c = conv.max()
    n = a.shape[0]
    return float(2.0 * n * max_c / (a.sum() ** 2))


def softmax_weights(conv: np.ndarray, gamma: float) -> np.ndarray:
    """Smooth approximation of the max‑operator (stable)."""
    m = conv.max()
    exps = np.exp(gamma * (conv - m))
    return exps / exps.sum()


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x ≥ 0, Σ x = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform vector
        return np.full(n, 1.0 / n, dtype=v.dtype)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def generate_seeds(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """
    Produce a diverse collection of candidate vectors.
    Includes the warm‑start, uniform, many Dirichlet draws,
    monotone exponential / power‑law shapes, a half‑wave sinusoid
    and a linear taper.
    """
    seeds = []

    # warm start (and its reverse, which is a different monotone order)
    if warm is not None:
        seeds.append(warm.copy())
        seeds.append(warm[::-1].copy())

    # uniform baseline
    seeds.append(np.full(n, 1.0 / n))

    # random Dirichlet with various concentration parameters
    for conc in (0.001, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        seeds.append(np.random.dirichlet(np.full(n, conc)))

    i = np.arange(n, dtype=np.float64)

    # monotone exponential (left‑hand side only)
    for alpha in np.logspace(-3, 0, num=12):   # 0.001 … 1
        shape = np.exp(-alpha * i)
        shape /= shape.sum()
        seeds.append(shape)

    # monotone power‑law (left‑hand side only)
    for p in np.linspace(0.1, 5.0, num=12):
        shape = (i + 1.0) ** (-p)
        shape /= shape.sum()
        seeds.append(shape)

    # half‑wave sinusoid (positive part only)
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    seeds.append(shape)

    # linear decreasing (triangle) a_i ∝ (n‑i)
    shape = (n - i).astype(np.float64)
    shape /= shape.sum()
    seeds.append(shape)

    return seeds

                            gamma_max: float = 80.0,
                            gamma_inc: float = 1.02,
                            eta_base: float = 0.5) -> tuple[np.ndarray | None, float]:

    Returns the best vector/value observed before the deadline.
    """

    gamma = gamma_start

    m = np.zeros_like(a)

    beta1 = 0.9

    eps = 1e-12

    while time.time() < deadline:

        max_c = conv.max()
