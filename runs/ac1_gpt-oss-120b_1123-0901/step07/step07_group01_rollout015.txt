def pg_optimize(a0, n, conv_fft_len, deadline, max_iters=200_000, eta0=0.5):
    a = a0.copy()
    best_a = a.copy()
    best_val = inf
    it = 0
    conv = full_convolution(a, conv_fft_len)
    while time.time() < deadline and it < max_iters:
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
        # compute w mask for active entries close to max
        # use tolerance 1e-12 relative maybe -less.
        active = np.isclose(conv, max_c, atol=1e-12)
        if not active.any():
            # fallback: assign weight to max index only
            w = np.zeros_like(conv); w[conv.argmax()] = 1.0
        else:
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        # compute gradient
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)   # need corr_fft_len for cross-correlation
        # step size schedule
        eta = eta0 / np.sqrt(it + 1)
        # gradient descent describe: a_new = a - eta * grad
        a_new = a - eta * grad
        # ensure non-negative
        a_new = np.maximum(a_new, 0.0)
        # project onto simplex to sum to 1
        s = a_new.sum()
        if s > 0:
            a_new = a_new / s
        else:
            # if all zero, reinitialize
            a_new = np.random.dirichlet(np.ones(n))
        # update conv if changed significantly; recompute
        a = a_new
        conv = _full_convolution(a, conv_fft_len)
        it += 1
    return best_a, best_val

def project_simplex(v):
    """Project onto probability simplex (sum=1, v>=0). Returns array."""
    # Duchi et al. algorithm
    n = len(v)
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    # find rho = max{i | u_i - cssv_i / i > 0}
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Hybrid random‑gradient‑annealing optimizer for the step‑function problem.

The routine proceeds in several stages inside the given time budget:

1. **Warm‑start** – if a previous solution `height_sequence_1` exists,
   use it as the initial point.
2. **Random exploration** – generate many Dirichlet draws with varying
   concentration to discover a decent starting shape.
3. **Adam on a smooth‑max surrogate** – fast first‑order optimisation
   with a slowly increasing temperature (log‑sum‑exp).
4. **Projected‑gradient descent on the true max** – uses the exact
   sub‑gradient of `max(a*a)` and projects onto the probability simplex.
5. **Simulated annealing** – a cheap O(n) move‑based local search that
   transfers a small amount of mass between two positions.
6. **Optional up‑scaling** – interpolate to a finer grid (8192 points)
   and repeat a short Adam+PG pass if time permits.
7. **Final Gaussian smoothing** – a few low‑pass blur trials are tried;
   the best smoothed vector is kept.

All vectors are kept non‑negative and normalised to unit sum, which makes
the objective
    F(a) = 2·n·max(a*a)     (because sum(a)=1)
scale‑invariant.  The code respects the 1000 s budget and always returns a
valid list of floats.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via FFT (length 2·len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for the gradient."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def _adam_optimize(a0: np.ndarray,
                   n: int,
                   conv_fft_len: int,
                   corr_fft_len: int,
                   deadline: float,
                   max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """Adam optimisation on the smooth‑max (log‑sum‑exp) surrogate."""
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.03               # base learning rate
    lam = 0.3               # start temperature
    lam_factor = 1.015      # annealing factor for λ

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        # soft‑max reparameterisation (ensures sum=1 automatically)
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate (via chain rule)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # centred gradient in log‑space (mirrors Adam on the simplex)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature anneal
        lam = min(lam * lam_factor, 1e6)

        # occasional jitter near deadline to escape tiny plateaus
        if it % 4000 == 0 and time.time() > deadline - 5:
            logits += (np.random.rand(n) - 0.5) * 1e-3

        it += 1

    return best_a, best_val

def _pg_optimize(a0: np.ndarray,
                 n: int,
                 conv_fft_len: int,
                 corr_fft_len: int,
                 deadline: float,
                 max_iters: int = 300_000,
                 eta0: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the *exact* max‑convolution objective.
    Uses the true sub‑gradient (active‑set weights) and projects onto the
    probability simplex after each step.
    """
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s

    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    cur_val = best_val = 2.0 * n * conv.max()

    it = 0
    while time.time() < deadline and it < max_iters:
        # active set for the current max
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-14)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()
        else:
            w[conv.argmax()] = 1.0

        # true sub‑gradient of max_c
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # diminishing step‑size
        eta = eta0 / np.sqrt(it + 1)

        # gradient step on the simplex (Euclidean projection)
        a_new = a - eta * grad
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_simplex(a_new)

        # evaluate new point
        conv_new = _full_convolution(a_new, conv_fft_len)
        val_new = 2.0 * n * conv_new.max()

        if val_new < best_val:
            best_val = val_new
            best_a = a_new.copy()
            print(f"[PG it={it:06d}] new best {val_new:.6f}")

        # accept the move if it does not increase the objective
        if val_new <= cur_val:
            a, conv, cur_val = a_new, conv_new, val_new

        it += 1

    return best_a, best_val

def _simulated_annealing(a0: np.ndarray,
                         n: int,
                         fft_len: int,
                         deadline: float,
                         cur_best_val: float) -> tuple[np.ndarray, float]:
    """Fast O(n) move‑based SA that transfers a small mass between two bins."""
    a = a0.copy()
    conv = _full_convolution(a, fft_len)
    cur_max = conv.max()
    cur_val = 2.0 * n * cur_max
    best_a = a.copy()
    best_val = cur_val

    start = time.time()
    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        fraction = elapsed / (deadline - start + 1e-12)

        # linearly decreasing temperature and move magnitude
        T = max(1e-6, 0.15 * (1.0 - fraction))
        move_scale = max(0.02, 0.3 * (1.0 - fraction))

        # source proportional to current mass
        if a.sum() == 0:
            break
        i = np.random.choice(n, p=a / a.sum())
        if a[i] < 1e-14:
            continue
        # distinct target
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * (np.random.rand() * move_scale)

        # apply move on copies
        a_new = a.copy()
        conv_new = conv.copy()
        a_new[i] -= delta
        a_new[j] += delta

        # update convolution locally (O(n) update)
        conv_new[i : i + n] -= 2.0 * delta * a
        conv_new[j : j + n] += 2.0 * delta * a
        conv_new[2 * i] += delta * delta
        conv_new[2 * j] += delta * delta
        conv_new[i + j] -= 2.0 * delta * delta

        new_max = conv_new.max()
        new_val = 2.0 * n * new_max

        # Metropolis acceptance
        if new_val < cur_val or np.random.rand() < np.exp(-(new_val - cur_val) / T):
            a, conv = a_new, conv_new
            cur_val, cur_max = new_val, new_max
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[SA it={it:06d}] new best {best_val:.6f}")

        it += 1
        if it % 5000 == 0:
            print(f"[SA it={it:06d}] best {best_val:.6f}")

    return best_a, best_val

def _gaussian_smoothing_candidates(a: np.ndarray,
                                   sigmas: list[float]) -> tuple[np.ndarray, float]:
    """Try a few Gaussian blur widths and keep the best."""
    n = a.shape[0]
    best_a, best_val = a.copy(), float("inf")
    for sigma in sigmas:
        # build a 1‑D Gaussian kernel (odd length)
        radius = int(3 * sigma)
        xs = np.arange(-radius, radius + 1)
        kernel = np.exp(-(xs ** 2) / (2 * sigma ** 2))
        kernel /= kernel.sum()
        a_s = np.convolve(a, kernel, mode="same")
        a_s = np.maximum(a_s, 0.0)
        a_s /= a_s.sum()
        conv = _full_convolution(a_s, _next_pow2(2 * n - 1))
        val = 2.0 * n * conv.max()
        if val < best_val:
            best_val = val
            best_a = a_s.copy()
    return best_a, best_val

        )
        if val_sa < best_val:
            best_val = val_sa
            best_seq = a_sa.copy()
            print(f"[SA] improved to {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – optional upscale to 8192 points (if time permits)
    if best_seq is not None and time.time() + 20 < deadline:
        print("Phase 5 – upscale to 8192 and refine")
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        # linear interpolation
        x_small = np.linspace(0.0, 1.0, best_seq.shape[0], endpoint=False)
        x_big = np.linspace(0.0, 1.0, n_big, endpoint=False)
        a_big = np.interp(x_big, x_small, best_seq)
        a_big = np.maximum(a_big, 0.0)
        a_big /= a_big.sum()

        # short Adam + PG on the larger grid
        a_big_adam, _ = _adam_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            max_iters=120_000,
        )
        if time.time() < deadline - 5:
            a_big_pg, val_big = _pg_optimize(
                a_big_adam,
                n_big,
                conv_fft_len_big,
                corr_fft_len_big,
                deadline,
                max_iters=150_000,
            )
            if val_big < best_val:
                best_val = val_big
                best_seq = a_big_pg.copy()
                print(f"[Big] final best {val_big:.6f}")

    # ------------------------------------------------------------------
    # Phase 6 – final light Gaussian smoothing sweep
    if best_seq is not None and time.time() < deadline - 1:
        sigmas = [0.5, 1.0, 1.5, 2.0]
        a_smooth, val_smooth = _gaussian_smoothing_candidates(best_seq, sigmas)
        if val_smooth < best_val:
            best_val = val_smooth
            best_seq = a_smooth.copy()
            print(f"[smooth] improved to {val_smooth:.6f}")

    # ------------------------------------------------------------------
    # Fallback – guarantee a valid return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the limits enforced by evaluate_sequence
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Report final objective (via the judge's routine if available)
    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"Final reported value {final_val:.6f} (length {len(best_seq)})")
    except Exception:
        print(f"Final internal value {best_val:.6f} (length {len(best_seq)})")

    return best_seq.tolist()
