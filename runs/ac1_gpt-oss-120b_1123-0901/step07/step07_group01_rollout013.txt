def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5

    # Choose n = 4096 (or maybe 8192 if time). Let's start n=4096.
    n = 4096
    # Initialize candidate distribution.
    warm = globals().get("height_sequence_1")
    if warm is not None and len(warm) >= n:
        a = np.asarray(warm[:n], dtype=float)
        a = np.clip(a, 0, 1000)
        if a.sum() < 0.01:
            a = np.random.dirichlet(np.ones(n))
        else:
            a = a / a.sum()
    else:
        # random initialization: Dirichlet with small concentration (promoting spikes) or uniform.
        # Let's attempt mixture.
        if np.random.rand() < 0.5:
            a = np.random.dirichlet(np.ones(n))
        else:
            # Beta-like shape: raise to power
            alpha = np.random.uniform(0.2, 0.8)
            x = np.random.rand(n) ** alpha
            a = x / x.sum()
    # Ensure sum=1
    a = a / a.sum()
    best_a = a.copy()
    best_val = evaluate_sequence(a.tolist())
    print(f"Initial val {best_val}")

    # helpers:
    conv_fft_len = _next_pow2(2 * n - 1)

    def conv(a1, a2):
        # compute linear convolution using FFT
        fa = np.fft.rfft(a1, conv_fft_len)
        fb = np.fft.rfft(a2, conv_fft_len)
        conv_res = np.fft.irfft(fa * fb, conv_fft_len)
        return conv_res[: 2 * n - 1]

    t = 0.5  # mixing ratio

    # iterate until deadline
    iter = 0
    while time.time() < deadline - 0.5:
        iter += 1
        # compute conv and M
        conv_f = conv(a, a)
        M = conv_f.max()
        val = 2 * n * M
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Iter {iter}] new best {val:.6f}")

        # active set threshold
        active_idx = np.where(conv_f >= M - 1e-12)[0]
        # start with these constraints
        constraint_idxs = active_idx.tolist()
        # cut generation loop
        max_cuts = 20
        for cut in range(max_cuts):
            # Build constraint matrix rows for indices in constraint_idxs
            m = len(constraint_idxs)
            A = np.zeros((m, n))
            for row_i, k in enumerate(constraint_idxs):
                i_start = max(0, k - (n - 1))
                i_end = min(k, n - 1)
                # j = k - i
                # assign reversed slice
                # length of segment
                length = i_end - i_start + 1
                j_start = k - i_end

                A[row_i, j_start : j_start + length] = a[i_start : i_end + 1][::-1]
            # solve LP
            c = -np.ones(n)
            bounds = [(0, None)] * n
            res = None
            try:
                res = linprog(c, A_ub=A, b_ub=np.full(m, M), bounds=bounds, method='highs')
            except Exception as e:
                print(f"LP error {e}")
                break
            if not res.success:
                # maybe unbounded - add more constraints or break
                print(f"LP not successful: {res.message}")
                break
            b0 = res.x
            # check feasibility on whole domain
            conv_fb = conv(a, b0)
            max_violation = conv_fb.max() - M
            if max_violation <= 1e-12:
                # feasible
                break
            # else find violating indices
            viol_idx = np.where(conv_fb > M + 1e-12)[0]
            # add the most violating index
            new_idx = int(viol_idx[np.argmax(conv_fb[viol_idx])])
            if new_idx not in constraint_idxs:
                constraint_idxs.append(new_idx)
            else:
                # already present, break to avoid infinite loop
                break
        else:

            pass

        # after LP, b0 is feasible for current constraints (maybe not globally)
        # compute sum to see scaling potential
        sum_b = b0.sum()
        if sum_b < 1e-12:
            # fallback random pert
            a = np.random.dirichlet(np.ones(n))
            continue

        # normalise b0 to probability distribution g
        g = b0 / sum_b

        # mixture update
        a = (1 - t) * a + t * g
        # renormalise due to numeric error
        a = a / a.sum()
        # maybe adjust t adaptively: decay
        t = max(0.1, t * 0.99)

    # final best candidate
    final_seq = best_a.tolist()
    # Clip to bounds
    final_seq = [float(min(1000.0, max(0.0, x))) for x in final_seq]
    try:
        final_val = evaluate_sequence(final_seq)
        print(f"Final reported value {final_val}")
    except Exception:
        print(f"Final internal best {best_val}")

    return final_seq

import time
import numpy as np
from scipy.optimize import linprog

def _next_pow2(x: int) -> int:
    """Return smallest power of two >= x."""
    return 1 << (x - 1).bit_length()

def _full_convolution_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]

def _convolution_fft(a: np.ndarray, b: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*b via FFT."""
    fa = np.fft.rfft(a, fft_len)
    fb = np.fft.rfft(b, fft_len)
    conv = np.fft.irfft(fa * fb, fft_len)
    return conv[: a.shape[0] + b.shape[0] - 1]

def _build_constraint_matrix(a: np.ndarray, idxs: list[int]) -> np.ndarray:
    """For each index k in idxs construct row with coefficients of convolution a*b at k."""
    n = a.shape[0]
    m = len(idxs)
    A = np.zeros((m, n), dtype=np.float64)
    for row, k in enumerate(idxs):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        length = i_end - i_start + 1
        j_start = k - i_end
        # Assign reversed slice of a
        A[row, j_start:j_start + length] = a[i_start:i_end + 1][::-1]
    return A

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimizes
        F(a) = 2·n·max(a∗a) / (sum a)^2 .
    The algorithm uses an inequality‑based cutting‑plane LP to enlarge the
    feasible region without increasing the current max‑convolution bound,
    then mixes the current distribution toward the LP solution.
    It repeatedly refines the distribution on a fixed grid (default 8192)
    while keeping the total mass normalized to 1.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5      # safety margin

    # ------------------------------------------------------------------
    # Grid size. 8192 offers a good trade‑off between resolution and speed.
    n = 8192
    fft_len = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Initialise a probability vector a.
    warm = globals().get("height_sequence_1")
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64).ravel()
        # Interpolate to the desired size.
        x_src = np.linspace(0.0, 1.0, w.shape[0], endpoint=False)
        x_dst = np.linspace(0.0, 1.0, n, endpoint=False)
        a = np.interp(x_dst, x_src, w)
        a = np.maximum(a, 0.0)
        if a.sum() < 0.01:
            a = np.random.dirichlet(np.ones(n))
        else:
            a = a / a.sum()
    else:
        # Random Dirichlet with a small concentration to get a non‑flat start.
        alpha = np.random.uniform(0.05, 0.5)
        a = np.random.dirichlet(np.full(n, alpha))
        a = a / a.sum()

    # Ensure strictly positive components – required for bounded LP.
    eps = 1e-12
    a = np.maximum(a, eps)
    a = a / a.sum()

    best_a = a.copy()
    # Compute initial objective directly (sum a = 1 ⇒ denominator = 1)
    conv_a = _full_convolution_fft(a, fft_len)
    M = conv_a.max()
    best_val = 2.0 * n * M
    print(f"Initial objective {best_val:.6f}")

    # ------------------------------------------------------------------
    # Mixing coefficient – will be slowly decayed.
    t = 0.5
    iteration = 0

    while time.time() < deadline - 2.0:
        iteration += 1

        # --- Current state -------------------------------------------------
        conv_a = _full_convolution_fft(a, fft_len)
        M = conv_a.max()
        cur_val = 2.0 * n * M
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[Iter {iteration:05d}] new best {cur_val:.6f}")

        # --- Cutting‑plane LP (maximize total mass under current bound) -----
        # Start with the indices where the current max is attained.
        active_idxs = np.where(conv_a >= M - 1e-12)[0].tolist()
        constraint_idxs = active_idxs.copy()

        # Iteratively add violated constraints.
        max_cuts = 30
        feasible_b = None
        for cut in range(max_cuts):
            A = _build_constraint_matrix(a, constraint_idxs)
            b_ub = np.full(len(constraint_idxs), M, dtype=np.float64)
            c = -np.ones(n, dtype=np.float64)                 # maximise sum(b)  ⇔ minimise -sum(b)
            bounds = [(0, None)] * n                           # b_j ≥ 0

            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
            if not res.success:
                # LP failed – abort the cutting‑plane loop.
                print(f"[Iter {iteration}] LP failure: {res.message}")
                break

            b0 = res.x
            # Verify feasibility on the whole convolution.
            conv_ab = _convolution_fft(a, b0, fft_len)
            violation = conv_ab - M
            max_violation = violation.max()
            if max_violation <= 1e-11:
                feasible_b = b0
                break

            # Add the most violating index to the active set.
            viol_idx = int(np.argmax(violation))
            if viol_idx not in constraint_idxs:
                constraint_idxs.append(viol_idx)
            else:
                # No progress – stop.
                break

        # If the LP never became feasible, fall back to a tiny random move.
        if feasible_b is None:
            jitter = (np.random.rand(n) - 0.5) * 1e-5
            a = np.clip(a + jitter, eps, None)
            a /= a.sum()
            continue

        # --- Mix the current distribution toward the LP solution ------------
        sum_b = feasible_b.sum()
        if sum_b < 1e-12:
            # Degenerate solution – skip mixing.
            continue
        g = feasible_b / sum_b                     # normalised probability vector
        a = (1.0 - t) * a + t * g
        a = np.maximum(a, eps)                      # keep positivity
        a /= a.sum()

        # Decay the mixing factor slowly to stabilise.
        t = max(0.1, t * 0.995)

        # Occasionally inject a very small random perturbation to escape plateaus.
        if iteration % 200 == 0:
            a += (np.random.rand(n) - 0.5) * 1e-6
            a = np.maximum(a, eps)
            a /= a.sum()

    # ------------------------------------------------------------------
    # Return the best sequence found.
    final_seq = best_a.tolist()
    final_seq = [float(min(1000.0, max(0.0, x))) for x in final_seq]

    try:
        final_val = evaluate_sequence(final_seq)
        print(f"Final reported value {final_val:.6f}")
    except Exception as e:
        print(f"Could not evaluate final sequence: {e}; best internal value {best_val:.6f}")

    return final_seq

            a += (np.random.rand(n) - 0.5) * 1e-6
            a = np.maximum(a, eps)
            a /= a.sum()

    # ------------------------------------------------------------------
    # Return the best sequence found (as plain Python floats).
    final_seq = [float(min(1000.0, max(0.0, x))) for x in best_a.tolist()]

    try:
        final_val = evaluate_sequence(final_seq)
        print(f"Final reported value {final_val:.6f}")
    except Exception as e:
        print(f"Could not evaluate final sequence ({e}); best internal value {best_val:.6f}")

    return final_seq
