"""
A fast two‑stage optimizer for the step‑function problem.

Stage 1 – a tiny linear program (LP) on a coarse grid (n≈512) that
pushes the mass to positions where the convolution is already low.
The LP maximises the total mass of a new vector `b` under the
constraints   a * b ≤ M   (a fixed, M the current worst convolution).
The resulting `b` is renormalised and mixed back into `a`.  A handful of
iterations dramatically reduces the worst convolution.

Stage 2 – refinement on the full grid (n=4096) with the same
smooth‑max / Adam and multiplicative sub‑gradient machinery that was
used in the reference implementation, but with longer runs and a
slightly more aggressive learning‑rate schedule.  Finally a cheap
hill‑climbing pass is performed.

The whole routine respects the caller’s time budget (default 1000 s)
and always returns the best sequence found.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)−1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a length‑len(a) vector:
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i = 0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
# ---------- Stage 1: cheap LP on a small grid ------------------------
def _build_conv_matrix(a: np.ndarray) -> np.ndarray:
    """
    Build the (2n‑1)×n matrix C such that (C @ b)[k] = (a * b)[k].
    a is assumed to be a 1‑D array of length n.
    """
    n = a.shape[0]
    rows = 2 * n - 1
    C = np.zeros((rows, n), dtype=np.float64)
    # C[k, j] = a[k‑j] if 0≤k‑j<n else 0
    for i in range(n):
        # column i of b contributes a[i] to rows i … i+n‑1
        C[i : i + n, i] = a
    return C


def _lp_step(a: np.ndarray, bound: float, t: float = 0.5) -> np.ndarray:
    """
    One LP‑based update.

    Parameters
    ----------
    a : ndarray, shape (n,)
        Current probability vector (sum = 1).
    bound : float
        Current worst convolution value M = max(a*a).
    t : float, optional
        Mixing coefficient 0<t≤1 (default 0.5).  The new vector is
        (1‑t)·a + t·b_opt, where b_opt solves the LP.

    Returns
    -------
    a_new : ndarray, shape (n,)
        Updated probability vector (still sums to 1).
    """
    import cvxpy as cp

    n = a.shape[0]
    # linear constraints: (a * b)[k] ≤ bound   for all k
    C = _build_conv_matrix(a)

    b = cp.Variable(n, nonneg=True)

    constraints = [C @ b <= bound]
    # No explicit sum‑to‑1 constraint – the LP will push the mass up
    # as far as the bound allows.
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)

    # ECOS is fast for modest‑size LPs; warm‑start speeds repeated solves.
    prob.solve(solver=cp.ECOS, warm_start=True, verbose=False, max_iters=2000)

    if b.value is None:
        # LP failed – just return the original vector.
        return a.copy()

    b_opt = np.maximum(b.value, 0.0)
    total = b_opt.sum()
    if total < 1e-12:            # degenerate solution – keep a.
        return a.copy()
    b_opt /= total                # re‑normalise to a probability vector

    a_new = (1.0 - t) * a + t * b_opt
    # re‑normalise (tiny numerical drift)
    a_new /= a_new.sum()
    return a_new

# ---------- Stage 2: Adam + sub‑gradient on the full grid ------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the smooth‑max (log‑sum‑exp) surrogate.
    Returns the best vector (∑a≈1) encountered and its exact objective.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.06                     # a bit more aggressive than the baseline
    lam = 0.5                     # start temperature
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max → simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective (∑a=1)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam iter={it:07d}] new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient wrt logits (chain rule through soft‑max)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature → true max
        lam = min(lam * lam_factor, 1e6)

        # tiny random jitter every few thousand iterations
        if it % 5000 == 0 and time.time() + 0.1 < deadline:
            logits += (np.random.rand(n) - 0.5) * 3e-3

        it += 1

    return best_a, best_val


def _subgradient_opt(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
    eta0: float = 0.7,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient method with a decaying step size.
    Returns the best vector seen and its exact objective.
    """
    a = np.maximum(a0, 0.0)
    a /= max(a.sum(), 1e-12)

    best_a = a.copy()
    best_val = float("inf")
    stall = 0
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            stall = 0
            if it % 5000 == 0:
                print(f"[Subgrad iter={it:08d}] new best {val:.6f}")
        else:
            stall += 1

        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # decaying step size
        eta = eta0 / np.sqrt(it + 1)

        # multiplicative update and renormalisation
        a = a * np.exp(-eta * grad_a)
        a_sum = a.sum()
        if a_sum < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            a_sum = a.sum()
        a /= a_sum

        # occasional shake if stuck
        if stall > 8000:
            a = np.maximum(a + 0.02 * (np.random.rand(n) - 0.5), 0.0)
            a /= a.sum()
            stall = 0

        it += 1
    return best_a, best_val


def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Cheap hill‑climbing: relocate a random fraction of mass from i→j.
    Keep the move if the objective improves.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    best_a = a.copy()
    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue
        # move a fraction (up to 30 %) of the source mass
        delta = a[i] * (np.random.rand() * 0.3)
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta
        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()
        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5000 == 0:
                print(f"[Local move {it}] improved to {val2:.6f}")
        it += 1
    return best_a, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Return a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence`` within the supplied time budget.
    The algorithm first runs a tiny LP on a 512‑point grid to obtain a
    good starting shape, then refines it on the full 4096‑point grid
    with Adam, a long sub‑gradient run and a final cheap hill‑climbing
    pass.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Stage 1 – LP on a coarse grid (n≈512)
    n_small = 512
    a = np.full(n_small, 1.0 / n_small, dtype=np.float64)

    # pre‑compute FFT length for the small grid (used only inside the LP)
    conv_fft_len_small = _next_pow2(2 * n_small - 1)

    # initial bound M = max(a*a)
    M = _full_convolution(a, conv_fft_len_small).max()
    print(f"[Stage 1] start M={M:.6e}")

    # a handful of LP‑mixing iterations
    for it_lp in range(8):
        if time.time() >= deadline:
            break
        a = _lp_step(a, M, t=0.5)
        M_new = _full_convolution(a, conv_fft_len_small).max()
        print(f"[LP iter {it_lp}] M={M_new:.6e}")
        # stop early if the bound does not improve appreciably
        if M_new >= M * (1 - 1e-5):
            break
        M = M_new

    # ------------------------------------------------------------------
    # Stage 2 – up‑sample to the target size (n = 4096) and refine
    n = 4096
    x_old = np.linspace(0.0, 1.0, n_small, endpoint=False)
    x_new = np.linspace(0.0, 1.0, n, endpoint=False)
    a_big = np.interp(x_new, x_old, a)

    a_big = np.maximum(a_big, 0.0)
    a_big /= a_big.sum()
    best_seq = a_big.copy()

    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Adam phase (≈30 s)
    if time.time() < deadline - 5:
        adam_deadline = min(deadline - 5.0, time.time() + 30.0)
        a_adam, val_adam = _adam_optimize(
            best_seq, n, conv_fft_len, corr_fft_len, adam_deadline,
            max_iters=400_000)
        best_seq = a_adam.copy()
        best_val = val_adam
        print(f"[Adam] best {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Long sub‑gradient phase (use up most of the remaining budget)
    if time.time() < deadline - 5:
        sub_deadline = deadline
        a_sub, val_sub = _subgradient_opt(
            best_seq, n, conv_fft_len, corr_fft_len,
            sub_deadline, max_iters=800_000, eta0=0.6)
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Subgrad long] best {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Cheap local random‑move hill‑climbing
    if time.time() < deadline - 5:
        a_loc, val_loc = _local_random_search(
            best_seq, n, conv_fft_len, deadline, max_moves=250_000)
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()
            print(f"[Local hill] improved best {val_loc:.6f}")

    # ------------------------------------------------------------------
    # Optional upscale to 8192 if time permits (very short final refinements)
    if time.time() + 30 < deadline:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        # linear interpolation of the current best
        x_old = np.linspace(0.0, 1.0, best_seq.shape[0], endpoint=False)
        x_new = np.linspace(0.0, 1.0, n_big, endpoint=False)
        a_big = np.interp(x_new, x_old, best_seq)

        a_big = np.maximum(a_big, 0.0)
        a_big /= a_big.sum()

        # short Adam on the larger grid
        adam_big_dead = min(deadline - 5.0, time.time() + 20.0)
        a_big_adam, val_big_adam = _adam_optimize(
            a_big, n_big, conv_fft_len_big, corr_fft_len_big,
            adam_big_dead, max_iters=180_000)
        if val_big_adam < best_val:
            best_val = val_big_adam
            best_seq = a_big_adam.copy()
            print(f"[Big‑8192] Adam best {val_big_adam:.6f}")

        # one quick sub‑gradient pass
        if time.time() < deadline:
            a_big_sub, val_big_sub = _subgradient_opt(
                best_seq, n_big, conv_fft_len_big, corr_fft_len_big,
                deadline, max_iters=300_000, eta0=0.5)
            if val_big_sub < best_val:
                best_val = val_big_sub
                best_seq = a_big_sub.copy()
                print(f"[Big‑8192] Subgrad best {val_big_sub:.6f}")

    # ------------------------------------------------------------------
    # Final safety: ensure a valid return value
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Clip to the evaluator’s limits (0 ≤ x ≤ 1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
