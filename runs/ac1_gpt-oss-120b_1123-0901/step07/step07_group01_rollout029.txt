def simulated_annealing(a, deadline, init_val=None):
    n = len(a)
    fft_len = _next_pow2(2*n-1)  # convert to power of 2 for FFT speed
    conv = np.convolve(a, a) # maybe use FFT for speed
    cur_val = 2 * n * conv.max()
    best_a = a.copy()
    best_val = cur_val
    while time.time() < deadline:
        # random source i (mass weighted)
        i = np.random.choice(n, p=a/a.sum())
        if a[i] < 1e-12:
            continue
        # target j random not equal
        j = np.random.randint(n-1)
        if j >= i:
            j += 1
        # move magnitude
        move_scale = 0.2  # maybe degrade with time
        delta = a[i] * np.random.rand() * move_scale
        if delta <= 0:
            continue
        # apply incremental conv update
        a_new = a.copy()
        conv_new = conv.copy()

"""
Hybrid optimisation for the step‑function bound

The routine proceeds in three stages:

1. **Small‑grid LP refinement** – a 256‑point distribution is iteratively
   improved by a linear programme that maximises the total mass while
   keeping the self‑convolution peak unchanged.  The update is a small
   convex combination with the LP solution; this quickly produces a
   “U‑shaped’’ profile that already reduces the objective.

2. **Upscaling + smooth‑max Adam + exact‑max sub‑gradient** – the
   refined 256‑vector is interpolated to the full resolution (4096 points)
   and then optimised with the Adam optimiser on a log‑sum‑exp surrogate
   (slowly sharpening) followed by an exact‑max exponentiated‑gradient
   descent.  Both algorithms run until the remaining time budget forces
   them to stop.

3. **Fast simulated annealing** – a final local search works directly on
   the linear convolution (O(n) updates per move) and cleans up any
   remaining peaks.

All heavy work uses FFT‑based convolutions and respects the global time
budget (default 1000 s).  The returned sequence is guaranteed to be
non‑negative and clipped to the limits enforced by evaluate_sequence().
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via FFT (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for the gradient."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   conv_fft_len: int,
                   corr_fft_len: int,
                   deadline: float,
                   max_iters: int = 600_000) -> tuple[np.ndarray, float]:
    """Adam on the log‑sum‑exp (smooth‑max) surrogate."""
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.03                # modest learning rate
    lam = 0.3                # start temperature for the smooth‑max
    lam_factor = 1.015       # slow anneal

    best_a = a0.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c          # sum(a)=1 → objective = 2·n·max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max (log‑sum‑exp) weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate (chain‑rule through soft‑max)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        lam = min(lam * lam_factor, 1e6)

        it += 1

    return best_a, best_val


def _subgradient_optimize(a0: np.ndarray,
                          n: int,
                          conv_fft_len: int,
                          corr_fft_len: int,
                          deadline: float,
                          max_iters: int = 500_000,
                          eta0: float = 0.6) -> tuple[np.ndarray, float]:
    """Exact‑max sub‑gradient descent using exponentiated‑gradient updates."""
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s

    best_a = a.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # sub‑gradient of the exact max (uniform weight over active entries)
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # diminishing step‑size
        eta = eta0 / np.sqrt(it + 1)

        # multiplicative (exponentiated‑gradient) update
        a = a * np.exp(-eta * grad_a)

        # renormalise onto the simplex
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray,
                         n: int,
                         fft_len: int,
                         deadline: float,
                         cur_best_val: float | None = None) -> tuple[np.ndarray, float]:
    """Fast SA that updates the convolution in O(n) per elementary move."""
    a = a0.copy()
    conv = _full_convolution(a, fft_len)
    cur_val = 2.0 * n * conv.max()
    best_a = a.copy()
    best_val = cur_val if cur_best_val is None else min(cur_val, cur_best_val)

    start = time.time()
    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        fraction = elapsed / (deadline - start + 1e-12)

        # linear cooling from 0.1 down to 1e‑6
        T = max(1e-6, 0.1 * (1.0 - fraction))
        # move magnitude shrinks as we cool
        move_scale = max(0.01, 0.3 * (1.0 - fraction))

        # pick source index proportionally to its mass
        if a.sum() == 0:
            break
        i = np.random.choice(n, p=a / a.sum())
        if a[i] < 1e-12:
            continue
        # pick a different target uniformly
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * (np.random.rand() * move_scale)

        a_new = a.copy()
        conv_new = conv.copy()
        a_new[i] -= delta
        a_new[j] += delta

        # update convolution using the *old* a (the one before the move)
        conv_new[i : i + n] -= 2.0 * delta * a
        conv_new[j : j + n] += 2.0 * delta * a
        conv_new[2 * i] += delta * delta
        conv_new[2 * j] += delta * delta
        conv_new[i + j] -= 2.0 * delta * delta

        new_val = 2.0 * n * conv_new.max()

        # Metropolis acceptance
        if new_val < cur_val or np.random.rand() < np.exp(-(new_val - cur_val) / T):
            a, conv = a_new, conv_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[SA it={it:06d}] new best {best_val:.6f}")

        it += 1
        if it % 5000 == 0:
            print(f"[SA it={it:06d}] best so far {best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
def _objective_small(a: np.ndarray) -> float:
    """Objective on a small grid (sum is normalised to 1)."""
    n = a.shape[0]
    conv = np.convolve(a, a)
    return 2.0 * n * conv.max()


def _lp_update(f: np.ndarray, eta: float = 0.2) -> tuple[np.ndarray, float]:
    """One LP‑guided update on a normalised vector `f`."""
    n = f.shape[0]
    eps = 1e-12
    # current peak of the self‑convolution
    conv = np.convolve(f, f)
    M = conv.max()

    # build the linear constraints A g <= M
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j : j + n, j] = f
    b = np.full(2 * n - 1, M)

    # maximise sum(g)  <=> minimise -1ᵀ g
    res = linprog(-np.ones(n), A_ub=A, b_ub=b,
                 bounds=[(0, None)] * n, method='highs', options={'presolve': True})
    if not res.success:
        # fall back to the current point
        return f, M

    g = res.x
    sum_g = g.sum()
    if sum_g <= eps:
        return f, M
    g /= sum_g                     # keep the simplex normalisation

    # convex combination step
    f_new = (1 - eta) * f + eta * g
    f_new = np.maximum(f_new, eps)
    f_new /= f_new.sum()

    # compute new peak (used only for bookkeeping)
    new_M = np.convolve(f_new, f_new).max()
    return f_new, new_M


def _lp_refine_small(start: np.ndarray,
                     max_iters: int,
                     deadline: float) -> np.ndarray:
    """Repeated LP‑guided refinement on the 256‑point grid."""
    f = start.copy()
    best_f = f.copy()
    best_val = _objective_small(f)
    it = 0
    while time.time() < deadline and it < max_iters:
        f, _ = _lp_update(f, eta=0.2)
        cur_val = _objective_small(f)
        if cur_val < best_val:
            best_val = cur_val
            best_f = f.copy()
            print(f"[LP it={it:04d}] new best {cur_val:.6f}")
        it += 1
    return best_f


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """Search for a non‑negative step‑function that minimises
       `2·n·max(a∗a)/(sum a)^2`.  The routine first runs a short LP phase on a
       256‑point grid, then up‑scales, performs long Adam + sub‑gradient
       optimisation on the full grid (4096 points), and finishes with a fast
       simulated‑annealing local search."""
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # 1. Small‑grid LP phase (≈256 points)
    n_small = 256
    # initialise: arcsine‑type edge‑heavy shape (good warm‑start)
    idx = np.arange(1, n_small + 1, dtype=np.float64)
    arcs = 1.0 / np.sqrt(idx * (n_small - idx + 1.0))
    arcs /= arcs.sum()
    start_small = arcs

    # give the LP phase at most 15 % of the total budget, but ensure
    # at least 30 seconds are left for the later stages.
    lp_deadline = min(deadline - 30.0,
                      time.time() + 0.15 * budget_s)
    best_small = _lp_refine_small(start_small,
                                  max_iters=80,
                                  deadline=lp_deadline)

    # --------------------------------------------------------------
    # 2. Upscale to the full grid and run Adam / sub‑gradient
    n_big = 4096
    x_small = np.linspace(0.0, 1.0, n_small, endpoint=False)
    x_big = np.linspace(0.0, 1.0, n_big, endpoint=False)
    a_big = np.interp(x_big, x_small, best_small)
    a_big = np.maximum(a_big, 0.0)
    a_big /= a_big.sum()

    conv_fft_len = _next_pow2(2 * n_big - 1)
    corr_fft_len = _next_pow2(3 * n_big - 2)

    # split the remaining time: 60 % Adam, 30 % sub‑gradient,
    # keep ~10 % for the final SA.
    now = time.time()
    remaining = deadline - now
    adam_deadline = now + 0.60 * remaining
    sub_deadline = adam_deadline + 0.30 * (deadline - adam_deadline)
    # (the SA phase will automatically use whatever time is left)

    # Adam on a smooth‑max surrogate
    a_adam, val_adam = _adam_optimize(a_big, n_big,
                                      conv_fft_len, corr_fft_len,
                                      adam_deadline,
                                      max_iters=800_000)
    # Sub‑gradient refinement (exact max)
    a_sub, val_sub = _subgradient_optimize(a_adam, n_big,
                                           conv_fft_len, corr_fft_len,
                                           sub_deadline,
                                           max_iters=800_000)

    # --------------------------------------------------------------
    # 3. Final simulated‑annealing refinement
    # give the SA the rest of the time budget
    a_final, final_val = _simulated_annealing(a_sub, n_big,
                                              conv_fft_len,
                                              deadline,
                                              cur_best_val=val_sub)

    # --------------------------------------------------------------
    # Clip to the limits enforced by evaluate_sequence
    best_seq = np.clip(a_final, 0.0, 1000.0)

    # Report the final objective (using the judge's routine if available)
    try:
        reported = evaluate_sequence(best_seq.tolist())
        print(f"Final reported value {reported:.6f} (length {len(best_seq)})")
    except Exception as exc:
        print(f"Could not call evaluate_sequence: {exc}")
        print(f"Internal best value {final_val:.6f}")

    return best_seq.tolist()
