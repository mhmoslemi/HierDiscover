"""
search_stepfunction.py

Search for a non‑negative step function (probability vector) that minimizes

        F(a) = 2·n·max(a∗a) / (sum(a)²)

The algorithm works in three stages:

1. **Parametric sweep** – evaluate many low‑dimensional families
   (edge‑exponential, edge‑power‑law, double‑Gaussian, sinusoid,
   uniform mixture) on a grid of parameters.  The best few candidates
   are kept.

2. **Refined mirror‑descent** – starting from each top candidate we run
   an exponentiated‑gradient (multiplicative‑weights) optimisation.
   The true max‑convolution is approximated by a soft‑max (temperature
   annealing) which provides a smooth sub‑gradient.  An Adam‑style
   momentum scheme is used on the log‑variables for faster convergence.

3. **Upscaling** – if time permits the best solution is linearly
   interpolated to a finer grid (n=8192) and refined again.

All vectors are kept on the simplex (sum = 1) so the objective reduces
to `2·n·max(conv)`.  The code respects the 1000 s budget, prints
progress, and finally returns the best sequence as a plain list of
floats.*

"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n-1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Fast evaluation of the objective for a (assumed normalised)."""
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c)   # sum(a) == 1


# ----------------------------------------------------------------------
# Parametric candidate generators
# ----------------------------------------------------------------------


def _candidate_edge_exponential(n: int, alpha: float, w: float) -> np.ndarray:
    """w·edge‑exp(alpha) + (1‑w)·uniform, non‑negative and normalised."""
    i = np.arange(n, dtype=np.float64)
    edge = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
    edge /= edge.sum()
    uniform = np.full(n, 1.0 / n)
    shape = w * edge + (1.0 - w) * uniform
    shape /= shape.sum()
    return shape


def _candidate_edge_powerlaw(n: int, p: float, w: float) -> np.ndarray:
    """w·edge‑power‑law(p) + (1‑w)·uniform."""
    i = np.arange(n, dtype=np.float64)
    edge = (i + 1.0) ** (-p) + (n - i) ** (-p)
    edge /= edge.sum()
    uniform = np.full(n, 1.0 / n)
    shape = w * edge + (1.0 - w) * uniform
    shape /= shape.sum()
    return shape


def _candidate_double_gaussian(n: int, gamma: float, w: float) -> np.ndarray:
    """w·(Gaussian left + Gaussian right) + (1‑w)·uniform."""
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-gamma * (i ** 2))
    right = np.exp(-gamma * ((n - 1 - i) ** 2))
    edge = left + right
    edge /= edge.sum()
    uniform = np.full(n, 1.0 / n)
    shape = w * edge + (1.0 - w) * uniform
    shape /= shape.sum()
    return shape


def _candidate_half_sinusoid(n: int, w: float) -> np.ndarray:
    """w·sin(π·(i+½)/n) (half‑wave) + (1‑w)·uniform."""
    i = np.arange(n, dtype=np.float64)
    sinus = np.sin(np.pi * (i + 0.5) / n)
    sinus = np.maximum(sinus, 0.0)
    sinus /= sinus.sum()
    uniform = np.full(n, 1.0 / n)
    shape = w * sinus + (1.0 - w) * uniform
    shape /= shape.sum()
    return shape


# ----------------------------------------------------------------------
# Mirror‑descent refinement (soft‑max sub‑gradient + Adam)
# ----------------------------------------------------------------------


def _refine_subgradient(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray | None, float]:
    """
    Refine a starting vector `a0` with a soft‑max based sub‑gradient.
    Uses an Adam optimiser on the log‑variables (exponential weights).
    """
    # initialise
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    # log‑domain variables for multiplicative updates
    log_a = np.log(a + 1e-16)

    # Adam parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # temperature schedule for the soft‑max (high → smooth, low → true max)
    T0 = 0.02          # initial temperature
    T_final = 0.001    # final temperature
    decay_T = (T_final / T0) ** (1.0 / 1e6)   # decay per iteration (~1e6 iters)

    # step‑size (learning rate) for log‑updates
    eta0 = 0.5

    it = 0
    last_improve = time.time()

    while time.time() < deadline:
        it += 1

        # convolution & objective
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c

        if val < best_val - 1e-12:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[n={n} refine iter={it:07d}] new best {val:.6f}")

        # ----- soft‑max weighting -----
        temperature = max(T_final, T0 * (decay_T ** it))
        # avoid overflow / underflow
        w_raw = np.exp((conv - max_c) / temperature)
        w = w_raw / w_raw.sum()

        # ----- gradient of weighted sum of convolutions -----
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)   # ∂F/∂a

        # ----- Adam on log‑variables -----
        # chain rule: ∂F/∂log_a = a * ∂F/∂a
        grad_log = grad * a

        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # adaptive learning rate
        eta = eta0 / (np.sqrt(v_hat) + eps)

        # log‑update
        log_a = log_a - eta * m_hat

        # back to primal, enforce simplex
        a = np.exp(log_a)
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s
        log_a = np.log(a + 1e-16)

        # occasional random shake if stagnated
        if time.time() - last_improve > 5.0:
            perturb = (np.random.rand(n) - 0.5) * 0.02  # log‑space jitter
            log_a = log_a + perturb
            a = np.exp(log_a)
            a = a / a.sum()
            log_a = np.log(a + 1e-16)
            last_improve = time.time()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a low‑evaluation‑score step function.
    Returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Primary grid size – 4096 gives a good trade‑off between quality and speed.
    # ------------------------------------------------------------------
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm start (if a previous best is stored globally)
    # ------------------------------------------------------------------
    warm_start = globals().get("height_sequence_1")
    if warm_start is not None:
        warm_start = np.asarray(warm_start, dtype=np.float64)
        # Trim / pad to length n
        if warm_start.shape[0] > n:
            warm_start = warm_start[:n]
        elif warm_start.shape[0] < n:
            warm_start = np.pad(warm_start, (0, n - warm_start.shape[0]))
        warm_start = np.maximum(warm_start, 0.0)
        if warm_start.sum() >= 0.01:
            warm_start = warm_start / warm_start.sum()
        else:
            warm_start = None
    else:
        warm_start = None

    # ------------------------------------------------------------------
    # Stage 1 – parametric sweep
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    evaluated = 0

    # Helper to evaluate a candidate and update best if needed
    def _eval_and_update(cand: np.ndarray):
        nonlocal best_seq, best_val, evaluated
        if time.time() >= deadline:
            return
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 0.01:                # re‑sample if degenerate
            cand = np.random.dirichlet(np.ones(n))
            s = cand.sum()
        cand = cand / s
        val = _objective(cand, n, conv_fft_len)
        evaluated += 1
        if val < best_val:
            best_val = val
            best_seq = cand.copy()
            print(f"[sweep #{evaluated:05d}] new best {val:.6f}")

    # 1) Warm‑start and simple rotations
    if warm_start is not None:
        _eval_and_update(warm_start)
        for shift in (n // 4, n // 2, 3 * n // 4):
            _eval_and_update(np.roll(warm_start, shift))

    # 2) Uniform baseline
    _eval_and_update(np.full(n, 1.0 / n))

    # 3) Edge‑exponential family
    alphas = np.logspace(-3, 1.0, num=30)          # 0.001 … 10
    ws = np.linspace(0.0, 1.0, num=11)           # mixture weight
    for alpha in alphas:
        for w in ws:
            cand = _candidate_edge_exponential(n, alpha, w)
            _eval_and_update(cand)
            if time.time() >= deadline:
                break
        if time.time() >= deadline:
            break

    # 4) Edge‑power‑law family
    ps = np.linspace(0.2, 5.0, num=30)
    for p in ps:
        for w in ws:
            cand = _candidate_edge_powerlaw(n, p, w)
            _eval_and_update(cand)
            if time.time() >= deadline:
                break
        if time.time() >= deadline:
            break

    # 5) Double‑Gaussian family
    gammas = (0.01, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0)
    for gamma in gammas:
        for w in ws:
            cand = _candidate_double_gaussian(n, gamma, w)
            _eval_and_update(cand)
            if time.time() >= deadline:
                break
        if time.time() >= deadline:
            break

    # 6) Half‑wave sinusoid + uniform mixture
    for w in ws:
        cand = _candidate_half_sinusoid(n, w)
        _eval_and_update(cand)
        if time.time() >= deadline:
            break

    # 7) Random Dirichlet draws (helps to explore non‑parametric regions)
    for conc in (0.005, 0.02, 0.07, 0.15, 0.4, 0.8):
        for _ in range(25):
            cand = np.random.dirichlet(np.full(n, conc))
            _eval_and_update(cand)
            if time.time() >= deadline:
                break
        if time.time() >= deadline:
            break

    if best_seq is None:
        # Fallback – uniform distribution (should never happen)
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = _objective(best_seq, n, conv_fft_len)
        print("*** No valid candidate found – falling back to uniform.")

    print(f"*** Sweep finished: evaluated {evaluated} candidates, best val≈{best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – deep refinement of the top‑k initialisations
    # ------------------------------------------------------------------
    # Keep a short list of the k best candidates from the sweep to
    # spend more time on them.
    # We re‑evaluate the sweep candidates (stored implicitly by best_seq)
    # by extracting the top few after sorting of values we already have.
    # For simplicity we reuse the best one only; the loop can be extended
    # to more starts if time permits.
    top_k = 3
    # Re‑run the sweep to collect top‑k (lightweight: only stored few best)
    # Since we already have best_seq, we will also add a few random Dirichlet
    # starts to diversify.
    extra_starts = []
    for _ in range(top_k - 1):
        extra = np.random.dirichlet(np.ones(n))
        extra_starts.append(extra)

    starts = [best_seq] + extra_starts

    for idx, a0 in enumerate(starts):
        if time.time() >= deadline:
            break
        print(f"--- Refinement from start #{idx + 1}/{len(starts)}")
        best_seq, best_val = _refine_subgradient(
            a0,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            best_seq,
            best_val,
        )

    # ------------------------------------------------------------------
    # Stage 3 – optional upscaling to a finer grid (n = 8192)
    # ------------------------------------------------------------------
    time_left = deadline - time.time()
    if time_left > 30.0 and best_seq is not None and best_seq.shape[0] == n:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        # Linear interpolation onto bigger grid
        x_old = np.linspace(0, 1, n, endpoint=False)
        x_big = np.linspace(0, 1, n_big, endpoint=False)
        a_big = np.interp(x_big, x_old, best_seq)
        a_big = np.maximum(a_big, 0.0)
        a_big /= a_big.sum()

        print("--- Upscaling to n=8192 and refinement")
        best_seq, best_val = _refine_subgradient(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            best_seq,
            best_val,
        )
        n = n_big  # final length

    # ------------------------------------------------------------------
    # Final safety clipping (evaluate_sequence rejects NaNs/negatives/inf,
    # clips >1000 automatically)
    # ------------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"*** Search finished: best objective ≈ {best_val:.6f} (n={best_seq.shape[0]})")
    return best_seq.tolist()
