    # Build constraints matrix A
    # Use for each shift k compute indices i where valid.
    # Use scipy.sparse to keep memory lower.
    import scipy.sparse as sp
    rows = []
    cols = []
    data = []
    for k in range(2*n-1):
        i_start = max(0, k - (n-1))
        i_end = min(k, n-1)
        # For i in range(i_start, i_end+1):
        #   A[k,i] = f[k-i]
        # Then j = k-i = index for g
        # So g index = j = k - i. Must be within [0,n-1]
        # Wait we want A shape (2n-1, n) multiply g (size n) to produce conv_k = sum_i f_i * g_{k-i}
        # So for each i, coefficient for g_{k-i} is f_i.
        # So we can loop over i as above and set entry A[k, k-i] = f_i.
        # Equivalent to for j in range(max(0,k-(n-1)), min(k,n-1)+1):
        #    i = k - j
        #    A[k,j] = f_i
        # So we can loop j.
        for j in range(i_start, i_end+1):
            i = k - j
            rows.append(k)
            cols.append(j)
            data.append(f[i])

    A = sp.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))

    # solve LP
    import cvxpy as cp
    g = cp.Variable(n, nonneg=True)
    constraints = [A @ g <= M]
    obj = cp.Maximize(cp.sum(g))
    prob = cp.Problem(obj, constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
    if g.value is None:
        return None
    return np.array(g.value)

def _lp_max_sum(f, M, deadline):
    # Solve max sum(g) subject to conv(f,g) <= M, g>=0.
    n = len(f)
    # Build sparse matrix in CSR format.
    from scipy.sparse import csr_matrix
    rows = []
    cols = []
    data = []
    # For k in [0, 2n-2]
    for k in range(2*n-1):
        i_start = max(0, k - (n-1))
        i_end = min(k, n-1)
        for j in range(i_start, i_end+1):
            i = k - j
            rows.append(k)
            cols.append(j)
            data.append(f[i])
    A = csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    # Linprog: maximize sum(g) => minimize -sum(g)
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    # Solve with high efficiency, possibly using method='highs-ds' if needed.
    # Provide max time? There's no direct param but we can check after solve.
    from scipy.optimize import linprog
    # Use options to limit iteration if needed but we rely on speed.
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        return None
    g = res.x
    # Ensure nonnegativity
    g = np.maximum(g, 0.0)
    return g

"""
Hybrid optimization for minimizing max self‑convolution.

The algorithm works in three phases:
1. A coarse optimisation on a modest grid (n≈1024) using Adam on a
   log‑sum‑exp surrogate and a sub‑gradient refinement.
2. A linear‑programming “exchange” step (maximise Σg subject to
   f∗g ≤ M) that flattens the autocorrelation.  The new vector is mixed
   with the current one.
3. Upscaling to a larger grid (≤8192) and a final fine‑grained Adam +
   hill‑climb.

All work is performed inside the 1000 s deadline; the best sequence is
returned as a plain Python list.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

# ------------------- FFT utilities -------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power‑of‑two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using a real FFT of length fft_len."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Return the evaluation function (ignoring the clamping performed later)."""
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c / (a.sum() ** 2))


# ------------------- Adam on smoothed max -------------------
def _adam_smooth_max(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    start_best_seq: np.ndarray | None,
    start_best_val: float,
) -> tuple[np.ndarray, float]:
    """Adam optimiser on a log‑sum‑exp smooth‑max surrogate."""
    eps = 1e-15
    logits = np.log(a0 + eps)               # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.07
    lam = 0.4
    lam_factor = 1.02
    lam_cap = 1e5

    best_seq = start_best_seq
    best_val = start_best_val
    last_improve = time.time()
    it = 0
    max_iter = 250_000

    while time.time() < deadline and it < max_iter:
        # simplex via soft‑max
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # convolution & surrogate objective
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[{n}] Adam it={it:06d} new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or not np.isfinite(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient w.r.t. a (before the soft‑max)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)               # ⟨a,∇⟩ term from the chain rule
        grad_log = a * (grad_a - avg)         # gradient wrt logits

        # Adam update in log‑space
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature schedule
        lam = min(lam * lam_factor, lam_cap)

        # occasional jitter if stuck
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1

    return best_seq, best_val


# ------------------- Sub‑gradient refinement -------------------
def _subgradient_refinement(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,

        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)

        if s < 1e-12:

        a /= s

def _lp_max_sum(f: np.ndarray, M: float, deadline: float) -> np.ndarray | None:
    """
    Solve   maximize Σ g_j   s.t.   (f ★ g)[k] ≤ M  for all k,
    g ≥ 0.  Returns the optimal g (may be all‑zero).
    """
    n = f.shape[0]
    rows = []
    cols = []
    data = []

    # Build sparse matrix A where (A @ g)[k] = (f ★ g)[k]
    for k in range(2 * n - 1):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        for j in range(i_start, i_end + 1):
            i = k - j
            rows.append(k)
            cols.append(j)
            data.append(f[i])

    A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

    c = -np.ones(n)                     # maximise Σ g → minimise −Σ g
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    # Early‑exit if we run out of time
    if time.time() >= deadline - 2.0:
        return None

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method="highs", options={"presolve": True})
    if not res.success:
        return None
    g = np.maximum(res.x, 0.0)
    return g

"""
Hybrid optimiser for the max‑self‑convolution problem.

The algorithm proceeds in three stages:

1. **Coarse optimisation** – a modest grid (n≈1024) is optimised with
   Adam on a log‑sum‑exp smooth‑max surrogate, followed by an
   exponentiated‑gradient sub‑gradient step.

2. **LP exchange step** – repeatedly solve a linear programme
   maximise Σ g  subject to f★g ≤ M (M = max f★f) and g≥0.
   The resulting g is mixed with the current f (mix≈0.25) and the bound
   is recomputed.  This flattens the autocorrelation.

3. **Upscaling & fine tuning** – the improved shape is interpolated to
   a larger grid (n=4096) and refined again with the Adam+sub‑gradient
   pipeline.  A short random hill‑climb finishes the search.

All work stops before the 1000 s deadline; the best non‑negative
sequence found is returned as a plain Python list.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
# FFT utilities
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using a real FFT of length fft_len."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Fast evaluation of the objective (scale‑invariant)."""
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c / (a.sum() ** 2))

# Adam optimiser on a smooth‑max surrogate
def _adam_smooth_max(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    start_best_seq: np.ndarray | None,
    start_best_val: float,
) -> tuple[np.ndarray, float]:
    eps = 1e-15
    logits = np.log(a0 + eps)               # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.07
    lam = 0.4
    lam_factor = 1.02
    lam_cap = 1e5

    best_seq = start_best_seq
    best_val = start_best_val
    last_improve = time.time()
    it = 0
    max_iter = 250_000

    while time.time() < deadline and it < max_iter:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[{n}] Adam it={it:06d} new best {val:.6f}")

        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or not np.isfinite(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        lam = min(lam * lam_factor, lam_cap)

        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1

    return best_seq, best_val

        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        eta = eta0 / np.sqrt(it + 1)

        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:
