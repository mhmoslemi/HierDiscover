import time
import numpy as np

def _next_pow2(x: int) -> int:
    """Return smallest power of two >= x."""
    return 1 << (x-1).bit_length()

def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using real FFT of length fft_len."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]

def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Cross‑correlation (w ★ a) needed for the sub‑gradient."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]

def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Fast evaluation of 2·n·max(conv) when sum(a)=1."""
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c)

def _subgradient_opt(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    init_best_seq: np.ndarray | None,
    init_best_val: float,
    max_iter: int = 200_000,
) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient sub‑gradient optimisation of max‑convolution.
    Works on the probability simplex (sum = 1)."""
    a = a0.copy()
    # enforce positivity and normalisation
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s

    best_seq = a.copy()
    best_val = init_best_val
    # initial best from init_best_seq if given
    if init_best_seq is not None:
        # compute its objective
        val0 = _objective(init_best_seq, n, conv_fft_len)
        if val0 < best_val:
            best_val = val0
            best_seq = init_best_seq.copy()

    eta0 = 0.6
    last_improve = time.time()

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = float(2.0 * n * max_c)  # sum(a)=1

        if val < best_val - 1e-12:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            # optional progress print
            print(f"[{n}] subgrad it={it:06d} new best {val:.6f}")

        # build weight vector w that isolates the active indices
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        # gradient of the current max (sub‑gradient)
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # exponentiated‑gradient update
        eta = eta0 / np.sqrt(it + 1.0)
        a = a * np.exp(-eta * grad)

        # renormalise onto the simplex
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        # early stop if stuck for a while
        if time.time() - last_improve > 8.0:
            break

    return best_seq, best_val

"""
Optimise a non‑negative step‑function for the black‑box evaluator.

The search proceeds in four stages:

1️⃣ **Seeding** – a large pool of random Dirichlet draws (various
   concentrations) together with a few deterministic shapes (uniform,
   warm‑start and asymmetric beta‑type) is generated.  All vectors are
   normalised to sum = 1 and scored with a cheap FFT‑based objective
   (2·n·max(conv) because the evaluator is scale‑invariant).

2️⃣ **Long sub‑gradient descent** – each of the best seeds is refined by
   an exponentiated‑gradient method that directly minimises the
   true max‑convolution.  The gradient is a cross‑correlation with a
   one‑hot weight on the active (max‑attaining) indices.  The loop runs
   until the time budget is exhausted or stagnation is detected.

3️⃣ **Upscaling** – if time permits the current best solution is linearly
   interpolated to a finer grid (n = 8192) and a further sub‑gradient
   refinement is performed.  The larger grid can reduce discretisation
   error and often yields a lower objective.

4️⃣ **Final hill‑climb** – a short random mass‑transfer local search
   (pairwise re‑allocation of probability mass) attempts a last tiny
   improvement.

All operations use NumPy/FFT, respect the 1000 s deadline and return a
list of floats suitable for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2·len(a)‑1) using an RFFT of length
    ``fft_len`` (must be ≥ 2·len(a)‑1).
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the sub‑gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """
    Fast evaluation of the objective when ``sum(a)=1``:
        2·n·max(a*a)
    """
    conv = _full_convolution(a, conv_fft_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
def _subgradient_opt(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iter: int = 1_000_000,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient minimisation of
    ``max(conv(a))`` on the probability simplex.
    """
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s

    best_seq = a.copy()
    best_val = _objective(a, n, conv_fft_len)

    eta0 = 0.6
    last_improve = time.time()

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = float(2.0 * n * max_c)

        if val < best_val - 1e-12:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[{n}] subgrad it={it:06d} new best {val:.6f}")

        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

            s = a.sum()
        a /= s

        if time.time() - last_improve > 8.0:
            break

        if time.time() >= deadline:
            break
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 1e-12:
            cand = np.random.dirichlet(np.ones(n))
            s = cand.sum()
        cand = cand / s
        val = _objective(cand, n, conv_fft_len)
        evaluated.append((val, cand))

    evaluated.sort(key=lambda x: x[0])
    top_k = min(25, len(evaluated))          # keep the most promising seeds
    seeds = [c for _, c in evaluated[:top_k]]
    print(f"Phase 1: generated {len(candidates)} seeds, kept {top_k} for refinement")

    # ------------------------------------------------------------------
    # ----------- Phase 2 : long sub‑gradient optimisation -------------
    best_seq = None
    best_val = float("inf")

    for start_id, a0 in enumerate(seeds):
        if time.time() >= deadline:
            break
        print(f"--- Sub‑gradient start #{start_id + 1}/{len(seeds)} ---")
        seq, val = _subgradient_opt(
            a0,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iter=200_000,
        )
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"*** New global best after start #{start_id+1}: {best_val:.6f}")

    # ------------------------------------------------------------------
    # --------------- Phase 3 : optional up‑scaling --------------------
    time_left = deadline - time.time()
    if time_left > 30.0:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)
        print("--- Upscaling to n=8192 for final polishing ---")
        if best_seq is not None:
            x_old = np.linspace(0, 1, n, endpoint=False)
            x_new = np.linspace(0, 1, n_big, endpoint=False)
            a_big = np.interp(x_new, x_old, best_seq)
        else:
            a_big = np.random.dirichlet(np.ones(n_big))
        a_big = np.maximum(a_big, 0.0)
        a_big /= a_big.sum()

        seq_big, val_big = _subgradient_opt(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            max_iter=120_000,
        )
        if val_big < best_val:
            best_val = val_big
            best_seq = seq_big.copy()
            n = n_big
            conv_fft_len = conv_fft_len_big
            corr_fft_len = corr_fft_len_big
            print(f"*** Improved on the larger grid: {best_val:.6f}")

    # ------------------------------------------------------------------
    # -------------------- Phase 4 : final hill‑climb ------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        a = best_seq.copy()
        best_local = best_val
        attempts = 0
        max_attempts = 10_000
        while time.time() < deadline and attempts < max_attempts:
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            max_transfer = min(a[i], a[j])
            if max_transfer <= 1e-14:
                continue
            delta = (np.random.rand() - 0.5) * 0.5 * max_transfer
            if delta > a[j]:
                delta = a[j]
            if -delta > a[i]:
                delta = -a[i]
            a[i] -= delta
            a[j] += delta
            conv = _full_convolution(a, conv_fft_len)
            val = float(2.0 * n * conv.max())
            if val < best_local:
                best_local = val
                best_seq = a.copy()
            else:
                a[i] += delta
                a[j] -= delta
            attempts += 1
        best_val = best_local
        print(f"Final hill‑climb finished after {attempts} attempts, best={best_val:.6f}")

    # ------------------------------------------------------------------
    # Safety fallback – should never be needed
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096)

    # Clip to the range accepted by the evaluator (0 … 1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"*** Search complete: best objective ≈ {best_val:.6f} (n={best_seq.shape[0]})")
    return best_seq.tolist()
