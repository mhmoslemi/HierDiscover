def _mass_transfer_refine(a, n, conv_fft_len, deadline, best_val):
    # Precompute convolution via FFT for current a
    conv = _full_convolution(a, conv_fft_len)
    current_val = _objective(a, n, conv_fft_len)
    best_seq = a.copy()
    best_val = current_val
    while time.time() < deadline:
        # choose src and dst indices; src from top 10% largest values
        # choose epsilon = random fraction of source value * factor 0.01
        src = np.random.choice(n, p=prob_src)  # probability proportional to a_i
        dst = np.random.randint(n)  # any index diff
        eps = a[src] * (np.random.rand()*0.02)  # up to 2% of src value
        # ensure nonneg
        if eps <= 0: continue
        a_new = a.copy()
        a_new[src] -= eps
        a_new[dst] += eps
        # if src==dst continue
        # renorm not needed because sum unchanged.
        # compute objective quickly: use convolution via FFT each iteration (~0.004 s)
        val = _objective(a_new, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_seq = a_new
            a = a_new
            # possibly update prob_src
            # print progress
    return best_seq, best_val
```

"""
Propose a low‑objective non‑negative step function.

The algorithm consists of four phases:
  1. Generate a rich pool of parametric candidates (uniform, Dirichlet,
     symmetric/asymmetric exponentials and power‑laws, simple windows,
     triangular/linear shapes, sparse spikes, …) and keep the top few.
  2. For each top start run a short Adam optimisation on a log‑sum‑exp
     surrogate of the max‑convolution, followed by a modest exponentiated‑
     gradient sub‑gradient phase.
  3. Take the globally best vector and spend the remaining budget on an
     extended sub‑gradient refinement and a cheap stochastic hill‑climb.
  4. Finally perform a pure “mass‑transfer” local search that repeatedly
     moves a small fraction of probability mass from a randomly chosen
     source index to a destination index; the exact objective is
     evaluated after every move and improvements are kept.

All operations respect the evaluator’s clipping (non‑negative, ≤ 1000) and
are performed on the simplex (∑a_i = 1), which makes the objective
scale‑invariant.  The code is written for a single process but respects the
global time‑budget (default 1000 s) and stops early if needed.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2*len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact evaluation  F = 2·n·max(a*a) / (sum(a)²).  Here sum(a)=1."""
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c / (a.sum() ** 2))


# ----------------------------------------------------------------------
def _adam_smooth_max(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    init_best_seq: np.ndarray | None,
    init_best_val: float,
    max_iter: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Adam on a log‑sum‑exp surrogate of the max‑convolution.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05

    lam = 2.0                # start temperature
    lam_factor = 1.012
    lam_cap = 1e6

    best_seq = init_best_seq
    best_val = init_best_val

    it = 0
    last_improve = time.time()

    while time.time() < deadline and it < max_iter:
        # ----- simplex via soft‑max -----
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # ----- exact objective (also needed for early‑stop) -----
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Adam] iter={it:06d} new best {val:.6f}")

        # ----- smooth‑max weights -----
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of surrogate -----
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ----- Adam update on logits -----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature schedule
        lam = min(lam * lam_factor, lam_cap)

        # occasional random shake if stuck for several seconds
        if time.time() - last_improve > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1

    return best_seq, best_val


def _subgradient_refinement(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    init_best_seq: np.ndarray | None,
    init_best_val: float,
    max_iter: int = 150_000,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient optimisation on the exact max‑convolution.
    """
    a = init_best_seq if init_best_seq is not None else a0.copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a /= s  # stay on simplex

    eta0 = 0.8
    decay = 0.99995
    eta = eta0

    best_seq = a.copy()
    best_val = init_best_val

    it = 0
    last_improve = time.time()

    while time.time() < deadline and it < max_iter:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Subgrad] iter={it:06d} new best {val:.6f}")

        # active set where convolution attains its maximum
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # exponentiated‑gradient update (keeps a ≥ 0)
        a = a * np.exp(-eta * grad_a)

        # renorm – protect against collapse
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        # decay step size
        eta *= decay
        it += 1

        # early stop if no progress for a while
        if time.time() - last_improve > 6.0:
            break

    return best_seq, best_val


def _deep_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Run a short Adam phase followed by a modest sub‑gradient phase.
    Returns the best sequence/value observed in this run.
    """
    # Adam – a few‑tens of thousands of iterations (fast)
    best_seq, best_val = _adam_smooth_max(
        a0,
        n,
        conv_fft_len,
        corr_fft_len,
        deadline,
        global_best_seq,
        global_best_val,
        max_iter=30_000,
    )
    # Sub‑gradient – a few‑tens of thousands of iterations
    best_seq, best_val = _subgradient_refinement(
        a0,
        n,
        conv_fft_len,
        corr_fft_len,
        deadline,
        best_seq,
        best_val,
        max_iter=15_000,
    )
    return best_seq, best_val

    Produce a diverse pool of candidate vectors.
    Families include uniform, Dirichlet, symmetric/asymmetric exponentials,
    power‑laws, simple windows (Hann, Hamming, Blackman, Gaussian),
    linear / triangular, sinusoidal half‑wave, and sparse spikes.
    All candidates are normalized to sum ≈ 1.
    """
    candidates = []

    # ----- warm‑start & shifts -----
    if warm_start is not None:
        candidates.append(warm_start.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            candidates.append(np.roll(warm_start, shift))

    # ----- uniform baseline -----
    candidates.append(np.full(n, 1.0 / n))

    # ----- random Dirichlet draws (various concentrations) -----
    for conc in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(12):
            cand = np.random.dirichlet(np.full(n, conc))
            candidates.append(cand)

    i = np.arange(n, dtype=np.float64)

    # ----- symmetric edge‑exponential family -----
    for alpha in np.logspace(-2, 1.0, num=10):      # 0.01 … 10
        edge = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
        edge /= edge.sum()
        for w in (0.0, 0.25, 0.5, 0.75, 1.0):
            shape = w * edge + (1.0 - w) / n
            candidates.append(shape)

    # ----- symmetric edge‑power‑law family -----
    for p in np.linspace(0.2, 6.0, num=10):
        edge = (i + 1.0) ** (-p) + (n - i) ** (-p)
        edge /= edge.sum()
        for w in (0.0, 0.3, 0.6, 1.0):
            shape = w * edge + (1.0 - w) / n
            candidates.append(shape)

    # ----- asymmetric two‑sided exponentials -----
    alphas = np.logspace(-2, 1.0, num=6)
    w_vals = np.linspace(0.0, 1.0, 5)
    for wL in w_vals:
        for wR in w_vals:
            if wL + wR > 1.0:
                continue
            wU = 1.0 - wL - wR
            for aL in alphas:
                for aR in alphas:
                    left = np.exp(-aL * i)
                    right = np.exp(-aR * (n - 1 - i))
                    shape = wL * left + wR * right + wU / n
                    shape /= shape.sum()
                    candidates.append(shape)

    # ----- asymmetric power‑law -----
    p_vals = np.linspace(0.2, 5.0, num=6)
    for wL in w_vals:
        for wR in w_vals:
            if wL + wR > 1.0:
                continue
            wU = 1.0 - wL - wR
            for pL in p_vals:
                for pR in p_vals:
                    left = (i + 1.0) ** (-pL)
                    right = (n - i) ** (-pR)
                    shape = wL * left + wR * right + wU / n
                    shape /= shape.sum()
                    candidates.append(shape)

    # ----- simple linear / triangular shapes -----
    lin = np.linspace(1.0, 0.0, n)
    tri = np.concatenate([np.linspace(0, 1, n // 2, endpoint=False),
                          np.linspace(1, 0, n - n // 2)])
    for base in (lin, tri):
        base = np.maximum(base, 0.0)
        base /= base.sum()
        for w in (0.0, 0.4, 0.8):
            cand = w * base + (1.0 - w) / n
            cand /= cand.sum()
            candidates.append(cand)

    # ----- sinusoidal half‑wave -----
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    candidates.append(shape)

    # ----- Hann, Hamming and Blackman windows (with uniform mixing) -----
    theta = 2 * np.pi * i / (n - 1)
    hann = 0.5 - 0.5 * np.cos(theta)
    hamming = 0.54 - 0.46 * np.cos(theta)
    blackman = 0.42 - 0.5 * np.cos(theta) + 0.08 * np.cos(2 * theta)
    for win in (hann, hamming, blackman):
        win = np.maximum(win, 0.0)
        win /= win.sum()
        for w in (0.0, 0.2, 0.5, 0.8, 1.0):
            cand = w * win + (1.0 - w) / n
            cand /= cand.sum()
            candidates.append(cand)

    # ----- Gaussian windows (different sigmas) -----
    centre = (n - 1) / 2.0
    for sigma in (0.20, 0.30, 0.45):
        dist = (i - centre) / (sigma * centre)
        gauss = np.exp(-0.5 * dist ** 2)
        gauss = np.maximum(gauss, 0.0)
        gauss /= gauss.sum()
        for w in (0.0, 0.25, 0.5, 0.75, 1.0):
            cand = w * gauss + (1.0 - w) / n
            cand /= cand.sum()
            candidates.append(cand)

    # ----- sparse spike vectors -----
    for spikes in (3, 5, 8):
        for _ in range(20):
            idx = np.random.choice(n, size=spikes, replace=False)
            weights = np.random.dirichlet(np.ones(spikes))
            vec = np.zeros(n)
            vec[idx] = weights
            vec /= vec.sum()
            candidates.append(vec)

    return candidates


# ----------------------------------------------------------------------
def _local_random_search(
    a_best: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climbing: add Gaussian noise to the logits of the current
    best vector, evaluate the exact objective and keep the improvement.
    Runs while time permits.
    """
    logits_best = np.log(a_best + 1e-15)

    while time.time() < deadline:
        noise = np.random.randn(n) * 0.03
        logits = logits_best + noise
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        val = _objective(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            a_best = a.copy()
            logits_best = np.log(a_best + 1e-15)
            print(f"[Local search] improved to {val:.6f}")

    return a_best, best_val


# ----------------------------------------------------------------------
def _mass_transfer_refine(
    a: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Pure mass‑transfer hill‑climbing:
    repeatedly move a small random fraction of probability mass from a
    source index to a destination index and keep improvements.
    """
    best_seq = a.copy()
    best = best_val
    # source distribution – initially proportional to current mass
    src_prob = a / a.sum()

    while time.time() < deadline:
        src = np.random.choice(n, p=src_prob)
        dst = np.random.randint(0, n)
        if src == dst:
            continue

        eps = a[src] * np.random.rand() * 0.04   # up to 4 % of src weight
        if eps <= 0.0:
            continue

        a_new = a.copy()
        a_new[src] -= eps
        a_new[dst] += eps

        # ensure non‑negative (src ≥ eps)
        if a_new[src] < 0.0:
            continue

        val = _objective(a_new, n, conv_fft_len)
        if val < best:
            best = val
            best_seq = a_new.copy()
            a = a_new
            src_prob = a / a.sum()
            print(f"[Transfer] new best {val:.6f}")

    return best_seq, best


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Main entry point – searches for a non‑negative step function that
    minimises the supplied ``evaluate_sequence`` objective.
    Returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Work with n = 4096 (a good compromise between quality and speed)
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Optional warm‑start from a previously found construction
    warm_start = globals().get("height_sequence_1")
    if warm_start is not None:
        warm_start = np.asarray(warm_start, dtype=np.float64)
        if warm_start.shape[0] > n:
            warm_start = warm_start[:n]
        elif warm_start.shape[0] < n:
            warm_start = np.pad(warm_start, (0, n - warm_start.shape[0]))
        warm_start = np.maximum(warm_start, 0.0)
        s = warm_start.sum()
        if s >= 0.01:
            warm_start = warm_start / s
        else:
            warm_start = None
    else:
        warm_start = None

    # ------------------------------------------------------------------
    # Stage 1 – generate a large pool of inexpensive candidates
    all_candidates = _generate_parametric_candidates(n, warm_start)

    cand_vals = []
    for cand in all_candidates:
        if time.time() >= deadline:
            break
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 0.01:          # safety net
            cand = np.random.dirichlet(np.ones(n))
            s = cand.sum()
        cand = cand / s
        val = _objective(cand, n, conv_fft_len)
        cand_vals.append((val, cand))

    # keep the top K for the expensive optimisation phase
    cand_vals.sort(key=lambda x: x[0])
    top_k = 10
    top_candidates = [c for _, c in cand_vals[:top_k]]
    print(f"Stage 1: evaluated {len(cand_vals)} candidates, best so far = {cand_vals[0][0]:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – quick deep optimisation on each top start
    for start_id, a0 in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        print(f"--- Deep optimisation on start #{start_id + 1}/{len(top_candidates)} ---")
        best_seq, best_val = _deep_optimize(
            a0,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            best_seq,
            best_val,
        )

    # ------------------------------------------------------------------
    # Stage 3 – extended sub‑gradient refinement on the best vector seen so far
    if best_seq is not None:
        print("--- Extended sub‑gradient refinement ---")
        best_seq, best_val = _subgradient_refinement(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            best_seq,
            best_val,
            max_iter=120_000,          # longer run if time allows
        )

    # ------------------------------------------------------------------
    # Stage 4 – cheap stochastic hill‑climbing in log‑space
    if best_seq is not None:
        print("--- Local random perturbation refinement ---")
        best_seq, best_val = _local_random_search(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            best_val,
        )

    # ------------------------------------------------------------------
    # Stage 5 – mass‑transfer hill‑climbing (pure exact objective)
    if best_seq is not None:
        print("--- Mass‑transfer refinement ---")
        best_seq, best_val = _mass_transfer_refine(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            best_val,
        )

    # ------------------------------------------------------------------
    # Stage 6 – if there is enough time left, try a finer grid (n = 8192)
    time_left = deadline - time.time()
    if time_left > 40.0:               # need a meaningful budget
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        if best_seq is not None:
            # linear interpolation from the current grid to the bigger one
            x_old = np.linspace(0, 1, n, endpoint=False)
            x_new = np.linspace(0, 1, n_big, endpoint=False)
            a_big = np.interp(x_new, x_old, best_seq)
            a_big = np.maximum(a_big, 0.0)
            a_big /= a_big.sum()
        else:
            a_big = np.random.dirichlet(np.ones(n_big))

        print("--- Optimising on larger grid (n=8192) ---")
        best_seq, best_val = _deep_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            best_seq,
            best_val,
        )
        # Keep the larger vector if it truly improved the bound
        if best_seq is not None and best_seq.shape[0] == n_big:
            n = n_big

    # ------------------------------------------------------------------
    # Fallback – if nothing succeeded, return a uniform vector.
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Clip to the evaluator’s allowed range (non‑negative, ≤ 1000) and return.
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Optional sanity check (does not affect the returned value)
    try:
        from __main__ import evaluate_sequence  # the evaluator is injected by the harness
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"*** Finished search: best objective ≈ {final_val:.6f} (n={best_seq.shape[0]})")
    except Exception:
        print(f"*** Finished search (no external evaluation): best internal objective ≈ {best_val:.6f}")

    return best_seq.tolist()
