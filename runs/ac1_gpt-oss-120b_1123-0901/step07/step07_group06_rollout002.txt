"""
Hybrid LP + gradient‐descent search for a low‑evaluation step function.

The algorithm has two phases.

1. **LP phase (small dimension).**  
   Starting from a warm‑start (the previous best sequence if available) we
   repeatedly
   * compute M = max(f * f),
   * solve the linear program  
        maximise   Σ g_i
        subject to 0 ≤ g,  f * g ≤ M                (convolution bound)
   * if the LP yields a larger ℓ₁‑mass we scale g back to the current total
     mass and replace f by a convex combination f←(1‑t) f + t g.

   The LP enlarges the feasible ℓ₁‑mass while keeping the worst
   self‑convolution unchanged, which provably does not increase the
   evaluation value.  A few iterations already improve the shape dramatically.

2. **Refinement phase (large dimension).**  
   The improved small‑scale vector is linearly interpolated to n = 4096 and
   refined with a fast exponentiated‑gradient/sub‑gradient method that works
   directly on the exact objective.  FFTs give O(n log n) per iteration,
   allowing hundreds of thousands of updates within the 1000 s budget.

The whole pipeline respects the time limit, prints intermediate bounds and
finally returns the best sequence as a plain list of non‑negative floats.
"""

import time
import numpy as np
import scipy.optimize as opt
import scipy.sparse as sp

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation_fft(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("v must be 1‑d")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _build_conv_matrix(f: np.ndarray) -> sp.csr_matrix:
    """
    Build the matrix A (size (2n‑1) × n) such that A @ g = f * g
    (linear convolution).  The sparsity pattern is independent of f,
    only the data values (the coefficients) change.
    """
    n = f.shape[0]
    # rows: for each column i the rows i … i+n‑1
    rows = (np.arange(n)[:, None] + np.arange(n)).ravel()
    cols = np.repeat(np.arange(n), n)
    data = np.repeat(f, n)          # each coefficient f[i] appears n times
    return sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


def _lp_step(f: np.ndarray, M: float, A_fixed: sp.csr_matrix) -> np.ndarray | None:
    """
    Solve  max Σ g_i  s.t.  g ≥ 0,  A_fixed @ g ≤ M·1 .
    Returns the optimal g (scaled to the original total mass) or None if
    the LP gives no improvement.
    """
    n = f.shape[0]
    c = -np.ones(n, dtype=np.float64)                   # maximise Σg_i → minimise −Σg_i
    b = np.full(2 * n - 1, M, dtype=np.float64)        # each convolution entry ≤ M
    res = opt.linprog(c, A_ub=A_fixed, b_ub=b,
                      bounds=[(0, None)] * n, method='highs')
    if not res.success:
        return None
    g_raw = res.x
    sum_g = g_raw.sum()
    sum_f = f.sum()
    if sum_g <= sum_f + 1e-12:      # no increase in ℓ₁‑mass → stop
        return None
    # Scale back to the current total mass (maintain Σ = Σf)
    return g_raw * (sum_f / sum_g)


def _lp_phase(initial: np.ndarray,
              deadline: float,
              t_mix: float = 0.2,
              max_iters: int = 30) -> np.ndarray:
    """
    Run the LP‑based improvement loop on a vector of length n_lp (≈1024).
    Returns the improved vector.
    """
    f = initial.copy()
    n = f.shape[0]

    # The sparsity pattern of the convolution matrix never changes.
    A_fixed = _build_conv_matrix(f)   # placeholders – data will be overwritten each iteration

    for it in range(max_iters):
        if time.time() > deadline - 5.0:
            break

        # current worst self‑convolution
        conv_ff = np.convolve(f, f)
        M = conv_ff.max()

        # update matrix data (the pattern stays the same)
        A_fixed.data = np.repeat(f, n)

        g = _lp_step(f, M, A_fixed)
        if g is None:
            break

        # convex combination and re‑normalise
        f = (1.0 - t_mix) * f + t_mix * g
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s > 0:
            f /= s
        else:                                   # fallback – uniform
            f = np.full(n, 1.0 / n)

        # slowly decay mixing factor
        t_mix = max(0.05, t_mix * 0.96)

        # report progress
        val = 2.0 * n * np.max(np.convolve(f, f)) / (f.sum() ** 2)
        print(f"[LP it={it:02d}]  n={n}  val≈{val:.6f}")

    return f


# ----------------------------------------------------------------------
def _gradient_refine(a: np.ndarray,
                     deadline: float,
                     max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient / sub‑gradient refinement on the exact objective.
    Works on the full size (default 4096).  Returns the best vector seen and its value.
    """
    n = a.shape[0]
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # initialise
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum

    best_seq = a.copy()
    best_val = 2.0 * n * _full_convolution_fft(a, conv_fft_len).max() / (a.sum() ** 2)

    eta = 0.6                       # initial step size for exponentiated‑gradient
    last_improve = time.time()
    it = 0

    while it < max_iters and time.time() < deadline:
        # convolution and current bound
        conv = _full_convolution_fft(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val + 1e-12 < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Grad n={n} it={it:06d}] new best {val:.6f}")

        # active set where the max is attained
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        # gradient of the (exact) max‑convolution term
        grad = 2.0 * _cross_correlation_fft(w, a, corr_fft_len)

        # exponentiated‑gradient step (keeps entries non‑negative)
        a = a * np.exp(-eta * grad)

        # project back onto the simplex (simple renormalisation is enough)
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a = np.full(n, 1.0 / n)

        # step‑size schedule – decay if we are stuck
        if time.time() - last_improve > 3.0:
            eta = max(1e-5, eta * 0.85)
        else:
            eta = min(2.0, eta * 1.001)

        it += 1

        # very early break if learning rate is tiny and no recent improvement
        if eta <= 1e-5 and time.time() - last_improve > 8.0:
            break

    return best_seq, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:

        # shrink / expand to the working size of the LP phase
        n_lp = 1024
        if warm.shape[0] != n_lp:
            x_old = np.linspace(0, 1, warm.shape[0], endpoint=False)
            x_new = np.linspace(0, 1, n_lp, endpoint=False)
            warm = np.interp(x_new, x_old, warm)
        f0 = warm.copy()
        f0 = np.maximum(f0, 0.0)

        if s0 > 0:
            f0 /= s0
        else:
            f0 = np.full(n_lp, 1.0 / n_lp)

        n_lp = 1024
        i = np.arange(n_lp, dtype=np.float64)
        edge = np.exp(-0.1 * i) + np.exp(-0.1 * (n_lp - 1 - i))
        f0 = edge / edge.sum()

    # --------------------------------------------------------------
    # 2️⃣  LP improvement phase (small dimension)
    print("[*] Starting LP improvement phase …")
    f_improved = _lp_phase(f0, deadline)

    a_start = np.maximum(a_start, 0.0)
