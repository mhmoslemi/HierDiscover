"""
Propose a non‑negative step‑function that (approximately) minimises

        F(a) = 2·n·max(a∗a) / (∑a)^2 .

The algorithm is a lightweight multi‑stage optimiser:

1.   Build a modest pool of diverse start vectors (Dirichlet, power‑law,
     double‑exponential, sinusoid, uniform, and spike patterns).  If a
     previous solution exists (`height_sequence_1`) it is used as a warm
     start.

2.   For each start vector run a short Adam optimiser on the smooth‑max
     (log‑sum‑exp) surrogate of the convolution maximum.  The parameters
     (learning‑rate, temperature schedule) are tuned for speed.

3.   Refine the best candidate with a longer exponentiated‑gradient
     sub‑gradient method that incorporates a momentum term (exponential
     moving‑average of the gradient) and a decaying step size.

4.   Upsample the best vector to a finer grid (8192 points) and repeat the
     sub‑gradient phase for many iterations (hundreds of thousands).

5.   Perform a fast simulated‑annealing style hill‑climbing phase on the
     finer grid.  The auto‑convolution is updated *incrementally* for each
     random mass transfer, avoiding a full FFT per move.

6.   If time permits, upscale once more to 16384 points and repeat a
     brief Adam + annealing refinement.

All phases respect the global time budget (default 1000 s) and the
function always returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a length‑len(a) vector:
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
def _initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Generate a modest set of diverse start vectors."""
    cand = []

    # warm start and a few rotations
    if warm is not None:
        cand.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    # Dirichlet (uniform concentration)
    cand.append(np.random.dirichlet(np.ones(n)))

    # Dirichlet with small concentration → spikier vectors (α=0.2)
    cand.append(np.random.dirichlet(0.2 * np.ones(n)))

    # Edge‑concentrated power‑law shapes
    idx = np.arange(n, dtype=np.float64) + 1.0
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-3.0, -0.2, 13):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    # Symmetric double‑exponential shapes
    i = np.arange(n, dtype=np.float64)
    for gamma in np.geomspace(0.2, 5.0, 8):
        shape = np.exp(-gamma * i) + np.exp(-gamma * (n - 1 - i))
        shape /= shape.sum()
        cand.append(shape)

    # Positive half‑wave sinusoid
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand.append(shape)

    # Uniform
    cand.append(np.full(n, 1.0 / n))

    return cand


def _spike_candidates(n: int) -> list[np.ndarray]:
    """Very sparse “spike’’ candidates with k equally spaced non‑zero entries."""
    out = []
    for k in (32, 64, 128, 256, 512):
        a = np.zeros(n, dtype=np.float64)
        if k > 0:
            pos = np.round(np.linspace(0, n - 1, k)).astype(int)
            a[pos] = 1.0 / k
        out.append(a)
    return out


# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max (log‑sum‑exp) surrogate.
    Returns the best vector (normalised to sum = 1) encountered and its
    exact objective value.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.12                    # a bit larger than before
    lam = 0.5                    # start temperature
    lam_factor = 1.05            # grow faster to true max

    best_a = a0.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max → simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective (sum(a)=1)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            # occasional progress report
            if it % 10_000 == 0:
                print(f"[Adam iter={it:06d}] new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate (chain‑rule through soft‑max)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        lam = min(lam * lam_factor, 1e6)

        # very occasional tiny random shake when time is short
        if it % 7_000 == 0 and time.time() + 5 > deadline:
            logits += (np.random.rand(n) - 0.5) * 5e-3

        it += 1

    return best_a, best_val


def _subgradient_momentum(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
    eta0: float = 0.6,
    beta: float = 0.9,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient with momentum (EMA of the gradient).
    Returns the best vector seen and its exact objective.
    """
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s

    grad_ma = np.zeros_like(a)          # exponential moving average of gradient
    best_a = a.copy()
    best_val = float("inf")
    stall = 0
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            stall = 0
            if it % 10_000 == 0:
                print(f"[SubgradM iter={it:07d}] new best {val:.6f}")
        else:
            stall += 1

        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # EMA of the gradient
        grad_ma = beta * grad_ma + (1.0 - beta) * grad

        # decaying step size
        eta = eta0 / np.sqrt(it + 1.0)

        # multiplicative update and renormalisation
        a = a * np.exp(-eta * grad_ma)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        # occasional random perturbation if we have been stuck
        if stall > 15_000:
            a = np.maximum(a + 0.02 * (np.random.rand(n) - 0.5), 0.0)
            a /= a.sum()
            stall = 0

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
def _annealing_random_moves(
    a: np.ndarray,
    n: int,
    conv: np.ndarray,
    deadline: float,
    max_moves: int = 200_000,
    init_temp_factor: float = 0.05,
    cooling: float = 0.99995,
    max_frac: float = 0.45,
) -> tuple[np.ndarray, float]:
    """
    Very cheap simulated‑annealing hill‑climber.
    The auto‑convolution `conv` (length 2·n‑1) is updated incrementally for
    each random mass transfer.  Returns the best vector and its exact
    objective value.
    """
    best_a = a.copy()
    best_val = 2.0 * n * conv.max()
    T = best_val * init_temp_factor
    moves = 0

    while time.time() < deadline and moves < max_moves:
        i = np.random.randint(n)
        if a[i] <= 1e-15:
            moves += 1
            continue
        j = np.random.randint(n)
        if i == j:
            moves += 1
            continue

        delta = a[i] * (np.random.rand() * max_frac)   # amount to move
        if delta < 1e-15:
            moves += 1
            continue

        # make a tentative copy of the convolution
        conv2 = conv.copy()
        # linear updates (see derivation in analysis)
        conv2[j:j + n] += 2.0 * delta * a
        conv2[i:i + n] -= 2.0 * delta * a
        conv2[2 * j] += delta * delta
        conv2[2 * i] += delta * delta
        conv2[i + j] -= 2.0 * delta * delta

        # evaluate the tentative move
        val2 = 2.0 * n * conv2.max()

        accept = False
        if val2 < best_val:
            accept = True
        else:
            prob = np.exp(-(val2 - best_val) / (T + 1e-30))
            if np.random.rand() < prob:
                accept = True

        if accept:
            # commit the move
            a[i] -= delta
            a[j] += delta
            conv = conv2
            if val2 < best_val:
                best_val = val2
                best_a = a.copy()
        # cooling
        T *= cooling
        moves += 1

        # renormalise occasionally to avoid drift
        if moves % 10_000 == 0:
            s = a.sum()
            if abs(s - 1.0) > 1e-12:
                a /= s
                conv = _full_convolution(a, _next_pow2(2 * n - 1))

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The function respects the given time budget and returns the best
    sequence (as a Python list) found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.3   # tiny safety margin

    # ------------------------------------------------------------------
    # Phase 0 – cheap optimisation on a moderate grid (n = 2048)
    n0 = 2048
    conv_fft_len0 = _next_pow2(2 * n0 - 1)
    corr_fft_len0 = _next_pow2(3 * n0 - 2)

    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n0:
            warm = warm[:n0]
        elif warm.shape[0] < n0:
            warm = np.pad(warm, (0, n0 - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None

    candidates = _initial_candidates(n0, warm) + _spike_candidates(n0)

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # run a quick Adam + short sub‑gradient on each candidate
    for cid, cand in enumerate(candidates):
        if time.time() >= deadline:
            break
        a0 = np.maximum(cand, 0.0)
        if a0.sum() < 0.01:
            a0 = np.random.dirichlet(np.ones(n0))
        else:
            a0 = a0 / a0.sum()

        # short Adam (≈6 s per candidate)
        cand_deadline = min(deadline - 1.0, time.time() + 6.0)
        a_adam, val_adam = _adam_optimize(
            a0, n0, conv_fft_len0, corr_fft_len0, cand_deadline, max_iters=120_000
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[cand {cid}] Adam best {val_adam:.6f}")

        # short momentum sub‑gradient (≈8 s)
        if time.time() < deadline:
            sub_deadline = min(deadline - 1.0, time.time() + 8.0)
            a_sub, val_sub = _subgradient_momentum(
                best_seq, n0, conv_fft_len0, corr_fft_len0, sub_deadline, max_iters=80_000
            )
            if val_sub < best_val:
                best_val = val_sub
                best_seq = a_sub.copy()
                print(f"[cand {cid}] SubgradM best {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – upscale to 8192 and long momentum sub‑gradient
    if best_seq is None:
        best_seq = np.full(n0, 1.0 / n0, dtype=np.float64)
        best_val = 2.0 * n0 * _full_convolution(best_seq, conv_fft_len0).max()

    n1 = 8192
    conv_fft_len1 = _next_pow2(2 * n1 - 1)
    corr_fft_len1 = _next_pow2(3 * n1 - 2)

    # linear interpolation to the finer grid
    x_old = np.linspace(0.0, 1.0, n0, endpoint=False)
    x_new = np.linspace(0.0, 1.0, n1, endpoint=False)
    a_big = np.interp(x_new, x_old, best_seq)
    a_big = np.maximum(a_big, 0.0)
    a_big /= a_big.sum()

    if time.time() + 5 < deadline:
        a_ref, val_ref = _subgradient_momentum(
            a_big, n1, conv_fft_len1, corr_fft_len1, deadline, max_iters=400_000
        )
        if val_ref < best_val:
            best_val = val_ref
            best_seq = a_ref.copy()
            print(f"[Refine8192] SubgradM best {val_ref:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – cheap annealing moves on the 8192‑grid
    if time.time() + 5 < deadline:
        conv_big = _full_convolution(best_seq, conv_fft_len1)
        remaining = deadline - time.time()
        # estimate moves we can afford (≈0.0004 s per move)
        max_moves = int(min(300_000, remaining / 0.00045))
        if max_moves > 0:
            a_best, val_best = _annealing_random_moves(
                best_seq, n1, conv_big, deadline, max_moves=max_moves
            )
            if val_best < best_val:
                best_val = val_best
                best_seq = a_best.copy()
                print(f"[Anneal8192] improved best {val_best:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – optional upscale to 16384 if time permits
    if time.time() + 30 < deadline:
        n2 = 16384
        conv_fft_len2 = _next_pow2(2 * n2 - 1)
        corr_fft_len2 = _next_pow2(3 * n2 - 2)

        x_old = np.linspace(0.0, 1.0, best_seq.shape[0], endpoint=False)
        x_new = np.linspace(0.0, 1.0, n2, endpoint=False)
        a_big2 = np.interp(x_new, x_old, best_seq)
        a_big2 = np.maximum(a_big2, 0.0)
        a_big2 /= a_big2.sum()

        # short Adam on the very fine grid
        if time.time() + 15 < deadline:
            sub_dead = min(deadline - 1.0, time.time() + 12.0)

                a_big2, n2, conv_fft_len2, corr_fft_len2, sub_dead, max_iters=120_000
            )
            if val_adam2 < best_val:
                best_val = val_adam2
                best_seq = a_adam2.copy()
                print(f"[Big16384] Adam best {val_adam2:.6f}")

        # final annealing on the 16384‑grid
        if time.time() + 10 < deadline:
            conv_big2 = _full_convolution(best_seq, conv_fft_len2)
            remaining = deadline - time.time()
            max_moves2 = int(min(200_000, remaining / 0.00055))
            if max_moves2 > 0:
                a_best2, val_best2 = _annealing_random_moves(
                    best_seq, n2, conv_big2, deadline, max_moves=max_moves2
                )
                if val_best2 < best_val:
                    best_val = val_best2
                    best_seq = a_best2.copy()
                    print(f"[Anneal16384] improved best {val_best2:.6f}")

    # ------------------------------------------------------------------
    # Final safety – make sure we have a valid return
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Clip to evaluator limits (non‑negative, ≤1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
