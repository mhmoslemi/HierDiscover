def project_simplex(v):
    """Project onto the probability simplex: x_i >= 0, sum x_i = 1."""
    # Classic algorithm O(n log n): sort descending.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    # Find rho = max{i | u_i > cssv_i / i}
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

Propose a low‑value step function.

Algorithm
---------
1. **Candidate pool** – generate a diverse set of normalized non‑negative
   vectors (uniform, many Dirichlet samples, anti‑pair ordering,
   edge‑focused shapes, triangular shape, etc.) and keep the best

# FFT utilities
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) – used for sub‑gradient computation."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Project a vector onto the probability simplex {x ≥ 0, sum x = 1}.
    Implementation of the algorithm from:
    Wang & Carreira‑Perpiñán, “Projection onto the Probability Simplex”.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# Objective (exact, without the outer checks)
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(a*a) / (sum(a)^2).  Assumes a is non‑negative."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    n = a.shape[0]
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    return 2.0 * n * conv.max() / (s * s)

def _anti_pair_arrangement(vals: np.ndarray) -> np.ndarray:
    """Largest opposite smallest, second‑largest opposite second‑smallest …"""
    n = len(vals)
    sorted_vals = np.sort(vals)[::-1]          # descending

    a[-half:] = sorted_vals[-half:][::-1]    # smallest half right side

    return a

    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return _anti_pair_arrangement(vals)


def _triangle_candidate(n: int) -> np.ndarray:
    """Tent shape (large at the edges, small in the middle)."""
    x = np.arange(n, dtype=np.float64)
    shape = np.minimum(x + 1, n - x)          # 1,2,…,⌈n/2⌉,…,2,1
    shape = shape / shape.sum()
    return shape


def _gaussian_edges_candidate(n: int, gamma: float = 0.5) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
    shape = shape / shape.sum()
    return shape


def _piecewise_edges_candidate(n: int, edge_frac: float = 0.1,
                               high: float = 0.8, low: float = 0.2) -> np.ndarray:
    """Two‑level shape: big mass near both ends, small mass in the centre."""
    edge_len = max(1, int(edge_frac * n))
    w = np.full(n, low, dtype=np.float64)
    w[:edge_len] = high
    w[-edge_len:] = high
    w = np.maximum(w, 0.0)
    w /= w.sum()
    return w

# Optimisation primitives
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   conv_fft_len: int,
                   corr_fft_len: int,
                   deadline: float,
                   max_iters: int = 300_000,
                   lr: float = 0.07,
                   lam_start: float = 1.0,
                   lam_factor: float = 1.07) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max (log‑sum‑exp) surrogate.
    Returns the best exact solution seen during the run.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lam = lam_start

    best_a = a0.copy()
    best_val = _objective(a0)

    it = 0
    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val_exact = 2.0 * n * max_c
        if val_exact < best_val:
            best_val = val_exact
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best exact {best_val:.6f}")

        # smooth‑max weights (log‑sum‑exp approximation)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of surrogate w.r.t. a
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature schedule → true max
        lam = min(lam * lam_factor, 1e6)

        # occasional jitter to escape plateaus
        if it % 5000 == 0 and (deadline - time.time()) > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2

        it += 1

    return best_a, best_val


def _subgradient_optimize(a0: np.ndarray,
                          n: int,
                          conv_fft_len: int,
                          corr_fft_len: int,
                          deadline: float,
                          max_iters: int = 300_000,
                          eta0: float = 0.7) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient optimiser on the exact max‑convolution.
    """
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s

    best_a = a.copy()
    best_val = _objective(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # step‑size schedule (slowly decaying)
        eta = eta0 / np.sqrt(it + 1)

        # multiplicative update
        a = a * np.exp(-eta * grad_a)

        # renormalise onto the simplex
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _proj_subgradient_optimize(a0: np.ndarray,
                               n: int,
                               conv_fft_len: int,
                               corr_fft_len: int,
                               deadline: float,
                               max_iters: int = 200_000,
                               eta0: float = 0.7) -> tuple[np.ndarray, float]:
    """
    Euclidean‑projected sub‑gradient optimiser.
    Uses the average of gradients from all active constraints.
    """
    a = np.maximum(a0, 0.0)
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[ProjSubgrad it={it:06d}] new best {val:.6f}")

        # active constraints
        active = np.isclose(conv, max_c, atol=1e-12)
        w = active.astype(float)
        sum_w = w.sum()
        if sum_w > 0:
            w /= sum_w
        else:
            w = np.zeros_like(conv)

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1)
        a = a - eta * grad
        a = _project_simplex(a)

        it += 1

    return best_a, best_val


def _local_random_search(a0: np.ndarray,
                         n: int,
                         conv_fft_len: int,
                         deadline: float,
                         max_moves: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing hill‑climbing.
    Moves: swap two entries or transfer a fraction of mass.
    Accept if improvement or with probability exp(-Δ/Temp).
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a)

    start_temp = 0.02  # a little hotter than before
    it = 0
    while time.time() < deadline and it < max_moves:
        # move type
        if np.random.rand() < 0.5:          # swap
            i, j = np.random.randint(n, size=2)
            if i != j:
                a2 = a.copy()
                a2[i], a2[j] = a2[j], a2[i]
            else:
                continue
        else:                               # transfer
            i = np.random.randint(n)
            if a[i] < 1e-14:
                continue
            j = np.random.randint(n)
            if i == j:
                continue
            delta = a[i] * (np.random.rand() * 0.6)   # up to 60 % move
            a2 = a.copy()
            a2[i] -= delta
            a2[j] += delta

        val2 = _objective(a2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
        else:
            temp = start_temp * (1.0 - (it / max_moves))
            if temp > 0.0 and np.random.rand() < np.exp(-(val2 - best_val) / temp):
                a = a2

        if it % 5000 == 0:
            print(f"[Local it={it:06d}] best {best_val:.6f}")

        it += 1

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function minimizing
    F(a)=2·n·max(a∗a)/(sum(a)²).

    The routine respects the time budget, mixes several optimisation
    techniques (Adam, exponentiated‑gradient, projected sub‑gradient,
    simulated‑annealing) and returns the best vector found as a Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # Core parameters
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # 1) Build a diverse pool of start vectors
    candidates = []

    # warm start (if available)
    warm = _warm_start(n)
    if warm is not None:
        candidates.append(warm.copy())
        # few rotations of the warm start
        for shift in (n // 4, n // 2, 3 * n // 4):
            candidates.append(np.roll(warm, shift))

    # uniform
    candidates.append(_uniform_candidate(n))

    # random Dirichlet samples with various concentrations
    for alpha in (0.05, 0.1, 0.3, 0.7, 1.0, 2.0, 5.0):
        candidates.append(_random_dirichlet_candidate(n, alpha))

    # anti‑pair Dirichlet samples (the strongest heuristic)
    for alpha in (0.05, 0.1, 0.3, 0.7, 1.0, 2.0):
        candidates.append(_anti_pair_candidate(n, alpha))

    # edge‑focused shapes
    for gamma in (0.1, 0.2, 0.5, 1.0):
        candidates.append(_gaussian_edges_candidate(n, gamma))

    # piecewise‑edge shapes
    for edge_frac in (0.05, 0.1, 0.2):
        candidates.append(_piecewise_edges_candidate(n, edge_frac=edge_frac))

    # simple triangle
    candidates.append(_triangle_candidate(n))

    # ------------------------------------------------------------------
    # 2) Quick evaluation of the pool – keep the best starter
    best_seq = None
    best_val = float("inf")
    for idx, cand in enumerate(candidates):
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 0.01:                     # fallback to fresh Dirichlet
            cand = np.random.dirichlet(np.ones(n))
            s = cand.sum()
        cand = cand / s                    # normalise to sum = 1
        val = _objective(cand)
        print(f"[Init cand {idx:02d}] value {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = cand.copy()

    # ------------------------------------------------------------------
    # 3) Continuous optimisation phases
    # allocate a few seconds for a final upscale (if any)
    final_upscale_deadline = deadline - 20.0

    # ---- Adam phase (smooth‑max surrogate)
    adam_deadline = min(deadline - 15.0, time.time() + 250.0)
    a_adam, val_adam = _adam_optimize(
        best_seq,
        n,
        conv_fft_len,
        corr_fft_len,
        adam_deadline,
        max_iters=400_000,
        lr=0.07,
        lam_start=1.0,
        lam_factor=1.07,
    )
    if val_adam < best_val:
        best_val = val_adam
        best_seq = a_adam.copy()
        print(f"[Adam] improved to {val_adam:.6f}")

    # ---- Exponentiated‑gradient sub‑gradient phase
    subgrad_deadline = min(deadline - 8.0, time.time() + 300.0)
    a_sub, val_sub = _subgradient_optimize(
        best_seq,
        n,
        conv_fft_len,
        corr_fft_len,
        subgrad_deadline,
        max_iters=400_000,
        eta0=0.7,
    )
    if val_sub < best_val:
        best_val = val_sub
        best_seq = a_sub.copy()
        print(f"[Subgrad] improved to {val_sub:.6f}")

    # ---- Projected sub‑gradient fine‑tuning
    proj_deadline = min(deadline - 5.0, time.time() + 200.0)
    a_proj, val_proj = _proj_subgradient_optimize(
        best_seq,
        n,
        conv_fft_len,
        corr_fft_len,
        proj_deadline,
        max_iters=300_000,
        eta0=0.7,
    )
    if val_proj < best_val:
        best_val = val_proj
        best_seq = a_proj.copy()
        print(f"[ProjSubgrad] improved to {val_proj:.6f}")

    # ------------------------------------------------------------------
    # 4) Cheap stochastic hill‑climbing (more moves than before)
    if best_seq is not None and time.time() < deadline - 5.0:
        local_deadline = deadline - 2.0
        # allocate moves roughly proportional to remaining time
        remaining = local_deadline - time.time()
        max_moves = int(max(50_000, remaining * 3000))  # ~3k moves per second
        a_local, val_local = _local_random_search(
            best_seq,
            n,
            conv_fft_len,
            local_deadline,
            max_moves=max_moves,
        )
        if val_local < best_val:
            best_val = val_local
            best_seq = a_local.copy()
            print(f"[Local] improved to {val_local:.6f}")

    # ------------------------------------------------------------------
    # 5) Optional upscale to a finer grid (8192) if time permits
    if time.time() < final_upscale_deadline:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        # linear interpolation of the current best solution
        x_old = np.linspace(0.0, 1.0, n, endpoint=False)
        x_new = np.linspace(0.0, 1.0, n_big, endpoint=False)
        a_big = np.interp(x_new, x_old, best_seq)

        a_big = np.maximum(a_big, 0.0)
        a_big = a_big / a_big.sum()

        # brief Adam on larger grid
        ad_big_deadline = min(deadline - 5.0, time.time() + 180.0)
        a_big_adam, val_big_adam = _adam_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            ad_big_deadline,
            max_iters=250_000,
            lr=0.07,
            lam_start=1.0,
            lam_factor=1.07,
        )
        # sub‑gradient refinement on big grid
        sub_big_deadline = min(deadline, time.time() + 200.0)
        a_big_sub, val_big_sub = _subgradient_optimize(
            a_big_adam,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            sub_big_deadline,
            max_iters=250_000,
            eta0=0.7,
        )
        if val_big_sub < best_val:
            best_val = val_big_sub
            best_seq = a_big_sub.copy()
            print(f"[Big] improved to {val_big_sub:.6f}")

    # ------------------------------------------------------------------
    # 6) Safeguard – guarantee a valid return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the range accepted by the evaluator
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
