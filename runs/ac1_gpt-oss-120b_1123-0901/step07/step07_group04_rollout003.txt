    deadline = time.time() + budget_s - 0.5

    n = 4096
    conv_len = _next_pow2(2*n - 1)
    corr_len = _next_pow2(3*n - 2)   # conv+len-1?

    warm_start = ... as before (height_sequence_1)
    if warm_start is None:
        # generate initial seeds
        # use uniform and random Dirichlet 
        seeds = [np.full(n, 1/n)]
        for _ in range(10):
            conc = np.random.uniform(0.05,1.0)
            seeds.append(np.random.dirichlet(np.full(n, conc)))
    else:
        seeds = [warm_start]

    # Evaluate seeds quickly using conv_fft; keep best few
    best_seq = None; best_val = np.inf
    for a0 in seeds:
        a0 = np.maximum(a0,0); a0 = a0 / a0.sum()
        val = _objective(a0, n, conv_len)
        if val < best_val:
            best_val = val; best_seq = a0

    # Now run softmax mirror descent from best_seq for remainder of time.
    a = best_seq.copy()
    # AdaGrad/Adam parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    eta = 0.1  # step size for mirror descent? need adjust
    # schedule gamma:
    gamma0 = 0.1
    gamma_inc = 0.001   # per iteration increase 

    it = 0
    while time.time() < deadline:
        # compute conv
        conv = _full_convolution(a, conv_len)  # len 2n-1
        max_c = conv.max()
        val = 2.0*n*max_c/(a.sum()**2) # sum==1 usually
        if val < best_val:
            best_val = val; best_seq = a.copy()
            print(f"[Iter {it}] new best {val:.6f}")

        # gamma schedule: gradually increase
        gamma = gamma0 + gamma_inc*it

        # compute softmax weights
        # To avoid overflow, subtract max
        scaled = gamma * conv
        max_scaled = scaled.max()
        exp_scaled = np.exp(scaled - max_scaled)
        w = exp_scaled / exp_scaled.sum()  # shape (2n-1)

        # gradient
        grad = 2.0 * _cross_correlation(w, a, corr_len)

        # Adam updates
        it += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**it)

            a = a / a.sum()
    # after loop, return best_seq as list

    print(f"*** Finished search: best objective ≈ {best_val:.6f} (n={len(best_seq)})")

def _cross_correlation(w, a, fft_len):
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]

"""
propose_candidate
=================

Search for a non‑negative step‑function (a probability vector) that minimizes

        F(a) = 2·n·max(a∗a) / (sum(a)²)

under the constraints enforced by ``evaluate_sequence``.
The algorithm consists of three stages:

1. **Warm‑start / seeding** – a small pool of diverse candidates
   (uniform, Dirichlet, edge‑exponential, edge‑power‑law, double Gaussian,
   sinusoid).  The best few are kept.

2. **Soft‑max mirror descent** – a smooth approximation of the
   ``max`` in the objective is used:
   ``softmax_γ(conv) = exp(γ·conv)/∑exp(γ·conv)``.
   The gradient of the smoothed objective is
   ``∇ = 2·(softmax ★ a)`` (★ = cross‑correlation).  An
   exponentiated‑gradient (mirror‑descent) update
   ``a ← a·exp(−η·∇)`` keeps the vector non‑negative; after each step we
   project onto the simplex (normalise to sum = 1).

   * γ (smoothness) is increased linearly with elapsed time, beginning
     with a very smooth landscape (γ≈0.01) and ending with a near‑exact
     max (γ≈15).
   * η (step size) decays geometrically but never falls below a floor,
     ensuring both rapid early progress and fine late‑stage tuning.

3. **Optional up‑sampling** – if a sizeable time margin remains,
   the best vector is interpolated to a finer grid (n = 8192) and a
   brief refinement with the same optimizer is performed.

The routine respects the overall 1000 s budget, prints progress,
stores the best vector found and finally returns it as a plain list.
All values are clipped to the range accepted by ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2·len(a)-1) using FFT.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a).  Returns a vector of length len(a)
    with (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)] for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """
    Fast evaluation of the objective (assuming a is already normalised).
    """
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c / (a.sum() ** 2))


def _generate_parametric_candidates(
    n: int,
    warm_start: np.ndarray | None,
) -> list[np.ndarray]:
    """
    Produce a diverse set of candidate vectors using several families:
    uniform, Dirichlet, edge‑exponential, edge‑power‑law, double‑Gaussian,
    half‑wave sinusoid.  All candidates are normalised to sum ≈ 1.
    """
    candidates = []

    # Warm‑start and simple rotations
    if warm_start is not None:
        candidates.append(warm_start.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            candidates.append(np.roll(warm_start, shift))

    # Uniform baseline
    candidates.append(np.full(n, 1.0 / n))

    # Random Dirichlet with a variety of concentration parameters
    for conc in (0.01, 0.05, 0.1, 0.3, 0.5, 1.0, 2.0):
        cand = np.random.dirichlet(np.full(n, conc))
        candidates.append(cand)

    i = np.arange(n, dtype=np.float64)

    # Edge‑exponential + optional uniform mixture
    for alpha in np.logspace(-2, 1.0, num=16):      # 0.01 … 10
        edge = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
        edge /= edge.sum()
        for w in (0.0, 0.25, 0.5, 0.75, 1.0):
            shape = w * edge + (1.0 - w) / n
            candidates.append(shape)

    # Edge‑power‑law
    for p in np.linspace(0.2, 5.0, num=16):
        edge = (i + 1.0) ** (-p) + (n - i) ** (-p)
        edge /= edge.sum()
        for w in (0.0, 0.33, 0.66, 1.0):
            shape = w * edge + (1.0 - w) / n
            candidates.append(shape)

    # Double‑Gaussian edges
    for gamma in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        shape /= shape.sum()
        candidates.append(shape)

    # Positive half‑wave sinusoid
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    candidates.append(shape)

    return candidates

# Core optimisation routine (soft‑max mirror descent)
# ----------------------------------------------------------------------


def _softmax_mirror_descent(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray | None, float]:
    """
    Perform mirror‑descent with a soft‑max approximation of the max‑convolution.
    ``a0`` is assumed to be non‑negative and normalised to sum = 1.
    Returns the best vector/value observed.
    """
    a = a0.copy()
    a = np.maximum(a, 0.0)
    if a.sum() < 0.01:
        a = np.random.dirichlet(np.ones(n))
    a /= a.sum()

    # Step‑size schedule (mirroring the original code)
    eta0 = 0.5
    decay = 0.99995
    min_eta = 0.001

    # Soft‑max schedule parameters
    min_gamma = 0.01
    max_gamma = 15.0

    start_time = time.time()
    it = 0
    m = np.zeros_like(a)          # first moment (Adam‑style)
    v = np.zeros_like(a)          # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    total_budget = deadline - start_time

    while time.time() < deadline:
        # Full convolution a*a
        conv = _full_convolution(a, conv_fft_len)           # length 2n‑1
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)              # sum≈1 → denominator ≈1

        if val < best_val - 1e-12:
            best_val = val
            best_seq = a.copy()
            print(f"[n={n} iter={it:07d}] new best {val:.6f}")

        # ----- soft‑max weight computation -----
        elapsed = time.time() - start_time
        gamma = min_gamma + (max_gamma - min_gamma) * (elapsed / total_budget)
        # numerical stable softmax
        scaled = gamma * conv
        max_scaled = np.max(scaled)
        exp_scaled = np.exp(scaled - max_scaled)
        w = exp_scaled / exp_scaled.sum()    # shape (2n‑1), sums to 1

        # ----- gradient of the smoothed objective -----
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # ----- Adam‑style adaptive step -----
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        eta = max(min_eta, eta0 * (decay ** it))
        step = eta * m_hat / (np.sqrt(v_hat) + eps)

        # Exponentiated‑gradient update (keeps positivity)
        # Clip the exponent to avoid overflow/underflow
        step_clip = np.clip(step, -20.0, 20.0)
        a = a * np.exp(-step_clip)

        # Projection onto the simplex (non‑negative, sum = 1)
        s = a.sum()
        if s <= 0.0 or np.isnan(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        # Occasionally inject a tiny jitter to escape plateaus
        if it % 5000 == 0:
            jitter = (np.random.rand(n) - 0.5) * 1e-3
            a = a * np.exp(jitter)
            a = a / a.sum()

    return best_seq, best_val
