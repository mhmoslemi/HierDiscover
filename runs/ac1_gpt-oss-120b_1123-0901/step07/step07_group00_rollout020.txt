"""
Search for a non‑negative step‑function (vector) that minimizes

        F(a) = 2·n·max(a∗a) / (sum(a)²)

The algorithm is deliberately simple yet aggressive:

1. Build a large and diverse pool of candidate shapes (warm‑start,
   Dirichlet with many concentration parameters, edge‑exponential,
   edge‑polynomial, double‑Gaussian, uniformly spaced spikes and a few
   random sparse spikes).  All candidates are normalised to sum = 1.
2. Scan the whole pool with a fast FFT‑based evaluation of the exact
   objective and keep the best one.
3. Refine the best candidate in three stages that respect a global
   deadline (default 1000 s):
   – Adam optimisation on a smooth‑max (log‑sum‑exp) surrogate with a
     slowly increasing temperature.
   – Exponentiated‑gradient sub‑gradient descent on the exact max,
     using a decreasing step size.
   – Very cheap random hill‑climbing that moves a random fraction of
     mass between two coordinates.
4. If time permits up‑sample the solution to 8192 points and repeat
   steps 3 on the finer grid.
5. Finally enforce symmetry (a_i = a_{n‑1‑i}), re‑normalise and clip
   to the evaluator’s allowed range before returning the sequence.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via FFT (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a length‑len(a) vector:
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i = 0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """
    Exact objective value 2·n·max(conv(a,a)) / (sum(a)²) .
    a is assumed non‑negative; it will be normalised internally.
    """
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    a_norm = a / s
    conv = _full_convolution(a_norm, conv_fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c / (a_norm.sum() ** 2))


def _symmetrize(a: np.ndarray) -> np.ndarray:
    """Force a to be symmetric: a_i = a_{n‑1‑i}."""
    return (a + a[::-1]) * 0.5


# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 600_000,
) -> tuple[np.ndarray, float]:
    """Adam on the smooth‑max (log‑sum‑exp) surrogate."""
    eps = 1e-15
    logits = np.log(a0 + eps)          # initialise on the simplex (soft‑max)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05                         # slightly smaller than before
    lam = 0.1                         # start temperature (soft)
    lam_factor = 1.05

    best_a = a0.copy()
    best_val = _objective(a0, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        # ---------- soft‑max projection ----------
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()          # a lies on the simplex

        # ---------- exact objective for bookkeeping ----------
        val = _objective(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # ---------- smooth‑max weights ----------
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---------- gradient of the surrogate ----------
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through the soft‑max (mirror‑descent form)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)            # ∇_logits

        # ---------- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # ---------- temperature schedule ----------
        lam = min(lam * lam_factor, 1e7)

        # occasional jitter near deadline
        if it % 4000 == 0 and time.time() > deadline - 60:
            logits += (np.random.rand(n) - 0.5) * 2e-2

        # gentle learning‑rate decay
        if it % 50_000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    # enforce symmetry before returning
    best_a = _symmetrize(best_a)
    best_a = best_a / max(best_a.sum(), 1e-12)
    best_val = _objective(best_a, n, conv_fft_len)
    return best_a, best_val


def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 800_000,
    eta0: float = 0.85,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient phase using exponentiated‑gradient updates.
    Step‑size η_t = η₀ / √(t+1).
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active set (indices where the max is attained)
        active = np.isclose(conv, max_c, atol=1e-14)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # decreasing step size
        eta = eta0 / math.sqrt(it + 1)

        # exponentiated‑gradient update (mirror descent on the simplex)
        a = a * np.exp(-eta * grad_a)

        # re‑normalise – protect against underflow
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    # enforce symmetry before returning
    best_a = _symmetrize(best_a)
    best_a = best_a / max(best_a.sum(), 1e-12)
    best_val = _objective(best_a, n, conv_fft_len)
    return best_a, best_val


def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climbing: move a random fraction of mass from one
    index to another, keep the move if the objective improves.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()

    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        # move up to 30 % of the source mass
        delta = a[i] * (np.random.rand() * 0.3)

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5000 == 0:
                print(f"[Local move {it:06d}] improved to {val2:.6f}")

        it += 1

    # final symmetrisation
    best_a = _symmetrize(best_a)
    best_a = best_a / max(best_a.sum(), 1e-12)
    best_val = _objective(best_a, n, conv_fft_len)
    return best_a, best_val

        else:

    pool = []

            pool.append(np.roll(warm, shift))

    for alpha in (0.01, 0.05, 0.1, 0.5, 1.0, 2.0, 5.0):

    for gamma in np.logspace(-2, 0.7, 30):               # ~0.01 … 5

        shape = left + left[::-1]
        shape /= shape.sum()

        if shape.sum() > 0:
            shape /= shape.sum()
            pool.append(shape)

    i = np.arange(n, dtype=np.float64)
    for sigma in np.linspace(0.01, 0.5, 20):
        shape = np.exp(-0.5 * ((i) / (sigma * n)) ** 2) \
                + np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
        shape /= shape.sum()
        pool.append(shape)

    # sparse spike patterns (equally spaced)
    for k in (32, 64, 128, 256, 512):
        a = np.zeros(n, dtype=np.float64)
        if k > 0:
            pos = np.round(np.linspace(0, n - 1, k)).astype(int)
            a[pos] = 1.0 / k
        pool.append(a)

    # a few random sparse spikes
    for _ in range(5):
        k = np.random.randint(16, 128)
        a = np.zeros(n, dtype=np.float64)
        pos = np.random.choice(n, k, replace=False)
        a[pos] = 1.0 / k
        pool.append(a)

    # --------------------------------------------------------------
    # quick scan of the pool
    best_seq = None
    best_val = float("inf")
    for cid, cand in enumerate(pool):
        if time.time() >= deadline:
            break
        a = np.maximum(cand, 0.0)
        if a.sum() < 1e-12:
            a = np.random.dirichlet(np.ones(n))
        a = a / a.sum()
        val = _objective(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Init cand {cid}] new best {val:.6f}")

    # safety fallback
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = _objective(best_seq, n, conv_fft_len)

    # --------------------------------------------------------------
    # Phase 1 – Adam on smooth‑max surrogate
    if time.time() < deadline - 5:
        a_adam, val_adam = _adam_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=600_000,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Phase Adam] best {val_adam:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – Exponentiated‑gradient sub‑gradient descent
    if time.time() < deadline - 5:
        a_sub, val_sub = _subgradient_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=800_000,
            eta0=0.85,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Phase Subgrad] best {val_sub:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – cheap random local search (many moves)
    if time.time() < deadline - 5:
        a_loc, val_loc = _local_random_search(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=300_000,
        )
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()
            print(f"[Phase Local] improved best {val_loc:.6f}")

    # --------------------------------------------------------------
    # Optional up‑sampling to a finer grid (8192) if time permits
    if time.time() + 30 < deadline:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        # linear interpolation to the finer grid
        x_old = np.linspace(0, 1, n, endpoint=False)
        x_new = np.linspace(0, 1, n_big, endpoint=False)
        a_big = np.interp(x_new, x_old, best_seq)
        a_big = np.clip(a_big, 0.0, 1000.0)
        a_big = a_big / a_big.sum()

        # short Adam on the larger grid
        a_big_adam, _ = _adam_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            max_iters=200_000,
        )

        # longer sub‑gradient refinement on the large grid
        a_big_ref, val_big = _subgradient_optimize(
            a_big_adam,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            max_iters=400_000,
            eta0=0.8,
        )
        if val_big < best_val:
            best_val = val_big
            best_seq = a_big_ref.copy()
            print(f"[Phase Big] refined best {val_big:.6f}")

    # --------------------------------------------------------------
    # final tidy‑up (symmetrisation, normalisation, clipping)
    best_seq = _symmetrize(best_seq)
    best_seq = best_seq / max(best_seq.sum(), 1e-12)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
