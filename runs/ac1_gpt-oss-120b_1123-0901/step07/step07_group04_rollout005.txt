def _smooth_obj_and_grad(x, n, conv_fft_len, corr_fft_len, epsilon):
    # compute a (softmax)
    max_x = np.max(x)
    exp_x = np.exp(x - max_x)
    a = exp_x / np.sum(exp_x)
    # convolution
    fa = np.fft.rfft(a, conv_fft_len)
    conv = np.fft.irfft(fa * fa, conv_fft_len)[:2*n-1]
    max_c = np.max(conv)
    # stable softmax for conv
    conv_shift = conv - max_c
    w_stable = np.exp(conv_shift / epsilon)
    sum_w = np.sum(w_stable)
    w = w_stable / sum_w
    # gradient w.r.t a:
    grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
    # gradient w.r.t x (softmax):
    # grad_x = a * (grad_a - np.dot(a, grad_a))
    grad_x = a * (grad_a - np.dot(a, grad_a))
    # smoothed objective:
    smooth_val = max_c + epsilon * np.log(sum_w)
    return smooth_val, grad_x
```

    max_conv = _full_convolution(cur_a, conv_fft_len).max()
    epsilon_vals = [max_conv * factor for factor in (0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005)]

            return val, grad

"""
Propose a low‑evaluation‑score step function.

The algorithm works in three phases:

1. **Broad, cheap seeding** – generate a diverse set of candidate vectors
   (uniform, Dirichlet, edge‑exponential/power‑law, Gaussian‑edges,
   half‑wave sinusoid) and keep the best few according to a fast FFT‑based
   proxy for the objective.

2. **Smoothing‑annealed Adam optimisation** – each retained start is
   represented in log‑it space (soft‑max → simplex).  The non‑smooth
   `max(conv)` is replaced by a smooth approximation
   `ε·log Σ exp(conv/ε)`; its gradient is obtained analytically via a
   weighted cross‑correlation.  An Adam optimiser runs for a limited number
   of iterations while the smoothing parameter ε is gradually reduced.
   The best vector (in terms of the true objective `2·n·max(conv)`) is kept.

3. **Optional up‑sampling** – if time permits, the current best vector is
   linearly interpolated to a finer grid (n = 8192) and refined once more
   with the same Adam schedule.

All numbers are forced to be non‑negative and ≤ 1000, which already holds
for the soft‑max representation.  The routine respects the 1000 s time
budget and prints progress information.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper FFT utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Fast proxy for the true objective (assumes sum(a)=1)."""
    conv = _full_convolution(a, conv_fft_len)
    return float(2.0 * n * conv.max())  # because sum(a)=1 → denominator = 1


# ----------------------------------------------------------------------
# Soft‑max and smoothed max‑convolution utilities
# ----------------------------------------------------------------------
def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max producing a simplex vector."""
    m = np.max(x)
    e = np.exp(x - m)
    return e / np.sum(e)


def _smooth_obj_and_grad(
    x: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    epsilon: float,
) -> tuple[float, np.ndarray, float]:
    """
    Returns (smooth_objective, gradient_wrt_x, max_convolution).

    smooth_objective = max(conv) + ε·log Σ_i exp((conv_i‑max(conv))/ε)
    Its gradient w.r.t. x (log‑it parameters) is:
        grad_x = a ⊙ (grad_a – (a·grad_a)·1)
    where a = softmax(x) and grad_a = 2·(w ★ a) with
    w_i = exp((conv_i‑max)/ε) / Σ_j exp((conv_j‑max)/ε).
    """
    a = _softmax(x)

    # convolution a*a
    fa = np.fft.rfft(a, conv_fft_len)
    conv = np.fft.irfft(fa * fa, conv_fft_len)[: 2 * n - 1]
    max_c = conv.max()

    # stable soft‑max over convolution entries
    shifted = conv - max_c          # ≤ 0
    w_unnorm = np.exp(shifted / epsilon)
    sum_w = w_unnorm.sum()
    w = w_unnorm / sum_w            # weights sum to 1

    # gradient wrt a
    grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

    # gradient wrt x (soft‑max Jacobian)
    dot = np.dot(a, grad_a)
    grad_x = a * (grad_a - dot)

    # smooth approximation of max
    smooth_val = max_c + epsilon * np.log(sum_w)
    return smooth_val, grad_x, max_c


# ----------------------------------------------------------------------
# Adam optimiser working on the smoothed objective
# ----------------------------------------------------------------------
def _adam_optimize(
    x0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    epsilon: float,
    max_iter: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smoothed max‑convolution.
    Returns (best_x, best_obj) where best_obj = 2·n·max(conv) for the
    corresponding simplex vector.
    """
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    lr = 0.1
    eps = 1e-8

    best_x = x.copy()
    best_val = float("inf")
    t = 0

    while t < max_iter and time.time() < deadline:
        t += 1
        smooth_val, grad_x, max_c = _smooth_obj_and_grad(
            x, n, conv_fft_len, corr_fft_len, epsilon
        )
        # true objective (sum(a)=1 → denominator = 1)
        cur_val = 2.0 * n * max_c

        if cur_val < best_val:
            best_val = cur_val
            best_x = x.copy()

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        x = x - step

    return best_x, best_val

def _optimise_from_start(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray | None, float]:
    """
    Run a multi‑stage Adam optimiser on a0.
    The smoothing parameter ε is annealed from a relatively large value
    down to 1e‑3.  The best solution (in terms of the true objective) is
    returned together with the updated best value.
    """
    # initialise log‑it parameters; avoid log(0)
    eps_small = 1e-12
    a0 = np.maximum(a0, eps_small)
    x = np.log(a0)

    # initial max‑convolution to set the ε‑schedule
    init_conv = _full_convolution(_softmax(x), conv_fft_len)
    init_max = init_conv.max()
    eps_schedule = [
        init_max * factor for factor in (0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002)
    ]

    for eps in eps_schedule:
        if time.time() >= deadline:
            break

        # allocate a modest budget per ε (adapt to remaining time)
        remaining = deadline - time.time()
        max_iter = max(1, int(min(4000, remaining / 0.05)))  # ~20 ms per iter

        x, cur_best = _adam_optimize(
            x,
            n,
            conv_fft_len,
            corr_fft_len,
            epsilon=eps,
            max_iter=max_iter,
            deadline=deadline,
        )

        # evaluate the best vector found at this ε
        a = _softmax(x)
        conv = _full_convolution(a, conv_fft_len)
        cur_val = 2.0 * n * conv.max()
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_seq = a.copy()
            print(
                f"[ε={eps:.3g}] new global best {best_val:.6f} (n={n})"
            )

    return best_seq, best_val

def _generate_parametric_candidates(
    n: int,
    warm_start: np.ndarray | None,
) -> list[np.ndarray]:
    """Return a list of diverse candidate probability vectors."""
    candidates = []

    if warm_start is not None:
        candidates.append(warm_start.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            candidates.append(np.roll(warm_start, shift))

    # uniform
    candidates.append(np.full(n, 1.0 / n))

    # Dirichlet draws with a few concentration parameters
    for conc in (0.01, 0.05, 0.1, 0.3, 0.5, 1.0, 2.0):
        cand = np.random.dirichlet(np.full(n, conc))
        candidates.append(cand)

    i = np.arange(n, dtype=np.float64)

    # edge‑exponential families
    for alpha in np.logspace(-2, 1.0, num=12):
        edge = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
        edge /= edge.sum()
        for w in (0.0, 0.25, 0.5, 0.75, 1.0):
            shape = w * edge + (1.0 - w) / n
            candidates.append(shape)

    # edge‑power‑law families
    for p in np.linspace(0.2, 5.0, num=12):
        edge = (i + 1.0) ** (-p) + (n - i) ** (-p)
        edge /= edge.sum()
        for w in (0.0, 0.33, 0.66, 1.0):
            shape = w * edge + (1.0 - w) / n
            candidates.append(shape)

    # double‑Gaussian edges
    for gamma in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        shape /= shape.sum()
        candidates.append(shape)

    # half‑wave sinusoid (positive part only)
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    candidates.append(shape)

    return candidates

        best_seq, best_val = _optimise_from_start(
            a0,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            best_seq,
            best_val,
        )

    # ------------------------------------------------------------------
    # Phase 3 – optional up‑sampling to a finer grid if time permits
    if time.time() < deadline - 10.0 and best_seq is not None:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        # linear interpolation onto finer grid
        x_old = np.linspace(0, 1, n, endpoint=False)
        x_big = np.linspace(0, 1, n_big, endpoint=False)
        a_big = np.interp(x_big, x_old, best_seq)
        a_big = np.maximum(a_big, 0.0)
        a_big /= a_big.sum()

        print("--- Refinement on larger grid (n=8192)")
        best_seq, best_val = _optimise_from_start(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            best_seq,
            best_val,
        )
        n = n_big  # final size for reporting

    # ------------------------------------------------------------------
    # Safety fallback
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to evaluator’s legal range
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"*** Finished search: best objective ≈ {best_val:.6f} (n={n})")
    return best_seq.tolist()
