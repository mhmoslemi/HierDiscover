"""
Search for a low‑evaluation step function.

The algorithm proceeds in stages:

1. **Candidate generation** – a rich pool of parametric shapes
   (uniform, Dirichlet draws, exponentials, power‑laws,
   asymmetric two‑sided exponentials, Gaussians, Beta‑pdfs,
   linear/triangular, half‑wave sinusoid and sparse spikes) is built.
   All candidates are normalised to sum 1 and evaluated cheaply
   via FFT‑based convolution.  The best ≈20 candidates are kept.

2. **Deep optimisation** – for each selected start we run
   *Adam* on a smooth‑max surrogate of the max‑convolution,
   followed by an *exponentiated‑gradient* sub‑gradient phase on the
   exact max.  Both phases stop as soon as the global deadline is
   reached.

3. **Pairwise stochastic hill‑climbing** – a cheap local search
   repeatedly transfers a small amount of mass between two random
   positions (preserving the simplex).  Improvements are accepted
   immediately; the step size decays over time.

4. **Multi‑resolution refinement** – if time permits the current
   best vector is interpolated to a finer grid (n = 8192) and the
   whole deep‑optimisation pipeline is applied again.

All FFT work uses the next power‑of‑two length for speed.
The routine is deterministic for a given seed and respects the
evaluator’s clipping (non‑negative, ≤ 1000).  It returns the best
sequence found within the 1000 s budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact evaluation function 2·n·max(a*a) / (sum(a)²)."""
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c / (a.sum() ** 2))


# ----------------------------------------------------------------------
def _adam_smooth_max(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on a log‑sum‑exp surrogate of the max‑convolution.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05

    lam = 2.0
    lam_factor = 1.012
    lam_cap = 1e6

    it = 0
    max_iter = 350_000
    last_improve = time.time()

    while it < max_iter and time.time() < deadline:
        # ---------- soft‑max ----------
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # ---------- exact objective ----------
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Adam] iter={it:06d} new best {val:.6f}")

        # ---------- smooth‑max weights ----------
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---------- gradient wrt a ----------
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # ---------- gradient wrt logits (soft‑max chain rule) ----------
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ---------- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---------- temperature schedule ----------
        lam = min(lam * lam_factor, lam_cap)

        # ---------- occasional shake ----------
        if time.time() - last_improve > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1

    return best_seq, best_val


def _subgradient_refinement(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient steps on the *exact* max‑convolution.
    """
    a = a0.copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a /= s

    eta = 0.8
    decay = 0.99995
    it = 0
    max_iter = 80_000
    last_improve = time.time()

    while it < max_iter and time.time() < deadline:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Subgrad] iter={it:06d} new best {val:.6f}")

        # active set where the maximum is attained
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # exponentiated‑gradient update (keeps a ≥ 0)
        a = a * np.exp(-eta * grad_a)

        # renormalise
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        eta *= decay
        it += 1

        # early stop if stagnant
        if time.time() - last_improve > 6.0:
            break

    return best_seq, best_val


def _pairwise_local_search(
    a_best: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climbing: transfer a small amount of mass between two
    random positions while preserving the simplex.  Accept only improvements.
    """
    a = a_best.copy()
    rng = np.random.default_rng()
    step_scale = 0.1  # relative to the source entry
    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # propose a signed transfer delta
        delta = step_scale * a[i] * (rng.random() - 0.5) * 2.0
        # keep the move feasible
        if delta > a[i]:
            delta = a[i]
        if a[i] - delta < 0.0:
            delta = a[i]  # clamp to zero

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        # evaluate exact objective
        conv = _full_convolution(a_new, fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a_new.sum() ** 2)

        if val < best_val:
            best_val = val
            a = a_new
            step_scale = max(step_scale * 0.99, 1e-5)
            print(f"[Pairwise] new best {val:.6f}")
        else:
            # slowly increase step size if we are stuck
            step_scale = min(step_scale * 1.001, 0.5)

    return a, best_val


def _deep_optimize(
    a0: np.ndarray,
    n: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Run Adam → sub‑gradient → pairwise local search.
    Updates the global best.
    """
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # Adam phase
    best_seq, best_val = _adam_smooth_max(
        a0, n, conv_fft_len, corr_fft_len, deadline, global_best_seq, global_best_val
    )

    # Sub‑gradient phase
    best_seq, best_val = _subgradient_refinement(
        a0, n, conv_fft_len, corr_fft_len, deadline, best_seq, best_val
    )

    # Pairwise local refinement
    best_seq, best_val = _pairwise_local_search(
        best_seq, n, conv_fft_len, deadline, best_val
    )

    return best_seq, best_val

def _generate_parametric_candidates(n: int, warm_start: np.ndarray | None) -> list[np.ndarray]:
    """
    Produce a diverse pool of candidate vectors.
    Families:
      - uniform
      - Dirichlet draws (various concentrations)
      - symmetric edge‑exponential / power‑law
      - asymmetric two‑sided exponentials & power‑laws
      - linear / triangular
      - half‑wave sinusoid
      - Gaussian bumps
      - Beta‑pdf shapes
      - sparse spike vectors
    All candidates are normalised to sum ≈ 1.
    """
    candidates = []

    # warm‑start and its shifts
    if warm_start is not None:
        candidates.append(warm_start.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            candidates.append(np.roll(warm_start, shift))

    # uniform baseline
    candidates.append(np.full(n, 1.0 / n))

    # random Dirichlet draws
    for conc in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(10):
            cand = np.random.dirichlet(np.full(n, conc))
            candidates.append(cand)

    # index vector for analytic families
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n  # for continuous shapes

    # symmetric edge‑exponential / power‑law
    alphas = np.logspace(-2, 1.0, num=12)          # 0.01 … 10
    for a in alphas:
        edge = np.exp(-a * i) + np.exp(-a * (n - 1 - i))
        edge /= edge.sum()
        for w in (0.0, 0.25, 0.5, 0.75, 1.0):
            shape = w * edge + (1.0 - w) / n
            candidates.append(shape / shape.sum())

    ps = np.linspace(0.2, 6.0, num=12)
    for p in ps:
        edge = (i + 1.0) ** (-p) + (n - i) ** (-p)
        edge /= edge.sum()
        for w in (0.0, 0.3, 0.6, 1.0):
            shape = w * edge + (1.0 - w) / n
            candidates.append(shape / shape.sum())

    # asymmetric two‑sided exponentials
    alphas = np.logspace(-2, 1.0, num=10)
    w_vals = np.linspace(0.0, 1.0, 6)
    for wL in w_vals:
        for wR in w_vals:
            if wL + wR > 1.0:
                continue
            wU = 1.0 - wL - wR
            for aL in alphas:
                for aR in alphas:
                    left = np.exp(-aL * i)
                    right = np.exp(-aR * (n - 1 - i))
                    shape = wL * left + wR * right + wU / n
                    shape /= shape.sum()
                    candidates.append(shape)

    # asymmetric power‑laws
    p_vals = np.linspace(0.2, 5.0, num=8)
    for wL in w_vals:
        for wR in w_vals:
            if wL + wR > 1.0:
                continue
            wU = 1.0 - wL - wR
            for pL in p_vals:
                for pR in p_vals:
                    left = (i + 1.0) ** (-pL)
                    right = (n - i) ** (-pR)
                    shape = wL * left + wR * right + wU / n
                    shape /= shape.sum()
                    candidates.append(shape)

    # linear / triangular shapes
    lin = np.linspace(1.0, 0.0, n)
    tri = np.concatenate(
        [np.linspace(0, 1, n // 2, endpoint=False), np.linspace(1, 0, n - n // 2)]
    )
    for base in (lin, tri):
        base = base / base.sum()
        for w in (0.0, 0.4, 0.8):
            cand = w * base + (1.0 - w) / n
            candidates.append(cand / cand.sum())

    # half‑wave sinusoid
    sinus = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    sinus = np.maximum(sinus, 0.0)
    sinus /= sinus.sum()
    candidates.append(sinus)

    # Gaussian bumps (centered, varied width)
    for sigma_frac in (0.02, 0.05, 0.1, 0.15, 0.2):
        sigma = sigma_frac * n
        center = n // 2
        gauss = np.exp(-0.5 * ((i - center) / sigma) ** 2)
        gauss /= gauss.sum()
        for w in (0.0, 0.3, 0.6, 1.0):
            cand = w * gauss + (1.0 - w) / n
            candidates.append(cand / cand.sum())

    # Beta‑pdf shapes
    for a in (0.3, 0.5, 1.0, 2.0, 4.0):
        for b in (0.3, 0.5, 1.0, 2.0, 4.0):
            beta = x ** (a - 1) * (1 - x) ** (b - 1)
            beta = np.maximum(beta, 0.0)
            beta /= beta.sum()
            for w in (0.0, 0.4, 0.8):
                cand = w * beta + (1.0 - w) / n
                candidates.append(cand / cand.sum())

    # sparse spike vectors
    for spikes in (3, 5, 8, 12):
        for _ in range(15):
            idx = np.random.choice(n, size=spikes, replace=False)
            weights = np.random.dirichlet(np.ones(spikes))
            vec = np.zeros(n)
            vec[idx] = weights
            vec /= vec.sum()
            candidates.append(vec)

    return candidates

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Main entry point – searches for a non‑negative step function that
    minimises the supplied evaluation function.  Returns the best
    sequence found (plain Python list).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # Determine a warm start if the previous run exported one.
    warm_start = globals().get("height_sequence_1")
    if warm_start is not None:
        warm_start = np.asarray(warm_start, dtype=np.float64)
        # make it length‑agnostic – we will later interpolate if needed
        warm_start = np.maximum(warm_start, 0.0)
        s = warm_start.sum()
        if s >= 0.01:
            warm_start /= s
        else:
            warm_start = None
    else:
        warm_start = None

    # ------------------------------------------------------------------
    # Stage 1 – cheap exploration on medium resolution (n = 4096)
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    print("[Main] Generating candidate pool (n = %d) ..." % n)

    candidates = _generate_parametric_candidates(n, warm_start)
    print("[Main] %d raw candidates generated." % len(candidates))

    # evaluate cheaply and keep top K
    cand_vals = []
    for cand in candidates:
        if time.time() >= deadline:
            break
        # guard against numerical issues
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 1e-8:
            continue
        cand = cand / s
        val = _objective(cand, n, conv_fft_len)
        cand_vals.append((val, cand))

    cand_vals.sort(key=lambda x: x[0])
    top_k = 20
    top_candidates = [c for _, c in cand_vals[:top_k]]
    print("[Main] Stage 1 completed: best candidate % .6f" % cand_vals[0][0])

    # ------------------------------------------------------------------
    # Stage 2 – deep optimisation of the selected starts
    best_seq = None
    best_val = float("inf")
    for idx, start_vec in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        print("[Main] Deep optimisation start #%d / %d" % (idx + 1, len(top_candidates)))
        best_seq, best_val = _deep_optimize(
            start_vec, n, deadline, best_seq, best_val
        )
        print("[Main] Current global best %.6f" % best_val)

    # ------------------------------------------------------------------
    # Stage 3 – upscale to a finer grid if time permits
    time_left = deadline - time.time()
    if time_left > 80.0 and best_seq is not None:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)

        # linear interpolation of the best vector to the larger grid
        x_old = np.linspace(0.0, 1.0, n, endpoint=False)
        x_new = np.linspace(0.0, 1.0, n_big, endpoint=False)
        a_big = np.interp(x_new, x_old, best_seq)
        a_big = np.maximum(a_big, 0.0)
        a_big /= a_big.sum()

        print("[Main] Upscaling to n = %d and continuing optimisation." % n_big)
        best_seq, best_val = _deep_optimize(
            a_big, n_big, deadline, best_seq, best_val
        )
        n = n_big
        conv_fft_len = conv_fft_len_big
        print("[Main] After upscaling best objective %.6f" % best_val)

    # ------------------------------------------------------------------
    # Stage 4 – final cheap stochastic refinement (pairwise moves)
    if best_seq is not None and time.time() < deadline:
        print("[Main] Final pairwise stochastic refinement")
        best_seq, best_val = _pairwise_local_search(
            best_seq, n, conv_fft_len, deadline, best_val
        )
        print("[Main] Post‑refinement best %.6f" % best_val)

    # ------------------------------------------------------------------
    # Fallback – if nothing succeeded, return a uniform vector.
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Apply the evaluator's clipping just to be safe.
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Optionally show the exact evaluation (uses the hidden function).
    try:
        final_obj = evaluate_sequence(best_seq.tolist())
        print(f"[Main] Final exact objective reported by evaluator: {final_obj:.6f}")
    except Exception:
        pass

    print("[Main] Search finished. Returning sequence of length %d." % len(best_seq))
    return best_seq.tolist()
