def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Fast evaluation assuming sum(a)≈1."""
    conv = _full_convolution(a, conv_fft_len)
    return float(2.0 * n * conv.max() / (a.sum() ** 2))

    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _adam_smooth_max(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑sum‑exp surrogate of max(a*a).
    The temperature λ grows slowly so the landscape stays smooth for longer.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)                     # work in log‑space (soft‑max)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.02                                    # a little more conservative

    lam = 0.3                                    # start temperature
    lam_factor = 1.015                          # slower annealing
    lam_cap = 5e7

    last_improve = time.time()
    it = 0
    max_iter = 1_500_000                         # generous upper bound

    while time.time() < deadline and it < max_iter:
        # ---- simplex via soft‑max ----
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # ---- surrogate evaluation (exact max for monitoring) ----
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Adam n={n} it={it:07d}] new best {val:.6f}")

        # ---- smooth‑max weights (log‑sum‑exp) ----
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        # ---- gradient of the surrogate (chain‑rule through soft‑max) ----
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---- temperature schedule ----
        lam = min(lam * lam_factor, lam_cap)

        # ---- occasional noise if stuck ----
        if time.time() - last_improve > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
def _subgradient_refinement(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient steps on the true max‑convolution.
    """
    a = best_seq if best_seq is not None else a0.copy()
    a = np.maximum(a, 0.0)
    a /= a.sum()                     # stay on the simplex

    eta = 0.6
    last_improve = time.time()
    max_iter = 400_000               # large, bounded by deadline

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Subgrad n={n} it={it:07d}] new best {val:.6f}")

        # ---- active set (where the max is attained) ----
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # ---- multiplicative (exponentiated‑gradient) update ----
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:                     # avoid collapse
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        # ---- early stop if stagnating ----
        if time.time() - last_improve > 6.0:
            break

        # slowly decay eta to fine‑tune
        eta *= 0.99995

    return best_seq, best_val


# ----------------------------------------------------------------------
def _projected_gradient_refine(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Gradient descent on the exact objective with explicit simplex projection.
    """
    a = a0.copy()
    a = np.maximum(a, 0.0)
    a /= a.sum()

    lr = 0.25
    last_improve = time.time()
    max_iter = 200_000

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[ProjGrad n={n} it={it:07d}] new best {val:.6f}")

        # active mask (tolerance to numerical errors)
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)   # dM/da
        step = -lr * grad_a
        a_candidate = _proj_simplex(a + step)

        # accept if improvement, otherwise shrink step size
        conv_cand = _full_convolution(a_candidate, conv_fft_len)
        val_cand = 2.0 * n * conv_cand.max() / (a_candidate.sum() ** 2)

        if val_cand < val:
            a = a_candidate
            lr = min(lr * 1.02, 1.0)    # gently enlarge when moving downhill
        else:
            lr *= 0.7
            if lr < 1e-5:
                break

        if time.time() - last_improve > 6.0:
            break

        # gentle decay of learning rate as iterations progress
        lr *= 0.99998

    return best_seq, best_val


# ----------------------------------------------------------------------
def _deep_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Full optimisation pipeline for a single start:
    Adam (smooth‑max) → sub‑gradient (exp‑grad) → projected gradient.
    """
    best_seq, best_val = _adam_smooth_max(
        a0,
        n,
        conv_fft_len,
        corr_fft_len,
        deadline,
        global_best_seq,
        global_best_val,
    )
    best_seq, best_val = _subgradient_refinement(
        a0,
        n,
        conv_fft_len,
        corr_fft_len,
        deadline,
        best_seq,
        best_val,
    )
    best_seq, best_val = _projected_gradient_refine(
        a0,
        n,
        conv_fft_len,
        corr_fft_len,
        deadline,
        best_seq,
        best_val,
    )
    return best_seq, best_val


# ----------------------------------------------------------------------
def _generate_parametric_candidates(n: int, warm_start: np.ndarray | None):
    """
    Produce a very rich set of parameterised vectors.
    All candidates are normalised to sum≈1 and contain only non‑negative entries.
    """
    cand_list = []

    # ---- warm‑start and simple cyclic shifts ----
    if warm_start is not None:
        cand_list.append(warm_start.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand_list.append(np.roll(warm_start, shift))

    # ---- uniform baseline ----
    cand_list.append(np.full(n, 1.0 / n))

    # ---- Dirichlet draws – wide range of concentrations ----
    for conc in (0.001, 0.005, 0.01, 0.02, 0.05, 0.1, 0.3, 0.5, 1.0, 2.0, 4.0, 10.0):
        cand_list.append(np.random.dirichlet(np.full(n, conc)))

    # ---- edge‑exponential families (optional uniform mix) ----
    i = np.arange(n, dtype=np.float64)
    for alpha in np.logspace(-2, 1.5, num=22):
        edge = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
        edge /= edge.sum()
        for w in (0.0, 0.25, 0.5, 0.75, 1.0):
            shape = w * edge + (1.0 - w) / n
            cand_list.append(shape)

    # ---- edge‑power‑law families ----
    for p in np.linspace(0.2, 5.5, num=22):
        edge = (i + 1.0) ** (-p) + (n - i) ** (-p)
        edge /= edge.sum()
        for w in (0.0, 0.33, 0.66, 1.0):
            shape = w * edge + (1.0 - w) / n
            cand_list.append(shape)

    # ---- two‑Gaussian edges (different widths) ----
    for gamma in (0.025, 0.05, 0.1, 0.2, 0.5, 1.0):
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        shape /= shape.sum()
        cand_list.append(shape)

    # ---- sinusoidal half‑wave (positive part) ----
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand_list.append(shape)

    # ---- Beta‑shaped families (flexible edge/center bias) ----
    for alpha in (0.2, 0.5, 1.0, 2.0, 5.0):
        for beta in (0.2, 0.5, 1.0, 2.0, 5.0):
            edge = (i + 1) ** (alpha - 1) * (n - i) ** (beta - 1)
            edge = np.maximum(edge, 0.0)
            edge /= edge.sum()
            cand_list.append(edge)

    # ---- simple triangular / quadratic shapes ----
    tri = np.minimum(i + 1, n - i)
    tri = tri.astype(np.float64)
    tri /= tri.sum()
    cand_list.append(tri)

    quad = (i + 1) * (n - i)
    quad = quad.astype(np.float64)
    quad /= quad.sum()
    cand_list.append(quad)

    # ---- arcsine‑like (heavy edges) ----
    x = (i + 0.5) / n
    arcs = 1.0 / np.sqrt(np.clip(x * (1.0 - x), 1e-12, None))
    arcs /= arcs.sum()
    cand_list.append(arcs)

    # ---- harmonic edge boost ----
    harm = 1.0 / (i + 1.0) + 1.0 / (n - i)
    harm = np.maximum(harm, 0.0)
    harm /= harm.sum()
    cand_list.append(harm)

    # ---- sparse binary patterns (random subsets) ----
    for m_frac in (0.01, 0.02, 0.04, 0.08, 0.12):
        m = max(2, int(m_frac * n))
        mask = np.zeros(n, dtype=np.float64)
        idx = np.random.choice(n, size=m, replace=False)
        mask[idx] = 1.0 / m
        cand_list.append(mask)

    # ---- two‑spike edge pattern ----
    twospike = np.zeros(n, dtype=np.float64)
    twospike[0] = twospike[-1] = 0.5
    cand_list.append(twospike)

    return cand_list


# ----------------------------------------------------------------------
def _local_perturbations(
    base_seq: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Stochastic local search: random jitter + short sub‑gradient refinement.
    """
    a0 = base_seq.copy()
    while time.time() < deadline:
        eps = 0.015
        a_pert = a0 + eps * (np.random.rand(n) - 0.5)
        a_pert = np.maximum(a_pert, 0.0)
        s = a_pert.sum()
        if s < 1e-8:
            continue
        a_pert /= s

        a_ref, val_ref = _subgradient_refinement(
            a_pert,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            best_seq,
            best_val,
        )
        if val_ref < best_val:
            best_val = val_ref
            best_seq = a_ref.copy()
            a0 = best_seq.copy()      # recentre perturbations
            print(f"[Perturb] new best {best_val:.6f}")
        else:
            # moderate pull towards the previous best to avoid drift
            a0 = 0.92 * a0 + 0.08 * base_seq
    return best_seq, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Entry point – searches for a step‑function with minimal evaluation score.
    Returns the best sequence as a plain Python list.
    """
    np.random.seed(seed)
    # leave a tiny safety margin for the evaluation harness
    deadline = time.time() + budget_s - 0.4

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # --------------------------------------------------------------
    # Base grid size
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Warm‑start from a previously found construction (if any)
    warm_start = globals().get("height_sequence_1")
    if warm_start is not None:
        warm_start = np.asarray(warm_start, dtype=np.float64)
        if warm_start.shape[0] > n:
            warm_start = warm_start[:n]
        elif warm_start.shape[0] < n:
            warm_start = np.pad(warm_start, (0, n - warm_start.shape[0]))
        warm_start = np.maximum(warm_start, 0.0)
        if warm_start.sum() >= 0.01:
            warm_start = warm_start / warm_start.sum()
        else:
            warm_start = None
    else:
        warm_start = None

    # --------------------------------------------------------------
    # Stage 1 – generate a large pool of parametric candidates
    all_candidates = _generate_parametric_candidates(n, warm_start)

    # quick objective evaluation for each candidate (using the cheap _objective)
    cand_vals = []
    for cand in all_candidates:
        if time.time() >= deadline:
            break
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 1e-8:
            cand = np.random.dirichlet(np.ones(n))
            s = cand.sum()
        cand /= s
        val = _objective(cand, n, conv_fft_len)
        cand_vals.append((val, cand))

    # --------------------------------------------------------------
    # keep the most promising starts (a bit more than before)
    cand_vals.sort(key=lambda x: x[0])
    top_k = 20
    top_candidates = [c for _, c in cand_vals[:top_k]]

    # --------------------------------------------------------------
    # Stage 2 – deep optimisation of each start
    for idx, start_vec in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        print(f"\n--- Deep optimisation on start #{idx + 1}/{len(top_candidates)} ---")
        best_seq, best_val = _deep_optimize(
            start_vec,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            best_seq,
            best_val,
        )

    # --------------------------------------------------------------
    # Stage 3 – stochastic local perturbations around the current best
    if best_seq is not None and time.time() < deadline - 5.0:
        print("\n--- Local stochastic perturbations ---")
        best_seq, best_val = _local_perturbations(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            best_seq,
            best_val,
        )

    # --------------------------------------------------------------
    # Stage 4 – upscale to a larger grid if time permits
    time_left = deadline - time.time()
    if time_left > 40.0:          # need a little head‑room for the bigger run
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        if best_seq is not None:
            # linear interpolation to the finer grid
            x_old = np.linspace(0, 1, n, endpoint=False)
            x_new = np.linspace(0, 1, n_big, endpoint=False)
            a_big = np.interp(x_new, x_old, best_seq)
            a_big = np.maximum(a_big, 0.0)
            a_big /= a_big.sum()
        else:
            a_big = np.random.dirichlet(np.ones(n_big))

        print("\n--- Optimising on larger grid (n=8192) ---")
        # on the large grid we run a slightly trimmed pipeline
        best_seq, best_val = _adam_smooth_max(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            best_seq,
            best_val,
        )
        best_seq, best_val = _subgradient_refinement(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            best_seq,
            best_val,
        )
        # a final short projected‑gradient pass
        best_seq, best_val = _projected_gradient_refine(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            best_seq,
            best_val,
        )
        if best_seq is not None and best_seq.shape[0] == n_big:
            n = n_big

    # --------------------------------------------------------------
    # Fallback – uniform vector if everything failed
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Clip to the evaluator's allowed range (0…1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"\n*** Finished search: best objective ≈ {best_val:.6f} (n={best_seq.shape[0]})")
    return best_seq.tolist()
