"""
Propose a step‑function (non‑negative vector) that minimizes

        F(a) = 2·n·max(a∗a) / (sum(a)^2)

Algorithm
---------

1. **Warm start** – reuse the previously best sequence (global
   ``height_sequence_1``) if it exists; otherwise start from a few
   simple handcrafted shapes (edge‑concentrated power‑law,
   double‑exponential, random Dirichlet).

2. **Adaptive Adam on a smooth‑max surrogate** – we optimise the
   soft‑max of the convolution using a soft‑max temperature that grows
   exponentially.  The variables are kept on the simplex by a
   soft‑max of unrestricted logits.  This phase is run for as many
   iterations as the time budget allows (up to several hundred‑thousand).

3. **Multiplicative sub‑gradient (exponentiated‑gradient)** – after the
   Adam phase we switch to the exact max‑convolution objective using a
   sub‑gradient that puts uniform weight on all active (maximal)
   convolution indices.  A step size ηₜ = η₀/√t is used together with
   occasional random perturbations to escape plateaus.

4. **Resolution lifting** – the best vector is linearly interpolated to a
   finer grid (8192 → 16384 → 32768 points) whenever enough time is left.
   For each new resolution we repeat (2) and (3) with longer budgets.

5. **Cheap hill‑climbing** – a final local search moves a small fraction
   of mass between two coordinates and keeps the move if the exact
   objective improves.

All phases respect the global time budget (default 1000 s).  The routine
always returns the best sequence found, clipped to the evaluator’s
allowed range.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a):
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i = 0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


# ----------------------------------------------------------------------
def _initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Generate a modest, diverse set of seeds."""
    cand = []

    # warm start and a few rotations
    if warm is not None:
        cand.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    # random Dirichlet
    cand.append(np.random.dirichlet(np.ones(n)))

    # edge‑concentrated power‑law (negative exponent)
    idx = np.arange(n, dtype=np.float64) + 1.0
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.5, -0.3, 9):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    # symmetric double‑exponential
    i = np.arange(n, dtype=np.float64)
    for gamma in np.geomspace(0.3, 8.0, 6):
        shape = np.exp(-gamma * i) + np.exp(-gamma * (n - 1 - i))
        shape /= shape.sum()
        cand.append(shape)

    # half‑wave sinusoid (positive part only)
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand.append(shape)

    # uniform
    cand.append(np.full(n, 1.0 / n))

    return cand


# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Adam on the log‑sum‑exp surrogate of max(convolution).
    Returns the best vector (sum≈1) encountered and its exact objective.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.07
    lam = 0.5           # temperature (small → smooth)
    lam_factor = 1.03   # exponential schedule

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        # ----- simplex via soft‑max -----
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()                # sum(a) == 1

        # ----- exact objective (for bookkeeping) -----
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5_000 == 0:
                print(f"[Adam iter={it:06d}] new best {val:.6f}")

        # ----- smooth‑max weights (log‑sum‑exp) -----
        # w ∝ exp(lam * (conv - max_c))
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or not np.isfinite(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of surrogate -----
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)               # ∇_logits of surrogate

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- temperature schedule -----
        lam = min(lam * lam_factor, 1e6)

        # ----- occasional tiny random shake (helps escape plateaus) -----
        if it % 8_000 == 0 and time.time() + 5 > deadline:
            logits += (np.random.rand(n) - 0.5) * 4e-3

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
def _subgradient_opt(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
    eta0: float = 0.6,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (multiplicative) sub‑gradient on the true max.
    Returns the best vector seen and its exact objective.
    """
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s

    best_a = a.copy()
    best_val = float("inf")
    stall = 0
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            stall = 0
            if it % 5_000 == 0:
                print(f"[Subgrad iter={it:07d}] new best {val:.6f}")
        else:
            stall += 1

        # ----- active set (indices attaining the max) -----
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # ----- decaying step size -----
        eta = eta0 / np.sqrt(it + 1)

        # ----- multiplicative update & renormalisation -----
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        # ----- random shake when stuck -----
        if stall > 12_000:
            a = np.maximum(a + 0.02 * (np.random.rand(n) - 0.5), 0.0)
            a /= a.sum()
            stall = 0

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 150_000,
) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climbing: transfer a random fraction of mass from one
    index to another, keep the move if the exact objective improves.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    best_a = a.copy()
    it = 0

    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-12:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        # move a random fraction (up to 30 %) of the source mass
        delta = a[i] * (np.random.rand() * 0.3)
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()
        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5_000 == 0:
                print(f"[Local move {it}] improved to {val2:.6f}")

        it += 1

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with low evaluation score.
    The function respects the given time budget and returns the best
    sequence (as a Python list) found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.4   # safety margin

    best_seq = None
    best_val = float("inf")

    # ----- Phase 0 : warm start (if any) -----
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.ndim != 1:
            warm = None
    # -------------------------------------------------------------
    # Phase 1 : coarse resolution (n = 4096)
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    candidates = _initial_candidates(n, warm)

    # Give each candidate a modest amount of optimisation time.
    for cid, cand in enumerate(candidates):
        if time.time() >= deadline:
            break

        a0 = np.maximum(cand, 0.0)
        s = a0.sum()
        if s < 0.01:
            a0 = np.random.dirichlet(np.ones(n))
        else:
            a0 = a0 / s

        # ---- short Adam (≈8 s) ----
        cand_deadline = min(deadline - 5.0, time.time() + 8.0)
        a_adam, val_adam = _adam_optimize(
            a0, n, conv_fft_len, corr_fft_len, cand_deadline, max_iters=120_000
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[cand {cid}] Adam best {val_adam:.6f}")

        # ---- short sub‑gradient refinement (≈8 s) ----
        if time.time() < deadline:
            sub_deadline = min(deadline - 5.0, time.time() + 8.0)
            a_sub, val_sub = _subgradient_opt(
                best_seq, n, conv_fft_len, corr_fft_len, sub_deadline, max_iters=50_000
            )
            if val_sub < best_val:
                best_val = val_sub
                best_seq = a_sub.copy()
                print(f"[cand {cid}] Subgrad best {val_sub:.6f}")

    # -------------------------------------------------------------
    # Phase 2 : long sub‑gradient at the current best (still n=4096)
    if best_seq is not None and time.time() < deadline - 5:
        a_ref, val_ref = _subgradient_opt(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=300_000,
        )
        if val_ref < best_val:
            best_val = val_ref
            best_seq = a_ref.copy()
            print(f"[Refine] Subgrad long best {val_ref:.6f}")

    # -------------------------------------------------------------
    # Phase 3 : cheap random local search (still n=4096)
    if best_seq is not None and time.time() < deadline - 5:
        a_loc, val_loc = _local_random_search(
            best_seq, n, conv_fft_len, deadline, max_moves=120_000
        )
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()
            print(f"[Local] improved best {val_loc:.6f}")

    # -------------------------------------------------------------
    # Phase 4 : upscale to larger resolutions if time permits
    # (8192 → 16384 → 32768).  At each resolution we redo a long Adam
    # run followed by a long sub‑gradient run.
    for target_n in (8192, 16384, 32768):
        if time.time() + 40 > deadline:
            break
        # interpolate current best to new grid
        x_old = np.linspace(0.0, 1.0, best_seq.shape[0], endpoint=False)
        x_new = np.linspace(0.0, 1.0, target_n, endpoint=False)
        a_big = np.interp(x_new, x_old, best_seq)
        a_big = np.maximum(a_big, 0.0)
        a_big /= a_big.sum()

        conv_fft_len_big = _next_pow2(2 * target_n - 1)
        corr_fft_len_big = _next_pow2(3 * target_n - 2)

        # ---- Adam on the big grid (≈30 s) ----
        big_deadline = min(deadline - 5.0, time.time() + 30.0)
        a_big_adam, val_big_adam = _adam_optimize(
            a_big,
            target_n,
            conv_fft_len_big,
            corr_fft_len_big,
            big_deadline,
            max_iters=250_000,
        )
        if val_big_adam < best_val:
            best_val = val_big_adam
            best_seq = a_big_adam.copy()
            print(f"[Big {target_n}] Adam best {val_big_adam:.6f}")

        # ---- sub‑gradient on the big grid (≈30 s) ----
        if time.time() < deadline - 5:
            sub_big_deadline = min(deadline - 5.0, time.time() + 30.0)
            a_big_sub, val_big_sub = _subgradient_opt(
                best_seq,
                target_n,
                conv_fft_len_big,
                corr_fft_len_big,
                sub_big_deadline,
                max_iters=200_000,
            )
            if val_big_sub < best_val:
                best_val = val_big_sub
                best_seq = a_big_sub.copy()
                print(f"[Big {target_n}] Subgrad best {val_big_sub:.6f}")

        # ---- final tiny local search (if still time) ----
        if time.time() < deadline - 5:
            a_loc2, val_loc2 = _local_random_search(
                best_seq, target_n, conv_fft_len_big, deadline, max_moves=80_000
            )
            if val_loc2 < best_val:
                best_val = val_loc2
                best_seq = a_loc2.copy()
                print(f"[Local {target_n}] improved best {val_loc2:.6f}")

    # -------------------------------------------------------------
    # Fallback – ensure a valid return value
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Clip to evaluator's allowed bounds and normalise for safety
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    total = best_seq.sum()
    if total > 0:
        best_seq = best_seq / total

    print(f"Final best value ≈ {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
