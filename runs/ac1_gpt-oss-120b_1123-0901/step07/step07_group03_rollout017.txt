def _lbfgs_optimize(a0, n, conv_fft_len, corr_fft_len, deadline, lam_start=1.0, lam_factor=10.0, max_outer=4):
    # a0 normalized sum=1
    logits = np.log(a0 + 1e-15)
    best_a = a0
    best_val = _objective(a0)
    lam = lam_start
    for outer in range(max_outer):
        # define objective function closure capturing lam
        def fun(logits_var):
            # compute a as softmax
            max_log = logits_var.max()
            exp_log = np.exp(logits_var - max_log)
            a = exp_log / exp_log.sum()
            conv = _full_convolution(a, conv_fft_len)
            max_c = conv.max()
            # compute smoothed value
            w = np.exp(lam * (conv - max_c))
            w_sum = w.sum()
            smooth_val = max_c + (1.0/lam) * np.log(w_sum)
            # gradient
            w = w / w_sum
            grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
            avg = np.dot(a, grad_a)
            grad_log = a * (grad_a - avg)
            return smooth_val, grad_log
        # wrapper for scipy: returns value and gradient
        def fun_wrap(logits_var):
            val, grad = fun(logits_var)
            return val, grad
        # set options such as maxiter based on time left
        remaining = deadline - time.time()
        if remaining < 5.0:
            break
        # allocate roughly limited iter: maybe maxiter = min(200, int(remaining*5)) (approx 5 iter per sec)
        maxiter = int(min(200, remaining * 5))
        try:
            res = scipy.optimize.minimize(fun_wrap, logits, method='L-BFGS-B', jac=True,
                                          options={'maxiter': maxiter, 'disp': False})
        except Exception as e:
            break
        logits = res.x
        # compute a from final logits
        max_log = logits.max()
        exp_log = np.exp(logits - max_log)
        a = exp_log / exp_log.sum()
        # compute exact objective
        val_exact = _objective(a)
        if val_exact < best_val:
            best_val = val_exact
            best_a = a.copy()
            print(f"[L-BFGS lam={lam:.1e} outer={outer}] new best {best_val:.6f}")
        # increase lam for next outer
        lam = min(lam * lam_factor, 1e6)
        # early stop if time out
        if time.time() >= deadline:
            break
    return best_a, best_val

def _cross_correlation(w, a, fft_len):
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]
```

"""
Propose a step‑function with a low value of

    F(a) = 2·n·max(a∗a) / (sum(a)²) ,   a_i ≥ 0 ,   sum(a) > 0.

The algorithm works in three stages:

1. **Diverse warm‑starts** – a pool of candidates is built from
   * the previous best solution,
   * several Dirichlet samples (different concentrations),
   * anti‑pair ordering, strided permutations and edge‑heavy shapes.
   The best of this pool is taken as an initial point.

2. **Smooth‑max optimisation** – the exact max in the objective is
   replaced by a log‑sum‑exp (smooth‑max) surrogate.
   Optimisation is performed on the log‑itself (logits) of a soft‑max
   parametrisation using L‑BFGS‑B.  The temperature (`lam`) is
   increased geometrically (1 → 10 → 100 → …) so that the surrogate
   approaches the true max.

3. **Exact‑max fine‑tuning** – a multiplicative exponentiated‑gradient
   (sub‑gradient) phase minimises the exact objective, followed by a
   cheap stochastic hill‑climber (swap / mass‑transfer moves).

All steps respect the overall time budget (default 1000 s).  The
function returns a plain Python list that can be fed to
`evaluate_sequence`.  The best sequence found is also stored in the
global variable `height_sequence_1` for warm‑starting later runs.
"""

import time
import math
import numpy as np

# FFT utilities (always on the top level – no closures)
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
# Exact objective (fast, no security checks)
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(a*a) / (sum(a)²);  assumes a is non‑negative."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    conv = _full_convolution(a, _next_pow2(2 * a.shape[0] - 1))
    return 2.0 * a.shape[0] * conv.max() / (s * s)

# Candidate generators ---------------------------------------------------
def _warm_start(n: int):
    """Return a resized, normalised copy of the global warm‑start if present."""
    warm = globals().get("height_sequence_1")
    if warm is None:
        return None
    warm = np.asarray(warm, dtype=np.float64)
    if warm.shape[0] > n:
        warm = warm[:n]
    elif warm.shape[0] < n:
        warm = np.pad(warm, (0, n - warm.shape[0]))
    warm = np.maximum(warm, 0.0)
    total = warm.sum()
    if total < 0.01:
        return None
    return warm / total


def _uniform_candidate(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _random_dirichlet_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    return np.random.dirichlet(alpha * np.ones(n)).astype(np.float64)


def _anti_pair_arrangement(vals: np.ndarray) -> np.ndarray:
    """Largest opposite smallest, second‑largest opposite second‑smallest, …"""
    n = len(vals)
    s = np.sort(vals)[::-1]               # descending
    a = np.empty(n, dtype=np.float64)
    half = n // 2
    a[:half] = s[:half]                   # biggest half on the left
    a[-half:] = s[-half:][::-1]           # smallest half on the right
    if n % 2 == 1:
        a[half] = s[half]                 # median stays in centre
    return a


def _anti_pair_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    vals = np.random.dirichlet(alpha * np.ones(n))
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return _anti_pair_arrangement(vals)


def _triangle_candidate(n: int) -> np.ndarray:
    """Tent shape (large at the edges, small in the centre)."""
    x = np.arange(n, dtype=np.float64)
    shape = np.minimum(x + 1, n - x)          # 1,2,…,⌈n/2⌉,…,2,1
    shape = shape / shape.sum()
    return shape


def _edge_heavy_candidate(n: int) -> np.ndarray:
    """Heavier mass near the two boundaries."""
    i = np.arange(n, dtype=np.float64)
    shape = 1.0 / (i + 1.0) + 1.0 / (n - i)
    shape = shape / shape.sum()
    return shape


def _strided_candidate(n: int, alpha: float, step: int) -> np.ndarray:
    """Dirichlet sample, sorted descending, then placed with stride `step`."""
    vals = np.random.dirichlet(alpha * np.ones(n))
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    sorted_vals = np.sort(vals)[::-1]
    idx = (np.arange(n) * step) % n
    perm = np.empty(n, dtype=np.float64)
    perm[idx] = sorted_vals
    # normalise again – numerical noise may break sum==1
    return perm / perm.sum()

# Smooth‑max objective needed by L‑BFGS
def _lbfgs_obj(logits: np.ndarray,
                n: int,
                conv_fft_len: int,
                corr_fft_len: int,
                lam: float):
    """Smooth‑max (log‑sum‑exp) surrogate and its gradient w.r.t. logits."""
    # soft‑max → a on the simplex
    mx = logits.max()
    exp_log = np.exp(logits - mx)
    a = exp_log / exp_log.sum()

    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()

    # smooth‑max weight vector
    w = np.exp(lam * (conv - max_c))
    w_sum = w.sum()
    if w_sum == 0 or np.isnan(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # gradient w.r.t. a
    grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

    # transform to gradient w.r.t. logits (soft‑max Jacobian)
    avg = np.dot(a, grad_a)
    grad_log = a * (grad_a - avg)

    # smooth‑max value
    smooth_val = max_c + (1.0 / lam) * np.log(w_sum)

    return smooth_val, grad_log


def _lbfgs_optimize(a0: np.ndarray,
                    n: int,
                    conv_fft_len: int,
                    corr_fft_len: int,
                    deadline: float,
                    lam_start: float = 1.0,
                    lam_factor: float = 10.0,
                    outer_steps: int = 4) -> tuple[np.ndarray, float]:
    """
    Multi‑stage L‑BFGS optimisation on the smooth‑max surrogate.
    The temperature `lam` is increased geometrically; after each stage
    the exact objective is evaluated and retained if improved.
    """
    import scipy.optimize as opt

    logits = np.log(a0 + 1e-15)          # start from a0 (already normalised)
    best_a = a0.copy()
    best_val = _objective(a0)
    lam = lam_start

    for outer in range(outer_steps):
        # time‑check
        if time.time() >= deadline - 5.0:
            break

        # allocate a modest iteration budget for each L‑BFGS call
        remaining = deadline - time.time()
        maxiter = int(min(150, remaining * 10))   # roughly 10 it/s is safe

        # minimise the smooth objective
        res = opt.minimize(
            fun=_lbfgs_obj,
            x0=logits,
            args=(n, conv_fft_len, corr_fft_len, lam),
            method='L-BFGS-B',
            jac=True,
            options={'maxiter': maxiter, 'disp': False}
        )
        logits = res.x

        # evaluate exact objective at the new point
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()
        val = _objective(a)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[L‑BFGS outer={outer} lam={lam:.1e}] new best {best_val:.6f}")

        # increase temperature for the next outer stage
        lam = min(lam * lam_factor, 1e6)

    return best_a, best_val

# Adam (smooth‑max) – kept as a fallback if SciPy is unavailable
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    lr: float = 0.05,
    lam_start: float = 1.0,
    lam_factor: float = 1.07,
) -> tuple[np.ndarray, float]:
    """Adam on the log‑sum‑exp surrogate (same flavour as the original runner)."""
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lam = lam_start

    best_a = a0.copy()
    best_val = _objective(a0)

    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max → simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact conv & objective (for tracking)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val_exact = 2.0 * n * max_c
        if val_exact < best_val:
            best_val = val_exact
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Adam it={it:06d}] new best exact {best_val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient w.r.t. a
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # anneal temperature → true max
        lam = min(lam * lam_factor, 1e6)

        # occasional jitter to avoid stagnation
        if it % 5000 == 0 and (deadline - time.time()) > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2

        it += 1

    return best_a, best_val

# Exact‑max exponentiated‑gradient sub‑gradient phase
def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 0.7,
) -> tuple[np.ndarray, float]:
    """Multiplicative (exponentiated‑gradient) refinement of the exact objective."""
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s

    best_a = a.copy()
    best_val = _objective(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active set for the exact max
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # step‑size schedule (slowly decaying)
        eta = eta0 / np.sqrt(it + 1)

        # multiplicative update & renormalisation
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val

# Simple simulated‑annealing‑style hill‑climber (swap / transfer)
def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 120_000,
) -> tuple[np.ndarray, float]:
    """Local stochastic search: swaps or fractional transfers of mass."""
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a)

    start_temp = 0.01
    it = 0
    while time.time() < deadline and it < max_moves:
        if np.random.rand() < 0.5:                     # swap two entries
            i, j = np.random.randint(n, size=2)
            if i != j:
                a2 = a.copy()
                a2[i], a2[j] = a2[j], a2[i]
            else:
                continue
        else:                                          # transfer a fraction
            i = np.random.randint(n)
            if a[i] < 1e-14:
                it += 1
                continue
            j = np.random.randint(n)
            if i == j:
                it += 1
                continue
            delta = a[i] * (np.random.rand() * 0.4)   # up to 40 % move
            a2 = a.copy()
            a2[i] -= delta
            a2[j] += delta

        val2 = _objective(a2)

        if val2 < best_val:               # accept improvement
            best_val = val2
            best_a = a2.copy()
            a = a2
        else:
            # probabilistic uphill move
            temp = start_temp * (1.0 - (it / max_moves))
            if temp > 0.0 and np.random.rand() < math.exp(-(val2 - best_val) / temp):
                a = a2

        if it % 5000 == 0:
            print(f"[Local move {it:06d}] best {best_val:.6f}")

        it += 1

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises
    F(a)=2·n·max(a∗a)/(sum(a)²).

    The routine respects the given time budget and returns the best
    sequence (as a Python list) found.
    """

    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Main resolution (larger grids give more flexibility)
    n = 8192
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # 1) Build a diverse pool of start vectors
    candidates = []

    warm = _warm_start(n)
    if warm is not None:
        candidates.append(warm.copy())
        # a few rotated copies – rotation does not change the max‑conv value
        for shift in (n // 4, n // 2, 3 * n // 4):
            candidates.append(np.roll(warm, shift))

    candidates.append(_uniform_candidate(n))
    candidates.append(_edge_heavy_candidate(n))
    candidates.append(_triangle_candidate(n))

    alphas = [0.2, 0.5, 0.8, 1.0, 2.0]
    steps = [n // 5, n // 3, n // 2, (2 * n) // 3, (4 * n) // 5]

    for alpha in alphas:
        # plain Dirichlet
        candidates.append(_random_dirichlet_candidate(n, alpha))
        # anti‑pair ordering
        candidates.append(_anti_pair_candidate(n, alpha))
        # three strided permutations
        for step in steps[:3]:
            candidates.append(_strided_candidate(n, alpha, step))

    # ------------------------------------------------------------------
    # 2) Quick evaluation of the pool, keep the best starter
    best_seq = None
    best_val = float("inf")
    for idx, cand in enumerate(candidates):
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 0.01:                     # fallback to a fresh Dirichlet
            cand = np.random.dirichlet(np.ones(n))
            s = cand.sum()
        cand = cand / s                    # enforce sum = 1
        val = _objective(cand)
        print(f"[Init cand {idx:02d}] value {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = cand.copy()

    # ------------------------------------------------------------------
    # 3) Smooth‑max optimisation (L‑BFGS) – primary optimisation engine
    use_scipy = True
    try:

    except Exception:

    if use_scipy:
        a_opt, val_opt = _lbfgs_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            lam_start=1.0,
            lam_factor=10.0,
            outer_steps=4,
        )
        if val_opt < best_val:
            best_val = val_opt
            best_seq = a_opt.copy()
            print(f"[L‑BFGS] improved to {val_opt:.6f}")
    else:
        # fall back to the Adam implementation from the original runner
        adam_deadline = min(deadline - 15.0, time.time() + 300.0)
        a_adam, val_adam = _adam_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            adam_deadline,

            lr=0.06,

        if val_adam < best_val:

    # ------------------------------------------------------------------

    a_sub, val_sub = _subgradient_optimize(

    total = best_seq.sum()
    if total < 0.01:
        best_seq = np.full_like(best_seq, 1.0 / n)
    else:
        best_seq = best_seq / total

    # Export as a warm start for possible later runs
    globals()["height_sequence_1"] = best_seq.tolist()

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
