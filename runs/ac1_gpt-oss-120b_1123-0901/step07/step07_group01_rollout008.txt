def _targeted_annealing(a0, n, fft_len, deadline, cur_best_val):
    a = a0.copy()
    conv = _full_convolution(a, fft_len)
    cur_max = conv.max()
    cur_val = 2.0 * n * cur_max
    best_a = a.copy()
    best_val = cur_val
    start = time.time()
    it = 0
    # parameters
    while time.time() < deadline:
        elapsed = time.time() - start
        fraction = elapsed / (deadline - start + 1e-12)
        # cooling schedule
        T = max(1e-6, 0.1 * (1.0 - fraction))
        move_scale = max(0.001, 0.2 * (1.0 - fraction))
        # find current peak index
        k0 = np.argmax(conv)
        # compute counterpart array b as described
        b = np.zeros(n, dtype=np.float64)
        i_start = max(0, k0 - (n-1))
        i_end = min(n-1, k0)
        # for i in range(i_start, i_end+1):
        # b[i] = a[k0 - i]
        idx = np.arange(i_start, i_end+1)
        b[idx] = a[k0 - idx]  # works using numpy broadcasting
        # contributions
        contrib = a * b
        i_src = np.argmax(contrib)  # source index to reduce
        if a[i_src] < 1e-15:
            # nothing to move
            break
        # find candidate target indices where b is minimal (prefer zero)
        min_b = b.min()
        cand_targets = np.where(b == min_b)[0]
        if cand_targets.size == 0:
            cand_targets = np.arange(n)
        j = np.random.choice(cand_targets)
        if i_src == j:
            # choose a different random target
            j = np.random.randint(n-1)
            if j >= i_src:
                j += 1
        # delta random fraction scaled
        delta = a[i_src] * move_scale * np.random.rand()
        # apply move
        a_new = a.copy()
        conv_new = conv.copy()
        a_new[i_src] -= delta
        a_new[j] += delta
        # update convolution using delta move
        # subtract contributions from i_src, add from j
        # conv_new[i_src : i_src + n] -= 2*delta*a
        conv_new[i_src : i_src + n] -= 2.0 * delta * a
        conv_new[j : j + n] += 2.0 * delta * a
        conv_new[2 * i_src] += delta * delta
        conv_new[2 * j] += delta * delta
        conv_new[i_src + j] -= 2.0 * delta * delta
        new_max = conv_new.max()
        new_val = 2.0 * n * new_max
        # accept if improvement, else Metropolis based on T
        if new_val < cur_val or np.random.rand() < np.exp(-(new_val - cur_val) / T):
            a, conv = a_new, conv_new
            cur_val = new_val
            cur_max = new_max
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[Targeted it={it}] new best {best_val:.6f}")
        it += 1
        if it % 5000 == 0:
            # progress print
            print(f"[Targeted it={it}] best {best_val:.6f}")
    return best_a, best_val

"""
Hybrid optimiser for the step‑function objective

    F(a) = 2·n·max(a∗a) / (sum(a)^2)

The routine builds a diverse pool of warm‑starts, runs a short Adam
optimisation on a smooth‑max surrogate, refines the result with an
exact‑max exponentiated‑gradient sub‑gradient method, and then performs
a *targeted* simulated‑annealing phase that deliberately moves mass
away from the current convolution peak.  If time permits the solution
is up‑sampled to a finer grid (8192 points) and refined again.

Key new idea: instead of random moves the annealer picks the index
`i` that contributes most to the current maximal convolution entry and
shifts a small amount of its mass to an index `j` whose contribution to
that entry is (almost) zero.  This dramatically accelerates the
reduction of the sup‑norm of the self‑convolution, allowing us to push
the bound below the previous 1.5077 → target ≈1.503.

All helper functions are top‑level, contain no closures and respect
the 1000 s time budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via FFT (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for the gradient."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
def _initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Create a diverse pool of start vectors."""
    cand = []

    # --- warm start ----------------------------------------------------
    if warm is not None:
        cand.append(warm.copy())
        # few simple rotations (helpful for periodic patterns)
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    # --- classic families ------------------------------------------------
    cand.append(np.random.dirichlet(np.ones(n)))          # uniform Dirichlet
    cand.append(np.random.dirichlet(0.2 * np.ones(n)))    # spiky Dirichlet

    # power‑law edge‑concentrated shapes
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in (-2.0, -1.5, -1.0, -0.75):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    # arcsine‑type (Beta(½,½) analogue)
    arcs = 1.0 / np.sqrt(np.arange(1, n + 1) *
                         np.arange(n, 0, -1, dtype=np.float64))
    arcs /= arcs.sum()
    cand.append(arcs)

    # double‑Gaussian edge shapes
    i = np.arange(n, dtype=np.float64)
    for gamma in (0.3, 0.7, 1.5):
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        shape /= shape.sum()
        cand.append(shape)

    # half‑sine (positive half‑wave)
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand.append(shape)

    # Hann window (raised cosine) – another smooth edge shape
    hann = 0.5 * (1 - np.cos(2 * np.pi * np.arange(n) / (n - 1)))
    hann /= hann.sum()
    cand.append(hann)

    # simple triangular (Beta(2,2) analogue)
    tri = (np.arange(1, n + 1) * np.arange(n, 0, -1)).astype(float)
    tri = np.maximum(tri, 0.0)
    tri /= tri.sum()
    cand.append(tri)

    # uniform
    cand.append(np.full(n, 1.0 / n))

    # spike families (k equally‑spaced non‑zero entries)
    for k in (64, 128, 256, 512, 1024):
        a = np.zeros(n, dtype=np.float64)
        pos = np.round(np.linspace(0, n - 1, k)).astype(int)
        a[pos] = 1.0 / k
        cand.append(a)

    return cand


# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   conv_fft_len: int,
                   corr_fft_len: int,
                   deadline: float,
                   max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """Adam on the log‑sum‑exp (smooth‑max) surrogate."""
    eps = 1e-15
    logits = np.log(a0 + eps)            # work in log‑space → positivity
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.04                # a touch larger than the original
    lam = 0.3                # initial temperature for smooth‑max
    lam_factor = 1.012       # slower anneal for stability

    best_a = a0.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        # ----- smooth‑max (log‑sum‑exp) -----
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # ----- gradient of smooth‑max -----
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)      # ∇log‑softmax

        # ----- Adam step -----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        lam = min(lam * lam_factor, 1e6)

        # occasional jitter near the deadline to escape plateaus
        if it % 4000 == 0 and time.time() > deadline - 5:
            logits += (np.random.rand(n) - 0.5) * 1e-3

        it += 1

    return best_a, best_val


def _subgradient_optimize(a0: np.ndarray,
                          n: int,
                          conv_fft_len: int,
                          corr_fft_len: int,
                          deadline: float,
                          max_iters: int = 300_000,
                          eta0: float = 0.6) -> tuple[np.ndarray, float]:
    """Exact‑max sub‑gradient descent using exponentiated‑gradient updates."""
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s

    best_a = a.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 2000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # diminishing step‑size (standard EG schedule)
        eta = eta0 / math.sqrt(it + 1)

        # multiplicative (exponentiated‑gradient) update
        a = a * np.exp(-eta * grad_a)

        # renormalise onto the simplex
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
def _targeted_annealing(a0: np.ndarray,
                        n: int,
                        fft_len: int,
                        deadline: float,
                        cur_best_val: float) -> tuple[np.ndarray, float]:
    """
    Fast annealer that *targets* the current convolution peak.
    Moves are deliberately chosen from the index with the largest product
    a[i]·a[k0‑i] (i.e. the biggest contributor to the peak) to a location
    that contributes almost nothing.
    """
    a = a0.copy()
    conv = _full_convolution(a, fft_len)
    cur_max = conv.max()
    cur_val = 2.0 * n * cur_max
    best_a = a.copy()
    best_val = cur_val

    start = time.time()
    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        fraction = elapsed / (deadline - start + 1e-12)

        # linear cooling (very low temps at the end)
        T = max(1e-6, 0.12 * (1.0 - fraction))
        # move magnitude shrinks as we cool
        move_scale = max(0.001, 0.25 * (1.0 - fraction))

        # peak index of the convolution
        k0 = int(conv.argmax())

        # counterpart array b[i] = a[k0‑i] (zero where out of range)
        b = np.zeros(n, dtype=np.float64)
        i_start = max(0, k0 - (n - 1))
        i_end = min(n - 1, k0)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        b[idx] = a[k0 - idx]

        # product contribution to the peak
        contrib = a * b
        i_src = int(contrib.argmax())
        if a[i_src] < 1e-15:
            # nothing to move – give up early
            break

        # candidate targets: indices where the counterpart is minimal
        min_b = b.min()
        cand_targets = np.where(b == min_b)[0]
        if cand_targets.size == 0:
            cand_targets = np.arange(n)
        j = int(np.random.choice(cand_targets))
        if j == i_src:
            # fall back to a random different index
            j = np.random.randint(n - 1)
            if j >= i_src:
                j += 1

        # random fraction of the source mass (scaled by move_scale)
        delta = a[i_src] * move_scale * np.random.rand()
        if delta <= 0.0:
            it += 1
            continue

        # ----- perform the move and update the convolution in O(n) -----
        a_new = a.copy()
        conv_new = conv.copy()
        a_new[i_src] -= delta
        a_new[j] += delta

        # linear update of the convolution (see derivation in the comment)
        conv_new[i_src : i_src + n] -= 2.0 * delta * a
        conv_new[j : j + n] += 2.0 * delta * a
        conv_new[2 * i_src] += delta * delta
        conv_new[2 * j] += delta * delta
        conv_new[i_src + j] -= 2.0 * delta * delta

        new_max = conv_new.max()
        new_val = 2.0 * n * new_max

        # Metropolis acceptance
        if new_val < cur_val or np.random.rand() < np.exp(-(new_val - cur_val) / T):
            a, conv = a_new, conv_new
            cur_val = new_val
            cur_max = new_max
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[Targeted it={it:06d}] new best {best_val:.6f}")

        it += 1
        if it % 5000 == 0:
            print(f"[Targeted it={it:06d}] best {best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
def _interpolate_and_refine(a_small: np.ndarray,
                            n_big: int,
                            deadline: float) -> tuple[np.ndarray, float]:
    """Upsample to a larger grid and run a short Adam + sub‑gradient pass."""
    n_small = a_small.shape[0]
    x_small = np.linspace(0.0, 1.0, n_small, endpoint=False)
    x_big = np.linspace(0.0, 1.0, n_big, endpoint=False)
    a_big = np.interp(x_big, x_small, a_small)
    a_big = np.maximum(a_big, 0.0)
    a_big /= a_big.sum()

    conv_fft_len_big = _next_pow2(2 * n_big - 1)
    corr_fft_len_big = _next_pow2(3 * n_big - 2)

    # a brief Adam run on the finer grid
    a_adam, _ = _adam_optimize(a_big, n_big, conv_fft_len_big,
                               corr_fft_len_big, deadline,
                               max_iters=80_000)

    # followed by a few sub‑gradient steps
    a_ref, val_ref = _subgradient_optimize(a_adam, n_big,
                                           conv_fft_len_big,
                                           corr_fft_len_big,
                                           deadline, max_iters=100_000)

    return a_ref, val_ref

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises
    `2·n·max(a∗a)/(sum a)^2`.  The routine uses Adam,
    exponentiated‑gradient sub‑gradient descent and a *targeted*
    annealing phase, followed by optional refinement on a finer grid.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ---- problem size ---------------------------------------------------
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ---- warm start ------------------------------------------------------
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # ---- build diverse pool of start vectors ----------------------------
    candidates = _initial_candidates(n, warm)

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ---- optimise each candidate (short budget per candidate) -----------
    for cid, cand in enumerate(candidates):
        if time.time() >= deadline:
            break

        a0 = np.maximum(cand, 0.0)
        if a0.sum() < 0.01:
            a0 = np.random.dirichlet(np.ones(n))
        else:
            a0 /= a0.sum()

        # Give each candidate at most ~20 s (adjustable)
        local_deadline = min(deadline - 5.0, time.time() + 20.0)

        # Adam (smooth‑max surrogate)
        a_adam, val_adam = _adam_optimize(a0, n, conv_fft_len,
                                          corr_fft_len, local_deadline,
                                          max_iters=200_000)
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[cand {cid}] Adam improved best to {val_adam:.6f}")

        # Sub‑gradient refinement on the *global* best so far
        if time.time() < deadline:
            a_sub, val_sub = _subgradient_optimize(best_seq, n,
                                                   conv_fft_len,
                                                   corr_fft_len,
                                                   deadline,
                                                   max_iters=120_000)
            if val_sub < best_val:
                best_val = val_sub
                best_seq = a_sub.copy()
                print(f"[cand {cid}] Sub‑grad improved best to {val_sub:.6f}")

    # --------------------------------------------------------------------
    # Heavier *targeted* annealing on the current global best
    if best_seq is not None and time.time() < deadline - 5:
        a_target, val_target = _targeted_annealing(best_seq, n,
                                                   conv_fft_len,
                                                   deadline,
                                                   best_val)
        if val_target < best_val:
            best_val = val_target
            best_seq = a_target.copy()
            print(f"[Targeted] best improved to {val_target:.6f}")

    # --------------------------------------------------------------------
    # If time allows, upscale to 8192 points and repeat a short refinement
    if time.time() + 30 < deadline:
        n_big = 8192
        a_big, val_big = _interpolate_and_refine(
            best_seq if best_seq is not None else np.random.dirichlet(np.ones(n_big)),
            n_big,
            deadline)
        if val_big < best_val:
            best_val = val_big
            best_seq = a_big.copy()
            print(f"[Big] final best {val_big:.6f}")

    # --------------------------------------------------------------------
    # Fallback – guarantee a valid return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the limits enforced by evaluate_sequence
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Report the final objective using the provided evaluator
    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"Final reported value {final_val:.6f} (length {len(best_seq)})")
    except Exception:
        print(f"Final internal best value {best_val:.6f} (length {len(best_seq)})")

    return best_seq.tolist()
