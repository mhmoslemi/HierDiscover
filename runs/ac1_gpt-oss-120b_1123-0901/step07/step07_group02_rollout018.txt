def propose_candidate(seed=0, budget_s=1000):
    import numpy as np, time, scipy.optimize as opt

    start = time.time()
    deadline = start + budget_s - 0.5
    np.random.seed(seed)
    # Choose n_init = 4096
    n = 4096
    conv_fft_len = _next_pow2(2*n - 1)
    corr_fft_len = _next_pow2(3*n - 2)

    # Warm start:
    warm = globals().get("height_sequence_1")
    if warm is not None:
        a0 = np.asarray(warm, dtype=np.float64)
        # ensure length n
        if a0.shape[0] != n:
            # resample
            x_old = np.linspace(0,1,a0.shape[0],endpoint=False)
            x_new = np.linspace(0,1,n,endpoint=False)
            a0 = np.interp(x_new, x_old, a0)
    else:
        # default: edge-concentrated shape using product^alpha
        idx = np.arange(1,n+1)
        prod = idx * (n - idx + 1)
        shape = prod ** -0.8
        shape = np.clip(shape, 0, None)
        shape = shape / shape.sum()
        a0 = shape

    # Ensure nonneg, sum >= 0.01 else random dirichlet.
    a0 = np.maximum(a0, 0.0)
    if a0.sum() < 0.01:
        a0 = np.random.dirichlet(np.ones(n))
    a0 = a0 / a0.sum()

    # Parameterization via logits
    eps = 1e-15
    logits = np.log(a0 + eps)

    # lambda schedule
    lambdas = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 50.0, 100.0]
    # for each lambda, run L-BFGS
    for lam in lambdas:
        if time.time() >= deadline:
            break
        # define objective function that returns (value, grad) for minimize
        def fun_and_grad(logits_vec, lam):
            # compute a
            mx = logits_vec.max()
            exp_log = np.exp(logits_vec - mx)
            a = exp_log / exp_log.sum()
            # conv
            conv = _full_convolution(a, conv_fft_len)
            max_c = conv.max()
            # stable log-sum-exp for smooth max
            exp_terms = np.exp(lam * (conv - max_c))
            sum_exp = exp_terms.sum()
            smooth_max = max_c + np.log(sum_exp) / lam
            # gradient wrt a
            w = exp_terms / sum_exp  # weight vector length 2n-1
            grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
            # gradient wrt logits via chain rule
            avg = np.dot(a, grad_a)
            grad_log = a * (grad_a - avg)
            return smooth_max, grad_log
        # Use SciPy minimize
        res = opt.minimize(fun_and_grad, logits, args=(lam,), method='L-BFGS-B',
                           jac=True, options={'maxiter': 300, 'disp': False})
        logits = res.x
        # logging

        a = a / a.sum()
        conv = _full_convolution(a, conv_fft_len)
        val = 2.0 * n * conv.max()
        print(f"[L-BFGS lam={lam}] val={val:.6f}")

    # after lam schedule, compute best a
    a = np.exp(logits - logits.max())
    a = a / a.sum()

    # Phase C: subgradient long
    # Determine remaining time
    sub_deadline = deadline - 10.0  # keep margin for SA
    if sub_deadline > time.time():

def _simulated_annealing(a, n, conv_fft_len, deadline, max_moves=None):
    # start with current a. Compute current best_val.
    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    max_moves = max_moves or 200_000
    # temp schedule: start T = best_val * 0.1 maybe
    T0 = best_val * 0.1  # relative scale
    # linear or exponential cooling: T = T0 * (0.99 ** move)
    T = T0
    for move in range(max_moves):
        if time.time() >= deadline:
            break
        # pick source i with a[i] > eps
        i = np.random.randint(n)
        if a[i] < 1e-12:
            continue
        # target j
        j = np.random.randint(n)
        if i == j:
            continue
        # delta fraction
        frac = np.random.rand() * 0.3  # up to 30% of source
        delta = a[i] * frac
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # compute new obj
        conv_new = _full_convolution(a_new, conv_fft_len)
        val_new = 2.0 * n * conv_new.max()
        # acceptance:
        if val_new < best_val:
            best_val = val_new
            best_a = a_new.copy()
            a = a_new
        else:
            # prob
            prob = np.exp(-(val_new - best_val) / T) if T > 0 else 0.0
            if np.random.rand() < prob:
                a = a_new
        # anneal temperature
        if move % 500 == 0 and move > 0:
            T *= 0.95
    return best_a, best_val

The program searches for a non‑negative step‑function `a` (length ≈ 4096) that minimises  

\[
F(a)=\frac{2\,n\,\max (a*a)}{(\sum a)^2}\; .
\]

Because the denominator is forced to 1 (the vector is normalised), the task is
equivalent to **reducing the peak of the autocorrelation** `a*a`.

The optimiser works in three stages:

1. **Warm‑start** – load the previous best solution (`height_sequence_1`) or
   build a simple edge‑concentrated shape.
2. **Deterministic optimisation** – minimise a *smooth* approximation of the
   peak using L‑BFGS‑B (via `scipy.optimize.minimize`).  
   The smooth surrogate is the log‑sum‑exp of the convolution with a slowly
   increasing temperature `λ`.  The optimisation is performed on the
   log‑soft‑max (logits) so the vector always stays on the probability simplex.
3. **Refinement** – a long exponentiated‑gradient sub‑gradient phase
   (the same update that was used in the reference code) followed by a
   cheap simulated‑annealing hill‑climber that is allowed to accept occasional
   worsening moves.  The two phases are executed until the time budget expires.
4. **Optional up‑sampling** – if enough time remains the solution is
   interpolated to 8192 points and the whole pipeline (L‑BFGS → sub‑gradient →
   SA) is repeated on the finer grid.

All heavy work (convolution, cross‑correlation) is performed with FFTs,
making each iteration cheap enough to run hundreds of thousands of steps
well inside the 1000 s limit.

```python
#!/usr/bin/env python
# -*- coding: utf-8 -*-

"""
Search for a non‑negative step‑function (vector) that minimises

        F(a) = 2·n·max(a*a) / (sum(a)^2)

The algorithm consists of:
  1.  Warm‑starting from the previous best (or a simple edge‑concentrated shape).
  2.  Deterministic optimisation of a smooth “log‑sum‑exp’’ surrogate using
      L‑BFGS‑B on the log‑soft‑max (logits) representation.
  3.  Long exponentiated‑gradient sub‑gradient refinement.
  4.  Simulated‑annealing hill‑climbing that can accept uphill moves.
  5.  If time permits the solution is interpolated to a finer grid (8192)
      and the whole pipeline is repeated.

All steps respect the global time budget (default 1000 s) and the routine
always returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT helpers -----------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2·len(a)-1) using FFT.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation needed for gradients.
    Returns a vector of length len(a):
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i = 0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
# Sub‑gradient optimisation (exponentiated‑gradient) -------------------
def _subgradient_opt2(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
    eta0: float = 0.7,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient descent on the exact peak.
    Returns the best vector seen and its objective value.
    """
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s

    best_a = a.copy()
    best_val = float("inf")
    stall = 0
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            stall = 0
            if it % 5_000 == 0:
                print(f"[Subgrad iter={it:07d}] new best {val:.6f}")
        else:
            stall += 1

        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        if stall > 10_000:
            a = np.maximum(a + 0.02 * (np.random.rand(n) - 0.5), 0.0)
            a /= a.sum()
            stall = 0

        it += 1
    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing local search --------------------------------------
def _simulated_annealing(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 150_000,
) -> tuple[np.ndarray, float]:
    """
    Cheap simulated‑annealing on the simplex.
    Randomly moves a fraction of mass from one index to another.
    Accepts uphill moves with a temperature that cools exponentially.
    Returns the best sequence and its exact objective.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    best_a = a.copy()

    # initialise temperature – a modest fraction of the current value
    T = best_val * 0.1
    move = 0
    while time.time() < deadline and move < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-12:
            move += 1
            continue
        j = np.random.randint(n)
        if i == j:
            move += 1
            continue

        # move up to 30 % of the source mass
        frac = np.random.rand() * 0.30
        delta = a[i] * frac
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
        else:
            # metropolis acceptance
            if T > 0.0 and np.random.rand() < np.exp(-(val2 - best_val) / T):
                a = a2

        # exponential cooling
        if move % 1_000 == 0 and move > 0:
            T *= 0.95
        move += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# L‑BFGS optimisation of the smooth surrogate ---------------------------
def _lbfgs_optimize(
    logits: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    lam: float,
    maxiter: int = 400,
):
    """
    One L‑BFGS‑B run on the smooth log‑sum‑exp surrogate for a given λ.
    Returns the updated logits.
    """
    import scipy.optimize as opt

    eps = 1e-15

    def fun_and_grad(logits_vec, lam):
        # soft‑max → simplex
        mx = logits_vec.max()
        exp_log = np.exp(logits_vec - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        # stable log‑sum‑exp
        exp_terms = np.exp(lam * (conv - max_c))
        sum_exp = exp_terms.sum()
        smooth_max = max_c + np.log(sum_exp) / lam

        w = exp_terms / sum_exp
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)
        return smooth_max, grad_log

    res = opt.minimize(
        fun_and_grad,
        logits,
        args=(lam,),
        method="L-BFGS-B",
        jac=True,
        options={"maxiter": maxiter, "disp": False},
    )
    return res.x


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Main entry point: returns a list of non‑negative heights (the step function)
    that minimises the evaluation function within the time budget.
    """
    import numpy as np
    import time

    # temporary import – SciPy may be unavailable
    try:
        import scipy.optimize as _  # noqa: F401
        SCIPY_AVAILABLE = True
    except Exception:
        SCIPY_AVAILABLE = False

    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # Base resolution (n = 4096)
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm start
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] != n:
            # linear interpolation to required length
            x_old = np.linspace(0, 1, warm.shape[0], endpoint=False)
            x_new = np.linspace(0, 1, n, endpoint=False)
            warm = np.interp(x_new, x_old, warm)
    else:
        # simple symmetric edge‑concentrated power‑law
        idx = np.arange(1, n + 1, dtype=np.float64)
        prod = idx * (n - idx + 1.0)
        shape = prod ** -0.7
        shape[np.isnan(shape)] = 0.0
        warm = shape / shape.sum()

    warm = np.maximum(warm, 0.0)
    if warm.sum() < 0.01:
        warm = np.random.dirichlet(np.ones(n))
    warm = warm / warm.sum()

    # ------------------------------------------------------------------
    # Parameterisation (logits → soft‑max)
    eps = 1e-15
    logits = np.log(warm + eps)

    # ------------------------------------------------------------------
    # Deterministic L‑BFGS‑B phase (smooth surrogate)
    if SCIPY_AVAILABLE:
        lambdas = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 50.0, 100.0]
        for lam in lambdas:
            if time.time() >= deadline:
                break
            logits = _lbfgs_optimize(
                logits,
                n,
                conv_fft_len,
                corr_fft_len,
                lam,
                maxiter=400,
            )
            # monitor exact objective
            a_tmp = np.exp(logits - logits.max())
            a_tmp = a_tmp / a_tmp.sum()
            val_tmp = 2.0 * n * _full_convolution(a_tmp, conv_fft_len).max()
            print(f"[L‑BFGS λ={lam:.1f}] objective={val_tmp:.6f}")

    # ------------------------------------------------------------------
    # Convert to probability vector
    a = np.exp(logits - logits.max())
    a = a / a.sum()

    # ------------------------------------------------------------------
    # Long sub‑gradient refinement (exact max)
    sub_deadline = deadline - 30.0  # leave room for SA / up‑sampling
    if time.time() < sub_deadline:
        a, val = _subgradient_opt2(
            a,
            n,
            conv_fft_len,
            corr_fft_len,
            sub_deadline,
            max_iters=1_000_000,
            eta0=0.7,
        )
        print(f"[Subgradient] best objective {val:.6f}")

    # ------------------------------------------------------------------
    # Simulated‑annealing local search
    if time.time() < deadline - 10.0:
        a, val = _simulated_annealing(
            a, n, conv_fft_len, deadline, max_moves=200_000
        )
        print(f"[SA] best objective {val:.6f}")

    # ------------------------------------------------------------------
    # Optional up‑sampling to 8192 points if time permits
    if time.time() + 40.0 < deadline:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        # linear interpolation to the finer grid
        x_old = np.linspace(0, 1, len(a), endpoint=False)
        x_new = np.linspace(0, 1, n_big, endpoint=False)
        a_big = np.interp(x_new, x_old, a)
        a_big = np.maximum(a_big, 0.0)
        a_big = a_big / a_big.sum()

        # L‑BFGS on the larger grid (fewer λ‑steps)
        if SCIPY_AVAILABLE:
            for lam in [1.0, 5.0, 15.0, 30.0]:
                if time.time() >= deadline:
                    break
                logits_big = np.log(a_big + eps)
                logits_big = _lbfgs_optimize(
                    logits_big,
                    n_big,
                    conv_fft_len_big,
                    corr_fft_len_big,
                    lam,
                    maxiter=300,
                )
                a_big = np.exp(logits_big - logits_big.max())
                a_big = a_big / a_big.sum()
                val_big = 2.0 * n_big * _full_convolution(a_big, conv_fft_len_big).max()
                print(f"[L‑BFGS‑big λ={lam:.1f}] objective={val_big:.6f}")

        # Sub‑gradient on the large grid
        sub_deadline_big = deadline - 5.0
        if time.time() < sub_deadline_big:
            a_big, val_big = _subgradient_opt2(
                a_big,
                n_big,
                conv_fft_len_big,
                corr_fft_len_big,
                sub_deadline_big,
                max_iters=600_000,
                eta0=0.6,
            )
            print(f"[Subgrad‑big] best objective {val_big:.6f}")

        # Final SA on the large grid
        if time.time() < deadline:
            a_big, val_big = _simulated_annealing(
                a_big, n_big, conv_fft_len_big, deadline, max_moves=150_000
            )
            print(f"[SA‑big] best objective {val_big:.6f}")

        best_seq = a_big
    else:
        best_seq = a

    # ------------------------------------------------------------------
    # Final sanity clamp (the evaluator caps at 1000 anyway)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Final log
    final_val = 2.0 * len(best_seq) * _full_convolution(
        best_seq, _next_pow2(2 * len(best_seq) - 1)
    ).max()
    print(f"Final objective {final_val:.6f} (length {len(best_seq)})")

    return best_seq.tolist()
```
