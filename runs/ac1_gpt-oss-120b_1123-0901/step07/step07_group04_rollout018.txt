"""
Propose a candidate sequence that (approximately) minimizes

        F(a) = 2·n·max(a∗a) / (sum(a)²)

using a multi‑start Adam optimiser on the simplex.
The algorithm works as follows:

1.  Choose a moderately large grid (default n = 8192, a power of two).
2.  Build a diverse pool of parametric seeds (uniform, Dirichlet,
    edge‑exponential, power‑law, double‑Gaussian, half‑wave sinusoid,
    optionally a warm‑start from the global ``height_sequence_1``).
3.  Evaluate every seed with a fast FFT‑based convolution and keep the
    best few (``top_k``).
4.  For each top seed run a constrained Adam optimiser that
    * represents the vector as a soft‑max of unrestricted parameters
      `p`  (so `a = softmax(p)` guarantees non‑negativity and sum‑to‑1),
    * computes the sub‑gradient of the true max‑convolution by
      taking a uniform weight over the active (maximal) shift(s),
    * updates `p` with Adam (per‑coordinate adaptive step sizes),
    * injects a small random jitter if no improvement is seen for a
      few seconds (helps to escape plateaus).
5.  If time permits, the best solution is linearly interpolated to a
    twice‑larger grid (n = 16384) and refined again for a short period.
6.  The best sequence found (clipped to the evaluator’s allowed range)
    is returned as a plain Python list.

The implementation respects the 1000 s budget, prints progress
information, and never uses closures or file/network I/O.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (all top‑level – no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n-1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


def _objective_fast(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Fast evaluation of the objective (identical to evaluate_sequence)."""
    conv = _full_convolution(a, conv_fft_len)
    return float(2.0 * n * conv.max() / (a.sum() ** 2))


def _generate_parametric_candidates(
    n: int,
    warm_start: np.ndarray | None,
) -> list[np.ndarray]:
    """
    Produce a diverse set of candidate vectors.
    All candidates are normalised to have sum≈1.
    """
    candidates = []

    # warm‑start and simple rotations (if provided)
    if warm_start is not None:
        candidates.append(warm_start.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            candidates.append(np.roll(warm_start, shift))

    # uniform baseline
    candidates.append(np.full(n, 1.0 / n))

    # random Dirichlet draws with several concentration parameters
    for conc in (0.01, 0.05, 0.1, 0.3, 0.5, 1.0, 2.0):
        cand = np.random.dirichlet(np.full(n, conc))
        candidates.append(cand)

    i = np.arange(n, dtype=np.float64)

    # edge‑exponential mixtures
    for alpha in np.logspace(-2, 1.0, num=12):
        edge = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
        edge /= edge.sum()
        for w in (0.0, 0.4, 0.8, 1.0):
            shape = w * edge + (1.0 - w) / n
            candidates.append(shape)

    # edge‑power‑law mixtures
    for p in np.linspace(0.2, 5.0, num=12):
        edge = (i + 1.0) ** (-p) + (n - i) ** (-p)
        edge /= edge.sum()
        for w in (0.0, 0.33, 0.66, 1.0):
            shape = w * edge + (1.0 - w) / n
            candidates.append(shape)

    # double‑Gaussian edges
    for gamma in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        shape /= shape.sum()
        candidates.append(shape)

    # positive half‑wave sinusoid
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    candidates.append(shape)

    return candidates


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (via soft‑max parametrisation)
# ----------------------------------------------------------------------


def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray | None, float]:
    """
    Adam optimisation of the max‑convolution objective.
    `a0` must be a non‑negative vector (not necessarily normalised).
    The routine returns the best vector/value seen before the deadline.
    """
    eps = 1e-12
    # initialise unrestricted parameters (log‑space). Adding eps avoids log(0).
    p = np.log(a0 + eps)

    # Adam hyper‑parameters (standard settings, tuned for this problem)
    lr = 0.05
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    m = np.zeros_like(p)
    v = np.zeros_like(p)

    it = 0
    last_improve = time.time()
    # Main loop
    while time.time() < deadline:
        # --- obtain a on the simplex (soft‑max) -------------------------
        max_p = p.max()
        a = np.exp(p - max_p)
        a_sum = a.sum()
        a = a / a_sum  # now sum(a) == 1

        # --- compute convolution & current objective --------------------
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        cur_val = 2.0 * n * max_c / (a_sum ** 2)  # a_sum == 1 but keep formula

        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[n={n} Adam it={it:07d}] new best {cur_val:.6f}")

        # --- build sub‑gradient of the max‑convolution -----------------
        # Use the active set (indices where the convolution attains its max)
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if not np.any(active):
            # Should never happen but guard against numerical issues
            it += 1
            continue
        w = np.zeros_like(conv)
        w[active] = 1.0 / active.sum()          # uniform weight over active shifts

        # gradient w.r.t. a (2 * weighted cross‑correlation)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain‑rule: gradient w.r.t. unrestricted parameters `p`
        mean_grad = np.dot(a, grad_a)           # ⟨a, grad_a⟩
        grad_p = a * (grad_a - mean_grad)       # same as ∂F/∂p for soft‑max

        # --- Adam parameter update ------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)

        # bias‑corrected estimates
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + epsilon)
        p = p - step

        # --- occasional random jitter to escape plateaus ---------------
        if time.time() - last_improve > 5.0:
            jitter = (np.random.rand(n) - 0.5) * 0.02   # log‑space jitter
            p = p + jitter
            last_improve = time.time()

        it += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation environment
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a low‑evaluation‑score non‑negative sequence.
    Returns the best sequence found (as a plain Python list).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Grid size – 8192 provides a finer search space while staying fast.
    n = 8192
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm start (global variable `height_sequence_1` if present)
    warm_start = globals().get("height_sequence_1")
    if warm_start is not None:
        warm_start = np.asarray(warm_start, dtype=np.float64)
        # linearly interpolate/extrapolate to the required length
        old_len = warm_start.shape[0]
        if old_len != n:
            x_old = np.linspace(0.0, 1.0, old_len, endpoint=False)
            x_new = np.linspace(0.0, 1.0, n, endpoint=False)
            warm_start = np.interp(x_new, x_old, warm_start)
        warm_start = np.maximum(warm_start, 0.0)
        if warm_start.sum() >= 0.01:
            warm_start = warm_start / warm_start.sum()
        else:
            warm_start = None
    else:
        warm_start = None

    # ------------------------------------------------------------------
    # Stage 1: generate a broad collection of seeds and keep the best few.
    candidates = _generate_parametric_candidates(n, warm_start)

    # add a handful of extra Dirichlet draws (different concentrations)
    for conc in (0.005, 0.02, 0.07, 0.15, 0.4, 0.8):
        for _ in range(12):
            cand = np.random.dirichlet(np.full(n, conc))
            candidates.append(cand)

    # fast evaluation of all candidates (FFT based)
    cand_vals = []
    for idx, cand in enumerate(candidates):
        if time.time() >= deadline:
            break
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 0.01:
            cand = np.random.dirichlet(np.ones(n))
            s = cand.sum()
        cand = cand / s
        val = _objective_fast(cand, n, conv_fft_len)
        cand_vals.append((val, cand))

    if not cand_vals:
        # fallback – uniform vector
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        print("*** No candidates evaluated – returning uniform vector.")
        return best_seq.tolist()

    # keep the top‑k seeds (they will be fed to the expensive optimiser)
    cand_vals.sort(key=lambda x: x[0])
    top_k = 6
    top_candidates = [c for _, c in cand_vals[:top_k]]

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Stage 2: deep optimisation of the selected starts.
    remaining = len(top_candidates)
    for start_id, a0 in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        # allocate a fair share of the remaining time to this start
        time_now = time.time()
        time_left = deadline - time_now
        if remaining <= 0:
            break
        # give each remaining start roughly an equal share (but at least 20 s)
        allocate = max(20.0, time_left / remaining)
        sub_deadline = time_now + allocate

        print(f"--- Optimising start #{start_id + 1}/{len(top_candidates)} "
              f"(allocated {allocate:.1f}s)")

        best_seq, best_val = _adam_optimize(
            a0,
            n,
            conv_fft_len,
            corr_fft_len,
            sub_deadline,
            best_seq,
            best_val,
        )
        remaining -= 1

    # ------------------------------------------------------------------
    # Stage 3: optional refinement on a doubled grid if time permits.
    time_left = deadline - time.time()
    if time_left > 30.0 and best_seq is not None:
        n_big = 16384
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        # linear interpolation of the current best onto the finer grid
        x_old = np.linspace(0.0, 1.0, n, endpoint=False)
        x_big = np.linspace(0.0, 1.0, n_big, endpoint=False)
        a_big = np.interp(x_big, x_old, best_seq)
        a_big = np.maximum(a_big, 0.0)
        if a_big.sum() < 0.01:
            a_big = np.random.dirichlet(np.ones(n_big))
        a_big = a_big / a_big.sum()

        print("--- Refinement on larger grid (n=16384)")
        best_seq, best_val = _adam_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            best_seq,
            best_val,
        )
        if best_seq is not None and best_seq.shape[0] == n_big:
            n = n_big

    # ------------------------------------------------------------------
    # Fallback – uniform vector if nothing succeeded (should never happen)
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the evaluator’s allowed range (non‑negative, ≤1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"*** Finished search: best objective ≈ {best_val:.6f} (n={best_seq.shape[0]})")
    return best_seq.tolist()
