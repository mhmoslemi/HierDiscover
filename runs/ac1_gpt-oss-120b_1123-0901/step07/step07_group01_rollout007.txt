"""
Hybrid multi‑scale optimizer for the step‑function objective

    F(a) = 2·n·max(a∗a) / (sum(a)^2) ,

where a is a non‑negative vector (the “heights”).  The routine works in
three phases:

1.  A moderate‑size grid (n≈4096) is explored with a diverse pool of
    warm‑starts (uniform, Dirichlet, power‑law, arcsine, classic
    window‑functions, sparse spike patterns).  Each start is refined by
    – Adam on a smooth‑max (log‑sum‑exp) surrogate with a very slow
      temperature anneal, – an exact‑max sub‑gradient descent using
      exponentiated‑gradient updates (larger initial step‑size).

2.  The best solution is interpolated to a larger grid (n≈8192) and the
    same two‑stage refinement is performed again (shorter budgets).

3.  A long simulated‑annealing phase works directly on the large grid,
    updating the convolution in O(n) per elementary mass‑transfer move.

All heavy work uses FFT‑based convolutions.  The algorithm respects the
global time budget (default 1000 s) and always returns a valid
non‑negative list.  The final value is printed via the provided
`evaluate_sequence` routine.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via FFT (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for the gradient."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Fast evaluation of 2·n·max(a∗a)/(sum a)^2 (no clipping)."""
    conv = _full_convolution(a, conv_fft_len)
    return 2.0 * n * conv.max() / (a.sum() ** 2)

def _initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Create a diverse pool of start vectors for a given length n."""
    cand = []

    # Warm‑start (if any) and a few rotations.
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    if warm is not None:
        cand.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    # Dirichlet families with several concentration parameters.
    for alpha in (0.05, 0.2, 0.5, 1.0, 5.0):
        cand.append(np.random.dirichlet(alpha * np.ones(n)))

    # Power‑law edge‑concentrated families (exponent = α, α∈[‑1.0,‑0.2]).
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in (-1.0, -0.75, -0.5, -0.4, -0.3, -0.2):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    # Classical window functions (Hann, Hamming, Blackman, Kaiser‑β=6).
    x = np.arange(n, dtype=np.float64) / (n - 1)
    hann = 0.5 * (1 - np.cos(2 * np.pi * x))
    hann /= hann.sum()
    cand.append(hann)

    hamming = 0.54 - 0.46 * np.cos(2 * np.pi * x)
    hamming /= hamming.sum()
    cand.append(hamming)

    blackman = (
        0.42
        - 0.5 * np.cos(2 * np.pi * x)
        + 0.08 * np.cos(4 * np.pi * x)
    )
    blackman /= blackman.sum()
    cand.append(blackman)

    # Kaiser window (beta=6) – approximated manually.
    beta = 6.0
    from math import i0  # Modified Bessel function I0.
    kaiser = i0(beta * np.sqrt(1 - (2 * x - 1) ** 2)) / i0(beta)
    kaiser /= kaiser.sum()
    cand.append(kaiser)

    # Half‑sine and quadratic “tent” shapes.
    half_sine = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    half_sine = np.maximum(half_sine, 0.0)
    half_sine /= half_sine.sum()
    cand.append(half_sine)

    tent = np.minimum(np.arange(n) + 1, n - np.arange(n))
    tent = tent.astype(float)
    tent /= tent.sum()
    cand.append(tent)

    # Sparse “spike” families (k equally spaced non‑zero entries).
    for k in (32, 64, 128, 256):
        a = np.zeros(n, dtype=np.float64)
        pos = np.round(np.linspace(0, n - 1, k)).astype(int)
        a[pos] = 1.0 / k
        cand.append(a)

    # Uniform.
    cand.append(np.full(n, 1.0 / n))

    return cand

def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """Adam on the log‑sum‑exp surrogate of max(a∗a)."""
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.04                      # a tad larger than the baseline
    lam = 0.1                      # start very smooth
    lam_factor = 1.02              # very slow anneal → smooth → max

    best_a = a0.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        # Soft‑max that gives the probability vector a.
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:05d}] new best {val:.6f}")

        # Smooth‑max weights (log‑sum‑exp).
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # Gradient of the surrogate.
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # Center the gradient (to stay on the simplex after exponentiated‑gradient).
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits.
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # Temperature schedule.
        lam = min(lam * lam_factor, 1e6)

        # Small jitter at the very end to escape tiny plateaus.
        if it % 5000 == 0 and time.time() > deadline - 10:
            logits += (np.random.rand(n) - 0.5) * 1e-3

        it += 1

    return best_a, best_val


def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 1.0,
) -> tuple[np.ndarray, float]:
    """Exact‑max sub‑gradient descent with exponentiated‑gradient updates."""
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s

    best_a = a.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 2000 == 0:
                print(f"[Subgrad it={it:05d}] new best {val:.6f}")

        # Active set of maximal convolution entries (a tiny tolerance).
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # Diminishing step‑size (larger initial step than baseline).
        eta = eta0 / np.sqrt(it + 1)

        # Multiplicative (exponentiated‑gradient) update.
        a = a * np.exp(-eta * grad_a)

        # Re‑normalise onto the simplex.
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
def _simulated_annealing(
    a0: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    cur_best_val: float,
) -> tuple[np.ndarray, float]:
    """Fast SA on the exact max (O(n) per elementary move)."""
    a = a0.copy()
    conv = _full_convolution(a, fft_len)
    cur_max = conv.max()
    cur_val = 2.0 * n * cur_max
    best_a = a.copy()
    best_val = cur_val

    start = time.time()
    it = 0
    # Temperature schedule (starts relatively hot and decays exponentially).
    T0 = 0.3
    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / (deadline - start + 1e-12)
        # Exponential cooling.
        T = max(1e-6, T0 * np.exp(-5.0 * frac))
        # Move magnitude (relative to source mass).
        move_scale = max(0.02, 0.4 * (1.0 - frac))

        # Choose a source index proportional to its current mass.
        if a.sum() == 0:
            break
        i = np.random.choice(n, p=a / a.sum())
        if a[i] < 1e-14:
            continue
        # Choose a distinct target uniformly.
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # Transfer a random fraction of the source mass.
        delta = a[i] * (np.random.rand() * move_scale)

        a_new = a.copy()
        conv_new = conv.copy()
        a_new[i] -= delta
        a_new[j] += delta

        # Update convolution using the *old* a (the one before the move).
        # The linear update of self‑convolution for a transfer from i→j.
        conv_new[i : i + n] -= 2.0 * delta * a
        conv_new[j : j + n] += 2.0 * delta * a
        conv_new[2 * i] += delta * delta
        conv_new[2 * j] += delta * delta
        conv_new[i + j] -= 2.0 * delta * delta

        new_max = conv_new.max()
        new_val = 2.0 * n * new_max

        # Metropolis acceptance.
        if new_val < cur_val or np.random.rand() < np.exp(-(new_val - cur_val) / T):
            a, conv = a_new, conv_new
            cur_val = new_val
            cur_max = new_max
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[SA it={it:06d}] new best {best_val:.6f}")

        it += 1
        if it % 5000 == 0 and time.time() < deadline:
            print(f"[SA it={it:06d}] best {best_val:.6f}")

    return best_a, best_val


def _interpolate_and_refine(
    a_small: np.ndarray,
    n_big: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """Upsample a solution to a larger grid and perform a short Adam+sub‑grad pass."""
    n_small = a_small.shape[0]
    x_small = np.linspace(0.0, 1.0, n_small, endpoint=False)
    x_big = np.linspace(0.0, 1.0, n_big, endpoint=False)
    a_big = np.interp(x_big, x_small, a_small)
    a_big = np.maximum(a_big, 0.0)
    a_big /= a_big.sum()

    conv_fft_len_big = _next_pow2(2 * n_big - 1)
    corr_fft_len_big = _next_pow2(3 * n_big - 2)

    # Short Adam (few 10 k iters) – enough to get a decent direction.
    a_adam, _ = _adam_optimize(
        a_big,
        n_big,
        conv_fft_len_big,
        corr_fft_len_big,
        deadline,
        max_iters=60_000,
    )
    # Follow‑up exact‑max sub‑gradient descent.
    a_ref, val_ref = _subgradient_optimize(
        a_adam,
        n_big,
        conv_fft_len_big,
        corr_fft_len_big,
        deadline,
        max_iters=80_000,
        eta0=1.0,
    )
    return a_ref, val_ref

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a step‑function that (approximately) minimises
        2·n·max(a∗a) / (sum a)^2
    within the allotted time budget.  The routine runs a three‑stage
    optimisation (medium grid → up‑sample → long simulated annealing) and
    always returns a valid non‑negative list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # Phase‑1 : medium resolution optimisation (n≈4096)
    n_mid = 4096
    conv_fft_len_mid = _next_pow2(2 * n_mid - 1)
    corr_fft_len_mid = _next_pow2(3 * n_mid - 2)

    warm = globals().get("height_sequence_1")
    candidates = _initial_candidates(n_mid, warm)

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # Limit the number of candidates to keep the runtime reasonable.
    max_cands = 12
    for cid, cand in enumerate(candidates[:max_cands]):
        now = time.time()
        if now >= deadline:
            break

        a0 = np.maximum(cand, 0.0)
        if a0.sum() < 0.01:
            a0 = np.random.dirichlet(np.ones(n_mid))
        else:
            a0 = a0 / a0.sum()

        # Allocate a tiny per‑candidate budget (≈ 30 s) but respect the global deadline.
        local_deadline = min(deadline - 5.0, now + 30.0)

        # Adam (smooth‑max) refinement.
        a_adam, val_adam = _adam_optimize(
            a0,
            n_mid,
            conv_fft_len_mid,
            corr_fft_len_mid,
            local_deadline,
            max_iters=120_000,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[cand {cid}] Adam improved best to {val_adam:.6f}")

        # Exact‑max sub‑gradient descent.
        if time.time() < deadline:
            a_sub, val_sub = _subgradient_optimize(
                best_seq,
                n_mid,
                conv_fft_len_mid,
                corr_fft_len_mid,
                deadline,
                max_iters=80_000,
                eta0=1.0,
            )
            if val_sub < best_val:
                best_val = val_sub
                best_seq = a_sub.copy()
                print(f"[cand {cid}] Sub‑grad improved best to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Phase‑2 : upscale to a larger grid (n≈8192) if time permits.
    n_big = 8192
    if best_seq is not None and time.time() + 20.0 < deadline:
        a_big, val_big = _interpolate_and_refine(best_seq, n_big, deadline)
        if val_big < best_val:
            best_val = val_big
            best_seq = a_big.copy()
            print(f"[Big] improved best to {val_big:.6f}")

    # ------------------------------------------------------------------
    # Phase‑3 : final long simulated‑annealing on the largest grid.
    if best_seq is not None and time.time() + 5.0 < deadline:
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        a_sa, val_sa = _simulated_annealing(
            best_seq,
            n_big,
            conv_fft_len_big,
            deadline,
            best_val,
        )
        if val_sa < best_val:
            best_val = val_sa
            best_seq = a_sa.copy()
            print(f"[SA] final best improved to {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Fallback – guarantee a return value if everything failed.
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Clip to the limits enforced by evaluate_sequence and produce a list.
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Report the final value using the judge's routine (if available).
    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"Final reported value {final_val:.6f} (length {len(best_seq)})")
    except Exception:
        print(f"Final internal value {best_val:.6f} (length {len(best_seq)})")

    return best_seq.tolist()
