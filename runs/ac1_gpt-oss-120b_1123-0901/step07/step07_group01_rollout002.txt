"""
Hybrid search for a non‑negative step function that minimizes

    F(a) = 2·n·max(a∗a) / (sum(a))²

Improvements over the reference implementation:
* A richer, stochastic pool of start vectors (random sparse masks,
  random Fourier‑phase vectors, Dirichlet with random concentration,
  Gaussian bumps, etc.).
* A simplified time‑budget manager that gives each candidate a small
  dedicated budget, then spends the remaining time on a global
  simulated‑annealing (SA) refinement.
* The same fast FFT‑based convolution utilities and the Adam /
  sub‑gradient optimisers from the reference code.
* The best sequence is stored in the global variable ``height_sequence_1``
  for warm‑starting the next run.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a by FFT (length 2·len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a)."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]

def _adam_optimize(a0: np.ndarray,
                   n: int,
                   conv_fft_len: int,
                   corr_fft_len: int,
                   deadline: float,
                   max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """Adam on the log‑sum‑exp (smooth‑max) surrogate."""
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.03
    lam = 0.3           # start temperature
    lam_factor = 1.015  # anneal speed

    best_a = a0.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        # softmax re‑parameterisation
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:05d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        lam = min(lam * lam_factor, 1e6)

        # occasional jitter near the end to escape plateaus
        if it % 4000 == 0 and time.time() > deadline - 5:
            logits += (np.random.rand(n) - 0.5) * 1e-3

        it += 1

    return best_a, best_val


def _subgradient_optimize(a0: np.ndarray,
                          n: int,
                          conv_fft_len: int,
                          corr_fft_len: int,
                          deadline: float,
                          max_iters: int = 300_000,
                          eta0: float = 0.6) -> tuple[np.ndarray, float]:
    """Exact‑max sub‑gradient descent with exponentiated‑gradient updates."""
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s

    best_a = a.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 2000 == 0:
                print(f"[Subgrad it={it:05d}] new best {val:.6f}")

        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # diminishing step‑size
        eta = eta0 / np.sqrt(it + 1)

        # multiplicative (exponentiated‑gradient) update
        a = a * np.exp(-eta * grad_a)

        # renormalise onto the simplex
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val

def _simulated_annealing(a0: np.ndarray,
                         n: int,
                         fft_len: int,
                         deadline: float,
                         cur_best_val: float) -> tuple[np.ndarray, float]:
    """Fast SA that updates the convolution in O(n) per elementary move."""
    a = a0.copy()
    conv = _full_convolution(a, fft_len)
    cur_val = 2.0 * n * conv.max()
    best_a = a.copy()
    best_val = cur_val

    start = time.time()
    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        fraction = elapsed / (deadline - start + 1e-12)

        # linear cooling schedule
        T = max(1e-6, 0.1 * (1.0 - fraction))
        move_scale = max(0.01, 0.3 * (1.0 - fraction))

        # source index proportional to its mass
        if a.sum() == 0:
            break
        i = np.random.choice(n, p=a / a.sum())
        if a[i] < 1e-12:
            continue
        # target index uniformly (different from i)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * (np.random.rand() * move_scale)

        a_new = a.copy()
        conv_new = conv.copy()
        a_new[i] -= delta
        a_new[j] += delta

        # update convolution incremental (see reference code)
        conv_new[i : i + n] -= 2.0 * delta * a
        conv_new[j : j + n] += 2.0 * delta * a
        conv_new[2 * i] += delta * delta
        conv_new[2 * j] += delta * delta
        conv_new[i + j] -= 2.0 * delta * delta

        new_val = 2.0 * n * conv_new.max()

        # Metropolis criterion
        if new_val < cur_val or np.random.rand() < np.exp(-(new_val - cur_val) / T):
            a, conv = a_new, conv_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[SA it={it:05d}] new best {best_val:.6f}")

        it += 1
        if it % 5000 == 0:
            print(f"[SA it={it:05d}] best {best_val:.6f}")

    return best_a, best_val


def _interpolate_and_refine(a_small: np.ndarray,
                            n_big: int,
                            deadline: float) -> tuple[np.ndarray, float]:
    """Upscale a solution to a larger grid and run a short Adam+sub‑gradient pass."""
    n_small = a_small.shape[0]
    x_small = np.linspace(0.0, 1.0, n_small, endpoint=False)
    x_big = np.linspace(0.0, 1.0, n_big, endpoint=False)
    a_big = np.interp(x_big, x_small, a_small)
    a_big = np.maximum(a_big, 0.0)
    a_big /= a_big.sum()

    conv_fft_len_big = _next_pow2(2 * n_big - 1)
    corr_fft_len_big = _next_pow2(3 * n_big - 2)

    a_adam, _ = _adam_optimize(a_big, n_big,
                               conv_fft_len_big,
                               corr_fft_len_big,
                               deadline,
                               max_iters=80_000)
    a_ref, val_ref = _subgradient_optimize(a_adam, n_big,
                                           conv_fft_len_big,
                                           corr_fft_len_big,
                                           deadline,
                                           max_iters=100_000)
    return a_ref, val_ref

def _random_sparse_candidate(n: int, k: int) -> np.ndarray:

def _random_phase_candidate(n: int) -> np.ndarray:
    """Take absolute‑square of an IFFT of a unit‑modulus random spectrum."""
    fft_len = _next_pow2(2 * n - 1)
    phi = np.random.rand(fft_len) * 2.0 * np.pi
    spectrum = np.exp(1j * phi)
    v = np.fft.ifft(spectrum)[:n]
    a = np.abs(v) ** 2
    if a.sum() == 0:
        a = np.random.rand(n)
    a /= a.sum()
    return a


def _random_gaussian_blob(n: int) -> np.ndarray:
    """Gaussian bump with random centre and width."""
    x = np.arange(n, dtype=np.float64)
    centre = np.random.uniform(0, n)
    sigma = np.random.uniform(0.03, 0.15) * n
    shape = np.exp(-0.5 * ((x - centre) ** 2) / (sigma ** 2))
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _random_dirichlet_candidate(n: int) -> np.ndarray:
    """Dirichlet draw with a random concentration vector."""
    alpha = np.random.rand(n) * 2.0 + 0.1   # each α in [0.1, 2.1]
    a = np.random.dirichlet(alpha)
    return a


def _initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Build a diverse pool of warm‑starts."""
    cand = []

    # warm start and rotations
    if warm is not None:
        cand.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    # classic families (kept from reference)
    cand.append(np.random.dirichlet(np.ones(n)))                # uniform Dirichlet
    cand.append(np.random.dirichlet(0.2 * np.ones(n)))          # spiky Dirichlet

    # power‑law edge shapes
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in (-2.0, -1.5, -1.0, -0.75):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    # arcsine‑type
    arcs = 1.0 / np.sqrt(np.arange(1, n + 1) * np.arange(n, 0, -1))
    arcs /= arcs.sum()
    cand.append(arcs)

    # two‑Gaussian edge shapes
    i = np.arange(n, dtype=np.float64)
    for gamma in (0.3, 0.7, 1.5):
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        shape /= shape.sum()
        cand.append(shape)

    # half‑sine
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand.append(shape)

    # uniform
    cand.append(np.full(n, 1.0 / n))

    # spike families (equally spaced)
    for k in (64, 128, 256, 512, 1024):
        a = np.zeros(n, dtype=np.float64)
        pos = np.round(np.linspace(0, n - 1, k)).astype(int)
        a[pos] = 1.0 / k
        cand.append(a)

    # --- New stochastic families ------------------------------------
    # random Dirichlet with random concentration
    for _ in range(5):
        cand.append(_random_dirichlet_candidate(n))

    # random sparse masks with various sparsities
    for k in (32, 64, 128, 256, 512, 1024):
        for _ in range(2):
            cand.append(_random_sparse_candidate(n, k))

    # random Fourier‑phase candidates
    for _ in range(5):
        cand.append(_random_phase_candidate(n))

    # random Gaussian blobs
    for _ in range(5):
        cand.append(_random_gaussian_blob(n))

    return cand


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises
    `2·n·max(a∗a)/(sum a)^2`.  The routine mixes diverse warm‑starts,
    short Adam+sub‑gradient passes, a global simulated‑annealing
    refinement and (if time permits) an upscale‑and‑refine stage at
    double resolution.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # obtain optional warm‑start from previous run
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    candidates = _initial_candidates(n, warm)

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Small dedicated budget per candidate (≈ 25 s or until time runs out)
    for cid, cand in enumerate(candidates):
        if time.time() >= deadline:
            break
        a0 = np.maximum(cand, 0.0)
        if a0.sum() < 0.01:
            a0 = np.random.dirichlet(np.ones(n))
        else:
            a0 = a0 / a0.sum()

        # give each candidate a little time slice
        local_deadline = min(deadline - 5.0, time.time() + 25.0)

        # Adam on smooth‑max surrogate
        a_adam, val_adam = _adam_optimize(a0, n, conv_fft_len,
                                          corr_fft_len, local_deadline,
                                          max_iters=150_000)

        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[cand {cid}] Adam improved best to {val_adam:.6f}")

        # Sub‑gradient refinement (exact max)
        if time.time() < deadline:
            a_sub, val_sub = _subgradient_optimize(best_seq, n,
                                                   conv_fft_len,
                                                   corr_fft_len,
                                                   deadline,
                                                   max_iters=80_000,
                                                   eta0=0.8)
            if val_sub < best_val:
                best_val = val_sub
                best_seq = a_sub.copy()
                print(f"[cand {cid}] Sub‑grad improved best to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Global simulated‑annealing refinement on the overall best candidate
    if best_seq is not None and time.time() < deadline - 5:
        a_sa, val_sa = _simulated_annealing(best_seq, n,
                                            conv_fft_len,
                                            deadline,
                                            best_val)
        if val_sa < best_val:
            best_val = val_sa
            best_seq = a_sa.copy()
            print(f"[SA] best improved to {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Optional upscale to 8192 points and a short refinement
    if time.time() + 20 < deadline:
        n_big = 8192
        a_big, val_big = _interpolate_and_refine(
            best_seq if best_seq is not None else np.random.dirichlet(np.ones(n_big)),
            n_big,
            deadline)
        if val_big < best_val:
            best_val = val_big
            best_seq = a_big.copy()
            print(f"[Big] final best {val_big:.6f}")

    # ------------------------------------------------------------------
    # Fallback – guarantee a valid return
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Clip to the limits enforced by evaluate_sequence
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Report the final objective (using the judge's routine if available)
    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"Final reported value {final_val:.6f} (length {len(best_seq)})")
    except Exception:
        print(f"Final internal value {best_val:.6f} (length {len(best_seq)})")

    # store for warm‑starting the next invocation
    globals()["height_sequence_1"] = best_seq.tolist()

    return best_seq.tolist()
