"""
Hybrid search for a low‑value step‐function.
The routine builds a rich pool of parametric candidates (Gaussian,
Beta‑type, power‑law, cosine, double‑edge), evaluates them quickly via
FFT, keeps the best, and then refines it with a short Adam run,
exact‑max sub‑gradient descent and a tiny simulated‑annealing sweep.
If time permits the solution is up‑sampled to a longer grid (8192)
and refined again.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for the gradient."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """
    Exact evaluation of 2·n·max(a∗a) / (sum a)².
    The vector a is assumed non‑negative.
    """
    s = a.sum()
    if s < 0.01:
        return float("inf")
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    return 2.0 * n * max_c / (s * s)


# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   conv_fft_len: int,
                   corr_fft_len: int,
                   deadline: float,
                   max_iters: int = 120_000) -> tuple[np.ndarray, float]:
    """
    Adam on a smooth‑max surrogate (log‑sum‑exp) of the convolution peak.
    Returns the best vector found (normalised to sum = 1) and its raw
    value 2·n·max(conv) (sum is 1, so this equals the objective).
    """
    eps = 1e-15
    logits = np.log(a0 + eps)                     # soft‑max parameters
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.03
    lam = 0.3               # initial temperature for the smooth max
    lam_factor = 1.015

    best_a = a0.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        # construct a from logits
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:05d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # move to log‑space (mirror‑descent on the simplex)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # anneal smooth‑max temperature
        lam = min(lam * lam_factor, 1e6)

        # occasional jitter to avoid plateaus
        if it % 4000 == 0 and time.time() > deadline - 5:
            logits += (np.random.rand(n) - 0.5) * 1e-3

        it += 1

    return best_a, best_val


def _subgradient_optimize(a0: np.ndarray,
                          n: int,
                          conv_fft_len: int,
                          corr_fft_len: int,
                          deadline: float,
                          max_iters: int = 120_000,
                          eta0: float = 0.6) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient descent using exponentiated‑gradient updates.
    The vector is kept on the simplex (sum = 1, non‑negative).
    Returns the best vector found and its raw 2·n·max(conv) value.
    """
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s

    best_a = a.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 2000 == 0:
                print(f"[Subgrad it={it:05d}] new best {val:.6f}")

        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # diminishing step size
        eta = eta0 / np.sqrt(it + 1)

        # exponentiated‑gradient update (multiplicative)
        a = a * np.exp(-eta * grad_a)

        # re‑normalise onto the simplex
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _simulated_annealing(a0: np.ndarray,
                         n: int,
                         fft_len: int,
                         deadline: float,
                         cur_best_val: float) -> tuple[np.ndarray, float]:
    """
    Fast O(n) per move SA that directly updates the convolution.
    Returns the best vector found and its raw 2·n·max(conv) value.
    """
    a = a0.copy()
    conv = _full_convolution(a, fft_len)
    cur_max = conv.max()
    cur_val = 2.0 * n * cur_max
    best_a = a.copy()
    best_val = cur_val

    start = time.time()
    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        fraction = elapsed / (deadline - start + 1e-12)

        # linear cooling schedule
        T = max(1e-6, 0.1 * (1.0 - fraction))
        move_scale = max(0.01, 0.3 * (1.0 - fraction))

        # choose a source index weighted by its current mass
        total = a.sum()
        if total == 0:
            break
        i = np.random.choice(n, p=a / total)
        if a[i] < 1e-12:
            continue

        # choose a different target uniformly
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * (np.random.rand() * move_scale)

        # propose move
        a_new = a.copy()
        conv_new = conv.copy()
        a_new[i] -= delta
        a_new[j] += delta

        # update convolution using the *old* a (linear update rule)
        conv_new[i:i + n] -= 2.0 * delta * a
        conv_new[j:j + n] += 2.0 * delta * a
        conv_new[2 * i] += delta * delta
        conv_new[2 * j] += delta * delta
        conv_new[i + j] -= 2.0 * delta * delta

        new_max = conv_new.max()
        new_val = 2.0 * n * new_max

        # Metropolis acceptance
        if new_val < cur_val or np.random.rand() < np.exp(-(new_val - cur_val) / T):
            a, conv = a_new, conv_new
            cur_val = new_val
            cur_max = new_max
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[SA it={it:05d}] new best {best_val:.6f}")

        it += 1
        if it % 5000 == 0:
            print(f"[SA it={it:05d}] best {best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function minimising
    `2·n·max(a∗a)/(sum a)^2`.  The method:
      * builds many parametric candidates (Gaussian, Beta, power‑law,
        cosine, double‑edge, mixtures);
      * keeps the best one;
      * refines it with Adam → sub‑gradient → a light SA;
      * finally interpolates to 8192 points if time permits.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    # --------------------------------------------------------------
    # problem size
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # optional warm‑start from previous run
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # --------------------------------------------------------------
    # generate a rich pool of candidates
    candidates = []

    if warm is not None:
        candidates.append(warm.copy())
        # some rotations of the warm start
        for shift in (n // 4, n // 2, 3 * n // 4):
            candidates.append(np.roll(warm, shift))

    # uniform baseline
    candidates.append(np.full(n, 1.0 / n))

    # Dirichlet random draws (light exploration)
    for _ in range(12):
        candidates.append(np.random.dirichlet(np.ones(n)))

    # --- parametric families ---------------------------------------
    idx = np.arange(n, dtype=np.float64)

    # 1. centred Gaussian
    mid = (n - 1) / 2.0
    for sigma in np.linspace(0.10 * n, 0.55 * n, 23):
        g = np.exp(-((idx - mid) ** 2) / (2.0 * sigma ** 2))
        s = g.sum()
        if s > 1e-12:
            candidates.append(g / s)

    # 2. symmetric Beta (α‑parameter)
    xs = (idx + 0.5) / n
    for alpha in np.linspace(0.20, 5.00, 24):
        b = (xs ** (alpha - 1.0)) * ((1.0 - xs) ** (alpha - 1.0))
        s = b.sum()
        if s > 1e-12:
            candidates.append(b / s)

    # 3. power‑law edge‑concentrated shapes (p‑parameter)
    prod = (idx + 1.0) * (n - idx)
    for p in np.linspace(-2.0, -0.2, 20):
        pl = np.where(prod > 0, prod ** p, 0.0)
        s = pl.sum()
        if s > 1e-12:
            candidates.append(pl / s)

    # 4. cosine‑powered (γ‑parameter)
    base = np.sin(np.pi * (idx + 0.5) / n)
    base = np.maximum(base, 0.0)
    for gamma in np.linspace(0.5, 4.0, 16):
        c = base ** gamma
        s = c.sum()
        if s > 1e-12:
            candidates.append(c / s)

    # 5. double‑Gaussian edges (σ‑parameter)
    for sigma in np.linspace(0.02 * n, 0.20 * n, 14):
        eg = np.exp(- (idx ** 2) / (2.0 * sigma ** 2))
        eg += np.exp(- ((n - 1 - idx) ** 2) / (2.0 * sigma ** 2))
        s = eg.sum()
        if s > 1e-12:
            candidates.append(eg / s)

    # 6. mixtures of uniform & arcsine
    arcs = 1.0 / np.sqrt((idx + 1) * (n - idx))
    arcs = arcs / arcs.sum()
    for lam in np.linspace(0.0, 1.0, 11):
        mix = lam * arcs + (1.0 - lam) * (1.0 / n)
        candidates.append(mix / mix.sum())

    # --------------------------------------------------------------
    # evaluate the pool (quick, FFT based) and keep the best
    best_seq = None
    best_val = float("inf")
    for cid, cand in enumerate(candidates):
        if time.time() >= deadline:
            break
        val = _objective(cand, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_seq = cand.copy()
            print(f"[pool {cid:04d}] new best {val:.6f}")

    # --------------------------------------------------------------
    # if we found nothing (should never happen) fall back to uniform
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)

    # --------------------------------------------------------------
    # allocate remaining time to refinement phases
    remaining = deadline - time.time()
    if remaining > 30.0:
        # 40% Adam, 40% sub‑gradient, 15% SA, rest for possible upscale
        t_adam = remaining * 0.40
        t_sub  = remaining * 0.40
        t_sa   = remaining * 0.15

        # ----- Adam (smooth‑max) ---------------------------------
        if t_adam > 6.0:
            local_deadline = time.time() + t_adam
            a_adam, val_adam = _adam_optimize(best_seq, n,
                                              conv_fft_len,
                                              corr_fft_len,
                                              local_deadline,
                                              max_iters=200_000)
            if val_adam < best_val:
                best_val = val_adam
                best_seq = a_adam.copy()
                print(f"[Adam] improved best to {val_adam:.6f}")

        # ----- exact sub‑gradient ---------------------------------
        if time.time() < deadline - 5.0 and t_sub > 6.0:
            local_deadline = time.time() + t_sub
            a_sub, val_sub = _subgradient_optimize(best_seq, n,
                                                   conv_fft_len,
                                                   corr_fft_len,
                                                   local_deadline,
                                                   max_iters=200_000)
            if val_sub < best_val:
                best_val = val_sub
                best_seq = a_sub.copy()
                print(f"[Subgrad] improved best to {val_sub:.6f}")

        # ----- lightweight simulated annealing --------------------
        if time.time() < deadline - 2.0 and t_sa > 6.0:
            a_sa, val_sa = _simulated_annealing(best_seq, n,
                                                conv_fft_len,
                                                deadline,
                                                best_val)
            if val_sa < best_val:
                best_val = val_sa
                best_seq = a_sa.copy()
                print(f"[SA] improved best to {val_sa:.6f}")

    # --------------------------------------------------------------
    # optional upscale to 8192 points and short refinement
    if best_seq is not None and time.time() + 18 < deadline:
        n_big = 8192
        x_small = np.linspace(0.0, 1.0, best_seq.shape[0], endpoint=False)
        x_big = np.linspace(0.0, 1.0, n_big, endpoint=False)
        a_big = np.interp(x_big, x_small, best_seq)
        a_big = np.maximum(a_big, 0.0)
        a_big = a_big / a_big.sum()

        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        # very short Adam + sub‑grad on the enlarged grid
        t_big_deadline = deadline - 10.0
        a_big_adam, _ = _adam_optimize(a_big, n_big,
                                       conv_fft_len_big,
                                       corr_fft_len_big,
                                       t_big_deadline,
                                       max_iters=80_000)

        a_big_ref, val_big = _subgradient_optimize(a_big_adam, n_big,
                                                   conv_fft_len_big,
                                                   corr_fft_len_big,
                                                   deadline,
                                                   max_iters=100_000)
        if val_big < best_val:
            best_val = val_big
            best_seq = a_big_ref.copy()
            print(f"[Upscale] final best {val_big:.6f}")

    # --------------------------------------------------------------
    # final housekeeping (clip to limits required by evaluate_sequence)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # report the final objective using the judge's routine (if available)
    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"Final reported value {final_val:.6f} (length {len(best_seq)})")
    except Exception:
        print(f"Final internal best {best_val:.6f} (length {len(best_seq)})")

    return best_seq.tolist()
