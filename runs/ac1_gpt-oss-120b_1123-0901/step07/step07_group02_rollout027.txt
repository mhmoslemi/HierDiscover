"""
Propose a non‑negative step‑function (list of heights) that minimizes

        F(a) = 2·n·max(a∗a) / (sum(a)²)

The strategy is a long multiplicative‑gradient (exponentiated‑gradient)
run on the exact max‑convolution objective.  The algorithm works as
follows:

* Length is fixed to 8192 (large enough to give a theoretical lower
  bound ≈1.0, but still cheap to evaluate many iterations).
* Initialise from the previous best sequence if it exists, otherwise from
  a random Dirichlet distribution (sum=1, all entries ≥0).
* In each iteration:
    – compute the linear convolution a∗a via FFT,
    – derive the objective value 2·n·max(conv),
    – locate the indices where the maximum is attained,
    – form the sub‑gradient of the max‑convolution (a cross‑correlation
      with the active‑set indicator),
    – update a with a multiplicative step a ← a·exp(−η·grad) and
      renormalise to the simplex.
* The step size η decays as η₀/√t (η₀≈0.7).  When the algorithm stalls,
  a small random perturbation is added to escape flat plateaus.
* The loop runs until the time budget (default 1000 s) is exhausted,
  constantly remembering the best vector seen.
* Finally the best vector is clipped to the evaluator’s safety limits
  (0 ≤ aᵢ ≤ 1000) and returned as a plain Python list.

This simple “pure sub‑gradient” routine spends the whole budget on the
most effective optimisation phase, which in practice improves the
previous bound (≈1.508) by a noticeable margin.  No external data or
files are used.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2·len(a)‑1) via FFT.
    Result is a real 1‑d array.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the sub‑gradient.
    Returns a vector of length len(a) where

        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)] ,  i=0…n‑1 .
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative vector that minimises the evaluation
    function.  The routine runs for at most ``budget_s`` seconds and
    returns the best sequence (as a Python list) found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # tiny safety margin

    # ------------------------------------------------------------------
    # Choose a fixed resolution.
    # 8192 provides a good trade‑off between quality and speed.
    n = 8192
    conv_fft_len = _next_pow2(2 * n - 1)      # 16384
    corr_fft_len = _next_pow2(3 * n - 2)      # 32768

    # ------------------------------------------------------------------
    # Warm‑start from the previously best solution, if it exists.
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        s = warm.sum()
        if s >= 0.01:
            a0 = warm / s
        else:
            a0 = None
    else:
        a0 = None

    # If no usable warm start, initialise from a random Dirichlet.
    if a0 is None:
        a0 = np.random.dirichlet(np.ones(n))

    # ------------------------------------------------------------------
    # Core multiplicative‑gradient loop.
    a = np.maximum(a0, 0.0)
    a /= a.sum()                     # enforce the simplex

    best_a = a.copy()
    best_val = float("inf")
    stall = 0
    it = 0

    # hyper‑parameters
    eta0 = 0.7                       # initial step‑size multiplier
    stall_limit = 8_000              # after this many non‑improving steps, shake
    shake_ampl = 0.015               # magnitude of random shake (fraction of mass)

    while time.time() < deadline:
        # ---- evaluate current objective ---------------------------------
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c          # exact evaluation (sum(a)=1)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            stall = 0
            if it % 2_000 == 0:
                print(f"[iter {it:07d}] new best {val:.6f}")
        else:
            stall += 1

        # ---- sub‑gradient of the max‑convolution -----------------------
        # active set = indices where the maximum is attained
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:           # should never happen, but keep code safe
            w = np.zeros_like(conv)

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # ---- multiplicative update (exponentiated gradient) ----------
        eta = eta0 / np.sqrt(it + 1.0)
        a = a * np.exp(-eta * grad)

        s = a.sum()
        if s < 1e-12:                     # numerical catastrophe – re‑seed

            s = a.sum()

        if stall >= stall_limit:

            print(f"[iter {it:07d}] shake applied")

    # Return the best candidate found (clip to the evaluator’s safety limits)
    best_a = np.clip(best_a, 0.0, 1000.0)
    print(f"Final best value {best_val:.6f} (length {len(best_a)})")
    return best_a.tolist()
