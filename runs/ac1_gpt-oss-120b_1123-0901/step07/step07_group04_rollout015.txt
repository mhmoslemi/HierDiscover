        sum_exp = exp_part.sum()
        w = exp_part / sum_exp

        val = 2.0 * n * smooth_max  # since sum=1
        if val < best_val - 1e-12:

        # renormalize

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility FFT helpers
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


# ----------------------------------------------------------------------
# Objective (exact max‑convolution)
def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Fast exact evaluation (assumes a is already normalised)."""
    conv = _full_convolution(a, conv_fft_len)
    return float(2.0 * n * conv.max() / (a.sum() ** 2))

    if warm_start is not None:
        candidates.append(warm_start.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            candidates.append(np.roll(warm_start, shift))

    # Uniform baseline
    candidates.append(np.full(n, 1.0 / n))

    # Random Dirichlet with many concentration values
    for conc in (0.001, 0.005, 0.01, 0.03, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        cand = np.random.dirichlet(np.full(n, conc))
        candidates.append(cand)

    i = np.arange(n, dtype=np.float64)

    # Edge‑exponential (with fine α‑grid)
    for alpha in np.logspace(-3, 1.2, num=32):           # 0.001 … ~16
        edge = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
        edge /= edge.sum()
        for w in np.linspace(0.0, 1.0, 9):               # 0,0.125,…,1
            shape = w * edge + (1.0 - w) / n
            candidates.append(shape)

    # Edge‑power‑law
    for p in np.linspace(0.1, 6.0, num=24):
        edge = (i + 1.0) ** (-p) + (n - i) ** (-p)
        edge /= edge.sum()
        for w in np.linspace(0.0, 1.0, 7):
            shape = w * edge + (1.0 - w) / n
            candidates.append(shape)

    # Double‑Gaussian edges
    for gamma in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0):
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        shape /= shape.sum()
        candidates.append(shape)

    # Positive half‑wave sinusoid
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    candidates.append(shape)

    # Two‑peak bump (symmetric Gaussians placed away from centre)
    for sigma in (0.05, 0.08, 0.12, 0.2):
        peak = np.exp(-0.5 * ((i - n // 4) / (sigma * n)) ** 2)
        peak += np.exp(-0.5 * ((i - 3 * n // 4) / (sigma * n)) ** 2)
        peak /= peak.sum()
        candidates.append(peak)

    return candidates

# Smoothed Adam optimiser (log‑space, soft‑max of convolution)
def _smooth_adam_opt(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
    init_alpha: float = 0.5,
    max_alpha: float = 200.0,
) -> tuple[np.ndarray | None, float]:
    """Adam on the log‑soft‑max approximation of the max‑convolution."""
    a = np.maximum(a0, 1e-16)
    a /= a.sum()

    m_arr = np.zeros_like(a)
    v_arr = np.zeros_like(a)

    t = 0
    alpha = init_alpha
    lr = 0.5
    min_lr = 0.001
    decay_lr = 0.99998
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    last_improve = time.time()

    while time.time() < deadline:
        t += 1

        # Convolution
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()

        # Soft‑max weights (stable)
        exp_part = np.exp(alpha * (conv - max_c))
        sum_exp = exp_part.sum()
        w = exp_part / sum_exp

        # Smooth max value
        smooth_max = (np.log(sum_exp) + alpha * max_c) / alpha
        val = 2.0 * n * smooth_max  # denominator = 1 because sum(a)=1

        if val < best_val - 1e-12:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(
                f"[SmoothAdam n={n} iter={t:07d}] new best {val:.6f} "
                f"alpha={alpha:.3f} lr={lr:.5f}"
            )

        # Gradient of smooth max w.r.t a
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # Adam moments
        m_arr = beta1 * m_arr + (1 - beta1) * grad
        v_arr = beta2 * v_arr + (1 - beta2) * (grad * grad)

        m_hat = m_arr / (1 - beta1 ** t)
        v_hat = v_arr / (1 - beta2 ** t)

        step = -lr * m_hat / (np.sqrt(v_hat) + eps)
        step = np.clip(step, -20.0, 20.0)   # avoid overflow

        # Exponentiated (multiplicative) update in log‑space
        a = a * np.exp(step)
        s = a.sum()
        if s <= 0 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        # Temperature schedule
        if alpha < max_alpha:
            alpha = min(max_alpha, alpha * 1.0006)

        # Learning‑rate decay
        lr = max(min_lr, lr * decay_lr)

        # Random jitter if stuck
        if time.time() - last_improve > 4.0:
            jitter = (np.random.rand(n) - 0.5) * 0.015
            a = a * np.exp(jitter)
            a /= a.sum()
            last_improve = time.time()

    return best_seq, best_val

# Original sub‑gradient optimiser (kept unchanged apart from minor cleanup)
def _long_subgradient_opt(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray | None, float]:
    """Mirror‑descent on the true max‑convolution (as in the reference code)."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 0.01:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    eta0 = 0.5
    decay = 0.99995
    min_eta = 0.001

    last_improve = time.time()
    it = 0

    while time.time() < deadline:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val - 1e-12:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Subgrad n={n} iter={it:07d}] new best {val:.6f}")

        # Active set (indices achieving the max)
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = max(min_eta, eta0 * (decay ** it))

        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        if time.time() - last_improve > 5.0:
            perturb = (np.random.rand(n) - 0.5) * 0.02
            a = a * np.exp(perturb)
            a = a / a.sum()
            last_improve = time.time()

        it += 1

    return best_seq, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """Main entry point – searches for a low‑evaluation‑score step function."""
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Stage 0 – small grid for quick exploration
    n_small = 2048
    conv_fft_small = _next_pow2(2 * n_small - 1)
    corr_fft_small = _next_pow2(3 * n_small - 2)

    # Warm‑start from previous run if available
    warm = globals().get("height_sequence_1")
    if warm is not None:
        ws = np.asarray(warm, dtype=np.float64)
        if ws.shape[0] > n_small:
            ws = ws[:n_small]
        elif ws.shape[0] < n_small:
            ws = np.pad(ws, (0, n_small - ws.shape[0]))
        ws = np.maximum(ws, 0.0)
        if ws.sum() >= 0.01:
            ws = ws / ws.sum()
        else:
            ws = None
    else:
        ws = None

    # Generate a broad candidate pool
    candidates = _generate_parametric_candidates(n_small, ws)

    # Fast evaluation (exact max‑convolution) – keep the best few
    cand_vals = []
    for cand in candidates:
        if time.time() >= deadline:
            break
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 0.01:
            cand = np.random.dirichlet(np.ones(n_small))
            s = cand.sum()
        cand = cand / s
        val = _objective(cand, n_small, conv_fft_small)
        cand_vals.append((val, cand))

    if not cand_vals:
        # Fallback: uniform vector
        best_seq = np.full(n_small, 1.0 / n_small, dtype=np.float64)
        best_val = _objective(best_seq, n_small, conv_fft_small)
        print("*** No candidates evaluated – returning uniform.")
        globals()["height_sequence_1"] = best_seq.tolist()
        return best_seq.tolist()

    cand_vals.sort(key=lambda x: x[0])
    top_k = 6
    top_candidates = [c for _, c in cand_vals[:top_k]]

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Stage 1 – smooth Adam optimisation on the small grid
    for idx, start in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        print(f"--- SmoothAdam start #{idx + 1}/{len(top_candidates)} (n={n_small})")
        best_seq, best_val = _smooth_adam_opt(
            start,
            n_small,
            conv_fft_small,
            corr_fft_small,
            deadline,
            best_seq,
            best_val,
            init_alpha=0.5,
            max_alpha=250.0,
        )

    # ------------------------------------------------------------------
    # Stage 2 – upscale to medium grid (n=4096) and refine with sub‑gradient
    if best_seq is not None and time.time() < deadline - 40.0:
        n_mid = 4096
        conv_fft_mid = _next_pow2(2 * n_mid - 1)
        corr_fft_mid = _next_pow2(3 * n_mid - 2)

        x_small = np.linspace(0, 1, n_small, endpoint=False)
        x_mid = np.linspace(0, 1, n_mid, endpoint=False)
        a_mid = np.interp(x_mid, x_small, best_seq)
        a_mid = np.maximum(a_mid, 0.0)
        a_mid /= a_mid.sum()

        init_val = _objective(a_mid, n_mid, conv_fft_mid)
        print(f"--- Subgradient refinement start (n={n_mid}, init val={init_val:.6f})")
        best_seq, best_val = _long_subgradient_opt(
            a_mid,
            n_mid,
            conv_fft_mid,
            corr_fft_mid,
            deadline,
            a_mid,
            init_val,
        )

    # ------------------------------------------------------------------
    # Stage 3 – optional final refinement on a finer grid (n=8192)
    time_left = deadline - time.time()
    if time_left > 30.0 and best_seq is not None and best_seq.shape[0] >= 4096:
        n_big = 8192
        conv_fft_big = _next_pow2(2 * n_big - 1)
        corr_fft_big = _next_pow2(3 * n_big - 2)

        x_mid = np.linspace(0, 1, best_seq.shape[0], endpoint=False)
        x_big = np.linspace(0, 1, n_big, endpoint=False)
        a_big = np.interp(x_big, x_mid, best_seq)
        a_big = np.maximum(a_big, 0.0)
        a_big /= a_big.sum()

        init_val = _objective(a_big, n_big, conv_fft_big)
        print(f"--- Final refinement on n={n_big} (init val={init_val:.6f})")
        best_seq, best_val = _long_subgradient_opt(
            a_big,
            n_big,
            conv_fft_big,
            corr_fft_big,
            deadline,
            a_big,
            init_val,
        )
        n_final = n_big
    else:
        n_final = best_seq.shape[0] if best_seq is not None else 4096

    # ------------------------------------------------------------------
    # Fallback handling
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)
        n_final = 4096

    # Clip as required by the evaluator
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Store for possible warm‑starts in subsequent runs
    globals()["height_sequence_1"] = best_seq.tolist()

    print(f"*** Finished search: best objective ≈ {best_val:.6f} (n={n_final})")
    return best_seq.tolist()
