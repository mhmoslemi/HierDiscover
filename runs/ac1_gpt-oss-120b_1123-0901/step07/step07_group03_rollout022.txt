    x0 = np.log(a0 + 1e-15)

    def fun(x):
        # compute a from softmax
        mx = np.max(x)
        exp_x = np.exp(x - mx)
        a = exp_x/exp_x.sum()
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        # smooth max using log-sum-exp weighting w = exp(lam*(conv-max_c))
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0/w.size)
        else:
            w /= w_sum

        surrogate = (1.0/lam) * np.log(np.mean(np.exp(lam*conv)))

        grad_x = a * (grad_a - avg)

def _project_simplex(v):
    # ensure v sums to 1 and non-negative
    # code from: 
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a step‐function (non‑negative list) that minimizes

    F(a) = 2·n·max(a*a) / (sum(a)**2)

The algorithm is a multi‑stage search:

1. **Massive random warm‑start:** generate thousands of Dirichlet
   samples (different concentration) and apply the “anti‑pair”
   ordering (largest opposite smallest).  Keep the best shape.

2. **Smooth‑max L‑BFGS refinement:** optimise a soft‑max surrogate
   (log‑sum‑exp of the convolution) using SciPy’s L‑BFGS‑B.
   The temperature (λ) is increased in stages.

3. **Exact‑max sub‑gradient phases:** two variants are tried
   – the original multiplicative (exponentiated‑gradient) update,
   – an additive projected‑gradient update (Euclidean projection onto
     the simplex).  Each runs as long as the remaining budget allows.

4. **Local stochastic hill‑climbing:** swaps or fractional mass transfers
   with a simple annealing acceptance rule.

5. **Optional up‑sampling:** if time permits, linearly interpolate the
   current best vector to 8192 points and repeat steps 2–4 on the finer grid.

All heavy work respects the global time limit (default 1000 s).  The
returned list is clipped to the range accepted by ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for gradients.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
# Objective (exact)
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(a*a) / (sum(a)^2).  a must be non‑negative."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    conv = _full_convolution(a, _next_pow2(2 * a.shape[0] - 1))
    return 2.0 * a.shape[0] * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = ind[cond][-1] - 1
        theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Warm‑start handling
def _warm_start(n: int):
    """Return a previously stored sequence (global `height_sequence_1`) if usable."""
    warm = globals().get("height_sequence_1")
    if warm is None:
        return None
    warm = np.asarray(warm, dtype=np.float64)
    if warm.shape[0] > n:
        warm = warm[:n]
    elif warm.shape[0] < n:
        warm = np.pad(warm, (0, n - warm.shape[0]))
    warm = np.maximum(warm, 0.0)
    if warm.sum() < 0.01:
        return None
    return warm / warm.sum()


# ----------------------------------------------------------------------
# Candidate generators
def _uniform_candidate(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _random_dirichlet_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    a = np.random.dirichlet(alpha * np.ones(n))
    return a.astype(np.float64)


def _anti_pair_arrangement(vals: np.ndarray) -> np.ndarray:
    """Place largest opposite smallest (anti‑pair ordering)."""
    n = len(vals)
    s = np.sort(vals)[::-1]                    # descending
    a = np.empty(n, dtype=np.float64)
    half = n // 2
    a[:half] = s[:half]
    a[-half:] = s[-half:][::-1]
    if n % 2:
        a[half] = s[half]
    return a


def _anti_pair_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    vals = np.random.dirichlet(alpha * np.ones(n))
    vals /= vals.sum()
    return _anti_pair_arrangement(vals)


def _gaussian_edges_candidate(n: int, gamma: float = 0.5) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
    shape /= shape.sum()
    return shape


def _triangle_candidate(n: int) -> np.ndarray:
    x = np.arange(n, dtype=np.float64)
    shape = np.minimum(x + 1, n - x)          # 1,2,…,⌈n/2⌉,…,2,1
    shape /= shape.sum()
    return shape


# ----------------------------------------------------------------------
# L‑BFGS optimisation of a smooth‑max surrogate
def _lbfgs_obj(logits: np.ndarray, n: int, conv_fft_len: int,
               corr_fft_len: int, lam: float):
    """Smooth‑max (log‑sum‑exp) objective and its gradient w.r.t. logits."""
    mx = np.max(logits)
    exp_l = np.exp(logits - mx)
    a = exp_l / exp_l.sum()

    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    # stable log‑sum‑exp
    z = np.exp(lam * (conv - max_c))
    sum_z = np.sum(z)
    surrogate = max_c + (1.0 / lam) * np.log(sum_z)

    w = z / sum_z                     # soft weights, sum to 1
    grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
    avg = np.dot(a, grad_a)
    grad_log = a * (grad_a - avg)    # chain rule through softmax
    return surrogate, grad_log.astype(np.float64)


def _lbfgs_optimize(a0: np.ndarray, n: int, conv_fft_len: int,
                    corr_fft_len: int, deadline: float,
                    lam_schedule=(5.0, 15.0, 40.0, 120.0),
                    max_iter_stage: int = 250) -> tuple[np.ndarray, float]:
    """Run L‑BFGS on the smooth‑max surrogate, increasing λ progressively."""
    import scipy.optimize as opt

    best_a = a0.copy()
    best_val = _objective(best_a)
    logits = np.log(best_a + 1e-12)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        # remaining time → bound the number of iterations
        remaining = deadline - time.time()
        iters = min(max_iter_stage, max(1, int(remaining * 10)))  # generous bound
        res = opt.minimize(_lbfgs_obj, logits,
                           args=(n, conv_fft_len, corr_fft_len, lam),
                           method='L-BFGS-B', jac=True,
                           options={'maxiter': iters, 'disp': False})
        logits = res.x
        mx = np.max(logits)
        exp_l = np.exp(logits - mx)
        a = exp_l / exp_l.sum()
        val = _objective(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[L‑BFGS λ={lam:.1f}] new best {val:.6f}")
    return best_a, best_val

# Multiplicative (exponentiated‑gradient) sub‑gradient optimisation
def _subgradient_optimize(a0: np.ndarray, n: int, conv_fft_len: int,
                          corr_fft_len: int, deadline: float,
                          max_iters: int = 800_000,
                          eta0: float = 0.7) -> tuple[np.ndarray, float]:
    a = np.maximum(a0, 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_val = _objective(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()

        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1)
        a = a * np.exp(-eta * grad_a)          # multiplicative update
        s = a.sum()
        if s < 1e-12:                         # occasional re‑initialisation
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        val = _objective(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[MulSubgrad it={it:06d}] new best {val:.6f}")

        it += 1
    return best_a, best_val

# Additive projected sub‑gradient optimisation
def _additive_subgradient_optimize(a0: np.ndarray, n: int,
                                  conv_fft_len: int, corr_fft_len: int,
                                  deadline: float,
                                  max_iters: int = 600_000,
                                  eta0: float = 0.5) -> tuple[np.ndarray, float]:
    a = np.maximum(a0, 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_val = _objective(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)  # sub‑gradient of max
        eta = eta0 / np.sqrt(it + 1)

        a_new = a - eta * grad_a
        a = _project_onto_simplex(a_new)

        val = _objective(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[AddSubgrad it={it:06d}] new best {val:.6f}")

        it += 1
    return best_a, best_val

# Local stochastic hill‑climbing (simulated annealing)
def _local_random_search(a0: np.ndarray, n: int,
                         conv_fft_len: int, deadline: float,
                         max_moves: int = 300_000) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a)

    start_temp = 0.02
    it = 0
    while time.time() < deadline and it < max_moves:
        if np.random.rand() < 0.5:               # swap two entries
            i, j = np.random.randint(n, size=2)
            if i != j:
                a2 = a.copy()
                a2[i], a2[j] = a2[j], a2[i]
            else:
                continue
        else:                                    # transfer a fraction
            i = np.random.randint(n)
            if a[i] < 1e-15:
                continue
            j = np.random.randint(n)
            if i == j:
                continue
            delta = a[i] * (np.random.rand() * 0.8)   # up to 80 % move
            a2 = a.copy()
            a2[i] -= delta
            a2[j] += delta

        val2 = _objective(a2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
        else:
            temp = start_temp * (1.0 - it / max_moves)
            if temp > 0.0 and np.random.rand() < np.exp(-(val2 - best_val) / temp):
                a = a2

        if it % 5000 == 0:
            print(f"[Local it={it:06d}] best {best_val:.6f}")

        it += 1
    return best_a, best_val

# Main entry point
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises
    F(a)=2·n·max(a∗a)/(sum(a)²) within the given time budget.
    Returns the best sequence found (as a plain list of floats).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Problem size
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # 1) Massive random warm‑starts
    best_seq = None
    best_val = float("inf")

    # include possible warm start provided by the harness
    warm = _warm_start(n)
    if warm is not None:
        cand_val = _objective(warm)
        best_seq, best_val = warm.copy(), cand_val
        print(f"[Warm start] value {cand_val:.6f}")

    # a few handcrafted starters (cheap)
    starters = [
        _uniform_candidate(n),
        _triangle_candidate(n),
        _gaussian_edges_candidate(n, 0.3),
        _gaussian_edges_candidate(n, 0.7),
        _gaussian_edges_candidate(n, 1.2),
    ]

    for c in starters:
        val = _objective(c)
        if val < best_val:
            best_val, best_seq = val, c.copy()
            print(f"[Starter] value {val:.6f}")

    # random anti‑pair Dirichlet pool
    n_samples = 2500
    print(f"Generating {n_samples} random anti‑pair candidates …")
    for _ in range(n_samples):
        # sample a concentration α in a wide logarithmic range
        log_alpha = np.random.uniform(-2.0, 1.5)   # α ∈ [0.01, ≈31.6]
        alpha = 10.0 ** log_alpha
        cand = _anti_pair_candidate(n, alpha)
        val = _objective(cand)
        if val < best_val:
            best_val, best_seq = val, cand.copy()
            print(f"[Random pool] new best {val:.6f}")

    # --------------------------------------------------------------
    # 2) L‑BFGS smooth‑max refinement
    # allocate about 12 % of remaining budget
    lbfgs_deadline = min(deadline - 120.0, time.time() + 180.0)
    if time.time() < lbfgs_deadline:
        seq_lb, val_lb = _lbfgs_optimize(best_seq, n, conv_fft_len,
                                         corr_fft_len, lbfgs_deadline)
        if val_lb < best_val:
            best_val, best_seq = val_lb, seq_lb.copy()
            print(f"[L‑BFGS] improved to {val_lb:.6f}")

    # --------------------------------------------------------------
    # 3) Exact‑max sub‑gradient phases
    # (a) multiplicative (exponentiated‑gradient)
    sub_deadline = min(deadline - 70.0, time.time() + 250.0)
    if time.time() < sub_deadline:
        seq_sub, val_sub = _subgradient_optimize(best_seq, n,
                                                 conv_fft_len, corr_fft_len,
                                                 sub_deadline,
                                                 max_iters=1_200_000,
                                                 eta0=0.7)
        if val_sub < best_val:
            best_val, best_seq = val_sub, seq_sub.copy()
            print(f"[MulSubgrad] improved to {val_sub:.6f}")

    # (b) additive projected sub‑gradient
    add_deadline = min(deadline - 40.0, time.time() + 180.0)
    if time.time() < add_deadline:
        seq_add, val_add = _additive_subgradient_optimize(best_seq, n,
                                                          conv_fft_len,
                                                          corr_fft_len,
                                                          add_deadline,
                                                          max_iters=800_000,
                                                          eta0=0.5)
        if val_add < best_val:
            best_val, best_seq = val_add, seq_add.copy()
            print(f"[AddSubgrad] improved to {val_add:.6f}")

    # --------------------------------------------------------------
    # 4) Local stochastic annealing
    if best_seq is not None and time.time() < deadline - 20.0:
        local_deadline = deadline - 5.0
        seq_loc, val_loc = _local_random_search(best_seq, n,
                                                conv_fft_len,
                                                local_deadline,
                                                max_moves=400_000)
        if val_loc < best_val:
            best_val, best_seq = val_loc, seq_loc.copy()
            print(f"[Local] improved to {val_loc:.6f}")

    # --------------------------------------------------------------
    # 5) Optional up‑sampling to a finer grid (8192)
    if time.time() < deadline - 10.0:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        x_old = np.linspace(0.0, 1.0, n, endpoint=False)
        x_new = np.linspace(0.0, 1.0, n_big, endpoint=False)
        a_big = np.interp(x_new, x_old, best_seq)
        a_big = np.maximum(a_big, 0.0)
        a_big = a_big / a_big.sum()

        big_deadline = min(deadline - 2.0, time.time() + 180.0)
        # a quick L‑BFGS on the big grid
        seq_big_lb, val_big_lb = _lbfgs_optimize(a_big, n_big,
                                                 conv_fft_len_big,
                                                 corr_fft_len_big,
                                                 big_deadline,
                                                 lam_schedule=(5, 20, 80, 200),
                                                 max_iter_stage=120)
        # final sub‑gradient fine‑tuning
        final_deadline = min(deadline, time.time() + 120.0)
        seq_big_sub, val_big_sub = _subgradient_optimize(seq_big_lb, n_big,
                                                         conv_fft_len_big,
                                                         corr_fft_len_big,
                                                         final_deadline,
                                                         max_iters=600_000,
                                                         eta0=0.6)
        if val_big_sub < best_val:
            best_val, best_seq = val_big_sub, seq_big_sub.copy()
            print(f"[Big grid] improved to {val_big_sub:.6f}")

    # --------------------------------------------------------------
    # 6) Safeguard – guarantee a valid return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to evaluator limits (non‑negative, ≤ 1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
