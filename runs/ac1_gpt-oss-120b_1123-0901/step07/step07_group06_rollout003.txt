"""
Simple iterative refinement with active-set linear program.

We start from a warm‑start (or uniform) distribution on the simplex.
At each iteration we compute the autocorrelation a⋆a, keep the
few largest entries (active set) and solve a linear program that
maximises the ℓ1‑norm of a new vector g under the constraints
(f⋆g) ≤ M for all active indices (M = max(f⋆f)).  The new vector
is mixed with the current f using a small step size; the step size
is adaptively reduced when the maximum autocorrelation does not
decrease.  After the LP phase we apply a short projected‑gradient
refinement on the exact objective.  If time allows we upscale the
solution to a finer grid (n=8192) and repeat the whole procedure.

The method is deterministic apart from the random‑initialised warm
start and occasional jitter; it works within the 1000 s budget and
produces a step‑function with a lower evaluation bound than the
previous baseline.
"""

import time
import numpy as np
from scipy.optimize import linprog


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("v must be 1‑d")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all non‑positive – return the uniform simplex point
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a⋆a (length 2·len(a)-1)."""
    return np.convolve(a, a)


def _objective_normed(a: np.ndarray) -> float:
    """Objective 2·n·max(a⋆a) / (sum(a)²) assuming a is already normalised."""
    n = a.shape[0]
    conv = _full_convolution(a)
    M = conv.max()
    return 2.0 * n * M / (a.sum() ** 2)


def _active_indices(conv: np.ndarray, max_idx: int, k: int = 7) -> np.ndarray:
    """
    Return a set of indices to be used as active constraints.
    We keep the k largest values and always include the global maximiser.
    """
    size = conv.shape[0]
    if k >= size:
        return np.arange(size, dtype=np.int64)
    # select k indices with largest values (unsorted)
    top = np.argpartition(-conv, k - 1)[:k]
    # ensure the absolute maximum is present
    if max_idx not in top:
        top = np.concatenate((top, [max_idx]))
    return top


def _build_constraint_matrix(f: np.ndarray, indices: np.ndarray) -> np.ndarray:
    """
    Build matrix A of shape (len(indices), len(f)) where each row i
    corresponds to the linear map g ↦ (f⋆g)[indices[i]].
    """
    n = f.shape[0]
    rows = []
    for j in indices:
        # row_k = f[j - k] if 0 ≤ j - k < n else 0
        idx = np.arange(n)
        valid = (j - idx >= 0) & (j - idx < n)
        row = np.zeros(n, dtype=np.float64)
        row[valid] = f[j - idx[valid]]
        rows.append(row)
    if rows:
        return np.vstack(rows)
    else:
        return np.empty((0, n), dtype=np.float64)


def _lp_step(f: np.ndarray, M: float, active_idx: np.ndarray, step: float) -> np.ndarray:
    """
    One iteration of the LP‑based improvement.
    Solves:
        max  Σ g_i   s.t.   A g ≤ M,   g ≥ 0
    where A encodes the active convolution constraints.
    The resulting g is mixed into f with mixing weight `step`
    and projected back onto the simplex.
    """
    n = f.shape[0]
    if active_idx.size == 0:
        # no constraints – return a random direction
        g = np.random.dirichlet(np.ones(n))
    else:
        A = _build_constraint_matrix(f, active_idx)
        b = np.full(A.shape[0], M, dtype=np.float64)

        c = -np.ones(n, dtype=np.float64)            # maximise Σ g_i  <=> minimise -Σ g_i
        bounds = [(0.0, None)] * n

        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.success:
            g = res.x
            # normalise – direction only matters
            total = g.sum()
            if total > 1e-12:
                g = g / total
            else:
                g = np.random.dirichlet(np.ones(n))
        else:
            # fallback – random direction
            g = np.random.dirichlet(np.ones(n))

    # convex combination and simplex projection
    f_new = (1.0 - step) * f + step * g
    f_new = _proj_simplex(f_new)
    return f_new


def _proj_gradient_refine(a: np.ndarray, n: int, deadline: float) -> tuple[np.ndarray, float]:
    """
    Simple projected‑gradient descent on the exact objective.
    Uses the sub‑gradient of max(a⋆a) based on the active set.
    """
    a = _proj_simplex(a.copy())
    best_seq = a.copy()
    best_val = float('inf')
    lr = 0.2
    last_improve = time.time()

    while time.time() < deadline:
        conv = _full_convolution(a)
        M = conv.max()
        val = 2.0 * n * M / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[ProjGrad] new best {val:.6f}")

        # active mask
        active = np.isclose(conv, M, atol=1e-12, rtol=0.0)
        if not np.any(active):
            # nothing to do – break
            break
        w = np.zeros_like(conv)
        w[active] = 1.0 / active.sum()

        # compute cross‑correlation w ★ a efficiently via convolution
        rev_a = a[::-1]
        corr = np.convolve(w, rev_a)               # length 3n‑2
        start = n - 1
        grad = 2.0 * corr[start:start + n]          # dM/da

        # simple gradient step followed by projection
        a_candidate = a - lr * grad
        a_candidate = _proj_simplex(a_candidate)

        # evaluate candidate
        conv_cand = _full_convolution(a_candidate)
        M_cand = conv_cand.max()
        val_cand = 2.0 * n * M_cand / (a_candidate.sum() ** 2)

        if val_cand < val:
            a = a_candidate
            lr = min(lr * 1.02, 1.0)   # increase step when moving downhill
        else:
            lr *= 0.7
            if lr < 1e-5:
                break

        if time.time() - last_improve > 5.0:
            # no improvement for a while – stop early
            break

        # gentle decay
        lr *= 0.99997

    return best_seq, best_val


def _run_lp_phase(start: np.ndarray, n: int, deadline: float) -> tuple[np.ndarray, float]:
    """
    Run the active‑set LP improvement until the time budget expires.
    Returns the best vector seen and its objective value.
    """
    f = _proj_simplex(start.copy())
    best_seq = f.copy()
    best_val = _objective_normed(f)
    step = 0.15          # initial mixing weight
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _full_convolution(f)
        M = conv.max()
        val = 2.0 * n * M / (f.sum() ** 2)
        if val < best_val:
            best_val = val
            best_seq = f.copy()
            print(f"[LP it={it:05d}] new best {val:.6f}")

        # choose active indices (centre heavy)
        max_idx = int(np.argmax(conv))
        active_idx = _active_indices(conv, max_idx, k=7)

        # one LP step
        f_new = _lp_step(f, M, active_idx, step)

        # evaluate whether we succeeded in decreasing the max
        conv_new = _full_convolution(f_new)
        M_new = conv_new.max()
        if M_new < M * (1.0 + 1e-8):
            f = f_new
            # slowly decay step size – we want fine moves later
            step = max(step * 0.995, 0.01)
        else:
            # reject – shrink step size aggressively
            step = max(step * 0.5, 0.01)

        # safety break – if step becomes tiny we are close to optimum
        if step < 0.02 and it > 200:
            # give a few more iterations then stop
            if time.time() > deadline - 2.0:
                break

    return best_seq, best_val


def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step function with minimal evaluation bound.
    The method combines an active‑set linear‑program improvement with a
    projected‑gradient refinement, and optionally upsamples to a finer grid.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # Choose an initial size. 4096 gives a good compromise between
    # resolution and speed; we later try 8192 if time permits.
    n = 4096

    # ------------------------------------------------------------------
    # Warm‑start: reuse a previously discovered construction if available.
    warm = globals().get("height_sequence_1")
    if warm is not None:
        f0 = np.asarray(warm, dtype=np.float64)
        if f0.shape[0] != n:
            if f0.shape[0] > n:
                f0 = f0[:n]
            else:
                f0 = np.pad(f0, (0, n - f0.shape[0]))
        f0 = np.maximum(f0, 0.0)
        total = f0.sum()
        if total < 0.01:
            f0 = np.full(n, 1.0 / n)
        else:
            f0 = f0 / total
    else:
        f0 = np.full(n, 1.0 / n)

    # ------------------------------------------------------------------
    # Phase 1 – active‑set LP based descent
    f_best, val_best = _run_lp_phase(f0, n, deadline)

    # ------------------------------------------------------------------
    # Phase 2 – short projected‑gradient refinement (exact objective)
    # We give it the remaining time budget.
    if time.time() < deadline - 1.0:
        f_best, val_best = _proj_gradient_refine(f_best, n, deadline)

    # ------------------------------------------------------------------
    # Phase 3 – optional up‑sampling to 8192 points if time permits.
    time_left = deadline - time.time()
    if time_left > 30.0:
        n_big = 8192
        # linear interpolation onto the finer grid
        x_old = np.linspace(0.0, 1.0, n, endpoint=False)
        x_big = np.linspace(0.0, 1.0, n_big, endpoint=False)
        f_big = np.interp(x_big, x_old, f_best)
        f_big = _proj_simplex(f_big)

        # repeat the LP phase on the larger grid (but with smaller iteration budget)
        sub_deadline = time.time() + min(20.0, time_left / 2.0)
        f_big, val_big = _run_lp_phase(f_big, n_big, sub_deadline)

        # final gradient refinement on the large grid
        if time.time() < deadline - 1.0:
            f_big, val_big = _proj_gradient_refine(f_big, n_big, deadline)

        # keep the better of the two resolutions
        if val_big < val_best:
            f_best, val_best = f_big, val_big
            n = n_big

    # ------------------------------------------------------------------
    # Clip to the evaluator's allowed range and return.
    f_best = np.clip(f_best, 0.0, 1000.0)
    print(f"\n*** Search finished: best objective ≈ {val_best:.6f} (n={n})")
    return f_best.tolist()

    for j in indices:
        idx = np.arange(n)
        valid = (j - idx >= 0) & (j - idx < n)
        row = np.zeros(n, dtype=np.float64)
        row[valid] = f[j - idx[valid]]
        rows.append(row)
    if rows:
        return np.vstack(rows)
    return np.empty((0, n), dtype=np.float64)


def _lp_step(f: np.ndarray, M: float, active_idx: np.ndarray, step: float) -> np.ndarray:
    """
    One linear‑program improvement step.
    Solve   max Σ g_i  s.t.  A·g ≤ M,  g ≥ 0,
    where A encodes the active convolution constraints.
    The feasible direction g is mixed into f with weight `step`,
    followed by simplex projection.
    """
    n = f.shape[0]
    if active_idx.size == 0:
        g = np.random.dirichlet(np.ones(n))
    else:
        A = _build_constraint_matrix(f, active_idx)
        b = np.full(A.shape[0], M, dtype=np.float64)

        c = -np.ones(n, dtype=np.float64)           # maximise Σg_i  ≡ minimise -Σg_i
        bounds = [(0.0, None)] * n

        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.success:
            g = res.x
            tot = g.sum()
            if tot > 1e-12:
                g = g / tot
            else:
                g = np.random.dirichlet(np.ones(n))
        else:
            g = np.random.dirichlet(np.ones(n))

    f_new = (1.0 - step) * f + step * g
    return _proj_simplex(f_new)


def _proj_gradient_refine(a: np.ndarray, n: int, deadline: float) -> tuple[np.ndarray, float]:
    """
    Short projected‑gradient descent on the exact objective.
    Uses the sub‑gradient of max(a⋆a) derived from the active set.
    """
    a = _proj_simplex(a.copy())
    best_seq = a.copy()
    best_val = float('inf')
    lr = 0.2
    last_improve = time.time()

    while time.time() < deadline:
        conv = _full_convolution(a)
        M = conv.max()
        val = 2.0 * n * M / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[ProjGrad] new best {val:.6f}")

        # active mask (exact max)
        active = np.isclose(conv, M, atol=1e-12, rtol=0.0)
        if not np.any(active):
            break
        w = np.zeros_like(conv)
        w[active] = 1.0 / active.sum()

        # cross‑correlation w ★ a = (w * rev(a))_{n-1 … 2n-2}
        rev_a = a[::-1]
        corr = np.convolve(w, rev_a)                # length 3n‑2
        start = n - 1
        grad = 2.0 * corr[start:start + n]          # ∂M/∂a

        a_candidate = a - lr * grad
        a_candidate = _proj_simplex(a_candidate)

        conv_cand = _full_convolution(a_candidate)
        val_cand = 2.0 * n * conv_cand.max() / (a_candidate.sum() ** 2)

        if val_cand < val:
            a = a_candidate
            lr = min(lr * 1.02, 1.0)
        else:
            lr *= 0.7
            if lr < 1e-5:
                break

        if time.time() - last_improve > 5.0:
            break
        lr *= 0.99997

    return best_seq, best_val


def _run_lp_phase(start: np.ndarray, n: int, deadline: float) -> tuple[np.ndarray, float]:
    """
    Run the active‑set LP descent until `deadline` is reached.
    Returns the best vector and its objective value.
    """
    f = _proj_simplex(start.copy())
    best_seq = f.copy()
    best_val = _objective_normed(f)
    step = 0.15
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _full_convolution(f)
        M = conv.max()
        val = 2.0 * n * M / (f.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = f.copy()
            print(f"[LP it={it:05d}] new best {val:.6f}")

        max_idx = int(np.argmax(conv))
        active_idx = _active_indices(conv, max_idx, k=7)

        f_new = _lp_step(f, M, active_idx, step)

        # accept only if the maximal convolution does not increase
        if _full_convolution(f_new).max() <= M * (1.0 + 1e-8):
            f = f_new
            step = max(step * 0.995, 0.01)          # slowly shrink step
        else:
            step = max(step * 0.5, 0.01)            # aggressive shrink on rejection

        # early exit when step becomes tiny
        if step < 0.02 and it > 200 and time.time() > deadline - 2.0:
            break

    return best_seq, best_val


def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Main entry point.  Searches for a step‑function with minimal
    evaluation bound under the given time budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # Base resolution (4096 points).  Larger grids are tried later if time permits.
    n = 4096

    # ------------------------------------------------------------------
    # Warm‑start from a previously discovered construction if available.
    warm = globals().get("height_sequence_1")
    if warm is not None:
        f0 = np.asarray(warm, dtype=np.float64)
        if f0.shape[0] > n:
            f0 = f0[:n]
        elif f0.shape[0] < n:
            f0 = np.pad(f0, (0, n - f0.shape[0]))
        f0 = np.maximum(f0, 0.0)
        if f0.sum() < 0.01:
            f0 = np.full(n, 1.0 / n)
        else:
            f0 = f0 / f0.sum()
    else:
        f0 = np.full(n, 1.0 / n)

    # ------------------------------------------------------------------
    # Phase 1 – active‑set LP descent.
    f_best, val_best = _run_lp_phase(f0, n, deadline)

    # ------------------------------------------------------------------
    # Phase 2 – projected‑gradient refinement on the exact objective.
    if time.time() < deadline - 1.0:
        f_best, val_best = _proj_gradient_refine(f_best, n, deadline)

    # ------------------------------------------------------------------
    # Phase 3 – optional up‑sampling to 8192 points if budget permits.
    time_left = deadline - time.time()
    if time_left > 30.0:
        n_big = 8192
        x_old = np.linspace(0.0, 1.0, n, endpoint=False)
        x_big = np.linspace(0.0, 1.0, n_big, endpoint=False)
        f_big = np.interp(x_big, x_old, f_best)
        f_big = _proj_simplex(f_big)

        # a shorter LP phase on the finer grid
        sub_deadline = time.time() + min(20.0, time_left / 2.0)
        f_big, val_big = _run_lp_phase(f_big, n_big, sub_deadline)

        # final gradient refinement on the large grid
        if time.time() < deadline - 1.0:
            f_big, val_big = _proj_gradient_refine(f_big, n_big, deadline)

        if val_big < val_best:
            f_best, val_best = f_big, val_big
            n = n_big

    # ------------------------------------------------------------------
    # Clip to the evaluator's allowed range and return.
    f_best = np.clip(f_best, 0.0, 1000.0)
    print(f"\n*** Search finished: best objective ≈ {val_best:.6f} (n={n})")
    return f_best.tolist()
