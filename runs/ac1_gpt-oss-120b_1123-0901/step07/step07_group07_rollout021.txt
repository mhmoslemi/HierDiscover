def _lp_iterative_refine(a0, n, deadline, t=0.2, max_iter=200):
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a, n, _next_pow2(2*n-1))
    for it in range(max_iter):
        if time.time() >= deadline:
            break
        # compute conv and M
        conv = np.convolve(a, a)
        M = conv.max()
        # build matrix A
        rows = []
        cols = []
        data = []
        for j in range(n):
            aj = a[j]
            # row indices where a_j appears
            start = j
            end = j + n
            # I'm going to use vectorized fill
            rows.extend(range(start, end))
            cols.extend([j] * n)
            data.extend([aj] * n)
        A = csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
        # RHS
        b_ub = np.full(2*n-1, M)
        c = -np.ones(n)  # maximize sum b
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
        if not res.success:
            print(f"LP iteration {it} failed, breaking")
            break
        b = res.x
        sum_b = b.sum()
        if sum_b <= a.sum() + 1e-12:
            # no improvement possible
            break
        # Normalized g
        g = b / sum_b
        # convex combination
        a = (1 - t) * a + t * g
        # maintain sum=1 (should be unchanged)
        # evaluate
        val = _objective(a, n, _next_pow2(2*n-1))
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"LP refine it {it} new best {val:.6f}")
    return best_a, best_val
```

def _small_candidates(n):
    i = np.arange(n)
    candidates = []
    candidates.append(np.full(n, 1.0/n))

    # arcsine
    shape = 1.0/np.sqrt((i+0.5)*(n-i-0.5))
    shape /= shape.sum()
    candidates.append(shape)

    # beta alpha 0.5,1.2,2.0
    for alpha in (0.2, 0.5, 1.0, 2.0):
        shape = (i+1)**(alpha-1)*(n-i)**(alpha-1)
        shape = np.maximum(shape, 0)
        shape /= shape.sum()
        candidates.append(shape)

    # exponential decline from left
    for alpha in (0.01, 0.05, 0.2, 0.5):
        shape = np.exp(-alpha*i) + np.exp(-alpha*(n-1-i))
        shape /= shape.sum()
        candidates.append(shape)

    # power law
    for p in (0.5, 1.0, 2.0, 3.0):
        shape = (i+1)**(-p) + (n-i)**(-p)
        shape /= shape.sum()
        candidates.append(shape)

    # Dirichlet random small ones
    for conc in (0.01, 0.05, 0.1):
        cand = np.random.dirichlet(np.full(n, conc))
        candidates.append(cand)
    return candidates

"""
Propose a low‑evaluation step‑function.

Algorithm outline
-----------------
1. **Coarse search on a small grid (n≈256).**
   - Generate a diverse set of parametric candidates
     (uniform, arcsine, beta‑shapes, exponentials, power‑laws,
     random Dirichlet draws and a possible warm‑start).
   - Keep the best one according to the objective.

2. **Iterative linear‑program refinement (LP).**
   For a fixed shape `f` we solve
       max   Σ b_i
       s.t.  (f * b)_k ≤ M   ∀k      (M = max(f*f))
             b_i ≥ 0
   This is a linear program because `f` is fixed.
   The solution `b` typically has a larger total mass while
   respecting the same peak `M`.  Rescaling `b` to the original
   sum and mixing it with `f` (convex combination) provably
   reduces the peak.  We repeat the LP step several times
   (tuned for speed on the small grid).

3. **Upscaling to a fine grid (n≈4096) and continuous optimisation.**
   - Linear interpolation of the refined small‑grid vector.
   - Run the two‑stage optimiser from the previous state‑of‑the‑art:
     *Adam* on a smooth‑max surrogate of the max‑convolution,
     followed by an exact sub‑gradient step on the true `max`.
     Both stages use FFT‑based convolutions for speed.

4. **Final optional random hill‑climb.**
   Tiny mass transfers between random indices are accepted
   only if they improve the true objective.

All steps respect the 1000 s budget; the best sequence found
is returned as a plain Python list.  The code works with only
NumPy, SciPy and the supplied `evaluate_sequence`.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using real FFT of length `fft_len`."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """
    Fast evaluation of the objective 2·n·max(a*a) / (sum(a)**2).
    """
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c / (a.sum() ** 2))

# ---------- LP‑based refinement on a small grid --------------------
def _build_conv_matrix(a: np.ndarray) -> csr_matrix:
    """Build the (2n‑1)×n matrix T with T[k,j] = a[j] when k‑j∈[0,n‑1]."""
    n = a.shape[0]
    total = n * n
    rows = np.empty(total, dtype=np.int32)
    cols = np.empty(total, dtype=np.int32)
    data = np.empty(total, dtype=np.float64)

    idx = 0
    for j in range(n):
        rows[idx : idx + n] = np.arange(j, j + n)
        cols[idx : idx + n] = j
        data[idx : idx + n] = a[j]
        idx += n
    return csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


def _lp_iterative_refine(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         t: float = 0.2,
                         max_iter: int = 150) -> tuple[np.ndarray, float]:
    """
    Repeatedly solve a linear programme that maximises Σb_i under the
    constraint (a0 * b) ≤ M, where M = max(a0*a0).  After each solve we
    mix the normalised solution back into the current vector.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a, n, _next_pow2(2 * n - 1))

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        # current peak of the self‑convolution
        conv = np.convolve(a, a)
        M = conv.max()

        # linear constraints A @ b ≤ M
        A = _build_conv_matrix(a).toarray()
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)

        # maximise Σ b_i  → minimise -Σ b_i
        c = -np.ones(n, dtype=np.float64)
        bounds = [(0, None)] * n

        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'tol': 1e-9})

        if not res.success:
            print(f"[LP] iteration {it} failed: {res.message}")
            break

        b = res.x
        sum_b = b.sum()

        # No improvement possible → stop
        if sum_b <= a.sum() + 1e-12:
            break

        # Normalise and convex‑combine
        g = b / sum_b
        a = (1.0 - t) * a + t * g
        # keep unit sum (actually preserved by construction)
        a /= a.sum()

        val = _objective(a, n, _next_pow2(2 * n - 1))
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[LP] it {it:03d} new best {val:.6f}")

        # gradually shrink the step size to stabilise
        t = max(t * 0.94, 0.02)

    return best_a, best_val

# ---------- Adam + sub‑gradient optimiser (large grid) ------------
def _adam_smooth_max(a0: np.ndarray,
                     n: int,
                     conv_fft_len: int,
                     corr_fft_len: int,
                     deadline: float,
                     global_best_seq: np.ndarray | None,
                     global_best_val: float) -> tuple[np.ndarray, float]:
    eps = 1e-15
    logits = np.log(a0 + eps)               # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.07
    lam = 0.4
    lam_factor = 1.02
    lam_cap = 1e6

    best_seq = global_best_seq
    best_val = global_best_val
    last_improve = time.time()
    it = 0
    max_iter = 300_000

    while time.time() < deadline and it < max_iter:
        # soft‑max → probability vector
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # surrogate objective (smooth max)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Adam] it={it:06d} new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature schedule
        lam = min(lam * lam_factor, lam_cap)

        # occasional jitter
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1

    return best_seq, best_val


def _subgradient_refinement(a0: np.ndarray,
                            n: int,
                            conv_fft_len: int,
                            corr_fft_len: int,
                            deadline: float,
                            init_best_seq: np.ndarray | None,
                            init_best_val: float) -> tuple[np.ndarray, float]:
    a = init_best_seq if init_best_seq is not None else a0.copy()
    a = np.maximum(a, 0.0)
    a /= a.sum()

    best_seq = a.copy()
    best_val = init_best_val
    eta0 = 0.6
    it = 0
    max_iter = 30_000
    last_improve = time.time()

    while time.time() < deadline and it < max_iter:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Subgrad] it={it:05d} new best {val:.6f}")

        # active set where the max is attained
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        if time.time() - last_improve > 6.0:
            break
        it += 1

    return best_seq, best_val


def _deep_optimize(a0: np.ndarray,
                   n: int,
                   conv_fft_len: int,
                   corr_fft_len: int,
                   deadline: float,
                   global_best_seq: np.ndarray | None,
                   global_best_val: float) -> tuple[np.ndarray, float]:
    """Adam (smooth‑max) followed by exact sub‑gradient refinement."""
    best_seq, best_val = _adam_smooth_max(a0, n, conv_fft_len,
                                          corr_fft_len, deadline,
                                          global_best_seq, global_best_val)
    best_seq, best_val = _subgradient_refinement(a0, n, conv_fft_len,
                                                 corr_fft_len, deadline,
                                                 best_seq, best_val)
    return best_seq, best_val

    cand.append(np.full(n, 1.0 / n))

    # arcsine‑type
    shape = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    shape /= shape.sum()
    cand.append(shape)

    # symmetric beta‑shapes
    for alpha in (0.2, 0.5, 1.0, 2.0):
        shape = (i + 1.0) ** (alpha - 1.0) * (n - i) ** (alpha - 1.0)
        shape = np.maximum(shape, 0.0)
        shape /= shape.sum()
        cand.append(shape)

    # double‑exponential edges
    for alpha in (0.01, 0.05, 0.2, 0.5):
        shape = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
        shape /= shape.sum()
        cand.append(shape)

    # power‑law edges
    for p in (0.5, 1.0, 2.0, 3.0):
        shape = (i + 1.0) ** (-p) + (n - i) ** (-p)
        shape /= shape.sum()
        cand.append(shape)

    # random Dirichlet draws (various concentrations)
    for conc in (0.01, 0.05, 0.1, 0.5, 1.0):
        shape = np.random.dirichlet(np.full(n, conc))
        shape /= shape.sum()
        cand.append(shape)

    if warm_start is not None:
        # align warm‑start to the small grid by interpolation
        x_old = np.linspace(0, 1, warm_start.shape[0], endpoint=False)
        x_new = np.linspace(0, 1, n, endpoint=False)
        ws = np.interp(x_new, x_old, warm_start)
        ws = np.maximum(ws, 0.0)
        ws /= ws.sum()
        cand.append(ws)

    return cand


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Entry point – search for a non‑negative step function that minimises
    ``evaluate_sequence`` within the allotted time budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Small grid initialisation
    n_small = 256
    conv_fft_len_small = _next_pow2(2 * n_small - 1)

    # optional warm‑start from a previously found solution
    warm_start = globals().get("height_sequence_1")
    ws_small = None
    if warm_start is not None:
        ws_arr = np.asarray(warm_start, dtype=np.float64)
        if ws_arr.ndim == 1:
            ws_small = ws_arr

    # generate candidates
    candidates = _small_candidates(n_small, ws_small)

    best_seq_small = None
    best_val_small = float("inf")

    # quick evaluation on the small grid
    for idx, cand in enumerate(candidates):
        if time.time() >= deadline:
            break
        a = np.maximum(cand, 0.0)
        s = a.sum()
        if s < 0.01:
            a = np.random.dirichlet(np.ones(n_small))
            s = a.sum()
        a /= s
        val = _objective(a, n_small, conv_fft_len_small)
        if val < best_val_small:
            best_val_small = val
            best_seq_small = a.copy()
        # occasional progress report
        # print(f"[small candidate {idx}] val={val:.6f}")

    print(f"Small‑grid best after seeding: {best_val_small:.6f}")

    # ------------------------------------------------------------------
    # LP‑based refinement on the small grid
    if best_seq_small is not None:
        best_seq_small, best_val_small = _lp_iterative_refine(
            best_seq_small, n_small, deadline, t=0.2, max_iter=150
        )
        print(f"After LP refinement best value: {best_val_small:.6f}")

    # ------------------------------------------------------------------
    # Upscale to a larger grid (the final resolution)
    n_big = 4096
    if time.time() < deadline - 30.0:
        # linear interpolation from the refined small vector
        x_small = np.linspace(0, 1, n_small, endpoint=False)
        x_big = np.linspace(0, 1, n_big, endpoint=False)
        a_big = np.interp(x_big, x_small, best_seq_small)
        a_big = np.maximum(a_big, 0.0)
        a_big /= a_big.sum()
    else:
        a_big = np.random.dirichlet(np.ones(n_big))

    conv_fft_len_big = _next_pow2(2 * n_big - 1)
    corr_fft_len_big = _next_pow2(3 * n_big - 2)

    # ------------------------------------------------------------------
    # Deep continuous optimisation (Adam + sub‑gradient)
    best_seq, best_val = _deep_optimize(
        a_big,
        n_big,
        conv_fft_len_big,
        corr_fft_len_big,
        deadline,
        global_best_seq=None,
        global_best_val=float("inf"),
    )
    print(f"After deep optimisation best value: {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final tiny random mass‑transfer hill‑climb
    if best_seq is not None and time.time() < deadline - 5.0:
        a = best_seq.copy()
        best_local_val = best_val
        attempts = 0
        max_attempts = 8000
        n = n_big
        while time.time() < deadline and attempts < max_attempts:
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            max_transfer = min(a[i], a[j])
            if max_transfer < 1e-12:
                continue
            delta = (np.random.rand() - 0.5) * 0.5 * max_transfer
            # enforce positivity
            if delta > a[j]:
                delta = a[j]
            if -delta > a[i]:
                delta = -a[i]
            if abs(delta) < 1e-14:
                continue
            a[i] += delta
            a[j] -= delta

            conv = _full_convolution(a, conv_fft_len_big)
            max_c = conv.max()
            val = 2.0 * n * max_c / (a.sum() ** 2)

            if val < best_local_val:
                best_local_val = val
                best_seq = a.copy()
            else:
                # revert move
                a[i] -= delta
                a[j] += delta
            attempts += 1

        best_val = best_local_val
        print(f"Final local refinement finished: best={best_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback safety
    if best_seq is None:
        best_seq = np.full(n_big, 1.0 / n_big, dtype=np.float64)

    # Clip to evaluator limits
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"*** Search complete: best objective ≈ {best_val:.6f} (n={len(best_seq)})")
    return best_seq.tolist()
