"""
Propose a low‑evaluation step function using active‑set linear programming.

The algorithm works in three phases:

1. **Warm‑start** – take the best known sequence (`height_sequence_1`) if
   available, otherwise start from a uniform distribution.  The vector is
   normalised to sum = 1 (the evaluation depends only on the ratio
   max conv / sum²).

2. **Active‑set LP refinement** – repeatedly
   * compute the full convolution `c = a * a`,
   * pick the `K` largest entries of `c` (the “active” constraints where the
     maximum is attained),
   * solve a tiny linear program  
        maximise   1ᵀ b   
        subject to f * b ≤ max(c) on the active indices,  b ≥ 0,
     obtaining a new candidate `b`,
   * move a small fraction `t` towards the normalised `b` (exponential‑mix)
     and keep the step only if the objective does not increase.
   The LP is tiny (≈ 40 × n) and is solved with SciPy’s `highs` solver,
   allowing thousands of cheap iterations.

3. **Optional up‑sampling & final hill‑climb** – if time remains, the best
   vector is interpolated to a finer grid (n = 8192) and the LP refinement
   is repeated, followed by a brief random‑mass‑transfer hill climb.

The whole search respects the 1000 s deadline and always returns a
non‑negative list of floats suitable for `evaluate_sequence`.
"""

import time
import numpy as np
from scipy.optimize import linprog


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using a real FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Fast evaluation of 2·n·max(a*a) / (sum(a)**2)."""
    conv = _full_convolution(a, conv_fft_len)
    return float(2.0 * n * conv.max() / (a.sum() ** 2))


def _build_constraint_matrix(f: np.ndarray, active_idxs: np.ndarray) -> np.ndarray:
    """
    Build the matrix A (|active| × n) such that
        (A @ b)[k] = (f * b)[active_idxs[k]].
    Row k contains the coefficients f[i] where i + j = active_idxs[k].
    """
    n = f.shape[0]
    K = active_idxs.shape[0]
    A = np.zeros((K, n), dtype=np.float64)

    for row, k in enumerate(active_idxs):
        # j runs over indices of b that actually appear in (f*b)[k]
        j_start = max(0, k - (n - 1))
        j_end = min(k, n - 1)
        # f-index i = k - j
        js = np.arange(j_start, j_end + 1, dtype=np.int64)
        is_ = k - js
        A[row, js] = f[is_]
    return A


def _lp_step(f: np.ndarray, M: float, active_idxs: np.ndarray) -> np.ndarray | None:
    """
    Solve the tiny LP:
        maximise   1ᵀ·b
        subject to A·b ≤ M,  b ≥ 0,
    where A is the convolution matrix restricted to ``active_idxs``.
    Returns the optimal vector `b` or ``None`` on failure.
    """
    A = _build_constraint_matrix(f, active_idxs)
    m = A.shape[0]
    c = -np.ones(f.shape[0], dtype=np.float64)       # maximise sum(b)
    b_ub = np.full(m, M, dtype=np.float64)
    bounds = [(0.0, None) for _ in range(f.shape[0])]

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method="highs")
    if res.success:
        return res.x
    return None


def _lp_refinement(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_iters: int = 2000,
    active_k: int = 40,
) -> tuple[np.ndarray, float]:
    """
    Perform many small LP‑driven updates starting from ``a0``.
    The iteration stops after ``max_iters`` steps or when the time budget
    expires.  Returns the best vector and its objective value.
    """
    a = a0.copy()
    # normalise to sum = 1 (the objective is scale‑invariant)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s < 1e-12:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = a / s

    best_seq = a.copy()
    best_val = _objective(a, n, conv_fft_len)
    cur_val = best_val

    it = 0
    t0 = 0.18                     # initial mixing weight
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        M = conv.max()

        # pick up to ``active_k`` indices with the largest convolution values
        if conv.size <= active_k:
            active_idxs = np.arange(conv.size, dtype=np.int64)
        else:
            active_idxs = np.argpartition(-conv, active_k - 1)[:active_k]

        b = _lp_step(a, M, active_idxs)
        if b is None:
            # fallback: tiny random jitter and continue
            i, j = np.random.randint(n, size=2)
            eps = (np.random.rand() - 0.5) * 1e-4
            eps = np.clip(eps, -a[i], a[j])
            a[i] -= eps
            a[j] += eps
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            continue

        # normalise b to sum = 1 (the same scale as a)
        b_sum = b.sum()
        if b_sum < 1e-12:
            break
        b = b / b_sum

        # decaying mixing weight
        t = t0 * (0.96 ** it)
        t = max(t, 0.001)

        a_new = (1.0 - t) * a + t * b
        a_new = np.clip(a_new, 0.0, None)
        a_new /= a_new.sum()

        new_val = _objective(a_new, n, conv_fft_len)

        # keep monotone (or accept tiny uphill moves with tiny probability)
        if new_val <= cur_val * (1.0 + 1e-12):
            a = a_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = a.copy()
                print(f"[LP] it={it:05d} new best {new_val:.6f}")
        else:
            # occasional stochastic acceptance to escape plateaus
            if np.random.rand() < 0.02:
                a = a_new
                cur_val = new_val

        # occasional random shaking to avoid stagnation
        if it % 50 == 0 and np.random.rand() < 0.25:
            i, j = np.random.randint(n, size=2)
            eps = (np.random.rand() - 0.5) * 1e-4
            eps = np.clip(eps, -a[i], a[j])
            a[i] -= eps
            a[j] += eps
            a = np.clip(a, 0.0, None)
            a /= a.sum()

        it += 1

    return best_seq, best_val


def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """Search for a step‑function minimising ``evaluate_sequence``."""
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # base size – start with 4096 (fast FFT) and enlarge if time permits
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # initialise from a previous good construction if available
    warm = globals().get("height_sequence_1")
    if warm is not None:
        a0 = np.asarray(warm, dtype=np.float64)
        if a0.shape[0] > n:
            a0 = a0[:n]
        elif a0.shape[0] < n:
            a0 = np.pad(a0, (0, n - a0.shape[0]))
        a0 = np.clip(a0, 0.0, None)
        if a0.sum() < 0.01:
            a0 = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a0 = a0 / a0.sum()
    else:
        a0 = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = a0.copy()
    best_val = _objective(best_seq, n, conv_fft_len)
    print(f"[init] objective ≈ {best_val:.6f}")

    # ------------------------------------------------------------------
    # main LP‑driven refinement
    best_seq, best_val = _lp_refinement(
        best_seq,
        n,
        conv_fft_len,
        deadline,
        max_iters=3000,
        active_k=40,
    )
    print(f"[LP‑refine] best after LP ≈ {best_val:.6f}")

    # ------------------------------------------------------------------
    # try a few random restarts if time permits
    while time.time() < deadline - 5.0:
        conc = np.random.uniform(0.05, 3.0)
        start = np.random.dirichlet(np.full(n, conc))
        start = start / start.sum()
        seq, val = _lp_refinement(start, n, conv_fft_len, deadline,
                                   max_iters=1500, active_k=35)
        if val < best_val:
            best_val, best_seq = val, seq
            print(f"[restart] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # optional upscale to 8192 points if we still have time
    if time.time() < deadline - 30.0:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)

        # linear interpolation of the current best sequence
        x_old = np.linspace(0, 1, n, endpoint=False)
        x_new = np.linspace(0, 1, n_big, endpoint=False)
        a_big = np.interp(x_new, x_old, best_seq)
        a_big = np.clip(a_big, 0.0, None)
        a_big /= a_big.sum()

        print("[upscale] starting LP refinement on larger grid")
        a_big, val_big = _lp_refinement(
            a_big,
            n_big,
            conv_fft_len_big,
            deadline,
            max_iters=2500,
            active_k=45,
        )
        if val_big < best_val:
            best_val, best_seq = val_big, a_big
            n, conv_fft_len = n_big, conv_fft_len_big
            print(f"[upscale] new best ≈ {best_val:.6f}")

    # ------------------------------------------------------------------
    # final short random hill‑climb (tiny mass transfers)
    if time.time() < deadline - 2.0:
        a = best_seq.copy()
        cur_val = best_val
        it = 0
        max_hc_iters = 8000
        while time.time() < deadline and it < max_hc_iters:
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            # propose moving up to 20 % of the smaller bucket
            max_transfer = min(a[i], a[j])
            if max_transfer < 1e-12:
                continue
            delta = (np.random.rand() - 0.5) * 0.4 * max_transfer
            delta = np.clip(delta, -a[i], a[j])
            a[i] += delta
            a[j] -= delta
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            val = _objective(a, n, conv_fft_len)
            if val < cur_val:
                cur_val = val
                best_seq = a.copy()
                best_val = val
            it += 1
        print(f"[hill‑climb] finished after {it} attempts, best≈{best_val:.6f}")

    # ------------------------------------------------------------------
    # final safety clipping
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    if best_seq.sum() < 0.01:
        best_seq = np.full(best_seq.shape, 1.0 / best_seq.shape[0])

    print(f"*** Search complete: best objective ≈ {best_val:.6f} (n={best_seq.shape[0]})")
    return best_seq.tolist()
