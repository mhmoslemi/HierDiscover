"""Fast soft‑max gradient descent for flattening a self‑convolution.

The algorithm keeps the vector on the probability simplex, computes a
soft‑max approximation of the convolution peak and updates with an
exponentiated‑gradient (mirror‑descent) step.  A tiny β‑schedule starts
with a smooth objective (β≈0.5) and gradually sharpens it (β→30) so that
the optimisation first finds a flat region and only later focuses on the
true max.  Several parametric seeds (uniform, Dirichlet, edge‑exponential,
edge‑power‑law, Gaussian‑edges, sinusoid, beta‑distribution) are
generated, the best few are refined, and, if time permits, the best
solution is interpolated to a finer grid (n=8192) and refined again.

The function ``propose_candidate`` respects the 1000 s budget, prints
progress and returns the best sequence as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper FFT utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) using real FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n-1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


# ----------------------------------------------------------------------
# Parametric seed generation
# ----------------------------------------------------------------------


def _generate_parametric_candidates(n: int, warm_start: np.ndarray | None) -> list[np.ndarray]:
    """Create a diverse pool of initial probability vectors."""
    candidates: list[np.ndarray] = []

    # Warm‑start and a few simple rotations
    if warm_start is not None:
        candidates.append(warm_start.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            candidates.append(np.roll(warm_start, shift))

    # Uniform baseline
    candidates.append(np.full(n, 1.0 / n))

    # Random Dirichlet draws with several concentration levels
    for conc in (0.01, 0.05, 0.1, 0.3, 0.5, 1.0, 2.0):
        candidates.append(np.random.dirichlet(np.full(n, conc)))

    i = np.arange(n, dtype=np.float64)

    # Edge‑exponential family
    for alpha in np.logspace(-2, 1.0, num=12):
        edge = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
        edge /= edge.sum()
        for w in (0.0, 0.25, 0.5, 0.75, 1.0):
            candidates.append(w * edge + (1.0 - w) / n)

    # Edge‑power‑law family
    for p in np.linspace(0.2, 5.0, num=12):
        edge = (i + 1.0) ** (-p) + (n - i) ** (-p)
        edge /= edge.sum()
        for w in (0.0, 0.33, 0.66, 1.0):
            candidates.append(w * edge + (1.0 - w) / n)

    # Double‑Gaussian edges
    for gamma in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        shape /= shape.sum()
        candidates.append(shape)

    # Positive half‑wave sinusoid
    sinusoid = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    sinusoid = np.maximum(sinusoid, 0.0)
    sinusoid /= sinusoid.sum()
    candidates.append(sinusoid)

    # Beta‑distribution family (α,β ∈ [0.2, 5])
    xs = (i + 0.5) / n
    for a in np.geomspace(0.2, 5.0, num=8):
        for b in np.geomspace(0.2, 5.0, num=8):
            shape = (xs ** (a - 1.0)) * ((1.0 - xs) ** (b - 1.0))
            if np.all(shape == 0):
                continue
            shape /= shape.sum()
            candidates.append(shape)

    return candidates


# ----------------------------------------------------------------------
# Core optimisation routine (soft‑max mirror descent)
# ----------------------------------------------------------------------


def _softmax_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
    *,
    beta_start: float = 0.5,
    beta_end: float = 30.0,
    eta_start: float = 0.5,
    eta_min: float = 0.001,
    decay: float = 0.99995,
    patience: float = 5.0,
) -> tuple[np.ndarray | None, float]:
    """
    Soft‑max mirror‑descent on the true convolution peak.

    * a0            – initial probability vector (already normalised)
    * beta_start    – initial smoothing factor for the max
    * beta_end      – final smoothing factor (sharper max)
    * eta_start, decay – step‑size schedule
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 0.01:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    beta = beta_start
    eta = eta_start
    last_improve = time.time()
    it = 0

    # Linear schedule for β based on remaining iterations
    # The number of iterations is not known a‑priori, so we simply
    # increase β by a tiny amount each step (≈ (beta_end‑beta_start)/5e5).
    beta_increment = (beta_end - beta_start) / 500_000.0

    while time.time() < deadline:
        # ---- convolution and current value ---------------------------------
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c   # sum(a)=1 ⇒ objective = 2·n·max_c

        if val < best_val - 1e-12:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[n={n} iter={it:07d}] new best {val:.6f}")

        # ---- soft‑max weights on the convolution ---------------------------
        # shift to avoid overflow; conv_shift ≤ 0
        conv_shift = conv - max_c
        w_raw = np.exp(beta * conv_shift)
        w_sum = w_raw.sum()
        if w_sum == 0.0 or np.isnan(w_sum):
            # fallback to uniform weight on active maximum indices
            active = np.isclose(conv, max_c, atol=1e-12)
            w = np.zeros_like(conv)
            w[active] = 1.0 / max(1, active.sum())
        else:
            w = w_raw / w_sum

        # ---- gradient of the smoothed max ----------------------------------
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # ---- exponentiated‑gradient update ---------------------------------
        eta = max(eta_min, eta * decay)
        a = a * np.exp(-eta * grad)

        # re‑normalise onto the simplex
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        # ---- β schedule ----------------------------------------------------
        if beta < beta_end:
            beta = min(beta_end, beta + beta_increment)

        # ---- occasional random shake if stuck -------------------------------
        if time.time() - last_improve > patience:
            perturb = (np.random.rand(n) - 0.5) * 0.02   # small log‑space jitter
            a = a * np.exp(perturb)
            a = a / a.sum()
            last_improve = time.time()

        it += 1

    return best_seq, best_val

        print("--- Refinement on larger grid (n=8192)")
        best_seq, best_val = _softmax_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            best_seq,
            best_val,
        )
        if best_seq is not None and best_seq.shape[0] == n_big:
            n = n_big  # final length is the larger one

    # ------------------------------------------------------------------
    # Final safety net
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Clamp to the evaluator’s allowed range
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Store for possible warm‑starts in a future call
    globals()["height_sequence_1"] = best_seq.tolist()

    print(f"*** Finished search: best objective ≈ {best_val:.6f} (n={best_seq.shape[0]})")
    return best_seq.tolist()
