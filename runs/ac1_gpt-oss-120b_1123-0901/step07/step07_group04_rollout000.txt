Implement projection onto simplex (standard algorithm):

def _project_to_simplex(v):
    # ensure v sum to 1 and v>=0
    # Implementation as described in Duchi et al.

    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a low‑objective step‑function (non‑negative vector) using a
smoothed‑max gradient descent (Adam) on the self‑convolution.
The algorithm:

1. Generate a diverse pool of candidate shapes (uniform, Dirichlet,
   edge‑exponential/power‑law, double‑Gaussian, sinusoid, Beta‑type,
   and the analytic arcsine law).  All candidates are normalised to sum = 1.

2. Evaluate each candidate quickly with an FFT‑based exact convolution and
   keep the best few (default 6).

3. Refine each of the selected starts with a smooth approximation of the
   max‑convolution:
      smooth_max_β(conv) = (1/β)·log∑exp(β·conv),
   whose gradient is 2·cross_correlation(w, a) with w the soft‑max
   weights.  Adam updates are applied and the iterate is projected onto the
   probability simplex after each step.

   The smoothing parameter β starts small (≈0.5) and is multiplied by a
   factor 1.001 each iteration until a cap (≈30).  The learning‑rate
   decays slowly.  A small random jitter is injected if no improvement
   occurs for a few seconds to escape plateaus.

4. If time permits, the best vector is interpolated onto a finer grid
   (n = 8192) and refined again with the same optimiser.

The routine respects the 1000 s budget, prints progress, and returns the
best sequence found as a plain Python list (values clipped to the range
accepted by ``evaluate_sequence``).  The implementation uses only NumPy
and the provided ``evaluate_sequence`` for final scoring.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (no closures, no lambdas)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n-1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact objective for a (assumed normalised to sum≈1)."""
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c / (a.sum() ** 2))


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex {x≥0, Σx=1}.
    Implementation of Duchi et al. (2008).
    """
    if v.ndim != 1:
        raise ValueError("Input must be a 1‑D array")
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # find the largest rho where u_i - (cssv_i / i) > 0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero or negative -> return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    # numeric safety: renormalise
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(v, 1.0 / n)
    else:
        w = w / w_sum
    return w


def _generate_parametric_candidates(n: int, warm_start: np.ndarray | None) -> list[np.ndarray]:
    """Create a diverse set of candidate vectors."""
    candidates = []

    # warm‑start and shifted copies
    if warm_start is not None:
        candidates.append(warm_start.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            candidates.append(np.roll(warm_start, shift))

    # uniform
    candidates.append(np.full(n, 1.0 / n))

    rng = np.random.default_rng()

    # random Dirichlet (various concentrations)
    for conc in (0.01, 0.05, 0.1, 0.3, 0.5, 1.0, 2.0):
        cand = rng.dirichlet(np.full(n, conc))
        candidates.append(cand)

    # edge‑exponential
    i = np.arange(n, dtype=np.float64)
    for alpha in np.logspace(-2, 1.0, num=12):
        edge = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
        edge /= edge.sum()
        for w in (0.0, 0.25, 0.5, 0.75, 1.0):
            shape = w * edge + (1.0 - w) / n
            candidates.append(shape)

    # edge‑power‑law
    for p in np.linspace(0.2, 5.0, num=12):
        edge = (i + 1.0) ** (-p) + (n - i) ** (-p)
        edge /= edge.sum()
        for w in (0.0, 0.33, 0.66, 1.0):
            shape = w * edge + (1.0 - w) / n
            candidates.append(shape)

    # double‑Gaussian edges
    for gamma in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        shape /= shape.sum()
        candidates.append(shape)

    # positive half‑wave sinusoid
    sinus = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    sinus = np.maximum(sinus, 0.0)
    sinus /= sinus.sum()
    candidates.append(sinus)

    # Beta‑type families (including arcsine = Beta(0.5,0.5))
    for a_b in [(0.5, 0.5), (0.3, 0.3), (1.0, 1.0), (2.0, 2.0),
                (0.5, 2.0), (2.0, 0.5), (0.2, 0.2), (0.8, 0.8)]:
        a_, b_ = a_b
        x = (np.arange(n, dtype=np.float64) + 0.5) / n
        shape = (x ** (a_ - 1)) * ((1 - x) ** (b_ - 1))
        shape_sum = shape.sum()
        if shape_sum > 0:
            shape = shape / shape_sum
            candidates.append(shape)

    # Pure random Dirichlet samples (extra)
    for _ in range(30):
        cand = rng.dirichlet(np.ones(n))
        candidates.append(cand)

    return candidates


def _smooth_opt(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray | None, float]:
    """
    Adam optimiser on the smoothed maximum of the self‑convolution.
    Returns the best vector/value observed during its run.
    """
    # initialise on the simplex
    a = np.maximum(a0, 0.0)
    if a.sum() < 0.01:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.05

    # smoothing parameter for the soft‑max (starts low, grows)
    beta = 0.5
    beta_max = 30.0
    beta_inc = 1.001

    it = 0
    last_improve = time.time()
    max_iter = 5_000_000  # stop by deadline

    while time.time() < deadline and it < max_iter:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        # sum(a) is 1 by construction, but keep formula for safety
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val - 1e-12:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[n={n} smooth iter={it:07d}] new best {val:.6f}")

        # soft‑max weights over convolution entries
        # stable computation: subtract max before exponentiation
        exp_vals = np.exp(beta * (conv - max_c))
        w = exp_vals / exp_vals.sum()

        # gradient of the smoothed max (2 * cross‑correlation)
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # Adam update (gradient descent)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # project back onto the probability simplex
        a = _project_to_simplex(a)

        # schedule updates
        beta = min(beta_max, beta * beta_inc)
        lr = max(1e-4, lr * 0.99995)

        # occasional random jitter if we have been idle
        if time.time() - last_improve > 4.0:
            jitter = (np.random.rand(n) - 0.5) * 0.01  # ±0.5 % in log‑space
            a = a * np.exp(jitter)
            a = _project_to_simplex(a)
            last_improve = time.time()

        it += 1

    return best_seq, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step function with minimal evaluation score.
    Returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # safety margin

    # --------------------------------------------------------------
    # Primary grid size – 4096 (good compromise)
    # --------------------------------------------------------------
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Warm start from a previous global variable if present
    # --------------------------------------------------------------
    warm_start = globals().get("height_sequence_1")
    if warm_start is not None:
        warm_start = np.asarray(warm_start, dtype=np.float64)
        if warm_start.shape[0] > n:
            warm_start = warm_start[:n]
        elif warm_start.shape[0] < n:
            warm_start = np.pad(warm_start, (0, n - warm_start.shape[0]))
        warm_start = np.maximum(warm_start, 0.0)
        if warm_start.sum() >= 0.01:
            warm_start = warm_start / warm_start.sum()
        else:
            warm_start = None
    else:
        warm_start = None

    # --------------------------------------------------------------
    # Stage 1 – generate a broad set of initial candidates
    # --------------------------------------------------------------
    candidates = _generate_parametric_candidates(n, warm_start)

    # --------------------------------------------------------------
    # Fast evaluation of all candidates, keep a few best
    # --------------------------------------------------------------
    cand_vals = []
    for cand in candidates:
        if time.time() >= deadline:
            break
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 0.01:
            cand = np.random.dirichlet(np.ones(n))
            s = cand.sum()
        cand = cand / s
        val = _objective(cand, n, conv_fft_len)
        cand_vals.append((val, cand))

    if not cand_vals:
        # fallback – uniform vector
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        print("*** No candidates evaluated – returning uniform vector.")
        return best_seq.tolist()

    cand_vals.sort(key=lambda x: x[0])
    top_k = 6
    top_candidates = [c for _, c in cand_vals[:top_k]]

    best_seq = None
    best_val = float("inf")

    # --------------------------------------------------------------
    # Stage 2 – deep optimisation of selected starts (smooth‑max Adam)
    # --------------------------------------------------------------
    for idx, a0 in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        init_val = _objective(a0, n, conv_fft_len)
        print(
            f"--- Optimising from start #{idx + 1}/{len(top_candidates)} "
            f"(init val={init_val:.6f})"
        )
        best_seq, best_val = _smooth_opt(
            a0,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            best_seq,
            best_val,
        )

    # --------------------------------------------------------------
    # Stage 3 – optional upscale to a finer grid if time permits
    # --------------------------------------------------------------
    time_left = deadline - time.time()
    if time_left > 30.0 and best_seq is not None:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        # linear interpolation onto the finer grid
        x_old = np.linspace(0, 1, n, endpoint=False)
        x_big = np.linspace(0, 1, n_big, endpoint=False)
        a_big = np.interp(x_big, x_old, best_seq)
        a_big = np.maximum(a_big, 0.0)
        a_big /= a_big.sum()

        print("--- Refinement on larger grid (n=8192)")
        best_seq, best_val = _smooth_opt(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            best_seq,
            best_val,
        )
        if best_seq is not None and best_seq.shape[0] == n_big:
            n = n_big

    # --------------------------------------------------------------
    # Final sanity clipping & return
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"*** Finished search: best objective ≈ {best_val:.6f} (n={best_seq.shape[0]})")
    return best_seq.tolist()
