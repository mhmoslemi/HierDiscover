"""
Propose a non‑negative step‑function (vector) that minimizes

        F(a) = 2·n·max(a∗a) / (sum(a)^2)

The routine follows a multi‑phase hybrid optimisation:

1.  Build a diverse pool of warm‑started candidates (uniform, Dirichlet,
    Gaussian bells, double‑exponential, power‑law edges, sinusoid, spikes).
2.  Run a short Adam optimisation on a smooth “log‑sum‑exp’’ surrogate of
    the max‑convolution for each candidate; keep the best.
3.  Refine the best candidate with a *long* Adam run (temperature is
    increased until the surrogate equals the true max) and a long
    multiplicative‑gradient (sub‑gradient) phase.
4.  Perform a cheap hill‑climbing random‑move phase.
5.  If time permits, interpolate the solution to a finer grid (8192 or
    16384 points) and repeat steps 3–4 on the larger problem.
All phases respect the global 1000 s budget and the best sequence found
is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via FFT (length 2·len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for a sub‑gradient.
    Returns length‑len(a) vector:
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i = 0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
def _initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Generate a modest set of diverse start vectors."""
    cand = []

    # warm start & rotations
    if warm is not None:
        cand.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    # random Dirichlet
    cand.append(np.random.dirichlet(np.ones(n)))

    # Gaussian bells of varying width
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    for sigma_rel in (0.02, 0.04, 0.07, 0.12, 0.20):
        sigma = sigma_rel * n
        shape = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    # edge‑concentrated power‑law shapes
    idx = np.arange(n, dtype=np.float64) + 1.0
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-3.0, -0.2, 13):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    # symmetric double‑exponential shapes
    i = np.arange(n, dtype=np.float64)
    for gamma in np.geomspace(0.2, 5.0, 8):
        shape = np.exp(-gamma * i) + np.exp(-gamma * (n - 1 - i))
        shape /= shape.sum()
        cand.append(shape)

    # half‑wave sinusoid
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand.append(shape)

    # uniform
    cand.append(np.full(n, 1.0 / n))

    return cand


def _spike_candidates(n: int) -> list[np.ndarray]:
    """A few “spike’’ candidates with k equally spaced non‑zero entries."""
    out = []
    for k in (32, 64, 128, 256, 512):
        a = np.zeros(n, dtype=np.float64)
        if k > 0:
            pos = np.round(np.linspace(0, n - 1, k)).astype(int)
            a[pos] = 1.0 / k
        out.append(a)
    return out

def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max (log‑sum‑exp) surrogate.
    Returns the best vector (∑a≈1) encountered and its exact objective.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)          # re‑parameterise via soft‑max
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.09                         # a slightly larger learning rate
    lam = 0.5                         # start temperature
    lam_factor = 1.02                 # gradual cooling → true max

    best_a = a0.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max → simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective (sum(a)=1)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam iter={it:07d}] new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature schedule → true max
        lam = min(lam * lam_factor, 1e7)

        # occasional tiny random shake to avoid premature stalling
        if it % 6_000 == 0 and time.time() + 5 > deadline:
            logits += (np.random.rand(n) - 0.5) * 3e-3

        it += 1

    return best_a, best_val


def _subgradient_opt2(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 1_000_000,
    eta0: float = 0.8,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient refinement with decaying step size.
    Returns the best vector seen and its exact objective.
    """
    a = np.maximum(a0, 0.0)
    a_sum = a.sum()
    if a_sum < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        a_sum = a.sum()
    a = a / a_sum

    best_a = a.copy()
    best_val = float("inf")
    stall = 0
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            stall = 0
            if it % 5_000 == 0:
                print(f"[Subgrad iter={it:07d}] new best {val:.6f}")
        else:
            stall += 1

        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # decaying step size
        eta = eta0 / np.sqrt(it + 1)

        # multiplicative update + renormalisation
        a = a * np.exp(-eta * grad_a)
        a_sum = a.sum()
        if a_sum < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            a_sum = a.sum()
        a /= a_sum

        # occasional random perturbation if stuck
        if stall > 12_000:
            a = np.maximum(a + 0.02 * (np.random.rand(n) - 0.5), 0.0)
            a /= a.sum()
            stall = 0

        it += 1

    return best_a, best_val


def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climbing: move a random fraction of mass from one index
    to another, keep the move if the objective improves.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    best_a = a.copy()
    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-12:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue
        delta = a[i] * (np.random.rand() * 0.4)          # move up to 40 % of source mass
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta
        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()
        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5_000 == 0:
                print(f"[Local move {it}] improved to {val2:.6f}")
        it += 1
    return best_a, best_val

    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Phase 1 – base resolution (n = 4096)
    n = 4096

    corr_fft_len = _next_pow2(3 * n - 2)

    # Optional warm‑start from previous run

        else:
            warm = None

    candidates = _initial_candidates(n, warm) + _spike_candidates(n)

    # ------------------------------------------------------------------
    # Short Adam screening of every candidate
    for cid, cand in enumerate(candidates):
        if time.time() >= deadline - 200:  # keep some slack for later phases
            break
        a0 = np.maximum(cand, 0.0)
        s = a0.sum()
        if s < 0.01:
            a0 = np.random.dirichlet(np.ones(n))
        else:
            a0 = a0 / s

        # allocate a few seconds per candidate
        cand_deadline = min(deadline - 120, time.time() + 8.0)
        a_adam, val_adam = _adam_optimize(
            a0, n, conv_fft_len, corr_fft_len, cand_deadline, max_iters=120_000
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[cand {cid}] Adam best {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – long Adam refinement on the current best
    if best_seq is not None and time.time() < deadline - 120:
        long_adam_deadline = deadline - 100
        a_long, val_long = _adam_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            long_adam_deadline,
            max_iters=800_000,
        )
        if val_long < best_val:
            best_val = val_long
            best_seq = a_long.copy()
            print(f"[Long Adam] best {val_long:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – long sub‑gradient on the best candidate
    if best_seq is not None and time.time() < deadline - 80:
        sub_deadline = deadline - 60
        a_ref, val_ref = _subgradient_opt2(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            sub_deadline,
            max_iters=800_000,
        )
        if val_ref < best_val:
            best_val = val_ref
            best_seq = a_ref.copy()
            print(f"[Subgrad long] best {val_ref:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – cheap random local search
    if best_seq is not None and time.time() < deadline - 30:
        loc_deadline = deadline - 20
        a_loc, val_loc = _local_random_search(
            best_seq, n, conv_fft_len, loc_deadline, max_moves=250_000
        )
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()
            print(f"[Local search] improved best {val_loc:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – upscale to 8192 if enough time remains
    if time.time() + 35 < deadline:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        # linear interpolation to fine grid
        if best_seq is not None:
            x_old = np.linspace(0, 1, n, endpoint=False)
            x_new = np.linspace(0, 1, n_big, endpoint=False)
            a_big = np.interp(x_new, x_old, best_seq)
        else:
            a_big = np.random.dirichlet(np.ones(n_big))

        a_big = np.maximum(a_big, 0.0)
        a_big /= a_big.sum()

        # short Adam on bigger grid
        big_deadline = min(deadline - 20, time.time() + 15.0)
        a_big_adam, val_big_adam = _adam_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            big_deadline,
            max_iters=200_000,
        )
        if val_big_adam < best_val:
            best_val = val_big_adam
            best_seq = a_big_adam.copy()
            print(f"[8192 Adam] best {val_big_adam:.6f}")

        # sub‑gradient refinement at 8192
        if time.time() < deadline - 10:
            a_big_ref, val_big_ref = _subgradient_opt2(
                best_seq,
                n_big,
                conv_fft_len_big,
                corr_fft_len_big,
                deadline,
                max_iters=400_000,
            )
            if val_big_ref < best_val:
                best_val = val_big_ref
                best_seq = a_big_ref.copy()
                print(f"[8192 Subgrad] best {val_big_ref:.6f}")

        # cheap local search at 8192
        if time.time() < deadline - 5:
            a_loc2, val_loc2 = _local_random_search(
                best_seq, n_big, conv_fft_len_big, deadline, max_moves=200_000
            )
            if val_loc2 < best_val:
                best_val = val_loc2
                best_seq = a_loc2.copy()
                print(f"[8192 Local] improved best {val_loc2:.6f}")

    # ------------------------------------------------------------------
    # Phase 6 – optional upscale to 16384 if any time left
    if time.time() + 30 < deadline:
        n_big = 16384
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        x_old = np.linspace(0, 1, best_seq.shape[0], endpoint=False)
        x_new = np.linspace(0, 1, n_big, endpoint=False)
        a_big = np.interp(x_new, x_old, best_seq)

        a_big = np.maximum(a_big, 0.0)
        a_big /= a_big.sum()

        # brief Adam on 16384
        big_deadline = min(deadline - 10, time.time() + 12.0)
        a_big_adam, val_big_adam = _adam_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            big_deadline,
            max_iters=120_000,
        )
        if val_big_adam < best_val:
            best_val = val_big_adam
            best_seq = a_big_adam.copy()
            print(f"[16384 Adam] best {val_big_adam:.6f}")

        # final sub‑gradient on 16384
        if time.time() < deadline:
            a_big_ref, val_big_ref = _subgradient_opt2(
                best_seq,
                n_big,
                conv_fft_len_big,
                corr_fft_len_big,
                deadline,
                max_iters=200_000,
            )
            if val_big_ref < best_val:
                best_val = val_big_ref
                best_seq = a_big_ref.copy()
                print(f"[16384 Subgrad] best {val_big_ref:.6f}")

    # ------------------------------------------------------------------
    # Fallback – guarantee a valid return value
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Clip to evaluator’s safety limits
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Store for possible warm‑starts in a subsequent run
    globals()["height_sequence_1"] = best_seq.copy()

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
