"""
Hybrid gradient‑based & stochastic search for a non‑negative step function
that minimises the black‑box objective

    F(a) = 2·n·max(a∗a) / (sum(a)²),

where * denotes linear convolution.  The algorithm proceeds in four stages:

1️⃣  Generate a large, diverse set of parametric candidates (exponential,
    power‑law, triangular, sinusoidal, sparse spikes, etc.) and keep the
    best few according to the exact objective.

2️⃣  Deep optimisation of each top candidate:
    – Adam on a log‑sum‑exp surrogate of the max‑convolution (smooth‑max).
    – Exponentiated‑gradient sub‑gradient descent on the exact max.
    The optimisation for each start respects a time‑budget that is split
    among the remaining starts, so the whole search fits into the 1000 s
    deadline.

3️⃣  Cheap stochastic hill‑climbing / simulated‑annealing in the log‑space
    of the current globally‑best vector.

4️⃣  If time permits, upscale the best vector to n=8192 (linear
    interpolation) and repeat step 2 on the finer grid – larger
    dimensions usually admit a lower bound.

All vectors are kept on the probability simplex (sum≈1) and are clipped
to the evaluator’s allowed range (0 ≤ value ≤ 1000).  The function
`propose_candidate` returns the best sequence found as a plain list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a length‑n vector where n = len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact objective F = 2·n·max(a*a) / (sum(a)²)."""
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c / (a.sum() ** 2))

def _generate_parametric_candidates(n: int, warm_start: np.ndarray | None):
    """
    Produce a rich pool of candidate vectors.
    Families include uniform, Dirichlet draws, symmetric / asymmetric
    exponentials and power‑laws, linear / triangular, half‑wave sinusoid,
    and sparse spikes.
    """
    candidates = []

    # warm‑start and its circular shifts
    if warm_start is not None:
        candidates.append(warm_start.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            candidates.append(np.roll(warm_start, shift))

    # uniform baseline
    candidates.append(np.full(n, 1.0 / n))

    # Dirichlet draws with a variety of concentrations
    for conc in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(8):
            cand = np.random.dirichlet(np.full(n, conc))
            candidates.append(cand)

    i = np.arange(n, dtype=np.float64)

    # symmetric edge‑exponential families
    for alpha in np.logspace(-2, 1.0, num=12):
        edge = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
        edge /= edge.sum()
        for w in (0.0, 0.25, 0.5, 0.75, 1.0):
            shape = w * edge + (1.0 - w) / n
            candidates.append(shape)

    # symmetric edge‑power‑law families
    for p in np.linspace(0.2, 6.0, num=12):
        edge = (i + 1.0) ** (-p) + (n - i) ** (-p)
        edge /= edge.sum()
        for w in (0.0, 0.3, 0.6, 1.0):
            shape = w * edge + (1.0 - w) / n
            candidates.append(shape)

    # asymmetric two‑sided exponentials
    alphas = np.logspace(-2, 1.0, num=8)
    w_vals = np.linspace(0.0, 1.0, 5)
    for wL in w_vals:
        for wR in w_vals:
            if wL + wR > 1.0:
                continue
            wU = 1.0 - wL - wR
            for aL in alphas:
                for aR in alphas:
                    left = np.exp(-aL * i)
                    right = np.exp(-aR * (n - 1 - i))
                    shape = wL * left + wR * right + wU / n
                    shape /= shape.sum()
                    candidates.append(shape)

    # asymmetric power‑law families
    p_vals = np.linspace(0.2, 5.0, num=6)
    for wL in w_vals:
        for wR in w_vals:
            if wL + wR > 1.0:
                continue
            wU = 1.0 - wL - wR
            for pL in p_vals:
                for pR in p_vals:
                    left = (i + 1.0) ** (-pL)
                    right = (n - i) ** (-pR)
                    shape = wL * left + wR * right + wU / n
                    shape /= shape.sum()
                    candidates.append(shape)

    # simple linear / triangular shapes
    lin = np.linspace(1.0, 0.0, n)
    tri = np.concatenate([np.linspace(0, 1, n // 2, endpoint=False),
                          np.linspace(1, 0, n - n // 2)])
    for base in (lin, tri):
        base = base / base.sum()
        for w in (0.0, 0.4, 0.8):
            cand = w * base + (1.0 - w) / n
            cand /= cand.sum()
            candidates.append(cand)

    # half‑wave sinusoid
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    candidates.append(shape)

    # sparse spike vectors
    for spikes in (3, 5, 8):
        for _ in range(12):
            idx = np.random.choice(n, size=spikes, replace=False)
            weights = np.random.dirichlet(np.ones(spikes))
            vec = np.zeros(n)
            vec[idx] = weights
            vec /= vec.sum()
            candidates.append(vec)

    return candidates

    while time.time() < deadline and it < max_iter:
        # -----------------------------------------------------------------
        # produce a on the simplex via soft‑max
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact convolution and objective (used for logging)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Adam] iter={it:06d} new best {val:.6f}")

        # -----------------------------------------------------------------
        # smooth‑max weights (log‑sum‑exp)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate w.r.t a (via cross‑correlation)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain‑rule for logits (entropy‑mirror map)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature schedule
        lam = min(lam * lam_factor, lam_cap)

        # occasional random shake if stuck
        if time.time() - last_improve > 6.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
def _subgradient_refinement(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    init_best_seq: np.ndarray | None,
    init_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient descent on the *exact* max
    convolution.  The number of iterations is larger than before and the
    step‑size decays more gently to allow fine‑grained adjustments.
    """
    a = init_best_seq if init_best_seq is not None else a0.copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a /= s  # keep on the simplex

    eta0 = 0.9
    decay = 0.99997
    eta = eta0

    best_seq = a.copy()
    best_val = init_best_val
    it = 0
    max_iter = 120_000
    last_improve = time.time()

    while time.time() < deadline and it < max_iter:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[SubGrad] iter={it:06d} new best {val:.6f}")

        # active set where the maximum is attained
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # exponentiated‑gradient update (keeps a ≥ 0)
        a = a * np.exp(-eta * grad_a)

        # renormalise
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        # decay step size
        eta *= decay
        it += 1

        # early stop if no progress for a while
        if time.time() - last_improve > 8.0:
            break

    return best_seq, best_val


# ----------------------------------------------------------------------
def _simulated_annealing(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    init_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing on the log‑space of a.
    Accepts worsening moves with probability exp(‑Δ / T); temperature T
    decays exponentially with the number of iterations.
    """
    a_best = a0.copy()
    best_val = init_best_val
    # temperature schedule
    T_start = 0.5
    T_end = 1e-4
    total_iters_est = 200_000  # rough upper bound for time budgeting
    it = 0
    log_T_decay = math.log(T_end / T_start) / total_iters_est

    logits_best = np.log(a_best + 1e-15)
    cur_val = best_val

    while time.time() < deadline:
        # temperature at this iteration
        T = T_start * math.exp(log_T_decay * it)

        # propose a gaussian perturbation of logits
        noise = np.random.randn(n) * 0.04
        logits = logits_best + noise
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        val = _objective(a, n, conv_fft_len)

        delta = val - cur_val
        if delta < 0 or np.random.rand() < math.exp(-delta / max(T, 1e-12)):
            # accept
            logits_best = logits
            cur_val = val
            if val < best_val:
                best_val = val
                a_best = a.copy()
                print(f"[SA] iter={it:06d} new best {val:.6f}")

        it += 1
        # safety break if temperature is already tiny and no improvement for a while
        if T < 1e-5 and it - best_val > 1e5:
            break

    return a_best, best_val


# ----------------------------------------------------------------------
def _deep_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_seq: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Run Adam‑smooth‑max → sub‑gradient refinement → short SA.
    Updates the globally best sequence/value.
    """
    # Adam phase
    best_seq, best_val = _adam_smooth_max(
        a0,
        n,
        conv_fft_len,
        corr_fft_len,
        deadline,
        global_best_seq,
        global_best_val,
    )
    # Sub‑gradient phase
    best_seq, best_val = _subgradient_refinement(
        a0,
        n,
        conv_fft_len,
        corr_fft_len,
        deadline,
        best_seq,
        best_val,
    )
    # Tiny simulated‑annealing refinement (only if time remains)
    if time.time() < deadline:
        best_seq, best_val = _simulated_annealing(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            best_val,
        )
    return best_seq, best_val


# ----------------------------------------------------------------------
def _local_random_search(
    a_best: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climbing: add small Gaussian noise to the logits,
    keep changes that strictly improve the exact objective.
    """
    logits_best = np.log(a_best + 1e-15)

    while time.time() < deadline:
        noise = np.random.randn(n) * 0.03
        logits = logits_best + noise
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        val = _objective(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            a_best = a.copy()
            logits_best = np.log(a_best + 1e-15)
            print(f"[LocalSearch] improved to {val:.6f}")

    return a_best, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Main entry point – searches for a non‑negative step function that
    minimises the supplied evaluation function.
    Returns the best sequence found (plain Python list).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # Base dimension (fast enough for many gradient steps)
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm‑start from previously found construction, if any
    warm_start = globals().get("height_sequence_1")
    if warm_start is not None:
        warm_start = np.asarray(warm_start, dtype=np.float64)
        if warm_start.shape[0] > n:
            warm_start = warm_start[:n]
        elif warm_start.shape[0] < n:
            warm_start = np.pad(warm_start, (0, n - warm_start.shape[0]))
        warm_start = np.maximum(warm_start, 0.0)
        s = warm_start.sum()
        if s >= 0.01:
            warm_start = warm_start / s
        else:
            warm_start = None
    else:
        warm_start = None

    # ------------------------------------------------------------------
    # Stage 1 – generate a large pool of cheap candidates
    all_candidates = _generate_parametric_candidates(n, warm_start)

    cand_vals = []
    for cand in all_candidates:
        if time.time() >= deadline:
            break
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 0.01:
            cand = np.random.dirichlet(np.ones(n))
            s = cand.sum()
        cand = cand / s
        val = _objective(cand, n, conv_fft_len)
        cand_vals.append((val, cand))

    # keep a small set of elite starts (fewer to give each more time)
    cand_vals.sort(key=lambda x: x[0])
    top_k = 2
    top_candidates = [c for _, c in cand_vals[:top_k]]
    print(f"[Stage 1] evaluated {len(cand_vals)} candidates, best = {cand_vals[0][0]:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – deep optimisation of each elite start, sharing the remaining time
    best_seq = None
    best_val = float("inf")
    for idx, a0 in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        remaining = deadline - time.time()
        # allocate roughly equal share of the remaining budget to each remaining start
        remaining_starts = len(top_candidates) - idx
        sub_deadline = time.time() + remaining / remaining_starts * 0.95
        print(f"[Stage 2] deep optimisation start #{idx + 1}/{len(top_candidates)} (budget ≈ {sub_deadline - time.time():.1f}s)")
        best_seq, best_val = _deep_optimize(
            a0,
            n,
            conv_fft_len,
            corr_fft_len,
            sub_deadline,
            best_seq,
            best_val,
        )

    # ------------------------------------------------------------------
    # Stage 3 – cheap stochastic hill‑climbing from the current best
    if best_seq is not None:
        print("[Stage 3] local stochastic refinement")
        best_seq, best_val = _local_random_search(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            best_val,
        )

    # ------------------------------------------------------------------
    # Stage 4 – if enough time remains, upscale to n=8192 and repeat deep optimisation
    time_left = deadline - time.time()
    if time_left > 80.0:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        if best_seq is not None:
            # linear interpolation to finer grid
            x_old = np.linspace(0, 1, n, endpoint=False)
            x_new = np.linspace(0, 1, n_big, endpoint=False)
            a_big = np.interp(x_new, x_old, best_seq)
        else:
            a_big = np.random.dirichlet(np.ones(n_big))

        a_big = np.maximum(a_big, 0.0)
        a_big /= a_big.sum()

        print("[Stage 4] optimisation on larger grid (n=8192)")
        best_seq, best_val = _deep_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            best_seq,
            best_val,
        )
        # keep the larger vector if it improves the bound
        if best_seq is not None and best_seq.shape[0] == n_big:
            n = n_big  # update size for final reporting

    # ------------------------------------------------------------------
    # Fallback – return uniform vector if nothing succeeded
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Final clipping (the optimiser already respects the bounds)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"*** Search finished: best objective ≈ {best_val:.6f} (n={best_seq.shape[0]})")
    return best_seq.tolist()
