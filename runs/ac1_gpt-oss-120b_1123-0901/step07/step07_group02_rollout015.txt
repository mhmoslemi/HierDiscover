def _adam_optimize(a0, n, conv_fft_len, corr_fft_len, deadline, max_iters=200_000):
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.08
    lam = 0.5
    lam_factor = 1.08
    best_a = a0.copy()
    best_val = float('inf')
    it=0
    while time.time() < deadline and it < max_iters:
        exp_log = np.exp(logits - np.max(logits))
        a = exp_log / exp_log.sum()
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
        # compute smooth max surrogate
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)
        m = beta1 * m + (1-beta1) * grad_log
        v = beta2 * v + (1-beta2) * (grad_log**2)
        m_hat = m / (1-beta1**(it+1))
        v_hat = v / (1-beta2**(it+1))
        logits -= lr * m_hat / (np.sqrt(v_hat)+eps_adam)
        lam = min(lam * lam_factor, 1e6)
        # optional random shake
        if it % 5000 == 0:
            logits += (np.random.rand(n)-0.5) * 1e-4
        it+=1
    return best_a, best_val
```

def _simulated_annealing(a0, n, conv_fft_len, deadline, max_moves=200_000, init_temp=1e-3, decay=0.9999):
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    best_a = a.copy()
    temp = init_temp
    i=0
    while time.time() < deadline and i < max_moves:
        # pick move
        src = np.random.randint(n)
        if a[src] < 1e-12:
            continue
        dst = np.random.randint(n)
        if src == dst:
            continue
        # delta fraction from source to dest
        frac = np.random.rand() * 0.5  # up to 50%
        delta = a[src] * frac
        a_new = a.copy()
        a_new[src] -= delta
        a_new[dst] += delta
        # renorm not needed as mass preserved
        conv_new = _full_convolution(a_new, conv_fft_len)
        val_new = 2.0 * n * conv_new.max()
        # acceptance
        if val_new < best_val or np.random.rand() < np.exp(-(val_new-best_val)/temp):
            a = a_new
            if val_new < best_val:
                best_val = val_new
                best_a = a_new.copy()
        temp *= decay
        i+=1
    return best_a, best_val
```

    candidates.append(np.full(n, 1.0/n))

"""
Propose a step‑function (non‑negative vector) with a low value of

        F(a) = 2·n·max(a∗a) / (∑a)²

The algorithm consists of several stages that respect a total time budget
(≈1000 s).  A diverse pool of hand‑crafted initial vectors (uniform,
Dirichlet, edge‑concentrated power‑laws, double‑exponential, sinusoid,
random spike patterns) is generated.  Each candidate is briefly refined
with Adam on a smooth log‑sum‑exp surrogate of the max‑convolution while
keeping the vector on the simplex via a soft‑max.  The best candidate so
far is then polished by a long exponentiated‑gradient sub‑gradient phase.
If time permits a simple simulated‑annealing hill‑climber makes a final
push, and the solution is optionally interpolated to a finer grid
(8192 or 16384 points) where the same optimisation steps are repeated.
All intermediate results are kept, the globally best sequence is finally
returned as a plain Python list.  The code uses only NumPy and runs on a
single CPU core.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a length‑len(a) vector:
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i = 0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _objective(a: np.ndarray, conv_fft_len: int) -> float:
    """Exact objective value for a normalised (∑a = 1) vector."""
    conv = _full_convolution(a, conv_fft_len)
    return 2.0 * a.shape[0] * conv.max()


# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max (log‑sum‑exp) surrogate.
    Returns the best vector (∑a≈1) encountered and its exact objective.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.08                    # base learning rate
    lam = 0.5                    # start temperature for smooth‑max
    lam_factor = 1.08            # grow temperature fast

    best_a = a0.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max → simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective (sum(a)≈1)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            # occasional progress report
            if it % 20_000 == 0:
                print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or not np.isfinite(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate (chain‑rule through soft‑max)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature schedule → true max
        lam = min(lam * lam_factor, 1e6)

        # tiny random shake every few thousand iterations
        if it % 10_000 == 0:
            logits += (np.random.rand(n) - 0.5) * 5e-4

        it += 1

    return best_a, best_val


def _subgradient_opt(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
    eta0: float = 0.7,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient optimiser with a decaying step size.
    Returns the best vector seen and its exact objective.
    """
    a = np.maximum(a0, 0.0)
    a_sum = a.sum()
    if a_sum < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    else:
        a = a / a_sum

    best_a = a.copy()
    best_val = float("inf")
    stall = 0
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            stall = 0
            if it % 5_000 == 0:
                print(f"[Subgrad it={it:07d}] new best {val:.6f}")
        else:
            stall += 1

        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # decaying step size
        eta = eta0 / np.sqrt(it + 1)

        # multiplicative update and renormalisation
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        # occasional random shake when stagnating
        if stall > 8_000:
            a = np.maximum(a + 0.02 * (np.random.rand(n) - 0.5), 0.0)
            a /= a.sum()
            stall = 0

        it += 1

    return best_a, best_val


def _simulated_annealing(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 30_000,
) -> tuple[np.ndarray, float]:
    """
    Cheap SA / hill‑climbing that moves a random fraction of mass from one
    index to another.  Moves that improve the objective are always kept;
    worsening moves are accepted with a temperature‑controlled probability.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    best_a = a.copy()
    temp = 0.01
    decay = 0.9999
    moves = 0
    while time.time() < deadline and moves < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-12:
            continue
        j = np.random.randint(n)
        if i == j:
            continue
        # move up to 40 % of the source mass
        delta = a[i] * (np.random.rand() * 0.4)
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta
        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()
        accept = False
        if val2 < best_val:
            accept = True
        else:
            prob = np.exp(-(val2 - best_val) / max(temp, 1e-12))
            if np.random.rand() < prob:
                accept = True
        if accept:
            a = a2
            if val2 < best_val:
                best_val = val2
                best_a = a2.copy()
        temp *= decay
        moves += 1
        # occasional progress report
        if moves % 5000 == 0:
            print(f"[SA move={moves}] best {best_val:.6f}")
    return best_a, best_val


# ----------------------------------------------------------------------
def _generate_initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Create a diverse set of start vectors."""
    cand = []

    # 1) warm start and a few rotations
    if warm is not None:
        cand.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    # 2) uniform
    cand.append(np.full(n, 1.0 / n))

    # 3) a few random Dirichlet draws
    for _ in range(5):
        cand.append(np.random.dirichlet(np.ones(n)))

    # 4) edge‑concentrated power‑law shapes
    idx = np.arange(n, dtype=np.float64) + 1.0
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-6.0, -0.2, 15):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    # 5) symmetric double‑exponential shapes
    i = np.arange(n, dtype=np.float64)
    for gamma in np.geomspace(0.2, 5.0, 12):
        shape = np.exp(-gamma * i) + np.exp(-gamma * (n - 1 - i))
        shape /= shape.sum()
        cand.append(shape)

    # 6) half‑wave sinusoid
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand.append(shape)

    # 7) deterministic spike patterns (equally spaced)
    for k in (32, 64, 128, 256, 512):
        a = np.zeros(n, dtype=np.float64)
        pos = np.round(np.linspace(0, n - 1, k)).astype(int)
        a[pos] = 1.0 / k
        cand.append(a)

    # 8) random sparse patterns (varying support size)
    for _ in range(12):
        k = np.random.randint(10, min(600, n // 2))
        pos = np.random.choice(n, k, replace=False)
        weights = np.random.dirichlet(np.ones(k))
        a = np.zeros(n, dtype=np.float64)
        a[pos] = weights
        cand.append(a)

    return cand


def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:

        else:

                a_big2_sub, val_big2_sub = _subgradient_opt(
                    best_seq,
                    n_big2,
                    conv_fft_len_big2,
                    corr_fft_len_big2,
                    deadline,
                    max_iters=200_000,
                )
                if val_big2_sub < best_val:
                    best_val = val_big2_sub
                    best_seq = a_big2_sub.copy()
                    print(f"[Big‑16384] Sub‑grad best {val_big2_sub:.6f}")

    # ------------------------------------------------------------------
    # Fallback – ensure a valid return value
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Clip values to the evaluator's safety limits
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    # expose the best sequence for possible future warm‑starts
    globals()["height_sequence_1"] = best_seq.copy()

    return best_seq.tolist()
