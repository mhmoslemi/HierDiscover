def _adam_expgd_opt(a0, n, conv_fft_len, corr_fft_len, deadline):
    a = np.maximum(a0, 0.0)
    a = a / a.sum()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    lr = 0.5  # initial learning rate
    t = 0
    best_a = a.copy()
    best_val = _objective(a, n, conv_fft_len)  # compute.
    while time.time() < deadline:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)  # a sum=1, but compute anyway.
        if val < best_val:
            best_val = val
            best_a = a.copy()
            # optionally print.
        # compute active set
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)  # shape (n,)
        # Adam updates on gradient
        t += 1
        m = beta1 * m + (1-beta1) * grad
        v = beta2 * v + (1-beta2) * (grad*grad)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = lr * m_hat / (np.sqrt(v_hat) + epsilon)
        # exponentiated gradient update
        a = a * np.exp(-step)
        # renormalize
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))

            # small random jitter:
            perturb = (np.random.rand(n)-0.5)*0.01
            a = a * np.exp(perturb)
            a = a / a.sum()
    return best_a, best_val

"""
Non‑negative step‑function optimizer.

The algorithm builds a diverse pool of parametric shapes (uniform,
Dirichlet, edge‑exponential, power‑law, Gaussian‑edges, sinusoid,
triangular, semi‑circular, generalized Gaussian).  All candidates are
evaluated quickly with an FFT‑based convolution.  The best few seeds are
refined with an Adam‑style exponentiated‑gradient (mirror‑descent) that
optimises the true max‑convolution.  If time permits the solution is
interpolated to a larger grid (n = 8192) and refined once more.

The optimizer respects the 1000 s budget, prints progress and returns
the best sequence (plain Python list) it found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using real FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the sub‑gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Fast evaluation of 2·n·max(a*a) / (sum(a)²) (sum is assumed =1)."""
    conv = _full_convolution(a, conv_fft_len)
    return float(2.0 * n * conv.max() / (a.sum() ** 2))


# ----------------------------------------------------------------------
# Adam‑style exponentiated‑gradient optimiser
def _adam_expgd_opt(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent with Adam momentum on the true max‑convolution.
    Returns the best vector encountered and its objective value.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 0.01:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.5          # base learning rate for the exponentiated step
    max_step = 10.0   # to avoid overflow in exp()

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_val = _objective(a, n, conv_fft_len)
    last_improve = time.time()
    it = 0

    while time.time() < deadline:
        # --- compute max‑convolution and its sub‑gradient ---
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val - 1e-12:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} Adam iter={it:07d}] new best {val:.6f}")

        # active set (indices achieving the maximum)
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        # sub‑gradient of the max‑convolution w.r.t. a
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # --- Adam update on the *gradient* (not on log‑a) ---
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        # Clamp step size to avoid overflow in the exponential map
        step = np.clip(step, -max_step, max_step)

        # exponentiated‑gradient multiplicative update
        a = a * np.exp(-step)

        # renormalise onto the simplex
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        # occasional jitter if we have been idle for a few seconds
        if time.time() - last_improve > 5.0:
            jitter = (np.random.rand(n) - 0.5) * 0.02   # log‑space jitter
            a = a * np.exp(jitter)
            a = a / a.sum()
            last_improve = time.time()

    return best_a, best_val

def _generate_candidates(n: int, warm_start: np.ndarray | None) -> list[np.ndarray]:
    """
    Produce a diverse list of initial sequences.
    Families:
        - warm start / rotations
        - uniform
        - random Dirichlet (various concentrations)
        - edge‑exponential (mixed with uniform)
        - edge‑power‑law (mixed)
        - double‑Gaussian edges
        - half‑wave sinusoid
        - triangular / semi‑circular
        - generalized Gaussian (p, sigma)
        - mixture of exponential & sinusoid
    All candidates are normalised to have sum≈1.
    """
    candidates = []

    # Warm start and simple cyclic shifts
    if warm_start is not None:
        candidates.append(warm_start.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            candidates.append(np.roll(warm_start, shift))

    # Uniform baseline
    candidates.append(np.full(n, 1.0 / n))

    # Random Dirichlet with a spectrum of concentration parameters
    for conc in (0.005, 0.01, 0.02, 0.05, 0.1, 0.3, 0.5, 1.0, 2.0):
        cand = np.random.dirichlet(np.full(n, conc))
        candidates.append(cand)

    i = np.arange(n, dtype=np.float64)

    # Edge‑exponential (optionally mixed with uniform)
    for alpha in np.logspace(-2, 1.0, num=12):          # 0.01 … 10
        edge = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
        edge /= edge.sum()
        for w in (0.0, 0.25, 0.5, 0.75, 1.0):
            shape = w * edge + (1.0 - w) / n
            shape /= shape.sum()
            candidates.append(shape)

    # Edge‑power‑law (mixed with uniform)
    for p in np.linspace(0.2, 5.0, num=12):
        edge = (i + 1.0) ** (-p) + (n - i) ** (-p)
        edge /= edge.sum()
        for w in (0.0, 0.33, 0.66, 1.0):
            shape = w * edge + (1.0 - w) / n
            shape /= shape.sum()
            candidates.append(shape)

    # Double‑Gaussian edges
    for gamma in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        shape /= shape.sum()
        candidates.append(shape)

    # Positive half‑wave sinusoid
    sin_shape = np.sin(np.pi * (i + 0.5) / n)
    sin_shape = np.maximum(sin_shape, 0.0)
    sin_shape /= sin_shape.sum()
    candidates.append(sin_shape)

    # Triangular (linear rise to centre, then fall)
    tri = np.minimum(i + 1, n - i)
    tri = tri.astype(np.float64)
    tri /= tri.sum()
    candidates.append(tri)

    # Semi‑circular (quarter‑circle density)
    x = (2.0 * i) / (n - 1) - 1.0          # maps i∈[0,n-1] → [-1,1]
    semi = np.sqrt(np.maximum(0.0, 1.0 - x ** 2))
    semi /= semi.sum()
    candidates.append(semi)

    # Generalised Gaussian family (shape = exp(- (i/σ)^p) + mirror)
    for p in np.linspace(0.5, 5.0, num=6):
        for sigma_frac in np.linspace(0.02, 0.5, num=6):
            sigma = sigma_frac * n
            shape = np.exp(- (i / sigma) ** p) + np.exp(- ((n - 1 - i) / sigma) ** p)
            shape = np.maximum(shape, 0.0)
            shape /= shape.sum()
            candidates.append(shape)

    # Mixture of exponential and sinusoid
    for alpha in np.logspace(-2, 1.0, num=8):
        exp_shape = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
        exp_shape /= exp_shape.sum()
        for w in np.linspace(0.0, 1.0, num=5):
            mix = w * exp_shape + (1.0 - w) * sin_shape
            mix /= mix.sum()
            candidates.append(mix)

    return candidates


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Main entry point – searches for a low‑evaluation‑score step function.
    Returns the best sequence found (as a plain Python list).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    # --------------------------------------------------------------
    # Primary grid size – 4096 gives a good speed / quality trade‑off.
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Warm start (if a previous solution is stored globally)
    warm_start = globals().get("height_sequence_1")
    if warm_start is not None:
        ws = np.asarray(warm_start, dtype=np.float64)
        if ws.shape[0] > n:
            ws = ws[:n]
        elif ws.shape[0] < n:
            ws = np.pad(ws, (0, n - ws.shape[0]))
        ws = np.maximum(ws, 0.0)
        if ws.sum() >= 0.01:
            ws = ws / ws.sum()
        else:
            ws = None
        warm_start = ws
    else:
        warm_start = None

    # --------------------------------------------------------------
    # Stage 1 – build a large pool of candidate vectors
    candidates = _generate_candidates(n, warm_start)

    # --------------------------------------------------------------
    # Stage 2 – fast evaluation of every candidate, keep the best few
    cand_vals = []
    for idx, cand in enumerate(candidates):
        if time.time() >= deadline:
            break
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 0.01:
            cand = np.random.dirichlet(np.ones(n))
            s = cand.sum()
        cand = cand / s
        val = _objective(cand, n, conv_fft_len)
        cand_vals.append((val, cand))

    if not cand_vals:
        # Fallback: uniform vector
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        print("*** No candidates evaluated – returning uniform vector.")
        return best_seq.tolist()

    cand_vals.sort(key=lambda x: x[0])
    top_k = min(8, len(cand_vals))
    top_candidates = [c for _, c in cand_vals[:top_k]]

    best_seq = None
    best_val = float("inf")

    # --------------------------------------------------------------
    # Stage 3 – deep optimisation of the selected seeds with Adam‑EG
    for start_id, a0 in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        init_val = _objective(a0, n, conv_fft_len)
        print(f"--- Optimising start #{start_id + 1}/{top_k} (init F={init_val:.6f})")
        seq, val = _adam_expgd_opt(
            a0,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
        )
        if val < best_val:
            best_val = val
            best_seq = seq
            print(f"*** New overall best F={best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 4 – if time permits, upscale to a finer grid and refine
    time_left = deadline - time.time()
    if time_left > 30.0 and best_seq is not None:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        # Linear interpolation onto the finer grid
        x_old = np.linspace(0, 1, n, endpoint=False)
        x_big = np.linspace(0, 1, n_big, endpoint=False)
        a_big = np.interp(x_big, x_old, best_seq)
        a_big = np.maximum(a_big, 0.0)
        a_big /= a_big.sum()

        print("--- Refinement on larger grid (n=8192)")
        seq_big, val_big = _adam_expgd_opt(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
        )
        if val_big < best_val:
            best_val = val_big
            best_seq = seq_big
            n = n_big  # remember final length

    # --------------------------------------------------------------
    # Final sanity: clip to allowed range and return
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"*** Finished search: best objective ≈ {best_val:.6f} (n={best_seq.shape[0]})")
    # store for possible warm‑start in a later run
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
