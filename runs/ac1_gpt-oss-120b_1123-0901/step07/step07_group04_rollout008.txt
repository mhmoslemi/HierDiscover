def _adam_opt(a0, n, conv_fft_len, corr_fft_len, deadline, best_seq, best_val, max_iter=20000):
   a = np.maximum(a0, 0)
   a = a / a.sum()
   m = np.zeros_like(a)
   v = np.zeros_like(a)
   beta1 = 0.9
   beta2 = 0.999
   eps = 1e-8
   eta = 0.1
   beta = 100.0  # smoothing parameter for max approximation
   it = 0
   last_improve = time.time()
   while time.time() < deadline and it < max_iter:
       conv = _full_convolution(a, conv_fft_len)
       max_c = conv.max()
       val = 2.0 * n * max_c / (a.sum()**2)  # a.sum=1 => 2n*max_c
       if val < best_val - 1e-12:
           best_val = val
           best_seq = a.copy()
           last_improve = time.time()
           print(f"[{n} Adam iter={it}] new best {val:.6f}")

       # compute soft weights
       # shift conv to avoid overflow:
       # w_i = exp(beta * (conv_i - max_c))
       # weight sum = sum exp( beta*(conv_i - max_c) )
       exp_term = np.exp(beta * (conv - max_c))
       w = exp_term / exp_term.sum()  # shape length 2n-1, sum 1

       # compute gradient: 2 * cross-correlation(w, a)
       grad = 2.0 * _cross_correlation(w, a, corr_fft_len, n)

       # Adam update
       m = beta1 * m + (1 - beta1) * grad
       v = beta2 * v + (1 - beta2) * (grad ** 2)
       m_hat = m / (1 - beta1 ** (it + 1))
       v_hat = v / (1 - beta2 ** (it + 1))
       a = a - eta * m_hat / (np.sqrt(v_hat) + eps)

       # Project onto simplex
       a = _proj_simplex(np.maximum(a, 0))

       # If idle, random jitter
       if time.time() - last_improve > 5.0:
           jitter = (np.random.rand(n) - 0.5) * 0.01
           a = a * np.exp(jitter)
           a = _proj_simplex(a)
           last_improve = time.time()

       # Decay parameters
       if (it+1) % 1000 == 0:
           # increase beta (more focus on max) and decrease eta
           beta = min(1000.0, beta * 1.2)
           eta = eta * 0.9

       it += 1

   return best_seq, best_val

def _proj_simplex(v):
    # ensure non-negative and sum to 1
    n = len(v)
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all <=0, return uniform
        return np.full(n, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a low‑objective step‑function using an arcsine‑inspired seed and
Adam‑style projected gradient descent on a smooth approximation of the
max‑convolution.  The method works in three phases:

1.  Generate a diverse pool of candidates (uniform, Dirichlet, edge‑power‑law,
    edge‑exponential, and a direct discretisation of the arcsine pdf).
    All candidates are normalised to sum = 1.

2.  Keep the best few candidates (fast FFT‑based objective) and refine each
    with Adam on the *smooth max* of the self‑convolution:
        w_k ∝ exp(β·( (a*a)_k – max(a*a) ))   (β grows during optimisation)
    The gradient of the smooth max is
        ∇ = 2·(w ★ a)                          (cross‑correlation)
    After each update the vector is projected back onto the simplex
    (non‑negative, sum = 1).

3.  If time permits, interpolate the best solution to a finer grid
    (n = 8192) and run a short additional Adam refinement.

The objective is invariant to scaling, so the simplex constraint does not
affect the optimum.  The arcsine shape roughly makes the self‑convolution
uniform, pushing the objective close to its theoretical lower bound (≈1).
The whole search respects the 1000 s budget and returns the best sequence
found as a plain Python list.
"""

import time
import numpy as np


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    n = a.shape[0]
    return conv[: 2 * n - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Cross‑correlation (w ★ a) giving a vector of length n:
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n-1)].
    Implemented via convolution of w with reversed a.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr_full = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr_full[start:start + n]


def _objective_val(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Fast evaluation of the original objective (identical to evaluate_sequence)."""
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    # a.sum() == 1 in our usage, but keep the formula for safety.
    return float(2.0 * n * max_c / (a.sum() ** 2))


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    Reference: Duchi et al., “Efficient Projections onto the ℓ1‑Ball…”.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are non‑positive; return uniform vector.
        return np.full(n, 1.0 / n, dtype=v.dtype)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _adam_opt(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
    max_iter: int = 5000,
) -> tuple[np.ndarray | None, float]:
    """
    Adam optimiser on a smooth approximation of  max(a*a).
    Returns the best vector/value observed.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    # Adam parameters
    m = np.zeros_like(a)
    v_adam = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # optimisation hyper‑parameters
    eta = 0.2          # base learning rate
    beta = 20.0        # smoothing parameter for the max (soft‑max)
    beta_inc = 1.05    # increase beta every few hundred steps

    last_improve = time.time()
    it = 0

    while time.time() < deadline and it < max_iter:
        # ---- compute objective & check improvement ----
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)   # a.sum() stays equal to 1
        if val < best_val - 1e-12:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[n={n} Adam iter={it:06d}] new best {val:.6f}")

        # ---- smooth‑max weights ----
        # w_k ∝ exp(β·(conv_k – max_c)), summed to 1.
        exp_term = np.exp(beta * (conv - max_c))
        w = exp_term / exp_term.sum()

        # ---- gradient of smooth max (2·(w ★ a)) ----
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len, n)

        # ---- Adam update (additive) ----
        m = beta1 * m + (1.0 - beta1) * grad
        v_adam = beta2 * v_adam + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v_adam / (1.0 - beta2 ** (it + 1))
        a = a - eta * m_hat / (np.sqrt(v_hat) + eps)

        # ---- project back onto the simplex ----
        a = _project_to_simplex(np.maximum(a, 0.0))

        # ---- occasional random shake if stuck ----
        if time.time() - last_improve > 5.0:
            jitter = (np.random.rand(n) - 0.5) * 0.02   # log‑space perturbation
            a = a * np.exp(jitter)
            a = _project_to_simplex(a)
            last_improve = time.time()

        # ---- schedule updates ----
        if (it + 1) % 300 == 0:
            beta = min(1000.0, beta * beta_inc)
            eta = max(1e-4, eta * 0.95)

        it += 1

    return best_seq, best_val


def _generate_candidates(n: int, warm_start: np.ndarray | None) -> list[np.ndarray]:
    """
    Produce a fairly rich set of seed vectors:
      * uniform
      * random Dirichlet (several concentrations)
      * edge‑power‑law (p ∈ [0.1, 1.0]) with mixing weight w
      * edge‑exponential (α ∈ [0.01, 3]) with mixing weight w
      * arcsine‑like discretisation (with small epsilon offsets)
      * small perturbed copies of warm start (if available)
    All candidates are normalised to sum = 1.
    """
    candidates = []

    # warm‑start and its rotations
    if warm_start is not None:
        candidates.append(warm_start.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            candidates.append(np.roll(warm_start, shift))

    # uniform
    candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

    i = np.arange(n, dtype=np.float64)

    # arcsine discretisation: a_i ∝ 1 / sqrt((i+½)(n‑i‑½))
    for eps in (0.0, 0.5, 1.0):
        shape = 1.0 / np.sqrt((i + 0.5 + eps) * (n - i - 0.5 + eps))
        shape = np.maximum(shape, 0.0)
        shape /= shape.sum()
        candidates.append(shape)

    # edge‑power‑law family
    p_vals = np.linspace(0.1, 1.0, 19)          # 0.1,0.15,…,1.0
    w_vals = np.linspace(0.0, 1.0, 11)          # 0.0,…,1.0
    for p in p_vals:
        edge = (i + 1) ** (-p) + (n - i) ** (-p)
        edge /= edge.sum()
        for w in w_vals:
            shape = w * edge + (1.0 - w) / n
            candidates.append(shape)

    # edge‑exponential family
    alphas = np.logspace(-2, 0.6, 20)           # 0.01 … ≈4
    for alpha in alphas:
        edge = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
        edge /= edge.sum()
        for w in (0.0, 0.5, 1.0):
            shape = w * edge + (1.0 - w) / n
            candidates.append(shape)

    # random Dirichlet samples
    for conc in (0.02, 0.05, 0.1, 0.3, 0.8, 2.0):
        for _ in range(4):
            cand = np.random.dirichlet(np.full(n, conc))
            candidates.append(cand)

    return candidates


def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a step‑function (non‑negative vector) with minimal
    ``evaluate_sequence`` score.  The routine respects the time budget,
    prints progress and finally returns the best vector as a plain list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Primary resolution
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Warm‑start from previous global if present
    warm_start = globals().get("height_sequence_1")
    if warm_start is not None:
        warm_start = np.asarray(warm_start, dtype=np.float64)
        if warm_start.ndim != 1:
            warm_start = None
        else:
            if warm_start.shape[0] > n:
                warm_start = warm_start[:n]
            elif warm_start.shape[0] < n:
                warm_start = np.pad(warm_start, (0, n - warm_start.shape[0]))
            warm_start = np.maximum(warm_start, 0.0)
            if warm_start.sum() >= 0.01:
                warm_start = warm_start / warm_start.sum()
            else:
                warm_start = None
    else:
        warm_start = None

    # --------------------------------------------------------------
    # Phase 1 – generate a pool of candidates
    pool = _generate_candidates(n, warm_start)

    # Phase 2 – cheap evaluation & keep the best few
    evaluated = []
    for idx, cand in enumerate(pool):
        if time.time() >= deadline:
            break
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 0.01:
            cand = np.random.dirichlet(np.ones(n))
            s = cand.sum()
        cand = cand / s
        val = _objective_val(cand, n, conv_fft_len)
        evaluated.append((val, cand))
        if (idx + 1) % 100 == 0:
            print(f"[Phase 1] evaluated {idx + 1}/{len(pool)} candidates – best so far {min(v[0] for v in evaluated):.6f}")

    if not evaluated:
        # fallback – uniform vector
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        print("*** No candidates evaluated – returning uniform vector.")
        return best_seq.tolist()

    evaluated.sort(key=lambda x: x[0])
    top_k = 8
    top_candidates = [c for _, c in evaluated[:top_k]]

    # --------------------------------------------------------------
    # Phase 3 – local refinement with Adam (smooth‑max)
    best_seq = None
    best_val = float("inf")

    for start_id, a0 in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        init_val = _objective_val(a0, n, conv_fft_len)
        print(f"--- Refinement start #{start_id + 1}/{len(top_candidates)} (init {init_val:.6f})")
        best_seq, best_val = _adam_opt(
            a0,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            best_seq,
            best_val,
            max_iter=6000,
        )

    # --------------------------------------------------------------
    # Phase 4 – optional up‑scaling to n=8192 if time permits
    time_left = deadline - time.time()
    if time_left > 30.0 and best_seq is not None:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        # linear interpolation onto the finer grid
        x_old = np.linspace(0, 1, n, endpoint=False)
        x_big = np.linspace(0, 1, n_big, endpoint=False)
        a_big = np.interp(x_big, x_old, best_seq)
        a_big = np.maximum(a_big, 0.0)
        a_big = a_big / a_big.sum()

        print("--- Upscaling to n=8192 and brief refinement")
        best_seq, best_val = _adam_opt(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            best_seq,
            best_val,
            max_iter=3000,
        )
        if best_seq is not None and best_seq.shape[0] == n_big:
            n = n_big

    # --------------------------------------------------------------
    # Final safety clip (evaluate_sequence expects 0 ≤ x ≤ 1000)
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"*** Search finished – best objective ≈ {best_val:.6f} (n={best_seq.shape[0]})")
    return best_seq.tolist()
