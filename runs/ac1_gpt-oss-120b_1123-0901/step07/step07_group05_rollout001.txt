"""
propose_candidate
-----------------
Search for a non‑negative step‑function that minimises

    2·n·max(conv(a,a)) / (sum(a))²

using a multi‑stage pipeline:

1. **Diverse cheap seeds** – a large pool of parametric families
   (uniform, Dirichlet, symmetric/asymmetric exponentials,
   power‑law, beta‑like shapes, triangular, sinusoidal and sparse spikes)
   is sampled and the best few are kept.

2. **Mirror‑descent on a smooth‑max surrogate** – Adam optimisation on a
   log‑sum‑exp approximation of the convolution maximum.  The loop runs
   until the time budget expires; hyper‑parameters (learning‑rate,
   temperature schedule) are chosen to be more patient than the previous
   baseline.

3. **Exact sub‑gradient phase** – exponentiated‑gradient updates on the
   true max‑convolution (active‑set weights).  This phase also runs
   until the deadline.

4. **Local stochastic refinement** – Gaussian perturbations of the
   logits are accepted if they improve the exact objective (hill‑climbing).

5. **Scale‑up** – if enough time remains the best vector is interpolated
   to a finer grid (n = 8192) and steps 2‑4 are repeated.

All operations stay on the probability simplex (sum = 1), which makes the
objective independent of the overall scale.  The routine finally returns
the best sequence found as a plain Python list.  No filesystem or network
access is used; only NumPy and the supplied ``evaluate_sequence`` are
required.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2*len(a)-1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact objective: 2·n·max(a*a) / (sum(a)²)."""
    conv = _full_convolution(a, fft_len)
    return float(2.0 * n * conv.max() / (a.sum() ** 2))


# ----------------------------------------------------------------------
def _adam_smooth_max(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    init_best_seq: np.ndarray | None,
    init_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Adam on a log‑sum‑exp surrogate of the max‑convolution.
    Runs until the deadline.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr = 0.05

    # temperature (smooth‑max parameter) schedule
    lam = 2.0
    lam_factor = 1.006      # slower increase → smoother gradients
    lam_cap = 1e6

    best_seq = init_best_seq
    best_val = init_best_val
    last_improve = time.time()

    it = 0
    max_iter = 10 ** 8            # effectively unlimited – loop stops by deadline

    while time.time() < deadline and it < max_iter:
        # ---- soft‑max to obtain a on the simplex ----
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # ---- exact objective (used only for best‑track) ----
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Adam] iter={it:07d} new best {val:.6f}")

        # ---- smooth‑max weights (log‑sum‑exp) ----
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---- gradient of the surrogate w.r.t. a ----
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)                     # chain rule through soft‑max
        grad_log = a * (grad_a - avg)

        # ---- Adam update on logits ----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---- temperature schedule ----
        lam = min(lam * lam_factor, lam_cap)

        # ---- occasional jitter if stuck for a while ----
        if time.time() - last_improve > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        it += 1

    return best_seq, best_val

def _subgradient_refinement(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    init_best_seq: np.ndarray | None,
    init_best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient steps on the *exact* max‑convolution.
    The loop stops when the deadline is reached or when no improvement
    occurs for several seconds.
    """
    a = init_best_seq if init_best_seq is not None else a0.copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a /= s                         # stay on the simplex

    eta0 = 0.8
    decay = 0.99995
    eta = eta0

    best_seq = a.copy()
    best_val = init_best_val
    it = 0
    max_iter = 10 ** 8
    last_improve = time.time()

    while time.time() < deadline and it < max_iter:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Subgrad] iter={it:07d} new best {val:.6f}")

        # ---- active set (indices attaining the maximum) ----
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)    # should not happen

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # ---- exponentiated‑gradient update (keeps a ≥ 0) ----
        a = a * np.exp(-eta * grad_a)

        # ---- renormalise ----
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        eta *= decay
        it += 1

        if time.time() - last_improve > 6.0:
            break

    return best_seq, best_val


# ----------------------------------------------------------------------
def _local_random_search(
    a_best: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climbing: Gaussian perturbation of the logits.
    Accepts only improvements.
    """
    logits_best = np.log(a_best + 1e-15)

    while time.time() < deadline:
        noise = np.random.randn(n) * 0.03
        logits = logits_best + noise
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        val = _objective(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            a_best = a.copy()
            logits_best = np.log(a_best + 1e-15)
            print(f"[Local] improved to {val:.6f}")

    return a_best, best_val

def _smooth_vector(a: np.ndarray, window: int = 5) -> np.ndarray:
    """
    Simple moving‑average smoothing; keeps the vector non‑negative.
    The result is re‑normalised to sum = 1.
    """
    if window <= 1:
        return a
    kernel = np.ones(window) / window
    a_s = np.convolve(a, kernel, mode='same')
    a_s = np.maximum(a_s, 0.0)
    s = a_s.sum()
    if s < 1e-12:
        return a
    return a_s / s

def _generate_parametric_candidates(n: int, warm_start: np.ndarray | None):
    """
    Return a list of diverse candidate vectors, each already normalised.
    """
    candidates = []

    # ---- warm‑start and simple shifts ----
    if warm_start is not None:
        candidates.append(warm_start.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            candidates.append(np.roll(warm_start, shift))

    # ---- uniform ----
    candidates.append(np.full(n, 1.0 / n))

    # ---- random Dirichlet draws with many concentrations ----
    for conc in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(10):
            candidates.append(np.random.dirichlet(np.full(n, conc)))

    # ---- symmetric exponential families ----
    i = np.arange(n, dtype=np.float64)
    alphas = np.logspace(-2, 1.0, num=12)          # 0.01 … 10
    for a in alphas:
        edge = np.exp(-a * i) + np.exp(-a * (n - 1 - i))
        edge /= edge.sum()
        for w in (0.0, 0.25, 0.5, 0.75, 1.0):
            cand = w * edge + (1.0 - w) / n
            candidates.append(cand / cand.sum())

    # ---- symmetric power‑law families ----
    ps = np.linspace(0.2, 6.0, num=12)
    for p in ps:
        edge = (i + 1.0) ** (-p) + (n - i) ** (-p)
        edge /= edge.sum()
        for w in (0.0, 0.3, 0.6, 1.0):
            cand = w * edge + (1.0 - w) / n
            candidates.append(cand / cand.sum())

    # ---- asymmetric two‑sided exponentials ----
    alphas = np.logspace(-2, 1.0, num=8)
    w_vals = np.linspace(0.0, 1.0, 5)
    for wL in w_vals:
        for wR in w_vals:
            if wL + wR > 1.0:
                continue
            wU = 1.0 - wL - wR
            for aL in alphas:
                for aR in alphas:
                    left = np.exp(-aL * i)
                    right = np.exp(-aR * (n - 1 - i))
                    shape = wL * left + wR * right + wU / n
                    shape /= shape.sum()
                    candidates.append(shape)

    # ---- beta‑like shapes (i^α (n-1‑i)^β) ----
    alphas = np.linspace(-0.9, 5.0, 9)   # allow negative to push mass to edges
    betas = np.linspace(-0.9, 5.0, 9)
    for A in alphas:
        for B in betas:
            shape = (i + 1) ** A * (n - i) ** B
            shape = np.maximum(shape, 0.0)
            if shape.sum() == 0:
                continue
            shape /= shape.sum()
            for w in (0.0, 0.4, 0.8):
                cand = w * shape + (1.0 - w) / n
                cand /= cand.sum()
                candidates.append(cand)

    # ---- simple linear / triangular ----
    lin = np.linspace(1.0, 0.0, n)
    tri = np.concatenate([np.linspace(0, 1, n // 2, endpoint=False),
                          np.linspace(1, 0, n - n // 2)])
    for base in (lin, tri):
        base = base / base.sum()
        for w in (0.0, 0.4, 0.8):
            cand = w * base + (1.0 - w) / n
            cand /= cand.sum()
            candidates.append(cand)

    # ---- half‑wave sinusoid ----
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    candidates.append(shape)

    # ---- sparse spike vectors (few non‑zeros) ----
    for spikes in (3, 5, 8):
        for _ in range(15):
            idx = np.random.choice(n, size=spikes, replace=False)
            weights = np.random.dirichlet(np.ones(spikes))
            vec = np.zeros(n)
            vec[idx] = weights
            vec /= vec.sum()
            candidates.append(vec)

    return candidates

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Main entry point.  Returns a list of non‑negative heights that (hopefully)
    achieves a better bound than the previous best (~1.503).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # keep a small safety margin

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # --------------------------------------------------------------
    # Problem size – start with n = 4096 (used by the previous baseline)
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Warm start from a previously found construction, if present
    warm_start = globals().get("height_sequence_1")
    if warm_start is not None:
        warm_start = np.asarray(warm_start, dtype=np.float64)
        if warm_start.shape[0] > n:
            warm_start = warm_start[:n]
        elif warm_start.shape[0] < n:
            warm_start = np.pad(warm_start, (0, n - warm_start.shape[0]))
        warm_start = np.maximum(warm_start, 0.0)
        s = warm_start.sum()
        if s >= 0.01:
            warm_start = warm_start / s
        else:
            warm_start = None
    else:
        warm_start = None

    # --------------------------------------------------------------
    # Stage 1 – generate a large pool of cheap candidates
    all_cands = _generate_parametric_candidates(n, warm_start)

    cand_vals = []
    for cand in all_cands:
        if time.time() >= deadline:
            break
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 0.01:
            cand = np.random.dirichlet(np.ones(n))
            s = cand.sum()
        cand = cand / s
        val = _objective(cand, n, conv_fft_len)
        cand_vals.append((val, cand))

    # keep the top K seeds for the expensive optimisation
    cand_vals.sort(key=lambda x: x[0])
    top_k = 24
    top_candidates = [c for _, c in cand_vals[:top_k]]
    if cand_vals:
        print(f"Stage 1: evaluated {len(cand_vals)} candidates, best = {cand_vals[0][0]:.6f}")

    # --------------------------------------------------------------
    # Stage 2 – deep optimisation (Adam → sub‑gradient) on each seed
    for idx, seed_vec in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        print(f"--- Deep optimisation on seed #{idx + 1}/{len(top_candidates)} ---")
        # Adam phase
        best_seq, best_val = _adam_smooth_max(
            seed_vec,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            best_seq,
            best_val,
        )
        # Sub‑gradient refinement
        best_seq, best_val = _subgradient_refinement(
            seed_vec,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            best_seq,
            best_val,
        )

    # --------------------------------------------------------------
    # Stage 3 – cheap stochastic hill‑climbing from the current best
    if best_seq is not None:
        print("--- Local stochastic refinement ---")
        best_seq, best_val = _local_random_search(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            best_val,
        )

    # --------------------------------------------------------------
    # Stage 4 – optional up‑sampling to a finer grid if time permits
    time_left = deadline - time.time()
    if time_left > 50.0:                 # need enough time for another round
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        if best_seq is not None:
            # linear interpolation to the finer grid
            x_old = np.linspace(0, 1, n, endpoint=False)
            x_new = np.linspace(0, 1, n_big, endpoint=False)
            a_big = np.interp(x_new, x_old, best_seq)
            a_big = np.maximum(a_big, 0.0)
            a_big /= a_big.sum()
        else:
            a_big = np.random.dirichlet(np.ones(n_big))

        print("--- Optimising on larger grid (n=8192) ---")
        best_seq, best_val = _adam_smooth_max(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            best_seq,
            best_val,
        )
        best_seq, best_val = _subgradient_refinement(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            best_seq,
            best_val,
        )
        # final cheap local search on the large grid
        best_seq, best_val = _local_random_search(
            best_seq,
            n_big,
            conv_fft_len_big,
            deadline,
            best_val,
        )
        n = n_big
        conv_fft_len = conv_fft_len_big

    # --------------------------------------------------------------
    # Optional smoothing step to break possible tiny spikes
    if best_seq is not None:
        best_seq = _smooth_vector(best_seq, window=7)
        best_val = _objective(best_seq, n, conv_fft_len)
        print(f"After smoothing: objective ≈ {best_val:.6f}")

    # --------------------------------------------------------------
    # Fallback – uniform vector if everything failed
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Clip to the evaluator’s allowed range (non‑negative, ≤ 1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"*** Finished search: best objective ≈ {best_val:.6f} (n={best_seq.shape[0]})")
    return best_seq.tolist()
