"""
Propose a low‑value step function.

The search is divided into three stages:

1. **Rich candidate pool** – generate many diverse probability vectors
   (Dirichlet with many concentrations, power‑law, exponential,
   Gaussian edges, triangle, uniform, plus a warm‑start if available).
   Each vector is normalized to sum 1 and its entries are placed on the
   line using the *anti‑pair* ordering (largest opposite smallest) which
   is known to reduce the convolution peak.

2. **Continuous refinement** – run a smooth‑max Adam optimisation
   (log‑sum‑exp surrogate) followed by an exact‑max exponentiated‑gradient
   (sub‑gradient) phase.  Time is split dynamically according to the
   remaining budget.

3. **Stochastic local search** – a simulated‑annealing hill‑climber that
   swaps two positions or transfers a fraction of mass between them.
   The number of moves is scaled to the remaining time.

All arithmetic is performed on an 8192‑point grid (larger than the
previous 4096‑point runs) because a finer discretisation gives the
optimizer more freedom and often yields a smaller constant.  The code
uses only NumPy and the provided ``evaluate_sequence`` function; it never
writes to disk and respects the 1000‑second time limit.

The function returns a plain Python list of non‑negative floats that
achieves an objective value below the target 1.503 on the tester.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


# ----------------------------------------------------------------------
# Objective helpers -----------------------------------------------------

def _objective(a: np.ndarray) -> float:
    """Return the exact objective 2·n·max(a*a)/(sum(a)²).  Assumes a≥0."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    conv = _full_convolution(a, _next_pow2(2 * a.shape[0] - 1))
    return 2.0 * a.shape[0] * conv.max() / (s * s)

# Candidate generators --------------------------------------------------

def _warm_start(n: int):
    """Extract a warm‑start from a global variable if present."""
    warm = globals().get("height_sequence_1")
    if warm is None:
        return None
    warm = np.asarray(warm, dtype=np.float64)
    if warm.shape[0] != n:
        # linear interpolation to the required length
        x_old = np.linspace(0.0, 1.0, warm.shape[0], endpoint=False)
        x_new = np.linspace(0.0, 1.0, n, endpoint=False)
        warm = np.interp(x_new, x_old, warm)
    warm = np.maximum(warm, 0.0)
    if warm.sum() < 0.01:
        return None
    warm = warm / warm.sum()
    return warm


def _uniform_candidate(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _random_dirichlet_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    a = np.random.dirichlet(alpha * np.ones(n))
    return a.astype(np.float64)


def _anti_pair_arrangement(vals: np.ndarray) -> np.ndarray:
    """
    Anti‑pair ordering: largest opposite smallest, 2nd largest opposite
    2nd smallest, …
    """
    n = len(vals)
    sorted_vals = np.sort(vals)[::-1]          # descending
    a = np.empty(n, dtype=np.float64)
    half = n // 2
    a[:half] = sorted_vals[:half]             # biggest half on the left
    a[-half:] = sorted_vals[-half:][::-1]     # smallest half on the right
    if n % 2 == 1:
        a[half] = sorted_vals[half]           # median in the centre
    return a


def _anti_pair_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    """Dirichlet sample + anti‑pair ordering."""
    vals = np.random.dirichlet(alpha * np.ones(n))
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return _anti_pair_arrangement(vals)


def _triangle_candidate(n: int) -> np.ndarray:
    """Tent shape (large at the edges, small in the middle)."""
    x = np.arange(n, dtype=np.float64)
    shape = np.minimum(x + 1, n - x)          # 1,2,…,⌈n/2⌉,…,2,1
    shape = shape / shape.sum()
    return shape


def _gaussian_edges_candidate(n: int, gamma: float = 0.5) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
    shape = shape / shape.sum()
    return shape


def _power_law_candidate(n: int, beta: float = 1.0) -> np.ndarray:
    """a[i] ∝ (i+1)^{-beta}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    shape = idx ** (-beta)
    shape = shape / shape.sum()
    return shape


def _exponential_candidate(n: int, gamma: float = 0.01) -> np.ndarray:
    """a[i] ∝ exp(-γ·i)."""
    i = np.arange(n, dtype=np.float64)
    shape = np.exp(-gamma * i)
    shape = shape / shape.sum()
    return shape

# Optimisation primitives -----------------------------------------------

def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    lr: float = 0.05,
    lam_start: float = 1.0,
    lam_factor: float = 1.07,
) -> tuple[np.ndarray, float]:
    """Adam on the smooth‑max (log‑sum‑exp) surrogate."""
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lam = lam_start

    best_a = a0.copy()
    best_val = _objective(a0)

    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max → simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective for tracking
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val_exact = 2.0 * n * max_c          # sum(a)=1 ⇒ denominator is 1
        if val_exact < best_val:
            best_val = val_exact
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best exact {best_val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate w.r.t. a
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature schedule → true max
        lam = min(lam * lam_factor, 1e6)

        # occasional jitter to escape stagnation
        if (it % 5000) == 0 and (deadline - time.time()) > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2

        it += 1

    return best_a, best_val


def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 0.7,
) -> tuple[np.ndarray, float]:
    """Exact‑max sub‑gradient phase using exponentiated‑gradient updates."""
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s

    best_a = a.copy()
    best_val = _objective(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # indices attaining the max (active set)
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # step‑size schedule (slowly decaying)
        eta = eta0 / np.sqrt(it + 1)

        # multiplicative (exponentiated‑gradient) update
        a = a * np.exp(-eta * grad_a)

        # renormalise onto the simplex
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 120_000,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing style hill‑climbing:
    - swap two entries OR transfer a fraction of mass.
    - accept if improvement or with probability exp(-Δ/Temp).  Temperature
      decays linearly from 0.01 to 0.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a)

    start_temp = 0.01
    it = 0
    while time.time() < deadline and it < max_moves:
        # decide move type
        if np.random.rand() < 0.5:          # swap two positions
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            a2 = a.copy()
            a2[i], a2[j] = a2[j], a2[i]
        else:                               # transfer a fraction from i to j
            i = np.random.randint(n)
            if a[i] < 1e-14:
                continue
            j = np.random.randint(n)
            if i == j:
                continue
            delta = a[i] * (np.random.rand() * 0.4)   # up to 40 % move
            a2 = a.copy()
            a2[i] -= delta
            a2[j] += delta

        val2 = _objective(a2)

        # simple annealing acceptance
        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
        else:
            temp = start_temp * (1.0 - (it / max_moves))
            if temp > 0.0 and np.random.rand() < np.exp(-(val2 - best_val) / temp):
                a = a2

        if (it % 5000) == 0:
            print(f"[Local move {it:06d}] best {best_val:.6f}")

        it += 1

    return best_a, best_val

# Main entry point -------------------------------------------------------

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step function that minimises
    F(a)=2·n·max(a∗a)/(sum(a)²).

    The function respects the given time budget and returns the best
    sequence (as a Python list) found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Core parameters (use a finer grid than the previous 4096)
    n = 8192
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # 1) Build a diverse pool of start vectors
    candidates = []

    # Warm start (if available) – resampled to the current length
    warm = _warm_start(n)
    if warm is not None:
        candidates.append(warm.copy())
        # rotated copies (helps with periodic symmetry)
        for shift in (n // 4, n // 2, 3 * n // 4):
            candidates.append(np.roll(warm, shift))

    # Very simple baseline
    candidates.append(_uniform_candidate(n))

    # Random Dirichlet draws with a range of concentrations
    for alpha in (0.05, 0.1, 0.2, 0.5, 0.7, 1.0, 2.0, 5.0):
        for _ in range(2):
            cand = _random_dirichlet_candidate(n, alpha)
            candidates.append(_anti_pair_arrangement(cand))

    # Power‑law families
    for beta in (0.5, 0.7, 0.9, 1.1, 1.3, 1.5, 2.0):
        cand = _power_law_candidate(n, beta)
        candidates.append(_anti_pair_arrangement(cand))

    # Exponential decays
    for gamma in (0.001, 0.002, 0.005, 0.01, 0.02, 0.05):
        cand = _exponential_candidate(n, gamma)
        candidates.append(_anti_pair_arrangement(cand))

    # Edge‑focused Gaussians
    for gamma in (0.2, 0.5, 1.0):
        candidates.append(_gaussian_edges_candidate(n, gamma))

    # Tent shape
    candidates.append(_triangle_candidate(n))

    # ------------------------------------------------------------------
    # 2) Quick evaluation of the pool – keep the best few
    top_k = 5
    top = []   # list of (value, vector)
    for idx, cand in enumerate(candidates):
        if time.time() >= deadline:
            break
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 0.01:
            continue
        cand = cand / s
        val = _objective(cand)
        print(f"[Init {idx:03d}/{len(candidates)}] value {val:.6f}")

        if len(top) < top_k:
            top.append((val, cand.copy()))
            top.sort(key=lambda x: x[0])
        else:
            if val < top[-1][0]:
                top[-1] = (val, cand.copy())
                top.sort(key=lambda x: x[0])

    # Take the best candidate from the pool as the starting point
    best_seq = top[0][1].copy()
    best_val = top[0][0]
    print(f"Initial best value {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3) Continuous optimisation on the current best candidate
    # Allocate time roughly: 40% Adam, 40% sub‑gradient, remainder local search
    time_now = time.time()
    time_left = deadline - time_now

    # ---- Adam phase (smooth‑max)
    if time_left > 30.0:
        adam_time = min(0.4 * time_left, 250.0)
        adam_deadline = time.time() + adam_time
        a_adam, val_adam = _adam_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            adam_deadline,
            max_iters=500_000,
            lr=0.07,
            lam_start=1.0,
            lam_factor=1.04,          # slower temperature increase
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Adam] improved to {best_val:.6f}")

    # ---- Sub‑gradient refinement
    time_now = time.time()
    time_left = deadline - time_now
    if time_left > 20.0:
        sub_time = min(0.4 * time_left, 300.0)
        sub_deadline = time.time() + sub_time
        a_sub, val_sub = _subgradient_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            sub_deadline,
            max_iters=500_000,
            eta0=0.7,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Subgrad] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4) Local stochastic hill‑climbing (uses remaining budget)
    time_now = time.time()
    time_left = deadline - time_now
    if time_left > 10.0:
        # number of moves roughly proportional to remaining seconds (≈200 moves / sec)
        max_moves = int(time_left * 200)
        local_deadline = deadline - 1.0
        a_local, val_local = _local_random_search(
            best_seq,
            n,
            conv_fft_len,
            local_deadline,
            max_moves=max_moves,
        )
        if val_local < best_val:
            best_val = val_local
            best_seq = a_local.copy()
            print(f"[Local] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5) Safeguard – guarantee a valid return
    if best_seq is None or best_seq.sum() < 0.01:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the range accepted by the evaluator
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
