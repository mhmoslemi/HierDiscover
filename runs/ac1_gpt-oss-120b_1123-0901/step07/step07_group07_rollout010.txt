    """Linear convolution a*a (length 2·len(a)-1) using a real FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _objective(a: np.ndarray, conv_fft_len: int) -> float:
    """Exact evaluation used for comparisons with evaluate_sequence."""
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    n = a.shape[0]
    return float(2.0 * n * max_c / (a.sum() ** 2))


def _smooth_max_and_grad(
    a: np.ndarray,
    lam: float,
    conv_fft_len: int,
    corr_fft_len: int,
) -> tuple[float, np.ndarray, float]:
    """
    Return (smooth_max, grad_wrt_a, true_max).
    smooth_max = max_c + log( Σ exp(lam·(conv - max_c)) ) / lam
    grad_wrt_a = 2·(w ★ a)   where w_i ∝ exp(lam·conv_i) and Σ w_i = 1.
    """
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    # stable log‑sum‑exp
    exp_term = np.exp(lam * (conv - max_c))
    w_sum = exp_term.sum()
    if w_sum == 0 or np.isnan(w_sum):
        w = np.full_like(conv, 1.0 / conv.size)
    else:
        w = exp_term / w_sum

    smooth = max_c + np.log(w_sum) / lam
    grad = 2.0 * _cross_correlation(w, a, corr_fft_len)
    return smooth, grad, max_c


def _exp_gradient_descent(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    lam_start: float = 0.5,
    lam_factor: float = 1.25,
    lam_cap: float = 200.0,
    base_eta: float = 0.8,
    min_iter: int = 5_000,
    max_iter: int = 120_000,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the smooth‑max surrogate.
    The iterate stays on the simplex (a ≥ 0, Σa = 1).
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a, conv_fft_len)

    lam = lam_start
    it = 0
    last_improve = time.time()

    while time.time() < deadline and it < max_iter:
        smooth, grad, _ = _smooth_max_and_grad(a, lam, conv_fft_len, corr_fft_len)

        # gradient of the *smoothed* objective f = 2·n·smooth (sum a = 1)
        grad_f = grad * (2.0 * n)

        # exponentiated‑gradient update (keep positivity)
        eta = base_eta / np.sqrt(it + 1)
        a = a * np.exp(-eta * grad_f)

        # renormalise onto the simplex
        s = a.sum()
        if s <= 1e-12:                     # numerical safe‑guard
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        # evaluate true (non‑smooth) objective
        val = _objective(a, conv_fft_len)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[{n}] exp‑grad it={it:06d} λ={lam:.2f} best={val:.6f}")

        # temperature schedule
        lam = min(lam * lam_factor, lam_cap)

        # early stop if no improvement for a while
        if time.time() - last_improve > 6.0 and it > min_iter:
            break

        it += 1

    return best_a, best_val


def _subgradient_refine(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iter: int = 30_000,
) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient steps on the true max‑convolution.
    Uses the active set (indices attaining the true maximum) as weights.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a, conv_fft_len)
    last_improve = time.time()
    it = 0

    while time.time() < deadline and it < max_iter:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[{n}] subgrad it={it:05d} best={val:.6f}")

        # active set where the convolution attains its maximum
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # exponentiated‑gradient step (keeps non‑negativity)
        eta = 0.6 / np.sqrt(it + 1)
        a = a * np.exp(-eta * grad)

        # renormalise
        s = a.sum()
        if s <= 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        if time.time() - last_improve > 8.0:
            break
        it += 1

    return best_a, best_val


def _deep_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    global_best_a: np.ndarray | None,
    global_best_val: float,
) -> tuple[np.ndarray, float]:
    """Run the two‑phase optimisation (smooth‑max + exact sub‑gradient)."""
    a, val = _exp_gradient_descent(
        a0,
        n,
        conv_fft_len,
        corr_fft_len,
        deadline,
    )
    if val < global_best_val:
        global_best_a, global_best_val = a.copy(), val

    a, val = _subgradient_refine(
        a,
        n,
        conv_fft_len,
        corr_fft_len,
        deadline,
    )
    if val < global_best_val:
        global_best_a, global_best_val = a.copy(), val

    return global_best_a, global_best_val

def _generate_parametric_candidates(n: int, warm_start: np.ndarray | None) -> list[np.ndarray]:
    """Create a rich set of symmetric candidates (all normalised to sum ≈ 1)."""
    candidates: list[np.ndarray] = []

    # warm‑start and simple shifts
    if warm_start is not None:
        ws = warm_start.copy()
        candidates.append(ws)
        for shift in (n // 4, n // 2, 3 * n // 4):
            candidates.append(np.roll(ws, shift))

    # uniform
    candidates.append(np.full(n, 1.0 / n))

    # random Dirichlet draws with various concentrations
    for conc in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        candidates.append(np.random.dirichlet(np.full(n, conc)))

    i = np.arange(n, dtype=np.float64)

    # edge exponential families
    for alpha in np.logspace(-2, 1.0, num=14):
        edge = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
        edge /= edge.sum()
        for w in (0.0, 0.25, 0.5, 0.75, 1.0):
            candidates.append(w * edge + (1.0 - w) / n)

    # edge power‑law families
    for p in np.linspace(0.2, 5.0, num=14):
        edge = (i + 1.0) ** (-p) + (n - i) ** (-p)
        edge /= edge.sum()
        for w in (0.0, 0.4, 0.8, 1.0):
            candidates.append(w * edge + (1.0 - w) / n)

    # symmetric beta‑shapes
    for alpha in (0.2, 0.4, 0.5, 0.8, 1.2, 2.0):
        shape = (i + 1.0) ** (alpha - 1.0) * (n - i) ** (alpha - 1.0)
        shape = np.maximum(shape, 0.0)
        shape /= shape.sum()
        candidates.append(shape)

    # raised‑cosine / sine‑power families
    for p in (1, 2, 4, 6):
        shape = np.sin(np.pi * (i + 0.5) / n) ** p
        shape = np.maximum(shape, 0.0)
        shape /= shape.sum()
        candidates.append(shape)

    # two‑Gaussian edge shapes
    for gamma in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        shape /= shape.sum()
        candidates.append(shape)

    # arcsine‑type shape
    shape = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    candidates.append(shape)

    return candidates


def _random_hill_climb(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_iter: int = 8_000,
) -> tuple[np.ndarray, float]:
    """
    Tiny random mass‑transfer moves, accepted only if they improve the true objective.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a, conv_fft_len)

    it = 0
    while time.time() < deadline - 1.0 and it < max_iter:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # move at most 25 % of the smaller bucket
        max_transfer = min(a[i], a[j])
        if max_transfer < 1e-12:
            continue
        delta = (np.random.rand() - 0.5) * 0.5 * max_transfer
        if delta > a[j]:
            delta = a[j]
        if -delta > a[i]:
            delta = -a[i]

        a[i] += delta
        a[j] -= delta

        val = _objective(a, conv_fft_len)
        if val < best_val:
            best_val = val
            best_a = a.copy()
        else:
            # revert
            a[i] -= delta
            a[j] += delta

        it += 1

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Entry point for the challenge.  Returns a list of non‑negative
    heights that (hopefully) yields a low value of ``evaluate_sequence``.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # a little safety margin

    # ------------------------------------------------------------------
    # Primary grid size (fast enough to explore many starts)
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm‑start from a previously discovered vector, if any
    warm_start = globals().get("height_sequence_1")
    if warm_start is not None:
        ws = np.asarray(warm_start, dtype=np.float64)
        if ws.shape[0] > n:
            ws = ws[:n]
        elif ws.shape[0] < n:
            ws = np.pad(ws, (0, n - ws.shape[0]))
        ws = np.clip(ws, 0.0, None)
        if ws.sum() >= 0.01:
            ws = ws / ws.sum()
        else:
            ws = None
    else:
        ws = None

    # ------------------------------------------------------------------
    # ---- Stage 1 : generate and quickly evaluate a large candidate pool ----
    cand_pool = _generate_parametric_candidates(n, ws)

    evaluated = []
    for idx, cand in enumerate(cand_pool):
        if time.time() >= deadline:
            break
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 0.01:
            cand = np.random.dirichlet(np.ones(n))
            s = cand.sum()
        cand = cand / s
        val = _objective(cand, conv_fft_len)
        evaluated.append((val, cand))
        # optional progress output
        # print(f"[candidate {idx}] val≈{val:.6f}")

    evaluated.sort(key=lambda x: x[0])
    top_k = min(6, len(evaluated))
    top_candidates = [c for _, c in evaluated[:top_k]]
    print(f"Stage 1: selected {top_k} promising starts (n={n})")

    # ------------------------------------------------------------------
    # ---- Stage 2 : deep optimisation on each selected start ---------------
    best_seq: np.ndarray | None = None
    best_val = float("inf")
    for start_id, a0 in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        print(f"--- Deep optimisation start #{start_id + 1}/{len(top_candidates)} ---")
        best_seq, best_val = _deep_optimize(
            a0,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            best_seq,
            best_val,
        )

    # ------------------------------------------------------------------
    # ---- Stage 3 : upscale to a larger grid if we still have time ---------
    time_left = deadline - time.time()
    if time_left > 30.0:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        if best_seq is not None:
            # linear interpolation to the finer grid
            x_old = np.linspace(0.0, 1.0, n, endpoint=False)
            x_new = np.linspace(0.0, 1.0, n_big, endpoint=False)
            a_big = np.interp(x_new, x_old, best_seq)
            a_big = np.maximum(a_big, 0.0)
            a_big = a_big / a_big.sum()
        else:
            a_big = np.random.dirichlet(np.ones(n_big))

        print("--- Optimising on larger grid (n=8192) ---")
        best_seq, best_val = _deep_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            best_seq,
            best_val,
        )
        # update bookkeeping for the final hill‑climb
        n = n_big
        conv_fft_len = conv_fft_len_big
        corr_fft_len = corr_fft_len_big

    # ------------------------------------------------------------------
    # ---- Final tiny random hill‑climb ------------------------------------
    if best_seq is not None and time.time() < deadline - 4.0:
        best_seq, best_val = _random_hill_climb(
            best_seq,
            n,
            conv_fft_len,
            deadline,
        )
        print(f"Final hill‑climb finished, best objective ≈ {best_val:.6f}")

    # ------------------------------------------------------------------
    # ---- Fallback in the unlikely event no candidate was produced -------
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Clip to the evaluator's allowed range and return a plain list
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    print(f"*** Search complete: best objective ≈ {best_val:.6f} (n={best_seq.shape[0]})")
    return best_seq.tolist()
