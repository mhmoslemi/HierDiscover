def _CE_optimize(n, K, conv_fft_len, deadline, max_iter=40, popsize=56, elite_frac=0.2):
    # Determine half length m = (n+1)//2. Precompute interpolation positions.
    m = (n+1)//2
    knot_positions = np.linspace(0, m-1, K)
    # Init mean and std
    mean = np.zeros(K)
    std = np.full(K, 1.0)  # log-space parameters

    best_val = float('inf')
    best_a = None

    for it in range(max_iter):
        if time.time() >= deadline:
            break
        # Sample
        p = np.random.randn(popsize, K) * std + mean
        # Evaluate each
        vals = np.empty(popsize)
        as_list = []
        for i in range(popsize):
            a = _shape_from_params(p[i], n, m, knot_positions)
            # Normalize
            a_sum = a.sum()
            if a_sum < 1e-12:
                continue
            a = a / a_sum
            conv = _full_convolution(a, conv_fft_len)
            max_c = conv.max()
            val = 2.0 * n * max_c
            vals[i] = val
            as_list.append((val, a))
        # sort
        idx = np.argsort(vals)
        elite_cnt = max(1, int(popsize * elite_frac))
        elite_idx = idx[:elite_cnt]
        elite_params = p[elite_idx]
        # update mean and std
        mean = np.mean(elite_params, axis=0)
        std = np.std(elite_params, axis=0) + 1e-6  # avoid zero

        # track best
        min_idx = idx[0]
        if vals[min_idx] < best_val:
            best_val = vals[min_idx]
            best_a = as_list[min_idx][1]  # produce correct mapping
            print(f"[CE it={it}] new best {best_val:.6f}")

    return best_a, best_val

def _subgradient_optimize_sym(a0, ...):
    a = np.maximum(a0,0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()
    # enforce symmetry
    a = (a + a[::-1]) / 2
    best_a = a.copy()
    best_val = float('inf')
    it=0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0*n*max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000==0:
                print(f"[SubgradSym it={it}] new best {val:.6f}")
        # compute gradient of max (w)
        active = np.isclose(conv, max_c, atol=1e-14)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        eta = eta0 / np.sqrt(it + 1)
        a = a * np.exp(-eta * grad_a)
        # Normalize and symmetrize
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s
        a = (a + a[::-1]) / 2
        it += 1
    return best_a, best_val

"""
Improved optimizer for the step‑function bound

The algorithm proceeds in three distinct phases.

1️⃣  **Low‑dimensional exploration** – a *cross‑entropy* (CE) optimisation
    over a small set of knots defines a symmetric, piecewise‑linear
    candidate on the full grid.  This cheap global search discovers
    promising shapes that are not covered by the hand‑crafted families
    used previously.

2️⃣  **Fine‑grained refinement** – the best CE candidate is fed to the
    exact‑max sub‑gradient (exponentiated‑gradient) scheme from the
    reference code.  Every iteration enforces symmetry
    `a ← (a + a[::-1]) / 2`, keeping the search inside the natural
    symmetric sub‑space.

3️⃣  **Cheap hill‑climbing** – finally a random‑move local search
    (again with symmetry projection) tries to squeeze out the last
    fraction of improvement.

All phases respect the global deadline (default 1000 s).  The
implementation uses only NumPy and the supplied `evaluate_sequence`
function, runs on a single core, and always returns a feasible
(non‑negative, normalised) sequence.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a length‑len(a) vector:
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i = 0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
def _shape_from_params(p: np.ndarray,
                       n: int,
                       half_len: int,
                       knot_pos: np.ndarray) -> np.ndarray:
    """
    Build a symmetric, piecewise‑linear vector of length n from
    log‑parameters `p` (size = number of knots).
    """
    # positive knot values
    knot_val = np.exp(p)

    # linear interpolation on the half‑grid
    x_half = np.arange(half_len, dtype=np.float64)
    half = np.interp(x_half, knot_pos, knot_val)
    half = np.maximum(half, 0.0)

    # mirror to the full grid (symmetry)
    if n % 2 == 0:
        a = np.concatenate([half, half[::-1]])
    else:
        a = np.concatenate([half, half[-2::-1]])
    return a[:n]


def _cross_entropy_optimize(n: int,
                            conv_fft_len: int,
                            deadline: float,
                            K: int = 32,
                            popsize: int = 64,
                            elite_frac: float = 0.2,
                            max_iters: int = 40,
                            warm: np.ndarray | None = None):
    """
    Cross‑entropy (CE) optimisation over a low‑dimensional knot
    representation.  Returns the best full‑length vector and its
    exact objective value.
    """
    half_len = n // 2 if n % 2 == 0 else n // 2 + 1
    knot_pos = np.linspace(0, half_len - 1, K)

    # initialise mean/std of log‑parameters
    if warm is not None:
        w = warm.copy()
        w = w / w.sum()
        # enforce symmetry for the warm start (helps the CE search)
        w = (w + w[::-1]) / 2
        half_w = w[:half_len]
        # sample the knot values from the warm shape (interpolation)
        knot_vals = np.interp(knot_pos, np.arange(half_len), half_w)
        mean = np.log(np.maximum(knot_vals, 1e-12))
    else:
        mean = np.zeros(K)

    std = np.full(K, 1.0)          # generous initial spread

    best_val = float("inf")
    best_a = None

    elite_sz = max(1, int(popsize * elite_frac))

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        # sample a population
        pop = np.random.randn(popsize, K) * std + mean
        vals = np.empty(popsize, dtype=np.float64)

        for i in range(popsize):
            a = _shape_from_params(pop[i], n, half_len, knot_pos)
            a_sum = a.sum()
            if a_sum < 1e-12:
                vals[i] = float("inf")
                continue
            a /= a_sum                     # simplex projection
            conv = _full_convolution(a, conv_fft_len)
            max_c = conv.max()
            vals[i] = 2.0 * n * max_c       # sum(a)=1 → exact objective

        # keep elite
        elite_idx = np.argsort(vals)[:elite_sz]
        elite = pop[elite_idx]

        # update distribution
        mean = elite.mean(axis=0)
        std = elite.std(axis=0) + 1e-8   # avoid collapse

        # remember global best
        best_idx = np.argmin(vals)
        if vals[best_idx] < best_val:
            best_val = float(vals[best_idx])
            best_a = _shape_from_params(pop[best_idx], n, half_len, knot_pos)
            best_a /= best_a.sum()
            print(f"[CE it={it:02d}] new best {best_val:.6f}")

    return best_a, best_val


def _subgradient_optimize_sym(a0: np.ndarray,
                              n: int,
                              conv_fft_len: int,
                              corr_fft_len: int,
                              deadline: float,
                              max_iters: int = 300_000,
                              eta0: float = 0.85):
    """
    Exact‑max sub‑gradient (exponentiated‑gradient) on the simplex,
    projecting onto the symmetric sub‑space after every update.
    Returns the best vector seen and its exact objective.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()
    # enforce symmetry immediately
    a = (a + a[::-1]) / 2

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 10_000 == 0:
                print(f"[SubgradSym it={it:06d}] new best {val:.6f}")

        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-14)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)
        a = a * np.exp(-eta * grad_a)

        # renormalise and re‑impose symmetry
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s
        a = (a + a[::-1]) / 2

        it += 1

    return best_a, best_val


def _local_random_search_sym(a0: np.ndarray,
                             n: int,
                             conv_fft_len: int,
                             deadline: float,
                             max_moves: int = 70_000):
    """
    Very cheap, symmetric hill‑climbing: move a random fraction of mass
    between two indices, keep the move if it improves the objective.
    After each accepted move the vector is symmetrised.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    best_a = a.copy()

    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n // 2)            # only need to pick in first half
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n // 2)
        if i == j:
            it += 1
            continue

        # move up to 30 % of the source mass
        delta = a[i] * (np.random.rand() * 0.3)

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta
        # symmetric copy for the mirrored half
        a2[n - 1 - i] = a2[i]
        a2[n - 1 - j] = a2[j]

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5_000 == 0:
                print(f"[LocalSym move={it:06d}] improved to {val2:.6f}")

        it += 1

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with the smallest evaluation
    score possible within the time budget.  The search consists of:
        * a low‑dimensional cross‑entropy global search,
        * a symmetric sub‑gradient refinement,
        * an optional cheap symmetric hill‑climb.
    The best vector is always returned.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    # problem size -------------------------------------------------------
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # warm start ---------------------------------------------------------
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None

    # ------------------------------------------------------------------
    # 1️⃣  Cross‑entropy global search over a symmetric piecewise‑linear
    #     representation (K knots on the first half of the domain).
    # ------------------------------------------------------------------
    print("[Phase 1] Cross‑entropy exploration …")
    best_seq, best_val = _cross_entropy_optimize(
        n=n,
        conv_fft_len=conv_fft_len,
        deadline=deadline,
        K=32,                           # number of knots (adjustable)
        popsize=64,
        elite_frac=0.2,
        max_iters=40,
        warm=warm,
    )
    if best_seq is None:
        # fallback to a uniform distribution
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = float("inf")

    # ------------------------------------------------------------------
    # 2️⃣  Sub‑gradient refinement (symmetrised)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        print("[Phase 2] Symmetric sub‑gradient refinement …")
        best_seq, best_val = _subgradient_optimize_sym(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=250_000,
            eta0=0.85,
        )
        print(f"[Phase 2] best value after refinement {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Cheap symmetric hill‑climbing (optional)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        print("[Phase 3] Symmetric local random search …")
        best_seq, best_val = _local_random_search_sym(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=70_000,
        )
        print(f"[Phase 3] best value after local search {best_val:.6f}")

    # ------------------------------------------------------------------
    # Ensure a valid return ------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the evaluator’s accepted range (non‑negative, ≤1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
