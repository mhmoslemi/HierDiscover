def _project_onto_simplex(v):
    """Projection of vector v onto the probability simplex."""
    # from Duchi et al. 2008
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0)
    return w

def _proj_simplex(v):
    # v: 1D array
    # onto simplex sum=1, nonnegative
    # using algorithm by Duchi et al.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, len(v)+1) > 0)[0]
    if len(rho) == 0:
        # all negative, return uniform?
        return np.full_like(v, 1.0/len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def _generate_candidates(n, warm):

    cand = []

    # warm start (if any)
    if warm is not None:
        cand.append(warm.copy())
        # rotations to diversify (although shape might be periodic)
        shifts = (n // 4, n // 2, 3 * n // 4)
        for shift in shifts:
            cand.append(np.roll(warm, shift))

    # Uniform
    cand.append(np.full(n, 1.0 / n))

    # Dirichlet random variations (maybe 10)
    for _ in range(5):
        cand.append(np.random.dirichlet(np.ones(n)))

    # Arcsine shape
    i = np.arange(n, dtype=np.float64) + 0.5
    arcsine = 1.0 / np.sqrt(i * (n + 0.5 - i))
    arcsine /= arcsine.sum()
    cand.append(arcsine)

    # Beta shapes: exponent alpha (shape parameter)
    for alpha in np.linspace(0.2, 2.0, 19):
        # shape = (i*(n+1-i))^(alpha-1)
        shape = np.power(i * (n + 0.5 - i), alpha - 1.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    # Double exponential (laplace-like) from center
    center = (n - 1) / 2.0
    for gamma in np.geomspace(0.01, 5.0, 8):
        shape = np.exp(-gamma * np.abs(i - (center+0.5)))  # shift
        shape /= shape.sum()
        cand.append(shape)

    # Sinusoid half-wave
    sinus = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    sinus = np.maximum(sinus, 0.0)
    sinus /= sinus.sum()
    cand.append(sinus)

    # Spike (spike pattern) as before
    for k in (16, 32, 64, 128, 256):
        a = np.zeros(n, dtype=np.float64)
        pos = np.linspace(0, n-1, k).astype(int)
        a[pos] = 1.0 / k
        cand.append(a)

    return cand

"""
search_step_function.py

Search for a non‑negative step function (vector) that minimizes

        F(a) = 2·n·max(a∗a) / (sum(a)^2)

The algorithm proceeds in a few lightweight stages:

1.  Build a diverse set of deterministic “shape’’ candidates
    (uniform, Dirichlet, Beta‑type, double‑exponential,
    half‑wave sinusoid, arcsine, sparse spikes, rotations of a warm start,
    …).

2.  Evaluate every candidate with an exact convolution (FFT based) and
    keep the best few (default 20).

3.  Refine each of the survivors with a fast exponentiated‑gradient
    sub‑gradient method that directly minimises the true max‑convolution.
    The step size decays as η₀/√t and a tiny random shake is injected when
    progress stalls.

4.  If time permits, up‑sample the current best sequence to a finer grid
    (8192 → 16384 points) using linear interpolation and repeat the
    refinement there.

All work respects the global time budget (default 1000 s) and the routine
always returns the best vector found as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two greater than or equal to ``x``."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2·len(a)‑1) via FFT.
    The result is truncated to the exact length.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a length‑len(a) vector:
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i = 0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


def _objective_value(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact objective 2·n·max(conv) / (sum(a)**2).  a is assumed non‑negative."""
    if a.sum() < 1e-12:
        return float("inf")
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    return 2.0 * n * max_c / (a.sum() ** 2)


def _proj_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        {x | x_i ≥ 0, Σ_i x_i = 1}
    Implementation of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries will be zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max() - 1  # zero‑based index
    theta = cssv[cond].max() / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # Numerical safety: re‑normalise to sum exactly 1
    w_sum = w.sum()
    if w_sum > 0:
        w /= w_sum
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _expgrad_subgradient_opt(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 0.7,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient descent on the true max‑convolution.
    Returns the best vector seen together with its exact objective.
    """
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s

    best_a = a.copy()
    best_val = _objective_value(a, n, conv_fft_len)

    stall = 0
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c  # sum(a)==1 by construction
        if val < best_val:
            best_val = val
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-14)
        if not active.any():
            w = np.zeros_like(conv)
        else:
            w = np.zeros_like(conv)
            w[active] = 1.0 / np.count_nonzero(active)

        # gradient of the true max (chain rule)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # decaying step size
        eta = eta0 / math.sqrt(it + 1)

        # multiplicative (exponentiated) update + renormalisation
        a = a * np.exp(-eta * grad_a)
        a_sum = a.sum()
        if a_sum < 1e-12:
            a = np.random.dirichlet(np.ones(n))
        else:
            a = a / a_sum

        # occasional random shake when we seem stuck
        if stall > 8_000:
            a = np.maximum(a + 0.01 * (np.random.rand(n) - 0.5), 0.0)
            a = a / a.sum()
            stall = 0

        it += 1
    return best_a, best_val


def _generate_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """
    Produce a modest but diverse collection of start vectors.
    """
    cand = []

    # Warm start (if available) and a few rotations
    if warm is not None:
        cand.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    # Uniform
    cand.append(np.full(n, 1.0 / n))

    # Random Dirichlet (few samples – cheap)
    for _ in range(6):
        cand.append(np.random.dirichlet(np.ones(n)))

    # Half‑wave sinusoid
    s = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    s = np.maximum(s, 0.0)
    s /= s.sum()
    cand.append(s)

    # Arcsine‑type (Beta(0.5,0.5)) – a classic low‑autocorrelation shape
    i = np.arange(n, dtype=np.float64) + 0.5
    arcsine = 1.0 / np.sqrt(i * (n + 0.5 - i))
    arcsine /= arcsine.sum()
    cand.append(arcsine)

    # Beta‑type families  (α in [0.2, 2.0])
    for alpha in np.linspace(0.2, 2.0, 19):
        shape = np.power(i * (n + 0.5 - i), alpha - 1.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    # Double‑exponential (Laplace‑like) from the centre
    centre = (n - 1) / 2.0
    for gamma in np.geomspace(0.02, 5.0, 9):
        shape = np.exp(-gamma * np.abs(i - (centre + 0.5)))
        shape /= shape.sum()
        cand.append(shape)

    # Sparse “spike’’ patterns (k equally spaced non‑zero entries)
    for k in (16, 32, 64, 128, 256):
        a = np.zeros(n, dtype=np.float64)
        pos = np.linspace(0, n - 1, k).astype(int)
        a[pos] = 1.0 / k
        cand.append(a)

    return cand


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The function respects the given time budget and returns the best
    sequence (as a Python list) found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # small safety margin

    # Base resolution – 8192 points give a good trade‑off between quality
    # and speed (FFT length = 16384).
    n = 8192
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm start from a previous solution, if available
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] != n:
            # linear interpolation to the target length
            x_old = np.linspace(0, 1, warm.shape[0], endpoint=False)
            x_new = np.linspace(0, 1, n, endpoint=False)
            warm = np.interp(x_new, x_old, warm)
        warm = np.maximum(warm, 0.0)
        total = warm.sum()
        if total > 1e-12:
            warm = warm / total
        else:
            warm = None
    else:
        warm = None

    # ------------------------------------------------------------------
    # 1 – generate a pool of diverse candidates
    candidates = _generate_candidates(n, warm)

    # 2 – evaluate all candidates, keep the best few
    evals = []
    for a in candidates:
        val = _objective_value(a, n, conv_fft_len)
        evals.append((val, a))
    evals.sort(key=lambda x: x[0])
    top_k = 20
    top_cands = [a for _, a in evals[:top_k]]
    best_val = evals[0][0]
    best_seq = evals[0][1].copy()
    print(f"[Init] best of pool = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3 – refine each top candidate with exponentiated‑gradient sub‑gradient
    for idx, cand in enumerate(top_cands):
        if time.time() >= deadline - 5.0:
            break
        # allocate a modest slice of the remaining time to each candidate
        sub_deadline = min(deadline - 0.5, time.time() + 12.0)
        a_refined, val_ref = _expgrad_subgradient_opt(
            cand, n, conv_fft_len, corr_fft_len, sub_deadline,
            max_iters=200_000, eta0=0.8
        )
        if val_ref < best_val:
            best_val = val_ref
            best_seq = a_refined.copy()
            print(f"[Refine #{idx}] new best {val_ref:.6f}")

    # ------------------------------------------------------------------
    # 4 – cheap random local moves (hill‑climbing with simulated‑annealing
    #     acceptance) to squeeze a little more out of the current best.
    if time.time() < deadline - 5.0:
        a = best_seq.copy()
        conv = _full_convolution(a, conv_fft_len)
        best_local_val = 2.0 * n * conv.max()
        best_local_a = a.copy()
        moves = 250_000
        it = 0
        stall = 0
        while time.time() < deadline and it < moves:
            i = np.random.randint(n)
            if a[i] < 1e-12:
                it += 1
                continue
            j = np.random.randint(n)
            if i == j:
                it += 1
                continue
            delta = a[i] * (np.random.rand() * 0.4)   # move up to 40 % of source mass
            a2 = a.copy()
            a2[i] -= delta
            a2[j] += delta
            conv2 = _full_convolution(a2, conv_fft_len)
            val2 = 2.0 * n * conv2.max()
            # simulated‑annealing acceptance
            if val2 < best_local_val or np.random.rand() < math.exp(-(val2 - best_local_val) / (0.02 + 0.001 * stall)):
                if val2 < best_local_val:
                    best_local_val = val2
                    best_local_a = a2.copy()
                    stall = 0
                else:
                    stall += 1
                a = a2
            else:
                stall += 1
            if it % 50_000 == 0:
                print(f"[Local move {it}] current best {best_local_val:.6f}")
            it += 1
        if best_local_val < best_val:
            best_val = best_local_val
            best_seq = best_local_a.copy()
            print(f"[Local] improved best to {best_local_val:.6f}")

    # ------------------------------------------------------------------
    # 5 – up‑sample to 16384 points (if time permits) and repeat the refinement.
    if time.time() + 30 < deadline:
        n_big = 16384
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        x_new = np.linspace(0, 1, n_big, endpoint=False)
        a_big = np.interp(x_new, x_old, best_seq)
        a_big = np.maximum(a_big, 0.0)
        a_big /= a_big.sum()

        # refinement on the larger grid

            a_big, n_big, conv_fft_len_big, corr_fft_len_big,
            big_deadline, max_iters=250_000, eta0=0.7
        )
        if val_big < best_val:
            best_val = val_big
            best_seq = a_big_ref.copy()
            print(f"[Big‑16384] refined best {val_big:.6f}")

    # ------------------------------------------------------------------
    # 6 – final safety clipping (the evaluator also clips to [0,1000])
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # sanity check through the official evaluator (optional, cheap)
    try:

        print(f"[Final] evaluator reports {final_check:.6f}")
    except Exception:
        # if evaluate_sequence is not available in this namespace we ignore it
        pass

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
