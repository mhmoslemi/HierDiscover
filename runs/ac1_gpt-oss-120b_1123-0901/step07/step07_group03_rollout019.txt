def surrogate_and_grad(logits, lam, conv_len):
    a = softmax(logits)
    conv = _full_convolution(a, conv_len)
    # compute max conv approx (smooth)
    # Actually we compute weight w as softmax lam*conv
    max_c = conv.max()
    exp_term = np.exp(lam * (conv - max_c))   # subtract max for stability
    w = exp_term / exp_term.sum()
    # gradient w.r.t. a
    grad_a = 2.0 * _cross_correlation(w, a, corr_len) # need appropriate corr_len
     (use same conv_len? maybe separate.)
    # gradient w.r.t. logits
    avg = np.dot(a, grad_a)  # dot product
    grad_log = a * (grad_a - avg)
    # surrogate value: (1/lam) * log(sum exp(lam*(conv - max_c))) + max_c? Actually we need original shape:
    # S = (1/lam) * log( sum_i exp(lam * conv_i) )
    # Since we may shift conv, we can compute as max_c + (1/lam) * log(sum_i exp(lam*(conv_i - max_c))) 
    # = max_c + (1/lam) * log(exp_sum)
    exp_sum = exp_term.sum()
    surrogate = max_c + (1.0/lam) * np.log(exp_sum)
    return surrogate, grad_log

Note: lam may be high, but if lam huge, exp terms may overflow; we subtract max for stability.

def _lbfgs_optimize(a0, n, conv_fft_len, corr_fft_len, deadline, max_iter=2000, lam=1.0, lam_factor=1.07):

    best_a = a0.copy()
    best_val = _objective(a0)
    iters_done = 0

    def func(logits_vec):
        nonlocal best_a, best_val, iters_done
        if time.time() >= deadline:

"""
Propose a low‑value step‑function.

The algorithm proceeds in several stages within the 1000 s budget:

1. **Warm‑start & diverse pool** – If a previous solution exists it is
   included along with many Dirichlet‑sampled candidates, anti‑pair
   ordered, Gaussian‑edge and simple tent shapes.  The best of these is
   selected as the initial point.

2. **Smooth‑max Adam** – Optimise a log‑sum‑exp surrogate of
   ``max(a*a)`` using Adam on the logits of a soft‑max (ensuring non‑negativity
   and unit sum).  The temperature λ is increased exponentially so the
   surrogate sharpens to the true max.

3. **L‑BFGS refinement** – A few thousand L‑BFGS‑B steps are applied to the
   same surrogate (again on the logits).  This gives a more precise
   “second‑order” push towards a local optimum.

4. **Exact‑max sub‑gradient** – Exponentiated‑gradient updates are performed
   on the exact max (active set) for the remainder of the time.

5. **Stochastic hill‑climbing** – A simulated‑annealing style local search
   (swaps and fractional mass transfers) further polishes the solution.

6. **Rotation search** – The final vector is rotated by a few random offsets;
   the orientation with the smallest exact objective is kept.

7. **Optional upscale** – If time permits the solution is interpolated to a
   finer grid (8192 points) and refined once more with Adam + sub‑gradient.

All operations respect the security checks of ``evaluate_sequence`` (non‑negative,
clipped to [0,1000]), and the best sequence found so far is always returned
before the deadline.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


# ----------------------------------------------------------------------
# Objective (exact, no safety checks)

def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(a*a) / (sum(a)^2).  Assumes a is non‑negative."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    conv = _full_convolution(a, _next_pow2(2 * a.shape[0] - 1))
    return 2.0 * a.shape[0] * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Simple helpers

def _softmax(logits: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (produces a probability vector)."""
    m = logits.max()
    e = np.exp(logits - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# Candidate generation

def _warm_start(n: int):
    """If a global ``height_sequence_1`` exists, adapt it to length ``n``."""
    warm = globals().get("height_sequence_1")
    if warm is None:
        return None
    warm = np.asarray(warm, dtype=np.float64)
    if warm.shape[0] > n:
        warm = warm[:n]
    elif warm.shape[0] < n:
        warm = np.pad(warm, (0, n - warm.shape[0]))
    warm = np.maximum(warm, 0.0)
    if warm.sum() >= 0.01:
        warm = warm / warm.sum()
        return warm
    return None


def _uniform_candidate(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _random_dirichlet_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    return np.random.dirichlet(alpha * np.ones(n)).astype(np.float64)


def _anti_pair_arrangement(vals: np.ndarray) -> np.ndarray:
    """Place largest opposite smallest (the ‘anti‑pair’ ordering)."""
    n = len(vals)
    sorted_vals = np.sort(vals)[::-1]          # descending
    a = np.empty(n, dtype=np.float64)
    half = n // 2
    a[:half] = sorted_vals[:half]             # biggest half on the left
    a[-half:] = sorted_vals[-half:][::-1]     # smallest half on the right
    if n % 2:
        a[half] = sorted_vals[half]
    return a


def _anti_pair_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    vals = np.random.dirichlet(alpha * np.ones(n))
    vals = np.maximum(vals, 0.0)
    vals = vals / vals.sum()
    return _anti_pair_arrangement(vals)


def _triangle_candidate(n: int) -> np.ndarray:
    """Tent shape (small in the middle, large toward the ends)."""
    x = np.arange(n, dtype=np.float64)
    shape = np.minimum(x + 1, n - x)          # 1,2,…,⌈n/2⌉,…,2,1
    shape = shape / shape.sum()
    return shape


def _gaussian_edges_candidate(n: int, gamma: float = 0.5) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
    shape = shape / shape.sum()
    return shape

# Optimisation primitives

def _adam_optimize(a0: np.ndarray, n: int,
                   conv_fft_len: int, corr_fft_len: int,
                   deadline: float,
                   max_iters: int = 300_000,
                   lr: float = 0.06,
                   lam_start: float = 1.0,
                   lam_factor: float = 1.07) -> tuple[np.ndarray, float]:
    """Adam on the smooth‑max (log‑sum‑exp) surrogate."""
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lam = lam_start

    best_a = a0.copy()
    best_val = _objective(a0)

    it = 0
    while time.time() < deadline and it < max_iters:
        a = _softmax(logits)

        # surrogate data
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()

        # track exact objective
        val_exact = 2.0 * n * max_c
        if val_exact < best_val:
            best_val = val_exact
            best_a = a.copy()
            if (it % 5000) == 0:
                print(f"[Adam it={it:06d}] new best exact {best_val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient w.r.t. a
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature schedule → true max
        lam = min(lam * lam_factor, 1e6)

        # occasional jitter
        if it % 4000 == 0 and (deadline - time.time()) > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2

        it += 1

    return best_a, best_val


def _smooth_max_obj(logits: np.ndarray, lam: float,
                    conv_fft_len: int, corr_fft_len: int) -> tuple[float, np.ndarray]:
    """
    Smooth‑max surrogate and its gradient w.r.t. logits.
    Returns (objective, grad_logit) where grad_logit has same shape as logits.
    """
    a = _softmax(logits)
    conv = _full_convolution(a, conv_fft_len)
    mx = conv.max()
    # stable soft‑max over the *shifted* values
    exp_term = np.exp(lam * (conv - mx))
    sumexp = exp_term.sum()
    # smooth max value
    obj = mx + (1.0 / lam) * np.log(sumexp)

    w = exp_term / sumexp
    grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
    avg = np.dot(a, grad_a)
    grad_log = a * (grad_a - avg)
    return obj, grad_log.astype(np.float64)


def _lbfgs_optimize(a0: np.ndarray, n: int,
                    conv_fft_len: int, corr_fft_len: int,
                    deadline: float,
                    max_iter: int = 2000,
                    lam: float = 1.0,
                    lam_factor: float = 1.07) -> tuple[np.ndarray, float]:
    """
    Limited‑memory BFGS on the smooth‑max surrogate.
    Optimises the logits (unconstrained) while keeping the simplex constraint via soft‑max.
    """
    try:
        import scipy.optimize
    except Exception as exc:
        print("[LBFGS] scipy not available – skipping.")
        return a0, _objective(a0)

    logits = np.log(a0 + 1e-15)
    best_a = a0.copy()
    best_val = _objective(a0)

    # wrapper that raises when we run out of time
    def fun(x):
        nonlocal best_a, best_val
        if time.time() >= deadline:
            raise KeyboardInterrupt
        obj, grad = _smooth_max_obj(x, lam, conv_fft_len, corr_fft_len)
        # track exact value occasionally
        a = _softmax(x)
        val = _objective(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[LBFGS] new best exact {val:.6f}")
        return float(obj), grad

    try:
        res = scipy.optimize.fmin_l_bfgs_b(
            func=fun,
            x0=logits,
            maxiter=max_iter,
            disp=0,
        )
        logits_opt = res[0]
        best_a = _softmax(logits_opt)
        best_val = _objective(best_a)
    except KeyboardInterrupt:
        # time limit hit – keep the best found so far
        pass
    except Exception as e:
        print(f"[LBFGS] warning: {e}")

    return best_a, best_val


def _subgradient_optimize(a0: np.ndarray, n: int,
                          conv_fft_len: int, corr_fft_len: int,
                          deadline: float,
                          max_iters: int = 500_000,
                          eta0: float = 0.9) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient (exponentiated‑gradient) phase.
    """
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s

    best_a = a.copy()
    best_val = _objective(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if (it % 5000) == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / math.sqrt(it + 1)

        a = a * np.exp(-eta * grad_a)

        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s
        it += 1

    return best_a, best_val


def _local_random_search(a0: np.ndarray, n: int,
                         conv_fft_len: int,
                         deadline: float,
                         max_moves: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing style hill‑climbing (swap or mass transfer).
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a)

    start_temp = 0.012
    it = 0
    while time.time() < deadline and it < max_moves:
        # move type
        if np.random.rand() < 0.5:           # swap
            i, j = np.random.randint(n, size=2)
            if i != j:
                a2 = a.copy()
                a2[i], a2[j] = a2[j], a2[i]
            else:
                continue
        else:                                # transfer a fraction
            i = np.random.randint(n)
            if a[i] < 1e-16:
                continue
            j = np.random.randint(n)
            if i == j:
                continue
            delta = a[i] * (np.random.rand() * 0.45)   # up to 45 % move
            a2 = a.copy()
            a2[i] -= delta
            a2[j] += delta

        val2 = _objective(a2)
        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
        else:
            # linear temperature schedule
            temp = start_temp * (1.0 - it / max_moves)
            if temp > 0.0 and np.random.rand() < math.exp(-(val2 - best_val) / temp):
                a = a2

        if (it % 5000) == 0:
            print(f"[Local move {it:06d}] best {best_val:.6f}")
        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Rotation search – try a few random cyclic shifts

def _rotation_search(a: np.ndarray, n: int, best_val: float, trials: int = 12) -> tuple[np.ndarray, float]:
    """Return the best rotation of ``a`` among ``trials`` random shifts."""
    best_seq = a
    for _ in range(trials):
        shift = np.random.randint(0, n)
        cand = np.roll(a, shift)
        val = _objective(cand)
        if val < best_val:
            best_val = val
            best_seq = cand
            print(f"[Rotate shift={shift}] new best {val:.6f}")
    return best_seq, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    F(a)=2·n·max(a∗a)/(sum(a)²) within the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Problem size – 4096 is a good trade‑off between quality and speed.
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # 1) Build a pool of starter vectors
    candidates = []

    warm = _warm_start(n)
    if warm is not None:
        candidates.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            candidates.append(np.roll(warm, shift))

    candidates.append(_uniform_candidate(n))

    for alpha in (0.2, 0.5, 0.9, 1.5, 2.5):
        candidates.append(_random_dirichlet_candidate(n, alpha))
        candidates.append(_anti_pair_candidate(n, alpha))

    for gamma in (0.15, 0.35, 0.6, 1.0):
        candidates.append(_gaussian_edges_candidate(n, gamma))

    candidates.append(_triangle_candidate(n))

    # ------------------------------------------------------------------
    # 2) Quick evaluation of the pool – keep the best starter
    best_seq = None
    best_val = np.inf
    for idx, cand in enumerate(candidates):
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 0.01:
            cand = np.random.dirichlet(np.ones(n))
            s = cand.sum()
        cand = cand / s
        val = _objective(cand)
        print(f"[Init cand {idx:02d}] value {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = cand.copy()

    # ------------------------------------------------------------------
    # 3) Optimisation phases (time‑aware)

    # Reserve a little slack for the final upscale (if any)
    final_upscale_deadline = deadline - 25.0

    # ---- Phase A: Adam on smooth‑max surrogate
    phase_a_deadline = min(deadline - 20.0, time.time() + 0.25 * (deadline - time.time()))
    a_adam, val_adam = _adam_optimize(
        best_seq, n, conv_fft_len, corr_fft_len,
        phase_a_deadline,
        max_iters=300_000,
        lr=0.06,
        lam_start=1.0,
        lam_factor=1.07,
    )
    if val_adam < best_val:
        best_val = val_adam
        best_seq = a_adam.copy()
        print(f"[Adam] improved to {val_adam:.6f}")

    # ---- Phase B: L‑BFGS refinement (sharpens the surrogate)
    phase_b_deadline = min(deadline - 15.0, time.time() + 0.15 * (deadline - time.time()))
    a_lbfgs, val_lbfgs = _lbfgs_optimize(
        best_seq, n, conv_fft_len, corr_fft_len,
        phase_b_deadline,
        max_iter=2500,
        lam=1.0,
        lam_factor=1.07,
    )
    if val_lbfgs < best_val:
        best_val = val_lbfgs
        best_seq = a_lbfgs.copy()
        print(f"[LBFGS] improved to {val_lbfgs:.6f}")

    # ---- Phase C: Exact‑max sub‑gradient (exponentiated gradient)
    phase_c_deadline = min(deadline - 8.0, time.time() + 0.30 * (deadline - time.time()))
    a_sub, val_sub = _subgradient_optimize(
        best_seq, n, conv_fft_len, corr_fft_len,
        phase_c_deadline,
        max_iters=600_000,
        eta0=0.9,
    )
    if val_sub < best_val:
        best_val = val_sub
        best_seq = a_sub.copy()
        print(f"[Subgrad] improved to {val_sub:.6f}")

    # ---- Phase D: Local stochastic hill‑climbing
    phase_d_deadline = min(deadline - 4.0, time.time() + 0.20 * (deadline - time.time()))
    a_local, val_local = _local_random_search(
        best_seq, n, conv_fft_len,
        phase_d_deadline,
        max_moves=250_000,
    )
    if val_local < best_val:
        best_val = val_local
        best_seq = a_local.copy()
        print(f"[Local] improved to {val_local:.6f}")

    # ---- Phase E: Random rotations (very cheap)
    best_seq, best_val = _rotation_search(best_seq, n, best_val, trials=12)

    # ------------------------------------------------------------------
    # 4) Optional upscale to a finer grid (8192) if time permits
    if time.time() < final_upscale_deadline:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        # Linear interpolation
        x_old = np.linspace(0.0, 1.0, n, endpoint=False)
        x_new = np.linspace(0.0, 1.0, n_big, endpoint=False)
        a_big = np.interp(x_new, x_old, best_seq)

        a_big = np.maximum(a_big, 0.0)
        a_big = a_big / a_big.sum()

        # Brief Adam on large grid
        big_adam_deadline = min(deadline - 2.0, time.time() + 0.12 * (deadline - time.time()))
        a_big_adam, val_big_adam = _adam_optimize(
            a_big, n_big, conv_fft_len_big, corr_fft_len_big,
            big_adam_deadline,
            max_iters=150_000,
            lr=0.07,
            lam_start=1.0,
            lam_factor=1.07,
        )
        # Sub‑gradient on large grid
        big_sub_deadline = min(deadline, time.time() + 0.10 * (deadline - time.time()))
        a_big_sub, val_big_sub = _subgradient_optimize(
            a_big_adam, n_big, conv_fft_len_big, corr_fft_len_big,
            big_sub_deadline,
            max_iters=250_000,
            eta0=0.9,
        )
        if val_big_sub < best_val:
            best_val = val_big_sub
            best_seq = a_big_sub.copy()
            print(f"[Big] improved to {val_big_sub:.6f}")

        # Final tiny local search on the big grid (if any time left)
        if time.time() < deadline - 1.0:
            big_local_deadline = deadline - 0.5
            a_big_local, val_big_local = _local_random_search(
                best_seq, n_big, conv_fft_len_big,
                big_local_deadline,
                max_moves=120_000,
            )
            if val_big_local < best_val:
                best_val = val_big_local
                best_seq = a_big_local.copy()
                print(f"[Big local] improved to {val_big_local:.6f}")

    # ------------------------------------------------------------------
    # 5) Safeguard – guarantee a valid return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Optional: report final value via the provided evaluator (for debugging)
    # Uncomment the line below if you want to see the exact score.
    # print("Final exact value:", evaluate_sequence(best_seq.tolist()))

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
