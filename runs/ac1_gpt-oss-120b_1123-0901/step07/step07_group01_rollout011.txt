"""
Hybrid random‑search + long‑run Adam / sub‑gradient / annealing.

* Generate many diverse parametric families (Beta, power‑law, edge‑Gaussians,
  Dirichlet) and keep the top‑k (k≈10) after a cheap FFT evaluation.
* Starting from the current best candidate run a *long* Adam optimisation on a
  smooth‑max (log‑sum‑exp) surrogate, then an exact‑max exponentiated‑gradient
  sub‑gradient descent.
* Finish with a slower simulated‑annealing local search.
* If time permits upscale to a finer grid (8192 points) and repeat a short
  refinement pass.
* All steps respect the 1000 s budget and the sequence is returned as a plain
  list of non‑negative floats.
"""

import time
import numpy as np
from operator import itemgetter

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for the gradient."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta(α,β) density on a regular grid, normalized to sum‑1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    shape = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    # Guard against NaNs (should not happen with the grid we use)
    shape = np.where(np.isnan(shape), 0.0, shape)
    total = shape.sum()
    if total <= 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return shape / total


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Return the value of 2·n·max(a∗a) (a is normalized inside)."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return np.inf
    a = a / s
    conv = _full_convolution(a, conv_fft_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   conv_fft_len: int,
                   corr_fft_len: int,
                   deadline: float,
                   max_iters: int = 400_000) -> tuple[np.ndarray, float]:
    """Adam on the log‑sum‑exp surrogate (smooth max)."""
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.025               # a little smaller to stay stable for long runs
    lam = 0.2                # start temperature
    lam_factor = 1.012       # anneal slowly

    best_a = a0.copy()
    best_val = np.inf
    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max probabilities
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w[:] = 1.0 / w.size
        else:
            w /= w_sum

        # gradient of the surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        lam = min(lam * lam_factor, 1e6)

        # occasional jitter when time is almost up
        if it % 5000 == 0 and time.time() > deadline - 10:
            logits += (np.random.rand(n) - 0.5) * 1e-4

        it += 1

    return best_a, best_val


def _subgradient_optimize(a0: np.ndarray,
                          n: int,
                          conv_fft_len: int,
                          corr_fft_len: int,
                          deadline: float,
                          max_iters: int = 400_000,
                          eta0: float = 0.55) -> tuple[np.ndarray, float]:
    """Exact‑max sub‑gradient descent with exponentiated‑gradient updates."""
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s

    best_a = a.copy()
    best_val = np.inf
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 2000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active set (indices where conv ≈ max)
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # diminishing step‑size
        eta = eta0 / np.sqrt(it + 1)

        # exponentiated‑gradient update
        a = a * np.exp(-eta * grad_a)

        # re‑normalise onto simplex
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray,
                         n: int,
                         fft_len: int,
                         deadline: float,
                         cur_best_val: float) -> tuple[np.ndarray, float]:
    """Fast O(n) SA that updates the convolution incrementally."""
    a = a0.copy()
    conv = _full_convolution(a, fft_len)
    cur_max = conv.max()
    cur_val = 2.0 * n * cur_max
    best_a = a.copy()
    best_val = cur_val

    start = time.time()
    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start
        fraction = elapsed / (deadline - start + 1e-12)

        # linear cooling schedule
        T = max(1e-6, 0.12 * (1.0 - fraction))
        move_scale = max(0.008, 0.25 * (1.0 - fraction))

        if a.sum() == 0.0:
            break
        # pick source proportional to current mass
        i = np.random.choice(n, p=a / a.sum())
        if a[i] < 1e-12:
            continue
        # pick distinct target uniformly
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * (np.random.rand() * move_scale)

        a_new = a.copy()
        conv_new = conv.copy()
        a_new[i] -= delta
        a_new[j] += delta

        # update convolution (exact linear update)
        conv_new[i : i + n] -= 2.0 * delta * a
        conv_new[j : j + n] += 2.0 * delta * a
        conv_new[2 * i] += delta * delta
        conv_new[2 * j] += delta * delta
        conv_new[i + j] -= 2.0 * delta * delta

        new_max = conv_new.max()
        new_val = 2.0 * n * new_max

        # Metropolis acceptance
        if new_val < cur_val or np.random.rand() < np.exp(-(new_val - cur_val) / T):
            a, conv = a_new, conv_new
            cur_val, cur_max = new_val, new_max
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[SA it={it:06d}] new best {best_val:.6f}")

        it += 1
        if it % 6000 == 0:
            print(f"[SA it={it:06d}] best so far {best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
def _interpolate_and_refine(a_small: np.ndarray,
                            n_big: int,
                            deadline: float) -> tuple[np.ndarray, float]:
    """Interpolate to a larger grid and run a short Adam + sub‑gradient pass."""
    n_small = a_small.shape[0]
    x_small = np.linspace(0.0, 1.0, n_small, endpoint=False)
    x_big = np.linspace(0.0, 1.0, n_big, endpoint=False)
    a_big = np.interp(x_big, x_small, a_small)
    a_big = np.maximum(a_big, 0.0)
    a_big /= a_big.sum()

    conv_fft_len_big = _next_pow2(2 * n_big - 1)
    corr_fft_len_big = _next_pow2(3 * n_big - 2)

    a_adam, _ = _adam_optimize(a_big, n_big,
                               conv_fft_len_big,
                               corr_fft_len_big,
                               deadline,
                               max_iters=80_000)
    a_ref, val_ref = _subgradient_optimize(a_adam, n_big,
                                           conv_fft_len_big,
                                           corr_fft_len_big,
                                           deadline,
                                           max_iters=100_000)
    return a_ref, val_ref


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises
        F(a) = 2·n·max(a∗a) / (sum a)² .
    The routine builds a large random pool of parametric families, keeps the
    best few, then runs a long Adam optimisation, an exact‑max sub‑gradient
    descent and finally a slow simulated‑annealing local search.  If time
    permits the solution is interpolated to a finer grid (8192 points) and
    refined again.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5        # tiny safety margin

    # --------------------------------------------------------------
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # optional warm start from previous run
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # --------------------------------------------------------------
    # keep the top‑k candidates after cheap evaluation
    top_k = 10
    best_cands: list[tuple[float, np.ndarray]] = []

    def _maybe_insert(val: float, a: np.ndarray):
        if not np.isfinite(val):
            return
        if len(best_cands) < top_k:
            best_cands.append((val, a.copy()))
            best_cands.sort(key=itemgetter(0))
        else:
            if val < best_cands[-1][0]:
                best_cands[-1] = (val, a.copy())
                best_cands.sort(key=itemgetter(0))

    # Warm start
    if warm is not None:
        v = _objective(warm, n, conv_fft_len)
        _maybe_insert(v, warm)

    # Deterministic baselines
    uniform = np.full(n, 1.0 / n)
    _maybe_insert(_objective(uniform, n, conv_fft_len), uniform)

    arcsine = _beta_shape(n, 0.5, 0.5)
    _maybe_insert(_objective(arcsine, n, conv_fft_len), arcsine)

    # Power‑law edge‑concentrated shapes
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in (-2.5, -2.0, -1.5, -1.0, -0.75):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            _maybe_insert(_objective(shape, n, conv_fft_len), shape)

    # Two‑Gaussian edge shapes
    i = np.arange(n, dtype=np.float64)
    for gamma in (0.2, 0.5, 0.9, 1.2):
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        shape = np.maximum(shape, 0.0)
        s = shape.sum()
        if s > 0:
            shape /= s
            _maybe_insert(_objective(shape, n, conv_fft_len), shape)

    # Random Beta shapes (very diverse)
    nbeta = 800
    for _ in range(nbeta):
        # log‑uniform α,β ∈ [0.01, 4]
        log_a = np.random.uniform(np.log10(0.01), np.log10(4.0))
        log_b = np.random.uniform(np.log10(0.01), np.log10(4.0))
        a_par = 10 ** log_a
        b_par = 10 ** log_b
        shape = _beta_shape(n, a_par, b_par)
        _maybe_insert(_objective(shape, n, conv_fft_len), shape)
        if time.time() > deadline - 30:
            break

    # Random Dirichlet shapes
    ndir = 200
    for _ in range(ndir):
        conc = 10 ** np.random.uniform(np.log10(0.05), np.log10(2.0))
        shape = np.random.dirichlet(np.full(n, conc))
        _maybe_insert(_objective(shape, n, conv_fft_len), shape)
        if time.time() > deadline - 30:
            break

    # ------------------------------------------------------------------
    # Pick the current best candidate
    best_seq, best_val = best_cands[0][1], best_cands[0][0]
    print(f"Initial best value {best_val:.6f}")

    # ------------------------------------------------------------------
    # Extended Adam optimisation (big slice of the remaining budget)
    time_left = deadline - time.time()
    if time_left > 30.0:
        adam_deadline = time.time() + min(0.4 * time_left, 300.0)
        best_seq, val_adam = _adam_optimize(best_seq, n,
                                            conv_fft_len,
                                            corr_fft_len,
                                            adam_deadline,
                                            max_iters=400_000)
        if val_adam < best_val:
            best_val = val_adam
            print(f"[Adam] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Exact‑max sub‑gradient descent (another big slice)
    if time.time() < deadline - 20.0:
        sub_deadline = deadline - 10.0
        best_seq, val_sub = _subgradient_optimize(best_seq, n,
                                                  conv_fft_len,
                                                  corr_fft_len,
                                                  sub_deadline,
                                                  max_iters=400_000)
        if val_sub < best_val:
            best_val = val_sub
            print(f"[Subgrad] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Simulated annealing with a slower schedule
    if time.time() < deadline - 5.0:
        sa_deadline = deadline - 0.5
        best_seq, val_sa = _simulated_annealing(best_seq, n,
                                                conv_fft_len,
                                                sa_deadline,
                                                best_val)
        if val_sa < best_val:
            best_val = val_sa
            print(f"[SA] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Optional upscale to 8192 points and quick refinement
    if time.time() + 15.0 < deadline:
        n_big = 8192
        a_big, val_big = _interpolate_and_refine(best_seq,
                                                 n_big,
                                                 deadline)
        if val_big < best_val:
            best_seq = a_big
            best_val = val_big
            print(f"[Upscale] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Clip to limits enforced by evaluate_sequence and report
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"Final reported value {final_val:.6f} (length {len(best_seq)})")
    except Exception as exc:
        print(f"Evaluation error: {exc}")
        print(f"Internal best value {best_val:.6f}")

    # store for the next round (if any)
    globals()["height_sequence_1"] = best_seq.tolist()

    return best_seq.tolist()
