"""
Propose a low‑objective step‑function using a multi‑start basin‑hopping
search.

The method builds a diverse pool of parametric seeds (uniform, Dirichlet,
edge‑exponential, edge‑power‑law, double‑Gaussian, sinusoid and symmetric
Beta shapes).  The best few seeds are refined with a soft‑max
sub‑gradient (exponentiated‑gradient) optimiser that works directly on the
true convolution via FFT.  The optimiser starts with a smooth
soft‑max approximation of the max‑operator (high temperature) and cools
the temperature, which helps escape shallow local minima.  After the
initial refinements a simple basin‑hopping loop repeatedly perturbs the
current best sequence and runs a short optimisation from each perturbation.
If time permits the best solution is interpolated to a finer grid (n=
16384) and refined again.

The whole search respects the 1000 s budget, prints progress and finally
returns the best sequence as a plain Python list.
"""

import time
import math
import numpy as np

# --------------------------------------------------------------
# FFT‑based helpers
# --------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)−1) using FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k−(n−1)]   for i=0…n−1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, n=fft_len)
    fr = np.fft.rfft(rev_a, n=fft_len)
    corr = np.fft.irfft(fw * fr, n=fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Fast evaluation of 2·n·max(a*a) (a is assumed normalised to sum≈1)."""
    conv = _full_convolution(a, conv_fft_len)
    return float(2.0 * n * conv.max() / (a.sum() ** 2))


# --------------------------------------------------------------
# Candidate generators
# --------------------------------------------------------------
def _generate_initial_candidates(n: int, warm_start: np.ndarray | None) -> list[np.ndarray]:
    """
    Produce a heterogeneous set of initial vectors.
    All vectors are returned normalised to sum≈1 and clipped to ≥0.
    """
    candidates = []

    # ------------------------------------------------------------------
    # Warm‑start and its cyclic shifts
    # ------------------------------------------------------------------
    if warm_start is not None:
        candidates.append(warm_start.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            candidates.append(np.roll(warm_start, shift))

    # ------------------------------------------------------------------
    # Uniform baseline
    # ------------------------------------------------------------------
    candidates.append(np.full(n, 1.0 / n))

    # ------------------------------------------------------------------
    # Random Dirichlet (various concentrations)
    # ------------------------------------------------------------------
    for conc in (0.01, 0.05, 0.1, 0.3, 0.5, 1.0, 2.0):
        cand = np.random.dirichlet(np.full(n, conc))
        candidates.append(cand)

    # ------------------------------------------------------------------
    # Edge‑exponential family
    # ------------------------------------------------------------------
    i = np.arange(n, dtype=np.float64)
    for alpha in np.logspace(-2, 1.0, num=12):          # 0.01 … 10
        edge = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
        edge /= edge.sum()
        for w in (0.0, 0.5, 1.0):
            shape = w * edge + (1.0 - w) / n
            candidates.append(shape)

    # ------------------------------------------------------------------
    # Edge‑power‑law family
    # ------------------------------------------------------------------
    for p in np.linspace(0.2, 5.0, num=12):
        edge = (i + 1.0) ** (-p) + (n - i) ** (-p)
        edge /= edge.sum()
        for w in (0.0, 0.5, 1.0):
            shape = w * edge + (1.0 - w) / n
            candidates.append(shape)

    # ------------------------------------------------------------------
    # Double‑Gaussian edges
    # ------------------------------------------------------------------
    for gamma in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        shape /= shape.sum()
        candidates.append(shape)

    # ------------------------------------------------------------------
    # Positive half‑wave sinusoid
    # ------------------------------------------------------------------
    sinus = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    sinus = np.maximum(sinus, 0.0)
    sinus /= sinus.sum()
    candidates.append(sinus)

    # ------------------------------------------------------------------
    # Symmetric Beta shapes (U‑ or ∩‑shapes)
    # ------------------------------------------------------------------
    x = (np.arange(n) + 0.5) / n
    eps = 1e-12
    for alpha in np.linspace(0.2, 5.0, num=12):
        shape = (x + eps) ** (alpha - 1) * ((1 - x) + eps) ** (alpha - 1)
        shape /= shape.sum()
        candidates.append(shape)

    # ------------------------------------------------------------------
    # Triangular (peak in the middle)
    # ------------------------------------------------------------------
    tri = np.minimum(i, n - 1 - i) + 1.0
    tri /= tri.sum()
    candidates.append(tri)

    return candidates

# Soft‑max sub‑gradient optimiser (exponentiated gradient)
# --------------------------------------------------------------
def _softmax_subgrad_opt(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
    target_val: float = 1.503,
) -> tuple[np.ndarray | None, float]:

    Exponentiated‑gradient descent on a smoothed version of the max‑convolution.
    The smoothing is a soft‑max with a temperature that gradually cools.
    Returns the best sequence/value observed before deadline.
    """
    a = np.maximum(a0, 1e-30)
    if a.sum() < 0.01:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    eta0 = 0.5
    eta_decay = 0.99995
    min_eta = 0.001

    # Temperature schedule (soft‑max)
    tau0 = 0.5          # start with a relatively high temperature
    tau_decay = 0.99990
    min_tau = 0.01

    it = 0
    last_improve = time.time()

    while time.time() < deadline:
        # ---- convolution & objective ---------------------------------
        conv = _full_convolution(a, conv_fft_len)                     # length 2n‑1
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        # ---- record best ---------------------------------------------
        if val < best_val - 1e-12:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[n={n} soft‑grad iter={it:07d}] new best {val:.6f}")

            # early exit if we already beat the target
            if val < target_val:
                print(f"*** Target {target_val:.3f} reached – exiting early.")
                return best_seq, best_val

        # ---- soft‑max weights ----------------------------------------
        tau = max(min_tau, tau0 * (tau_decay ** it))
        temp = max(tau * max_c, 1e-12)          # avoid division by zero

        # Use (conv - max_c) for numerical stability
        exp_arg = (conv - max_c) / temp
        w = np.exp(exp_arg)
        ws = w.sum()
        if ws == 0 or not np.isfinite(ws):
            # fallback to uniform distribution over the active set
            active = np.isclose(conv, max_c, atol=1e-12)
            w = np.zeros_like(conv)
            if active.any():
                w[active] = 1.0 / active.sum()
        else:
            w = w / ws

        # ---- gradient (log‑domain) ----------------------------------
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # ---- step size ------------------------------------------------
        eta = max(min_eta, eta0 * (eta_decay ** it))

        # ---- exponentiated‑gradient update ----------------------------
        a = a * np.exp(-eta * grad)

        # ---- renormalise onto the simplex -----------------------------
        s = a.sum()
        if s < 1e-12 or not np.isfinite(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        # ---- occasional random shake when stalled --------------------
        if time.time() - last_improve > 5.0:
            # log‑space jitter of ±2 %
            jitter = (np.random.rand(n) - 0.5) * 0.04
            a = a * np.exp(jitter)
            a = a / a.sum()
            last_improve = time.time()

        it += 1

    return best_seq, best_val

# Main entry point
# --------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a step‑function (non‑negative sequence) that minimises
    F(a) = 2·n·max(a∗a) / (sum(a)²).

    The routine respects the time budget, prints progress and returns the
    best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    # safety margin – leave a tiny buffer for the caller
    deadline = time.time() + budget_s - 0.5

    # --------------------------------------------------------------
    # Primary grid size
    # --------------------------------------------------------------
    n = 8192                     # a good trade‑off between resolution & speed
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Warm start (if a previous best lives in the global namespace)
    # --------------------------------------------------------------
    warm_start = globals().get("height_sequence_1")
    if warm_start is not None:
        warm_start = np.asarray(warm_start, dtype=np.float64)
        if warm_start.shape[0] > n:
            warm_start = warm_start[:n]
        elif warm_start.shape[0] < n:
            warm_start = np.pad(warm_start, (0, n - warm_start.shape[0]))
        warm_start = np.maximum(warm_start, 0.0)
        if warm_start.sum() >= 0.01:
            warm_start = warm_start / warm_start.sum()
        else:
            warm_start = None
    else:
        warm_start = None

    # --------------------------------------------------------------
    # Stage 1 – generate a broad pool of seeds
    # --------------------------------------------------------------
    candidates = _generate_initial_candidates(n, warm_start)

    # Add a few extra Dirichlet samples with diverse concentrations
    for conc in (0.02, 0.07, 0.15, 0.4, 0.8):
        for _ in range(20):
            candidates.append(np.random.dirichlet(np.full(n, conc)))

    # --------------------------------------------------------------
    # Fast evaluation of all seeds – keep the most promising
    # --------------------------------------------------------------
    seed_vals = []
    for cand in candidates:
        if time.time() >= deadline:
            break
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 0.01:
            cand = np.random.dirichlet(np.ones(n))
            s = cand.sum()
        cand = cand / s
        val = _objective(cand, n, conv_fft_len)
        seed_vals.append((val, cand))

    if not seed_vals:
        # fallback – uniform vector
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        print("*** No candidates evaluated – returning uniform vector.")
        return best_seq.tolist()

    seed_vals.sort(key=lambda x: x[0])
    top_k = min(10, len(seed_vals))
    top_candidates = [c for _, c in seed_vals[:top_k]]

    best_seq = None
    best_val = float("inf")

    # --------------------------------------------------------------
    # Stage 2 – deep optimisation of the top seeds
    # --------------------------------------------------------------
    for idx, start_vec in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        print(f"--- Optimising from seed #{idx + 1}/{len(top_candidates)} ---")
        best_seq, best_val = _softmax_subgrad_opt(
            start_vec,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            best_seq,
            best_val,
        )
        # Stop early if target already achieved
        if best_val < 1.503:
            break

    # --------------------------------------------------------------
    # Stage 3 – basin‑hopping via random perturbations
    # --------------------------------------------------------------
    while time.time() < deadline - 2.0 and best_seq is not None:
        # log‑space jitter magnitude slowly reduced with time
        time_frac = (deadline - time.time()) / (deadline - (deadline - budget_s))
        sigma = 0.04 * time_frac + 0.005   # between ~0.045 and ~0.005
        # a handful of perturbations in parallel
        for _ in range(6):
            if time.time() >= deadline - 2.0:
                break
            noise = (np.random.rand(n) - 0.5) * sigma
            pert = best_seq * np.exp(noise)
            pert = pert / pert.sum()
            sub_dead = min(time.time() + 4.0, deadline)
            best_seq, best_val = _softmax_subgrad_opt(
                pert,
                n,
                conv_fft_len,
                corr_fft_len,
                sub_dead,
                best_seq,
                best_val,
            )
            if best_val < 1.503:
                break
        if best_val < 1.503:
            break

    # --------------------------------------------------------------
    # Stage 4 – final refinement on a finer grid (if time permits)
    # --------------------------------------------------------------
    time_left = deadline - time.time()
    if time_left > 30.0 and best_seq is not None:
        n_big = 16384
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        # Linear interpolation of the current best onto the larger grid
        x_old = np.linspace(0.0, 1.0, n, endpoint=False)
        x_big = np.linspace(0.0, 1.0, n_big, endpoint=False)
        a_big = np.interp(x_big, x_old, best_seq)
        a_big = np.maximum(a_big, 0.0)
        a_big = a_big / a_big.sum()

        print("*** Refinement on larger grid (n=16384) ***")
        best_seq_big, best_val_big = _softmax_subgrad_opt(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            None,
            float("inf"),
        )
        if best_seq_big is not None and best_val_big < best_val:
            best_seq = best_seq_big
            best_val = best_val_big
            n = n_big

    # --------------------------------------------------------------
    # Final safety checks
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(8192, 1.0 / 8192, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)
    print(f"*** Finished search – best objective ≈ {best_val:.6f} (n={best_seq.shape[0]})")
    return best_seq.tolist()
