def _adam_exact_optimize(a0, n, conv_fft_len, corr_fft_len, deadline, max_iters=300_000):
    eps=1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1=0.9; beta2=0.999; eps_adam=1e-8
    lr = 0.05 (maybe start lower).
    best_a=a0.copy(); best_val=inf; i=0
    while time.time()<deadline and i<max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum()**2)
        if val < best_val:
            best_val=val; best_a=a.copy(); print...
        
        # Active set
        active = np.isclose(conv, max_c, atol=1e-14)
        # if many active; weight uniform over them
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0/active.sum()
        # gradient wrt a
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        # gradient wrt logits (mirror descent)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)
        # Adam update
        m = beta1*m + (1-beta1)*grad_log
        v = beta2*v + (1-beta2)*(grad_log**2)
        m_hat = m/(1-beta1**(i+1))
        v_hat = v/(1-beta2**(i+1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step
        # Learning rate decay maybe
        if i%20000==0 and i>0: lr *= 0.9
        i+=1

    return best_a, best_val

def _annealing_local_search(a0, n, conv_fft_len, deadline, max_moves=100000):
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    best_a = a.copy()
    best_val_float = best_val

    # schedule temperature: start_T = maybe 0.01? compute based on initial improvement range?
    # We'll use linear schedule T = T0 * (1 - t / total_moves)
    total_moves = max_moves
    move = 0
    while time.time() < deadline and move < max_moves:
        # sample i and j indices where a_i > epsilon
        i = np.random.randint(n)
        if a[i] < 1e-12: 
            continue
        j = np.random.randint(n)
        if i == j: continue
        # delta up to some fraction of a[i], encourage small steps
        delta = a[i] * (np.random.rand() * 0.2) # up to 20% of mass
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        conv_new = _full_convolution(a_new, conv_fft_len)
        val_new = 2.0 * n * conv_new.max()  # denom =1
        # accept if better or by probability
        delta_val = val_new - best_val_float
        # time-dependent temperature: compute fraction of elapsed
        elapsed = move / total_moves
        T = max(0.001, (0.02 * (1 - elapsed)))  # decreasing temperature
        accept = False
        if val_new < best_val_float:
            accept = True
        else:
            if np.exp(-delta_val / T) > np.random.rand():
                accept = True
        if accept:
            a = a_new
            conv = conv_new
            best_val_float = val_new
            best_a = a_new
            if move % 5000 == 0:
                print(f"[Anneal move={move}] current best {best_val_float}")
        move += 1

    return best_a, best_val_float

"""
Enhanced step‑function optimizer for the objective

    F(a) = 2·n·max(a∗a) / (sum(a)²)

The algorithm combines three complementary phases:

1. **Smooth‑Adam** –   Optimises a soft‑max surrogate
   (log‑sum‑exp) of the convolution peak.  The temperature grows very
   slowly (lam_factor=1.001) so that the optimiser explores a large
   neighbourhood before focusing on the true maximum.

2. **Exact‑Adam** –   Uses the true max (active‑set) as a sub‑gradient
   but updates the *logits* of a soft‑max parametrisation via Adam.
   This gives the speed of Adam while respecting the simplex constraint.

3. **Simulated‑annealing local search** – Random mass transfers between
   two coordinates are accepted either if they improve the exact objective
   or with a temperature‑controlled probability, allowing the search to
   escape shallow local minima.

If time permits the best solution is linearly interpolated to a finer
grid (8192 points) and the three phases are repeated on the refined grid.

All phases respect a global deadline (default 1000 s) and the best
sequence found so far is always returned as a plain Python list.
"""

import time
import numpy as np


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a length‑len(a) vector:
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i = 0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


def _adam_smooth_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth‑max (log‑sum‑exp) surrogate.
    Returns best vector (sum ≈1) and its exact objective.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)                 # initialise on the simplex
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.08                                 # slightly larger than before
    lam = 0.04                                 # start temperature (soft)
    lam_factor = 1.001                        # very slow growth

    best_a = a0.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        # ---------- simplex projection (soft‑max) ----------
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # ---------- exact objective ----------
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)   # sum ≈1
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Smooth Adam it={it:06d}] new best {val:.6f}")

        # ---------- smooth‑max weights ----------
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---------- gradient of the surrogate ----------
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through the soft‑max (mirror‑descent form)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ---------- Adam update on logits ----------
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # ---------- temperature schedule ----------
        lam = min(lam * lam_factor, 1e7)

        # occasional tiny random shake near the deadline
        if it % 4000 == 0 and time.time() > deadline - 60:
            logits += (np.random.rand(n) - 0.5) * 2e-2

        # simple learning‑rate decay
        if it % 50_000 == 0 and it > 0:
            lr *= 0.93

        it += 1

    return best_a, best_val


def _adam_exact_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 400_000,
) -> tuple[np.ndarray, float]:
    """
    Adam on the *exact* max (sub‑gradient) surrogate.
    Uses the active set (indices attaining the current max) to form a
    sub‑gradient, then performs an Adam update on the simplex‑logits.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.07
    best_a = a0.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        # simplex projection
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Exact Adam it={it:06d}] new best {val:.6f}")

        # active set (all indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-14)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()
        # gradient of the true max (sub‑gradient)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # mirror‑descent gradient (log‑space)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # learning‑rate decay
        if it % 30_000 == 0 and it > 0:
            lr *= 0.90

        it += 1

    return best_a, best_val


def _annealing_local_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 80_000,
) -> tuple[np.ndarray, float]:
    """
    Cheap stochastic hill‑climbing with simulated‑annealing acceptance.
    Moves a random fraction of mass from one index to another.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()

    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        # move up to 20 % of the source mass
        delta = a[i] * (np.random.rand() * 0.20)

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        # linear temperature schedule (starts at 0.02, ends at 0.001)
        T = 0.02 * (1.0 - it / max_moves) + 0.001
        improve = val2 < best_val
        accept = improve or (np.exp(-(val2 - best_val) / T) > np.random.rand())

        if accept:
            a = a2
            conv = conv2
            if val2 < best_val:
                best_val = val2
                best_a = a2.copy()
                if it % 5000 == 0:
                    print(f"[Anneal move={it:06d}] improved to {best_val:.6f}")

        it += 1

    return best_a, best_val


def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Main entry point.  Runs the three‑phase optimiser described above
    and returns the best non‑negative step‑function (list) found
    within the allotted time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # tiny safety margin

    # ------------------------------------------------------------------
    # problem size
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # warm start (previous best if available)
    warm = globals().get("height_sequence_1")
    a0 = None
    if warm is not None:
        a0 = np.asarray(warm, dtype=np.float64)
        if a0.shape[0] > n:
            a0 = a0[:n]
        elif a0.shape[0] < n:
            a0 = np.pad(a0, (0, n - a0.shape[0]))
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        if s >= 0.01:
            a0 = a0 / s
        else:
            a0 = None
    if a0 is None:
        a0 = np.full(n, 1.0 / n, dtype=np.float64)   # uniform start

    # record initial value
    conv0 = _full_convolution(a0, conv_fft_len)
    best_val = 2.0 * n * conv0.max() / (a0.sum() ** 2)
    best_seq = a0.copy()
    print(f"Initial value {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – smooth Adam (log‑sum‑exp surrogate)
    if time.time() < deadline:
        phase_deadline = time.time() + (deadline - time.time()) * 0.50
        a_smooth, val_smooth = _adam_smooth_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            phase_deadline,
            max_iters=600_000,
        )
        if val_smooth < best_val:
            best_val, best_seq = val_smooth, a_smooth.copy()
            print(f"[Phase 1] best {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – exact Adam (sub‑gradient on the true max)
    if time.time() < deadline:
        phase_deadline = time.time() + (deadline - time.time()) * 0.55
        a_exact, val_exact = _adam_exact_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            phase_deadline,
            max_iters=500_000,
        )
        if val_exact < best_val:
            best_val, best_seq = val_exact, a_exact.copy()
            print(f"[Phase 2] best {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – simulated‑annealing local search
    if time.time() < deadline - 5:
        phase_deadline = time.time() + (deadline - time.time()) * 0.40
        a_loc, val_loc = _annealing_local_search(
            best_seq,
            n,
            conv_fft_len,
            phase_deadline,
            max_moves=120_000,
        )
        if val_loc < best_val:
            best_val, best_seq = val_loc, a_loc.copy()
            print(f"[Phase 3] best {best_val:.6f}")

    # ------------------------------------------------------------------
    # Optional Phase 4 – up‑sample to finer grid (8192 points) and repeat
    if time.time() + 30 < deadline:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        # linear interpolation to the finer grid
        x_old = np.linspace(0, 1, n, endpoint=False)
        x_new = np.linspace(0, 1, n_big, endpoint=False)
        a_big = np.interp(x_new, x_old, best_seq)
        a_big = np.maximum(a_big, 0.0)
        a_big = a_big / a_big.sum()

        # Smooth Adam on the larger grid
        phase_deadline = time.time() + (deadline - time.time()) * 0.45
        a_big_smooth, _ = _adam_smooth_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            phase_deadline,
            max_iters=300_000,
        )

        # Exact Adam on the larger grid
        if time.time() < deadline:
            phase_deadline = time.time() + (deadline - time.time()) * 0.55
            a_big_exact, val_big = _adam_exact_optimize(
                a_big_smooth,
                n_big,
                conv_fft_len_big,
                corr_fft_len_big,
                phase_deadline,
                max_iters=300_000,
            )
            if val_big < best_val:
                best_val, best_seq = val_big, a_big_exact.copy()
                print(f"[Phase 4] improved best {best_val:.6f}")

    # ------------------------------------------------------------------
    # safety fallback – guarantee a non‑empty return
    if best_seq is None or best_seq.size == 0:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Clip to evaluator's accepted range (non‑negative, ≤1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
