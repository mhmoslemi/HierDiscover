"""
Multi‑start search for a low‑value non‑negative step function.

Algorithm
---------
1.  Generate many diverse starting vectors (Dirichlet samples with
    different concentration parameters, anti‑pair ordering, alternating
    large/small, random permutation, Gaussian‑edge, triangle, uniform).
    All vectors are normalised to have sum = 1.

2.  For each start we run a *short* optimisation pipeline:
    * Adam on a smooth‑max surrogate (log‑sum‑exp) – a few ×10⁴ iterations.
    * Exact‑max sub‑gradient (exponentiated‑gradient) – a few ×10⁴ iterations.
    * Simulated‑annealing hill‑climbing that swaps two positions or moves a
      fraction of mass between them.

    The best sequence obtained from the pipeline is kept if its exact
    objective value improves the global best.

3.  After the allotted time for the short runs is exhausted, we take the
    current global best and run a *long* refinement (more Adam/sub‑gradient
    iterations followed by a larger hill‑climbing phase).  This is similar
    to the previous runner but benefits from a better starting point and
    more time.

4.  The final sequence is clipped to the range accepted by the evaluator
    and returned as a plain Python list.

All heavy work is done with NumPy and FFT (linear‑time convolution).  The
search respects the 1000 s budget; a small safety margin is kept so that
the function always returns a valid sequence.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility FFT helpers ---------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
# Exact objective (without safety checks) -------------------------------
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(a*a)/(sum(a)²).  Assumes a ≥ 0 and sum(a)>0."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    conv = _full_convolution(a, _next_pow2(2 * a.shape[0] - 1))
    return 2.0 * a.shape[0] * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Arrangement helpers ---------------------------------------------------
def _anti_pair_arrangement(vals: np.ndarray) -> np.ndarray:
    """Largest opposite smallest, 2nd‑largest opposite 2nd‑smallest, …"""
    n = len(vals)
    sorted_vals = np.sort(vals)[::-1]          # descending
    a = np.empty(n, dtype=np.float64)
    half = n // 2
    a[:half] = sorted_vals[:half]             # biggest half left side
    a[-half:] = sorted_vals[-half:][::-1]     # smallest half right side
    if n % 2 == 1:
        a[half] = sorted_vals[half]
    return a


def _alternating_arrangement(vals: np.ndarray) -> np.ndarray:
    """
    Interleave largest, smallest, second‑largest, second‑smallest, …
    """
    n = len(vals)
    asc = np.sort(vals)        # ascending
    desc = asc[::-1]           # descending
    a = np.empty(n, dtype=np.float64)
    i = 0
    left = 0
    right = 0
    # fill alternating
    for idx in range(n):
        if idx % 2 == 0:
            a[idx] = desc[left]
            left += 1
        else:
            a[idx] = asc[right]
            right += 1
    return a


# ----------------------------------------------------------------------
# Candidate generators -------------------------------------------------
def _uniform_candidate(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _dirichlet_candidate(n: int, rng: np.random.Generator,
                         alpha: float = 1.0) -> np.ndarray:
    a = rng.dirichlet(alpha * np.ones(n))
    return a.astype(np.float64)


def _gaussian_edges_candidate(n: int, rng: np.random.Generator,
                              gamma: float = 0.5) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
    shape /= shape.sum()
    return shape


def _triangle_candidate(n: int) -> np.ndarray:
    x = np.arange(n, dtype=np.float64)
    shape = np.minimum(x + 1, n - x)          # 1,2,…,⌈n/2⌉,…,2,1
    shape /= shape.sum()
    return shape


def _random_permutation_candidate(n: int, rng: np.random.Generator,
                                 alpha: float = 1.0) -> np.ndarray:
    a = rng.dirichlet(alpha * np.ones(n))
    rng.shuffle(a)
    return a.astype(np.float64)


def _generate_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a random starting vector using a random mixture of the
    generation strategies described above.
    """
    # pick a strategy uniformly
    strategies = [
        "uniform",
        "dirichlet",
        "anti_pair",
        "alternating",
        "perm",
        "gauss",
        "triangle",
    ]
    choice = rng.choice(strategies)

    if choice == "uniform":
        return _uniform_candidate(n)

    # Choose concentration parameter on a log‑scale for Dirichlet‑based methods
    log_alpha = rng.uniform(-2.0, 1.0)   # α ∈ [0.01, 10]
    alpha = 10 ** log_alpha

    if choice == "dirichlet":
        return _dirichlet_candidate(n, rng, alpha)

    if choice == "anti_pair":
        v = _dirichlet_candidate(n, rng, alpha)
        return _anti_pair_arrangement(v)

    if choice == "alternating":
        v = _dirichlet_candidate(n, rng, alpha)
        return _alternating_arrangement(v)

    if choice == "perm":
        return _random_permutation_candidate(n, rng, alpha)

    if choice == "gauss":
        # pick γ on log‑scale as well
        log_gamma = rng.uniform(-1.0, 1.0)   # γ ∈ [0.1, 10]
        gamma = 10 ** log_gamma
        return _gaussian_edges_candidate(n, rng, gamma)

    # triangle is deterministic
    return _triangle_candidate(n)


# ----------------------------------------------------------------------
# Optimisation primitives -----------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
    lr: float = 0.06,
    lam_start: float = 1.0,
    lam_factor: float = 1.05,
) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth‑max (log‑sum‑exp) surrogate.
    Returns (best_vector, best_exact_value).
    """
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lam = lam_start

    best_a = a0.copy()
    best_val = _objective(a0)

    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max → simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val_exact = 2.0 * n * max_c
        if val_exact < best_val:
            best_val = val_exact
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Adam it={it:06d}] new best exact {best_val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        lam = min(lam * lam_factor, 1e6)

        # occasional jitter to avoid stagnation
        if it % 4000 == 0 and (deadline - time.time()) > 2.0:
            logits += (rng.random(n) - 0.5) * 5e-3

        it += 1

    return best_a, best_val


def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
    eta0: float = 0.75,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient with exponentiated‑gradient updates.
    Returns (best_vector, best_exact_value).
    """
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = rng.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s

    best_a = a.copy()
    best_val = _objective(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1)

        a = a * np.exp(-eta * grad_a)

        # renormalise
        s = a.sum()
        if s < 1e-12:
            a = rng.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 150_000,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing hill climbing on the exact objective.
    Two move types:
        * swap two entries
        * transfer a fraction (≤40 %) of mass from i to j
    Temperature decays linearly from 0.015 to 0.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a)

    start_temp = 0.015
    it = 0
    while time.time() < deadline and it < max_moves:
        # choose move type
        if rng.random() < 0.5:                # swap
            i, j = rng.integers(n, size=2)
            if i == j:
                continue
            a2 = a.copy()
            a2[i], a2[j] = a2[j], a2[i]
        else:                                 # transfer fraction
            i = rng.integers(n)
            if a[i] < 1e-14:
                continue
            j = rng.integers(n)
            if i == j:
                continue
            delta = a[i] * (rng.random() * 0.4)   # up to 40 % move
            a2 = a.copy()
            a2[i] -= delta
            a2[j] += delta

        val2 = _objective(a2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
        else:
            # acceptance probability decreasing with temperature
            temp = start_temp * (1.0 - (it / max_moves))
            if temp > 0.0 and rng.random() < np.exp(-(val2 - best_val) / temp):
                a = a2

        if it % 5000 == 0:
            print(f"[Local move {it:06d}] best {best_val:.6f}")

        it += 1

    return best_a, best_val

    It respects the given time budget and returns the best sequence found.
    """

    global rng
    rng = np.random.default_rng(seed)

    # Problem size (use the same size as the previous best)
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------

    best_seq = None
    best_val = float("inf")

    def _maybe_update(seq, val, label=""):
        nonlocal best_seq, best_val

            best_val = val
            best_seq = seq.copy()

    # ------------------------------------------------------------------

    warm = globals().get("height_sequence_1")
    if isinstance(warm, list) and warm:

        warm_arr = np.maximum(warm_arr, 0.0)
        if warm_arr.sum() >= 0.01:

    # ------------------------------------------------------------------

    multi_start_deadline = time.time() + (budget_s * 0.45)
