def _expgrad_optimize_v(v0, n, conv_fft_len, corr_fft_len, deadline, max_iters=inf, eta0=0.5, lam_start=1.0, lam_factor=1.04):
    v = v0.copy()
    best_v = v.copy()
    best_val = _objective(_anti_pair_from_sorted(v))
    lam = lam_start
    it = 0
    while time.time() < deadline:
        a = _anti_pair_from_sorted(v)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        # compute w
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum
        # gradient w.r.t a
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        # map to grad_v
        grad_v = np.zeros_like(v)
        half = n // 2
        # left part
        grad_v[:half] = grad_a[:half]
        # right part
        if n % 2 == 0:
            # even
            for l in range(half):
                grad_v[n-1 - l] = grad_a[half + l]
        else:
            # odd
            grad_v[half] = grad_a[half]
            for l in range(half):
                grad_v[n-1 - l] = grad_a[half + 1 + l]
        # Exponential update
        eta = eta0 / np.sqrt(it + 1)
        v = v * np.exp(-eta * grad_v)
        # renorm and sort
        s = v.sum()
        if s <= 0:
            v = np.random.dirichlet(np.ones(n))
        else:
            v = v / s
        v = np.sort(v)[::-1]

        # evaluate objective
        val = _objective(_anti_pair_from_sorted(v))
        if val < best_val:
            best_val = val
            best_v = v.copy()
            print(f"[ExpGrad it={it}] new best {val:.6f}")

        # increase lam
        lam = min(lam * lam_factor, 1e6)
        it += 1
        if max_iters and it >= max_iters:
            break
    return best_v, best_val

def _expgrad_optimize_v(v0, n, conv_fft_len, corr_fft_len, deadline, eta0=0.5, lam_start=1.0, lam_factor=1.03, max_iters=10**9):
    v = v0.copy()
    best_v = v.copy()
    best_val = _objective(_anti_pair_from_sorted(v))
    lam = lam_start
    it = 0
    while time.time() < deadline and it < max_iters:
        # compute a from v
        a = _anti_pair_from_sorted(v)
        # conv
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        # compute weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum
        # gradient w.r.t a
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        # map to grad_v
        grad_v = np.zeros_like(v)
        half = n // 2
        # left half
        grad_v[:half] = grad_a[:half]
        if n % 2 == 0:
            # even
            for l in range(half):
                grad_v[n-1 - l] = grad_a[half + l]
        else:
            # odd
            grad_v[half] = grad_a[half]
            for l in range(half):
                grad_v[n-1 - l] = grad_a[half + 1 + l]
        # step size
        eta = eta0 / np.sqrt(it + 1)
        # update by multiplicative rule
        v = v * np.exp(-eta * grad_v)
        # renormalize
        sum_v = v.sum()
        if sum_v <= 0:
            v = np.random.dirichlet(np.ones(n))
        else:
            v = v / sum_v
        # sorting
        v = np.sort(v)[::-1]
        # evaluate
        val = _objective(_anti_pair_from_sorted(v))
        if val < best_val:
            best_val = val
            best_v = v.copy()
            print(f"[ExpGrad it={it}] new best {val:.6f}")
        # increase lam
        lam = min(lam * lam_factor, 1e6)
        # occasional jitter

            # add small noise
            v += (np.random.rand(n) - 0.5) * 1e-3
            v = np.maximum(v, 0)
            v = v / v.sum()
            v = np.sort(v)[::-1]
        it += 1
    return best_v, best_val

"""
Propose a low‐value step function.

The algorithm works entirely on the *sorted* multiset of heights.
For any multiset the optimal ordering (minimising the maximal
self‑convolution) is the “anti‑pair’’ ordering: the largest mass
faces the smallest, the second largest the second smallest, etc.
We therefore optimise only the sorted vector `v` (non‑negative,
∑v = 1, decreasing).  At each iteration we

  • build the anti‑pair vector a = anti_pair(v),
  • evaluate the smooth‑max surrogate of max(a∗a) using a
    temperature λ and the log‑sum‑exp trick,
  • obtain the gradient w.r.t. a,
  • map it to a gradient w.r.t. the sorted vector v,
  • perform an exponentiated‑gradient (mirror‑descent) update,
  • renormalise, sort, and repeat while the time budget permits.

The temperature λ is increased slowly so that the surrogate
smoothly turns into the exact max, yielding a high‑quality
solution without ever leaving the anti‑pair feasible set.
After the main optimisation a short simulated‑annealing hill‑climb
on v is performed, and finally the solution is up‑sampled to a
finer grid (8192 points) and refined once more.

All steps respect the 1000 s budget and return a valid
non‑negative list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


# ----------------------------------------------------------------------
# Objective helpers -----------------------------------------------------

def _objective(a: np.ndarray) -> float:
    """Exact objective: 2·n·max(a*a) (assumes ∑a = 1, a≥0)."""
    n = a.shape[0]
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Anti‑pair arrangement -------------------------------------------------
def _anti_pair_from_sorted(v: np.ndarray) -> np.ndarray:
    """
    Build the anti‑pair step function from a sorted (descending) vector v.
    The largest half of v occupies the left side, the smallest half the
    right side (reversed).  For odd length the median stays in the centre.
    """
    n = v.shape[0]
    half = n // 2
    a = np.empty_like(v)
    # left side: biggest half (or biggest floor(n/2) entries)
    a[:half] = v[:half]
    if n % 2 == 0:
        # even length – right side uses the remaining half reversed
        a[half:] = v[:half - 1:-1]   # v[half:][::-1]
    else:
        # odd length – median stays, right side uses the tail reversed
        a[half] = v[half]
        a[half + 1:] = v[:half - 1:-1]   # v[half+1:][::-1]
    return a


def _grad_v_from_grad_a(grad_a: np.ndarray, n: int) -> np.ndarray:
    """
    Map the gradient w.r.t. the anti‑pair vector `a`
    back to the gradient w.r.t. the sorted vector `v`.
    """
    half = n // 2
    grad_v = np.zeros(n, dtype=grad_a.dtype)

    # left half (largest entries)
    grad_v[:half] = grad_a[:half]

    if n % 2 == 0:
        # even: right side maps indices half .. n-1
        for l in range(half):
            grad_v[n - 1 - l] = grad_a[half + l]
    else:
        # odd: centre element
        grad_v[half] = grad_a[half]
        for l in range(half):
            grad_v[n - 1 - l] = grad_a[half + 1 + l]

    return grad_v


# ----------------------------------------------------------------------
# Candidate generators --------------------------------------------------

def _warm_start_sorted(n: int):
    """Extract a sorted warm start from the global variable if present."""
    warm = globals().get("height_sequence_1")
    if warm is None:
        return None
    warm = np.asarray(warm, dtype=np.float64)
    if warm.ndim != 1:
        return None
    warm = np.maximum(warm, 0.0)
    if warm.sum() < 0.01:
        return None
    warm = warm / warm.sum()
    v = np.sort(warm)[::-1]
    if v.shape[0] != n:
        # resample / pad / truncate while preserving order
        if v.shape[0] > n:
            v = v[:n]
        else:
            # repeat the smallest entries to reach n
            extra = np.full(n - v.shape[0], v[-1] if v.size else 0.0)
            v = np.concatenate([v, extra])
        v = v / v.sum()
    return v


def _uniform_candidate(n: int) -> np.ndarray:
    """Uniform distribution (already sorted)."""
    v = np.full(n, 1.0 / n, dtype=np.float64)
    return v


def _random_dirichlet_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    """Random Dirichlet sample, then sorted."""
    vals = np.random.dirichlet(alpha * np.ones(n))
    v = np.sort(vals)[::-1]
    return v


def _powerlaw_candidate(n: int, p: float = 1.0) -> np.ndarray:
    """v_i ∝ (i+1)^{-p}, normalised and sorted."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    vals = idx ** (-p)
    v = np.sort(vals)[::-1]
    v = v / v.sum()
    return v


def _exponential_candidate(n: int, beta: float = 0.5) -> np.ndarray:
    """v_i ∝ exp(-beta·i), sorted (already decreasing)."""
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(-beta * i)
    v = vals / vals.sum()
    return v


# ----------------------------------------------------------------------
# Optimisation primitives -----------------------------------------------

def _expgrad_optimize_v(
    v0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    eta0: float = 0.5,
    lam_start: float = 1.0,
    lam_factor: float = 1.04,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on the sorted vector v.
    The surrogate objective uses a soft‑max with temperature λ.
    """
    v = v0.copy()
    best_v = v.copy()
    best_val = _objective(_anti_pair_from_sorted(v))
    lam = lam_start
    it = 0

    while time.time() < deadline:
        # Build anti‑pair arrangement
        a = _anti_pair_from_sorted(v)

        # Convolution and smooth max weights
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # Gradient w.r.t. a
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # Map to gradient w.r.t. sorted v
        grad_v = _grad_v_from_grad_a(grad_a, n)

        # Step size (decays like 1/√t)
        eta = eta0 / np.sqrt(it + 1)

        # Exponentiated‑gradient update (keeps positivity)
        v = v * np.exp(-eta * grad_v)

        # Normalise onto the simplex and enforce monotonicity
        s = v.sum()
        if s <= 0:
            v = np.random.dirichlet(np.ones(n))
        else:
            v = v / s
        v = np.sort(v)[::-1]

        # Track best exact objective
        val = _objective(_anti_pair_from_sorted(v))
        if val < best_val:
            best_val = val
            best_v = v.copy()
            print(f"[ExpGrad it={it:06d}] new best {best_val:.6f}")

        # Temperature schedule – gradually approach the true max
        lam = min(lam * lam_factor, 1e6)

        # Occasionally inject a tiny jitter to escape plateaus
        if it % 5000 == 0:
            jitter = (np.random.rand(n) - 0.5) * 1e-3
            v = np.maximum(v + jitter, 0.0)
            v = v / v.sum()
            v = np.sort(v)[::-1]

        it += 1

    return best_v, best_val


def _local_random_search_v(
    v0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 150_000,
) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing hill‑climbing on the sorted vector v.
    Moves consist of transferring a fraction of mass between two entries.
    """
    v = v0.copy()
    best_v = v.copy()
    best_val = _objective(_anti_pair_from_sorted(v))
    start_temp = 0.01

    it = 0
    while time.time() < deadline and it < max_moves:
        # transfer a random fraction from i to j
        i, j = np.random.randint(n, size=2)
        if i == j or v[i] < 1e-14:
            it += 1
            continue
        delta = v[i] * (np.random.rand() * 0.4)   # up to 40 % move
        v2 = v.copy()
        v2[i] -= delta
        v2[j] += delta

        # renorm & sort
        v2 = np.maximum(v2, 0.0)
        v2 /= v2.sum()
        v2 = np.sort(v2)[::-1]

        val2 = _objective(_anti_pair_from_sorted(v2))

        # acceptance rule
        if val2 < best_val:
            best_val = val2
            best_v = v2.copy()
            v = v2
        else:
            temp = start_temp * (1.0 - it / max_moves)
            if temp > 0 and np.random.rand() < np.exp(-(val2 - best_val) / temp):
                v = v2

        if it % 5000 == 0:
            print(f"[LocalV {it:06d}] best {best_val:.6f}")

        it += 1

    return best_v, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function minimising
        F(a) = 2·n·max(a∗a)/(∑a)²
    (the evaluator normalises the sum automatically).
    The routine respects the given time budget and returns the best
    sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # Core parameters
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # 1) Build a diverse pool of sorted starts (vectors v)
    candidates = []

    warm = _warm_start_sorted(n)
    if warm is not None:
        candidates.append(warm)

    candidates.append(_uniform_candidate(n))

    for alpha in (0.3, 0.7, 1.0, 2.0):
        candidates.append(_random_dirichlet_candidate(n, alpha))

    for p in (0.5, 1.0, 1.5, 2.0, 3.0):
        candidates.append(_powerlaw_candidate(n, p))

    for beta in (0.2, 0.5, 1.0):
        candidates.append(_exponential_candidate(n, beta))

    # ------------------------------------------------------------------
    # 2) Quick evaluation – keep the best start
    best_v = None
    best_val = float("inf")
    for idx, v in enumerate(candidates):
        if v.sum() < 1e-12:
            continue
        val = _objective(_anti_pair_from_sorted(v))
        print(f"[Init {idx:02d}] value {val:.6f}")
        if val < best_val:
            best_val = val
            best_v = v.copy()

    # ------------------------------------------------------------------
    # 3) Main optimisation on the best start
    # Reserve a few seconds for the final up‑sampling stage
    upscale_deadline = deadline - 20.0

    # ----- Exponentiated‑gradient optimisation (smooth then exact max)
    opt_deadline = min(deadline - 10.0, time.time() + 300.0)   # up to ~5 min
    v_opt, val_opt = _expgrad_optimize_v(
        best_v,
        n,
        conv_fft_len,
        corr_fft_len,
        opt_deadline,
        eta0=0.5,
        lam_start=1.0,
        lam_factor=1.04,
    )
    if val_opt < best_val:
        best_val = val_opt
        best_v = v_opt.copy()
        print(f"[ExpGrad] improved to {val_opt:.6f}")

    # ----- Local stochastic hill‑climbing
    if time.time() < deadline - 5.0:
        local_deadline = deadline - 2.0
        v_loc, val_loc = _local_random_search_v(
            best_v,
            n,
            conv_fft_len,
            local_deadline,
            max_moves=150_000,
        )
        if val_loc < best_val:
            best_val = val_loc
            best_v = v_loc.copy()
            print(f"[Local] improved to {val_loc:.6f}")

    # ------------------------------------------------------------------
    # 4) Optional up‑sampling to a finer grid (8192 entries)
    if time.time() < upscale_deadline:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        # Linear interpolation of the sorted vector (preserves monotonicity)
        x_old = np.linspace(0.0, 1.0, n, endpoint=False)
        x_new = np.linspace(0.0, 1.0, n_big, endpoint=False)
        v_big = np.interp(x_new, x_old, best_v)

        v_big = np.maximum(v_big, 0.0)
        v_big = v_big / v_big.sum()

        # Brief optimisation on the larger grid
        big_opt_deadline = min(deadline - 5.0, time.time() + 200.0)
        v_big_opt, val_big_opt = _expgrad_optimize_v(
            v_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            big_opt_deadline,
            eta0=0.5,
            lam_start=1.0,
            lam_factor=1.04,
        )
        if val_big_opt < best_val:
            best_val = val_big_opt
            best_v = v_big_opt.copy()
            print(f"[Big] improved to {val_big_opt:.6f}")

        # Optional tiny local search on the large grid (if time remains)
        if time.time() < deadline - 2.0:
            big_local_deadline = deadline - 1.0
            v_big_loc, val_big_loc = _local_random_search_v(
                best_v,
                n_big,
                conv_fft_len_big,
                big_local_deadline,
                max_moves=100_000,
            )
            if val_big_loc < best_val:
                best_val = val_big_loc
                best_v = v_big_loc.copy()
                print(f"[BigLocal] improved to {val_big_loc:.6f}")

    # ------------------------------------------------------------------
    # 5) Finalise: build the anti‑pair step function and clip
    best_a = _anti_pair_from_sorted(best_v)
    best_a = np.clip(best_a, 0.0, 1000.0)   # safety for the evaluator

    print(f"Final best value {best_val:.6f} (length {len(best_a)})")
    return best_a.tolist()
