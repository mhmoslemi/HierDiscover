assistantfinal**Solution Overview**

The goal is to find a non‑negative vector `a` (the “step heights”) that minimises  

\[
\Phi(a)=\frac{2n\;\max (a*a)}{(\sum a)^2},
\]

where `*` denotes linear convolution.  
Because the objective is invariant to scaling we can fix `∑a = 1`.  
Thus the problem is equivalent to **minimising the maximum entry of the
self‑convolution**.

The previous pipeline searched a large set of random candidates and then
performed a few hundred thousand gradient‑type steps.  
To push the bound a little further we combine two ideas:

1. **Fine‑grained parametric search** – families that are known to be good
   (exponential‑edges, power‑law edges) are explored with a small number of
   continuous parameters (`α` for the decay and a mixture weight `w` with the
   uniform distribution).  A cheap grid followed by a few refinement
   passes gives a starting vector that is already close to the optimum.

2. **Long‑run projected gradient descent** – from the best parametric
   vector we run a pure *projected‑gradient* method on the exact objective.
   The gradient of `max(a*a)` is obtained by the sub‑gradient
   `∂max/∂a = 2·corr(w,a)` where `w` is the uniform weight on the indices
   attaining the maximum.  After each gradient step the vector is projected
   onto the probability simplex (non‑negative, sum = 1).  The loop runs
   until the time budget expires, adaptively shrinking the step‑size when
   no improvement is observed.

Both stages are cheap compared with the 1000 s budget, leaving ample time
for the long gradient phase.  The algorithm is deterministic except for the
initial random Dirichlet warm‑start when no previous solution is available,
so it works reliably under the competition constraints.

The code below implements the whole pipeline.  All helper functions are
defined at top level, no closures are used, and the function
`propose_candidate` is the entry point required by the evaluator.

```python
"""
Propose a low‑evaluation step‑function.

Algorithm
---------
1. Initialise a probability vector (warm‑start from the global
   variable ``height_sequence_1`` if present, otherwise uniform).
2. Fine‑grained parametric search on two‑parameter families:
   – two‑sided exponential edge   a_i ∝ exp(-α·i)+exp(-α·(n‑1‑i))
   – two‑sided power‑law edge    a_i ∝ (i+1)^(-p)+(n‑i)^(-p)
   For each mixture weight w∈{0,0.2,…,1.0} the best decay parameter is
   found by successive grid refinements.  The best candidate becomes the
   start for the next stage.
3. Long projected‑gradient descent on the exact objective.
   The sub‑gradient of max(a*a) is computed by an FFT‑based cross‑correlation.
   After each step the iterate is projected onto the probability simplex.
   Step size is adapted and the loop stops when the deadline is reached.
4. Return the best vector (clipped to the allowed range).

The whole routine respects the 1000 s time budget and prints progress.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact objective value (assumes a is non‑negative, sum≈1)."""
    conv = _full_convolution(a, conv_fft_len)
    return float(2.0 * n * conv.max() / (a.sum() ** 2))


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Returns the projected vector (non‑negative, sum=1).
    """
    if v.ndim != 1:
        raise ValueError("v must be 1‑d")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _projected_gradient_refine(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray, float]:
    """
    Gradient descent on the exact objective with explicit simplex projection.
    """
    a = np.copy(a0)
    a = np.maximum(a, 0.0)
    a /= a.sum()                     # stay on the simplex

    lr = 0.2
    last_improve = time.time()
    max_iter = 500_000               # large upper bound – will stop on deadline

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[ProjGrad n={n} it={it:06d}] new best {val:.6f}")

        # active mask (tolerance to numerical errors)
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()

        # gradient of max(conv) w.r.t a
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # gradient step and projection onto the simplex
        a_candidate = _proj_simplex(a - lr * grad_a)

        # check improvement
        conv_cand = _full_convolution(a_candidate, conv_fft_len)
        val_cand = 2.0 * n * conv_cand.max() / (a_candidate.sum() ** 2)

        if val_cand < val:
            a = a_candidate
            lr = min(lr * 1.02, 1.0)  # slowly enlarge step when moving downhill
        else:
            lr *= 0.7
            if lr < 1e-5:
                break

        if time.time() - last_improve > 5.0:
            # no improvement for a while – stop early
            break

        # gentle decay of learning rate as iterations progress
        lr *= 0.99998

    return best_seq, best_val


# ----------------------------------------------------------------------
def _shape_exponential(alpha: float, n: int) -> np.ndarray:
    """Two‑sided exponential edge (normalized)."""
    i = np.arange(n, dtype=np.float64)
    edge = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
    edge_sum = edge.sum()
    if edge_sum == 0.0:
        return np.full(n, 1.0 / n)
    return edge / edge_sum


def _shape_power(p: float, n: int) -> np.ndarray:
    """Two‑sided power‑law edge (normalized)."""
    i = np.arange(n, dtype=np.float64)
    edge = (i + 1.0) ** (-p) + (n - i) ** (-p)
    edge_sum = edge.sum()
    if edge_sum == 0.0:
        return np.full(n, 1.0 / n)
    return edge / edge_sum

                print(f"[Perturb] new best {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity checks & return
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    # ensure sum not too small (evaluate_sequence would reject)
    if best_seq.sum() < 0.01:
        best_seq = uniform

    total_time = time.time() - start_time
    print(f"\n*** Finished search (time {total_time:.1f}s) – best value ≈ {best_val:.6f} (n={best_seq.shape[0]})")
    return best_seq.tolist()
```
