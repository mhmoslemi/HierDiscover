    conv = _full_convolution(a, conv_fft_len)

    best_val = cur_val

    old_i = np.empty(n)
    old_j = np.empty(n)

    moves = 0
    stall = 0
    inner_steps = 5000
    while time.time() < deadline and (max_total_moves is None or moves < max_total_moves):
        # compute gradient
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-12)
        if not np.any(active):

        w[active] = 1.0 / active.sum()

"""
Search for a non‑negative step‑function that minimizes

        F(a) = 2·n·max(a∗a) / (sum(a)²)

The algorithm works in three stages:

1. **Warm‑start** – use the previously found best sequence (`height_sequence_1`);
   otherwise start from a uniform distribution.

2. **Directed hill‑climbing** – repeatedly move a small amount of mass from a
   “bad” index to a “good” index.  The indices are obtained from the
   sub‑gradient of the current maximal convolution value:
        w  = uniform weight on the active (maximal) positions,
        g  = 2·(w ★ a)   (cross‑correlation, see `_cross_correlation`),
   `src = argmax(g)` is the index whose increase would raise the max the most,
   `tgt = argmin(g)` the index whose increase would lower it the most.
   A move transfers `δ ≤ step_frac·a[src]` from `src` to `tgt`.  The
   convolution is updated *incrementally* in O(n) time, so millions of moves
   fit easily into the 1000 s budget.

   When the search stalls a small random shake is applied and the gradient
   is recomputed.

3. **Refinement on a finer grid** – if time permits the best vector is
   interpolated to 8192 points and the directed hill‑climbing is repeated.
   The final best vector (any length) is returned as a plain Python list.

All helper routines are defined at module level; there are no closures.
Only NumPy is required.
"""

import time
import numpy as np


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via FFT (length 2·len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a length‑len(a) vector.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _directed_hill_climb(
    a: np.ndarray,
    n: int,
    deadline: float,
    step_frac: float = 0.5,
    inner_steps: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climbing that moves mass from the index with the largest
    sub‑gradient (most harmful) to the one with the smallest (most helpful).
    The convolution is updated incrementally, making each move O(n).

    Returns the best vector seen and its exact objective value.
    """
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # initial convolution
    conv = _full_convolution(a, conv_fft_len)
    cur_val = 2.0 * n * conv.max()
    best_val = cur_val
    best_a = a.copy()

    # temporary buffers for slice backup (avoid repeated allocations)
    buf_i = np.empty(n, dtype=np.float64)
    buf_j = np.empty(n, dtype=np.float64)

    moves_done = 0
    stall = 0

    while time.time() < deadline:
        # --- compute sub‑gradient based on the active (maximal) set ---
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-12)
        if not np.any(active):
            break
        w = np.zeros_like(conv)
        w[active] = 1.0 / active.sum()
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        src = int(np.argmax(grad))
        tgt = int(np.argmin(grad))
        if src == tgt:
            # fallback – choose a random source weighted by mass
            src = np.random.choice(n, p=a)
            tgt = np.random.randint(n - 1)
            if tgt >= src:
                tgt += 1

        # ----- inner move loop (several small transfers using the same src/tgt) -----
        for _ in range(inner_steps):
            if time.time() >= deadline:
                break

            if a[src] < 1e-15:
                break

            delta = a[src] * np.random.rand() * step_frac
            if delta <= 0.0:
                continue

            i_start, i_end = src, src + n
            j_start, j_end = tgt, tgt + n

            # backup the two affected slices
            np.copyto(buf_i, conv[i_start:i_end])
            np.copyto(buf_j, conv[j_start:j_end])

            # apply the move (conv update uses the *old* a)
            conv[i_start:i_end] -= 2.0 * delta * a
            conv[j_start:j_end] += 2.0 * delta * a
            a[src] -= delta
            a[tgt] += delta

            new_val = 2.0 * n * conv.max()

            if new_val < cur_val - 1e-14:
                # improvement – keep the change
                cur_val = new_val
                stall = 0
                if new_val < best_val:
                    best_val = new_val
                    best_a = a.copy()
            else:
                # revert
                a[src] += delta
                a[tgt] -= delta
                np.copyto(conv[i_start:i_end], buf_i)
                np.copyto(conv[j_start:j_end], buf_j)
                stall += 1
                if stall > 3000:
                    # too many non‑improving moves – break to recompute gradient
                    break

            moves_done += 1

        # random shake when we have been stuck for a while
        if stall > 5000:
            noise = (np.random.rand(n) - 0.5) * 0.02
            a += noise
            a = np.maximum(a, 0.0)
            a /= a.sum()
            conv = _full_convolution(a, conv_fft_len)
            cur_val = 2.0 * n * conv.max()
            stall = 0

    return best_a, best_val


def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Entry point used by the grader.  Performs the three‑stage optimisation
    described in the module docstring and returns the best sequence found.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5     # safety margin

    # ------------------------------------------------------------------
    # Warm start
    warm = globals().get("height_sequence_1")
    if warm is not None:
        a0 = np.asarray(warm, dtype=np.float64)
        # make sure the vector is not too long; cut or pad to 4096
        n0 = 4096
        if a0.shape[0] > n0:
            a0 = a0[:n0]
        elif a0.shape[0] < n0:
            a0 = np.pad(a0, (0, n0 - a0.shape[0]))
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        if s < 0.01:
            a0 = np.random.dirichlet(np.ones(n0))
            s = a0.sum()
        a0 /= s
    else:
        # fall back to uniform on 4096 points
        n0 = 4096
        a0 = np.full(n0, 1.0 / n0, dtype=np.float64)

    # ------------------------------------------------------------------
    # Phase 1 – hill‑climbing on the base grid (n = 4096)
    print(f"[Phase 1] start hill‑climbing on {n0} points")
    best_a, best_val = _directed_hill_climb(
        a0.copy(), n0, deadline, step_frac=0.5, inner_steps=3000
    )
    print(f"[Phase 1] best value {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – upscale if time permits and repeat the optimisation
    if time.time() + 30.0 < deadline:
        n_big = 8192
        print(f"[Phase 2] upscale to {n_big} points")
        x_old = np.linspace(0.0, 1.0, n0, endpoint=False)
        x_new = np.linspace(0.0, 1.0, n_big, endpoint=False)
        a_big = np.interp(x_new, x_old, best_a)
        a_big = np.maximum(a_big, 0.0)
        a_big /= a_big.sum()

        best_a_big, best_val_big = _directed_hill_climb(
            a_big, n_big, deadline, step_frac=0.4, inner_steps=2500
        )
        if best_val_big < best_val:
            best_a, best_val = best_a_big, best_val_big
        print(f"[Phase 2] best value {best_val:.6f}")

    # ------------------------------------------------------------------
    # Ensure a legal return (non‑negative, bounded)
    best_a = np.clip(best_a, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_a)})")
    return best_a.tolist()
