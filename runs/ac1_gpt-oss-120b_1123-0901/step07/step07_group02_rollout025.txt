"""
Propose a step‑function (non‑negative vector) that minimizes

        F(a) = 2·n·max(a∗a) / (sum(a)²)

The routine is a multi‑phase optimiser:

1.  Generate a diverse pool of starters (warm‑start, Dirichlet,
    power‑law, double‑exponential, sinusoidal, sparse “spike” patterns).
2.  For every starter run a short Adam optimisation on the smooth
    “log‑sum‑exp’’ surrogate of the max‑convolution (soft‑max weights).
    The parameters are kept on the simplex via a soft‑max.
3.  Refine the best candidate with a long exponentiated‑gradient
    sub‑gradient method (multiplicative updates, 1/√t step decay).
4.  Finally perform a simulated‑annealing / hill‑climbing phase that
    moves a random fraction of mass between two indices and accepts
    worsening moves with a temperature that slowly cools.
5.  All phases respect the global budget (default 1000 s) and the
    routine always returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a length‑len(a) vector:
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i = 0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑sum‑exp surrogate of max(conv).
    Returns the best vector (sum≈1) seen and the exact objective value
    2·n·max(conv).
    """
    eps = 1e-15
    logits = np.log(a0 + eps)          # unconstrained parameters
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.07                         # a bit smaller – more stable
    lam = 0.5                         # start temperature for soft‑max
    lam_factor = 1.03

    best_a = a0.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max → simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective (sum(a)=1)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam iter={it:06d}] new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)                 # ∇⟨a,w⟩·a
        grad_log = a * (grad_a - avg)           # chain‑rule through soft‑max

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature schedule → true max
        lam = min(lam * lam_factor, 1e6)

        # occasional tiny random shake when we seem to stall
        if it % 8_000 == 0 and time.time() + 5 > deadline:
            logits += (np.random.rand(n) - 0.5) * 5e-3

        it += 1

    return best_a, best_val


def _subgradient_opt2(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 1_000_000,
    eta0: float = 0.7,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (multiplicative) sub‑gradient descent.
    Returns the best vector encountered and its exact objective.
    """
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s

    best_a = a.copy()
    best_val = float("inf")
    stall = 0
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            stall = 0
            if it % 5_000 == 0:
                print(f"[Subgrad iter={it:07d}] new best {val:.6f}")
        else:
            stall += 1

        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # decaying step size
        eta = eta0 / np.sqrt(it + 1)

        # multiplicative update & renormalise
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        # random shake when stuck
        if stall > 12_000:
            a = np.maximum(a + 0.02 * (np.random.rand(n) - 0.5), 0.0)
            a /= a.sum()
            stall = 0

        it += 1
    return best_a, best_val


def _spike_candidates(n: int) -> list[np.ndarray]:
    """A few “spike’’ candidates with k equally spaced non‑zero entries."""
    out = []
    for k in (32, 64, 128, 256, 512, 1024):
        a = np.zeros(n, dtype=np.float64)
        if k > 0:
            pos = np.round(np.linspace(0, n - 1, k)).astype(int)
            a[pos] = 1.0 / k
        out.append(a)
    return out


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The function respects the given time budget and returns the best
    sequence (as a Python list) found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # Base resolution (n = 4096)
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm start (if any)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None

    # ------------------------------------------------------------------
    # Build a modest but diverse candidate pool
    candidates = []

    # warm start + rotations
    if warm is not None:
        candidates.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            candidates.append(np.roll(warm, shift))

    # uniform
    candidates.append(np.full(n, 1.0 / n))

    # Dirichlet with several concentration parameters
    for alpha in (0.2, 0.5, 1.0, 2.0, 5.0):
        candidates.append(np.random.dirichlet(np.full(n, alpha)))

    # Edge‑concentrated power‑law shapes
    idx = np.arange(n, dtype=np.float64) + 1.0
    prod = idx * (n - idx + 1.0)
    for alpha in (-3.0, -2.0, -1.0, -0.5):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            candidates.append(shape)

    # Symmetric double‑exponential shapes
    i = np.arange(n, dtype=np.float64)
    for gamma in (0.2, 0.5, 1.0, 2.0, 5.0):
        shape = np.exp(-gamma * i) + np.exp(-gamma * (n - 1 - i))
        shape /= shape.sum()
        candidates.append(shape)

    # Positive half‑wave sinusoid
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    candidates.append(shape)

    # Antisymmetric random starters (one side dominant, the other tiny)
    half = n // 2
    for _ in range(3):
        left = np.random.dirichlet(np.ones(half))
        right_noise = np.random.rand(half) * 1e-3    # almost zero
        a = np.concatenate([left, right_noise])
        a /= a.sum()
        candidates.append(a)
        # also the opposite orientation
        a_rev = np.concatenate([right_noise, left])
        a_rev /= a_rev.sum()
        candidates.append(a_rev)

    # Spike patterns
    candidates += _spike_candidates(n)

    # ------------------------------------------------------------------
    # Helper to evaluate the exact objective (sum is 1 → 2·n·max(conv))
    def _objective(a_vec: np.ndarray) -> float:
        conv = _full_convolution(a_vec, conv_fft_len)
        return float(2.0 * n * conv.max())

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Phase 1 – short optimisation of every candidate
    for cid, cand in enumerate(candidates):
        if time.time() >= deadline:
            break
        a0 = np.maximum(cand, 0.0)
        s = a0.sum()
        if s < 0.01:
            a0 = np.random.dirichlet(np.ones(n))
            s = a0.sum()
        a0 /= s

        # allocate a few seconds per candidate, min‑max budgeting
        cand_deadline = min(deadline - 4.0, time.time() + 7.0)

        a_adam, val_adam = _adam_optimize(
            a0, n, conv_fft_len, corr_fft_len, cand_deadline, max_iters=80_000
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Cand {cid}] Adam best {val_adam:.6f}")

        # short sub‑gradient polishing
        if time.time() < deadline:
            sub_deadline = min(deadline - 4.0, time.time() + 7.0)
            a_sub, val_sub = _subgradient_opt2(
                best_seq, n, conv_fft_len, corr_fft_len, sub_deadline, max_iters=150_000
            )
            if val_sub < best_val:
                best_val = val_sub
                best_seq = a_sub.copy()
                print(f"[Cand {cid}] Subgrad best {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – long sub‑gradient run on the current best
    if best_seq is not None and time.time() < deadline - 5:
        a_ref, val_ref = _subgradient_opt2(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=800_000,
        )
        if val_ref < best_val:
            best_val = val_ref
            best_seq = a_ref.copy()
            print(f"[Refine] Subgrad long best {val_ref:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – simulated‑annealing / random hill‑climbing
    if best_seq is not None and time.time() < deadline - 5:
        a = best_seq.copy()
        cur_val = _objective(a)
        best_local = a.copy()
        best_local_val = cur_val
        temperature = 0.05 * cur_val  # initial temperature

        while time.time() < deadline - 2:
            i = np.random.randint(n)
            if a[i] < 1e-12:
                continue
            j = np.random.randint(n)
            if i == j:
                continue
            # move a random fraction (up to 40 %) of mass from i to j
            delta = a[i] * (np.random.rand() * 0.4)
            a2 = a.copy()
            a2[i] -= delta
            a2[j] += delta
            val2 = _objective(a2)

            # Metropolis acceptance
            if val2 < cur_val or np.random.rand() < np.exp((cur_val - val2) / max(1e-12, temperature)):
                a = a2
                cur_val = val2
                if val2 < best_local_val:
                    best_local_val = val2
                    best_local = a2.copy()
                    print(f"[Anneal] new best {val2:.6f}")

            # temperature schedule (slow cooling)
            temperature *= 0.99995
            if temperature < 1e-8:
                temperature = 1e-8

        if best_local_val < best_val:
            best_val = best_local_val
            best_seq = best_local.copy()
            print(f"[Anneal] improved global best {best_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – guarantee a valid return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip for the evaluator's safety limits
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
