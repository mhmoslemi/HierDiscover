def _softmax_gradient_opt(a0, n, conv_fft_len, corr_fft_len, deadline, best_seq, best_val):
    a = np.maximum(a0, 0)
    a = a / a.sum()
    eta0 = 0.5
    decay = 0.9999
    beta = 2.0
    beta_growth = 1.0005
    min_eta = 0.001
    momentum = np.zeros_like(a)
    beta_max = 50.0  # cap
    step = 0
    last_improve = time.time()
    while time.time() < deadline:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        # evaluate objective
        if val < best_val - 1e-12:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[n={n} Opt iter={step}] new best {val:.6f}")

        # softmax weight
        diff = conv - max_c
        # clip diff for numerical stability
        exp_arg = beta * diff
        exp_arg = np.clip(exp_arg, -50, 0)  # negative values
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0:
            # fallback to active set (like earlier)
            active = np.isclose(conv, max_c, atol=1e-12)
            w = np.zeros_like(conv)
            if active.any():
                w[active] = 1.0 / active.sum()
            else:
                w[conv.argmax()] = 1.0
            w_sum = w.sum()
        w = w / w_sum

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # maybe use momentum:
        momentum = 0.9 * momentum + 0.1 * grad

        eta = max(min_eta, eta0 * (decay ** step))

        a = a * np.exp(-eta * momentum)

        a_sum = a.sum()
        if a_sum <= 0 or np.isnan(a_sum):
            a = np.random.dirichlet(np.ones(n))
            a_sum = a.sum()
        a = a / a_sum

        # optionally gradually increase beta
        if beta < beta_max:
            beta = min(beta_max, beta * beta_growth)

        # random perturb if stagnating
        if time.time() - last_improve > 5.0:
            perturb = (np.random.rand(n) - 0.5) * 0.02
            a = a * np.exp(perturb)
            a = a / a.sum()
            last_improve = time.time()

        step += 1

    return best_seq, best_val

"""
search_proposal.py

Search for a non‑negative step‑function that minimises

        F(a) = 2·n·max(a∗a) / (sum(a)²)

The key observation is that monotone power‑law shapes a_i ∝ (i+1)^p
produce a much smaller peak of the autocorrelation than symmetric
templates (uniform, Hann‐windows, etc.).  The script therefore:

  1. Generates a large family of simple parametric candidates:
       – uniform
       – power‑law (i+1)^p   for many real p
       – exponential   exp(α·i)      for many α
  2. Evaluates each candidate fast using FFT‑based convolution and keeps
     the best few.
  3. Refines the best candidate with a smoothed (soft‑max) sub‑gradient
     descent that works directly on the true max‑convolution.  The
     update is mirror‑descent (exponentiated‑gradient) so the vector
     stays non‑negative; a tiny momentum term (Adam‑style) speeds
     convergence.
  4. Returns the best sequence found as a plain Python list.
"""

import time
import numpy as np
import math

# ----------------------------------------------------------------------
# FFT helpers ------------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n-1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """
    Fast evaluation of the objective (scale invariant version).
    The vector a is assumed to be normalised to sum(a)=1.
    Returns 2·n·max(conv(a,a)).
    """
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c)


# ----------------------------------------------------------------------
# Parameteric families ---------------------------------------------------

def _candidate_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^p, normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64) + 1.0
    a = np.power(i, p, out=None)
    a /= a.sum()
    return a


def _candidate_exponential(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ exp(alpha * i), normalised to sum = 1 (alpha≥0)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(alpha * i, out=None)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed mirror‑descent optimiser -------------------------------------

def _softmax_mirror_descent(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray | None,
    best_val: float,
) -> tuple[np.ndarray | None, float]:
    """
    Exponentiated‑gradient optimisation using a soft‑max weighting of the
    convolution entries.  This provides a smoother sub‑gradient than the
    “hard‑active‑set” used in the reference implementation.
    """
    # start from a safe point (probability simplex)
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 0.01 or np.isnan(s):
        a = np.random.dirichlet(np.ones(n))
    else:
        a = a / s

    # hyper‑parameters (tuned for a few seconds of work)
    eta0 = 0.7          # initial step size
    decay = 0.99996     # per‑iteration decay
    min_eta = 0.001
    beta = 2.0          # soft‑max temperature (starts low → smooth)
    beta_mul = 1.001    # slowly increase temperature
    beta_max = 80.0

    # momentum / Adam‑style buffers
    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment
    eps = 1e-12
    beta1, beta2 = 0.9, 0.999

    it = 0
    last_improve = time.time()

    while time.time() < deadline:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c                # sum(a)=1 → scale invariant

        # record best
        if val < best_val - 1e-12:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[n={n} iter={it:07d}] new best {val:.6f}")

        # ----- soft‑max weighting ------------------------------------------------
        # w[k] = exp(beta * (conv[k] - max_c))  (>=0, max weight = 1)
        exp_arg = beta * (conv - max_c)
        # numeric stability: all entries ≤ 0, clip to avoid underflow
        exp_arg = np.clip(exp_arg, -50.0, 0.0)
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0 or np.isnan(w_sum):
            # fallback to hard active set (should not happen)
            active = np.isclose(conv, max_c, atol=1e-12)
            w = np.zeros_like(conv)
            if active.any():
                w[active] = 1.0 / active.sum()
            else:
                w[conv.argmax()] = 1.0
        else:
            w /= w_sum

        # gradient of the weighted max‑convolution
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # ----- Adam‑style momentum update ---------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        # step size (decaying)
        eta = max(min_eta, eta0 * (decay ** it))

        # exponentiated‑gradient (mirror descent on the simplex)
        a = a * np.exp(-eta * m_hat / (np.sqrt(v_hat) + eps))

        # renormalise onto the probability simplex
        s = a.sum()
        if s <= 0.0 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        # slowly increase the soft‑max temperature – towards the true max
        if beta < beta_max:
            beta = min(beta_max, beta * beta_mul)

        # occasional random shake if we have stalled
        if time.time() - last_improve > 5.0:
            jitter = (np.random.rand(n) - 0.5) * 0.02
            a = a * np.exp(jitter)
            a = a / a.sum()
            last_improve = time.time()

        it += 1

    return best_seq, best_val

            break
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 0.01:
            cand = np.random.dirichlet(np.ones(n))
            s = cand.sum()
        cand = cand / s
        val = _objective(cand, n, conv_fft_len)

        scores.append((val, cand))
        if val < best_val:
            best_val = val
            best_seq = cand.copy()

    scores.sort(key=lambda x: x[0])
    top_candidates = [c for _, c in scores[:top_k]]

    # ------------------------------------------------------------------
    # Refine each top candidate with the soft‑max mirror descent.
    for start_id, a0 in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        init_val = _objective(a0, n, conv_fft_len)
        print(f"--- Refinement start #{start_id + 1}/{len(top_candidates)} "
              f"(initial val={init_val:.6f})")
        best_seq, best_val = _softmax_mirror_descent(
            a0,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            best_seq,
            best_val,
        )

    # ------------------------------------------------------------------
    # If time permits, upscale to a finer grid and polish once more.
    time_left = deadline - time.time()
    if time_left > 30.0 and best_seq is not None:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        # linear interpolation onto larger grid
        x_old = np.linspace(0.0, 1.0, n, endpoint=False)
        x_big = np.linspace(0.0, 1.0, n_big, endpoint=False)
        a_big = np.interp(x_big, x_old, best_seq)
        a_big = np.maximum(a_big, 0.0)
        a_big /= a_big.sum()

        print("--- Final polishing on n=8192")
        best_seq, best_val = _softmax_mirror_descent(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            best_seq,
            best_val,
        )
        n = n_big

    # ------------------------------------------------------------------
    # Safety fallback
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the evaluator's admissible range
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"*** search finished – best objective ≈ {best_val:.6f} (n={n})")
    # Return plain Python list for the external harness
    return best_seq.tolist()
