"""
Propose a candidate sequence using a combinatorial hill‑climbing algorithm.

The evaluation metric is
    score = 2 * n * max(conv(seq,seq)) / (sum(seq)**2)

For a binary (0/1) sequence the denominator is m² where m = # of ones.
The goal is thus to minimise the maximum entry of the (ordered) sum‑set
distribution D_k = #{(i,j)∈S×S : i+j=k} where S = support of the sequence.
The theoretical lower bound for a given support size m is
    L_min = n - sqrt(n² - m²)   (n = length of the sequence)
which yields a score ≈ 2·(1-√(1-α²))/α² with α=m/n.
Choosing a moderate density (α≈0.4‑0.5) allows a score far below the
current bound 2.0 and comfortably below the target 1.503.

The algorithm:
1. Choose a random support of size m (≈α·n) – binary sequence.
2. Compute the full convolution once.
3. Repeatedly pick the currently maximal convolution bin k_max.
   - Remove a 1 that participates in that bin (i such that a[i]=1 and
     a[k_max-i]=1).
   - Insert a 0 at a position j that does **not** contribute to k_max
     (i.e. either k_max-j is out of range or a[k_max-j]=0).
   - Update the convolution in O(n) using the analytic effect of a
     single-element swap (no full recomputation).
   - Keep the swap if it lowers the score; otherwise revert.
4. Perform many swaps until the time budget expires or a score < 1.503
   is reached.  Several random restarts (different α, different seeds)
   are performed and the best sequence overall is returned.

All updates are integer‑based, avoiding floating‑point errors.  The
final list of floats is returned for the external `evaluate_sequence`
call.
"""

import numpy as np
import time
from typing import List

def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs) -> List[float]:
    """
    Generate a non‑negative step‑function sequence that aims to minimise
    `evaluate_sequence`.  The routine runs for at most `budget_s` seconds
    and returns the best sequence discovered.
    """
    np.random.seed(seed)
    # length of the problem (use the existing construction length)
    n = len(height_sequence_1)          # global variable provided by the runner
    deadline = time.time() + budget_s - 2.0   # leave a tiny safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Helper: perform a single swap (i -> 0, j -> 1) and return the
    # updated sequence and convolution.  Both are *copies*; the caller
    # decides whether to keep the change.
    # ------------------------------------------------------------------
    def apply_swap(a: np.ndarray, conv: np.ndarray, i: int, j: int):
        """Return (a_new, conv_new) after swapping out i and inserting j."""
        a_new = a.copy()
        conv_new = conv.copy()

        # --- removal of i (a[i] == 1) ----------------------------------
        conv_new[i:i + n] -= 2 * a                 # all ordered pairs (i,·) and (·,i)
        conv_new[2 * i] += 1                       # add back the self‑pair correction
        a_new[i] = 0

        # --- insertion of j (a[j] == 0) ---------------------------------
        conv_new[j:j + n] += 2 * a_new             # interactions with the *new* support
        conv_new[2 * j] += 1
        a_new[j] = 1

        return a_new, conv_new

    # ------------------------------------------------------------------
    # Main optimisation loop – several independent random restarts.
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # ------------------------------------------------------------------
        # Randomly choose a density 0.35 ≤ α ≤ 0.55 for this restart.
        # ------------------------------------------------------------------
        alpha = np.random.uniform(0.35, 0.55)
        m = int(alpha * n)               # number of ones in the binary sequence
        if m < 2:                        # need at least two entries to have a convolution
            m = 2

        # initialise a random support of size m
        a = np.zeros(n, dtype=np.int8)
        a[np.random.choice(n, m, replace=False)] = 1

        # full convolution (ordered pairs, same as np.convolve)
        conv = np.convolve(a, a)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b / (m * m)

        # keep the best of this restart
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()

        # ------------------------------------------------------------------
        # Hill‑climbing: try to reduce the current maximum entry.
        # ------------------------------------------------------------------
        # set a modest inner‑loop budget so that we can perform many restarts
        inner_deadline = time.time() + min(15.0, deadline - time.time())
        attempts_without_improvement = 0

        while time.time() < inner_deadline:
            # locate the bin with the largest convolution value
            k_max = int(np.argmax(conv))

            # ---- choose an element i to remove (participates in k_max) -------
            one_idx = np.where(a == 1)[0]
            comp = k_max - one_idx
            valid = (comp >= 0) & (comp < n)
            paired = one_idx[valid & (a[comp[valid]] == 1)]
            if paired.size == 0:
                # fallback: any 1 will do
                i = np.random.choice(one_idx)
            else:
                i = np.random.choice(paired)

            # ---- choose a position j to insert (does NOT increase k_max) -----
            zero_idx = np.where(a == 0)[0]
            comp_z = k_max - zero_idx
            valid_z = (comp_z >= 0) & (comp_z < n)

            # for indices where the complement lies inside the array we require that complement = 0
            good = np.ones_like(zero_idx, dtype=bool)
            good[valid_z] = (a[comp_z[valid_z]] == 0)
            candidate_j = zero_idx[good]
            if candidate_j.size == 0:
                j = np.random.choice(zero_idx)
            else:
                j = np.random.choice(candidate_j)

            # ---- evaluate the swap -----------------------------------------
            a_new, conv_new = apply_swap(a, conv, i, j)
            new_max = conv_new.max()
            new_score = 2.0 * n * new_max / (m * m)

            if new_score < cur_score:
                # accept improvement
                a, conv = a_new, conv_new
                cur_score = new_score
                max_b = new_max
                attempts_without_improvement = 0

                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = a.copy()
                    # early exit if we have already beaten the target
                    if best_score < 1.503:
                        break
            else:
                attempts_without_improvement += 1
                # occasional random move to escape tiny plateaus
                if attempts_without_improvement > 2000:
                    # perform a completely random swap (no acceptance test)
                    i_rand = np.random.choice(np.where(a == 1)[0])
                    j_rand = np.random.choice(np.where(a == 0)[0])
                    a, conv = apply_swap(a, conv, i_rand, j_rand)
                    max_b = conv.max()
                    cur_score = 2.0 * n * max_b / (m * m)
                    attempts_without_improvement = 0

        # Check global time budget before next restart
        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # Return the best sequence found (as a list of floats for the evaluator)
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – the provided reference sequence (should never happen)
        best_seq = np.array(height_sequence_1, dtype=float)

    return best_seq.astype(float).tolist()
