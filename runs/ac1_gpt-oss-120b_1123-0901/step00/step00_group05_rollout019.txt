Pseudo:

def gradient_step(a, epsilon, lr):
    b = np.convolve(a, a)   # shape (2n-1,)
    b_max = np.max(b)
    # softmax weights
    w = np.exp((b - b_max) / epsilon)
    w /= np.sum(w)
    # compute gradient
    conv = np.convolve(w, a[::-1])
    g = 2 * conv[len(a)-1: len(a)-1 + len(a)]  # slice of length n
    # step
    a_new = a - lr * g
    # ensure nonnegative
    a_new = np.maximum(a_new, 0)
    # renorm to sum=1
    sum_a = np.sum(a_new)
    if sum_a == 0:
        a_new = np.full_like(a, 1.0 / len(a))
    else:
        a_new /= sum_a
    return a_new

"""
Propose a step‑function sequence with a low value of the evaluation
function (2 * n * max(conv(a,a)) / sum(a)^2).  The algorithm performs a
large number of short random restarts.  Each restart starts from a random
probability vector (the sequence is normalised to sum 1) and runs a short
gradient‑descent on a smooth approximation of the max‑convolution.  The
gradient of the soft‑max of the convolution is easy to compute with a
single convolution.  After each gradient step the vector is projected back
onto the probability simplex (non‑negative, sum = 1).  The best sequence
found is kept globally.  Occasionally the current best sequence is
interpolated to a larger length (to exploit a shape that works for any
resolution) and the larger candidate is also evaluated.

The method uses only NumPy (and the already imported evaluate_sequence)
and respects the 1000 s budget.  Progress is printed to the console.
"""

import numpy as np
import time

def _project_onto_simplex(v):
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("simplex projection expects a 1‑d array")
    n = v.shape[0]
    # shift by the mean of the sorted values (algorithm of Wang & Carreira‑Perpinán)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return the uniform vector
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1  # zero‑based index of the last positive element
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # numerical safety
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(v, 1.0 / n)
    else:
        w /= w_sum
    return w


def _gradient_step(a, eps, lr):
    """
    One gradient‑descent step on a smoothed max‑convolution objective.
    The gradient is computed as 2 * conv(w, a[::-1]) where w is a soft‑max
    over the convolution a*a.
    """
    n = a.shape[0]
    # convolution a with itself (full length = 2*n-1)
    b = np.convolve(a, a)
    # soft‑max over the convolution (focuses on the current maximum)
    b_max = b.max()
    # keep eps strictly positive
    w = np.exp((b - b_max) / max(eps, 1e-12))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(b, 1.0 / b.size)
    else:
        w /= w_sum
    # gradient of the smoothed max (size n)
    conv = np.convolve(w, a[::-1])
    grad = 2.0 * conv[n - 1:n - 1 + n]
    # move opposite to the gradient (we want to reduce the max)
    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0.0)
    # project back onto the simplex (sum = 1, non‑negative)
    a_proj = _project_onto_simplex(a_new)
    return a_proj

    smoothed version of the max‑convolution objective.  The best sequence
    found before the time budget expires is returned.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # small safety margin

    # global best
    best_seq = None
    best_score = float("inf")

    # try the provided construction if it exists
    try:
        init = list(height_sequence_1)  # may raise NameError
        sc = evaluate_sequence(init)
        if sc < best_score:
            best_score = sc
            best_seq = init
            print(f"[init] height_sequence_1 score = {best_score:.6f}")
    except Exception:
        pass

    # main time‑bounded search loop
    while time.time() < deadline:
        # pick a length for this restart (moderate size for speed)
        n = np.random.randint(150, 801)  # 150 – 800
        # start from a random probability vector (Dirichlet‑like)
        a = np.random.rand(n)
        a /= a.sum()

        # hyper‑parameters for the inner optimisation
        lr = 0.2                     # initial learning rate
        eps = 0.1                    # initial smoothing parameter
        lr_decay = 0.997
        eps_decay = 0.990
        inner_iters = 180            # number of gradient steps

        # inner optimisation
        for it in range(inner_iters):
            if time.time() >= deadline:
                break

            a = _gradient_step(a, eps, lr)

            # occasional random perturbation to escape plateaus
            if np.random.rand() < 0.08:
                i, j = np.random.randint(n, size=2)
                if i != j:
                    delta = lr * 0.2 * a[i]
                    a[i] -= delta
                    a[j] += delta
                    a = _project_onto_simplex(a)

            # decay schedules
            lr = max(lr * lr_decay, 1e-5)
            eps = max(eps * eps_decay, 1e-6)

            # evaluate every few iterations
            if (it % 10) == 0:
                score = evaluate_sequence(a.tolist())
                if score < best_score:
                    best_score = score
                    best_seq = a.tolist()
                    print(f"[improve] n={n} iter={it} score={best_score:.6f}")

        # after the short run we try to upscale the current best shape
        if best_seq is not None:
            src = np.array(best_seq, dtype=float)
            src_n = src.shape[0]
            # choose a larger target length (does not need to be huge)
            target_len = 3000
            if src_n < target_len:
                src_x = np.linspace(0.0, 1.0, src_n)
                tgt_x = np.linspace(0.0, 1.0, target_len)
                # linear interpolation (fast, no external libs)
                tgt = np.interp(tgt_x, src_x, src)
                # ensure validity and normalise
                tgt = np.maximum(tgt, 0.0)
                s = tgt.sum()
                if s > 0:
                    tgt /= s
                else:
                    tgt = np.full(target_len, 1.0 / target_len)
                up_score = evaluate_sequence(tgt.tolist())
                if up_score < best_score:
                    best_score = up_score
                    best_seq = tgt.tolist()
                    print(f"[upscale] up_len={target_len} score={best_score:.6f}")

    # fallback in the unlikely case nothing was found
    if best_seq is None:
        n = 200
        best_seq = [1.0 / n] * n
        best_score = evaluate_sequence(best_seq)
        print(f"[fallback] using uniform length {n}, score={best_score:.6f}")

    # final sanity check (evaluate once more)
    try:
        final_score = evaluate_sequence(best_seq)
        if final_score > best_score:
            print("[warning] final evaluation higher than recorded best")
    except Exception:
        print("[error] final evaluation failed, returning fallback")
        best_seq = [1.0 / len(best_seq)] * len(best_seq)

    return best_seq
