"""
Propose a step‐function sequence with a low evaluation score.

The main idea is to use a power‑law decreasing sequence a_i = (i+1)^(-α).
For 0 < α < 1 the convolution a*a becomes flatter than the uniform case,
and the quantity 2·n·max(a*a)/sum(a)^2 can be driven below 1.5.
We sweep a few values of α, keep the best, and then apply a tiny
swap‑based local search to possibly improve it further.
All numbers stay in [0,1000] and the sum is far above the 0.01 safety limit.
"""

import time
import math
import numpy as np

def _generate_power_law(n: int, alpha: float) -> list[float]:
    """Return [ (i+1)^(-alpha) for i in 0..n-1 ]."""
    # using a list‑comprehension is fast and avoids numpy overhead
    return [(i + 1) ** (-alpha) for i in range(n)]

def _try_sequence(seq: list[float], best_score: float, best_seq: list[float]) -> tuple[float, list[float]]:
    """Evaluate seq, keep it if it improves the current best."""
    score = evaluate_sequence(seq)
    if score < best_score:
        print(f"  New best score {score:.6f}")
        return score, seq.copy()
    return best_score, best_seq

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a sequence whose evaluation score is as low as possible.
    The function runs for at most `budget_s` seconds and returns the
    best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin for final prints
    deadline = start_time + budget_s - 0.5

    # Length of the construction – try to reuse the existing length
    n = len(globals().get('height_sequence_1', [])) or 2100

    # ------------------------------------------------------------------
    # 1) Sweep a few power‑law exponents α
    # ------------------------------------------------------------------
    alphas = np.linspace(0.45, 0.85, 41)               # 41 candidates
    best_score = float('inf')
    best_seq = None

    print(f"Starting sweep over {len(alphas)} alphas, sequence length {n}")

    for alpha in alphas:
        # generate two candidates: normal and reversed (mirrored) order
        seq = _generate_power_law(n, alpha)
        # forward order
        best_score, best_seq = _try_sequence(seq, best_score, best_seq)
        # reversed order (large values at the opposite end)
        best_score, best_seq = _try_sequence(seq[::-1], best_score, best_seq)

        if time.time() >= deadline:
            print("Time limit reached during α‑sweep")
            break

    # ------------------------------------------------------------------
    # 2) Very cheap local refinement: random swaps and tiny perturbations
    # ------------------------------------------------------------------
    curr_seq = best_seq
    attempts = 0
    # a simple hill‑climbing loop; we stop when time runs out
    while time.time() < deadline:
        attempts += 1
        # Occasionally perform a full random perturbation to escape tiny plateaus
        if np.random.rand() < 0.02:
            idx = np.random.randint(n)
            new_seq = curr_seq.copy()
            delta = np.random.randn() * 0.01          # tiny Gaussian step
            new_seq[idx] = max(0.0, new_seq[idx] + delta)
        else:
            # swap two random positions – this changes only the convolution
            i, j = np.random.randint(0, n, size=2)
            if i == j:
                continue
            new_seq = curr_seq.copy()
            new_seq[i], new_seq[j] = new_seq[j], new_seq[i]

        score = evaluate_sequence(new_seq)
        if score < best_score:
            best_score = score
            best_seq = new_seq.copy()
            curr_seq = new_seq
            print(f"  Improvement after {attempts} attempts: {best_score:.6f}")

    print(f"Search finished after {time.time() - start_time:.2f}s.")
    print(f"Best score obtained: {best_score:.6f}")
    return best_seq
