def projected_gradient_descent(n, steps, step_size, lambda_val):
    a = np.full(n, 1/n)
    for t in range(steps):
        conv = np.convolve(a, a)
        max_conv = conv.max()
        # compute weights
        w = np.exp(lambda_val * (conv - max_conv))
        w /= w.sum()
        # compute gradient via correlation
        corr = np.correlate(w, a, mode='full')
        # extract relevant slice
        grad = 2 * corr[len(conv)-1 : len(conv)-1 + n]
        a -= step_size * grad
        # clip non-negative
        a = np.maximum(a, 0)
        # renorm to sum=1
        total = a.sum()
        if total == 0:
            a = np.full(n, 1/n)
        else:
            a /= total
        # optional step adjust step_size decreasing
    return a

"""
Propose a candidate step‑function heights sequence.
-----------------------------------------------------------------
The objective to minimise is

    obj(a) = 2 * n * max(conv(a,a)) / (sum(a)**2)

with a_i ≥ 0.  The term is scale‑invariant, therefore we keep the
vector normalised to sum = 1 during the optimisation.  The algorithm
uses a projected gradient descent on a smooth approximation of the
“max” (soft‑max) together with a simple annealing schedule.  The
gradient of the smoothed objective can be expressed by a single
correlation, so each iteration costs only a few fast NumPy
convolutions / correlations (O(n log n)).  Random restarts and a
tiny amount of stochastic perturbation are employed to escape
possible local minima.  The best sequence found within the allocated
time is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """Exact objective value (no clipping) for a normalised sequence."""
    n = seq.shape[0]
    conv = np.convolve(seq, seq)
    max_conv = conv.max()
    # sum(seq) == 1 after normalisation → denominator = 1
    return 2.0 * n * max_conv


def _softmax_weights(conv: np.ndarray, lam: float) -> np.ndarray:
    """
    Compute a probability distribution that concentrates on the largest
    entries of `conv`.  The distribution is the soft‑max with parameter
    `lam` (larger → sharper).  Numerical stability is ensured by subtracting
    the maximum before exponentiation.
    """
    max_c = conv.max()
    # all entries are ≤ max_c, therefore the exponent is non‑positive
    w = np.exp(lam * (conv - max_c))
    w_sum = w.sum()
    if w_sum == 0.0:                     # should never happen, but be safe
        w = np.ones_like(w) / w.size
    else:
        w /= w_sum
    return w


def _gradient(seq: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the smoothed objective (soft‑max of the convolution).
    For a given weight vector `w` the gradient equals
        2 * corr(w, seq)
    where `corr` denotes the full discrete correlation.
    """
    n = seq.shape[0]
    conv = np.convolve(seq, seq)
    w = _softmax_weights(conv, lam)

    # full correlation (length = len(w)+len(seq)-1 = 3*n-2)
    corr = np.correlate(w, seq, mode='full')
    # slice that corresponds to shifts 0 … n‑1
    start = len(conv) - 1                # = 2*n-2
    grad = 2.0 * corr[start:start + n]   # shape (n,)
    return grad


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the simplex {x ≥ 0, sum(x)=1}."""
    # Simple scaling works because we never allow negative entries after clipping.
    s = v.sum()
    if s <= 0.0:
        # fallback to uniform distribution
        return np.full_like(v, 1.0 / v.size)
    return v / s


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 950) -> list[float]:
    """
    Search for a sequence of non‑negative heights that minimises
    `evaluate_sequence`.  The function returns the best sequence found
    before the time budget expires.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Maximum wall‑clock time (seconds) allocated to the search.

    Returns
    -------
    list[float]
        The best sequence discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s

    # --------------------------------------------------------------
    # Determine the problem size (take the length of the provided
    # construction if it exists, otherwise fall back to a moderate size).
    try:
        init_seq = np.array(list(height_sequence_1), dtype=float)
        n = init_seq.size
        # normalise to sum‑one for the internal optimiser
        init_seq = _project_onto_simplex(init_seq)
    except NameError:
        # No global `height_sequence_1`; use a default length.
        n = 1500
        init_seq = np.full(n, 1.0 / n)

    # Evaluate the starting point with the official evaluator.
    best_seq = init_seq.copy()
    try:
        best_score = evaluate_sequence(best_seq.tolist())
    except Exception:
        best_score = float('inf')
    print(f"[init] length={n}, baseline score={best_score:.6f}")

    # --------------------------------------------------------------
    # Hyper‑parameters for the inner descent.
    max_iters_per_restart = 300          # iterations for each random start
    max_restarts = 2000                  # upper bound – stopped by time
    step_initial = 0.5                   # initial learning‑rate factor
    lam_initial = 2.0                    # initial soft‑max sharpness
    step_decay = 0.995                   # geometric decay per iteration
    lam_growth = 1.005                   # increase `lam` slowly

    restarts = 0
    while time.time() < deadline and restarts < max_restarts:
        restarts += 1

        # start from a random point on the simplex (Dirichlet)
        a = np.random.dirichlet(np.ones(n))

        step = step_initial
        lam = lam_initial

        for it in range(max_iters_per_restart):
            if time.time() >= deadline:
                break

            # ----- evaluate current point (fast exact objective)
            cur_obj = _objective(a)

            # Compare with the best known (using the official evaluator only
            # for the final candidate – this cheap surrogate is sufficient
            # for the inner loop).
            if cur_obj < best_score:
                # Store a copy before any later mutation.
                best_score = cur_obj
                best_seq = a.copy()
                print(f"[improve] restart={restarts}, iter={it}, "
                      f"obj={cur_obj:.6f}")

            # ----- gradient step (projected)
            grad = _gradient(a, lam)

            # Normalise the step to avoid overshooting.
            max_abs = np.max(np.abs(grad)) + 1e-12
            a = a - step * (grad / max_abs)

            # Enforce non‑negativity and re‑normalise onto the simplex.
            a = np.maximum(a, 0.0)
            a = _project_onto_simplex(a)

            # ----- schedule updates
            step *= step_decay
            lam = min(30.0, lam * lam_growth)

            perturb = np.random.normal(scale=0.01, size=n)
            a = np.maximum(best_seq + perturb, 0.0)
            a = _project_onto_simplex(a)

    # --------------------------------------------------------------
    # Return the best sequence found (as a plain Python list).
    # The official evaluator is scale‑invariant; we keep the sum equal
    # to the original normalisation used by the reference solutions
    # (sqrt(2*n)) to avoid unnecessary clipping.
    scale_factor = np.sqrt(2.0 * n) / best_seq.sum()
    final_seq = (best_seq * scale_factor).tolist()

    # Quick sanity check (will be printed by the harness if something goes wrong)
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[final] best_score={best_score:.6f}, evaluator={final_score:.6f}")
    except Exception:
        print("[final] evaluation failed – returning the raw sequence.")

    return final_seq
