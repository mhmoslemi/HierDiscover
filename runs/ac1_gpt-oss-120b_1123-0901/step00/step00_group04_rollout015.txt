def propose_candidate(seed=0, budget_s=1000, **kwargs):
    import numpy as np, time, random
    # set seed
    np.random.seed(seed)
    random.seed(seed)
    deadline = time.time() + budget_s - 5
    best_seq = None
    best_score = float('inf')
    lengths = [50, 80, 120, 200]  # can adjust within time
    # First try known shape: arcsine-like
    for n in lengths:
        # some deterministic shapes
        for (alpha, beta) in [( -0.5, -0.5), (-1, -1), (-0.8, -0.8), ( -0.3, -0.3), (0.5,0.5)]:
            a = ((np.arange(n)+1) ** alpha) * ((n - np.arange(n)) ** beta)
            a = a / a.sum()
            score = evaluate_sequence(a.tolist())
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"Initial shape n={n} alpha={alpha} beta={beta} score={score}")
        # random shape sampling
        for _ in range(200):
            alpha = np.random.uniform(-2, 2)
            beta = np.random.uniform(-2, 2)
            a = ((np.arange(n)+1) ** alpha) * ((n - np.arange(n)) ** beta)
            # ensure positivity (should be positive)
            a = a / a.sum()
            score = evaluate_sequence(a.tolist())
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"Random shape n={n} alpha={alpha:.3f} beta={beta:.3f} score={score}")
        # gradient descent from best of this n
        a = best_seq if best_seq is not None and len(best_seq)==n else np.random.rand(n)
        a = a / a.sum()
        it = 0
        last_impr = 0
        while time.time() < deadline:
            # compute score via evaluation function to be accurate
            score = evaluate_sequence(a.tolist())
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"Grad descent n={n} it={it} score={score}")
                last_impr = it
            # compute gradient
            b = np.convolve(a, a)
            max_b = b.max()
            # Use tolerance to detect ties
            active_idx = np.where(np.abs(b - max_b) < 1e-12)[0]
            g = np.zeros_like(a)
            for k in active_idx:
                lo = max(0, k-(n-1))
                hi = min(k, n-1)
                i_vals = np.arange(lo, hi+1)
                j_vals = k - i_vals
                g[i_vals] += 2 * a[j_vals]
            # step size schedule
            step = 0.5 / np.sqrt(it + 1)
            a_new = a - step * g
            # ensure non-negative and renorm
            a_new = np.maximum(a_new, 0)
            if a_new.sum() == 0:
                a_new = np.random.rand(n)
            a = a_new / a_new.sum()
            it += 1
            # restart if stagnated
            if it - last_impr > 2000:
                a = np.random.rand(n)
                a = a / a.sum()
                last_impr = it
                print(f"Restarted at iteration {it} for n={n}")
            if it > 20000:  # limit per n
                break
    # final refinement with LP as earlier approach optional if time left
    if time.time() < deadline:
        if best_seq is not None:
            a = best_seq.copy()
            # run get_good_direction_to_move_into a few times
            for _ in range(20):
                new_seq = get_good_direction_to_move_into(a.tolist())
                if new_seq is None:

                new_score = evaluate_sequence(new_seq)
                if new_score < best_score:
                    best_score = new_score
                    best_seq = np.array(new_seq)
                    print(f"LP refinement improved to {new_score}")
                a = np.array(new_seq)
                if time.time() >= deadline:
                    break
    # ensure final sequence as list

"""
Propose a step‑function height sequence with a small evaluation score.

The algorithm combines three ideas:
1. Randomly generate simple analytical shapes (power‑law / beta‑like) and keep
   the best one found.
2. Refine promising candidates with a projected sub‑gradient descent that
   reduces the maximal convolution value (the only term influencing the
   evaluation function after normalisation).
3. Optionally improve the current best using the LP‑based direction from the
   reference implementation (a small convex step that keeps the convolution
   bound while increasing the total sum).

All operations respect the non‑negativity constraint, the sum is kept equal
to 1 (any other normalisation yields the same evaluation because the objective
is scale‑invariant) and the function stops well before the 1000 s budget.
"""
import time
import random

import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper functions (no closures, no lambdas)
# ----------------------------------------------------------------------


def project_to_simplex(v):
    """Project a non‑negative vector onto the simplex {x≥0, Σx = 1}."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0:
        # fallback to uniform distribution
        return np.full_like(v, 1.0 / v.size)
    return v / s


def subgradient_step(a, step):
    """
    Perform one projected sub‑gradient step on a.
    a – current probability vector (sum equals 1)
    step – positive step size
    Returns a new probability vector.
    """
    n = a.size
    conv = np.convolve(a, a)
    max_val = conv.max()
    # indices where the maximal value is attained (allow tiny tolerance)
    active = np.where(np.abs(conv - max_val) <= 1e-12)[0]

    g = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        i = np.arange(lo, hi + 1)
        j = k - i
        g[i] += 2.0 * a[j]

    a_new = a - step * g
    a_new = np.maximum(a_new, 0.0)
    return project_to_simplex(a_new)


def random_beta_shape(n, alpha, beta):
    """Generate a shape a_i ∝ (i+1)^α·(n‑i)^β and normalise to sum 1."""
    i = np.arange(n)
    a = ((i + 1) ** alpha) * ((n - i) ** beta)
    # avoid division by zero for extreme negative exponents
    a = np.where(np.isfinite(a), a, 0.0)
    if a.sum() == 0:
        a = np.ones(n)
    return a / a.sum()


def solve_convolution_lp(f_seq, rhs):
    """
    Solve the LP:
        maximise   Σ b_j
        subject to   (f_seq * b)_k ≤ rhs   (for all k)
                     b_j ≥ 0
    Returns the optimal b (numpy array) or None if the LP fails.
    """
    n = len(f_seq)
    c = -np.ones(n)                     # maximise → minimise -Σb
    A = []
    b = []
    # build the convolution constraints
    for k in range(2 * n - 1):
        row = np.zeros(n)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                row[j] = f_seq[i]
        A.append(row)
        b.append(rhs)
    A = np.vstack(A)
    b = np.array(b)

    # use HiGHS (fast, reliable) – non‑negativity via bounds
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method="highs")
    if res.success:
        return res.x
    return None


def get_good_direction_to_move_into(seq):
    """
    Find a direction g using the LP from the reference method.
    The returned sequence is a small convex combination (t=0.01) of the
    original sequence and the normalised LP solution.
    """
    n = len(seq)
    sum_seq = np.sum(seq)
    if sum_seq < 1e-12:
        return None
    # normalise to the standard ℓ1‑norm √(2n)
    scale = np.sqrt(2.0 * n) / sum_seq
    f_norm = np.array(seq) * scale
    rhs = np.max(np.convolve(f_norm, f_norm))
    g = solve_convolution_lp(f_norm, rhs)
    if g is None:
        return None
    # re‑normalise g to the same ℓ1‑norm
    g_scale = np.sqrt(2.0 * n) / np.sum(g)
    g_norm = g * g_scale
    t = 0.01
    new_seq = (1.0 - t) * np.array(seq) + t * g_norm
    return new_seq.tolist()

# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed=0, budget_s=1000, **kwargs):
    """
    Search for a height sequence with a low evaluation score.

    The search strategy:
    * try a few analytic families (power‑law / beta shapes) with random
      exponents,
    * refine the best candidates with projected sub‑gradient descent,
    * finish with a few LP‑based refinement steps (the method from the
      reference implementation).
    The function stops slightly before the allocated time and returns the
    best sequence found.
    """
    np.random.seed(seed)
    random.seed(seed)

    deadline = time.time() + budget_s - 5.0  # keep a small safety margin
    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------

    lengths = [50, 80, 120, 200]          # a handful of moderate sizes

            a = random_beta_shape(n, alpha, beta)
            score = evaluate_sequence(a.tolist())
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"[init] n={n} α={alpha:.2f} β={beta:.2f} → score={score:.6f}")

        # try many random exponent pairs
        for _ in range(250):
            if time.time() >= deadline:
                break
            alpha = np.random.uniform(-2.0, 2.0)
            beta = np.random.uniform(-2.0, 2.0)
            a = random_beta_shape(n, alpha, beta)
            score = evaluate_sequence(a.tolist())
            if score < best_score:
                best_score = score

        # ------------------------------------------------------------------

        # ------------------------------------------------------------------

            a = project_to_simplex(a)

                a = subgradient_step(a, step)

                it += 1
                # restart if no improvement for a while
                if it - last_improve > 2000:
                    a = np.random.rand(n)
                    a = project_to_simplex(a)
                    last_improve = it
                    print(f"[restart] n={n} start={start_id} it={it}")

    # ------------------------------------------------------------------

            if new_score < best_score:
                best_score = new_score
                best_seq = np.array(refined)
                a = best_seq.copy()
                print(f"[LP] improvement → score={new_score:.6f}")
            else:
                a = np.array(refined)

    if best_seq is None:
        # fallback: uniform distribution of length 1 (score = 2)
        best_seq = np.array([1.0])
    return [float(x) for x in best_seq.tolist()]
