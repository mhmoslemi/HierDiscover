        # randomly pick n size from a set
        n = np.random.choice([50,100,200,500,1000,2000])   # choose random
        # generate random initial sequence: Dirichlet
        a = np.random.rand(n)
        a = a/ a.sum()    # Normalize sum to 1 (scale doesn't matter)
        # after scaling, we may multiply by arbitrary constant? not needed.
        # Evaluate initial
        cur_seq = a.copy()
        cur_score = evaluate_sequence(cur_seq.tolist())
        # local search iterations limited by time
        # set number of iterations according to n of steps?
        # For moderate n ~500, each evaluate ~O(500^2)=250k operations ~ small.
        # Lets run e.g. 2000 iterations per restart, adjusting steps.
        # We'll adapt derived step size decreasing.
        # Use simulated annealing:
        steps = 0
        T0 = 0.5
        while time.time() < deadline and steps < 2000:
            steps += 1
            T = T0 * (1 - steps/2000) # linear cooling
            # propose neighbor: move small mass epsilon from idx i to j
            i = np.random.randint(n)
            j = np.random.randint(n)
            if i == j: continue
            # epsilon is proportion of current a_i
            eps = np.random.exponential(scale=0.01) * a[i] # small fraction
            eps = min(eps, a[i])  # ensure nonneg
            new_seq = cur_seq.copy()
            new_seq[i] -= eps
            new_seq[j] += eps
            # evaluate
            new_score = evaluate_sequence(new_seq.tolist())
            # if better or random acceptance
            if new_score < cur_score or np.random.rand() < np.exp(-(new_score - cur_score)/max(T,1e-8)):
                cur_seq = new_seq
                cur_score = new_score
                # update best
                if new_score < best_score:
                    best_score = new_score
                    best_seq = cur_seq.copy()
                    print(f"New best score: {best_score:.6f} n={n} at time {time.time():.2f}")
        # after local search, maybe also apply a small smoothing / mixing patch:
        # optionally, try to smooth sequence: replace each element with average of neighbors.
        # We'll try a few smoothing passes.
        for _ in range(3):
            smoothed = np.convolve(cur_seq, [0.25,0.5,0.25], mode='same')
            # keep sum same: adjust scaling
            smoothed = smoothed * (cur_seq.sum() / smoothed.sum())
            cur_seq = smoothed
            cur_score = evaluate_sequence(cur_seq.tolist())
            if cur_score < best_score:
                best_score = cur_score
                best_seq = cur_seq.copy()

    best_score = np.inf
    best_seq = None

    # function to evaluate
    def eval_obj(x):
        # ensure non-negative
        x = np.maximum(x, 0.0)
        # avoid all zero
        if np.sum(x) < 1e-8:
            # give large penalty
            return np.inf
        return evaluate_sequence(x.tolist())

    for n in [2,3,4,5,6,7,8,9,10,12,15,20]:

            break

            result = differential_evolution(eval_obj, bounds, seed=seed,
                                            maxiter=200, popsize=5,
                                            polish=False, updating='deferred',
                                            disp=False)
            seq = result.x
            score = result.fun
            print(f"DE n={n} score={score}")
            if score < best_score:
                best_score = score
                best_seq = seq.tolist()
        except Exception as e:
            print(f"DE error n={n}: {e}")

        # optional local refinement using L-BFGS-B
        if time.time() < deadline:
            try:
                lbseq = np.maximum(seq, 0.0)

import time

def propose_candidate(seed=0, budget_s=1000, **kwargs):
    """
    Simple heuristic that searches small‑dimensional sequences.
    It always includes the analytically optimal length‑2 construction
    a = (2/3, 1/3) which yields the evaluation value 16/9 ≈ 1.777777.
    The routine also performs a few random hill‑climbing trials on
    sequences of size 2‑6 and keeps the best result found before the
    time budget expires.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5

    best_seq = None
    best_score = float('inf')

    # Known optimal 2‑element solution
    seq2 = [2.0/3.0, 1.0/3.0]
    score2 = evaluate_sequence(seq2)
    best_seq = seq2
    best_score = score2
    # print progress
    print(f"Initial length‑2 solution score={best_score:.12f}")

    # Random small‑n search (optional, may improve further)
    while time.time() < deadline:
        # pick a small dimension
        n = np.random.randint(2, 7)       # 2 <= n <= 6
        # random Dirichlet point
        x = np.random.rand(n)
        x = x / x.sum()
        # small random perturbation to escape uniform
        x += np.random.normal(scale=0.02, size=n)
        x = np.clip(x, 0, None)
        if x.sum() == 0:
            continue
        x = x / x.sum()
        cur_score = evaluate_sequence(x.tolist())
        if cur_score < best_score:
            best_score = cur_score
            best_seq = x.tolist()
            print(f"Random init n={n} improved to {best_score:.12f}")

        # a few hill‑climbing steps
        for _ in range(10):
            i, j = np.random.choice(n, 2, replace=False)
            eps = np.random.exponential(scale=0.01) * x[i]
            eps = min(eps, x[i])
            y = x.copy()
            y[i] -= eps

            y = np.clip(y, 0, None)

            if sc < cur_score:
                x = y
                cur_score = sc
                if sc < best_score:
                    best_score = sc
                    best_seq = y.tolist()
                    print(f"Hill‑climb n={n} improved to {best_score:.12f}")

"""
Simple heuristic to beat the trivial bound (eval = 2).

The analytically optimal 2‑element construction (a0=2/3, a1=1/3) yields
    eval = 2·2·max(conv)/(sum**2) = 16/9 ≈ 1.777777…
which is already strictly better than the current best value 2.0.
The routine returns this construction and, while time permits,
performs a few cheap random searches on very short sequences (length
2‑6) with a tiny hill‑climbing step.  If any of those trials improve
the score they replace the current best candidate.

The search is deliberately lightweight so it always finishes well inside
the 1000 s budget.
"""

import time
import numpy as np

def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Return a non‑negative coefficient sequence that minimizes
    ``evaluate_sequence``.

    The algorithm:
    1. Start with the known optimal 2‑element sequence [2/3, 1/3].
    2. While time remains, sample short random vectors (length 2‑6),
       normalise them to unit L1‑norm and evaluate them.
    3. For each sampled vector perform a handful of tiny “mass‑transfer”
       hill‑climbing steps (move a small amount from one index to another)
       and keep any improvement.
    4. Keep the best sequence seen and return it.

    The result is guaranteed to have an evaluation ≤ 16/9 ≈ 1.777777,
    i.e. an improvement over the previous upper bound of 2.0.
    """
    # make the run reproducible
    np.random.seed(seed)

    # deadline a little before the real limit to avoid accidental overruns
    deadline = time.time() + budget_s - 0.5

    # ------------------------------------------------------------------
    # 1. start from the analytically optimal 2‑element construction
    # ------------------------------------------------------------------
    best_seq = [2.0 / 3.0, 1.0 / 3.0]          # sum = 1, max(conv) = 4/9
    best_score = evaluate_sequence(best_seq)
    print(f"Initial 2‑element candidate score = {best_score:.12f}")

    # ------------------------------------------------------------------
    # 2. cheap random search on very short vectors (2‑6 entries)
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # pick a small dimension – the problem is cheap enough to try many
        n = np.random.randint(2, 7)           # length 2 … 6

        # draw a random point on the simplex and add a tiny perturbation
        x = np.random.rand(n)
        x = x / x.sum()
        x += np.random.normal(scale=0.02, size=n)   # break symmetry a bit
        x = np.clip(x, 0, None)
        if x.sum() == 0.0:
            continue
        x = x / x.sum()                       # renormalise to L1 = 1

        cur_score = evaluate_sequence(x.tolist())
        if cur_score < best_score:
            best_score = cur_score
            best_seq = x.tolist()
            print(f"Random init (n={n}) improved score to {best_score:.12f}")

        # ------------------------------------------------------------------
        # 3. a few tiny hill‑climbing refinements on the current vector
        # ------------------------------------------------------------------
        for _ in range(10):
            i, j = np.random.choice(n, 2, replace=False)
            # move a small exponential amount from i to j
            eps = np.random.exponential(scale=0.01) * x[i]
            eps = min(eps, x[i])
            y = x.copy()
            y[i] -= eps
            y[j] += eps
            y = np.clip(y, 0, None)
            if y.sum() == 0.0:
                continue
            y = y / y.sum()
            sc = evaluate_sequence(y.tolist())
            if sc < cur_score:
                x = y
                cur_score = sc
                if sc < best_score:
                    best_score = sc
                    best_seq = y.tolist()
                    print(f"Hill‑climb (n={n}) improved score to {best_score:.12f}")

    # ------------------------------------------------------------------
    # Return the best sequence we have found.
    # ------------------------------------------------------------------
    return best_seq
