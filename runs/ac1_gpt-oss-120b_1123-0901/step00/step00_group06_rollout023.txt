def project_onto_simplex(v, s):
    # v: np.array
    # returns a projected vector onto {x >=0, sum x = s}

Standard algorithm: sort v descending, find rho s.t. v_i - ( (sum_{j<=i} v_j - s) / i ) > 0.

Implementation steps:

- cumsum = np.cumsum(u)
- find rho = max i (1-indexed) such that u[i] - (cumsum[i] - s) / (i+1) > 0.
- theta = (cumsum[rho] - s) / (rho+1)
- w = np.maximum(v - theta, 0)

Return w.

def project_simplex(v, s=1.0):
    """Projects vector v onto the simplex {x >=0, sum x = s}."""
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n+1)
    # find rho = max i such that u_i - cssv_i / i > 0
    # compute condition
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0]
    if len(rho) == 0:
        theta = 0
    else:
        rho = rho[-1]  # last true index
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a sequence of non‑negative heights whose self‑convolution infinity norm
is reduced below the uniform (value 2) bound.

Algorithm
---------
* Work with a probability vector `p` (sum = 1).  After scaling by
  `sqrt(2·n)` the evaluator computes the exact score, but the ratio is
  invariant under scaling, so we minimise `max(conv(p,p))`.
* Initialise `p` with a small random perturbation of the uniform distribution.
* Repeat while time permits:
    - Compute the convolution via FFT (O(n log n)).
    - Find the index/indices where the convolution attains its maximum.
    - The gradient of a single active constraint `d` is
          g_i = 2·p_{d‑i} (if the index is inside the support).
      Average the gradients of all active constraints.
    - Take a projected sub‑gradient step:
          p ← proj_{simplex}(p – α·g)   (α decreases like 1/√t)
    - Keep the best vector seen so far.
    - Every few thousand iterations add a tiny random jitter to avoid dead‑ends.
* Return the best vector scaled by `sqrt(2·n)`.

The method pushes mass from the centre of the sequence towards the ends,
flattening the distribution of sums of two independent draws and thereby
lowering the maximum convolution entry.  It runs entirely with NumPy, needs
no external files and respects the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex {x ≥ 0, sum x = s}
def _project_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Project vector v onto the simplex of non‑negative vectors with sum s."""
    if s <= 0:
        raise ValueError("Simplex radius must be positive")
    n = v.shape[0]

    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]          # last true index
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Helper: maximum of the self‑convolution of a probability vector (sum=1)
def _max_convolution(p: np.ndarray, fft_len: int) -> float:
    """Return max_{k} (p * p)[k] using FFT of length fft_len."""
    # real FFT is sufficient because p is real
    f = np.fft.rfft(p, fft_len)
    conv = np.fft.irfft(f * f, fft_len)[: 2 * p.size - 1]
    return float(conv.max())

# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of heights) with a low evaluation score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds.
    kwargs : dict
        Ignored; present for compatibility with the competition harness.

    Returns
    -------
    list[float]
        The best sequence found before the time limit.
    """
    np.random.seed(seed)

    # Use the length of the previously found construction if available,
    # otherwise fall back to a modest size.
    try:
        n = len(height_sequence_1)          # type: ignore[name-defined]
    except Exception:
        n = 2000
    sqrt_2n = np.sqrt(2.0 * n)

    # FFT length – next power of two not smaller than 2n‑1 (fast convolution)
    fft_len = 1 << ((2 * n - 1).bit_length())

    # --------------------------------------------------------------
    # Initialise a probability vector (sum = 1) with a tiny random noise
    p = np.full(n, 1.0 / n)
    p += 1e-4 * np.random.randn(n)          # break symmetry
    p = np.maximum(p, 0.0)
    p /= p.sum()

    # Best solution seen so far
    best_p = p.copy()
    best_score = evaluate_sequence((sqrt_2n * best_p).tolist())

    # Pre‑compute an array of indices for vectorised gradient computation
    idx = np.arange(n)

    # Simple decreasing schedule for the step size
    step0 = 0.5

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    iteration = 0
    while time.time() < deadline:
        # ---- convolution and active set ---------------------------------
        f = np.fft.rfft(p, fft_len)
        conv = np.fft.irfft(f * f, fft_len)[: 2 * n - 1]
        max_val = float(conv.max())
        # treat values within 1e-12 relative tolerance as active
        tol = 1e-12 * max_val if max_val > 0 else 1e-12
        active = np.where(conv >= max_val - tol)[0]

        # ---- sub‑gradient of the active max‑constraint -----------------
        g = np.zeros_like(p)
        for d in active:
            j = d - idx
            mask = (j >= 0) & (j < n)
            # contribution 2 * p[j] for every valid pair
            g[mask] += 2.0 * p[j[mask]]
        g /= float(len(active))               # average over the active set

        # ---- projected sub‑gradient step --------------------------------
        step = step0 / np.sqrt(iteration + 1.0)
        p_new = p - step * g
        p_new = _project_simplex(p_new, s=1.0)

        # Evaluate the new point (fast via FFT)
        f_new = np.fft.rfft(p_new, fft_len)
        conv_new = np.fft.irfft(f_new * f_new, fft_len)[: 2 * n - 1]
        max_new = float(conv_new.max())

        # Accept if we made progress; otherwise keep the old point.
        if max_new <= max_val:
            p = p_new
            max_val = max_new

            # Compute the true score only when an improvement occurs.
            cur_score = float(2.0 * n * max_val)   # equivalent to evaluate_sequence
            if cur_score < best_score:
                best_score = cur_score
                best_p = p.copy()
                print(f"[{iteration:7d}] improved score = {best_score:.6f}")

        # ---- occasional jitter to avoid stagnation --------------------
        if (iteration & 1023) == 0 and iteration > 0:
            # add a very small random perturbation and re‑project
            jitter = 5e-5 * np.random.rand(n)
            p = _project_simplex(p + jitter, s=1.0)

        iteration += 1

    # ------------------------------------------------------------------
    # Scale the best probability vector to the required L1 norm
    best_sequence = (sqrt_2n * best_p).tolist()
    # Final sanity check (optional, does not affect the returned value)
    try:
        final_score = evaluate_sequence(best_sequence)
        print(f"Final returned score = {final_score:.6f}")
    except Exception:
        pass

    return best_sequence
