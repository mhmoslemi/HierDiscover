def max_convolution(seq):
    a = np.array(seq, dtype=np.float64)
    n = len(a)
    size = 2*n-1
    # find next power of two for efficient FFT
    N = 1 << (size-1).bit_length()
    fa = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(fa * fa, n=N)[:size]  # real result
    return np.max(conv)

def compute_score(seq):
    a = np.asarray(seq, dtype=np.float64)
    n = a.shape[0]
    size = 2*n -1
    N = 1 << (size-1).bit_length()
    fa = np.fft.rfft(a, N)  # real FFT length N

"""
Search for a low‑evaluation step function.

The idea is simple: pick roughly half of the positions, give each of them the
same weight (so the whole sequence sums to 1) and keep the best random
construction.  For a random half‑dense set the self‑convolution is close to its
average value 1/(2 n) and the evaluation 2 n·max (conv) is typically close to 1.
A cheap hill‑climbing phase swaps a selected and an unselected index to push
the maximal convolution even lower.
"""

import time
import numpy as np

def _eval_score(seq):
    """
    Fast implementation of evaluate_sequence (no safety checks).
    Returns 2·n·max(conv(seq,seq)) / (sum(seq)^2).
    """
    a = np.asarray(seq, dtype=np.float64)
    n = a.shape[0]
    size = 2 * n - 1
    # size for a zero‑padded FFT (next power of two)
    N = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, N)
    conv = np.fft.irfft(fa * fa, N)[:size]
    max_b = np.max(conv)
    sum_a = np.sum(a)
