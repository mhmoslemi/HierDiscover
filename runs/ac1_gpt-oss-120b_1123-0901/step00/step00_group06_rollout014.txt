For a fixed length n the maximal autocorrelation of a Gaussian is
approximately n / (σ·√π); it therefore decreases when σ grows.
The algorithm scans a modest set of σ‑values, keeps the best Gaussian,
and then performs a tiny stochastic hill‑climbing phase to polish
the result.  All operations respect the non‑negative and bound
constraints enforced inside ``evaluate_sequence``.
"""

import time
import numpy as np

def _make_gaussian(n: int, sigma: float) -> list[float]:
    """Return a normalised Gaussian of length n and width sigma."""
    # coordinate vector
    i = np.arange(n, dtype=np.float64)
    mu = (n - 1) / 2.0
    # raw (unscaled) Gaussian
    raw = np.exp(- ((i - mu) ** 2) / (2.0 * sigma * sigma))
    # normalise so that sum = sqrt(2·n) (the scale used by the benchmark)
    total = np.sum(raw)
    if total == 0.0:                     # safety – should never happen
        return [0.0] * n
    scale = np.sqrt(2.0 * n) / total
    seq = (raw * scale).tolist()
    return seq


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Build a Gaussian‑shaped sequence and improve it with a tiny random walk.

    The function runs for at most ``budget_s`` seconds (leaving a 5‑second margin)
    and returns the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0

    # ------------------------------------------------------------
    # 1. Determine the working length.
    # ------------------------------------------------------------
    # ``height_sequence_1`` is provided by the evaluation harness; fall back to 6850.
    try:
        n = len(height_sequence_1)      # type: ignore[name-defined]
    except Exception:
        n = 6850

    # ------------------------------------------------------------
    # 2. Scan a few Gaussian widths.
    # ------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    # width range: from a fairly narrow bump to a very wide one.
    sigma_min = n / 20.0                # ~5 % of the length
    sigma_max = n                       # up to a flat distribution
    sigma_vals = np.linspace(sigma_min, sigma_max, 30)

    for sigma in sigma_vals:
        seq = _make_gaussian(n, sigma)
        # quick evaluation (the expensive part is the convolution inside evaluate_sequence)
        score = evaluate_sequence(seq)
        if score < best_score:
            best_score = score
            best_seq = seq
        # stop early if we run out of time
        if time.time() >= deadline:
            break

    # ------------------------------------------------------------
    # 3. Refine the best Gaussian with a tiny stochastic hill‑climb.
    #    The sum is re‑normalised after every perturbation to keep the
    #    evaluation invariant under scaling.
    # ------------------------------------------------------------
    if best_seq is None:                     # safety fallback
        best_seq = _make_gaussian(n, sigma_max)
        best_score = evaluate_sequence(best_seq)

    # pre‑compute the target L1‑norm for normalisation
    target_norm = np.sqrt(2.0 * n)

    while time.time() < deadline:
        # pick a random index and propose a small relative change
        i = np.random.randint(n)
        delta = (np.random.rand() - 0.5) * 0.02 * best_seq[i]   # ≤ ± 1 % change
        new_val = max(0.0, best_seq[i] + delta)                # stay non‑negative
        if new_val == best_seq[i]:
            continue

        cand = best_seq.copy()
        cand[i] = new_val

        # re‑normalise to the required L1‑norm (the clipping inside
        # evaluate_sequence will also enforce the 0 – 1000 bounds)
        cur_sum = sum(cand)
        if cur_sum == 0.0:
            continue
        scale = target_norm / cur_sum
        cand = (np.array(cand) * scale).tolist()

        cand_score = evaluate_sequence(cand)

        # accept only improvements (simple greedy hill‑climb)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand
            # occasional debug output
            print(f"Improved score: {best_score:.6f}  (sigma≈{sigma:.1f})")

    # final report
    print(f"Final best score: {best_score:.6f}  (length={n})")
    return best_seq
