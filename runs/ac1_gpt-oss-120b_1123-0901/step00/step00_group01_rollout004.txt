def simplex_projection(v):
    # v is array
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a new step‑function sequence.

The algorithm is a continuous optimisation on the probability simplex.
We minimise  max(conv(a,a))  under the constraint  sum(a)=1 , a_i≥0.
The objective is invariant under scaling, therefore we keep the
ℓ₁–norm fixed to 1.

We use a smooth approximation of the max (log‑sum‑exp) with a
temperature parameter β.  The gradient of this smooth max is a
weighted sum of the gradients of the individual convolution entries.
For a fixed β the gradient can be computed efficiently via a
cross‑correlation.  After a gradient step the vector is projected back
onto the simplex (standard Euclidean projection).  β is increased
gradually, the step size is decreased, and occasional random
perturbations are performed to escape plateaux.

The whole procedure runs until the allotted time expires and returns
the best sequence seen so far (as a plain Python list).  The final
score is printed using the provided `evaluate_sequence` function.
"""

import time
import math
import numpy as np

# ------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex.
# ------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Return the Euclidean projection of v onto {x≥0, sum x = 1}."""
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # find the largest rho satisfying the condition
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / float(rho)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Core optimisation step (gradient of the smooth max)
# ------------------------------------------------------------
def _smooth_max_gradient(a: np.ndarray, beta: float):
    """
    Compute convolution a*a, the smooth‑max value and its gradient.

    Returns:
        b (np.ndarray)      : full convolution (length 2*n-1)
        max_b (float)       : max(b)
        grad (np.ndarray)   : gradient of the smooth‑max w.r.t. a
    """
    n = a.shape[0]
    # full self‑convolution
    b = np.convolve(a, a)
    max_b = np.max(b)

    # stable log‑sum‑exp weights
    shift = b - max_b               # ≤ 0, max entry becomes 0
    exp_shift = np.exp(beta * shift)
    w = exp_shift / exp_shift.sum()  # probabilities over the 2n‑1 slots

    # gradient = 2 * (w ⋆ a)  (cross‑correlation, mode='valid' gives length n)
    # np.correlate implements exactly (w ⋆ a)[i] = Σ_j w[i+j] a[j]
    grad = 2.0 * np.correlate(w, a, mode='valid')
    return b, max_b, grad


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low ``evaluate_sequence`` score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Time budget in seconds (the function returns before this deadline).

    Returns
    -------
    list[float]
        The best sequence found (list of plain Python floats).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0   # a small safety margin

    # ------------------------------------------------------------------
    # Initialise the problem (length, starting point)
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        # Use the supplied SOTA sequence as a warm start
        init = np.array(height_sequence_1, dtype=np.float64)
        n = init.shape[0]
        init /= init.sum()                     # normalise to sum = 1
    else:
        # Fallback: random length roughly comparable to the SOTA one
        n = np.random.randint(5000, 7000)
        init = np.random.rand(n).astype(np.float64)
        init /= init.sum()

    a_curr = init.copy()
    best_a = a_curr.copy()
    # simple “raw” score (identical to evaluate_sequence up to a scale)
    b_curr = np.convolve(a_curr, a_curr)
    best_score = 2.0 * n * np.max(b_curr)      # sum(a)=1 → denominator = 1

    # hyper‑parameters
    step_initial = 0.5
    beta = 1.0
    max_beta = 30.0
    beta_growth = 1.005          # multiply every few iterations
    no_improve_counter = 0
    iteration = 0
    last_print = time.time()

    while time.time() < deadline:
        iteration += 1

        # --------------------------------------------------------------
        # Gradient descent step on the smooth max
        # --------------------------------------------------------------
        b, max_b, grad = _smooth_max_gradient(a_curr, beta)

        # Simple diminishing step size
        step = step_initial / math.sqrt(iteration)

        a_new = a_curr - step * grad
        a_new = _project_onto_simplex(a_new)

        # --------------------------------------------------------------
        # Evaluate the (non‑smooth) objective for monitoring / acceptance
        # --------------------------------------------------------------
        b_new = np.convolve(a_new, a_new)
        score_new = 2.0 * n * np.max(b_new)        # sum(a_new) == 1

        # Keep the best ever seen (according to the true max‑convolution)
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            no_improve_counter = 0
            # occasional progress report
            if time.time() - last_print > 5.0:
                print(f"[{iteration:6d}] New best score ≈ {best_score:.6f}")
                last_print = time.time()
        else:
            no_improve_counter += 1

        # Accept the new point only if it improves the true objective;
        # otherwise stay where we are (this is a simple greedy scheme).
        if score_new <= (2.0 * n * np.max(b_curr)):
            a_curr = a_new
            b_curr = b_new
        else:
            # Occasionally move even without improvement to avoid stagnation
            if no_improve_counter % 2000 == 0 and no_improve_counter > 0:
                noise = (np.random.rand(n) - 0.5) * 0.02
                a_curr = _project_onto_simplex(best_a + noise)

        # --------------------------------------------------------------
        # Schedule changes for β (sharpen the approximation)
        # --------------------------------------------------------------
        if iteration % 100 == 0:
            beta = min(beta * beta_growth, max_beta)

        # --------------------------------------------------------------
        # Random restart if stuck for a long time
        # --------------------------------------------------------------
        if no_improve_counter > 15000:
            # Re‑initialise around the current best with a larger perturbation
            perturb = (np.random.rand(n) - 0.5) * 0.1
            a_curr = _project_onto_simplex(best_a + perturb)
            no_improve_counter = 0
            beta = 1.0                # reset β to a smoother regime

    # ------------------------------------------------------------------
    # Final evaluation with the official function (for the log)
    # ------------------------------------------------------------------
    final_list = best_a.tolist()
    try:
        final_val = evaluate_sequence(final_list)
        print(f"Final sequence length {n}, evaluate_sequence = {final_val:.6f}")
    except Exception as exc:          # safety – the environment may not expose it
        print("Could not evaluate final sequence:", exc)

    return final_list
