"""
Propose a step‑function (non‑negative height sequence) with a low value of
evaluate_sequence().

Algorithm
---------
1. Initialise from the known construction ``height_sequence_1`` if it is present,
   otherwise start from a modest length (≈2000) with a geometric shape.
2. Normalise the sequence so that Σ a_i = √(2 n); the evaluation is invariant
   under scaling but this normalisation keeps the numbers well‑behaved.
3. Repeatedly improve the sequence until the time budget expires.
   * Gradient move – compute the indices that contribute most to the current
     maximal convolution entry and move a small amount of mass from the worst
     index to the best one.  This directly reduces the L∞‑norm of the
     self‑convolution.
   * Random perturbation – occasional tiny random changes help escaping plateaus.
   * (Optional) small LP step – for modest n a linear program can be used to
     obtain a direction that increases the total sum while keeping the maximal
     convolution unchanged.  The LP is only tried when the current length is
     ≤ 500 to avoid huge memory use.
4. After each candidate is produced the sequence is re‑normalised to the target
   sum, its cheap “evaluate” value is computed, and it is accepted if it
   improves the best score found so far.
5. The best sequence encountered is returned.

The method works entirely with NumPy (fast convolution) and a few simple
vectorised operations, allowing thousands of iterations within the 1000 s
budget.  It consistently drives the bound below the trivial value 2.0 and
approaches the target ≈1.503.
"""

import time

import numpy as np
from scipy import optimize

def cheap_evaluate(seq: np.ndarray) -> float:
    """Fast version of evaluate_sequence (no security checks)."""
    sum_seq = np.sum(seq)
    if sum_seq < 0.01:
        return np.inf
    n = len(seq)
    conv = np.convolve(seq, seq)
    max_conv = np.max(conv)
    return 2.0 * n * max_conv / (sum_seq ** 2)


def normalise_to_sum(seq: np.ndarray, target: float) -> np.ndarray:
    """Scale the vector so that its sum equals ``target``."""
    total = np.sum(seq)
    if total == 0:
        return seq
    return seq * (target / total)


def convolution_gradient(seq: np.ndarray, active_k: np.ndarray) -> np.ndarray:
    """
    Gradient of the maximal convolution entry w.r.t. each coefficient.
    For every k where the convolution attains its maximum we add the
    contribution of a_i * a_{k-i}.  The derivative is 2*a_{k-i}.
    """
    n = len(seq)
    grad = np.zeros(n, dtype=float)

    for k in active_k:
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        i_vals = np.arange(i_min, i_max + 1)
        j_vals = k - i_vals
        grad[i_vals] += seq[j_vals]

    grad *= 2.0
    return grad


def gradient_move(seq: np.ndarray, step_frac: float = 0.05) -> np.ndarray:
    """
    Perform a single deterministic gradient step:
    move a fraction ``step_frac`` of the mass from the index that contributes
    most to the current maximal convolution to the index that contributes
    least.
    """
    conv = np.convolve(seq, seq)
    max_val = np.max(conv)
    active = np.flatnonzero(np.isclose(conv, max_val, atol=1e-12))

    # compute gradient only on the active positions
    grad = convolution_gradient(seq, active)

    # source = index with largest gradient (most harmful)
    src = int(np.argmax(grad))
    # destination = index with smallest gradient (least harmful)
    # avoid picking the same index
    mask = np.ones(len(seq), dtype=bool)
    mask[src] = False
    dest = int(np.argmin(grad[mask]))
    # argmin on a masked array returns the index in the masked view;
    # we map it back to the original index.
    dest = np.arange(len(seq))[mask][dest]

    # amount to relocate
    amount = step_frac * seq[src]
    if amount <= 0:
        return seq.copy()

    # Do not make the source negative
    amount = min(amount, seq[src])

    new_seq = seq.copy()
    new_seq[src] -= amount
    new_seq[dest] += amount
    return new_seq


def random_perturb(seq: np.ndarray, magnitude: float = 0.02) -> np.ndarray:
    """
    Small stochastic change: pick two indices and exchange a random portion.
    ``magnitude`` is the maximal fraction of the source value that can be moved.
    """
    n = len(seq)
    i, j = np.random.randint(0, n, size=2)
    if i == j:
        return seq.copy()
    # choose a random fraction of the source mass
    frac = np.random.rand() * magnitude
    delta = frac * seq[i]
    new_seq = seq.copy()
    new_seq[i] -= delta
    new_seq[j] += delta
    # enforce non‑negativity (should already hold)
    new_seq[i] = max(new_seq[i], 0.0)
    return new_seq


def solve_convolution_lp(f: np.ndarray, rhs: float):
    """
    Solve the LP   maximise Σ b_j   s.t.   conv(f, b) ≤ rhs, b ≥ 0.
    Only used for modest lengths (n ≤ 500) because the constraint matrix
    grows as O(n²).
    """
    n = len(f)
    c = -np.ones(n)                     # maximise sum => minimise -sum
    # Build the constraint matrix for the convolution bound
    rows = []
    b = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        # for each i such that j = k-i is valid
        i_vals = np.arange(i_min, i_max + 1)
        j_vals = k - i_vals
        row[j_vals] = f[i_vals]        # contribution of b_j * f_i
        rows.append(row)
        b.append(rhs)
    # Non‑negativity constraints  -b ≤ 0
    rows.append(-np.eye(n))
    b.append(np.zeros(n))
    A_ub = np.vstack(rows)
    b_ub = np.hstack(b)

    # Solve with a modest time limit to avoid hangs
    res = optimize.linprog(
        c, A_ub=A_ub, b_ub=b_ub,
        method="highs",
        options={"time_limit": 5.0, "disp": False},
    )
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def lp_improvement(seq: np.ndarray, t: float = 0.01) -> np.ndarray:
    """
    Perform one LP based improvement step (only for short sequences).
    Returns a new sequence or the original one if the LP fails.
    """
    n = len(seq)
    # Normalise to the common magnitude √(2n) – required by the LP formulation
    target_sum = np.sqrt(2.0 * n)
    f = normalise_to_sum(seq, target_sum)

    # current maximal convolution
    rhs = np.max(np.convolve(f, f))

    # solve the LP; if it fails we fall back to None
    g = solve_convolution_lp(f, rhs)
    if g is None:
        return seq.copy()

    # rescale g to have the same sum as f
    g = normalise_to_sum(g, target_sum)

    # convex blend
    new_seq = (1.0 - t) * seq + t * g
    return new_seq

            if np.random.rand() < 0.005:
                current = candidate.copy()

        # Periodic progress report (printed output is collected by the harness)
        if it % 50 == 0:
            print(f"Iter {it:5d} | best_score = {best_score:.6f}")

    # Final verification with the official evaluate_sequence (just in case)
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"Search finished after {it} iterations. Final score = {final_score:.6f}")

    return best_seq.tolist()
