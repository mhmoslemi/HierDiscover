def hill_climb(seq, deadline):
    best_seq = seq[:]
    best_score = evaluate_sequence(seq)
    n = len(seq)
    temperature = 0.1
    while time.time() < deadline:
        # maybe update temperature based on iteration count.
        i, j = np.random.randint(n), np.random.randint(n)
        if i == j: continue
        # propose moving delta from i to j
        delta = (np.random.rand() - 0.5) * 0.1 * best_seq[i]  # up to 5% of value
        new_seq = best_seq[:]
        new_seq[i] = max(0.0, new_seq[i] + delta)
        new_seq[j] = max(0.0, new_seq[j] - delta)
        # renormalize to avoid sum too small? scaling invariance means not needed.
        new_score = evaluate_sequence(new_seq)
        if new_score < best_score or np.random.rand() < np.exp(-(new_score - best_score)/temperature):
            best_seq, best_score = new_seq, new_score
        # cooling
        temperature *= 0.9999
        if temperature < 1e-6:
            temperature = 0.1
    return best_seq, best_score

"""
Search for a non‑negative step function that minimises
evaluate_sequence().

The routine samples a few candidate families (geometric,
sinusoidal, power‑law and random Dirichlet vectors) for many
different lengths, then refines each candidate with a very
light simulated‑annealing hill‑climber.  The best sequence found
before the time budget expires is returned.

The approach is deliberately different from the LP‑based update
used in the reference implementation – it relies on stochastic
local optimisation rather than linear programming.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper generators for simple families of sequences
# ----------------------------------------------------------------------
def generate_geometric(n: int, r: float) -> list[float]:
    """Geometric decay a_i = r**i, normalised to sum = 1."""
    a = np.power(r, np.arange(n, dtype=float))
    s = a.sum()
    if s == 0.0:                # very unlikely, but keep safe
        a[0] = 1.0
        s = 1.0
    return (a / s).tolist()


def generate_sine(n: int) -> list[float]:
    """a_i = sin(pi * (i+1)/(n+1)), normalised."""
    i = np.arange(1, n + 1, dtype=float)
    a = np.sin(np.pi * i / (n + 1))
    s = a.sum()
    return (a / s).tolist()


def generate_powerlaw(n: int, p: float) -> list[float]:
    """a_i = 1/(i**p), normalised."""
    i = np.arange(1, n + 1, dtype=float)
    a = 1.0 / np.power(i, p)
    s = a.sum()
    return (a / s).tolist()


# ----------------------------------------------------------------------
# Simple simulated‑annealing hill‑climber
# ----------------------------------------------------------------------
def hill_climb(start_seq: list[float],
               deadline: float,
               init_temp: float = 1.0,
               cooling: float = 0.9995) -> tuple[list[float], float]:
    """
    Starting from ``start_seq`` try to lower the evaluation score.
    Mutations move a small amount of mass between two random positions.
    Acceptance uses a temperature schedule for occasional uphill moves.
    Returns the best sequence encountered and its score.
    """
    cur_seq = list(start_seq)
    try:
        cur_score = evaluate_sequence(cur_seq)
    except Exception:                # safety – should not happen
        cur_score = float("inf")
    best_seq, best_score = cur_seq[:], cur_score

    n = len(cur_seq)
    temp = init_temp

    while time.time() < deadline:
        # pick two distinct indices
        i, j = np.random.randint(n), np.random.randint(n)
        if i == j:
            continue

        # propose a transfer of a fraction of the larger entry
        max_val = max(cur_seq[i], cur_seq[j], 1e-12)
        delta = (np.random.rand() - 0.5) * 0.2 * max_val   # ±10 % roughly
        new_seq = cur_seq[:]
        new_seq[i] = max(0.0, new_seq[i] + delta)
        new_seq[j] = max(0.0, new_seq[j] - delta)

        # evaluate candidate
        try:
            new_score = evaluate_sequence(new_seq)
        except Exception:
            continue

        # accept if better, otherwise with Metropolis probability
        if new_score < cur_score or np.random.rand() < np.exp(-(new_score - cur_score) / max(temp, 1e-12)):
            cur_seq, cur_score = new_seq, new_score
            if new_score < best_score:
                best_seq, best_score = new_seq[:], new_score

        # cool down
        temp *= cooling
        if temp < 1e-6:
            temp = init_temp

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point required by the framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a height sequence that minimises evaluate_sequence().
    The function runs for at most ``budget_s`` seconds and returns the
    best sequence discovered.  It explores several lengths (3…30 and a
    few larger ones), builds a handful of seed candidates for each
    length, and refines them with a lightweight simulated‑annealing
    hill‑climber.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # small safety margin

    # ------------------------------------------------------------------
    # start from the known construction if it exists
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        best_seq = list(height_sequence_1)
        try:
            best_score = evaluate_sequence(best_seq)
        except Exception:
            best_score = float("inf")
    else:
        best_seq = [0.5, 0.5]
        best_score = evaluate_sequence(best_seq)

    print(f"[init] score={best_score:.6f} length={len(best_seq)}")

    # ------------------------------------------------------------------
    # lengths to explore – short enough for fast convolution,
    # long enough to give the optimiser freedom.
    # ------------------------------------------------------------------
    lengths = list(range(3, 31))          # 3 … 30
    lengths += [40, 50, 60, 80, 100]      # a few larger sizes

    # ------------------------------------------------------------------
    # main outer loop over lengths
    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() >= deadline:
            break

        # --------------------------------------------------------------
        # generate a small collection of seed candidates for this n
        # --------------------------------------------------------------
        seeds = []

        # random Dirichlet draws (balanced and un‑balanced)
        for _ in range(4):
            seeds.append(np.random.dirichlet(np.ones(n)).tolist())

        # a handful of geometric families
        for r in np.linspace(0.2, 0.9, 5):
            seeds.append(generate_geometric(n, r))

        # sinusoidal shape
        seeds.append(generate_sine(n))

        # power‑law shapes
        for p in (0.5, 1.0, 1.5):
            seeds.append(generate_powerlaw(n, p))

        # --------------------------------------------------------------
        # evaluate each seed and run a local optimiser from it
        # --------------------------------------------------------------
        for seed_seq in seeds:
            if time.time() >= deadline:
                break

            # evaluate the raw seed (quick way to catch obvious winners)
            try:
                seed_score = evaluate_sequence(seed_seq)
            except Exception:
                continue
            if seed_score < best_score:
                best_score, best_seq = seed_score, seed_seq[:]
                print(f"[seed] new best score={best_score:.6f} len={n}")

            # run hill climbing from this seed
            opt_seq, opt_score = hill_climb(seed_seq, deadline)
            if opt_score < best_score:
                best_score, best_seq = opt_score, opt_seq[:]
                print(f"[local] new best score={best_score:.6f} len={n}")

    # ------------------------------------------------------------------
    # final tidy‑up: strip leading/trailing zeros (they increase n without benefit)
    # ------------------------------------------------------------------
    eps = 1e-12
    lo = 0
    while lo < len(best_seq) and best_seq[lo] < eps:
        lo += 1
    hi = len(best_seq) - 1
    while hi >= 0 and best_seq[hi] < eps:
        hi -= 1
    if lo <= hi:
        best_seq = best_seq[lo:hi + 1]
    else:
        best_seq = [0.0]   # fallback – should never happen

    print(f"[done] best score={best_score:.6f} final length={len(best_seq)}")
    return best_seq
