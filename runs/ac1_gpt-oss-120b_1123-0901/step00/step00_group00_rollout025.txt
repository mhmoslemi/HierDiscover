def compute_eval(a):
    conv = np.fft.ifft(np.fft.fft(a, 2*len(a)) * np.fft.fft(a, 2*len(a))).real[:2*len(a)-1]
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 1e-8:
        return float('inf')
    return float(2 * len(a) * max_b / (sum_a**2))

def propose_candidate(seed=12345, budget_s=1000, **kwargs):
    import numpy as np, time

    np.random.seed(seed)
    deadline = time.time() + budget_s - 5

    best_seq = None
    best_score = float('inf')
    # define lengths to test
    lengths = [256, 512, 1024, 2048, 2304]
    # Simulated annealing parameters
    step_initial = 0.02
    step_min = 1e-4
    for n in lengths:
        if time.time() >= deadline:
            break
        # number of restarts per length
        restarts = 5
        for r in range(restarts):
            if time.time() >= deadline:
                break
            # initial distribution: Dirichlet with low concentration for variety
            a = np.random.dirichlet(np.full(n, 0.5))
            # optionally, apply monotone decreasing deterministic init on first restart
            if r == 0:
                a = np.linspace(1.0, 0.0, n)
                a = np.maximum(a, 0)
                a /= a.sum()
            # inner optimization loop
            step = step_initial
            it = 0
            while time.time() < deadline and it < 3000:
                # compute convolution via FFT
                size = 1
                while size < 2 * n:
                    size <<= 1
                fa = np.fft.rfft(a, n=size)
                conv = np.fft.irfft(fa * fa, n=size)[:2*n-1]
                max_idx = int(np.argmax(conv))
                max_val = conv[max_idx]
                # compute contributions from each index to the max bin
                idx = np.arange(n)
                j = max_idx - idx
                mask = (j >= 0) & (j < n)
                # product contributions
                contrib = np.zeros(n)
                contrib[mask] = a[idx[mask]] * a[j[mask]]
                # find source index with biggest contribution
                src = int(np.argmax(contrib))
                # find destination index with smallest mass (but not same)
                dst = int(np.argmin(a))
                if src == dst:
                    # pick a random different destination
                    dst = (dst + 1) % n
                delta = min(step, a[src])
                if delta <= 1e-12:
                    break
                a[src] -= delta
                a[dst] += delta
                # re‑normalize to avoid drift (sum should still be 1, but float errors)
                a_sum = a.sum()
                if abs(a_sum - 1.0) > 1e-12:
                    a /= a_sum
                # gradual step reduction
                if it % 500 == 0 and step > step_min:
                    step *= 0.7
                it += 1
            # evaluate final candidate using the provided evaluate_sequence for robustness
            seq = a.tolist()
            score = evaluate_sequence(seq)
            if score < best_score:
                best_score = score
                best_seq = seq
                print(f'New best score {best_score:.6f} with length {n}')
    # as fallback if never improved, return a uniform sequence of moderate length
    if best_seq is None:
        n_fallback = 256
        best_seq = [1.0 / n_fallback] * n_fallback
    return best_seq
