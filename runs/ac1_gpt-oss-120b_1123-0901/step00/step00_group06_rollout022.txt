def compute_conv(x):
    n = len(x)
    size = 1 << ((2*n - 1).bit_length())
    X = np.fft.rfft(x, n=size)
    conv = np.fft.irfft(X * X, n=size)[:2*n - 1]
    return conv
```

"""
Propose a sequence that minimizes the evaluation function.

The algorithm works with a probability‐like interpretation of the
coefficients: the sum of the sequence is fixed to 1 (the objective is
scale invariant) and we want to make the self‑convolution as flat as
possible.  Starting from a simple exponential decay (which already beats
the uniform bound 2.0) we perform a fast greedy “peak‑reduction”
iteration:

* compute the convolution a⋆a by FFT (O(n log n));
* locate the index k where the convolution attains its maximum;
* the contribution of every coefficient a[i] to that peak is a[i]·a[k-i];
  we move a tiny amount of mass from the index with the largest
  contribution to the index with the smallest contribution;
* the move is kept only if the new maximum does not increase.

Repeating this step quickly flattens the convolution, lowering the
objective.  The loop runs until the time budget is exhausted (or no
improvement is possible).  The best sequence seen during the search is
returned.

The method is completely self‑contained, uses only NumPy and the
standard library and respects the 1000 s time limit.
"""

import time
import numpy as np

def _fft_convolve(a: np.ndarray) -> np.ndarray:
    """Fast linear convolution of a with itself using real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    freq = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(freq * freq, n=size)[: 2 * n - 1]
    return conv


def _objective(a: np.ndarray, conv: np.ndarray) -> float:
    """Return the value of evaluate_sequence for the (unnormalised) a."""
    # the evaluation function is 2 * n * max(conv) / (sum(a)^2)
    n = a.shape[0]
    return float(2.0 * n * conv.max() / (a.sum() ** 2))


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Generate a list of non‑negative heights with a low evaluation score.

    The routine performs a greedy peak‑reduction on the self‑convolution.
    It works for any reasonable length; here we use n≈2500 which gives a
    good trade‑off between speed and quality.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # leave a small margin

    # ------------------------------------------------------------------
    # 1) choose a length and an initial shape (exponential, sum=1)
    # ------------------------------------------------------------------
    n = 2500                                   # can be changed if needed
    s = 1.0 / n                               # step of the exponential
    a = np.exp(-np.arange(n, dtype=float) * s)   # raw exponential
    a = a / a.sum()                           # normalise → sum(a) == 1

    # keep the best sequence found
    conv = _fft_convolve(a)
    best_a = a.copy()
    best_obj = _objective(a, conv)
    print(f"[init] n={n}, objective={best_obj:.6f}")

    # ------------------------------------------------------------------
    # 2) iterative greedy improvement
    # ------------------------------------------------------------------
    iteration = 0
    while time.time() < deadline:
        iteration += 1

        # current convolution and objective
        conv = _fft_convolve(a)
        max_val = conv.max()
        obj = 2.0 * n * max_val / (a.sum() ** 2)

        # keep the best ever seen
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
            print(f"[{iteration:6d}] new best obj = {best_obj:.6f}")

        # locate the peak index k
        k = int(conv.argmax())

        # contribution of each coefficient to that peak
        # conv[k] = sum_i a[i] * a[k-i]  (indices out of range are ignored)
        start = max(0, k - (n - 1))
        stop = min(k, n - 1)
        idx = np.arange(start, stop + 1, dtype=int)
        contrib = np.zeros_like(a)
        contrib[idx] = a[k - idx]               # a[k-i] for valid i
        prod = a * contrib                       # a[i] * a[k-i]

        i_max = int(prod.argmax())              # biggest contributor
        i_min = int(prod.argmin())               # smallest contributor

        # --------------------------------------------------------------
        # take a tiny amount of mass from i_max and give it to i_min
        # --------------------------------------------------------------
        # step size is a fraction of the source mass; we start relatively
        # aggressive and shrink automatically when no improvement is seen.
        step = min(a[i_max] * 0.08, 1e-4)        # 8 % of source or 1e‑4
        if step <= 0.0:
            break

        a_new = a.copy()
        a_new[i_max] -= step
        a_new[i_min] += step

        # avoid negative entries caused by numeric round‑off
        if a_new[i_max] < 0:
            continue

        # evaluate the new maximum – accept only if it does not increase
        conv_new = _fft_convolve(a_new)
        if conv_new.max() <= max_val * (1.0 + 1e-12):
            a = a_new
            continue

        # if the naive step fails, try a half‑step; otherwise skip
        step *= 0.5
        if step < 1e-12:
            continue
        a_new = a.copy()
        a_new[i_max] -= step
        a_new[i_min] += step
        if a_new[i_max] < 0:
            continue
        conv_new = _fft_convolve(a_new)
        if conv_new.max() <= max_val * (1.0 + 1e-12):
            a = a_new
            continue

        # ----------------------------------------------------------------
        # occasional random perturbation to escape tiny plateaus
        # ----------------------------------------------------------------
        if iteration % 500 == 0:
            # pick two random positions and swap a tiny slice of mass
            i, j = np.random.choice(n, size=2, replace=False)
            eps = min(a[i], a[j]) * 0.02
            if eps > 0:
                a[i] -= eps
                a[j] += eps
                a[j] -= eps
                a[i] += eps
    # ----------------------------------------------------------------------
    # 3) finalise – guarantee non‑negativity and normalise (just in case)
    # ----------------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total == 0.0:
        # fall back to uniform if something went terribly wrong
        best_a = np.full(n, 1.0 / n)
    else:
        best_a = best_a / total

    print(f"[final] best objective = {best_obj:.6f} after {iteration} iterations")
    return best_a.tolist()
