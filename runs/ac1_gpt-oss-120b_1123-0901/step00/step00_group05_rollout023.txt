"""
Propose a candidate step function using a block‐parameterisation and a simple
gradient‑based hill‑climber.

The idea:
* Represent a long sequence (≈ 6000 entries) as K blocks of equal length L.
  The full sequence is obtained by repeating each block height.
* For a block vector `h` (sum(h)=K) the evaluation score satisfies

        score ≈ 2 * max(conv(h, h)) / K

  (the factor L cancels out).  Thus minimising the maximum entry of the
  self‑convolution of the short vector directly reduces the original
  objective.
* Starting from a uniform block vector we repeatedly move a small amount of
  mass from a block that contributes most to the current maximal convolution
  entry towards a block that contributes least.  This follows the gradient of
  the peak of the convolution while keeping the total sum constant.
* The process is run until the time budget expires.  The best block vector
  found is expanded to the full length and returned.

The algorithm stays within the allowed time, uses only NumPy and the provided
`evaluate_sequence` and does not perform any I/O.
"""

import time
import numpy as np

def _objective_from_blocks(blocks: np.ndarray) -> float:
    """Return the proxy objective 2*max(conv(blocks,blocks))/len(blocks)."""
    K = blocks.shape[0]
    # convolution of the short vector with itself
    max_conv = np.max(np.convolve(blocks, blocks))
    return 2.0 * max_conv / K


def _build_full_sequence(blocks: np.ndarray, target_len: int) -> np.ndarray:
    """Repeat each block entry to obtain a sequence of length target_len."""
    K = blocks.shape[0]
    L = target_len // K
    remainder = target_len - K * L
    full = np.repeat(blocks, L)
    if remainder:
        full = np.concatenate([full, blocks[:remainder]])
    return full

    # timing
    end_time = time.time() + budget_s - 5.0   # leave a safety margin

    # annealing / step‑size schedule
    temperature = 1.0
    min_temp = 1e-3
    decay = 0.9995

    # counters for occasional status output
    last_print = time.time()
    it = 0
    no_improve = 0

    while time.time() < end_time:
        it += 1

        # current convolution and its maximal index
        conv = np.convolve(blocks, blocks)
        max_conv = conv.max()
        k_max = int(conv.argmax())

        # gradient of the maximal convolution term w.r.t each block entry
        grad = np.zeros(K, dtype=float)
        i_start = max(0, k_max - (K - 1))
        i_end = min(K - 1, k_max)
        for i in range(i_start, i_end + 1):
            j = k_max - i
            grad[i] = 2.0 * blocks[j]          # d / d h_i (h_i * h_j * 2)

        # choose a source (largest gradient) and a sink (smallest gradient)
        i_source = int(np.argmax(grad))
        i_sink = int(np.argmin(grad))
        if i_source == i_sink:
            # pick a different sink at random
            cand = np.arange(K)
            cand = cand[cand != i_source]
            i_sink = int(np.random.choice(cand))

        # propose moving a small amount of mass
        step_frac = 0.06 * np.random.rand()     # up to 6 % of source value
        delta = step_frac * blocks[i_source]
        if delta <= 0.0:
            continue

        # perform the transfer while keeping non‑negativity
        new_blocks = blocks.copy()
        new_blocks[i_source] -= delta
        new_blocks[i_sink] += delta

        # compute new objective (exact proxy)
        new_score = _objective_from_blocks(new_blocks)

        # greedy acceptance with a tiny simulated‑annealing component
        accept = False
        if new_score < best_score:
            accept = True
            best_score = new_score
            best_blocks = new_blocks.copy()
            no_improve = 0
        elif new_score < _objective_from_blocks(blocks):
            accept = True
        else:
            # occasional uphill moves to escape plateaus
            if np.random.rand() < np.exp(-(new_score - _objective_from_blocks(blocks)) / temperature):
                accept = True

        if accept:
            blocks = new_blocks
            temperature = max(min_temp, temperature * decay)
            no_improve = 0
        else:
            no_improve += 1

        # occasional random reshuffle to diversify the search
        if no_improve > 5000:
            # add a random perturbation and re‑normalise the sum to K
            perturb = 0.05 * np.random.rand(K) * blocks
            blocks += perturb
            blocks = np.maximum(blocks, 0.0)
            blocks *= K / blocks.sum()
            no_improve = 0

        # status logging every few seconds
        if time.time() - last_print > 5.0:
            print(f"Iter {it:7d} | proxy score {best_score:.6f} | temp {temperature:.4f}")
            last_print = time.time()

    # ----------------------------------------------------------------------
    # Build the final full‑length sequence and return it
    # ----------------------------------------------------------------------
    final_seq = _build_full_sequence(best_blocks, N)

    # Ensure the sequence respects the bounds enforced by evaluate_sequence
    final_seq = np.clip(final_seq, 0.0, 1000.0)

    # final sanity check (optional – does not affect the returned value)
    try:
        true_score = evaluate_sequence(final_seq.tolist())
        print(f"Final true score: {true_score:.6f} (proxy {best_score:.6f})")
    except Exception as exc:
        print(f"Error evaluating final sequence: {exc}")

    return final_seq.tolist()
