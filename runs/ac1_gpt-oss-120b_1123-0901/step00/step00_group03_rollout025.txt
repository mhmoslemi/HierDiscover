"""
Propose a candidate sequence of non‑negative heights that improves the
evaluation bound.  The algorithm searches over simple geometric
distributions

    a[i] = (1‑r)·r**i   (0 ≤ i < n‑1)
    a[n‑1] = r**(n‑1)

which have total sum exactly 1 for any 0<r<1.  For a given length *n* the
parameter *r* controls the shape: a smaller *r* concentrates mass at the
beginning, a larger *r* pushes mass toward the end.  The convolution
a∗a then has a peak that is roughly

    max ≈ max( c/e , r**(2·(n‑1)) )  with  c = –ln(r),

so by scanning *n* and *r* we can minimise the evaluation function

    2·n·max / (sum²) = 2·n·max      (sum==1)

which can be made substantially smaller than the uniform bound 2.
The search enumerates a modest grid of (n,r) pairs, keeps the best
sequence, and then performs a short random refinement around that pair.
All work respects the 1000 s time limit.

The returned sequence is a Python list of floats ready for
evaluate_sequence().
"""

import time
import numpy as np
from scipy import optimize


def _build_geometric_seq(n: int, r: float) -> list[float]:
    """Return the geometric sequence of length n with ratio r and sum=1."""
    # first n-1 entries follow the (1‑r)·r**i law, the last entry holds the
    # remaining mass.
    seq = [(1.0 - r) * (r ** i) for i in range(n - 1)]
    seq.append(r ** (n - 1))
    return seq


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative heights) that produces a
    low value of evaluate_sequence.  The search explores geometric families
    a[i] = (1‑r)·r**i (with a tail term) across many lengths and ratios and
    keeps the best sequence found before the time budget expires.
    """
    import numpy as np
    import time

    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5          # safety margin

    best_seq = None
    best_score = float("inf")
    best_n = None
    best_r = None

    # --------------------------------------------------------------
    # 1.) Coarse grid search over lengths and ratios
    # --------------------------------------------------------------
    # Lengths we try – small lengths keep the factor 2·n low, but must be
    # large enough to accommodate the interior peak of the convolution.
    length_candidates = list(range(10, 101, 5))      # 10,15,...,100
    # Ratio candidates – we avoid very extreme ratios that give obviously bad
    # behaviour (r close to 0 or 1).  300 points give a fine enough resolution.
    ratio_candidates = np.linspace(0.70, 0.99, 250)

    for n in length_candidates:
        if time.time() > deadline:
            break
        for r in ratio_candidates:
            # build the geometric sequence
            seq = _build_geometric_seq(n, r)
            # evaluate – evaluate_sequence does the safety checks and the
            # actual objective value.
            score = evaluate_sequence(seq)
            if score < best_score:
                best_score = score
                best_seq = seq
                best_n = n
                best_r = r
                print(
                    f"New best: n={n}, r={r:.5f}, score={best_score:.6f}"
                )

    # --------------------------------------------------------------
    # 2.) Local random refinement around the best pair found
    # --------------------------------------------------------------
    if best_seq is not None:
        # A small window for random perturbations – we keep the length fixed
        # because changing n changes the objective factor 2·n dramatically.
        for _ in range(2000):
            if time.time() > deadline:
                break
            # propose a new ratio by a relative perturbation (up to ±10%)
            r_pert = best_r * (1.0 + np.random.uniform(-0.10, 0.10))
            r_pert = max(0.001, min(0.999, r_pert))

            seq = _build_geometric_seq(best_n, r_pert)
            score = evaluate_sequence(seq)
            if score < best_score:
                best_score = score
                best_seq = seq
                best_r = r_pert
                print(
                    f"Refined best: n={best_n}, r={best_r:.5f}, score={best_score:.6f}"
                )

        # Occasionally try a slightly different length as well.
        for _ in range(500):
            if time.time() > deadline:
                break
            n_pert = best_n + np.random.randint(-3, 4)   # change length by ≤3
            if n_pert < 5:
                continue
            r_pert = best_r * (1.0 + np.random.uniform(-0.05, 0.05))
            r_pert = max(0.001, min(0.999, r_pert))
            seq = _build_geometric_seq(int(n_pert), r_pert)
            score = evaluate_sequence(seq)
            if score < best_score:
                best_score = score
                best_seq = seq
                best_n = len(seq)
                best_r = r_pert
                print(
                    f"Length‑adjusted best: n={best_n}, r={best_r:.5f}, score={best_score:.6f}"
                )

    # --------------------------------------------------------------
    # 3.) Fallback – if the search somehow failed, return a simple uniform
    # --------------------------------------------------------------
    if best_seq is None:
        # Uniform sequence of modest length (guaranteed to be feasible)
        n = 50
        best_seq = [1.0 / n] * n
        best_score = evaluate_sequence(best_seq)
        print(f"Fallback uniform: n={n}, score={best_score:.6f}")

    print(f"Final best: length={len(best_seq)}, score={best_score:.6f}")
    return best_seq
