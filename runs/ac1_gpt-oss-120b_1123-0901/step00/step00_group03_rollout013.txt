def propose_candidate(seed=0, budget_s=1000):
    """Heuristic optimizer for sequence minimizing evaluate_sequence.
    Uses gradient descent on softmax-approximated max convolution, occasional LP-based
    directions, and random perturbations. Runs until deadline and returns best found.
    """
    # set random seed
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2

    # initialize sequence length
    if 'height_sequence_1' in globals():
        init_seq = np.array(height_sequence_1, dtype=float)
        n = init_seq.size
    else:
        n = np.random.randint(800, 1200)
        init_seq = np.random.rand(n)

    # enforce positivity and bounds (evaluate_sequence does that anyway)
    init_seq = np.maximum(init_seq, 0.0)
    # normalize sum to sqrt(2n)
    target_sum = np.sqrt(2.0 * n)
    if np.sum(init_seq) == 0:
        init_seq = np.ones(n)
    init_seq = init_seq * (target_sum / np.sum(init_seq))

    best_seq = init_seq.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    cur_seq = best_seq.copy()
    cur_score = best_score

    # parameters
    base_lr = 0.02 * target_sum / n   # ~0.02 average step
    beta = 12.0  # softmax temperature
    iter_count = 0

    # main loop
    while time.time() < deadline:
        iter_count += 1

        rand = np.random.rand()
        if rand < 0.6:
            # gradient descent step
            # compute convolution using FFT
            Nfft = 1 << ((2 * n - 1).bit_length())
            a_fft = np.fft.fft(cur_seq, Nfft)
            conv_full = np.real(np.fft.ifft(a_fft * a_fft))[:2 * n - 1]

            max_b = np.max(conv_full)
            # softmax weights
            exp_vals = np.exp(beta * (conv_full - max_b))
            w = exp_vals / np.sum(exp_vals)

            # compute dM via convolution with reversed weights
            Nfft2 = 1 << ((3 * n - 2).bit_length())
            a_fft2 = np.fft.fft(cur_seq, Nfft2)

            L = 2 * n - 2
            dM = 2.0 * conv_rev[L - np.arange(n)]

"""
Heuristic optimizer for the step‑function evaluation problem.

The algorithm mixes three complementary strategies:
1. **Soft‑max gradient descent** – approximates the max‑convolution with a
   smooth soft‑max, computes the exact gradient of the resulting objective,
   and takes a small step opposite to the gradient.  The step size decays
   slowly and the sequence is re‑normalised after each step (the objective is
   scale‑invariant).

2. **LP‑based direction** – solves a linear program that finds a non‑negative
   vector `g` with the maximal possible ℓ₁‑sum while keeping the convolution
   `f * g` bounded by the current maximal convolution value `M`.  A short line
   search over the convex combination ` (1‑t) f + t g ` is performed and the
   best point is kept.

3. **Random perturbations** – occasional Gaussian jitter of a small fraction
   of the entries to escape local minima.

The search runs until the allotted budget (≈ 1000 s) is exhausted, keeping
the best sequence seen so far.  The final sequence is returned as a plain
Python list of floats, ready to be fed to `evaluate_sequence`.
"""

import time
import numpy as np
from scipy.optimize import linprog


def _next_pow2(x: int) -> int:
    """Next power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _solve_convolution_lp(f_seq, rhs):
    """
    Linear program:
        maximise   sum(b)
        subject to  (f * b)[k] ≤ rhs   for all k
                    b ≥ 0

    Returns the optimal vector `b` or None if the LP fails.
    """
    n = len(f_seq)
    c = -np.ones(n)                     # minimise -sum(b)  → maximise sum(b)

    # Build convolution constraints: (f * b)[k] = Σ_i f[i] * b[k‑i]
    rows = []
    bk = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        # fill only the entries that actually appear
        for i in range(i_min, i_max + 1):
            j = k - i
            row[j] = f_seq[i]
        rows.append(row)
        bk.append(rhs)

    A_ub = np.vstack(rows)
    b_ub = np.array(bk)

    # Use the high‑performance HiGHS solver (default in recent SciPy)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                  bounds=[(0, None)] * n,
                  method='highs')
    if res.success:
        return res.x
    return None


def _softmax_weights(arr, beta):
    """Stable soft‑max on a 1‑D array."""
    max_val = np.max(arr)
    exp_vals = np.exp(beta * (arr - max_val))
    w = exp_vals / np.sum(exp_vals)
    return w


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a sequence of non‑negative heights that minimises
    `evaluate_sequence`.  The function respects the time budget and returns the
    best sequence found as a list of floats.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0   # a tiny safety margin

    # ------------------------------------------------------------------
    # 1️⃣ Initialise ----------------------------------------------------
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        init = np.array(height_sequence_1, dtype=float).copy()
        n = init.size
    else:
        n = np.random.randint(800, 1200)
        init = np.random.rand(n)

    init = np.maximum(init, 0.0)
    target_sum = np.sqrt(2.0 * n)            # any positive value works (scale‑invariant)
    if np.sum(init) == 0.0:
        init = np.ones(n)
    init = init * (target_sum / np.sum(init))

    cur_seq = init.copy()
    best_seq = cur_seq.copy()
    try:
        best_score = evaluate_sequence(best_seq.tolist())
    except Exception:   # safety – should never happen with the above safeguards
        best_score = float('inf')
    cur_score = best_score

    # ------------------------------------------------------------------
    # 2️⃣ Parameters ----------------------------------------------------
    # ------------------------------------------------------------------
    base_lr = 0.02 * target_sum / n          # base learning‑rate for gradient step
    beta = 12.0                               # soft‑max temperature (larger → sharper)
    iteration = 0

    # ------------------------------------------------------------------
    # 3️⃣ Main optimisation loop ----------------------------------------
    # ------------------------------------------------------------------
    while time.time() < deadline:
        iteration += 1
        choice = np.random.rand()

        # --------------------------------------------------------------
        # 3.A Gradient descent (≈ 60 % of the time)
        # --------------------------------------------------------------
        if choice < 0.6:
            # --- convolution f * f via FFT ---------------------------------
            Nfft = _next_pow2(2 * n - 1)
            f_fft = np.fft.fft(cur_seq, Nfft)
            conv_full = np.real(np.fft.ifft(f_fft * f_fft))[:2 * n - 1]

            max_b = np.max(conv_full)

            # --- soft‑max approximation of the max -------------------------
            w = _softmax_weights(conv_full, beta)

            # --- gradient of the smooth objective -------------------------
            # dM_i = 2 * Σ_j a_j * w_{i+j}
            Nfft2 = _next_pow2(3 * n - 2)
            f_fft2 = np.fft.fft(cur_seq, Nfft2)
            w_rev_fft = np.fft.fft(w[::-1], Nfft2)
            conv_rev = np.real(np.fft.ifft(f_fft2 * w_rev_fft))
            L = 2 * n - 2
            dM = 2.0 * conv_rev[L - np.arange(n)]

            S = np.sum(cur_seq)
            M_approx = np.sum(w * conv_full)

            grad = (2.0 * n / (S ** 3)) * (S * dM - 2.0 * M_approx)

            # --- step (learning‑rate decays very slowly) ------------------
            lr = base_lr * (0.9 ** (iteration // 200))
            new_seq = cur_seq - lr * grad
            new_seq = np.maximum(new_seq, 0.0)

            if np.sum(new_seq) > 0.0:
                new_seq = new_seq * (target_sum / np.sum(new_seq))
            else:                               # re‑initialise if we collapsed to zero
                new_seq = np.random.rand(n)
                new_seq = new_seq * (target_sum / np.sum(new_seq))

            # --- evaluate -------------------------------------------------
            try:
                new_score = evaluate_sequence(new_seq.tolist())
                if new_score < best_score:
                    best_score = new_score
                    best_seq = new_seq.copy()
                    print(f"[{iteration:05d}] New best (grad)  = {best_score:.6f}")

                # accept improving moves; occasionally accept a worse move
                if new_score <= cur_score or np.random.rand() < 0.02:
                    cur_seq = new_seq
                    cur_score = new_score
            except Exception:
                pass

        # --------------------------------------------------------------
        # 3.B LP‑based direction (≈ 30 % of the time)
        # --------------------------------------------------------------
        elif choice < 0.9:
            M_cur = np.max(np.convolve(cur_seq, cur_seq))
            g = _solve_convolution_lp(cur_seq.tolist(), M_cur)

            if g is not None:
                g_arr = np.array(g)
                # line‑search over t ∈ (0,1] (grid coarse → cheap)
                best_local_seq = cur_seq
                best_local_score = cur_score

                for t in np.linspace(0.05, 1.0, 10):
                    cand = (1.0 - t) * cur_seq + t * g_arr
                    if np.sum(cand) == 0.0:
                        continue
                    cand = cand * (target_sum / np.sum(cand))
                    try:
                        cand_score = evaluate_sequence(cand.tolist())
                    except Exception:
                        continue
                    if cand_score < best_local_score:
                        best_local_score = cand_score
                        best_local_seq = cand.copy()

                if best_local_score < best_score:
                    best_score = best_local_score
                    best_seq = best_local_seq.copy()
                    print(f"[{iteration:05d}] New best (LP)    = {best_score:.6f}")

                if best_local_score < cur_score:
                    cur_seq = best_local_seq
                    cur_score = best_local_score

        # --------------------------------------------------------------
        # 3.C Random perturbation (≈ 10 % of the time)
        # --------------------------------------------------------------
        else:
            pert = cur_seq.copy()
            k = max(1, int(0.02 * n))
            idx = np.random.choice(n, size=k, replace=False)
            scale = np.median(pert) if np.median(pert) > 0 else target_sum / n
            pert[idx] += np.random.randn(k) * 0.1 * scale
            pert = np.maximum(pert, 0.0)
            pert = pert * (target_sum / np.sum(pert))

            try:
                pert_score = evaluate_sequence(pert.tolist())
                if pert_score < best_score:
                    best_score = pert_score
                    best_seq = pert.copy()
                    print(f"[{iteration:05d}] New best (pert) = {best_score:.6f}")
                if pert_score < cur_score:
                    cur_seq = pert
                    cur_score = pert_score
            except Exception:
                pass

        # --------------------------------------------------------------
        # 3.D Optional restart if stagnating (very cheap)
        # --------------------------------------------------------------
        if iteration % 500 == 0 and best_score > 1.9:
            cur_seq = np.random.rand(n)
            cur_seq = cur_seq * (target_sum / np.sum(cur_seq))
            cur_score = evaluate_sequence(cur_seq.tolist())

    # ------------------------------------------------------------------
    # 4️⃣ Return the best sequence found
    # ------------------------------------------------------------------
    print(f"Search finished after {iteration} iterations – best score: {best_score:.6f}")
    return best_seq.tolist()
