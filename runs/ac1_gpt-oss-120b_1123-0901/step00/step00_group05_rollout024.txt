    # initialize sequence random Dirichlet
    alpha = 0.5  # parameter controlling sparsity (smaller => more extreme)
    seq = np.random.gamma(alpha, 1.0, size=n)
    seq = seq / seq.sum()

    best_seq = seq.copy()
    best_score = evaluate_sequence(seq.tolist())

    # Optimization loop
    step = 0.1  # step size for gradient updates
    while time.time() < deadline:
        # compute convolution and find max offset(s)
        conv = np.convolve(seq, seq)
        max_val = conv.max()
        max_idxs = np.flatnonzero(conv == max_val)

        # compute gradient w.r.t. max_b (average of grads for all max idx)
        grad = np.zeros_like(seq)
        for k in max_idxs:
            lower = max(0, k - (n - 1))
            upper = min(n - 1, k)
            for i in range(lower, upper + 1):
                j = k - i
                if i != j:
                    grad[i] += seq[j]
                    grad[j] += seq[i]

                    grad[i] += 2 * seq[i]

function compute_gradient(seq, max_idx):
    n = len(seq)
    grad = np.zeros(n)
    for k in max_idx:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            j = k - i
            if i < j:
                grad[i] += seq[j]
                grad[j] += seq[i]
            elif i == j:
                grad[i] += 2 * seq[i]
    return grad / len(max_idx)
