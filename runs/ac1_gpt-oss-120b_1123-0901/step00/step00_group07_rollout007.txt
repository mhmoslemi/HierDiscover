def propose_candidate(seed=42, budget_s=1000):
    import numpy as np, time, random
    random.seed(seed)
    np.random.seed(seed)
    best_seq = None
    best_score = float('inf')
    deadline = time.time() + budget_s - 5

    # try different lengths
    lengths = [256, 512, 768, 1024, 1536, 2048, 3072]
    for N in lengths:
        # choose m about sqrt(N)*something: maybe int(0.7 * N) ??? we need not exceed N.
        # set target density to about 0.4
        m = int(0.4 * N)  # ensure at least 20?
        # ensure m>=10
        m = max(10, m)
        # generate initial random positions distinct
        positions = random.sample(range(N), m)
        positions_set = set(positions)
        # precompute counts
        cnt = np.zeros(2 * N - 1, dtype=int)
        pos_list = positions
        for i in range(m):
            pi = pos_list[i]
            for j in range(m):
                pj = pos_list[j]
                cnt[pi+pj] += 1  # ordered pairs (i,j) -> includes both i=j and i!=j
        # equivalent to double loop; we could use fast method but fine.

        # compute current score
        max_c = int(cnt.max())
        curr_score = 2 * N * max_c / (m * m)
        if curr_score < best_score:
            best_score = curr_score
            best_seq = np.zeros(N)
            best_seq[positions] = 1.0

        # annealing/hillclimbing:
        temperature = 1.0
        while time.time() < deadline:
            # propose a move: choose random pos to remove, random new pos absent
            p_remove = random.choice(pos_list)
            p_add = random.randrange(N)
            while p_add in positions_set:
                p_add = random.randrange(N)
            # compute new max after move without fully recompute?
            # We'll apply move and measure.
            # Remove contributions
            for j in pos_list:
                if j == p_remove:
                    cnt[2 * p_remove] -= 1
                else:
                    cnt[p_remove + j] -= 2
            # Now positions set update: remove p_remove
            positions_set.remove(p_remove)
            pos_list.remove(p_remove)
            # Add contributions for p_add
            for j in pos_list:
                cnt[p_add + j] += 2
            cnt[2 * p_add] += 1
            # add p_add to set/list
            positions_set.add(p_add)
            pos_list.append(p_add)
            # compute updated max
            new_max_c = int(cnt.max())
            # evaluate new score
            new_score = 2 * N * new_max_c / (m * m)
            # acceptance criterion
            if new_score < curr_score or random.random() < np.exp((curr_score - new_score) / temperature):
                # accept move
                curr_score = new_score
                max_c = new_max_c
                # update best
                if new_score < best_score:
                    best_score = new_score
                    best_seq = np.zeros(N)
                    best_seq[pos_list] = 1.0
            else:
                # revert move
                # Remove contributions of new added p_add
                for j in pos_list:
                    if j == p_add:
                        cnt[2 * p_add] -= 1
                    else:
                        cnt[p_add + j] -= 2
                positions_set.remove(p_add)
                pos_list.remove(p_add)
                # Add contributions back of p_remove
                for j in pos_list:
                    cnt[p_remove + j] += 2
                cnt[2 * p_remove] += 1
                positions_set.add(p_remove)
                pos_list.append(p_remove)
                # recompute max after revert
                max_c = int(cnt.max())
                # curr_score unchanged
            # cooling
            temperature *= 0.9999
            # break if time is near up
            if time.time() > deadline - 0.1:
                break

    # Convert best_seq to list of floats
    best_seq_list = [float(x) for x in best_seq.tolist()]
    print(f"Best score: {best_score}, length: {len(best_seq_list)}")
    return best_seq_list

"""
Search for a binary step function (non‑negative heights) that minimizes the
``evaluate_sequence`` objective.  The algorithm picks a length N and a target
density m/N, creates a random support, and repeatedly tries to improve the
maximum self‑convolution value by swapping a ``1``‑position with a ``0``‑position.
Only moves that strictly decrease the objective are accepted (greedy hill‑climb).
Multiple lengths and densities are tried within the allowed time budget.
The best sequence found is returned.
"""

import time
import random
import numpy as np

def _initial_counts(N, positions):
    """Return the array of ordered‑pair counts for a binary sequence of length N."""
    m = len(positions)
    cnt = np.zeros(2 * N - 1, dtype=np.int32)
    # ordered pairs (i,j)
    for i in range(m):
        pi = positions[i]
        for j in range(m):
            cnt[pi + positions[j]] += 1
    return cnt

def _try_swap(N, positions, pos_set, cnt, m):
    """
    Propose a random swap (remove one index, insert a new free index),
    compute the new count array and return information needed to accept it.
    Returns (new_cnt, new_positions, new_set, new_max) or None if the swap
    does not improve the objective.
    """
    # pick an index to remove and a free index to add
    p_remove = random.choice(positions)
    p_add = random.randrange(N)
    while p_add in pos_set:
        p_add = random.randrange(N)

    # copy the count array
    cnt2 = cnt.copy()

    # subtract contributions of the removed index
    for pj in positions:
        if pj == p_remove:
            cnt2[2 * p_remove] -= 1          # self term
        else:
            cnt2[p_remove + pj] -= 2          # ordered pair (p_remove,pj) and (pj,p_remove)

    # add contributions of the new index (pair with the remaining positions)
    for pj in positions:
        if pj == p_remove:
            continue
        cnt2[p_add + pj] += 2
    cnt2[2 * p_add] += 1                     # self term for the new index

    new_max = int(cnt2.max())
    # compute the objective value for binary weights (each weight = 1)
    #   f = 2 * N * max_c / (m * m)
    new_score = 2.0 * N * new_max / (m * m)

    # current score (derived from current cnt)
    cur_max = int(cnt.max())
    cur_score = 2.0 * N * cur_max / (m * m)

    if new_score < cur_score - 1e-12:        # strict improvement (tiny tolerance)
        # build new position list
        new_positions = positions.copy()
        idx = new_positions.index(p_remove)
        new_positions[idx] = p_add
        new_set = pos_set.copy()
        new_set.remove(p_remove)
        new_set.add(p_add)
        return cnt2, new_positions, new_set, new_max, new_score
    return None
