"""
Propose a step‑function (non‑negative sequence) that minimizes the evaluation
function:

    score = 2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm uses a simple projected gradient descent on a smooth
approximation of the max‑convolution.  The idea is to flatten the
self‑convolution: a uniform distribution gives a triangular convolution
with a maximal value 1/n (score = 2).  By moving mass away from the region
that creates the current peak we can reduce that peak and obtain a lower
score (target ≈ 1.5).

Key steps
---------

* Initialise several candidates (uniform sequences of various lengths and,
  if available, the previously discovered `height_sequence_1`).
* For each candidate perform many gradient‑descent iterations.
    - Compute the self‑convolution `b = a * a`.
    - Build a soft‑max weight vector `w` that concentrates on the largest
      entries of `b` (parameter `lam` controls the sharpness).
    - The gradient of the smooth max w.r.t. `a` is
        g_i = 2 * Σ_k w_k * a_{k‑i}
      which can be evaluated efficiently by a single convolution.
    - Take a step `a ← a – step * g`, project onto the non‑negative orthant
      and renormalise to keep Σ a_i = 1 (the score is homogeneous, any
      positive normalisation works).
* Adapt the step size and the smoothing parameter; if no improvement is
  seen for a while, perform a random Dirichlet restart.
* Keep the best sequence found across all lengths and restarts.
* Stop when the allotted time (≈ 1000 s) is exhausted and return the best
  sequence.

The method is completely self‑contained, uses only `numpy` and `time`,
and respects all platform restrictions.
"""

import time
import numpy as np
import random

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _score(seq):
    """
    Compute the score used by `evaluate_sequence` (without the safety caps).
    The sequence may be any iterable of non‑negative numbers.
    """
    a = np.asarray(seq, dtype=float)
    total = a.sum()
    if total < 1e-12:
        return float("inf")
    b = np.convolve(a, a)
    maxb = b.max()
    n = a.size
    # identical to evaluate_sequence when sum_a = total
    return 2.0 * n * maxb / (total * total)


def _gradient_step(a, lam, step):
    """
    One projected gradient step that aims at reducing the maximal entry of
    the self‑convolution.

    Parameters
    ----------
    a : np.ndarray
        Current (non‑negative) sequence.  Its sum does not need to be 1;
        the function normalises it internally.
    lam : float
        Soft‑max temperature (>0).  Larger values focus more on the true
        maximum.
    step : float
        Step length for the gradient descent.

    Returns
    -------
    np.ndarray
        Updated sequence (still non‑negative and normalised to sum 1).
    """
    # Normalise to a probability vector (the score is homogeneous)
    total = a.sum()
    if total <= 0.0:
        a = np.full_like(a, 1.0 / a.size)
    else:
        a = a / total

    n = a.size

    # self‑convolution and smooth weights
    b = np.convolve(a, a)                     # length 2n‑1
    maxb = b.max()
    # soft‑max (subtract max for stability)
    exp_part = np.exp(lam * (b - maxb))
    w = exp_part / exp_part.sum()             # sums to 1

    # gradient = 2 * Σ_k w_k * a_{k‑i}
    # This is a cross‑correlation; compute via convolution of reversed a.
    conv = np.convolve(a[::-1], w)            # length 3n‑2
    g = 2.0 * conv[n - 1 : n - 1 + n]         # same length as a

    # gradient descent step
    a_new = a - step * g
    a_new = np.maximum(a_new, 0.0)

    # renormalise again (projection onto the simplex)
    total_new = a_new.sum()
    if total_new <= 0.0:
        a_new = np.full_like(a_new, 1.0 / n)
    else:
        a_new = a_new / total_new

    return a_new
