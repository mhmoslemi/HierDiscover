def project_simplex(v):
    # from literature: sort v descending, find rho, theta.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, len(v)+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

def improve_sequence(a, max_iter, step_factor=0.1):
    n = len(a)
    for it in range(max_iter):
        b = np.convolve(a, a)
        M = b.max()
        active = np.where(b >= M - 1e-12)[0]
        C = np.zeros(n)
        for k in active:
            i_start = max(0, k-(n-1))
            i_end = min(k, n-1)
            # vectorized? But loops fine.
            for i in range(i_start, i_end+1):
                j = k - i
                C[i] += a[j]   # as described
        # find indices for transfer
        i_high = np.argmax(C)
        i_low = np.argmin(C)
        if i_high == i_low:
            break
        # transfer delta
        delta = step_factor * a[i_high]  # adjust factor maybe decreasing
        if delta < 1e-12:
            break
        a[i_high] -= delta
        a[i_low] += delta
        # ensure non-negative
        if a[i_high] < 0:
            # adjust
            a[i_low] += a[i_high]
            a[i_high] = 0
        # Optionally renormalize to sum 1 (should remain same)
        # continue
    return a

    for it in range(max_iter):
        b = np.convolve(a, a)
        M = b.max()
        active = np.where(b >= M - 1e-12)[0]
        # compute contribution gradients
        C = np.zeros(n)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            # vectorized extraction
            i_indices = np.arange(i_start, i_end+1)
            j_indices = k - i_indices
            C[i_indices] += a[j_indices]
        # find high and low contributions
        i_high = np.argmax(C)
        i_low = np.argmin(C)
        if C[i_high] <= C[i_low] + 1e-12:
            break
        delta = step_factor * a[i_high]
        if delta <= 1e-12:
            break
        a[i_high] -= delta
        a[i_low] += delta
        # ensure non-negativity
        if a[i_high] < 0:
            a[i_low] += a[i_high]
            a[i_high] = 0.0
        # renormalize to sum 1
        a = a / a.sum()
        # gradually reduce step_factor
        step_factor *= 0.99
    return a

"""
Simple parametric search for a low‐value step function.

The evaluation to minimise is
    2 * n * max(conv(a, a)) / sum(a)^2

Because the expression is scale invariant we fix sum(a)=1.
A decreasing sequence makes the self‑convolution at the centre
small (the largest coefficients are paired with the smallest ones).
The algorithm enumerates a few families of monotone sequences
(linear, quadratic, exponential, power law) for various lengths,
keeps the best one and finally performs a short random‑perturbation
local search to possibly improve it further.

The whole search stops when the allocated time budget (≈1000 s) is
exhausted, guaranteeing that a valid sequence is always returned.
"""

import numpy as np
import time

# ----------------------------------------------------------------------
# Helper functions (top level – no closures)
# ----------------------------------------------------------------------


def project_to_simplex(v):
    """Project a non‑negative vector onto the probability simplex (sum=1)."""
    # Implementation from https://arxiv.org/abs/1309.1541
    if np.allclose(v.sum(), 1) and np.min(v) >= 0:
        return v
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w


def linear_decreasing(n):
    """a_i ∝ (n‑i)  (i = 0 … n‑1)."""
    a = np.arange(n, 0, -1, dtype=float)
    return a / a.sum()


def quadratic_decreasing(n):
    """a_i ∝ (n‑i)^2."""
    a = np.arange(n, 0, -1, dtype=float) ** 2
    return a / a.sum()


def exponential_decreasing(n, rate=0.01):
    """a_i ∝ exp(-rate * i).  rate>0 → decreasing."""
    a = np.exp(-rate * np.arange(n, dtype=float))
    return a / a.sum()


def power_decreasing(n, gamma):
    """a_i ∝ (i+1)^{-gamma}  (gamma≥0)."""
    idx = np.arange(1, n + 1, dtype=float)
    a = idx ** (-gamma)
    return a / a.sum()


def random_perturbation(seq, step=0.02):
    """Small random mass transfer between two positions (keeps sum = 1)."""
    a = np.array(seq, copy=True)
    n = len(a)
    if n < 2:
        return a.tolist()
    i, j = np.random.choice(n, size=2, replace=False)
    delta = step * a[i] * np.random.rand()
    a[i] = max(0.0, a[i] - delta)
    a[j] += delta
    # re‑normalise to avoid tiny drift
    a = project_to_simplex(a)
    return a.tolist()


# ----------------------------------------------------------------------
# Main candidate generator
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 990, **kwargs):
    """Search for a sequence with a low evaluation score.

    The function explores a few parametric families of monotone decreasing
    sequences of various lengths, keeps the best one, and then performs a
    short random‑local‑search phase.  It stops before the time budget expires
    and returns the best sequence found.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + max(0.0, budget_s - 5.0)   # leave a little margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 1) deterministic scan over a handful of families and lengths
    # ------------------------------------------------------------------
    lengths = [50, 80, 120, 200, 300, 500]          # diverse sizes (still fast)
    gammas = [0.2, 0.5, 1.0, 2.0, 4.0]               # for the power family
    rates = [0.005, 0.01, 0.02, 0.05]               # for the exponential family

    # linear decreasing (p=1)
    for n in lengths:
        seq = linear_decreasing(n)
        try:
            score = evaluate_sequence(seq.tolist())
            if score < best_score:
                best_score, best_seq = score, seq.tolist()
                print(f"[init] linear n={n} score={score:.6f}")
        except Exception:
            pass

    # quadratic decreasing
    for n in lengths:
        seq = quadratic_decreasing(n)
        try:
            score = evaluate_sequence(seq.tolist())
            if score < best_score:
                best_score, best_seq = score, seq.tolist()
                print(f"[init] quadratic n={n} score={score:.6f}")
        except Exception:
            pass

    # exponential families
    for n in lengths:
        for r in rates:
            seq = exponential_decreasing(n, rate=r)
            try:
                score = evaluate_sequence(seq.tolist())
                if score < best_score:
                    best_score, best_seq = score, seq.tolist()
                    print(f"[init] exp(rate={r:.3f}) n={n} score={score:.6f}")
            except Exception:
                pass

    # power‑law families
    for n in lengths:
        for gamma in gammas:
            seq = power_decreasing(n, gamma=gamma)
            try:
                score = evaluate_sequence(seq.tolist())
                if score < best_score:
                    best_score, best_seq = score, seq.tolist()
                    print(f"[init] power(gamma={gamma}) n={n} score={score:.6f}")
            except Exception:
                pass

    # ------------------------------------------------------------------
    # 2) short stochastic refinement (random perturbations + occasional
    #    re‑initialisation) while time permits.
    # ------------------------------------------------------------------
    iteration = 0
    while time.time() < deadline:
        iteration += 1

        # occasionally restart from a fresh random Dirichlet vector
        if best_seq is None or np.random.rand() < 0.03:
            n = np.random.randint(60, 500)
            a = np.random.rand(n)
            a = a / a.sum()
        else:
            # start from the current best and apply a small perturbation
            a = np.array(best_seq, copy=True)
            # random transfer of a modest fraction of mass
            i, j = np.random.choice(len(a), size=2, replace=False)
            delta = 0.01 * a[i] * np.random.rand()
            a[i] = max(0.0, a[i] - delta)
            a[j] += delta
            a = project_to_simplex(a)

        # evaluate the candidate
        try:
            score = evaluate_sequence(a.tolist())
            if score < best_score:
                best_score, best_seq = score, a.tolist()
                print(f"[improve] iter={iteration} new best={score:.6f} n={len(a)}")
                # early exit if we already beat the target
                if best_score < 1.503:
                    break
        except Exception:
            # any invalid sequence is simply ignored
            pass

    # Fallback – should never happen, but guard against empty result.
    if best_seq is None:
        best_seq = [1.0] * 10

    print(f"Finished search: best score={best_score:.6f}, length={len(best_seq)}")
    return best_seq
