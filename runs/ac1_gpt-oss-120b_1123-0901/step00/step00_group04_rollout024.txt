def flatten_sequence(a, steps=2000, step_size=0.01):
    n=len(a)
    for t in range(steps):
        b = np.convolve(a, a)
        max_k = int(b.argmax())
        # compute contributions
        contrib = np.zeros(n)
        for i in range(n):
            j = max_k - i
            if 0 <= j < n:
                contrib[i] = a[j]  # g_i = 2*a[j], but we ignore factor 2
        # positions with largest contribution
        idx_max = np.argmax(contrib)
        idx_min = np.argmin(contrib)  # may be zero
        # Transfer amount
        delta = step_size * a[idx_max]  # small fraction
        if delta > a[idx_max]:
            delta = a[idx_max]
        # ensure not too small
        if delta <= 0:
            break
        a[idx_max] -= delta
        a[idx_min] += delta
        # optional re-normalize (should stay sum as same)
        # also ensure nonneg
    return a

def anneal(a, max_iters, init_temp, final_temp):
    n = len(a)
    temp = init_temp
    cooling = (final_temp/init_temp)**(1.0/max_iters)
    best_a = a.copy()
    best_score = compute_score(a)
    curr_score = best_score
    for iter in range(max_iters):
        i, j = np.random.randint(n), np.random.randint(n)
        if i == j:
            continue
        # propose transfer
        delta = (np.random.rand() - 0.5) * 2 * temp * a[i]  # maybe scale
        # ensure we don't go negative beyond a[i]
        if delta > a[i]:
            delta = a[i]
        # apply
        a[i] -= delta
        a[j] += delta
        # compute new score
        new_score = compute_score(a)
        # accept?
        if new_score < curr_score or np.random.rand() < np.exp((curr_score - new_score)/temp):

            # revert

            a[j] -= delta
        temp *= cooling

import numpy as np, time, math, random

def compute_score(seq):
    n = len(seq)
    b = np.convolve(seq, seq)
    max_b = np.max(b)
    sum_a = np.sum(seq)
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a * sum_a)

def anneal(seq, iter_budget, start_temp=1.0, end_temp=1e-3):
    n = len(seq)
    a = seq.copy()
    best_seq = a.copy()
    best_score = compute_score(a)
    curr_score = best_score
    temp = start_temp
    # Cooling schedule geometric
    cooling = (end_temp/start_temp) ** (1.0/iter_budget) if iter_budget>0 else 1.0
    for it in range(iter_budget):
        # pick two distinct indices
        i, j = np.random.randint(n), np.random.randint(n)
        if i == j:
            continue
        # choose direction
        if np.random.rand() < 0.5:
            donor, receiver = i, j
        else:
            donor, receiver = j, i
        if a[donor] <= 0:
            continue
        # step size proportional to temp and donor amount
        max_transfer = a[donor]
        delta = np.random.rand() * max_transfer * temp
        # move delta
        a[donor] -= delta
        a[receiver] += delta
        # re-normalize to avoid drift (optional small)
        # not needed because sum conserved exactly
        new_score = compute_score(a)
        # accept if better or with probability
        if new_score < curr_score or np.random.rand() < math.exp((curr_score - new_score) / (temp + 1e-12)):
            curr_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = a.copy()
        else:
            # revert
            a[donor] += delta
            a[receiver] -= delta
        # decrease temperature
        temp *= cooling
        # occasional re-normalization to sum 1 (to fix floating errors)
        if it % 100 == 0:
            total = a.sum()
            if total != 0:
                a /= total
    return best_seq, best_score

    """

    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5  # leave a little margin
    # keep global best
    best_seq = None
    best_score = float('inf')
    # optionally start from the known sequence if provided
    try:
        # height_sequence_1 might be defined globally
        init = list(height_sequence_1)
        # ensure it's a list of floats and normalised
        init = [float(x) for x in init]
        init = np.maximum(0, init)  # just in case
        init = np.minimum(1000.0, init)
        init_sum = np.sum(init)
        if init_sum > 0:
            init = init / init_sum   # scale to sum 1 (scale invariant)
            score = compute_score(init)
            best_seq = init.tolist()
            best_score = score
    except Exception:
        pass

    # search parameters
    lengths = [2,3,4,5,6,7,8,9,10,12,15,20,30,40,50,70,100,150,200,300]
    # main loop
    while time.time() < deadline:
        n = np.random.choice(lengths)
        # random Dirichlet initialisation
        a = np.random.rand(n)
        a = a / a.sum()
        # use a few iterations of annealing
        # allocate iteration count based on remaining time
        remaining = deadline - time.time()
        # aim for maybe 2000 iterations per try but limit to time
        iters = min(3000, max(200, int(remaining * 5)))  # rough
        seq, score = anneal(a, iters, start_temp=1.0, end_temp=1e-4)
        if score < best_score:
            best_score = score
            best_seq = seq.tolist()
            print(f"New best score {best_score:.6f} with length {n}")
        # occasional small random perturb to global best and continue
        if best_seq is not None and np.random.rand() < 0.2:
            # take a copy of current best and jitter a bit
            a2 = np.array(best_seq)
            # add Gaussian noise, keep nonnegative, renorm
            noise = np.random.randn(len(a2)) * 0.01
            a2 = a2 + noise
            a2 = np.clip(a2, 0, None)
            if a2.sum() == 0:
                a2 = np.random.rand(len(a2))
            a2 = a2 / a2.sum()
            seq2, score2 = anneal(a2, 1500, start_temp=0.5, end_temp=1e-5)
            if score2 < best_score:
                best_score = score2
                best_seq = seq2.tolist()
                print(f"Improved via perturbation to {best_score:.6f}")
    # final guard: ensure we return a list of floats
    if best_seq is None:
        # fallback to simple two-element optimum (1/3,2/3)
        best_seq = [1/3, 2/3]
    return best_seq

def anneal_sequence(n, iter_budget, seed):
    rng = np.random.RandomState(seed)
    # initialize random Dirichlet
    a = rng.rand(n)
    a = a / a.sum()
    curr_score = compute_score(a)
    best_a = a.copy()
    best_score = curr_score
    temp = 0.5
    cooling = 0.9995  # may need many iterations
    for it in range(iter_budget):
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        donor = i
        receiver = j
        # move fraction up to temp
        max_frac = min(1.0, temp)
        frac = rng.rand() * max_frac
        delta = a[donor] * frac
        if delta <= 0:
            continue
        a_new = a.copy()
        a_new[donor] -= delta
        a_new[receiver] += delta
        # compute score
        new_score = compute_score(a_new)
        if new_score < curr_score or rng.rand() < math.exp((curr_score - new_score) / (temp + 1e-12)):
            a = a_new
            curr_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
        # cooling
        temp *= cooling
        # periodic renorm
        if it % 100 == 0:
            s = a.sum()
            if s != 0:
                a = a / s
    return best_a, best_score
```

"""
Simple stochastic optimizer for the convolution‑based objective.

The objective is scale invariant, therefore we normalise every candidate
to have a unit L1‑norm (sum == 1).  For a fixed length `n` we try to
minimise  max(conv(a,a))  which is exactly 0.5·score/n after the normalisation.
A simulated‑annealing scheme moves a small amount of mass from one index to
another while keeping the total sum constant.  Moves that decrease the
maximum convolution value are always accepted; otherwise they are accepted
with the usual Metropolis probability.  Several random restarts with
different lengths are performed within the time budget and the best
sequence found is returned.

The routine is deliberately lightweight and uses only NumPy; no external
solvers are required.
"""

import time
import math
import numpy as np

# -------------------------------------------------------------
# Helper: fast convolution (real FFT) – works for all n we use.
def _conv(a: np.ndarray) -> np.ndarray:
    """Return np.convolve(a, a) using real FFT (fast for n > ~50)."""
    n = a.shape[0]
    # length of the linear convolution
    m = 2 * n - 1
    # next power of two for efficient FFT
    size = 1 << (m - 1).bit_length()
    A = np.fft.rfft(a, size)
    B = A * A
    conv = np.fft.irfft(B, size)[:m]
    return conv

# -------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """
    Compute the evaluation score for a normalised (sum == 1) non‑negative vector.
    The formula is 2 * n * max(conv(a,a)) because the denominator sum^2 equals 1.
    """
    n = a.shape[0]
    b = _conv(a)
    max_b = np.max(b)
    return 2.0 * n * max_b

# -------------------------------------------------------------
def _anneal_one(n: int, iters: int, seed: int) -> tuple[np.ndarray, float]:
    """
    Simulated annealing for a fixed length `n`.

    Returns the best vector found (already normalised) and its score.
    """
    rng = np.random.RandomState(seed)
    # start from a random Dirichlet distribution
    a = rng.rand(n)
    a = a / a.sum()
    cur_score = _score(a)

    best_a = a.copy()
    best_score = cur_score

    # annealing schedule – start relatively hot and cool geometrically
    temp = 0.5                     # initial temperature (max fraction moved)
    cooling = 0.9995               # per‑iteration cooling factor

    for it in range(iters):
        # pick two distinct indices
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue

        donor, receiver = (i, j) if rng.rand() < 0.5 else (j, i)

        if a[donor] == 0.0:
            continue

        # the fraction of `donor` we try to move; limited by the current temperature
        max_frac = min(1.0, temp)
        frac = rng.rand() * max_frac
        delta = a[donor] * frac   # amount to transfer

        # propose new vector
        a_new = a.copy()
        a_new[donor] -= delta
        a_new[receiver] += delta

        # because of floating point drift we renormalise occasionally
        if (it & 31) == 0:               # every 32 steps
            a_new /= a_new.sum()

        new_score = _score(a_new)

        # Metropolis acceptance
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            # probability declines with temperature
            if rng.rand() < math.exp((cur_score - new_score) / (temp + 1e-12)):
                accept = True

        if accept:
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()

        # cool down
        temp *= cooling

    return best_a, best_score

    """

    deadline = start_time + budget_s - 5.0         # keep a small safety margin

    # keep the globally best sequence discovered
    best_seq = None
    best_score = float("inf")

    # If the previous best construction is available, use it as a seed.
    try:

            init_score = _score(init)
            best_seq = init.tolist()
            best_score = init_score
    except Exception:

    lengths = [2, 3, 4, 5, 6, 8, 10, 12, 15, 20, 30,
