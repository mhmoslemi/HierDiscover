Projection onto simplex:

Standard algorithm:

def simplex_projection(v, z=1):
    """Project v onto simplex sum(v)=z, v>=0."""
    if np.sum(v) == z and np.all(v >= 0): return v
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)

    np.random.seed(seed)
    deadline = time.time() + budget_s - 2

    best_seq = None
    best_score = float('inf')
    iter_counter = 0

    while time.time() < deadline:
        # pick sequence length randomly within range 200-2000
        n = np.random.randint(200, 2000)
        # random init vector on simplex (sum=1)
        a = np.random.rand(n)
        a = a / np.sum(a)
        # optionally some extra smoothing: apply moving average to start
        # run gradient descent
        max_iters = 2000
        # step size schedule:
        step0 = 0.5
        for it in range(max_iters):
            # compute convolution via FFT
            conv = conv_fft(a)
            max_val = conv.max()
            # compute current score
            score = 2 * n * max_val / (np.sum(a) ** 2)
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                # print improvement
                print(f"Improved: n={n}, iter={iter_counter}, it={it}, score={score:.6f}")
            # termination early if enough improvement?
            # compute gradient from max index
            kmax = int(np.argmax(conv))
            # compute gradient
            i = np.arange(n)
            idx = kmax - i
            mask = (idx >= 0) & (idx < n)
            grad = np.zeros_like(a)
            grad[mask] = 2 * a[idx[mask]]
            # step size decays
            step = step0 / np.sqrt(it + 1)

            # check time
            if time.time() >= deadline:

    # final best sequence may be list of floats

def conv_fft(a):
    n = len(a)
    size = 1 << ((2*n - 1).bit_length())
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:2*n - 1]
    return conv

def local_refine(seq, max_iters=5000, time_deadline=None):
    a = seq.copy()
    n = len(a)
    best_a = a.copy()
    best_sc = score(a)
    step0 = 0.5
    it = 0
    while it < max_iters:
        if time_deadline is not None and time.time() > time_deadline:
            break
        conv = conv_fft(a)
        max_idx = int(np.argmax(conv))
        grad = np.zeros_like(a)
        idx = max_idx - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad[mask] = 2 * a[idx[mask]]
        step = step0 / np.sqrt(it + 1)
        a_new = a - step * grad
        a_new = simplex_projection(a_new)
        sc = score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        # Accept improvement
        if sc < score(a):
            a = a_new
        else:
            # maybe random perturb occasionally to escape
            if np.random.rand() < 0.02:
                a = a_new
        it += 1
        # occasional random jitter decreasing over time
        if it % 200 == 0:
            eps = step0 * 0.01 / np.sqrt(it)
            a = a + eps * np.random.randn(n)
            a = np.maximum(a, 0)
            a = simplex_projection(a)
    return best_a

def conv_fft(a):
    n = a.shape[0]
    size = 1 << ((2*n-1).bit_length())
    # Use rfft for real numbers
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:2*n-1]
    return conv.real
```

Note the product of transforms is the convolution of a with a (circular? Actually zero padding ensures linear convolution). This yields correct.

```
def simplex_projection(v, z=1.0):
    # from https://github.com/MLDawn/Utility/blob/master/simplex_projection.py
    if np.sum(v) == z and np.all(v >= 0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    rho = np.where(u - cssv / (np.arange(1, n+1)) > 0)[0]
    if rho.size == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

Now implement score:

```
def score(seq):
    a = np.array(seq, dtype=float)
    n = len(a)
    # ensure sum > 0 ; if sum < 1e-12 return inf
    s = a.sum()
    if s < 1e-8:
        return float('inf')
    conv = conv_fft(a)
    max_c = conv.max()
    return 2.0 * n * max_c / (s * s)
```

"""
Search for a step‑function (non‑negative sequence) that minimises

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The basic idea is a two‑stage stochastic search:

1. **Parametric exploration** – generate many candidate sequences from a few
   simple families (exponential decay, power‑law decay, random Dirichlet,
   symmetric “U” shapes).  The length `n` is sampled between 200 and 1500.
   Each candidate is normalised to unit sum (the objective is scale‑invariant).

2. **Local refinement** – from each candidate run a fast projected gradient
   descent that tries to reduce the current maximum convolution entry.
   The gradient of the (non‑smooth) max can be approximated by the gradient
   at the active index (the index where the convolution attains its maximum).
   After each step we project back onto the probability simplex to keep
   the coefficients non‑negative and the sum equal to 1.

Both stages are repeated until the time budget (≈990 s) is exhausted.
The best sequence found is returned as a plain Python list of floats.
"""

import numpy as np
import time


def _simplex_projection(v: np.ndarray, z: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x >= 0, sum(x)=z}."""
    if np.sum(v) == z and np.all(v >= 0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a using FFT (real inputs)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())   # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return conv.real


def _score(a: np.ndarray) -> float:
    """Objective value for a (sum(a) is assumed > 0)."""
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    conv = _conv_fft(a)
    return 2.0 * a.shape[0] * conv.max() / (s * s)


def _local_refine(a0: np.ndarray, max_iters: int, deadline: float) -> np.ndarray:
    """
    Simple projected gradient descent that reduces the current max
    convolution entry.  The time budget is respected via `deadline`.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _score(a)

    step0 = 0.5
    for it in range(max_iters):
        if time.time() >= deadline:
            break

        conv = _conv_fft(a)
        kmax = int(np.argmax(conv))

        # gradient of conv[kmax] w.r.t a : 2 * a_{kmax - i}
        idx = kmax - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]

        step = step0 / np.sqrt(it + 1.0)
        a_candidate = a - step * grad
        a_candidate = _simplex_projection(a_candidate)

        val = _score(a_candidate)
        if val < best_val:
            best_val = val
            best_a = a_candidate.copy()

        # accept only if we improve (otherwise keep current a)
        if val < _score(a):
            a = a_candidate

        # occasional jitter to escape shallow plateaus
        if it % 200 == 0 and it > 0:
            jitter = (step0 * 0.01) / np.sqrt(it + 1.0)
            a = a + jitter * np.random.randn(n)
            a = np.maximum(a, 0.0)
            a = _simplex_projection(a)

    return best_a


def _candidate_from_params(n: int, kind: str, param) -> np.ndarray:
    """Generate a normalised candidate sequence of length n."""
    if kind == "exp_decay":
        # a_i = exp(-alpha * i)
        alpha = param
        a = np.exp(-alpha * np.arange(n, dtype=float))
    elif kind == "exp_sym":
        # symmetric decay from both ends: a_i = exp(-alpha * min(i, n-1-i))
        alpha = param
        d = np.minimum(np.arange(n), n - 1 - np.arange(n))
        a = np.exp(-alpha * d.astype(float))
    elif kind == "pow_decay":
        # a_i = (i+1)^{-beta}
        beta = param
        a = 1.0 / (np.arange(1, n + 1, dtype=float) ** beta)
    elif kind == "pow_sym":
        # a_i = (min(i, n-1-i)+1)^{beta}
        beta = param
        d = np.minimum(np.arange(n), n - 1 - np.arange(n)) + 1.0
        a = d ** beta
    elif kind == "random":
        a = np.random.rand(n)
    else:                     # uniform fallback
        a = np.ones(n, dtype=float)

    a = a / a.sum()
    return a


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Main entry point used by the evaluation harness.
    Returns a list of non‑negative floats (the best sequence found before the
    time budget expires).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # keep a small safety margin

    # ------------------------------------------------------------------
    # start with a safe uniform baseline (score = 2.0)
    # ------------------------------------------------------------------
    n0 = 500
    best_seq = np.full(n0, 1.0 / n0)
    best_score = _score(best_seq)

    # ------------------------------------------------------------------
    # optional: try to improve the known SOTA sequence (if it exists)
    # ------------------------------------------------------------------
    try:
        import __main__
        if hasattr(__main__, "height_sequence_1"):
            seq0 = np.array(__main__.height_sequence_1, dtype=float)
            seq0 = seq0 / seq0.sum()
            sc0 = _score(seq0)
            if sc0 < best_score:
                best_score, best_seq = sc0, seq0.copy()
            # a quick local refinement of the SOTA sequence
            refined = _local_refine(seq0, max_iters=2500, deadline=deadline)
            sc_ref = _score(refined)
            if sc_ref < best_score:
                best_score, best_seq = sc_ref, refined.copy()
    except Exception:
        pass

    # ------------------------------------------------------------------
    # stochastic search over parametric families + local refinement
    # ------------------------------------------------------------------
    families = ["exp_decay", "exp_sym", "pow_decay", "pow_sym", "random"]
    while time.time() < deadline:
        n = np.random.randint(200, 1500)          # length of candidate
        kind = np.random.choice(families)

        # draw a sensible parameter for the selected family
        if kind in ("exp_decay", "exp_sym"):
            param = 10 ** np.random.uniform(-3, 2)   # alpha in [1e-3, 1e2]
        elif kind in ("pow_decay",):
            param = np.random.uniform(0.1, 4.0)       # beta > 0
        elif kind in ("pow_sym",):
            param = np.random.uniform(-2.0, 2.0)      # can be negative
        else:
            param = None

        cand = _candidate_from_params(n, kind, param)
        sc = _score(cand)
        if sc < best_score:
            best_score, best_seq = sc, cand.copy()
            print(f"[{time.time():.2f}] New best from {kind} (p={param:.4g}) "
                  f"n={n} score={sc:.6f}")

        # local refinement (few hundred iterations – cheap)
        refined = _local_refine(cand, max_iters=1500, deadline=deadline)
        sc_ref = _score(refined)
        if sc_ref < best_score:
            best_score, best_seq = sc_ref, refined.copy()
            print(f"[{time.time():.2f}] New best after refine n={n} "
                  f"score={sc_ref:.6f}")

        # occasional aggressive refinement of the current global best
        if np.random.rand() < 0.07:
            aggressive = _local_refine(best_seq, max_iters=3000, deadline=deadline)
            sc_agg = _score(aggressive)
            if sc_agg < best_score:
                best_score, best_seq = sc_agg, aggressive.copy()
                print(f"[{time.time():.2f}] Aggressive refine improved score={sc_agg:.6f}")

    # ------------------------------------------------------------------
    # final sanity – ensure the returned list is non‑empty and non‑negative
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    if best_seq.size == 0:
        best_seq = np.full(10, 0.1)   # fallback trivial sequence

    print(f"Search finished. Best score ≈ {best_score:.6f} with length {len(best_seq)}")
    return best_seq.tolist()
