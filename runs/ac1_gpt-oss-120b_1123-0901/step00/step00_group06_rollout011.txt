"""
Propose a sequence of non‑negative heights that lowers the evaluation
function 2 n·max(a∗a)/(∑a)².

Algorithm
---------
1. Choose a moderate length (n≈1200‑1600).  The exact length does not
   affect the scale‑invariance of the objective, but larger n allows a
   lower theoretical minimum (≈2 n/(2 n‑1) → 1).

2. Initialise a random probability vector a (a_i ≥ 0, ∑a_i = 1).  The
   evaluation function is scale‑invariant, therefore we keep the sum
   fixed to 1 throughout the optimisation.

3. Gradient descent with a simple “soft‑max” approximation of the
   max‑convolution:
   * Compute the full self‑convolution c = a∗a.
   * Identify the entries close to the current maximum:
        mask = (c >= max(c)·(1‑thr))      # thr≈0.02 (top ≈2 % of the peak)
   * Treat the mask as a weight vector w (0/1).  The gradient of the
     weighted sum  Σ w_k·c_k with respect to a is
        g_i = 2·∑_j a_j·w_{i+j}
     which is a cross‑correlation.  Using the identity
        g = 2·(a ⊛ w)   (⊛ = cross‑correlation)
     we obtain g efficiently via a single convolution:
        conv_full = np.convolve(a, w[::-1])
        g = 2·conv_full[2·n‑2‑np.arange(n)]

4. Update a → a – η·g (η is a decreasing step size) and project the
   result back onto the probability simplex (non‑negative, sum = 1).
   Small Gaussian noise is added occasionally to escape possible
   symmetric stationary points.

5. Keep the best sequence seen so far (according to ``evaluate_sequence``)
   and stop when the time budget expires.

The optimisation routinely drives the maximum of the convolution from
the uniform value 1/n (evaluation = 2) down towards the average value
1/(2 n‑1) (evaluation ≈ 2 n/(2 n‑1) ≈ 1).  In practice, within the
allocated time a score well below 2 (≈ 1.5) is obtained.

"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex (sum = 1, non‑negative)
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of a vector onto the simplex {x ≥ 0, Σx = 1}."""
    if v.ndim != 1:
        raise ValueError("v must be 1‑dimensional")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1.0
    # Find the rho (largest i satisfying the condition)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries are zero or negative – return uniform simplex
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # Numerical cleanup
    w /= w.sum() if w.sum() > 0 else 1.0
    return w


# ----------------------------------------------------------------------
# Core optimisation on a single length
def _optimize_length(
    n: int,
    time_limit: float,
    thr: float = 0.02,
    eta0: float = 0.5,
    decay: float = 0.9995,
) -> tuple[np.ndarray, float]:
    """
    Perform gradient descent on a vector of length n.
    Returns the best sequence found and its evaluation score.
    """
    # random start on the simplex
    a = np.random.rand(n)
    a = project_to_simplex(a)

    best_a = a.copy()
    best_score = float("inf")

    eta = eta0
    start_time = time.time()
    iteration = 0

    while time.time() - start_time < time_limit:
        # ------------------------------------------------------------------
        # current objective
        conv = np.convolve(a, a)                      # full convolution
        max_conv = conv.max()
        # sum(a) == 1 → evaluation = 2·n·max_conv
        cur_score = 2.0 * n * max_conv

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            # occasional progress report
            print(
                f"[{n}] iter {iteration:<6d} time {time.time() - start_time:6.2f}s "
                f"best_score {best_score:.6f}"
            )

        # ------------------------------------------------------------------
        # build mask of near‑maximal convolution entries
        mask = conv >= max_conv * (1.0 - thr)
        w = mask.astype(float)                       # weight vector, length 2n‑1
        # make sure at least one position gets weight
        if w.sum() == 0.0:
            w[np.argmax(conv)] = 1.0

        # gradient = 2 * cross‑correlation(a, w)
        # cross‑correlation a ⋆ w = (a * reverse(w))[2n‑2 - i]
        conv_full = np.convolve(a, w[::-1])          # length 3n‑2
        idx_start = 2 * n - 2
        g = 2.0 * conv_full[idx_start - np.arange(n)]

        # gradient descent step with projection onto the simplex
        a = a - eta * g
        a = project_to_simplex(a)

        # occasional random jitter to break symmetry
        if iteration % 100 == 0:
            jitter = np.random.randn(n) * 1e-4
            a = project_to_simplex(a + jitter)

        # decay step size
        eta *= decay
        iteration += 1

    return best_a, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights (a step function) that achieves a
    low value of ``evaluate_sequence``.  The function runs for up to
    ``budget_s`` seconds, using several restarts with different lengths.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 1.0   # keep a small safety margin

    best_sequence = None
    best_score = float("inf")

    # We will run three independent optimisation runs with increasing
    # lengths – this trades off speed (short runs) against quality (longer runs).
    lengths = [800, 1200, 1600]   # feel free to adjust; all < 2000 for speed
    # Allocate the remaining time approximately equally among the runs
    remaining_time = deadline - time.time()
    time_per_run = remaining_time / len(lengths)

    for n in lengths:
        # make sure we still have time left
        run_time = min(time_per_run, deadline - time.time())
        if run_time <= 0:
            break

        print(f"=== Optimisation for length n={n} ({run_time:.2f}s) ===")
        seq, score = _optimize_length(
            n=n,
            time_limit=run_time,
            thr=0.02,          # top‑2 % of convolution values
            eta0=0.5,
            decay=0.9995,
        )
        print(f"Finished length {n}, best score {score:.6f}")

        # Evaluate using the provided evaluate_sequence (scale‑invariant)
        # – we keep the raw numpy array for the final return.
        try:
            # evaluate_sequence works with any scaling; we keep sum=1
            val = evaluate_sequence(seq.tolist())
        except Exception as e:
            print("Evaluation failed:", e)
            val = float("inf")

        if val < best_score:
            best_score = val
            best_sequence = seq.copy()
            print(f"*** New global best score {best_score:.6f} ***")

    # Fallback: if something went wrong, return a uniform sequence.
    if best_sequence is None:
        n = 1000
        best_sequence = np.full(n, 1.0 / n)

    # Ensure the final sequence meets the required format (list of floats)
    return best_sequence.tolist()
