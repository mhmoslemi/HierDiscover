def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Gradient‑descent search for a step‑function sequence with a low
evaluate_sequence() value.

The algorithm works on a probability simplex (sum of coefficients = 1,
all non‑negative) because the evaluation function is homogeneous:
scaling the whole sequence does not change the score.
For a fixed length n the score simplifies to

    score = 2 * n * max(conv(a, a))

where conv(a,a) is the self‑convolution.  To minimise the score we try
to flatten the convolution, i.e. make its largest entry as close as
possible to the average value 1/(2n‑1).  This is done by a simple
projected gradient descent on the index `kmax` where the current
convolution reaches its maximum.

The gradient of the objective w.r.t. a_i (when kmax is unique) is

    g_i = 2 * a_{kmax-i}          (if 0 ≤ kmax-i < n, else 0)

We take a step in the opposite direction, project back onto the
simplex (clip negatives and renormalise), and repeat.  Momentum,
a decaying step‑size and occasional random perturbations help escape
local minima.  Several random restarts with different lengths are
performed until the allotted time (≈ 1000 s) expires; the best sequence
found overall is returned.

The convolution is computed by an FFT (zero‑padded to the next power
of two) which is much faster than the naïve O(n²) method for the
typical lengths (800‑1500) we use.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: project a vector onto the probability simplex {x>=0, sum(x)=1}
# Simple and fast: clip negatives then renormalise (exact enough for our
# optimisation because the step direction already points towards higher
# sum).
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Clip to non‑negative and renormalise to sum 1."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0.0:
        # fallback – uniform distribution (any point on the simplex)
        return np.full_like(v, 1.0 / v.size)
    return v / s


# ----------------------------------------------------------------------
def _conv_self_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Full (non‑circular) convolution a*a using FFT.
    `fft_len` must be a power of two ≥ 2*len(a)-1.
    """
    A = np.fft.rfft(a, n=fft_len)
    B = A * A
    conv = np.fft.irfft(B, n=fft_len)[: 2 * a.size - 1]
    return conv

    Parameters

    seed : int
        Random seed for reproducibility.

    list[float]
        The best sequence discovered within the time budget.
    """

    deadline = time.time() + budget_s - 5   # stay a bit safe

    best_sequence = None
    best_score = float("inf")

    min_len = 800      # minimal length of a run
    max_len = 1500     # maximal length of a run
    init_eta = 0.25    # initial step size
    eta_decay = 1e-4   # decay per iteration
    momentum_beta = 0.9

    jitter_scale = 0.02     # std. dev. of jitter (relative to current values)

    # Main outer loop – random restarts until time runs out

    while time.time() < deadline:

        n = np.random.randint(min_len, max_len + 1)

        if have_known and np.random.rand() < 0.2:

            if known.size > n:

                a = np.concatenate([known, pad])
            a = _project_simplex(a)
        else:
            # Random point on the simplex
            a = np.random.dirichlet(np.ones(n)).astype(np.float64)

        # Pre‑compute the FFT length (next power of two)
        fft_len = 1 << ((2 * n - 1).bit_length())

        # Gradient‑descent state

        while time.time() < deadline:

            a = _project_simplex(a)

            it += 1
            eta = init_eta / (1.0 + eta_decay * it)

        # (The outer loop will also check the global time budget.)

    # If the time budget expires, just return the best sequence found.
    if best_sequence is None:

        best_sequence = np.full(min_len, 1.0 / min_len)
    return list(best_sequence)
