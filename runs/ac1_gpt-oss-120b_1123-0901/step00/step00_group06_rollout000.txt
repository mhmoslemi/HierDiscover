"""
Propose a step‑function sequence that minimizes the evaluation function.

The algorithm is a refined version of the published LP‑based approach:

1. **Normalization** – every candidate is scaled so that its sum equals
   √(2·n).  With this normalization the evaluation reduces to
   `max(conv(a,a))`, i.e. the maximal entry of the self‑convolution.

2. **LP direction** – for the current normalized sequence `f` we solve the
   linear program

        maximise   Σ b_j
        subject to  (f * b)_k ≤ M   for all k
                    b_j ≥ 0

   where `M = max(conv(f,f))`.  The optimum provides a feasible partner
   `g` that has the same ℓ₁‑norm as `f` but a (weakly) larger sum before
   the ℓ₁‑renormalization.

3. **Optimal mixing** – the new sequence is a convex combination
   `h(t) = (1‑t)·f + t·g` (0 ≤ t ≤ 1).  Because both `f` and `g` satisfy the
   same convolution bound, `max(conv(h,h)) ≤ M` for any `t`.  The optimal
   `t` is searched with a cheap 1‑D line search that evaluates the exact
   convolution
   `conv(h,h) = (1‑t)²·conv(f,f) + 2t(1‑t)·conv(f,g) + t²·conv(g,g)`.

   The value of `t` giving the smallest maximal entry is used.

4. **Iterative improvement & random restarts** – the above steps are
   repeated until the time budget expires.  If no improvement is observed
   for a while, a new random sequence (different length) is generated to
   escape possible local minima.

The whole procedure runs inside the allotted 1000 s and returns the best
sequence found.  All helper functions are defined at module level as
required.
"""

from scipy.optimize import linprog


def _normalize(seq):
    """Scale a non‑negative sequence so that sum = sqrt(2·len(seq))."""
    arr = np.asarray(seq, dtype=np.float64)
    arr = np.maximum(arr, 0.0)
    n = len(arr)
    target = np.sqrt(2.0 * n)
    total = arr.sum()
    if total == 0.0:
        arr = np.full(n, target / n, dtype=np.float64)
    else:
        arr = arr * (target / total)
    return arr


def _solve_lp(f_seq, rhs):
    """
    Solve   maximise Σ b_j   subject to conv(f,b) ≤ rhs , b ≥ 0.
    Returns the optimal b as a numpy array or None on failure.
    """
    n = len(f_seq)
    c = -np.ones(n)                     # maximise sum → minimise -sum
    # build convolution constraints
    a_ub = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        for i in range(i_min, i_max + 1):
            j = k - i
            a_ub[k, j] = f_seq[i]
    # non‑negativity constraints  –b ≤ 0
    a_ub = np.vstack([a_ub, -np.eye(n, dtype=np.float64)])
    b_ub = np.hstack([np.full(2 * n - 1, rhs, dtype=np.float64),
                      np.zeros(n, dtype=np.float64)])

    # use the HiGHS interior‑point solver; limit the running time a bit
    res = linprog(
        c,
        A_ub=a_ub,
        b_ub=b_ub,
        bounds=(0, None),
        method='highs',
        options={'time_limit': 5.0, 'disp': False},
    )
    if res.success:
        # Numerical noise can produce tiny negatives – clip them
        return np.maximum(res.x, 0.0)
    return None


def _best_t(b_f, b_g, c_fg):
    """
    One‑dimensional search for the mixing coefficient t ∈ [0,1] that
    minimises max_k ((1‑t)²·b_f[k] + 2t(1‑t)·c_fg[k] + t²·b_g[k]).
    """
    # coarse grid
    t_grid = np.linspace(0.0, 1.0, 101)
    values = np.empty_like(t_grid)
    for i, t in enumerate(t_grid):
        cur = (1.0 - t) ** 2 * b_f + 2.0 * t * (1.0 - t) * c_fg + t ** 2 * b_g
        values[i] = cur.max()
    best_idx = int(values.argmin())
    t_best = t_grid[best_idx]

    # local refinement using a few golden‑section steps
    left = max(t_best - 0.01, 0.0)
    right = min(t_best + 0.01, 1.0)
    for _ in range(12):
        t1 = left + (right - left) * 0.38196601125  # 1-φ
        t2 = left + (right - left) * 0.61803398875  # φ
        v1 = ((1.0 - t1) ** 2 * b_f + 2.0 * t1 * (1.0 - t1) * c_fg + t1 ** 2 * b_g).max()
        v2 = ((1.0 - t2) ** 2 * b_f + 2.0 * t2 * (1.0 - t2) * c_fg + t2 ** 2 * b_g).max()
        if v1 < v2:
            right = t2
        else:
            left = t1
    return (left + right) * 0.5


def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """Return a sequence of non‑negative heights with a low evaluation score."""
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # small safety margin

    # ------------------------------------------------------------------
    # initialise – try the provided SOTA construction first
    # ------------------------------------------------------------------
    try:
        init_seq = list(height_sequence_1)   # may be defined globally
    except Exception:
        init_seq = None

    if init_seq is None:
        # random initial sequence (length between 300 and 1500)
        n0 = np.random.randint(300, 1500)
        init_seq = np.random.rand(n0).tolist()

    curr_seq = _normalize(init_seq)
    best_seq = curr_seq.copy()
    best_score = evaluate_sequence(best_seq.tolist())

    # counters for restart logic
    no_improve_streak = 0
    it = 0

    while time.time() < deadline:
        it += 1

        # occasionally perform a random restart to avoid being stuck
        if no_improve_streak >= 40 and np.random.rand() < 0.3:
            n_new = np.random.randint(300, 2000)
            rand_seq = np.random.rand(n_new).tolist()
            curr_seq = _normalize(rand_seq)
            no_improve_streak = 0
            continue

        # ----- main LP‑based improvement step -----
        n = len(curr_seq)
        f = _normalize(curr_seq)                     # ensure exact normalisation
        b_f = np.convolve(f, f)
        M = b_f.max()

        # solve the LP for a partner sequence g
        g_raw = _solve_lp(f, M)
        if g_raw is None:
            # LP failed – try a tiny random perturbation instead
            idx = np.random.randint(n)
            pert = f.copy()
            pert[idx] = max(0.0, pert[idx] + np.random.randn() * 0.005)
            curr_seq = pert
            continue

        # normalise g to the same ℓ₁‑norm as f
        g = _normalize(g_raw)

        # pre‑compute needed convolutions
        b_g = np.convolve(g, g)
        c_fg = np.convolve(f, g)

        # find the best mixing coefficient
        t_opt = _best_t(b_f, b_g, c_fg)

        # new candidate = convex combination
        new_seq = (1.0 - t_opt) * f + t_opt * g
        new_seq = _normalize(new_seq)   # re‑normalise to avoid drift

        # evaluate
        try:
            new_score = evaluate_sequence(new_seq.tolist())
        except Exception:
            new_score = np.inf

        # keep the best seen so far
        if new_score < best_score - 1e-12:
            best_score = new_score
            best_seq = new_seq.tolist()
            no_improve_streak = 0
            print(f"Iter {it:04d}: improved score = {best_score:.6f} (t={t_opt:.4f})")
        else:
            no_improve_streak += 1

        # continue from the new point (even if not strictly better)
        curr_seq = new_seq.tolist()

    print(f"Search finished after {it} iterations. Best score = {best_score:.6f}")
    return best_seq
