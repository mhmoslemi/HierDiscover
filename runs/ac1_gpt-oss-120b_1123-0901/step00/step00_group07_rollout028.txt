def propose_candidate(seed=42, budget_s=1000, **kwargs):
    import numpy as np, time
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5

    # Determine length
    n = len(height_sequence_1) if 'height_sequence_1' in globals() else 3000
    # Initialize uniform
    a = np.full(n, 1.0/n)

    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f'Initial score: {best_score}')

    # parameters
    base_lr = 0.5
    alpha = 5.0
    alpha_inc = 0.02

    it = 0
    while time.time() < deadline:
        # compute M
        b = np.convolve(a, a)
        M = np.max(b)

        # compute softmax weights
        maxb = M
        w = np.exp(alpha * (b - maxb))
        w_sum = np.sum(w)
        w = w / w_sum

        # compute shift via convolution
        shift = np.convolve(a[::-1], w)[n-1:2*n-1]   # slice size n
        grad = shift + a   # shape n

        # attempt update
        lr = base_lr / np.sqrt(it+1)
        improved = False
        for _ in range(5):
            a_new = a - lr * grad
            a_new = np.clip(a_new, 0, None)
            s = np.sum(a_new)
            if s == 0:
                a_new = np.full(n, 1.0/n)
                s = 1.0
            a_new /= s

            M_new = np.max(np.convolve(a_new, a_new))
            if M_new < M:
                a = a_new
                M = M_new
                improved = True
                break
            else:
                lr *= 0.5
        if not improved:
            # occasionally random perturbation to escape
            idx = np.random.randint(n)
            a[idx] = max(0.0, a[idx] + np.random.randn() * 0.001)
            a /= np.sum(a)

        # evaluate occasionally
        if it % 20 == 0:
            score = evaluate_sequence(a.tolist())
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f'Iter {it}: improved score {best_score:.6f}')
        # increase alpha slowly
        alpha = min(alpha + alpha_inc, 1000.0)
        it += 1

    return best_seq.tolist()

import numpy as np
import time

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """Projected gradient descent on a smooth max-convolution objective.
    The algorithm maintains a probability vector `a` (sum=1, a_i>=0)
    and iteratively reduces the maximum entry of its self-convolution.
    A soft‑max (controlled by `alpha`) provides a differentiable
    approximation of the max, and a simple projected gradient step
    with line‑search guarantees non‑increase of the true objective.
    Random perturbations and a diminishing learning‑rate help escape
    local minima. """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5

    # Determine length of the step function.
    # Use the existing construction if available, otherwise 3000.
    n = globals().get('height_sequence_1')
    if n is not None:
        n = len(n)
    else:
        n = 3000

    # Initialise a random probability vector (Dirichlet)
    a = np.random.dirichlet(np.ones(n))
    # Keep the best candidate found.
    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f'Initial score: {best_score:.6f}')

    # Hyper‑parameters
    base_lr = 0.8          # initial learning‑rate scale
    alpha = 5.0            # start with a smooth max
    alpha_inc = 0.02       # slowly increase the temperature
    max_alpha = 200.0
    it = 0

    while time.time() < deadline:
        # current convolution and its maximum
        b = np.convolve(a, a)
        M = np.max(b)

        # soft‑max weights over the convolution entries
        # (subtracting the maximum for numerical stability)
        w = np.exp(alpha * (b - M))
        w_sum = np.sum(w)
        w = w / w_sum

        # shift term = sum_i a_i * w_{i+j} for each j
        # This can be obtained via one convolution of reversed a with w
        shift = np.convolve(a[::-1], w)[n - 1: 2 * n - 1]   # length n

        # gradient of the smooth max w.r.t. a (before projection)
        grad = shift + a   # because Σ w_k = 1

        # learning‑rate that decays with the number of iterations
        lr = base_lr / np.sqrt(it + 1)

        # simple back‑tracking line‑search: accept the first step that reduces M
        improved = False
        for _ in range(8):
            a_new = a - lr * grad
            a_new = np.clip(a_new, 0.0, None)
            s = a_new.sum()
            if s == 0.0:
                # fallback to uniform – should be very unlikely
                a_new = np.full(n, 1.0 / n)
                s = 1.0
            a_new /= s

            M_new = np.max(np.convolve(a_new, a_new))
            if M_new < M:
                a = a_new
                M = M_new
                improved = True
                break
            lr *= 0.5   # reduce step and try again

        if not improved:
            # occasional random jitter to escape flat regions
            if it % 20 == 0:
                jitter = np.random.randn(n) * 1e-4
                a = np.clip(a + jitter, 0.0, None)
                a /= a.sum()

        # Periodically evaluate the true objective and keep the best sequence.
        if it % 10 == 0:
            score = evaluate_sequence(a.tolist())
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f'Iter {it:5d}: improved score = {best_score:.6f}  max_conv = {M:.8f}')

        # slowly increase the temperature to sharpen the approximation
        alpha = min(alpha + alpha_inc, max_alpha)

        it += 1

    print(f'Finished after {it} iterations. Best score = {best_score:.6f}')
    return best_seq.tolist()

                break
            lr *= 0.5                         # shrink step if it didn't improve

        if not improved:
            # Inject a tiny random perturbation every few iterations to escape flat regions.
            if it % 20 == 0:
                jitter = np.random.randn(n) * 1e-4
                a = np.clip(a + jitter, 0.0, None)
                a /= a.sum()

        # Periodically evaluate the true objective (expensive) and keep the best.
        if it % 10 == 0:
            score = evaluate_sequence(a.tolist())
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                no_improve = 0
                print(f'Iter {it:5d}: improved score = {best_score:.6f}  max_conv = {M:.8f}')
            else:
                no_improve += 1

        # If we have gone a long time without improvement, restart from a fresh random point.
        if no_improve >= 500:
            a = np.random.dirichlet(np.ones(n))
            no_improve = 0
            print(f'Iter {it:5d}: restart due to stagnation')

        # Gradually increase the temperature so that the surrogate gets closer to the true max.
        alpha = min(alpha + alpha_inc, max_alpha)

        it += 1

    print(f'Finished after {it} iterations. Best score = {best_score:.6f}')
    return best_seq.tolist()
