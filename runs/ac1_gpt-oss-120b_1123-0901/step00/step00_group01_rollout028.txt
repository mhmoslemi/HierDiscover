def propose_candidate(seed=42, budget_s=1000, **kwargs):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5

    try:
        n = len(height_sequence_1)
    except NameError:
        n = 2000

    best_seq = None
    best_score = float('inf')
    # parameter search
    for p in np.linspace(0.4, 0.6, 41):
        for shift in np.linspace(0.5, 5.0, 19):
            # generate sequence
            idx = np.arange(n)
            seq = (idx + shift) ** (-p)
            # normalize
            total = seq.sum()
            if total < 1e-12:
                continue
            seq = seq / total
            score = evaluate_sequence(seq.tolist())
            if score < best_score:
                best_score = score
                best_seq = seq.tolist()
                print(f"New best: p={p:.3f}, shift={shift:.3f}, score={score:.6f}")
            # early exit if time
            if time.time() > deadline:
                break
        if time.time() > deadline:
            break

    # optional refinement via random perturbations
    # use simple simulated annealing around best_seq
    curr_seq = np.array(best_seq)
    curr_score = best_score
    t0 = 0.5
    while time.time() < deadline:
        # propose perturbation: choose i,j random, move epsilon
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        eps = np.random.randn() * 0.01 * curr_seq[i]
        new_seq = curr_seq.copy()
        new_seq[i] = max(0.0, new_seq[i] - eps)
        new_seq[j] = max(0.0, new_seq[j] + eps)
        # renormalize
        s = new_seq.sum()
        if s == 0:
            continue
        new_seq = new_seq / s
        score = evaluate_sequence(new_seq.tolist())
        # accept if better or with probability
        if score < curr_score or (np.random.rand() < np.exp((curr_score - score) / (t0+1e-8))):
            curr_seq = new_seq
            curr_score = score
            if score < best_score:
                best_score = score
                best_seq = new_seq.tolist()
                print(f"Improved via perturb: score={best_score:.6f}")
        # cooling
        t0 *= 0.9999

    return best_seq

    total = raw.sum()
    if total <= 0.0:

    return evaluate_sequence(seq)

# ----------------------------------------------------------------------
def _make_power_law(n, p, offset):
    """Return a normalised numpy array of length n."""
    idx = np.arange(n, dtype=float)
    # avoid negative or zero exponent issues
    seq = (idx + offset) ** (-p)
    s = seq.sum()
    if s == 0.0:
        return None
    return seq / s

# ----------------------------------------------------------------------
def _refine_sa(seq, budget, initial_temp=0.5):
    """Return a (possibly) improved sequence using simple SA."""
    n = len(seq)
    best_seq = seq.copy()
    best_score = _score_from_raw(best_seq)
    cur_seq = best_seq.copy()
    cur_score = best_score
    t = initial_temp
    deadline = time.time() + budget
    while time.time() < deadline:
        # pick two distinct indices
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # propose moving a small fraction of the mass at i to j
        delta = cur_seq[i] * np.random.uniform(0.0, 0.01)   # ≤1 % of a_i

        new_seq[i] = max(0.0, new_seq[i] - delta)
        new_seq[j] = new_seq[j] + delta
        # re‑normalise (preserves sum)
        s = new_seq.sum()
        if s == 0.0:

        new_seq /= s
        score = _score_from_raw(new_seq)

        # accept if better, otherwise with Metropolis probability
        accept = False
        if score < cur_score:
            accept = True
        else:
            # probability ~ exp(-(score‑cur_score)/t)
            if np.random.rand() < np.exp(-(score - cur_score) / (t + 1e-12)):

            cur_score = score

                best_seq = new_seq.copy()
                best_score = score
                print(f"[SA] New best score: {best_score:.6f}")

        # exponential cooling
        t *= 0.9999
    return best_seq
