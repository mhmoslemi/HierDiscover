def propose_candidate(seed=0, budget_s=1000):
    start_time = time.time()
    deadline = start_time + budget_s - 0.1
    best_seq = None
    best_score = float('inf')
    np.random.seed(seed)
    while time.time() < deadline:
        # Choose random N (maybe 200-1000)
        N = np.random.randint(150, 800) # small for speed
        # Random initial simplex
        a = np.random.rand(N)
        a = a / a.sum()
        # gradient descent
        step = 0.5
        for it in range(2000):
            # compute conv
            conv = np.convolve(a,a)
            k = np.argmax(conv)
            max_b = conv[k]
            score = 2 * N * max_b
            if score < best_score - 1e-12:
                best_score = score
                best_seq = a.copy()
                # optional print
                print(f"New best score {best_score:.6f} N={N} iter={it}")
            # compute gradient
            # g_i = 2 * a_{k - i}
            g = np.zeros_like(a)
            for i in range(N):
                j = k - i
                if 0 <= j < N:
                    g[i] = 2 * a[j]
                else:
                    g[i] = 0.0
            # Update
            a_new = a - step * g
            # Project onto simplex
            a = simplex_projection(a_new)
            # Decrease step gradually
            step *= 0.9995
            # optionally break if step tiny
        # maybe random perturb after gradient descent
    # after time, return best_seq, maybe convert to list, scaling not required
    return best_seq.tolist()

def propose_candidate(seed=42, budget_s=1000):
    import time, numpy as np
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5
    best_seq = None
    best_score = float('inf')
    # main loop of restarts
    while time.time() < deadline:
        N = np.random.randint(80, 400)  # choose moderate length
        # initialize a
        # start with arcsine distribution as initial guess
        x = np.arange(N) + 0.5
        a = 1.0/ np.sqrt(x*(N+1 - x))
        a = a / a.sum()
        # Optionally random perturb some entries
        a += 0.01 * np.random.rand(N)
        a = np.maximum(a, 0)
        a = a / a.sum()
        # subgradient descent
        step0 = 0.5
        for it in range(2000):
            # compute conv and score
            conv = np.convolve(a, a)
            max_idx = np.argmax(conv)
            max_val = conv[max_idx]
            score = 2 * N * max_val
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                # logging
                print(f"New best {best_score:.6f} N={N} it={it}")
                # maybe break if good enough
                if best_score < 1.503:
                    # can stop early? but keep exploring
                    pass
            # compute gradient for max index
            # gradient g_i = 2 * a[max_idx - i] if valid
            indices = np.arange(N)
            j = max_idx - indices
            mask = (j >= 0) & (j < N)
            g = np.zeros_like(a)
            g[mask] = 2 * a[j[mask]]
            # step schedule
            eta = step0 / np.sqrt(it + 1)
            # exponential update
            a = a * np.exp(-eta * g)
            # re-normalize to sum 1
            sum_a = a.sum()
            if sum_a == 0:
                # reinit
                a = np.random.rand(N)
                a = a / a.sum()
                continue
            a = a / sum_a
        # optional random perturbation of best to new start
    # final: return best_seq as list
    return best_seq.tolist()

def propose_candidate(seed=0, budget_s=1000):
    import time, numpy as np, random
    np.random.seed(seed)
    random.seed(seed)
    deadline = time.time() + budget_s - 0.1
    best_seq = None
    best_score = float('inf')
    # parameters
    while time.time() < deadline:
        N = np.random.randint(100, 600)  # moderate length
        # initialize with random Dirichlet distribution
        a = np.random.rand(N)
        a = a / np.sum(a)
        cur_score = 2 * N * np.max(np.convolve(a, a))
        cur_seq = a
        # inner loop for local improvements
        T0 = 0.1
        decay = 0.9999
        t_start = time.time()
        it = 0
        while time.time() < deadline and it < 2000:
            # propose perturbation
            i, j = np.random.randint(N, size=2)
            if i == j:
                continue
            delta = (np.random.rand() - 0.5) * 0.01 # small
            # move delta from i to j; ensure positivity
            if cur_seq[i] - delta < 0 or cur_seq[j] + delta < 0:
                continue
            new_seq = cur_seq.copy()
            new_seq[i] -= delta
            new_seq[j] += delta
            # renormalize to sum=1 (sum unchanged, but due to floating errors ensure)
            # compute score
            conv = np.convolve(new_seq, new_seq)
            new_score = 2 * N * np.max(conv)
            # acceptance
            if new_score < cur_score or np.random.rand() < np.exp(-(new_score - cur_score) / T0):
                cur_seq = new_seq
                cur_score = new_score
                if new_score < best_score:
                    best_score = new_score
                    best_seq = cur_seq.copy()
                    print(f"New best {best_score:.6f} N={N} it={it}")
            # temperature decay
            T0 *= decay
            it += 1
    # after search, compute final score using evaluate_sequence, and maybe scale to outside range?
    return best_seq.tolist()

"""
Hybrid optimizer for the step‑function evaluation problem.

The objective 2·n·max(conv(a,a))/(sum(a)^2) is invariant under uniform scaling,
so we work with sequences that sum to 1 (a simplex).  Starting from a uniform
distribution we repeatedly:

1. Compute the current maximal self‑convolution M.
2. Solve a linear program that finds a “direction” g (a feasible probability
   vector) with maximal total mass while keeping the mixed convolution
   f * g ≤ M.  For a uniform f this LP simply returns an extreme point,
   i.e. a unit vector at some index.
3. Perform a one‑dimensional line‑search on t∈(0,1] for the convex
   combination  f₊ = (1‑t)·f + t·g .  The new sequence is accepted if it
   strictly reduces the max‑convolution.
4. Project the result back onto the simplex (normalisation).

The loop is run for several lengths N (small enough to keep the quadratic
convolution cheap, but large enough to obtain a good bound) and for a fixed
time budget.  The best sequence found over all restarts is returned.

The implementation relies only on NumPy and SciPy (high‑level LP solver
‘highs’), uses no file or network I/O and provides progress output via
`print`.
"""

import time
import numpy as np
from scipy import optimize


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.sum(v) == 1 and np.all(v >= 0):
        return v
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero – return uniform tiny vector
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _solve_conv_lp(f: np.ndarray, bound: float) -> np.ndarray | None:
    """
    Linear program:
        maximise   sum(g)
        subject to  (f * g)[k] <= bound   for all k
                    g >= 0
    Returns the optimal g (as a raw numpy array) or None if the solve fails.
    """
    n = f.shape[0]
    c = -np.ones(n)                 # maximise sum(g)  <=> minimise -sum(g)

    # build the convolution constraints a_ub @ g <= bound
    rows = []
    rhs = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        # (f * g)[k] = sum_i f[i] * g[k-i]
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                row[j] += f[i]
        rows.append(row)
        rhs.append(bound)

    # non‑negativity constraints via -I * g <= 0
    rows.append(-np.eye(n))
    rhs.append(np.zeros(n))

    A = np.vstack(rows)
    b = np.hstack(rhs)

    # use the efficient HiGHS interior‑point method
    res = optimize.linprog(
        c,
        A_ub=A,
        b_ub=b,
        bounds=(0, None),
        method="highs",
        options={"presolve": True, "time_limit": 5.0},
    )
    if res.success:
        return res.x
    return None


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step sequence that minimises the provided
    evaluation function.  The function stops after `budget_s` seconds and
    returns the best sequence found (as a Python list).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # a small safety margin

    best_seq: np.ndarray | None = None
    best_score = float("inf")

    # try a few different lengths – small enough for fast LP solves
    for N in (150, 250, 350, 500):
        # initialise with the uniform distribution (feasible and often near optimal)
        f = np.full(N, 1.0 / N)

        # main optimisation loop
        it = 0
        while time.time() < deadline and it < 300:
            it += 1

            # current max‑convolution and corresponding score
            conv_f = np.convolve(f, f)
            M = float(conv_f.max())
            cur_score = 2.0 * N * M          # sum(f) == 1 → simplification
            if cur_score < best_score:
                best_score = cur_score
                best_seq = f.copy()
                print(f"[{time.time() - start_time:6.2f}s] New best: {best_score:.6f} (N={N}, it={it})")

            # solve the LP to obtain a promising direction g
            g0 = _solve_conv_lp(f, M)
            if g0 is None:
                break  # LP failed – give up on this N

            # normalise g to lie on the simplex
            if g0.sum() == 0:
                break
            g = _project_simplex(g0)

            # line‑search for the best mixing coefficient t ∈ (0,1]
            # we evaluate a modest grid – enough to see improvement
            best_local_t = 0.0
            best_local_score = cur_score
            for t in np.linspace(0.05, 1.0, 20):
                cand = (1.0 - t) * f + t * g
                cand = _project_simplex(cand)          # keep on the simplex
                conv_c = np.convolve(cand, cand)
                M_c = float(conv_c.max())
                s_c = 2.0 * N * M_c
                if s_c < best_local_score:
                    best_local_score = s_c
                    best_local_t = t

            # if no improvement, inject a tiny random perturbation to escape plateaus
            if best_local_t == 0.0:
                perturb = np.random.randn(N) * 1e-4
                f = _project_simplex(f + perturb)
                continue

            # accept the best convex combination
            f = (1.0 - best_local_t) * f + best_local_t * g
            f = _project_simplex(f)

        # early exit if time is exhausted
        if time.time() >= deadline:
            break

    # fallback – uniform sequence – should never happen but kept for safety
    if best_seq is None:
        best_seq = np.full(100, 0.01)
        best_seq = best_seq / best_seq.sum()

    # return a plain Python list as required by the framework
    return best_seq.tolist()
