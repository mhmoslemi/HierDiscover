def simplex_projection(v):
    # projecting onto probability simplex
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u > cssv / np.arange(1, len(v)+1))[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Simple yet effective maximisation of the peak of the self‑convolution.

The evaluation function can be written (after the sum‑normalisation
performed inside ``evaluate_sequence``) as

    score = 2 * n * max( (x * x) ),
    where x_i ≥ 0 , Σ x_i = 1.

Thus we need a probability vector *x* whose autocorrelation
(x⋆x) is as flat as possible.
The algorithm is a projected gradient (mirror) descent on the simplex:

* start from a uniform distribution plus a tiny random perturbation
* repeatedly compute the full convolution, a soft‑max weighting of the
  convolution entries (temperature τ) and the gradient of the weighted
  sum;
* perform an exponentiated‑gradient update (which automatically keeps
  the vector non‑negative) and renormalise to Σ x=1;
* slowly cool the temperature τ and slowly decay the step size;
* occasionally inject a small amount of noise to avoid getting stuck
  at symmetric points such as the perfectly uniform vector.

The method works entirely with NumPy, needs only O(n²) operations per
iteration (≈ 5·10⁶ FLOPs for n≈2300) and therefore can run many
thousands of iterations in the 1000 s budget.
The best sequence encountered is returned.
"""

import numpy as np
import time

# ----------------------------------------------------------------------
# Helper – exponentiated‑gradient update with projection onto the simplex
# ----------------------------------------------------------------------
def exp_gradient_step(x, grad, step):
    """
    Multiplicative (exponentiated) update followed by a projection onto
    the probability simplex.  ``x`` and ``grad`` are 1‑D NumPy arrays.
    """
    # multiplicative update; the exponential guarantees non‑negativity
    x_new = x * np.exp(-step * grad)

    # optional clipping to avoid extremely large spikes (empirical)
    # we never want a single entry to be more than a few times the
    # uniform value 1/n – this keeps the convolution well behaved.
    n = x_new.shape[0]
    max_allowed = 5.0 / n                 # 5 × uniform is a safe bound
    np.clip(x_new, 0.0, max_allowed, out=x_new)

    # project onto the simplex by normalisation (the vector stays non‑negative)
    s = np.sum(x_new)
    if s == 0.0:                          # very unlikely, but guard against it
        return np.full_like(x_new, 1.0 / n)
    return x_new / s


# ----------------------------------------------------------------------
# Main entry point required by the judging harness
# ----------------------------------------------------------------------
def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Search for a step‑function (a non‑negative sequence) that minimises
    ``evaluate_sequence``.  The method uses a soft‑max weighted gradient
    descent on the probability simplex as described in the module
    docstring.
    """
    np.random.seed(seed)

    # --------------------------------------------------------------
    # Determine the length of the sequence (the "SOTA" length is
    # available through the global variable ``height_sequence_1``).
    # --------------------------------------------------------------
    try:
        n = len(height_sequence_1)            # provided by the environment
    except NameError:
        # fallback – a reasonably sized sequence if the global is missing
        n = 1024

    # --------------------------------------------------------------
    # Parameters for the optimisation loop (tuned for the 1000 s budget)
    # --------------------------------------------------------------
    max_total_time = budget_s - 0.5            # small safety margin
    start_time = time.time()
    best_score = np.inf
    best_seq = None

    # We perform a few independent restarts – each may discover a
    # better basin of attraction.
    while time.time() - start_time < max_total_time:
        # ---- initial vector: uniform + small random noise -----------------
        x = np.full(n, 1.0 / n)
        x += 0.01 * np.random.rand(n)        # break symmetry of the uniform vector
        x = np.maximum(x, 0.0)
        x /= np.sum(x)

        # ---- optimisation hyper‑parameters (they adapt during the run) --
        tau     = 0.5                         # temperature for the soft‑max
        eta0    = 0.2                         # base step size for the exponentiated update
        iteration = 0
        # we cannot afford an infinite inner loop; cap it loosely
        max_inner_iters = 200000

        while (iteration < max_inner_iters and
               time.time() - start_time < max_total_time):
            # ----- full self‑convolution (O(n²), but fast in C) ----------
            conv = np.convolve(x, x)          # length 2n‑1
            max_conv = np.max(conv)

            # ----- compute the current score (exactly the same as evaluate_sequence) -----
            # because Σ x = 1, the denominator of evaluate_sequence equals 1
            cur_score = 2.0 * n * max_conv
            if cur_score < best_score:
                best_score = cur_score
                best_seq = x.copy()
                print(f"[{time.time()-start_time:6.2f}s] iter {iteration:6d}  new best = {best_score:.6f}")

                # early stop if we already beat the target
                if best_score <= 1.503:
                    return best_seq.tolist()

            # ----- soft‑max weights over the convolution entries -------------
            # w_k ∝ exp( (conv_k - max_conv) / τ )
            # the subtraction of max_conv avoids overflow.
            w = np.exp((conv - max_conv) / tau)
            w_sum = np.sum(w)
            w /= w_sum

            # ----- gradient of the weighted sum of convolutions -------------
            # ∇_i  Σ_k w_k (x⋆x)_k = 2 Σ_k w_k x_{k-i}
            # This can be written as a convolution of w with the reversed x.
            rev_x = x[::-1]                    # reversed vector, length n
            grad_full = np.convolve(w, rev_x)  # length (2n‑1) + n‑1
            # extract the n entries that correspond to ∑_k w_k x_{k-i}
            grad = 2.0 * grad_full[n-1:n-1+n]

            # ----- exponentiated gradient step -------------------------------
            step = eta0 / np.sqrt(iteration + 1)   # diminishing step size
            x = exp_gradient_step(x, grad, step)

            # ----- occasional random jitter to avoid stagnation -----------
            if iteration % 500 == 0:
                jitter = (np.random.rand(n) - 0.5) * 1e-4
                x = np.maximum(x + jitter, 0.0)
                x /= np.sum(x)

            # ----- temperature schedule (cooling) ----------------------------
            tau = max(1e-4, tau * 0.9995)     # slowly tighten the soft‑max

            iteration += 1

        # after a restart we try again only if time permits
        if best_score <= 1.503:
            break

    # --------------------------------------------------------------
    # Return the best sequence found (as a plain Python list of floats)
    # --------------------------------------------------------------
    if best_seq is None:
        # should never happen, but a fallback uniform sequence is safe
        best_seq = np.full(n, 1.0 / n)
    return best_seq.tolist()
