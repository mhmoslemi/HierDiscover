def project_to_simplex(v):
    # Implementation from Duchi et al.
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1,n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def project_to_simplex(v):
    """Projects vector v onto the probability simplex."""
    # from Duchi et al. (2008)
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    # find rho
    rho = np.nonzero(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        # all v <= something, return uniform? Actually set tau = max(v) - ??? but safe:
        tau = cssv[-1] / n
    else:
        rho = rho[-1]
        tau = cssv[rho] / (rho + 1)
    w = np.maximum(v - tau, 0)
    return w

    f(a) = 2·n·max(conv(a,a)) / (sum(a)^2) .
Because the objective is scale‑invariant we keep the vector on the simplex
(sum = 1, a_i ≥ 0).  At each iteration we:
  * compute the linear convolution a*a (FFT for speed),
  * locate the index k where the convolution attains its maximum,
  * compute the sub‑gradient of max(conv) w.r.t. a,
  * take an Adam step in the opposite direction,
  * project the result back onto the probability simplex.

Multiple restarts with different lengths are performed while respecting the
overall time budget.  The best sequence found is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, sum x_i = 1}.
    Implementation of Duchi et al. (2008).
    """
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1  # cumulative sum minus 1
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = cssv[-1] / n
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Small utility returning the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_fft(a: np.ndarray, size: int) -> np.ndarray:
    """
    Linear convolution a * a using FFT of length `size`.
    Returns the full convolution (length 2*len(a)-1).
    """
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * a.shape[0] - 1]
    return conv


def _max_conv_gradient(a: np.ndarray, k_max: int) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) at the index `k_max`.
    For each i, grad_i = 2 * a_j where j = k_max - i (if 0≤j<n).
    The diagonal case (i == j) also yields 2 * a_i.
    """
    n = a.shape[0]
    i_arr = np.arange(n)
    j_arr = k_max - i_arr
    mask = (j_arr >= 0) & (j_arr < n)
    i_sel = i_arr[mask]
    j_sel = j_arr[mask]

    grad = np.zeros_like(a)
    # diagonal entries
    diag = i_sel == j_sel
    if np.any(diag):
        idx = i_sel[diag]
        grad[idx] = 2.0 * a[idx]
    # off‑diagonal entries
    if np.any(~diag):
        idx_i = i_sel[~diag]
        idx_j = j_sel[~diag]
        grad[idx_i] = 2.0 * a[idx_j]
    return grad


# ----------------------------------------------------------------------
# Main candidate generator
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a sequence of non‑negative heights that lowers the
    evaluation function.  The function respects a wall‑clock budget
    (default 1000 s) and returns the best sequence found as a plain
    Python list.
    """
    start_time = time.time()
    # Keep a tiny safety margin
    deadline = start_time + budget_s - 0.5

    np.random.seed(seed)

    best_seq = None
    best_val = float("inf")

    # Lengths tried in increasing order; the last one is close to the
    # length of the previously discovered construction.
    lengths = [256, 512, 1024, 2048, 4096, 6000]

    # Adam hyper‑parameters (same for all runs)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    base_lr = 0.5   # learning rate before time‑decay

    for n in lengths:
        if time.time() >= deadline:
            break

        # allocate a slice of the remaining budget to this length
        remaining = deadline - time.time()
        # at most 30 s per length, but also share proportionally
        time_for_n = min(30.0, remaining / (len(lengths) - lengths.index(n)))
        t_start_n = time.time()

        # FFT size (next power of two ≥ 2·n‑1)
        fft_size = _next_pow2(2 * n - 1)

        # ------------------------------------------------------------------
        # Initialise a starting point for this length
        # ------------------------------------------------------------------
        if best_seq is not None:
            # linearly interpolate the best sequence seen so far to the new size
            old = np.asarray(best_seq, dtype=float)
            old_n = old.shape[0]
            new_pos = np.linspace(0, old_n - 1, n)
            old_pos = np.arange(old_n)
            a = np.interp(new_pos, old_pos, old)
        else:
            a = np.random.rand(n)

        # small random perturbation to break symmetry
        a += 0.01 * np.random.randn(n)
        a = np.clip(a, 0.0, None)
        if a.sum() == 0.0:
            a = np.random.rand(n)
        a /= a.sum()

        # Adam state
        m = np.zeros_like(a)
        v = np.zeros_like(a)

        iteration = 0

        # ------------------------------------------------------------------
        # Main optimisation loop for the current length
        # ------------------------------------------------------------------
        while time.time() - t_start_n < time_for_n:
            iteration += 1

            # ----- compute current convolution and locate the max -----
            conv = _conv_fft(a, fft_size)
            k_max = int(np.argmax(conv))

            # ----- sub‑gradient of the max entry -----
            grad = _max_conv_gradient(a, k_max)

            # ----- Adam update -----
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** iteration)
            v_hat = v / (1.0 - beta2 ** iteration)
            step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

            a_new = a - step
            # ----- projection onto the simplex (non‑negative & sum=1) -----
            a = _project_to_simplex(a_new)

            # occasional tiny jitter to help escape flat regions
            if iteration % 2000 == 0:
                a += 0.001 * np.random.randn(n)
                a = np.clip(a, 0.0, None)
                a /= a.sum()

            # ----- evaluate the (exact) objective for the new point -----
            conv_after = _conv_fft(a, fft_size)
            maxb = conv_after.max()
            sum_a = a.sum()
            if sum_a < 1e-12:   # should never happen after projection
                continue
            val = 2.0 * n * maxb / (sum_a * sum_a)

            # ----- keep the best sequence seen so far -----
            if val < best_val:
                best_val = float(val)
                best_seq = a.copy()
                print(
                    f"[{time.time() - start_time:6.2f}s] "
                    f"New best: {best_val:.6f} (n={n}, it={iteration})"
                )

            # ----- early exit if overall time is exhausted -----
            if time.time() >= deadline:
                break

        # end while (time for this length)

    # ----------------------------------------------------------------------
    # Final polishing on the globally best sequence (if time permits)
    # ----------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        a = best_seq.copy()
        n = a.shape[0]
        fft_size = _next_pow2(2 * n - 1)
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        iteration = 0
        # a few thousand fine‑tuning steps with a smaller learning rate
        while time.time() < deadline:
            iteration += 1
            conv = _conv_fft(a, fft_size)
            k_max = int(np.argmax(conv))
            grad = _max_conv_gradient(a, k_max)

            # smaller learning‑rate for polishing
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** iteration)
            v_hat = v / (1.0 - beta2 ** iteration)
            step = 0.1 * m_hat / (np.sqrt(v_hat) + eps)

            a = _project_to_simplex(a - step)

            conv_after = _conv_fft(a, fft_size)
            maxb = conv_after.max()
            val = 2.0 * n * maxb / (a.sum() ** 2)
            if val < best_val:
                best_val = float(val)
                best_seq = a.copy()
                print(
                    f"[{time.time() - start_time:6.2f}s] "
                    f"Polish improvement: {best_val:.6f} (n={n})"
                )

    # ----------------------------------------------------------------------
    # Return the result as a plain list (fallback to the provided seed if needed)
    # ----------------------------------------------------------------------
    if best_seq is None:
        # fallback – use the previously known construction if it exists
        try:
            best_seq = list(height_sequence_1)
        except Exception:
            best_seq = [1.0]

    return list(best_seq.tolist())
