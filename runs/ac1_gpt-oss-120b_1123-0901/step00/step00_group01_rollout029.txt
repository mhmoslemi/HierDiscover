def fast_evaluation(seq):
    seq_arr = np.array(seq, dtype=np.float64)
    # apply clipping as evaluate does
    seq_arr = np.maximum(0.0, np.minimum(1000.0, seq_arr))
    n = len(seq_arr)
    sum_a = seq_arr.sum()
    if sum_a < 0.01:
        return np.inf
    # compute convolution via FFT
    # Compute size as next power of 2 for efficiency
    m = 1 << ((2*n - 1).bit_length())
    fa = np.fft.rfft(seq_arr, m)
    conv = np.fft.irfft(fa * fa, m)[: 2*n - 1]
    max_b = np.max(conv)
    return 2.0 * n * max_b / (sum_a * sum_a)

def fast_evaluate(seq_array):
    # already clipped to [0,1000]?
    # Sequence passed from generate already in [0,1000] since r <=1.
    seq = np.asarray(seq_array, dtype=np.float64)
    n = seq.shape[0]
    sum_a = seq.sum()
    if sum_a < 0.01:
        return np.inf
    # compute convolution using FFT
    # length for conv = 2*n-1
    m = 1 << ((2*n-1).bit_length())
    fa = np.fft.rfft(seq, m)
    conv = np.fft.irfft(fa * fa, m)[:2*n-1]
    max_b = conv.max()
    return 2.0 * n * max_b / (sum_a * sum_a)

"""
Propose a step‑function (non‑negative list) that minimizes the
`evaluate_sequence` score.

The algorithm mixes three ideas:

1. **Fast evaluation** – convolution via FFT (identical to the
   reference implementation but much faster).

2. **Exponential families** – a one‑parameter family
   a_i = r**i (with r close to 1) is known to beat the uniform
   construction.  A small sweep over the parameter provides a quick
   good baseline (< 2.0).

3. **LP‑guided improvement** – following the SOTA idea, we solve a
   linear program that, keeping the current maximal convolution value,
   maximises the ℓ₁‑norm.  The new direction is then mixed with the
   current sequence (several step sizes are tried) and the best
   improvement is kept.

The whole loop runs until the allotted time (`budget_s`) is exhausted.
Any sequence found with a lower score than the previous best is saved
and finally returned.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper: fast evaluation that mimics `evaluate_sequence` exactly.
# ----------------------------------------------------------------------
def _fast_evaluate(seq: np.ndarray) -> float:
    """Return the same score as `evaluate_sequence` but using FFT."""
    # Clip to [0,1000] as the validator does.
    a = np.clip(seq, 0.0, 1000.0)
    n = a.shape[0]
    s = a.sum()
    if s < 0.01:          # matches the safety check
        return float("inf")
    # convolution via FFT (real‑to‑real)
    m = 1 << ((2 * n - 1).bit_length())        # next power of two
    fa = np.fft.rfft(a, m)
    conv = np.fft.irfft(fa * fa, m)[: 2 * n - 1]
    max_b = conv.max()
    return 2.0 * n * max_b / (s * s)


# ----------------------------------------------------------------------
# Helper: solve the LP  max Σ b_j  s.t.  (f * b)_k ≤ rhs,  b ≥ 0
# ----------------------------------------------------------------------
def _solve_convolution_lp(f: np.ndarray, rhs: float):
    """
    Linear program used in the original approach.
    Returns a feasible vector `b` (numpy array) or None.
    """
    n = f.shape[0]
    c = -np.ones(n)                # maximise Σ b_j  → minimise -Σ b_j
    # Build A_ub row for every convolution index k
    rows = []
    bounds = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        # (f * b)_k = Σ_i f_i * b_{k-i}
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        # j = k-i runs from i_min..i_max
        # the coefficient of b_j is f_{k-j}
        for i in range(i_min, i_max + 1):
            j = k - i
            row[j] = f[i]
        rows.append(row)
        bounds.append(rhs)
    # non‑negativity  –b ≤ 0
    rows.append(-np.eye(n))
    bounds.extend([0.0] * n)

    A_ub = np.vstack(rows)
    b_ub = np.array(bounds)

    # use the high‑performance HiGHS solver; give it a modest time limit
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, method="highs",
                  options={"time_limit": 4.0, "presolve": True, "disp": False})
    if res.success:
        return res.x
    return None


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Returns a list of non‑negative heights (the step function) that gives a
    low `evaluate_sequence` score.  The routine respects the time budget
    (`budget_s` seconds) and performs a quick global sweep over an
    exponential family followed by several LP‑guided refinement steps.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # leave a tiny margin

    # ------------------------------------------------------------------
    # Determine a working length.  If the reference construction is
    # available we use at most its length (to keep the LP size tractable).
    # ------------------------------------------------------------------
    try:
        base_len = len(height_sequence_1)          # provided globally
        n = min(base_len, 2000)                    # limit for LP speed
    except Exception:
        n = 2000

    # ------------------------------------------------------------------
    # Start from a simple uniform vector (this already scores 2.0)
    # ------------------------------------------------------------------
    best_seq = np.full(n, 1.0, dtype=np.float64)
    best_score = _fast_evaluate(best_seq)
    print(f"[init] uniform length={n} score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 1) Scan an exponential family a_i = r**i .
    #    Parameter r is expressed via d = n * (1 - r)  (d ≈ 1 … 2 gives
    #    the best theoretical trade‑off for this family).
    # ------------------------------------------------------------------
    for d in np.linspace(0.4, 2.0, 33):
        r = 1.0 - d / n
        if r <= 0.0:
            continue
        cand = np.power(r, np.arange(n, dtype=np.float64))
        sc = _fast_evaluate(cand)
        if sc < best_score:

    curr_seq = best_seq.copy()

            # keep a minimum mass to avoid the <0.01 guard

            scale_g = sqrt_2n / sum_g
            g = g_norm * scale_g

            # try several convex‑combination step sizes and keep the best
            best_local_seq = None
            best_local_score = float("inf")
            for t in (0.02, 0.05, 0.10, 0.20):
                cand = (1.0 - t) * curr_seq + t * g
                sc = _fast_evaluate(cand)
                if sc < best_local_score:
                    best_local_score, best_local_seq = sc, cand
            new_seq = best_local_seq if best_local_seq is not None else curr_seq

        # ----------------------------------------------------------------
        # Evaluate the candidate and keep it if it improves the global best.
        # ----------------------------------------------------------------
        cand_score = _fast_evaluate(new_seq)
        if cand_score < best_score:
            best_score, best_seq = cand_score, new_seq.copy()
            print(f"[improve] new best score={best_score:.6f}")

                if sc < best_score:

                    print(f"[exp‑rand] d={d:.3f}  score={sc:.6f}")

    print(f"[final] best score achieved: {best_score:.6f}")
    # Return a plain Python list as required.
    return best_seq.tolist()
