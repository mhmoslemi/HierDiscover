def propose_candidate(seed=0, budget_s=1000, **kwargs):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2
    # Use length n from global variable height_sequence_1
    global height_sequence_1
    n = len(height_sequence_1)

    # Initialize a uniform distribution
    a = np.full(n, 1.0/n, dtype=float)

    # Evaluate initial
    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"Initial score: {best_score}")

    # parameters
    beta = 20.0
    eta = 0.02  # multiplicative step size
    # learning schedule: gradually increase beta to focus more on max
    beta_inc = (200.0 - beta) / 2000  # e.g., after many iterations increase.
    # We'll define iteration count.
    iter_count = 0

    # Pre-allocate size for FFT lengths
    Lconv = 1 << ((2*n - 1).bit_length())
    Lcorr = 1 << ((3*n - 2).bit_length())

    while time.time() < deadline:
        # Compute convolution via FFT
        A = np.fft.rfft(a, Lconv)
        conv_full = np.fft.irfft(A * A, Lconv)
        conv = conv_full[:2*n-1]  # exact linear conv length

        # Compute softmax weights approximating max
        max_c = np.max(conv)
        # To handle potential overflow, subtract max
        exp_vals = np.exp(beta * (conv - max_c))
        sum_exp = np.sum(exp_vals)
        p = exp_vals / sum_exp

        # Compute correlation for gradient
        revp = p[::-1]
        A_corr = np.fft.rfft(a, Lcorr)
        RevP_corr = np.fft.rfft(revp, Lcorr)
        conv_rev = np.fft.irfft(A_corr * RevP_corr, Lcorr)
        # correlation_i = conv_rev[2*n-2 - i] for i=0..n-1
        idx_start = 2*n - 2
        corr = conv_rev[idx_start - np.arange(n)]

        # Gradient = 2 * beta * corr
        grad = 2.0 * beta * corr

        # Multiplicative update (exponential mirror descent)
        # To keep numbers stable, we can clip grad
        # Ensure grad is not too large
        # Use eta scaling
        a = a * np.exp(-eta * grad)

        # Normalize to sum=1 (project onto simplex)
        sum_a = np.sum(a)
        if sum_a == 0.0 or np.isnan(sum_a):
            a = np.full(n, 1.0/n)
        else:
            a = a / sum_a

        # Occasionally increase beta to focus on max
        if iter_count % 100 == 0 and beta < 200:
            beta = min(beta * 1.05, 200)

        # Evaluate candidate
        try:
            score = evaluate_sequence(a.tolist())
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"Iter {iter_count}: new best {best_score:.6f}")
        except Exception as e:
            pass

        iter_count += 1
        # Optionally break early if improvement small.
        if iter_count % 500 == 0 and time.time() > deadline - 1:
            # final few sec
            pass

    print(f"Finished after {iter_count} iterations, best score {best_score}")
    return best_seq.tolist()

"""
Projected gradient descent to flatten the self‑convolution of a non‑negative
vector.

The algorithm works on the simplex (all entries ≥0 and sum = 1).  At each
iteration we compute the linear convolution a*a, locate the indices where the
convolution attains its (few) largest values and take a sub‑gradient of the
maximum (∂(max_k (a*a)_k)/∂a_i = 2·a_{k−i}).  A small step in the opposite
direction is taken, the vector is projected back onto the simplex and the
new objective value `evaluate_sequence` is recorded.  The step size is
adaptively reduced when a trial does not improve the score and increased
slightly when it does.  Random jitter is added from time to time to avoid
getting stuck in a flat region.  The search stops when the allotted time
budget is exhausted and the best sequence found is returned.
"""

import numpy as np
import time

def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the probability simplex."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0 or np.isnan(s):
        # fallback to uniform distribution
        n = len(v)
        return np.full(n, 1.0 / n, dtype=v.dtype)
    return v / s


def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """Search for a step‑function heights vector that minimises the evaluation metric.

    The function respects the 1000 s time budget, returning the best sequence it
    discovered.  The length `n` is taken from the global variable
    `height_sequence_1`.  The returned list consists of non‑negative floats.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # leave a little margin

    # ----------------------------------------------------------------------
    # 1. Problem size – use the length of the previously known construction.
    # ----------------------------------------------------------------------
    try:
        n = len(height_sequence_1)          # provided by the surrounding environment
    except Exception:
        # fallback: a reasonable size if the global is missing
        n = 2000

    # ----------------------------------------------------------------------
    # 2. Initial point – uniform distribution with a tiny random perturbation.
    # ----------------------------------------------------------------------
    a = np.full(n, 1.0 / n, dtype=np.float64)
    a += 1e-4 * np.random.rand(n)            # break symmetry
    a = _project_to_simplex(a)

    # ----------------------------------------------------------------------
    # 3. Book‑keeping.
    # ----------------------------------------------------------------------
    best_seq = a.copy()
    try:
        best_score = evaluate_sequence(best_seq.tolist())
    except Exception:
        best_score = float("inf")
    print(f"[init] n={n}, score={best_score:.6f}")

    # ----------------------------------------------------------------------
    # 4. Parameters for the optimisation loop.
    # ----------------------------------------------------------------------
    # FFT sizes (powers of two) for the two convolutions we need.
    L_conv = 1 << ((2 * n - 1).bit_length())          # for a*a
    L_corr = 1 << ((3 * n - 2).bit_length())          # for correlation a * p_rev
    # Learning‑rate – will be adapted during the run.
    lr = 0.2                                          # initial additive step size
    min_lr = 1e-5
    max_lr = 1.0
    # How many of the largest convolution entries we treat as active.
    K_active = 3
    # Counters used for adaptation.
    it = 0
    no_improve = 0

    # ----------------------------------------------------------------------
    # 5. Main optimisation loop.
    # ----------------------------------------------------------------------
    while time.time() < deadline:
        # ----- a) current convolution (FFT based) -------------------------
        A = np.fft.rfft(a, L_conv)
        conv_full = np.fft.irfft(A * A, L_conv)
        conv = conv_full[: 2 * n - 1]               # linear convolution length

        # ----- b) indices where the convolution is (almost) maximal ------
        # use np.argpartition for speed; we take the K_active largest values.
        if K_active == 1:
            k_star = np.argmax(conv)
            active_idxs = np.array([k_star], dtype=int)
        else:
            # partition to get K_active biggest values (unsorted)
            partition = np.argpartition(conv, -K_active)[-K_active:]
            # sort them descending for deterministic behaviour (optional)
            active_idxs = partition[np.argsort(-conv[partition])]

        # ----- c) sub‑gradient of max_k conv_k ---------------------------
        grad = np.zeros(n, dtype=np.float64)
        for k in active_idxs:
            # i ranges where a_{k-i} is defined.
            i_min = max(0, k - (n - 1))
            i_max = min(n - 1, k)
            i_range = np.arange(i_min, i_max + 1, dtype=int)
            grad[i_range] += 2.0 * a[k - i_range]

        # Normalise the gradient by the number of active indices (optional)
        grad /= float(len(active_idxs))

        # ----- d) trial step (additive) ----------------------------------
        a_trial = a - lr * grad
        a_trial = _project_to_simplex(a_trial)

        # ----- e) evaluate ------------------------------------------------
        try:
            score = evaluate_sequence(a_trial.tolist())
        except Exception:
            # a numerical glitch – treat as bad step
            score = float("inf")

        # ----- f) accept / reject logic -----------------------------------
        if score < best_score:
            best_score = score
            best_seq = a_trial.copy()
            a = a_trial                               # continue from the improved point
            # enlarge learning rate a little (but keep it bounded)
            lr = min(lr * 1.05, max_lr)
            no_improve = 0
            print(f"[{it:05d}] new best {best_score:.6f}, lr={lr:.6f}")
        else:
            # reject – shrink the step size
            lr = max(lr * 0.5, min_lr)
            no_improve += 1

        # ----- g) occasional random jitter to get out of plateaus ----------
        if no_improve >= 200:
            # inject a tiny random perturbation and reset the counter
            jitter = 1e-5 * np.random.randn(n)
            a = _project_to_simplex(a + jitter)
            no_improve = 0
            lr = max(lr, 1e-3)   # give the algorithm a fresh start

        # ----- h) housekeeping --------------------------------------------
        it += 1

        # optional early‑stop if we already hit the theoretical lower bound
        # (the bound is 2*n/(2*n-1) ≈ 1.0).  Stop once we are very close.
        if best_score < 1.01:
            break

    # ----------------------------------------------------------------------
    # 6. Return the best sequence discovered.
    # ----------------------------------------------------------------------
    print(f"[done] iterations={it}, best_score={best_score:.6f}")
    return best_seq.tolist()
