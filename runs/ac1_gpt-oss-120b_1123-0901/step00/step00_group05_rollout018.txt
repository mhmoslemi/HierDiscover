"""
Propose a candidate step‑function (non‑negative sequence) that minimizes

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

where conv is the full discrete convolution.  The algorithm starts from a
uniform sequence (the trivial optimal point for the naïve bound) and then
iteratively improves it using a linear‑programming “direction‑finding”
step described in the problem statement.  In each iteration we:

1.   Compute the current maximal convolution value M.
2.   Solve a linear program that finds a non‑negative sequence g with the
     largest possible ℓ₁‑norm under the linear constraints
         conv(f, g) ≤ M .
     (f is the current sequence.)
3.   Rescale g so that it has the same ℓ₁‑norm as f and take a convex
     combination `f ← (1‑t)·f + t·g`.  The step size *t* is decreased
     adaptively until an improvement of the evaluation score is observed.
4.   If the LP fails or no improvement is found, perform a tiny random
     perturbation of a single entry.
5.   Occasionally restart from a fresh random distribution to escape
     possible stagnation.

The loop runs until the allotted budget (default 1000 s) is exhausted.
The best sequence discovered is returned.  All numbers stay well below the
1000‑clamp in `evaluate_sequence`, so the ratio is scale‑invariant and the
search works purely on the shape of the sequence.
"""

import time
import numpy as np
from scipy import optimize

linprog = optimize.linprog


def solve_convolution_lp(f_seq, rhs):
    """
    Solve the LP:
        maximise   Σ_j b_j
        subject to conv(f_seq, b) ≤ rhs  (for every convolution index)
                    b_j ≥ 0 .
    Returns the optimal b as a NumPy array, or None if the LP fails.
    """
    n = len(f_seq)
    # objective: maximise sum(b)  → minimise -sum(b)
    c = -np.ones(n)

    # Build the upper‑bound matrix for the convolution constraints.
    # There are (2*n-1) constraints, one for each possible sum index.
    a_ub = []
    b_ub = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        # row[j] = f[i] where i + j == k  ⇔  i = k - j
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                row[j] = f_seq[i]
        a_ub.append(row)
        b_ub.append(rhs)

    # non‑negativity constraints  -b_j ≤ 0
    a_ub_nonneg = -np.eye(n)
    b_ub_nonneg = np.zeros(n)

    a_ub = np.vstack([a_ub, a_ub_nonneg])
    b_ub = np.hstack([b_ub, b_ub_nonneg])

    try:
        res = linprog(
            c,
            A_ub=a_ub,
            b_ub=b_ub,
            method="highs",
            options={"presolve": True, "time_limit": 5.0},
        )
    except Exception:          # pragma: no cover – safety net
        return None

    if res.success:
        # Numerical noise may produce tiny negative entries; clip them.
        return np.maximum(res.x, 0.0)
    return None


def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Search for a sequence with a small evaluation score.
    Returns a list of non‑negative floats.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # tiny safety margin

    # --- initialise ---------------------------------------------------------
    # start from a uniform distribution (the baseline with score = 2)
    n = np.random.randint(200, 350)          # moderate size – fast LPs
    uniform_val = 1.0 / n
    best_seq = [uniform_val] * n
    best_score = evaluate_sequence(best_seq)

    curr_seq = best_seq.copy()
    print(f"Start: n={n}, initial score={best_score:.6f}")

    # adaptive step‑size schedule (larger steps early, finer later)
    step_schedule = [0.2, 0.1, 0.05, 0.02, 0.01, 0.005]

    # --- main optimisation loop --------------------------------------------
    while time.time() < deadline:
        # Compute current maximal convolution value
        conv = np.convolve(curr_seq, curr_seq)
        max_b = np.max(conv)

        # LP to get a direction that can increase the ℓ₁‑norm while keeping
        # the convolution bound unchanged.
        g = solve_convolution_lp(curr_seq, max_b)

        if g is not None and np.sum(g) > 0:
            # Rescale g to have the same ℓ₁‑norm as the current sequence.
            sum_curr = np.sum(curr_seq)
            sum_g = np.sum(g)
            if sum_g == 0:          # extremely unlikely, but guard division
                continue
            g_scaled = (g * (sum_curr / sum_g)).tolist()

            improved = False
            # try the predefined step sizes from large to small
            for t in step_schedule:
                # convex combination
                new_seq = [(1 - t) * x + t * y for x, y in zip(curr_seq, g_scaled)]
                # ensure non‑negativity (the convex combo already guarantees it)
                score = evaluate_sequence(new_seq)
                if score < best_score:
                    best_score = score
                    best_seq = new_seq.copy()
                    curr_seq = new_seq
                    improved = True

                    )

                idx = np.random.randint(len(curr_seq))

                new_seq = curr_seq.copy()
                new_seq[idx] = max(0.0, new_seq[idx] + delta)
                score = evaluate_sequence(new_seq)
                if score < best_score:

                    curr_seq = new_seq

            if score < best_score:
                best_score = score

                curr_seq = new_seq
                print(f"LP‑fallback improvement: score={best_score:.6f}")

        # occasional full restart to avoid being stuck in a local optimum

            score = evaluate_sequence(curr_seq)
            if score < best_score:
                best_score = score
                best_seq = curr_seq.copy()
                print(f"Random restart: score={best_score:.6f}, n={m}")

    print(f"Finished: best score={best_score:.6f}, length={len(best_seq)}")
    return best_seq
