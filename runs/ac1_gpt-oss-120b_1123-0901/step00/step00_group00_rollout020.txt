import numpy as np
import time
from scipy.optimize import linprog

def solve_convolution_lp(f_seq, rhs):
    n = len(f_seq)
    c = -np.ones(n)  # maximize sum(b) => minimize -sum(b)
    # Build constraints
    # For each k in 0..2n-2, sum_i f_i * b_{k-i} <= rhs
    A = []
    b = []
    for k in range(2*n-1):
        row = np.zeros(n)
        # For all i where 0 <= k-i < n --> j = k-i
        lo = max(0, k-(n-1))
        hi = min(n-1, k)
        # i runs from lo..hi
        # correspondingly b_j indices
        # row[j] = f_i
        # but we need to set row[j] = f_i, j = k-i
        # iterate i
        for i in range(lo, hi+1):
            j = k - i
            row[j] = f_seq[i]
        A.append(row)
        b.append(rhs)
    # Non-negative constraints: -I * b <= 0
    A_ub = np.vstack([A, -np.eye(n)])
    b_ub = np.concatenate([b, np.zeros(n)])
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs', options={'time_limit': 10})
    if res.success:
        return np.maximum(0.0, res.x)   # ensure nonneg
    else:
        return None

    lo = max(0, k - (n-1))
    hi = min(n-1, k)
    # i from lo..hi
    for i in range(lo, hi+1):
        j = k - i

import numpy as np
import time
from scipy.optimize import linprog

def solve_convolution_lp(f_seq, rhs):
    n = len(f_seq)
    c = -np.ones(n)
    A_rows = []
    b_ub = []
    for k in range(2*n-1):
        row = np.zeros(n)
        lo = max(0, k-(n-1))
        hi = min(n-1, k)
        for i in range(lo, hi+1):
            j = k - i
            row[j] = f_seq[i]
        A_rows.append(row)
        b_ub.append(rhs)
    A = np.vstack([A_rows, -np.eye(n)])
    b = np.concatenate([b_ub, np.zeros(n)])
    res = linprog(c, A_ub=A, b_ub=b, method='highs', options={'time_limit': 10})
    if res.success:
        return np.maximum(res.x, 0.0)
    return None

"""
A simple stochastic optimizer for the step‑function evaluation problem.

The objective after the evaluator’s internal normalisation is to minimise
the maximal value of the discrete convolution a*a while the L1‑norm of a
is fixed (≈√(2·n)).  The algorithm works with a fixed length `n` and
alternates two cheap operations:

1. **Sub‑gradient step** – the index `k*` where the convolution attains
   its maximum gives a sub‑gradient `g_i = 2·a_{k*‑i}` (zero outside the
   feasible range).  Moving a little opposite to this gradient and
   projecting back onto the simplex (non‑negative entries with the required
   sum) never increases the maximal convolution and often reduces it.

2. **Linear‑programming jump** – occasionally we solve the LP used in the
   published SOTA (maximise the total sum of a non‑negative vector `b`
   under `conv(a,b) ≤ max(conv(a,a))`).  The optimal `b` is mixed with the
   current candidate using a line‑search over the mixing factor `t`.  This
   provides larger, safe moves that the plain gradient step may miss.

The optimiser runs for the allotted time budget (default 1000 s), keeps the
best sequence seen, and finally returns it as a plain Python list.
"""

import time
import numpy as np
from scipy.optimize import linprog

# --------------------------------------------------------------
# Linear programming helper (identical to the one used in the SOTA)
# --------------------------------------------------------------
def solve_convolution_lp(f_seq: np.ndarray, rhs: float):
    """
    Solve:
        maximise   Σ b_i
        subject to  conv(f_seq, b) ≤ rhs   (for every shift)
                    b_i ≥ 0
    Returns the optimal vector b (as a NumPy array) or None on failure.
    """
    n = len(f_seq)
    # objective: minimise -Σ b_i  (i.e. maximise Σ b_i)
    c = -np.ones(n)

    # Build the convolution constraints.
    rows = []
    bounds = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        # i runs from lo .. hi  and j = k-i
        for i in range(lo, hi + 1):
            j = k - i
            row[j] = f_seq[i]
        rows.append(row)
        bounds.append(rhs)

    # non‑negativity constraints  (-I) b ≤ 0
    rows.append(-np.eye(n))
    bounds = np.concatenate([bounds, np.zeros(n)])

    A = np.vstack(rows)

    # HiGHS (default) is very fast for these modest dimensions.
    res = linprog(
        c,
        A_ub=A,
        b_ub=bounds,
        method="highs",
        options={"time_limit": 10.0, "disp": False},
    )
    if res.success:
        # Numerical noise may produce tiny negatives – clip them.
        return np.maximum(res.x, 0.0)
    return None


# --------------------------------------------------------------
# Gradient / LP based optimisation loop
# --------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **_):
    """
    Return a list of non‑negative heights that (hopefully) gives a
    low value of `evaluate_sequence`.  The optimiser uses a mixture of
    sub‑gradient descent and occasional LP jumps under a strict time budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5  # keep a small safety margin

    # ----------------------------------------------------------
    # 1. Choose a problem size.
    #    Larger n gives more freedom but each iteration becomes
    #    slower (O(n²) convolution).  600–800 is a good trade‑off.
    # ----------------------------------------------------------
    n = 720                     # fixed length for the whole run
    target_sum = np.sqrt(2.0 * n)

    # ----------------------------------------------------------
    # 2. Initialise a random feasible point.
    #    Dirichlet distribution (exponential samples) gives a dense,
    #    non‑uniform start that breaks the symmetry of the uniform vector.
    # ----------------------------------------------------------
    random_weights = np.random.exponential(scale=1.0, size=n)
    seq = random_weights * (target_sum / random_weights.sum())

    # Ensure feasibility after any rounding errors.
    seq = np.maximum(seq, 0.0)

    best_seq = seq.copy()
    # direct evaluation (faster than the provided wrapper because we already have the convolution)
    conv = np.convolve(seq, seq)
    best_score = float(2 * n * np.max(conv) / (np.sum(seq) ** 2))

    # algorithmic parameters
    step_size = 0.05 * target_sum          # initial sub‑gradient step
    min_step = 1e-7
    lp_every = 15                          # attempt an LP jump every this many iterations
    iter_counter = 0

    while time.time() < deadline:
        # ------------------------------------------------------
        # 2a) Sub‑gradient step (always performed)
        # ------------------------------------------------------
        conv = np.convolve(seq, seq)
        max_val = np.max(conv)
        k_star = np.argmax(conv)                # shift where max occurs

        # sub‑gradient of max(conv) wrt a_i is 2·a_{k*‑i} (if the index exists)
        grad = np.zeros_like(seq)
        lo = max(0, k_star - (n - 1))
        hi = min(n - 1, k_star)
        # only indices i in [lo, hi] have a partner j = k_star - i
        for i in range(lo, hi + 1):
            j = k_star - i
            grad[i] = 2.0 * seq[j]

        # candidate move
        cand = seq - step_size * grad
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s > 0:
            cand *= (target_sum / s)          # project onto the L1‑sphere

        # evaluate candidate
        conv_cand = np.convolve(cand, cand)
        cand_score = float(2 * n * np.max(conv_cand) / (np.sum(cand) ** 2))

        # accept if we improve
        if cand_score < best_score:
            seq = cand
            best_seq = cand.copy()
            best_score = cand_score
            # increase step a little – we are heading downhill
            step_size = min(step_size * 1.15, target_sum)
            print(f"[{iter_counter:4d}] improvement → {best_score:.6f}  step={step_size:.6f}")
        else:
            # no improvement – shrink the step
            step_size = max(step_size * 0.55, min_step)

        # ------------------------------------------------------
        # 2b) Periodic LP jump (more aggressive move)
        # ------------------------------------------------------
        if iter_counter % lp_every == 0 and iter_counter > 0:
            # Normalise the current vector (its sum is already target_sum,
            # but we keep the normalisation step for numerical safety)
            cur_sum = seq.sum()
            f_norm = seq * (target_sum / cur_sum)

            # current bound on the convolution
            rhs = np.max(np.convolve(f_norm, f_norm))

            b = solve_convolution_lp(f_norm, rhs)
            if b is not None:
                # scale b to the same L1‑norm as f_norm
                b_sum = b.sum()
                if b_sum > 0:
                    g = b * (target_sum / b_sum)

                    # line‑search on the mixing factor t∈[0,1]
                    best_local = seq
                    best_local_score = best_score
                    for t in (0.05, 0.10, 0.20, 0.35, 0.50, 0.70, 0.90, 1.0):
                        mixed = (1.0 - t) * f_norm + t * g
                        mixed = np.maximum(mixed, 0.0)          # safety
                        mixed_sum = mixed.sum()
                        if mixed_sum == 0:
                            continue
                        mixed *= (target_sum / mixed_sum)

                        conv_mix = np.convolve(mixed, mixed)
                        mix_score = float(2 * n * np.max(conv_mix) / (np.sum(mixed) ** 2))

                        if mix_score < best_local_score:
                            best_local_score = mix_score
                            best_local = mixed.copy()

                    # accept the best mixing found
                    if best_local_score < best_score:
                        seq = best_local
                        best_seq = best_local.copy()
                        best_score = best_local_score
                        print(f"[{iter_counter:4d}] LP‑mix improvement → {best_score:.6f}")

        iter_counter += 1

    # --------------------------------------------------------------
    # Return the best sequence found (as a plain Python list)
    # --------------------------------------------------------------
    return best_seq.tolist()
