def flatten_sequence(seq, steps, step_size):
    for _ in range(steps):
        b = np.convolve(seq, seq)
        k_max = int(np.argmax(b))
        # compute contributions for each index i that has a partner j = k_max - i within range
        indices = []
        contribs = []
        for i in range(len(seq)):
            j = k_max - i
            if 0 <= j < len(seq):
                contribs.append(seq[i]*seq[j])
                indices.append(i)
        # find index with max contribution
        idx_max = np.argmax(contribs)
        i_max = indices[idx_max]
        j_max = k_max - i_max
        # Determine amount to move
        delta = step_size * min(seq[i_max], seq[j_max])
        if delta <=0:
            continue
        # Subtract from both i_max and j_max
        seq[i_max] = max(0, seq[i_max] - delta)
        seq[j_max] = max(0, seq[j_max] - delta)
        # Add to random other indices (maybe two random distinct)
        # total removed = 2*delta, we can distribute equally among two random indices not in {i_max,j_max}
        # or uniformly to all other indices
        # Simpler: Add to random i_rand1 and i_rand2
        for _ in range(2):
            r = np.random.randint(len(seq))
            if r != i_max and r != j_max:
                seq[r] += delta
        # renormalize sum to 1
        total = np.sum(seq)
        if total > 0:
            seq = seq / total

import numpy as np
import time
from scipy import optimize

linprog = optimize.linprog

def compute_score(a):
    """
    Compute the evaluation score equal to:
        2 * n * max(conv(a, a)) / (sum(a) ** 2)
    This is identical to evaluate_sequence but much faster because we
    avoid the defensive checks and clipping.
    """
    b = np.convolve(a, a)
    max_b = b.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * len(a) * max_b / (s * s))

def solve_convolution_lp(f_sequence, rhs):
    """Solve the linear program:
       maximise sum(b)  s.t.  conv(f, b) <= rhs, b >= 0.
    Returns the optimal b (as a numpy array) or None if LP fails.
    """
    n = len(f_sequence)
    c = -np.ones(n)  # maximise sum(b)
    # Build the convolution constraints
    a_ub = []
    b_ub = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        # for each i that contributes to index k
        # we need b_j with j = k - i
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        # when i runs from lo..hi, j = k-i is in [0,n-1]
        # row[j] += f[i]
        # This can be done with a loop
        for i in range(lo, hi + 1):
            j = k - i
            row[j] += f_sequence[i]
        a_ub.append(row)
        b_ub.append(rhs)
    # non‑negativity constraints  b >= 0  →  -I b <= 0
    a_ub = np.vstack([a_ub, -np.eye(n)])
    b_ub = np.hstack([b_ub, np.zeros(n)])
    # Solve with a modest time limit so that a single LP never blocks
    result = linprog(c, A_ub=a_ub, b_ub=b_ub, options={'time_limit': 5.0, 'disp': False})
    if result.success:
        return result.x
    return None

def get_good_direction_to_move_into(sequence):
    """
    Perform one LP step of the classic SOTA approach:
    given a sequence f, find a sequence g that maximises the total mass
    while keeping the convolution f * g below the current bound.
    Return a convex combination (1‑t)·f + t·g with a small step size t.
    If the LP fails we return None.
    """
    n = len(sequence)
    sum_sequence = float(np.sum(sequence))
    # Normalise to the standard SOTA scaling (sum = sqrt(2 n))
    if sum_sequence == 0:
        return None
    factor = np.sqrt(2.0 * n) / sum_sequence
    f_norm = sequence * factor
    rhs = np.max(np.convolve(f_norm, f_norm))
    g = solve_convolution_lp(f_norm, rhs)
    if g is None:
        return None
    sum_g = float(np.sum(g))
    if sum_g == 0:
        return None
    # Rescale g to the same normalisation as f
    g_norm = g * (np.sqrt(2.0 * n) / sum_g)
    # Small step towards g
    t = 0.05
    new_seq = (1.0 - t) * sequence + t * g_norm
    return new_seq

"""
Hybrid optimiser for the convolution‑maximisation problem.

The algorithm works as follows
--------------------------------
* Initialise a fairly smooth, exponentially decaying distribution.
* Repeatedly apply a cheap *projected gradient* step that directly
  reduces the currently maximal convolution entry.
  The gradient of the active entry is simply a reflected copy of the
  sequence (factor 2 because each unordered pair appears twice).
* With a smaller probability we perform the classic LP “direction”
  step (the one used by the previous SOTA) which injects additional
  mass while preserving the current bound.
* All moves are projected onto the feasible set (non‑negative,
  re‑normalised to sum = 1).  The best sequence seen so far is kept.
* The loop stops after ``budget_s`` seconds and returns the best
  sequence as a plain Python list.
"""

import time
import numpy as np
from scipy import optimize

linprog = optimize.linprog


def _score(seq: np.ndarray) -> float:
    """Fast replica of ``evaluate_sequence`` without the defensive checks."""
    b = np.convolve(seq, seq)
    max_b = b.max()
    s = seq.sum()
    if s < 0.01:          # mirrors the “sum too small” guard
        return np.inf
    return float(2.0 * len(seq) * max_b / (s * s))


def solve_convolution_lp(f_seq: np.ndarray, rhs: float):
    """
    Solve   maximise Σ b_j    s.t.   (f * b)_k ≤ rhs,   b ≥ 0.
    Returns the optimal b as a NumPy array or ``None`` if the LP fails.
    """
    n = len(f_seq)
    c = -np.ones(n)                     # maximise Σ b_j  → minimise -Σ b_j
    a_ub = []
    b_ub = []

    # Build the convolution constraints: for each k we need Σ_i f_i b_{k-i} ≤ rhs
    for k in range(2 * n - 1):
        row = np.zeros(n)
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            j = k - i
            row[j] += f_seq[i]
        a_ub.append(row)
        b_ub.append(rhs)

    # non‑negativity: –I·b ≤ 0
    a_ub = np.vstack([a_ub, -np.eye(n)])
    b_ub = np.hstack([b_ub, np.zeros(n)])

    try:
        res = linprog(c, A_ub=a_ub, b_ub=b_ub,
                      options={"time_limit": 5.0, "disp": False})
    except Exception:                # linprog version differences
        return None

    if res.success:
        return res.x
    return None


def get_good_direction_to_move_into(seq: np.ndarray):
    """
    One LP‑based “direction” step (the SOTA primitive).

    Normalises ``seq`` to the usual SOTA scale (Σ = √(2n)),
    solves the LP to maximise the total mass under the current convolution
    bound and returns a convex combination (1‑t)·seq + t·g with a small t.
    Returns ``None`` if the LP is infeasible.
    """
    n = len(seq)
    sum_seq = float(seq.sum())
    if sum_seq == 0.0:
        return None

    # standard SOTA normalisation
    factor = np.sqrt(2.0 * n) / sum_seq
    f_norm = seq * factor

    rhs = np.max(np.convolve(f_norm, f_norm))
    g = solve_convolution_lp(f_norm, rhs)
    if g is None:
        return None

    sum_g = float(g.sum())
    if sum_g == 0.0:
        return None

    # re‑normalise g to the same scaling as f_norm
    g_norm = g * (np.sqrt(2.0 * n) / sum_g)

    # small step towards the LP solution
    t = 0.05
    new_seq = (1.0 - t) * seq + t * g_norm
    return new_seq


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) yields a low
    ``evaluate_sequence`` value.  The search runs for roughly ``budget_s``
    seconds, using the hybrid method described in the module docstring.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5  # tiny safety margin

    # Length – large enough to give a fine discretisation, but cheap to convolve.
    n = int(rng.integers(1500, 2500))

    def random_exponential():
        """Create a normalised exponentially‑decaying vector with tiny noise."""
        c = rng.uniform(0.5, 5.0)          # decay strength
        p = np.exp(-c / n)                # geometric base
        a = np.power(p, np.arange(n))
        a = a / a.sum()
        a += rng.random(n) * 1e-4         # break perfect symmetry
        a = np.maximum(a, 0.0)
        a /= a.sum()
        return a

    # Initial sequence and bookkeeping
    cur_seq = random_exponential()
    cur_score = _score(cur_seq)
    best_seq = cur_seq.copy()
    best_score = cur_score
    print(f"[init] n={n}, initial score={best_score:.6f}")

    while time.time() < deadline:
        # -----------------------------------------------------------------
        # 60 % of the time we use the cheap projected‑gradient primitive,
        # 40 % we fall back to the LP primitive.
        # -----------------------------------------------------------------
        if rng.random() < 0.6:
            # ---- Projected gradient step on the active maximal index ----
            conv = np.convolve(cur_seq, cur_seq)
            max_idx = int(conv.argmax())

            # gradient of the active convolution entry (factor 2 because of symmetry)
            grad = np.zeros_like(cur_seq)
            i_array = np.arange(n)
            j_array = max_idx - i_array
            mask = (j_array >= 0) & (j_array < n)
            grad[mask] = 2.0 * cur_seq[j_array[mask]]

            improved = False
            # try a few step‑size candidates, stop at the first improvement
            for step in (0.5, 0.2, 0.1, 0.05, 0.02, 0.01):
                cand = cur_seq - step * grad
                cand = np.maximum(cand, 0.0)
                if cand.sum() == 0.0:
                    continue
                cand /= cand.sum()
                cand_score = _score(cand)
                if cand_score < cur_score:
                    cur_seq = cand
                    cur_score = cand_score
                    improved = True
                    break

            if not improved:
                # occasional tiny random jitter to shake off flat plateaus
                if rng.random() < 0.02:
                    i = rng.integers(n)
                    delta = rng.random() * 0.01 * cur_seq[i]
                    j = rng.integers(n)
                    if i != j:
                        cur_seq[i] = max(0.0, cur_seq[i] - delta)
                        cur_seq[j] += delta
                        cur_seq /= cur_seq.sum()
                        cur_score = _score(cur_seq)
        else:
            # ---- Classical LP direction step (SOTA primitive) ----
            cand = get_good_direction_to_move_into(cur_seq)
            if cand is not None:
                cand = np.asarray(cand, dtype=float)
                cand_score = _score(cand)
                if cand_score < cur_score:
                    cur_seq = cand
                    cur_score = cand_score

        # -----------------------------------------------------------------
        # Keep the best ever seen
        # -----------------------------------------------------------------
        if cur_score < best_score:
            best_score = cur_score
            best_seq = cur_seq.copy()
            elapsed = time.time() - start
            print(f"[update] new best {best_score:.6f} after {elapsed:.1f}s")

        # Very occasional full restart to explore a new basin
        if rng.random() < 0.001:
            cur_seq = random_exponential()
            cur_score = _score(cur_seq)

    # Return a plain Python list for the external harness
    return best_seq.tolist()
