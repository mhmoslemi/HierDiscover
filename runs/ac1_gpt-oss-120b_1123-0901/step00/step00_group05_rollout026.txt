import numpy as np, time
from scipy.optimize import linprog

def solve_convolution_lp(f, bound):
    n = len(f)
    # maximize sum(b)  -> minimize -sum(b)
    c = -np.ones(n)
    # Build A_ub, b_ub
    # conv constraint: for each k, sum_{i} f[i] * b[k-i] <= bound
    # This is linear in b: coefficient of b[j] = f[k-j] (if 0<=k-j<n)
    rows = []
    b_vec = []
    for k in range(2*n - 1):
        row = np.zeros(n)
        # compute indices j where 0 <= k - j < n  => max(0, k-(n-1)) <= j <= min(k, n-1)
        j_start = max(0, k - (n - 1))
        j_end = min(k, n - 1)
        # For these j, coefficient is f[k - j]
        ks = k - np.arange(j_start, j_end + 1)
        row[j_start:j_end + 1] = f[ks]
        rows.append(row)
        b_vec.append(bound)

    rows.append(-np.eye(n))
    b_vec.append(np.zeros(n))

    b_ub = np.concatenate(b_vec)
    # Solve
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    if res.success:
        return res.x
    return None

"""
Propose a better step‑function using an iterative LP‑driven smoothing scheme.

The algorithm works with a modest length (≈200) to keep the LP cheap.
Starting from a uniform (flat) sequence – which gives the baseline score
2.0 – we repeatedly:

1. Compute the current convolution bound M = max(a*a).
2. Solve a linear program that **maximises the ℓ1‑norm** of a new
   sequence `b` while keeping the mixed convolution `a*b` bounded by M.
   The LP naturally pushes mass into indices where `a` is small.
3. Rescale `b` to the same total sum as `a` and blend the two sequences:
   `a ← (1‑t)·a + t·b`.  A back‑tracking line‑search on `t` (starting at
   0.5) accepts the step only if the evaluation score improves.
4. Keep the best sequence seen so far.

Randomised tiny redistributions are used when the LP fails or when a
step does not improve the score, helping the search escape stagnation.
The whole loop runs until the allotted time is exhausted (≤ budget_s
seconds) and finally returns the best sequence discovered.
"""

import time
import numpy as np
from scipy.optimize import linprog


def _solve_convolution_lp(f_seq: np.ndarray, bound: float):
    """
    Linear programme:
        maximise   Σ b_j
        subject to conv(f_seq, b) ≤ bound   (for every shift)
                   b_j ≥ 0
    Returns the optimal b as a NumPy array or None on failure.
    """
    n = f_seq.size
    # maximise sum(b)  →  minimise -sum(b)
    c = -np.ones(n)

    # Build inequality matrix A_ub x ≤ b_ub representing conv(f,b) ≤ bound.
    rows = []
    rhs = []

    # For each possible sum index k = 0 … 2n‑2 we have
    #   Σ_j f_seq[k‑j]·b_j  ≤  bound
    for k in range(2 * n - 1):
        row = np.zeros(n)
        # indices j that appear in this convolution entry
        j_start = max(0, k - (n - 1))
        j_end = min(k, n - 1)
        # coefficients are f_seq[k‑j]
        idx = k - np.arange(j_start, j_end + 1)
        row[j_start:j_end + 1] = f_seq[idx]
        rows.append(row)
        rhs.append(bound)

    # Non‑negativity of b  →  -b ≤ 0
    rows.append(-np.eye(n))
    rhs.append(np.zeros(n))

    A_ub = np.vstack(rows)
    b_ub = np.concatenate(rhs)

    # Solve with the high‑performance HiGHS interior‑point method.
    res = linprog(
        c,
        A_ub=A_ub,
        b_ub=b_ub,
        bounds=[(0, None)] * n,
        method="highs",
        options={"presolve": True},
    )
    if res.success:
        return res.x
    return None


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """Return a step‑function (list of non‑negative heights) with a low score."""
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin so we always finish before the hard timeout
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------ #
    # 1.  Problem size – a trade‑off between expressiveness and LP cost.
    # ------------------------------------------------------------------ #
    n = 200                         # length of the candidate sequence
    a = np.ones(n, dtype=float)    # start from the flat (uniform) sequence

    # Initial evaluation (used only for bookkeeping)
    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    cur_score = best_score
    print(f"[{time.time() - start_time:.1f}s] start score = {best_score:.6f}")

    # ------------------------------------------------------------------ #
    # 2.  Main optimisation loop
    # ------------------------------------------------------------------ #
    iteration = 0
    while time.time() < deadline:
        iteration += 1

        # current convolution bound
        conv = np.convolve(a, a)
        M = conv.max()

        # -------------------------------------------------------------- #
        # 2.1  LP that tries to increase the ℓ1‑norm without raising M
        # -------------------------------------------------------------- #
        b = _solve_convolution_lp(a, M)
        if b is None:
            # LP failed – perform a tiny random mass transfer.
            i, j = np.random.randint(n, size=2)
            delta = 0.02 * a[i]
            a[i] = max(a[i] - delta, 0.0)
            a[j] += delta
            continue

        # Rescale b so that its total mass equals the current mass of a.
        sum_a = a.sum()
        sum_b = b.sum()
        if sum_b == 0.0:      # degenerate – skip this iteration
            continue
        g = b * (sum_a / sum_b)

        # -------------------------------------------------------------- #
        # 2.2  Blend a towards g.  Accept only if the score improves.
        # -------------------------------------------------------------- #
        t = 0.5                      # initial step size
        improved = False
        while t > 1e-4:
            a_candidate = (1.0 - t) * a + t * g
            a_candidate = np.maximum(a_candidate, 0.0)   # safety
            cand_score = evaluate_sequence(a_candidate.tolist())
            if cand_score < cur_score:
                a = a_candidate
                cur_score = cand_score
                improved = True
                if cand_score < best_score:
                    best_score = cand_score
                    best_seq = a_candidate.copy()
                    print(
                        f"[{time.time() - start_time:.1f}s] "
                        f"iter {iteration:4d} new best {best_score:.6f}"
                    )
                break
            t *= 0.5  # back‑track

        # If no improvement was possible, inject a small random perturbation.
        if not improved:
            i, j = np.random.randint(n, size=2)
            delta = 0.02 * a[i]
            a[i] = max(a[i] - delta, 0.0)
            a[j] += delta

    # ------------------------------------------------------------------ #
    # 3.  Return the best sequence discovered.
    # ------------------------------------------------------------------ #
    return best_seq.tolist()
