def propose_candidate(seed=42, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0
    n = 1200  # maybe adjust based on time
    # initial shape
    idx = np.arange(n)
    centre = (n-1)/2
    dist = np.abs(idx - centre)
    a = dist.astype(np.float64) + 1e-6  # avoid zeroes?
    a = a / a.sum()
    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    # pre compute target constant average for conv length
    length = 2*n - 1
    target = np.full(length, 1.0/length, dtype=np.float64)
    it = 0
    lr = 0.3
    while time.time() < deadline:
        # gradient step
        b = np.convolve(a, a)
        diff = b - target
        grad = 4.0 * np.convolve(diff, a[::-1], mode='full')
        grad = grad[n-1:n-1+n]  # slice length n
        # normalized step
        max_abs = np.max(np.abs(grad)) + 1e-12
        step = lr * grad / max_abs
        a = a - step
        # enforce non-negativity and renormalize sum to 1
        a = np.maximum(a, 0)
        s = a.sum()
        if s == 0:
            # reinitialize to uniform
            a = np.full(n, 1.0/n)
        else:
            a = a / s
        # evaluate
        score = evaluate_sequence(a.tolist())
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            # optionally print
            print(f"Iter {it}: improved score {best_score:.6f}")
        # decay learning rate occasionally
        if it % 1000 == 0 and it > 0:
            lr *= 0.9
        # occasional random perturb
        if it % 5000 == 0 and it > 0:
            # random small noise to avoid local minima
            a = a + np.random.randn(n) * 0.01
            a = np.maximum(a, 0)
            a /= a.sum()
        it += 1
    return best_seq.tolist()

"""
Propose a coefficient sequence that minimizes evaluate_sequence.

The evaluation metric is:
    2 * n * max(conv(a, a)) / (sum(a)^2)

Because the metric is homogeneous of degree 0, we may fix sum(a)=1.
The goal is to make the self‑convolution as flat as possible,
i.e. to approach a uniform distribution whose value is 1/(2n‑1).
A simple gradient‑descent scheme on the L2 distance between the current
convolution and the uniform target drives the sequence toward this flat
shape.  After each gradient step we project back onto the probability
simplex (non‑negative entries summing to 1).  Small random jitter and a
final local‑search phase help escape shallow minima.

The algorithm works for any length; a length of about 2000 gives a good
trade‑off between runtime and quality.  The routine runs until the
time budget expires and returns the best sequence found.
"""

import time
import numpy as np

def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the probability simplex (sum = 1)."""
    # Simple renormalisation works because we never get a zero‑vector after
    # clipping (the sum is forced to be positive by a tiny epsilon if needed).
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0.0:
        # fallback to uniform distribution
        return np.full_like(v, 1.0 / v.size)
    return v / s


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Generate a sequence of non‑negative heights that (hopefully) improves
    upon the current best bound.  The function respects the given time
    budget and returns the best sequence discovered.
    """
    np.random.seed(seed)
    # leave a few seconds margin for the caller
    deadline = time.time() + budget_s - 5.0

    # --- parameters -------------------------------------------------
    n = 2000                     # length of the sequence (tunable)
    length_conv = 2 * n - 1      # length of the linear convolution
    target = np.full(length_conv, 1.0 / length_conv, dtype=np.float64)  # flat target
    step_size = 0.1              # initial step magnitude
    min_step = 1e-6
    # ---------------------------------------------------------------

    # start from the uniform distribution (sum = 1)
    a = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"initial score: {best_score:.6f}")

    it = 0
    while time.time() < deadline:
        # 1. compute current convolution
        conv = np.convolve(a, a)                     # shape (2n‑1,)

        # 2. gradient of L2 loss to a uniform target
        diff = conv - target                         # shape (2n‑1,)
        # grad_i = 4 * Σ_k diff_k * a_{k-i}
        grad_full = 4.0 * np.convolve(diff, a[::-1], mode='full')
        grad = grad_full[n - 1:n - 1 + n]            # keep only the n relevant components

        # 3. take a normalized step
        norm = np.linalg.norm(grad) + 1e-12
        step = step_size * grad / norm

        a = a - step
        a = _project_to_simplex(a)

        # occasional random jitter to avoid stagnation
        if it % 1000 == 0 and it > 0:
            jitter = np.random.randn(n) * 0.001
            a = _project_to_simplex(a + jitter)

        # evaluate the new candidate
        try:
            score = evaluate_sequence(a.tolist())
        except Exception:
            score = float('inf')

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"iter {it:6d}: improved score {best_score:.6f}")
            # slowly reduce step size when a new record is found
            step_size = max(step_size * 0.98, min_step)

        # periodic step‑size decay
        if it % 5000 == 0 and it > 0:
            step_size = max(step_size * 0.9, min_step)

        it += 1

    # ----- final local refinement (hill‑climbing) -------------------
    # Use small random moves on the best sequence found so far.
    a = best_seq.copy()
    while time.time() < deadline:
        idx = np.random.randint(n)
        delta = (np.random.rand() - 0.5) * 0.02   # modest perturbation
        a_new = a.copy()
        a_new[idx] = max(0.0, a_new[idx] + delta)
        a_new = _project_to_simplex(a_new)

        try:
            sc = evaluate_sequence(a_new.tolist())
        except Exception:
            continue

        if sc < best_score:
            best_score = sc
            best_seq = a_new.copy()
            a = a_new
            print(f"refine: improved score {best_score:.6f}")
        else:
            # occasionally accept a move to keep exploration alive
            if np.random.rand() < 0.001:
                a = a_new

    print(f"final best score: {best_score:.6f}")
    return best_seq.tolist()
