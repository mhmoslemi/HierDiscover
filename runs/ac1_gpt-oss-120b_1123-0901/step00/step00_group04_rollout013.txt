"""
Propose a step‑function (non‑negative sequence) that minimizes the
evaluation function

    E(a) = 2 * n * max(conv(a, a)) / (sum(a) ** 2)

where conv is the linear convolution (zero padding).  The naïve uniform
sequence gives E = 2 for any length, but a decreasing exponential shape can
reduce the peak of the self‑convolution.  Starting from such a shape we
apply a simple projected gradient descent that always keeps the L1‑norm
fixed (the value of the sum does not matter because the score is scale‑
invariant).  The gradient of the maximal convolution entry is easily
computed; we move mass away from the locations that contribute most to the
current peak, clip negatives to zero and renormalise.  An adaptive step
size together with occasional random restarts (different exponential
rates) lets the algorithm escape poor local minima.  The whole procedure
runs until the allotted wall‑time (≈ 1000 s) is exhausted and returns the
best sequence found.

The implementation uses FFT‑based convolution (≈ O(N log N)) so that many
iterations are affordable even for N ≈ 6000.
"""

import time
import numpy as np
from typing import List

# The evaluator is assumed to be already imported in the execution
# environment:
#    from <module> import evaluate_sequence


def _conv_self_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution of a with itself using real FFT.
    Returns an array of length 2*len(a)-1.
    """
    A = np.fft.rfft(a, n=fft_len)
    c = np.fft.irfft(A * A, n=fft_len)
    return c[: 2 * a.size - 1]


def _initial_exponential(n: int, target_sum: float, c: float) -> np.ndarray:
    """
    Exponential (geometric) start: a_i ∝ exp(-c * i / n)
    then scaled to obtain the prescribed L1 norm.
    """
    idx = np.arange(n, dtype=np.float64)
    raw = np.exp(-c * idx / n)          # decreasing exponential
    scale = target_sum / raw.sum()
    return raw * scale
