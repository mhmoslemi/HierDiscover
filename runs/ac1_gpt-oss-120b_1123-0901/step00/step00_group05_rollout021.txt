"""
Gradient descent on the variance of the full convolution.

The loss L(a) = ½∑_k (conv(a,a)_k - μ)², where μ is the mean of the
convolution, is differentiable and its gradient can be written as a
single convolution:

    ∇L_i = 4·∑_k (conv_k - μ)·a_{k-i}
          = 4·( (conv - μ)  *  a[::-1] )[n-1+i] .

For a fixed ℓ₁‑norm (we keep Σ a_i = √(2·n) – the same normalisation used
by the reference algorithm) the evaluation function
    evaluate_sequence(a) = 2·n·max(conv(a,a))/Σa²
reduces to the maximum entry of the convolution, since Σa² = 2·n.

The algorithm starts from a random non‑negative vector (or from the
provided `height_sequence_1`), repeatedly:
  * computes the convolution and its mean,
  * evaluates the gradient above,
  * performs a multiplicative update a ← a·exp(−η·∇L) which keeps all
    entries non‑negative,
  * re‑normalises the vector to the target ℓ₁‑norm,
  * keeps the best sequence seen so far.

The step size η is adapted: it is increased after a successful improvement
and decreased otherwise.  Random jitter is injected if the search stalls.
The method runs for the allotted time (≈ 1000 s) and finally returns the
best sequence found.
"""

import numpy as np
import time

def _project_to_target_sum(x: np.ndarray, target: float) -> np.ndarray:
    """Scale a non‑negative vector to have the given ℓ₁‑norm."""
    s = x.sum()
    if s <= 0.0:
        # extremely unlikely, but fall back to uniform
        x = np.full_like(x, target / len(x))
        return x
    return x * (target / s)


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a sequence of non‑negative heights that (hopefully) beats the
    current best upper bound.  The implementation follows the description
    above.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0  # leave a little safety margin

    # ------------------------------------------------------------------
    # Determine the length n to work with.
    # Prefer the length of the known construction, otherwise pick a
    # reasonable default (~6000).
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        base_seq = list(height_sequence_1)
        n = len(base_seq)
    else:
        n = 6000                     # a safe size for the time budget
        base_seq = None

    # Target ℓ₁‑norm (the normalisation used by the SOTA code)
    target_sum = np.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # Initialise the search vector.
    # If a previous construction is available, start from it (after
    # normalising).  Otherwise start from a random positive vector.
    # ------------------------------------------------------------------
    if base_seq is not None:
        a = np.asarray(base_seq, dtype=float)
        a = _project_to_target_sum(a, target_sum)
    else:
        a = np.random.rand(n) + 1e-8      # avoid exact zeros
        a = _project_to_target_sum(a, target_sum)

    best_a = a.copy()
    best_score = evaluate_sequence(best_a.tolist())

    # Parameters for the optimisation loop
    eta = 0.1          # learning rate for the multiplicative update
    no_improve = 0    # counter for stagnation detection
    report_interval = 30.0   # seconds between progress reports
    next_report = start_time + report_interval

    while time.time() < deadline:
        # --------------------------------------------------------------
        # 1) Full convolution and its mean
        # --------------------------------------------------------------
        conv = np.convolve(a, a)               # length 2n‑1
        mu = conv.mean()

        # --------------------------------------------------------------
        # 2) Gradient of the variance loss
        #    ∇L_i = 4 * ( (conv - mu) * a[::-1] )[n-1+i]
        # --------------------------------------------------------------
        residual = conv - mu
        # convolution of residual with reversed a
        grad_full = np.convolve(residual, a[::-1], mode='full')
        grad = 4.0 * grad_full[n - 1 : n - 1 + n]   # length n

        # --------------------------------------------------------------
        # 3) Multiplicative (positive) update and projection onto the simplex
        # --------------------------------------------------------------
        # Guard against overflow/underflow in the exponential
        max_exp = 50.0
        grad = np.clip(grad, -max_exp, max_exp)

        a_new = a * np.exp(-eta * grad)

        # Non‑negativity is automatic; re‑normalise the ℓ₁‑norm
        a_new = _project_to_target_sum(a_new, target_sum)

        # --------------------------------------------------------------
        # 4) Evaluate and possibly accept the new point
        # --------------------------------------------------------------
        try:
            new_score = evaluate_sequence(a_new.tolist())
        except Exception:
            # Numerical issues – reject this candidate
            new_score = np.inf

        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()
            a = a_new                     # continue from the improved point
            eta = min(eta * 1.2, 1.0)     # become a bit more aggressive
            no_improve = 0
        else:
            # No improvement – be more conservative
            a = a_new                     # still move (helps escape plateaus)
            eta *= 0.7
            no_improve += 1

        # Occasionally inject a small random jitter if we have stalled
        if no_improve >= 300:
            jitter = np.random.randn(n) * (target_sum * 1e-4)
            a = np.maximum(a + jitter, 0.0)
            a = _project_to_target_sum(a, target_sum)
            no_improve = 0

        # --------------------------------------------------------------
        # 5) Periodic progress report
        # --------------------------------------------------------------
        now = time.time()
        if now >= next_report:
            print(f"[{now - start_time:6.1f}s] best_score={best_score:.6f} "
                  f"eta={eta:.4f}")
            next_report = now + report_interval

    # ------------------------------------------------------------------
    # Fallback: if the gradient descent failed to improve over the uniform
    # baseline (score ≈ 2), return a simple exponential sequence that is
    # known to give ~1.8.
    # ------------------------------------------------------------------
    if best_score >= 2.0:

        # where x* solves e^x = 1 + 2x (x*≈1.256)
        x_star = 1.25643
        p_opt = x_star / n
        r = 1.0 - p_opt
        geo = np.array([r ** i for i in range(n)], dtype=float)
        geo = _project_to_target_sum(geo, target_sum)
        fallback_score = evaluate_sequence(geo.tolist())
        if fallback_score < best_score:
            best_a = geo
            best_score = fallback_score
        print(f"Fallback exponential sequence used, score={best_score:.6f}")

    print(f"Finished. Best score: {best_score:.6f}")
    return best_a.tolist()
