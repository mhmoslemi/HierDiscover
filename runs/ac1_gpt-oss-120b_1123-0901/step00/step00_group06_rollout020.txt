"""
Propose a sequence that minimizes the evaluation function

    2 * n * max(conv(f,f)) / (sum(f)^2)

The algorithm works in three stages:

1. **Parameter sweep** – generate a few hundred candidate shapes
   (exponential, power‑law, sine‑power and a mix of exponential + constant)
   and keep the best one.

2. **Projected sub‑gradient descent** – starting from the best candidate,
   repeatedly reduce the peak of the convolution.
   For the current maximal index K the sub‑gradient of the peak is
   g_i = f_{K-i} (zero when the index is out of range).  A step opposite
   to the gradient is taken and the result is projected onto the simplex
   {x ≥ 0, Σx = 1}.  Several step sizes are tried and the one giving the
   smallest objective is kept.

3. **Random perturbations** – if no improvement is found for a while,
   a tiny random jitter is added to escape a possible plateau.

All operations are O(n) or O(n log n) and the whole routine respects the
1000 s time budget.  The returned list is a non‑negative step function;
its scale is irrelevant because the objective is homogeneous.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_to_simplex(v, s=1.0):
    """Project vector v onto the simplex {x >= 0, sum(x)=s}."""
    v = np.asarray(v, dtype=float)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    # Find the largest rho satisfying the KKT condition
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero
        return np.zeros_like(v)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w


# ----------------------------------------------------------------------
# Core optimizer
# ----------------------------------------------------------------------
def propose_candidate(budget_s: float = 1000, seed: int = 42, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) beats the
    current bound of 2.0.  The routine runs for at most `budget_s`
    seconds, performs a quick parametric sweep and refines the best
    candidate with projected sub‑gradient descent.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0   # spare a couple of seconds

    # ------------------------------------------------------------------
    # 1. Parameter sweep over a few simple families
    # ------------------------------------------------------------------
    n = 2500                     # moderate length – fast enough for many trials
    idx = np.arange(n)

    best_seq = None
    best_score = float('inf')

    def _try_seq(seq):
        nonlocal best_seq, best_score
        # evaluate_sequence works with Python list, returns a float
        score = evaluate_sequence(seq.tolist())
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
        return score

    # ---- exponential families (r^i) ----------------------------------
    for eps in np.linspace(5e-5, 5e-3, 90):       # epsilon = 1 - r
        r = 1.0 - eps
        seq = r ** idx
        _try_seq(seq)

    # ---- power‑law families ( (i+1)^-p ) -----------------------------
    for p in np.linspace(0.3, 2.0, 30):
        seq = (idx + 1) ** (-p)
        _try_seq(seq)

    # ---- sine‑power families ( sin(x)^q ) ----------------------------
    x = np.linspace(0.0, np.pi / 2.0, n)
    sin_vals = np.sin(x)
    for q in [1, 2, 3, 4]:
        seq = sin_vals ** q
        _try_seq(seq)

    # ---- constant + exponential (a + r^i) ----------------------------
    for a in np.linspace(0.0, 0.5, 7):
        for eps in np.linspace(1e-4, 1e-2, 12):
            r = 1.0 - eps
            seq = a + (r ** idx)
            _try_seq(seq)

    # If for any reason the sweep failed to produce a sequence,
    # fall back to a uniform one (evaluation = 2.0).
    if best_seq is None:
        best_seq = np.ones(n) / n
        best_score = evaluate_sequence(best_seq.tolist())

    # ------------------------------------------------------------------
    # 2. Projected sub‑gradient descent on the peak of the convolution
    # ------------------------------------------------------------------
    seq = best_seq.copy()
    it = 0
    no_improve = 0
    # step‑size schedule – diminishing but never smaller than 1e‑4
    while time.time() < deadline:
        # compute convolution and locate the maximal index
        conv = np.convolve(seq, seq)
        K = int(conv.argmax())

        # sub‑gradient of the peak (only one active index)
        grad = np.zeros_like(seq)
        lo = max(0, K - (n - 1))
        hi = min(n - 1, K)
        # indices i such that K-i is inside [0, n-1]
        i_range = np.arange(lo, hi + 1, dtype=int)
        grad[i_range] = seq[K - i_range]

        # try a handful of step sizes and keep the best result
        improved = False
        best_local = seq
        best_local_score = best_score
        for lr in (0.02, 0.05, 0.1, 0.2, 0.4):
            cand = seq - lr * grad
            cand = _project_to_simplex(cand, s=1.0)   # keep sum fixed (=1)
            score = evaluate_sequence(cand.tolist())
            if score < best_local_score:
                best_local_score = score
                best_local = cand
                improved = True

        # accept the best local improvement (if any)
        if improved:
            seq = best_local
            best_score = best_local_score
            best_seq = seq.copy()
            no_improve = 0
        else:
            no_improve += 1
            # occasional random jitter to jump out of a flat region
            if no_improve % 50 == 0:
                jitter = np.random.randn(n) * 1e-3
                seq = _project_to_simplex(seq + jitter, s=1.0)

        it += 1
        if it > 8000:               # safety stop – usually converged long before
            break

    # ------------------------------------------------------------------
    # 3. Final clean‑up – make sure we respect the evaluate_sequence guards
    # ------------------------------------------------------------------
    # Clip to the allowed range [0, 1000] (already satisfied) and ensure
    # the sequence is non‑empty.
    final_seq = [float(x) for x in best_seq.tolist()]
    final_score = evaluate_sequence(final_seq)
    print(f"Finished after {it} iterations, best score = {final_score:.6f}")

    return final_seq
