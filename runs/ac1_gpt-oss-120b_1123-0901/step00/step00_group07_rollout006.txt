"""
Simulated‑annealing improvement of a non‑negative step function.

The algorithm works with a fixed length `n` (chosen randomly in a reasonable
range) and a total sum equal to √(2 n) – this normalisation makes the
evaluation function equal to the maximal entry of the self‑convolution.
Starting from a (possibly noisy) uniform sequence we repeatedly move a
small amount of mass from one coordinate to another.  Moves that lower the
objective are always accepted, otherwise they are accepted with a
temperature‑dependent probability.  The temperature is slowly reduced;
occasional random jitter and restarts avoid getting stuck in local minima.
The best sequence found before the time budget expires is returned.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Helper: fast evaluation (identical to evaluate_sequence for admissible data)
def _fast_evaluate(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq))/sum(seq)^2 ."""
    n = len(seq)
    s = seq.sum()
    if s < 0.01:                     # protection used in evaluate_sequence
        return float('inf')
    conv = np.convolve(seq, seq)
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑value step‑function using simulated annealing.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Maximum running time in seconds.
    kwargs : dict
        Ignored – present for compatibility with the evaluation harness.

    Returns
    -------
    list[float]
        The best sequence discovered before the time limit.
    """
    np.random.seed(seed)
    random.seed(seed)

    # reserve a little margin for the final return/printing
    deadline = time.time() + budget_s - 0.2

    best_seq = None
    best_score = float('inf')
    total_iters = 0

    # ------------------------------------------------------------------
    # Core SA loop – we restart many times until the budget expires.
    while time.time() < deadline:
        # ---- pick a length -------------------------------------------------
        # The existing construction has length 3428; we also allow shorter
        # runs for speed.  Lengths between 800 and 1800 work well.
        n = np.random.randint(800, 1800)

        # ---- initialise a starting sequence --------------------------------
        if 'height_sequence_1' in globals() and np.random.rand() < 0.5:
            # reuse part of the previous best construction (if available)
            base = np.asarray(height_sequence_1, dtype=float)
            if len(base) >= n:
                start = np.random.randint(0, len(base) - n + 1)
                a = base[start:start + n].copy()
            else:
                # pad with zeros and then renormalise
                a = np.concatenate([base, np.zeros(n - len(base))])
        else:
            # start from an (almost) uniform distribution plus a small perturbation
            a = np.full(n, np.sqrt(2.0 * n) / n, dtype=float)
            a += np.random.normal(scale=0.01, size=n)   # tiny noise to break symmetry
            a = np.maximum(a, 0.0)

        # renormalise the total sum to √(2 n) (scaling does not change the metric)
        target_sum = np.sqrt(2.0 * n)
        a *= target_sum / a.sum()

        cur = a.astype(float)
        cur_score = _fast_evaluate(cur)

        # keep the best of the very first state
        if cur_score < best_score:
            best_score = cur_score
            best_seq = cur.tolist()

        # ---- annealing schedule -------------------------------------------
        # initial temperature proportional to the current score
        T0 = max(0.1, cur_score * 0.15)
        # geometric cooling factor – tuned for roughly 10⁵‑10⁶ iterations per run
        cooling = 0.9999

        # counter for stagnation detection
        stagnation = 0

        # inner SA loop
        while time.time() < deadline:
            total_iters += 1
            # ----- propose a move ------------------------------------------------
            i, j = np.random.choice(n, 2, replace=False)
            if cur[i] <= 0.0:
                continue
            # move a random fraction (0‑10 %) of mass from i to j
            frac = np.random.rand() * 0.10
            delta = cur[i] * frac
            new = cur.copy()
            new[i] -= delta
            new[j] += delta

            # ----- evaluate ------------------------------------------------------
            new_score = _fast_evaluate(new)

            # ----- Metropolis acceptance ----------------------------------------
            if new_score < cur_score:
                accept = True
            else:
                # temperature decays after each attempted move
                T = T0
                if T > 0:
                    prob = math.exp(-(new_score - cur_score) / T)
                else:
                    prob = 0.0
                accept = np.random.rand() < prob

            if accept:
                cur = new
                cur_score = new_score
                stagnation = 0
                if new_score < best_score:
                    best_score = new_score
                    best_seq = new.tolist()
            else:
                stagnation += 1

            # ----- cooling -------------------------------------------------------
            T0 *= cooling

            # ----- occasional jitter to escape flat regions ----------------------
            if total_iters % 5000 == 0:
                jitter = np.random.normal(scale=0.001 * cur.mean(), size=n)
                cur = np.maximum(cur + jitter, 0.0)
                # renormalise (keeps sum constant)
                cur *= target_sum / cur.sum()
                cur_score = _fast_evaluate(cur)

            # ----- restart if we have been stuck for a while --------------------
            if stagnation > 8000:
                break

        # end of one SA run – loop again with a fresh start if time permits

    # ------------------------------------------------------------------
    # Final reporting
    elapsed = time.time() - (deadline + 0.2 - budget_s)
    print(f"[INFO] SA finished after {elapsed:.1f}s, total iterations {total_iters}")
    print(f"[INFO] Best score achieved: {best_score:.6f} (length {len(best_seq)})")
    return best_seq
