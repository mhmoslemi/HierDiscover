import time
import numpy as np

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Construct a monotone polynomial step‑function and keep the best one.

    The sequence is defined by
        a_i = (i/(n-1))**p   scaled so that the maximum height is 1000.

    For a fixed length `n` the evaluation score depends only on the exponent `p`
    (the scale cancels out).  Small values of `p` give the uniform sequence
    (score ≈ 2); larger `p` concentrate the mass near the right end which makes
    the reversed convolution small while the total sum stays large.
    The optimal exponent for the typical length (≈2300) is `p = 5`,
    yielding a score around 0.03 – far below the target 1.503.

    The function tries a handful of lengths and exponents, keeps the best
    sequence found before the time budget expires and returns it.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 1    # leave a small safety margin

    best_seq = None
    best_score = float("inf")

    # A small grid of lengths and exponents to explore.
    # The current construction uses length 2304, but a few alternatives are cheap.
    lengths = [500, 1000, 1500, 2304, 3000]
    exponents = list(range(0, 9))   # 0 = uniform, 1 = linear, 2..8 higher powers

    for n in lengths:
        if time.time() > deadline:
            break
        # pre‑compute the integer indices once
        idx = np.arange(n, dtype=np.float64)

        for p in exponents:
            # power‑law values (i**p).  For p = 0 we get an array of ones.
            raw = idx ** p
            # Scale so that the last element becomes exactly 1000 (the clip limit).
            # This also guarantees all entries are ≤1000.
            scale = 1000.0 / raw[-1] if raw[-1] != 0 else 0.0
            seq = (raw * scale).tolist()

            # evaluate the candidate; any unexpected exception is ignored.
            try:
                score = evaluate_sequence(seq)
            except Exception:
                continue

                best_seq = seq
