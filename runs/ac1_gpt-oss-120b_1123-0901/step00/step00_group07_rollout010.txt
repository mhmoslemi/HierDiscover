def propose_candidate(seed=...):

    np.random.seed(seed)
    random.seed(seed)
    deadline = time.time() + budget_s - 2
    n = 2000
    # start sequence
    if "height_sequence_1" in globals():
        seq = np.array(height_sequence_1, dtype=float)
        # Normalize sum to 1
        seq = seq / seq.sum()
        # maybe trim or pad to length n?
        if len(seq) > n:
            seq = seq[:n]   # keep first n
        elif len(seq) < n:
            # pad zeros
            seq = np.pad(seq, (0, n-len(seq)), mode='constant')
    else:
        seq = np.full(n, 1.0/n)
    # Ensure nonneg
    seq = np.maximum(seq, 0.0)
    sum_seq = seq.sum()
    if sum_seq < 1e-12:
        seq = np.full(n, 1.0/n)
    else:
        seq = seq / sum_seq
    best_seq = seq.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"Initial score: {best_score}")
    # parameters
    temperature = 0.1
    decay = 0.9995  # per iteration
    iter = 0
    while time.time() < deadline:
        iter += 1
        # compute convolution and max index
        conv = np.convolve(seq, seq)
        max_idx = int(np.argmax(conv))
        # compute contributions for each position to the peak
        idxs = np.arange(len(seq))
        # mask valid indices that contribute to max_idx
        mask = (max_idx - idxs >= 0) & (max_idx - idxs < len(seq))
        contrib = np.zeros_like(seq)
        contrib[mask] = seq[idxs[mask]] * seq[max_idx - idxs[mask]]
        # choose donor index with highest contribution (likely large product)
        donor_candidates = np.where(contrib > np.percentile(contrib, 80))[0]
        if donor_candidates.size == 0:
            donor = np.random.randint(len(seq))
        else:
            donor = np.random.choice(donor_candidates)
        # choose receiver from low contribution indices
        receiver_candidates = np.where(contrib < np.percentile(contrib, 20))[0]
        if receiver_candidates.size == 0:
            receiver = np.random.randint(len(seq))
        else:
            receiver = np.random.choice(receiver_candidates)
        if donor == receiver:
            continue
        # Determine amount to move: based on temperature and donor size
        delta_max = min(seq[donor], temperature * seq[donor])
        if delta_max <= 0:
            continue
        delta = np.random.rand() * delta_max
        # propose new sequence
        new_seq = seq.copy()
        new_seq[donor] -= delta
        new_seq[receiver] += delta
        # evaluate
        score = evaluate_sequence(new_seq.tolist())
        # Accept if better
        if score < best_score:
            best_score = score
            best_seq = new_seq.copy()
            seq = new_seq
            print(f"Iter {iter}, improved score: {best_score:.6f}")
        else:
            # Accept with probability e^{-(score-best)/temp}? optional
            # but we may just reject to keep greedy
            # optionally small chance to accept for diversification
            if np.random.rand() < 0.001:
                seq = new_seq
        # tempering
        temperature *= decay
        # Periodically recompute best by resetting to best if stuck
        if iter % 1000 == 0:
            # small perturbation to escape local minima
            seq = best_seq.copy()
            # random shuffle small amount
            idxs = np.random.choice(len(seq), size=10, replace=False)
            for _ in range(5):
                i, j = np.random.choice(idxs, size=2, replace=False)
                delta = min(seq[i], 0.01) * np.random.rand()
                seq[i] -= delta
                seq[j] += delta

    print(f"Finished after {iter} iterations, best score: {best_score}")
    # Return as list
    return best_seq.tolist()

"""
Propose a candidate sequence of non‑negative heights that minimizes the
evaluation function 2*n*max(conv)/sum^2.

The algorithm starts from a (possibly existing) construction, adds a small
random perturbation and then performs a projected sub‑gradient descent that
targets the peak of the self‑convolution.  In each iteration we:

1. Compute the convolution a*a and locate the index of its maximum.
2. For that index compute the product a[i]·a[max‑i] (zero if out of range);
   this is proportional to the sub‑gradient of the peak.
3. Move a small amount of mass from the index with the largest product
   (largest contribution to the peak) to the index with the smallest product.
4. Project the resulting vector onto the probability simplex (non‑negative,
   sum = 1) to keep the invariance of the objective.
5. Keep the best sequence seen so far.

The step size decays over time and occasional random perturbations are added
to escape plateaus.  The whole search runs until the time budget is exhausted
or the target value (≈1.503) is reached.
"""

import time
import numpy as np

def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i >= 0, sum x_i = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries are non‑positive – return the uniform distribution.
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def compute_score(a: np.ndarray) -> float:
    """Exact score used by evaluate_sequence (without the safety wrappers)."""
    s = a.sum()
    if s < 0.01:
        return np.inf
    max_b = np.max(np.convolve(a, a))
    return float(2.0 * len(a) * max_b / (s * s))


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # 1. Choose a starting sequence
    # ------------------------------------------------------------------
    # Try to use the previously found construction if it exists.
    init = globals().get("height_sequence_1", None)
    if init is not None:
        a = np.array(init, dtype=np.float64)
        # Clip to the limits enforced inside evaluate_sequence.
        a = np.clip(a, 0.0, 1000.0)
    else:
        # fallback: uniform on a moderate size (≈2000)
        a = None

    # If the previous construction is too short/long we adapt its size.
    # We aim for a length between 1500 and 3500 (fast enough and close to
    # the current best length 3428).
    if a is None or a.size < 1500:
        n = 2500
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        n = a.size
        # Ensure we have at least 1500 entries; pad with zeros if needed.
        if n < 1500:
            a = np.pad(a, (0, 1500 - n), "constant")
            n = 1500
        # Normalise to sum = 1 (scaling does not affect the objective)
        a = a / max(a.sum(), 1e-12)

    # Small random perturbation to break symmetry (uniform sequences are
    # stationary points of the squared‑error objective).
    noise = 0.001 * np.random.randn(n)
    a = a + noise
    a = np.maximum(a, 0.0)
    a = proj_simplex(a)

    best_seq = a.copy()
    best_score = compute_score(best_seq)
    print(f"[init] length={n}, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Main optimisation loop (projected sub‑gradient descent on the peak)
    # ------------------------------------------------------------------
    deadline = time.time() + budget_s - 2.0  # leave a tiny margin
    it = 0
    # Step‑size control: start relatively aggressive and decay.
    step_factor = 0.2   # fraction of the donor’s mass to move each step
    decay = 0.9995
    # For occasional exploration
    rand_prob = 0.004

    while time.time() < deadline:
        it += 1

        # ----- compute convolution and locate the current peak -----
        conv = np.convolve(a, a)
        max_idx = int(np.argmax(conv))
        max_val = conv[max_idx]

        # ----- sub‑gradient of the peak w.r.t. a -----
        # product_i = a[i] * a[max_idx - i]   if the partner index exists
        idx = np.arange(n)
        partner = max_idx - idx
        valid = (partner >= 0) & (partner < n)
        prod = np.zeros_like(a)
        prod[valid] = a[idx[valid]] * a[partner[valid]]

        # Choose donor (large contribution) and receiver (small contribution)
        donor = int(np.argmax(prod))
        receiver = int(np.argmin(prod))
        if donor == receiver:
            # fallback to a random different pair
            receiver = (donor + np.random.randint(1, n)) % n

        # Amount to move: a fraction of the donor’s current value,
        # but never more than the donor has.
        delta = step_factor * a[donor]
        delta = min(delta, a[donor])

        # ----- propose new sequence -----
        a_new = a.copy()
        a_new[donor] -= delta
        a_new[receiver] += delta
        # Projection back to the simplex (keeps sum = 1 and non‑negativity)
        a_new = proj_simplex(a_new)

        # ----- compute new score -----
        cur_score = compute_score(a_new)

        # ----- acceptance -----
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a_new.copy()
            a = a_new
            # Slightly increase aggressiveness after a success
            step_factor = min(step_factor * 1.02, 0.5)
            print(f"[{it:07d}] Improve → score={best_score:.6f}")
            if best_score < 1.503:    # reached the target, stop early
                break
        else:
            # Occasionally accept a worse move (simulated‑annealing style)
            if np.random.rand() < 0.001:
                a = a_new
            # otherwise keep the current a unchanged

        # ----- decay step size -----
        step_factor *= decay

        # ----- occasional random reshuffle to escape plateaus -----
        if np.random.rand() < rand_prob:
            # add a tiny Gaussian jitter to all coordinates
            jitter = 0.0005 * np.random.randn(n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = proj_simplex(a)

        # Periodic logging (every ~2000 iterations)
        if it % 2000 == 0:
            elapsed = time.time() - (deadline - budget_s + 2.0)
            print(f"[{it:07d}] time={elapsed:.1f}s, best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Return the best sequence found.
    # ------------------------------------------------------------------
    print(f"[done] iterations={it}, best_score={best_score:.6f}")
    # Convert back to a plain Python list of floats (as required by the judge)
    return best_seq.tolist()
