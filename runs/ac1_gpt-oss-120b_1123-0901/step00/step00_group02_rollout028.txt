"""
Propose a sequence of non‑negative heights that (hopefully) beats the
trivial bound 2.0.

Algorithm
---------
1.  Initialise the sequence with a simple exponential profile
    a_i ∝ q^i (the parameter q is swept to give the best starting point).
    This already reduces the evaluation to ≈1.8 for the target length.

2.  Refine the profile by directly minimising the L∞‑norm of the
    self‑convolution.  We use a smooth surrogate:
        J(a) = log( Σ_k exp( λ·(a*a)_k ) )
    whose gradient can be written using a correlation of the soft‑max
    weights with the current sequence.  After a gradient step we project
    back onto the probability simplex (non‑negative, unit‑sum).

    * λ starts small and is slowly increased (focuses more on the true
      maximum as optimisation proceeds).
    * Step size η is decayed gently.
    * Occasional random perturbations help escaping shallow local minima.

3.  Keep the best sequence seen (according to the original
    `evaluate_sequence` function) and return it before the time budget
    expires.

All heavy work (convolution, correlation) is done with NumPy which is
fast enough to allow thousands of iterations within the 1000‑second
budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def fast_evaluate(seq: np.ndarray) -> float:
    """Fast analogue of evaluate_sequence (no clipping, no checks)."""
    n = seq.size
    conv = np.convolve(seq, seq)
    max_b = np.max(conv)
    sum_a = np.sum(seq)
    if sum_a < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a ** 2))


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex (non‑negative, sum=1)."""
    if np.all(v <= 0):
        # fallback to uniform distribution
        return np.full_like(v, 1.0 / v.size)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(v.size) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w


def exponential_sequence(n: int, q: float) -> np.ndarray:
    """Return a normalised exponential profile a_i ∝ q^i (sum = 1)."""
    if q >= 1.0:
        # uniform fallback
        return np.full(n, 1.0 / n)
    # geometric series: sum_{i=0}^{n-1} q^i = (1 - q**n) / (1 - q)
    denom = (1.0 - q ** n)
    if denom == 0.0:  # numerically unstable, treat as uniform
        return np.full(n, 1.0 / n)
    factor = (1.0 - q) / denom
    return factor * (q ** np.arange(n))


def softmax_weights(b: np.ndarray, lam: float) -> np.ndarray:
    """Stable soft‑max over the convolution vector b."""
    b_max = np.max(b)
    # subtract the max to avoid overflow
    exp_term = np.exp(lam * (b - b_max))
    w = exp_term / np.sum(exp_term)
    return w

    """
    Search for a step‑function (list of non‑negative heights) that minimises
    the evaluation function defined in the problem statement.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0  # leave a tiny safety margin

    # --------------------------------------------------------------
    # Choose a length.  If a previous construction is available we reuse it,
    # otherwise we fall back to the canonical length 2100.
    # --------------------------------------------------------------
    try:
        n = len(height_sequence_1)          # type: ignore[name-defined]
    except Exception:
        n = 2100

    # --------------------------------------------------------------
    # Phase 1 – sweep a simple exponential family for a good start.
    # --------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    # sweep q in a sensible interval (more weight on small q gives
    # peaked distributions, q close to 1 approaches uniform)
    for q in np.linspace(0.90, 0.9999, 200):
        seq = exponential_sequence(n, q)
        score = fast_evaluate(seq)
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
        if time.time() > deadline:
            break

    # --------------------------------------------------------------
    # Phase 2 – smooth L∞ optimisation via a soft‑max surrogate.
    # --------------------------------------------------------------
    a = best_seq.copy()
    lam = 5.0            # start with a gentle smooth approximation
    eta = 0.12           # initial gradient‑step size
    iteration = 0
    while time.time() < deadline:
        iteration += 1

        # current exact score (used for bookkeeping)
        cur_score = fast_evaluate(a)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            # occasional progress output
            print(f"Iter {iteration:6d} – new best score: {best_score:.6f}")

        # compute full convolution (length 2n-1)
        b = np.convolve(a, a)

        # soft‑max weights that concentrate on the largest entries of b
        w = softmax_weights(b, lam)

    # --------------------------------------------------------------
