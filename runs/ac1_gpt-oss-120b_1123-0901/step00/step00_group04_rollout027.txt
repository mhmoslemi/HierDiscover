    sums = set()

import numpy as np
import time
from scipy import optimize, sparse

def project_simplex(v):
    # returns projection onto simplex sum=1
    # from Duchi et al.
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def compute_score(a):
    n = len(a)
    conv = np.convolve(a, a)
    M = np.max(conv)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    return float(2 * n * M / (S**2))

def build_lp_matrix(f):
    n = len(f)
    rows = []
    cols = []
    data = []
    # convolution constraints
    for k in range(2*n - 1):
        j_start = max(0, k - (n-1))
        j_end = min(k, n-1)
        # i = k - j
        for j in range(j_start, j_end+1):
            i = k - j
            rows.append(k)
            cols.append(j)
            data.append(f[i])
    # nonnegativity constraints not needed with bounds
    A = sparse.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    return A

def solve_direction(f, M):
    n = len(f)
    A = build_lp_matrix(f)
    b = np.full(2*n-1, M)
    c = -np.ones(n)  # maximize sum(b)
    # Set bounds for variables to be >=0
    bounds = [(0, None)] * n
    # Use high efficiency method
    res = optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                          method='highs',
                          options={'time_limit': 5.0})
    if not res.success:
        return None
    return res.x

def propose_candidate(seed=123, budget_s=990, **kwargs):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5.0
    # choose length
    n = 1200
    target = 1.503
    # initial sequence: random positive, normalized to sum sqrt(2n)
    raw = np.random.random(n)
    f = raw / raw.sum()
    f = f * np.sqrt(2 * n)   # normalize sum
    best_f = f.copy()
    best_score = compute_score(best_f)
    print(f"Initial score: {best_score:.6f}")
    iteration = 0
    while time.time() < deadline:
        iteration += 1
        # compute current max convolution
        conv = np.convolve(f, f)
        M = conv.max()
        # solve LP for direction
        b = solve_direction(f, M)
        if b is None:
            # fallback: random perturbation
            idx = np.random.randint(n)
            f[idx] = max(0, f[idx] + np.random.randn() * 0.01)
            continue
        # normalize g to same sum as f
        sum_b = b.sum()
        if sum_b == 0:
            continue
        g = b * (np.sqrt(2 * n) / sum_b)
        # try several mixing factors
        for t in [0.1, 0.25, 0.5, 0.75]:
            new_f = (1 - t) * f + t * g
            score = compute_score(new_f)
            if score < best_score:
                best_score = score
                best_f = new_f.copy()
                print(f"Iter {iteration} t={t:.2f} improved score to {score:.6f}")
            # accept if not worse
            if score <= compute_score(f) or np.random.rand() < 0.1:
                f = new_f
                break
        # early stop if target reached
        if best_score < target:
            print(f"Target reached at iteration {iteration} with score {best_score:.6f}")
            break
    # ensure nonnegativity and clip to bounds
    best_f = np.maximum(best_f, 0)
    # convert to python list of floats
    return best_f.tolist()

"""
Propose a step‑function sequence that beats the current upper bound (2.0).

Approach
--------
The objective to minimise is

    f(a) = 2·n·max(a∗a) / (∑a)²               (n = len(a))

Uniform heights give f = 2 for any length, but the true optimum is
≈ 1 as n → ∞.  The improvement comes from redistributing mass to positions
that do **not** increase the current maximal convolution value.
For a fixed sequence `f` with maximal convolution `M = max(f∗f)`,
the linear programme

    maximise   Σ b_j
    subject to (f ∗ b)_k ≤ M   for all k
               b_j ≥ 0

finds a non‑negative direction `b` that can increase the total mass
without raising the current peak `M`.  After normalising `b` to the same
ℓ₁‑norm as `f` we move a fraction `t` towards it.  Repeating this step
with a line‑search on `t` steadily reduces the peak convolution.

Implementation details
----------------------
* Length `n = 1200` – large enough to get a theoretical bound ≈ 1,
  yet small enough for the LP (≈ 3 M non‑zeros) to solve quickly.
* The LP is built with a sparse matrix (SciPy `csr_matrix`) and solved
  by the HiGHS interior‑point method.
* After each LP solve several mixing factors `t ∈ {0.1,0.25,0.5,0.75}`
  are tried; the best improvement is kept.
* The routine runs until the time budget expires or the target
  `1.503` is reached, keeping the best sequence seen.
* The final sequence is returned as a plain Python list of floats,
  ready for `evaluate_sequence`.
"""

import numpy as np
from scipy import optimize, sparse


def _build_lp_matrix(f: np.ndarray) -> sparse.csr_matrix:
    """Construct the sparse matrix A for constraints (f * b)_k ≤ M."""
    n = f.shape[0]
    rows = []
    cols = []
    data = []

    for k in range(2 * n - 1):
        # j runs over all positions of b that appear in (f*b)_k
        j_start = max(0, k - (n - 1))
        j_end = min(k, n - 1)
        # i = k - j
        for j in range(j_start, j_end + 1):
            i = k - j
            rows.append(k)
            cols.append(j)
            data.append(f[i])
    return sparse.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


def _solve_direction(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve the LP:
        maximise   Σ b_j
        subject to (f * b)_k ≤ M  ∀k
                   b_j ≥ 0
    Returns the optimal `b` (or None on failure).
    """
    n = f.shape[0]
    A = _build_lp_matrix(f)
    ub = np.full(2 * n - 1, M, dtype=np.float64)          # RHS for each k
    c = -np.ones(n, dtype=np.float64)                    # maximise Σb  → minimise -Σb
    bounds = [(0.0, None)] * n                           # b_j ≥ 0

    # HiGHS is fast and accepts sparse matrices.
    res = optimize.linprog(
        c,
        A_ub=A,
        b_ub=ub,
        bounds=bounds,
        method="highs",
        options={"time_limit": 5.0, "presolve": True},
    )
    if not res.success:
        return None
    return res.x


def _score(seq: np.ndarray) -> float:
    """Compute the evaluation function (lower is better)."""
    n = seq.shape[0]
    conv = np.convolve(seq, seq)
    M = np.max(conv)
    S = np.sum(seq)
    if S < 0.01:
        return float("inf")
    return float(2.0 * n * M / (S * S))


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with evaluation < 1.503.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds (the function will stop a few seconds
        before the limit to avoid termination).

    Returns
    -------
    list[float]
        The best sequence found within the time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin

    # ------------------------------------------------------------------
    # 1. Initial sequence (random, then normalised to a fixed ℓ₁‑norm)
    # ------------------------------------------------------------------
    n = 1200                                        # length of the construction
    target_score = 1.503

    # random positive values, normalise to the standard sum √(2n)
    raw = np.random.random(n)
    f = raw / raw.sum()
    f *= np.sqrt(2.0 * n)                           # ℓ₁‑norm = √(2n)

    best_seq = f.copy()
    best_val = _score(best_seq)
    print(f"[init] score = {best_val:.6f}")

    iteration = 0
    # ------------------------------------------------------------------
    # 2. Main optimisation loop – LP direction + line search on t
    # ------------------------------------------------------------------
    while time.time() < deadline:
        iteration += 1

        # current peak of the convolution
        cur_conv = np.convolve(f, f)
        M = np.max(cur_conv)

        # --------------------------------------------------------------
        # Solve LP to obtain a direction that can increase the total mass
        # without exceeding the current peak M.
        # --------------------------------------------------------------
        b = _solve_direction(f, M)
        if b is None:
            # If LP failed, apply a tiny random perturbation to escape.
            idx = np.random.randint(n)
            f[idx] = max(0.0, f[idx] + np.random.randn() * 0.001)
            continue

        # Normalise the new direction so that it has the same ℓ₁‑norm as `f`
        sum_b = b.sum()
        if sum_b == 0.0:
            continue
        g = b * (np.sqrt(2.0 * n) / sum_b)

        # --------------------------------------------------------------
        # Try several mixing factors t and keep any improvement.
        # --------------------------------------------------------------
        improved = False
        for t in (0.1, 0.25, 0.5, 0.75):
            candidate = (1.0 - t) * f + t * g
            cand_val = _score(candidate)
            if cand_val < best_val:
                best_val = cand_val
                best_seq = candidate.copy()
                print(
                    f"[{iteration:04d}] t={t:.2f}  improved score = {cand_val:.6f}"
                )
                improved = True
            # Accept the move if it does not worsen the objective
            if cand_val <= _score(f):
                f = candidate
                break

        # If none of the tried t‑values improved the objective,
        # fall back to a tiny random step to keep the search alive.
        if not improved:
            idx = np.random.randint(n)
            f[idx] = max(0.0, f[idx] + np.random.randn() * 0.001)

        # Termination check – we have already hit the target.
        if best_val < target_score:
            print(
                f"[{iteration:04d}] target reached – score {best_val:.6f}"
            )
            break

    # ------------------------------------------------------------------
    # 3. Return the best sequence (as a plain Python list)
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)   # enforce non‑negativity
    return best_seq.tolist()
