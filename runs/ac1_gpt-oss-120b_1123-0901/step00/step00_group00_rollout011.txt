"""
Propose a candidate step‑function by stochastic optimisation.

The objective (see `evaluate_sequence`) is  
    score = 2 * n * max( a * a ) / (sum(a)**2)

Because the score is scale‑invariant we keep the L¹‑norm fixed to the
canonical value `sqrt(2*n)`.  The optimisation therefore reduces to
minimising the maximal entry of the self‑convolution under a simplex
constraint.

The algorithm is a simple simulated‑annealing / hill‑climbing loop:

* start from a uniform sequence (the known 2‑bound);
* repeatedly apply cheap random moves that preserve the total mass:
  – transfer a small amount of weight between two positions,
  – add a tiny Gaussian perturbation to a few coordinates and renormalise,
  – occasionally draw a completely new Dirichlet‑distributed sequence.
* accept moves that improve the score, otherwise accept with a
  Metropolis probability that cools down over time.
* keep the best sequence seen so far and return it when the time budget
  expires.

Only NumPy and the provided `evaluate_sequence` are used, no external
files or network access.  The routine prints occasional progress updates
so the user can see the current best bound.
"""

import time
import numpy as np

# The evaluation function is assumed to be already imported in the
# execution environment:
# from <somewhere> import evaluate_sequence

def _normalise_to_target_sum(seq: np.ndarray, target_sum: float) -> np.ndarray:
    """Rescale a non‑negative vector so that its L¹‑norm equals target_sum."""
    s = seq.sum()
    if s == 0:
        # avoid division by zero – fallback to uniform distribution
        return np.full_like(seq, target_sum / len(seq))
    return seq * (target_sum / s)


def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Return a list of non‑negative heights (the step function) that
    minimises `evaluate_sequence`.  The function runs for at most
    `budget_s` seconds and keeps the best sequence found.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Time budget in seconds.
    kwargs : dict
        Ignored – kept for compatibility with the surrounding framework.

    Returns
    -------
    list[float]
        The best sequence encountered within the allocated time.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.1   # a tiny safety margin

    # ------------------------------------------------------------
    # Determine problem size.
    # If the global `height_sequence_1` is available we adopt its length,
    # otherwise we fall back to a modest default (512).
    # ------------------------------------------------------------
    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)
    else:
        n = 512

    target_sum = np.sqrt(2.0 * n)   # the normalisation used in the paper

    # ------------------------------------------------------------
    # Initial solution – uniform distribution (the obvious 2‑bound).
    # ------------------------------------------------------------
    cur_seq = np.full(n, target_sum / n, dtype=np.float64)
    cur_score = evaluate_sequence(cur_seq.tolist())

    best_seq = cur_seq.copy()
    best_score = cur_score

    # ------------------------------------------------------------
    # Simple simulated‑annealing schedule.
    # ------------------------------------------------------------
    init_temp = 0.1
    final_temp = 1e-6
    it = 0

    while time.time() < deadline:
        it += 1

        # Linear interpolation of temperature on a log‑scale time axis
        fraction = (time.time() - (deadline - budget_s)) / budget_s
        temperature = init_temp * (1.0 - fraction) + final_temp * fraction

        # --------------------------------------------------------
        # Pick a move type.
        # --------------------------------------------------------
        move = np.random.rand()
        new_seq = cur_seq.copy()

        if move < 0.60:
            # ------ transfer a fraction of mass from i -> j ------------
            i, j = np.random.randint(n, size=2)
            if i != j and new_seq[i] > 0:
                delta = np.random.rand() * 0.05 * new_seq[i]   # ≤ 5 % of source
                new_seq[i] -= delta
                new_seq[j] += delta

        elif move < 0.90:
            # ------ small Gaussian perturbation of a few entries -------
            k = np.random.randint(1, 5)                     # 1‑4 positions
            idx = np.random.choice(n, size=k, replace=False)
            perturb = np.random.randn(k) * 0.01 * target_sum / n
            new_seq[idx] = np.maximum(0.0, new_seq[idx] + perturb)
            # Renormalise to keep the L¹‑norm fixed
            new_seq = _normalise_to_target_sum(new_seq, target_sum)

        else:
            # ------ full random restart (Dirichlet / exponential) -----
            # exponential draws give a Dirichlet distribution after normalisation
            exp_vals = np.random.exponential(scale=1.0, size=n)
            new_seq = _normalise_to_target_sum(exp_vals, target_sum)

        # --------------------------------------------------------
        # Evaluate the candidate.
        # --------------------------------------------------------
        try:
            new_score = evaluate_sequence(new_seq.tolist())
        except Exception:
            # If anything goes wrong (should not happen) just skip this move.
            continue

        # --------------------------------------------------------
        # Metropolis acceptance.
        # --------------------------------------------------------
        if new_score < cur_score:
            cur_seq, cur_score = new_seq, new_score
        else:
            # Accept worse moves with a probability that decays with temperature.
            prob = np.exp(-(new_score - cur_score) / max(temperature, 1e-12))
            if np.random.rand() < prob:
                cur_seq, cur_score = new_seq, new_score

        # --------------------------------------------------------
        # Keep the best overall sequence.
        # --------------------------------------------------------
        if new_score < best_score:
            best_seq, best_score = new_seq, new_score
            print(f"[Iter {it}] New best score: {best_score:.6f}")

    print(f"Search finished after {it} iterations. Best score = {best_score:.6f}")
    return best_seq.tolist()
