def simplex_projection(v, z=1):
    """Project vector v onto the simplex {x >= 0, sum x = z}."""
    n = len(v)
    if v.sum() == z and np.all(v >= 0):
        return v
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n+1)
    # find rho
    condition = u - cssv / ind > 0
    rho = np.where(condition)[0]
    if len(rho) == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

import time
import numpy as np

def simplex_projection(v, z):
    # v may be numpy array
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    if np.sum(v) == z and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n+1)
    # find rho
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # numerics adjust
    # ensure sum exactly
    diff = z - w.sum()
    if abs(diff) > 1e-10:
        # distribute diff proportionally to nonzero entries
        nonzero = w > 0
        w[nonzero] += diff / nonzero.sum()
    return w

def arcsine_initial(n, target_sum):
    i = np.arange(n, dtype=float)
    # use i+0.5 and (n-0.5-i)
    w = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    w = w * (target_sum / np.sum(w))
    return w

def propose_candidate(seed=42, budget_s=1000):
    """Search for a non‑negative step‑function that minimizes
    evaluate_sequence.  Uses a projected gradient descent on the
    maximum convolution coefficient, starting from an arcsine‑shaped
    distribution and performing several random restarts."""
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # leave a tiny safety margin

    best_seq = None
    best_score = float('inf')
    it = 0

    # parameters
    MIN_N = 1200
    MAX_N = 4000
    MAX_RESTARTS = 100
    MAX_INNER = 5000

    while time.time() < deadline and it < MAX_RESTARTS:
        it += 1
        # choose length
        n = np.random.randint(MIN_N, MAX_N + 1)
        target_sum = np.sqrt(2.0 * n)

        # initial sequence: arcsine shape with small random perturbation
        a = arcsine_initial(n, target_sum)

        # occasional random perturbation to avoid pathological cases
        if np.random.rand() < 0.2:
            a += 0.01 * np.random.randn(n)
            a = np.maximum(a, 0.0)
            a = simplex_projection(a, target_sum)

        cur_score = evaluate_sequence(a.tolist())
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.tolist()
            print(f"[{time.time() - start_time:.1f}s] Restart {it}: init score {cur_score:.6f}")

        # inner projected gradient descent
        step_factor = 0.5          # base scaling for the step size
        patience = 0
        last_improve = time.time()

        for inner in range(MAX_INNER):
            if time.time() >= deadline:
                break

            # convolution
            b = np.convolve(a, a)
            max_b = b.max()
            kmax = int(np.argmax(b))

            # early stopping if we are already close to the theoretical lower bound
            # (avg = sum_a^2/(2n-1) = 2n/(2n-1) approx 1)
            # we stop when max_b is not much larger than avg.
            avg_b = (target_sum ** 2) / (2 * n - 1)
            if max_b <= avg_b * 1.05:   # within 5%
                break

            # gradient of the active coefficient
            g = np.zeros_like(a)
            i_start = max(0, kmax - (n - 1))
            i_end = min(n - 1, kmax)
            # i range inclusive
            idx = np.arange(i_start, i_end + 1, dtype=int)
            jdx = kmax - idx
            g[idx] = 2.0 * a[jdx]

            # dynamic step size – larger when the gap is large
            gap = max_b - avg_b
            step = step_factor * gap / max_b   # relative step
            a = a - step * g
            a = np.maximum(a, 0.0)
            a = simplex_projection(a, target_sum)

            # evaluate
            cur_score = evaluate_sequence(a.tolist())
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_seq = a.tolist()
                patience = 0
                last_improve = time.time()
                print(f"[{time.time() - start_time:.1f}s] Restart {it} inner {inner} => new best {best_score:.6f}")
                # if we already beat the target we can stop early
                if best_score < 1.503:
                    deadline = time.time()   # break outer loops
                    break
            else:
                patience += 1

            # adapt step_factor if we are stuck
            if patience >= 200:
                step_factor *= 0.5
                patience = 0
                if step_factor < 1e-6:
                    break

            # occasional random jitter to escape flat regions
            if inner % 500 == 0 and np.random.rand() < 0.1:
                noise = 0.005 * np.random.randn(n)
                a = np.maximum(a + noise, 0.0)
                a = simplex_projection(a, target_sum)

        # end inner loop
    # end while

    if best_seq is None:
        # fallback: return a simple uniform sequence
        n = 1000
        target_sum = np.sqrt(2.0 * n)
        a = np.full(n, target_sum / n)
        best_seq = a.tolist()
        best_score = evaluate_sequence(best_seq)

    print(f"Finished search. Best score = {best_score:.6f} with length {len(best_seq)}")
    return best_seq

"""
Propose a step‑function (non‑negative list of heights) that minimises
`evaluate_sequence`.  The algorithm is a simple projected gradient descent
on the quantity  max_k (a * a)[k]  while keeping the ℓ₁‑norm fixed to
sqrt(2·n).  The gradient of the active (largest) convolution coefficient
is cheap to compute, and after each gradient step we project back onto the
simplex (non‑negative entries with the prescribed sum) by a scaling step.
Several random restarts with different lengths and initialisations are
performed until the time budget expires.  The best sequence seen is
returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions
# ----------------------------------------------------------------------


def _project_to_sum(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Scale a non‑negative vector so that its sum equals ``target_sum``.
    Zeros stay zero; if the whole vector is zero a uniform vector is
    returned.
    """
    s = v.sum()
    if s <= 0.0:
        # fallback – uniform distribution
        return np.full_like(v, target_sum / v.size, dtype=float)
    return v * (target_sum / s)


def _initial_random(n: int, target_sum: float) -> np.ndarray:
    """
    Return a random starting point: exponential (Dirichlet) samples
    normalised to the required ℓ₁‑norm.
    """
    a = np.random.exponential(scale=1.0, size=n)
    a = _project_to_sum(a, target_sum)
    return a


def _initial_uniform_noise(n: int, target_sum: float) -> np.ndarray:
    """
    Uniform distribution perturbed by a tiny Gaussian noise – this breaks
    the symmetry of the uniform fixed point while staying close to it.
    """
    a = np.full(n, target_sum / n, dtype=float)
    a += 0.001 * target_sum * np.random.randn(n)   # tiny noise
    a = np.maximum(a, 0.0)
    a = _project_to_sum(a, target_sum)
    return a


def _initial_arcsine(n: int, target_sum: float) -> np.ndarray:
    """
    A discretised arcsine density (more mass near the ends).  It is *not*
    optimal on its own but provides a non‑trivial, non‑symmetric start.
    """
    i = np.arange(n, dtype=float)
    # 1 / sqrt( (i+0.5)*(n-i-0.5) )
    w = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    w = _project_to_sum(w, target_sum)
    return w


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a sequence that minimises ``evaluate_sequence``.
    The function respects the 1000 s time budget and returns the best
    sequence discovered (as a plain Python list of floats).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    best_seq: list[float] | None = None
    best_score = float("inf")

    # parameters for the search
    MIN_N = 1500
    MAX_N = 3500
    MAX_RESTARTS = 200          # far more than enough; loop stops by time
    MAX_INNER = 8000            # gradient steps per restart
    STEP_FACTOR = 0.5           # base size of a gradient step (will shrink)

    # ------------------------------------------------------------------
    # outer loop – random restarts
    # ------------------------------------------------------------------
    for restart_idx in range(1, MAX_RESTARTS + 1):
        if time.time() >= deadline:
            break

        n = np.random.randint(MIN_N, MAX_N + 1)
        target_sum = np.sqrt(2.0 * n)

        # ----------------------------------------------------------------
        # pick an initialisation method
        # ----------------------------------------------------------------
        init_choice = np.random.choice(
            ["arcsine", "uniform_noise", "random"], p=[0.4, 0.3, 0.3]
        )
        if init_choice == "arcsine":
            a = _initial_arcsine(n, target_sum)
        elif init_choice == "uniform_noise":
            a = _initial_uniform_noise(n, target_sum)
        else:
            a = _initial_random(n, target_sum)

        # evaluate the starting point
        cur_score = evaluate_sequence(a.tolist())
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.tolist()
            print(
                f"[{time.time() - start_time:6.2f}s] Restart {restart_idx:3d} "
                f"init ({init_choice}) – n={n} – score={cur_score:.6f}"
            )
        # --------------------------------------------------------------
        # inner projected‑gradient descent
        # --------------------------------------------------------------
        step_factor = STEP_FACTOR
        patience = 0               # counts iterations without improvement
        last_improve = time.time()

        for it in range(1, MAX_INNER + 1):
            if time.time() >= deadline:
                break

            # convolution and the active index
            b = np.convolve(a, a)
            max_b = b.max()
            k_max = int(b.argmax())

            # theoretical average (constant for our normalisation)
            avg_b = (target_sum ** 2) / (2 * n - 1)

            gap = max_b - avg_b

                    break
            else:
                patience += 1

            # adapt step factor if we are stuck for a while
            if patience >= 300:
                step_factor *= 0.5
                patience = 0
                if step_factor < 1e-8:
                    break

            # occasional random jitter to escape flat plateaus

                jitter = 0.003 * target_sum * np.random.randn(n)
                a = np.maximum(a + jitter, 0.0)
                a = _project_to_sum(a, target_sum)

        # end of inner loop for this restart
        # (go on to the next restart automatically)

    # ----------------------------------------------------------------------
    # fallback – should never happen but keeps the contract safe
    # ----------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        target_sum = np.sqrt(2.0 * n)
        a = np.full(n, target_sum / n, dtype=float)
        best_seq = a.tolist()
        best_score = evaluate_sequence(best_seq)

    print(
        f"Search finished. Best score = {best_score:.6f} "
        f"with length {len(best_seq)}."
    )
    return best_seq
