"""
Propose a step‑function (non‑negative sequence) that lowers the
evaluation metric

    2 * n * max(conv(a,a)) / (sum(a)**2)

below the trivial uniform bound (=2).  The algorithm mixes three
simple search ideas:

1.  **Fast objective** – the convolution is evaluated with an FFT so
    that each iteration costs O(n log n) instead of O(n²).

2.  **Sub‑gradient move** – for the index `k` where the convolution
    attains its maximum we compute the (sub‑)gradient
    ∇ₖ b = a_{k‑i}.  A small step opposite to the gradient (with a
    decreasing step‑size) reduces the contribution of the “worst”
    sum while keeping the vector non‑negative and renormalised.

3.  **Exchange move** – the pair (i_out, i_in) is chosen from the
    largest and a random (or smallest) contributor to the maximal
    sum; a tiny amount of mass `δ` is transferred from `i_out` to
    `i_in`.  This greedy perturbation can break the symmetry of the
    uniform vector.

Both moves are accepted if they improve the current objective,
otherwise they are accepted with a small probability (simulated
annealing) to avoid getting stuck in a local optimum.  The procedure
runs for the allotted wall‑time (default 1000 s) and returns the best
sequence found.  The code is self‑contained, uses only NumPy / SciPy
and prints occasional progress reports.
"""

import time
import math
import numpy as np
from scipy import optimize

# ----------------------------------------------------------------------
# Helper: fast convolution (a * a) via FFT
def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Return the linear convolution a*a using an FFT of power‑of‑2 length."""
    n = a.shape[0]
    # length of the zero‑padded FFT (next power of two >= 2*n-1)
    L = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, L)
    conv = np.fft.irfft(A * A, L)[: 2 * n - 1]
    # Numerical noise may produce tiny negatives – clip them
    return np.maximum(conv, 0.0)


# ----------------------------------------------------------------------
# Fast evaluation of the objective (identical to evaluate_sequence)
def _objective(a: np.ndarray) -> float:
    """2 * n * max(conv(a,a)) / (sum(a)**2)  –  assumes a is non‑negative."""
    n = a.shape[0]
    s = a.sum()
    if s < 0.01:                # safeguard – mirrors evaluate_sequence
        return np.inf
    conv = _conv_fft(a)
    return float(2.0 * n * conv.max() / (s * s))


# ----------------------------------------------------------------------
# Linear‑programming direction (same LP used in the reference code)
def _solve_lp(f: np.ndarray, rhs: float):
    """
    Solve   max 1ᵀb   subject to   conv(f,b) ≤ rhs ,   b ≥ 0.
    Returns the optimal b (as a numpy array) or None on failure.
    """
    n = f.shape[0]
    c = -np.ones(n)                     # maximise sum(b)  → minimise -sum(b)
    # Build the inequality matrix for each convolution entry k
    A = []
    b = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        # row[j] = f[i] where i = k-j
        j_vals = np.arange(i_min, i_max + 1)
        i_vals = k - j_vals
        row[j_vals] = f[i_vals]
        A.append(row)
        b.append(rhs)
    # non‑negativity constraints  -b ≤ 0
    A.append(-np.eye(n))
    b.append(np.zeros(n))
    A = np.vstack(A)
    b = np.concatenate(b)
    # Linprog options: a small time limit per call prevents hangs
    res = optimize.linprog(c, A_ub=A, b_ub=b,
                          bounds=[(0, None)] * n,
                          method='highs',
                          options={'presolve': True, 'time_limit': 5.0})
    if res.success:
        return res.x
    return None


# ----------------------------------------------------------------------
# One iteration of the "LP move" – similar to the SOTA direction but
# with a tunable mixing factor.
def _lp_move(a: np.ndarray, mix: float):
    """
    Return a new sequence obtained by mixing the current vector `a`
    with the LP direction `g`.  `mix` is the interpolation parameter
    (0 < mix ≤ 1).  If the LP fails `None` is returned.
    """
    n = a.shape[0]
    # normalise to sum = sqrt(2n) as in the reference description
    scale = math.sqrt(2.0 * n) / a.sum()
    f_norm = a * scale
    rhs = _conv_fft(f_norm).max()
    g = _solve_lp(f_norm, rhs)
    if g is None:
        return None
    # bring g back to the original normalisation
    g = g * scale
    # linear interpolation
    new_a = (1.0 - mix) * a + mix * g
    # enforce non‑negativity and re‑normalise to avoid drift
    new_a = np.maximum(new_a, 0.0)
    if new_a.sum() == 0.0:
        return a.copy()
    new_a /= new_a.sum()
    return new_a

                a_candidate /= a_candidate.sum()
            else:
                # ---- sub‑gradient step ---------------------------------
                conv = _conv_fft(a)
                k = int(conv.argmax())
                i_low = max(0, k - n + 1)
                i_high = min(n - 1, k)
                idx_i = np.arange(i_low, i_high + 1)
                idx_j = k - idx_i
                g = np.zeros_like(a)
                g[idx_i] = a[idx_j]                      # ∇b_k_i = a_{k-i}
                # decreasing step size (sqrt schedule)
                step = step0 / math.sqrt(iter_cnt + 1.0)
                a_candidate = a - step * g
                a_candidate = np.maximum(a_candidate, 0.0)
                if a_candidate.sum() == 0.0:
                    # fallback to uniform if we collapsed to zeros
                    a_candidate = np.full(n, 1.0 / n)
                else:
                    a_candidate /= a_candidate.sum()

            # ---- evaluate ------------------------------------------------
            cand_score = _objective(a_candidate)

            # Acceptance rule (simulated annealing)
            if cand_score < cur_score or np.random.rand() < math.exp(
                -(cand_score - cur_score) / (temperature + 1e-12)
            ):
                a = a_candidate
                cur_score = cand_score
                # keep the global best
                if cand_score < best_score:
                    best_score = cand_score
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(
                        f"  iter {iter_cnt:6d}  new best = {best_score:.6f}  "
                        f"temp={temperature:.4f}"
                    )
            # temperature decay
            temperature *= cooling

            # ---- occasional random perturbation to shake stagnation -----
            if iter_cnt % 2000 == 0:
                # a tiny jitter on two random positions
                i, j = np.random.choice(n, 2, replace=False)
                eps = 0.005 * a[i]
                a[i] = max(0.0, a[i] - eps)
                a[j] += eps
                a /= a.sum()
                cur_score = _objective(a)

            # ---- stop early if we haven't improved for a while ---------
            if time.time() - last_improve > 60.0:   # one minute of silence
                print("  No improvement for 60 s – switching length or stopping.")
                break

        # end of while‑loop for this length
        print(f"Finished length {n}: best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the best sequence as a plain Python list (evaluate_sequence
    # will be called later by the evaluation harness).
    print("\n=== Search finished ===")
    print(f"Overall best score found: {best_score:.6f}")
    return best_seq.tolist()
