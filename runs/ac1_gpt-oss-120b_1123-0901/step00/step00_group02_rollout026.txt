def fft_convolve(a,b):
    n = len(a) + len(b) - 1
    size = 1 << (n-1).bit_length()
    A = np.fft.rfft(a, size)
    B = np.fft.rfft(b, size)

"""
Propose a step‑function (non‑negative sequence) that tries to minimise

    2 * n * max(conv(seq,seq)) / (sum(seq)**2)

The algorithm is a simple projected exponential‑gradient descent that
flattens the self‑convolution.  At each iteration we:
  * compute the convolution using an FFT (O(n log n)),
  * find a few indices with the largest and the smallest convolution values,
  * build a sub‑gradient that pushes mass away from the large‑value shifts
    and towards the small‑value shifts,
  * update the sequence with an exponentiated step,
  * re‑normalise to keep the total mass = 1 (the objective is scale‑invariant).

Multiple random restarts with different lengths are tried until the time
budget expires.  The best sequence found (according to the true
``evaluate_sequence`` function) is returned.
"""

import time
import numpy as np
import math

# --------------------------------------------------------------
# Helper: fast (real) convolution via FFT, length = len(x)+len(y)-1
def fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    n = len(x) + len(y) - 1
    m = 1 << (n - 1).bit_length()          # next power of two
    X = np.fft.rfft(x, m)
    Y = np.fft.rfft(y, m)
    Z = X * Y
    conv = np.fft.irfft(Z, m)[:n]
    return conv


# --------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list:
    """Return a sequence of non‑negative heights that (hopefully) beats
    the current bound 2.0.  The routine runs for at most *budget_s* seconds.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 5.0   # a few seconds safety margin

    best_seq = None
    best_score = math.inf

    # ------------------------------------------------------------------
    # If the previous best sequence is available, use it as a starting point.
    try:
        init = np.asarray(height_sequence_1, dtype=float)
        if init.size > 0:
            init = np.maximum(init, 0.0)
            init = init / init.sum()           # normalise to sum = 1
            best_seq = init.tolist()
            best_score = evaluate_sequence(best_seq)
            print(f"[init] starting score from height_sequence_1 = {best_score:.6f}")
    except Exception:
        init = None

    # ------------------------------------------------------------------
    # Lengths that we will explore (they are all modest enough for FFT).
    lengths = [800, 1200, 1600, 2100, 2600]

    # Main time‑limited search loop
    while time.time() < deadline:
        # --------------------------------------------------------------
        # pick a length and initialise a random distribution
        n = int(rng.choice(lengths))
        a = rng.random(n).astype(float)
        a += 1e-12                     # avoid exact zeros
        a /= a.sum()                   # normalise (scale does not matter)

        # occasionally start from the previous best sequence (if sizes match)
        if init is not None and init.size == n and rng.random() < 0.2:
            a = init.copy()
            a = a / a.sum()

        # hyper‑parameters for this run
        max_iters = 2000
        eta0 = 0.15                    # base step size for the exp‑gradient
        top_k = 7                      # how many of the largest conv entries to act on
        bot_k = 7                      # how many of the smallest conv entries to act on

        # --------------------------------------------------------------
        for it in range(max_iters):
            if time.time() >= deadline:
                break

            # 1) compute the full convolution (FFT is fast)
            conv = fft_convolve(a, a)

            # 2) evaluate the current objective (exact, no clipping)
            max_b = conv.max()
            sum_a = a.sum()
            cur_score = 2.0 * n * max_b / (sum_a * sum_a)

            # 3) keep the best ever seen (using the official evaluator)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.tolist()
                print(f"[new best] score={best_score:.6f}  n={n}  it={it}")

            # 4) stop early if we already have a respectable bound
            # (the target is about 1.5, we keep searching anyway)
            # ----------------------------------------------------------

            # 5) build a sub‑gradient that reduces the largest and raises the smallest
            #    convolution entries.
            #    For a given shift k, the derivative d conv_k / d a_i = 2 * a_{k-i}
            #    (zero when the index is out of range).  We only sum over a few k.
            grad = np.zeros_like(a)

            # indices of the largest values
            if top_k >= conv.size:
                top_idx = np.arange(conv.size)
            else:
                top_idx = np.argpartition(conv, -top_k)[-top_k:]

            for k in top_idx:
                i_start = max(0, k - (n - 1))
                i_end = min(n - 1, k)
                i_vals = np.arange(i_start, i_end + 1, dtype=int)
                j_vals = k - i_vals
                # contribution = a_j (the factor multiplying a_i)
                grad[i_vals] += a[j_vals]

            # indices of the smallest values (but we ignore the zero‑only extremes)
            if bot_k >= conv.size:
                bot_idx = np.arange(conv.size)
            else:
                bot_idx = np.argpartition(conv, bot_k)[:bot_k]

            for k in bot_idx:
                i_start = max(0, k - (n - 1))
                i_end = min(n - 1, k)
                i_vals = np.arange(i_start, i_end + 1, dtype=int)
                j_vals = k - i_vals
                grad[i_vals] -= a[j_vals]      # negative sign → push mass there

            # 6) exponential update (guarantees positivity)
            eta = eta0 / math.sqrt(it + 1.0)   # decaying step size
            # clip gradient to avoid overflow in the exponent
            np.clip(grad, -10, 10, out=grad)
            a = a * np.exp(-eta * grad)

            # 7) small random perturbation every few iterations to escape plateaus
            if (it + 1) % 120 == 0:
                noise = rng.normal(scale=1e-3, size=n)
                a = np.abs(a + noise)

            # 8) re‑normalise (the objective is homogeneous, any scale works)
            a_sum = a.sum()
            if a_sum == 0:
                a = rng.random(n) + 1e-12
                a_sum = a.sum()
            a /= a_sum

    # ------------------------------------------------------------------
    # Return the best sequence we discovered (as a plain Python list)
    if best_seq is None:
        # fallback: a simple uniform sequence (guaranteed to be valid)
        n = 2100
        best_seq = [1.0] * n
    return best_seq
