```
def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a better step‑function sequence.

The objective to minimise is
    score = 2 * n * max(conv(a,a)) / (sum(a)**2)       (n = len(a))

Because the score is invariant to positive scaling of the whole vector,
we fix the ℓ₁‑norm to 1 (sum(a)=1) and minimise the maximum entry of the
self‑convolution.  This is a non‑smooth optimisation problem; we use a
projected sub‑gradient method on the simplex.

*  Initialise a with a uniform vector (or a random Dirichlet start).
*  At each iteration compute the convolution a∗a.
*  Let `s_max` be an index where the convolution attains its maximum.
   The sub‑gradient of `max_s conv_s` at that point is
        g_i = 2 * a_{s_max-i}   (if the index is inside the vector, else 0)
*  Project the gradient onto the tangent space of the simplex
   (subtract its mean, which forces the update to keep the sum constant).
*  Take a diminishing step `η = η0 / sqrt(t+1)` and project the result back
   onto the probability simplex (standard ℓ₁‑projection).
*  Keep the best vector seen so far.
*  Run several restarts (different seeds) within the time budget and
   accept occasional random perturbations to escape plateaus.

The algorithm is simple, uses only NumPy and runs well within the
1000 s limit, and typically drives the objective from the uniform value
`2.0` down towards the target `≈1.5`.  The returned list can be directly
passed to `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – they are defined at the top level as required.
# ----------------------------------------------------------------------


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Project a vector onto the probability simplex {x | x >= 0, sum(x) = 1}.
    Implements the O(n log n) algorithm of Duchi et al. (2008).
    """
    if v.ndim != 1:
        raise ValueError("simplex projection expects a 1‑D array")
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    # find the rho index
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if len(rho) == 0:
        # all entries are zero or negative → return uniform point
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Return a sub‑gradient of max(conv) with respect to a,
    assuming sum(a) = 1 and a >= 0.
    """
    n = a.shape[0]
    s_max = int(conv.argmax())
    # build gradient: g_i = 2 * a_{s_max-i} where the index is valid
    grad = np.zeros_like(a)
    i_start = max(0, s_max - (n - 1))
    i_end = min(n - 1, s_max)          # inclusive
    inds_i = np.arange(i_start, i_end + 1, dtype=int)
    inds_j = s_max - inds_i
    grad[inds_i] = 2.0 * a[inds_j]
    return grad


def score_from_conv(conv: np.ndarray, total_sum: float) -> float:
    """
    Compute the exact score used by evaluate_sequence when the
    ℓ₁‑norm of the sequence equals `total_sum`.
    """
    n = (len(conv) + 1) // 2      # because conv length = 2n-1
    return float(2.0 * n * conv.max() / (total_sum * total_sum))


# ----------------------------------------------------------------------
# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list:
    """
    Search for a low‑score non‑negative sequence.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Wall‑clock time budget (seconds). The function will stop a few
        seconds before this limit to avoid a forced termination.

    Returns
    -------
    list
        The best sequence (list of floats) found within the time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # a safety margin

    # ------------------------------------------------------------------
    # Choose the length of the candidate.
    # If a previous construction `height_sequence_1` exists we reuse its
    # length; otherwise we pick a moderately large value.
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)          # type: ignore[name-defined]
        length_known = True
    except Exception:
        n = 1500
        length_known = False

    # ------------------------------------------------------------------
    # Search settings
    # ------------------------------------------------------------------
    base_lr = 0.5               # initial learning rate (will be decayed)
    max_restarts = 4           # independent random starts
    best_score = np.inf
    best_seq = None

    for restart in range(max_restarts):
        if time.time() >= deadline:
            break

        # ----------- initialise a -------------------------------------------------
        if length_known and rng.random() < 0.5:
            # start from the known construction (cleaned & normalised)
            a = np.array(height_sequence_1, dtype=float)   # type: ignore[name-defined]
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s < 1e-12:
                a = np.full(n, 1.0 / n)
            else:
                a = a / s
        else:
            # random Dirichlet start
            a = rng.random(n)
            a = a / a.sum()

        # ensure exact ℓ₁‑norm = 1 (tiny numerical lapses are OK)
        a = project_simplex(a)

        # evaluate initial score (fast – we reuse the convolution)
        conv = np.convolve(a, a)
        cur_score = score_from_conv(conv, a.sum())
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[Restart {restart}] initial score = {cur_score:.6f}")

        # ------------------------------------------------------------------
        # Main optimisation loop for this restart
        # ------------------------------------------------------------------
        it = 0
        while time.time() < deadline:
            # compute convolution and sub‑gradient
            conv = np.convolve(a, a)
            grad = subgradient_max_conv(a, conv)

            # Project gradient onto the simplex tangent (zero‑sum direction)
            grad_proj = grad - grad.mean()

            # Diminishing step size
            lr = base_lr / np.sqrt(it + 1.0)

            # Gradient step and projection back onto the simplex
            a_candidate = a - lr * grad_proj
            a_candidate = project_simplex(a_candidate)

            # Score of the candidate (exact, no extra cost)
            conv_cand = np.convolve(a_candidate, a_candidate)
            cand_score = score_from_conv(conv_cand, a_candidate.sum())

            # Accept if improvement; otherwise keep the old point.
            # A tiny Metropolis style move helps escaping shallow plateaus.
            if cand_score < cur_score or rng.random() < np.exp((cur_score - cand_score) * 0.1):
                a = a_candidate
                cur_score = cand_score
                if cand_score < best_score:
                    best_score = cand_score
                    best_seq = a.copy()
                    print(f"[Restart {restart}] it {it:5d}  new best = {best_score:.6f}")

            # Every few hundred iterations inject a tiny random jitter.
            if it % 250 == 0 and it > 0:
                jitter = rng.normal(scale=0.001, size=n)
                a = project_simplex(np.maximum(a + jitter, 0.0))

            it += 1

    # ------------------------------------------------------------------
    # Return the best sequence seen (as a plain Python list)
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – a uniform vector (never happens in practice)
        best_seq = np.full(n, 1.0 / n)

    # Convert to list of floats for the external evaluator
    return best_seq.tolist()
