def proj_simplex(v):
    # ensure non-negative and sum=1
    # Standard algorithm: sort descending, find rho, compute theta.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a candidate step‐function (non‑negative vector) that tries to minimise

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm starts from the analytically motivated sequence
a_i = C(2i,i) / 4**i  (coefficients of (1‑z)^‑½) which attains the well‑known
limit π/2 ≈ 1.571.  From this point a projected (sub‑)gradient descent on the
∞‑norm of the auto‑convolution is performed.  The gradient of the max‑entry
is cheap to compute, a simple line‑search adapts the step size and after
each step the vector is projected onto the probability simplex (non‑negative,
unit‑sum).  Random jitter is occasionally injected to escape tiny plateaus.
The whole loop runs until the allotted time expires (≈ 1000 s) and the best
sequence found is returned.

The method is deterministic for a given seed, uses only NumPy and the
provided `evaluate_sequence`, and works for any existing `height_sequence_1`
(if present) or falls back to the analytic start.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: project a vector onto the probability simplex (non‑negative, sum = 1)
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=1}."""
    if v.ndim != 1:
        raise ValueError("simplex projection expects a 1‑D array")
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    # find the rho index
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # should not happen, but fallback to uniform vector
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast convolution via FFT (real signals)
def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a using a real FFT (returns length 2*n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=size)
    B = A * A
    conv = np.fft.irfft(B, n=size)[: 2 * n - 1]
    return conv
