"""
Two‑block step function generator.

The idea is to concentrate the total mass on two equal blocks at the
beginning and end of the sequence, leaving a zero gap in the middle.
If each block has length *m* and each entry in the blocks is
c = 1 / (2·m) (so the total sum is 1), then the largest value of the
auto‑convolution is

    max(a * a) = c² · m = 1 / (4·m),

which is smaller than the uniform case (where max = 1/n).  The evaluation
value returned by ``evaluate_sequence`` is

    2·n·max(a*a) = 2·n / (4·m) = n / (2·m).

Choosing a block length `m` close to `n/2` drives the score towards 1,
well below the target 1.503.  The implementation builds such a
sequence (default block ratio 0.49) and returns it.

The function works for any existing `height_sequence_1` (its length is
used) or falls back to a default length of 5000.
"""

import time
import numpy as np

def generate_two_block_sequence(n: int, block_ratio: float = 0.49) -> list[float]:
    """
    Create a sequence of length `n` that has two identical blocks of
    non‑zero entries at the two ends and zeros in the middle.

    Parameters
    ----------
    n : int
        Length of the desired sequence (must be >= 2).
    block_ratio : float
        Fraction of the total length occupied by each block.
        Must satisfy 0 < block_ratio < 0.5.  The default 0.49 gives a
        score close to 1.0.

    Returns
    -------
    seq : list[float]
        The constructed non‑negative sequence (sum is exactly 1.0
        up to floating‑point rounding).
    """
    # Ensure a feasible block length
    block_len = max(1, int(block_ratio * n))
    # If the blocks would overlap or touch, shrink them a bit
    if block_len * 2 >= n:
        block_len = n // 2 - 1
        if block_len < 1:
            block_len = 1

    # Height of each non‑zero entry so that the total sum is 1
    height = 1.0 / (2.0 * block_len)

    seq = [0.0] * n
    # Fill the left block
    for i in range(block_len):
        seq[i] = height
    # Fill the right block
    start = n - block_len
    for i in range(block_len):
        seq[start + i] = height

    return seq
