    w = np.maximum(v - theta, 0)

    conv = np.convolve(arr, arr)
    diff = conv - target_avg
    grad_err = 4.0 * np.convolve(diff, arr[::-1])[n-1:n-1+n]

    idx = np.arange(n)

    grad_max = np.zeros_like(arr)
    grad_max[idx[mask]] = 2.0 * arr[k_star - idx[mask]]
    combined = weight_err * grad_err + weight_max * grad_max
    norm = np.linalg.norm(combined)
    if norm == 0:
        return arr
    step = 0.5 / (norm + 1e-12)   # step scaling may be tuned
    new_arr = arr - step * combined
    new_arr = np.maximum(new_arr, 0.0)
    s = np.sum(new_arr)
    if s == 0:
        return np.full_like(arr, 1.0 / n)
    return new_arr / s

"""
Propose a candidate step‑function sequence.

The algorithm performs many independent stochastic restarts.
Each restart starts from a (normalized) random vector and applies a
small number of gradient‑descent updates that try to flatten the
self‑convolution.  The loss being minimised is a combination of

* the squared error to the uniform target convolution (which pushes the
  whole autocorrelation towards a constant), and
* a sub‑gradient for the maximum entry of the convolution (which directly
  reduces the quantity appearing in the evaluation function).

After each gradient step the vector is projected back onto the probability
simplex (non‑negative entries that sum to one).  A short hill‑climbing
phase with random mass transfers is then applied.  The best sequence seen
throughout the allotted time is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (must be top‑level – no closures)
# ----------------------------------------------------------------------


def compute_objective(arr: np.ndarray) -> float:
    """Return the value of evaluate_sequence for a non‑negative array."""
    n = len(arr)
    s = np.sum(arr)
    if s <= 0.0:
        return float("inf")
    conv = np.convolve(arr, arr)
    max_c = np.max(conv)
    # the exact formula used by evaluate_sequence (scale‑invariant)
    return 2.0 * n * max_c / (s * s)


def gradient_update(
    arr: np.ndarray,
    target_avg: float,
    weight_err: float = 0.5,
    weight_max: float = 0.5,
) -> np.ndarray:
    """
    Perform one gradient‑descent step.

    * `target_avg`  – desired average value of the convolution,
                      i.e. 1 / (2*n‑1) when the sum of `arr` is 1.
    * `weight_err` – weight of the L2‑error gradient.
    * `weight_max` – weight of the sub‑gradient of the maximal entry.
    """
    n = len(arr)

    # current convolution
    conv = np.convolve(arr, arr)

    # ---------- error‑gradient (squared distance to the uniform target) ----------
    diff = conv - target_avg                # shape (2n‑1,)
    # grad_err = 4 * (diff * arr_rev)   (central part only)
    grad_err = 4.0 * np.convolve(diff, arr[::-1])[n - 1 : n - 1 + n]

    # ---------- sub‑gradient for the maximal entry ----------
    k_star = int(np.argmax(conv))           # index where conv is maximal
    idx = np.arange(n)
    mask = (k_star - idx >= 0) & (k_star - idx < n)
    grad_max = np.zeros_like(arr)
    grad_max[idx[mask]] = 2.0 * arr[k_star - idx[mask]]

    # combined direction
    direction = weight_err * grad_err + weight_max * grad_max
    norm = np.linalg.norm(direction)
    if norm == 0.0:
        return arr

    # step size – chose a scale that makes changes comparable to the current entries
    step = 0.5 / (norm + 1e-12)

    # gradient step
    new_arr = arr - step * direction

    # projection onto the (non‑negative) simplex
    new_arr = np.maximum(new_arr, 0.0)
    total = np.sum(new_arr)
    if total == 0.0:
        # fallback to uniform distribution if everything vanished
        return np.full_like(arr, 1.0 / n)
    return new_arr / total


def random_hill_climb(arr: np.ndarray, trials: int = 100) -> np.ndarray:
    """
    Small random local search: move a tiny amount of mass between two
    random positions and keep the change if it improves the objective.
    """
    n = len(arr)
    best = arr.copy()
    best_val = compute_objective(best)

    for _ in range(trials):
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # transfer a small fraction of the mass from i to j
        delta = 0.01 * best[i]          # move 1 % of the source entry
        if delta <= 0.0:
            continue
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        # renormalise – sum is unchanged, but keep numerical safety
        cand = np.maximum(cand, 0.0)
        cand /= np.sum(cand)

        val = compute_objective(cand)
        if val < best_val:
            best_val = val
            best = cand
    return best


# ----------------------------------------------------------------------
# Main entry point required by the platform
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a sequence of non‑negative heights that minimises
    `evaluate_sequence`.  The function runs for at most `budget_s` seconds
    (leaving a small safety margin) and returns the best sequence found.
    """
    np.random.seed(seed)
    # small safety margin so we finish before the hard timeout
    deadline = time.time() + budget_s - 5.0

    # length of the sequence – we keep the length of the provided construction
    n = len(height_sequence_1)  # globals are assumed to exist in the environment

    # target average value of the convolution when the sum of the sequence is 1
    target_avg = 1.0 / (2 * n - 1)

    # ------------------------------------------------------------------
    # initialise best solution (uniform is a known baseline with objective 2)
    # ------------------------------------------------------------------
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.copy()
    best_score = compute_objective(best_seq)

    # ------------------------------------------------------------------
    # main stochastic‑restart loop
    # ------------------------------------------------------------------
    restart = 0
    while time.time() < deadline:
        # ----- start point -------------------------------------------------
        if np.random.rand() < 0.3:
            # occasionally start from the uniform baseline
            cur = uniform.copy()
        else:
            # random Dirichlet‑like start (positive, normalised)
            cur = np.random.rand(n)
            cur /= np.sum(cur)

        # ----- gradient descent phase --------------------------------------
        # number of internal steps limited both by time and by a hard cap
        max_inner = 3000
        inner = 0
        while time.time() < deadline and inner < max_inner:
            # take one gradient step
            cur = gradient_update(cur, target_avg, weight_err=0.5, weight_max=0.5)

            # evaluate current objective (fast, no security checks)
            cur_score = compute_objective(cur)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = cur.copy()
                print(
                    f"[restart {restart}] iter {inner}: new best {best_score:.6f}"
                )
            inner += 1

        # ----- short random hill‑climbing refinement ------------------------
        cur = random_hill_climb(cur, trials=200)
        cur_score = compute_objective(cur)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = cur.copy()
            print(f"[restart {restart}] hill‑climb improvement: {best_score:.6f}")

        restart += 1

    # ------------------------------------------------------------------
    # final conversion to Python list and final verification
    # ------------------------------------------------------------------
    final_seq = [float(x) for x in best_seq.tolist()]
    # final score using the provided (secure) evaluator
    final_score = evaluate_sequence(final_seq)
    print(f"Best final score reported by evaluate_sequence: {final_score:.6f}")

    return final_seq
