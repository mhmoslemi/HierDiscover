def propose_candidate(seed=0, budget_s=1000):
    import numpy as np, time, random
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5

    best_seq = None
    best_score = np.inf

    def evaluate(seq):
        s = np.sum(seq)
        if s < 0.01:
            return np.inf
        c = np.convolve(seq, seq)
        return float(2 * len(seq) * np.max(c) / (s * s))

    # set of lengths to try
    lengths = [200, 500, 1000, 2000]   # maybe adjust based on time
    for n in lengths:
        # start with geometric init
        # choose c (parameter) based on n to target evaluation around 1.5?
        # We'll set c = 1.3
        c_param = 1.3
        r = 1 - c_param / n
        # ensure r>0
        if r <= 0:
            r = 0.5
        # generate geometric decreasing a_i = r^i
        raw = np.power(r, np.arange(n))
        raw = raw / raw.sum()
        seq = raw.astype(float)
        cur_seq = seq.copy()
        cur_score = evaluate(cur_seq)
        # local search
        it = 0
        while time.time() < deadline:
            # compute conv and max index
            conv = np.convolve(cur_seq, cur_seq)
            max_idx = int(np.argmax(conv))
            max_val = conv[max_idx]

            # build gradient
            grad = np.zeros_like(cur_seq)
            # indices where j = max_idx - i within range
            # vectorized
            i = np.arange(n)
            j = max_idx - i
            mask = (j >= 0) & (j < n)
            grad[mask] = cur_seq[j[mask]]

            # center gradient to keep sum invariant
            grad_centered = grad - grad.mean()
            # step size
            step = 0.02 * np.random.rand() * (cur_seq.max() - cur_seq.min() + 1e-12)
            new_seq = cur_seq - step * grad_centered
            new_seq = np.clip(new_seq, 0, None)
            tot = new_seq.sum()
            if tot < 1e-12:
                # fallback random perturb
                new_seq = np.random.rand(n)
                new_seq = new_seq / new_seq.sum()
                tot = 1.0
            new_seq = new_seq / tot  # renormalize sum to 1

            new_score = evaluate(new_seq)

            if new_score < cur_score:
                cur_seq = new_seq
                cur_score = new_score
                if new_score < best_score:
                    best_score = new_score
                    best_seq = new_seq.tolist()
                    print(f"New best score {best_score:.6f} at length {n} iteration {it}")
            else:
                # occasionally accept worse move to escape
                if np.random.rand() < 0.01:
                    cur_seq = new_seq
                    cur_score = new_score

def solve_lp_step(f, M):
    n = len(f)
    c = -np.ones(n)   # maximize sum
    # Build A_ub, b_ub for each k
    A_ub = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        # for i from max(0,k-(n-1))..min(k,n-1):
        i_start = max(0, k-(n-1))
        i_end = min(k, n-1)
        for i in range(i_start, i_end+1):
            j = k - i
            # term f[i] * b[j]
            A_ub[k, j] = f[i]   # coefficient for b[j] (since conv(f,b)[k] = sum_i f[i] * b[k-i])
    b_ub = np.full(2*n-1, M)
    # also nonnegative constraints are implicit in linprog if bounds set
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs', options={'time_limit': 1})
    if result.success:
        return result.x
    else:
        return None
```

"""
Propose a step‑function sequence with a low evaluation score.

The algorithm starts from a short geometric sequence (the known near‑optimal
shape) and repeatedly improves it using a linear‑programming direction.
For a fixed sequence `f` we solve

    maximize    Σ b_j
    subject to  (f * b)_k ≤ M   for all k,
                b_j ≥ 0,

where `M` is the current maximum of the convolution `f * f`.  The LP
produces a new non‑negative vector `b` that can increase the total sum
without increasing the worst convolution entry.  Mixing the old vector
with `b` (controlled by a step size) strictly reduces the evaluation
value `2·n·max(conv)/(sum²)`.  The whole process is repeated for a few
different lengths; the best sequence found (including the provided
`height_sequence_1` if it is already better) is returned.

The implementation is time‑boxed: it stops before `budget_s` seconds have
elapsed, prints progress, and finally outputs the best sequence.
"""

import time
import numpy as np
from scipy.optimize import linprog


def _evaluate(seq: np.ndarray) -> float:
    """Exact same formula as `evaluate_sequence` (avoids the security wrapper)."""
    s = seq.sum()
    if s < 0.01:
        return np.inf
    conv = np.convolve(seq, seq)
    return float(2.0 * len(seq) * conv.max() / (s * s))


def _solve_lp(f: np.ndarray, M: float, time_limit: float = 0.8) -> np.ndarray | None:
    """
    Linear program:
        max    Σ b_j
        s.t.   (f * b)_k ≤ M   for all k = 0 … 2n‑2
               b_j ≥ 0

    Returns the optimal `b` or ``None`` if the solver fails.
    """
    n = len(f)
    # objective (maximise sum → minimise negative sum)
    c = -np.ones(n)

    # Build the inequality matrix A_ub · b ≤ b_ub.
    # Row k corresponds to the inequality Σ_i f[i]·b[k‑i] ≤ M
    rows = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            j = k - i
            row[j] = f[i]
        rows.append(row)
    A_ub = np.vstack(rows)
    b_ub = np.full(2 * n - 1, M)

    bounds = [(0, None)] * n
    # use HiGHS (fast and reliable) and enforce a per‑LP time limit
    options = {"time_limit": time_limit}
    res = linprog(
        c,
        A_ub=A_ub,
        b_ub=b_ub,
        bounds=bounds,
        method="highs",
        options=options,
    )
    if res.success:
        return res.x
    return None

    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5

    best_seq: list[float] | None = None
    best_score = np.inf

    # ----------------------------------------------------------------------
    # 1. Try a few lengths with the LP‑based refinement.
    # ----------------------------------------------------------------------
    lengths = [150, 200, 250]               # small enough to keep LP cheap
    # a geometric seed that is close to the theoretical optimum of the
    # exponential family (c ≈ 1.256 solves e^c = 1 + 2c)
    c_opt = 1.256

    for n in lengths:
        # geometric start: a_i ∝ r^i,   r = 1 - c_opt/n  (so that sum = 1)
        r = max(0.0, 1.0 - c_opt / n)

        if raw.sum() == 0.0:
            raw = np.ones(n, dtype=float)
        seq = raw / raw.sum()                       # normalised, sum = 1.0

        step = 0.2                                   # mixing weight
        # main optimisation loop

            cur_score = _evaluate(seq)

                seq[i] -= delta
                seq[j] += delta
