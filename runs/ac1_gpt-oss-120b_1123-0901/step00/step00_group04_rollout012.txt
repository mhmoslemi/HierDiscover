    # Set learning rate

        conv = np.convolve(a, a)
        max_idx = int(np.argmax(conv))
        max_val = conv[max_idx]

        g = np.zeros_like(a)

            a = np.full(n, 1.0 / n)
        else:

        score = evaluate_sequence(seq)

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Heuristic optimizer that minimizes   2 * n * max(conv(a,a)) / (sum(a)^2)
    by using an exponentiated‑gradient descent on the soft‑max approximation
    of the convolution max.  The algorithm works on the (probability) simplex
    sum_i a_i = 1, keeps the variables non‑negative and periodically records
    the best sequence found.  Multiple random restarts with different lengths
    are performed until the time budget expires.
    """
    import time, numpy as np
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5

    best_seq = None
    best_score = float('inf')

    # helper to evaluate quickly when sum = 1
    def fast_score(a):
        # 2 * n * max(conv)
        n = a.shape[0]
        c = np.convolve(a, a)
        return 2.0 * n * np.max(c)

    # list of candidate lengths – we try a few and later increase if time permits
    lengths = [500, 1000, 2000, 4000, 6000]

    # main loop over lengths and restarts
    for n in lengths:
        if time.time() >= deadline:
            break
        # number of restarts we can afford for this length
        # loosely bound by remaining time / (estimated time per run)
        while time.time() < deadline:
            # random initialization on the simplex
            a = np.random.rand(n)
            a /= a.sum()

            # parameters for the exponential‑gradient iteration
            eta = 0.6          # base learning rate
            alpha = 15.0       # softness of the max (higher -> closer to true max)
            max_inner = 3000   # safety ceiling on inner iterations

            for it in range(max_inner):
                # time guard
                if time.time() >= deadline:
                    break

                # convolution of a with itself
                conv = np.convolve(a, a)
                max_c = conv.max()
                # compute current objective (sum==1 -> score = 2*n*max_c)
                cur_score = 2.0 * n * max_c
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = a.tolist()
                    print(f"[{time.time():.1f}] new best score={best_score:.6f} "
                          f"n={n} it={it}")

                # soft‑max weighting of the convolution entries
                # shift by max_c for numeric stability (all exponents ≤ 0)
                weights = np.exp(alpha * (conv - max_c))
                w_sum = weights.sum()
                if w_sum == 0.0:
                    # degenerate – break out of inner loop
                    break
                weights /= w_sum

                # gradient of the smoothed max: 2 * Σ_k w_k a_{k‑i}

                # reversed probability vector.

                # exponentiated‑gradient update (keeps positivity)
                a = a * np.exp(-eta * grad)

                # project back onto the simplex (simple renormalisation works because
                # all entries are non‑negative after the multiplicative step)
                s = a.sum()
                if s == 0.0:
                    a = np.full(n, 1.0 / n)
                else:

                # simple schedules for the hyper‑parameters
                eta *= 0.9995

                if it % 250 == 0:

                    a = np.clip(a, 0.0, None)

            # move to a larger size.  If enough time remains we perform a second

            if time.time() < deadline and np.random.rand() < 0.3:

            else:
                break

"""Propose a coefficient sequence that minimises
   2·n·max(conv(a,a)) / (sum(a)^2)

The optimiser works on the probability simplex (∑a_i = 1) and uses an
exponentiated‑gradient (EG) scheme on a soft‑max approximation of the
maximum convolution entry.  A few random restarts with increasing
lengths are performed until the time budget expires; the best sequence
found is returned.

All helpers are defined at top level (no closures) and only standard
numpy/scipy are used.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – they are top‑level as required by the rules
# ----------------------------------------------------------------------


def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray | None:
    """
    Return a probability vector proportional to exp(alpha*(conv-max(conv))).
    The shift by max(conv) guarantees numerical stability (all exponents ≤ 0).
    If the sum of the unnormalised weights is zero (should never happen) we
    return ``None`` to signal a failure.
    """
    max_c = conv.max()
    w = np.exp(alpha * (conv - max_c))
    w_sum = w.sum()
    if w_sum == 0.0:
        return None
    return w / w_sum


def _gradient_from_weights(weights: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    For a given weight vector ``weights`` (length 2·n−1) and a probability
    vector ``a`` (length n) compute the gradient of the smoothed maximum:

        g_i = 2 * Σ_k weights_k * a_{k-i}   (indices out of range are ignored)

    This Toeplitz‑matrix product can be performed with a single convolution
    of ``weights`` with the reversed ``a``.
    """
    n = a.shape[0]
    # convolution of weights with reversed a
    full = np.convolve(weights, a[::-1])
    # the slice (n-1) … (2n-2) (inclusive) gives n entries
    grad = 2.0 * full[(n - 1) : (2 * n - 1)]
    return grad


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the simplex {x≥0, Σx = 1}."""
    s = v.sum()
    if s == 0.0:
        # fallback to uniform distribution
        return np.full_like(v, 1.0 / v.size)
    return v / s


def _exp_gradient_step(a: np.ndarray,
                       weights: np.ndarray,
                       eta: float) -> np.ndarray:
    """
    Perform one EG update:   a ← a * exp(−η·grad)  and re‑normalise.
    """
    grad = _gradient_from_weights(weights, a)
    a_new = a * np.exp(-eta * grad)
    return _project_simplex(a_new)

        print("fallback uniform sequence, score =", best_score)

    # final sanitisation – the security checks of evaluate_sequence will
    # clip negatives and values >1000, but we keep the sequence clean.
    best_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    return best_seq
