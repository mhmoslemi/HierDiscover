            if time.time() > deadline:

def project_simplex(v):
    v = np.asarray(v)
    if v.sum() == 1 and np.all(v >= 0):
        return v
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def project_simplex(v):
    # v: np.ndarray
    # returns projected vector onto probability simplex sum=1, nonnegative.
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n+1) > 0)[0]
    if len(rho) == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""Search for a step function minimizing the evaluation metric.

Algorithm:
1. Generate a set of candidate sequences using simple parametric families:
   - power‑law decay a_i ∝ (i+1)^{-α}
   - exponential decay a_i ∝ r^{i}
   - random Dirichlet draws.
   All candidates are projected onto the probability simplex (sum = 1).
2. Evaluate each candidate with ``evaluate_sequence`` and keep the best one.
3. Starting from the best candidate, run a projected sub‑gradient descent
   on the non‑smooth objective  f(x)=max_k (x*x)[k]  (the denominator is fixed
   by the simplex constraint).  For the active convolution indices the
   sub‑gradient has a closed form, so each iteration is O(n).
4. Whenever a new lower evaluation score is found it is saved.
5. The routine respects a time budget (default 1000 s) and returns the best
   sequence found before the deadline.
"""

import time
import numpy as np

def _project_to_simplex(v):
    """Project a vector onto the probability simplex {x>=0, sum(x)=1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    # find the last index where u_i > (cumsum_i -1)/i
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will become zero, return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1  # zero‑based index
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _make_powerlaw(n, alpha):
    """Return a positive vector of length n with entries (i+1)^{-alpha}, normalised."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    raw = np.power(idx, -alpha)
    raw_sum = raw.sum()
    return raw / raw_sum

def _make_exponential(n, r):
    """Return a positive vector of length n with entries r**i, normalised."""
    raw = np.power(r, np.arange(n, dtype=np.float64))
    raw_sum = raw.sum()
    return raw / raw_sum

    Search for a low‑score step‑function.
    
    Parameters

    seed : int, optional
        Random seed for reproducibility.
    budget_s : float, optional
        Time budget in seconds (the caller usually sets this to 1000).
    """

    # start from the previously known construction if it exists
    best_seq = None
    best_score = float('inf')
    if 'height_sequence_1' in globals():
        try:
            cand = list(height_sequence_1)   # assume it is a list‑like object
            score = evaluate_sequence(cand)
            best_seq = cand
            best_score = score
            print(f"Initial from height_sequence_1: {score:.6f}")
        except Exception:
            pass

    # --------------------- 1. parametric sampling ---------------------
    def _try_candidate(seq):
        nonlocal best_seq, best_score
        try:
            sc = evaluate_sequence(seq)
        except Exception:
            return
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(f"New best {sc:.6f} (len={len(seq)})")

    # a few lengths – search breadth is cheap
    lengths = [300, 600, 900, 1200, 1500, 2100]

    alphas = np.linspace(0.0, 2.0, 41)   # 0,0.05,…,2
    rs = np.linspace(0.90, 0.9999, 30)  # exponential bases

    for n in lengths:
        if time.time() > deadline:
            break

        # power‑law family
        for a in alphas:
            seq = _make_powerlaw(n, a).tolist()
            _try_candidate(seq)
            if time.time() > deadline:
                break
        if time.time() > deadline:
            break

            seq = _make_exponential(n, r).tolist()
            _try_candidate(seq)
            if time.time() > deadline:
                break
        if time.time() > deadline:

            raw = np.random.rand(n)
            raw = raw / raw.sum()
            _try_candidate(raw.tolist())
            if time.time() > deadline:
                break

"""Search for a step function that minimises evaluate_sequence.

The algorithm works in two stages.

1.  Generate many candidate sequences from simple parametric families:
    * power‑law decay  a_i ∝ (i+1)^(-α)
    * exponential decay a_i ∝ r^i
    * a few random Dirichlet draws.
    All candidates are projected onto the probability simplex (∑a_i = 1, a_i ≥ 0)
    and the best one according to evaluate_sequence is kept.

2.  Starting from the best candidate, run a projected sub‑gradient descent on the
    non‑smooth objective  f(x)=max_k (x*x)[k]  (the denominator of the
    evaluation function is constant on the simplex).  For the active
    convolution indices the sub‑gradient has a closed form, so each iteration
    costs O(n).  The iterate is projected back onto the simplex after every step.
    The loop terminates when the time budget expires or after a fixed number of
    iterations.  The best sequence found is returned.

The routine respects a time budget (default 960 s, usually called with 1000 s)
and prints progress information.  It never returns an invalid sequence,
so the caller always receives a finite score (or -inf if the time limit is
exceeded before a candidate is found)."""
import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _project_to_simplex(v):
    """Project vector v onto the probability simplex {x ≥ 0, sum(x)=1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # descending sort
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # extreme case – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1                  # zero‑based index of the last positive
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _make_powerlaw(n, alpha):
    """Return a length‑n vector proportional to (i+1)^(-alpha), normalised to sum = 1."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    raw = np.power(idx, -alpha)
    raw_sum = raw.sum()
    return raw / raw_sum


def _make_exponential(n, r):
    """Return a length‑n vector proportional to r**i, normalised to sum = 1."""
    raw = np.power(r, np.arange(n, dtype=np.float64))
    raw_sum = raw.sum()
    return raw / raw_sum

# ----------------------------------------------------------------------
def propose_candidate(seed=0, budget_s=960, **kwargs):
    """

    budget_s : float, optional

    """
    np.random.seed(seed)

    # leave a small safety margin before the hard timeout
    deadline = time.time() + budget_s - 2.0

    # ------------------------------------------------------------------
    # 0.   Keep the best sequence found so far
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float('inf')

    # if a previous construction is available, start from it
    if 'height_sequence_1' in globals():
        try:

            sc = evaluate_sequence(cand)
            best_seq = cand

            print(f"Started from height_sequence_1 → score {sc:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1.   Sample from simple parametric families
    # ------------------------------------------------------------------
    def _test_candidate(seq):

        try:

        except Exception:
            return
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()

    # a few lengths – the search is cheap, so we can afford several values
    lengths = [300, 600, 900, 1200, 1500, 2100]

    alphas = np.linspace(0.0, 2.0, 41)   # 0, 0.05, …, 2

            seg = x[lo:hi + 1][::-1] * 2.0

            x = _project_to_simplex(x + jitter)

        it += 1

    print(f"Search finished – best score {best_score:.6f} (length={len(best_seq)})")
    return best_seq
