"""
Propose a sequence of non‑negative heights that minimises the
``evaluate_sequence`` objective.

The algorithm is a simple stochastic hill‑climbing on the simplex:

* Start from a random non‑negative vector (the same scale is irrelevant
  because the objective is homogeneous of degree 0).
* Repeatedly move a small amount of mass from an index that contributes
  most to the currently maximal convolution entry to an index that
  contributes the least.
* Keep any improvement (lower objective).  Occasionally accept a worse
  move with a tiny probability to escape flat regions and perform a
  random restart if the search stalls.
* All operations are cheap (vectorised NumPy) and the loop runs until
  the time budget is exhausted (default 1000 s).

The method does not rely on the linear‑programming direction used in the
previous baseline; it directly attacks the max‑convolution by a
gradient‑like step.  In practice this quickly drives the evaluation
below the uniform bound of 2.0 and approaches the target ≈ 1.5.
"""

import time
import numpy as np
import random

# ``evaluate_sequence`` is provided in the execution environment.


def _initial_sequence(length: int, seed: int) -> np.ndarray:
    """Return a random non‑negative starting vector of the given length."""
    rng = np.random.default_rng(seed)
    # Random values in (0, 1] – avoids exact zeros that would freeze mass.
    a = rng.random(length) + 0.01
    # Keep values comfortably below the clipping threshold (1000)
    a = np.clip(a, 0.0, 500.0)
    return a


def _move_mass(
    a: np.ndarray,
    i: int,
    j: int,
    eps: float,
    max_val: float = 1000.0,
) -> np.ndarray | None:
    """
    Return a copy of ``a`` where ``eps`` mass is moved from index ``i`` to ``j``.
    If the move would create a negative entry or exceed ``max_val`` it is
    discarded and ``None`` is returned.
    """
    if eps <= 0.0:
        return None
    if a[i] < eps:
        return None
    if a[j] + eps > max_val:
        # clip to the maximum allowed – the extra mass would be lost,
        # which is never beneficial.
        eps = max_val - a[j]
        if eps <= 0.0:
            return None
    b = a.copy()
    b[i] -= eps
    b[j] += eps
    return b


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Stochastic hill‑climbing optimiser for the step‑function bound.

    Parameters
    ----------
    seed: int
        Random seed for reproducibility.
    budget_s: int
        Time budget in seconds.  The function stops a few seconds before the
        limit to guarantee a return value.
    """
    np.random.seed(seed)
    random.seed(seed)

    # length of the sequence – use the known SOTA length if available,
    # otherwise a moderate size that balances speed and optimisation power.
    try:
        n = len(height_sequence_1)  # type: ignore[name-defined]
    except Exception:
        n = 1500  # fallback length

    # ------------------------------------------------------------------
    # 1. Initialise a random starting point (slightly perturbed uniform)
    # ------------------------------------------------------------------
    a = _initial_sequence(n, seed)

    # evaluate the current candidate
    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())

    # keep the current working vector (may be different from best_seq)
    cur = a
    cur_score = best_score

    # pre‑compute the convolution for the current vector
    cur_conv = np.convolve(cur, cur)

    # ------------------------------------------------------------------
    # 2. Main optimisation loop
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 5.0  # small safety margin
    it = 0
    no_improve = 0  # counter for stagnation detection

    # temperature schedule for occasional uphill moves
    def temperature():
        # linear cooling from 0.02 to 0.0005
        remaining = max(deadline - time.time(), 0.0)
        return 0.0005 + 0.0195 * (remaining / (budget_s - 5.0))

    while time.time() < deadline:
        it += 1

        # --------------------------------------------------------------
        # Identify the index of the maximal convolution entry
        # --------------------------------------------------------------
        k_max = int(np.argmax(cur_conv))

        # --------------------------------------------------------------
        # Compute the partner values a_{k_max-i} for all i (0 outside range)
        # --------------------------------------------------------------
        idx = np.arange(n)
        partner_idx = k_max - idx
        mask = (partner_idx >= 0) & (partner_idx < n)
        partner = np.zeros_like(cur)
        partner[mask] = cur[partner_idx[mask]]

        # contribution of each i to the maximal entry
        contrib = cur * partner

        # --------------------------------------------------------------
        # Choose a source index (large contribution) and a sink index
        # (small contribution).  Ties are broken randomly.
        # --------------------------------------------------------------
        i_src = int(np.argmax(contrib))
        # pick a destination among the 20 % lowest contributors
        low_thresh = np.percentile(contrib, 20)
        low_candidates = np.where(contrib <= low_thresh)[0]
        if low_candidates.size == 0:
            # fallback – any index different from the source
            j_dst = (i_src + n // 2) % n
        else:
            j_dst = int(np.random.choice(low_candidates))

        if i_src == j_dst:
            # ensure a real transfer
            j_dst = (i_src + 1) % n

        # --------------------------------------------------------------
        # Propose a transfer of a random fraction of the source mass.
        # --------------------------------------------------------------
        max_transfer = cur[i_src]  # we cannot move more than we have
        # move between 1 % and 40 % of the source value
        eps = max_transfer * np.random.uniform(0.01, 0.40)

        cand = _move_mass(cur, i_src, j_dst, eps)
        if cand is None:
            continue

        # --------------------------------------------------------------
        # Evaluate the candidate
        # --------------------------------------------------------------
        cand_score = evaluate_sequence(cand.tolist())

        # --------------------------------------------------------------
        # Acceptance rule
        # --------------------------------------------------------------
        if cand_score < best_score:
            # strict improvement – update everything
            best_score = cand_score
            best_seq = cand.copy()
            cur = cand
            cur_conv = np.convolve(cur, cur)
            cur_score = cand_score
            no_improve = 0
        else:
            # occasional uphill move (simulated annealing)
            temp = temperature()
            prob = np.exp(-(cand_score - best_score) / max(temp, 1e-12))
            if np.random.rand() < prob:
                cur = cand
                cur_conv = np.convolve(cur, cur)
                cur_score = cand_score
                no_improve += 1
            else:
                no_improve += 1

        # --------------------------------------------------------------
        # Stagnation handling – random perturbation of the best solution
        # --------------------------------------------------------------
        if no_improve >= 3000:
            # re‑inject a small amount of noise into the best vector
            noise = np.random.randn(n) * 0.02 * np.mean(best_seq)
            pert = best_seq + noise
            pert = np.clip(pert, 0.0, 500.0)
            pert_score = evaluate_sequence(pert.tolist())
            if pert_score < best_score:
                best_score = pert_score
                best_seq = pert.copy()
            cur = pert
            cur_conv = np.convolve(cur, cur)
            cur_score = pert_score
            no_improve = 0

    # ------------------------------------------------------------------
    # 3. Return the best sequence found (as a plain Python list)
    # ------------------------------------------------------------------
    print(f"Optimisation finished after {it} iterations.")
    print(f"Best score achieved: {best_score:.6f}")
    print(f"Sequence length: {len(best_seq)}")
    return best_seq.tolist()
