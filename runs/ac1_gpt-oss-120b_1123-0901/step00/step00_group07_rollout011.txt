def generate_sidon_set(n, max_trials=1000, seed=None):
    best_set = []
    best_K = 0
    start = time.time()
    trial = 0
    while time.time() - start < max_trials:
        np.random.seed(seed + trial if seed is not None else None)
        order = np.random.permutation(n)
        current_set = []
        sums_used = set()
        for idx in order:
            # compute sums with idx and each element in current_set, also 2*idx
            conflict = False
            # 2*idx sum (self)
            if 2*idx in sums_used:
                conflict = True
            else:
                # check distinct pairs
                for j in current_set:
                    if idx + j in sums_used:
                        conflict = True
                        break
            if not conflict:
                # Add idx
                current_set.append(idx)
                # Add sums
                sums_used.add(2*idx)  # diagonal sum
                for j in current_set[:-1]:  # all previous j
                    sums_used.add(idx + j)
        if len(current_set) > best_K:
            best_K = len(current_set)
            best_set = list(current_set)
        trial += 1
        # maybe break if we achieve certain size threshold e.g., >= 100
    return best_set

The evaluation function is scale‑invariant:
    score = 2 * n * max(conv(a,a)) / (sum(a))**2 .
For a binary (0/1) sequence the sum equals the number of non‑zero entries K,
and the maximum of the convolution is exactly the maximum number of ordered
pairs (i,j) with the same sum i+j.  If the set of indices with a_i=1 is a
*Sidon set* (all unordered sums i+j are distinct), then each unordered sum
contributes at most two ordered pairs, and the diagonal sums 2*i are also
unique.  Hence max(conv) = 2 and the score reduces to

    score = 4 * n / K²               (K > 1, binary heights).

Thus the problem becomes: find a large Sidon set inside a short interval.
For a range of length n the theoretical maximum is K ≈ 2·√n, giving a score
close to 1.  We use a simple random‑greedy construction with a post‑processing
phase that repeatedly tries to insert missing numbers without breaking the
Sidon property.  Many independent attempts are performed within the time
budget; the best (largest K / smallest score) construction is returned as a
list of floats (1.0 on the selected positions, 0.0 elsewhere).

The algorithm is completely self‑contained, uses only the standard library
and NumPy (which is already imported in the evaluation environment), and
does not rely on any external files or solvers.
"""

import time
import random
from typing import List, Set, Tuple

# ----------------------------------------------------------------------
# Helper functions
# ----------------------------------------------------------------------

def greedy_sidon(bound_n: int, rng: random.Random) -> Set[int]:
    """
    Build a Sidon set inside {0,…,bound_n‑1} using a random ordering.
    The construction guarantees that all unordered sums i+j (including i=j)
    are distinct, i.e. max(conv) = 2 for the binary indicator.
    """
    order = list(range(bound_n))
    rng.shuffle(order)

    chosen: Set[int] = set()
    used_sums: Set[int] = set()          # stores unordered sums already taken

    for idx in order:
        # diagonal sum 2*idx must be unused
        if 2 * idx in used_sums:
            continue
        # check sums with existing elements
        conflict = False
        for j in chosen:
            if idx + j in used_sums:
                conflict = True
                break
        if conflict:
            continue

        # safe to add idx
        for j in chosen:
            used_sums.add(idx + j)
        used_sums.add(2 * idx)
        chosen.add(idx)

    # ------------------------------------------------------------------
    # Optional improvement phase: try to insert any remaining indices.
    # This loop runs until no further insertion is possible.
    # ------------------------------------------------------------------
    added = True
    while added:
        added = False
        missing = [i for i in range(bound_n) if i not in chosen]
        rng.shuffle(missing)
        for idx in missing:
            if 2 * idx in used_sums:
                continue
            conflict = False
            for j in chosen:
                if idx + j in used_sums:
                    conflict = True
                    break
            if not conflict:
                for j in chosen:
                    used_sums.add(idx + j)
                used_sums.add(2 * idx)
                chosen.add(idx)
                added = True
                break  # restart scanning after each successful insertion
    return chosen


def sidon_score(chosen: Set[int]) -> Tuple[float, int, int]:
    """
    Compute the evaluation score for a binary Sidon set.
    Returns (score, n_needed, K) where
        n_needed = max(chosen) + 1   (effective length of the step function)
        K        = |chosen|          (number of ones)
    For a proper Sidon set max(conv) = 2 (or 1 if K == 1).
    """
    K = len(chosen)
    if K == 0:
        return float('inf'), 0, 0
    n_needed = max(chosen) + 1
    max_b = 2 if K > 1 else 1
    score = 2.0 * n_needed * max_b / (K * K)      # = 4 n / K² for K>1
    return score, n_needed, K


def build_sequence(chosen: Set[int]) -> List[float]:
    """Create the final step‑function list from the index set."""
    if not chosen:
        return [0.0]                     # should never happen (invalid)
    length = max(chosen) + 1
    seq = [0.0] * length
    for idx in chosen:
        seq[idx] = 1.0
    return seq


# ----------------------------------------------------------------------
# Main entry point required by the competition framework
# ----------------------------------------------------------------------

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a dense Sidon set within the allowed time budget.
    The result is a binary step‑function (list of floats) whose evaluation
    score is (usually) well below the uniform‑sequence baseline of 2.0.
    """
    # ------------------------------------------------------------------
    # initialise random engines and timing
    # ------------------------------------------------------------------
    random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # small safety margin

    # Length of the reference construction (if available).  We do not need
    # to keep that length – the algorithm chooses its own optimal length.
    ref_len = len(globals().get('height_sequence_1', []))
    if ref_len == 0:
        ref_len = 2000   # fallback, never used for the final answer

    best_chosen: Set[int] = set()
    best_score = float('inf')
    best_K = 0
    best_n = 0
    attempts = 0

    # ------------------------------------------------------------------
    # Main random‑restart loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        attempts += 1
        # Choose an interval size.  Larger intervals give more room for a
        # bigger Sidon set, but also increase the score via the factor n.
        # Empirically values between 800 and 4000 work well.
        bound_n = random.randint(800, 4000)

        # Build a Sidon set for this interval
        chosen = greedy_sidon(bound_n, random)

        # Compute its theoretical score (fast, no convolution)
        score, n_needed, K = sidon_score(chosen)

        # Keep the best construction seen so far
        if score < best_score:
            best_score = score
            best_chosen = chosen.copy()
            best_K = K
            best_n = n_needed
            print(f"[{time.time() - start_time:6.2f}s] "
                  f"New best after {attempts} attempts: K={K}, n={n_needed}, "
                  f"score≈{score:.6f}")

        # Occasionally force a short pause to let the time check happen
        if attempts % 50 == 0 and time.time() > deadline:
            break

    # ------------------------------------------------------------------
    # Build the final sequence from the best Sidon set.
    # Use the exact evaluation function for double‑checking.
    # ------------------------------------------------------------------
    final_seq = build_sequence(best_chosen)

    # The challenge uses the imported evaluate_sequence function.
    # This call also serves as a sanity check (it should match the
    # theoretical score up to rounding errors).
    try:
        final_score = evaluate_sequence(final_seq)
    except Exception as exc:          # defensive; should never happen
        final_score = float('inf')
        print("Evaluation error:", exc)

    print(f"Search finished after {attempts} attempts. "
          f"Best set: K={best_K}, n={best_n}, score≈{best_score:.6f}, "
          f"final evaluate_sequence={final_score:.6f}")

    return final_seq
