import numpy as np, time, sys
from scipy.optimize import linprog, OptimizeResult

def solve_convolution_lp(f):
    n = len(f)
    # Build sparse matrix A_ub of shape (2*n-1 + n, n)
    # first 2n-1 rows for conv constraints: sum_i f[i] * b[k-i] <= M
    M = np.max(np.convolve(f, f))
    # Build data for conv constraints:
    rows = []
    cols = []
    data = []
    for i in range(n):
        # f[i] contributes to rows i + j for each j in 0..n-1, column j
        row_start = i
        rows.extend(range(row_start, row_start + n))
        cols.extend(range(n))
        data.extend([f[i]] * n)
    # Build using scipy.sparse.coo_matrix
    from scipy import sparse
    A_conv = sparse.coo_matrix((data, (rows, cols)), shape=(2*n-1, n)).tocsr()
    # bound rows: with <= M
    b_conv = np.full(2*n-1, M)
    # nonnegativity constraints: -I b <= 0
    A_nonneg = -sparse.identity(n, format='csr')
    b_nonneg = np.zeros(n)
    A_ub = sparse.vstack([A_conv, A_nonneg])
    b_ub = np.concatenate([b_conv, b_nonneg])
    # objective: maximize sum b => minimize -sum b
    c = -np.ones(n)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs', options={'time_limit': 5.0})
    if res.success:
        return res.x
    else:
        return None

def get_good_direction_to_move_into(seq):
    n = len(seq)
    sum_seq = np.sum(seq)
    # Normalize sequence to have sum sqrt(2n) (so evaluation = max conv)
    target_sum = np.sqrt(2 * n)
    seq_norm = seq * (target_sum / sum_seq) if sum_seq !=0 else seq
    M = np.max(np.convolve(seq_norm, seq_norm))
    b = solve_convolution_lp(seq_norm)
    if b is None:
        return None
    # Normalize b to target sum
    b = b * (target_sum / np.sum(b))
    # mix with step size candidates
    best_mix = None
    best_val = evaluate_sequence(seq_norm.tolist())
    # try different mixing fractions
    for t in [0.25, 0.5, 0.75]:
        mixed = (1 - t) * seq_norm + t * b
        # ensure nonnegative (should be)
        val = evaluate_sequence(mixed.tolist())
        if val < best_val:
            best_val = val
            best_mix = mixed
    return best_mix if best_mix is not None else seq_norm

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    import numpy as np, time, random
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 5.0
    global height_sequence_1
    if 'height_sequence_1' in globals() and height_sequence_1:
        n = len(height_sequence_1)
        curr = np.array(height_sequence_1, dtype=float)
    else:
        n = 2000
        curr = rng.random(n)
    # normalize sum
    target_sum = np.sqrt(2 * n)
    curr = curr * (target_sum / np.sum(curr))
    best_seq = curr.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f'Initial score: {best_score:.6f}')
    # main loop
    iter_count = 0
    while time.time() < deadline:
        iter_count += 1
        # attempt LP direction
        if rng.random() < 0.7:
            new_seq = get_good_direction_to_move_into(curr)
        else:
            # random local perturbation
            new_seq = curr.copy()
            idx = rng.integers(0, n)
            delta = rng.normal(scale=0.1)
            new_seq[idx] = max(0.0, new_seq[idx] + delta)
            # renormalize sum
            new_seq = new_seq * (target_sum / np.sum(new_seq))
        if new_seq is None:
            continue
        try:
            score = evaluate_sequence(new_seq.tolist())
        except Exception:
            continue
        if score < best_score:
            best_score = score
            best_seq = new_seq.copy()
            print(f'Iter {iter_count}: new best {best_score:.6f}')
        # update current sequence for next iteration
        # accept if improvement else possibly accept with small prob
        if score < evaluate_sequence(curr.tolist()):
            curr = new_seq
        else:
            # occasional acceptance to escape local minima
            if rng.random() < 0.05:
                curr = new_seq
    print(f'Finished after {iter_count} iterations, best score {best_score:.6f}')
    return best_seq.tolist()

"""
Simple stochastic optimizer for the step‑function bound.

The algorithm works with a fixed length `n` (taken from the global
`height_sequence_1` when present).  The sequence is normalised so that
its L1‑norm equals `sqrt(2*n)`.  Under that normalisation the evaluation
value returned by ``evaluate_sequence`` is exactly the maximal
convolution value, so minimising the evaluation is equivalent to
minimising `max(conv(a,a))`.

The optimiser alternates a cheap hill‑climbing move (transfer a tiny
amount of mass from one position to another while keeping the total
mass constant) and an occasional linear‑programming step that finds a
direction `b` with a larger L1‑norm while respecting the current
convolution bound.  The two strategies complement each other:
random moves explore the neighbourhood quickly, while the LP step can
produce a more global improvement.

The code respects the 1000 s budget, prints occasional progress, and
returns the best sequence found.
"""

import time
import numpy as np
from scipy import optimize, sparse

# ----------------------------------------------------------------------
# Helper: linear programme that, given a fixed sequence f and a bound M,
# finds a non‑negative sequence b maximising sum(b) under the constraint
#   (f * b)[k] <= M   for all k = 0 .. 2n‑2
# ----------------------------------------------------------------------
def solve_convolution_lp(f, M, time_limit=5.0):
    """
    Solve the LP:
        maximise    sum(b)
        subject to  conv(f, b) <= M   (for every lag)
                    b >= 0
    Returns the optimal b as a dense 1‑d NumPy array, or ``None`` on
    failure / timeout.
    """
    n = f.shape[0]

    # ----- build convolution constraints (2n‑1 rows) -----------------
    # For each column j (the variable b_j) we need the entries
    #   f[i]  for rows i+j (i = 0..n‑1)
    # This is a Toeplitz pattern and can be built efficiently with
    # sparse matrices.
    row_idx = np.concatenate([np.arange(j, j + n) for j in range(n)])
    col_idx = np.concatenate([np.full(n, j, dtype=int) for j in range(n)])
    data = np.concatenate([f] * n)                     # repeat f for each column

    A_conv = sparse.csr_matrix((data, (row_idx, col_idx)),
                               shape=(2 * n - 1, n))
    b_conv = np.full(2 * n - 1, M, dtype=float)

    # ----- non‑negativity constraints  -b <= 0 -----------------------
    A_nonneg = -sparse.identity(n, format='csr')
    b_nonneg = np.zeros(n, dtype=float)

    # ----- combine all constraints ----------------------------------
    A_ub = sparse.vstack([A_conv, A_nonneg])
    b_ub = np.concatenate([b_conv, b_nonneg])

    # ----- objective (maximise sum(b) == minimise -sum(b)) ----------
    c = -np.ones(n, dtype=float)

    # use HiGHS solver; give it a modest time limit per call
    res = optimize.linprog(c,
                           A_ub=A_ub,
                           b_ub=b_ub,
                           method='highs',
                           options={'time_limit': time_limit, 'disp': False})

    if res.success:
        return res.x
    return None


# ----------------------------------------------------------------------
# Helper: produce a candidate sequence by mixing the current one with an
# LP‑derived direction.
# ----------------------------------------------------------------------
def lp_improvement_step(cur, target_sum):
    """
    Perform one LP‑based improvement step.
    Returns a new sequence (already normalised to ``target_sum``) or
    ``None`` if the LP fails.
    """
    n = cur.shape[0]
    # current bound M = max(conv(cur,cur))
    M = np.max(np.convolve(cur, cur))

    # solve LP to obtain a higher‑L1 direction
    b = solve_convolution_lp(cur, M, time_limit=4.0)
    if b is None:
        return None

    # normalise the LP solution to the required L1‑norm
    b = b * (target_sum / np.sum(b))

    # try a few mixing fractions and keep the best
    best_seq = cur
    best_score = evaluate_sequence(cur.tolist())
    for t in (0.25, 0.5, 0.75):
        cand = (1.0 - t) * cur + t * b
        # mixing two normalised vectors already preserves the sum
        score = evaluate_sequence(cand.tolist())
        if score < best_score:
            best_score = score
            best_seq = cand
    return best_seq


# ----------------------------------------------------------------------
# Helper: a cheap random hill‑climbing move that transfers a small amount
# of mass from one index to another while keeping the total sum fixed.
# ----------------------------------------------------------------------
def random_mass_transfer(seq, avg, max_frac):
    """
    Return a new sequence obtained by moving a random amount of mass
    from index i to index j.  The amount moved is at most
    ``max_frac * avg`` where ``avg`` = target_sum / n.
    The returned sequence respects non‑negativity.
    """
    n = seq.shape[0]
    i, j = np.random.randint(0, n, size=2)
    if i == j:
        return seq
    # amount to move (cannot exceed what is available at i)
    delta = np.random.uniform(0.0, max_frac * avg)
    delta = min(delta, seq[i])
    new_seq = seq.copy()
    new_seq[i] -= delta
    new_seq[j] += delta
    return new_seq

            best_score = cand_score
            best_seq = cand_seq.copy()
            cur_seq = cand_seq.copy()
            print(f"[{it:07d}] New best = {best_score:.6f} (LP calls: {lp_calls})")
        else:
            # occasional non‑improving acceptance to escape shallow traps
            # temperature decreases linearly with progress
            temperature = max(0.001, 0.05 * (1.0 - progress))
            if np.exp(-(cand_score - best_score) / temperature) > rng.random():
                cur_seq = cand_seq.copy()

        # very occasional full random restart (helps when stuck)
        if it % 5000 == 0 and rng.random() < 0.1:
            cur_seq = rng.random(n)
            cur_seq = cur_seq * (target_sum / np.sum(cur_seq))
            print(f"[{it:07d}] Random restart triggered.")

    print(f"[DONE] Iterations={it}, best score={best_score:.6f}")
    return best_seq.tolist()
