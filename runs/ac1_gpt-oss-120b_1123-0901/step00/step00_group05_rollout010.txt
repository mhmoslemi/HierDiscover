            if prods.sum() == 0:
                # all zero? then random move
                i = np.random.randint(len(seq))
                j = np.random.randint(len(seq))
            else:
                # choose pair with probability proportional to product
                probs = prods / prods.sum()

            # but we removed eps from src and added to dst, sum unchanged already.
            # Ensure non-negative
            if new_seq[src] < 0:

            # compute new score
            conv_new = np.convolve(new_seq, new_seq)
            max_new = conv_new.max()
            new_score = 2 * len(seq) * max_new / (new_seq.sum()**2)

            # acceptance
            if new_score < score or np.random.rand() < np.exp((score - new_score) / temp):
                seq = new_seq
                score = new_score

            # cooling
            temp *= cooling
            it += 1
            if it % 2000 == 0:

                # occasional random perturbation
                if np.random.rand() < 0.2:

                    idx1, idx2 = np.random.choice(len(seq), 2, replace=False)
                    delta = np.random.normal(scale=0.02) * seq[idx1]
                    delta = np.clip(delta, -seq[idx1], seq[idx2])
                    seq[idx1] -= delta
                    seq[idx2] += delta

def project_simplex(v):
    n = len(v)
    # sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

sequence found within the time budget is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the simplex {x ≥ 0, sum x = 1}."""
    if v.ndim != 1:
        raise ValueError("project_simplex expects a 1‑D array")
    n = v.shape[0]
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def generate_binomial_coeffs(n: int) -> np.ndarray:
    """
    Produce the first n coefficients of the series
        c_i = binom(2i, i) / 4**i  =  (1 - 1/(2i)) * c_{i‑1},  c_0 = 1.
    They are the positive coefficients of (1 - z)^‑½.
    """
    coeffs = np.empty(n, dtype=np.float64)
    coeffs[0] = 1.0
    for i in range(1, n):
        coeffs[i] = coeffs[i - 1] * (1.0 - 1.0 / (2.0 * i))
    return coeffs


def fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Convolution using FFT (real inputs).
    Returns a vector of length len(a)+len(b)-1.
    """
    n = a.shape[0]
    m = b.shape[0]
    L = 1 << ((n + m - 1).bit_length())   # next power of two
    Fa = np.fft.rfft(a, L)
    Fb = np.fft.rfft(b, L)
    conv = np.fft.irfft(Fa * Fb, L)[: n + m - 1]
    return conv


def smooth_max_and_gradient(a: np.ndarray, beta: float):
    """
    Compute:
      - conv   = a * a  (full convolution)
      - maxc   = max(conv)
      - grad   = gradient of the smooth‐max (softmax with temperature 1/beta)
                w.r.t. a, i.e.  ∇_a  ( 1/β log Σ exp(β conv) )
    The gradient formula reduces to:
        grad_j = 2 * Σ_i a_i * p_{i+j},
    where p is the soft‑max distribution over the convolution entries.
    """
    n = a.shape[0]
    conv = fft_convolve(a, a)                     # length 2n‑1
    # soft‑max weights (stable version)
    maxc = conv.max()
    shifted = conv - maxc                         # prevents overflow
    w = np.exp(beta * shifted)
    p = w / w.sum()                               # shape (2n‑1,)

    # cross‑correlation a ⋆ p  (equivalently convolution of reversed a with p)
    a_rev = a[::-1]
    cross = fft_convolve(a_rev, p)                # length 3n‑2
    start = n - 1
    grad = 2.0 * cross[start:start + n]           # size n
    return conv, grad, maxc


def evaluate_objective(seq: np.ndarray) -> float:
    """
    Directly compute the evaluation function used by the competition.
    The sequence is assumed to be non‑negative.  Scaling does not matter,
    but we keep the natural ℓ1‑norm = 1 to keep numbers well‑behaved.
    """
    n = seq.shape[0]
    conv = np.convolve(seq, seq)                  # plain fast enough for n ≤ 4000
    maxb = conv.max()
    sum_a = seq.sum()
    if sum_a < 1e-12:                              # guard against degenerate case
        return np.inf
    return float(2.0 * n * maxb / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """Return a sequence that (hopefully) beats the current bound 2.00."""
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # a few seconds safety margin

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # try several lengths; longer sequences give a theoretical bound ≈π/2
    # ------------------------------------------------------------------
    lengths = [200, 500, 1000, 2000]           # feel free to adjust
    # ensure we also test the previously known sequence if available
    if "height_sequence_1" in globals():
        try:
            seq0 = np.array(height_sequence_1, dtype=float)
            seq0 = np.maximum(seq0, 0.0)
            if seq0.sum() == 0:
                seq0 = np.ones_like(seq0) / seq0.size
            seq0 = project_simplex(seq0)      # normalise to sum=1
            sc0 = evaluate_objective(seq0)
            best_seq, best_score = seq0, sc0
            print(f"Initial candidate from previous work: score={sc0:.6f}")
        except Exception as e:
            print("Failed to import height_sequence_1:", e)

    # ------------------------------------------------------------------
    # Core optimisation loop over the selected lengths
    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"\n=== Optimising length n={n} ===")

        # --------------------------------------------------------------
        # initialise with analytic coefficients (or random if n very small)
        # --------------------------------------------------------------
        a = generate_binomial_coeffs(n)
        a = a / a.sum()               # project onto the simplex (sum = 1)

        # optional random perturbation to escape tiny symmetries
        a += 0.001 * np.random.randn(n)
        a = np.maximum(a, 0.0)
        a = project_simplex(a)

        # hyper‑parameters for the gradient descent
        beta = 0.5                     # start temperature (softmax sharpness)
        beta_max = 40.0
        beta_factor = 1.005            # slowly increase beta
        eta0 = 0.8                     # initial learning rate
        decay = 0.999                  # learning‑rate decay per iteration
        it = 0
        last_improve = 0
        max_iters = 20000              # generous upper bound (will be cut by time)

        while time.time() < deadline and it < max_iters:
            # ----------------------------------------------------------
            # compute smooth max, gradient, and the true objective
            # ----------------------------------------------------------
            conv, grad, maxc = smooth_max_and_gradient(a, beta)
            score = 2.0 * n * maxc / (a.sum() ** 2)   # same as evaluate_sequence for sum=1

            # keep the best overall sequence
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"[n={n}] iter={it}  score={score:.6f}  beta={beta:.2f}")

                # early exit if we already passed the target
                if best_score < 1.5:
                    print("Target reached, terminating early.")
                    break

            # ----------------------------------------------------------
            # gradient step (projected gradient descent)
            # ----------------------------------------------------------
            eta = eta0 * (decay ** it)          # decreasing stepsize
            a_new = a - eta * grad
            # projection onto the simplex keeps sum=1 and non‑negativity
            a_new = project_simplex(a_new)

            # optional occasional random “jump” to escape plateaus
            if it % 2000 == 0 and it > 0:
                if np.random.rand() < 0.3:
                    # move a random fraction of mass from one index to another
                    i, j = np.random.choice(n, 2, replace=False)
                    delta = 0.05 * a_new[i]    # up to 5 % of entry i
                    a_new[i] -= delta
                    a_new[j] += delta
                    a_new = project_simplex(a_new)

            a = a_new

            # increase soft‑max sharpness slowly
            if beta < beta_max:
                beta *= beta_factor

            # stop if we have not improved for a long time
            if it - last_improve > 5000:
                # re‑initialize from a fresh random point to explore new basins
                a = np.random.rand(n)
                a = project_simplex(a)
                last_improve = it
                beta = 0.5

            it += 1

        # finished this length; report best for it
        print(f"Finished length {n}. Best score seen: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the best sequence found (as a plain Python list)
    # ------------------------------------------------------------------
    if best_seq is None:
        # as a safety fallback, return a simple constant sequence
        n = 300
        best_seq = np.ones(n) / n
        print("No candidate found – returning a fallback constant sequence.")
    else:
        print(f"\n=== Overall best score: {best_score:.6f} (length {best_seq.size}) ===")

    # Clip to the limits enforced by evaluate_sequence (0 … 1000) for safety
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    return best_seq.tolist()
