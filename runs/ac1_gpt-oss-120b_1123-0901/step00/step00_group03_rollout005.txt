import numpy as np
import time
import math

def _project_simplex(v):
    """Project v onto the probability simplex (nonnegative, sum=1)."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

    Gradient‑descent / simulated‑annealing search for a non‑negative step

    The algorithm starts from a random (or the provided) sequence,
    repeatedly moves mass away from the indices that dominate the
    peak of the self‑convolution, and keeps the best sequence found
    before the time budget expires.
    """

    start_time = time.time()

        a = a / a.sum()

        n = 600
        a = rng.random(n)
        a = a / a.sum()

    # Ensure non‑negativity (just in case)
    a = np.maximum(a, 0.0)

    def objective(seq, conv=None):

        return float(2.0 * len(seq) * maxb / (s * s))

    # initial state
    conv_a = np.convolve(a, a)
    best_seq = a.copy()
    best_score = objective(a, conv_a)
    curr_score = best_score
    curr_conv = conv_a

    # ------------------------------------------------------------------
    # 3) hyper‑parameters
    # ------------------------------------------------------------------
    step_scale = 0.5          # initial step length
    step_decay = 0.9995       # geometric decay of the step size
    stagnation = 0            # counter of iterations without improvement
    max_stagnation = 2000     # after this we re‑initialise a fresh random start
    top_k = 3                 # how many of the largest convolution entries we use
    start = time.time()
    deadline = start + budget_s - 0.5   # small safety margin

    iteration = 0
    print(f"[{0:.1f}s] start optimisation – n={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) main optimisation loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        iteration += 1

        # ---- identify a few dominant convolution entries -----------------
        # we use the top‑k indices (k = top_k); this gives a smoother direction
        # than using only the single maximum.
        if top_k >= curr_conv.size:
            idx_peaks = np.arange(curr_conv.size)
        else:
            idx_peaks = np.argpartition(curr_conv, -top_k)[-top_k:]

        # weight vector w that has a unit weight on the selected indices
        w = np.zeros_like(curr_conv)
        w[idx_peaks] = 1.0

        # ---- compute the sub‑gradient of the (soft) max -----------------
        # grad_i = Σ_k w_k * 2 * a_{k-i}  (valid indices only)
        grad = 2.0 * np.correlate(w, a, mode='valid')   # length = n

        # ---- direction that attempts to lower the peak -------------------
        direction = -grad
        direction -= direction.mean()   # enforce Σ direction = 0  (sum stays constant)

        # If the direction is numerically zero we inject a tiny random push.
        if np.linalg.norm(direction) < 1e-12:
            direction = rng.normal(scale=0.01, size=n)
            direction -= direction.mean()

        # ---- maximum feasible step before any component would become negative
        neg = direction < 0
        if np.any(neg):
            max_step = np.min(-a[neg] / direction[neg])
        else:
            max_step = np.inf

        # ---- actual step size (shrink a little for safety) -------------
        step = min(step_scale, 0.99 * max_step)

        # ---- propose the new point ---------------------------------------
        a_new = a + step * direction
        a_new = np.maximum(a_new, 0.0)                # clip possible tiny negatives
        total = a_new.sum()
        if total == 0.0:                     # degenerate – skip this move
            continue
        a_new = a_new * (a.sum() / total)    # renormalise to keep the original sum

        # ---- evaluate ----------------------------------------------------
        conv_new = np.convolve(a_new, a_new)
        new_score = objective(a_new, conv_new)

        # ---- accept / reject -----------------------------------------------
        improved = new_score <= curr_score
        if improved:
            # keep the better (or equal) point
            a = a_new
            curr_score = new_score
            curr_conv = conv_new
            if new_score < best_score:
                best_score = new_score
                best_seq = a_new.copy()
                stagnation = 0
                print(f"[{time.time() - start:.1f}s] it {iteration:6d} – new best {best_score:.6f}")
            else:
                stagnation += 1
        else:
            # occasional uphill move with a temperature that shrinks over time
            temperature = max(1e-6, step_scale)   # a simple proxy for temperature
            prob = math.exp(-(new_score - curr_score) / temperature)
            if rng.random() < prob:
                a = a_new
                curr_score = new_score
                curr_conv = conv_new
            stagnation += 1

        # ---- step‑size decay ------------------------------------------------
        step_scale *= step_decay

        # ---- restart if we have been stuck for too long -------------------
        if stagnation >= max_stagnation:
            a = rng.random(n)
            a = _project_simplex(a)
            curr_conv = np.convolve(a, a)
            curr_score = objective(a, curr_conv)
            stagnation = 0
            step_scale = 0.5
            print(f"[{time.time() - start:.1f}s] it {iteration:6d} – restart (stagnation)")

        # ---- occasional progress report ------------------------------------
        if iteration % 500 == 0:
            elapsed = time.time() - start
            print(f"[{elapsed:.1f}s] iteration {iteration}, best {best_score:.6f}")

    # ----------------------------------------------------------------------
    # 5) final tiny local search around the best point found
    # ----------------------------------------------------------------------
    a_local = best_seq.copy()
    for _ in range(300):
        i, j = rng.integers(0, n, size=2)
        if a_local[i] == 0:
            continue
        # move a small random amount from i to j
        eps = min(a_local[i], rng.random() * 0.01)
        a_local[i] -= eps
        a_local[j] += eps
        a_local = _project_simplex(a_local)
        conv_loc = np.convolve(a_local, a_local)
        val = objective(a_local, conv_loc)
        if val < best_score:
            best_score = val
            best_seq = a_local.copy()
            print(f"[{time.time() - start:.1f}s] local tweak improved to {best_score:.6f}")

    print(f"[{time.time() - start:.1f}s] optimisation finished – best score {best_score:.6f}")
    return best_seq.tolist()
