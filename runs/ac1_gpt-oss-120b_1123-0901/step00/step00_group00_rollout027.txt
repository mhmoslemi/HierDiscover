def propose_candidate(seed=123, budget_s=1000, n=2304):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5
    # initialize shape
    seq = np.sqrt(2) * (np.sqrt(np.arange(1, n+1)) - np.sqrt(np.arange(n)))  # vectorized
    # ensure sum already sqrt(2n); due to telescoping sum
    # optionally clip to max 1000
    best_seq = seq.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    # track number of iterations
    iter = 0
    temperature = 1.0
    while time.time() < deadline:
        # compute convolution and max shift
        conv = np.convolve(seq, seq)
        max_idx = int(np.argmax(conv))
        max_val = conv[max_idx]
        # propose move
        # select i with probability proportional to contribution to max_idx
        start = max(0, max_idx - (n-1))
        end = min(n-1, max_idx)
        idx_range = np.arange(start, end+1)
        contributions = seq[idx_range] * seq[max_idx - idx_range]
        # choose i weighted
        if contributions.sum() > 0:
            probs = contributions / contributions.sum()
            i = np.random.choice(idx_range, p=probs)
        else:
            i = np.random.randint(n)
        # pick random target j not equal to i
        j = i
        while j == i:
            j = np.random.randint(n)
        # move some mass
        delta = seq[i] * (0.01 + 0.09 * np.random.rand())
        # ensure not exceed seq[i]
        delta = min(delta, seq[i] * 0.5)
        # adjust
        new_seq = seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        # enforce nonnegative
        if new_seq[i] < 0:
            new_seq[i] = 0.0
        # evaluate
        score = evaluate_sequence(new_seq.tolist())
        # accept if better
        if score < best_score:
            best_score = score
            best_seq = new_seq.copy()
            seq = new_seq
            print(f"Iter {iter}: improved score {best_score:.6f}")
        else:
            # maybe accept with probability
            # temperature schedule
            prob = np.exp((best_score - score) / (temperature + 1e-12))
            if np.random.rand() < prob:
                seq = new_seq
        # temperature decay
        iter += 1
        temperature *= 0.99995  # slowly reduce
        # occasional random reset
        if iter % 5000 == 0:
            # small random perturbation across seq
            seq += np.random.randn(n) * 0.001
            seq = np.maximum(seq, 0.0)
    return best_seq.tolist()

"""
Simulated‑annealing optimiser for the step‑function evaluation problem.

The algorithm starts from the analytically motivated sequence

    a_i = √2·(√(i+1) – √i)   (i = 0,…,n‑1)

which corresponds to a discretised density f(t)=1/(2√t) whose
autoconvolution is (asymptotically) constant.  This sequence already yields
a score ≈1.65 for n≈2300, better than the uniform construction (score = 2).

From this starting point the optimiser performs many small “mass‑transfer”
moves: a random amount is taken from an index that contributes heavily to the
current convolution peak and added to another (random) index.  Moves are
accepted if they improve the evaluation; otherwise they are accepted with a
Metropolis probability that gradually cools down.  The total sum of the
coefficients is preserved, so the scaling invariance of the objective is
respected automatically.

The routine runs for the allotted time budget (default 1000 s) and returns
the best sequence found.
"""

import numpy as np
import time


def _initial_sqrt_derivative(n: int) -> np.ndarray:
    """
    Initialise the sequence a_i = √2·(√(i+1) – √i).

    The sum of the produced vector equals √(2·n), which is the normalisation
    used by ``evaluate_sequence`` (the objective is invariant under global
    scaling, but keeping the sum fixed avoids the “sum too close to zero”
    guard in the evaluator).
    """
    i = np.arange(n, dtype=float)
    a = np.sqrt(2.0) * (np.sqrt(i + 1.0) - np.sqrt(i))
    # guard against rounding error – force the exact sum
    target = np.sqrt(2.0 * n)
    a *= target / a.sum()
    return a


def _evaluate(seq: np.ndarray) -> float:
    """Wrap the external ``evaluate_sequence`` function."""
    # ``evaluate_sequence`` expects a plain Python list of floats.
    return evaluate_sequence(seq.tolist())


def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs) -> list[float]:
    """
    Search for a low‑scoring step‑function.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Time budget in seconds.
    kwargs : dict
        Optional arguments – currently only ``length`` (default = 2304) is
        honoured.

    Returns
    -------
    list[float]
        The best sequence discovered before the time limit expires.
    """
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # Problem size – the reference construction uses 2304 coefficients.
    # Larger n gives a larger search space but also a more expensive
    # convolution (O(n²)).  The default is the known size.
    # ------------------------------------------------------------------
    n = int(kwargs.get("length", 2304))

    # time management
    deadline = time.time() + budget_s - 5.0   # keep a few seconds margin

    # ------------------------------------------------------------------
    # 1.  Initialise from the analytic shape.
    # ------------------------------------------------------------------
    seq = _initial_sqrt_derivative(n)

    best_seq = seq.copy()
    best_score = _evaluate(best_seq)
    print(f"initial score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2.  Simulated‑annealing loop.
    # ------------------------------------------------------------------
    temperature = 1.0          # initial temperature for Metropolis steps
    iteration = 0

    while time.time() < deadline:
        # Full convolution – the bottleneck (≈5 · 10⁶ operations for n≈2300)
        conv = np.convolve(seq, seq)
        k_max = int(np.argmax(conv))      # index of the current peak

        # --------------------------------------------------------------
        # Choose an index that participates strongly in the peak.
        # For the chosen k_max the term a_i·a_{k_max-i} is the contribution
        # of index i.  Weight the selection by the size of that contribution.
        # --------------------------------------------------------------
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idxs = np.arange(i_low, i_high + 1)
        contrib = seq[idxs] * seq[k_max - idxs]
        if contrib.sum() > 0.0:
            probs = contrib / contrib.sum()
            i = np.random.choice(idxs, p=probs)
        else:
            i = np.random.randint(n)

        # --------------------------------------------------------------
        # Destination index – pick a (different) random position.
        # --------------------------------------------------------------
        j = i
        while j == i:
            j = np.random.randint(n)

        # --------------------------------------------------------------
        # Propose moving a small amount of mass from i to j.
        # The step size shrinks with the temperature.
        # --------------------------------------------------------------
        step = max(temperature * 0.1, 1e-4)          # never zero
        delta = step * seq[i] * np.random.rand()    # ≤ step·a_i
        delta = min(delta, seq[i])                  # safety

        new_seq = seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        # safeguard – floating‑point round‑off could produce a tiny negative
        if new_seq[i] < 0.0:
            new_seq[i] = 0.0

        # ------------------------------------------------------------------
        # 3.  Evaluate the candidate.
        # ------------------------------------------------------------------
        score = _evaluate(new_seq)

        # ------------------------------------------------------------------
        # 4.  Acceptance rule.
        #    - If the new score is better we always accept.
        #    - Otherwise accept with Metropolis probability to keep the walk
        #      from getting stuck.
        # ------------------------------------------------------------------
        if score < best_score:
            best_score = score
            best_seq = new_seq.copy()
            seq = new_seq
            print(f"Iter {iteration:6d}: new best {best_score:.6f}")
        else:
            # Metropolis probability (temperature controls the acceptance)
            prob = np.exp((best_score - score) / (temperature + 1e-12))
            if np.random.rand() < prob:
                seq = new_seq

        # ------------------------------------------------------------------
        # 5.  Temperature decay & occasional jitter.
        # ------------------------------------------------------------------
        temperature *= 0.9999                     # exponential cooling

        if iteration % 2000 == 0 and iteration > 0:
            # tiny random jitter to escape shallow plateaus
            jitter = np.random.randn(n) * 5e-4
            seq = np.maximum(seq + jitter, 0.0)
            # renormalise the total sum (scaling does not affect the score)
            target_sum = np.sqrt(2.0 * n)
            seq *= target_sum / seq.sum()

        iteration += 1

    # ------------------------------------------------------------------
    # Return the best sequence found (as a plain Python list of floats).
    # ------------------------------------------------------------------
    print(f"finished after {iteration} iterations, best score = {best_score:.6f}")
    return best_seq.tolist()
