    # Ensure v >=0 and sum = 1.
    # If v may be all zeros, return uniform.
    n = len(v)

    """Generate a nonnegative sequence whose self‑convolution has a small maximal value.

    The routine uses a smooth approximation of the max‑operator (softmax) and
    performs projected gradient descent on the simplex.  By slowly increasing
    the softmax temperature the algorithm first flattens the whole convolution
    and then focuses on the current peak.  Several random restarts are tried
    within the allotted time; the best sequence seen is returned.
    """
    import numpy as np, time
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5

    # Choose a length – use the supplied reference if present.
    try:
        n = len(height_sequence_1)
    except NameError:
        n = 1500   # fallback
    best_seq = None
    best_score = float('inf')

    # Helper: project onto the probability simplex (sum = 1, non‑negative)
    def simplex_proj(v):
        """Euclidean projection of v onto the simplex {x>=0, sum(x)=1}."""
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - 1
        ind = np.arange(len(v)) + 1
        cond = u - cssv / ind > 0
        if not np.any(cond):
            # all zeros – return uniform
            return np.full_like(v, 1.0 / len(v))
        rho = ind[cond].max()
        theta = cssv[cond].max() / rho
        w = np.maximum(v - theta, 0)
        return w

    # Main optimisation loop – we do several restarts.
    restart = 0
    while time.time() < deadline:
        restart += 1
        # initialise: uniform or small random perturbation
        a = np.full(n, 1.0 / n)
        # add a small random perturbation on a few coordinates to diversify starts
        if restart % 5 == 0:
            idx = np.random.choice(n, size=min(10, n), replace=False)
            a[idx] += np.random.rand(len(idx)) * 0.1
            a = simplex_proj(a)

        # hyper‑parameters for this restart
        beta = 0.5          # temperature of soft‑max (low → smooth)
        step = 0.2          # initial step size
        it = 0

        while time.time() < deadline:
            # convolution
            b = np.convolve(a, a)
            # current score
            cur_score = 2.0 * n * b.max() / (np.sum(a) ** 2)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time():.2f}] restart {restart}, iter {it}, score {cur_score:.6f}, beta {beta:.3f}")

            # stop early if we already beat the target by a comfortable margin
            if best_score < 1.50:
                # good enough – break out early
                break

            # soft‑max weights approximating the max
            # subtract the maximum to avoid overflow
            shifted = b - b.max()
            w = np.exp(beta * shifted)
            w /= w.sum()

            # gradient of the smoothed loss L = log‑sum‑exp(beta * b)
            # ∂L/∂a_i = 2 * Σ_k w_k a_{k‑i}  (indices outside [0,n‑1] are ignored)

"""
Simple projected‑gradient algorithm for minimizing the maximal value of the
self‑convolution of a non‑negative step‑function.

The objective 2·n·max(conv(a,a)) / (sum(a)^2) is invariant under scaling, so we
keep the coefficients on the probability simplex (sum = 1, a_i ≥ 0).  We use a
smooth approximation of the max‑operator (soft‑max) and perform gradient steps
with a Euclidean projection onto the simplex.  The soft‑max temperature β is
started low (smooth loss) and slowly increased, allowing the algorithm to
first flatten the whole convolution and later focus on the current peak.
Several random restarts are performed while time permits and the best sequence
encountered is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the simplex {x≥0, Σx = 1}
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector v onto the probability simplex."""
    if np.all(v == 0):
        return np.full_like(v, 1.0 / v.size)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(v.size) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # Numerical safety – fall back to uniform distribution
        return np.full_like(v, 1.0 / v.size)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w
