def propose_candidate(...):

    start = time.time()
    deadline = start + budget_s - 0.5

    best_seq = None
    best_score = float('inf')
    # baseline uniform
    for n in [200,400,600,800,1000,1200,1500,1800,2000]:
        seq = [1.0] * n  # uniform
        score = evaluate_sequence(seq)
        if score < best_score:
            best_score = score
            best_seq = seq

    # grid search over geometric decreasing from start
    for n in [200,400,600,800,1000,1200,1500,1800,2000]:
        for s in np.linspace(0.5, 3.0, 26):  # 0.5 to 3 step ~0.1
            if time.time() > deadline:
                break
            r = 1 - s / n
            if r <= 0:
                continue
            seq = [r**i for i in range(n)]
            score = evaluate_sequence(seq)
            if score < best_score:
                best_score = score
                best_seq = seq

    # symmetric exponential
    for n in [200,400,600,800,1000,1200,1500,1800,2000]:
        for s in np.linspace(0.5, 3.0, 26):
            if time.time() > deadline:
                break
            r = 1 - s / n
            if r <= 0:
                continue
            # symmetric
            seq = [r**min(i, n-1-i) for i in range(n)]
            score = evaluate_sequence(seq)
            if score < best_score:
                best_score = score
                best_seq = seq

    # local refinement: use LP to improve best_seq
    curr_seq = best_seq[:]
    # compute M = max(convolve(curr_seq,curr_seq))
    while time.time() < deadline:
        # solve LP with current seq to find g that maximizes sum
        n = len(curr_seq)
        sum_seq = np.sum(curr_seq)
        if sum_seq == 0:
            break
        # normalization not needed because step mixing
        # But we need M
        max_b = max(np.convolve(curr_seq, curr_seq))
        g = solve_convolution_lp(curr_seq, max_b)
        if g is None:
            break
        # mix
        t = 0.05
        # ensure new seq is mixture, maybe scale to maintain sum? Not needed.
        new_seq = [(1 - t) * a + t * b for a, b in zip(curr_seq, g)]
        curr_score = evaluate_sequence(new_seq)
        if curr_score < best_score:
            best_score = curr_score
            best_seq = new_seq[:]
        # update curr_seq for next iteration
        curr_seq = new_seq

    print(f'Best score {best_score} with length {len(best_seq)}')

def propose_candidate(budget_s=1000, **kwargs):

        return ((n - np.arange(n)) / n).tolist()  # values from 1 to 0

        return (1.0 / (np.arange(n) + 1)).tolist()

            seq = geometric_seq(n, s)

            score = evaluate_sequence(seq)

evaluate_sequence.  The approach is a three‑stage search:

1.   Quick grid search over simple analytic families (geometric,
     linear, harmonic) for a few different lengths.
2.   Keep the best few candidates.
3.   Refine the best candidate by an iterative LP‑based improvement
     (the same idea used by the reference method, but with a larger
     step size and many more iterations) while respecting the time
     budget.

The final sequence is returned as a plain Python list of floats.
"""

import time
import numpy as np
from scipy import optimize

# ----------------------------------------------------------------------
# Helper: solve the LP  max sum(b)  s.t.  conv(f, b) <= M  and  b >= 0
# ----------------------------------------------------------------------
def solve_convolution_lp(f_seq, M):
    """Linear programme: maximise sum(b) under convolution bound."""
    n = len(f_seq)
    # objective: maximise sum(b)  <=> minimise -sum(b)
    c = -np.ones(n)

    # Build the convolution constraints A_ub @ b <= M
    rows = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        # conv(f,b)[k] = sum_i f[i] * b[k-i]
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                row[j] = f_seq[i]
        rows.append(row)
    A_ub = np.vstack([rows, -np.eye(n)])          # also b >= 0  -> -b <= 0
    b_ub = np.hstack([np.full(2 * n - 1, M), np.zeros(n)])

    # linprog with the HiGHS solver (default) – fast for moderate n
    res = optimize.linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs')
    if res.success:
        return res.x
    return None


# ----------------------------------------------------------------------
# Simple families of candidate sequences
# ----------------------------------------------------------------------
def geometric_seq(n, s):
    """a_i = (1 - s/n)^i   (decreasing exponential)."""
    r = 1.0 - s / n
    if r <= 0.0:
        return None
    return (r ** np.arange(n)).tolist()


def linear_seq(n):
    """a_i = (n-i)/n   (linear decay from 1 to 0)."""
    return ((n - np.arange(n)) / n).tolist()


def harmonic_seq(n):
    """a_i = 1/(i+1)   (slow decay)."""
    return (1.0 / (np.arange(n) + 1)).tolist()


# ----------------------------------------------------------------------
# Score using the same formula as evaluate_sequence (no extra caps)
# ----------------------------------------------------------------------
def score_of(seq):
    n = len(seq)
    if n == 0:
        return float('inf')
    conv = np.convolve(seq, seq)
    max_b = np.max(conv)
    sum_a = np.sum(seq)
    if sum_a < 0.01:
        return float('inf')
    return float(2.0 * n * max_b / (sum_a ** 2))


# ----------------------------------------------------------------------
# LP‑based refinement of a single sequence
# ----------------------------------------------------------------------
def refine_by_lp(seq, deadline):
    """Iteratively increase the sum while keeping the max convolution bounded."""
    a = np.array(seq, dtype=float)
    n = len(a)
    best_seq = a.copy()
    best_score = score_of(best_seq.tolist())
    iteration = 0

    while time.time() < deadline:
        # current bound M = max conv(a,a)
        M = np.max(np.convolve(a, a))

        # solve LP to get a direction with larger sum
        g = solve_convolution_lp(a.tolist(), M)
        if g is None:
            break

        # a small convex combination moves us towards g
        # start with a relatively aggressive step, slowly shrink it
        t = max(0.02, 0.1 / (iteration + 1))
        new_a = (1.0 - t) * a + t * g

        # evaluate the new candidate
        cur_score = score_of(new_a.tolist())
        if cur_score < best_score:
            best_score = cur_score
            best_seq = new_a.copy()
            print(f"[LP] iteration {iteration:3d}  new best {best_score:.6f}")
        a = new_a
        iteration += 1

    return best_seq.tolist(), best_score


# ----------------------------------------------------------------------
# Main entry point required by the platform
# ----------------------------------------------------------------------
def propose_candidate(budget_s=1000, **kwargs):
    """
    Search for a good step‑function sequence.
    The function returns a list of non‑negative floats.
    """
    # ------------------------------------------------------------------
    # 1. quick grid search over a few families and lengths
    # ------------------------------------------------------------------
    start_time = time.time()
    end_time = start_time + budget_s - 5.0          # keep a small safety margin
    candidate_seqs = []

    # a few lengths – larger lengths give a better theoretical bound,
    # but also cost more for the convolution.
    lengths = [800, 1200, 1500, 1800]

    for n in lengths:
        # uniform (baseline)
        uniform = [1.0] * n
        candidate_seqs.append(uniform)

        # linear decay
        candidate_seqs.append(linear_seq(n))

        # harmonic decay (only for moderate n)
        candidate_seqs.append(harmonic_seq(n))

        # geometric decay – sweep the shape parameter s
        for s in np.linspace(0.5, 2.5, 12):
            if time.time() > end_time:
                break
            seq = geometric_seq(n, s)
            if seq is not None:
                candidate_seqs.append(seq)

    # evaluate all candidates (fast, O(n²) each)
    best_seq = None
    best_score = float('inf')
    for seq in candidate_seqs:
        if time.time() > end_time:
            break
        sc = score_of(seq)
        if sc < best_score:
            best_score = sc
            best_seq = seq
            print(f"[grid] new best {best_score:.6f} (len={len(seq)})")

    # ------------------------------------------------------------------
    # 2. LP‑based refinement of the best candidate
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – a short uniform sequence
        best_seq = [1.0] * 500
        best_score = score_of(best_seq)

    # allocate the remaining time for refinement
    refine_deadline = time.time() + max(0.0, end_time - time.time())
    refined_seq, refined_score = refine_by_lp(best_seq, refine_deadline)
    if refined_score < best_score:
        best_seq = refined_seq
        best_score = refined_score

    # ------------------------------------------------------------------
    # 3. final sanity check (evaluate_sequence will also be called by
    #    the outer framework, but we print the score for logging)
    # ------------------------------------------------------------------
    final_score = score_of(best_seq)
    print(f"[final] best score {final_score:.6f} with length {len(best_seq)}")
    # ensure the returned value is a plain Python list
    return list(best_seq)
