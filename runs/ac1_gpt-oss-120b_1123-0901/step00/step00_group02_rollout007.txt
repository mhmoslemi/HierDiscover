"""
Projected gradient descent on the probability simplex.

We work with a probability vector w (sum = 1, w_i ≥ 0).  For a sequence of
length N we set the heights a_i = w_i * sqrt(2*N).  The evaluation function
for a normalized sequence reduces to

    score = max_k (a * a)_k = 2*N * max_k (w * w)_k .

Thus we minimise the maximum entry of the self‑convolution of w.
The optimisation uses a smooth approximation of the max via a soft‑max
with a temperature parameter α.  The gradient of
   L(w) = Σ_k p_k (w*w)_k ,   p_k ∝ exp(α·(w*w)_k)

is  ∇L = 2·conv_rev(p, w_rev)  (restricted to the first N components).
We perform a projected gradient step (additive update) and project the
result back onto the probability simplex.  The step‑size and the soft‑max
sharpness α are scheduled during the run.  Random perturbations and
re‑initialisations are applied occasionally to avoid stagnation.

The routine runs for at most `budget_s` seconds and returns the best
sequence found (as a plain Python list of floats).  All heights are
non‑negative and well below the 1000‑limit of ``evaluate_sequence``.
"""

import time
import numpy as np


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _linear_convolve_fft(a: np.ndarray, b: np.ndarray, L: int) -> np.ndarray:
    """
    Linear convolution of real vectors a and b using FFT of length L.
    The result has length len(a) + len(b) - 1 (truncated from the FFT output).
    """
    A = np.fft.rfft(a, L)
    B = np.fft.rfft(b, L)
    conv = np.fft.irfft(A * B, L)
    return conv[: len(a) + len(b) - 1]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex {x >= 0, Σx = 1}.
    Implements the O(n log n) algorithm of Duchi et al.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

        # ---------- gradient of the soft‑max objective ----------------
        w_rev = w[::-1]
        conv_rev = _linear_convolve_fft(w_rev, p, L_fft)[: 3 * N - 2]
        grad = 2.0 * conv_rev[N - 1 : N - 1 + N]         # length N

        # ---------- projected gradient step ---------------------------
        w_new = w - step * grad
        w_new = _project_simplex(w_new)

        # ---------- occasional random perturbation --------------------
        if iter_cnt % 500 == 0:
            perturb = 0.02 * np.random.randn(N) * w_new.mean()
            w_new = np.maximum(w_new + perturb, 0.0)
            w_new = _project_simplex(w_new)

        # ---------- restart if stuck for long ------------------------
        if iter_cnt - last_improve_iter > 5000:
            # re‑initialise with a fresh random simplex point
            w_new = np.random.rand(N)
            w_new = _project_simplex(w_new)
            last_improve_iter = iter_cnt
            print(f"Iter {iter_cnt}: restart after stagnation.")

        w = w_new

        # ---------- schedule parameters --------------------------------
        step *= step_decay
        if iter_cnt % 200 == 0:
            alpha = min(alpha_max, alpha + alpha_inc)

    # ------------------------------------------------------------------
    # Convert the best probability vector back to heights and return.
    # ------------------------------------------------------------------
    best_heights = (best_w * S).tolist()
    print(f"Finished after {iter_cnt} iterations. Best score = {best_score:.6f}")
    return best_heights
