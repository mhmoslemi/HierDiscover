    Simulated annealing / hill climbing search for a non‑negative step
    function that minimizes 2·n·max(conv(a,a))/(sum(a))².
    The algorithm starts from either the known SOTA sequence or a random
    sequence, then repeatedly moves mass from entries that contribute most
    to the current peak of the autocorrelation toward entries that contribute
    little.  Moves are accepted greedily or with a temperature‑dependent
    probability to escape local minima.  Small random perturbations and occasional
    restarts are performed.  At the end, near‑zero entries are pruned.
    """
    import numpy as np
    import time

    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # keep safety margin

    # Initialize sequence
    if 'height_sequence_1' in globals() and np.random.rand() < 0.6:

        init_seq = np.random.rand(n0).tolist()
        print(f"Starting from random sequence, length {n0}")

    # prune tiny entries
    init_seq = [x for x in init_seq if x > 1e-9]

    # cap length to reasonable size (avoid huge LP later)
    max_len = 1500
    if len(init_seq) > max_len:
        # simple truncation (keeps head of sequence)
        init_seq = init_seq[:max_len]
        print(f"Truncated start sequence to length {max_len}")

    # ensure sum is not too small (evaluate_sequence would return inf otherwise)
    if sum(init_seq) < 0.1:
        scale = 0.1 / (sum(init_seq) + 1e-12)
        init_seq = [x * scale for x in init_seq]

    # evaluate initial sequence
    try:
        best_score = evaluate_sequence(init_seq)
    except Exception:
        best_score = float('inf')
    if not np.isfinite(best_score):
        # fallback to uniform non‑zero sequence
        n = max(2, len(init_seq))
        init_seq = [1.0] * n
        best_score = evaluate_sequence(init_seq)
        print("Fallback to uniform sequence due to invalid start.")

    best_seq = init_seq[:]
    current_seq = init_seq[:]
    current_score = best_score

    print(f"Initial score: {best_score:.6f}")

    iteration = 0
    # Main optimisation loop
    while time.time() < deadline:

        move_type = np.random.rand()

            candidate = move_reduce_peak(current_seq, eps)

                # protect from overflow
                prob = np.exp(delta / max(temperature, 1e-12))
                if np.random.rand() < prob:
                    accept = True
            if accept:
                current_seq = candidate
                current_score = cand_score
                if cand_score < best_score:
                    best_score = cand_score
                    best_seq = candidate[:]
                    print(f"Iter {iteration:6d} – new best: {best_score:.6f} (temp {temperature:.5f})")

"""
Simulated‑annealing / hill‑climbing search for a non‑negative step function
that minimises

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

where ``conv`` is the ordinary discrete convolution.  The algorithm works as
follows:

1. **Initialisation** – start from the known SOTA sequence
   (``height_sequence_1``) with probability 0.6, otherwise from a random
   sequence of moderate length (≈ 500‑1500 elements).  Tiny entries are
   discarded and the length is capped to keep each iteration fast.

2. **Mass‑transfer moves** – each iteration either
   * moves mass from indices that contribute most to the current peak of the
     autocorrelation (the convolution at the index where it is maximal) to
     indices that contribute nothing, or
   * performs a simple random transfer of mass between two indices.
   The step size `eps` is tied to a temperature schedule that linearly
   cools from ~0.1 to 0.001 over the allotted time.

3. **Acceptance** – a move is always accepted if it improves the score.
   Otherwise it is accepted with Metropolis probability
   ``exp((old – new) / temperature)`` to allow occasional uphill steps.

4. **Exploration tricks** – tiny Gaussian jitter is added with low probability
   and every 5000 iterations a small random perturbation of the current best
   solution is tried (a “restart”).  After the search finishes, entries that
   are effectively zero are pruned.

The routine respects all constraints of ``evaluate_sequence`` (non‑negative,
finite entries).  It returns the best sequence found before the time budget
expires.
"""

import time
import numpy as np

def move_random_transfer(seq, eps):
    """
    Transfer a small fraction `eps` of mass from a randomly chosen index
    (biased by its current weight) to another random index.
    """
    n = len(seq)
    if n == 0:
        return seq[:]

    a = np.array(seq, dtype=float)
    total = a.sum()
    if total == 0:
        i = np.random.randint(n)
    else:
        probs = a / total
        i = np.random.choice(n, p=probs)

    # choose a destination index different from i
    j = np.random.randint(n - 1)
    if j >= i:
        j += 1

    delta = eps * a[i]
    delta = min(delta, a[i])  # keep non‑negative
    new_seq = seq[:]
    new_seq[i] = max(0.0, new_seq[i] - delta)
    new_seq[j] = new_seq[j] + delta
    return new_seq


def move_reduce_peak(seq, eps):
    """
    Identify the current peak of the autocorrelation (conv(seq, seq)),
    then move a small fraction of mass from an index that contributes most
    to that peak towards an index that does not contribute.
    """
    n = len(seq)
    if n == 0:
        return seq[:]

    a = np.array(seq, dtype=float)

    # convolution to locate the maximal entry
    conv = np.convolve(a, a)
    k_max = int(np.argmax(conv))

    # contribution of each position to the peak at k_max
    contrib = np.zeros(n, dtype=float)
    for i in range(n):
        j = k_max - i
        if 0 <= j < n:
            contrib[i] = a[i] * a[j]

    total_contrib = contrib.sum()
    if total_contrib == 0.0:
        # fallback to a generic random transfer
        return move_random_transfer(seq, eps)

    # source index: weighted by contribution size
    probs_src = contrib / total_contrib
    i = np.random.choice(n, p=probs_src)

    # destination index: ideally one that does NOT affect the current peak
    zero_inds = np.where(contrib == 0.0)[0]
    if zero_inds.size > 0:
        j = np.random.choice(zero_inds)
    else:
        # if every index contributes, pick the one with the smallest contribution
        min_val = contrib.min()
        cand = np.where(contrib == min_val)[0]
        j = np.random.choice(cand)

    delta = eps * a[i]
    delta = min(delta, a[i])
    new_seq = seq[:]
    new_seq[i] = max(0.0, new_seq[i] - delta)
    new_seq[j] = new_seq[j] + delta
    return new_seq


def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Search for a sequence of non‑negative heights that minimises the
    evaluation function.  Returns the best sequence discovered before the
    time budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # 1. Initialise the candidate sequence
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals() and np.random.rand() < 0.6:
        init_seq = list(height_sequence_1)
        print(f"[Init] Using height_sequence_1 (len={len(init_seq)})")
    else:
        n0 = np.random.randint(500, 1500)
        init_seq = np.random.rand(n0).tolist()
        print(f"[Init] Random start (len={n0})")

    # discard negligible entries
    init_seq = [x for x in init_seq if x > 1e-9]

    # keep length reasonable for fast iterations
    MAX_LEN = 1500
    if len(init_seq) > MAX_LEN:
        init_seq = init_seq[:MAX_LEN]
        print(f"[Init] Truncated to max length {MAX_LEN}")

    # guarantee a non‑trivial sum (evaluate_sequence returns ∞ for tiny sums)
    if sum(init_seq) < 0.1:
        scaling = 0.1 / (sum(init_seq) + 1e-12)
        init_seq = [x * scaling for x in init_seq]

    # initial evaluation
    try:
        best_score = evaluate_sequence(init_seq)
    except Exception:
        best_score = float('inf')
    if not np.isfinite(best_score):
        # fallback to a simple uniform sequence
        n = max(2, len(init_seq))
        init_seq = [1.0] * n
        best_score = evaluate_sequence(init_seq)
        print("[Init] Fallback to uniform sequence due to invalid start.")

    best_seq = init_seq[:]
    current_seq = init_seq[:]
    current_score = best_score

    print(f"[Init] Initial score = {best_score:.6f}")

    iteration = 0

    # ------------------------------------------------------------------
    # 2. Main optimisation loop (simulated annealing / hill climbing)
    # ------------------------------------------------------------------
    while time.time() < deadline:
        iteration += 1

        # --- temperature schedule (linear cooling) ---
        elapsed = time.time() - start_time
        fraction = elapsed / (budget_s - 0.5)
        temperature = max(0.1 * (1.0 - fraction), 0.001)  # from ~0.1 down to 0.001
        eps = temperature  # step size tied to temperature

        # --- decide which move to apply ---
        if np.random.rand() < 0.75:
            candidate = move_reduce_peak(current_seq, eps)
        else:
            candidate = move_random_transfer(current_seq, eps)

        # occasional tiny Gaussian jitter on a random index
        if np.random.rand() < 0.02:
            idx = np.random.randint(len(candidate))
            candidate[idx] = max(0.0, candidate[idx] + np.random.randn() * eps * 0.1)

        # --- evaluate the new candidate ---
        try:
            cand_score = evaluate_sequence(candidate)
        except Exception:
            cand_score = float('inf')

        # --- acceptance rule (Metropolis) ---
        if np.isfinite(cand_score):
            accept = False
            if cand_score < current_score:
                accept = True
            else:
                delta = current_score - cand_score
                # probability = exp(delta / T)
                prob = np.exp(delta / max(temperature, 1e-12))
                if np.random.rand() < prob:
                    accept = True

            if accept:
                current_seq = candidate
                current_score = cand_score
                if cand_score < best_score:
                    best_score = cand_score
                    best_seq = candidate[:]
                    print(f"[Iter {iteration:6d}] New best = {best_score:.6f} (temp={temperature:.5f})")

        # --- occasional restart with a jittered copy of the best solution ---
        if iteration % 5000 == 0:
            if best_seq:
                jitter_scale = 0.01 * max(best_seq)
                perturbed = [
                    max(0.0, x + np.random.randn() * jitter_scale)
                    for x in best_seq
                ]
                # prune near‑zero entries and enforce length cap
                perturbed = [x for x in perturbed if x > 1e-9]
                if len(perturbed) > MAX_LEN:
                    perturbed = perturbed[:MAX_LEN]
                if len(perturbed) < 2:
                    perturbed = [0.5, 0.5]

                try:
                    pert_score = evaluate_sequence(perturbed)
                except Exception:
                    pert_score = float('inf')

                if pert_score < best_score:
                    best_score = pert_score
                    best_seq = perturbed[:]
                    current_seq = perturbed[:]
                    current_score = pert_score
                    print(f"[Iter {iteration:6d}] Restart improved best to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Post‑processing: prune negligible entries
    # ------------------------------------------------------------------
    final_seq = [x for x in best_seq if x > 1e-8]
    if len(final_seq) < 2:
        final_seq = [0.5, 0.5]  # guarantee a feasible non‑empty sequence

    print(f"[Done] Iterations: {iteration}, best score = {best_score:.6f}, final length = {len(final_seq)}")
    return final_seq
