def simplex_projection(v, target):
    # ensure non-negative
    v = np.maximum(v, 0)
    if v.sum() == target:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a candidate step‑function sequence.

The algorithm proceeds in two stages:

1. **Parametric random search** – generate many candidate sequences from
   simple parametric families (exponential, power‑law, linear decay,
   alternating two levels, Dirichlet‑random).  For each length n a target
   sum `sqrt(2*n)` is enforced by projecting onto the simplex, and the
   candidate is evaluated with ``evaluate_sequence``.  The best candidate
   found is kept.

2. **Local simulated‑annealing refinement** – starting from the best
   candidate, repeatedly move a random amount of mass from one index to
   another (preserving the total sum).  Moves are accepted if they improve
   the score or with a temperature‑dependent probability to escape local
   minima.  The temperature decays linearly to zero as the time budget
   expires.

Both stages respect the non‑negative constraint and stay well below the
clipping bound (1000).  The time budget (default 1000 s) is respected by
checking the deadline after each iteration.  The returned sequence is the
best one found before termination.
"""

import time
import math
import random
import numpy as np

# Helper functions

def _simplex_projection(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x >= 0, sum(x)=target_sum}."""
    if target_sum <= 0:
        return np.zeros_like(v)
    # ensure non‑negative
    v = np.maximum(v, 0.0)
    s = v.sum()
    if abs(s - target_sum) < 1e-12:
        return v
    # if all zero we just spread equally
    if s == 0:
        return np.full_like(v, target_sum / len(v))

    # standard algorithm (see Wang & Carreira‑Perpiñán 2013)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        rho = 0
    else:
        rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho if rho > 0 else 0.0
    w = np.maximum(v - theta, 0.0)
    return w

def _gen_exponential(n: int) -> np.ndarray:
    """a_i = exp(-beta * i)"""
    beta = np.random.uniform(0.0, 5.0)          # rate
    i = np.arange(n, dtype=np.float64)
    return np.exp(-beta * i)


def _gen_powerlaw(n: int) -> np.ndarray:
    """a_i = (i+1)^(-alpha)"""
    alpha = np.random.uniform(0.0, 5.0)
    i = np.arange(1, n + 1, dtype=np.float64)
    return 1.0 / (i ** alpha)


def _gen_linear_decay(n: int) -> np.ndarray:
    """a_i = max(0, a0 - slope * i)"""
    # Choose slope so that the line stays non‑negative for most of the range
    slope = np.random.uniform(0.0, 1.0)
    # Choose a0 large enough to keep at least a few positive entries
    a0 = slope * (n - 1) + np.random.uniform(0.1, 2.0)
    i = np.arange(n, dtype=np.float64)
    a = a0 - slope * i
    return np.maximum(a, 0.0)


def _gen_alternating_two_levels(n: int) -> np.ndarray:
    """high on even positions, low on odd positions (or vice‑versa)."""
    high = np.random.rand() + 0.5          # between 0.5 and 1.5

    a[1::2] = low
    # possibly flip parity with 50% chance
    if np.random.rand() < 0.5:
        a = a[::-1]
    return a

            if sc0 < best_score:
                best_score = sc0
                best_seq = seq0.copy()
                print(f"[init] height_sequence_1 score = {sc0:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Stage 1 – random parametric search
    # ------------------------------------------------------------------
    # we will spend roughly the first 30% of the time on this stage
    stage1_deadline = time.time() + 0.3 * (deadline - time.time())

    while time.time() < stage1_deadline:
        # pick a length; longer lengths give finer granularity but cost more
        n = np.random.randint(200, 1500)        # 200‑1499 elements
        target_sum = math.sqrt(2.0 * n)

        # pick a generator
        gen = random.choice(_GENERATORS)
        raw = gen(n)

        # project onto the simplex to enforce the required sum
        a = _simplex_projection(raw, target_sum)

        # evaluate
        score = _eval(a)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[random] new best {best_score:.6f} (len={n})")

    # ------------------------------------------------------------------
    # Stage 2 – local simulated‑annealing refinement
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback: generate a simple uniform vector
        n = 500
        best_seq = np.full(n, math.sqrt(2.0 * n) / n)
        best_score = _eval(best_seq)

    current_seq = best_seq.copy()
    current_score = best_score

    # some parameters for the annealing schedule
    initial_temp = 0.5
    final_temp = 1e-4

    # Pre‑compute the target sum – it stays constant for a given length
    n = len(current_seq)
    target_sum = math.sqrt(2.0 * n)

    iteration = 0
    while time.time() < deadline:
        # linear temperature decay based on remaining time
        remaining = max(deadline - time.time(), 1e-6)
        fraction = remaining / (deadline - (deadline - budget_s + 5.0))
        temperature = final_temp + (initial_temp - final_temp) * fraction

        # ---- propose a mass‑transfer move -------------------------------------------------
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue

        # transfer up to 5 % of the source mass (or less if source is tiny)
        max_transfer = min(current_seq[i], 0.05 * target_sum)
        if max_transfer <= 0:
            continue
        delta = np.random.rand() * max_transfer

        cand = current_seq.copy()
        cand[i] -= delta
        cand[j] += delta   # sum stays exactly the same

        # (no need to re‑project: the move preserves non‑negativity and the sum)

        # ---- evaluate ----
        cand_score = _eval(cand)

        # keep the globally best
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            print(f"[anneal] new global best {best_score:.6f} (iter {iteration})")

        # accept/reject for the current state
        delta_score = cand_score - current_score
        if delta_score < 0 or random.random() < math.exp(-delta_score / temperature):
            current_seq = cand
            current_score = cand_score

        # occasionally perform a tiny random perturbation to escape plateaus
        if iteration % 2000 == 0 and iteration > 0:
            perturb = np.random.randn(n) * 0.001 * target_sum
            perturbed = current_seq + perturb
            perturbed = np.maximum(perturbed, 0.0)
            # re‑project to keep the sum exactly target_sum
            perturbed = _simplex_projection(perturbed, target_sum)
            pert_score = _eval(perturbed)
            if pert_score < best_score:
                best_score = pert_score
                best_seq = perturbed.copy()
                print(f"[perturb] new best {best_score:.6f} (iter {iteration})")
            if pert_score < current_score:
                current_seq = perturbed
                current_score = pert_score

        iteration += 1

    # ------------------------------------------------------------------
    # Return the best sequence found (as plain Python list)
    # ------------------------------------------------------------------
    if best_seq is None:
        # extremely unlikely, but return a safe uniform sequence
        n = 500
        best_seq = np.full(n, math.sqrt(2.0 * n) / n)

    print(f"[final] best score = {best_score:.6f} with length {len(best_seq)}")
    return best_seq.tolist()
