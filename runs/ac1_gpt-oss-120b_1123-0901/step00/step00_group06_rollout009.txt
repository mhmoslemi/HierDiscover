def project_onto_simplex(v, s):
    # Standard algorithm: sort v descending, compute cumulative sum, find rho.
    if s < 0: raise error.
    u = np.sort(v)[::-1] # descending
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, len(v)+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def project_onto_simplex(v, s):
    """Project vector v onto the simplex {x >=0, sum x = s}."""
    n = len(v)
    # v may have negative entries
    # Sorted descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    # find rho: last index where u_i > cssv_i / (i+1)
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if len(rho) == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a sequence of non‑negative heights that drives the evaluation function

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

as low as possible.  The algorithm is a simple projected gradient descent on the
index that currently attains the maximal convolution value.

*   The sequence is always projected onto the simplex
        { x_i ≥ 0 , Σ x_i = √(2 n) }
    which makes the objective scale‑invariant.
*   At each iteration the gradient of the maximal convolution entry is
    computed analytically (∂/∂a_i max_conv = 2 * a_{k‑i} when the index k is
    inside the overlap region).  The current vector is moved a small step in
    the opposite direction and projected back onto the simplex.
*   A few random restarts with different lengths (200 … 1500) are performed
    within the time budget.  Occasionally a tiny Gaussian perturbation is
    added to avoid getting stuck in a local minimum.
*   The best sequence seen so far is returned.

The method is completely self‑contained and relies only on NumPy, time and
the already imported ``evaluate_sequence`` function.
"""

import time
import numpy as np

def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        { x | x_i >= 0, sum(x_i) = s }.
    Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    n = v.shape[0]
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) yields a low
    evaluation score.  The routine runs for at most ``budget_s`` seconds and
    keeps the best sequence discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    best_seq = None
    best_score = float("inf")

    # -----------------------------------------------------------------
    # Helper: one optimisation run on a given length ``n`` and an
    # initial vector ``init``.
    # -----------------------------------------------------------------
    def _run_opt(n, init):
        nonlocal best_seq, best_score

        target_sum = np.sqrt(2.0 * n)          # the normalisation used by the LP method
        a = init.astype(np.float64)

        # ensure the initial point is on the simplex
        a = _project_onto_simplex(a, target_sum)

        # step size (will be adapted a bit during the run)
        base_step = 0.05

        # allocate a slice of the remaining time for this run
        run_deadline = min(deadline, time.time() + 12.0)

        while time.time() < run_deadline:
            # current convolution and the index of its maximum
            conv = np.convolve(a, a)
            k_max = int(np.argmax(conv))
            max_val = conv[k_max]

            # compute the gradient of the *selected* max entry
            # only indices that actually appear in the sum have a non‑zero gradient
            low = max(0, k_max - (n - 1))
            high = min(n - 1, k_max)
            idx = np.arange(low, high + 1, dtype=int)
            partner = k_max - idx                     # partner indices inside the overlap
            grad = np.zeros_like(a)
            grad[idx] = 2.0 * a[partner]

            # adapt step size so that we do not move too far (avoid overshoot)
            max_grad = np.max(grad) if np.max(grad) > 0 else 1.0
            eta = base_step * max_val / max_grad

            # take a step opposite to the gradient
            a_new = a - eta * grad

            # keep the vector feasible (non‑negative, correct sum)
            a_new = _project_onto_simplex(a_new, target_sum)

            # evaluate the candidate
            try:
                score = evaluate_sequence(a_new.tolist())
            except Exception:                     # safety – treat as infinite
                score = float("inf")

            # keep the global best
            if score < best_score:
                best_score = score
                best_seq = a_new.copy()
                print(f"[{time.time():.2f}] New best = {best_score:.6f} (n={n})")

            # simple acceptance rule – always continue from the new point
            # occasional random shake‑up to escape flat regions
            if np.random.rand() < 0.02:
                noise = np.random.randn(n) * (0.01 * target_sum / n)
                a = _project_onto_simplex(np.maximum(a_new + noise, 0.0), target_sum)
                # reduce step a little after a shake‑up
                base_step *= 0.9
            else:
                a = a_new

    # -----------------------------------------------------------------
    # Main loop – several independent restarts
    # -----------------------------------------------------------------
    while time.time() < deadline:
        # pick a problem size (short runs for very large n to keep each iteration cheap)
        n = np.random.randint(200, 1500)

        # initialise either from the provided SOTA sequence or from a random point
        if np.random.rand() < 0.3 and 'height_sequence_1' in globals():
            # use the existing construction (trim / pad if necessary)
            src = np.array(height_sequence_1, dtype=float)
            if src.size >= n:
                init = src[:n].copy()
            else:
                # pad with small random noise to reach length n
                pad = np.random.rand(n - src.size) * 1e-3
                init = np.concatenate([src, pad])
        else:
            init = np.random.rand(n)

        # run optimisation on this restart
        _run_opt(n, init)

        # if we are already very close to the target we can stop early
        if best_score < 1.55:
            break

    # -----------------------------------------------------------------
    # Fallback – if something went wrong return a safe uniform sequence
    # -----------------------------------------------------------------
    if best_seq is None:
        n = 1000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n, dtype=float)

    return best_seq.tolist()
