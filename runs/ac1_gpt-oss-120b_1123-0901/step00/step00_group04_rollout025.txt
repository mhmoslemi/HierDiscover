def fast_max_convolution(a):
    n = len(a)
    n2 = 1 << ((2*n-1).bit_length())
    A = np.fft.rfft(a, n2)
    conv = np.fft.irfft(A * A, n2)
    # conv is length n2, but only first 2*n-1 entries are valid (others due to zero padding).
    # take the first 2*n -1 . max.
    return np.max(conv[:2*n-1])

def max_conv(seq):
    n = len(seq)
    n2 = 1 << ((2*n - 1).bit_length())
    fft_seq = np.fft.rfft(seq, n2)
    conv = np.fft.irfft(fft_seq * fft_seq, n2)
    # may have floating errors: ensure real; get first 2*n-1 elements
    conv = conv[:2*n-1]
    return np.max(conv)

Now fast_score(seq):
    s = np.sum(seq)
    if s < 1e-12: return np.inf
    maxb = max_conv(seq)
    return 2.0 * len(seq) * maxb / (s*s)

def propose_candidate(seed=0, budget_s=1000, **kwargs):
    import numpy as np, time, math
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2

    best_seq = None
    best_score = float('inf')

    # families
    def try_shape(seq):
        nonlocal best_seq, best_score
        score = fast_score(seq)
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"Found new best {best_score:.6f} length {len(seq)}")

    lengths = [200, 500, 1000, 2000, 4000, 6000]

    # try exponential shapes
    for n in lengths:
        # compute exponent parameter a in logspace: we search a in [0.1, 5] by step 0.1
        for a in np.linspace(0.1, 5.0, 50):
            r = math.exp(-a / n)
            seq = np.array([r**i for i in range(n)], dtype=float)
            try_shape(seq)

    # power law shapes
    for n in lengths:
        for p in np.linspace(0.0, 2.0, 41):
            seq = np.array([ (i+1) ** (-p) for i in range(n)], dtype=float)
            try_shape(seq)

    # arcsine-like shape
    for n in lengths:
        idx = np.arange(n)
        seq = 1.0 / np.sqrt((idx+1)*(n-idx))
        try_shape(seq)

    # random initial sequences
    for n in [500, 1000, 2000, 4000]:
        for _ in range(3):
            seq = np.random.rand(n)
            try_shape(seq)

    # Now use simulated annealing / local optimization starting from best_seq
    if best_seq is None:
        # fallback: uniform
        best_seq = np.ones(1000)
        best_score = fast_score(best_seq)

    curr_seq = best_seq.copy()
    curr_score = fast_score(curr_seq)
    iter_num = 0
    # temperature schedule
    T0 = 0.2
    while time.time() < deadline:
        # propose neighbor
        n = len(curr_seq)
        i = np.random.randint(n)
        j = np.random.randint(n-1)
        if j >= i:
            j += 1
        delta = np.random.normal(scale=0.01 * np.mean(curr_seq) + 1e-6)
        new_seq = curr_seq.copy()
        # adjust i and j to preserve sum (optional)
        new_seq[i] = max(0.0, new_seq[i] + delta)
        new_seq[j] = max(0.0, new_seq[j] - delta)
        # ensure sum >0
        s = np.sum(new_seq)
        if s < 1e-8:
            continue
        # compute score
        new_score = fast_score(new_seq)
        # accept if better or with prob
        if new_score < curr_score or np.random.rand() < math.exp(-(new_score - curr_score) / max(T0 * (0.9995 ** iter_num), 1e-9)):
            curr_seq = new_seq
            curr_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = new_seq.copy()
                print(f"Iter {iter_num} new best {best_score:.6f}, len {len(best_seq)}")
        iter_num += 1

    # final evaluation through evaluate_sequence (the given function) for best_seq
    try:

def project_simplex(v):
    # ensure non-negative and sum to 1.
    # if all v >=0 and sum(v) = 1, return v.
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
