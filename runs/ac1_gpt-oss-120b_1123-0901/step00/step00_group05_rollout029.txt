Implementation of project onto simplex:

def simplex_projection(v):
    """Project onto probability simplex (non-negative sum=1)."""
    # Using algorithm: sort descending, find rho, theta.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, len(v)+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Simple gradient‑based optimizer for the “max‑convolution” problem.

The goal is to minimise

    2·n·max_k (a * a)[k] / (∑ a_i)²

for a non‑negative vector a.  The score is invariant under scaling,
so we fix the L1‑norm to 1 (∑ a_i = 1) and minimise the maximum
convolution entry directly.

The optimiser works as follows

* start from a uniform distribution (the known 2‑score)
* use a smooth approximation of the max (log‑sum‑exp) to obtain a
  differentiable surrogate loss,
* compute the gradient of the surrogate analytically (it is a
  convolution of the soft‑max weights with the current vector),
* perform a projected gradient step onto the probability simplex,
* slowly increase the smoothness parameter γ so that the surrogate
  approaches the true max,
* keep the best vector seen (according to the exact ``evaluate_sequence``).

The routine runs for at most ``budget_s`` seconds and returns the
best‑scoring sequence it discovered.
"""

import time
import numpy as np

def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of a vector onto the probability simplex (non‑negative, sum = 1)."""
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """Return a sequence of non‑negative heights that improves the score.

    The function uses a simple projected gradient scheme with a
    smooth approximation of the max‑convolution.  It tries several
    problem sizes and keeps the best result found before the time
    budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0  # a safety margin

    best_score = float("inf")
    best_seq = None

    # problem sizes to explore – small to medium n give the biggest
    # chance of beating the current bound 2.0
    sizes = [10, 20, 30, 40, 50, 80, 120, 200, 300]

    for n in sizes:
        if time.time() > deadline:
            break

        # initial vector: uniform + tiny random perturbation
        a = np.full(n, 1.0 / n, dtype=np.float64)
        a += 1e-4 * np.random.rand(n)
        a = _project_onto_simplex(a)

        # hyper‑parameters – they are annealed during the run
        lr = 0.6            # learning rate
        gamma = 5.0         # smooth‑max temperature
        max_iter = 3000

        # main optimisation loop
        for it in range(max_iter):
            if time.time() > deadline:
                break

            # exact convolution (used for a cheap progress check)
            if it % 100 == 0:
                cur_score = evaluate_sequence(a.tolist())
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = a.copy()
                    print(f"[n={n}] iter={it:04d}  score={cur_score:.6f}  lr={lr:.4f}  γ={gamma:.2f}")

            # smooth max (log‑sum‑exp) – use a numerically stable formulation
            conv = np.convolve(a, a)                      # length 2n‑1
            conv_max = conv.max()
            # w are the soft‑max weights over the convolution entries
            exponents = np.exp(gamma * (conv - conv_max))
            w = exponents / exponents.sum()

            # gradient of the smooth surrogate:
            #   ∂/∂a_i  Σ_k w_k (a * a)[k]  =  2 * Σ_k w_k a_{k-i}
            rev_w = w[::-1]                               # reverse for the needed shift
            grad_full = np.convolve(rev_w, a)            # length 3n‑2
            idx = (2 * n - 2) - np.arange(n)              # pick the needed slice
            grad = 2.0 * grad_full[idx]

            # projected gradient step
            a_new = a - lr * grad
            a_new = np.maximum(a_new, 0.0)                # enforce non‑negativity
            a = _project_onto_simplex(a_new)

            # slowly tighten the approximation and decay the step size
            if (it + 1) % 200 == 0:
                lr *= 0.85
                gamma *= 1.15

        # final evaluation for this size (in case the best was never hit inside the loop)
        final_score = evaluate_sequence(a.tolist())
        if final_score < best_score:
            best_score = final_score
            best_seq = a.copy()
            print(f"[n={n}] final  score={final_score:.6f}")

    # safety fallback – should never happen
    if best_seq is None:
        fallback_n = 100
        best_seq = np.full(fallback_n, 1.0 / fallback_n)

    # the caller expects a plain Python list
    return best_seq.tolist()
