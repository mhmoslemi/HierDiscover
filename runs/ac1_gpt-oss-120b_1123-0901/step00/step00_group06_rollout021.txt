"""
Gaussian‑shaped step function optimizer.

We generate a discrete Gaussian (bell‑shaped) sequence:
    a_i = exp(- (i‑μ)^2 / (2·σ²))

where μ is the centre of the vector and σ controls the width.
For a Gaussian the convolution a*a is again a (broader) Gaussian,
its maximum M = ∫ a² ≈ √π·σ while the total sum S = ∫ a ≈ √(2π)·σ.
Consequently the evaluation metric

    2·n·M / S²  ≈  n / (√π·σ)

is minimised by taking σ as large as possible while keeping the
shape non‑uniform (σ≈n/2 gives a value ≈2/√π≈1.13, well below the
target 1.503).

The function performs a tiny grid search on σ (expressed as a fraction
of n) to find the best Gaussian within the allotted time and returns the
corresponding sequence.  No external files or network access are used.
"""

import time
import numpy as np

def _gaussian_sequence(n: int, sigma: float) -> np.ndarray:
    """Return a length‑n Gaussian centred at (n‑1)/2 with width sigma."""
    mu = (n - 1) / 2.0
    x = np.arange(n, dtype=np.float64)
    return np.exp(-((x - mu) ** 2) / (2.0 * sigma * sigma))

    evaluate_sequence score is as low as possible.

    keeps the best sequence found before the time budget expires.
    """

    # -----------------------------------------------------------------

    # and would make each evaluation slower.

    sigma_factors = np.linspace(0.25, 0.5, 21)  # 21 candidates

    deadline = time.time() + budget_s - 0.1   # leave a tiny margin
    best_seq = None
    best_score = float('inf')

    # -----------------------------------------------------------------
    # Search over the few σ candidates
    # -----------------------------------------------------------------
    for factor in sigma_factors:
        if time.time() >= deadline:
            break

        sigma = factor * n
        seq = _gaussian_sequence(n, sigma)

        # evaluate_sequence is provided by the testing harness
        try:
            score = evaluate_sequence(seq.tolist())
        except Exception:
            # any unexpected error – just skip this candidate
            continue

        if score < best_score:
            best_score = score
            best_seq = seq.tolist()
            # Early exit if we already beat the target
            if best_score < 1.503:
                break

    # -----------------------------------------------------------------
    # Fallback (should never be needed)
    # -----------------------------------------------------------------
    if best_seq is None:
        # Uniform sequence – yields score 2, never worse than “inf”.
        best_seq = [1.0] * n
        best_score = evaluate_sequence(best_seq)

    # -----------------------------------------------------------------
    # Log the result for the host system (print output is captured)
    # -----------------------------------------------------------------
    print(f"Gaussian optimizer finished. n={n}, best_score={best_score:.6f}")

    return best_seq
