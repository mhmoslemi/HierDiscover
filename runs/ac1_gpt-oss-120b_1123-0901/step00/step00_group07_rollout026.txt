def greedy_sidb(n, L, order):
    sum_counts = np.zeros(2*n - 1, dtype=np.int16)
    selected = []
    for i in order:
        # check capacity
        ok = True
        for j in selected:
            s = i + j
            if sum_counts[s] >= L:
                ok = False
                break
        if ok:
            # also diagonal
            if sum_counts[2*i] >= L:
                ok = False
        if not ok:

        # add i
        for j in selected:

            sum_counts[s] += 1
        sum_counts[2*i] += 1
        selected.append(i)
    return selected, sum_counts

def local_search(seq, deadline, rng):
    n = len(seq)
    a = seq.copy()
    best = a.copy()
    best_score = evaluate_sequence(a.tolist())
    temp = 1.0
    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        if a[i] < 1e-12:
            continue
        # propose move
        delta = a[i] * rng.random() * 0.5  # up to 50% of a[i]
        # randomly decide sign to move from i to j
        # ensure non-negative
        a_new = a.copy()

        if score < best_score or rng.random() < math.exp((best_score - score)/temp):
            a = a_new

"""
Propose a candidate sequence that minimizes

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The evaluation is scale‑invariant, so we fix the total sum to 1.
The idea is to build a large subset of indices whose pairwise sums collide
as little as possible (a “Sidon‑like” set).  For a given maximum allowed
collision count L we greedily add indices in a random order while ensuring
that every sum i + j (including i = j) is used at most L times.
The larger the set and the smaller L, the smaller the maximal convolution
value becomes (max conv ≈ 2·L·(1/m)² where m is the set size).
After the greedy construction we assign equal weight 1/m to the selected
indices and evaluate the sequence.  Several random restarts with different
values of L (1…5) are performed within the provided time budget.  The best
sequence found is further refined by a short simulated‑annealing style
local search that moves a small amount of mass between two positions while
keeping the total sum fixed.

The algorithm runs entirely with NumPy and uses only the supplied
`evaluate_sequence` function.
"""

import numpy as np
import time
import math

# ----------------------------------------------------------------------
# Helper: greedy construction of a set with limited collisions
# ----------------------------------------------------------------------
def greedy_set(n: int, L: int, rng: np.random.Generator):
    """
    Build a set of indices S ⊆ {0,…,n‑1} such that each unordered sum i+j
    appears at most L times.
    Returns the list of selected indices and the array of sum counts.
    """
    order = rng.permutation(n)
    selected = []                     # list of chosen indices
    sum_counts = np.zeros(2 * n - 1, dtype=np.int16)   # how many pairs give each sum

    for i in order:
        # diagonal sum 2*i must also respect the limit
        if sum_counts[2 * i] >= L:
            continue

        ok = True
        for j in selected:
            if sum_counts[i + j] >= L:
                ok = False
                break
        if not ok:
            continue

        # add i to the set
        selected.append(i)
        sum_counts[2 * i] += 1                     # diagonal pair (i,i)
        for j in selected[:-1]:                    # pairs with previous members
            sum_counts[i + j] += 1

    return selected, sum_counts


# ----------------------------------------------------------------------
# Helper: very light local search (simulated annealing)
# ----------------------------------------------------------------------
def local_search(seq: np.ndarray, deadline: float, rng: np.random.Generator,
                 max_iter: int = 3000):
    """
    Starting from `seq` (non‑negative, sum≈1) try to improve the score by
    moving a small amount of mass between two positions.
    The procedure stops when the time budget is exhausted or after
    `max_iter` attempts.  Returns the best sequence found and its score.
    """
    a = seq.copy()
    best = a.copy()
    best_score = evaluate_sequence(a.tolist())
    temperature = 1.0

    n = len(a)
    for _ in range(max_iter):
        if time.time() >= deadline:
            break

        i = rng.integers(0, n)
        if a[i] <= 1e-12:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue

        # move a random fraction (up to 30%) of a[i] to position j
        delta = a[i] * rng.random() * 0.3
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        # evaluate candidate
        try:
            score = evaluate_sequence(a_new.tolist())
        except Exception:
            continue

        # accept if better or with a Metropolis probability
        if score < best_score or rng.random() < math.exp((best_score - score) / temperature):
            a = a_new
            if score < best_score:
                best_score = score
                best = a_new.copy()

        # cool down
        temperature *= 0.9995
        if temperature < 1e-4:
            temperature = 1e-4

    return best, best_score
