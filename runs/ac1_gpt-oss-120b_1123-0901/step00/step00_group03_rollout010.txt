The algorithm builds a sequence a_i = i+1 for i=0..n-1.
For this shape the self–convolution peak is analytically:
   max (a*a) = n(n+1)(n+2)/6,

This is below the required 1.503 target.
The function also performs a tiny local random refinement

import time
import numpy as np

def linear_sequence(n):
    """Return a non‑negative sequence a_i = i+1 (i starting at 1)."""
    # values are from 1 to n inclusive, all ≤ 1000 for n≤1000
    return np.arange(1, n + 1, dtype=float)

def compute_score(seq):
    """Compute the same score used by evaluate_sequence (without the safeguards)."""
    a = np.asarray(seq, dtype=float)
    n = a.size
    # convolution via FFT (real)
    size = 2 * n - 1
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    max_b = conv.max()
    sum_a = a.sum()
    return float(2.0 * n * max_b / (sum_a ** 2))

"""Linear (1,2,…,n) construction.

def compute_score(seq: np.ndarray) -> float:

    a = seq

    fa = np.fft.rfft(a, n=size)

    """

    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.1   # small safety margin

    # ------------------------------------------------------------------
    current = seq.copy()
