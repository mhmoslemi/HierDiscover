"""
Hybrid softmax-guided projected gradient descent with intermittent
linear‑program (LP) “mass‑reallocation” steps.  The algorithm
optimises a non‑negative vector a on the simplex Σa_i=√(2n) so that
the maximal entry of its self‑convolution is as small as possible.
The smooth surrogate (soft‑max) of the max‑convolution is minimised
by a projected gradient step; every few iterations a linear programme
maximises the ℓ₁‑norm under the current convolution bound and the
result is mixed back into a.  Multiple random restarts of varying
length are performed until the time budget expires.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

# --------------------------------------------------------------------------- #
# Simplex projection (Euclidean) – ensures Σx = s and x_i ≥ 0.
# --------------------------------------------------------------------------- #
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex {x >= 0, Σ x = s}."""
    n = v.shape[0]
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    # sort in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    # find the rho where u_i > cssv_i / i
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------------------- #
# Fast full convolution (real FFT) – returns length 2*n‑1 array.
# --------------------------------------------------------------------------- #
def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    # next power of two for efficient FFT
    n_fft = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    conv = np.fft.irfft(A * A, n=n_fft)[: 2 * n - 1]
    return conv


# --------------------------------------------------------------------------- #
# Build the Toeplitz‐style constraint matrix for the LP:
#   (a * b)_k = Σ_i a_i b_{k-i} ≤ M      for all k
# For a given vector a (length n) this matrix has shape (2n‑1, n) and
# entry (k, i) = a_{k-i} if 0 ≤ k-i < n, else 0.
# --------------------------------------------------------------------------- #
def _build_constraint_matrix(a: np.ndarray) -> csr_matrix:
    n = a.shape[0]
    # rows = i + j  for column i (0≤i<n) and row offset j (0≤j<n)
    # using broadcasting to generate all (= i + j) pairs
    i_idx = np.arange(n, dtype=np.int64)
    j_idx = np.arange(n, dtype=np.int64)
    row = (i_idx[:, None] + j_idx[None, :]).ravel()
    col = np.repeat(i_idx, n)
    data = np.tile(a, n)
    mat = csr_matrix((data, (row, col)), shape=(2 * n - 1, n))
    return mat


# --------------------------------------------------------------------------- #
# Soft‑max (log‑sum‑exp) weights for the current convolution.
# Returns the gradient of the smooth surrogate ϵ·log(∑exp(conv/ϵ)).
# --------------------------------------------------------------------------- #
def _softmax_gradient(a: np.ndarray, eps: float) -> np.ndarray:
    """Gradient of the smooth max of the self‑convolution."""
    n = a.shape[0]
    conv = _conv_full_fft(a)                     # length 2n‑1
    # stabilised softmax
    max_c = conv.max()
    w = np.exp((conv - max_c) / eps)             # w >= 0
    w_sum = w.sum()
    if w_sum == 0:
        w = np.ones_like(w) / w.size
    else:
        w /= w_sum
    # gradient = 2 * (w * a_rev)⟂ where a_rev = a[::-1]
    a_rev = a[::-1]
    grad_full = np.convolve(w, a_rev)            # length 3n‑2
    start = n - 1
    grad = 2.0 * grad_full[start : start + n]
    return grad


# --------------------------------------------------------------------------- #
# One optimisation run for a fixed length.
# --------------------------------------------------------------------------- #
def _optimise_length(n: int, init: np.ndarray, deadline: float,
                     best_score: float, best_seq: np.ndarray) -> tuple[float, np.ndarray]:
    """
    Perform a hybrid optimisation for a vector of length n.
    Returns the best score / sequence discovered during the run.
    """
    target_sum = np.sqrt(2.0 * n)

    # ensure feasibility
    a = _project_onto_simplex(init.astype(float), target_sum)

    # algorithmic hyper‑parameters
    eps = 0.5                # initial soft‑max temperature
    eps_min = 1e-4
    eps_decay = 0.998

    lr = 0.05                # base learning rate for gradient step
    lr_decay = 0.999

    max_iter = 300          # safeguard
    iter_cnt = 0
    lp_every = 12           # run an LP step every this many iterations
    lp_counter = 0

    # pre‑compute row/col pattern for the LP matrix (independent of a)
    i_idx = np.arange(n, dtype=np.int64)
    j_idx = np.arange(n, dtype=np.int64)
    row_pat = (i_idx[:, None] + j_idx[None, :]).ravel()
    col_pat = np.repeat(i_idx, n)

    while time.time() < deadline and iter_cnt < max_iter:
        # ---------- current convolution & score ----------
        conv = _conv_full_fft(a)
        M = conv.max()
        cur_score = float(M)            # same as evaluate_sequence()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.2f}] New global best = {best_score:.6f} (n={n})")
        if cur_score < 1.44:            # aggressive early exit
            break

        # ---------- gradient descent (soft‑max) ----------
        grad = _softmax_gradient(a, eps)

        # basic back‑tracking line search
        step = lr
        improved = False
        while step > 1e-7:
            a_candidate = a - step * grad
            a_candidate = np.maximum(a_candidate, 0.0)
            a_candidate = _project_onto_simplex(a_candidate, target_sum)

            conv_cand = _conv_full_fft(a_candidate)
            M_cand = conv_cand.max()
            if M_cand <= M:
                a = a_candidate
                improved = True
                break
            step *= 0.5

        if not improved:
            # fallback: keep current a but shrink learning‑rate
            pass

        # ---------- occasional LP re‑allocation ----------
        lp_counter += 1
        if lp_counter % lp_every == 0:
            # Build constraint matrix for current a
            data = np.tile(a, n)
            A_mat = csr_matrix((data, (row_pat, col_pat)), shape=(2 * n - 1, n))

            # Linear programme: maximise Σ b_j  subject to a * b ≤ M, b ≥ 0
            # In scipy.linprog we minimise cᵀx, so use c = –1
            lp_res = linprog(
                c=-np.ones(n),
                A_ub=A_mat,
                b_ub=np.full(2 * n - 1, M),
                bounds=[(0, None)] * n,
                method='highs',
                options={'presolve': True, 'time_limit': max(0.0, deadline - time.time())}
            )
            if lp_res.success:
                b = lp_res.x
                sum_b = b.sum()
                if sum_b > 0:
                    scale = target_sum / sum_b
                    g = b * scale
                    conv_g = _conv_full_fft(g)
                    M_g = conv_g.max()
                    if M_g < M:
                        a = g
                        # after a successful LP update we reset learning‑rate a bit
                        lr *= 0.9
                        # also make the temperature a touch larger to escape potential plateaus
                        eps = min(eps * 1.05, 1.0)

        # ---------- schedule updates ----------
        eps = max(eps * eps_decay, eps_min)
        lr *= lr_decay

        # occasional random perturbation (simulated‑annealing style)
        if np.random.rand() < 0.03:
            noise = np.random.randn(n) * (0.005 * target_sum / n)
            a = _project_onto_simplex(np.maximum(a + noise, 0.0), target_sum)
            # allow a bigger step afterwards
            lr *= 0.95

        iter_cnt += 1

    return best_score, best_seq

"""
Hybrid softmax‑guided projected gradient descent with intermittent linear‑program
(LP) “mass‑reallocation” steps.  The algorithm searches for a non‑negative vector
`a` on the simplex Σa_i = √(2 n) that minimises the largest entry of its
self‑convolution.  A smooth surrogate (soft‑max) of the max‑convolution is
optimised by projected gradient steps; every few iterations a linear programme
maximises the ℓ₁‑norm under the current convolution bound and the resulting
vector is mixed back into `a`.  Multiple random restarts of varying length are
performed until the time budget expires.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

# --------------------------------------------------------------------------- #
# Euclidean projection onto the simplex {x ≥ 0, Σx = s}.
# --------------------------------------------------------------------------- #
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex of radius s."""
    n = v.shape[0]
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------------------- #
# Fast full convolution (real FFT) – length 2 n‑1.
# --------------------------------------------------------------------------- #
def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    n_fft = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    conv = np.fft.irfft(A * A, n=n_fft)[: 2 * n - 1]
    return conv


# --------------------------------------------------------------------------- #
# Build the Toeplitz‑style matrix for the LP:
#   (a * b)_k = Σ_i a_i b_{k-i} ≤ M   for all k.
#   Entry (k, i) = a_{k-i} if 0 ≤ k-i < n, else 0.
# --------------------------------------------------------------------------- #
def _build_constraint_pattern(n: int):
    """Return the (row, col) index arrays that are independent of a."""
    i = np.arange(n, dtype=np.int64)
    j = np.arange(n, dtype=np.int64)
    row = (i[:, None] + j[None, :]).ravel()
    col = np.repeat(i, n)
    return row, col


# --------------------------------------------------------------------------- #
# Soft‑max (log‑sum‑exp) gradient of the smooth max of the self‑convolution.
# --------------------------------------------------------------------------- #
def _softmax_gradient(a: np.ndarray, eps: float) -> np.ndarray:
    """Gradient of ε·log(∑exp(conv/ε))."""
    n = a.shape[0]
    conv = _conv_full_fft(a)                      # length 2n‑1
    max_c = conv.max()
    w = np.exp((conv - max_c) / eps)              # stabilised softmax
    w_sum = w.sum()
    if w_sum == 0:
        w = np.ones_like(w) / w.size
    else:
        w /= w_sum
    a_rev = a[::-1]
    grad_full = np.convolve(w, a_rev)             # length 3n‑2
    start = n - 1
    grad = 2.0 * grad_full[start:start + n]
    return grad


# --------------------------------------------------------------------------- #
# Optimisation loop for a single length n.
# --------------------------------------------------------------------------- #
def _optimise_length(
    n: int,
    init: np.ndarray,
    deadline: float,
    best_score: float,
    best_seq: np.ndarray,
) -> tuple[float, np.ndarray]:
    """
    Perform a hybrid optimisation for a vector of length n.
    Returns the best (score, sequence) seen in this run.
    """
    target_sum = np.sqrt(2.0 * n)

    a = _project_onto_simplex(init.astype(float), target_sum)

    # hyper‑parameters
    eps = 0.5                # soft‑max temperature
    eps_min = 1e-4
    eps_decay = 0.998

    lr = 0.05                # gradient‑step scale
    lr_decay = 0.999

    max_iter = 300
    lp_every = 12            # LP step frequency
    lp_counter = 0

    row_pat, col_pat = _build_constraint_pattern(n)   # reusable pattern

    iter_cnt = 0
    while time.time() < deadline and iter_cnt < max_iter:
        # ----- current objective -----
        conv = _conv_full_fft(a)
        M = conv.max()
        cur_score = float(M)                     # equal to evaluate_sequence()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.2f}] New global best = {best_score:.6f} (n={n})")
        if cur_score < 1.44:                     # early exit if already tiny
            break

        # ----- soft‑max gradient descent -----
        grad = _softmax_gradient(a, eps)

        # simple back‑tracking line search
        step = lr
        improved = False
        while step > 1e-7:
            a_candidate = a - step * grad
            a_candidate = np.maximum(a_candidate, 0.0)
            a_candidate = _project_onto_simplex(a_candidate, target_sum)

            conv_cand = _conv_full_fft(a_candidate)
            if conv_cand.max() <= M:
                a = a_candidate
                improved = True
                break
            step *= 0.5

        if not improved:
            # no progress – keep current a, possibly shrink learning rate later
            pass

        # ----- occasional LP “mass re‑allocation” -----
        lp_counter += 1
        if lp_counter % lp_every == 0:
            data = np.tile(a, n)
            A_mat = csr_matrix((data, (row_pat, col_pat)), shape=(2 * n - 1, n))

            lp_res = linprog(
                c=-np.ones(n),                     # maximise Σb  → minimise −Σb
                A_ub=A_mat,

                method="highs",
                options={"presolve": True, "time_limit": max(0.0, deadline - time.time())},
            )
            if lp_res.success:
                b = lp_res.x
                sum_b = b.sum()
                if sum_b > 0:
                    scale = target_sum / sum_b
                    g = b * scale
                    M_g = _conv_full_fft(g).max()
                    if M_g < M:
                        a = g
                        lr *= 0.9
                        eps = min(eps * 1.05, 1.0)

        # ----- schedule updates -----
        eps = max(eps * eps_decay, eps_min)
        lr *= lr_decay

        # ----- occasional random perturbation -----
        if np.random.rand() < 0.03:
            noise = np.random.randn(n) * (0.005 * target_sum / n)
            a = _project_onto_simplex(np.maximum(a + noise, 0.0), target_sum)
            lr *= 0.95

        iter_cnt += 1

    return best_score, best_seq
