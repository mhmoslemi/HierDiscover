"""
Propose a step‑function (non‑negative heights) with a small
2·n·max(conv(a,a))/sum(a)^2 value.

The search combines three ideas:

1.   Projected stochastic gradient descent on the active (maximal)
     convolution entries, using a momentum term.
2.   A simple “mass‑transfer” heuristic that moves probability mass from
     indices with a large sub‑gradient (they increase the current max)
     to indices with a small sub‑gradient.
3.   Occasional random shakes (simulated‑annealing style) to escape
     shallow local minima.

All vectors are kept on the simplex Σa_i = sqrt(2·n) by an
Euclidean projection routine.  The algorithm runs many independent
restarts with varying lengths (800–3000) and stops as soon as the
target bound 1.503 is beaten.  The best sequence found is returned as a
plain Python list.
"""

import time
import math
import numpy as np
from typing import List

# --------------------------------------------------------------------------- #
# Helper: Euclidean projection onto the simplex { x_i ≥ 0 , Σ x_i = s }.
# --------------------------------------------------------------------------- #
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project the vector ``v`` onto the simplex of radius ``s``."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, v.shape[0] + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------------------- #
# Fast full convolution via real FFT – returns a length 2*n‑1 array.
# --------------------------------------------------------------------------- #
def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    n_fft = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    conv = np.fft.irfft(A * A, n=n_fft)[: 2 * n - 1]
    return conv


# --------------------------------------------------------------------------- #
# Gradient of the *maximal* convolution entry (or a set of near‑maximal ones).
# --------------------------------------------------------------------------- #
def _gradient_of_active(a: np.ndarray, active_idxs: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in active_idxs:
        low = max(0, k - (n - 1))
        high = min(n - 1, k)
        idx = np.arange(low, high + 1, dtype=int)
        partner = k - idx
        grad[idx] += 2.0 * a[partner]
    return grad


# --------------------------------------------------------------------------- #
# Main entry point expected by the grader.
# --------------------------------------------------------------------------- #
def propose_candidate(seed: int = 42, budget_s: float = 1000.0) -> List[float]:
    """
    Search for a low‑value step‑function within ``budget_s`` seconds.
    Returns a plain Python list of non‑negative heights.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    best_seq = None            # best numpy array discovered
    best_score = float("inf")  # corresponds to max(conv)

    # ------------------------------------------------------------------- #
    # Core optimisation routine for a given length ``n``.
    # ------------------------------------------------------------------- #
    def _optimise_one_length(n: int, init_vec: np.ndarray) -> None:
        nonlocal best_seq, best_score

        target_sum = math.sqrt(2.0 * n)          # Σ a_i = √(2n)
        a = _project_onto_simplex(init_vec.astype(float), target_sum)

        # hyper‑parameters (they decay automatically)
        base_lr = 0.8
        momentum = np.zeros_like(a)
        beta_mom = 0.9

        # temperature for simulated‑annealing acceptance
        temp = 0.12

        # allocate a slice of the remaining budget for this run
        run_deadline = min(deadline, time.time() + 18.0)   # ≈18 s per run

        while time.time() < run_deadline:
            # ---- current convolution and objective ---------------------------------
            conv = _conv_full_fft(a)
            max_val = conv.max()
            if max_val < best_score:
                best_score = max_val
                best_seq = a.copy()
                print(f"[{time.time():.2f}] New best = {best_score:.6f} (n={n})")
                if best_score < 1.503:          # early stop if target is reached
                    return

            # ---- active (near‑maximal) indices ------------------------------------
            # use a relative tolerance to gather a small active set
            rel_tol = 1e-6
            active = np.where(conv >= max_val * (1.0 - rel_tol))[0]

            # ---- sub‑gradient of the (near) maximal entry -------------------------
            grad = _gradient_of_active(a, active)
            max_grad = grad.max()

            # ---- momentum‑based gradient step ------------------------------------
            if max_grad > 0:
                eta = base_lr * max_val / max_grad
                momentum = beta_mom * momentum - eta * grad
                a_proposed = a + momentum
            else:
                a_proposed = a.copy()

            # ---- simple mass‑transfer heuristic ------------------------------------
            # move a little mass from the index with the largest gradient to the
            # one with the smallest gradient (keeps Σ a_i unchanged).
            i_high = int(np.argmax(grad))
            i_low = int(np.argmin(grad))
            if grad[i_high] > grad[i_low] + 1e-9:
                # amount is a fraction of the source entry, limited by the lr
                delta = min(a_proposed[i_high],
                            base_lr * (grad[i_high] - grad[i_low]) / (max_grad + 1e-12))
                a_proposed[i_high] -= delta
                a_proposed[i_low] += delta

            # ---- occasional random perturbation (annealing step) -------------------
            if np.random.rand() < 0.05:
                noise = np.random.randn(n) * (0.012 * target_sum / n)
                a_proposed = np.maximum(a_proposed + noise, 0.0)
                base_lr *= 0.93    # after a shake we shrink the learning rate

            # ---- feasibility restoration -------------------------------------------
            a_proposed = _project_onto_simplex(np.maximum(a_proposed, 0.0), target_sum)

            # ---- optional acceptance test (allows occasional uphill moves) -------
            if max_val > best_score:
                # accept with a Metropolis probability
                accept_prob = math.exp(-(max_val - best_score) / max(1e-12, temp))
                if np.random.rand() >= accept_prob:
                    # reject, keep the old vector
                    a_proposed = a

            # ---- update for next iteration -----------------------------------------
            a = a_proposed
            # decay schedules
            base_lr *= 0.9996
            temp *= 0.9994

    # ------------------------------------------------------------------- #
    # Outer loop – many independent restarts with varying lengths.
    # ------------------------------------------------------------------- #
    while time.time() < deadline:
        # pick a length; larger n gives more freedom but is slower
        n = np.random.randint(800, 3001)

        # initialise – reuse a previously good construction if possible
        if np.random.rand() < 0.35 and "height_sequence_1" in globals():
            src = np.asarray(height_sequence_1, dtype=float)
            if src.size >= n:
                init = src[:n].copy()
            else:
                pad = np.random.rand(n - src.size) * 1e-3
                init = np.concatenate([src, pad])
        else:
            init = np.random.rand(n)

        _optimise_one_length(n, init)

        # early exit if we already beat the target bound
        if best_score < 1.503:
            break

    # ------------------------------------------------------------------- #
    # Fallback – if nothing was found (should never happen).
    # ------------------------------------------------------------------- #
    if best_seq is None:
        n = 1000
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)

    # Convert to plain Python list as required by the grader
    return best_seq.tolist()
