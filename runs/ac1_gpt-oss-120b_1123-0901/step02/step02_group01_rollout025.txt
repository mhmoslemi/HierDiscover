def lp_step(current_seq, M):
    import cvxpy as cp
    n = len(current_seq)
    b = cp.Variable(n)
    # conv f*b
    # Use cp.convolve
    conv_expr = cp.conv(current_seq, b)
    constraints = [conv_expr <= M, b >= 0]
    objective = cp.Maximize(cp.sum(b))
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.ECOS, abstol=1e-8, reltol=1e-8, max_iters=5000)
    if b.value is None:

    # normalize sum to 1

    if sum_b == 0:

    b_seq = b_seq / sum_b

"""
Propose a step‑function sequence with a small L∞ self‑convolution ratio.

The algorithm mixes three ideas:

1. **Two‑phase Adam optimisation**
   – Phase 1 (first ~1000 iterations) minimises the variance of the full
     self‑convolution (flattens the whole triangle).
   – Phase 2 uses a smooth‑max (log‑sum‑exp) surrogate that converges to
     the true max.  The temperature λ is increased slowly (continuation).
   – Gradients are obtained by a single FFT‑based cross‑correlation
     (O(n log n)).  After each step the point is projected onto the
     probability simplex (non‑negative, sum = 1) using the Euclidean
     projection.

2. **Stagnation handling**
   – If no improvement is seen for >2 s a tiny random perturbation is
     added.
   – If stagnation lasts >5 s a cheap linear programme (via CVXPY) is
     solved: maximise the total mass of a non‑negative partner `g`
     subject to the current convolution bound.  Mixing the current
     sequence with the LP solution (10 % weight) often breaks the
     plateau.

3. **Multi‑restart & refinement**
   – Several random initialisations (geometric, Dirichlet, uniform) are
     tried on a moderate length `n=2000`.  The best result is kept.
   – If time permits the best sequence is interpolated to a larger
     size (`n=4000`) and a short second optimisation run is performed,
     allowing the extra degrees of freedom to lower the objective
     further.

All operations respect the strict checks of `evaluate_sequence`.  The
routine honours the 1000 s budget – it stops a few tenths of a second
early to avoid timeout.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Projection onto the probability simplex {x ≥ 0, sum(x)=1}
def _project_onto_simplex(v):
    """Euclidean projection of a vector onto the simplex."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        # all entries non‑positive ⇒ uniform simplex point
        return np.full_like(v, 1.0 / len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based full linear convolution a*a (length 2·n‑1)
def _next_pow2(x: int) -> int:
    return 1 << (x - 1).bit_length()


def _full_convolution(a, size=None):
    """Linear self‑convolution a*a (full, length 2·len(a)‑1)."""
    n = len(a)
    if size is None:
        size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Exact objective used by `evaluate_sequence`
def _objective(a):
    """2·n·max(conv) / (sum(a)²).  Assumes a ≥ 0 and sum(a) > 0."""
    n = len(a)
    conv = _full_convolution(a)
    max_b = conv.max()
    return 2.0 * n * max_b / (a.sum() ** 2)


# ----------------------------------------------------------------------
# Gradient of the variance ½·∑(conv-avg)² (flattens the whole triangle)
def _variance_gradient(a, conv, size_corr):
    """
    Returns ∇ ½·∑(conv-avg)² = 4·( (conv-avg) ★ a ),
    where ★ denotes linear cross‑correlation.
    """
    n = len(a)
    avg = 1.0 / (2 * n - 1)
    w = conv - avg                     # weight vector (length 2n‑1)
    rev_a = a[::-1]
    fw = np.fft.rfft(w, size_corr)
    frev = np.fft.rfft(rev_a, size_corr)
    corr = np.fft.irfft(fw * frev, size_corr)
    grad = 4.0 * corr[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Gradient of the smooth‑max surrogate  (1/λ)·log∑exp(λ·conv)
def _smoothmax_gradient(a, conv, lam, size_corr):
    """
    Returns ∇φ_λ(a) = 2·(softmax(λ·conv) ★ a)
    """
    n = len(a)
    m = conv.max()
    w = np.exp(lam * (conv - m))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(conv, 1.0 / len(conv))
    else:
        w /= w_sum
    rev_a = a[::-1]
    fw = np.fft.rfft(w, size_corr)
    frev = np.fft.rfft(rev_a, size_corr)
    corr = np.fft.irfft(fw * frev, size_corr)
    grad = 2.0 * corr[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Optional LP step (requires cvxpy).  Returns a new feasible sequence
# or None if the solver fails.
def _lp_step(f_seq, M):
    try:
        import cvxpy as cp
    except Exception:                 # cvxpy not available → skip
        return None

    n = len(f_seq)
    b = cp.Variable(n)
    f = cp.Parameter(n, value=f_seq, name="f")
    # linear convolution f * b (full)
    conv_expr = cp.conv(f, b)          # length 2n‑1
    constraints = [conv_expr <= M, b >= 0]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
    # a fast, reliable interior‑point solver
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000,
               abstol=1e-8, reltol=1e-8, feastol=1e-8)

    if b.value is None:
        return None
    g = np.maximum(b.value, 0.0)
    s = g.sum()
    if s == 0:
        return None
    return g / s                       # normalise to sum = 1

            g = _lp_step(seq, M)
            if g is not None:
                # convex combination (10 % of the LP solution)
                seq = 0.9 * seq + 0.1 * g
                seq = _project_onto_simplex(seq)
                val_lp = _objective(seq)
                if val_lp < best_val:
                    best_val, best_seq = val_lp, seq.copy()
                    last_improve = time.time()
                    print(f"[LP step] improved to {val_lp:.6f}")

        # ----- decay the learning rate --------------------------------------
        lr *= lr_factor
        it += 1

        # early exit if we are already under the target
        if best_val < 1.48:
            break

    return best_seq, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises
    `evaluate_sequence`.  Uses a multi‑restart Adam optimiser,
    a variance‑flattening phase, a smooth‑max phase and occasional
    linear‑programming refinements.  Returns the best sequence found
    as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.4   # safety margin

    # ------------------------------------------------------------------
    # 1.  Fast work on a moderate size (n≈2000)
    n = 2000
    best_seq = None
    best_val = float("inf")

    # helper to start a new optimisation run with a given initial vector
    def _run_one(init_seq):
        nonlocal best_seq, best_val
        # give each run a personal deadline (roughly 15 s per run or less)
        sub_deadline = time.time() + min(25.0, deadline - time.time())
        seq_opt, val_opt = _optimise_one_start(init_seq, sub_deadline)
        if val_opt < best_val:
            best_val, best_seq = val_opt, seq_opt.copy()
            print(f"[Global] improved global best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2.  Generate a few diverse starting points and optimise them
    start_time = time.time()
    while time.time() < deadline - 5.0:          # reserve a few seconds for final refinement
        # choose a random model for the start point
        model = np.random.choice(["geom", "dirichlet", "uniform"])
        if model == "geom":
            r = np.random.uniform(0.6, 0.98)     # decay factor
            powers = np.power(r, np.arange(n, dtype=np.float64))
            init = powers / powers.sum()
        elif model == "dirichlet":
            init = np.random.dirichlet(np.ones(n))
        else:  # uniform
            init = np.full(n, 1.0 / n)

        _run_one(init)

        # stop early if we already beat the target comfortably
        if best_val < 1.45:
            break

        # keep an eye on the overall time budget
        if time.time() - start_time > budget_s * 0.6:
            break

    # ------------------------------------------------------------------
    # 3.  If time permits, upscale to a larger problem size and refine
    if best_seq is not None and time.time() < deadline - 2.0:
        # upscale to n = 4000 via linear interpolation
        n_big = 4000
        x_old = np.linspace(0.0, 1.0, len(best_seq))
        x_new = np.linspace(0.0, 1.0, n_big)
        big_seq = np.interp(x_new, x_old, best_seq)
        big_seq = np.maximum(big_seq, 0.0)
        big_seq /= big_seq.sum()
        # run a short optimisation on the larger vector
        sub_deadline = time.time() + min(30.0, deadline - time.time())
        seq_opt, val_opt = _optimise_one_start(big_seq, sub_deadline)
        if val_opt < best_val:
            best_val, best_seq = val_opt, seq_opt.copy()
            print(f"[Refine] improved after up‑scaling = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4.  Return the best sequence found (as plain Python list)
    if best_seq is None:
        # should never happen, but guard against it
        raise RuntimeError("No feasible sequence was generated.")
    return best_seq.tolist()
