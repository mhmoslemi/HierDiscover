def propose_candidate(seed=123, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5
    
    # we can try length list
    lengths = [500, 1000, 2000]  # maybe from small to large
    best_seq = None
    best_val = float('inf')
    
    for n in lengths:
        # define number of inner iterations dependent on time left
        # start with random Dirichlet (size n)
        seq = np.random.dirichlet(np.ones(n))
        # optionally use previous best sequence truncated/padded to length n?
        # Not needed; just random start.
        # Use Adam for this length.
        # Parameter schedule
        lam = 1.0
        lam_factor = 1.01
        alpha = 0.05   # base step for Adam (learning rate)
        beta1 = 0.9
        beta2 = 0.999
        eps = 1e-8
        m = np.zeros_like(seq)
        v = np.zeros_like(seq)
        t = 0
        size_fft = 1 << int(np.ceil(np.log2(3*n)))  # for convolution/correlation
        last_improve = time.time()
        while time.time() < deadline:
            conv = _full_convolution(seq)  # length 2n-1
            # compute gradient of smooth max
            grad = _softmax_gradient(seq, conv, lam, size_corr=size_fft)
            # Adam update
            t += 1
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad * grad)
            m_hat = m / (1 - beta1**t)
            v_hat = v / (1 - beta2**t)
            step = alpha * m_hat / (np.sqrt(v_hat) + eps)
            seq = seq - step
            # project onto simplex
            seq = _project_onto_simplex(seq)
            # compute exact objective
            val = _objective(seq)
            if val < best_val:
                best_val = val
                best_seq = seq.copy()
                last_improve = time.time()

"""
Propose a step‑function sequence with a low evaluation score.

The algorithm is a projected Adam optimiser that minimises the
smooth‑max surrogate of the self‑convolution peak.  It works as follows:

1. **Initial candidates** – a collection of geometric progressions
   (a_i ∝ r^i) and random Dirichlet vectors are examined; the best of
   them (according to the exact objective) becomes the starting point.

2. **Smooth‑max surrogate** – instead of the hard `max` we use
   `ϕ_λ(a) = (1/λ)·log Σ_k exp(λ·(a*a)[k])`.  Its gradient is
   `∇ϕ = 2·(w ★ a)` where `w` are the soft‑max weights of the
   convolution values.  For large λ the surrogate approaches the true
   max.

3. **Adam update + simplex projection** – an Adam‑style adaptive step
   is taken and the result is projected onto the probability simplex
   `{x ≥ 0, Σx = 1}` (the feasible set of the original problem).

4. **Continuation** – λ starts small (smooth landscape) and is
   multiplied by a factor each iteration, gradually sharpening the
   objective.  When λ reaches a preset maximum a sub‑gradient based on
   the active set (indices attaining the current max) is used instead
   of the soft‑max.

5. **Escaping stagnation** – if no improvement has been seen for a few
   seconds a tiny random perturbation (“shake”) is applied.

The loop runs until the time budget (≈1000 s) is exhausted and the best
sequence found is returned as a plain list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v):
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:                            # all non‑positive
        return np.full_like(v, 1.0 / len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _full_convolution(a, size=None):
    """Linear self‑convolution a*a (length 2·len(a)-1) using FFT."""
    n = a.shape[0]
    if size is None:
        size = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
def _objective(a):
    """Exact objective used by evaluate_sequence (sum(a) = 1)."""
    n = a.shape[0]
    conv = _full_convolution(a)
    max_b = conv.max()
    return 2.0 * n * max_b  # sum(a)=1 ⇒ denominator = 1


# ----------------------------------------------------------------------
def _softmax_gradient(a, conv, lam, size_corr):
    """
    Gradient of the smooth‑max surrogate:
        ∇ϕ_λ(a) = 2·(w ★ a)
    where w_k ∝ exp(λ·conv_k) (soft‑max weights).
    """
    # soft‑max weights – numerically stable
    m = conv.max()
    w = np.exp(lam * (conv - m))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # cross‑correlation w ★ a  (linear, not circular)
    rev_a = a[::-1]
    fw = np.fft.rfft(w, size_corr)
    frev = np.fft.rfft(rev_a, size_corr)
    corr = np.fft.irfft(fw * frev, size_corr)

    n = a.shape[0]
    grad = 2.0 * corr[n - 1 : n - 1 + n]
    return grad

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = alpha * m_hat / (np.sqrt(v_hat) + eps)

        seq = seq - step
        seq = _project_onto_simplex(seq)

        # 3d) exact objective evaluation
        val = _objective(seq)
        if val < best_val:
            best_val, best_seq = val, seq.copy()
            last_improve = time.time()
            print(f"[Iter {it:06d}] new best = {val:.6f}")

        # 3e) random shake if we have been idle for a few seconds
        if time.time() - last_improve > 3.0:
            noise = (np.random.rand(n) - 0.5) * 1e-4
            seq = _project_onto_simplex(seq + noise)
            last_improve = time.time()

        # 3f) increase λ (sharpen the surrogate)
        lam = min(lam_max, lam * lam_factor)

    print(f"Final best objective = {best_val:.6f} (length {n})")
    # The evaluator expects a plain Python list
    return best_seq.tolist()
