def _softmax_gradient(a: np.ndarray, conv: np.ndarray, beta: float) -> np.ndarray:
    w = np.exp(beta * (conv - np.max(conv)))   # subtract max for stability
    w /= np.sum(w)
    w_rev = w[::-1]
    conv_rev = np.convolve(a, w_rev)
    offset = len(w_rev) - 1
    grad = 2.0 * conv_rev[offset - np.arange(len(a))]
    return grad

"""
Proposes a low‑peak autocorrelation (step) function.

The search proceeds in two stages:

1. **Soft‑max smoothed gradient descent** –  
   The non‑smooth objective `max(conv(a,a))` is approximated by the
   smoothed functional  L_β(a)= (1/β)·log∑exp(β·conv_k).  
   Its gradient w.r.t. a is `2·conv_rev`, where `conv_rev` is the
   convolution of the current sequence with the reversed soft‑max
   weights.  Using an increasing β (starting at 0.5 and ending at 8)
   the algorithm first smooths the landscape (global moves) and then
   hones in on the true maximum.  After each step the vector is projected
   onto the simplex {x≥0, Σx = √(2·n)} so the evaluation formula simplifies
   to `max(conv)`.

2. **Short projected‑gradient refinement** –  
   When a promising candidate is found the routine switches to the
   classic sub‑gradient descent on the actual `max(conv)` for a few
   seconds, polishing the solution.

Multi‑restart is used: several lengths n (≈2000) are tried,
the current best solution is resized to the new length and used as a warm‑start.
Random mutations are injected occasionally to escape plateaus.

The whole routine respects the 1000 s budget, prints progress and finally
returns the best sequence (list of non‑negative floats) it discovered.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simple Euclidean projection onto the simplex {x ≥ 0, Σ x = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex of non‑negative vectors with sum s."""
    n = v.shape[0]
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Very small mutation helper (Gaussian noise + simplex projection)
# ----------------------------------------------------------------------
def _mutate(a: np.ndarray, target_sum: float, sigma: float) -> np.ndarray:
    """Return a mutated copy of a (Gaussian noise, then simplex projection)."""
    b = a + np.random.randn(a.size) * sigma
    b = np.maximum(b, 0.0)
    b = _project_onto_simplex(b, target_sum)
    return b


# ----------------------------------------------------------------------
# Max‑convolution and its sub‑gradient (used for the final polish)
# ----------------------------------------------------------------------
def _max_conv_and_grad(a: np.ndarray):
    """
    Compute max(conv(a, a)), its index and the sub‑gradient of that entry.
    Returns (max_val, grad, max_idx, full_conv).
    """
    n = a.shape[0]
    conv = np.convolve(a, a)
    max_idx = int(np.argmax(conv))
    max_val = conv[max_idx]

    low = max(0, max_idx - (n - 1))
    high = min(n - 1, max_idx)
    idx = np.arange(low, high + 1, dtype=int)
    partner = max_idx - idx
    grad = np.zeros_like(a)
    grad[idx] = 2.0 * a[partner]
    return max_val, grad, max_idx, conv


# ----------------------------------------------------------------------
# Projected gradient descent on the *actual* max‑convolution (polishing)
# ----------------------------------------------------------------------
def _local_descent(
    init: np.ndarray,
    target_sum: float,
    time_limit: float,
    base_step: float = 0.05,
    tag: str = "",
) -> tuple[np.ndarray, float]:
    """
    Short projected‑gradient descent using the sub‑gradient of the maximal entry.
    """
    a = _project_onto_simplex(init.astype(float), target_sum)

    try:
        best_score = float(evaluate_sequence(a.tolist()))
    except Exception:
        best_score = math.inf
    best_seq = a.copy()

    step = base_step
    it = 0
    t0 = time.time()

    while time.time() < t0 + time_limit:
        max_val, grad, _, _ = _max_conv_and_grad(a)

        # make the step size proportional to the current max entry
        max_abs = np.max(np.abs(grad)) if np.max(np.abs(grad)) > 0 else 1.0
        eta = step * max_val / max_abs

        a_new = a - eta * grad
        a_new = _project_onto_simplex(a_new, target_sum)

        try:
            score = float(evaluate_sequence(a_new.tolist()))
        except Exception:
            score = math.inf

        if score < best_score:
            best_score = score
            best_seq = a_new.copy()
            print(f"[{time.time():.2f}] {tag} local best = {score:.6f}")

        # occasional random shake‑up
        if np.random.rand() < 0.03:
            sigma = 0.01 * target_sum / len(a)
            a = _mutate(a_new, target_sum, sigma)
        else:
            a = a_new

        if it % 250 == 0 and it > 0:

        it += 1

    return best_seq, best_score


# ----------------------------------------------------------------------
# Soft‑max weighted gradient (smooth approximation of max‑conv)
# ----------------------------------------------------------------------
def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Return the soft‑max weights of conv with temperature 1/β."""
    # shift by the maximum to avoid overflow
    m = np.max(conv)
    exp_vals = np.exp(beta * (conv - m))
    total = np.sum(exp_vals)
    if total == 0.0:
        # fallback to uniform weights (should not happen)
        return np.full_like(conv, 1.0 / conv.size)
    return exp_vals / total


def _softmax_gradient(a: np.ndarray, conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Gradient of the soft‑max‑smoothed objective:
        g = 2 * conv_rev,
    where conv_rev = (a * w_rev) and w = softmax(conv, beta).
    """
    w = _softmax_weights(conv, beta)
    w_rev = w[::-1]
    conv_rev = np.convolve(a, w_rev)            # length = n + (2n‑1)‑1 = 3n‑2
    offset = len(w_rev) - 1                     # = 2n‑2
    # select the n entries that correspond to shifts 0 … n‑1
    grad = 2.0 * conv_rev[offset - np.arange(len(a))]
    return grad

# ----------------------------------------------------------------------
# Helper to resize a known good sequence to a different length
# ----------------------------------------------------------------------
def _resize_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize ``seq`` to ``new_n`` using linear interpolation (preserves non‑negativity)."""
    old_n = seq.size
    if old_n == new_n:
        return seq.copy()
    if old_n > new_n:
        idx = np.linspace(0, old_n - 1, new_n).astype(int)
        return seq[idx]
    else:
        x_old = np.arange(old_n)
        x_new = np.linspace(0, old_n - 1, new_n)
        return np.interp(x_new, x_old, seq)

            print(f"[{time.time():.2f}] start from historic seq, score={sc:.6f}")
        except Exception:
            pass

    if best_seq is None:
        # fallback: uniform sequence of moderate size
        n0 = 1200
        target0 = math.sqrt(2.0 * n0)
        best_seq = np.full(n0, target0 / n0, dtype=float)
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"[{time.time():.2f}] start from uniform, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Main multi‑restart loop
    # ------------------------------------------------------------------
    while time.time() < deadline - 2.0:
        # pick a length – bias towards current best but wander a bit
        if np.random.rand() < 0.7 and best_seq is not None:
            n = best_seq.size
            if np.random.rand() < 0.25:          # occasional stretch/shrink
                n = int(n * np.random.uniform(0.85, 1.15))
                n = max(300, min(2500, n))
        else:
            n = np.random.randint(600, 2500)

        target_sum = math.sqrt(2.0 * n)

        # initialise candidate (resize current best)
        if best_seq is not None:
            init_vec = _resize_sequence(best_seq, n)
        else:
            init_vec = np.random.rand(n)
            init_vec = init_vec / init_vec.sum() * target_sum

        # allocate a time slice for this run
        time_left = deadline - time.time()
        if time_left < 8.0:
            break
        slice_time = min(22.0, time_left * 0.55)   # leave room for a final polish

        # --- Phase 1 : smoothed optimisation --------------------------------
        cand_seq, cand_score = _optimize_softmax(
            n=n,
            time_budget=slice_time,
            init=init_vec,
        )

        # --- Phase 2 : short exact max‑conv descent (polish) ---------------
        remaining = deadline - time.time()
        if remaining > 4.0:
            cand_seq, cand_score = _local_descent(
                init=cand_seq,
                target_sum=target_sum,
                time_limit=min(5.0, remaining),
                base_step=0.07,
                tag=f"polish n={n}"
            )

        # keep global best
        if cand_seq is not None and cand_score < best_score:
            best_score = cand_score
            best_seq = cand_seq.copy()
            print(f"[{time.time():.2f}] *** GLOBAL NEW BEST = {best_score:.6f} (n={n})")
            if best_score < 1.503:               # early exit if target passed
                break

    # ------------------------------------------------------------------
    # Final safety fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1200
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=float)

    return best_seq.tolist()
