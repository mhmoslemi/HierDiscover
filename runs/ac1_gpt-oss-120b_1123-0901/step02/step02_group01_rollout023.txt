"""
Propose a low‑evaluation step‑function using a two‑phase optimizer:

1. **Adam on a smooth‑max surrogate** –  
   The hard max in the objective
       F(a)=2·n·max_k (a*a)[k]/(∑a_i)²
   is replaced by a log‑sum‑exp with a temperature λ.  The parameters are
   re‑parameterised by a soft‑max (`a_i = exp(z_i)/∑exp(z)`) which guarantees
   non‑negativity and the simplex constraint.  An Adam optimiser (first‑order
   with adaptive step‑size) drives the surrogate while λ is increased
   geometrically, turning the surrogate gradually into the true max.

2. **Fast local‑search refinement** –  
   After the Adam phase the current sequence is fine‑tuned by many tiny
   “mass‑transfer’’ moves: pick two indices *i*, *j* and move a small amount
   δ from *i* to *j*.  The effect on the self‑convolution can be updated in
   O(n) using a padded view of the current sequence, so millions of trials
   fit easily into the remaining time.  Improvements are kept and the best
   sequence found during the whole run is returned.

Both phases respect the strict input checks of `evaluate_sequence` and stay
well within the 1000 s wall‑clock budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex (not used in the final
# optimiser – kept for completeness and possible fall‑backs).
def _project_onto_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear (non‑circular) self‑convolution via FFT
def _full_convolution(a, size=None):
    n = len(a)
    if size is None:
        size = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Exact objective used by evaluate_sequence (sum‑normalised)
def _objective(a):
    n = len(a)
    conv = _full_convolution(a)
    max_b = conv.max()
    sum_a = a.sum()
    # sum_a is guaranteed > 0 for our candidates
    return 2.0 * n * max_b / (sum_a * sum_a)


# ----------------------------------------------------------------------
# Gradient of the smooth‑max surrogate φ_λ(a) = (1/λ)·log∑exp(λ·c_k)
def _softmax_gradient(a, conv, lam, size_corr=None):
    """
    Returns ∇_a φ_λ(a) = 2·(w ★ a), where w are the soft‑max weights over c=conv.
    The result is a vector of length len(a).
    """
    n = len(a)
    m = conv.max()
    # stable soft‑max
    w = np.exp(lam * (conv - m))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.zeros_like(w)
        w[conv.argmax()] = 1.0
    else:
        w /= w_sum

    rev_a = a[::-1]
    if size_corr is None:
        size_corr = 1 << int(np.ceil(np.log2(len(w) + n - 1)))
    fw = np.fft.rfft(w, size_corr)
    frev = np.fft.rfft(rev_a, size_corr)
    corr = np.fft.irfft(fw * frev, size_corr)
    grad = 2.0 * corr[n - 1 : n - 1 + n]   # slice that matches a's indices
    return grad


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence``.  The routine runs for at most ``budget_s`` seconds
    and returns the best sequence found (as a plain Python list).
    """
    np.random.seed(seed)
    # safety margin for the interpreter / printing
    deadline = time.time() + budget_s - 0.5

    # --------------------------------------------------------------
    # 1. Length selection – 2000 is a good compromise (fast FFT, enough
    #    degrees of freedom).  If a global ``height_sequence_1`` exists,
    #    we adopt its length.
    n = 2000
    if 'height_sequence_1' in globals():
        try:
            tmp = np.asarray(height_sequence_1, dtype=float)
            if tmp.size >= 10:               # sanity check
                n = int(tmp.size)
        except Exception:
            pass

    # --------------------------------------------------------------
    # 2. Gather a few cheap initial points (geometric series + Dirichlet)
    best_seq = None
    best_val = float("inf")

    # geometric progressions (a_i ∝ r^i)
    for r in np.linspace(0.90, 0.99, 15):
        if time.time() > deadline:
            break
        powers = np.power(r, np.arange(n, dtype=np.float64))
        seq = powers / powers.sum()
        val = _objective(seq)
        if val < best_val:
            best_val, best_seq = val, seq.copy()
            print(f"[Init geom] r={r:.4f}  obj={val:.6f}")

    # random Dirichlet seeds
    for _ in range(12):
        if time.time() > deadline:
            break
        seq = np.random.dirichlet(np.ones(n))
        val = _objective(seq)
        if val < best_val:
            best_val, best_seq = val, seq.copy()
            print(f"[Init rand] obj={val:.6f}")

    # fall‑back (uniform) – should never be chosen because above already set
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)
        best_val = _objective(best_seq)

    # --------------------------------------------------------------
    # 3. Adam optimisation on the smooth‑max surrogate
    #    – parameters are stored in the “logits’’ z so that a = softmax(z)
    eps = 1e-12
    z = np.log(best_seq + eps)        # initialise logits
    m = np.zeros_like(z)             # first moment
    v = np.zeros_like(z)             # second moment
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr = 0.5                         # base learning rate
    lam = 2.0                        # start temperature (smoothness)
    lam_factor = 1.02                # geometric increase per iteration
    size_corr = 1 << int(np.ceil(np.log2(3 * n)))   # FFT size for correlation
    iter_cnt = 1
    last_improve = time.time()
    # allocate about 65 % of the budget to the Adam phase
    adam_deadline = time.time() + 0.65 * (budget_s - 1.0)

    while time.time() < min(deadline, adam_deadline):
        # current simplex point
        exp_z = np.exp(z - np.max(z))          # stabilise soft‑max
        a = exp_z / exp_z.sum()                # a lives on the probability simplex

        # 3a) self‑convolution
        conv = _full_convolution(a)

        # 3b) gradient of smooth‑max surrogate
        grad_a = _softmax_gradient(a, conv, lam, size_corr=size_corr)

        # 3c) chain rule: gradient w.r.t logits z
        dot = np.dot(grad_a, a)                # ∑_i a_i·∂φ/∂a_i
        grad_z = a * (grad_a - dot)            # dφ/dz_i = a_i (∂φ/∂a_i - dot)

        # 3d) Adam update (bias‑corrected)
        m = beta1 * m + (1 - beta1) * grad_z
        v = beta2 * v + (1 - beta2) * (grad_z ** 2)
        m_hat = m / (1 - beta1 ** iter_cnt)
        v_hat = v / (1 - beta2 ** iter_cnt)
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        z -= step

        # 3e) evaluate the *exact* objective (not the surrogate)
        val = _objective(a)
        if val < best_val:
            best_val, best_seq = val, a.copy()
            last_improve = time.time()
            print(f"[Adam {iter_cnt:06d}] new best obj={val:.6f}")

        # 3f) schedule updates
        lam *= lam_factor
        iter_cnt += 1

        # occasional tiny random shake when stuck
        if time.time() - last_improve > 2.0:
            # add a small zero‑mean Gaussian perturbation in logit space
            z += (np.random.randn(n) * 1e-3)
            last_improve = time.time()

    # --------------------------------------------------------------
    # 4. Fast local‑search refinement (pairwise mass transfers)
    #    – keep the best sequence obtained so far and try to lower the
    #    exact objective by tiny redistributions of probability mass.
    a = best_seq.copy()
    conv = _full_convolution(a)
    conv_len = 2 * n - 1
    pad_len = n - 1
    best_val_local = best_val
    last_improve_loc = time.time()

    while time.time() < deadline:
        # pick two distinct indices
        i, j = np.random.choice(n, size=2, replace=False)
        if a[i] <= 0:
            continue

        # propose moving a modest fraction (≤5 %) of mass from i to j
        frac = np.random.rand() * 0.05
        delta = a[i] * frac                     # amount transferred (always ≥0)

        # fast view of shifted copies of a (zero‑padded)
        padded = np.concatenate([np.zeros(pad_len), a, np.zeros(pad_len)])
        shift_i = padded[i : i + conv_len]
        shift_j = padded[j : j + conv_len]

        # update of the convolution using the linearised formula:
        #   c_new = c + 2·δ·(shift_j - shift_i)
        new_conv = conv + 2.0 * delta * (shift_j - shift_i)
        new_val = 2.0 * n * new_conv.max()

        # accept if we improve the exact objective
        if new_val < best_val_local:
            a[i] -= delta
            a[j] += delta
            conv = new_conv
            best_val_local = new_val
            best_seq = a.copy()
            last_improve_loc = time.time()
            print(f"[Local] new best obj={new_val:.6f}")

        # if we have been idle for >3 s, apply a tiny random shake
        if time.time() - last_improve_loc > 3.0:
            # add a small Gaussian jitter and renormalise
            a += (np.random.randn(n) * 1e-4)
            a = _project_onto_simplex(a)    # cheap projection back onto the simplex
            conv = _full_convolution(a)
            last_improve_loc = time.time()

    # --------------------------------------------------------------
    # 5. Return the best sequence (as plain Python list)
    #    – the evaluator will normalise internally, so we keep sum=1.
    print(f"Final best objective = {best_val_local:.6f} (length {n})")
    return best_seq.tolist()
