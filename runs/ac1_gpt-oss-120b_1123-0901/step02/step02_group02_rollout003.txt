def _conv_fft_pair(x, y):
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    return conv

def propose_candidate(seed=42, budget_s=1000.0, **kwargs):
   np.random.seed(seed)
   start_time = time.time()
   deadline = start_time + budget_s - 0.01 (?) we need to avoid overshoot.

   best_seq = None
   best_score = float('inf')
   try:
       global height_sequence_1
       hs = height_sequence_1
       # convert to list or np
       seq = np.array(hs, dtype=float)
       # ensure nonneg
       seq = np.maximum(seq, 0.0)
       # Cap large
       seq = np.clip(seq, 0.0, 1000.0)
       # Evaluate
       score = evaluate_sequence(seq.tolist())
       if score < best_score:
           best_score = score
           best_seq = seq.copy()
   except Exception:
       pass

   # main loop
   while time.time() < deadline:
       remaining = deadline - time.time()
       # Adapt length based on remaining time: earlier many restarts, later longer ones
       if remaining > 300:
           n = np.random.randint(800, 3500)  # typical
       elif remaining > 120:
           n = np.random.randint(600, 3000)
       else:
           n = np.random.randint(400, 2500)

           tau_exp = np.random.uniform(0.5 * n, 2.0 * n)  # decay length

       a += np.random.rand(n) * 1e-5  # jitter

       # Warm start from best

"""
Optimisation of a non‑negative step‑function to minimise

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm follows a soft‑max (smooth‑max) sub‑gradient descent:

* The sequence is always projected onto the simplex
  Σ a_i = √(2 n)  (this makes the objective equal to max(conv)).
* In each iteration the full self‑convolution `c = a*a` is computed by FFT.
* A smooth approximation of the max is used:
      w_k ∝ exp((c_k – max(c)) / τ)
  with a temperature τ that decays during optimisation.
  The gradient of the weighted sum ``Σ w_k c_k`` is obtained analytically
  as a cross‑correlation of ``a`` with the reversed weight vector;
  it is also computed via FFT (O(n log n)).
* A (optional) momentum term and an adaptive step size are applied.
* The procedure is run many times with random lengths and random
  power‑law initialisations (optionally with an exponential tail).  The
  best sequence found over the whole time budget is returned.

The implementation respects the 1000 s budget, prints progress, and
uses only NumPy and the provided ``evaluate_sequence`` function.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Helper – Euclidean projection onto the simplex {x_i ≥ 0, Σ x_i = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex of radius s (sum = s, non‑negative)."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper – Full linear convolution using real FFT (fast for any length)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of real vectors x and y via FFT."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power‑of‑2
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    # remove tiny negative artefacts caused by rounding
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Optimisation of a single length using a smooth‑max (soft‑max) gradient
# ----------------------------------------------------------------------
def _optimise_one_length_softmax(
    n: int,
    init: np.ndarray,
    run_deadline: float,
    *,
    init_tau_factor: float = 0.10,
    min_tau_factor: float = 0.001,
    step_factor: float = 0.6,
    step_decay: float = 0.9995,
    momentum_beta: float = 0.9,
) -> tuple[np.ndarray, float]:
    """
    Perform soft‑max sub‑gradient descent for a fixed length `n`.
    Returns the best sequence found (as a NumPy array) and its
    evaluation score obtained from ``evaluate_sequence``.
    """
    target_sum = np.sqrt(2.0 * n)          # Σ a_i = √(2 n)
    a = _project_onto_simplex(init.astype(np.float64), target_sum)

    best_seq = a.copy()
    # Because we keep the sum fixed, the objective equals the maximal
    # convolution value.
    conv = _full_convolution_fft(a, a)
    best_score = 2.0 * n * conv.max() / (target_sum ** 2)

    velocity = np.zeros_like(a)           # momentum term
    iteration = 0

    while time.time() < run_deadline:
        # --------------------------------------------------------------
        # 1) self‑convolution and maximum value
        # --------------------------------------------------------------
        conv = _full_convolution_fft(a, a)
        max_val = conv.max()

        # --------------------------------------------------------------
        # 2) smooth‑max weights (soft‑max) – temperature τ decays
        # --------------------------------------------------------------
        tau = max(init_tau_factor * max_val * (0.95 ** iteration),
                  min_tau_factor * max_val)
        # shift for numerical stability, then exponentiate
        scaled = (conv - max_val) / tau
        w = np.exp(scaled)                 # all entries ≤ 1
        w_sum = w.sum()
        if w_sum == 0.0:                   # fallback – uniform weights
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # --------------------------------------------------------------
        # 3) gradient of the weighted sum Σ w_k conv_k
        #    grad_i = 2 * Σ_j a_j w_{i+j}
        #    This is a cross‑correlation – computed via FFT.
        # --------------------------------------------------------------
        w_rev = w[::-1]
        conv2 = _full_convolution_fft(a, w_rev)   # length = 3n‑2
        idx = (2 * n - 2) - np.arange(n)           # positions giving Σ_j a_j w_{i+j}
        grad = 2.0 * conv2[idx]

        # --------------------------------------------------------------
        # 4) momentum and adaptive step size
        # --------------------------------------------------------------
        if momentum_beta > 0.0:
            velocity = momentum_beta * velocity + (1.0 - momentum_beta) * grad
            step_dir = velocity
        else:
            step_dir = grad

        # Scale step so that we move roughly a fraction of the current max
        max_grad = np.max(np.abs(step_dir)) + 1e-12
        eta = step_factor * max_val / max_grad

        a_new = a - eta * step_dir
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_onto_simplex(a_new, target_sum)

        # --------------------------------------------------------------
        # 5) evaluate and possibly accept as new best
        # --------------------------------------------------------------
        conv_new = _full_convolution_fft(a_new, a_new)
        score_new = 2.0 * n * conv_new.max() / (target_sum ** 2)

        if score_new < best_score:
            best_score = score_new
            best_seq = a_new.copy()
            print(f"[{time.time():.2f}] n={n}  new local best = {score_new:.6f}")

        a = a_new
        step_factor *= step_decay
        iteration += 1

        # occasional random shake‑up to escape plateaus
        if np.random.rand() < 0.01:
            noise = np.random.randn(n) * (target_sum * 0.0005 / n)
            a = _project_onto_simplex(np.maximum(a + noise, 0.0), target_sum)
            # after a shake‑up we become a little more cautious
            step_factor *= 0.9

    # Final evaluation via the provided evaluator (ensures identical rounding)
    final_score = evaluate_sequence(best_seq.tolist())
    return best_seq, final_score

def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative height sequence that minimises the given
    evaluation function.  The routine runs for at most ``budget_s`` seconds
    and returns the best sequence discovered.

    Strategy
    --------
    * Randomly pick a length `n` (≈ 800 … 3500) and initialise a
      power‑law shape a_i ∝ (i+1)^{-p} (p sampled around 0.5).  Optionally
      apply a mild exponential taper.
    * Run the soft‑max gradient descent defined in ``_optimise_one_length_softmax``.
    * Keep the overall best across many independent restarts.
    * Occasionally warm‑start from the current global best (truncating or
      padding as needed).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # tiny safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Initialise from a previously known construction if available.
    # ------------------------------------------------------------------
    try:
        # ``height_sequence_1`` may be provided in the global namespace.
        hs = globals()["height_sequence_1"]
        hs_arr = np.array(hs, dtype=float)
        hs_arr = np.clip(hs_arr, 0.0, 1000.0)
        score = evaluate_sequence(hs_arr.tolist())
        if score < best_score:
            best_score = score
            best_seq = hs_arr.copy()
            print(f"[{time.time():.2f}] Warm‑start from global seed = {score:.6f}")
    except Exception:
        pass  # no warm‑start available

    # ------------------------------------------------------------------
    # Main restart loop – each restart gets a time slice.
    # ------------------------------------------------------------------
    while time.time() < deadline:
        remaining = deadline - time.time()

        # Choose a problem size.  Early on we explore a wide range,
        # later we focus on moderately sized problems to finish quickly.
        if remaining > 300:
            n = np.random.randint(800, 3500 + 1)
        elif remaining > 120:
            n = np.random.randint(600, 3000 + 1)
        else:
            n = np.random.randint(400, 2200 + 1)

        # ------------------- initial shape -----------------------------
        p = np.random.uniform(0.3, 0.7)               # power‑law exponent
        idx = np.arange(1, n + 1, dtype=float)
        init = idx ** (-p)

        # optional exponential taper (adds extra flexibility)
        if np.random.rand() < 0.35:
            tau_exp = np.random.uniform(0.4, 2.0) * n
            init *= np.exp(-idx / tau_exp)

        init += np.random.rand(n) * 1e-4   # tiny jitter to break symmetry

        # occasional warm‑start from the current global best
        if best_seq is not None and np.random.rand() < 0.45:
            bs = best_seq
            if bs.size >= n:
                init = bs[:n].copy()
            else:
                pad = np.random.rand(n - bs.size) * 1e-4
                init = np.concatenate([bs, pad])

        # allocate a time slice (≈ 8 s, but never exceed the global deadline)
        run_deadline = min(deadline, time.time() + min(8.0, remaining * 0.5))

        # run the optimiser for this length
        seq_local, score_local = _optimise_one_length_softmax(
            n,
            init,
            run_deadline,
            init_tau_factor=0.12,
            min_tau_factor=0.001,
            step_factor=0.6,
            step_decay=0.9994,
            momentum_beta=0.9,
        )

        # update the global best if improvement was found
        if score_local < best_score:
            best_score = score_local
            best_seq = seq_local.copy()
            print(
                f"[{time.time():.2f}] GLOBAL new best = {best_score:.6f} (length={n})"
            )
            # early exit if we comfortably beat the target bound
            if best_score < 1.49:
                break

    # ------------------------------------------------------------------
    # Fallback in the unlikely event that no candidate was generated.
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        target_sum = np.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)

    # The evaluator expects a plain Python list.
    return best_seq.tolist()
