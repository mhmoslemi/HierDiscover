"""
Propose a step‑function sequence with a low evaluation score.

The algorithm consists of three stages:

1. **Diverse initialisation** – a short scan of geometric progressions and
   random Dirichlet vectors on a modest length (≈2000).  The best of these
   candidates is kept as a starting point.

2. **Projected Adam optimisation** – a smooth‑max surrogate
   ``φ_λ(a) = (1/λ)·log Σ exp(λ·(a*a)_k)`` is minimised by Adam with a
   projection onto the probability simplex after each update.
   The smoothing parameter λ is increased gradually (continuation) so that
   early iterations move freely while later ones concentrate on the true
   `max` term.

3. **Simulated‑annealing fine‑tuning** – starting from the best Adam
   solution we perform many cheap random perturbations that preserve the
   simplex constraint.  Moves are accepted if they improve the exact
   objective or with a temperature‑controlled probability, allowing us to
   escape shallow local minima.

All heavy work (convolution, gradient) uses FFT‑based O(n log n)
implementations, so the routine comfortably fits into the 1000 s budget.
The final sequence is returned as a plain Python list and satisfies all
security checks of `evaluate_sequence`.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Projection onto the simplex {x ≥ 0, sum(x)=1}
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        # all entries non‑positive – return uniform distribution
        return np.full_like(v, 1.0 / len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full linear convolution via FFT (power‑of‑two length is used)
def _full_convolution(a: np.ndarray, size: int | None = None) -> np.ndarray:
    """Return the linear self‑convolution a*a (length 2·len(a)‑1)."""
    n = len(a)
    if size is None:
        size = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Exact objective used by `evaluate_sequence`
def _objective(a: np.ndarray) -> float:
    n = len(a)
    conv = _full_convolution(a)
    max_b = conv.max()
    sum_a = a.sum()
    # guard against division by zero – caller always ensures sum_a>0
    return 2.0 * n * max_b / (sum_a * sum_a)


# ----------------------------------------------------------------------
# Gradient of the smooth‑max surrogate (log‑sum‑exp of the convolution)
def _softmax_gradient(a: np.ndarray, conv: np.ndarray, lam: float,
                      size_corr: int | None = None) -> np.ndarray:
    """
    Gradient of (1/λ)·log Σ exp(λ·conv_k) w.r.t. a.
    It equals 2·(w ★ a) where w_k = softmax(λ·conv_k).
    """
    n = len(a)
    # soft‑max weights (numerically stable)
    m = conv.max()
    w = np.exp(lam * (conv - m))
    w /= w.sum()

    # linear cross‑correlation w ★ a
    rev_a = a[::-1]
    if size_corr is None:
        size_corr = 1 << int(np.ceil(np.log2(len(w) + n - 1)))
    fw = np.fft.rfft(w, size_corr)
    frev = np.fft.rfft(rev_a, size_corr)
    corr = np.fft.irfft(fw * frev, size_corr)

    # slice the part corresponding to indices 0 … n‑1 of the original vector
    grad = 2.0 * corr[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of heights) with a small
    evaluation score.  The routine respects the 1000 s time budget, prints
    progress, and finally returns the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    # --------------------------------------------------------------
    # 1. Choose a working length.
    #    2000 is fast enough for many iterations and already yields good
    #    candidates; later we optionally upscale.
    n0 = 2000
    best_seq = None
    best_val = float("inf")

    # --------------------------------------------------------------
    # 2. Initialise with a few geometric progressions and Dirichlet draws.
    print("[Init] scanning simple families (n =", n0, ")")
    geometric_factors = np.linspace(0.8, 3.0, 30)
    for fac in geometric_factors:
        if time.time() > deadline:
            break
        eps = fac / n0
        r = 1.0 - eps
        if r <= 0.0:
            continue
        powers = np.power(r, np.arange(n0, dtype=np.float64))
        seq = powers / powers.sum()
        val = _objective(seq)
        if val < best_val:
            best_val, best_seq = val, seq.copy()
            print(f"[Init geom] fac={fac:.3f} eps={eps:.2e} obj={val:.6f}")

    # random Dirichlet candidates
    for _ in range(20):
        if time.time() > deadline:
            break
        seq = np.random.dirichlet(np.ones(n0))
        val = _objective(seq)
        if val < best_val:
            best_val, best_seq = val, seq.copy()
            print(f"[Init dir] obj={val:.6f}")

    # --------------------------------------------------------------
    # 3. Projected Adam optimisation on the smooth‑max surrogate.
    #    This is the main refinement stage.
    def _adam_opt(seq: np.ndarray, time_budget: float):
        """Run Adam on the surrogate for the given time budget."""
        nonlocal best_seq, best_val
        n = len(seq)
        # hyper‑parameters
        lam = 0.5                     # start with a fairly smooth approximation
        lam_factor = 1.02
        lr = 0.5                      # base learning rate
        lr_factor = 0.9995
        beta1, beta2 = 0.9, 0.999
        eps = 1e-8

        # Adam state
        m = np.zeros_like(seq)
        v = np.zeros_like(seq)
        t = 0
        # FFT size for the cross‑correlation used in the gradient
        size_corr = 1 << int(np.ceil(np.log2(3 * n)))

        stop_time = time.time() + time_budget
        last_improve = time.time()
        it = 0
        while time.time() < stop_time:
            it += 1
            conv = _full_convolution(seq)
            grad = _softmax_gradient(seq, conv, lam, size_corr=size_corr)

            t += 1
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)

            new_seq = seq - step
            new_seq = _project_onto_simplex(new_seq)

            val = _objective(new_seq)
            if val < best_val:
                best_val, best_seq = val, new_seq.copy()
                last_improve = time.time()
                print(f"[Adam it {it:05d}] new best obj={val:.6f}")

            # simple restart if stuck for a while
            if time.time() - last_improve > 3.0:
                noise = (np.random.rand(n) - 0.5) * 1e-3
                new_seq = _project_onto_simplex(new_seq + noise)
                last_improve = time.time()

            seq = new_seq
            lam *= lam_factor
            lr *= lr_factor

        return seq

    # allocate roughly half of the remaining budget to the Adam stage
    remaining = deadline - time.time()
    adam_budget = remaining * 0.45
    if adam_budget > 5.0:            # need at least a few seconds
        print("[Phase] Adam optimisation (budget {:.1f}s)".format(adam_budget))
        best_seq = _adam_opt(best_seq.copy(), adam_budget)

    # --------------------------------------------------------------
    # 4. Optional up‑scaling to a larger length (more expressive power).
    #    If enough time is left we interpolate the current best solution
    #    to a longer vector and run a second Adam pass.
    remaining = deadline - time.time()
    if remaining > 250.0:            # enough time to merit the extra work
        n_big = 4000
        print("[Upscale] interpolating to n =", n_big)
        # linear interpolation preserving the mass (the simplex property)
        x_small = np.linspace(0.0, 1.0, len(best_seq))
        x_big = np.linspace(0.0, 1.0, n_big)
        seq_big = np.interp(x_big, x_small, best_seq)
        seq_big = _project_onto_simplex(seq_big)

        # second Adam run on the larger problem
        remaining = deadline - time.time()
        adam_budget2 = remaining * 0.45
        if adam_budget2 > 5.0:
            print("[Phase] Adam on up‑scaled problem (budget {:.1f}s)".format(adam_budget2))
            best_seq = _adam_opt(seq_big, adam_budget2)

    # --------------------------------------------------------------
    # 5. Simulated‑annealing fine‑tuning.
    #    Random perturbations respecting the simplex constraint are proposed.
    n = len(best_seq)
    current_seq = best_seq.copy()
    current_val = _objective(current_seq)

    # annealing schedule
    T = 0.05                      # initial temperature
    T_factor = 0.9998
    eps = 0.05                    # maximal amplitude of the raw noise
    eps_factor = 0.9995

    print("[Phase] Simulated annealing")
    while time.time() < deadline:
        # propose a random move (small white‑noise perturbation)
        noise = (np.random.rand(n) - 0.5) * eps
        candidate = _project_onto_simplex(current_seq + noise)
        cand_val = _objective(candidate)

        # Metropolis acceptance
        accept = False
        if cand_val < current_val:
            accept = True
        else:
            prob = np.exp(-(cand_val - current_val) / max(T, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            current_seq, current_val = candidate, cand_val
            if cand_val < best_val:
                best_val, best_seq = cand_val, candidate.copy()
                print(f"[Anneal] new best obj={cand_val:.6f}")

        # cool down
        T *= T_factor
        eps *= eps_factor

    print(f"Finished. Best objective = {best_val:.6f} (length {len(best_seq)})")
    # evaluator expects a plain Python list
    return best_seq.tolist()
