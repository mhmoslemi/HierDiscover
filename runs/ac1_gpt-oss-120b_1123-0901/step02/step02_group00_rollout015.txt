def _conv_full_fft(a):
    n = a.shape[0]
    N = 1 << ((2*n-1).bit_length())
    A = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(A*A, n=N)[:2*n-1]
    return conv

def _softmax_weights(conv, tau):
    # stable softmax
    maxc = np.max(conv)
    if tau <= 0:
        # fallback to argmax
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w, maxc
    exp = np.exp((conv - maxc)/tau)
    sumexp = np.sum(exp)
    w = exp / sumexp
    # compute Ltau = maxc + tau * log(sumexp)
    Ltau = maxc + tau * np.log(sumexp)
    return w, Ltau
```

def _weighted_rev(a, w):
    n = a.shape[0]
    # compute conv of reversed w with a using FFT
    N = 1 << ((len(w) + n - 1).bit_length())
    W_rev = np.fft.rfft(w[::-1], n=N)
    A = np.fft.rfft(a, n=N)
    conv_rev = np.fft.irfft(W_rev * A, n=N)
    conv_rev = conv_rev[:len(w)+n-1]
    idx_start = 2*n-2
    rev = conv_rev[idx_start - np.arange(n)]
    return rev
```

def _optimize_length_smt(n, init_a, best_seq, best_score, deadline):
    target_sum = math.sqrt(2.0*n)
    a = init_a.copy()
    a = a * (target_sum / a.sum())
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1=0.9; beta2=0.999; eps=1e-8
    lr0=0.05
    iteration=0
    jitter_interval=5000
    # initial temperature scale: start with tau_factor * current max (maybe 0.2)
    tau_factor = 0.2
    # tau decay rate
    decay_rate = 20000.0
    while time.time()<deadline:
        conv = _conv_full_fft(a)
        M = conv.max()
        # temperature
        tau = max(1e-8, tau_factor * M * np.exp(-iteration/decay_rate))
        w, Ltau = _softmax_weights(conv, tau)
        rev = _weighted_rev(a, w)

        coeff = 4.0 * n / (target_sum * target_sum)
        const = 4.0 * n * Ltau / (target_sum ** 3)
        grad = coeff * rev - const

        iteration+=1
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad*grad)
        m_hat = m/(1-beta1**iteration)
        v_hat = v/(1-beta2**iteration)
        lr = lr0/(1+1e-4*iteration)
        a = a - lr* m_hat/(np.sqrt(v_hat)+eps)

        a = np.maximum(a, 0.0)
        s = a.sum()
        if s>0:
            a = a * (target_sum/s)
        else:
            a = np.full_like(a, target_sum/n)

        if iteration%jitter_interval==0:
            jitter = np.random.randn(n) * 1e-3 * target_sum / n
            a = np.maximum(a+jitter,0.0)
            a = a * (target_sum / a.sum())

        # Evaluate actual score:
        cur_score = evaluate_sequence(a.tolist())
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[len {n} iter {iteration}] new best = {best_score:.6f}")

    return best_seq, best_score

"""
Propose a step‑function (non‑negative vector) that minimises the
``evaluate_sequence`` objective.

Algorithm
---------
* Initialise from the known construction ``height_sequence_1`` (if any) and
  normalise it to Σa = √(2 n).
* Search several lengths (original, 1.25×, 1.5× and 2×).  For each length
  run a *soft‑max Adam* optimiser:
    – The maximal convolution value M = max (a∗a) is replaced by the
      smooth approximation Lτ = τ·log ∑exp((a∗a)/τ) (temperature τ).
    – The gradient of Lτ w.r.t. a is computed analytically via a single
      FFT (convolution of the soft‑max weights with a).
    – Adam updates are projected onto the feasible set (a≥0,
      Σa = √(2 n)).
    – Temperature τ is annealed (starts at 20 % of the current M and
      decays exponentially) so that the optimisation first sees a
      smooth landscape and then refines the exact max.
    – Occasional tiny Gaussian jitter helps to escape shallow plateaus.
* The best sequence over all runs (measured by the exact objective
  2 n·max(a∗a)/Σa² = max(a∗a) after normalisation) is returned.
* The routine respects a 1000 s wall‑clock limit.

The implementation uses only ``numpy`` and the already‑imported
``evaluate_sequence`` function; all heavy work is done with FFTs for
speed.
"""

import time
import math
import numpy as np


def _softmax_weights(conv: np.ndarray, tau: float):
    """
    Stable soft‑max of the convolution values.
    Returns the weight vector w (summing to 1) and the smoothed maximum
    Lτ = τ·log ∑exp((conv)/τ).
    """
    maxc = conv.max()
    if tau <= 0.0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w, maxc
    # subtract the maximum for numerical stability
    ex = np.exp((conv - maxc) / tau)
    sum_ex = ex.sum()
    w = ex / sum_ex
    Ltau = maxc + tau * np.log(sum_ex)
    return w, Ltau


def _weighted_rev(a: np.ndarray, w: np.ndarray, Nfft: int):
    """
    Compute the vector
        rev_i = Σ_k w_k a_{k‑i}
    (indices outside 0…len(a)-1 are ignored) using FFTs.
    """
    n = a.shape[0]
    # FFT of a (already zero‑padded to length Nfft)
    A_fft = np.fft.rfft(a, n=Nfft)
    # FFT of the reversed weight vector
    W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)
    # Convolution of w_rev with a
    conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
    conv_rev = conv_rev[: len(w) + n - 1]          # keep the valid part
    idx_start = 2 * n - 2
    rev = conv_rev[idx_start - np.arange(n)]      # pick the needed slice
    return rev


def _optimize_length(
    n: int,
    init_a: np.ndarray,
    best_seq: np.ndarray,
    best_score: float,
    deadline: float,
    max_iters: int = 10_000_000,
):
    """
    Adam optimiser that minimises the *smooth* maximal self‑convolution.
    The exact objective value (max (a∗a)) is used to update the global best.
    """
    target_sum = math.sqrt(2.0 * n)

    # start from the supplied initial vector and enforce the normalisation
    a = init_a.copy()
    a = a * (target_sum / a.sum())

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr0 = 0.05
    iteration = 0
    jitter_interval = 5000

    # parameters for the annealed temperature
    tau_factor = 0.20               # initial τ = τ_factor·M
    decay_rate = 20000.0           # controls exponential decay of τ

    # FFT size (next power‑of‑two ≥ 2·n‑1, re‑used for both convolutions)
    Nfft = 1 << ((2 * n - 1).bit_length())

    while time.time() < deadline and iteration < max_iters:
        # --------------------------------------------------------------
        # 1) exact convolution a∗a (full) via FFT
        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        M = float(conv.max())

        # --------------------------------------------------------------
        # 2) soft‑max smoothing of the max
        tau = max(1e-12, tau_factor * M * math.exp(-iteration / decay_rate))
        w, Ltau = _softmax_weights(conv, tau)

        # --------------------------------------------------------------
        # 3) weighted reverse convolution (sub‑gradient of the smooth max)
        rev = _weighted_rev(a, w, Nfft)

        # --------------------------------------------------------------
        # 4) Adam gradient step (including the normalisation term)
        coeff = 4.0 * n / (target_sum * target_sum)
        const = 4.0 * n * Ltau / (target_sum ** 3)
        grad = coeff * rev - const

        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        lr = lr0 / (1.0 + 1e-4 * iteration)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # --------------------------------------------------------------
        # 5) projection onto the feasible set (a≥0, Σa = √(2 n))
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a = a * (target_sum / s)
        else:
            a = np.full_like(a, target_sum / n)

        # --------------------------------------------------------------
        # 6) occasional jitter to escape plateaus
        if iteration % jitter_interval == 0:
            jitter = np.random.randn(n) * (1e-3 * target_sum / n)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        # --------------------------------------------------------------
        # 7) exact objective value (after normalisation it equals M)
        cur_score = M
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[len {n:5d} iter {iteration:8d}] new best = {best_score:.6f}")

    return best_seq, best_score
