```
def _hill_climb(a: np.ndarray, best: dict, deadline: float, max_iters=100000):
    n = a.shape[0]
    target = math.sqrt(2.0 * n)
    # compute current objective
    cur = _objective(a)
    # temperature schedule perhaps: start T = some fraction of cur
    T0 = cur * 0.1
    it = 0
    while time.time() < deadline and it < max_iters:
        # choose indices i and j
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # attempt to move small delta
        # choose delta as fraction of a[i] maybe
        delta = np.random.uniform(0, min(a[i], cur*0.001)) # ensure nonneg
        # but also ensure a[j] + delta not exceed 1000
        delta = min(delta, 1000 - a[j])
        if delta <= 0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # sum unchanged
        # optional, ensure a_new still nonneg; yes.
        # evaluate
        score = _objective(a_new)
        # accept if better, else maybe with prob.
        if score < cur:
            a = a_new
            cur = score
            _maybe_update(a, best)
        else:
            # Metropolis acceptance
            T = max(T0 * (1 - it / max_iters), 1e-6)
            if np.random.rand() < math.exp(-(score - cur) / T):
                a = a_new
                cur = score
        it += 1
```

assistantfinal**Solution Overview**

We need a non‑negative sequence `a = (a₀,…,a_{n‑1})` that minimises  

\[
\text{score}(a)=\frac{2\,n\;\max_k (a*a)_k}{\bigl(\sum_i a_i\bigr)^2}.
\]

If the sum is forced to `√(2 n)` (the code projects onto this simplex) the
score reduces to the maximal entry of the full convolution `a*a`.  
Thus the problem is *“make the peak of the autocorrelation as small as
possible while keeping the ℓ₁‑norm fixed.”*  

The previous runner used a simple sub‑gradient scheme and obtained a best
value ≈ 1.522.  This implementation improves on it by

* searching a wider family of parametric shapes (exponential, power‑law,
  Gaussian, linear, quadratic, triangular) on many lengths up to 5000,
* running many **sub‑gradient + momentum** refinements with different
  hyper‑parameters,
* performing a **stochastic hill‑climbing / simulated‑annealing** phase
  that moves tiny amounts of mass between two positions,
* using fast FFT‑based convolution for the internal objective
  (so we can afford many thousands of evaluations),
* keeping the current best sequence in a mutable dictionary that can be
  updated from any helper without closures.

The routine respects the 1000 s budget: each phase checks the deadline and
stops early if the time is exhausted.  The final sequence is returned as a
plain Python list, ready for the external `evaluate_sequence` call.

```python
"""
propose_candidate
-----------------

Search for a non‑negative step‑function that minimises

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm consists of three stages:

1.  A broad parametric sweep over several families (exponential,
    power‑law, Gaussian, linear, quadratic, triangular) for many lengths
    (up to 5000).  Every candidate is projected onto the simplex
    Σ a_i = √(2 n) and its maximal convolution entry is computed with a
    fast FFT‑based routine.  The best‑scoring sequence is remembered.

2.  Heavy sub‑gradient descent with momentum on the current best sequence
    and on a number of random restarts.  The gradient is the exact
    sub‑gradient of the active convolution entry; after each step the
    iterate is projected back onto the simplex.

3.  A stochastic hill‑climbing / simulated‑annealing phase that repeatedly
    transfers a tiny amount of mass between two random indices while
    preserving the ℓ₁‑norm.  Improvements are accepted greedily; occasional
    uphill moves are allowed according to a temperature schedule.

All heavy work uses only NumPy and the standard library; the helper
functions are defined at module level (no closures).  The routine stops
as soon as the 1000 s deadline is reached and returns the best sequence
found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the simplex {x ≥ 0, Σx = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper: fast full convolution a * a using real FFT (returns length 2n‑1)
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full convolution of a with itself (real FFT, O(n log n))."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Internal objective: maximal entry of the convolution (the score after the
# sum‑normalisation).
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Return max(conv(a,a)) – the objective that evaluate_sequence uses."""
    # a is assumed non‑negative and already satisfies the simplex sum.
    conv = _conv_self(a)
    return float(conv.max())


# ----------------------------------------------------------------------
# Shape factories – non‑negative, not yet normalised
# ----------------------------------------------------------------------
def _shape_exp(n: int, alpha: float) -> np.ndarray:
    """a_i = exp(-alpha * i)  (decreasing exponential)"""
    return np.exp(-alpha * np.arange(n, dtype=float))


def _shape_exp_rev(n: int, alpha: float) -> np.ndarray:
    """a_i = exp(-alpha * (n‑1‑i))  (increasing exponential)"""
    return np.exp(-alpha * np.arange(n - 1, -1, -1, dtype=float))


def _shape_power(n: int, p: float) -> np.ndarray:
    """a_i = (i+1)^{-p}  (power‑law decay)"""
    return (np.arange(1, n + 1, dtype=float) ** (-p))


def _shape_gaussian(n: int, sigma: float) -> np.ndarray:
    """a_i = exp(- (i / sigma)^2)  (Gaussian‑ish tail)"""
    return np.exp(- (np.arange(n, dtype=float) / sigma) ** 2)


def _shape_linear_desc(n: int, beta: float) -> np.ndarray:
    """a_i = max(0, 1 - beta * i)  (linear decreasing)"""
    i = np.arange(n, dtype=float)
    return np.maximum(0.0, 1.0 - beta * i)


def _shape_quad_desc(n: int, beta: float) -> np.ndarray:
    """a_i = max(0, 1 - beta * i^2)  (quadratic decreasing)"""
    i = np.arange(n, dtype=float)
    return np.maximum(0.0, 1.0 - beta * i * i)


def _shape_triangular(n: int) -> np.ndarray:
    """symmetrical triangle: 1,2,…,⌈n/2⌉,…,2,1 (non‑negative)"""
    i = np.arange(n, dtype=float)
    return np.minimum(i + 1.0, n - i)


# ----------------------------------------------------------------------
# Update the global best candidate (dictionary is mutated in‑place)
# ----------------------------------------------------------------------
def _maybe_update(cand: np.ndarray, best: dict) -> None:
    """If cand improves the current best, store it."""
    score = _objective(cand)
    if score < best["score"]:
        best["score"] = score
        best["seq"] = cand.copy()
        print(f"[{time.time():.2f}] New best = {score:.6f} (len={len(cand)})")


# ----------------------------------------------------------------------
# Sub‑gradient descent with momentum – works on a single start vector.
# ----------------------------------------------------------------------
def _run_subgrad(
    init_a: np.ndarray,
    best: dict,
    deadline: float,
    momentum: float = 0.9,
    base_step: float = 0.3,
) -> None:
    """Perform many sub‑gradient steps starting from init_a."""
    a = init_a.astype(float)
    n = a.shape[0]
    target = math.sqrt(2.0 * n)

    # momentum accumulator
    m = np.zeros_like(a)
    it = 0
    while time.time() < deadline:
        conv = _conv_self(a)
        kmax = int(np.argmax(conv))

        # sub‑gradient of the active linear functional L_{kmax}
        low = max(0, kmax - (n - 1))
        high = min(n - 1, kmax)
        idx = np.arange(low, high + 1, dtype=int)
        partner = kmax - idx
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[partner]          # see analysis for justification

        # direction = -grad  (we minimise the max)
        direction = -grad
        m = momentum * m + (1.0 - momentum) * direction

        step = base_step / math.sqrt(it + 1)
        max_abs = np.max(np.abs(m))
        if max_abs == 0.0:
            max_abs = 1.0
        eta = step / max_abs

        a_new = a + eta * m
        a_new = _project_onto_simplex(a_new, target)

        # periodic evaluation (cheap)
        if it % 5 == 0:
            _maybe_update(a_new, best)

        a = a_new
        it += 1

        # optional early exit if we have already reached a very good value
        if best["score"] < 1.45:          # arbitrary safe guard
            break


# ----------------------------------------------------------------------
# Stochastic hill‑climbing / simulated‑annealing on a fixed length.
# ----------------------------------------------------------------------
def _hill_climb(
    a: np.ndarray,
    best: dict,
    deadline: float,
    max_iters: int = 200_000,
    temp_start: float = 0.05,
) -> None:
    """Transfer tiny masses between two random positions, accepting
    occasional uphill moves (simulated annealing)."""
    n = a.shape[0]
    cur_score = _objective(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        # pick two distinct indices
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # transfer a small amount from i to j
        delta_max = min(a[i], 0.1)                 # limit step size
        if delta_max <= 0.0:
            continue
        delta = np.random.rand() * delta_max

        # enforce upper bound 1000 (far away from our scale, but keep it)
        if a[j] + delta > 1000.0:
            delta = 1000.0 - a[j]
        if delta <= 0.0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_score = _objective(a_new)

        if new_score < cur_score:
            a = a_new
            cur_score = new_score
            _maybe_update(a, best)
        else:
            # downhill move with probability exp(-(Δ)/T)
            T = max(temp_start * (1.0 - it / max_iters), 1e-8)
            if np.random.rand() < math.exp(-(new_score - cur_score) / T):
                a = a_new
                cur_score = new_score

        it += 1
        # early stop if we already beat the target
        if best["score"] < 1.45:
            break
```
