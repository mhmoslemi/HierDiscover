"""
Propose a step‑function (non‑negative sequence) that minimizes the
`evaluate_sequence` objective.

Algorithm
---------
We treat the problem as a sequence f (∑f_i = 1, f_i ≥ 0) and try to make the
self‑convolution `c = f * f` as flat as possible, because the objective is

    obj(f) = 2·n·max(c)                (n = len(f), sum(f)=1)

The classic “alternating LP” approach works well:

1. Start from a smooth distribution (arcsine law) that already gives a
   low value.
2. For the current f compute M = max(f*f).
3. Solve a linear program
       maximise   Σ g_i
       subject to g_i ≥ 0
                  (f * g)(k) ≤ M   for every k = 0 … 2n‑2
   The convolution (f * g) is linear in g, so the constraints are linear.
   A feasible point is g = f, therefore the LP always has a solution with
   Σ g_i ≥ 1.
4. Scale the LP solution to unit sum (the objective is scale‑invariant) and
   form a convex combination f ← (1‑t)·f + t·g  (t slowly decays).
5. Repeat while time permits.  The maximum convolution value never grows,
   and in practice it decreases, giving a better upper bound.
6. If the LP solver is unavailable we fall back to the sub‑gradient method
   from the previous baseline.

The implementation keeps everything in plain NumPy, uses `scipy.optimize.linprog`
with the efficient HiGHS backend, and respects the 1000‑second budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v):
    """Project a vector onto the probability simplex {x ≥ 0, Σx = 1}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _build_conv_matrix(f):
    """
    Build the matrix A (shape (2n‑1, n)) such that (f * g) = A @ g.
    Row k contains the coefficients f_i that multiply g_{k‑i}.
    """
    n = len(f)
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = f[i]
    return A


# ----------------------------------------------------------------------
def _solve_lp(f, M):
    """
    Solve  max Σ g_i   s.t.   (f * g) ≤ M,  g ≥ 0.
    Returns the optimal g (as a dense numpy array) or None on failure.
    """
    try:
        from scipy.optimize import linprog
    except Exception:          # SciPy not available
        return None

    n = len(f)
    rows = 2 * n - 1
    # Build the convolution matrix once per call
    A = _build_conv_matrix(f)
    b = np.full(rows, M, dtype=np.float64)

    # maximise Σ g_i  → minimise -Σ g_i
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0, None)] * n

    # HiGHS is very fast for dense problems of this size
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method='highs', options={'presolve': True, 'disp': False})
    if res.success:
        return res.x
    return None


# ----------------------------------------------------------------------
def _initial_arcsine(n):
    """Arcsine‑law initial guess (already normalised to sum = 1)."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimises
    `evaluate_sequence` within the given time budget.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5    # safety margin

    # --------------------------------------------------------------
    # 1. problem size – 1500 is a good compromise (memory & quality)
    n = 1500

    # 2. start from a smooth arcsine distribution
    f = _initial_arcsine(n).astype(np.float64)
    best_seq = f.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[Init] n={n}, initial objective = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3. main optimisation loop (alternating LP)
    t = 0.30                # initial mixing weight
    decay = 0.985           # slowly reduce the step size
    it = 0
    stagnation = 0

    while time.time() < deadline:
        # current convolution and its peak
        conv = np.convolve(f, f)
        M = conv.max()

        # Linear programme to enlarge the feasible mass without raising M
        g0 = _solve_lp(f, M)
        if g0 is None:
            # fallback to simple sub‑gradient refinement (very cheap)
            # – this path is only taken if SciPy is missing.
            # The code below mirrors the previous baseline method.
            # ------------------------------------------------------
            # Compute a rough sub‑gradient using the K largest conv entries
            K = 5
            top_idx = np.argpartition(conv, -K)[-K:]
            grad = np.zeros_like(f)
            for k in top_idx:
                lo = max(0, k - (n - 1))
                hi = min(n - 1, k)
                idx = np.arange(lo, hi + 1)
                grad[idx] += f[k - idx]
            grad /= K
            # heavy‑ball update
            beta = 0.9
            if it == 0:
                momentum = np.zeros_like(f)
            else:
                momentum = beta * momentum + (1.0 - beta) * grad
            step = 0.15 * (0.9999 ** it)
            f_new = f - step * momentum
            f_new = _project_onto_simplex(f_new)
        else:
            # normalise the LP solution to unit sum (objective is scale‑invariant)
            g = g0 / g0.sum()
            # convex combination with the current f
            f_new = (1.0 - t) * f + t * g
            f_new = _project_onto_simplex(f_new)

        # evaluate the candidate
        score = evaluate_sequence(f_new.tolist())
        if score < best_score:
            best_score = score
            best_seq = f_new.copy()
            print(f"[Iter {it:04d}] new best = {best_score:.6f}")
            stagnation = 0
        else:
            stagnation += 1

        # decay the mixing weight – we want big steps early, fine steps later
        t = max(0.02, t * decay)

        # occasional random jitter to escape possible plateaus
        if stagnation > 2500:
            jitter = rng.normal(scale=0.005, size=n)
            f_new = _project_onto_simplex(f_new + jitter)
            stagnation = 0
            print(f"[Iter {it:04d}] jitter applied")

        f = f_new
        it += 1

    print(f"Finished after {it} iterations. Best objective = {best_score:.6f}")
    return best_seq.tolist()
