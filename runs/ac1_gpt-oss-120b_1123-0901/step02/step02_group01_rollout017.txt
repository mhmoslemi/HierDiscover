"""
Search method for step function sequences using Adam optimizer
with a smooth‐max surrogate of the max convolution. The method
runs several restarts on various lengths, projects onto the
probability simplex after each Adam update and gradually tightens
the surrogate (increasing λ). It keeps the best sequence found
within the allotted time budget.
"""

import time
import numpy as np

def _project_onto_simplex(v):
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all non‑positive, return uniform distribution
        return np.full_like(v, 1.0 / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def _full_convolution(a, size=None):
    """Linear self‑convolution a*a using FFT (output length 2·len(a)-1)."""
    n = a.shape[0]
    if size is None:
        size = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return conv

def _objective(a):
    """Exact objective used by evaluate_sequence."""
    n = a.shape[0]
    conv = _full_convolution(a)
    max_b = conv.max()
    sum_a = a.sum()
    # sum_a is guaranteed > 0 because a is a probability vector
    return 2.0 * n * max_b / (sum_a * sum_a)

def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimizes the evaluation
    function.  The routine runs several Adam‑based refinements with a
    smooth‑max surrogate, projects onto the simplex after each step,
    and returns the best sequence found before the time budget expires.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_val = float("inf")

    # Try a few different lengths – longer sequences give more flexibility
    # but also a larger factor n in the objective.  Experience shows that
    # 1800‑2500 is a good trade‑off.
    lengths = [1500, 2000, 2500]

    # Hyper‑parameters that are useful for a wide range of lengths
    lam_start = 0.5          # start of smooth‑max parameter
    lam_factor = 1.02        # geometric increase each iteration
    lr = 0.2                 # base learning rate for Adam
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    shake_interval = 2.0     # seconds without improvement -> shake
    noise_scale = 5e-4

    for n in lengths:
        if time.time() > deadline:
            break

        size_fft = 1 << int(np.ceil(np.log2(2 * n - 1)))
        # correlation size: len(w) + n - 1 = (2n-1) + n -1 = 3n-2
        size_corr = 1 << int(np.ceil(np.log2(3 * n)))

        # a few independent restarts for each length
        for restart in range(5):
            if time.time() > deadline:
                break

            # ----- 1. initialise sequence ------------------------------------
            if restart == 0:
                # geometric decay
                r = 0.999
                a = np.power(r, np.arange(n, dtype=np.float64))
            elif restart == 1:
                # Dirichlet (dense random)
                a = np.random.dirichlet(np.ones(n))
            elif restart == 2:
                # linear ramp from high to low
                a = np.linspace(0.2, 0.001, n)
            else:
                # completely random positive vector
                a = np.random.rand(n)

            a = np.maximum(a, 1e-12)
            a = a / a.sum()          # enforce simplex

            # Adam moments
            m = np.zeros_like(a)
            v = np.zeros_like(a)

            lam = lam_start
            t = 0
            last_improve = time.time()

            # ----- 2. optimisation loop ------------------------------------
            while time.time() < deadline:
                t += 1

                # (a) self‑convolution
                conv = _full_convolution(a, size=size_fft)

                # (b) smooth‑max weights w = softmax(λ·conv)
                maxc = conv.max()
                w = np.exp(lam * (conv - maxc))
                w_sum = w.sum()
                if w_sum == 0:
                    w = np.full_like(w, 1.0 / w.size)
                else:
                    w /= w_sum

                # (c) gradient = 2·(w ★ a)
                rev_a = a[::-1]
                fw = np.fft.rfft(w, size_corr)
                frev = np.fft.rfft(rev_a, size_corr)
                corr = np.fft.irfft(fw * frev, size_corr)
                grad = 2.0 * corr[n - 1 : n - 1 + n]

                # (d) Adam update
                m = beta1 * m + (1.0 - beta1) * grad
                v = beta2 * v + (1.0 - beta2) * (grad * grad)
                m_hat = m / (1.0 - beta1 ** t)
                v_hat = v / (1.0 - beta2 ** t)
                step = lr * m_hat / (np.sqrt(v_hat) + eps)

                a_new = a - step
                a_new = _project_onto_simplex(a_new)

                # (e) evaluate exact objective
                val = _objective(a_new)

                if val < best_val:
                    best_val = val
                    best_seq = a_new.copy()
                    last_improve = time.time()
                    print(f"[n={n}, rs={restart}, it={t:06d}] new best obj={val:.6f}")

                # (f) occasional random shake to escape flat regions
                if time.time() - last_improve > shake_interval:
                    noise = (np.random.rand(n) - 0.5) * noise_scale
                    a_new = _project_onto_simplex(a_new + noise)
                    # reset Adam moments because the direction changed abruptly
                    m.fill(0.0)
                    v.fill(0.0)
                    last_improve = time.time()

                # (g) prepare next iteration
                a = a_new
                lam *= lam_factor

                # early stop if we already beat the target
                if best_val < 1.503:
                    # we have a solution good enough – exit early
                    print("Target reached, exiting early.")
                    return best_seq.tolist()

    # Fallback: if nothing was found (should not happen) return a uniform vector
    if best_seq is None:
        best_seq = np.full(1, 1.0)   # a single element sequence is always valid
    print(f"Final best objective = {best_val:.6f}")
    return best_seq.tolist()

The algorithm

* works on the probability simplex (∑a_i = 1, a_i ≥ 0);
* starts from several diverse initialisations (geometric decay, Dirichlet,

    - takes an Adam step and projects back onto the simplex,
    - occasionally adds a tiny random “shake” when progress stalls;
