def solve_lp_max_sum(f, M):
    n = f.shape[0]
    # build A_ub matrix in sparse format
    i = np.arange(n)
    j = np.arange(n)
    row = np.repeat(i, n) + np.tile(j, n)   # shape n*n
    col = np.tile(j, n)
    data = np.repeat(f, n)  # f[i] repeated n times
    A = sp.coo_matrix((data, (row, col)), shape=(2*n-1, n)).tocsr()
    b = np.full(2*n-1, M, dtype=float)

    c = -np.ones(n, dtype=float)  # maximize sum(g)
    bounds = [(0, None)] * n

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        # fallback: return zeros
        return None
    return res.x
```

"""
Hybrid search for a low‑autocorrelation step function.

The algorithm works in three stages for a collection of sequence lengths.

1. **Adam optimisation** – minimise the maximal self‑convolution
   (the objective after the normalisation Σa = √(2n) is exactly M = max(a*a)).
   The gradient is the exact sub‑gradient of the max‑convolution.
   Adam’s adaptive learning‑rate and occasional Gaussian jitter help escape
   shallow plateaux.

2. **Linear‑programming refinement** – for the current vector `f` we solve
   a linear programme

        maximise   Σ g_j
        subject to  (f * g)_k ≤ M   (for all k)
                    g_j ≥ 0

   where `M = max(f*f)` is the current worst autocorrelation.
   The LP gives a “direction” `g` that can be mixed with `f` to obtain a
   new feasible point with a strictly smaller maximal autocorrelation.
   The LP is built as a sparse Toeplitz matrix and solved with SciPy’s
   HiGHS optimiser.  It is applied only to moderately sized vectors
   (n ≤ 2500) to keep memory usage reasonable.

3. **Fine‑tuning** – after the LP step a short second Adam run finishes
   the optimisation.

The whole search respects a wall‑clock budget (default 1000 s).  It tries
several lengths (including interpolations of the supplied `height_sequence_1`
and a few random lengths), keeps the best sequence found and returns it as a
plain Python list.
"""

import time
import math
import numpy as np
from scipy import sparse as sp
from scipy.optimize import linprog


# ----------------------------------------------------------------------
def _conv_max_fft(a: np.ndarray):
    """Full linear convolution a*a via FFT and return (max, full_conv)."""
    n = a.shape[0]
    N = 1 << ((2 * n - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(A * A, n=N)[: 2 * n - 1]
    M = float(conv.max())
    return M, conv


# ----------------------------------------------------------------------
def _subgradient(a: np.ndarray, active_idxs: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the max‑convolution.
    Returns the average reversed vector over all active shift indices.
    """
    n = a.shape[0]
    idx = np.arange(n)
    rev = np.zeros_like(a)
    for k in active_idxs:
        j = k - idx
        mask = (j >= 0) & (j < n)
        rev[mask] += a[j[mask]]
    if active_idxs.size > 0:
        rev /= active_idxs.size
    return rev


# ----------------------------------------------------------------------
def _adam_optimize(
    a_init: np.ndarray,
    target_sum: float,
    deadline: float,
    max_iters: int | None = None,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser that minimises max(a*a) with the exact sub‑gradient.
    Returns the best local vector and its objective value (M).
    """
    a = a_init.copy()
    n = a.shape[0]

    # enforce the normalisation Σa = target_sum
    s = a.sum()
    if s > 0:
        a *= target_sum / s
    else:                                     # extremely unlikely
        a = np.full_like(a, target_sum / n)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr0 = 0.05
    iteration = 0
    jitter_interval = 4000
    best_seq = a.copy()
    best_val = float("inf")
    const_den = target_sum ** 3               # used for the scalar term

    while time.time() < deadline and (max_iters is None or iteration < max_iters):
        M, conv = _conv_max_fft(a)

        # active set of indices attaining the maximum (within tolerance)
        tol = 1e-12
        active = np.where(conv >= M - tol)[0]
        rev = _subgradient(a, active)

        # sub‑gradient of the objective (which equals M after normalisation)
        # coeff = 4 n / target_sum^2 = 2   (because target_sum^2 = 2n)
        # const = 4 n M / target_sum^3 = 2 M / target_sum
        coeff = 2.0
        const = (2.0 * M) / target_sum
        grad = coeff * rev - const

        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        lr = lr0 / (1.0 + 1e-4 * iteration)
        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto non‑negative orthant and renormalisation
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0:
            a *= target_sum / s
        else:                                 # fallback to uniform
            a = np.full_like(a, target_sum / n)

        # occasional jitter to escape plateaux
        if iteration % jitter_interval == 0:
            jitter = np.random.randn(n) * 0.001 * target_sum / n
            a = np.maximum(a + jitter, 0.0)
            a *= target_sum / a.sum()

        # keep the best found locally
        if M < best_val:
            best_val = M
            best_seq = a.copy()
            print(
                f"[adam] length {n:5d} iter {iteration:7d}  new best M = {best_val:.6f}"
            )

    return best_seq, best_val


# ----------------------------------------------------------------------
def _solve_lp_max_sum(
    f: np.ndarray,
    M: float,
    row_idx: np.ndarray,
    col_idx: np.ndarray,
) -> np.ndarray | None:
    """
    Solve   maximise Σ g_j   s.t. (f * g)_k ≤ M, g ≥ 0.
    The coefficient matrix is built from the current `f` using the pre‑computed
    sparse pattern (row_idx, col_idx).  Returns the optimal g or None on failure.
    """
    n = f.shape[0]

    # data for the Toeplitz‑like constraint matrix
    data = np.repeat(f, n)          # length n*n

    A = sp.csr_matrix((data, (row_idx, col_idx)), shape=(2 * n - 1, n))
    b = np.full(2 * n - 1, M, dtype=float)

    c = -np.ones(n, dtype=float)    # maximise Σ g  → minimise -Σ g
    bounds = [(0.0, None)] * n

    # HiGHS is the default method for recent SciPy versions
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method="highs")
    if not res.success:
        return None
    return res.x


# ----------------------------------------------------------------------
def _lp_refine(
    a_init: np.ndarray,
    target_sum: float,
    deadline: float,
    max_outer: int = 3,
) -> tuple[np.ndarray, float]:
    """
    Perform a few LP‑based improvement steps.
    Returns the best sequence found and its objective value (M).
    """
    n = a_init.shape[0]
    if n > 2500:                # skip LP on very large vectors (memory)
        return a_init, float("inf")

    # pre‑compute the sparse pattern of the convolution matrix – it does not
    # depend on the actual values of `f`
    i = np.arange(n, dtype=np.int32)
    j = np.arange(n, dtype=np.int32)
    row_idx = np.repeat(i, n) + np.tile(j, n)      # shape n*n
    col_idx = np.tile(j, n)

    a = a_init.copy()
    best_seq = a.copy()
    best_val = float("inf")
    outer = 0

    while time.time() < deadline and outer < max_outer:
        M, _ = _conv_max_fft(a)

        g0 = _solve_lp_max_sum(a, M, row_idx, col_idx)
        if g0 is None or g0.sum() == 0.0:
            break

        # scale to the proper total mass
        g = g0 * (target_sum / g0.sum())

        # evaluate the pure LP candidate
        M_g, _ = _conv_max_fft(g)

        if M_g < M:
            a = g
            M = M_g
        else:
            # try a convex combination; 0.5 works well in practice
            t = 0.5
            a_mix = (1 - t) * a + t * g
            a_mix = np.maximum(a_mix, 0.0)
            a_mix *= target_sum / a_mix.sum()
            M_mix, _ = _conv_max_fft(a_mix)

            if M_mix < M:
                a = a_mix
                M = M_mix
            else:
                # cannot improve any further with this LP
                break

        if M < best_val:
            best_val = M
            best_seq = a.copy()
            print(f"[lp] length {n:5d} iter {outer+1}  improved M = {best_val:.6f}")

        outer += 1

    return best_seq, best_val

def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    Returns the best sequence found before the time budget expires.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 5.0        # safety margin

    # ------------------------------------------------------------------
    # Load a baseline construction if it exists
    try:
        base = np.array(height_sequence_1, dtype=float)   # type: ignore
    except Exception:
        # fallback: modest uniform vector
        base = np.ones(1500, dtype=float)

    best_seq = None
    best_score = float("inf")

    # normalise the base sequence once
    n0 = base.shape[0]
    S0 = math.sqrt(2.0 * n0)
    base = base * (S0 / base.sum())
    init_score = evaluate_sequence(base.tolist())
    best_seq = base.copy()
    best_score = init_score
    print(f"[init] length = {n0}, score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Choose a diverse set of lengths (original, interpolations, random)
    length_factors = [0.5, 0.75, 1.0, 1.25, 1.5, 2.0]
    lengths = {max(300, int(n0 * f)) for f in length_factors}
    # a few random lengths to diversify the search
    for _ in range(4):
        lengths.add(int(rng.integers(300, 5000)))
    lengths = sorted(min(l, 8000) for l in lengths)

    # ------------------------------------------------------------------
    # Main optimisation loop over the selected lengths
    for n in lengths:
        if time.time() >= deadline:
            break

        # initialise a vector for this length
        if n == n0:
            a0 = base.copy()
        else:
            # linear interpolation of the base pattern onto the new grid
            old_x = np.linspace(0.0, 1.0, n0)
            new_x = np.linspace(0.0, 1.0, n)
            a0 = np.interp(new_x, old_x, base)
            a0 = np.maximum(a0, 0.0)

        # add a small random perturbation to avoid exact replicas
        a0 += rng.normal(scale=0.01, size=n)
        a0 = np.maximum(a0, 0.0)

        target_sum = math.sqrt(2.0 * n)
        if a0.sum() == 0:
            a0 = np.full(n, target_sum / n)
        else:
            a0 = a0 * (target_sum / a0.sum())

        print(f"[run] starting length {n}")

        # ----- first Adam phase -------------------------------------------------
        a_opt, local_best = _adam_optimize(
            a_init=a0,
            target_sum=target_sum,
            deadline=deadline,
            max_iters=400000,                      # generous but bounded
        )

        # ----- LP refinement ----------------------------------------------------
        a_opt, local_best = _lp_refine(
            a_init=a_opt,
            target_sum=target_sum,
            deadline=deadline,
            max_outer=3,
        )

        # ----- short second Adam phase -----------------------------------------
        a_opt, local_best = _adam_optimize(
            a_init=a_opt,
            target_sum=target_sum,
            deadline=deadline,
            max_iters=150000,
        )

        # ----- global bookkeeping -----------------------------------------------
        cur_score = evaluate_sequence(a_opt.tolist())
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a_opt.copy()
            print(f"[global] new best score = {best_score:.6f} (length {n})")

    # ------------------------------------------------------------------
    # Fallback in the unlikely case that nothing was found
    if best_seq is None:
        n = 2100
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n)
        best_score = evaluate_sequence(best_seq.tolist())
        print("[fallback] using uniform sequence")

    print(f"[done] best score = {best_score:.6f}, length = {best_seq.shape[0]}")
    # Return a plain Python list – the harness expects this format
    return best_seq.tolist()
