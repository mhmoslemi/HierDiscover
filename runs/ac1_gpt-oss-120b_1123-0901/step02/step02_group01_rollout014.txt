"""
Propose a step‑function sequence that (approximately) minimises

        F(a) = 2·n·max(a*a) / (sum(a)²)

under a_i ≥ 0.  The algorithm is a continuous optimisation in the
probability simplex using a smooth‑max surrogate and Adam‑style
updates.  It works for arbitrary lengths (here we try several sizes)
and restarts from random points and from the previously best known
construction (if the global variable ``height_sequence_1`` exists).

The main ideas
--------------
* Parameterise the non‑negative simplex point a with unconstrained logits
  x (a = softmax(x)).  This guarantees a_i≥0 and Σa_i=1 exactly.
* Replace the hard max in the objective by the log‑sum‑exp
  φ_λ(a) = (1/λ)·log Σ_k exp(λ·c_k) where c = a*a.
  Its gradient is ∇φ = 2·(w ★ a) with
       w_k = exp(λ·(c_k–max(c))) / Σ_j exp(λ·(c_j–max(c))).
* Use Adam to update the logits, increase the temperature λ slowly
  (continuation), and keep track of the best *exact* objective
  (2·n·max(c) because Σa_i=1).
* If progress stalls, add a small random “shake’’ to the logits.
* Run a few restarts for several sequence lengths (1024…8192) inside
  the 1000 s budget and return the best sequence found.
"""

# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) using the FFT."""
    n = a.shape[0]
    if fft_len is None:
        fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Compute the linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) equal to
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    n = a.shape[0]
    start = n - 1
    return corr[start:start + n]  # length n


def _objective_from_conv(conv: np.ndarray, n: int) -> float:
    """
    Exact objective used by evaluate_sequence for a vector whose sum is 1.
    (Because Σa_i = 1, the denominator is 1 and the objective reduces to
    2·n·max(conv).)
    """
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative sequence with a low evaluation score.

    The routine respects the given time budget, prints intermediate
    progress and finally returns the best sequence (as a plain Python list)
    that it found.
    """
    np.random.seed(seed)
    # safety margin for interpreter shutdown
    deadline = time.time() + budget_s - 0.5

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Helper: initialise logits from a probability vector
    def _logits_from_probs(p: np.ndarray) -> np.ndarray:
        return np.log(p + 1e-15)

    # ------------------------------------------------------------------
    # Search over several problem sizes – larger n gives more freedom.
    lengths = [1024, 2048, 4096, 8192]

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")

        # FFT sizes (powers of two) for the two convolutions we need.
        fft_len_conv = 1 << int(np.ceil(np.log2(2 * n - 1)))
        fft_len_corr = 1 << int(np.ceil(np.log2(3 * n)))  # w length + a length - 1

        # Number of independent restarts for this length.
        restarts = 3

        for r in range(restarts):
            if time.time() > deadline:
                break

            # ---------- initialise logits ---------------------------------
            if r == 0 and best_seq is not None and len(best_seq) == n:
                # Warm‑start from the globally best sequence (if length matches)
                init_a = np.array(best_seq, dtype=np.float64)
                init_a /= init_a.sum()
                x = _logits_from_probs(init_a)
                print("[restart 0] warm start from previous best")
            elif 'height_sequence_1' in globals():
                # Optional: use the provided earlier construction (trim / pad)
                cand = np.array(height_sequence_1, dtype=np.float64)
                if cand.shape[0] >= n:
                    cand = cand[:n]
                else:
                    cand = np.pad(cand, (0, n - cand.shape[0]))
                cand = np.maximum(cand, 0.0)
                cand_sum = cand.sum()
                if cand_sum < 0.01:
                    cand = np.random.dirichlet(np.ones(n))
                else:
                    cand /= cand_sum
                x = _logits_from_probs(cand)
                print("[restart 0] start from height_sequence_1")
            else:
                # Pure random start
                x = np.random.randn(n) * 0.1

            # Adam state
            m = np.zeros(n, dtype=np.float64)
            v = np.zeros(n, dtype=np.float64)
            beta1, beta2, eps = 0.9, 0.999, 1e-8
            lr = 0.05

            # Smooth‑max temperature
            lam = 0.5
            lam_factor = 1.03  # geometric increase

            t = 0
            last_improve = time.time()
            max_iters = 20000  # safety cap per restart

            while time.time() < deadline and t < max_iters:
                # ----- logits → probabilities (softmax, guarantees simplex) -----
                max_x = np.max(x)                         # for numerical stability
                exp_x = np.exp(x - max_x)
                a = exp_x / exp_x.sum()                  # a_i ≥ 0, Σa_i = 1

                # ----- self‑convolution -------------------------------------------------
                conv = _full_convolution(a, fft_len=fft_len_conv)
                max_c = conv.max()

                # ----- smooth‑max weights ------------------------------------------------
                # w_k ∝ exp(λ·(conv_k – max_c))  (numerically stable)
                w = np.exp(lam * (conv - max_c))
                w /= w.sum()                               # Σ w_k = 1

                # ----- gradient of the surrogate w.r.t. a --------------------------------
                grad_a = 2.0 * _cross_correlation(w, a, fft_len=fft_len_corr)

                # ----- back‑prop through softmax (gradient w.r.t. logits) -----------------
                avg = np.dot(a, grad_a)                    # scalar ⟨a,∇φ⟩
                grad_x = a * (grad_a - avg)                # ∇_x φ

                # ----- Adam update -------------------------------------------------------
                t += 1
                m = beta1 * m + (1.0 - beta1) * grad_x
                v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
                m_hat = m / (1.0 - beta1 ** t)
                v_hat = v / (1.0 - beta2 ** t)
                x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

                # ----- temperature schedule ------------------------------------------------
                lam = min(lam * lam_factor, 1e6)   # cap to avoid overflow

                # ----- exact objective evaluation ----------------------------------------
                # Σa_i ≈ 1, so the exact objective equals 2·n·max_c
                val = 2.0 * n * max_c / (a.sum() ** 2)

                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} restart={r} iter={t:05d}] new best: {val:.6f}")

                # ----- stagnation handling ------------------------------------------------
                if time.time() - last_improve > 5.0:
                    # small random perturbation of the logits
                    x += (np.random.rand(n) - 0.5) * 1e-2
                    last_improve = time.time()
                    print(f"[n={n} restart={r} iter={t}] shake due to stagnation")

    # ------------------------------------------------------------------
    # Fallback – if nothing succeeded (very unlikely) return a simple uniform vector.
    if best_seq is None:
        n_fallback = 2000
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)

    # Convert to plain Python list as required by the external evaluator.
    return best_seq.tolist()
