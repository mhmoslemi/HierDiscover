def propose_candidate(seed=0, budget_s=1000):
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5

    # Determine length n0: maybe try to start with 2000 then gradually increase to 5000 or 8000.
    # Let's set n=5000.
    n = 5000

    # Initial sequence: if global height_sequence_1 exists use it and interpolate; else use arcsine.
    try:
        base = np.array(height_sequence_1, dtype=float)
        # normalize sum =1
        base = base / base.sum()
        # interpolate
        old_n = len(base)
        x_old = np.linspace(0,1,old_n)
        x_new = np.linspace(0,1,n)
        seq0 = np.interp(x_new, x_old, base)
    except NameError:
        seq0 = np.full(n, 1.0/n) # fallback uniform

    seq = seq0.copy()
    # Ensure non-negative and sum=1
    seq = np.maximum(seq, 0)
    seq = seq / seq.sum()
    # Initialize Adam parameters
    m = np.zeros_like(seq)
    v = np.zeros_like(seq)
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    alpha = 0.5   # base learning rate
    it = 0
    best_seq = seq.copy()
    best_score = evaluate_sequence(seq.tolist())
    print("[Init] initial score", best_score)

    while time.time() < deadline:
        # Convolution via FFT
        mlen = 1 << (2*n-1).bit_length()
        fft_seq = np.fft.rfft(seq, n=mlen)
        conv = np.fft.irfft(fft_seq * fft_seq, n=mlen)[:2*n-1]   # length 2n-1
        # Determine top K indices
        K = 3
        if K >= conv.size:
            top_idx = np.arange(conv.size)
        else:
            top_idx = np.argpartition(conv, -K)[-K:]

        # Compute gradient as sum of a_{k-i} contributions for each top k
        grad = np.zeros_like(seq)
        for k in top_idx:
            lo = max(0, k - (n - 1))
            hi = min(n-1, k)
            idx = np.arange(lo, hi+1)
            grad[idx] += seq[k - idx]  # a_{k-i}
        grad = grad / K

        # Adam update
        it += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        step = alpha * m_hat / (np.sqrt(v_hat) + epsilon)

        new_seq = seq - step
        # Project onto simplex
        new_seq = _project_onto_simplex(new_seq)

        # Evaluate
        score = evaluate_sequence(new_seq.tolist())
        if score < best_score:
            best_score = score
            best_seq = new_seq.copy()
            print(f"[Iter {it}] new best {best_score:.6f}")

        # Epoch wise random restart or jitter
        if it % 20000 == 0:
            # jitter
            jitter = rng.normal(scale=0.02, size=n)
            new_seq = new_seq + jitter
            new_seq = np.clip(new_seq, 0, None)
            new_seq = new_seq / new_seq.sum()
            m.fill(0)
            v.fill(0)
            print(f"[Iter {it}] jitter applied")
        seq = new_seq

    print("Finished best score", best_score)
    return best_seq.tolist()
