def _solve_max_sum_lp(f, M):
    n = len(f)
    # Build A_ub
    rows = 2*n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    # Fill each column
    for j in range(n):
        A[j:j+n, j] = f
    # b_ub
    b = np.full(rows, M, dtype=np.float64)
    # Objective c
    c = -np.ones(n)
    # bounds
    bounds = [(0, None)]*n
    # Solve
    result = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not result.success:
        # If fails, fallback to uniform?
        return np.full(n, 1.0/n)
    g = result.x
    # Ensure nonnegatives due to solver tolerance
    g = np.maximum(g, 0.0)
    return g

"""
Propose a step‑function (non‑negative sequence) that minimises
`evaluate_sequence`.  The algorithm alternates two phases:

1. **Linear‑programming step** –  given the current distribution `f`
   (normalised to sum = 1) we compute the current peak
   `M = max(f * f)`.  We then solve a linear program  

        maximise    Σ g_j
        subject to  (f * g)_k ≤ M   for all k
                     g_j ≥ 0

   The solution `g` is the "most massive" sequence that never makes the
   convolution with `f` exceed the current peak.  Mixing a small amount of
   `g` into `f` (convex combination) is known to never increase the peak
   (see the alternating‑LP method in the statement).

2. **Local‑search polishing** – after the LP phase we perform a fast
   random‑perturbation hill‑climb: small Gaussian jitter is added,
   the vector is projected back onto the probability simplex and kept
   iff it improves the objective.

Both phases respect the time budget (≈ 1000 s).  The best sequence ever
encountered is returned as a plain Python list of floats.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        {x | x_i ≥ 0, Σ x_i = 1}
    """
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:                     # all non‑positive
        return np.full_like(v, 1.0 / len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _initial_candidates(n: int, rng: np.random.Generator):
    """
    Produce a small pool of diversified starting points, already normalised
    to sum = 1.  The global variable `height_sequence_1` (if present) is also
    added to the pool.
    """
    cand = []

    # Uniform
    cand.append(np.full(n, 1.0 / n))

    # Geometric decays
    for factor in [0.6, 0.8, 1.0, 1.5, 2.0, 3.0]:
        r = max(0.0, 1.0 - factor / n)
        seq = np.power(r, np.arange(n, dtype=np.float64))
        cand.append(seq / seq.sum())

    # Arcsine‑shaped (discrete analogue of the continuous arcsine law)
    i = np.arange(n, dtype=np.float64)
    arcs = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    cand.append(arcs / arcs.sum())

    # Random Dirichlet draws
    for _ in range(5):
        rand = rng.random(n)
        cand.append(rand / rand.sum())

    # If a previous best is available, include it
    global_seq = globals().get("height_sequence_1")
    if isinstance(global_seq, list) and len(global_seq) > 0:
        arr = np.asarray(global_seq, dtype=np.float64)
        if arr.ndim == 1:
            if arr.size != n:
                # Resize by interpolation if necessary
                x_old = np.linspace(0, 1, arr.size)
                x_new = np.linspace(0, 1, n)
                arr = np.interp(x_new, x_old, arr)
            cand.append(arr / arr.sum())

    return cand


# ----------------------------------------------------------------------
def _solve_lp_step(f: np.ndarray, M: float):
    """
    Solve the linear program

        maximise    Σ g_j
        subject to  (f * g)_k ≤ M   for k = 0 … 2n‑2
                    g_j ≥ 0

    The convolution matrix (f * g) is Toeplitz with each column equal to `f`
    placed on a diagonal.  The function returns the optimal `g` (not normalised).
    """
    n = f.shape[0]
    rows = 2 * n - 1
    # Build the constraint matrix A_ub (dense, but still modest for n≈2000)
    A = np.zeros((rows, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    b = np.full(rows, M, dtype=np.float64)

    # maximise Σ g_j  ⇒ minimise -Σ g_j
    c = -np.ones(n, dtype=np.float64)

    # bounds: g_j ≥ 0
    bounds = [(0.0, None) for _ in range(n)]

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method='highs', options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})

    if not res.success or res.x is None:
        # Fallback: return the current f (feasible) – this will never improve,
        # but keeps the loop alive.
        return f.copy()
    g = res.x
    # Numerical safety
    g[g < 0] = 0.0
    return g


# ----------------------------------------------------------------------
def _local_search(seq: np.ndarray, rng: np.random.Generator,
                  deadline: float, best_score: float):
    """
    Simple hill‑climbing with Gaussian jitter.  The function mutates `seq`
    in‑place only when an improvement is observed, and returns the best
    sequence found together with its score.
    """
    n = seq.shape[0]
    cur = seq.copy()
    cur_score = best_score
    it = 0
    # The jitter magnitude shrinks slowly – start with 1 % of the total mass.
    sigma = 0.01

    while time.time() < deadline:
        jitter = rng.normal(scale=sigma, size=n)
        cand = cur + jitter
        cand = np.clip(cand, 0.0, None)
        if cand.sum() == 0:
            continue
        cand = cand / cand.sum()
        score = evaluate_sequence(cand.tolist())
        it += 1
        if score < cur_score:
            cur = cand
            cur_score = score
            print(f"[Local {it:06d}] improved to {cur_score:.6f}")
            # tighten sigma a bit after a success
            sigma = max(sigma * 0.98, 1e-4)
        else:
            # slowly increase exploration if we are stuck
            sigma = min(sigma * 1.001, 0.05)

    return cur, cur_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Main entry point required by the grader.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Time budget in seconds (the function must return before this time).

    Returns
    -------
    list[float]
        The best non‑negative step‑function found (as a Python list).
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # 1. Problem size – a modest length that balances quality and speed.
    n = 2000

    # ------------------------------------------------------------------
    # 2. Build a diverse pool of starting points and pick the best one.
    best_seq = None
    best_score = float("inf")
    for cand in _initial_candidates(n, rng):
        score = evaluate_sequence(cand.tolist())
        if score < best_score:
            best_score = score
            best_seq = cand.copy()
    print(f"[Init] best initial score = {best_score:.6f}")

    # Normalise the current iterate to sum = 1 (scaling is irrelevant for the
    # objective, but a fixed normalisation simplifies the LP constraints).
    f = best_seq.copy()
    f = f / f.sum()

    # ------------------------------------------------------------------
    # 3. Alternating Linear‑Programming improvement.
    #    t is the mixing coefficient (initially fairly aggressive, then
    #    decayed).
    t = 0.5
    it = 0
    max_lp_iters = 300   # generous upper bound; actual count is time‑controlled

    while time.time() < deadline and it < max_lp_iters:
        conv_f = np.convolve(f, f)
        M = conv_f.max()

        # Solve the LP: maximise total mass of g under the convolution bound M.
        g = _solve_lp_step(f, M)

        # Normalise g to a probability vector.
        g_sum = g.sum()
        if g_sum == 0:
            # Edge case – stop the LP phase.
            print("[LP] LP returned zero vector; terminating LP phase.")
            break
        g = g / g_sum

        # Convex combination (mixing).  The classic theory uses a small t;
        # we start with 0.5 and slowly decay.
        f_new = (1.0 - t) * f + t * g
        f_new = f_new / f_new.sum()   # re‑normalise to avoid drift

        # Evaluate the new candidate.
        score = evaluate_sequence(f_new.tolist())
        if score < best_score:
            best_score = score
            best_seq = f_new.copy()
            print(f"[LP Iter {it:04d}] new best = {best_score:.6f}")
        # Prepare for next iteration.
        f = f_new
        t = max(t * 0.995, 0.01)   # slowly shrink the mixing factor
        it += 1

    print(f"[LP] finished after {it} iterations, best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Local stochastic polishing (hill‑climbing with Gaussian noise).
    #    Allocate the remaining time to this cheap phase.
    if time.time() < deadline:
        remaining = deadline - time.time()
        # Give at most half of the leftover time to the local search.
        ls_deadline = time.time() + min(remaining * 0.5, 30.0)
        best_seq, best_score = _local_search(best_seq, rng,
                                             ls_deadline, best_score)
        print(f"[Local] final best after polishing = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Return the best sequence found (as a plain list of Python floats).
    #    The evaluation function will internally clip negatives and enforce the
    #    upper bound of 1000, so we simply return the raw numbers.
    return best_seq.tolist()
