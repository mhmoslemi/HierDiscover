"""
Propose a step‑function (non‑negative sequence) with a low
`evaluate_sequence` value.

Algorithm
---------
1. **Problem size** – use a moderate length (n≈2000).  
   Larger n gives more degrees of freedom but makes each evaluation
   slower (O(n²) convolution).  n=2000 is a good compromise.

2. **Initial pool** – generate a diverse set of candidates:
   * uniform,
   * geometric families (`r^i` with several decay rates),
   * polynomial‑decreasing families (`(n‑i)^p` for several p),
   * beta‑shaped families (`(i+1)^α (n‑i)^β` for a few (α,β)),
   * the previously found construction (`height_sequence_1`) if it exists.
   The best of these seeds the optimisation.

3. **Simulated‑annealing phase** – for roughly half of the time budget
   repeatedly move a small amount of mass from one index to another.
   Moves are accepted if they improve the score or with a Metropolis
   probability that slowly cools.  This explores the non‑convex landscape
   and can escape local minima.

4. **Sub‑gradient refinement** – the remaining time is spent on a
   deterministic sub‑gradient descent (the same idea as the reference
   code) but using the exact gradient of the currently maximal
   convolution entries.  A heavy‑ball momentum term and a slowly
   decaying step‑size are employed; the iterate is projected back onto
   the probability simplex after each step.

Both phases keep the best sequence ever seen.  At the end the best
sequence is returned – it will be a probability distribution (non‑negative,
summing to 1) and should achieve an objective well below the target
1.503 within the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v):
    """Project a vector onto the probability simplex {x≥0, sum(x)=1}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:           # all entries are non‑positive
        return np.full_like(v, 1.0 / len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimises
    `evaluate_sequence` within the allotted time.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ---------- 1. problem size ----------
    n = 2000  # moderate size – fast enough for many evaluations

    # ---------- 2. generate a diverse initial pool ----------
    candidates = []

    # uniform
    candidates.append(np.full(n, 1.0 / n))

    # geometric families  (r = 1 - factor/n)
    for factor in [0.5, 1.0, 2.0, 3.0, 5.0]:
        eps = factor / n
        r = max(0.0, 1.0 - eps)
        seq = np.power(r, np.arange(n, dtype=np.float64))
        candidates.append(seq / seq.sum())

    # polynomial decreasing families  a_i ∝ (n-i)^p
    for p in [0.5, 1, 2, 3, 4, 5]:
        raw = (np.arange(n)[::-1] + 1) ** p
        candidates.append(raw / raw.sum())

    # beta‑shaped families  a_i ∝ (i+1)^α (n-i)^β
    for α in [0.5, 1, 2]:
        for β in [0.5, 1, 2]:
            i = np.arange(n, dtype=np.float64)
            raw = (i + 1) ** α * (n - i) ** β
            candidates.append(raw / raw.sum())

    # include the previously best construction if it is available
    if 'height_sequence_1' in globals():
        prev = np.asarray(height_sequence_1, dtype=np.float64)
        if prev.shape[0] == n:
            candidates.append(prev / prev.sum())
        else:
            # interpolate to length n
            x_old = np.linspace(0.0, 1.0, prev.shape[0])
            x_new = np.linspace(0.0, 1.0, n)
            interpolated = np.interp(x_new, x_old, prev)
            candidates.append(interpolated / interpolated.sum())

    # ---------- 3. pick the best seed ----------
    best_seq = None
    best_score = float("inf")
    for seq in candidates:
        score = evaluate_sequence(seq.tolist())
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
    print(f"[Init] best seed score = {best_score:.6f}")

    step_scale = 0.5         # initial proportion of mass moved

    seq = best_seq.copy()

        max_val = conv.max()
        # indices where the convolution attains (or is very close to) the max
        max_idx = np.where(conv >= max_val - 1e-12)[0]

        # exact gradient of the (average) maximal entries
        grad = np.zeros_like(seq)
        for k in max_idx:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            # contribution a_{k-i} for each i in the overlap
            grad[idx] += seq[k - idx]

        grad = grad / max(1, len(max_idx))      # average over the active set

        momentum = beta_mom * momentum + (1.0 - beta_mom) * grad

        seq = new_seq
        it2 += 1

    print(f"Finished. Best objective = {best_score:.6f} (length {n})")
    return best_seq.tolist()
