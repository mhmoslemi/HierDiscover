"""
A fresh optimiser for the step‑function bound

The routine explores a few parametric families (exponential, linear,
triangular, Gaussian) to obtain quick seeds, then refines them with a
smooth‑max gradient descent.  The smooth maximum is realised by a
soft‑max over the autocorrelation entries; its gradient can be computed
by a single FFT‑based cross‑correlation, giving O(n log n) per
iteration.  The smoothing parameter α is increased from a small value
to a larger one (homotopy), allowing the optimiser to escape shallow
local minima that the plain sub‑gradient method suffers from.
Finally, if SciPy is available, a short linear‑programming refinement
is performed (the classic LP‑step).  The best sequence found within the
time budget is returned as a plain list of non‑negative floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the simplex {x ≥ 0, Σx = s}
# ----------------------------------------------------------------------
def project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Projection of a vector onto the simplex with prescribed sum `s`."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Safe wrapper around the external evaluation routine
# ----------------------------------------------------------------------
def safe_evaluate(seq: list) -> float:
    """Call evaluate_sequence and return +inf on any exception."""
    try:
        return evaluate_sequence(seq)          # type: ignore[name-defined]
    except Exception:
        return float("inf")


# ----------------------------------------------------------------------
# Convolution a * a (full linear convolution) via real FFT
# ----------------------------------------------------------------------
def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of a with itself."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Gradient of the smooth‑max (soft‑max) of the autocorrelation
# ----------------------------------------------------------------------
def smooth_max_gradient(a: np.ndarray, alpha: float) -> np.ndarray:
    """
    Compute ∇_a  (1/α) log Σ exp(α·(a*a)) using FFT‑based cross‑correlation.
    Returns a vector of length n (same as a).
    """
    n = a.shape[0]
    conv = conv_self_fft(a)
    max_c = conv.max()
    # soft‑max weights (stable)
    exp_args = alpha * (conv - max_c)          # ≤ 0, no overflow
    w = np.exp(exp_args)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # cross‑correlation a ⋆ w = conv(a_rev, w) shifted by (n‑1)
    a_rev = a[::-1]
    size = 1 << ((3 * n - 2).bit_length())    # enough for (n) + (2n-1) -1
    fa_rev = np.fft.rfft(a_rev, n=size)
    fw = np.fft.rfft(w, n=size)
    corr_full = np.fft.irfft(fa_rev * fw, n=size)
    grad = 2.0 * corr_full[n - 1 : n - 1 + n]  # slice to length n
    return grad


# ----------------------------------------------------------------------
# Candidate update helper – maintains the global best
# ----------------------------------------------------------------------
def maybe_update(best_seq: np.ndarray, best_score: float,
                 cand_seq: np.ndarray) -> tuple[np.ndarray, float]:
    """Evaluate `cand_seq`; if it improves, update and print progress."""
    score = safe_evaluate(cand_seq.tolist())
    if score < best_score:
        print(f"[{time.time():.2f}] New best = {score:.6f} (len={len(cand_seq)})")
        return cand_seq.copy(), score
    return best_seq, best_score


# ----------------------------------------------------------------------
# Shape factories (non‑negative, un‑normalised)
# ----------------------------------------------------------------------
def shape_exponential(n: int, alpha: float) -> np.ndarray:
    return np.exp(-alpha * np.arange(n, dtype=float))


def shape_exponential_rev(n: int, alpha: float) -> np.ndarray:
    return np.exp(-alpha * np.arange(n - 1, -1, -1, dtype=float))


def shape_linear_desc(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    return np.maximum(0.0, 1.0 - beta * i)


def shape_quadratic_desc(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    return np.maximum(0.0, 1.0 - beta * i * i)


def shape_triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    return np.minimum(i + 1.0, n - i)


def shape_gaussian(n: int, sigma: float) -> np.ndarray:
    """Discrete Gaussian centred in the middle of the vector."""
    x = np.arange(n, dtype=float)
    centre = (n - 1) / 2.0
    return np.exp(-0.5 * ((x - centre) / sigma) ** 2)


# ----------------------------------------------------------------------
# Core smooth‑max gradient descent optimizer
# ----------------------------------------------------------------------
def gradient_descent_optimize(start_seq: np.ndarray,
                              target_sum: float,
                              deadline: float,
                              best_seq: np.ndarray,
                              best_score: float,
                              max_iters: int = 2500,
                              eval_interval: int = 10) -> tuple[np.ndarray, float]:
    """
    Refine `start_seq` with a smooth‑max gradient descent.
    The method uses a homotopy on the soft‑max temperature α.
    Returns possibly improved (best_seq, best_score).
    """
    a = start_seq.astype(float).copy()
    n = a.shape[0]
    m = np.zeros_like(a)                     # momentum
    beta = 0.9                                 # momentum decay
    base_step = 0.25
    alpha_start = 0.05
    alpha_end = 5.0

    for it in range(max_iters):
        # time check
        if time.time() >= deadline:
            break

        # homotopic temperature
        alpha = alpha_start * (alpha_end / alpha_start) ** (it / max_iters)

        # gradient of smooth max
        grad = smooth_max_gradient(a, alpha)

        # momentum update
        m = beta * m + (1.0 - beta) * grad

        # diminishing step size (≈ 1/√it)
        step = base_step / math.sqrt(it + 1)
        max_abs = np.max(np.abs(m))
        if max_abs == 0.0:
            max_abs = 1.0
        eta = step / max_abs

        # gradient step and projection onto the simplex
        a_new = a - eta * m
        a_new = project_onto_simplex(a_new, target_sum)

        # occasional evaluation
        if it % eval_interval == 0:
            best_seq, best_score = maybe_update(best_seq, best_score, a_new)

        # early exit if we have already crossed the target bound
        if best_score < 1.503:
            break

        a = a_new

    return best_seq, best_score


# ----------------------------------------------------------------------
# (Optional) linear‑programming refinement – the classic “LP‑step”
# ----------------------------------------------------------------------
def lp_refinement(current_seq: np.ndarray,
                  target_sum: float,
                  deadline: float,
                  best_seq: np.ndarray,
                  best_score: float) -> tuple[np.ndarray, float]:
    """
    Perform a few LP refinement iterations using SciPy (if available).
    """
    try:
        from scipy.optimize import linprog
    except Exception:                         # SciPy not present
        return best_seq, best_score

    a = current_seq.copy()
    n = a.shape[0]
    rows = 2 * n - 1

    while time.time() < deadline:
        conv = conv_self_fft(a)
        M = conv.max()

        # Build matrix A such that (a * g)_k = A_k · g
        A = np.zeros((rows, n), dtype=float)
        for k in range(rows):
            low = max(0, k - (n - 1))
            high = min(n - 1, k)
            cols = np.arange(low, high + 1, dtype=int)
            A[k, cols] = a[k - cols]

        res = linprog(
            c=-np.ones(n),                 # maximise Σ g_j  → minimise -Σ g_j
            A_ub=A,
            b_ub=np.full(rows, M),
            bounds=(0, None),
            method="highs",
            options={"presolve": True},
        )
        if not res.success:
            break

        g = res.x
        s = g.sum()
        if s <= 0.0:
            break
        # rescale g to the same simplex used everywhere
        g = g * (target_sum / s)

        # convex combination step
        t = 0.2
        a = (1.0 - t) * a + t * g

        best_seq, best_score = maybe_update(best_seq, best_score, a)

        # stop if improvement stalls or we run out of time
        if best_score < 1.503 or time.time() + 0.5 >= deadline:
            break

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: float = 1000.0,
                     **kwargs) -> list:
    """
    Search for a non‑negative height sequence that minimises
    2 * n * max(conv(a, a)) / (sum(a)**2).

    The algorithm consists of:
      1. quick sweep over a few parametric families,
      2. smooth‑max gradient descent with homotopy (multiple restarts),
      3. optional LP refinement (if SciPy is available).
    The best sequence found before the deadline is returned.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0. incorporate a previously discovered sequence, if supplied
    # ------------------------------------------------------------------
    prev = globals().get('height_sequence_1')
    if isinstance(prev, list) and prev:
        cand = np.array(prev, dtype=float)
        # normalise onto our internal simplex (scale does not affect the score)
        n_prev = cand.shape[0]
        tgt = math.sqrt(2.0 * n_prev)
        cand = project_onto_simplex(cand, tgt)
        best_seq, best_score = maybe_update(np.empty(0), float("inf"), cand)

    # ------------------------------------------------------------------
    # 1. quick parametric scan – cheap candidates
    # ------------------------------------------------------------------
    lengths = list(range(200, 1601, 200)) + [1326, 1500, 1800]
    for n in lengths:
        if time.time() >= deadline:
            break
        target = math.sqrt(2.0 * n)

        # exponential decays
        for alpha in np.logspace(-4, -0.3, num=10):
            a = shape_exponential(n, alpha)
            a = project_onto_simplex(a, target)
            best_seq, best_score = maybe_update(best_seq, best_score, a)

        # reversed exponential
        for alpha in np.logspace(-4, -0.3, num=8):
            a = shape_exponential_rev(n, alpha)
            a = project_onto_simplex(a, target)
            best_seq, best_score = maybe_update(best_seq, best_score, a)

        # linear descending
        for beta in np.linspace(0.0, 2.0 / max(1, n - 1), num=8):
            a = shape_linear_desc(n, beta)
            if a.sum() == 0:
                continue
            a = project_onto_simplex(a, target)
            best_seq, best_score = maybe_update(best_seq, best_score, a)

        # quadratic descending
        for beta in np.linspace(0.0, 1.0 / max(1, n ** 2), num=8):
            a = shape_quadratic_desc(n, beta)
            if a.sum() == 0:
                continue
            a = project_onto_simplex(a, target)
            best_seq, best_score = maybe_update(best_seq, best_score, a)

        # triangular shape
        a = shape_triangular(n)
        a = project_onto_simplex(a, target)
        best_seq, best_score = maybe_update(best_seq, best_score, a)

        # Gaussian bumps (few σ values)
        for sigma in [n / 6.0, n / 4.0, n / 3.0]:
            a = shape_gaussian(n, sigma)
            a = project_onto_simplex(a, target)
            best_seq, best_score = maybe_update(best_seq, best_score, a)

    # ------------------------------------------------------------------
    # 2. smooth‑max gradient descent on the current best (if any)
    # ------------------------------------------------------------------
    if best_seq is not None:
        n = best_seq.shape[0]
        tgt = math.sqrt(2.0 * n)
        best_seq, best_score = gradient_descent_optimize(
            best_seq, tgt, deadline, best_seq, best_score)

    # ------------------------------------------------------------------
    # 3. random restarts + gradient descent (tuned)
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # pick a moderate length (fast FFT) – 600 … 2000
        n = np.random.randint(600, 2001)
        target = math.sqrt(2.0 * n)

        # Choose a random seed shape:
        if np.random.rand() < 0.5:
            # random exponential‑like
            alpha = np.random.rand() * 0.02 + 1e-5
            a0 = shape_exponential(n, alpha)
        else:
            # random Dirichlet – more irregular
            a0 = np.random.dirichlet(np.ones(n)) * target

        a0 = project_onto_simplex(a0, target)
        best_seq, best_score = gradient_descent_optimize(
            a0, target, deadline, best_seq, best_score)

        if best_score < 1.503:      # early win, stop searching
            break

    # ------------------------------------------------------------------
    # 4. LP refinement (if SciPy is present)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        n = best_seq.shape[0]
        tgt = math.sqrt(2.0 * n)
        best_seq, best_score = lp_refinement(best_seq, tgt, deadline,
                                             best_seq, best_score)

    # ------------------------------------------------------------------
    # 5. fallback – uniform sequence (should never happen)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=float)

    # Return plain Python list (the harness expects this)
    return best_seq.tolist()
