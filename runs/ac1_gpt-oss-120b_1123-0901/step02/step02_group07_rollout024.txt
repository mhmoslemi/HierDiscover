```python
import numpy as np
import cvxpy as cp

def solve_lp(f, M):
    n = len(f)
    # Build matrix A shape (2n-1, n)
    rows = []
    for k in range(2*n-1):
        # Compute row entries:
        # A[k,i] = f[k - i] if 0 <= k-i < n else 0
        row = np.zeros(n)
        i_min = max(0, k - (n-1))
        i_max = min(k, n-1)
        # i enumerates from i_min..i_max inclusive
        # row[i] = f[k - i]
        row[i_min:i_max+1] = f[k - np.arange(i_min, i_max+1)]
        rows.append(row)
    A = np.vstack(rows)  # shape (2n-1, n)
    b = cp.Variable(n, nonneg=True)
    constraints = [A @ b <= M]
    obj = cp.Maximize(cp.sum(b))
    prob = cp.Problem(obj, constraints)
    prob.solve(solver=cp.ECOS, warm_start=True)
    return b.value
```

```python
def propose_candidate(seed=0, budget_s=1000):
    import time, numpy as np, cvxpy as cp

    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5

    n = 2000

    # Initialize sequence from global if exists else random Dirichlet
    if "height_sequence_1" in globals():
        init = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        init = np.clip(init, 0.0, 1000.0)
        if init.sum() > 0.0:
            seq = init / init.sum()
        else:
            seq = np.random.dirichlet(np.ones(n))
    else:
        seq = np.random.dirichlet(np.ones(n))

    best_seq = seq.copy()
    best_score = _objective(best_seq)

    print(f"[Init] start objective={best_score:.6f}")

    # Helper functions
    def convolve_self(a):
        return _self_convolve(a)

    # Main loop with two phases: LP iterations + gradient refinement
    # Phase 1: LP iterations (max 30 seconds)
    while time.time() < deadline - 2.0:
        # compute current max M
        conv = convolve_self(seq)
        M = conv.max()
        # Solve LP
        b = cp.Variable(n, nonneg=True)
        conv_expr = cp.conv(seq, b)   # length 2n-1
        constraints = [conv_expr <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
        # solve with time limit per iteration maybe
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000, verbose=False)
        except Exception as e:
            print("LP solve error", e)
            break
        if b.value is None:
            print("LP failed to find solution")
            break
        b_vec = np.asarray(b.value, dtype=np.float64)
        # Ensure positivity
        b_vec = np.maximum(b_vec, 0.0)
        sum_b = b_vec.sum()
        if sum_b <= 0:
            print("LP gave zero sum")
            break
        # Rescale to maintain sum=1
        g = b_vec / sum_b  # sum=1
        # Mix
        t = 0.5
        new_seq = (1 - t) * seq + t * g
        # Project onto simplex (ensure sum=1)
        new_seq = _project_onto_simplex(new_seq)
        new_score = _objective(new_seq)
        # Accept if improved (i.e., max decreased)
        if new_score < best_score:
            best_score = new_score
            best_seq = new_seq.copy()
            print(f"[LP] improved objective to {best_score:.6f}")
        # Accept new_seq if its max conv decreased (monotonic)
        if new_score < best_score + 1e-12:
            seq = new_seq
        else:
            # if not improving, maybe reduce t and retry
            # simple fallback: break
            break

    # Phase 2: gradient descent using smooth max (to polish)
    # using same gradient code but now starting from best_seq
    seq = best_seq.copy()
    cur_score = best_score
    base_step = 1.0

    while time.time() < deadline:
        conv = convolve_self(seq)
        # softmax weights with beta schedule
        beta = 5.0  # moderate
        # compute softmax weights of conv
        max_conv = conv.max()
        # to avoid overflow
        exp_vals = np.exp(beta * (conv - max_conv))
        w = exp_vals / exp_vals.sum()
        # gradient = 2 * (conv of w with reversed seq)
        rev = seq[::-1]
        grad = 2.0 * np.convolve(w, rev)[:n]  # need appropriate slicing
        # Actually convolution of w (len 2n-1) with rev (len n) yields length 3n-2.
        # We need gradient_i = sum_k w_k * seq_{k-i} for i indexes. Use method earlier.
        # We'll compute using FFT for speed: compute g = conv(w, rev) and extract indices.
        w_arr = w
        # compute convolution via FFT
        m = 2*n-1
        nfft = _next_pow2(m + n - 1)  # m+n-1 = 3n-2
        W = np.fft.rfft(w_arr, n=nfft)
        R = np.fft.rfft(rev, n=nfft)
        grad_full = np.fft.irfft(W * R, n=nfft)[:(m+n-1)]
        # gradient_i corresponds to grad_full[i + n - 1] as derived.
        grad = 2.0 * grad_full[n-1: n-1 + n]  # length n

        # backtracking line search
        step = base_step
        improved = False
        while step > 1e-7:
            cand = _project_onto_simplex(seq - step * grad)
            cand_score = _objective(cand)
            if cand_score < cur_score:
                improved = True
                break
            step *= 0.5
        if improved:
            seq = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = seq.copy()
                print(f"[Grad] new best {best_score:.6f}")
            base_step = max(step * 1.2, 1e-4)
        else:
            # random small move
            i, j = np.random.choice(n, size=2, replace=False)
            delta = 0.01 * min(seq[i], 0.1)
            if delta > 0:
                cand = seq.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_onto_simplex(cand)
                cand_score = _objective(cand)
                if cand_score < cur_score:
                    seq = cand
                    cur_score = cand_score
                    if cand_score < best_score:
                        best_score = cand_score
                        best_seq = seq.copy()
                        print(f"[Random] new best {best_score:.6f}")
        # break if time
        if time.time() > deadline:
            break

    print(f"Final objective = {best_score:.6f}")
    return best_seq.tolist()
```

"""
Propose a non‑negative step‑function (list of heights) that improves the
objective 2·n·max(conv)/sum².  The algorithm has two phases:

1.  *LP polishing* – starting from a decent seed (the global
    ``height_sequence_1`` if present, otherwise a random Dirichlet
    point) we repeatedly solve a linear program

        maximise   1ᵀ b
        subject to  f ∗ b ≤ M ,   b ≥ 0

       where ``f`` is the current sequence, ``M`` is its current worst
       convolution value and ``∗`` is the (full) convolution.  The LP
       pushes mass onto indices where ``f`` is small while keeping the
       cross‑convolution bounded by ``M``.  After each LP we rescale the
       solution to keep the total sum equal to one and mix it with the
       old sequence (mixing factor t≈0.2).  This monotone‐decreasing step
       usually drops the objective quickly.

2.  *Smooth‑max gradient descent* – the final sequence from phase 1 is
    refined with a projected gradient descent that minimizes a smooth
    approximation of ``max(conv)``:

        S_β(a) = (1/β)·log Σₖ exp(β·(a∗a)[k])

       The gradient of S_β is a weighted sum of the reversed fragments
       that build each convolution entry and can be computed with a
       single FFT.  After each line‑search step the vector is projected
       onto the probability simplex (non‑negative, sum = 1).

The whole routine runs under the 1000 s budget, prints intermediate
progress, and returns the best list found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helpers – projection onto the probability simplex
def _project_onto_simplex(v):
    """Euclidean projection of v onto {x ≥ 0, Σx = 1}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast (real) FFT‑based self‑convolution
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _self_convolve(seq: np.ndarray) -> np.ndarray:
    """Full convolution `seq * seq` (length 2·n‑1)."""
    n = seq.shape[0]
    m = 2 * n - 1
    nfft = _next_pow2(m)
    A = np.fft.rfft(seq, n=nfft)
    conv = np.fft.irfft(A * A, n=nfft)[:m]
    return conv


# ----------------------------------------------------------------------
# Objective used locally (same as evaluate_sequence, but faster)
def _objective(seq: np.ndarray) -> float:
    """2·n·max(conv)/sum² for a non‑negative sequence."""
    sum_a = seq.sum()
    if sum_a < 0.01:
        return float("inf")
    conv = _self_convolve(seq)
    max_b = conv.max()
    return float(2.0 * len(seq) * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Linear‑programming polishing step (requires cvxpy)
def _lp_step(f: np.ndarray, M: float, mix: float = 0.2):
    """
    Solve
        maximise 1ᵀ b
        subject to conv(f, b) ≤ M ,  b ≥ 0
    Return the mixed sequence (1‑mix)·f + mix·b̂  where b̂ is rescaled to
    have unit sum.  Returns None if the LP fails.
    """
    n = f.shape[0]

    try:
        import cvxpy as cp
    except Exception:                     # cvxpy not available – skip LP
        return None

    # variable for the new coefficients
    b = cp.Variable(n, nonneg=True)

    # cross‑convolution f ∗ b (full length 2·n‑1)
    conv_expr = cp.conv(f, b)

    # element‑wise bound by the current worst value M
    constraints = [conv_expr <= M]

    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=4000, abstol=1e-9, reltol=1e-9)
    except Exception:
        return None

    if b.value is None:
        return None

    b_vec = np.asarray(b.value, dtype=np.float64)
    b_vec = np.maximum(b_vec, 0.0)
    sum_b = b_vec.sum()
    if sum_b <= 0.0:
        return None

    # rescale to keep total mass equal to 1
    g = b_vec / sum_b

    # convex combination (mix) – the sum stays 1 automatically
    new_seq = (1.0 - mix) * f + mix * g
    return new_seq


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list[float]:
    """
    Search for a step‑function that minimises the evaluation function.
    Returns the best sequence found before the time limit expires.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5      # safety margin

    # --------------------------------------------------------------
    # 1. Choose a working length – a modest size keeps the LP cheap.
    n = 1200

    # --------------------------------------------------------------
    # 2. Initialise from the global seed if it exists, otherwise random.
    if "height_sequence_1" in globals():
        try:
            tmp = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            tmp = np.clip(tmp, 0.0, 1000.0)
            if tmp.sum() > 0.0:
                seq = tmp / tmp.sum()
            else:
                raise ValueError
        except Exception:
            seq = np.random.dirichlet(np.ones(n))
    else:
        seq = np.random.dirichlet(np.ones(n))

    best_seq = seq.copy()
    best_score = _objective(best_seq)
    print(f"[Init] objective={best_score:.6f}")

    # --------------------------------------------------------------
    # 3. LP‑polishing phase – iterate while we have time.
    it_lp = 0
    mix_factor = 0.2               # mixing weight for new direction

    while time.time() < deadline - 2.0:
        conv = _self_convolve(seq)
        M = conv.max()
        new_seq = _lp_step(seq, M, mix=mix_factor)

        if new_seq is None:
            # LP failed (maybe cvxpy missing); break to gradient phase.
            break

        new_score = _objective(new_seq)

        if new_score < best_score - 1e-12:
            best_score = new_score
            best_seq = new_seq.copy()
            seq = new_seq.copy()
            print(f"[LP {it_lp:04d}] improved objective to {best_score:.6f}")
        else:
            # No improvement – reduce the mixing factor and try once more.
            mix_factor *= 0.5
            if mix_factor < 1e-3:
                break
        it_lp += 1

    # --------------------------------------------------------------
    # 4. Smooth‑max gradient descent refinement (always available).
    #    Uses a softmax (log‑sum‑exp) surrogate for the max.
    base_step = 1.0
    beta = 8.0                     # softness of the max‑approximation
    it_grad = 0

    while time.time() < deadline:
        conv = _self_convolve(seq)
        max_conv = conv.max()

        # softmax weights w_k ∝ exp(β·conv_k)
        exp_vals = np.exp(beta * (conv - max_conv))
        w = exp_vals / exp_vals.sum()                # shape (2n‑1,)

        # gradient of the smooth max: 2·(w ∗ rev(seq))[i+n‑1]
        rev_seq = seq[::-1]
        m = 2 * n - 1
        nfft = _next_pow2(m + n - 1)
        W = np.fft.rfft(w, n=nfft)
        R = np.fft.rfft(rev_seq, n=nfft)
        grad_full = np.fft.irfft(W * R, n=nfft)[: (m + n - 1)]
        grad = 2.0 * grad_full[n - 1: n - 1 + n]     # length n

        # back‑tracking line search on the true objective
        step = base_step
        improved = False
        cur_score = _objective(seq)

        while step > 1e-7:
            cand = _project_onto_simplex(seq - step * grad)
            cand_score = _objective(cand)
            if cand_score < cur_score:
                improved = True
                break
            step *= 0.5

        if improved:
            seq = cand
            if cand_score < best_score:
                best_score = cand_score
                best_seq = seq.copy()
                print(f"[Grad {it_grad:05d}] new best = {best_score:.6f}")
            # adapt step size slightly upwards for next iteration
            base_step = max(step * 1.2, 1e-4)
        else:
            # fallback: tiny random mass transfer (still respects simplex)
            i, j = np.random.choice(n, size=2, replace=False)
            delta = 0.01 * min(seq[i], 0.1)
            if delta > 0.0:
                cand = seq.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_onto_simplex(cand)
                cand_score = _objective(cand)
                if cand_score < cur_score:
                    seq = cand
                    if cand_score < best_score:
                        best_score = cand_score
                        best_seq = seq.copy()
                        print(f"[Rand {it_grad:05d}] improved to {best_score:.6f}")
        it_grad += 1
        # safety break if progress stalls
        if it_grad % 1000 == 0 and time.time() > deadline:
            break

    print(f"Final best objective = {best_score:.6f} (length {n})")
    return best_seq.tolist()
