"""
Propose a non‑negative step‑function that minimizes
``evaluate_sequence``.

Algorithm
---------
* The objective after normalising the sum to √(2n) is simply the
  maximal entry of the linear convolution a*a.
* For each candidate length we first run a cheap (1+λ) Evolution‑Strategy
  on a low‑dimensional symmetric parametrisation (log‑heights at a few
  knots, linear interpolation) to explore the global landscape.
* The best candidate is then refined with a few hundred Adam‑updates
  on the exact sub‑gradient of the max‑convolution (the same optimiser
  used in the reference code) to reach a high‑precision local optimum.
* Several lengths (original length and a few larger interpolated
  versions) are tried; the best sequence among all runs is returned.
* All timing respects a 1000 s wall‑clock budget; progress is printed.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Fast full convolution via FFT – returns max value, its index and the array
def _conv_max_fft(a: np.ndarray):
    """Return (M, k_max, full_convolution) for a real vector a."""
    n = a.shape[0]
    N = 1 << ((2 * n - 1).bit_length())  # next power of two
    A = np.fft.rfft(a, n=N)
    B = A * A
    conv = np.fft.irfft(B, n=N)[: 2 * n - 1]
    k_max = int(conv.argmax())
    return float(conv.max()), k_max, conv


# ----------------------------------------------------------------------
# Sub‑gradient of the max‑convolution (average of reversed active shifts)
def _subgradient(a: np.ndarray, active_idxs: np.ndarray):
    """Compute the sub‑gradient of max_k (a*a)[k] for the active set."""
    n = a.shape[0]
    idx = np.arange(n)
    rev = np.zeros_like(a)
    for k in active_idxs:
        j = k - idx
        mask = (j >= 0) & (j < n)
        rev[mask] += a[j[mask]]
    rev /= max(1, active_idxs.size)
    return rev


# ----------------------------------------------------------------------
# Adam optimiser for a full‑length vector (identical to the reference
# optimiser but with a smaller iteration budget – it will be called only
# after the cheap global search).
def _adam_optimize_on_length(
    n: int,
    init_a: np.ndarray,
    best_seq: np.ndarray,
    best_score: float,
    deadline: float,
):
    """
    Run a short Adam optimisation starting from ``init_a``.
    ``best_seq``/``best_score`` are updated in‑place if an improvement is found.
    """
    target_sum = math.sqrt(2.0 * n)

    a = init_a.copy()
    a = a * (target_sum / a.sum())

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.06
    iteration = 0
    jitter_interval = 3000

    # we deliberately limit the number of iterations – the outer loop
    # already guarantees we have enough time for a handful of Adam runs.
    max_iter = 50000

    while time.time() < deadline - 2.0 and iteration < max_iter:
        M, k_max, conv_full = _conv_max_fft(a)

        # treat all entries virtually equal to the max as active
        tol = 1e-12
        active = np.where(conv_full >= M - tol)[0]
        rev = _subgradient(a, active)

        coeff = 4.0 * n / (target_sum * target_sum)
        const = 4.0 * n * M / (target_sum ** 3)
        grad = coeff * rev - const

        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        lr = lr0 / (1.0 + 1e-4 * iteration)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto feasible set
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a = a * (target_sum / s)
        else:
            a = np.full_like(a, target_sum / n)

        # occasional small jitter
        if iteration % jitter_interval == 0:
            jitter = np.random.randn(n) * 0.001 * target_sum / n
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        cur_score = M  # because sum = sqrt(2n) ⇒ 2·n·M/(sum²) = M
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[len {n:5d} Adam iter {iteration:6d}] new best = {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Low‑dimensional symmetric parametrisation based on log‑heights at knots
def _seq_from_params(y: np.ndarray, n: int, k: int) -> np.ndarray:
    """
    Build a non‑negative sequence of length n from the log‑heights y.
    The shape is symmetric, defined by k equally spaced knots.
    """
    left = np.exp(y)                # positive heights on the left half
    if k % 2 == 0:
        full = np.concatenate([left, left[::-1]])
    else:                           # odd number of knots – centre not duplicated
        full = np.concatenate([left, left[-2::-1]])
    nodes = np.linspace(0.0, 1.0, k)
    t = np.linspace(0.0, 1.0, n)
    seq = np.interp(t, nodes, full)
    seq = np.maximum(seq, 0.0)

    # normalise to the required total sum √(2n)
    target = math.sqrt(2.0 * n)
    s = seq.sum()
    if s > 0.0:
        seq = seq * (target / s)
    else:
        seq = np.full_like(seq, target / n)
    return seq


def _init_y_from_seq(seq: np.ndarray, n: int, k: int) -> np.ndarray:
    """
    Initialise the log‑parameter vector y from an existing full‑length sequence.
    """
    nodes = np.linspace(0.0, 1.0, k)
    t = np.linspace(0.0, 1.0, n)
    knot_heights = np.interp(nodes, t, seq)
    d = k // 2
    left = knot_heights[:d]
    left = np.maximum(left, 1e-12)
    return np.log(left)

            sigma = min(sigma * 1.3, 5.0)
        else:
            sigma = max(sigma * 0.85, 1e-6)

        iteration += 1

    # short Adam refinement (few thousand iterations) if time permits
    if time.time() < deadline - 3.0:
        local_best_seq, local_best_score = _adam_optimize_on_length(
            n=n,
            init_a=local_best_seq,
            best_seq=best_seq,
            best_score=best_score,
            deadline=deadline,
        )
        # propagate any improvement to the global record
        if local_best_score < best_score:
            best_score = local_best_score
            best_seq = local_best_seq.copy()
            print(f"[len {n:5d}] post‑ES Adam refined best = {best_score:.6f}")

    return best_seq, best_score

def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The function respects the time budget and returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin for the harness

    # ------------------------------------------------------------------
    # Load the existing construction if it exists
    try:
        base = np.array(height_sequence_1, dtype=float)  # type: ignore
    except Exception:
        base = np.ones(2100, dtype=float)

    n0 = base.shape[0]
    target0 = math.sqrt(2.0 * n0)
    base = base * (target0 / base.sum())
    init_M, _, _ = _conv_max_fft(base)
    best_seq = base.copy()
    best_score = init_M
    print(f"[init] length={n0}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Choose a few lengths to explore – original plus larger versions
    length_factors = [1.0, 1.5, 2.0, 2.5]
    lengths = sorted({int(n0 * f) for f in length_factors})
    # keep the size reasonable (FFT cost grows slowly, but we avoid huge arrays)
    lengths = [min(l, 8000) for l in lengths if l >= 300]

    # ------------------------------------------------------------------
    # Main optimisation loop over the selected lengths
    for n in lengths:
        if time.time() >= deadline:
            break

        if n == n0:
            init_seq = base.copy()
        else:
            # linear interpolation of the current best shape onto the new grid
            old_x = np.linspace(0.0, 1.0, n0)
            new_x = np.linspace(0.0, 1.0, n)
            init_seq = np.interp(new_x, old_x, base)
            init_seq = np.maximum(init_seq, 0.0)

        # normalise the interpolated sequence
        target_sum = math.sqrt(2.0 * n)
        if init_seq.sum() > 0.0:
            init_seq = init_seq * (target_sum / init_seq.sum())
        else:
            init_seq = np.full(n, target_sum / n)

        print(f"[run] starting ES/Adam for length = {n}")

        # ---- global (ES) + local (Adam) optimisation for this length ----
        best_seq, best_score = _es_optimize_length(
            n=n,
            init_seq=init_seq,
            best_seq=best_seq,
            best_score=best_score,
            deadline=deadline,
        )
        # ----------------------------------------------------------------

    # ------------------------------------------------------------------
    # Fallback – should never be needed
    if best_seq is None:
        n = 2100
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n)
        best_score = float(_conv_max_fft(best_seq)[0])
        print("[fallback] using uniform sequence")

    print(f"[done] best score = {best_score:.6f}, length = {best_seq.shape[0]}")
    # Return a plain Python list as required by the harness
    return best_seq.tolist()
