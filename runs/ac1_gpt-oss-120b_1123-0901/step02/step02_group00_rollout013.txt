"""
Propose a non‑negative step‑function (vector) that minimizes
``evaluate_sequence``.

Algorithm
---------
* Load the previously discovered best sequence (``height_sequence_1``) if
  available and use it as a seed.
* Select several candidate lengths (including the original length and a few
  larger interpolated versions up to 8000).  Longer vectors give more
  degrees of freedom and can lower the max‑convolution.
* For each length run a few random restarts.
  – **Phase 1 (smooth optimisation).**  minimise a smooth approximation of
    the max‑convolution:  ``smooth_max = (1/β)·log Σ exp(β·(a*a))``.
    The gradient of the smooth objective is computed analytically using
    FFTs (the weights are ``w_k ∝ exp(β·(a*a)_k)``).  An Adam optimiser
    updates the vector, after each step we project onto the feasible set
    (non‑negative, fixed ℓ₁‑sum) and enforce symmetry ``a = a[::-1]`` to
    help the shape.
  – The smoothing parameter ``β`` is increased gradually
    (2 → 5 → 10 → 20) so that the search first explores globally and then
    refines the true max‑objective.
* **Phase 2 (refinement).**  Starting from the best point of Phase 1 we
  run a few thousand iterations of the original “active‑set” Adam optimiser
  (max‑convolution gradient) to polish the solution.
* The best vector over all runs (real score = max‑convolution) is kept and
  finally returned as a plain list.  The code respects the 1000 s wall‑clock
  limit and prints progress.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _fft_len(min_len):
    """Small helper: next power of two >= min_len."""
    return 1 << (min_len - 1).bit_length()


def _conv_self(a, N=None):
    """
    Full convolution a * a (linear, not circular) using FFT.
    Returns an array of length 2*len(a)-1.
    """
    n = a.shape[0]
    if N is None:
        N = _fft_len(3 * n)          # enough for later gradient step as well
    A = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(A * A, n=N)[: 2 * n - 1]
    return conv


def _softmax_weights(conv, beta):
    """
    Given ``conv = a*a`` and temperature ``beta``, return the smoothing
    weights w_k = exp(beta*conv_k) / Σ exp(beta*conv).
    Also returns the smooth max value.
    """
    max_c = conv.max()
    # subtract max for numerical stability
    exp_vals = np.exp(beta * (conv - max_c))
    Z = exp_vals.sum()
    w = exp_vals / Z
    smooth_max = max_c + np.log(Z) / beta
    return w, smooth_max


def _grad_softmax(a, w, N):
    """
    Gradient of the smooth max (see docstring above).
    Implements  grad = 2 * Σ_j a_j w_{j+i} for each i.
    """
    n = a.shape[0]
    # reversed weight vector
    w_rev = w[::-1]
    A_fft = np.fft.rfft(a, n=N)
    W_rev_fft = np.fft.rfft(w_rev, n=N)
    conv = np.fft.irfft(A_fft * W_rev_fft, n=N)
    # indices corresponding to (2*n-2)-i
    idx = (2 * n - 2) - np.arange(n)
    grad = 2.0 * conv[idx]
    return grad


def _adam_softmax(
    a0,
    target_sum,
    beta_schedule,
    max_total_iters,
    deadline,
    best_seq,
    best_score,
    length,
):
    """
    Phase‑1 optimisation: minimise the smooth max using Adam.
    Returns possibly improved best_seq / best_score.
    """
    a = a0.copy()
    a = a * (target_sum / a.sum())            # enforce ℓ₁‑sum exactly
    n = length
    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.05
    iteration = 0
    # plan the schedule
    schedule = []
    total_iters = 0
    for beta, iters in beta_schedule:
        schedule.extend([(beta, iters)])
        total_iters += iters
    # truncate if too many iterations for the remaining time budget
    # (very rough: assume ~0.0005 s per iteration for n≈2000)
    max_iters = min(max_total_iters, total_iters)

    for beta, iters in schedule:
        for _ in range(iters):
            if iteration >= max_iters or time.time() >= deadline:
                return best_seq, best_score, iteration
            iteration += 1

            # --- convolution and weights ---
            N = _fft_len(3 * n)               # length for both convolutions
            conv = _conv_self(a, N=N)
            w, smooth_max = _softmax_weights(conv, beta)

            # --- gradient of smooth max ---
            grad = _grad_softmax(a, w, N)

            # --- Adam update ---
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** iteration)
            v_hat = v / (1.0 - beta2 ** iteration)

            # learning rate gets smaller with β (the objective is steeper)
            lr = lr0 / (1.0 + 1e-4 * iteration) / (beta ** 0.5)
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

            # --- projection onto feasible set ---
            a = np.maximum(a, 0.0)                     # non‑negative
            # enforce symmetry (helps flatten the autocorrelation)
            a = (a + a[::-1]) * 0.5
            s = a.sum()
            if s > 0.0:
                a = a * (target_sum / s)               # fix ℓ₁‑sum
            else:                                      # should never happen
                a = np.full_like(a, target_sum / n)

            # occasional tiny jitter to escape plateaus
            if iteration % 5000 == 0:
                jitter = np.random.randn(n) * 0.001 * target_sum / n
                a = np.maximum(a + jitter, 0.0)
                a = (a + a[::-1]) * 0.5
                a = a * (target_sum / a.sum())

            # --- evaluate (exact) ---
            cur_M = conv.max()
            cur_score = cur_M                                   # sum = target_sum ⇒ evaluate = max
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(
                    f"[soft len{n:5d} it{iteration:6d} β{beta:.1f}] "
                    f"new best = {best_score:.6f}"
                )
    return best_seq, best_score, iteration


def _adam_max_convolution(
    a0,
    target_sum,
    max_iters,
    deadline,
    best_seq,
    best_score,
    length,
):
    """
    Phase‑2 refinement: minimise the exact max‑convolution using the active‑set
    sub‑gradient (exact same update as the reference implementation).
    """
    a = a0.copy()
    a = a * (target_sum / a.sum())
    n = length
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.05
    iteration = 0
    jitter_interval = 5000

    N_conv = _fft_len(2 * n)            # only needed for a*a
    while iteration < max_iters and time.time() < deadline:
        iteration += 1
        # --- max‑convolution via FFT (fast) ---
        A = np.fft.rfft(a, n=N_conv)
        conv = np.fft.irfft(A * A, n=N_conv)[: 2 * n - 1]
        M = conv.max()
        # active set (indices where conv is close to the max)
        active = np.where(conv >= M - 1e-12)[0]
        # compute sub‑gradient: average reversed a over active shifts
        rev = np.zeros_like(a)
        idx = np.arange(n)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            rev[mask] += a[j[mask]]
        rev /= max(1, active.size)

        # gradient of the original objective (see comment in the prompt)
        coeff = 4.0 * n / (target_sum * target_sum)
        const = 4.0 * n * M / (target_sum ** 3)
        grad = coeff * rev - const

        # Adam step
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)
        lr = lr0 / (1.0 + 1e-4 * iteration)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection + symmetry
        a = np.maximum(a, 0.0)
        a = (a + a[::-1]) * 0.5
        s = a.sum()
        if s > 0.0:
            a = a * (target_sum / s)
        else:
            a = np.full_like(a, target_sum / n)

        # jitter
        if iteration % jitter_interval == 0:
            jitter = np.random.randn(n) * 0.001 * target_sum / n
            a = np.maximum(a + jitter, 0.0)
            a = (a + a[::-1]) * 0.5
            a = a * (target_sum / a.sum())

        # evaluate
        cur_score = M
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(
                f"[max  len{n:5d} it{iteration:6d}] new best = {best_score:.6f}"
            )
    return best_seq, best_score

def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for the best non‑negative height sequence within the time budget.
    Returns a plain Python list.
    """
    np.random.seed(seed)
    start = time.time()
    # keep a small safety margin
    deadline = start + budget_s - 5.0

    # ------------------------------------------------------------------
    # Load a known good construction if available, otherwise start from uniform
    try:
        base = np.array(height_sequence_1, dtype=float)  # type: ignore
    except Exception:
        base = np.ones(2100, dtype=float)

    # Global best tracker
    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Helper to evaluate quickly (max‑convolution) – uses the exact formula
    def _exact_score(vec):
        n = len(vec)
        conv = np.convolve(vec, vec)
        M = conv.max()
        # evaluate_sequence returns 2*n*M/(sum^2); we enforce sum≈sqrt(2n)
        return float(2 * n * M / (np.sum(vec) ** 2))

    # ------------------------------------------------------------------
    # Initialise from the loaded base (normalise it)
    n0 = base.shape[0]
    S0 = math.sqrt(2.0 * n0)
    base = base * (S0 / base.sum())
    init_score = _exact_score(base)
    best_seq = base.copy()
    best_score = init_score
    print(f"[init] length={n0}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Choose lengths to explore – include a few larger interpolations
    length_factors = [1.0, 1.25, 1.5, 2.0, 3.0, 4.0]
    lengths = sorted({int(n0 * f) for f in length_factors})
    # cap and filter
    lengths = [min(l, 8000) for l in lengths if 300 <= l <= 8000]

    # ------------------------------------------------------------------
    # Main optimisation loop
    for n in lengths:
        if time.time() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)
        # number of restarts for this length (more for larger n)
        n_restarts = 3 if n <= 3000 else 2

        for restart in range(n_restarts):
            if time.time() >= deadline:
                break

            # ---- initialise a starting vector for this run ----
            if n == n0 and restart == 0:
                init_a = base.copy()
            else:
                # linear interpolation from the base shape + jitter
                old_x = np.linspace(0.0, 1.0, n0)
                new_x = np.linspace(0.0, 1.0, n)
                init_a = np.interp(new_x, old_x, base)
                # add a small random perturbation before normalisation
                jitter = np.random.randn(n) * 0.01 * target_sum / n
                init_a = np.maximum(init_a + jitter, 0.0)

            # random Dirichlet component (adds diversity)
            if restart > 0:
                rnd = np.random.rand(n)
                init_a = rnd
                init_a = np.maximum(init_a, 0.0)

            init_a = init_a * (target_sum / init_a.sum())

            # --------------------------------------------------
            # Phase 1 – smooth optimisation (soft‑max)
            beta_sched = [(2.0, 2000), (5.0, 3000), (10.0, 4000), (20.0, 3000)]
            max_iters_phase1 = sum(st[1] for st in beta_sched)

            best_seq, best_score, _ = _adam_softmax(
                a0=init_a,
                target_sum=target_sum,
                beta_schedule=beta_sched,
                max_total_iters=max_iters_phase1,
                deadline=deadline,
                best_seq=best_seq,
                best_score=best_score,
                length=n,
            )

            # --------------------------------------------------
            # Phase 2 – refine with exact max‑objective (short run)
            # we restrict the number of refinement iterations to keep the total time reasonable
            max_refine_iters = 5000
            best_seq, best_score = _adam_max_convolution(
                a0=best_seq if best_seq.shape[0] == n else init_a,
                target_sum=target_sum,
                max_iters=max_refine_iters,
                deadline=deadline,
                best_seq=best_seq,
                best_score=best_score,
                length=n,
            )

            # print intermediate status
            cur_score = _exact_score(best_seq)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = best_seq.copy()
            print(
                f"[len {n:5d} restart {restart+1}/{n_restarts}] "
                f"current best = {best_score:.6f}"
            )

    # ------------------------------------------------------------------
    # Fallback if nothing was found (should not happen)
    if best_seq is None:
        n = 2100
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n)
        best_score = _exact_score(best_seq)
        print("[fallback] using uniform sequence")

    print(f"[done] best score = {best_score:.6f}, length = {best_seq.shape[0]}")
    # Return a plain Python list for the harness
    return best_seq.tolist()
