def _fft_conv(a):
    n = a.shape[0]

    A = np.fft.rfft(a, n=N)

def _crosscorr_fft(a, w):
    n = a.shape[0]
    m = w.shape[0]
    N = 1 << ((n + m - 1).bit_length())
    A = np.fft.rfft(a, n=N)

    # cross-correlation via conj: r = ifft(A * conj(W))
    corr = np.fft.irfft(A * np.conj(W), n=N)[:n]
    return corr

"""
Propose a step‑function (non‑negative vector) that minimizes the
``evaluate_sequence`` objective.

Algorithm – smooth maximum + Adam
---------------------------------
* The objective (after normalising the sum) is the peak of the
  self‑convolution  M = max_k (a*a)[k].  Direct optimisation of the
  non‑smooth max is difficult, so we replace it by a smooth
  approximation  Lγ = (1/γ)·log∑_k exp(γ·conv_k)  (log‑sum‑exp).
  As γ→∞ the smooth loss converges to the true maximum.
* For a given γ the loss gradient has a closed form:
      ∂Lγ/∂a_i = 2·∑_k w_k·a_{k−i},
  where w_k = exp(γ·conv_k)/∑_j exp(γ·conv_j) are the soft‑max
  weights.  The weighted sum is exactly a (linear) cross‑correlation
  between the current vector and the weight vector.
* Both the full convolution and the cross‑correlation are computed
  with a single FFT per vector (real‑valued RFFT), so each optimisation
  step costs only a few fast FFTs.
* Adam (adaptive moment estimation) is used on the raw gradient.
  After every update we project onto the feasible set:
      – non‑negative orthant,
      – total sum = √(2·n) (the normalisation required by the
        evaluation function).
* To avoid getting trapped we:
      – increase γ gradually (annealing),
      – add a tiny Gaussian jitter every few thousand steps,
      – run several random restarts and a few different lengths
        (the original length and larger interpolated versions).
* The best sequence found (according to the exact ``evaluate_sequence``
  value) is kept and returned when the time budget expires.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_full_fft(a: np.ndarray, N: int) -> np.ndarray:
    """
    Full linear convolution a * a using a pre‑allocated FFT size N.
    Returns an array of length 2*len(a)-1.
    """
    A = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(A * A, n=N)[: 2 * a.shape[0] - 1]
    return conv


def _crosscorr_fft(a: np.ndarray, w: np.ndarray, N: int) -> np.ndarray:
    """
    Linear cross‑correlation (a ⋆ w)[i] = Σ_j a_j·w_{i+j}
    for i = 0 … len(a)-1, using FFT size N.
    Returns an array of length len(a).
    """
    A = np.fft.rfft(a, n=N)
    W = np.fft.rfft(w, n=N)
    corr = np.fft.irfft(A * np.conj(W), n=N)[: a.shape[0]]
    return corr.real


# ----------------------------------------------------------------------
# Adam optimiser with smooth‑max gradient ------------------------------

def _optimize_smooth_max(
    n: int,
    init_a: np.ndarray,
    best_seq: np.ndarray,
    best_score: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation for a vector of length `n` starting from `init_a`.
    The loss is a smooth approximation of the maximal self‑convolution.
    Updates the global best sequence/score while respecting the wall‑time
    limit.
    """
    target_sum = math.sqrt(2.0 * n)

    # normalise the start vector
    a = init_a.copy()
    a = np.maximum(a, 0.0)
    a = a * (target_sum / a.sum())

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr0 = 0.07
    iteration = 0

    # smoothing parameter – start small, grow to a large value
    gamma = 0.3
    gamma_max = 15.0
    gamma_mul = 1.001

    # FFT size that works for both convolution and cross‑correlation:
    # we need at least 3·n‑2 points (n + (2n‑1) - 1).
    N_fft = _next_pow2(3 * n - 2)

    jitter_interval = 4000

    while time.time() < deadline:
        # --------------------------------------------------------------
        # 1) full self‑convolution
        conv = _conv_full_fft(a, N_fft)           # length 2·n‑1
        M = conv.max()                            # current peak (exact objective)

        # --------------------------------------------------------------
        # 2) smooth‑max weights
        #   w_k = exp(gamma·(conv_k−M)) / Σ exp(...)
        #   using M for numerical stability
        exp_vals = np.exp(gamma * (conv - M))
        sum_exp = exp_vals.sum()
        w = exp_vals / sum_exp                     # length 2·n‑1, sums to 1

        # --------------------------------------------------------------
        # 3) gradient = 2·(a ⋆ w)   (cross‑correlation)
        grad = 2.0 * _crosscorr_fft(a, w, N_fft)

        # --------------------------------------------------------------
        # 4) Adam update
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        # simple decay of the learning‑rate
        lr = lr0 / (1.0 + 1e-4 * iteration)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # --------------------------------------------------------------
        # 5) projection onto feasible set (non‑negative + fixed sum)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a = a * (target_sum / s)
        else:
            a = np.full_like(a, target_sum / n)

        # --------------------------------------------------------------
        # 6) (optional) enforce symmetry – the problem is invariant under
        #    reversal, a symmetric vector often gives a better bound.
        a = (a + a[::-1]) * 0.5
        a = a * (target_sum / a.sum())

        # --------------------------------------------------------------
        # 7) schedule the smoothing parameter
        gamma = min(gamma * gamma_mul, gamma_max)

        # --------------------------------------------------------------
        # 8) occasional jitter to escape shallow plateaus
        if iteration % jitter_interval == 0:
            jitter = np.random.randn(n) * 0.001 * target_sum / n
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        # --------------------------------------------------------------
        # 9) keep the best exact score seen so far
        #     (with the normalised sum the objective equals the peak M)
        cur_score = M
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[len {n:5d} iter {iteration:7d}] new best score = {best_score:.6f}")

        # --------------------------------------------------------------
        # stop condition is the outer time check
    return best_seq, best_score
