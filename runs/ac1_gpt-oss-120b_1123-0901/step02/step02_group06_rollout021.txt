"""
Propose a step‑function (non‑negative sequence) minimizing ``evaluate_sequence``.
The algorithm:
1. Generate a rich set of initial candidates (uniform, geometric,
   arcsine, and a grid of Beta‑shaped densities) and keep the best.
2. Perform a fast heavy‑ball sub‑gradient descent that uses the true
   sub‑gradient of the *maximum* convolution entry (both terms of the
   derivative are taken into account).  This quickly drives the objective
   below the previous record.
3. Switch to a smoothed Adam optimiser.  The L∞ norm is approximated by
   a soft‑max with an annealing parameter ``p``; the exact gradient of this
   smooth surrogate is computed by a linear correlation (FFT‑based) and a
   simple sliding‑window sum.  After each step the vector is projected onto
   the probability simplex (non‑negative, sum = 1).
4. The best sequence found before the time budget expires is returned.
"""

import time
import numpy as np

def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x ≥ 0, sum(x)=1}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:          # all entries non‑positive
        return np.full_like(v, 1.0 / len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def _objective(seq: np.ndarray) -> float:
    """Exact objective 2*n*max(conv)/sum(seq)^2 (no clipping needed)."""
    n = len(seq)
    conv = np.convolve(seq, seq)
    max_b = np.max(conv)
    sum_a = np.sum(seq)
    if sum_a < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (sum_a * sum_a))

def _beta_candidates(n: int, rng: np.random.Generator):
    """Generate Beta‑type densities for a grid of α,β > 0."""
    cand = []
    xs = (np.arange(n, dtype=np.float64) + 0.5) / n   # points in (0,1)
    alphas = np.concatenate([np.linspace(0.15, 0.5, 8),
                             np.linspace(0.6, 2.5, 7)])
    betas  = alphas  # start with symmetric
    for a in alphas:
        for b in betas:
            # unnormalised density x^{a-1} (1-x)^{b-1}
            vals = np.power(xs, a - 1.0) * np.power(1.0 - xs, b - 1.0)
            # protect against NaNs at the exact edges when a<1 or b<1
            vals = np.nan_to_num(vals, nan=0.0, posinf=0.0, neginf=0.0)
            if vals.sum() == 0:
                continue
            cand.append(vals / vals.sum())
    # add a few random skewed beta shapes
    for _ in range(5):
        a = rng.uniform(0.1, 3.0)
        b = rng.uniform(0.1, 3.0)
        vals = np.power(xs, a - 1.0) * np.power(1.0 - xs, b - 1.0)
        vals = np.nan_to_num(vals, nan=0.0, posinf=0.0, neginf=0.0)
        if vals.sum() > 0:
            cand.append(vals / vals.sum())
    return cand

def _initial_candidates(n: int, rng: np.random.Generator):
    """Generate a diverse pool of starting sequences (already normalised)."""
    cand = []

    # 1. uniform
    cand.append(np.full(n, 1.0 / n))

    # 2. geometric decays
    for factor in [0.5, 0.7, 0.8, 0.9, 1.0, 1.2, 1.5, 2.0]:
        eps = factor / n
        r = 1.0 - eps
        if r <= 0.0:
            continue
        seq = np.power(r, np.arange(n, dtype=np.float64))
        if seq.sum() > 0:
            cand.append(seq / seq.sum())

    # 3. arcsine shape (Beta(0.5,0.5))
    i = np.arange(n, dtype=np.float64)
    arcs = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    cand.append(arcs / arcs.sum())

    # 4. Beta‑family shapes
    cand.extend(_beta_candidates(n, rng))

    # 5. random Dirichlet draws (concentration 1)
    for _ in range(8):
        rand = rng.random(n)
        cand.append(rand / rand.sum())

    return cand

def _linear_correlation(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear (non‑circular) correlation r[i] = sum_t a[t] * b[i+t]
    for i = 0 … len(a)-1 .  Both a and b are real, non‑negative.
    Implemented via FFT to keep O(n log n).
    """
    n = a.shape[0]
    m = b.shape[0]
    L = 1
    while L < n + m - 1:
        L <<= 1
    A = np.fft.fft(a, L)
    B = np.fft.fft(b, L)
    r = np.fft.ifft(np.conj(A) * B).real
    # the first n entries correspond to i = 0 … n-1
    return r[:n]

    best_seq = None

        sc = _objective(seq)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
    print(f"[Init] best initial objective = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Stage‑1: fast heavy‑ball sub‑gradient descent on the true max.
    seq = best_seq.copy()
    step = 0.25                     # initial step size
    beta = 0.9                      # momentum factor
    momentum = np.zeros_like(seq)
    K = 12                          # number of top convolution entries examined
    it = 0
    no_improve = 0
    while time.time() < deadline - 30.0:   # reserve ~30 s for the second stage
        conv = np.convolve(seq, seq)
        # indices of K largest entries
        if K >= conv.size:
            top_idx = np.arange(conv.size)
        else:
            top_idx = np.argpartition(conv, -K)[-K:]

            idx = np.arange(lo, hi + 1)

            grad[idx] += seq[k - idx]
            # a_i term (appears once per selected shift)
            grad[idx] += seq[idx]

        momentum = beta * momentum + (1.0 - beta) * grad

        # gradient step and simplex projection
        new_seq = seq - step * momentum
        new_seq = _project_onto_simplex(new_seq)

            seq = new_seq
            no_improve = 0
            print(f"[Subgrad {it:06d}] stagnation – jitter applied")

    seq = best_seq.copy()         # restart from the best found so far

    v = np.zeros_like(seq)       # second‑moment estimate

"""
Propose a step‑function (non‑negative sequence) minimizing ``evaluate_sequence``.

Algorithm
---------
1. **Initial pool** – generate a diverse set of candidates:
   uniform, geometric progressions, the arcsine shape (Beta(0.5,0.5)),
   a grid of symmetric and skewed Beta(α,β) densities and a few random
   Dirichlet draws.  The best of those (according to the exact objective)
   becomes the starting point.

2. **Stage 1 – fast heavy‑ball sub‑gradient descent** on the true
   ‑∞‑norm of the self‑convolution.  For each of the `K` largest
   convolution entries we add the *full* sub‑gradient
   (both `a_{k‑i}` and `a_i` terms) and use a momentum update.
   This quickly brings the objective well below the previous record.

3. **Stage 2 – smooth Adam optimisation**.
   The ‑∞‑norm is approximated by a soft‑max with temperature `p`;
   an exact gradient of this smooth surrogate is obtained via a linear
   correlation (FFT‑based) plus a sliding‑window sum.  Adam with
   projection onto the probability simplex (sum = 1, non‑negative) is
   then applied while slowly increasing `p` so the surrogate tightens.
   The best sequence seen during the whole run is kept.

4. Return the best sequence found before the time budget expires.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x ≥ 0, sum(x)=1}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:                     # all entries non‑positive
        return np.full_like(v, 1.0 / len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq))/sum(seq)² (no clipping needed)."""
    n = len(seq)
    conv = np.convolve(seq, seq)
    max_b = np.max(conv)
    sum_a = np.sum(seq)
    if sum_a < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
def _beta_candidates(n: int, rng: np.random.Generator):
    """Generate a collection of discretised Beta(α,β) densities."""
    cand = []
    xs = (np.arange(n, dtype=np.float64) + 0.5) / n   # points in (0,1)

    # a modest grid covering heavily‑U‑shaped to bell‑shaped families
    alphas = np.concatenate([np.linspace(0.15, 0.5, 8),
                             np.linspace(0.6, 2.5, 7)])
    betas = alphas  # start with the symmetric case
    for a in alphas:
        for b in betas:
            vals = np.power(xs, a - 1.0) * np.power(1.0 - xs, b - 1.0)
            vals = np.nan_to_num(vals, nan=0.0, posinf=0.0, neginf=0.0)
            if vals.sum() == 0:
                continue
            cand.append(vals / vals.sum())

    # a few random skewed shapes
    for _ in range(5):
        a = rng.uniform(0.1, 3.0)
        b = rng.uniform(0.1, 3.0)
        vals = np.power(xs, a - 1.0) * np.power(1.0 - xs, b - 1.0)
        vals = np.nan_to_num(vals, nan=0.0, posinf=0.0, neginf=0.0)
        if vals.sum() > 0:
            cand.append(vals / vals.sum())
    return cand


# ----------------------------------------------------------------------
def _initial_candidates(n: int, rng: np.random.Generator):
    """Assemble a diverse pool of initial sequences (already normalised)."""
    cand = []

    # uniform
    cand.append(np.full(n, 1.0 / n))

    # geometric decays (different rates)
    for factor in [0.5, 0.7, 0.8, 0.9, 1.0, 1.2, 1.5, 2.0]:
        eps = factor / n
        r = 1.0 - eps
        if r <= 0.0:
            continue
        seq = np.power(r, np.arange(n, dtype=np.float64))
        cand.append(seq / seq.sum())

    # arcsine (Beta(½,½))
    i = np.arange(n, dtype=np.float64)
    arcs = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    cand.append(arcs / arcs.sum())

    # Beta family
    cand.extend(_beta_candidates(n, rng))

    # random Dirichlet draws (concentration 1)
    for _ in range(8):
        rand = rng.random(n)
        cand.append(rand / rand.sum())

    return cand


# ----------------------------------------------------------------------
def _linear_correlation(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear (non‑circular) correlation r[i] = Σ_t a[t] * b[i+t] for i=0…len(a)-1.
    Implemented via FFT → O(n log n).
    """
    n = a.shape[0]
    m = b.shape[0]
    L = 1
    while L < n + m - 1:
        L <<= 1
    A = np.fft.fft(a, L)
    B = np.fft.fft(b, L)
    r = np.fft.ifft(np.conj(A) * B).real
    return r[:n]                     # first n entries are the valid correlation


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimises
    ``evaluate_sequence`` within the allotted time.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5          # tiny safety margin

    # ------------------------------------------------------------------
    # 1. problem size
    n = 2000                                          # fast enough for many iterations

    # ------------------------------------------------------------------
    # 2. generate a pool of candidates and keep the best start point
    best_seq = None
    best_score = float("inf")
    for seq in _initial_candidates(n, rng):
        sc = _objective(seq)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
    print(f"[Init] best initial objective = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Stage 1 – heavy‑ball sub‑gradient descent on the true max
    seq = best_seq.copy()
    step = 0.25                       # initial step magnitude
    beta = 0.9                        # momentum factor
    momentum = np.zeros_like(seq)
    K = 12                            # number of top convolution entries examined
    it = 0
    no_improve = 0
    # reserve roughly 30 s for the second stage (or less if budget is tiny)
    reserve = min(30.0, max(5.0, budget_s * 0.05))
    while time.time() < deadline - reserve:
        conv = np.convolve(seq, seq)

        # indices of the K largest convolution values
        if K >= conv.size:
            top_idx = np.arange(conv.size)
        else:
            top_idx = np.argpartition(conv, -K)[-K:]

        # exact sub‑gradient for the chosen shifts (both terms a_{k-i} and a_i)
        grad = np.zeros_like(seq)
        for k in top_idx:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            grad[idx] += seq[k - idx]        # a_{k-i} part
            grad[idx] += seq[idx]            # a_i part

        grad = grad / K                     # normalise

        # heavy‑ball update
        momentum = beta * momentum + (1.0 - beta) * grad

        # gradient step + projection onto simplex
        new_seq = seq - step * momentum
        new_seq = _project_onto_simplex(new_seq)

        # cheap exact evaluation
        sc = _objective(new_seq)
        if sc < best_score:
            best_score = sc
            best_seq = new_seq.copy()
            print(f"[Subgrad {it:06d}] new best = {best_score:.6f}")
            no_improve = 0
        else:
            no_improve += 1

        # very slow decay – keeps the method exploratory
        step *= 0.9997

        # random jitter if we have been stuck for a while
        if no_improve > 5000:
            jitter = rng.normal(scale=0.01, size=n)
            new_seq = np.clip(new_seq + jitter, 0.0, None)
            new_seq = new_seq / new_seq.sum()
            momentum.fill(0.0)
            seq = new_seq
            no_improve = 0
            print(f"[Subgrad {it:06d}] stagnation – jitter applied")
            continue

        seq = new_seq
        it += 1

    # ------------------------------------------------------------------
    # 4. Stage 2 – smooth Adam optimisation (soft‑max surrogate of the max)
    seq = best_seq.copy()                     # restart from the best found
    m = np.zeros_like(seq)                    # first‑moment estimate
    v = np.zeros_like(seq)                    # second‑moment estimate
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    p = 5.0                                   # temperature of the soft‑max
    step_size = 0.06

        conv = np.convolve(seq, seq)
        max_val = conv.max()

        # soft‑max approximation of the ‑∞‑norm
        w = np.exp(p * (conv - max_val))      # shift for numerical stability
        s = w / w.sum()                       # shape (2n‑1,)

        # gradient of the smooth objective
        first = _linear_correlation(seq, s)                # Σ_t a[t]·s[i+t]
        slice_sum = np.convolve(s, np.ones(n), mode='valid')  # Σ_t s[i+t]
        grad = first + seq * slice_sum

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        seq = _project_onto_simplex(seq)

        sc = _objective(seq)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(f"[Adam {t:06d}] new best = {best_score:.6f}")

        # anneal temperature – makes the surrogate tighter
        if t % 250 == 0 and p < 30.0:
            p = min(30.0, p * 1.25)

        # gentle decay of the learning rate
        step_size *= 0.99995

    print(f"Finished. Best objective = {best_score:.6f} (length {n})")
    return best_seq.tolist()
