to improve on the previous best bound.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x ≥ 0, sum(x)=1}."""
    v = v.astype(np.float64, copy=True)
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        # all entries non‑positive – return the uniform distribution
        return np.full_like(v, 1.0 / v.shape[0])
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _max_conv_and_gradient(seq: np.ndarray):
    """
    Compute the self‑convolution of `seq`, the maximal entry,
    and a sub‑gradient of the maximal entry w.r.t. `seq`.

    Returns
    -------
    conv : np.ndarray
        Full convolution (length 2·n‑1).
    max_val : float
        Maximum of `conv`.
    grad : np.ndarray
        Sub‑gradient (same shape as `seq`).
    """
    n = seq.shape[0]
    conv = np.convolve(seq, seq)
    max_val = conv.max()
    # treat all entries that are numerically equal to the max as active
    eps = 1e-12
    active = np.where(conv >= max_val - eps)[0]

    grad = np.zeros_like(seq)

    # For each active shift k we add contributions a_{k-i} to grad[i]
    # and a_i to grad[k-i]; doing both simultaneously yields the correct
    # sub‑gradient (double contribution when i == k-i).
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=np.int64)
        j = k - i
        grad[i] += seq[j]
        grad[j] += seq[i]

    # Average if more than one shift is active (sub‑gradient of max)
    grad = grad / max(1, active.size)
    return conv, max_val, grad


# ----------------------------------------------------------------------
def _initial_candidates(n: int, rng: np.random.Generator):
    """Create a list of promising initial sequences (already normalised)."""
    cand = []

    # 1. uniform
    cand.append(np.full(n, 1.0 / n))

    # 2. arcsine shape (p = 0.5)
    i = np.arange(n, dtype=np.float64)
    arcs = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    cand.append(arcs / arcs.sum())

    # 3. one‑parameter family  a_i ∝ ((i+½)(n‑i‑½))^{-p}
    for p in [0.30, 0.40, 0.45, 0.55, 0.60, 0.70]:
        seq = ((i + 0.5) * (n - i - 0.5)) ** (-p)
        cand.append(seq / seq.sum())

    # 4. geometric progressions
    for r in [0.75, 0.85, 0.95, 1.05, 1.15, 1.25]:
        # generate a decaying / growing geometric sequence
        base = np.power(r, i)
        cand.append(base / base.sum())

    # 5. a few random Dirichlet draws
    for _ in range(8):
        rand = rng.random(n)
        cand.append(rand / rand.sum())

    # 6. optional global fallback
    global_seq = globals().get("height_sequence_1")
    if global_seq is not None:
        arr = np.asarray(global_seq, dtype=np.float64)
        if arr.size > 0:
            # normalise, then pad/trim to length n
            arr = np.maximum(arr, 0.0)
            if arr.sum() < 1e-12:   # avoid degenerate case
                arr = np.full_like(arr, 1.0 / arr.size)
            arr = arr / arr.sum()
            if arr.size < n:
                arr = np.concatenate([arr, np.zeros(n - arr.size)])
            elif arr.size > n:
                arr = arr[:n]
            cand.append(arr / arr.sum())

    return cand


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Return a list of non‑negative heights (a step function) that
    minimizes `evaluate_sequence` within the given time budget.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # 1. Problem size
    n = 3000                                 # modest size, fast convolutions

    # ------------------------------------------------------------------
    # 2. Build an initial pool and keep the best start
    best_seq = None
    best_score = float("inf")
    for seq in _initial_candidates(n, rng):
        score = evaluate_sequence(seq.tolist())
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
    print(f"[Init] best initial score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Exact projected sub‑gradient descent with momentum
    seq = best_seq.copy()
    step = 0.02               # initial learning rate
    beta = 0.90               # momentum coefficient
    momentum = np.zeros_like(seq)
    it = 0
    stagnation = 0

    while time.time() < deadline:
        # full convolution + exact max‑gradient
        conv, max_val, grad = _max_conv_and_gradient(seq)

        # momentum update (heavy‑ball)
        momentum = beta * momentum + (1.0 - beta) * grad

        # ----- additive projected gradient step -----
        new_seq = seq - step * momentum
        new_seq = np.maximum(new_seq, 0.0)                    # positivity
        new_seq = _project_onto_simplex(new_seq)            # sum = 1

        # ----- occasional exponential (multiplicative) step -----
        if it % 1200 == 0 and it != 0:
            exp_seq = seq * np.exp(-step * momentum)
            exp_seq = _project_onto_simplex(exp_seq)
            exp_score = evaluate_sequence(exp_seq.tolist())
            if exp_score < best_score:
                best_score = exp_score
                best_seq = exp_seq.copy()
                print(f"[Iter {it}] exp‑step improved best = {best_score:.6f}")
            # keep the better of additive vs exponential for the next iteration
            add_score = evaluate_sequence(new_seq.tolist())
            if exp_score < add_score:
                new_seq = exp_seq
                # reset momentum after a successful multiplicative move
                momentum[:] = 0.0

        # evaluate additive candidate
        score = evaluate_sequence(new_seq.tolist())
        if score < best_score:
            best_score = score
            best_seq = new_seq.copy()
            print(f"[Iter {it:06d}] new best = {best_score:.6f}")
            stagnation = 0
        else:
            stagnation += 1

        # stop early if target achieved
        if best_score <= 1.503:
            print("[Term] target reached, exiting early.")
            break

        # decay the step size very slowly
        step *= 0.9997

        # if we have not improved for a while, add a small jitter
        if stagnation > 6000:
            jitter = rng.normal(scale=0.008, size=n)
            new_seq = new_seq + jitter
            new_seq = np.clip(new_seq, 0.0, None)
            new_seq = _project_onto_simplex(new_seq)
            momentum[:] = 0.0
            stagnation = 0
            print(f"[Iter {it}] jitter applied to escape stagnation")

        seq = new_seq
        it += 1

    print(f"Finished after {it} iterations. Best objective = {best_score:.6f}")
    return best_seq.tolist()
```

This implementation builds a rich set of starting shapes (including a tunable power‑law family), then performs **exact** sub‑gradient descent on the max‑convolution with momentum, occasional multiplicative updates, and jitter‑based escape from plateaus.  
The algorithm respects all constraints of `evaluate_sequence` (non‑negative, sum≥0.01, values clipped to [0, 1000]) and runs within the 1000 s budget, aiming to beat the previous bound of 1.535603 and reach the target 1.503.
