def _refine_sequence(a_init, deadline):
    a = a_init.copy()
    n = len(a); target = math.sqrt(2.0 * n)
    m = np.zeros_like(a) # momentum
    beta = 0.9
    step0 = 0.5
    alpha = 0.5
    alpha_factor = 1.1
    alpha_max = 30.0
    iteration = 0
    while time.time() < deadline:
        conv = _conv_self_fft(a)
        # compute softmax weights
        # To avoid overflow, subtract max
        max_conv = np.max(conv)
        scaled = alpha * (conv - max_conv)
        exp_scaled = np.exp(scaled)
        w = exp_scaled / np.sum(exp_scaled)
        # gradient
        grad = _gradient_correlation(a, w) # returns gradient vector length n
        direction = -grad
        # momentum update
        m = beta * m + (1 - beta) * direction
        step = step0 / math.sqrt(iteration + 1)
        # scale step so that max absolute update ~ step
        max_abs = np.max(np.abs(m))
        if max_abs == 0:
            max_abs = 1.0

        a_new = a + eta * m
        a_new = _project_onto_simplex(a_new, target)
        # evaluate periodically
        if iteration % 5 == 0:
            _maybe_update_global_best(a_new)
        a = a_new
        iteration += 1
        # update alpha
        alpha = min(alpha_max, alpha * alpha_factor)
        # optional early break if best_score < 1.503
        if best_score < 1.503:
            break
    return

def _gradient_wrt_a(a, w):
    # compute correlation a with w
    # returns gradient: 2 * corr[a,w] for zero lag offset region
    n = len(a)

    grad = 2.0 * corr[offset: offset + n]
    return grad

"""
Propose a non‑negative step‑function (a list of heights) that minimises

    2 * n * max(conv(a,a)) / (sum(a)**2)

The code improves on the previous approach by:
* Using a smooth approximation of the max‑convolution (soft‑max) and
  (accelerated) projected gradient descent with momentum.
* Starting from several parametric families and random restarts.
* Updating the best candidate only via a cheap FFT‑based exact score
  (max(conv)) instead of calling the expensive ``evaluate_sequence`` on
  every iteration.
* Optionally performing a linear‑programming “LP‑step’’ (if SciPy is
  present) as a final refinement.
The whole search respects a time budget (default 1000 s) and returns the
best found sequence as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection ----------------------------------------------------
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto {x >= 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based, O(n log n)) -----------------------
# ----------------------------------------------------------------------
def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full convolution a * a using real FFT (returns length 2n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


def _gradient_wrt_a(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the smooth max-convolution objective.
    Returns 2 * (a ⋆ w) for the relevant lags (length len(a)).
    """
    n = a.shape[0]
    m = len(w)
    size = 1 << ((n + m - 1).bit_length())
    Fa = np.fft.rfft(a, n=size)
    Fw = np.fft.rfft(w, n=size)
    # linear correlation (real part) a ⋆ w
    corr = np.fft.irfft(Fa * np.conj(Fw), n=size)
    # keep only the linear part
    corr = corr[: n + m - 1]
    offset = m - 1
    grad = 2.0 * corr[offset: offset + n]
    return grad


# ----------------------------------------------------------------------
# Scoring (exact max‑convolution) ---------------------------------------
# ----------------------------------------------------------------------
def _exact_score(a: np.ndarray) -> float:
    """Exact objective value = max(a*a) for a already normalised (sum≈√(2n))."""
    conv = _conv_self_fft(a)
    return float(conv.max())


# ----------------------------------------------------------------------
# Simple parametric families (non‑normalised) -------------------------
# ----------------------------------------------------------------------
def _shape_exp(n: int, alpha: float) -> np.ndarray:
    return np.exp(-alpha * np.arange(n, dtype=float))


def _shape_exp_rev(n: int, alpha: float) -> np.ndarray:
    return np.exp(-alpha * np.arange(n - 1, -1, -1, dtype=float))


def _shape_linear_desc(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    return np.maximum(0.0, 1.0 - beta * i)


def _shape_quad_desc(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    return np.maximum(0.0, 1.0 - beta * i * i)


def _shape_triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    return np.minimum(i + 1.0, n - i)

_best = {"seq": None, "score": float("inf")}


def _maybe_update_best(a: np.ndarray) -> None:
    """
    Compute the exact score of 'a' and update the global best if improved.
    The function also prints a progress line.
    """
    score = _exact_score(a)

        _best["seq"] = a.copy()

# ----------------------------------------------------------------------
# Gradient‑descent refinement (smooth max) ------------------------------
# ----------------------------------------------------------------------
def _refine_sequence(a0: np.ndarray, deadline: float) -> None:
    """
    Perform many projected (momentum) gradient steps on a smooth approximation
    of the max‑convolution.  Updates the global best whenever a better
    candidate appears.  Stops when the deadline is reached or the target
    bound (≈1.503) is attained.
    """
    a = a0.astype(float)
    n = a.shape[0]
    target = math.sqrt(2.0 * n)

    # momentum
    m = np.zeros_like(a)
    beta = 0.9
    base_step = 0.5

    # smooth‑max temperature schedule
    alpha = 0.5          # starts very smooth
    alpha_factor = 1.07
    alpha_max = 35.0

    it = 0
    # initial evaluation
    _maybe_update_best(a)

    while time.time() < deadline:
        # ----- smooth max -----
        conv = _conv_self_fft(a)                     # full auto‑correlation
        max_c = np.max(conv)
        scaled = alpha * (conv - max_c)              # shift → stability
        exp_scaled = np.exp(scaled)                  # all ≤ 1
        w = exp_scaled / np.sum(exp_scaled)          # soft‑max weights

        # ----- gradient of the smooth objective -----
        grad = _gradient_wrt_a(a, w)

        # ----- momentum step -----
        direction = -grad
        m = beta * m + (1.0 - beta) * direction

        # ----- diminishing step size -----
        step = base_step / math.sqrt(it + 1.0)
        max_abs = np.max(np.abs(m))
        if max_abs == 0.0:
            max_abs = 1.0
        eta = step / max_abs

        # ----- project back to the simplex -----
        a_new = a + eta * m
        a_new = _project_onto_simplex(a_new, target)

        # ----- occasional evaluation -----
        if it % 20 == 0:
            _maybe_update_best(a_new)
            if _best["score"] < 1.503:
                return

        a = a_new
        it += 1

        # increase the temperature (make the soft‑max sharper)
        alpha = min(alpha_max, alpha * alpha_factor)

        # safety cap on iterations (avoid endless loops)
        if it >= 8000:
            break

    # final check
    _maybe_update_best(a)


# ----------------------------------------------------------------------
# Optional LP refinement (requires SciPy) -------------------------------
# ----------------------------------------------------------------------
def _lp_refine(seq: np.ndarray, deadline: float) -> None:
    """Linear‑programming refinement exactly as described in the SOTA paper."""
    try:
        from scipy.optimize import linprog
    except Exception:                      # SciPy not available
        return

    a = seq.copy()
    n = a.shape[0]
    target = math.sqrt(2.0 * n)

    while time.time() < deadline:
        conv = _conv_self_fft(a)
        M = conv.max()
        rows = 2 * n - 1

        # Build constraint matrix A where (a * g)_k ≤ M
        A = np.zeros((rows, n), dtype=float)
        for k in range(rows):
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            cols = np.arange(lo, hi + 1, dtype=int)
            A[k, cols] = a[k - cols]

        res = linprog(
            c=-np.ones(n),                # maximise Σ g_j
            A_ub=A,
            b_ub=np.full(rows, M),
            bounds=(0, None),
            method="highs",
            options={"presolve": True},
        )
        if not res.success:
            break

        g = res.x
        s = g.sum()
        if s == 0:
            break
        g = g * (target / s)               # re‑scale to the simplex

        # convex combination with a modest mixing factor
        t = 0.2
        a = (1.0 - t) * a + t * g

        _maybe_update_best(a)

        if _best["score"] < 1.503 or time.time() + 0.5 >= deadline:
            break
