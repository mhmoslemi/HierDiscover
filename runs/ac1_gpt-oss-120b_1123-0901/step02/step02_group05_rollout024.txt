"""
Propose a step‑function (non‑negative heights) with a low value of

        2 * n * max(conv(a, a)) / (sum(a) ** 2)

The optimizer combines several ideas that were not used together before:

*   **AdaGrad/Adam‑style adaptive steps** on the sub‑gradient of the maximal
    convolution entry – this converges faster than the plain fixed‑step
    projected gradient descent used previously.
*   **Geometric initialisations** (a_i ∝ r^i) – the analytic model
    (a_i = C·r^i) is a good starting point because each convolution term
    a_i·a_{k‑i} = C²·r^k, i.e. the dependence on i disappears.
*   **Multi‑scale search** – we start from short sequences (200–500 points),
    optimise them, then repeatedly resize (linear interpolation) to a larger
    length while re‑optimising.  This gives the algorithm many degrees of
    freedom without paying the full price of a high‑dimensional search from
    scratch.
*   **Random restarts & mutations** – occasional Gaussian perturbations
    and completely new random seeds help escape local minima.
*   **Optional LP refinement** (if cvxpy is present) – a cheap LP update is
    applied a few times after the Adam phase.

The whole routine respects the 1000 s budget, always returns a feasible
sequence and prints progress information.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simple Euclidean projection onto the simplex {x ≥ 0, Σ x = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex of non‑negative vectors with sum s."""
    n = v.shape[0]
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast full convolution (FFT for large n)
# ----------------------------------------------------------------------
def _full_convolve(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    if n <= 1024:
        return np.convolve(a, a)
    size = 2 * n - 1
    # next power of two for FFT speed
    N = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, n=N)
    B = A * A
    conv = np.fft.irfft(B, n=N)[:size]
    # numerical noise can produce tiny negative values – clip
    return np.clip(conv.real, 0.0, None)


# ----------------------------------------------------------------------
# Sub‑gradient of the maximal convolution entry
# ----------------------------------------------------------------------
def _max_conv_and_grad(a: np.ndarray):
    """
    Compute max(conv(a, a)), its index and the sub‑gradient of that entry.
    Returns (max_val, grad, max_idx, full_conv).
    """
    n = a.shape[0]
    conv = _full_convolve(a)
    max_idx = int(np.argmax(conv))
    max_val = conv[max_idx]

    # gradient of the entry at max_idx:
    low = max(0, max_idx - (n - 1))
    high = min(n - 1, max_idx)
    idx = np.arange(low, high + 1, dtype=int)
    partner = max_idx - idx
    grad = np.zeros_like(a)
    grad[idx] = 2.0 * a[partner]
    return max_val, grad, max_idx, conv


# ----------------------------------------------------------------------
# Simple mutation helper (Gaussian noise + projection)
# ----------------------------------------------------------------------
def _mutate(a: np.ndarray, target_sum: float, sigma: float) -> np.ndarray:
    """Return a mutated copy of a (Gaussian noise, then simplex projection)."""
    b = a + np.random.randn(a.size) * sigma
    b = np.maximum(b, 0.0)
    b = _project_onto_simplex(b, target_sum)
    return b


# ----------------------------------------------------------------------
# Safe wrapper around the provided evaluate_sequence()
# ----------------------------------------------------------------------
def _evaluate(seq: np.ndarray) -> float:
    try:
        return float(evaluate_sequence(seq.tolist()))
    except Exception:
        return math.inf


# ----------------------------------------------------------------------
# Adam‑style projected optimisation on the sub‑gradient of max(conv)
# ----------------------------------------------------------------------
def _adam_descent(
    init: np.ndarray,
    target_sum: float,
    time_limit: float,
    lr: float = 0.05,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    tag: str = "",
) -> tuple[np.ndarray, float]:
    """
    Perform adaptive‑gradient descent (Adam) on the sub‑gradient of the
    maximal convolution entry, with Euclidean projection onto the simplex
    after each step.  Returns the best sequence seen and its score.
    """
    a = _project_onto_simplex(init.astype(float), target_sum)

    best_score = _evaluate(a)
    best_seq = a.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    start = time.time()
    it = 0
    while time.time() < start + time_limit:
        it += 1
        # sub‑gradient of the current maximum
        max_val, grad, _, _ = _max_conv_and_grad(a)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = -lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a + step
        a_new = _project_onto_simplex(a_new, target_sum)

        # occasional random shake‑up
        if np.random.rand() < 0.02:
            sigma = 0.02 * target_sum / a.size
            a_new = _mutate(a_new, target_sum, sigma)

        score = _evaluate(a_new)
        if score < best_score:
            best_score = score
            best_seq = a_new.copy()
            print(
                f"[{time.time():.2f}] {tag} new best = {score:.6f} (n={a.size})"
            )

        a = a_new

        # simple learning‑rate decay
        if it % 500 == 0:
            lr *= 0.9

    return best_seq, best_score

# Optional LP refinement (if cvxpy is available)
# ----------------------------------------------------------------------
try:
    import cvxpy as cp

    _CVX_AVAILABLE = True
except Exception:  # pragma: no cover
    _CVX_AVAILABLE = False


def _solve_lp_for_g(f: np.ndarray, M: float):
    """Solve the LP: maximise Σ g_j subject to g ≥ 0 and f*g ≤ M."""
    if not _CVX_AVAILABLE:
        return None
    n = f.shape[0]
    g = cp.Variable(n, nonneg=True)

    constraints = []
    for k in range(2 * n - 1):
        low = max(0, k - (n - 1))
        high = min(n - 1, k)
        idx = np.arange(low, high + 1, dtype=int)
        partner = k - idx
        expr = cp.sum(cp.multiply(f[low : high + 1], g[partner]))
        constraints.append(expr <= M)

    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.GLPK, warm_start=True, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, verbose=False, max_iters=2000)
        except Exception:
            return None

    if prob.status not in ("optimal", "optimal_inaccurate"):
        return None
    g_val = np.asarray(g.value).flatten()
    if g_val is None:
        return None
    return np.maximum(g_val, 0.0)


def _lp_refine(
    init: np.ndarray,
    target_sum: float,
    time_limit: float,
    mix: float = 0.2,
    max_iter: int = 4,
) -> tuple[np.ndarray, float]:
    """Apply a few LP‑updates starting from ``init``."""
    a = _project_onto_simplex(init.astype(float), target_sum)
    best_score = _evaluate(a)
    best_seq = a.copy()
    start = time.time()
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        M = np.max(_full_convolve(a))
        g0 = _solve_lp_for_g(a, float(M))
        if g0 is None or np.sum(g0) == 0:
            break
        scale = target_sum / np.sum(g0)
        g = g0 * scale
        a = (1 - mix) * a + mix * g
        a = _project_onto_simplex(a, target_sum)
        sc = _evaluate(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(
                f"[{time.time():.2f}] LP‑refine iter {it+1} new best = {sc:.6f} (n={a.size})"
            )
    return best_seq, best_score


# ----------------------------------------------------------------------
# Resize helper (linear interpolation / down‑sampling)
# ----------------------------------------------------------------------
def _resize_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize seq to length new_n preserving non‑negativity (linear interpolation)."""
    old_n = seq.size
    if old_n == new_n:
        return seq.copy()
    if old_n > new_n:
        # pick equally spaced indices (fast down‑sampling)
        idx = np.linspace(0, old_n - 1, new_n).astype(int)
        return seq[idx]
    else:
        # linear interpolation to upscale
        x_old = np.arange(old_n)
        x_new = np.linspace(0, old_n - 1, new_n)
        return np.interp(x_new, x_old, seq)
