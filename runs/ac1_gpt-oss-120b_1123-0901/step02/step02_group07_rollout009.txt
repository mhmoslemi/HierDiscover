"""
Propose a step‑function (non‑negative sequence) that improves the
evaluation score below the previous best 1.5449.  The algorithm works in
two stages:

1. **Broad seeding** – generate a few hundred candidate sequences from
   simple parametric families (geometric, power‑law, beta‑like,
   random Dirichlet).  The best of these seeds the subsequent optimisation.

2. **Smoothed max‑gradient descent** – at each iteration the full
   self‑convolution is computed, a soft‑max weighting of the convolution
   entries is built and the corresponding (sub‑)gradient is obtained via
   a second FFT (the convolution of the weight vector with the reversed
   sequence).  A back‑tracking line‑search together with a projection onto
   the probability simplex yields a new feasible candidate.  When a line
   search fails a tiny random mass‑transfer move is tried; after many
   consecutive failures the current iterate is re‑initialised from the
   best solution with a small Gaussian perturbation.

The whole loop runs until the 1000 s budget is exhausted (a small safety
margin is kept).  The best sequence found is returned as a plain
Python list.  The code uses only ``numpy`` and ``time`` and respects all
security checks of ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex {x ≥ 0, sum(x)=1}
def _project_onto_simplex(v):
    """Project vector v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        # all entries are zero – give uniform distribution
        return np.full_like(v, 1.0 / len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


# ----------------------------------------------------------------------
def _self_convolve(seq: np.ndarray) -> np.ndarray:
    """Full convolution of seq with itself (length 2·n‑1)."""
    n = seq.shape[0]
    m = 2 * n - 1
    nfft = _next_pow2(m)
    A = np.fft.rfft(seq, n=nfft)
    conv = np.fft.irfft(A * A, n=nfft)[:m]
    return conv


# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """Exact objective used by evaluate_sequence (fast version)."""
    sum_a = seq.sum()
    if sum_a < 0.01:
        return float("inf")
    conv = _self_convolve(seq)
    max_b = conv.max()
    return float(2.0 * len(seq) * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Parametric families (used for seeding)
def _geom_seq(n: int, alpha: float) -> np.ndarray | None:
    """Geometric progression a_i ∝ r^i with r = 1 - alpha/n."""
    eps = alpha / n
    r = 1.0 - eps
    if r <= 0.0:
        return None
    powers = r ** np.arange(n, dtype=np.float64)
    seq = powers / powers.sum()
    return seq


def _powerlaw_seq(n: int, p: float) -> np.ndarray:
    """Power‑law a_i ∝ (i+1)^{-p}."""
    seq = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    return seq / seq.sum()


def _beta_seq(n: int, a: float, b: float) -> np.ndarray:
    """Beta‑like shape a_i ∝ (i+1)^{a-1} (n-i)^{b-1}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    seq = (i ** (a - 1.0)) * ((n + 1 - i) ** (b - 1.0))
    seq[seq < 0] = 0.0
    return seq / seq.sum()


def _random_simplex(n: int) -> np.ndarray:
    """Draw a random point on the simplex (Dirichlet(1,…,1))."""
    x = np.random.exponential(1.0, size=n)
    return x / x.sum()


# ----------------------------------------------------------------------
def _grad_from_weights(seq: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute the gradient of the smooth‑max objective:
    grad_i = sum_k w_k * a_{k-i}  (i.e. convolution of w with the reversed seq).
    """
    n = seq.shape[0]
    a_rev = seq[::-1]                     # reversed sequence
    L = _next_pow2((2 * n - 1) + (n - 1))  # length needed for convolution
    W = np.fft.rfft(w, n=L)
    A = np.fft.rfft(a_rev, n=L)
    grad_full = np.fft.irfft(W * A, n=L)
    start = n - 1
    grad = grad_full[start:start + n]
    # factor 2 from the true derivative is irrelevant for direction
    return grad


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Build and locally optimise a non‑negative sequence to minimise
    evaluate_sequence(...).  Returns the best list found within the time
    budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # choose a length that balances expressiveness and speed
    n = 4000

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0.  Global seed (if supplied by the runner)
    if "height_sequence_1" in globals():
        try:
            init = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            init = np.clip(init, 0.0, 1000.0)
            if init.sum() > 0.0:
                init = init / init.sum()
                score = _objective(init)
                if score < best_score:
                    best_score = score
                    best_seq = init.copy()
                    print(f"[Init] global seed score = {score:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1.  Systematic parametric sweep
    def _maybe_update(seq):
        nonlocal best_seq, best_score
        sc = _objective(seq)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(f"[Init] new best seed score = {sc:.6f}")

    # geometric family
    for alpha in np.linspace(0.5, 3.0, 12):
        if time.time() > deadline:
            break
        seq = _geom_seq(n, alpha)
        if seq is not None:
            _maybe_update(seq)

    # power‑law family
    for p in np.linspace(0.2, 2.5, 12):
        if time.time() > deadline:
            break
        seq = _powerlaw_seq(n, p)
        _maybe_update(seq)

    # beta‑like family
    for a in np.linspace(0.5, 4.0, 4):
        for b in np.linspace(0.5, 4.0, 4):
            if time.time() > deadline:
                break
            seq = _beta_seq(n, a, b)
            _maybe_update(seq)

    # random Dirichlet points
    for _ in range(30):
        if time.time() > deadline:
            break
        seq = _random_simplex(n)
        _maybe_update(seq)

    # Fallback to uniform if nothing was generated
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)
        best_score = _objective(best_seq)

    # ------------------------------------------------------------------
    # 2.  Smoothed max‑gradient descent
    seq = best_seq.copy()
    cur_score = best_score
    it = 0
    base_step = 5.0
    no_improve = 0

    while time.time() < deadline:
        # full convolution and current max
        conv = _self_convolve(seq)
        max_conv = conv.max()

        # sanity – compute current objective directly (sum is ~1)
        cur_score = 2.0 * n * max_conv / (seq.sum() ** 2)

        # ----------- smooth weighting (soft‑max) ----------
        diff = max_conv - np.median(conv)
        eps = max(diff * 0.1, max_conv * 0.001, 1e-12)
        # stabilised exponent to avoid overflow
        w_raw = np.exp((conv - max_conv) / eps)
        w = w_raw / w_raw.sum()                     # soft‑max weights

        # ----------- gradient from weights ----------
        grad = _grad_from_weights(seq, w)

        # ----------- back‑tracking line search ----------
        step = base_step
        improved = False
        while step > 1e-7:
            cand = _project_onto_simplex(seq - step * grad)
            cand_score = _objective(cand)
            if cand_score < cur_score - 1e-12:
                improved = True
                break
            step *= 0.5

        if improved:
            seq = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = seq.copy()
                print(f"[Iter {it:06d}] new best score = {best_score:.6f}")
            # gently increase step size for next iteration
            base_step = max(step * 1.2, 1e-4)
            no_improve = 0
        else:
            # ---- random tiny mass‑transfer move ----
            i, j = np.random.choice(n, size=2, replace=False)
            delta = 0.01 * min(seq[i], 0.1)
            if delta > 0:
                cand = seq.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_onto_simplex(cand)
                cand_score = _objective(cand)
                if cand_score < cur_score:
                    seq = cand
                    cur_score = cand_score
                    if cand_score < best_score:
                        best_score = cand_score
                        best_seq = seq.copy()
                        print(f"[Iter {it:06d}] random move improved to {best_score:.6f}")
            no_improve += 1

            # occasional re‑initialisation from the best solution with noise
            if no_improve > 10000 and time.time() < deadline - 5.0:
                noise = np.random.normal(scale=0.02, size=n)
                cand = best_seq + noise
                cand = np.clip(cand, 0.0, None)
                cand = _project_onto_simplex(cand)
                cand_score = _objective(cand)
                if cand_score < cur_score:
                    seq = cand
                    cur_score = cand_score
                # shrink step size to become more cautious
                base_step = max(base_step * 0.5, 1e-4)
                no_improve = 0

        it += 1

    print(f"Final best objective = {best_score:.6f} (length {n})")
    return best_seq.tolist()
