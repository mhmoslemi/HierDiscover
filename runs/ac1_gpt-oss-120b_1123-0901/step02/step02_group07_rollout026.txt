"""
Propose a step function using a Frank‑Wolfe (LP‑based) refinement.
...
"""
import time
import numpy as np
from scipy.optimize import linprog

def _project_onto_simplex(v):
    """Project onto the probability simplex {x >= 0, sum(x)=1}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def _next_pow2(x: int) -> int:
    return 1 << (x - 1).bit_length()

def _self_convolve(seq: np.ndarray) -> np.ndarray:
    n = seq.shape[0]
    m = 2 * n - 1
    nfft = _next_pow2(m)
    A = np.fft.rfft(seq, n=nfft)
    conv = np.fft.irfft(A * A, n=nfft)[:m]
    return conv

def _objective(seq: np.ndarray) -> float:
    sum_a = seq.sum()
    if sum_a < 0.01:
        return float('inf')
    conv = _self_convolve(seq)
    max_b = conv.max()
    return float(2.0 * len(seq) * max_b / (sum_a * sum_a))

def _build_constraint_matrix(f: np.ndarray) -> np.ndarray:

Propose a step‑function (non‑negative sequence) that minimises the
``evaluate_sequence`` objective.

The algorithm works in three phases:

1️⃣ **Broad seed generation** – several parametric families (geometric,
   power‑law, beta‑like, random Dirichlet) are swept; the best seed is kept.

2️⃣ **Frank‑Wolfe–style LP refinement** – for the current sequence `f`
   (normalised to sum = 1) we compute the current bound `M = max(f*f)`.
   We then solve a linear programme

        maximise   1ᵀg
        subject to  f * g ≤ M   (component‑wise)
                    g ≥ 0

   which yields a direction `g` that pushes mass away from the active
   convolution constraints.  A short line‑search on the mixing coefficient
   `t ∈ [0,1]` updates `f ← (1‑t)f + t g`.  The process is repeated a few
   times – each iteration provably does not increase the objective.

3️⃣ **Tiny random hill‑climb** – after the LP phase we perform many small
   random mass‑transfer moves (projected back onto the probability simplex)
   and keep any improvement.

All operations stay well within the 1000 s budget; the routine returns the
best list found.  The code is fully self‑contained and uses only NumPy and
SciPy (the latter for the linear programme).
"""

import time

# ----------------------------------------------------------------------
def _project_onto_simplex(v):
    """Project a vector onto the probability simplex {x ≥ 0, sum(x)=1}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for zero‑padding FFT)."""
    return 1 << (x - 1).bit_length()


def _self_convolve(seq: np.ndarray) -> np.ndarray:
    """Full convolution of `seq` with itself (length 2 n‑1) via FFT."""
    n = seq.shape[0]
    m = 2 * n - 1
    nfft = _next_pow2(m)
    A = np.fft.rfft(seq, n=nfft)
    conv = np.fft.irfft(A * A, n=nfft)[:m]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective used by the competition (scale‑invariant)."""
    sum_a = seq.sum()
    if sum_a < 0.01:
        return float("inf")
    conv = _self_convolve(seq)
    max_b = conv.max()
    return float(2.0 * len(seq) * max_b / (sum_a * sum_a))

    A = np.zeros((rows, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = f[i]
    return A
