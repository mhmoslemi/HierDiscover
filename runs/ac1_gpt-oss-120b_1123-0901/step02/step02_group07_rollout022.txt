def _softmax_gradient(seq: np.ndarray, beta: float) -> np.ndarray:
    conv = _self_convolve(seq)
    # compute stable softmax
    maxc = conv.max()
    w = np.exp(beta * (conv - maxc))
    w_sum = w.sum()
    w /= w_sum
    w_rev = w[::-1]
    # compute correlation using FFT
    n = seq.shape[0]
    m = w_rev.shape[0]  # 2*n-1
    nfft = _next_pow2(n + m - 1)
    A = np.fft.rfft(seq, n=nfft)
    B = np.fft.rfft(w_rev, n=nfft)
    corr_full = np.fft.irfft(A * B, n=nfft)[:n + m - 1]
    # slice: indices (2*n-2) - j for j=0..n-1
    idx = (2 * n - 2) - np.arange(n)
    grad = 2.0 * corr_full[idx]
    return grad

    beta = 0.5

        # compute gradient via softmax

"""
Propose a step-function (non‑negative sequence) that improves the
previous best objective (~1.5449).  The algorithm uses a smooth
approximation of the max‑convolution term (soft‑max) and an Adam‑style
projected gradient descent on the probability simplex.  Several
parameterised seeds (uniform, geometric, power‑law, beta‑like and
random Dirichlet) are generated, the best few are refined, and the
overall best sequence is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x)=1}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / v.shape[0])
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << (x - 1).bit_length()

# ----------------------------------------------------------------------
def _self_convolve(seq: np.ndarray) -> np.ndarray:
    """Full linear convolution of seq with itself (length 2*len-1)."""
    n = seq.shape[0]
    m = 2 * n - 1
    nfft = _next_pow2(m)
    A = np.fft.rfft(seq, n=nfft)
    conv = np.fft.irfft(A * A, n=nfft)[:m]
    return conv

# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """Exact objective used by evaluate_sequence (scale invariant)."""
    conv = _self_convolve(seq)
    max_b = conv.max()
    return float(2.0 * len(seq) * max_b)  # sum(seq)=1 after projection

# ----------------------------------------------------------------------
def _softmax_gradient(seq: np.ndarray, beta: float) -> np.ndarray:
    """
    Gradient of the smooth max objective using a soft‑max with
    temperature 1/beta.  Returns ∂/∂seq ( 2*n * log‑sum‑exp(beta*conv) ).
    """
    n = seq.shape[0]
    conv = _self_convolve(seq)                # shape (2n‑1,)
    # soft‑max weights, shifted for numerical stability
    maxc = conv.max()
    w = np.exp(beta * (conv - maxc))
    w /= w.sum()                               # shape (2n‑1,)

    # correlation with the reversed weight vector:
    w_rev = w[::-1]                            # also length (2n‑1)
    m = w_rev.shape[0]
    nfft = _next_pow2(n + m - 1)                # length for correlation
    A = np.fft.rfft(seq, n=nfft)
    B = np.fft.rfft(w_rev, n=nfft)
    corr_full = np.fft.irfft(A * B, n=nfft)[: n + m - 1]

    # pick the n entries that correspond to the formula
    idx = (2 * n - 2) - np.arange(n)           # descending indices
    grad = 2.0 * corr_full[idx]
    return grad

# ----------------------------------------------------------------------
def _adam_optimize(seq0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Adam‑style projected gradient descent on the simplex using the
    smooth‑max gradient.  The temperature beta is slowly increased so
    that the optimisation starts with a smooth landscape and ends
    focusing on the true max.
    """
    seq = seq0.copy()
    n = seq.shape[0]
    # Adam hyper‑parameters
    lr = 0.5
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(seq)
    v = np.zeros_like(seq)

    t = 0
    beta = 0.5                     # start very smooth
    best_local = seq.copy()
    best_score = _objective(best_local)

    while time.time() < deadline:
        t += 1
        # gradient of the smooth approximation
        grad = _softmax_gradient(seq, beta)

        # Adam moment updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        cand = _project_onto_simplex(seq - step)
        cand_score = _objective(cand)

        # keep the better of the two (simple acceptance rule)
        if cand_score < _objective(seq):
            seq = cand
        else:
            # occasional random shake‑up to escape flat plateaus
            if np.random.rand() < 0.005:
                i, j = np.random.choice(n, size=2, replace=False)
                delta = 0.01 * min(seq[i], 0.1)
                tmp = seq.copy()
                tmp[i] -= delta
                tmp[j] += delta
                seq = _project_onto_simplex(tmp)

        # update global best found for this run
        if cand_score < best_score:
            best_score = cand_score
            best_local = cand.copy()
            print(f"[Local] iteration {t:06d}  score={best_score:.6f}")

        # slowly tighten the soft‑max (increase beta)
        beta = min(beta * 1.002, 30.0)

        # early exit if time is almost up
        if time.time() > deadline - 0.1:
            break

    return best_local

# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000) -> list[float]:
    """
    Generate a family of initial step‑functions, improve each of the
    most promising ones with a smooth‑max Adam optimiser and return the
    best sequence found before the time budget expires.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5   # safety margin

    # length of the step function – a moderate size keeps each iteration cheap
    n = 3000

    # ------------------------------------------------------------------
    # helper to build a normalised candidate
    def _norm(arr):
        arr = np.asarray(arr, dtype=np.float64)
        arr = np.clip(arr, 0.0, 1000.0)
        s = arr.sum()
        if s == 0:
            arr = np.full_like(arr, 1.0 / len(arr))
        else:
            arr = arr / s
        return arr

    # ------------------------------------------------------------------
    # 1. generate a diverse pool of seeds
    seed_pool = []

    # uniform
    seed_pool.append(_norm(np.full(n, 1.0 / n)))

    # geometric families
    for alpha in np.linspace(0.5, 4.0, 10):
        eps = alpha / n
        r = 1.0 - eps
        if r <= 0:
            continue
        powers = r ** np.arange(n, dtype=np.float64)
        seq = _norm(powers)
        seed_pool.append(seq)

    # power‑law families
    for p in np.linspace(0.1, 2.5, 12):
        idx = np.arange(1, n + 1, dtype=np.float64)
        seq = _norm(idx ** (-p))
        seed_pool.append(seq)

    # beta‑like families (edge‑heavy when a,b<1, centre‑heavy when >1)
    for a in np.linspace(0.2, 3.0, 6):
        for b in np.linspace(0.2, 3.0, 6):
            i = np.arange(1, n + 1, dtype=np.float64)
            seq = _norm((i ** (a - 1.0)) * ((n + 1 - i) ** (b - 1.0)))
            seed_pool.append(seq)

    # random Dirichlet (different concentration values)
    for concentration in [0.05, 0.2, 0.5, 1.0, 2.0]:
        x = np.random.gamma(concentration, 1.0, size=n)
        seed_pool.append(_norm(x))

    # ------------------------------------------------------------------
    # evaluate pool and keep the best few (we will refine them)
    scores = [_objective(s) for s in seed_pool]
    ranked = sorted(zip(scores, seed_pool), key=lambda p: p[0])
    top_k = 5
    top_candidates = [s for _, s in ranked[:top_k]]
    best_seq = ranked[0][1].copy()
    best_score = ranked[0][0]
    print(f"[Init] best seed score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Local optimisation of the top candidates (time‑sliced)
    # distribute remaining budget evenly among runs, but stop early if
    # we run out of time.
    remaining = deadline - time.time()
    per_run = max(5.0, remaining / (top_k + 1))

    for idx, init_seq in enumerate(top_candidates):
        if time.time() > deadline - 0.5:
            break
        run_deadline = time.time() + per_run
        print(f"[Run {idx}] optimisation start, allocated {per_run:.1f}s")
        refined = _adam_optimize(init_seq, run_deadline)
        refined_score = _objective(refined)
        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined.copy()
            print(f"[Run {idx}] new global best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Fallback: if nothing found, return uniform
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)

    print(f"[Done] final best objective = {best_score:.6f} (length {n})")
    # Convert to plain Python list (the caller may re‑scale, that does not matter)
    return best_seq.tolist()

    beta = 0.5                     # start with a very smooth surrogate
    cur_score = _objective(seq)

    best_local = seq.copy()
    best_score = cur_score

    while time.time() < deadline:
        t += 1
        grad = _softmax_gradient(seq, beta)

        # Adam moment updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        cand = _project_onto_simplex(seq - step)
        cand_score = _objective(cand)

        # simple accept‑if‑better rule (otherwise keep the current point)
        if cand_score < cur_score:
            seq = cand
            cur_score = cand_score
        else:
            # occasional random shake‑up to escape shallow plateaus
            if np.random.rand() < 0.005:
                i, j = np.random.choice(n, size=2, replace=False)
                delta = 0.01 * min(seq[i], 0.1)
                tmp = seq.copy()
                tmp[i] -= delta
                tmp[j] += delta
                seq = _project_onto_simplex(tmp)
                cur_score = _objective(seq)

        # keep track of the best point ever seen during this run
        if cur_score < best_score:
            best_score = cur_score
            best_local = seq.copy()
            print(f"[Optim] iter {t:06d}  best local score = {best_score:.6f}")

        # tighten the soft‑max (increase β) – this makes the surrogate
        # converge to the true max
        beta = min(beta * 1.002, 30.0)

        # break early if the overall wall‑time is almost exhausted
        if time.time() > deadline - 0.1:
            break

    return best_local


# ----------------------------------------------------------------------
def _norm(arr: np.ndarray) -> np.ndarray:
    """Clip to [0,1000] and project onto the simplex (sum = 1)."""
    arr = np.asarray(arr, dtype=np.float64)
    arr = np.clip(arr, 0.0, 1000.0)
    s = arr.sum()
    if s == 0:
        arr = np.full_like(arr, 1.0 / arr.size)
    else:
        arr = arr / s
    return arr

def propose_candidate(seed: int = 42, budget_s: float = 1000) -> list[float]:
    """
    Build a diverse set of seeds, refine the most promising ones with a
    smooth‑max Adam optimiser and return the best sequence discovered
    before the time budget expires.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # length of the step function – large enough to give flexibility,
    # still cheap for FFT‑based convolutions.
    n = 3000

    # ------------------------------------------------------------------
    # collect a pool of candidate seeds
    seed_pool = []

    # uniform distribution (the baseline)
    seed_pool.append(_norm(np.full(n, 1.0 / n)))

    # geometric progressions (decreasing)
    for alpha in np.linspace(0.5, 4.0, 12):
        eps = alpha / n
        r = 1.0 - eps
        if r <= 0:
            continue
        powers = r ** np.arange(n, dtype=np.float64)
        seed_pool.append(_norm(powers))

    # power‑law families
    for p in np.linspace(0.1, 2.5, 14):
        idx = np.arange(1, n + 1, dtype=np.float64)
        seed_pool.append(_norm(idx ** (-p)))

    # beta‑like shapes (edge‑heavy when a,b<1, centre‑heavy when >1)
    for a in np.linspace(0.2, 3.0, 6):
        for b in np.linspace(0.2, 3.0, 6):
            i = np.arange(1, n + 1, dtype=np.float64)
            shape = (i ** (a - 1.0)) * ((n + 1 - i) ** (b - 1.0))
            seed_pool.append(_norm(shape))

    # random Dirichlet draws with various concentration parameters
    for conc in [0.05, 0.2, 0.5, 1.0, 2.0]:
        x = np.random.gamma(conc, 1.0, size=n)
        seed_pool.append(_norm(x))

    # optionally use the globally supplied construction if it exists
    if "height_sequence_1" in globals():
        try:
            g = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            seed_pool.append(_norm(g))
        except Exception:
            pass

    # ------------------------------------------------------------------
    # evaluate the pool and select the best few for refinement
    scores = [_objective(s) for s in seed_pool]
    ranked = sorted(zip(scores, seed_pool), key=lambda p: p[0])
    top_k = 5
    top_candidates = [s for _, s in ranked[:top_k]]

    best_seq = ranked[0][1].copy()
    best_score = ranked[0][0]
    print(f"[Init] best seed score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # allocate time to the local optimiser for each top candidate
    remaining_time = deadline - time.time()
    per_candidate = max(5.0, remaining_time / (top_k + 1))

    for idx, cand in enumerate(top_candidates):
        if time.time() > deadline - 0.5:
            break
        run_deadline = time.time() + per_candidate
        print(f"[Run {idx}] start optimisation (≈{per_candidate:.1f}s budget)")
        refined = _adam_optimize(cand, run_deadline)
        refined_score = _objective(refined)
        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined.copy()
            print(f"[Run {idx}] new global best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # final sanity – if everything failed, fall back to uniform
    if best_seq is None or best_seq.size == 0:
        best_seq = np.full(n, 1.0 / n)

    print(f"[Done] final best objective = {best_score:.6f} (length {n})")
    return best_seq.tolist()
