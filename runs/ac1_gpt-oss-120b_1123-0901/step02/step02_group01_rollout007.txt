def propose_candidate():
    set seed
    deadline = time.time()+budget - margin

    n = 5000
    best_seq = None; best_val = inf

    # initialize from several candidate sequences 
       - geometric with different decays (like previous)
       - random Dirichlet
    for each init:
         evaluate exact objective (using _objective)
         if better: update best

    # pick seq = best_seq

    # Adam optimizer
    lam = 2.0; lam_factor = 1.015
    lr = 0.5  # base step size (learning rate) for Adam's update BEFORE projection.
    beta1=0.9; beta2=0.999
    eps=1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    last_improve = time.time()
    while time.time() < deadline:
        conv = _full_convolution(seq)
        grad = _softmax_gradient(seq, conv, lam, size_corr=size_fft)
        t += 1
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad*grad)
        m_hat = m/(1 - beta1**t)
        v_hat = v/(1 - beta2**t)
        # Adam update: seq_new = seq - lr * m_hat / (sqrt(v_hat) + eps)
        seq_new = seq - lr * m_hat / (np.sqrt(v_hat) + eps)
        seq_new = _project_onto_simplex(seq_new)

        # evaluate
        val = _objective(seq_new)
        if val < best_val:
            best_val = val; best_seq = seq_new.copy()
            last_improve = time.time()
            print(f"[Iter {t}] new best {val:.6f}")

        # if stagnated >2 sec, add random noise
        if time.time() - last_improve > 2.0:
            noise = np.random.uniform(-1e-3, 1e-3, size=n)
            seq_new = _project_onto_simplex(seq_new + noise)
            last_improve = time.time()

        # update for next iter
        seq = seq_new

        # schedule: increase lam slowly; maybe decrease lr slowly.
        lam *= lam_factor
        lr *= 0.9999

        # maybe adjust size_fft if n changes (no)

    return best_seq.tolist()

"""
Propose a low‑score step‑function by Adam optimisation on a smooth‑max
surrogate of the max‑self‑convolution.  The algorithm

1. chooses a moderately large length (default 5000) and normalises the
   vector to the probability simplex (∑a_i = 1);
2. initialises from several cheap candidates (geometric decays,
   Dirichlet draws and – if available – the previous best sequence);
3. runs a continuation scheme: a smooth‑max temperature λ starts small
   (≈2) and is multiplied by ≈1.02 each iteration, while an Adam optimiser
   (β1=0.9, β2=0.999) updates the coefficients;
4. after each step the vector is projected back onto the simplex;
5. the exact objective 2·n·max(a∗a)/(∑a)² is monitored, the best vector is
   kept and occasional tiny random “shakes’’ are injected if the search
   stalls;
6. when the time budget (≈1000 s) is exhausted the best sequence found is
   returned as a plain Python list.

The use of Adam (momentum and adaptive step‑size) together with the
gradual increase of λ generally yields a lower max‑convolution than plain
gradient descent, and the larger dimension gives the optimiser more
degrees of freedom to flatten the auto‑correlation.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    # sort decreasing
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        # all entries non‑positive -> uniform distribution
        return np.full_like(v, 1.0 / len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, size_fft: int) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) using an FFT of size `size_fft`."""
    n = len(a)
    fa = np.fft.rfft(a, size_fft)
    conv = np.fft.irfft(fa * fa, size_fft)[: 2 * n - 1]
    return conv


def _objective(a: np.ndarray) -> float:
    """
    Exact objective used by evaluate_sequence:
        2·n·max(a*a) / (sum(a)²)
    a must be non‑negative; the function does **not** enforce the simplex
    constraint – it works for any scaling.
    """
    n = len(a)
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 1e-12:
        return float("inf")
    return 2.0 * n * max_b / (sum_a * sum_a)


def _softmax_gradient(a: np.ndarray,
                      conv: np.ndarray,
                      lam: float,
                      size_fft: int) -> np.ndarray:
    """
    Gradient of the smooth‑max surrogate
        φ_λ(a) = (1/λ)·log Σ_k exp(λ·(a*a)[k])
    Returns the n‑dimensional vector ∇φ_λ(a) = 2·(w ★ a),
    where w are the soft‑max weights.
    """
    n = len(a)
    # numerically stable soft‑max on the convolution entries
    max_c = conv.max()
    w = np.exp(lam * (conv - max_c))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # correlation w ★ a  (linear, not circular)
    rev_a = a[::-1]
    fw = np.fft.rfft(w, size_fft)
    frev = np.fft.rfft(rev_a, size_fft)
    corr = np.fft.irfft(fw * frev, size_fft)

    # slice the part that corresponds to indices 0 … n‑1
    grad = 2.0 * corr[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the given time budget, prints progress, and
    returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    # time budget – keep a tiny safety margin
    deadline = time.time() + budget_s - 0.5

    # ------------------------------------------------------------------
    # 1. Choose a length.  5000 offers a good trade‑off between expressivity
    #    and per‑iteration cost on a modern CPU.
    n = 5000
    size_fft = _next_pow2(3 * n)         # enough for both convolution & correlation

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # 2. Generate a set of cheap initial candidates.
    #    (a) geometric decays a_i ∝ r^i
    for ratio in np.linspace(0.85, 0.999, 25):
        if time.time() > deadline:
            break
        r = ratio
        powers = r ** np.arange(n, dtype=np.float64)
        seq = powers
        seq /= seq.sum()                 # put on the simplex
        val = _objective(seq)
        if val < best_val:
            best_val, best_seq = val, seq.copy()
            print(f"[Init geom] r={r:.5f}  obj={val:.6f}")

    #    (b) random Dirichlet vectors
    for _ in range(20):
        if time.time() > deadline:
            break
        seq = np.random.dirichlet(np.ones(n))
        val = _objective(seq)
        if val < best_val:
            best_val, best_seq = val, seq.copy()
            print(f"[Init dir] obj={val:.6f}")

    #    (c) up‑sample the previous best if it exists
    if "height_sequence_1" in globals():
        old = np.asarray(height_sequence_1, dtype=np.float64)
        if old.ndim == 1 and old.size > 1:
            # linear interpolation onto the new grid
            xs_old = np.linspace(0.0, 1.0, old.size)
            xs_new = np.linspace(0.0, 1.0, n)
            seq = np.interp(xs_new, xs_old, old)
            seq = np.maximum(seq, 0.0)
            if seq.sum() == 0.0:
                seq = np.full(n, 1.0 / n)
            else:
                seq /= seq.sum()
            val = _objective(seq)
            if val < best_val:
                best_val, best_seq = val, seq.copy()
                print(f"[Init upsample] obj={val:.6f}")

    # Fallback: uniform distribution
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)
        best_val = _objective(best_seq)
        print(f"[Init uniform] obj={best_val:.6f}")

    # ------------------------------------------------------------------
    # 3. Adam optimisation on the smooth‑max surrogate with continuation.
    seq = best_seq.copy()
    lam = 2.0                       # start temperature (smooth)
    lam_factor = 1.018              # growth per iteration
    lam_max = 250.0                 # cap to avoid overflow
    lr = 0.12                       # base learning rate for Adam
    lr_factor = 0.9997              # slow decay
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    it = 0
    last_improve = time.time()

    while time.time() < deadline:
        # 3a) convolution a*a
        conv = _full_convolution(seq, size_fft)

        # 3b) gradient of the smooth‑max surrogate
        grad = _softmax_gradient(seq, conv, lam, size_fft)

        # 3c) Adam update (pre‑projection)
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        seq_new = seq - step
        seq_new = _project_onto_simplex(seq_new)

        # 3d) exact objective
        val = _objective(seq_new)

        # 3e) keep the best ever seen
        if val < best_val:
            best_val, best_seq = val, seq_new.copy()
            last_improve = time.time()
            print(f"[Iter {it:06d}] new best obj={val:.6f}")

        # 3f) shake if we have been idle for a while
        if time.time() - last_improve > 2.0:
            noise = (np.random.rand(n) - 0.5) * 5e-4
            seq_new = _project_onto_simplex(seq_new + noise)
            last_improve = time.time()

        # 3g) advance state
        seq = seq_new

        # 3h) schedule updates
        lam = min(lam_max, lam * lam_factor)
        lr *= lr_factor

    # ------------------------------------------------------------------
    # 4. Final short local‑search polishing (optional – cheap tries)
    #    Randomly move a tiny amount of mass between two entries.
    attempts = 0
    while time.time() < deadline and attempts < 3000:
        attempts += 1
        i, j = np.random.choice(n, 2, replace=False)
        eps_move = np.random.rand() * 2e-3          # up to 0.002 of total mass
        if best_seq[i] < eps_move:
            continue
        cand = best_seq.copy()
        cand[i] -= eps_move
        cand[j] += eps_move
        # projection not needed – sum unchanged and all non‑negative by construction
        val = _objective(cand)
        if val < best_val:
            best_val, best_seq = val, cand.copy()
            print(f"[Polish] improve to {val:.6f} (swap {i}->{j})")

    print(f"Final best objective = {best_val:.6f} (length {n})")
    # Return a plain Python list, as required by the external evaluator.
    return best_seq.tolist()
