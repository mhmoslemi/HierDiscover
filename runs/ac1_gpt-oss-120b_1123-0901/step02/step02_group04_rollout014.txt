"""
Propose candidate: multi‑phase optimizer for minimizing max autocorrelation.
Phase 0 – optional seed from global `height_sequence_1`.
Phase 1 – exhaustive sweep over several analytic families (exponential,
         power‑law, Gaussian, sinusoidal, triangular, linear, quadratic).
Phase 2 – smoothed gradient descent on a log‑sum‑exp surrogate of the
         infinity‑norm, with projection onto the simplex (sum = √(2 n)).
Phase 3 – stochastic local search (hill‑climbing with simulated annealing)
         that moves a tiny amount of mass between two positions.
Phase 4 – optional linear‑programming refinement (as in the state‑of‑the‑art).

The routine respects a wall‑clock budget (`budget_s`) and always returns
the best sequence seen.  All helper functions are defined at module level
and do not capture outer variables.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _conv_full(a: np.ndarray) -> np.ndarray:
    """Full (linear) convolution a * a."""
    return np.convolve(a, a)

def _evaluate(seq: list) -> float:
    """Wrapper around the supplied evaluate_sequence, returns +inf on error."""
    try:
        return evaluate_sequence(seq)
    except Exception:
        return float("inf")

def _maybe_update_best(cand: np.ndarray, best_seq: np.ndarray | None, best_score: float):
    """Return possibly updated (best_seq, best_score) after evaluating cand."""
    score = _evaluate(cand.tolist())
    if score < best_score:
        print(f"[{time.time():.2f}] New best = {score:.6f} (len={len(cand)})")
        return cand.copy(), score
    return best_seq, best_score

# ----------------------------------------------------------------------
# Shape factories (non‑normalised, non‑negative)
# ----------------------------------------------------------------------
def _shape_exp(n: int, alpha: float) -> np.ndarray:
    """Monotonically decreasing exponential."""
    return np.exp(-alpha * np.arange(n, dtype=float))

def _shape_exp_rev(n: int, alpha: float) -> np.ndarray:
    """Exponential that grows towards the tail (mirror of _shape_exp)."""
    return np.exp(-alpha * np.arange(n - 1, -1, -1, dtype=float))

def _shape_linear_desc(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    return np.maximum(0.0, 1.0 - beta * i)

def _shape_quad_desc(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    return np.maximum(0.0, 1.0 - beta * i * i)

def _shape_triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    return np.minimum(i + 1.0, n - i)

def _shape_power_law(n: int, p: float) -> np.ndarray:
    """a_i = (i+1)^{-p}."""
    i = np.arange(1, n + 1, dtype=float)
    return i ** (-p)

def _run_smooth_gradient_descent(
    init_a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray | None,
    best_score: float,
    target_score: float = 1.503,
    max_iters: int = 4000,
) -> tuple[np.ndarray | None, float]:
    """Smoothed gradient descent on log‑sum‑exp surrogate of max convolution."""
    a = init_a.astype(float)
    n = a.shape[0]
    target = math.sqrt(2.0 * n)

    # momentum parameters
    m = np.zeros_like(a)
    mu = 0.9
    base_step = 0.2

    # schedule for the smoothness parameter β (beta)
    beta_start = 0.5
    beta_end = 12.0

    # pre‑compute index array for diagonal entries (2*j)
    diag_idx = 2 * np.arange(n)

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        # full self‑convolution
        conv = _conv_full(a)

        # smoothness parameter evolves linearly with iteration
        beta = beta_start + (beta_end - beta_start) * (it / max_iters)

        # log‑sum‑exp weights (numerically stable)
        conv_max = conv.max()
        w = np.exp(beta * (conv - conv_max))
        w_sum = w.sum()
        if w_sum == 0:
            # should not happen, protect against division by zero
            w = np.full_like(w, 1.0 / w.shape[0])
        else:
            w /= w_sum

        # correlation a ◦ w (cross‑correlation)
        # compute via convolution with reversed w
        corr = np.convolve(a, w[::-1])
        offset = len(w) - 1                # = 2*n-2
        corr_sub = corr[offset : offset + n]

        # diagonal weights (where k = 2*j) – needed for exact gradient
        diag_w = w[diag_idx]

        # gradient of the smoothed objective (we minimise, so we move opposite)
        grad = 2.0 * corr_sub - diag_w * a

        # momentum update and step size
        direction = -grad
        m = mu * m + (1.0 - mu) * direction
        step = base_step / math.sqrt(it + 1.0)
        max_abs = np.max(np.abs(m))
        if max_abs == 0.0:
            max_abs = 1.0
        eta = step / max_abs

        a_new = a + eta * m
        a_new = _project_onto_simplex(a_new, target)

        # occasional evaluation (every 5 iterations)
        if it % 5 == 0:
            best_seq, best_score = _maybe_update_best(a_new, best_seq, best_score)
            if best_score <= target_score:
                return best_seq, best_score

        a = a_new

    # final evaluation after loop
    best_seq, best_score = _maybe_update_best(a, best_seq, best_score)
    return best_seq, best_score


def _run_annealing(
    init_a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray | None,
    best_score: float,
    target_score: float = 1.503,
    iters: int = 5000,
) -> tuple[np.ndarray | None, float]:
    """Simple simulated‑annealing hill climbing that moves a little mass."""
    a = init_a.copy()
    n = a.shape[0]
    cur_score = best_score if best_seq is not None else _evaluate(a.tolist())
    # temperature schedule (linear decay)
    temp_start = max(0.01 * cur_score, 0.001)
    temp_end = 0.0001 * cur_score

    for it in range(iters):
        if time.time() >= deadline:
            break
        # linear temperature schedule
        temp = temp_start + (temp_end - temp_start) * (it / iters)

        # pick two distinct indices
        i = np.random.randint(0, n)
        j = np.random.randint(0, n - 1)
        if j >= i:
            j += 1

        # amount to transfer – up to a fraction of the donor weight
        max_move = 0.4 * a[i]          # move at most 40 % of the source
        if max_move <= 0.0:
            continue
        eps = np.random.rand() * max_move

        a_new = a.copy()
        a_new[i] -= eps
        a_new[j] += eps

        # evaluate candidate
        score = _evaluate(a_new.tolist())
        if score < cur_score:
            a = a_new
            cur_score = score
            best_seq, best_score = _maybe_update_best(a, best_seq, best_score)
            if best_score <= target_score:
                break
        else:
            # Metropolis acceptance
            if temp > 0 and np.random.rand() < math.exp(-(score - cur_score) / temp):
                a = a_new
                cur_score = score

    # final update
    best_seq, best_score = _maybe_update_best(a, best_seq, best_score)
    return best_seq, best_score


def _run_lp_refinement(a: np.ndarray, deadline: float, best_seq: np.ndarray | None,
                       best_score: float, target_score: float = 1.503):
    """Perform one or a few LP refinements (the SOTA “LP‑step”)."""
    try:
        from scipy.optimize import linprog
    except Exception:
        return best_seq, best_score

    n = a.shape[0]
    target = math.sqrt(2.0 * n)
    while time.time() < deadline:
        conv = _conv_full(a)
        M = conv.max()
        rows = 2 * n - 1

        # Build constraint matrix A with shape (rows, n)
        A = np.zeros((rows, n), dtype=float)
        for k in range(rows):
            low = max(0, k - (n - 1))
            high = min(n - 1, k)
            cols = np.arange(low, high + 1, dtype=int)
            A[k, cols] = a[k - cols]

        res = linprog(
            c=-np.ones(n),          # maximise Σ g_j  → minimise -Σ g_j
            A_ub=A,
            b_ub=np.full(rows, M),
            bounds=(0, None),
            method="highs",
            options={"presolve": True},
        )
        if not res.success:
            break

        g = res.x
        s = g.sum()
        if s <= 0:
            break
        g = g * (target / s)

        # convex blend
        t = 0.25
        a = (1 - t) * a + t * g

        best_seq, best_score = _maybe_update_best(a, best_seq, best_score)
        if best_score <= target_score:
            break

        # stop if very little progress
        if time.time() + 0.5 > deadline:
            break
    return best_seq, best_score

def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex {x≥0, Σx = s}."""
    if s < 0:

    return np.maximum(v - theta, 0.0)

                best_seq, best_score = _maybe_update_best(cand_arr, best_seq, best_score)
        except Exception:
            pass

    # --------------------------------------------------------------
    # Phase 1 – exhaustive sweep over parametric families
    # --------------------------------------------------------------
    lengths = list(range(200, 1601, 200)) + [1326, 1500]
    for n in lengths:
        if time.time() >= deadline:
            break
        target = math.sqrt(2.0 * n)

        # exponential decay
        for alpha in np.logspace(-4, -0.3, num=14):
            a = _shape_exp(n, alpha)
            if a.sum() == 0:
                continue
            a = a / a.sum() * target
            best_seq, best_score = _maybe_update_best(a, best_seq, best_score)

        # reversed exponential
        for alpha in np.logspace(-4, -0.3, num=14):
            a = _shape_exp_rev(n, alpha)
            if a.sum() == 0:
                continue
            a = a / a.sum() * target
            best_seq, best_score = _maybe_update_best(a, best_seq, best_score)

        # linear descending
        for beta in np.linspace(0.0, 2.0 / max(1, n - 1), num=12):
            a = _shape_linear_desc(n, beta)
            if a.sum() == 0:
                continue
            a = a / a.sum() * target
            best_seq, best_score = _maybe_update_best(a, best_seq, best_score)

        # quadratic descending
        for beta in np.linspace(0.0, 1.0 / max(1, n ** 2), num=12):
            a = _shape_quad_desc(n, beta)
            if a.sum() == 0:
                continue
            a = a / a.sum() * target
            best_seq, best_score = _maybe_update_best(a, best_seq, best_score)

        # triangular (symmetric)
        a = _shape_triangular(n)
        a = a / a.sum() * target
        best_seq, best_score = _maybe_update_best(a, best_seq, best_score)

        # power‑law (inverse polynomial)
        for p in np.linspace(0.0, 2.0, num=13):
            a = _shape_power_law(n, p)
            if a.sum() == 0:
                continue
            a = a / a.sum() * target
            best_seq, best_score = _maybe_update_best(a, best_seq, best_score)

        # Gaussian bell
        for sigma_factor in np.linspace(0.1, 0.5, num=9):
            sigma = sigma_factor * n
            a = _shape_gaussian(n, sigma)
            if a.sum() == 0:
                continue
            a = a / a.sum() * target
            best_seq, best_score = _maybe_update_best(a, best_seq, best_score)

        # cosine half‑wave
        a = _shape_cosine(n)
        a = a / a.sum() * target
        best_seq, best_score = _maybe_update_best(a, best_seq, best_score)

        if best_score <= TARGET:
            break

    # --------------------------------------------------------------
    # Phase 2 – smoothed gradient descent (multiple restarts)
    # --------------------------------------------------------------
    if best_seq is None:
        n0 = 1000
        target0 = math.sqrt(2.0 * n0)
        best_seq = np.full(n0, target0 / n0, dtype=float)
        best_score = _evaluate(best_seq.tolist())

    # first run from the current best
    best_seq, best_score = _run_smooth_gradient_descent(
        init_a=best_seq,
        deadline=deadline,
        best_seq=best_seq,
        best_score=best_score,
        target_score=TARGET,
    )
    if best_score <= TARGET:
        return best_seq.tolist()

    # random restarts (shorter runs)
    while time.time() < deadline:
        n = np.random.randint(300, 2001)
        target = math.sqrt(2.0 * n)
        alpha = np.random.rand() * 0.025 + 1e-5
        a0 = _shape_exp(n, alpha)
        a0 = a0 / a0.sum() * target
        best_seq, best_score = _run_smooth_gradient_descent(
            init_a=a0,
            deadline=deadline,
            best_seq=best_seq,
            best_score=best_score,
            target_score=TARGET,
            max_iters=1800,
        )
        if best_score <= TARGET:
            break
        if best_score < TARGET + 0.001:
            break

    # --------------------------------------------------------------
    # Phase 3 – stochastic annealing / hill climbing
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_seq, best_score = _run_annealing(
            init_a=best_seq,
            deadline=deadline,
            best_seq=best_seq,
            best_score=best_score,
            target_score=TARGET,
        )
        if best_score <= TARGET:
            return best_seq.tolist()

    # --------------------------------------------------------------
    # Phase 4 – optional LP refinement (the “LP‑step” from the SOTA paper)
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_seq, best_score = _run_lp_refinement(
            a=best_seq,
            deadline=deadline,
            best_seq=best_seq,
            best_score=best_score,
            target_score=TARGET,
        )

    # --------------------------------------------------------------
    # Fallback – return a uniform sequence if nothing succeeded
    # --------------------------------------------------------------
    if best_seq is None:
        n = 1000
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=float)

    return best_seq.tolist()
