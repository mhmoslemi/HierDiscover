        n_final = best_seq.shape[0]
        target_sum_final = math.sqrt(2.0 * n_final)
        print(f"[final] fine‑tuning length {n_final}")
        best_seq, best_M = _adam_optimize(
            init_a=best_seq,
            target_sum=target_sum_final,
            deadline=deadline,
            lr0=0.02,
            jitter_interval=8000,
            jitter_scale=0.0008,
        )
        best_score = best_M
        print(f"[final] final best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Clip to the evaluator's allowed range and return.
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # One final exact evaluation (optional, only for logging).
    try:
        exact = evaluate_sequence(best_seq.tolist())
        print(f"[final metric] evaluate_sequence = {exact:.6f}")
    except Exception:
        print("[final metric] evaluation failed")

    return best_seq.tolist()
