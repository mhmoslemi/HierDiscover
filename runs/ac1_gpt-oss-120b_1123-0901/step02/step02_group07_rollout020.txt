def _grad_active(seq, conv, max_val, tol=1e-6):
    n = len(seq)
    # Identify active shifts with conv >= max_val * (1 - rel_tol)
    active = np.where(conv >= max_val * (1 - tol))[0]
    grad = np.zeros_like(seq)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        # indices from lo to hi inclusive
        idx = np.arange(lo, hi+1)
        grad[idx] += seq[k - idx]
    # optionally average
    grad /= max(1, len(active))
    return grad

def _refine_gradient(seq, time_deadline):
    n = len(seq)
    cur_score = _objective(seq)
    best_seq = seq.copy()
    best_score = cur_score
    step_max = 1.0
    while time.time() < time_deadline:
        conv = _self_convolve(seq)
        max_val = conv.max()
        # active set: indices where conv >= max_val * 0.99
        active = np.where(conv >= max_val * 0.99)[0]
        grad = np.zeros_like(seq)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            grad[lo:hi+1] += seq[k - np.arange(lo, hi+1)]
        grad /= max(1, len(active))
        # line search
        step = step_max
        improved = False
        while step > 1e-6:
            cand = _project_onto_simplex(seq - step * grad)
            cand_score = _objective(cand)
            if cand_score < cur_score:
                improved = True
                break
            step *= 0.5
        if improved:
            seq = cand
            cur_score = cand_score
            step_max = max(step*1.2, 1e-3)

                best_score = cand_score

Propose a non‑negative step‑function (list of floats) that minimizes
the evaluate_sequence() objective.

The algorithm consists of three stages:

1. **Broad parametric sweep** – generate candidates from many simple
   families (beta‑like, sine‑power, Gaussian‑like, power‑law, geometric,
   Dirichlet draws with various concentrations).  The best of these
   seeds the next stage.

2. **Fine‑grained projected gradient descent** – on the current best
   sequence we repeatedly compute the full self‑convolution, collect
   all (near‑maximal) active shifts, form the aggregated sub‑gradient
   (the reversed fragment of the sequence for each active shift) and
   perform a back‑tracking line‑search with projection onto the
   probability simplex.  Random mass‑transfer moves break stagnation.

3. **Population‑based refinement (steady‑state EA)** – a small
   population of sequences evolves by convex combination of two parents,
   a tiny random mass‑transfer mutation, simplex projection and
   replacement of the worst individual.  The best individual is kept
   throughout.  This stage explores the neighbourhood of the incumbent
   solution and often finds a modest further improvement.

All operations use fast FFT‑based convolution (O(n log n)) and
projection onto the simplex (O(n log n)).  The total running time is
kept under the 1000 s budget; intermediate progress is printed.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex {x ≥ 0, sum(x)=1}
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast FFT based self‑convolution (real inputs)
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _self_convolve(seq: np.ndarray) -> np.ndarray:
    """Linear convolution of seq with itself (length 2·len(seq)‑1)."""
    n = seq.shape[0]
    m = 2 * n - 1
    nfft = _next_pow2(m)
    A = np.fft.rfft(seq, n=nfft)
    conv = np.fft.irfft(A * A, n=nfft)[:m]
    return conv


# ----------------------------------------------------------------------
# Objective (scale‑invariant; we keep sum(seq)=1)
def _objective(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) (sum(seq) is assumed = 1)."""
    n = len(seq)
    conv = _self_convolve(seq)
    max_b = conv.max()
    return float(2.0 * n * max_b)


# ----------------------------------------------------------------------
# Parametric families ----------------------------------------------------
def _geom_seq(n: int, alpha: float) -> np.ndarray | None:
    """Geometric progression a_i ∝ r^i with r = 1‑α/n."""
    eps = alpha / n
    r = 1.0 - eps
    if r <= 0.0:
        return None
    powers = r ** np.arange(n, dtype=np.float64)
    seq = powers / powers.sum()
    return seq


def _powerlaw_seq(n: int, p: float) -> np.ndarray:
    """Power‑law a_i ∝ (i+1)^‑p."""
    seq = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    return seq / seq.sum()


def _beta_seq(n: int, a: float, b: float) -> np.ndarray:
    """Beta‑like a_i ∝ (i+1)^{a‑1}·(n‑i)^{b‑1}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    seq = (i ** (a - 1.0)) * ((n + 1 - i) ** (b - 1.0))
    seq[seq < 0] = 0.0
    return seq / seq.sum()


def _sine_power_seq(n: int, p: float) -> np.ndarray:
    """a_i ∝ sin(π·(i+0.5)/n)^p."""
    angles = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    seq = np.sin(angles) ** p
    seq[seq < 0] = 0.0
    return seq / seq.sum()


def _gaussian_seq(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-λ·(i‑mid)²)."""
    mid = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    seq = np.exp(-lam * (i - mid) ** 2)
    return seq / seq.sum()


def _random_simplex(n: int, conc: float = 1.0) -> np.ndarray:
    """Dirichlet draw with scalar concentration conc."""
    alpha = np.full(n, conc, dtype=np.float64)
    x = np.random.gamma(alpha, 1.0)
    if x.sum() == 0:
        return np.full(n, 1.0 / n)
    return x / x.sum()


# ----------------------------------------------------------------------
# Gradient refinement ----------------------------------------------------
def _refine_gradient(start_seq: np.ndarray, deadline: float):
    """Projected sub‑gradient descent with back‑tracking line‑search."""
    seq = start_seq.copy()
    n = len(seq)
    cur_score = _objective(seq)
    best_seq = seq.copy()
    best_score = cur_score
    step_base = 5.0  # initial exploratory step size

    it = 0
    while time.time() < deadline:
        conv = _self_convolve(seq)
        max_val = conv.max()
        # all shifts within 1 % of the maximum are considered active
        active = np.where(conv >= max_val * 0.99)[0]
        grad = np.zeros_like(seq)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=int)
            grad[idx] += seq[k - idx]
        if active.size > 0:
            grad /= float(active.size)

        # --- line search on the opposite direction of the sub‑gradient ---
        step = step_base
        improved = False
        while step > 1e-6:
            cand = _project_onto_simplex(seq - step * grad)
            cand_score = _objective(cand)
            if cand_score < cur_score:
                improved = True
                break
            step *= 0.5

        if improved:
            seq = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = seq.copy()
                print(f"[Grad {it:05d}] new best = {best_score:.6f}")
            # modestly increase step size after a success
            step_base = max(step * 1.2, 1e-3)
        else:
            # --- occasional random mass‑transfer move to escape stagnation ---
            i, j = np.random.choice(n, size=2, replace=False)
            delta = 0.01 * min(seq[i], 0.1)  # at most 1 % of total mass
            if delta > 0.0:
                cand = seq.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_onto_simplex(cand)
                cand_score = _objective(cand)
                if cand_score < cur_score:
                    seq = cand
                    cur_score = cand_score
                    if cand_score < best_score:
                        best_score = cand_score
                        best_seq = seq.copy()
                        print(f"[Grad {it:05d}] random improve = {best_score:.6f}")
        it += 1
    return best_seq, best_score

# Population (steady‑state evolutionary) refinement -----------------------
def _refine_population(initial_seq: np.ndarray, init_score: float, deadline: float):
    """Maintain a small population that evolves by convex combination + mutation."""
    n = len(initial_seq)
    pop_size = 30
    # initialise population (including the supplied initial sequence)
    population = []
    # first entrant – the current best
    population.append((init_score, initial_seq.copy()))
    # fill the rest with random Dirichlet draws of varied concentration
    while len(population) < pop_size and time.time() < deadline:
        conc = 10 ** np.random.uniform(-2, 1)  # from 0.01 to 10
        seq = _random_simplex(n, conc)
        score = _objective(seq)
        population.append((score, seq))
    # keep sorted (best first)
    population.sort(key=lambda x: x[0])
    best_seq, best_score = population[0]

    it = 0
    while time.time() < deadline:
        # tournament selection of two parents
        i1, i2 = np.random.choice(pop_size, size=2, replace=False)
        parent1 = population[i1][1]
        parent2 = population[i2][1]
        # convex combination
        alpha = np.random.rand()
        child = alpha * parent1 + (1.0 - alpha) * parent2
        # tiny mutation: move a small amount of mass between two random indices
        if np.random.rand() < 0.7:
            i, j = np.random.choice(n, size=2, replace=False)
            delta = 0.005 * min(child[i], 0.2)  # up to 0.5 % of the total mass
            if delta > 0:
                child[i] -= delta
                child[j] += delta
        child = _project_onto_simplex(child)
        child_score = _objective(child)

        # replace worst if the child is better
        if child_score < population[-1][0]:
            population[-1] = (child_score, child)
            population.sort(key=lambda x: x[0])
            if child_score < best_score:
                best_score = child_score
                best_seq = child.copy()
                print(f"[Pop {it:05d}] new best = {best_score:.6f}")

        # occasional aggressive local optimisation on the current best
        if it % 200 == 0 and time.time() < deadline - 5.0:
            # a short gradient run (≈ 5 s) on the best individual
            sub_deadline = time.time() + 5.0
            refined, refined_score = _refine_gradient(best_seq, sub_deadline)
            if refined_score < best_score:
                best_seq, best_score = refined, refined_score
                # replace the worst entry with this refined version
                population[-1] = (best_score, best_seq.copy())
                population.sort(key=lambda x: x[0])
                print(f"[Pop {it:05d}] refined best = {best_score:.6f}")

        it += 1
    return best_seq, best_score

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list[float]:
    """Search for a step‑function with a low evaluate_sequence() score.

    The returned list is a probability vector (non‑negative, sum = 1).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Choose a moderate length – long enough to be expressive but cheap.
    n = 2000

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # 0) Warm‑start from a previously found global variable if available.
    if "height_sequence_1" in globals():
        try:
            init = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            init = np.clip(init, 0.0, 1000.0)
            if init.sum() > 0.0:
                init = init / init.sum()
                score = _objective(init)
                if score < best_score:
                    best_score, best_seq = score, init.copy()
                    print(f"[Init] global seed score = {best_score:.6f}")
        except Exception:
            pass

    # --------------------------------------------------------------
    # 1. Broad parametric sweep (time‑limited)
    families = []

    # geometric
    families.append(("geom", [(alpha,) for alpha in np.linspace(0.5, 4.0, 20)]))
    # power‑law (including negative exponents for increasing shapes)
    families.append(
        ("power", [(p,) for p in np.linspace(-2.0, 2.5, 20)])
    )
    # beta‑like (a,b) spanning (<1) to >1
    a_vals = np.logspace(-2, 0.7, 12)   # ~0.01 … 5
    b_vals = np.logspace(-2, 0.7, 12)
    families.append(("beta", [(a, b) for a in a_vals for b in b_vals]))
    # sine‑power
    families.append(
        ("sine", [(p,) for p in np.linspace(0.1, 5.0, 15)])
    )
    # Gaussian‑like (λ)
    families.append(
        ("gauss", [(lam,) for lam in np.logspace(-4, -0.5, 12)])
    )
    # random Dirichlet with various concentrations
    families.append(
        ("dirichlet", [(conc,) for conc in [0.01, 0.05, 0.1, 0.3, 0.7, 1.0, 2.0, 5.0]])
    )

    for tag, param_list in families:
        if time.time() > deadline:
            break
        for params in param_list:
            if time.time() > deadline:
                break
            if tag == "geom":
                seq = _geom_seq(n, *params)
            elif tag == "power":
                seq = _powerlaw_seq(n, *params)
            elif tag == "beta":
                seq = _beta_seq(n, *params)
            elif tag == "sine":
                seq = _sine_power_seq(n, *params)
            elif tag == "gauss":
                seq = _gaussian_seq(n, *params)
            elif tag == "dirichlet":
                seq = _random_simplex(n, *params)
            else:
                continue
            if seq is None or np.isnan(seq).any():
                continue
            score = _objective(seq)
            if score < best_score:
                best_score, best_seq = score, seq.copy()
                print(f"[Init {tag}] params={params}  score={best_score:.6f}")

    # --------------------------------------------------------------
    # Fallback to uniform if nothing was generated (unlikely)
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)
        best_score = _objective(best_seq)

    # --------------------------------------------------------------
    # 2. Local gradient refinement (projected sub‑gradient descent)
    if time.time() < deadline - 5.0:
        seq_refined, score_refined = _refine_gradient(best_seq, deadline - 2.0)
        if score_refined < best_score:
            best_score, best_seq = score_refined, seq_refined.copy()
            print(f"[Refine] gradient best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3. Population‑based steady‑state evolutionary refinement
    if time.time() < deadline - 5.0:
        seq_pop, score_pop = _refine_population(best_seq, best_score, deadline - 1.0)
        if score_pop < best_score:
            best_score, best_seq = score_pop, seq_pop.copy()
            print(f"[Refine] population best = {best_score:.6f}")

    # --------------------------------------------------------------
    print(f"Final best objective = {best_score:.6f} (length {n})")
    # convert to plain Python list, keep the scaling unchanged (sum = 1)
    return best_seq.tolist()
