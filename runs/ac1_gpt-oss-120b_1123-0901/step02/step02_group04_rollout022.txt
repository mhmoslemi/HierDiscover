    deadline = time.time() + budget_s - 1.0

    best_seq = None
    best_score = float('inf')
    
    # Helper to update best
    def maybe_update(seq):
        nonlocal best_seq, best_score
        # seq is np.ndarray
        score = float(np.max(_conv_self(seq)))  # compute maximum convolution
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"[{time.time():.2f}] New best = {best_score:.6f} (len={len(seq)})")
        return score

    # List of lengths to try initially
    lengths = list(range(300, 1801, 200)) + [1200, 1500]  # include 1200 as baseline

    # Parametric families generate shapes
    for n in lengths:
        target = math.sqrt(2.0 * n)

        # exponential decays
        for alpha in np.logspace(-5, -0.3, num=15):
            a = np.exp(-alpha * np.arange(n))
            a = a / a.sum() * target
            maybe_update(a)

        # reversed exponential
        for alpha in np.logspace(-5, -0.3, num=15):
            a = np.exp(-alpha * np.arange(n-1, -1, -1))
            a = a / a.sum() * target
            maybe_update(a)

            a = 1.0 / np.power(np.arange(1, n+1), p)
            if a.sum() == 0:
                continue
            a = a / a.sum() * target
            maybe_update(a)

        for beta in np.linspace(0.0, 2.0/(n-1), num=8):
            a = np.maximum(0.0, 1.0 - beta * np.arange(n))

def _conv_full_fft(u, v):
    size = 1 << ((len(u) + len(v) - 1).bit_length())
    fu = np.fft.rfft(u, n=size)
    fv = np.fft.rfft(v, n=size)
    conv = np.fft.irfft(fu * fv, n=size)
    return conv[: len(u) + len(v) - 1]

"""
Propose a low‑autocorrelation step‑function.

The algorithm combines several ideas:

1.  A quick sweep of simple parametric families (exponential,
    reversed exponential, power‑law, linear/quadratic decay, triangular).
    All candidates are projected onto the simplex
    Σ a_i = √(2 n).

2.  Lp‑smooth gradient descent.  For a fixed exponent p the objective
    Fₚ(a)=∑(a∗a)_k^p has a smooth gradient that can be written with
    two FFT‑based convolutions.  Starting from a candidate we run a few
    hundred steps for a sequence of exponents p=4,8,…,40, decreasing the
    step size with 1/√t and projecting onto the simplex after each update.
    The current best sequence is recorded regularly.

3.  Random restarts – random Dirichlet draws and random exponentials –
    are fed to the Lp‑descent to escape local minima.

4.  Optional linear‑programming refinement (SciPy required).  From the best
    sequence we solve
        maximise   Σ g_j    s.t.   a∗g ≤ M,  g ≥ 0
    where M = max(a∗a).  The feasible g is rescaled to the simplex and
    blended with the current a (t≈0.3).  A few iterations of this step are
    performed.

All helper routines are defined at module level (no closures).  The search
uses the remaining wall‑clock time (default budget = 1000 s) and returns
the best non‑negative height sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# 1.  Projection onto the simplex {x ≥ 0, Σx = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex with sum `s`."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # descending sort
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# 2.  Convolution utilities (FFT‑based, full mode)
# ----------------------------------------------------------------------
def _conv_full_fft(u: np.ndarray, v: np.ndarray) -> np.ndarray:
    """Full convolution (len(u)+len(v)-1) using real FFT."""
    size = 1 << ((len(u) + len(v) - 1).bit_length())
    fu = np.fft.rfft(u, n=size)
    fv = np.fft.rfft(v, n=size)
    conv = np.fft.irfft(fu * fv, n=size)
    return conv[: len(u) + len(v) - 1]


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full convolution a * a."""
    return _conv_full_fft(a, a)


# ----------------------------------------------------------------------
# 3.  Objective helpers
# ----------------------------------------------------------------------
def _max_convolution(a: np.ndarray) -> float:
    """Return max_k (a∗a)_k – the true score after normalisation."""
    return float(_conv_self(a).max())


def _gradient_lp(a: np.ndarray, p: int) -> np.ndarray:
    """
    Gradient of   Fₚ(a)=∑(a∗a)_k^p   (p ≥ 2, integer).
    Formula: ∇Fₚ = 2·p·(a ∗ w_rev)_{m‑1‑i}, where w = (a∗a)^{p‑1} and m=2·n‑1.
    """
    n = a.shape[0]
    conv = _conv_self(a)                     # length m = 2n‑1
    w = np.power(conv, p - 1, dtype=np.float64)   # w_k = conv_k^{p‑1}
    w_rev = w[::-1]
    full = _conv_full_fft(a, w_rev)          # length n + m - 1
    m = conv.shape[0]                        # =2n‑1
    start = m - 1
    # slice = full[start - i] for i = 0..n‑1  (reversed order)
    grad = 2.0 * p * full[start : start - n : -1]
    return grad


# ----------------------------------------------------------------------
# 4.  Main optimiser
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs) -> list:
    """Search for a non‑negative step‑function with minimal max autocorrelation."""
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    best_seq = None          # numpy array
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Helper to update the current best (no closure, inline code)
    # ------------------------------------------------------------------
    def _maybe_update(candidate: np.ndarray) -> None:
        nonlocal best_seq, best_score
        sc = _max_convolution(candidate)
        if sc < best_score:
            best_score = sc
            best_seq = candidate.copy()
            print(f"[{time.time():.2f}] New best = {best_score:.6f} (len={len(candidate)})")

    # ------------------------------------------------------------------
    # 1.  Deterministic parametric sweep
    # ------------------------------------------------------------------
    lengths = list(range(300, 1801, 200)) + [1200, 1500, 1800]
    for n in lengths:
        target = math.sqrt(2.0 * n)

        # exponential decay
        for alpha in np.logspace(-5, -0.3, num=15):
            a = np.exp(-alpha * np.arange(n, dtype=float))
            a = a / a.sum() * target
            _maybe_update(a)

        # reversed exponential
        for alpha in np.logspace(-5, -0.3, num=15):
            a = np.exp(-alpha * np.arange(n - 1, -1, -1, dtype=float))
            a = a / a.sum() * target
            _maybe_update(a)

        # power‑law decay
        for pwr in np.linspace(0.5, 5.0, num=12):
            a = 1.0 / np.power(np.arange(1, n + 1, dtype=float), pwr)
            if a.sum() == 0:
                continue
            a = a / a.sum() * target
            _maybe_update(a)

        # linear descending (a_i = max(0, 1-β·i))
        for beta in np.linspace(0.0, 2.0 / max(1, n - 1), num=10):
            a = np.maximum(0.0, 1.0 - beta * np.arange(n, dtype=float))
            if a.sum() == 0:
                continue
            a = a / a.sum() * target
            _maybe_update(a)

        # quadratic descending (a_i = max(0, 1-β·i²))
        for beta in np.linspace(0.0, 1.0 / max(1, n * n), num=10):
            a = np.maximum(0.0, 1.0 - beta * np.arange(n, dtype=float) ** 2)
            if a.sum() == 0:
                continue
            a = a / a.sum() * target
            _maybe_update(a)

        # symmetric triangular shape
        a = np.minimum(np.arange(n, dtype=float) + 1.0, n - np.arange(n, dtype=float))
        a = a / a.sum() * target
        _maybe_update(a)

        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # 2.  Lp‑smooth gradient descent (single start + random restarts)
    # ------------------------------------------------------------------
    def _run_lp_descent(start_vec: np.ndarray) -> None:
        """Run the Lp‑gradient descent from `start_vec` and update the global best."""
        nonlocal best_seq, best_score
        a = start_vec.astype(float)
        n = a.shape[0]
        target = math.sqrt(2.0 * n)

        p_list = [4, 8, 12, 16, 20, 30, 40]      # increasing smoothness
        max_iters_per_p = 250                    # cap per exponent
        base_step = 0.5

        for p in p_list:
            it = 0
            while time.time() < deadline and it < max_iters_per_p:
                # gradient of Σ(conv^p)
                grad = _gradient_lp(a, p)

                # normalise step size by the largest component
                maxabs = np.max(np.abs(grad))
                if maxabs == 0.0:
                    break
                step = base_step / math.sqrt(it + 1)
                eta = step / maxabs

                a = a - eta * grad
                a = _project_onto_simplex(a, target)

                # occasional evaluation
                if it % 4 == 0:
                    _maybe_update(a)
                    if best_score < 1.503:    # early exit if target reached
                        return

                it += 1
            if time.time() >= deadline or best_score < 1.503:
                break

    # start from the best found so far (if any)
    if best_seq is not None:
        _run_lp_descent(best_seq)

    # random restarts – mixture of Dirichlet and exponential seeds
    while time.time() < deadline:
        n = np.random.randint(500, 2501)                # moderate size
        target = math.sqrt(2.0 * n)

        # Dirichlet‑type start
        gamma = np.random.gamma(shape=1.0, scale=1.0, size=n)
        a0 = gamma / gamma.sum() * target
        _run_lp_descent(a0)
        if time.time() >= deadline or best_score < 1.503:
            break

        # random exponential tail start
        alpha = np.random.rand() * 0.02 + 1e-5
        a_exp = np.exp(-alpha * np.arange(n, dtype=float))
        a_exp = a_exp / a_exp.sum() * target
        _run_lp_descent(a_exp)
        if best_score < 1.503:
            break

    # ------------------------------------------------------------------
    # 3.  Optional LP refinement (requires SciPy)
    # ------------------------------------------------------------------
    try:
        from scipy.optimize import linprog
    except Exception:                       # SciPy unavailable
        linprog = None

    if linprog is not None and best_seq is not None and time.time() < deadline:
        a = best_seq.copy()
        n = a.shape[0]
        target = math.sqrt(2.0 * n)

        while time.time() < deadline:
            conv = _conv_self(a)
            M = conv.max()
            rows = 2 * n - 1

            # Build the Toeplitz‑style constraint matrix A (size rows × n)
            A = np.zeros((rows, n), dtype=np.float64)
            for k in range(rows):
                low = max(0, k - (n - 1))
                high = min(n - 1, k)
                cols = np.arange(low, high + 1, dtype=int)
                A[k, cols] = a[k - cols]

            res = linprog(
                c=-np.ones(n),                # maximise Σg_j
                A_ub=A,
                b_ub=np.full(rows, M, dtype=np.float64),
                bounds=(0, None),
                method="highs",
                options={"presolve": True, "time_limit": 30},
            )
            if not res.success:
                break

            g = res.x
            s = g.sum()
            if s == 0:
                break
            g = g * (target / s)               # rescale onto simplex

            # convex blend (more aggressive than the original 0.2)
            t = 0.3
            a = (1.0 - t) * a + t * g
            _maybe_update(a)

            # stop if progress stalls
            if best_score < 1.503:
                break
            # simple stagnation test
            new_M = _conv_self(a).max()
            if new_M >= M - 1e-6:
                break

    # ------------------------------------------------------------------
    # 4.  Fallback to a safe uniform exponential shape (should never trigger)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        target = math.sqrt(2.0 * n)
        a = np.exp(-0.01 * np.arange(n, dtype=float))
        a = a / a.sum() * target
        best_seq = a

    # return a plain Python list (exactly the type expected by the harness)
    return best_seq.tolist()
