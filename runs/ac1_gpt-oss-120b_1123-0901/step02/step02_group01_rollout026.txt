def propose_candidate(seed=..., budget_s=1000):
    import time, numpy as np
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5

    best_seq = None; best_val = inf

    n = 2500

    def evaluate(seq): return _objective(seq)   # defined above


    # helper functions (project, conv, gradient) defined earlier.

    # 1. Generate initial seeds
    init_candidates = []

    # uniform
    uniform_seq = np.ones(n)/n
    init_candidates.append(uniform_seq)

    # geometric with epsilon values (list)
    for eps in np.logspace(-4, -1, 20):  # epsilon from 1e-4 to 0.1? ensure eps*n ~ 0.2 to 250? Not too large.
        r = 1 - eps
        if r <=0: continue
        # compute unnormalized powers
        i = np.arange(n, dtype=np.float64)
        powers = r**i
        seq = powers / powers.sum()
        init_candidates.append(seq)

    # power-law
    for p in [0.5, 1.0, 1.5, 2.0, 3.0]:
        idx = np.arange(1, n+1)
        seq = idx**(-p)
        seq = seq / seq.sum()
        init_candidates.append(seq)

    # random Dirichlet with concentration parameter 0.5, 1, 2
    for alpha in [0.3, 0.5, 1.0, 2.0]:
        seq = np.random.dirichlet(np.full(n, alpha))
        init_candidates.append(seq)

    # derived shaped Dirichlet where alpha_i decreasing
    for gamma in [0.5, 1.0, 2.0]:
        alphas = np.linspace(gamma, 0.0, n) + 0.01
        seq = np.random.dirichlet(alphas)
        init_candidates.append(seq)

    # Evaluate each, keep best
    for seq in init_candidates:
        val = _objective(seq)
        if val < best_val:
            best_val = val; best_seq = seq.copy()
            print(f"[Init] val={val:.6f}")

    # Optimization loop: process each init candidate maybe for some steps
    # For simplicity we will perform many iterations starting from best_seq and apply Adam.

    # define optimizer parameters
    lam = 1.0
    lam_factor = 1.03
    lr0 = 0.05
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    # Initialize m, v
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0

    seq = best_seq.copy()
    last_improve_time = time.time()
    iteration = 0

    # Pre-allocate FFT size
    fft_size = 1 << int(np.ceil(np.log2(3 * n)))   # enough for correlation

    while time.time() < deadline:
        # compute conv
        conv = _full_convolution(seq, size=fft_size)
        # compute gradient via smooth-max
        grad = _softmax_gradient(seq, conv, lam, size_corr=fft_size)
        # Adam update
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step_vec = lr0 * m_hat / (np.sqrt(v_hat) + eps_adam)
        # apply step
        seq = seq - step_vec
        # projection onto simplex
        seq = _project_onto_simplex(seq)
        # evaluate objective
        val = _objective(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            last_improve_time = time.time()
            print(f"[Iter {iteration:06d}] new best obj={val:.6f}")

        # lambda schedule

            # apply random shake
            noise = (np.random.rand(n) - 0.5) * 1e-3

            # reset Adam moments maybe
            m.fill(0)
            v.fill(0)
            t = 0

        iteration += 1

    print(f"Final best objective = {best_val:.6f} (length {n})")

"""
Propose a step‑function sequence with a small evaluation score.

The algorithm is a multi‑start Adam‑optimiser on a smooth‑max surrogate of the
true objective

    F(a) = 2·n·max_k (a*a)[k] / (sum a)^2 ,

subject to a_i ≥ 0 and sum a_i = 1 (the objective is scale‑invariant, therefore
the simplex constraint is imposed).  The surrogate replaces the hard max by the
log‑sum‑exp

    φ_λ(a) = (1/λ)·log Σ_k exp(λ·c_k) ,   c = a*a ,

whose gradient is  2·(w ★ a)  with soft‑max weights w_k ∝ exp(λ·c_k).
A continuation on λ (starting smooth, ending sharp) together with Adam‑style
adaptation and occasional random “shakes’’ provides a fast and robust descent.
Several analytically‑motivated initialisations (geometric, power‑law,
Dirichlet) are tried; the best result found within the time budget is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Projection onto the probability simplex {x≥0, sum(x)=1}
def _project_onto_simplex(v):
    """Euclidean projection of a vector onto the simplex."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        # all numbers non‑positive – return uniform distribution
        return np.full_like(v, 1.0 / len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear (non‑circular) convolution using FFT
def _full_convolution(a, size=None):
    """Linear self‑convolution a*a (length 2·len(a)‑1)."""
    n = len(a)
    if size is None:
        size = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Exact objective used by `evaluate_sequence`
def _objective(a):
    """2·n·max(conv) / (sum(a)^2), a is assumed non‑negative."""
    n = len(a)
    conv = _full_convolution(a)
    max_b = conv.max()
    sum_a = a.sum()
    # sum_a cannot be zero (we always keep a on the simplex)
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Gradient of the smooth‑max surrogate φ_λ
def _softmax_gradient(a, conv, lam, size_corr=None):
    """
    Gradient of (1/λ)·log Σ_k e^{λ·conv_k}
    Returns 2·(w ★ a) where w is the soft‑max over conv.
    """
    # numerical‑stable soft‑max over the convolution values
    m = conv.max()
    w = np.exp(lam * (conv - m))
    w /= w.sum()

    # cross‑correlation w ★ a  (linear, not circular)
    rev_a = a[::-1]
    if size_corr is None:
        size_corr = 1 << int(np.ceil(np.log2(len(w) + len(a) - 1)))
    fw = np.fft.rfft(w, size_corr)
    frev = np.fft.rfft(rev_a, size_corr)
    corr = np.fft.irfft(fw * frev, size_corr)

    n = len(a)
    # slice to obtain the part that corresponds to indices 0 … n‑1 of a
    grad = 2.0 * corr[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) with a small
    evaluation score.  The routine respects a 1000 s time limit, prints progress
    and finally returns the best sequence found (as a plain Python list).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5          # safety margin

    # --------------------------------------------------------------
    # choose a length that gives a good trade‑off between expressiveness
    # and speed.  2500 works well on typical hardware.
    n = 2500

    # initialise best holder (uniform is a safe fall‑back)
    best_seq = np.ones(n) / n
    best_val = _objective(best_seq)

    # --------------------------------------------------------------
    # 1.  Build a diverse set of seed vectors
    seeds = []

    # uniform
    seeds.append(np.ones(n) / n)

    # geometric families –  ε = 1‑r  (ε·n = α controls the shape)
    for eps in np.logspace(-4, -1, 25):          # ε from 1e‑4 … 0.1
        r = 1.0 - eps
        if r <= 0.0:
            continue
        powers = np.power(r, np.arange(n, dtype=np.float64))
        seq = powers / powers.sum()
        seeds.append(seq)

    # power‑law families a_i ∝ (i+1)^{-p}
    for p in [0.3, 0.6, 1.0, 1.5, 2.0]:
        idx = np.arange(1, n + 1, dtype=np.float64)
        seq = idx ** (-p)
        seq = seq / seq.sum()
        seeds.append(seq)

    # Dirichlet draws with various concentration parameters
    for conc in [0.2, 0.5, 1.0, 2.0]:
        seq = np.random.dirichlet(np.full(n, conc))
        seeds.append(seq)

    # decreasing concentration – larger mass at the left end
    for gamma in [0.5, 1.0, 2.0]:
        alphas = np.linspace(gamma + 0.01, 0.01, n)
        seq = np.random.dirichlet(alphas)
        seeds.append(seq)

    # --------------------------------------------------------------
    # 2.  Evaluate all seeds and keep the best one as a starting point
    for s in seeds:
        val = _objective(s)
        if val < best_val:
            best_val, best_seq = val, s.copy()
            print(f"[Init] objective={val:.6f}")

    # --------------------------------------------------------------
    # 3.  Refine the best seed with Adam on the smooth‑max surrogate.
    #    Continuation on λ drives the surrogate towards the true max.
    seq = best_seq.copy()
    lam = 1.0                     # start with a smooth surrogate
    lam_factor = 1.025            # increase λ slowly
    lr = 0.05                     # base learning rate for Adam
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0

    # FFT size that accommodates both convolution and correlation
    fft_size = 1 << int(np.ceil(np.log2(3 * n)))   # safe upper bound

    it = 0
    last_improve = time.time()

    while time.time() < deadline:
        # ---- convolution -------------------------------------------------
        conv = _full_convolution(seq, size=fft_size)

        # ---- smooth‑max gradient -----------------------------------------
        grad = _softmax_gradient(seq, conv, lam, size_corr=fft_size)

        # ---- Adam update -------------------------------------------------
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        seq = seq - step

        # ---- project back onto the simplex --------------------------------
        seq = _project_onto_simplex(seq)

        # ---- exact objective ------------------------------------------------
        val = _objective(seq)
        if val < best_val:
            best_val, best_seq = val, seq.copy()
            last_improve = time.time()
            print(f"[Iter {it:06d}] new best objective={val:.6f}")

        # ---- continuation of λ -------------------------------------------
        lam *= lam_factor

        # ---- occasional random shake if stagnated -------------------------
        if time.time() - last_improve > 5.0:
            noise = (np.random.rand(n) - 0.5) * 1e-3
            seq = _project_onto_simplex(seq + noise)
            # reset Adam moments after a shake – helps to escape flat regions
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            last_improve = time.time()

        it += 1

    print(f"Final best objective = {best_val:.6f} (length {n})")
    # convert to plain Python list as required by the evaluator
    return best_seq.tolist()
