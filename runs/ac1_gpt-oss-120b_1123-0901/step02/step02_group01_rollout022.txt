"""
Improved step‑function search using Adam‑based projected
gradient descent on a smooth‑max surrogate of the max
self‑convolution, combined with multi‑scale restarts.
The method first optimises a moderate length (≈2000) and then
interpolates to a larger length (≈5000) to exploit the extra
degrees of freedom.  Non‑negativity and the sum‑to‑one
constraint are enforced by a Euclidean projection onto the
probability simplex after each update.  Random “shakes’’ and
re‑initialisation on stagnation help escape shallow local
minima.  All operations are FFT‑based (O(n log n)) so that many
thousands of iterations fit into the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Projection onto the probability simplex {x ≥ 0, Σ x = 1}
def _project_onto_simplex(v):
    """ Euclidean projection of a vector onto the simplex. """
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        # all numbers are non‑positive -> uniform distribution
        return np.full_like(v, 1.0 / len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear self‑convolution a*a (length 2·len(a)-1) using FFT
def _full_convolution(a, size=None):
    n = len(a)
    if size is None:
        size = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Gradient of the smooth‑max surrogate (log‑sum‑exp)
def _softmax_gradient(a, conv, lam, size_corr=None):
    """
    Returns the gradient of (1/lam)·log ∑_k exp(lam·conv_k)
    with respect to the vector a (non‑negative, length n).
    """
    n = len(a)
    # soft‑max weights, numerically stable
    m = conv.max()
    w = np.exp(lam * (conv - m))
    w /= w.sum()

    # cross‑correlation w ★ a (linear, not circular)
    rev_a = a[::-1]
    if size_corr is None:
        size_corr = 1 << int(np.ceil(np.log2(len(w) + n - 1)))
    fw = np.fft.rfft(w, size_corr)
    frev = np.fft.rfft(rev_a, size_corr)
    corr = np.fft.irfft(fw * frev, size_corr)

    # gradient = 2·(w ★ a) taken on indices n‑1 … n‑1+n‑1
    grad = 2.0 * corr[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Exact objective used by `evaluate_sequence`
def _objective(a):
    """
    2·n·max(a*a) / (Σ a)^2   (a is assumed non‑negative)
    """
    n = len(a)
    conv = _full_convolution(a)
    max_b = conv.max()
    sum_a = a.sum()
    # sum_a is forced to be 1 in our construction, but keep formula generic
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Simple candidate generators (return a normalized np.ndarray)
def _geom_seq(n, r):
    """Geometric progression a_i = r^i  (r<1) normalised to sum 1."""
    powers = np.power(r, np.arange(n, dtype=np.float64))
    return powers / powers.sum()


def _v_shape_seq(n):
    """V‑shaped sequence, large at the ends, small in the centre."""
    mid = (n - 1) / 2.0
    a = np.abs(np.arange(n, dtype=np.float64) - mid) + 1.0
    return a / a.sum()


def _linear_desc_seq(n):
    """Linearly decreasing from left to right."""
    a = np.arange(n, 0, -1, dtype=np.float64)
    return a / a.sum()


def _random_dirichlet_seq(n):
    """Random point on the simplex."""
    return np.random.dirichlet(np.ones(n))


# ----------------------------------------------------------------------
def _interpolate_seq(seq, new_n):
    """Linear interpolation of a normalised sequence to a new length."""
    old_n = len(seq)
    old_x = np.arange(old_n, dtype=np.float64)
    new_x = np.linspace(0, old_n - 1, new_n)
    new_seq = np.interp(new_x, old_x, seq)
    # re‑normalise (tiny numerical error may appear)
    return new_seq / new_seq.sum()


# ----------------------------------------------------------------------
def _run_optimize(start_seq, deadline, best_val_global):
    """
    Projected Adam optimisation of the smooth‑max surrogate.
    Returns the best sequence found and its exact objective.
    """
    n = len(start_seq)
    # FFT sizes (fixed for the whole run)
    conv_fft_size = 1 << int(np.ceil(np.log2(2 * n - 1)))
    corr_fft_size = 1 << int(np.ceil(np.log2(3 * n)))   # >= (2n‑1)+n‑1

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.2                     # base learning rate
    max_abs_step = 0.5           # hard clip for the absolute step per coordinate
    lam = 5.0
    lam_factor = 1.02
    lam_max = 500.0

    seq = start_seq.copy()
    m = np.zeros_like(seq)
    v = np.zeros_like(seq)
    beta1_pow = 1.0
    beta2_pow = 1.0
    t = 0

    best_seq = seq.copy()
    best_val = best_val_global
    last_improve = time.time()
    last_shake = time.time()

    while time.time() < deadline:
        # --- compute convolution and gradient ---
        conv = _full_convolution(seq, size=conv_fft_size)
        grad = _softmax_gradient(seq, conv, lam, size_corr=corr_fft_size)

        # --- Adam update ---
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        # Clip the step to keep changes moderate
        max_abs = np.max(np.abs(step))
        if max_abs > max_abs_step:
            step = step * (max_abs_step / max_abs)

        new_seq = seq - step
        new_seq = _project_onto_simplex(new_seq)

        # --- exact objective ---
        val = _objective(new_seq)

        # keep the best ever found
        if val < best_val:
            best_val = val
            best_seq = new_seq.copy()
            last_improve = time.time()
            print(f"[Iter {t:07d}] new best = {val:.6f}")

        # --- occasional random shake to escape plateaus ---
        now = time.time()
        if now - last_shake > 2.0:
            noise = (np.random.rand(n) - 0.5) * 1e-3
            new_seq = _project_onto_simplex(new_seq + noise)
            last_shake = now

        # --- restart if we have not improved for a while ---
        if now - last_improve > 10.0:
            # fresh random start while preserving the global best
            seq = np.random.dirichlet(np.ones(n))
            m.fill(0.0)
            v.fill(0.0)
            beta1_pow = 1.0
            beta2_pow = 1.0
            t = 0
            lam = 5.0
            last_improve = now
            # skip the assignment at the end of the loop – we already have a fresh seq
            continue

        # prepare for next iteration
        seq = new_seq
        lam = min(lam * lam_factor, lam_max)

        # mild decay of the learning rate (optional)
        if t % 1000 == 0:
            lr *= 0.99

    return best_seq, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step‑function sequence with a small
    evaluation score.  The routine respects the 1000 s time budget
    (a short safety margin is kept), prints progress, and finally
    returns the best sequence discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ---- Stage 1: modest length, fast exploration --------------------
    n_small = 2000
    best_seq = None
    best_val = float("inf")

    # generate a few diverse starting points
    init_fns = [
        lambda n: _geom_seq(n, 0.9999),
        lambda n: _geom_seq(n, 0.995),
        lambda n: _geom_seq(n, 0.98),
        _v_shape_seq,
        _linear_desc_seq,
        _random_dirichlet_seq,
    ]

    for fn in init_fns:
        if time.time() > deadline:
            break
        seq = fn(n_small) if fn.__code__.co_argcount == 1 else fn(n_small)
        val = _objective(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[Init] {fn.__name__ if hasattr(fn, '__name__') else 'geom'} val={val:.6f}")

    # give the optimiser a fraction of the total budget on the small problem
    time_small = min(250.0, (deadline - time.time()) * 0.3)
    if time_small > 5.0 and best_seq is not None:
        small_deadline = time.time() + time_small
        best_seq, best_val = _run_optimize(best_seq, small_deadline, best_val)

    # ---- Stage 2: larger length, refined optimisation ---------------
    if time.time() < deadline:
        n_big = 5000
        # up‑sample the best small‑scale solution
        seq_big = _interpolate_seq(best_seq, n_big)
        # a few extra random restarts can be helpful
        extra_seq = _random_dirichlet_seq(n_big)
        extra_val = _objective(extra_seq)
        if extra_val < best_val:
            best_val = extra_val
            best_seq = extra_seq.copy()
        else:
            best_seq = seq_big

        # run optimisation on the large problem for the remaining time
        best_seq, best_val = _run_optimize(best_seq, deadline, best_val)

    print(f"Final best objective = {best_val:.6f} (length {len(best_seq)})")
    # convert to plain Python list, as required by the evaluator
    return best_seq.tolist()

        best_seq, best_val = _run_optimize(best_seq, small_deadline, best_val)

    # ---------- Stage 2 : larger size (≈5000) ----------
    if time.time() < deadline:
        n_big = 5000
        # up‑sample the best small‑scale solution
        seq_big = _interpolate_seq(best_seq, n_big)

        # a fresh random start can sometimes be beneficial
        rand_seq = _random_dirichlet_seq(n_big)
        rand_val = _objective(rand_seq)

        if rand_val < best_val:
            best_seq, best_val = rand_seq, rand_val
        else:
            best_seq = seq_big

        # optimise for the rest of the budget
        best_seq, best_val = _run_optimize(best_seq, deadline, best_val)

    print(f"Final best objective = {best_val:.6f} (length {len(best_seq)})")
    # convert to plain Python list as required by the evaluator
    return best_seq.tolist()
