   # correlation for gradient via FFT
   n = a.shape[0]
   m = w.shape[0]

Propose a step function (non‑negative heights) that minimizes
    2 * n * max(conv(a, a)) / (sum(a) ** 2)

using projected Adam on a smoothed version of the max‑convolution.
The smooth approximation is the log‑sum‑exp (softmax) of the

import numpy as np

# --------------------------------------------------------------------------- #
# Simplex projection (Euclidean) – Σ x_i = s, x_i ≥ 0
# --------------------------------------------------------------------------- #
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex of radius s."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------------------- #
# Full auto‑convolution (real FFT) – length 2*n‑1
# --------------------------------------------------------------------------- #
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Return a * a (full convolution) using a real FFT."""
    n = a.shape[0]
    n_fft = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    conv = np.fft.irfft(A * A, n=n_fft)[: 2 * n - 1]
    return conv


# --------------------------------------------------------------------------- #
# Gradient of the smoothed max‑convolution (log‑sum‑exp) using FFT correlation
# --------------------------------------------------------------------------- #
def _gradient_via_correlation(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute 2 * Σ_i a_i * w_{i+lag} for lag = 0..n‑1.
    w has length 2*n‑1 (the same length as the auto‑convolution).
    The result is the sub‑gradient of the soft‑max objective.
    """
    n = a.shape[0]
    m = w.shape[0]          # = 2*n‑1
    n_fft = 1 << ((n + m - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    W = np.fft.rfft(w, n=n_fft)
    # linear correlation (real part)
    corr = np.fft.irfft(A * np.conj(W), n=n_fft)[: n + m - 1]
    # the required lags are 0 … n‑1
    grad = 2.0 * corr[:n]
    return grad


# --------------------------------------------------------------------------- #
# One optimisation run (projected Adam on the soft‑max of the convolution)
# --------------------------------------------------------------------------- #
def _optimize_one_instance(
    n: int,
    init_vec: np.ndarray,
    run_deadline: float,
    *,
    target_sum: float,
) -> tuple[np.ndarray, float]:
    """
    Returns (best_sequence, best_score) found before *run_deadline*.
    ``best_score`` is the maximal auto‑convolution value (equal to the
    value returned by ``evaluate_sequence`` when the sum is fixed to
    sqrt(2 n)).
    """
    a = _project_onto_simplex(init_vec.astype(float), target_sum)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.05
    t = 0

    # smoothing parameter for the log‑sum‑exp
    alpha = 1.0
    alpha_max = 30.0
    # schedule to increase alpha a bit each iteration
    alpha_inc = 0.005

    best_local_seq = a.copy()
    best_local_score = float("inf")

    while time.time() < run_deadline:
        # --------------------------------------------------- #
        # 1) auto‑convolution and current objective value
        # --------------------------------------------------- #
        conv = _full_convolution(a)
        max_val = conv.max()
        if max_val < best_local_score:
            best_local_score = max_val
            best_local_seq = a.copy()
            print(
                f"[{time.time():.2f}] New local best = {max_val:.6f} (n={n})"
            )

        # --------------------------------------------------- #
        # 2) soft‑max weights for a smooth approximation of the max
        # --------------------------------------------------- #
        # subtract max to keep exponentials in [0,1]
        exp_arg = alpha * (conv - max_val)
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            # fall back to a uniform weight on the active (max‑tying) set
            active = np.where(conv >= max_val - 1e-12)[0]
            w = np.zeros_like(conv)
            if active.size:
                w[active] = 1.0 / active.size
        else:
            w /= w_sum

        # --------------------------------------------------- #
        # 3) gradient of the smoothed objective (Adam step)
        # --------------------------------------------------- #
        grad = _gradient_via_correlation(a, w)

        # Adam update (descent: move opposite to gradient)
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step

        # --------------------------------------------------- #
        # 4) occasional random perturbation (simulated annealing)
        # --------------------------------------------------- #
        if np.random.rand() < 0.01:
            noise = np.random.randn(n) * (0.02 * target_sum / n)
            a_new = a_new + noise
            # shrink learning rate a little after a shake
            lr *= 0.95

        # --------------------------------------------------- #
        # 5) project back onto the simplex and continue
        # --------------------------------------------------- #
        a = _project_onto_simplex(a_new, target_sum)

        # --------------------------------------------------- #
        # 6) schedule updates
        # --------------------------------------------------- #
        lr *= 0.9995                # slow decay
        alpha = min(alpha_max, alpha + alpha_inc)

    return best_local_seq, best_local_score

"""
Propose a step function (non‑negative heights) that minimizes

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

using a projected Adam optimiser on a *smoothed* version of the
max‑convolution (log‑sum‑exp).  The gradient of the smooth objective is
obtained via a single FFT‑based correlation, which is cheap
(O(N log N)).  Many short independent runs with random lengths
(≈1300‑3000) are performed; the best sequence is stored.  If time
remains, a final longer refinement is run on the current best
candidate.  Projecting onto the simplex {x_i ≥ 0, Σx_i = √(2 n)} keeps
the problem scale‑invariant, so the evaluated objective equals the
maximal auto‑convolution entry.
"""

import time
import numpy as np

# --------------------------------------------------------------------------- #
# Euclidean projection onto the simplex { x_i ≥ 0 , Σ x_i = s }.
# --------------------------------------------------------------------------- #
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector ``v`` onto the simplex of radius ``s``."""
    if s < 0.0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    # Sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1, dtype=float)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# --------------------------------------------------------------------------- #
def _full_convolution(a: np.ndarray) -> np.ndarray:

    n = a.shape[0]

    A = np.fft.rfft(a, n=n_fft)
    conv = np.fft.irfft(A * A, n=n_fft)[: 2 * n - 1]
    return conv
