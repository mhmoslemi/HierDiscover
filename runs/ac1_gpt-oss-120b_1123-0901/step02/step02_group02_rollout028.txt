def _conv_fft_general(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    m = b.shape[0]
    size = 1 << ((n + m - 1 - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa * fb, n=size)[:n+m-1]
    np.maximum(conv, 0.0, out=conv)
    return conv
```

def _conv_fft_general(a,b):
    n = a.shape[0]
    m = b.shape[0]
    size = 1 << ((n + m - 1 - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa * fb, n=size)[:n+m-1]
    np.maximum(conv, 0.0, out=conv)
    return conv
```

def _optimise_one_length(
    n,
    init,
    run_deadline,
    *,
    target_sum=1.0,
    base_step=0.12,
    momentum_beta=0.9,
    alpha_start=5.0,
    alpha_factor=1.02,
    alpha_max=50.0,
    step_decay=0.9997,
    jitter_prob=0.01,
    jitter_scale=1e-3,
):
    # ensure init is array float
    a = np.array(init, dtype=np.float64)
    a = _project_onto_simplex(a, target_sum)
    best = a.copy()
    conv = _conv_fft(a)
    max_val = conv.max()
    best_score = 2 * n * max_val / (a.sum() ** 2)
    velocity = np.zeros_like(a)
    step = base_step
    alpha = alpha_start
    while time.time() < run_deadline:
        conv = _conv_fft(a)
        max_val = conv.max()
        # Softmax weights
        # shift by max to avoid overflow
        w = np.exp(alpha * (conv - max_val))
        # gradient via convolution of w with reversed a
        a_rev = a[::-1]
        grad_full = _conv_fft_general(w, a_rev)
        # extract middle part (indices n-1 .. n-1+n-1)
        grad = 2.0 * grad_full[n-1:n-1+n]
        # momentum
        velocity = momentum_beta * velocity + (1.0 - momentum_beta) * grad
        # step size based on gradient magnitude
        norm = np.linalg.norm(velocity)
        if norm == 0:
            step_factor = 0.0
        else:
            step_factor = step / norm
        a_new = a - step_factor * velocity
        # projection onto simplex
        a_new = _project_onto_simplex(a_new, target_sum)
        # occasional random jitter to escape flats
        if np.random.rand() < jitter_prob:
            noise = np.random.randn(n) * (target_sum * jitter_scale / n)
            a_new = _project_onto_simplex(np.maximum(a_new + noise, 0.0), target_sum)
            step *= 0.92
        # evaluate
        conv_new = _conv_fft(a_new)
        max_val_new = conv_new.max()
        score = 2 * n * max_val_new / (a_new.sum() ** 2)
        if score < best_score:
            best_score = score
            best = a_new.copy()
            # Print progress
            print(f"[{time.time():.2f}] n={n}   improved local best: {score:.6f}")
        a = a_new
        # update schedules
        step *= step_decay
        alpha = min(alpha * alpha_factor, alpha_max)
    return best, best_score

def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative height sequence that minimises
    2 * n * max(conv(a, a)) / (sum(a)^2).

    The algorithm performs:
        1. a quick grid/random search over lengths and power‑law exponents
           to obtain a good starting shape;
        2. many short projected‑gradient runs (soft‑max weighted sub‑gradient
           descent) on varied lengths;
        3. a final long refinement on the current best length.
    """

    deadline = time.time() + budget_s - 0.3

"""
Propose a step‑function (non‑negative vector) that minimizes

        2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm works in three stages:

1. **Fast power‑law sweep** – random lengths in [800, 3500] and
   exponents p∈[0.4,0.6] are tried; the best shape becomes the seed for
   the later optimisation.

2. **Projected soft‑max gradient descent** – for a fixed length we minimise
   the smooth approximation of the ℓ∞‑norm of the autocorrelation.
   The gradient of  Σ_k w_k·(a*a)_k  (weights w_k are a softmax of the
   convolution entries) is obtained by a single FFT‑based convolution.
   An exponential moving‑average (momentum) is used, the iterate is
   projected onto the simplex {x≥0, Σx = 1} and occasional jitter helps
   to escape plateaus.  Many independent short runs with different
   initialisations explore the search space.

3. **Long refinement** – after the budget is mostly spent the best
   discovered length is refined for the remaining time with a slower
   step‑size schedule.

The whole routine runs for at most ``budget_s`` seconds and returns the
best sequence found (as a plain Python list).  The implementation uses
only NumPy and the standard library, respects all constraints of
``evaluate_sequence`` and prints progress information.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper – Euclidean projection onto the simplex {x_i ≥ 0, Σ x_i = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex of radius s (s ≥ 0)."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper – full (linear) convolution via real FFT
# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full convolution a * a (real inputs) using FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    size = 1 << ((m - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[:m]
    np.maximum(conv, 0.0, out=conv)            # clip tiny negatives
    return conv


def _conv_fft_general(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two (real) vectors a and b using FFT."""
    n, m = a.shape[0], b.shape[0]
    size = 1 << ((n + m - 1 - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa * fb, n=size)[: n + m - 1]
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Helper – power‑law initialisation
# ----------------------------------------------------------------------
def _power_law_init(n: int, p: float) -> np.ndarray:
    """Return a vector a_i ∝ (i+1)^{-p} with a tiny random jitter."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    a += np.random.rand(n) * 1e-5
    return a


# ----------------------------------------------------------------------
# Core optimisation for a fixed length using a soft‑max weighted gradient
# ----------------------------------------------------------------------
def _optimise_one_length(
    n: int,
    init: np.ndarray,
    run_deadline: float,
    *,
    target_sum: float = 1.0,
    base_step: float = 0.12,
    momentum_beta: float = 0.9,
    alpha_start: float = 5.0,
    alpha_factor: float = 1.02,
    alpha_max: float = 50.0,
    step_decay: float = 0.9997,
    jitter_prob: float = 0.01,
    jitter_scale: float = 1e-3,
) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent with a soft‑max approximation of the max‑convolution.
    Returns (best_vector, best_score) for the given length.
    """
    a = np.array(init, dtype=np.float64)
    a = _project_onto_simplex(a, target_sum)

    # keep the best vector seen in this run
    conv = _conv_fft(a)
    max_val = conv.max()
    best_local = a.copy()
    best_score = 2.0 * n * max_val / (a.sum() ** 2)

    velocity = np.zeros_like(a)
    step = base_step
    alpha = alpha_start

    while time.time() < run_deadline:
        # 1) current convolution and its maximum
        conv = _conv_fft(a)
        max_val = conv.max()

        # 2) soft‑max weights (more weight on larger entries)
        w = np.exp(alpha * (conv - max_val))

        # 3) gradient of the weighted sum of convolutions
        a_rev = a[::-1]                              # reversed a
        grad_full = _conv_fft_general(w, a_rev)      # length 3n‑2
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]    # extract n components

        # 4) momentum update
        velocity = momentum_beta * velocity + (1.0 - momentum_beta) * grad

        # 5) take a step opposite to the gradient and project back
        a_new = a - step * velocity
        a_new = _project_onto_simplex(a_new, target_sum)

        # 6) occasional random jitter to escape plateaus
        if np.random.rand() < jitter_prob:
            noise = np.random.randn(n) * (target_sum * jitter_scale / n)
            a_new = _project_onto_simplex(np.maximum(a_new + noise, 0.0), target_sum)
            step *= 0.92

        # 7) evaluate new candidate
        conv_new = _conv_fft(a_new)
        max_val_new = conv_new.max()
        score = 2.0 * n * max_val_new / (a_new.sum() ** 2)

        # 8) keep the best seen locally
        if score < best_score:
            best_score = score
            best_local = a_new.copy()
            print(f"[{time.time():.2f}] n={n}  local improvement → {score:.6f}")

        a = a_new
        step *= step_decay
        alpha = min(alpha * alpha_factor, alpha_max)

    return best_local, best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative height sequence that minimises
    `2 * n * max(conv(a, a)) / (sum(a) ** 2)`.
    The routine runs for at most ``budget_s`` seconds and returns the best
    sequence discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.3   # tiny safety margin

    # ------------------------------------------------------------------
    # Initialise with the (optional) provided global sequence
    # ------------------------------------------------------------------
    global_seq = globals().get("height_sequence_1")
    if isinstance(global_seq, (list, np.ndarray)):
        best_seq = np.asarray(global_seq, dtype=np.float64)
        best_seq = best_seq / best_seq.sum()          # normalise to sum=1
        best_score = 2.0 * best_seq.size * np.max(_conv_fft(best_seq)) / (best_seq.sum() ** 2)
        print(f"[{time.time():.2f}] Using supplied sequence, score={best_score:.6f}")
    else:
        best_seq = None
        best_score = float("inf")

    # ------------------------------------------------------------------
    # Stage 1 – cheap exploration of power‑law families (≈30 s)
    # ------------------------------------------------------------------
    explore_deadline = min(deadline, time.time() + 30.0)
    while time.time() < explore_deadline:
        n = np.random.randint(800, 3500)
        p = np.random.uniform(0.4, 0.6)
        init = _power_law_init(n, p)
        init = init / init.sum()                       # sum = 1
        conv = _conv_fft(init)
        score = 2.0 * n * conv.max() / (init.sum() ** 2)
        if score < best_score:
            best_score = score
            best_seq = init.copy()
            print(f"[{time.time():.2f}] Exploration new best {score:.6f} (n={n}, p={p:.3f})")
        if best_score < 1.48:                         # early stop if already good
            break

    # ------------------------------------------------------------------
    # Stage 2 – many short restarts with projected soft‑max gradient descent
    # ------------------------------------------------------------------
    while time.time() < deadline:
        remaining = deadline - time.time()
        # pick a length; keep it modest when little time is left
        if remaining < 20.0:
            n = np.random.randint(800, 1500)
        else:
            n = np.random.randint(800, 3500)

        # initialise: power‑law + occasional warm‑start from the current best
        p = np.random.uniform(0.45, 0.55)
        init = _power_law_init(n, p)
        if best_seq is not None and np.random.rand() < 0.35:
            if best_seq.size >= n:
                init = best_seq[:n].copy()
            else:
                pad = np.random.rand(n - best_seq.size) * 1e-5
                init = np.concatenate([best_seq, pad])

        # allocate a short time slice (≈6 s) for this run
        run_deadline = min(deadline, time.time() + 6.0)

        seq_local, score_local = _optimise_one_length(
            n,
            init,
            run_deadline,
            target_sum=1.0,
            base_step=0.14,
            momentum_beta=0.9,
            alpha_start=4.0,
        )

        if score_local < best_score:
            best_score = score_local
            best_seq = seq_local.copy()
            print(f"[{time.time():.2f}] GLOBAL new best = {best_score:.6f} (length={n})")

        if best_score < 1.45:               # comfortably below the target
            break

    # ------------------------------------------------------------------
    # Stage 3 – final long refinement of the best‑so‑far length
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        n = best_seq.size
        seq_refined, refined_score = _optimise_one_length(
            n,
            best_seq,
            deadline,                         # use whatever time is left
            target_sum=1.0,
            base_step=0.09,
            momentum_beta=0.95,
            alpha_start=6.0,
        )
        if refined_score < best_score:
            best_score = refined_score
            best_seq = seq_refined.copy()
            print(f"[{time.time():.2f}] FINAL refined best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback in the very unlikely event that nothing was found
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the range accepted by ``evaluate_sequence`` (non‑negative, ≤1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Return a plain Python list as required by the evaluator
    return best_seq.tolist()
