def _gradient_via_softmax(a, lam):
    n = a.shape[0]
    # conv of a with a
    conv = _conv_full_fft(a) # length 2n-1
    max_conv = conv.max()
    # compute softmax weights
    # shift for numeric stability
    m = conv.max()
    exps = np.exp(lam * (conv - m))
    w = exps / np.sum(exps)
    # compute gradient
    a_rev = a[::-1]
    conv_full = _conv_full_fft_two(a_rev, w)  # length 3n-2
    start = n - 1
    grad = 2.0 * conv_full[start:start + n]
    return grad, max_conv, conv
```

def _conv_full_fft_two(x, y):
    n = x.shape[0] + y.shape[0] - 1
    n_fft = 1 << ((n).bit_length())
    X = np.fft.rfft(x, n_fft)
    Y = np.fft.rfft(y, n_fft)
    conv = np.fft.irfft(X * Y, n_fft)[:n]
    return conv
```

"""
Propose a step‑function (non‑negative heights) that minimizes

        2 * n * max(conv(a, a)) / (sum(a) ** 2)

using a *smooth‑max* surrogate and Adam optimisation.

The algorithm:
    1. Work on the simplex {a_i ≥ 0, Σ a_i = √(2 n)} – this makes the
       objective equal to max(conv(a,a)).
    2. Replace the hard max by a soft‑max:
           w_i = exp(λ·c_i) / Σ_j exp(λ·c_j)    (c = a⋆a)
       The gradient of the surrogate is
           ∇ = 2· (flip(a) ⋆ w)  (slice to length n)
       which can be evaluated with two FFT‑based convolutions.
    3. Apply Adam (adaptive moment) updates and project back onto the simplex.
    4. λ is increased slowly (continuation) so the method first “sees” a
       smooth objective and later focuses on the true maximum.
    5. Many independent restarts with different lengths (≈1500‑6000) are
       performed until the time budget expires.  The best sequence seen
       is returned.

The code is fully self‑contained, uses only NumPy and the provided
`evaluate_sequence` for final verification, and respects the 1000 s budget.
"""

import time
import math
import numpy as np

# --------------------------------------------------------------------------- #
# Helper: Euclidean projection onto the simplex {x_i ≥ 0 , Σ x_i = s }.
# --------------------------------------------------------------------------- #
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector `v` onto the simplex of radius `s`."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1, dtype=float)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------------------- #
# Fast full convolution (real FFT) – returns length (len(x)+len(y)-1) array.
# --------------------------------------------------------------------------- #
def _conv_full_fft_two(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Convolution of two real vectors using a power‑of‑two FFT."""
    n = x.shape[0] + y.shape[0] - 1
    n_fft = 1 << ((n - 1).bit_length())          # next power of two
    X = np.fft.rfft(x, n=n_fft)
    Y = np.fft.rfft(y, n=n_fft)
    conv = np.fft.irfft(X * Y, n=n_fft)[:n]
    return conv


def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    """Convenient wrapper for a⋆a."""
    return _conv_full_fft_two(a, a)


# --------------------------------------------------------------------------- #
# Gradient of the soft‑max surrogate.
# --------------------------------------------------------------------------- #
def _gradient_softmax(a: np.ndarray, lam: float) -> tuple[np.ndarray, float, np.ndarray]:
    """
    Returns (grad, max_conv, full_conv) where
        grad    – ∇ of  (1/λ)·log Σ exp(λ·conv)
        max_conv – max entry of the convolution a⋆a
        full_conv – the whole convolution vector (length 2·n‑1)
    """
    n = a.shape[0]
    # full convolution a⋆a
    conv = _conv_full_fft(a)               # length 2n‑1
    max_val = conv.max()

    # soft‑max weights (stable)
    shift = max_val
    exps = np.exp(lam * (conv - shift))
    w = exps / exps.sum()                  # sum_i w_i = 1, w_i > 0

    # gradient = 2 * (flip(a) ⋆ w)  restricted to indices n‑1 … 2n‑2
    a_rev = a[::-1]
    conv_grad_full = _conv_full_fft_two(a_rev, w)   # length 3n‑2
    start = n - 1
    grad = 2.0 * conv_grad_full[start:start + n]
    return grad, max_val, conv


# --------------------------------------------------------------------------- #
# Main entry point expected by the grader.
# --------------------------------------------------------------------------- #
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a low‑value step‑function within ``budget_s`` seconds.
    The best sequence discovered is returned as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # small safety margin

    best_seq = None          # numpy array of the best candidate
    best_score = float("inf")  # objective value (max convolution)

    # If a previous construction is available, keep a reference.
    previous = globals().get("height_sequence_1", None)
    if previous is not None:
        previous = np.asarray(previous, dtype=float)

    # ------------------------------------------------------------------- #
    # Helper: run a single optimisation instance on a given length ``n``.
    # ------------------------------------------------------------------- #
    def _run_one_instance(n: int, init_vec: np.ndarray):
        nonlocal best_seq, best_score

        target_sum = math.sqrt(2.0 * n)          # Σ a_i = √(2n)

        # initialise on simplex
        a = _project_onto_simplex(init_vec.astype(float), target_sum)

        # Adam hyper‑parameters
        lr = 0.05
        beta1 = 0.90
        beta2 = 0.999
        eps = 1e-8
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        t_iter = 0

        # continuation parameter for the soft‑max
        lam = 0.5
        lam_factor = 1.03
        lam_max = 30.0

        # allocate a slice of the overall budget for this run
        run_deadline = min(deadline, time.time() + 20.0)  # ~20 s per run

        while time.time() < run_deadline:
            grad, max_conv, conv = _gradient_softmax(a, lam)

            # current objective (identical to evaluate_sequence for our scaling)
            sum_a = a.sum()
            cur_score = 2.0 * n * max_conv / (sum_a * sum_a)

            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time():.2f}] New best {best_score:.6f} (n={n}, λ={lam:.2f})")
                # early stop if target reached
                if best_score < 1.503:
                    return

            # Adam update (gradient descent)
            t_iter += 1
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)

            # bias‑correction
            m_hat = m / (1.0 - beta1 ** t_iter)
            v_hat = v / (1.0 - beta2 ** t_iter)

            step = lr * m_hat / (np.sqrt(v_hat) + eps)
            a = a - step

            # project back onto the feasible simplex
            a = _project_onto_simplex(np.maximum(a, 0.0), target_sum)

            # increase λ (smooth‑max becomes sharper)
            lam = min(lam * lam_factor, lam_max)

            # occasional random shake to escape shallow minima
            if np.random.rand() < 0.01:
                noise = np.random.randn(n) * (0.01 * target_sum / n)
                a = _project_onto_simplex(np.maximum(a + noise, 0.0), target_sum)

    # ------------------------------------------------------------------- #
    # Outer loop – many independent restarts with varying lengths.
    # ------------------------------------------------------------------- #
    while time.time() < deadline:
        # Choose problem size.  Larger n provides more flexibility but costs
        # more per iteration; we sample a wide range.
        n = np.random.randint(1500, 6001)

        # Initialise either from the previous good construction or randomly.
        if previous is not None and np.random.rand() < 0.4:
            src = previous
            if src.size >= n:
                init = src[:n].copy()
            else:
                pad = np.random.rand(n - src.size) * 1e-3
                init = np.concatenate([src, pad])
        else:
            init = np.random.rand(n)

        _run_one_instance(n, init)

        # Exit early if we already beat the target bound.
        if best_score < 1.503:
            break

    # ------------------------------------------------------------------- #
    # Fallback – if for some reason nothing was found.
    # ------------------------------------------------------------------- #
    if best_seq is None:
        n = 2000
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)

    # Return a plain Python list as required by the grader.
    return best_seq.tolist()
