"""
Propose a step‑function (non‑negative sequence) that improves the known
record (objective ≈ 1.5449).  The search combines three ingredients:

1. **Broad parametric seeding** – many simple families (geometric,
   power‑law, beta‑like, triangular, random Dirichlet) are sampled and the
   best member is kept.

2. **Hybrid descent** – at each iteration we
   * compute the active set of convolution entries that are within a tiny
     tolerance of the current maximum,
   * build a sub‑gradient of the maximal convolution entry from *all* active
     shifts,
   * try two kinds of simplex steps:
        – additive (projected gradient with back‑tracking line‑search),
        – multiplicative (exponentiated gradient, automatically stays on the
          simplex),
   * if neither gives improvement we perform a small random mass‑transfer
     move, and occasionally a full random restart.

3. **Occasional LP boost** – when the descent stagnates we solve a small
   linear program (if SciPy is available) that maximises the total mass of a
   partner sequence `g` under the current convolution bound.  Mixing the
   current point with the normalised solution often lowers the maximal
   autocorrelation.

The algorithm works on the probability simplex (sum = 1), because the
objective is homogeneous.  All operations are vectorised and use FFT‑based
convolution for speed.  The time budget (≈ 1000 s) is respected by a global
deadline check.

The function returns a list of floats that can be fed directly to
`evaluate_sequence`.
"""

import time
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex {x ≥ 0, sum(x)=1}
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.size + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        # All entries are zero; return uniform distribution
        return np.full_like(v, 1.0 / v.size)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast self‑convolution using real FFT
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _self_convolve(seq: np.ndarray) -> np.ndarray:
    """Full linear convolution of seq with itself (length 2n‑1)."""
    n = seq.shape[0]
    m = 2 * n - 1
    nfft = _next_pow2(m)
    A = np.fft.rfft(seq, n=nfft)
    conv = np.fft.irfft(A * A, n=nfft)[:m]
    return conv


# ----------------------------------------------------------------------
# Local fast objective (same as evaluate_sequence for sum(seq)=1)
def _objective(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) for a simplex vector."""
    sum_a = seq.sum()
    if sum_a < 0.01:
        return float("inf")
    conv = _self_convolve(seq)
    max_b = conv.max()
    return float(2.0 * len(seq) * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Parametric families ---------------------------------------------------
def _geom_seq(n: int, alpha: float) -> np.ndarray | None:
    """Geometric progression a_i ∝ r^i with r = 1 - alpha/n."""
    eps = alpha / n
    r = 1.0 - eps
    if r <= 0.0:
        return None
    powers = r ** np.arange(n, dtype=np.float64)
    seq = powers / powers.sum()
    return seq


def _powerlaw_seq(n: int, p: float) -> np.ndarray:
    """Power‑law a_i ∝ (i+1)^{-p}."""
    seq = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    return seq / seq.sum()


def _beta_seq(n: int, a: float, b: float) -> np.ndarray:
    """Beta‑like shape a_i ∝ (i+1)^{a‑1}(n‑i)^{b‑1}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    seq = (i ** (a - 1.0)) * ((n + 1 - i) ** (b - 1.0))
    seq = np.where(np.isfinite(seq), seq, 0.0)
    seq[seq < 0] = 0.0
    if seq.sum() == 0:
        seq = np.full(n, 1.0 / n)
    return seq / seq.sum()


def _triangular_seq(n: int) -> np.ndarray:
    """Symmetric triangular shape (rise then fall)."""
    # peak at the centre
    half = (n + 1) // 2
    inc = np.arange(1, half + 1, dtype=np.float64)
    dec = inc[:-1][::-1] if n % 2 == 1 else inc[::-1]
    seq = np.concatenate([inc, dec])
    return seq / seq.sum()


def _random_simplex(n: int) -> np.ndarray:
    """Draw a random point on the simplex (Dirichlet(1,…,1))."""
    x = np.random.exponential(1.0, size=n)
    return x / x.sum()


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative sequence (step function) that minimises
    evaluate_sequence(...).  Returns the best sequence found within the
    allotted time.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 1.0   # a little safety margin

    # 1️⃣ Choose a moderate length – long enough to express structure,
    #    short enough for fast convolutions.
    n = 2500

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 2️⃣ Systematic sweep over parametric families (more granular than before)
    # ------------------------------------------------------------------
    # geometric family
    for alpha in np.linspace(0.2, 6.0, 30):
        if time.time() > deadline:
            break
        seq = _geom_seq(n, alpha)
        if seq is None:
            continue
        score = _objective(seq)
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"[Init] geom α={alpha:.3f}  score={score:.6f}")

    # power‑law family
    for p in np.linspace(0.1, 3.5, 30):
        if time.time() > deadline:
            break
        seq = _powerlaw_seq(n, p)
        score = _objective(seq)
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"[Init] power p={p:.3f}  score={score:.6f}")

    # beta‑like family (two parameters)
    for a in np.linspace(0.2, 5.0, 7):
        for b in np.linspace(0.2, 5.0, 7):
            if time.time() > deadline:
                break
            seq = _beta_seq(n, a, b)
            score = _objective(seq)
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"[Init] beta a={a:.2f} b={b:.2f}  score={score:.6f}")

    # triangular shape (deterministic)
    seq = _triangular_seq(n)
    score = _objective(seq)
    if score < best_score:
        best_score = score
        best_seq = seq.copy()
        print(f"[Init] triangular   score={score:.6f}")

    # pure random Dirichlet points
    for _ in range(50):
        if time.time() > deadline:
            break
        seq = _random_simplex(n)
        score = _objective(seq)
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"[Init] random   score={score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣ Fallback (should never be needed)
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)
        best_score = _objective(best_seq)

    seq = best_seq.copy()
    cur_score = best_score

    # params for the descent loops
    base_step = 1.0          # additive step size
    base_eta = 0.5           # multiplicative (exponential) step size
    iter_cnt = 0
    stagnation = 0
    last_improve_time = time.time()

    # optional LP import (may be absent)
    try:
        import scipy.optimize as _opt
        have_lp = True
    except Exception:
        have_lp = False

    # ------------------------------------------------------------------
    # 4️⃣ Main optimisation loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # ----- compute current convolution and identify active shifts -----
        conv = _self_convolve(seq)
        M = conv.max()
        # Using a relative tolerance selects all entries that are essentially maximal
        rel_tol = 1e-4
        active_idx = np.where(conv >= M * (1.0 - rel_tol))[0]

        # ----- sub‑gradient from all active shifts -----
        grad = np.zeros_like(seq)
        for k in active_idx:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=int)
            # contribution a_{k-i}
            grad[idx] += seq[k - idx]

        # ----- try additive projected gradient step -----
        step = base_step
        cand = None
        cand_score = None
        improved = False
        while step > 1e-6:
            cand_try = _project_onto_simplex(seq - step * grad)
            cand_try_score = _objective(cand_try)
            if cand_try_score < cur_score:
                cand = cand_try
                cand_score = cand_try_score
                improved = True
                break
            step *= 0.5

        # ----- if additive fails, try multiplicative (exponentiated) step -----
        if not improved:
            eta = base_eta
            while eta > 1e-4:
                # exponential move: x_i ← x_i·exp(−η·grad_i)
                exp_term = np.exp(-eta * grad)
                cand_try = seq * exp_term
                if cand_try.sum() == 0:
                    eta *= 0.5
                    continue
                cand_try = cand_try / cand_try.sum()
                cand_try_score = _objective(cand_try)
                if cand_try_score < cur_score:
                    cand = cand_try
                    cand_score = cand_try_score
                    improved = True
                    break
                eta *= 0.5

        # ----- accept the best candidate if it improves -----
        if improved and cand_score < cur_score:
            seq = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = seq.copy()
                print(f"[Iter {iter_cnt:05d}] new best = {best_score:.6f}")
                last_improve_time = time.time()
                stagnation = 0
                # slightly enlarge additive step after a win
                base_step = max(step * 1.5, 1e-4)
                base_eta = max(eta * 1.2, 1e-4)
            else:
                stagnation += 1
        else:
            # ----- no improvement: random mass‑transfer move -----
            stagnation += 1
            i, j = np.random.choice(n, size=2, replace=False)
            delta = 0.01 * np.random.rand() * min(seq[i], 0.1)
            if delta > 0:
                cand = seq.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_onto_simplex(cand)
                cand_score = _objective(cand)
                if cand_score < cur_score:
                    seq = cand
                    cur_score = cand_score
                    if cand_score < best_score:
                        best_score = cand_score
                        best_seq = seq.copy()
                        print(f"[Iter {iter_cnt:05d}] random move improved to {best_score:.6f}")
                        last_improve_time = time.time()
                        stagnation = 0
                    else:
                        # keep the move even if not globally best
                        pass

        # ----- occasional LP boost when we appear stuck -----
        if stagnation >= 200 and have_lp and (time.time() < deadline - 5):
            try:
                rows = 2 * n - 1
                # Build sparse constraint matrix A (shape rows × n)
                from scipy.sparse import lil_matrix
                A = lil_matrix((rows, n), dtype=np.float64)
                # For each shift k we need: sum_i seq[i] * g_{k-i} ≤ M
                for k in range(rows):
                    lo = max(0, k - (n - 1))
                    hi = min(k, n - 1)
                    # variable index j runs from lo..hi, coefficient = seq[k-j]
                    # Fill the row k
                    for j in range(lo, hi + 1):
                        A[k, j] = seq[k - j]
                A_csr = A.tocsr()
                b = np.full(rows, M, dtype=np.float64)
                c = -np.ones(n, dtype=np.float64)   # maximise sum(g)  ⇔ minimise -sum(g)
                bounds = [(0, None) for _ in range(n)]
                # Solve with HiGHS (fast interior‑point)
                res = _opt.linprog(c, A_ub=A_csr, b_ub=b, bounds=bounds,
                                   method='highs', options={'presolve': True})
                if res.success:
                    g_raw = res.x
                    total_g = g_raw.sum()
                    if total_g > 1e-12:
                        g_norm = g_raw / total_g
                        # Try mixing with a few step sizes
                        for t in (0.4, 0.2, 0.1, 0.05):
                            mixed = (1 - t) * seq + t * g_norm
                            mixed = _project_onto_simplex(mixed)
                            mixed_score = _objective(mixed)
                            if mixed_score < cur_score:
                                seq = mixed
                                cur_score = mixed_score
                                if mixed_score < best_score:
                                    best_score = mixed_score
                                    best_seq = seq.copy()
                                    print(f"[Iter {iter_cnt:05d}] LP‑mix improved to {best_score:.6f}")
                                    last_improve_time = time.time()
                                    stagnation = 0
                                break
                else:
                    # LP failed – ignore
                    pass
            except Exception as e:
                # any unexpected issue just abort the LP attempt
                print(f"[Iter {iter_cnt:05d}] LP step exception: {e}")

        # ----- occasional full restart if we have been stagnant a long time -----
        if stagnation >= 500:
            seq = _random_simplex(n)
            cur_score = _objective(seq)
            print(f"[Iter {iter_cnt:05d}] restart triggered, score={cur_score:.6f}")
            stagnation = 0

        # ----- early exit if we have already reached the target -----
        if best_score < 1.503:
            print(f"Target reached (score={best_score:.6f}) at iteration {iter_cnt}")
            break

        iter_cnt += 1

    print(f"Final best objective = {best_score:.6f} (length {n})")
    # convert to plain Python list of floats as required by the evaluator
    return best_seq.tolist()
