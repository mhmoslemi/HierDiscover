def _linear_correlation(w, a, size_corr=None):
    n = len(a)
    rev_a = a[::-1]
    if size_corr is None:
        size_corr = 1 << int(np.ceil(np.log2(len(w) + n - 1)))
    fw = np.fft.rfft(w, size_corr)
    frev = np.fft.rfft(rev_a, size_corr)
    corr = np.fft.irfft(fw * frev, size_corr)
    # slice
    return corr[n-1:n-1+n]

Thus gradient = 2*corr.

"""
Propose a step‑function sequence minimizing the evaluation score.

Algorithm Overview
------------------
1. **Initialisation** – generate a few geometric sequences, random
   Dirichlet vectors and (if available) the provided global solution
   `height_sequence_1`.  Keep the best one (objective evaluated with the
   exact formula).

2. **Smooth‑max optimisation** – minimise a smooth surrogate of the
   max‑convolution using projected Adam on the probability simplex.
   The surrogate is
        φ_λ(a) = (1/λ)·log Σ_k exp(λ·(a*a)[k]),
   whose gradient is 2·(w ★ a) where w are the soft‑max weights.
   λ is increased slowly (continuation) until it becomes large
   (≈200).  Then we switch to the true sub‑gradient of the max.

3. **Random Shake** – if no improvement for >2 s a tiny random
   perturbation (projected onto the simplex) is added to escape shallow
   plateaus.

4. **Final local search** – with the remaining time we perform a simple
   hill‑climbing / simulated‑annealing phase that adds Gaussian noise
   of decreasing magnitude, always re‑projecting onto the simplex.
   This often yields a few × 10⁻⁴ improvement.

All operations use FFT‑based linear convolutions, run in O(n log n)
time (n≈2000).  The routine respects the 1000 s budget and prints
progress.  The returned list conforms to the strict checks of
`evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex {x ≥ 0, sum(x)=1}
def _project_onto_simplex(v):
    """Euclidean projection of a vector onto the simplex."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        # All entries are non‑positive – return uniform distribution
        return np.full_like(v, 1.0 / len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full linear convolution using FFT (output length 2·n‑1)
def _full_convolution(a, size=None):
    n = len(a)
    if size is None:
        size = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Exact objective used by evaluate_sequence (sum of a is assumed >0)
def _objective(a):
    n = len(a)
    conv = _full_convolution(a)
    max_b = conv.max()
    sum_a = a.sum()
    return 2.0 * n * max_b / (sum_a * sum_a)


# ----------------------------------------------------------------------
# Gradient of the smooth‑max surrogate φ_λ
def _softmax_gradient(a, conv, lam, size_corr=None):
    """
    Gradient of (1/lam)·log Σ_k exp(lam·conv_k) w.r.t. `a`.
    Returns a vector of length n.
    """
    n = len(a)
    # Soft‑max weights (numerically stable)
    m = conv.max()
    w = np.exp(lam * (conv - m))
    w /= w.sum()

    # Linear cross‑correlation w ★ a (via FFT)
    rev_a = a[::-1]
    if size_corr is None:
        size_corr = 1 << int(np.ceil(np.log2(len(w) + n - 1)))
    fw = np.fft.rfft(w, size_corr)
    frev = np.fft.rfft(rev_a, size_corr)
    corr = np.fft.irfft(fw * frev, size_corr)

    # Extract the slice corresponding to indices 0…n‑1
    grad = 2.0 * corr[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Linear correlation w ★ a (used for the true max sub‑gradient)
def _linear_correlation(w, a, size_corr=None):
    n = len(a)
    rev_a = a[::-1]
    if size_corr is None:
        size_corr = 1 << int(np.ceil(np.log2(len(w) + n - 1)))
    fw = np.fft.rfft(w, size_corr)
    frev = np.fft.rfft(rev_a, size_corr)
    corr = np.fft.irfft(fw * frev, size_corr)
    return corr[n - 1 : n - 1 + n]


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that
    minimises `evaluate_sequence`.  The function respects a time
    budget (default 1000 s) and prints progress.
    """
    np.random.seed(seed)
    # Safety margin for interpreter overhead
    deadline = time.time() + budget_s - 0.5

    # --------------------------------------------------------------
    # 1. Choose a problem size.
    n = 2000  # short enough for many FFT iterations, long enough for expressive power

    # --------------------------------------------------------------
    # 2. Gather initial candidates.
    best_seq = None
    best_val = float("inf")

    # 2a) Use the global sequence if it exists (resample if needed)
    try:
        global height_sequence_1
        cand = np.asarray(height_sequence_1, dtype=float)
        if cand.size != n:
            # Simple linear interpolation to required length
            xs_old = np.linspace(0, 1, cand.size)
            xs_new = np.linspace(0, 1, n)
            cand = np.interp(xs_new, xs_old, cand)
        cand = cand / cand.sum()
        val = _objective(cand)
        best_seq, best_val = cand.copy(), val
        print(f"[Init global] obj={val:.6f}")
    except Exception:
        pass

    # 2b) Geometric progressions a_i ∝ r^i
    for f in np.linspace(0.8, 3.0, 30):
        if time.time() > deadline:
            break
        eps = f / n
        r = 1.0 - eps
        if r <= 0.0:
            continue
        powers = np.power(r, np.arange(n, dtype=np.float64))
        seq = powers / powers.sum()
        val = _objective(seq)
        if val < best_val:
            best_val, best_seq = val, seq.copy()
            print(f"[Init geom] factor={f:5.3f} obj={val:.6f}")

    # 2c) Random Dirichlet vectors
    for _ in range(15):
        if time.time() > deadline:
            break
        seq = np.random.dirichlet(np.ones(n))
        val = _objective(seq)
        if val < best_val:
            best_val, best_seq = val, seq.copy()
            print(f"[Init rand] obj={val:.6f}")

    # Fallback if nothing succeeded (unlikely)
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)
        best_val = _objective(best_seq)

    # --------------------------------------------------------------
    # 3. Main optimisation: smooth‑max → true max, projected Adam.
    seq = best_seq.copy()

    # Adam hyper‑parameters
    lr = 0.3            # base learning rate
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    m = np.zeros_like(seq)
    v = np.zeros_like(seq)

    # Continuation schedule for λ
    lam = 5.0
    lam_factor = 1.02
    lambda_switch = 200.0  # after this we use the exact max sub‑gradient

    # FFT size for correlation (w ★ a).  len(w)=2n‑1, rev_a length=n.
    size_corr = 1 << int(np.ceil(np.log2(3 * n)))

    it = 0
    last_improve = time.time()

    while time.time() < deadline:
        # ---- a) self‑convolution
        conv = _full_convolution(seq)

        # ---- b) gradient (smooth‑max or max)
        if lam < lambda_switch:
            grad = _softmax_gradient(seq, conv, lam, size_corr=size_corr)
        else:
            # Sub‑gradient of the true max
            maxc = conv.max()
            mask = (conv >= maxc - 1e-12)
            w = mask.astype(float)
            w_sum = w.sum()
            if w_sum == 0:
                w = np.ones_like(conv) / conv.size
            else:
                w /= w_sum
            grad = 2.0 * _linear_correlation(w, seq, size_corr=size_corr)

        # ---- c) Adam update + projection onto simplex
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        seq_new = seq - step
        seq_new = _project_onto_simplex(seq_new)

        # ---- d) exact objective evaluation
        val = _objective(seq_new)

        # ---- e) bookkeeping
        if val < best_val:
            best_val, best_seq = val, seq_new.copy()
            last_improve = time.time()
            print(f"[Iter {it:05d}] new best obj={val:.6f}  λ={lam:.2f}")

        # ---- f) random shake if stuck
        if time.time() - last_improve > 2.0:
            noise = (np.random.rand(n) - 0.5) * 1e-3
            seq_new = _project_onto_simplex(seq_new + noise)
            last_improve = time.time()

        # ---- g) prepare for next iteration
        seq = seq_new
        lam *= lam_factor

    # --------------------------------------------------------------
    # 4. Final local refinement (simple stochastic hill‑climbing)
    # Use the remaining slack time, if any.
    remain = deadline - time.time()
    if remain > 5.0:
        sigma = 0.01  # initial perturbation scale
        while time.time() < deadline:
            noise = (np.random.randn(n)) * sigma
            cand = seq + noise
            cand = np.maximum(cand, 0.0)
            if cand.sum() == 0.0:
                continue
            cand = cand / cand.sum()
            val = _objective(cand)
            if val < best_val:
                best_val, best_seq = val, cand.copy()
                seq = cand  # keep the improved point as starting point
                print(f"[Local] improved obj={val:.6f}")
                last_improve = time.time()
            # slowly decay perturbation magnitude
            sigma *= 0.9995
            # occasionally inject a larger shake to escape possible plateaus
            if time.time() - last_improve > 3.0:
                sigma = max(sigma, 0.02)
                last_improve = time.time()

    print(f"Final best objective = {best_val:.6f} (length {n})")
    # Convert to plain Python list as required by the evaluator
    return best_seq.tolist()
