def _beta_distribution_seq(n, alpha):
    # compute a_i ∝ (i+0.5)^(alpha-1) * (n-i-0.5)^(alpha-1)
    i = np.arange(n, dtype=float)
    seq = (i + 0.5) ** (alpha - 1) * (n - i - 0.5) ** (alpha - 1)
    # ensure non-negative
    # set zero for any negative due to rounding?
    seq = np.maximum(seq, 0)
    seq /= seq.sum()
    return seq

def _evaluate_obj(seq):
    # sum assumed 1
    conv = np.convolve(seq, seq)
    max_b = conv.max()
    return 2 * len(seq) * max_b

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5

    # Choose a base size n; maybe 4000
    n = 4000

    # Phase 1: param search over alpha (U-shaped families)
    best_seq = None
    best_val = float('inf')
    alphas = np.linspace(0.2, 0.9, 15) # maybe 15.
    for alpha in alphas:
        seq = _beta_distribution_seq(n, alpha)
        val = _evaluate_obj(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
    # Also include arcsine (alpha=0.5) and uniform baseline
    # Uniform
    seq_uniform = np.full(n, 1.0/n)
    val = _evaluate_obj(seq_uniform)
    if val < best_val:
        best_val = val; best_seq = seq_uniform.copy()
    # Random Dirichlet start
    for _ in range(3):
        rand = rng.random(n)
        seq = rand / rand.sum()
        val = _evaluate_obj(seq)
        if val < best_val:
            best_val = val; best_seq = seq.copy()

    print(f"[Init] Best after param scan: {best_val:.6f}")

    # Phase 2: Adam optimizer on simplex
    seq = best_seq.copy()
    # Ensure sum=1
    seq = seq / seq.sum()
    m = np.zeros_like(seq)
    v = np.zeros_like(seq)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    step = 0.5   # learning rate; may tune.

    it = 0
    no_improve = 0
    while time.time() < deadline:
        # compute convolution and objective
        conv = np.convolve(seq, seq)
        max_b = conv.max()
        val = 2 * n * max_b
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            no_improve = 0
            print(f"[Iter {it:06d}] new best {best_val:.6f}")
        else:
            no_improve += 1

        # gradient (subgradient of max entry)
        k = np.argmax(conv)
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        grad = np.zeros_like(seq)
        # gradient = 2 * seq[k - idx]
        grad[idx] = 2.0 * seq[k - idx]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        it_adj = it + 1
        m_hat = m / (1 - beta1 ** it_adj)
        v_hat = v / (1 - beta2 ** it_adj)

        step_adj = step / (np.sqrt(v_hat) + eps)
        seq_new = seq - step_adj * m_hat

        # projection onto simplex
        seq_new = _project_onto_simplex(seq_new)

        # if stagnation, reinitialize from best_seq with jitter
        if no_improve > 20000:
            jitter = rng.normal(scale=0.005, size=n)  # small noise
            seq_new = best_seq + jitter
            seq_new = np.clip(seq_new, 0, None)
            if seq_new.sum() == 0:
                seq_new = np.full(n, 1.0/n)
            else:
                seq_new = seq_new / seq_new.sum()
            m[:] = 0.0
            v[:] = 0.0
            no_improve = 0
            print(f"[Iter {it:06d}] stochastic perturbation applied")

        seq = seq_new
        it += 1

    print(f"Finished after {it} iterations. Best {best_val:.6f}")
    # Convert best_seq to list (ensuring float within bounds)
    # Clip to range [0,1000] because evaluate_sequence will clip anyway.
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    # Convert to list of Python floats
    return best_seq.tolist()

"""
Propose a step‑function (non‑negative sequence) that minimises
`evaluate_sequence`.  

The objective is 2·n·max(conv(seq,seq)) / (sum(seq)**2).  Because the
objective is homogeneous we may keep the sequence normalised to a
probability distribution (∑a_i = 1).  The algorithm consists of three
phases:

1. **Parameterized seed** – generate a family of “U‑shaped’’ sequences
   a_i ∝ (i+½)^{α‑1}(n‑i‑½)^{α‑1} (Beta(α,α) on a grid).  Scan a few
   exponents and keep the best one (plus a few random Dirichlet draws
   and the uniform vector).

2. **Projected Adam optimisation** – minimise the ℓ∞‑norm of the
   self‑convolution.  The sub‑gradient of the maximal entry
   (k = argmax (a*a)) is
   ∂(a*a)_k/∂a_i = 2·a_{k‑i} (zero when the index is out of range).
   Adam’s adaptive steps are applied to the raw vector and the result
   is projected back onto the probability simplex after each step.

3. **Short local random‑search (annealing)** – after the main loop a
   cheap hill‑climbing stage moves a tiny amount of mass between two
   random positions; improvements are kept.  The whole procedure stops
   just before the supplied time budget expires.

The code never calls `evaluate_sequence` inside the optimisation
loops – it computes the objective directly (the same formula that
`evaluate_sequence` uses after normalisation) – which keeps the inner
iterations fast enough to explore many candidates within the 1000 s
budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x ≥ 0, sum(x)=1}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    # Find the last index where the “rho” condition holds
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        # All entries are non‑positive – return the centre of the simplex
        return np.full_like(v, 1.0 / len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _beta_u_shape(n: int, alpha: float) -> np.ndarray:
    """
    Return a U‑shaped probability vector of length n.
    a_i ∝ (i+½)^{α‑1} (n‑i‑½)^{α‑1}.
    For 0 < α < 1 the mass concentrates near the endpoints.
    """
    i = np.arange(n, dtype=np.float64)
    # avoid overflow for extreme alphas – np.clip is safe because exponents are modest
    seq = (i + 0.5) ** (alpha - 1.0) * (n - i - 0.5) ** (alpha - 1.0)
    # numerical safety
    seq = np.clip(seq, 0.0, None)
    total = seq.sum()
    if total == 0.0:
        seq = np.full(n, 1.0 / n)
    else:
        seq /= total
    return seq


# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """
    Compute the objective 2·n·max(conv(seq,seq))  (seq is assumed to sum to 1).
    """
    n = len(seq)
    conv = np.convolve(seq, seq)
    return 2.0 * n * conv.max()
