"""
Propose a non‑negative step‑function (list of heights) that minimizes

        2 * n * max(conv(a, a)) / (sum(a) ** 2)

using a projected heavy‑ball / stochastic sub‑gradient descent.
Key ideas
---------

* Work on the simplex Σ a_i = √(2 n) – then the evaluation reduces to
  max(conv(a, a)).
* At each iteration take the *active* convolution entries – those within a
  small tolerance of the current maximum (or the top‑k largest values) –
  and compute the sub‑gradient of their sum.
* Heavy‑ball momentum + a decaying learning‑rate gives fast descent.
* Occasionally inject a small random shake to escape shallow minima.
* Multiple independent restarts with varied lengths (≈2 000–5 000) are
  performed until the time budget (≈1 000 s) expires.
* If a previous good construction is available in the global variable
  `height_sequence_1` it is used as a warm start (truncated or padded with
  tiny noise).

The algorithm improves on the previous baseline by (i) using a
relative tolerance for the active set (giving more informative gradients),
(ii) a slightly larger learning‑rate schedule and (iii) a broader
range of problem sizes, which allows the bound to approach the theoretical
limit ≈1.5.
"""

import time
import math
import numpy as np

# --------------------------------------------------------------------------- #
# Helper: Euclidean projection onto the simplex {x_i ≥ 0, Σ x_i = s}.
# --------------------------------------------------------------------------- #
def project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector ``v`` onto the simplex of radius ``s``."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------------------- #
# Fast full self‑convolution via real FFT (length 2 n‑1).
# --------------------------------------------------------------------------- #
def conv_full_fft(a: np.ndarray) -> np.ndarray:
    """Return a * a using a real FFT (full convolution, length 2·len(a)-1)."""
    n = a.shape[0]
    n_fft = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    conv = np.fft.irfft(A * A, n=n_fft)[: 2 * n - 1]
    return conv


# --------------------------------------------------------------------------- #
# Gradient of the *maximal* (or top‑k) convolution entries.
# --------------------------------------------------------------------------- #
def gradient_of_active(a: np.ndarray, active_idxs: np.ndarray) -> np.ndarray:
    """
    Compute the sub‑gradient of the sum of convolution entries whose
    indices are given in ``active_idxs``.
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in active_idxs:
        low = max(0, k - (n - 1))
        high = min(n - 1, k)
        idx = np.arange(low, high + 1, dtype=int)
        partner = k - idx
        grad[idx] += 2.0 * a[partner]
    return grad


# --------------------------------------------------------------------------- #
# Run a single optimisation instance for a fixed length ``n``.
# --------------------------------------------------------------------------- #
def run_one_instance(
    n: int,
    init_vec: np.ndarray,
    global_deadline: float,
    max_time_per_run: float = 30.0,
) -> tuple[np.ndarray, float]:
    """
    Optimise a sequence of length ``n`` starting from ``init_vec``.
    Returns the best vector found inside this run and its max‑convolution
    value (the objective under the normalization Σa=√(2n)).
    The run stops after ``max_time_per_run`` seconds or when the global
    deadline is reached.
    """
    target_sum = math.sqrt(2.0 * n)
    a = project_onto_simplex(init_vec.astype(float), target_sum)

    momentum = np.zeros_like(a)
    beta_mom = 0.9

    # initial learning‑rate (will be decayed)
    base_lr = 0.45

    # small safety margin for each run
    run_deadline = min(global_deadline, time.time() + max_time_per_run)

    best_local_seq = a.copy()
    best_local_score = float("inf")

    iter_cnt = 0
    while time.time() < run_deadline:
        # ----- convolution & current score ---------------------------------
        conv = conv_full_fft(a)
        max_val = float(conv.max())
        cur_score = max_val  # because Σa = √(2n) ⇒ evaluate_sequence == max_val

        if cur_score < best_local_score:
            best_local_score = cur_score
            best_local_seq = a.copy()

        # ----- active set --------------------------------------------------
        # tolerance proportional to the current max (≈1 % of it)
        tol = max(1e-8, 0.01 * max_val)
        active = np.where(conv >= max_val - tol)[0]

        # limit the active set size to keep the gradient cheap
        max_active = 200
        if active.size > max_active:
            # keep the truly largest entries only
            active = np.argpartition(-conv, max_active)[:max_active]

        # ----- gradient ----------------------------------------------------
        grad = gradient_of_active(a, active)
        max_grad = grad.max()
        if max_grad <= 0.0:
            break

        # step size adapts to the scale of the objective
        eta = base_lr * max_val / max_grad

        # ----- heavy‑ball momentum update ----------------------------------
        momentum = beta_mom * momentum - eta * grad
        a_new = a + momentum

        # ----- feasibility: non‑negative + simplex -------------------------
        a_new = np.maximum(a_new, 0.0)
        a_new = project_onto_simplex(a_new, target_sum)

        # ----- occasional random shake (simulated‑annealing style) ---------
        if np.random.rand() < 0.02:
            # noise magnitude scales with the simplex radius
            noise = np.random.randn(n) * (0.01 * target_sum / math.sqrt(n))
            a_new = project_onto_simplex(np.maximum(a_new + noise, 0.0), target_sum)
            base_lr *= 0.9  # shrink learning‑rate after a shake

        a = a_new

        # ----- learning‑rate decay -----------------------------------------
        base_lr *= 0.9997

        iter_cnt += 1

    return best_local_seq, best_local_score
