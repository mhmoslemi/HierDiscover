def _full_convolution(a, size=None):
    n = len(a)
    if size is None:
        size = 1 << int(np.ceil(np.log2(2*n-1)))
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa*fa, size)[:2*n-1]
    return conv

Now gradient:

def _softmax_gradient(a, conv, lam, size_corr=None):
    n = len(a)
    c_max = conv.max()
    w = np.exp(lam * (conv - c_max))
    w_sum = w.sum()
    if w_sum == 0:
        # fallback uniform weights
        w = np.full_like(w, 1.0/len(w))
    else:
        w /= w_sum

    # correlation: w ★ a
    rev_a = a[::-1]
    if size_corr is None:
        size_corr = 1 << int(np.ceil(np.log2(len(w) + n - 1)))
    fw = np.fft.rfft(w, size_corr)
    frev = np.fft.rfft(rev_a, size_corr)
    corr = np.fft.irfft(fw * frev, size_corr)
    grad = 2.0 * corr[n-1 : n-1 + n]
    return grad

"""
Propose a step‑function (non‑negative sequence) with a small
evaluate_sequence() score.

Algorithm – Adam on a smooth‑max surrogate
------------------------------------------------
* Work on the probability simplex (∑a_i = 1, a_i ≥ 0).
* Initialise the vector with several cheap candidates
  (geometric sequences, Dirichlet draws with different concentration
  parameters) and keep the best.
* Minimise the smooth approximation of the true objective

        φ_λ(a) = (1/λ)·log∑_k exp(λ·(a*a)[k])

  whose gradient is   ∇φ_λ(a) = 2·(w ★ a)  with
  w_k ∝ exp(λ·(a*a)[k]).

* Optimise φ_λ using the Adam optimiser (adaptive first‑ and
  second‑moment estimates) together with a Euclidean projection onto
  the simplex.
* λ starts small (very smooth) and is multiplied by a factor each
  iteration – a continuation scheme that gradually focuses on the
  true max.
* If no improvement is seen for a couple of seconds a tiny random
  shake is applied to escape shallow minima.
* At the end a short greedy local‑search (tiny mass transfers) is
  performed to polish the best vector.
* The best sequence found within the 1000 s budget is returned as a
  plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# 1.  Projection onto the probability simplex {x ≥ 0, sum(x)=1}
def _project_onto_simplex(v):
    """Euclidean projection of a vector onto the simplex."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:                      # all numbers non‑positive
        return np.full_like(v, 1.0 / len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# 2.  Linear (non‑circular) convolution via FFT
def _full_convolution(a, size=None):
    """Return the linear self‑convolution a*a (length 2·len(a)-1)."""
    n = len(a)
    if size is None:
        size = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# 3.  Exact objective (the same function used in evaluate_sequence
#    when the vector is already normalised to sum = 1)
def _objective(a):
    """2·n·max(a*a)  (sum(a) is assumed to be 1)."""
    n = len(a)
    conv = _full_convolution(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# 4.  Gradient of the smooth‑max surrogate
def _softmax_gradient(a, conv, lam, size_corr=None):
    """
    Gradient of φ_λ(a) = (1/λ)·log∑_k exp(λ·conv_k).
    Returns ∇φ_λ(a) = 2·(w ★ a) where w is the soft‑max over conv.
    """
    n = len(a)
    c_max = conv.max()
    # soft‑max weights, numerically stable
    w = np.exp(lam * (conv - c_max))
    w_sum = w.sum()
    if w_sum == 0.0:          # can only happen for extreme λ → ∞
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # linear cross‑correlation w ★ a
    rev_a = a[::-1]
    if size_corr is None:
        size_corr = 1 << int(np.ceil(np.log2(len(w) + n - 1)))
    fw = np.fft.rfft(w, size_corr)
    frev = np.fft.rfft(rev_a, size_corr)
    corr = np.fft.irfft(fw * frev, size_corr)

    # slice the part that corresponds to indices 0 … n‑1
    grad = 2.0 * corr[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# 5.  A tiny greedy local‑search used after the main optimisation
def _local_refine(seq, best_val, time_deadline):
    """
    Move a tiny amount of mass between random coordinates.
    Accept only if the true objective improves.
    """
    n = len(seq)
    cur_seq = seq.copy()
    cur_val = best_val
    rng = np.random.default_rng()
    while time.time() < time_deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # transfer at most 1 % of the source mass
        delta = 0.01 * cur_seq[i]
        if delta <= 1e-12:
            continue
        new_seq = cur_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        # projection is not required – the operation preserves sum and
        # non‑negativity (by construction)
        val = _objective(new_seq)
        if val < cur_val:
            cur_seq = new_seq
            cur_val = val
    return cur_seq, cur_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step‑function with a minimal evaluate_sequence()
    score.  The routine respects the given time budget, prints progress,
    and finally returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    # leave a small safety margin for the interpreter / printing
    deadline = time.time() + budget_s - 0.5

    # --------------------------------------------------------------
    # 1.  Choose a length – 2500 offers a good trade‑off between expressivity
    #     and speed on the available time budget.
    n = 2500

    best_seq = None
    best_val = float("inf")

    # --------------------------------------------------------------
    # 2.  Cheap initial candidates (geometric families + Dirichlet draws)

    # geometric progressions a_i ∝ r^i
    for eps_factor in np.linspace(5, 250, 30):          # eps = factor / n
        if time.time() > deadline:
            break
        eps = eps_factor / n
        r = 1.0 - eps
        if r <= 0.0:
            continue
        powers = np.power(r, np.arange(n, dtype=np.float64))
        seq = powers / powers.sum()
        val = _objective(seq)
        if val < best_val:
            best_val, best_seq = val, seq.copy()
            print(f"[Init geom] eps={eps:.2e}  obj={val:.6f}")

    # Dirichlet vectors with several concentration parameters
    for alpha_val in (0.2, 0.5, 1.0, 2.0):
        for _ in range(12):
            if time.time() > deadline:
                break
            seq = np.random.dirichlet(np.full(n, alpha_val))
            val = _objective(seq)
            if val < best_val:
                best_val, best_seq = val, seq.copy()
                print(f"[Init Dir] α={alpha_val:.1f}  obj={val:.6f}")

    # If nothing was generated (should never happen) fall back to uniform
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)
        best_val = _objective(best_seq)

    # --------------------------------------------------------------
    # 3.  Adam optimiser on the smooth‑max surrogate
    seq = best_seq.copy()

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.05                     # learning rate before Adam scaling
    m = np.zeros_like(seq)
    v = np.zeros_like(seq)

    # λ‑continuation
    lam = 0.5                     # start very smooth
    lam_factor = 1.04
    lam_max = 200.0

    # FFT sizes (pre‑computed, powers of two)
    size_conv = 1 << int(np.ceil(np.log2(2 * n - 1)))   # for a*a
    size_corr = 1 << int(np.ceil(np.log2(3 * n)))       # for w ★ a

    it = 0
    last_improve = time.time()

    while time.time() < deadline:
        # ----- a) current self‑convolution
        conv = _full_convolution(seq, size=size_conv)

        # ----- b) gradient of the smooth‑max surrogate
        grad = _softmax_gradient(seq, conv, lam, size_corr=size_corr)

        # ----- c) Adam update (negative gradient → minimisation)
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        new_seq = seq - step
        new_seq = _project_onto_simplex(new_seq)

        # ----- d) exact objective (i.e. evaluate_sequence without the
        #       clamp‑and‑scale checks)
        val = _objective(new_seq)

        if val < best_val:
            best_val, best_seq = val, new_seq.copy()
            last_improve = time.time()
            print(f"[Iter {it:06d}] new best obj={val:.6f}")

        # ----- e) shake if we have been idle for a while
        if time.time() - last_improve > 2.0:
            noise = (np.random.rand(n) - 0.5) * 1e-3
            new_seq = _project_onto_simplex(new_seq + noise)
            last_improve = time.time()

        seq = new_seq

        # ----- f) update λ (smooth‑max becomes sharper)
        if lam < lam_max:
            lam = min(lam * lam_factor, lam_max)

        # occasional progress report
        if it % 5000 == 0:
            print(f"[Iter {it}] best={best_val:.6f}  λ={lam:.2f}")

    # --------------------------------------------------------------
    # 4.  Final local polishing (tiny random mass transfers)
    if time.time() < deadline - 0.5:
        best_seq, best_val = _local_refine(best_seq, best_val, deadline)

    print(f"Final best objective = {best_val:.6f} (length {n})")
    # evaluate_sequence expects a plain Python list
    return best_seq.tolist()
