def _active_gradient(seq, conv, top_k=3):
    # find indices of top_k largest conv values
    idx = np.argpartition(-conv, range(top_k))[:top_k]  # unsorted
    # compute aggregated gradient
    grad = np.zeros_like(seq)
    n = len(seq)
    for k in idx:
        lo = max(0, k-(n-1))
        hi = min(k, n-1)
        idx_range = np.arange(lo, hi+1)
        grad[idx_range] += seq[k - idx_range]
    return grad

"""
Propose a step‑function (non‑negative sequence) with a low value of
evaluate_sequence.  The algorithm consists of three phases:

1. **Broad seeding** – generate a diverse set of candidate distributions
   using several parametric families (geometric, power‑law, beta‑like,
   arcsine, random Dirichlet, mixtures of Gaussians) on a moderate length
   `n` (≈ 2500).  The best of these seeds is kept for refinement.

2. **Refinement by multi‑active projected gradient descent** – at each
   iteration the full self‑convolution is computed (FFT).  All shifts whose
   convolution value lies within a tiny tolerance of the current maximum
   are collected (active set).  The gradient is the sum of the reversed
   fragments that contribute to those active shifts.  A back‑tracking line
   search together with projection onto the probability simplex drives the
   sequence towards a flatter convolution.  When a line‑search fails a
   small random mass‑transfer move is attempted.

3. **Final polishing** – the best sequence found so far is further
   optimised until the time budget runs out.  The resulting list is
   returned.

The implementation avoids closures, uses only NumPy and standard library,
and respects the 1000 s time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
def _project_onto_simplex(v):
    """Project vector v onto the probability simplex {x≥0, sum(x)=1}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        # fallback to uniform
        return np.full_like(v, 1.0 / len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# FFT utilities for fast self‑convolution
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _self_convolve(seq: np.ndarray) -> np.ndarray:
    """Full linear convolution of seq with itself (length 2n‑1)."""
    n = seq.shape[0]
    m = 2 * n - 1
    nfft = _next_pow2(m)
    A = np.fft.rfft(seq, n=nfft)
    conv = np.fft.irfft(A * A, n=nfft)[:m]
    return conv


# ----------------------------------------------------------------------
# Local objective (identical to evaluate_sequence without clipping)
def _objective(seq: np.ndarray) -> float:
    """Return 2·n·max(conv)/(sum(seq)**2) for a non‑negative sequence."""
    sum_a = seq.sum()
    if sum_a < 0.01:
        return float("inf")
    conv = _self_convolve(seq)
    max_b = conv.max()
    return float(2.0 * len(seq) * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Parametric families ---------------------------------------------------
def _geom_seq(n: int, alpha: float) -> np.ndarray | None:
    """Geometric progression a_i ∝ r^i with r = 1 - alpha/n."""
    eps = alpha / n
    r = 1.0 - eps
    if r <= 0.0:
        return None
    powers = r ** np.arange(n, dtype=np.float64)
    seq = powers / powers.sum()
    return seq


def _powerlaw_seq(n: int, p: float) -> np.ndarray:
    """Power‑law a_i ∝ (i+1)^{-p}."""
    seq = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    return seq / seq.sum()


def _beta_seq(n: int, a: float, b: float) -> np.ndarray:
    """Beta‑like shape a_i ∝ (i+1)^{a-1}(n-i)^{b-1}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    seq = (i ** (a - 1.0)) * ((n + 1 - i) ** (b - 1.0))
    seq = np.where(np.isfinite(seq), seq, 0.0)
    seq[seq < 0] = 0.0
    total = seq.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return seq / total


def _arcsine_seq(n: int) -> np.ndarray:
    """Arcsine‑like shape a_i ∝ 1/√(i·(n‑1‑i)).  Peaks at the ends, dip in the centre."""
    i = np.arange(n, dtype=np.float64)
    # shift by 0.5 to avoid exact zero denominator
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5) + 1e-12)
    return vals / vals.sum()


def _random_simplex(n: int) -> np.ndarray:
    """Draw a random point on the simplex (Dirichlet(1,…,1))."""
    x = np.random.exponential(1.0, size=n)
    return x / x.sum()


def _gaussian_mixture_seq(n: int, components: int = 3) -> np.ndarray:
    """Mixture of a few Gaussian bumps (positive weights) on [0,n)."""
    xs = np.arange(n, dtype=np.float64)
    weights = np.random.dirichlet(np.ones(components))
    mixture = np.zeros(n, dtype=np.float64)
    for w in weights:
        center = np.random.uniform(0, n - 1)
        width = np.random.uniform(0.05 * n, 0.15 * n)
        mixture += w * np.exp(-0.5 * ((xs - center) / width) ** 2)
    # clip tiny negatives that may appear due to floating point
    mixture = np.maximum(mixture, 0.0)
    total = mixture.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return mixture / total

def _refine_sequence(seq: np.ndarray,
                     deadline: float,
                     max_no_improve: int = 5000) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent using a multi‑active set.
    Returns (best_sequence, best_score) found before timeout.
    """
    n = len(seq)
    best_seq = seq.copy()
    best_score = _objective(seq)
    cur_seq = seq.copy()
    cur_score = best_score

    # parameters for line‑search
    base_step = 2.0
    no_improve_counter = 0
    iteration = 0

    while time.time() < deadline and no_improve_counter < max_no_improve:
        conv = _self_convolve(cur_seq)
        max_val = conv.max()

        # active set: all k whose value is within 0.1 % of the maximum
        tol = 0.001  # 0.1 %
        active = np.where(conv >= max_val * (1.0 - tol))[0]
        # if the active set is huge, keep only the top few
        if active.size > 7:
            # pick the 7 largest entries
            top_idx = np.argpartition(-conv[active], 7)[:7]
            active = active[top_idx]

        # aggregate sub‑gradient from all active shifts
        grad = np.zeros_like(cur_seq)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            # contribution a_{k-i} for i∈[lo,hi]
            grad[idx] += cur_seq[k - idx]

        # ---- back‑tracking line search ----
        step = base_step
        improved = False
        while step > 1e-7:
            cand = _project_onto_simplex(cur_seq - step * grad)
            cand_score = _objective(cand)
            if cand_score < cur_score - 1e-12:
                improved = True
                break
            step *= 0.5

        if improved:
            cur_seq = cand
            cur_score = cand_score
            no_improve_counter = 0
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                print(f"[Refine] iter {iteration:06d} new best = {best_score:.6f}")
            # slightly enlarge step for next iteration

        else:

            i, j = np.random.choice(n, size=2, replace=False)
            delta = 0.01 * min(cur_seq[i], 0.02)  # move up to ~1 % of total mass
            if delta > 0:
                cand = cur_seq.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_onto_simplex(cand)
                cand_score = _objective(cand)
                if cand_score < cur_score - 1e-12:
                    cur_seq = cand
                    cur_score = cand_score
                    no_improve_counter = 0
                    if cand_score < best_score:
                        best_score = cand_score
                        best_seq = cand.copy()
                        print(f"[Refine] iter {iteration:06d} rand‑move → {best_score:.6f}")
                else:
                    no_improve_counter += 1
            else:
                no_improve_counter += 1

        iteration += 1

    return best_seq, best_score

                sc = _objective(init)
                if sc < best_score:
                    best_score = sc
                    best_seq = init.copy()
                    print(f"[Init] global seed score = {sc:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1️⃣  Build a diverse pool of seed sequences.
    seeds = []

    # uniform baseline
    seeds.append(np.full(n, 1.0 / n))

    # arcsine‑type shape (edges heavy)
    seeds.append(_arcsine_seq(n))

    # geometric family
    for alpha in np.linspace(0.5, 5.0, 12):
        seq = _geom_seq(n, alpha)
        if seq is not None:
            seeds.append(seq)

    # power‑law family
    for p in np.linspace(0.2, 3.5, 14):
        seeds.append(_powerlaw_seq(n, p))

    # beta‑like family – explore a wide range, incl. a,b<1
    for a in np.linspace(0.1, 2.2, 7):
        for b in np.linspace(0.1, 2.2, 7):
            seeds.append(_beta_seq(n, a, b))

    # a few random Dirichlet points
    for _ in range(25):
        seeds.append(_random_simplex(n))

    # Gaussian mixture candidates
    for _ in range(20):
        components = np.random.choice([2, 3, 4])
        seeds.append(_gaussian_mixture_seq(n, components))

    # ------------------------------------------------------------------
    # 2️⃣  Evaluate each seed and retain the best one before local refinement.
    for s in seeds:
        if time.time() >= deadline:
            break
        s = np.clip(s, 0.0, 1000.0)
        total = s.sum()
        if total == 0.0:
            continue
        s = s / total
        sc = _objective(s)
        if sc < best_score:
            best_score = sc
            best_seq = s.copy()
            print(f"[Seed] initial best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Local refinement of promising seeds.
    # Allocate a fraction of the remaining budget for quick refinement of a few
    # top seeds, then devote the rest to polishing the overall best.
    time_remaining = deadline - time.time()
    # Determine how many seeds we can afford to refine (≈2 s each).
    quick_refine_budget = min(30.0, 0.1 * time_remaining)  # at most 30 s or 10 % of remaining
    per_seed_budget = 2.0  # seconds per quick refine
    quick_deadline = time.time() + quick_refine_budget

    # Sort seeds by their initial score (ascending) and take a few best.
    seed_scores = [(s, _objective(s)) for s in seeds]
    seed_scores.sort(key=lambda x: x[1])
    top_seeds = [s for s, _ in seed_scores[:min(8, len(seed_scores))]]

    for s in top_seeds:
        if time.time() >= quick_deadline:
            break
        # allocate a small time slice for each quick refinement
        seed_deadline = time.time() + per_seed_budget
        refined, refined_score = _refine_sequence(s, seed_deadline, max_no_improve=2000)
        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined.copy()
            print(f"[Quick] improved best to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Final polishing: run a longer optimisation starting from the best
    # discovered sequence.
    if best_seq is None:
        # fallback uniform
        best_seq = np.full(n, 1.0 / n)
        best_score = _objective(best_seq)

    final_deadline = deadline
    best_seq, best_score = _refine_sequence(best_seq, final_deadline, max_no_improve=20000)

    # ------------------------------------------------------------------
    print(f"Final best objective = {best_score:.6f} (length {n})")
    # Return as a plain Python list of floats.
    return best_seq.tolist()
