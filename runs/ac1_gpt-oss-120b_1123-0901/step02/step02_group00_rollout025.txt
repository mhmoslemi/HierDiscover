    A = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(A*A, n=N)[:2*n-1]
    M = float(conv.max())
    return M, conv

def _compute_objective(a):
    n = a.shape[0]
    M, _ = _conv_max_fft(a)
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return 2.0 * n * M / (s * s)

def _subgradient(a, active):
    n = a.shape[0]
    idx = np.arange(n)
    rev = np.zeros_like(a)
    for k in active:
        j = k - idx
        mask = (j >= 0) & (j < n)
        rev[mask] += a[j[mask]]
    if active.size > 0:
        rev /= active.size
    return rev

def _adam_optimize(init_a, max_iters, deadline, verbose_label):
    a = init_a.copy()
    n = a.shape[0]
    target_sum = math.sqrt(2.0 * n)
    # normalize
    if a.sum() == 0:
        a = np.full_like(a, target_sum / n)
    else:
        a = a * (target_sum / a.sum())
    
    # hyperparameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.05
    
    best_a = a.copy()
    best_val = _compute_objective(a)

    for it in range(1, max_iters+1):
        if time.time() >= deadline:
            break
        
        # compute convolution and active set
        M, conv_full = _conv_max_fft(a)
        # active indices within tolerance
        tol = 1e-12
        active = np.where(conv_full >= M - tol)[0]
        rev = _subgradient(a, active)
        coeff = 4.0 * n / (target_sum * target_sum)
        const = 4.0 * n * M / (target_sum ** 3)
        grad = coeff * rev - const
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        lr = lr0 / (1 + 1e-4 * it)  # decaying
        
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # project
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0:
            a = a * (target_sum / s)
        else:
            a = np.full_like(a, target_sum / n)
        
        # occasional jitter
        if it % 5000 == 0:
            jitter = np.random.randn(n) * 0.001 * target_sum / n
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())
        
        # evaluate
        cur_val = 2.0 * n * M / (target_sum ** 2)
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{verbose_label} it {it:6d}] new best {best_val:.6f}")
    return best_a, best_val

def _random_initial(n, target_sum):
    # Dirichlet with random concentration
    conc = np.random.rand()*3 + 0.1  # between 0.1 and 3.1
    # as simple gamma draws
    draws = np.random.gamma(conc, 1, size=n)
    a = draws
    if a.sum() == 0:
        a = np.full_like(a, target_sum / n)
    else:
        a = a * (target_sum / a.sum())
    return a

   time slice so that many restarts fit into the 1000 s budget.

3. **Refinement** – the overall best vector is further improved by a
   longer Adam run (using the remaining time) and a short stochastic
   hill‑climbing phase.

All vectors are kept non‑negative and re‑normalised after each update
to the target sum √(2 n) (the objective is scale‑invariant, this merely
avoids the ``evaluate_sequence`` clamping).  The best sequence found
before the deadline is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution (full) using FFT – returns max value and the full array
def _conv_max_fft(a: np.ndarray):
    """Return (M, full_convolution) for a real vector a."""
    n = a.shape[0]
    # next power of two >= 2*n-1
    N = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(A * A, n=N)[: 2 * n - 1]
    M = float(conv.max())
    return M, conv


# ----------------------------------------------------------------------
# Sub‑gradient of the max‑convolution for a (possibly non‑unique) active set
def _subgradient(a: np.ndarray, active_idxs: np.ndarray):
    """
    Compute the average reversed vector a_{k‑i} over all active shift indices.
    Returns a vector of length len(a).
    """
    n = a.shape[0]
    idx = np.arange(n)
    rev = np.zeros_like(a)
    for k in active_idxs:
        j = k - idx
        mask = (j >= 0) & (j < n)
        rev[mask] += a[j[mask]]
    if active_idxs.size > 0:
        rev /= active_idxs.size
    return rev


# ----------------------------------------------------------------------
# Adam optimiser that works directly on the sub‑gradient of the objective
def _adam_optimize(
    init_a: np.ndarray,
    max_iters: int,
    deadline: float,
    label: str,
):
    """
    Run Adam optimisation on a vector while respecting the non‑negative
    and L1‑normalisation constraints.  Returns the best vector seen and
    its objective value.
    """
    n = init_a.shape[0]
    target_sum = math.sqrt(2.0 * n)

    # normalise initial vector
    a = init_a.copy()
    s0 = a.sum()
    if s0 > 0.0:
        a = a * (target_sum / s0)
    else:
        a = np.full_like(a, target_sum / n)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.05
    best_a = a.copy()
    # objective after normalisation = max(conv) because 2n / target_sum**2 = 1
    M, _ = _conv_max_fft(a)
    best_val = 2.0 * n * M / (target_sum ** 2)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # 1) convolution, max, active set
        M, conv_full = _conv_max_fft(a)
        tol = 1e-12
        active = np.where(conv_full >= M - tol)[0]
        rev = _subgradient(a, active)

        # 2) sub‑gradient of the *scaled* objective
        coeff = 4.0 * n / (target_sum * target_sum)
        const = 4.0 * n * M / (target_sum ** 3)
        grad = coeff * rev - const

        # 3) Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # simple decay for learning rate
        lr = lr0 / (1.0 + 1e-4 * it)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # 4) projection onto feasible set
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a = a * (target_sum / s)
        else:
            a = np.full_like(a, target_sum / n)

        # 5) occasional jitter to escape plateaus
        if it % 5000 == 0:
            jitter = np.random.randn(n) * 0.001 * target_sum / n
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        # 6) evaluate current objective (fast, using the just‑computed M)
        cur_val = 2.0 * n * M / (target_sum ** 2)
        if cur_val + 1e-12 < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{label} it {it:6d}] new best = {best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple families of candidate shapes (all non‑negative)
def _shape_exponential(n, alpha):
    return np.exp(-alpha * np.arange(n, dtype=float))


def _shape_power(n, beta):
    # i starts at 1 to avoid division by zero
    return (np.arange(1, n + 1, dtype=float) ** (-beta))


def _shape_linear(n, _):
    return (n - np.arange(n, dtype=float)) / n


def _shape_quadratic(n, _):
    return 1.0 - (np.arange(n, dtype=float) / (n - 1)) ** 2


def _shape_sine(n, _):
    # half‑sine, strictly positive on (0,π)
    return np.sin(np.pi * (np.arange(1, n + 1, dtype=float) / (2 * n + 2)))


# ----------------------------------------------------------------------
# Convenience: fast objective (identical to evaluate_sequence up to clipping)
def _fast_objective(a: np.ndarray) -> float:
    n = a.shape[0]
    M, _ = _conv_max_fft(a)
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return 2.0 * n * M / (s * s)


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the evaluation
    function.  The search consists of a quick parametric scan, several
    short Adam runs with random restarts, a final long Adam refinement,
    and a tiny stochastic hill‑climbing phase.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Load previous best construction if available
    try:
        base_seq = np.array(height_sequence_1, dtype=float)  # type: ignore
    except Exception:
        base_seq = None

    global_best_a = None
    global_best_val = float("inf")

    if base_seq is not None and base_seq.size > 0:
        init_score = evaluate_sequence(base_seq.tolist())
        global_best_a = base_seq.copy()
        global_best_val = init_score
        print(f"[init] length={base_seq.size}, score={init_score:.6f}")

    # ------------------------------------------------------------------
    # Lengths to explore (original and a few scaled versions, capped)
    base_len = base_seq.shape[0] if base_seq is not None else 2100
    length_factors = [1.0, 1.25, 1.5, 2.0, 3.0]
    lengths = sorted({int(base_len * f) for f in length_factors})
    lengths = [l for l in lengths if 300 <= l <= 8000]

    # ------------------------------------------------------------------
    # Stage 1 – fast parametric scan
    for n in lengths:
        if time.time() >= deadline:
            break
        target_sum = math.sqrt(2.0 * n)
        print(f"[scan] exploring length {n}")

        # exponential family
        for alpha in np.logspace(-4, 0.5, 24):
            a = _shape_exponential(n, alpha)
            a = np.maximum(a, 0.0)
            a = a * (target_sum / a.sum())
            val = _fast_objective(a)
            if val < global_best_val:
                global_best_val, global_best_a = val, a.copy()
                print(f"[scan] exp α={alpha:.5f} → {val:.6f}")

        # power‑law family
        for beta in np.linspace(0.1, 3.0, 24):
            a = _shape_power(n, beta)
            a = np.maximum(a, 0.0)
            a = a * (target_sum / a.sum())
            val = _fast_objective(a)
            if val < global_best_val:
                global_best_val, global_best_a = val, a.copy()
                print(f"[scan] pow β={beta:.5f} → {val:.6f}")

        # linear, quadratic, sine (no extra parameters)
        for kind, shape_fn in [
            ("lin", _shape_linear),
            ("quad", _shape_quadratic),
            ("sine", _shape_sine),
        ]:
            a = shape_fn(n, None)
            a = a * (target_sum / a.sum())
            val = _fast_objective(a)
            if val < global_best_val:
                global_best_val, global_best_a = val, a.copy()
                print(f"[scan] {kind} → {val:.6f}")

    # ------------------------------------------------------------------
    # Helper: generate a random Dirichlet‑type start vector
    def _random_start(n, target_sum):
        conc = np.random.rand() * 3.0 + 0.1  # concentration in [0.1,3.1]
        draws = np.random.gamma(conc, 1.0, size=n)
        a = draws
        if a.sum() == 0.0:
            a = np.full(n, target_sum / n)
        else:
            a = a * (target_sum / a.sum())
        return a

    # ------------------------------------------------------------------
    # Stage 2 – short Adam runs with random restarts
    for n in lengths:
        if time.time() >= deadline:
            break
        target_sum = math.sqrt(2.0 * n)
        remaining = deadline - time.time()
        # allocate at most 30 s per length (or less if time is scarce)
        per_len_budget = min(30.0, remaining / (len(lengths) * 2 + 1))
        run_deadline = time.time() + per_len_budget

        # a) start from the best scanned shape for this length (if any)
        if global_best_a is not None and global_best_a.shape[0] == n:
            init = global_best_a.copy()
        else:
            init = _random_start(n, target_sum)

        a_opt, val_opt = _adam_optimize(
            init_a=init,
            max_iters=int(5e6),  # very large – will stop at run_deadline
            deadline=run_deadline,
            label=f"len{n}_scan",
        )
        if val_opt < global_best_val:
            global_best_val, global_best_a = val_opt, a_opt.copy()
            print(f"[adam] length {n} improved to {val_opt:.6f}")

        # b) a few pure random restarts
        for restart_idx in range(3):
            if time.time() >= deadline:
                break
            # give each restart a small slice of the remaining time
            remaining = deadline - time.time()
            if remaining <= 0.0:
                break
            slice_budget = min(10.0, remaining / 4.0)
            slice_deadline = time.time() + slice_budget
            init_rand = _random_start(n, target_sum)
            a_opt, val_opt = _adam_optimize(
                init_a=init_rand,
                max_iters=int(5e6),
                deadline=slice_deadline,
                label=f"len{n}_rnd{restart_idx}",
            )
            if val_opt < global_best_val:
                global_best_val, global_best_a = val_opt, a_opt.copy()
                print(f"[adam] length {n} rnd{restart_idx} → {val_opt:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – final long Adam refinement on the current global best
    if global_best_a is not None and time.time() < deadline:
        n = global_best_a.shape[0]
        target_sum = math.sqrt(2.0 * n)
        final_deadline = deadline  # use whatever time is left
        print(f"[refine] final Adam on length {n}, time left {final_deadline - time.time():.1f}s")
        a_ref, val_ref = _adam_optimize(
            init_a=global_best_a,
            max_iters=int(5e7),
            deadline=final_deadline,
            label="final",
        )
        if val_ref < global_best_val:
            global_best_val, global_best_a = val_ref, a_ref.copy()
            print(f"[refine] improved to {val_ref:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – tiny stochastic hill‑climbing (optional, lightweight)
    if global_best_a is not None and time.time() < deadline:
        n = global_best_a.shape[0]
        target_sum = math.sqrt(2.0 * n)
        a = global_best_a.copy()
        cur_val = _fast_objective(a)
        it = 0
        print("[hill] starting stochastic refinement")
        while time.time() < deadline and it < 200_000:
            it += 1
            # small Gaussian perturbation of all entries
            pert = np.random.randn(n) * 0.001 * target_sum / n
            cand = a + pert
            cand = np.maximum(cand, 0.0)
            s = cand.sum()
            if s > 0.0:
                cand = cand * (target_sum / s)
            else:
                continue
            new_val = _fast_objective(cand)
            if new_val + 1e-12 < cur_val:
                a, cur_val = cand, new_val
                if new_val < global_best_val:
                    global_best_val, global_best_a = new_val, cand.copy()
                    print(f"[hill] iter {it:6d} improved to {new_val:.6f}")

    # ------------------------------------------------------------------
    # Final safety fallback
    if global_best_a is None:
        n = 2100
        target_sum = math.sqrt(2.0 * n)
        global_best_a = np.full(n, target_sum / n)
        global_best_val = evaluate_sequence(global_best_a.tolist())
        print("[fallback] using uniform sequence")
    else:
        # compute the official score (includes the clamping checks of evaluate_sequence)
        global_best_val = evaluate_sequence(global_best_a.tolist())
        print(f"[final] official score = {global_best_val:.6f}, length = {global_best_a.shape[0]}")

    return global_best_a.tolist()
