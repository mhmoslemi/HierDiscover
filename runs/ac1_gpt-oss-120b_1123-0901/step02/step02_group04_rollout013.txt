def _conv_weight_rev(weight, a_rev):
    # compute convolution of weight and a_rev via FFT
    size = 1 << ((len(weight) + len(a_rev) - 1).bit_length())

    fa = np.fft.rfft(a_rev, n=size)

"""
Optimisation of a non‑negative step‑function sequence.

The score to minimise is
    2 * n * max(conv(a, a)) / (sum(a)**2)

which for a normalised sequence (sum(a)=sqrt(2·n)) equals the maximal
autocorrelation entry.  The programme below

1.  tests a few cheap parametric families (exponential, reversed,
    triangular) and keeps the best.
2.  Refines any candidate with a projected gradient descent on a smooth
    Lp‑approximation of the infinity‑norm (p = 2, 4, 8, …).  The gradient
    of  Σ(conv_k^p) is computed analytically and the updates are
    performed with momentum and a diminishing step size.
3.  Performs several random restarts (Dirichlet‑initialised vectors)
    and runs the same refinement while time permits.
4.  Optionally runs a single linear‑programme step (SciPy) if the
    library is available.
5.  Returns the best sequence found within the allotted budget.

All helper functions are defined at module level (no closures) and the
routine respects the given time budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the simplex {x ≥ 0, Σx = s}
# ----------------------------------------------------------------------
def project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project a vector onto the simplex with sum `s` (non‑negative)."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper: fast full convolution using real FFT (zero‑padded to next power of two)
# ----------------------------------------------------------------------
def conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real 1‑D arrays using FFT."""
    n = len(x) + len(y) - 1
    size = 1 << ((n - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)
    return conv[:n]


def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a * a (real, using FFT)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 2).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Helper: evaluate a sequence using the provided external routine.
# ----------------------------------------------------------------------
def evaluate_external(seq: list) -> float:
    """Call the already imported evaluate_sequence, guarding against errors."""
    try:
        return evaluate_sequence(seq)          # type: ignore[name-defined]
    except Exception:
        return float("inf")


def compute_score(a: np.ndarray) -> float:
    """
    Compute the same score as evaluate_sequence for a normalised vector.
    This avoids the overhead of the external call.
    """
    n = a.shape[0]
    conv = conv_self_fft(a)
    max_b = np.max(conv)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Shape factories (non‑normalised)
# ----------------------------------------------------------------------
def shape_exp(n: int, alpha: float) -> np.ndarray:
    return np.exp(-alpha * np.arange(n, dtype=float))


def shape_exp_rev(n: int, alpha: float) -> np.ndarray:
    return np.exp(-alpha * np.arange(n - 1, -1, -1, dtype=float))


def shape_triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    return np.minimum(i + 1.0, n - i)


# ----------------------------------------------------------------------
# Gradient descent on a smooth Lp approximation of the max‑norm.
# ----------------------------------------------------------------------
def lp_gradient_descent(
    a0: np.ndarray,
    target_sum: float,
    p_schedule: list[int],
    iters_per_p: int,
    deadline: float,
    eval_every: int = 10,
    early_stop: float = 1.48,
) -> tuple[np.ndarray, float]:
    """
    Perform projected gradient descent on Σ(conv_k**p) (p in p_schedule).
    Returns the best vector seen and its (true) score.
    """
    a = a0.copy()
    n = a.shape[0]
    best_seq = a.copy()
    best_score = compute_score(a)

    for p in p_schedule:
        # momentum vector
        m = np.zeros_like(a)
        beta = 0.9
        base_step = 0.4

        for it in range(iters_per_p):
            if time.time() >= deadline:
                return best_seq, best_score

            conv = conv_self_fft(a)                     # length 2n‑1
            # weight = conv^{p‑1}
            w = np.power(conv, p - 1, out=np.empty_like(conv))

            # gradient = 2·p·(w * rev(a))[n‑1 : n‑1+n]
            rev_a = a[::-1]
            grad_full = conv_fft(w, rev_a)             # length 3n‑2
            grad = 2.0 * p * grad_full[(n - 1) : (n - 1 + n)]

            direction = -grad
            m = beta * m + (1.0 - beta) * direction

            # diminishing step size, normalised by max component
            step = base_step / math.sqrt(it + 1)
            max_abs = np.max(np.abs(m))
            if max_abs < 1e-12:
                max_abs = 1.0
            eta = step / max_abs

            a = a + eta * m
            a = project_onto_simplex(a, target_sum)

            if (it % eval_every) == 0:
                sc = compute_score(a)
                if sc < best_score:
                    best_score = sc
                    best_seq = a.copy()
                    print(f"[{time.time():.2f}]   Local best = {best_score:.6f} (n={n}, p={p})")
                if best_score <= early_stop:
                    return best_seq, best_score

    return best_seq, best_score


# ----------------------------------------------------------------------
# Optional LP refinement (requires SciPy)
# ----------------------------------------------------------------------
def lp_refinement(seq: np.ndarray, target_sum: float, deadline: float) -> np.ndarray:
    """
    One iteration of the LP‑step from the SOTA description.
    Returns a possibly improved sequence (or the original if LP fails).
    """
    try:
        from scipy.optimize import linprog
    except Exception:
        return seq

    n = seq.shape[0]
    conv = conv_self_fft(seq)
    M = float(np.max(conv))

    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=float)

    # Fill the convolution matrix A so that (seq * g)_k = Σ_i A[k,i] * g_i
    for k in range(rows):
        low = max(0, k - (n - 1))
        high = min(n - 1, k)
        cols = np.arange(low, high + 1, dtype=int)
        A[k, cols] = seq[k - cols]

    res = linprog(
        c=-np.ones(n),                # maximise Σ g_i  → minimise –Σ g_i
        A_ub=A,
        b_ub=np.full(rows, M),
        bounds=(0, None),
        method="highs",
        options={"presolve": True},
    )
    if not res.success:
        return seq

    g = res.x
    s = np.sum(g)
    if s <= 0:
        return seq
    g = g * (target_sum / s)

    # convex combination
    t = 0.2
    new_seq = (1.0 - t) * seq + t * g
    new_seq = project_onto_simplex(new_seq, target_sum)
    return new_seq

# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs) -> list:
    """
    Search for a non‑negative step‑function that minimises the evaluation
    function within the given time budget.  The best sequence found is
    returned as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Phase 1 – cheap parametric families
    # ------------------------------------------------------------------
    lengths = [800, 1000, 1200, 1500, 1800, 2000, 2500]
    alphas = [1e-4, 5e-4, 1e-3, 5e-3, 1e-2, 2e-2]
    for n in lengths:
        target = math.sqrt(2.0 * n)

        # exponential decay
        for a in alphas:
            a_vec = shape_exp(n, a)
            a_vec = a_vec / np.sum(a_vec) * target
            sc = compute_score(a_vec)
            if sc < best_score:
                best_score, best_seq = sc, a_vec.copy()
                print(f"[{time.time():.2f}] New best = {best_score:.6f} (len={n})")
        # reversed exponential
        for a in alphas:
            a_vec = shape_exp_rev(n, a)
            a_vec = a_vec / np.sum(a_vec) * target
            sc = compute_score(a_vec)
            if sc < best_score:
                best_score, best_seq = sc, a_vec.copy()
                print(f"[{time.time():.2f}] New best = {best_score:.6f} (len={n})")
        # triangular symmetric shape
        a_vec = shape_triangular(n)
        a_vec = a_vec / np.sum(a_vec) * target
        sc = compute_score(a_vec)
        if sc < best_score:
            best_score, best_seq = sc, a_vec.copy()
            print(f"[{time.time():.2f}] New best = {best_score:.6f} (len={n})")

        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # Helper: try to improve a given candidate via LP‑gradient descent
    # ------------------------------------------------------------------
    def improve_candidate(candidate: np.ndarray, n: int) -> None:
        nonlocal best_seq, best_score
        target = math.sqrt(2.0 * n)
        # parameter schedule (smooth → sharp)
        p_sched = [2, 4, 8, 16, 32, 64]
        iters = 2000        # per p value (will be cut by deadline)
        cand, cand_sc = lp_gradient_descent(
            candidate, target, p_sched, iters, deadline
        )
        if cand_sc < best_score:
            best_score, best_seq = cand_sc, cand.copy()
            print(f"[{time.time():.2f}] New best (post‑GD) = {best_score:.6f} (len={n})")
        # optional LP refinement
        if time.time() < deadline:
            cand_lp = lp_refinement(cand, target, deadline)
            sc_lp = compute_score(cand_lp)
            if sc_lp < best_score:
                best_score, best_seq = sc_lp, cand_lp.copy()
                print(f"[{time.time():.2f}] New best (LP) = {best_score:.6f} (len={n})")

    # ------------------------------------------------------------------
    # Phase 2 – refine the best candidate from Phase 1
    # ------------------------------------------------------------------
    if best_seq is not None:
        improve_candidate(best_seq, best_seq.shape[0])

    # ------------------------------------------------------------------
    # Phase 3 – random restarts with gradient descent
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # pick a random length (kept moderate)
        n = np.random.randint(900, 2600)
        target = math.sqrt(2.0 * n)

        # random Dirichlet‑like start
        a0 = np.random.rand(n)
        a0 = a0 / np.sum(a0) * target

        improve_candidate(a0, n)

        # stop early if we have already crossed a reasonable target
        if best_score <= 1.48:
            break

    # ------------------------------------------------------------------
    # Fallback – uniform sequence if nothing succeeded
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1500
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=float)

    # Convert to plain list for the harness
    return best_seq.tolist()
