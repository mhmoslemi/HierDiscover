def _run_softmax(init_vec):
    a = init_vec.copy()
    n = len(a)
    target = math.sqrt(2.0*n)
    m = np.zeros_like(a)
    beta = 0.9
    max_iters_per_restart = 3000
    # schedule for c
    c0 = 0.3
    c_max = 8.0
    for it in range(max_iters_per_restart):
        # compute step size
        step = 0.5 / math.sqrt(it + 1)
        # schedule c: linear increase
        c = c0 + (c_max - c0) * it / max_iters_per_restart
        # compute conv
        conv = _conv_self(a)
        # compute weights
        w = _softmax_weights(conv, c)
        # compute grad via conv of w and a_rev
        a_rev = a[::-1]
        grad_full = _fft_convolve_full(w, a_rev)
        start = n - 1
        grad = 2.0 * grad_full[start:start+n]
        # momentum update
        m = beta * m + (1.0 - beta) * (-grad)
        eta = step / (np.max(np.abs(m)) + 1e-12)

        # projection onto simplex

        a = a_new
        if it % 10 == 0:

import time, math, numpy as np
try:
    from scipy.optimize import linprog
except Exception:
    linprog = None

def _project_onto_simplex(v, s):
    """Euclidean projection onto the simplex {x >= 0, sum = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _fft_convolve_full(x, y):
    """Full convolution of two real 1‑D arrays via FFT."""
    n = x.shape[0]
    m = y.shape[0]
    size = 1 << ((n + m - 1).bit_length())
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(X * Y, n=size)
    return conv[: n + m - 1]

def _conv_self(a):
    """Full autocorrelation a * a."""
    return _fft_convolve_full(a, a)

# ----------------------------------------------------------------------
# Shape factories (non‑negative, not normalised)
# ----------------------------------------------------------------------
def _shape_exp(n, alpha):
    return np.exp(-alpha * np.arange(n, dtype=float))

def _shape_exp_rev(n, alpha):
    return np.exp(-alpha * np.arange(n - 1, -1, -1, dtype=float))

def _shape_linear_desc(n, beta):
    i = np.arange(n, dtype=float)
    return np.maximum(0.0, 1.0 - beta * i)

def _shape_quad_desc(n, beta):
    i = np.arange(n, dtype=float)
    return np.maximum(0.0, 1.0 - beta * i * i)

def _shape_triangular(n):
    i = np.arange(n, dtype=float)
    return np.minimum(i + 1.0, n - i)

def _shape_sin(n):
    i = np.arange(n, dtype=float)
    return np.sin(np.pi * (i + 1) / (n + 1))

def _shape_gaussian(n, sigma):
    i = np.arange(n, dtype=float) - (n - 1) / 2.0
    return np.exp(-0.5 * (i / sigma) ** 2)

        return evaluate_sequence(seq)
    except Exception:
        return float('inf')

class Optimizer:
    """Encapsulates the search state and provides sub‑routines."""

    def __init__(self, seed: int, deadline: float):
        self.rng = np.random.RandomState(seed)
        self.deadline = deadline
        self.best_seq = None   # list of floats
        self.best_score = float('inf')

    # ------------------------------------------------------------------
    def maybe_update(self, seq: np.ndarray):
        """Project seq to list, evaluate and store as new best if improved."""
        # Ensure seq is a python list of floats for evaluate_sequence
        cand = [float(x) for x in seq]
        score = _evaluate(cand)
        if score < self.best_score:
            self.best_score = score
            self.best_seq = cand.copy()
            print(f"[{time.time():.2f}] New best = {self.best_score:.6f} (len={len(cand)})")
        return score

    # ------------------------------------------------------------------
    def run_subgrad(self, init_vec: np.ndarray):
        """Sub‑gradient descent on max‑autocorrelation (original approach)."""
        a = init_vec.astype(float)
        n = a.shape[0]
        target = math.sqrt(2.0 * n)

        m = np.zeros_like(a)          # momentum
        beta = 0.9
        base_step = 0.2
        it = 0
        while time.time() < self.deadline:
            conv = _conv_self(a)
            kmax = int(np.argmax(conv))

            # gradient of the peak entry
            low = max(0, kmax - (n - 1))
            high = min(n - 1, kmax)
            idx = np.arange(low, high + 1, dtype=int)
            partner = kmax - idx
            grad = np.zeros_like(a)
            grad[idx] = 2.0 * a[partner]

            direction = -grad
            m = beta * m + (1.0 - beta) * direction

            step = base_step / math.sqrt(it + 1)
            max_abs = np.max(np.abs(m))
            if max_abs == 0.0:
                max_abs = 1.0
            eta = step / max_abs

            a_new = a + eta * m
            a_new = _project_onto_simplex(a_new, target)

            if it % 5 == 0:
                self.maybe_update(a_new)

            a = a_new
            it += 1

            if self.best_score < 1.503:
                break

    # ------------------------------------------------------------------
    def run_softmax_grad(self, init_vec: np.ndarray):
        """Gradient descent on a soft‑max smoothed max‑convolution."""
        a = init_vec.astype(float)
        n = a.shape[0]
        target = math.sqrt(2.0 * n)

        m = np.zeros_like(a)          # momentum
        beta = 0.9
        base_step = 0.5
        it = 0
        max_iters = 5000
        # temperature schedule parameters
        c0 = 0.3
        c_max = 8.0

        while time.time() < self.deadline and it < max_iters:
            # step size schedule
            step = base_step / math.sqrt(it + 1)
            # temperature schedule (linear)
            c = c0 + (c_max - c0) * (it / max_iters)

            conv = _conv_self(a)
            # compute soft‑max weights
            max_conv = np.max(conv)
            exps = np.exp(c * (conv - max_conv))
            sum_exps = np.sum(exps)
            if sum_exps == 0 or np.isnan(sum_exps):
                w = np.full_like(conv, 1.0 / conv.size)
            else:
                w = exps / sum_exps

            # gradient = 2 * (w * a_rev) convolved
            a_rev = a[::-1]
            grad_full = _fft_convolve_full(w, a_rev)
            start = n - 1
            grad = 2.0 * grad_full[start:start + n]

            direction = -grad
            m = beta * m + (1.0 - beta) * direction

            max_abs = np.max(np.abs(m))
            if max_abs == 0.0:
                max_abs = 1.0
            eta = step / max_abs

            a_new = a + eta * m
            a_new = _project_onto_simplex(a_new, target)

            if it % 10 == 0:
                self.maybe_update(a_new)

            a = a_new
            it += 1

            if self.best_score < 1.503:
                break

    # ------------------------------------------------------------------
    def lp_refine(self):
        """Linear‑programming refinement (highs solver)."""
        if linprog is None:
            return
        if self.best_seq is None:
            return

        a = np.array(self.best_seq, dtype=float)

Optimiser for the step–function bound

The routine combines several ideas:

1. A quick sweep of low‑dimensional parametric families (exponential,
   linear, quadratic, triangular, sinusoidal, Gaussian).  Every candidate
   is projected onto the simplex Σa_i = √(2 n) and evaluated; the best
   candidate is retained.

2. Two heavy local‑optimisation procedures:
   * a classic sub‑gradient descent that follows the maximal autocorrelation
     entry;
   * a “soft‑max” gradient descent in which the max‑operator is smoothed
     with a temperature parameter.  The gradient of the smoothed objective
     is obtained by a second FFT‑based convolution, making each iteration
     O(n log n).

   Both procedures use Euclidean projection onto the simplex and a simple
   momentum scheme.

3. Random restarts with diverse shape initialisers (including a random
   Dirichlet draw) followed by the two refinements.

4. An optional linear‑programming polishing step (SciPy required) that
   solves the LP from the state‑of‑the‑art description.

The whole search stops after ``budget_s`` seconds (default 1000) and
returns the best sequence found.  All helper functions are defined at the
module level; no closures or lambdas are used.

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT based full convolution
# ----------------------------------------------------------------------
def _fft_convolve_full(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full (linear) convolution of two real 1‑D arrays using FFT."""
    n = x.shape[0]
    m = y.shape[0]
    size = 1 << ((n + m - 1).bit_length())   # next power of two
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(X * Y, n=size)
    return conv[: n + m - 1]


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Autocorrelation a * a (full length)."""
    return _fft_convolve_full(a, a)


# ----------------------------------------------------------------------
# Shape factories (non‑normalised)
# ----------------------------------------------------------------------
def _shape_exp(n: int, alpha: float) -> np.ndarray:
    return np.exp(-alpha * np.arange(n, dtype=float))


def _shape_exp_rev(n: int, alpha: float) -> np.ndarray:
    return np.exp(-alpha * np.arange(n - 1, -1, -1, dtype=float))


def _shape_linear_desc(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    return np.maximum(0.0, 1.0 - beta * i)


def _shape_quad_desc(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    return np.maximum(0.0, 1.0 - beta * i * i)


def _shape_triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    return np.minimum(i + 1.0, n - i)


def _shape_sin(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    return np.sin(np.pi * (i + 1) / (n + 1))


def _shape_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=float) - (n - 1) / 2.0
