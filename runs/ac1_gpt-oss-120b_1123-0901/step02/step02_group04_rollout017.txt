def _smooth_max_gradient(a, epsilon):
    conv = conv_self(a)   # length 2n-1
    max_c = conv.max()
    # compute weights
    exp_terms = np.exp((conv - max_c) / epsilon)
    w = exp_terms / exp_terms.sum()
    # gradient = 2 * (a * w) convolution truncated
    # Use FFT convolution
    n = a.shape[0]
    size = 1 << ((n + (2*n - 1) - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    fw = np.fft.rfft(w, n=size)
    conv_full = np.fft.irfft(fa * fw, n=size)
    grad = 2 * conv_full[:n]
    return grad

"""
Propose a step‑function height sequence that minimises

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The optimizer works in three stages:

1️⃣   Quick sweep over a few low‑dimensional parametric families
     (exponential, linear, triangular).  Every candidate is projected
     onto the simplex Σa_i = √(2 n) and evaluated; the best one is kept.

2️⃣   Smooth‑max gradient descent with momentum.
     The non‑smooth objective “max(conv)’’ is replaced by a
     temperature‑controlled soft‑max:
          f_ε(a) = ε·log Σ_k exp(conv_k / ε) .
     Its gradient can be written in closed form and is evaluated by
     FFT‑based convolutions, allowing thousands of inexpensive iterations.
     The temperature ε is slowly decreased (homotopy) so that the
     optimisation progressively approximates the true max.

3️⃣   (Optional) a single linear‑programming refinement exactly as in the
     SOTA description – maximise Σ g_j under the constraint
     (a * g) ≤ max(a * a).  The resulting g is blended into the current
     best sequence.

All helper routines are defined at module level (no closures).
The routine respects the supplied time budget (default 1000 s) and
returns the best list of non‑negative floats found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the simplex {x ≥ 0, Σx = s}
# ----------------------------------------------------------------------
def project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper: full self‑convolution (via FFT, O(n log n))
# ----------------------------------------------------------------------
def conv_self(a: np.ndarray) -> np.ndarray:
    """Return the full convolution a * a (real FFT version)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())  # next power of two
    fa = np.fft.rfft(a, n=size)
    conv_full = np.fft.irfft(fa * fa, n=size)
    return conv_full[: 2 * n - 1]


# ----------------------------------------------------------------------

# ----------------------------------------------------------------------
def _evaluate(seq: list) -> float:
    """Return the score from evaluate_sequence or +inf on error."""
    try:
        return evaluate_sequence(seq)
    except Exception:  # pragma: no cover
        return float("inf")


# ----------------------------------------------------------------------
# Helper: gradient of the ε‑smoothed max‑convolution objective
# ----------------------------------------------------------------------
def smooth_max_gradient(a: np.ndarray, eps: float) -> np.ndarray:
    """
    Compute ∇ f_ε(a) where
        f_ε(a) = ε·log Σ_k exp(conv_k/ε)
    The gradient is
        g_i = 2· Σ_k w_k·a_{k‑i},
    where w_k ∝ exp(conv_k/ε) and Σ w_k = 1.
    It is evaluated via FFT‑based convolution.
    """
    n = a.shape[0]
    conv = conv_self(a)                     # length 2n‑1
    max_c = conv.max()
    # soft‑max weights (stable)
    weights = np.exp((conv - max_c) / eps)
    weights /= weights.sum()

    # gradient: g = 2 * (a * weights_rev) slice, reversed order
    w_rev = weights[::-1]                   # reverse for the required alignment
    size = 1 << ((n + (2 * n - 1) - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    fw = np.fft.rfft(w_rev, n=size)
    conv_full = np.fft.irfft(fa * fw, n=size)

    # valid slice corresponds to indices n‑1 … 2n‑2 (length n)
    slice_ = conv_full[n - 1 : 2 * n - 1]   # shape (n,)
    grad = 2.0 * slice_[::-1]               # reverse to obtain g_i
    return grad

    return np.exp(-alpha * np.arange(n, dtype=float))

    i = np.arange(n, dtype=float)
    return np.maximum(0.0, 1.0 - beta * i)


def shape_triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    return np.minimum(i + 1.0, n - i)

            a0 = shape_exp(n, alpha)
        elif init_type == 'lin':
            beta = np.random.rand() * (2.0 / max(1, n - 1))
            a0 = shape_linear_desc(n, beta)
        else:  # completely random
            a0 = np.random.rand(n)

        a0 = a0 / a0.sum() * target
        run_smooth_descent(a0)

        if best_score < 1.503:
            break

    # ------------------------------------------------------------------
    # 3️⃣  Optional LP refinement (requires SciPy)
    # ------------------------------------------------------------------
    try:
        from scipy.optimize import linprog
    except Exception:  # pragma: no cover
        linprog = None

    if linprog is not None and best_seq is not None and time.time() < deadline:
        a = best_seq.copy()
        n = a.shape[0]
        target = math.sqrt(2.0 * n)
        rows = 2 * n - 1

        # Pre‑allocate constraint matrix once (it changes each iteration because a changes)
        while time.time() < deadline:
            conv = conv_self(a)
            M = conv.max()

            # Build A such that (a * g)_k = Σ_i a_i g_{k-i}
            A = np.zeros((rows, n), dtype=float)
            for k in range(rows):
                lo = max(0, k - (n - 1))
                hi = min(n - 1, k)
                cols = np.arange(lo, hi + 1, dtype=int)
                A[k, cols] = a[k - cols]

            res = linprog(
                c=-np.ones(n),          # maximise Σ g_j  → minimise -Σ g_j
                A_ub=A,
                b_ub=np.full(rows, M),
                bounds=(0, None),
                method="highs",
                options={"presolve": True},
            )
            if not res.success:
                break

            g = res.x
            s = g.sum()
            if s == 0:
                break
            g = g * (target / s)

            # convex combination (t is modest to keep stability)
            t = 0.2
            a = (1 - t) * a + t * g

            sc = _evaluate(a.tolist())
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
                print(f"[{time.time():.2f}] LP refinement improved = {best_score:.6f}")

            if best_score < 1.503 or time.time() + 0.5 > deadline:
                break

    # ------------------------------------------------------------------
    # Fallback – uniform distribution if everything else failed
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=float)

    return best_seq.tolist()
