def propose_candidate(seed=0, budget_s=1000.0, **kwargs):
    start = time.time()
    deadline = start + budget_s - 0.5
    np.random.seed(seed)

    # get initial best
    best_seq = None; best_score = np.inf
    if 'height_sequence_1' in globals():
        try:
            cand = np.asarray(height_sequence_1, dtype=float)
            sc = float(evaluate_sequence(cand.tolist()))
            best_seq = cand copy; best_score = sc
            print(f"Start from historic seq {sc}")
        except:
            pass
    if best_seq is None:
        # uniform baseline for length 1000
        n0 = 1000
        target0 = math.sqrt(2.0 * n0)
        best_seq = np.full(n0, target0 / n0)
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print("Start from uniform", best_score)

    # define helper functions: _project_onto_simplex (copy earlier), _max_conv_and_grad, _local_descent, _lp_refine etc.

    # Phase 1: systematic family search
    alphas = np.linspace(0.001, 0.03, 30) # maybe 30 grid points
    ns = [400,600,800,1000,1200,1500,1800,2000]
    for n in ns:
        target = math.sqrt(2.0 * n)
        for alpha in alphas:
            if time.time() > deadline - 5:
                break
            a = np.exp(-alpha * np.arange(n))
            a = a * (target / a.sum())
            try:
                sc = float(evaluate_sequence(a.tolist()))
            except Exception:
                continue
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
                print(f"[{time.time():.2f}] Exponential n={n} alpha={alpha:.4f} new best={sc:.6f}")

    # Phase 2: gradient descent refinement
    while time.time() < deadline - 5:
        # choose candidate init: best_seq resized to random length around best length
        n = best_seq.size
        # occasional random change of length
        if np.random.rand() < 0.2:
            n = int(n * np.random.uniform(0.9, 1.1))
            n = max(200, min(3000, n))
        init = _resize_sequence(best_seq, n)
        # allocate time for run
        remaining = deadline - time.time()
        run_time = min(remaining * 0.5, 30.0) # use up to 30s per run
        cand_seq, cand_score = _local_descent(init, math.sqrt(2.0 * n), run_time, base_step=0.08, print_tag=f"refine n={n}")
        if cand_seq is not None and cand_score < best_score - 1e-6:
            best_score = cand_score
            best_seq = cand_seq.copy()
            print(f"[{time.time():.2f}] *** UPDATED GLOBAL BEST = {best_score:.6f} (n={n})")
        # occasional LP refinement
        if _CVX_AVAILABLE and n <= 800 and np.random.rand() < 0.3:
            lp_time = min(remaining * 0.2, 15.0)

            if cand_seq2 is not None and cand_score2 < best_score:
                best_score = cand_score2
                best_seq = cand_seq2.copy()
                print(f"LP refined new best {best_score}")

    # final fallback if not set
    if best_seq is None:
        n = 1000
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n)

    return best_seq.tolist()

def _max_conv_and_grad(a):
    n = a.size
    # compute conv via FFT
    L = 1 << ((2*n-1).bit_length())
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa*fa, n=L)[:2*n-1]
    max_idx = int(np.argmax(conv))
    max_val = conv[max_idx]

    low = max(0, max_idx - (n - 1))
    high = min(n - 1, max_idx)
    idx = np.arange(low, high+1)
    partner = max_idx - idx
    grad = np.zeros_like(a)
    grad[idx] = 2.0 * a[partner]
    return max_val, grad, max_idx, conv
```

"""
A fast optimizer for the step‑function bound.

The objective after normalisation is simply the maximum entry of the
self‑convolution a*a.  The algorithm consists of three stages:

1.   Use a cheap family search (exponential / power law shapes) to
     obtain a decent starting point.
2.   Run a *flattening* gradient descent that minimises
     Σ (a*a - μ)² where μ is the current mean of the convolution.
     This drives the whole convolution curve towards a constant (the
     theoretical lower bound is the mean ≈1), thus reduces the peak.
3.   Refine the best candidate with a short projected‑gradient descent
     on the true max‑entry (sub‑gradient) and an optional LP‑update
     (if cvxpy is available).

All updates respect the simplex constraint Σ a_i = √(2n) and non‑negativity
by projecting onto the simplex after each step.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simple Euclidean projection onto the simplex {x ≥ 0, Σ x = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex of non‑negative vectors with sum s."""
    n = v.shape[0]
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper: convolution via FFT (full length)
# ----------------------------------------------------------------------
def _conv_full_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full discrete convolution of x and y using FFT."""
    n = x.size + y.size - 1
    L = 1 << ((n - 1).bit_length())
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    Z = X * Y
    conv = np.fft.irfft(Z, n=L)[:n]
    return conv


# ----------------------------------------------------------------------
# Max‑convolution and its sub‑gradient (used by the local descent)
# ----------------------------------------------------------------------
def _max_conv_and_grad(a: np.ndarray):
    """Return (max_val, grad, max_idx, full_conv) for the entry that attains max."""
    n = a.shape[0]
    conv = _conv_full_fft(a, a)
    max_idx = int(np.argmax(conv))
    max_val = conv[max_idx]

    low = max(0, max_idx - (n - 1))
    high = min(n - 1, max_idx)
    idx = np.arange(low, high + 1, dtype=int)
    partner = max_idx - idx
    grad = np.zeros_like(a)
    grad[idx] = 2.0 * a[partner]
    return max_val, grad, max_idx, conv


# ----------------------------------------------------------------------
# Very cheap mutation (Gaussian noise + simplex projection)
# ----------------------------------------------------------------------
def _mutate(a: np.ndarray, target_sum: float, sigma: float) -> np.ndarray:
    b = a + np.random.randn(a.size) * sigma
    b = np.maximum(b, 0.0)
    b = _project_onto_simplex(b, target_sum)
    return b


# ----------------------------------------------------------------------
# Local projected‑gradient descent on the true max‑entry (sub‑gradient)
# ----------------------------------------------------------------------
def _local_descent(
    init: np.ndarray,
    target_sum: float,
    time_limit: float,
    base_step: float = 0.05,
    print_tag: str = "",
) -> tuple[np.ndarray, float]:
    a = _project_onto_simplex(init.astype(float), target_sum)
    best_seq = a.copy()
    best_score = float(evaluate_sequence(a.tolist()))
    step = base_step
    it = 0
    t0 = time.time()

    while time.time() < t0 + time_limit:
        max_val, grad, _, _ = _max_conv_and_grad(a)
        mean_g = np.mean(grad)
        d = -grad + mean_g                # zero‑sum direction
        max_abs = np.max(np.abs(d)) if np.max(np.abs(d)) > 0 else 1.0
        eta = step * max_val / max_abs

        a_new = a + eta * d
        a_new = _project_onto_simplex(a_new, target_sum)

        score = float(evaluate_sequence(a_new.tolist()))
        if score < best_score:
            best_score = score
            best_seq = a_new.copy()
            print(f"[{time.time():.2f}] {print_tag} new local best = {score:.6f}")

        if np.random.rand() < 0.02:                 # occasional shake‑up
            sigma = 0.01 * target_sum / len(a)
            a = _mutate(a_new, target_sum, sigma)
            step *= 0.9
        else:
            a = a_new

        if it % 200 == 0 and it > 0:
            step *= 0.9
        it += 1

    return best_seq, best_score


# ----------------------------------------------------------------------
# Flattening gradient descent (minimise Σ (a*a - μ)² )
# ----------------------------------------------------------------------
def _flatten_descent(
    init: np.ndarray,
    target_sum: float,
    time_limit: float,
    base_step: float = 0.03,
    momentum: float = 0.9,
    print_tag: str = "",
) -> tuple[np.ndarray, float]:
    a = _project_onto_simplex(init.astype(float), target_sum)
    best_seq = a.copy()
    best_score = float(evaluate_sequence(a.tolist()))
    vel = np.zeros_like(a)
    t0 = time.time()
    it = 0

    while time.time() < t0 + time_limit:
        # full convolution
        conv = _conv_full_fft(a, a)
        mu = conv.mean()
        err = conv - mu                     # zero‑mean error

        # gradient of Σ err²  is  4 * (err * a_rev)  (slice of length n)
        a_rev = a[::-1]
        grad_full = _conv_full_fft(err, a_rev)          # length 3n‑2
        grad = 4.0 * grad_full[len(a) - 1 : len(a) - 1 + len(a)]

        # momentum update
        vel = momentum * vel - base_step * grad
        a_new = a + vel
        a_new = _project_onto_simplex(a_new, target_sum)

        score = float(evaluate_sequence(a_new.tolist()))
        if score < best_score:
            best_score = score
            best_seq = a_new.copy()
            print(f"[{time.time():.2f}] {print_tag} new flatten best = {score:.6f}")

        a = a_new
        # simple step‑size decay
        if it % 200 == 0 and it > 0:
            base_step *= 0.85
        it += 1

    return best_seq, best_score

def _solve_lp_for_g(f: np.ndarray, M: float):
    if not _CVX_AVAILABLE:
        return None
    n = f.shape[0]
    g = cp.Variable(n, nonneg=True)
    constraints = []
    for k in range(2 * n - 1):
        low = max(0, k - (n - 1))
        high = min(n - 1, k)
        idx = np.arange(low, high + 1, dtype=int)
        partner = k - idx
        expr = cp.sum(cp.multiply(f[low : high + 1], g[partner]))
        constraints.append(expr <= M)
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.GLPK, warm_start=True, glpkk=1e-08, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000, verbose=False)
        except Exception:
            return None
    if prob.status not in ("optimal", "optimal_inaccurate"):
        return None
    g_val = np.asarray(g.value).flatten()
    if g_val is None:
        return None
    return np.maximum(g_val, 0.0)


def _lp_refine(
    init: np.ndarray,
    target_sum: float,
    time_limit: float,
    mix: float = 0.2,
    max_iter: int = 6,
) -> tuple[np.ndarray, float]:
    a = _project_onto_simplex(init.astype(float), target_sum)
    best_score = float(evaluate_sequence(a.tolist()))
    best_seq = a.copy()
    t_start = time.time()
    for it in range(max_iter):
        if time.time() - t_start > time_limit:
            break
        conv = _conv_full_fft(a, a)
        M = float(conv.max())
        g0 = _solve_lp_for_g(a, M)
        if g0 is None or np.sum(g0) == 0:
            break
        scale = target_sum / np.sum(g0)
        g = g0 * scale
        a = (1.0 - mix) * a + mix * g
        a = _project_onto_simplex(a, target_sum)
        score = float(evaluate_sequence(a.tolist()))
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time():.2f}] LP‑refine iter {it+1} new best = {score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Resize helper – linear interpolation / simple down‑sampling
# ----------------------------------------------------------------------
def _resize_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    old_n = seq.size
    if old_n == new_n:
        return seq.copy()
    if old_n > new_n:
        idx = np.linspace(0, old_n - 1, new_n).astype(int)
        return seq[idx]
    else:
        x_old = np.arange(old_n)
        x_new = np.linspace(0, old_n - 1, new_n)
        return np.interp(x_new, x_old, seq)


# ----------------------------------------------------------------------
# Main entry point – propose a candidate sequence
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative step‑function with a low evaluation score.
    Returns the best sequence found before the time budget expires.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Global best holder (use historic solution if present)
    # ------------------------------------------------------------------
    best_seq = None
    best_score = math.inf
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(height_sequence_1, dtype=float)
            sc = float(evaluate_sequence(cand.tolist()))
            best_seq = cand.copy()
            best_score = sc
            print(f"[{time.time():.2f}] historic start score = {sc:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Fallback – uniform sequence of length 1200 (a decent size)
    # ------------------------------------------------------------------
    if best_seq is None:
        n0 = 1200
        target0 = math.sqrt(2.0 * n0)
        best_seq = np.full(n0, target0 / n0, dtype=float)
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"[{time.time():.2f}] uniform start (n={n0}) score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Helper: quick family search (exponential / power‑law shapes)
    # ------------------------------------------------------------------
    def _family_search(time_left: float):
        nonlocal best_seq, best_score
        # try a few lengths and a handful of shape parameters
        ns = [400, 800, 1200, 1600, 2000]
        alphas = np.linspace(0.001, 0.02, 15)   # exponential decay rates
        ps = np.linspace(-2.0, -0.1, 12)       # power‑law exponents
        for n in ns:
            target = math.sqrt(2.0 * n)
            idx = np.arange(n, dtype=float)

            # exponential shapes
            for a in alphas:
                if time.time() > deadline - 2.0:
                    return
                raw = np.exp(-a * idx)
                seq = raw * (target / raw.sum())
                sc = float(evaluate_sequence(seq.tolist()))
                if sc < best_score:
                    best_score = sc
                    best_seq = seq.copy()
                    print(f"[{time.time():.2f}] exp n={n} a={a:.4f} new best = {sc:.6f}")

            # power‑law shapes (a_i = (i+1)^p)
            for p in ps:
                if time.time() > deadline - 2.0:
                    return
                raw = (idx + 1) ** p
                seq = raw * (target / raw.sum())
                sc = float(evaluate_sequence(seq.tolist()))
                if sc < best_score:
                    best_score = sc
                    best_seq = seq.copy()
                    print(f"[{time.time():.2f}] pow n={n} p={p:.3f} new best = {sc:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – cheap family scan (use up to 10 % of the budget)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 20.0:
        _family_search(remaining * 0.1)

    # ------------------------------------------------------------------
    # Phase 2 – flattening gradient descent (main workhorse)
    # ------------------------------------------------------------------
    while time.time() < deadline - 5.0:
        # pick a length (prefer current best length but allow variation)
        if np.random.rand() < 0.7 and best_seq is not None:
            n = best_seq.size
            if np.random.rand() < 0.3:
                n = int(n * np.random.uniform(0.9, 1.1))
                n = max(200, min(2500, n))
        else:
            n = np.random.randint(300, 2000)

        target = math.sqrt(2.0 * n)

        # initialise candidate (resize best or random)
        if best_seq is not None:
            init = _resize_sequence(best_seq, n)
        else:
            init = np.random.rand(n)

        # allocate a time slice for this run
        time_left = deadline - time.time()
        if time_left < 6.0:
            break
        run_time = min(time_left * 0.6, 30.0)

        # flattening descent
        cand_seq, cand_score = _flatten_descent(
            init, target, run_time, base_step=0.04, momentum=0.9,
            print_tag=f"flat n={n}"
        )
        if cand_seq is not None and cand_score < best_score:
            best_score = cand_score
            best_seq = cand_seq.copy()
            print(f"[{time.time():.2f}] *** GLOBAL NEW BEST (flatten) = {best_score:.6f} (n={n})")

        # occasional local refinement on the same length
        if np.random.rand() < 0.25:
            cand2, sc2 = _local_descent(
                cand_seq, target, run_time * 0.4, base_step=0.07,
                print_tag=f"local n={n}"
            )
            if cand2 is not None and sc2 < best_score:
                best_score = sc2
                best_seq = cand2.copy()
                print(f"[{time.time():.2f}] *** GLOBAL NEW BEST (local) = {best_score:.6f} (n={n})")

        # optional LP refinement for modest lengths
        if _CVX_AVAILABLE and n <= 800 and np.random.rand() < 0.15:
            cand3, sc3 = _lp_refine(
                cand_seq, target, run_time * 0.3, mix=0.18, max_iter=5
            )
            if cand3 is not None and sc3 < best_score:
                best_score = sc3
                best_seq = cand3.copy()
                print(f"[{time.time():.2f}] *** GLOBAL NEW BEST (LP) = {best_score:.6f} (n={n})")

    # ------------------------------------------------------------------
    # Final safety net (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1200
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=float)

    return best_seq.tolist()
