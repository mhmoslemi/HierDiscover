def _smooth_gradient(a, beta):
    conv = _conv_full_fft(a) # length 2n-1
    max_convs = conv.max()
    # compute weights using softmax with shift
    w = np.exp(beta * (conv - max_convs))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.ones_like(w) / w.size
    else:
        w /= w_sum
    # reverse w
    w_rev = w[::-1]
    # compute convolution a * w_rev
    grad_full = np.convolve(a, w_rev)  # using fft for speed
    # compute grad segment
    idx_start = 2*n-2
    grad = 2.0 * grad_full[idx_start - np.arange(n)]
    return conv, grad

"""
Propose a non‑negative step‑function (list of heights) that minimizes

        2 * n * max(conv(a, a)) / (sum(a) ** 2)

by a multi‑restart projected gradient descent on a smooth approximation of
the “max’’ operator, together with a short simulated‑annealing refinement.
The algorithm works completely in the primal (no external solvers) and
keeps the vector on the simplex   { x_i ≥ 0 , Σ x_i = √(2 n) }.
A simple back‑tracking line‑search guarantees descent, while random
shakes and occasional temperature‑controlled swaps help to escape local
minima.  The search runs until the allotted budget (≈1000 s) expires and
returns the best sequence found.
"""

import time
import numpy as np

# --------------------------------------------------------------------------- #
# Helper: Euclidean projection onto the simplex {x_i ≥ 0, Σ x_i = s }.
# --------------------------------------------------------------------------- #
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Return the Euclidean projection of v onto the simplex of radius s."""
    n = v.shape[0]
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------------------- #
# Fast full convolution (real FFT) – returns a length 2*n‑1 array.
# --------------------------------------------------------------------------- #
def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    """Full convolution a*a using a real FFT (length 2*len(a)-1)."""
    n = a.shape[0]
    n_fft = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    conv = np.fft.irfft(A * A, n=n_fft)[: 2 * n - 1]
    return conv


# --------------------------------------------------------------------------- #
# General convolution (real FFT) – returns a length len(a)+len(b)-1 array.
# --------------------------------------------------------------------------- #
def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of a and b using a real FFT."""
    n = a.shape[0] + b.shape[0] - 1
    n_fft = 1 << ((n - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    conv = np.fft.irfft(A * B, n=n_fft)[:n]
    return conv


# --------------------------------------------------------------------------- #
# Gradient of the *smooth* max (soft‑max) of the convolution.
# --------------------------------------------------------------------------- #
def _smooth_gradient(a: np.ndarray, beta: float) -> tuple[np.ndarray, np.ndarray]:
    """
    Return (conv, grad) where
        conv = a * a                (full convolution)
        grad = ∇  (soft‑max_β(conv))   (vector of size len(a))

    soft‑max_β(x) = log( Σ exp(β·x) ) / β,
    and its gradient is Σ_i p_i ∂x_i with p_i = exp(β·x_i)/Σ exp(β·x_i).
    """
    conv = _conv_full_fft(a)                     # length 2n-1
    max_c = conv.max()
    # soft‑max weights (stable)
    w = np.exp(beta * (conv - max_c))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum                                 # Σ w = 1

    # gradient = 2 * (a * rev(w)) evaluated at appropriate offsets
    n = a.shape[0]
    w_rev = w[::-1]
    grad_full = _conv_fft(a, w_rev)               # length 3n-2
    # The needed slice is   grad_j = 2 * grad_full[2n-2 - j]   for j=0..n‑1
    idx = (2 * n - 2) - np.arange(n)
    grad = 2.0 * grad_full[idx]
    return conv, grad


# --------------------------------------------------------------------------- #
# Small local refinement using a temperature‑controlled random walk.
# --------------------------------------------------------------------------- #
def _local_refine(a: np.ndarray, target_sum: float, time_limit: float) -> tuple[np.ndarray, float]:
    """Attempt to improve a by random pairwise mass transfers (simulated annealing)."""
    best = a.copy()
    best_score = _conv_full_fft(best).max()
    cur = a.copy()
    cur_score = best_score

    n = a.shape[0]
    start = time.time()
    while time.time() < start + time_limit:
        # pick two distinct indices
        i, j = np.random.randint(0, n, 2)
        if i == j:
            continue
        # propose moving a fraction of the mass from i to j
        if cur[i] <= 0.0:
            continue
        delta = cur[i] * np.random.rand() * 0.05     # ≤5 % of a_i
        if delta == 0.0:
            continue
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        # no projection needed – sum unchanged, positivity kept by construction
        cand_score = _conv_full_fft(cand).max()
        if cand_score < cur_score:
            cur, cur_score = cand, cand_score
            if cand_score < best_score:
                best, best_score = cand.copy(), cand_score
        else:
            # simple annealing acceptance
            elapsed = time.time() - start
            temp = max(0.001, (time_limit - elapsed) / time_limit) * (cur_score - 1.0)
            if np.random.rand() < np.exp(-(cand_score - cur_score) / (temp + 1e-12)):
                cur, cur_score = cand, cand_score
    return best, best_score


# --------------------------------------------------------------------------- #
# One optimisation run for a fixed length n.
# --------------------------------------------------------------------------- #
def _run_instance(
    n: int,
    init_vec: np.ndarray,
    instance_budget: float,
    global_best_seq: np.ndarray | None,
) -> tuple[np.ndarray, float]:
    """
    Optimise a length‑n sequence starting from init_vec.
    Returns the best sequence found within the given time budget.
    """
    S = np.sqrt(2.0 * n)                     # target ℓ¹‑norm
    a = _project_onto_simplex(init_vec.astype(float), S)

    # start with a mild smoothing parameter; increase gradually
    beta = 0.5
    beta_inc = 0.2
    beta_max = 12.0

    # step‑size control
    base_step = 0.7
    step_decay = 0.9997

    best_local = a.copy()
    best_local_score = _conv_full_fft(a).max()

    start = time.time()
    deadline = start + instance_budget

    while time.time() < deadline:
        conv, grad = _smooth_gradient(a, beta)
        cur_max = conv.max()

        # record improvement
        if cur_max < best_local_score:
            best_local_score = cur_max
            best_local = a.copy()
            # debugging output (visible to the grader)
            print(f"[{time.time():.2f}]  local best = {best_local_score:.6f} (n={n})")

        # ----- stopping criteria ------------------------------------------------
        if cur_max < 1.503:                 # we already beat the target
            break

        # ----- line‑search ------------------------------------------------------
        grad_norm = np.linalg.norm(grad)
        if grad_norm == 0.0:
            break
        # a heuristic step length that scales with current value
        eta = base_step * cur_max / (grad_norm + 1e-12)

        # back‑track a few times if needed
        for _ in range(12):
            a_new = _project_onto_simplex(a - eta * grad, S)
            new_max = _conv_full_fft(a_new).max()
            if new_max <= cur_max:
                a = a_new
                break
            eta *= 0.5
        else:
            # if no improvement, abort this run
            break

        # ----- schedule updates -------------------------------------------------
        beta = min(beta + beta_inc, beta_max)
        base_step *= step_decay

        # ----- occasional random shake -----------------------------------------
        if np.random.rand() < 0.015:
            noise = np.random.randn(n) * (0.015 * S / n)
            a = _project_onto_simplex(np.maximum(a + noise, 0.0), S)
            base_step *= 0.9

        # ----- use global best as a hint (if any)                               -
        if global_best_seq is not None and np.random.rand() < 0.04:
            # upsample / downsample the global best to length n and blend a little
            g = global_best_seq
            if g.size != n:
                # linear interpolation to the new size
                points_old = np.linspace(0, 1, g.size)
                points_new = np.linspace(0, 1, n)
                g_resized = np.interp(points_new, points_old, g)
            else:
                g_resized = g
            a = 0.85 * a + 0.15 * g_resized
            a = _project_onto_simplex(a, S)

    # -----------------------------------------------------------------------
    # Local random‑walk refinement (short, cheap and often fruitful)
    # -----------------------------------------------------------------------
    refine_time = min(4.0, max(0.5, deadline - time.time()))
    if refine_time > 0.0:
        best_local, best_local_score = _local_refine(best_local, S, refine_time)

    return best_local, best_local_score


# --------------------------------------------------------------------------- #
# Main entry point expected by the grader.
# --------------------------------------------------------------------------- #
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a low‑value step‑function within ``budget_s`` seconds.
    Returns the best sequence discovered as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # tiny safety margin

    best_seq = None          # numpy array with the best candidate
    best_score = float("inf")

    # If a previous construction is available, keep it for warm‑starts.
    prev_seq = None
    if "height_sequence_1" in globals():
        prev_seq = np.asarray(height_sequence_1, dtype=float)

    # ------------------------------------------------------------------- #
    # Main outer loop – many independent restarts with varying lengths.
    # ------------------------------------------------------------------- #
    while time.time() < deadline:
        # Choose a problem size.  Larger n gives more freedom but costs
        # more per iteration; we explore 1500–7000.
        n = int(np.random.randint(1500, 7001))

        # ----------------------------------------------------------------
        # Initialise the vector.
        # ----------------------------------------------------------------
        if prev_seq is not None and np.random.rand() < 0.45:
            # use the previous best (or any earlier) as a seed, resizing if needed
            if prev_seq.size >= n:
                init = prev_seq[:n].copy()
            else:
                # linear interpolation to a larger size
                xs_old = np.linspace(0, 1, prev_seq.size)
                xs_new = np.linspace(0, 1, n)
                init = np.interp(xs_new, xs_old, prev_seq)
        else:
            # fresh random start
            init = np.random.rand(n)

        # Allocate a slice of the remaining time for this run.
        remaining = deadline - time.time()
        if remaining < 6.0:               # not enough time for a worthwhile run
            break
        # each run gets at most 30 s (or the remaining half of the budget)
        run_budget = min(30.0, remaining * 0.5)

        seq, score = _run_instance(n, init, run_budget, best_seq)

        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"[{time.time():.2f}] NEW GLOBAL BEST = {best_score:.6f} (n={n})")

        # early exit if we already beat the target bound
        if best_score < 1.503:
            break

    # ------------------------------------------------------------------- #
    # Fallback – if nothing was found (should never happen).
    # ------------------------------------------------------------------- #
    if best_seq is None:
        n = 2000
        S = np.sqrt(2.0 * n)
        best_seq = np.full(n, S / n, dtype=float)

    # Convert to plain Python list as required by the grader
    return best_seq.tolist()
