"""
Propose a step‑function (list of non‑negative heights) with a low
evaluation score.  The optimizer combines several ideas that differ from
the previous hybrid approach:

* A small population (≈30) of candidates of various lengths.
* Simple evolutionary operators (mutation, convex crossover, random
  re‑initialisation) plus a *gaussian‑annealing* schedule for the
  mutation strength.
* Short projected‑gradient refinements (sub‑gradient of the maximal
  convolution entry) for every offspring.
* Optional LP‑refinement (if cvxpy and a fast LP‑solver are available)
  for modest problem sizes.
* An adaptive resizing step that tries larger lengths when a good
  shape is found.

All candidates live on the simplex {x≥0, Σx = √(2 n)} so the evaluation
function reduces to the maximal entry of the linear convolution and
the algorithm respects the constraints automatically.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Simple Euclidean projection onto the simplex {x ≥ 0, Σ x = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex of non‑negative vectors with sum s."""
    n = v.shape[0]
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution helpers (max entry and its sub‑gradient)
# ----------------------------------------------------------------------
def _max_conv_and_grad(a: np.ndarray):
    """
    Compute the maximal entry of conv(a,a), its index and a sub‑gradient.
    Returns (max_val, grad, max_idx, full_conv).
    """
    n = a.shape[0]
    conv = np.convolve(a, a)
    max_idx = int(np.argmax(conv))
    max_val = conv[max_idx]

    # gradient of the chosen entry:
    low = max(0, max_idx - (n - 1))
    high = min(n - 1, max_idx)
    idx = np.arange(low, high + 1, dtype=int)
    partner = max_idx - idx
    grad = np.zeros_like(a)
    grad[idx] = 2.0 * a[partner]
    return max_val, grad, max_idx, conv


# ----------------------------------------------------------------------
# Small mutation helper (Gaussian noise + simplex projection)
# ----------------------------------------------------------------------
def _mutate(a: np.ndarray, target_sum: float, sigma: float) -> np.ndarray:
    """Return a mutated copy of a (Gaussian noise, then simplex projection)."""
    b = a + np.random.randn(a.size) * sigma
    b = np.maximum(b, 0.0)
    b = _project_onto_simplex(b, target_sum)
    return b


# ----------------------------------------------------------------------
# Local projected‑gradient descent (sub‑gradient of the maximal entry)
# ----------------------------------------------------------------------
def _local_descent(
    init: np.ndarray,
    target_sum: float,
    time_limit: float,
    base_step: float = 0.05,
    momentum: float = 0.9,
    print_tag: str = "",
) -> tuple[np.ndarray, float]:
    """
    Short gradient‑descent run from ``init``.
    Returns the best sequence seen and its evaluation score.
    """
    a = _project_onto_simplex(init.astype(float), target_sum)

    # evaluate start
    try:
        best_score = float(evaluate_sequence(a.tolist()))
    except Exception:
        best_score = math.inf
    best_seq = a.copy()

    step = base_step
    vel = np.zeros_like(a)  # momentum buffer
    it = 0
    t0 = time.time()

    while time.time() < t0 + time_limit:
        max_val, grad, _, _ = _max_conv_and_grad(a)

        # direction that preserves the sum (mean‑zero)
        mean_g = np.mean(grad)
        d = -grad + mean_g

        # momentum update
        vel = momentum * vel + d
        # adapt step length to gradient magnitude
        max_abs = np.max(np.abs(vel)) if np.max(np.abs(vel)) > 0 else 1.0
        eta = step * max_val / max_abs

        a_new = a + eta * vel
        a_new = _project_onto_simplex(a_new, target_sum)

        try:
            score = float(evaluate_sequence(a_new.tolist()))
        except Exception:
            score = math.inf

        if score < best_score:
            best_score = score
            best_seq = a_new.copy()
            print(f"[{time.time():.2f}] {print_tag} new local best = {score:.6f} (n={len(a)})")

        # occasional random shake‑up
        if np.random.rand() < 0.01:
            sigma = 0.02 * target_sum / len(a)
            a = _mutate(a_new, target_sum, sigma)
            step *= 0.9
            vel[:] = 0.0
        else:
            a = a_new

        # decay step size every few hundred iterations
        if it % 300 == 0 and it > 0:
            step *= 0.85
        it += 1

    return best_seq, best_score

# Optional LP refinement – uses cvxpy if present

try:
    import cvxpy as cp

    _CVX_AVAILABLE = True
except Exception:  # pragma: no cover
    _CVX_AVAILABLE = False


def _solve_lp_for_g(f: np.ndarray, M: float):
    """Solve the LP: maximise Σ g_j subject to g ≥ 0 and f*g ≤ M."""
    if not _CVX_AVAILABLE:
        return None
    n = f.shape[0]
    g = cp.Variable(n, nonneg=True)
    constraints = []
    for k in range(2 * n - 1):
        low = max(0, k - (n - 1))
        high = min(n - 1, k)
        idx = np.arange(low, high + 1, dtype=int)
        partner = k - idx
        expr = cp.sum(cp.multiply(f[low : high + 1], g[partner]))
        constraints.append(expr <= M)
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.GLPK, warm_start=True, verbose=False, glpkk=1e-08)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, verbose=False, max_iters=2000)
        except Exception:
            return None
    if prob.status not in ("optimal", "optimal_inaccurate"):
        return None
    g_val = np.asarray(g.value).flatten()
    if g_val is None:
        return None
    return np.maximum(g_val, 0.0)


def _lp_refine(
    init: np.ndarray,
    target_sum: float,
    time_limit: float,
    mix: float = 0.2,
    max_iter: int = 4,
) -> tuple[np.ndarray, float]:
    """
    Perform a few LP‑updates starting from ``init``.
    Returns the best sequence and its score.
    """
    a = _project_onto_simplex(init.astype(float), target_sum)
    try:
        best_score = float(evaluate_sequence(a.tolist()))
    except Exception:
        best_score = math.inf
    best_seq = a.copy()

    t_start = time.time()
    for it in range(max_iter):
        if time.time() - t_start > time_limit:
            break
        M = float(np.convolve(a, a).max())
        g0 = _solve_lp_for_g(a, M)
        if g0 is None or np.sum(g0) == 0:
            break
        scale = target_sum / np.sum(g0)
        g = g0 * scale
        a = (1.0 - mix) * a + mix * g
        a = _project_onto_simplex(a, target_sum)
        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            score = math.inf
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time():.2f}] LP‑refine it {it+1} new best = {score:.6f} (n={len(a)})")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Simple resize (linear interpolation / down‑sampling)
# ----------------------------------------------------------------------
def _resize_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize ``seq`` to length ``new_n`` preserving non‑negativity."""
    old_n = seq.size
    if old_n == new_n:
        return seq.copy()
    if old_n > new_n:
        # pick equally spaced indices
        idx = np.linspace(0, old_n - 1, new_n).astype(int)
        return seq[idx]
    else:
        # linear interpolation
        x_old = np.arange(old_n)
        x_new = np.linspace(0, old_n - 1, new_n)
        return np.interp(x_new, x_old, seq)

    # ------------------------------------------------------------------
    population = [(best_seq, best_score)]
    # add a few random seeds of different lengths
    for n in [400, 800, 1200, 1600]:
        a = np.random.rand(n)
        a = _project_onto_simplex(a, math.sqrt(2.0 * n))
        try:
            sc = float(evaluate_sequence(a.tolist()))
        except Exception:
            sc = math.inf
        population.append((a, sc))

    # keep the best 30 candidates only
    population.sort(key=lambda tup: tup[1])
    population = population[:30]

    # ------------------------------------------------------------------
    # Helper to evaluate safely
    # ------------------------------------------------------------------
    def _eval(seq: np.ndarray) -> float:
        try:
            return float(evaluate_sequence(seq.tolist()))
        except Exception:
            return math.inf

    # ------------------------------------------------------------------
    # Main evolutionary loop
    # ------------------------------------------------------------------
    while time.time() < deadline - 1.0:
        # schedule: mutation strength slowly cools down
        elapsed = time.time() - (deadline - budget_s)
        fraction = elapsed / budget_s
        # sigma for mutation (larger early, smaller later)
        sigma_factor = max(0.02, 0.1 * (1 - fraction))

        # Choose parent(s)
        elite = population[:5]                     # top few
        parent_seq, _ = random.choice(elite)

        # Decide which operator to apply
        op = random.random()
        if op < 0.55:
            # mutation
            n = parent_seq.shape[0]
            sigma = sigma_factor * math.sqrt(2.0 * n) / n
            child = _mutate(parent_seq, math.sqrt(2.0 * n), sigma)
        elif op < 0.80:
            # convex crossover between two elites (may need resizing)
            p1, _ = random.choice(elite)
            p2, _ = random.choice(elite)
            if p1.shape[0] != p2.shape[0]:
                n = max(p1.shape[0], p2.shape[0])
                p1 = _resize_sequence(p1, n)
                p2 = _resize_sequence(p2, n)
            w = random.random()
            child = w * p1 + (1 - w) * p2
            child = _project_onto_simplex(child, math.sqrt(2.0 * child.shape[0]))
        elif op < 0.95:
            # random re‑initialisation (different size)
            n = random.randint(400, 2500)
            child = np.random.rand(n)
            child = _project_onto_simplex(child, math.sqrt(2.0 * n))
        else:
            # aggressive resize of the current global best
            n = int(parent_seq.shape[0] * random.uniform(0.9, 1.2))
            n = max(400, min(3000, n))
            child = _resize_sequence(parent_seq, n)
            child = _project_onto_simplex(child, math.sqrt(2.0 * n))

        # ------------------------------------------------------------------
        # Local refinement (short projected‑gradient descent)
        # ------------------------------------------------------------------
        time_left = deadline - time.time()
        if time_left < 1.5:
            break
        # allocate a few seconds for refinement; cap to 4 s
        refine_time = min(3.0, time_left * 0.4)
        child_ref, child_score = _local_descent(
            child,
            math.sqrt(2.0 * child.shape[0]),
            refine_time,
            base_step=0.07,
            momentum=0.85,
            print_tag=f"ref n={child.shape[0]}",
        )

        # ------------------------------------------------------------------
        # Optional LP refinement for modest sizes
        # ------------------------------------------------------------------
        if _CVX_AVAILABLE and child_ref.shape[0] <= 700:
            lp_time = min(1.0, time_left * 0.1)
            child_ref, child_score = _lp_refine(
                child_ref,
                math.sqrt(2.0 * child_ref.shape[0]),
                lp_time,
                mix=0.25,
                max_iter=3,
            )

        # ------------------------------------------------------------------
        # Update global best and population
        # ------------------------------------------------------------------
        if child_score < best_score:
            best_score = child_score
            best_seq = child_ref.copy()
            print(
                f"[{time.time():.2f}] *** GLOBAL NEW BEST = {best_score:.6f} (n={best_seq.size})"
            )
            # early exit if we beat the target bound
            if best_score < 1.503:
                break

        population.append((child_ref, child_score))
        population.sort(key=lambda tup: tup[1])
        if len(population) > 30:
            population = population[:30]

        # Occasionally try to upscale the current best (if we have slack time)
        if random.random() < 0.07 and time.time() < deadline - 5.0:
            # upscale by linear interpolation
            new_n = int(best_seq.shape[0] * random.uniform(1.05, 1.3))
            new_n = min(new_n, 3000)
            up_seq = _resize_sequence(best_seq, new_n)
            up_seq = _project_onto_simplex(up_seq, math.sqrt(2.0 * new_n))
            # short refinement
            up_ref, up_score = _local_descent(
                up_seq,
                math.sqrt(2.0 * new_n),
                min(4.0, deadline - time.time()),
                base_step=0.06,
                momentum=0.8,
                print_tag=f"upscale n={new_n}",
            )
            if up_score < best_score:
                best_score = up_score
                best_seq = up_ref.copy()
                print(
                    f"[{time.time():.2f}] *** GLOBAL UPSCALED BEST = {best_score:.6f} (n={new_n})"
                )

    # ------------------------------------------------------------------
    # Safety fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=float)

    return best_seq.tolist()
