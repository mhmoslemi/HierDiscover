"""
Propose a step‑function sequence with a low evaluation score.

The algorithm works in two stages:
  1.  Fast Adam‑based optimisation on a moderate length (≈2000) using a
      smooth‑max surrogate (soft‑max of the self‑convolution).  The
      iterates stay on the probability simplex (∑a_i = 1, a_i ≥ 0) via
      Euclidean projection.
  2.  The best sequence from stage 1 is linearly up‑sampled to a larger
      length (≈4000) and refined again with the same optimiser.  A
      continuation on the soft‑max temperature (λ) and a decaying learning
      rate allow us to first flatten the whole convolution and then focus
      on the true peak.  Random “shakes’’ are injected when progress stalls.

Both stages respect the 1000 s wall‑time budget and keep the overall best
sequence found.  The returned list satisfies all security checks of
`evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Projection onto the probability simplex {x ≥ 0, ∑x = 1}
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:                       # all numbers non‑positive
        return np.full_like(v, 1.0 / len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear (non‑circular) self‑convolution using FFT
def _full_convolution(a: np.ndarray, size: int = None) -> np.ndarray:
    """Return a*a (length 2*len(a)-1) via FFT."""
    n = len(a)
    if size is None:
        size = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Exact objective used by evaluate_sequence (sum(a) is assumed >0)
def _objective(a: np.ndarray) -> float:
    """2·n·max(self‑convolution) / (∑a)² (with ∑a=1 ⇒ 2·n·max)."""
    n = len(a)
    conv = _full_convolution(a)
    max_b = conv.max()
    sum_a = a.sum()
    return 2.0 * n * max_b / (sum_a * sum_a)


# ----------------------------------------------------------------------
# Gradient of the smooth‑max (log‑sum‑exp) surrogate.
def _softmax_gradient(a: np.ndarray,
                      conv: np.ndarray,
                      lam: float,
                      size_corr: int = None) -> np.ndarray:
    """
    Gradient of φ_λ(a) = (1/λ)·log∑_k e^{λ·c_k}
    where c = a*a .  Returns ∇φ_λ(a) (shape = len(a)).
    """
    n = len(a)
    # soft‑max weights (numerically stable)
    m = conv.max()
    w = np.exp(lam * (conv - m))
    w_sum = w.sum()
    if w_sum == 0.0:                     # should not happen, but guard
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # cross‑correlation w ★ a  (linear)
    rev_a = a[::-1]
    if size_corr is None:
        size_corr = 1 << int(np.ceil(np.log2(len(w) + n - 1)))
    fw = np.fft.rfft(w, size_corr)
    frev = np.fft.rfft(rev_a, size_corr)
    corr = np.fft.irfft(fw * frev, size_corr)

    # slice that corresponds to the n components of the gradient
    grad = 2.0 * corr[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) with a low
    evaluation score.  The routine runs for at most `budget_s` seconds
    and returns the best sequence discovered.
    """
    np.random.seed(seed)
    # safety margin for interpreter & prints
    deadline = time.time() + budget_s - 0.5

    # ------------------------------------------------------------------
    # 0. Helper to keep track of the best solution
    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # 1.  Stage‑1 parameters (moderate length)
    n1 = 2048                       # power of two → fast FFTs
    size_corr_1 = 1 << int(np.ceil(np.log2(3 * n1)))   # for correlation

    # ------------------------------------------------------------------
    # 2.  Initialise a few candidates (geometric + random Dirichlet)
    #    and keep the best of them as a warm start.
    #    All candidates are projected onto the simplex (∑ = 1).
    def _evaluate_and_maybe_update(seq: np.ndarray, label: str):
        nonlocal best_seq, best_val
        val = _objective(seq)
        if val < best_val:
            best_val, best_seq = val, seq.copy()
            print(f"[Init {label}] obj={val:.6f}")

    # geometric families (a_i ∝ r^i)

    # random Dirichlet seeds
    for _ in range(12):

    # use the global variable `height_sequence_1` if present
