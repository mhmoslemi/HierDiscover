def max_conv_and_grad(a):
    n = a.shape[0]
    # compute conv via FFT: l = int(2**np.ceil(np.log2(2*n-1))) maybe using np.fft.rfft length

def _max_conv_and_grad_fft(a):
    n = a.shape[0]
    m = 1 << ((2*n-1).bit_length())
    fft_a = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(fft_a * fft_a, n=m)
    conv = conv[:2*n-1]
    max_idx = int(np.argmax(conv))
    max_val = conv[max_idx]
    # gradient:
    low = max_idx - (n - 1)
    if low < 0:
        low = 0
    high = min(n-1, max_idx)
    idx = np.arange(low, high+1)
    partner = max_idx - idx
    grad = np.zeros_like(a)
    grad[idx] = 2.0 * a[partner]
    return max_val, grad, max_idx, conv

def _max_conv_and_grad_fft(a):
    n = a.shape[0]
    m = 1 << ((2 * n - 1).bit_length())
    # compute rfft of a padded to length m
    A = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(A * A, n=m)[:2*n-1]
    # due to rounding, ensure real
    conv = np.real(conv)
    max_idx = int(np.argmax(conv))
    max_val = conv[max_idx]
    low = max(0, max_idx - (n - 1))
    high = min(n - 1, max_idx)
    idx = np.arange(low, high + 1, dtype=int)
    partner = max_idx - idx
    grad = np.zeros_like(a)
    grad[idx] = 2.0 * a[partner]
    return max_val, grad, max_idx, conv

"""
propose_candidate
-----------------

Search for a non‑negative step‑function (list of heights) that minimises

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

Because the sum is forced to sqrt(2 n), the objective equals the maximal
entry of the linear convolution a * a.  The optimiser uses a hierarchy:

* **Stage 0 – warm start** –  take the historic sequence (if any) or a
  uniform vector and normalise it to the simplex {x ≥ 0, Σx = √(2n)}.

* **Stage 1 – Adam‑style projected descent** – for a given length *n* we
  repeatedly evaluate the maximal convolution entry, compute its (sub‑)
  gradient, and perform an Adam update.  After each update the vector is
  projected back onto the simplex.  The step size is adaptive and the
  algorithm occasionally injects Gaussian mutations to escape shallow
  minima.

* **Stage 2 – multiscale refinement** – the current best vector is
  repeatedly resized (linear interpolation) to a larger length (up to
  a few thousand entries) and refined again.  The schedule automatically
  balances exploration (random initialisations, Gaussian shapes) and
  exploitation (long runs on the current best length).

All heavy work is done with FFT‑based convolution (≈ O(n log n)), allowing
hundreds of thousands of iterations within the 1000 s budget.  The code
keeps the global best and stops early if the target bound of 1.503 is
beaten.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simple Euclidean projection onto the simplex {x ≥ 0, Σ x = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project the vector v onto the simplex of non‑negative vectors with sum s."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution + gradient of the *maximal* entry (FFT implementation)
# ----------------------------------------------------------------------
def _max_conv_and_grad_fft(a: np.ndarray):
    """
    Return (max_val, grad, max_idx, full_conv) for the linear convolution a*a.
    grad is the sub‑gradient of the maximal entry with respect to a.
    """
    n = a.shape[0]
    # length of the padded FFT (next power of two ≥ 2n‑1)
    m = 1 << ((2 * n - 1).bit_length())
    # real FFT, convolution via pointwise product
    A = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(A * A, n=m)[: 2 * n - 1]
    conv = np.real(conv)               # numerical noise removal
    max_idx = int(np.argmax(conv))
    max_val = conv[max_idx]

    # gradient for the chosen index
    low = max(0, max_idx - (n - 1))
    high = min(n - 1, max_idx)
    idx = np.arange(low, high + 1, dtype=int)
    partner = max_idx - idx
    grad = np.zeros_like(a)
    grad[idx] = 2.0 * a[partner]
    return max_val, grad, max_idx, conv


# ----------------------------------------------------------------------
# Tiny mutation (Gaussian noise + simplex projection)
# ----------------------------------------------------------------------
def _mutate(a: np.ndarray, target_sum: float, sigma: float) -> np.ndarray:
    """Return a mutated copy of a (add Gaussian noise then project onto the simplex)."""
    b = a + np.random.randn(a.size) * sigma
    b = np.maximum(b, 0.0)
    return _project_onto_simplex(b, target_sum)


# ----------------------------------------------------------------------
# Simple resize (linear interpolation) – preserves non‑negativity
# ----------------------------------------------------------------------
def _resize_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a sequence to length new_n using linear interpolation (or down‑sampling)."""
    old_n = seq.size
    if old_n == new_n:
        return seq.copy()
    if old_n > new_n:
        # equally spaced down‑sampling
        idx = np.linspace(0, old_n - 1, new_n).astype(int)
        return seq[idx]
    else:
        # linear interpolation
        x_old = np.arange(old_n)
        x_new = np.linspace(0, old_n - 1, new_n)
        return np.interp(x_new, x_old, seq)


# ----------------------------------------------------------------------
# Gaussian‑shaped initialisation
# ----------------------------------------------------------------------
def _init_gaussian(n: int, sigma_factor: float = 0.3) -> np.ndarray:
    """Return a non‑negative Gaussian‐shaped vector of length n (not yet normalised)."""
    x = np.arange(n)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    g = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
    return g


# ----------------------------------------------------------------------
# Adam‑style projected gradient descent on max(conv)
# ----------------------------------------------------------------------
def _adam_descent(
    init: np.ndarray,
    target_sum: float,
    time_limit: float,
    lr: float = 0.25,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    print_tag: str = "",
) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimisation of max(conv(a, a)) on the simplex.
    Returns the best sequence seen during the run and its objective value.
    """
    a = _project_onto_simplex(init.astype(float), target_sum)
    # evaluate initial point
    cur_val, _, _, _ = _max_conv_and_grad_fft(a)
    best_seq = a.copy()
    best_val = float(cur_val)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    # tracking for adaptive learning‑rate decay
    no_improve = 0
    decay_patience = 300  # iterations without improvement before shrinking lr

    t0 = time.time()
    while time.time() < t0 + time_limit:
        max_val, grad, _, _ = _max_conv_and_grad_fft(a)

        # Adam moment updates (we minimise → move opposite to grad)
        t += 1
        g = grad
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step
        a_new = _project_onto_simplex(a_new, target_sum)

        # evaluate new candidate
        new_val, _, _, _ = _max_conv_and_grad_fft(a_new)

        if new_val < best_val:
            best_val = float(new_val)
            best_seq = a_new.copy()
            no_improve = 0
            print(
                f"[{time.time():.2f}] {print_tag} new local best = {best_val:.6f} (n={len(a)})"
            )
        else:
            no_improve += 1

        # occasional random shake‑up to escape flat regions
        if np.random.rand() < 0.01:
            sigma = 0.05 * target_sum / len(a)
            a_new = _mutate(a_new, target_sum, sigma)
            # reset Adam moments to avoid stale direction
            m.fill(0.0)
            v.fill(0.0)

        a = a_new

        # simple learning‑rate decay if stuck
        if no_improve >= decay_patience:
            lr *= 0.5
            no_improve = 0
            # tiny safety: keep lr above a floor
            lr = max(lr, 1e-4)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) minimise the
    evaluation function within the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # small safety margin

    # --------------------------------------------------------------
    # Initialise the global best – use historic solution if available
    # --------------------------------------------------------------
    best_seq = None
    best_score = math.inf

    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(height_sequence_1, dtype=float)
            n0 = cand.size
            target0 = math.sqrt(2.0 * n0)
            cand = _project_onto_simplex(cand, target0)
            best_seq = cand.copy()
            best_score = float(evaluate_sequence(best_seq.tolist()))
            print(f"[{time.time():.2f}] Using historic sequence (n={n0}) score={best_score:.6f}")
        except Exception:
            best_seq = None
            best_score = math.inf

    if best_seq is None:
        # fallback – uniform start at a modest size
        n0 = 500
        target0 = math.sqrt(2.0 * n0)
        best_seq = np.full(n0, target0 / n0, dtype=float)
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"[{time.time():.2f}] No historic data – uniform start (n={n0}) score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Helper: run one optimisation “run” for a chosen length
    # ------------------------------------------------------------------
    def _run_one(n: int, init_vec: np.ndarray, time_allowed: float):
        target = math.sqrt(2.0 * n)
        # run the Adam descent – allocate most of the remaining time but keep it bounded
        run_time = min(time_allowed, 30.0)
        cand_seq, cand_val = _adam_descent(
            init_vec, target, run_time, lr=0.25, print_tag=f"run n={n}"
        )
        return cand_seq, float(cand_val)

    # ------------------------------------------------------------------
    # Hierarchical/multiscale loop
    # ------------------------------------------------------------------
    MIN_N = 200
    MAX_N = 5000

    while time.time() < deadline - 1.0:
        # decide a length – bias toward the current best but allow changes
        if np.random.rand() < 0.7 and best_seq is not None:
            n = best_seq.size
            if np.random.rand() < 0.3:
                factor = np.random.uniform(0.8, 1.2)
                n = int(n * factor)
        else:
            n = np.random.randint(MIN_N, MAX_N + 1)

        n = max(MIN_N, min(MAX_N, n))
        target_sum = math.sqrt(2.0 * n)

        # initialise candidate
        if best_seq is not None:
            init = _resize_sequence(best_seq, n)
        else:
            init = np.random.rand(n)

        # occasional alternative initialisations
        if np.random.rand() < 0.2:
            # Gaussian shaped start
            sigma_f = np.random.uniform(0.15, 0.4)
            init = _init_gaussian(n, sigma_factor=sigma_f)
            init = _project_onto_simplex(init, target_sum)

        # allocate time slice for this run
        time_left = deadline - time.time()
        if time_left < 8.0:
            break
        cand_seq, cand_score = _run_one(n, init, time_left)

        # update global best
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand_seq.copy()
            print(
                f"[{time.time():.2f}] *** GLOBAL NEW BEST = {best_score:.6f} (n={n})"
            )
            if best_score < 1.503:
                break

    # ------------------------------------------------------------------
    # Final refinement on the best length (use any remaining time)
    # ------------------------------------------------------------------
    time_left = deadline - time.time()
    if time_left > 6.0 and best_seq is not None:
        n = best_seq.size
        target = math.sqrt(2.0 * n)
        refined_seq, refined_score = _adam_descent(
            best_seq, target, time_left, lr=0.15, print_tag="final_refine"
        )
        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined_seq.copy()
            print(
                f"[{time.time():.2f}] FINAL REFINEMENT improved best to {best_score:.6f}"
            )

    # safety fallback
    if best_seq is None:
        n = 500
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=float)

    # final report
    print(
        f"[{time.time():.2f}] DONE – best score {best_score:.6f} with length {len(best_seq)}"
    )
    # the caller expects a Python list
    return best_seq.tolist()
