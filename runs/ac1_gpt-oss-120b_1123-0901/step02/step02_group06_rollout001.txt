"""
Propose a non‑negative step‑function with a low `evaluate_sequence` value.

Algorithm
---------
1. **Rich initial pool** – generate many simple shapes (uniform,
   geometric, arcsine‐type with a tunable singularity, Beta‑type,
   random Dirichlet draws and the previously found `height_sequence_1`).
   All candidates are normalised to sum‑1 (the objective is scale‑invariant)
   and the best one is kept as the starting point.

2. **Exact sub‑gradient descent** – the objective after normalisation is simply
   the maximum entry of the self‑convolution.  For the shift `k*` where this
   maximum is attained the sub‑gradient is
       g_i = 2·a_{k*‑i}   (if the index is inside the array, else 0).
   At each iteration we:
        * compute the full convolution,
        * locate `k*`,
        * build `g`,
        * take a step `a ← a – η·g`,
        * project back onto the probability simplex (∑a_i = 1, a_i ≥ 0).

   The step size follows a classic diminishing schedule `η_t = η0 / sqrt(t+1)`.
   Momentum is omitted for simplicity – the exact sub‑gradient already gives
   a strong descent direction.

3. **Stagnation handling** – if the best objective does not improve for a
   while we apply a small Gaussian jitter to escape flat plateaus and reset
   the step size.

4. **Termination** – the loop runs until the allocated time budget (≈ 1000 s)
   is exhausted; the best sequence seen is returned.

The method works with a larger problem size (`n≈4000`) than the reference
implementation, uses the exact sub‑gradient (rather than a top‑K approximation)
and performs a cheap exhaustive search over a one‑parameter family of
arcsine‑type shapes, which helps to start much closer to the optimum.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x ≥ 0, sum(x)=1}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all non‑positive – return uniform simplex point
        return np.full_like(v, 1.0 / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _make_arcsine_family(n: int, delta: float) -> np.ndarray:
    """
    Return a symmetric arcsine‑type distribution with exponent `delta`:

        a_i ∝ ((i+½)(n‑i‑½))^{-delta}

    For delta=0 we obtain the uniform distribution,
    for delta=0.5 the classic arcsine law.
    """
    i = np.arange(n, dtype=np.float64)
    # add 0.5 offset to avoid singularities at exact borders when delta>0
    base = (i + 0.5) * (n - i - 0.5)
    # guard against zero (should never happen)
    seq = np.power(base, -delta, where=base > 0, out=np.ones_like(base))
    seq_sum = seq.sum()
    if seq_sum == 0:
        return np.full(n, 1.0 / n)
    return seq / seq_sum


# ----------------------------------------------------------------------
def _initial_candidates(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Collect a diverse set of normalized starting sequences."""
    cand = []

    # 1. uniform
    cand.append(np.full(n, 1.0 / n))

    # 2. geometric decays (different ratios)
    for factor in [0.8, 1.0, 1.5, 2.0, 2.5, 3.0]:
        r = 1.0 - factor / n
        if r <= 0:
            continue
        g = np.power(r, np.arange(n, dtype=np.float64))
        cand.append(g / g.sum())

    # 3. arcsine family with a sweep over the singularity exponent
    for delta in np.linspace(0.0, 1.0, 11):        # 0.,0.1,…,1.0
        cand.append(_make_arcsine_family(n, delta))

    # 4. Beta‑type PDFs (symmetrical, shape parameter a)
    x = (np.arange(n) + 0.5) / n
    for a in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0]:
        beta = np.power(x, a - 1) * np.power(1 - x, a - 1)
        beta_sum = beta.sum()
        if beta_sum > 0:
            cand.append(beta / beta_sum)

    # 5. a few random Dirichlet draws
    for _ in range(5):
        rand = rng.random(n)
        cand.append(rand / rand.sum())

    # 6. possibly the previously discovered best sequence
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.ndim == 1 and len(arr) == n:
                cand.append(arr / arr.sum())
        except Exception:
            pass

    return cand


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Return a list of non‑negative heights (probability simplex) that minimizes
    the evaluation function within the allotted time.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.05   # tiny safety margin

    # --------------------------------------------------------------
    # 1. problem size – a few thousand entries give a good trade‑off
    n = 4000

    # --------------------------------------------------------------
    # 2. generate initial pool and pick the best start
    best_seq = None
    best_score = float("inf")
    for seq in _initial_candidates(n, rng):
        # the evaluation is homogeneous → we can use the simplified formula
        conv = np.convolve(seq, seq)
        max_b = conv.max()
        score = 2.0 * n * max_b / (seq.sum() ** 2)   # sum should be 1
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
    print(f"[Init] best score after seeding = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3. exact sub‑gradient descent on the max‑convolution
    seq = best_seq.copy()
    curr_score = best_score
    it = 0
    no_improve = 0
    # step schedule parameters
    eta0 = 0.25          # initial learning rate
    # optional momentum (commented out – not needed for exact sub‑gradient)
    # momentum = np.zeros_like(seq)

    while time.time() < deadline:
        # full self‑convolution
        conv = np.convolve(seq, seq)
        max_b = conv.max()
        k_max = int(conv.argmax())

        # construct exact sub‑gradient of the maximal entry
        grad = np.zeros_like(seq)
        left = max(0, k_max - (n - 1))
        right = min(n - 1, k_max)
        # indices that actually contribute
        idx = np.arange(left, right + 1, dtype=np.int64)
        grad[idx] = 2.0 * seq[k_max - idx]

        # diminishing step size (η_t = η0 / sqrt(t+1))
        eta = eta0 / np.sqrt(it + 1)

        # gradient step and projection onto simplex
        new_seq = seq - eta * grad
        new_seq = _project_onto_simplex(new_seq)

        # evaluate the new point (cheap internal formula)
        new_conv = np.convolve(new_seq, new_seq)
        new_max = new_conv.max()
        new_score = 2.0 * n * new_max / (new_seq.sum() ** 2)

        # keep the best ever seen
        if new_score < best_score - 1e-12:
            best_score = new_score
            best_seq = new_seq.copy()
            print(f"[Iter {it:06d}] new best = {best_score:.6f}")
            no_improve = 0
        else:
            no_improve += 1

        # accept the move unconditionally – sub‑gradient descent works this way
        seq = new_seq
        curr_score = new_score
        it += 1

        # occasional jitter to escape plateaus
        if no_improve > 25000:
            jitter = rng.normal(scale=0.003, size=n)
            seq = seq + jitter
            seq = np.clip(seq, 0.0, None)
            if seq.sum() == 0:
                seq = np.full(n, 1.0 / n)
            else:
                seq = seq / seq.sum()
            # reset counters and step schedule
            no_improve = 0
            it = max(it, 1)          # keep denominator non‑zero
            print(f"[Iter {it:06d}] stagnation – random jitter applied")

    print(f"Finished. Best objective = {best_score:.6f} (length {n})")
    return best_seq.tolist()
