"""
Hybrid optimisation for the step‑function inequality.
----------------------------------------------------
1. Initialise a candidate sequence with a short parametric sweep
   (geometric, power‑law, beta‑like families and random Dirichlet draws).
   If a previous best `height_sequence_1` is present we re‑sample it to the
   chosen length and use it as a seed.

2. Refine the seed by minimising a smooth approximation of the
   max‑convolution using projected gradient descent on the probability
   simplex.  The smooth max is
        S(a) = (1/β)·log Σ_k exp(β·(a*a)[k])
   whose gradient is
        ∇S = 2·corr(w, a)      with  w_k ∝ exp(β·(a*a)[k]).

   The correlation `corr` is obtained by a single FFT‑based convolution.
   A back‑tracking line‑search together with a projection onto the simplex
   guarantees feasibility.  Occasionally a tiny random mass‑transfer move
   is performed to escape plateaux.

3. The best sequence encountered (lowest value of
   `evaluate_sequence`) is returned.  The whole routine respects a
   1000 s time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
def _project_onto_simplex(v):
    """Project a vector onto the probability simplex {x≥0, Σx=1}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:          # all non‑positive -> uniform
        return np.full_like(v, 1.0 / len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT helpers
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _self_convolve(seq: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using real FFT."""
    n = seq.shape[0]
    m = 2 * n - 1
    nfft = _next_pow2(m)
    A = np.fft.rfft(seq, n=nfft)
    conv = np.fft.irfft(A * A, n=nfft)[:m]
    return conv


# ----------------------------------------------------------------------
# Objective used locally (faster than evaluate_sequence)
def _objective(seq: np.ndarray) -> float:
    """2·n·max(a*a) for a non‑negative sequence with Σa = 1."""
    sum_a = seq.sum()
    if sum_a < 0.01:
        return float("inf")
    conv = _self_convolve(seq)
    max_b = conv.max()
    return float(2.0 * len(seq) * max_b)


# ----------------------------------------------------------------------
# Parametric families (useful for seeding)
def _geom_seq(n: int, alpha: float) -> np.ndarray | None:
    """Geometric progression a_i ∝ r^i with r = 1 - α/n."""
    eps = alpha / n
    r = 1.0 - eps
    if r <= 0.0:
        return None
    powers = r ** np.arange(n, dtype=np.float64)
    seq = powers / powers.sum()
    return seq


def _powerlaw_seq(n: int, p: float) -> np.ndarray:
    """Power‑law a_i ∝ (i+1)^{-p}."""
    seq = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    return seq / seq.sum()


def _beta_seq(n: int, a: float, b: float) -> np.ndarray:
    """Beta‑like shape a_i ∝ (i+1)^{a-1} (n-i)^{b-1}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    seq = (i ** (a - 1.0)) * ((n + 1 - i) ** (b - 1.0))
    seq[seq < 0] = 0.0
    return seq / seq.sum()


def _random_simplex(n: int) -> np.ndarray:
    """Draw a random point on the simplex (Dirichlet(1,…,1))."""
    x = np.random.exponential(1.0, size=n)
    return x / x.sum()


# ----------------------------------------------------------------------
# Correlation of sequence with a weight vector (valid region, length n)
def _corr_weights(seq: np.ndarray, weights: np.ndarray) -> np.ndarray:
    """
    Compute (valid) correlation of `weights` (length 2n‑1) with `seq`
    i.e.   c_i = Σ_j weights[i + j] * seq[j]   for i = 0,…,n‑1.
    """
    n = seq.shape[0]
    rev = seq[::-1]                     # reversed sequence
    L = _next_pow2(len(weights) + n - 1)
    W = np.fft.rfft(weights, n=L)
    R = np.fft.rfft(rev, n=L)
    full = np.fft.irfft(W * R, n=L)    # length L ≥ 3n‑2
    start = n - 1
    return full[start:start + n]


# ----------------------------------------------------------------------
# Gradient of the smooth‑max (log‑sum‑exp) of the convolution
def _smooth_max_grad(seq: np.ndarray, beta: float):
    """
    Return gradient of  S(a)= (1/β)·log Σ exp(β·(a*a))   and also the
    convolution and its current maximum (for logging).
    """
    conv = _self_convolve(seq)                 # length 2n‑1
    maxc = conv.max()
    # stable soft‑max on the convolution values
    exps = np.exp(beta * (conv - maxc))
    weights = exps / exps.sum()
    grad = 2.0 * _corr_weights(seq, weights)   # ∇S = 2·corr(weights, a)
    return grad, conv, maxc


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Generate and locally optimise a non‑negative sequence that minimises
    `evaluate_sequence`.  The routine respects the 1000 s time budget and
    returns the best list found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.2      # safety margin

    # --------------------------------------------------------------
    # 1. Choose a moderate length – large enough to give freedom,
    #    small enough for fast FFTs.
    n = 2500

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # 2. If a previous global solution exists, re‑sample it to length n.
    old = globals().get("height_sequence_1", None)
    if isinstance(old, (list, tuple, np.ndarray)):
        try:
            init = np.asarray(old, dtype=np.float64)
            init = np.clip(init, 0.0, 1000.0)
            if init.sum() > 0.0:
                if init.size != n:
                    # linear interpolation to the new size
                    x_old = np.linspace(0, 1, init.size)
                    x_new = np.linspace(0, 1, n)
                    init = np.interp(x_new, x_old, init)
                init = init / init.sum()
                score = _objective(init)
                best_seq = init.copy()
                best_score = score
                print(f"[Init] global seed objective={score:.6f}")
        except Exception:
            pass

    # --------------------------------------------------------------
    # 3. Systematic parametric sweep – keep the best candidate.
    #    (These loops terminate early if the budget expires.)

    # geometric family
    for alpha in np.linspace(0.5, 3.0, 12):

    # --------------------------------------------------------------
    print(f"Final best objective = {best_score:.6f} (length {n})")
