def _fast_score(a):
    a = np.asarray(a, dtype=float)
    n = a.shape[0]
    # Compute full convolution using FFT (real)
    size = 1 << ((2*n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[:2*n-1]
    max_conv = conv.max()
    s = a.sum()
    if s < 0.01:
        return float('inf')
    return 2.0 * n * max_conv / (s**2)
```

def subgrad_optimize(a, target, deadline, best_seq, best_score):
    n = a.shape[0]
    m = np.zeros_like(a)
    beta = 0.9
    base_step = 0.5
    iteration = 0
    while time.time() < deadline:
        conv = _full_convolution_fft(a)  # shape 2n-1
        max_c = conv.max()
        # threshold selection
        thresh = max_c * 0.99
        active = np.where(conv >= thresh)[0]
        grad = np.zeros_like(a)
        for k in active:
            low = max(0, k-(n-1))
            high = min(n-1, k)
            idx = np.arange(low, high+1)
            partner = k - idx
            # factor 2 for all except when idx == partner to avoid double
            # just use 2*a[partner] as before (approx)
            grad[idx] += 2.0 * a[partner]
        # direction = -grad
        direction = -grad
        # momentum update
        m = beta * m + (1.0 - beta) * direction
        # step size
        step = base_step / np.sqrt(iteration + 1)
        max_abs = np.max(np.abs(m))
        if max_abs == 0:
            max_abs = 1.0
        eta = step / max_abs
        a_new = a + eta * m
        a_new = _project_onto_simplex(a_new, target)
        # evaluate periodic update
        if iteration % 5 == 0:
            score = _score_seq(a_new)
            if score < best_score:
                best_score = score
                best_seq = a_new.copy()
                # Optionally print
                print(f"[{time.time():.2f}] Subgrad new best = {best_score:.6f} (len={n})")
                # if best_score achieves target bound break
                if best_score < 1.503:
                    return best_seq, best_score
        a = a_new
        iteration += 1
        if time.time() >= deadline:
            break
    return best_seq, best_score

"""
Simple but aggressive black‑box optimiser for the step‑function bound.

The optimiser works in four stages:

1.  A cheap sweep over a few parametric families (exponential,
    reversed exponential, linear / quadratic decay, triangular,
    power‑law, Gaussian‑like) for several lengths.  Every candidate is
    normalised onto the simplex Σ a_i = √(2 n) and evaluated with a fast
    FFT based score (identical to evaluate_sequence).

2.  A sub‑gradient descent on the infinity‑norm of the
    autocorrelation.  The descent uses momentum, a diminishing step
    size and aggregates sub‑gradients from all entries whose convolution
    value lies within 2 % of the current maximum – this yields a smoother
    direction than using a single active constraint.

3.  Simulated‑annealing style random mass‑transfer moves.  A source
    index is chosen with probability proportional to its current height,
    a small amount of mass is transferred to a random destination and the
    new sequence is accepted either if it improves the score or with a
    probability exp(−Δ/T) where the temperature drops linearly to zero.

4.  If SciPy is available a single linear‑programming refinement
    (the “LP‑step’’ from the SOTA description) is performed on the final
    sequence.

All helper functions are defined at module level – there are no closures
or lambdas.  The routine respects the supplied time budget and returns
the best sequence discovered.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the simplex {x ≥ 0, Σx = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Projection of a vector onto the simplex Σx = s, x ≥ 0."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # sort descending
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based full convolution a*a (real input)
# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Return the full (2·n‑1) convolution a*a using a power‑of‑two FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())   # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Fast evaluation – identical to evaluate_sequence (but without the safety checks)
# ----------------------------------------------------------------------
def _fast_score(a: np.ndarray) -> float:
    """Compute 2·n·max(conv(a,a))/sum(a)^2 for a non‑negative sequence."""
    a = np.asarray(a, dtype=float)
    sum_a = a.sum()
    if sum_a < 0.01:                     # protect against division by zero
        return float("inf")
    n = a.shape[0]
    conv = _full_convolution_fft(a)
    max_conv = conv.max()
    return 2.0 * n * max_conv / (sum_a * sum_a)


# ----------------------------------------------------------------------
# Sub‑gradient descent (active set = all near‑max entries)
# ----------------------------------------------------------------------
def _subgrad_optimize(a: np.ndarray,
                      target: float,
                      deadline: float,
                      best_seq: np.ndarray,
                      best_score: float) -> tuple[np.ndarray, float]:
    """Momentum sub‑gradient descent on max(a*a)."""
    n = a.shape[0]
    m = np.zeros_like(a)
    beta = 0.85               # momentum decay
    base_step = 0.30
    it = 0
    while time.time() < deadline:
        conv = _full_convolution_fft(a)
        max_val = conv.max()
        # treat all entries within 2 % of the maximum as active
        thresh = max_val * 0.98
        active = np.where(conv >= thresh)[0]

        grad = np.zeros_like(a)
        for k in active:
            low = max(0, k - (n - 1))
            high = min(n - 1, k)
            idx = np.arange(low, high + 1, dtype=int)
            partner = k - idx
            grad[idx] += 2.0 * a[partner]          # approximate sub‑gradient

        direction = -grad
        m = beta * m + (1.0 - beta) * direction
        step = base_step / math.sqrt(it + 1)
        max_abs = np.max(np.abs(m))
        if max_abs == 0.0:
            max_abs = 1.0
        eta = step / max_abs
        a = a + eta * m
        a = _project_onto_simplex(a, target)

        if it % 10 == 0:               # periodic cheap evaluation
            cur_score = _fast_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time():.2f}] Sub‑grad new best = {best_score:.6f} (len={n})")
                if best_score < 1.503:
                    return best_seq, best_score
        it += 1
    return best_seq, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing style random mass‑transfer moves
# ----------------------------------------------------------------------
def _annealing_optimize(a: np.ndarray,
                        target: float,
                        deadline: float,
                        best_seq: np.ndarray,
                        best_score: float) -> tuple[np.ndarray, float]:
    """Random pairwise mass transfers with a cooling schedule."""
    n = a.shape[0]
    cur = a.copy()
    cur_score = _fast_score(cur)

    start = time.time()
    total = deadline - start
    if total <= 0:
        return best_seq, best_score

    # temperature and step size start relatively large, then cool linearly
    T0 = max(0.015, (best_score - 1.45))       # ensure a positive start
    sigma0 = max(target * 0.02, 0.01)

    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / total
        T = T0 * (1.0 - frac) + 1e-6
        sigma = sigma0 * (1.0 - frac) + 1e-6

        # source index – choose proportionally to current height (more mass, more likely)
        prob = cur / cur.sum()
        src = np.random.choice(n, p=prob)
        dst = np.random.randint(0, n)
        if src == dst:
            continue

        max_transfer = min(cur[src], sigma)
        if max_transfer <= 0:
            continue
        delta = np.random.rand() * max_transfer          # amount to move
        cand = cur.copy()
        cand[src] -= delta
        cand[dst] += delta     # sum unchanged, non‑negativity guaranteed

        cand_score = _fast_score(cand)
        # accept if better, otherwise with Boltzmann probability
        if cand_score < cur_score or np.random.rand() < math.exp(-(cand_score - cur_score) / T):
            cur = cand
            cur_score = cand_score

        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            print(f"[{time.time():.2f}] Anneal new best = {best_score:.6f} (len={n})")
            if best_score < 1.503:
                break
    return best_seq, best_score


# ----------------------------------------------------------------------
# Linear‑programming refinement (optional, requires SciPy)
# ----------------------------------------------------------------------
def _lp_refine(a: np.ndarray,
               target: float,
               deadline: float,
               best_seq: np.ndarray,
               best_score: float) -> tuple[np.ndarray, float]:
    """Perform the SOTA LP step once (or a few times) if SciPy is available."""
    try:
        from scipy.optimize import linprog
    except Exception:
        return best_seq, best_score

    n = a.shape[0]
    rows = 2 * n - 1
    # build the convolution constraint matrix A once – it depends on a only
    A = np.zeros((rows, n), dtype=float)
    for k in range(rows):
        low = max(0, k - (n - 1))
        high = min(n - 1, k)
        inds = np.arange(low, high + 1, dtype=int)
        A[k, inds] = a[k - inds]                # (a * g)_k ≤ M

    while time.time() < deadline:
        conv = _full_convolution_fft(a)
        M = conv.max()
        res = linprog(c=-np.ones(n),            # maximise Σ g_j
                      A_ub=A,
                      b_ub=np.full(rows, M),
                      bounds=(0, None),
                      method="highs",
                      options={"presolve": True})
        if not res.success:
            break
        g = np.asarray(res.x, dtype=float)
        s = g.sum()
        if s <= 0:
            break
        g = g * (target / s)                     # re‑scale to simplex
        t = 0.2                                   # modest convex blend
        a = (1 - t) * a + t * g
        # cheap evaluation
        score = _fast_score(a)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time():.2f}] LP refinement = {best_score:.6f} (len={n})")
            if best_score < 1.503:
                break
        # rebuild A for the new a (required for next LP)
        for k in range(rows):
            low = max(0, k - (n - 1))
            high = min(n - 1, k)
            inds = np.arange(low, high + 1, dtype=int)
            A[k, inds] = a[k - inds]
    return best_seq, best_score


# ----------------------------------------------------------------------
# Main routine required by the evaluator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs) -> list:
    """
    Search for a non‑negative height sequence that minimises the evaluation
    function.  The routine respects the given time budget and returns the
    best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    random.seed(seed)
    deadline = time.time() + budget_s - 0.5     # small safety margin

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # 0.  Seed with a previously discovered construction if available
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        init = globals()["height_sequence_1"]
        if isinstance(init, (list, tuple, np.ndarray)):
            seq0 = np.array(init, dtype=float)
            if seq0.size > 0:
                n0 = seq0.shape[0]
                target0 = math.sqrt(2.0 * n0)
                seq0 = _project_onto_simplex(seq0, target0)
                score0 = _fast_score(seq0)
                best_seq = seq0.copy()
                best_score = score0
                print(f"[{time.time():.2f}] Seeded from height_sequence_1 = {best_score:.6f} (len={n0})")

    # --------------------------------------------------------------
    # 1.  Parametric sweep (very cheap)
    # --------------------------------------------------------------
    lengths = list(range(200, 2001, 200)) + [1200, 1500]
    for n in lengths:
        if time.time() >= deadline:
            break
        target = math.sqrt(2.0 * n)

        # ----- exponential decay -------------------------------------------------
        for alpha in np.logspace(-4, -0.2, num=12):
            a = np.exp(-alpha * np.arange(n, dtype=float))
            if a.sum() == 0:
                continue
            a = a / a.sum() * target
            score = _fast_score(a)
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"[{time.time():.2f}] Exp len={n} α={alpha:.5f} → {best_score:.6f}")

        # ----- reversed exponential ------------------------------------------------
        for alpha in np.logspace(-4, -0.2, num=12):
            a = np.exp(-alpha * np.arange(n - 1, -1, -1, dtype=float))
            if a.sum() == 0:
                continue
            a = a / a.sum() * target
            score = _fast_score(a)
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"[{time.time():.2f}] Rev‑exp len={n} α={alpha:.5f} → {best_score:.6f}")

        # ----- linear descending --------------------------------------------------
        for beta in np.linspace(0.0, 2.0 / max(1, n - 1), num=12):
            a = np.maximum(0.0, 1.0 - beta * np.arange(n, dtype=float))
            if a.sum() == 0:
                continue
            a = a / a.sum() * target
            score = _fast_score(a)
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"[{time.time():.2f}] Lin‑desc len={n} β={beta:.5f} → {best_score:.6f}")

        # ----- quadratic descending -----------------------------------------------
        for beta in np.linspace(0.0, 1.0 / max(1, n ** 2), num=12):
            a = np.maximum(0.0, 1.0 - beta * (np.arange(n, dtype=float) ** 2))
            if a.sum() == 0:
                continue
            a = a / a.sum() * target
            score = _fast_score(a)
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"[{time.time():.2f}] Quad‑desc len={n} β={beta:.7f} → {best_score:.6f}")

        # ----- triangular ---------------------------------------------------------
        a = np.minimum(np.arange(n, dtype=float) + 1.0, n - np.arange(n, dtype=float))
        if a.sum() != 0:
            a = a / a.sum() * target
            score = _fast_score(a)
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"[{time.time():.2f}] Triangular len={n} → {best_score:.6f}")

        # ----- power‑law ---------------------------------------------------------
        for p in np.linspace(0.1, 2.0, num=12):
            a = (np.arange(1, n + 1, dtype=float) ** (-p))
            if a.sum() == 0:
                continue
            a = a / a.sum() * target
            score = _fast_score(a)
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"[{time.time():.2f}] Power‑law len={n} p={p:.3f} → {best_score:.6f}")

        # ----- Gaussian‑like (centered at 0, decreasing) ------------------------
        for sigma_frac in np.linspace(0.15, 0.45, num=5):
            sigma = sigma_frac * n
            i = np.arange(n, dtype=float)
            a = np.exp(- (i ** 2) / (2.0 * sigma ** 2))
            if a.sum() == 0:
                continue
            a = a / a.sum() * target
            score = _fast_score(a)
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"[{time.time():.2f}] Gaussian len={n} σ={sigma_frac:.3f} → {best_score:.6f}")

        # ----- mixture of two exponentials --------------------------------------
        for _ in range(6):
            alpha1 = 10 ** np.random.uniform(-4, -0.2)
            alpha2 = 10 ** np.random.uniform(-4, -0.2)
            w = np.random.rand()
            a = w * np.exp(-alpha1 * np.arange(n, dtype=float)) + \
                (1 - w) * np.exp(-alpha2 * np.arange(n, dtype=float))
            if a.sum() == 0:
                continue
            a = a / a.sum() * target
            score = _fast_score(a)
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"[{time.time():.2f}] Mix‑exp len={n} w={w:.2f} → {best_score:.6f}")

        if time.time() >= deadline:
            break

    # --------------------------------------------------------------
    # 2.  Sub‑gradient refinement (momentum, active set)
    # --------------------------------------------------------------
    if best_seq is not None:
        remaining = deadline - time.time()
        subgrad_deadline = time.time() + 0.45 * remaining
        best_seq, best_score = _subgrad_optimize(best_seq,
                                                 math.sqrt(2.0 * len(best_seq)),
                                                 subgrad_deadline,
                                                 best_seq,
                                                 best_score)

    # --------------------------------------------------------------
    # 3.  Simulated annealing (mass‑transfer moves)
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        remaining = deadline - time.time()
        anneal_deadline = time.time() + 0.40 * remaining
        best_seq, best_score = _annealing_optimize(best_seq,
                                                   math.sqrt(2.0 * len(best_seq)),
                                                   anneal_deadline,
                                                   best_seq,
                                                   best_score)

    # --------------------------------------------------------------
    # 4.  Optional LP refinement (if SciPy is present)
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        remaining = deadline - time.time()
        lp_deadline = time.time() + 0.10 * remaining
        best_seq, best_score = _lp_refine(best_seq,
                                          math.sqrt(2.0 * len(best_seq)),
                                          lp_deadline,
                                          best_seq,
                                          best_score)

    # --------------------------------------------------------------
    # 5.  Simple length‑adjustment pass (try shortening the sequence)
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        n0 = len(best_seq)
        for cut in range(20, int(0.2 * n0) + 1, 20):
            n = n0 - cut
            if n < 50:
                break
            cand = best_seq[:n].copy()
            target = math.sqrt(2.0 * n)
            cand = _project_onto_simplex(cand, target)
            score = _fast_score(cand)
            if score < best_score:
                best_score = score
                best_seq = cand.copy()
                print(f"[{time.time():.2f}] Trimmed to len={n} → {best_score:.6f}")
                if best_score < 1.503:
                    break

    # --------------------------------------------------------------
    # 6.  Fallback (uniform) – should never be needed
    # --------------------------------------------------------------
    if best_seq is None:
        n = 1000
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=float)

    # Return as plain Python list
    return best_seq.tolist()
