"""
Propose a step‑function (list of non‑negative heights) with a low
evaluation score.  The algorithm mixes several strategies:

*  A fast projected gradient descent on the sub‑gradient of the current
   maximal convolution entry, with simple momentum and an adaptive
   step‑size.
*  Occasionally the gradient is enriched by a “push‑pull” term that also
   tries to raise the minimal convolution entry – this flattens the
   autocorrelation and often reduces the peak.
*  Random mutations (Gaussian noise + simplex projection) are injected
   to escape shallow local minima.
*  Multi‑scale search: the current best sequence is resized to new
   lengths (both larger and smaller) and refined again.  The length is
   allowed to vary from 400 up to 4000 – larger lengths give a finer
   granularity for distributing the fixed total mass.
*  Many short runs with different random seeds are performed until the
   time budget (default 1000 s) expires.  The best sequence found is
   returned.

All helper functions are defined at the top level (no closures).
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Simple Euclidean projection onto the simplex {x ≥ 0, Σ x = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex of non‑negative vectors with sum s."""
    n = v.shape[0]
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    # Sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Gradient of the convolution entry at a given index
# ----------------------------------------------------------------------
def _grad_at_index(a: np.ndarray, idx: int) -> np.ndarray:
    """Return the gradient of conv(a,a)[idx] w.r.t. a (2*a[partner] on the overlap)."""
    n = a.shape[0]
    low = max(0, idx - (n - 1))
    high = min(n - 1, idx)
    idxs = np.arange(low, high + 1, dtype=int)
    partner = idx - idxs
    grad = np.zeros_like(a)
    grad[idxs] = 2.0 * a[partner]
    return grad


# ----------------------------------------------------------------------
# Max convolution entry and its gradient (used frequently)
# ----------------------------------------------------------------------
def _max_conv_and_grad(a: np.ndarray):
    """Return (max_val, grad_at_max_idx, max_idx, full_conv)."""
    conv = np.convolve(a, a)
    max_idx = int(np.argmax(conv))
    max_val = conv[max_idx]
    grad = _grad_at_index(a, max_idx)
    return max_val, grad, max_idx, conv


# ----------------------------------------------------------------------
# Very small mutation helper (adds Gaussian noise and re‑projects)
# ----------------------------------------------------------------------
def _mutate(a: np.ndarray, target_sum: float, sigma: float) -> np.ndarray:
    """Return a mutated copy of a (Gaussian noise, then simplex projection)."""
    b = a + np.random.randn(a.size) * sigma
    b = np.maximum(b, 0.0)
    b = _project_onto_simplex(b, target_sum)
    return b


# ----------------------------------------------------------------------
# Local projected‑gradient descent with momentum and optional push‑pull
# ----------------------------------------------------------------------
def _local_descent_improved(
    init: np.ndarray,
    target_sum: float,
    time_limit: float,
    base_step: float = 0.15,
    momentum: float = 0.85,
    print_tag: str = "",
) -> tuple[np.ndarray, float]:
    """
    Run a gradient descent starting from ``init``.
    Returns the best sequence seen and its evaluation score.
    """
    a = _project_onto_simplex(init.astype(float), target_sum)

    # initial evaluation
    try:
        best_score = float(evaluate_sequence(a.tolist()))
    except Exception:
        best_score = math.inf
    best_seq = a.copy()

    v = np.zeros_like(a)          # momentum accumulator
    step = base_step
    t0 = time.time()
    it = 0

    while time.time() < t0 + time_limit:
        # ------------------------------------------------------------------
        # compute convolution data
        # ------------------------------------------------------------------
        conv = np.convolve(a, a)
        max_idx = int(np.argmax(conv))
        max_val = conv[max_idx]
        grad_max = _grad_at_index(a, max_idx)

        # Occasionally also consider the minimal entry (push‑pull)
        if np.random.rand() < 0.35:
            min_idx = int(np.argmin(conv))
            grad_min = _grad_at_index(a, min_idx)
            # combine: reduce max, increase min
            combined = -grad_max + 0.17 * grad_min
        else:
            combined = -grad_max

        # Zero‑mean direction (preserves the sum after projection)
        d = combined - np.mean(combined)

        # Momentum update
        v = momentum * v + (1.0 - momentum) * d

        # Adaptive step size; scale so that typical move size is comparable
        max_abs = np.max(np.abs(v))
        if max_abs < 1e-12:
            max_abs = 1.0
        eta = step * max_val / max_abs

        a_new = a + eta * v
        a_new = _project_onto_simplex(a_new, target_sum)

        # ------------------------------------------------------------------
        # evaluate
        # ------------------------------------------------------------------
        try:
            score = float(evaluate_sequence(a_new.tolist()))
        except Exception:
            score = math.inf

        # accept improvement
        if score < best_score:
            best_score = score
            best_seq = a_new.copy()
            # modestly increase step after a good move
            step = min(step * 1.07, 2.0)
            print(
                f"[{time.time():.2f}] {print_tag} new local best = {score:.6f} (n={len(a)})"
            )
        else:
            # shrink step when no progress
            step = max(step * 0.94, 1e-4)

        a = a_new

        # occasional random shake‑up to avoid stagnation
        if np.random.rand() < 0.015:
            sigma = 0.012 * target_sum / len(a)
            a = _mutate(a, target_sum, sigma)
            v = np.zeros_like(a)  # reset momentum after a shake

        it += 1

    return best_seq, best_score


# ----------------------------------------------------------------------
# Resize a sequence to a new length (interpolation / decimation)
# ----------------------------------------------------------------------
def _resize_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize seq to length new_n preserving non‑negativity (linear interpolation)."""
    old_n = seq.size
    if old_n == new_n:
        return seq.copy()
    if old_n > new_n:
        # down‑sample by picking equally spaced indices
        idx = np.linspace(0, old_n - 1, new_n).astype(int)
        return seq[idx]
    else:
        # up‑sample by linear interpolation
        x_old = np.arange(old_n)
        x_new = np.linspace(0, old_n - 1, new_n)
        return np.interp(x_new, x_old, seq)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a step‑function (list of non‑negative heights) that minimises
    the evaluation function.  The routine respects the given time budget.
    """
    np.random.seed(seed)
    random.seed(seed)
    deadline = time.time() + budget_s - 0.4  # keep a small safety margin

    # ------------------------------------------------------------------
    # Global best holder (maybe initialise from historic solution)
    # ------------------------------------------------------------------
    best_seq = None
    best_score = math.inf

    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(height_sequence_1, dtype=float)
            sc = float(evaluate_sequence(cand.tolist()))
            best_seq = cand.copy()
            best_score = sc
            print(f"[{time.time():.2f}] Started from historic seq, score={sc:.6f}")
        except Exception:
            best_seq = None
            best_score = math.inf

    # fallback: uniform distribution of moderate size
    if best_seq is None:
        n0 = 1500
        target0 = math.sqrt(2.0 * n0)
        best_seq = np.full(n0, target0 / n0, dtype=float)
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"[{time.time():.2f}] No historic seq – using uniform, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Helper to run a single optimisation episode
    # ------------------------------------------------------------------
    def _run_episode(n: int, init_vec: np.ndarray, remaining: float):
        """
        Run one optimisation episode (gradient descent + a few mutation bursts)
        and return (sequence, score).
        """
        target = math.sqrt(2.0 * n)

        # Allocate most of the remaining time to the main descent
        main_time = min(remaining * 0.65, 28.0)
        cand, sc = _local_descent_improved(
            init_vec, target, main_time, base_step=0.16, momentum=0.84,
            print_tag=f"n={n}"
        )

        # If time permits, perform a few short mutation‑refinement bursts
        elapsed = time.time()
        time_left = deadline - elapsed
        bursts = 2
        for _ in range(bursts):
            if time_left < 4.0:
                break
            sigma = 0.025 * target / n
            mutated = _mutate(cand, target, sigma)
            short_time = min(time_left * 0.35, 6.0)
            cand2, sc2 = _local_descent_improved(
                mutated, target, short_time,
                base_step=0.12, momentum=0.80,
                print_tag=f"burst n={n}"
            )
            if sc2 < sc:
                cand, sc = cand2, sc2
            time_left = deadline - time.time()

        return cand, sc

    # ------------------------------------------------------------------
    # Main search loop
    # ------------------------------------------------------------------
    while time.time() < deadline - 1.0:
        # Choose a length – bias towards current best but allow diversity
        if best_seq is not None and np.random.rand() < 0.68:
            n = best_seq.size
            # Occasionally stretch/shrink the length a bit
            if np.random.rand() < 0.27:
                n = int(n * np.random.uniform(0.88, 1.12))
                n = max(400, min(4200, n))
        else:
            n = np.random.randint(400, 4200)

        target_sum = math.sqrt(2.0 * n)

        # Initialise candidate: resized best or random
        if best_seq is not None:
            init = _resize_sequence(best_seq, n)
        else:
            init = np.random.rand(n)

        # Allocate a chunk of remaining time to this run
        time_left = deadline - time.time()
        if time_left < 8.0:
            break
        cand_seq, cand_score = _run_episode(n, init, time_left)

        if cand_seq is not None and cand_score < best_score:
            best_score = cand_score
            best_seq = cand_seq.copy()
            print(
                f"[{time.time():.2f}] *** GLOBAL NEW BEST = {best_score:.6f} (n={n})"
            )
            # Early exit if we have already beaten the target bound
            if best_score < 1.503:
                break

    # ------------------------------------------------------------------
    # Final safety fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1500
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=float)

    return best_seq.tolist()
