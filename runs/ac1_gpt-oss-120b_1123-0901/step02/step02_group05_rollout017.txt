def _smooth_descent(a_init, target_sum, beta_schedule, total_time, print_tag=""):
    a = _project_onto_simplex(a_init.copy(), target_sum)
    best_seq = a.copy()
    best_score = float(evaluate_sequence(a.tolist()))
    start = time.time()
    for beta in beta_schedule:
        # inner iterations allocate portion of time
        inner_time = total_time * 0.9 / len(beta_schedule) # roughly equal share, leftover for final refine
        it = 0
        step = 0.2 / (1 + beta)
        while time.time() - start < total_time and time.time() - start < start + inner_time:
            # compute grad
            conv = np.convolve(a, a)
            max_conv = conv.max()
            exp_vals = np.exp(beta * (conv - max_conv))
            w = exp_vals / exp_vals.sum()
            grad_full = np.convolve(w, a[::-1])
            grad = 2.0 * grad_full[len(a)-1:len(a)-1+len(a)]
            # gradient descent step
            # direction to move opposite gradient: a - eta * grad
            # adapt step magnitude:
            grad_norm = np.linalg.norm(grad)
            eta = step / (grad_norm + 1e-12)
            a_new = a - eta * grad
            # Project onto simplex (non-negative) to keep sum
            a_new = _project_onto_simplex(a_new, target_sum)
            a = a_new
            # evaluate occasionally
            if it % 5 == 0:
                score = float(evaluate_sequence(a.tolist()))
                if score < best_score:
                    best_score = score
                    best_seq = a.copy()
                    print(f"[{time.time():.2f}] {print_tag} beta={beta:.2f} iter={it} new best {score:.6f}")
            it += 1
            # optional random perturbation
            if np.random.rand() < 0.01:
                a = _mutate(a, target_sum, sigma=0.05 * target_sum/len(a))
                step *= 0.9
    # final local descent optionally
    return best_seq, best_score

"""
Propose a non‑negative step‑function (list of heights) that minimizes

    2 * n * max(conv(a, a)) / sum(a)**2

The optimizer works in three stages:

1. **Smooth max gradient descent** – we minimise a softmax
   approximation of the maximal convolution entry.
   The temperature (β) is increased gradually, turning the smooth
   problem into the original non‑smooth one while keeping the
   optimisation landscape friendly for gradient descent.

2. **Projected sub‑gradient refinement** – a short run of the classic
   sub‑gradient descent on the true maximal entry (as used in the
   reference implementation) cleans up the final result.

3. **Optional LP refinement** – if cvxpy is available we perform a few
   LP updates exactly as described in the paper sketch.

The algorithm keeps a global best, starts new runs from a resized copy
of the current best, from a Gaussian‑shaped guess or from a random
vector, and respects the 1000 s time budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simple Euclidean projection onto the simplex {x ≥ 0, Σ x = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex {x ≥ 0, Σ x = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Small mutation helper (Gaussian noise + projection)
# ----------------------------------------------------------------------
def _mutate(a: np.ndarray, target_sum: float, sigma: float) -> np.ndarray:
    """Gaussian perturbation followed by simplex projection."""
    b = a + np.random.randn(a.size) * sigma
    b = np.maximum(b, 0.0)
    b = _project_onto_simplex(b, target_sum)
    return b


# ----------------------------------------------------------------------
# Resize helper (linear interpolation / down‑sampling)
# ----------------------------------------------------------------------
def _resize_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize seq to length new_n preserving non‑negativity."""
    old_n = seq.size
    if old_n == new_n:
        return seq.copy()
    if old_n > new_n:
        idx = np.linspace(0, old_n - 1, new_n).astype(int)
        return seq[idx]
    else:
        x_old = np.arange(old_n)
        x_new = np.linspace(0, old_n - 1, new_n)
        return np.interp(x_new, x_old, seq)


# ----------------------------------------------------------------------
# Convolution utilities (numpy's convolve is sufficient)
# ----------------------------------------------------------------------
def _conv_full(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (length 2n‑1)."""
    return np.convolve(a, a)


def _grad_from_softmax_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute ∇_a Σ_i w_i (a * a)_i for a given weight vector w.
    The result is 2 * (w ⊛ rev(a))_valid where ⊛ denotes convolution.
    """
    # convolution of w (length 2n‑1) with flipped a (length n)
    grad_full = np.convolve(w, a[::-1])
    n = a.size
    # valid part: indices n‑1 … 2n‑2  (exactly n entries)
    grad_valid = grad_full[n - 1 : n - 1 + n]
    return 2.0 * grad_valid


# ----------------------------------------------------------------------
# Smooth‑max (softmax) gradient descent
# ----------------------------------------------------------------------
def _smooth_descent(
    init: np.ndarray,
    target_sum: float,
    time_limit: float,
    beta_schedule=None,
    base_step: float = 0.25,
    print_tag: str = "",
) -> tuple[np.ndarray, float]:
    """
    Gradient descent on the softmax approximation of max(conv).
    Returns the best sequence found and its exact evaluation score.
    """
    if beta_schedule is None:
        beta_schedule = [0.2, 0.5, 1.0, 2.0, 4.0, 8.0]

    a = _project_onto_simplex(init.astype(float), target_sum)

    # initial score
    try:
        best_score = float(evaluate_sequence(a.tolist()))
    except Exception:  # safety net
        best_score = math.inf
    best_seq = a.copy()

    start = time.time()
    it = 0
    for beta in beta_schedule:
        # a simple schedule for step size – smaller for larger beta
        step = base_step / (1.0 + beta)

        # inner loop; break if total budget exhausted
        while time.time() - start < time_limit:
            # full convolution a*a
            conv = np.convolve(a, a)
            max_conv = conv.max()

            # softmax weights (numerically stable)
            exp_vals = np.exp(beta * (conv - max_conv))
            w = exp_vals / exp_vals.sum()

            # gradient of the soft‑max objective
            grad = _grad_from_softmax_weights(a, w)

            # normalise step to keep updates moderate
            grad_norm = np.linalg.norm(grad)
            if grad_norm == 0:
                eta = 0.0
            else:
                eta = step / grad_norm

            a_new = a - eta * grad
            a_new = _project_onto_simplex(a_new, target_sum)

            # occasional random shake‑up to escape shallow traps
            if np.random.rand() < 0.01:
                a_new = _mutate(a_new, target_sum, sigma=0.04 * target_sum / len(a))
                step *= 0.9

            a = a_new

            # evaluate every few iterations (cheap enough)
            if it % 5 == 0:
                try:
                    score = float(evaluate_sequence(a.tolist()))
                except Exception:
                    score = math.inf
                if score < best_score:
                    best_score = score
                    best_seq = a.copy()
                    print(
                        f"[{time.time():.2f}] {print_tag} β={beta:.2f} it={it} new best={score:.6f}"
                    )
            it += 1

            # cut the inner loop if we spent too much on this β
            if time.time() - start > time_limit:
                break

    return best_seq, best_score


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true maximal convolution entry
# (same as the reference implementation)
# ----------------------------------------------------------------------
def _max_conv_and_grad(a: np.ndarray):
    """Maximum convolution value, its gradient (sub‑gradient) and full conv."""
    n = a.shape[0]
    conv = np.convolve(a, a)
    max_idx = int(np.argmax(conv))
    max_val = conv[max_idx]

    # gradient only on overlapping region
    low = max(0, max_idx - (n - 1))
    high = min(n - 1, max_idx)
    idx = np.arange(low, high + 1, dtype=int)
    partner = max_idx - idx
    grad = np.zeros_like(a)
    grad[idx] = 2.0 * a[partner]
    return max_val, grad, max_idx, conv


def _local_descent(
    init: np.ndarray,
    target_sum: float,
    time_limit: float,
    base_step: float = 0.07,
    print_tag: str = "",
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on max(conv)."""
    a = _project_onto_simplex(init.astype(float), target_sum)

    try:
        best_score = float(evaluate_sequence(a.tolist()))
    except Exception:
        best_score = math.inf
    best_seq = a.copy()

    step = base_step
    it = 0
    t0 = time.time()

    while time.time() - t0 < time_limit:
        max_val, grad, _, _ = _max_conv_and_grad(a)

        # make step direction sum‑zero; this respects the simplex after projection
        mean_g = np.mean(grad)
        d = -grad + mean_g

        # step size (scale with magnitude of max entry)
        max_abs = np.max(np.abs(d)) if np.max(np.abs(d)) > 0 else 1.0
        eta = step * max_val / max_abs

        a_new = a + eta * d
        a_new = _project_onto_simplex(a_new, target_sum)

        # occasional shake‑up
        if np.random.rand() < 0.02:
            sigma = 0.01 * target_sum / len(a)
            a_new = _mutate(a, target_sum, sigma)
            step *= 0.9

        a = a_new

        if it % 7 == 0:
            try:
                score = float(evaluate_sequence(a.tolist()))
            except Exception:
                score = math.inf
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(
                    f"[{time.time():.2f}] {print_tag} sub‑grad it={it} new best={score:.6f}"
                )
        # simple step decay
        if it % 200 == 0 and it > 0:
            step *= 0.9
        it += 1

    return best_seq, best_score


# ----------------------------------------------------------------------
# Optional LP refinement (same as the reference code)
# ----------------------------------------------------------------------
try:
    import cvxpy as cp

    _CVX_AVAILABLE = True
except Exception:  # pragma: no cover
    _CVX_AVAILABLE = False


def _solve_lp_for_g(f: np.ndarray, M: float):
    """Solve the LP: maximise Σ g_j s.t. g ≥ 0 and f*g ≤ M."""
    if not _CVX_AVAILABLE:
        return None
    n = f.shape[0]
    g = cp.Variable(n, nonneg=True)

    constraints = []
    for k in range(2 * n - 1):
        low = max(0, k - (n - 1))
        high = min(n - 1, k)
        idx = np.arange(low, high + 1, dtype=int)
        partner = k - idx
        expr = cp.sum(cp.multiply(f[low : high + 1], g[partner]))
        constraints.append(expr <= M)

    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.GLPK, warm_start=True, verbose=False, glpkk=1e-08)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, verbose=False, max_iters=2000)
        except Exception:
            return None

    if prob.status not in ("optimal", "optimal_inaccurate"):
        return None
    val = np.asarray(g.value).flatten()
    if val is None:
        return None
    return np.maximum(val, 0.0)


def _lp_refine(
    init: np.ndarray,
    target_sum: float,
    time_limit: float,
    mix: float = 0.2,
    max_iter: int = 6,
) -> tuple[np.ndarray, float]:
    """Perform several LP‑updates starting from init."""
    a = _project_onto_simplex(init.astype(float), target_sum)

    try:
        best_score = float(evaluate_sequence(a.tolist()))
    except Exception:
        best_score = math.inf
    best_seq = a.copy()

    t_start = time.time()
    for it in range(max_iter):
        if time.time() - t_start > time_limit:
            break
        conv = np.convolve(a, a)
        M = float(conv.max())

        g0 = _solve_lp_for_g(a, M)
        if g0 is None or np.sum(g0) == 0:
            break
        g = g0 * (target_sum / np.sum(g0))

        a = (1.0 - mix) * a + mix * g
        a = _project_onto_simplex(a, target_sum)

        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            score = math.inf
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(
                f"[{time.time():.2f}] LP‑refine iter {it+1} new best={score:.6f}"
            )
    return best_seq, best_score


# ----------------------------------------------------------------------
# Candidate generator – Gaussian-shaped initial guess
# ----------------------------------------------------------------------
def _gaussian_init(n: int, target_sum: float, sigma_factor: float = 0.3) -> np.ndarray:
    """Return a bell‑shaped vector (discrete Gaussian) of length n."""
    x = np.arange(n)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
    return _project_onto_simplex(a, target_sum)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a step‑function (list of non‑negative heights) with a low
    evaluation score. The routine respects a total time budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # Global best ( initialise from historic data if present )
    # ------------------------------------------------------------------
    best_seq = None
    best_score = math.inf

    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(height_sequence_1, dtype=float)
            sc = float(evaluate_sequence(cand.tolist()))
            best_seq = cand.copy()
            best_score = sc
            print(f"[{time.time():.2f}] Starting from historic seq, score={sc:.6f}")
        except Exception:
            best_seq = None
            best_score = math.inf

    # fallback: uniform of length 1000
    if best_seq is None:
        n0 = 1000
        target0 = math.sqrt(2.0 * n0)
        best_seq = np.full(n0, target0 / n0, dtype=float)
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"[{time.time():.2f}] No historic seq – using uniform, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Helper to run a single optimisation trial
    # ------------------------------------------------------------------
    def _run_one_trial(n: int, init_vec: np.ndarray, remaining: float):
        """Run a full optimisation (smooth → sub‑grad → optional LP)."""
        target = math.sqrt(2.0 * n)

        # 1) smooth‑max descent (≈ 12 s)
        t1 = min(remaining * 0.55, 15.0)
        seq_s, sc_s = _smooth_descent(
            init_vec,
            target,
            t1,
            beta_schedule=[0.2, 0.5, 1.0, 2.0, 4.0, 8.0],
            base_step=0.25,
            print_tag=f"smooth n={n}",
        )
        remaining -= (time.time() - (deadline - remaining))

        # 2) short sub‑gradient refinement (≈ 4 s)
        if remaining > 4.0:
            t2 = min(remaining * 0.4, 6.0)
            seq_l, sc_l = _local_descent(
                seq_s,
                target,
                t2,
                base_step=0.07,
                print_tag=f"subgrad n={n}",
            )
        else:
            seq_l, sc_l = seq_s, sc_s

        # 3) optional LP refinement (cheap, only for modest n)
        seq_out, sc_out = seq_l, sc_l
        if _CVX_AVAILABLE and n <= 800 and remaining > 5.0 and np.random.rand() < 0.2:
            t3 = min(remaining * 0.25, 6.0)
            seq_lp, sc_lp = _lp_refine(seq_l, target, t3, mix=0.2, max_iter=4)
            if sc_lp < sc_out:
                seq_out, sc_out = seq_lp, sc_lp

        return seq_out, sc_out

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while time.time() < deadline - 1.0:
        # Choose a length – favour the current best but allow exploration
        if best_seq is not None and np.random.rand() < 0.65:
            n = best_seq.size
            if np.random.rand() < 0.35:
                n = int(n * np.random.uniform(0.85, 1.15))
                n = max(200, min(4000, n))
        else:
            n = np.random.randint(200, 2500)

        target_sum = math.sqrt(2.0 * n)

        # initialise candidate
        r = np.random.rand()
        if r < 0.5 and best_seq is not None:
            init = _resize_sequence(best_seq, n)
        elif r < 0.8:
            init = _gaussian_init(n, target_sum, sigma_factor=np.random.uniform(0.15, 0.45))
        else:
            init = np.random.rand(n)

        # allocate time slice for this trial
        time_left = deadline - time.time()
        if time_left < 8.0:
            break
        cand_seq, cand_score = _run_one_trial(n, init, time_left)

        if cand_seq is not None and cand_score < best_score:
            best_score = cand_score
            best_seq = cand_seq.copy()
            print(
                f"[{time.time():.2f}] *** GLOBAL NEW BEST = {best_score:.6f} (n={n})"
            )
            if best_score < 1.503:
                break

    # ------------------------------------------------------------------
    # Final safety fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=float)

    return best_seq.tolist()
