def _smooth_descent(p_init, n, time_limit, base_lr=0.1, beta_max=20.0):
    p = _project_onto_simplex(p_init, 1.0)
    best_p = p.copy()
    conv = np.convolve(p,p)
    best_max = conv.max()
    it=0
    t0 = time.time()
    while time.time() < t0 + time_limit:
        conv = np.convolve(p,p)
        max_val = conv.max()
        # schedule beta: linearly increase up to beta_max over iterations
        beta = beta_max * (1.0 - np.exp(-it/200.0))
        # compute softmax weights
        exp_arg = beta * (conv - max_val)
        # avoid overflow by clipping
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.ones_like(w) / w.size
        else:
            w /= w_sum
        # gradient of weighted sum
        grad = 2.0 * np.convolve(w, p[::-1])[n-1:2*n-1]
        # make gradient orthogonal to simplex (zero sum)
        mean_g = np.mean(grad)
        d = -grad + mean_g
        # step size scaling
        max_abs = np.max(np.abs(d)) if np.max(np.abs(d))>0 else 1.0
        eta = base_lr * max_val / max_abs
        # update
        p_new = p + eta * d
        p_new = _project_onto_simplex(p_new, 1.0)
        # evaluate
        conv_new = np.convolve(p_new,p_new)
        max_new = conv_new.max()
        if max_new < best_max:
            best_max = max_new
            best_p = p_new.copy()
        p = p_new
        it += 1
    return best_p, best_max

"""
Improved optimiser for the step‑function bound.

The pipeline for each trial is
  1.   Turn the current candidate into a probability vector p (∑p=1).
  2.   Run a *smooth* descent on p that minimises a
       soft‑max of the convolution p∗p.  The soft‑max weight
       focuses the gradient on the largest convolution entries while
       keeping the update direction smooth.
  3.   Scale p back to the required ℓ¹‑norm (√(2n)) and polish the
       result with a *multi‑peak* projected‑gradient descent that
       uses the gradients of the top‑k convolution entries
       (instead of only the absolute maximum as in the reference code).
  4.   Optionally perform one cheap LP‑refinement (if cvxpy is present)
       for moderate problem sizes.
The search repeatedly restarts with random or resized “best‑so‑far”
initialisations and keeps the globally best sequence found before the
budget expires.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Projection on the simplex {x ≥ 0 , Σ x = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex of non‑negative vectors with sum s."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Small random mutation (Gaussian noise + simplex projection)
# ----------------------------------------------------------------------
def _mutate(a: np.ndarray, target_sum: float, sigma: float) -> np.ndarray:
    """Return a mutated copy of a (Gaussian noise, then simplex projection)."""
    b = a + np.random.randn(a.size) * sigma
    b = np.maximum(b, 0.0)
    b = _project_onto_simplex(b, target_sum)
    return b


# ----------------------------------------------------------------------
# Gradient that aggregates the top‑k convolution entries
# ----------------------------------------------------------------------
def _gradient_topk(a: np.ndarray, conv: np.ndarray, topk: int) -> np.ndarray:
    """
    Gradient of the sum of the top‑k convolution values.
    Returns a vector of length len(a).
    """
    n = a.shape[0]
    L = conv.shape[0]               # = 2n‑1
    topk = min(topk, L)
    # indices of the largest entries (unsorted)
    idx = np.argpartition(-conv, topk - 1)[:topk]
    grad = np.zeros_like(a)
    for k in idx:
        low = max(0, k - (n - 1))
        high = min(n - 1, k)
        ii = np.arange(low, high + 1, dtype=int)
        partner = k - ii
        grad[ii] += 2.0 * a[partner]
    return grad


# ----------------------------------------------------------------------
# Multi‑peak projected gradient descent (uses the top‑k gradient)
# ----------------------------------------------------------------------
def _multi_peak_descent(
    init: np.ndarray,
    target_sum: float,
    time_limit: float,
    topk: int = 5,
    base_step: float = 0.07,
    print_tag: str = "",
) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent that reduces the top‑k convolution entries.
    Returns the best sequence seen and its score.
    """
    a = _project_onto_simplex(init.astype(float), target_sum)

    try:
        best_score = float(evaluate_sequence(a.tolist()))
    except Exception:
        best_score = math.inf
    best_seq = a.copy()

    t0 = time.time()
    it = 0
    step = base_step

    while time.time() < t0 + time_limit:
        conv = np.convolve(a, a)
        grad = _gradient_topk(a, conv, topk)

        # keep the sum constant (mean‑zero direction)
        mean_g = np.mean(grad)
        d = -grad + mean_g
        max_abs = np.max(np.abs(d))
        if max_abs == 0:
            max_abs = 1.0
        eta = step * np.max(conv) / max_abs

        a_new = a + eta * d
        a_new = _project_onto_simplex(a_new, target_sum)

        try:
            score = float(evaluate_sequence(a_new.tolist()))
        except Exception:
            score = math.inf

        if score < best_score:
            best_score = score
            best_seq = a_new.copy()
            print(f"[{time.time():.2f}] {print_tag} new multi‑peak best = {score:.6f} (n={len(a)})")

        # occasional random shake‑up to escape plateaux
        if np.random.rand() < 0.02:
            sigma = 0.01 * target_sum / len(a)
            a = _mutate(a_new, target_sum, sigma)
            step *= 0.9
        else:
            a = a_new

        # simple step‑size decay
        if it % 200 == 0 and it > 0:
            step *= 0.9
        it += 1

    return best_seq, best_score


# ----------------------------------------------------------------------
# Smooth (soft‑max) descent on the probability vector p (∑p=1)
# ----------------------------------------------------------------------
def _smooth_descent(
    p_init: np.ndarray,
    n: int,
    time_limit: float,
    base_lr: float = 0.05,
    beta_max: float = 20.0,
) -> tuple[np.ndarray, float]:
    """
    Minimise the soft‑max of the convolution p∗p.
    Returns the best probability vector and its maximum convolution value.
    """
    p = _project_onto_simplex(p_init.astype(float), 1.0)

    best_p = p.copy()
    best_val = np.max(np.convolve(p, p))

    t0 = time.time()
    it = 0
    while time.time() < t0 + time_limit:
        conv = np.convolve(p, p)
        max_val = conv.max()

        # temperature schedule – slowly increase β to focus on the true max
        beta = beta_max * (1.0 - math.exp(-it / 200.0))

        # soft‑max weights (numerically stable)
        exp_arg = beta * (conv - max_val)
        w = np.exp(exp_arg - np.max(exp_arg))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w = w / w_sum

        # gradient of Σ w_i·conv_i  (see derivation in analysis)
        grad = 2.0 * np.convolve(w, p[::-1])[n - 1 : 2 * n - 1]

        # keep the ℓ¹‑norm fixed
        mean_g = np.mean(grad)
        d = -grad + mean_g

        max_abs = np.max(np.abs(d))
        if max_abs == 0:
            max_abs = 1.0
        eta = base_lr * max_val / max_abs

        p_new = p + eta * d
        p_new = _project_onto_simplex(p_new, 1.0)

        conv_new = np.convolve(p_new, p_new)
        max_new = conv_new.max()
        if max_new < best_val:
            best_val = max_new
            best_p = p_new.copy()

        p = p_new
        it += 1

    return best_p, best_val

def _solve_lp_for_g(f: np.ndarray, M: float):
    """Solve: maximise Σ g_j  s.t. g ≥ 0 and f∗g ≤ M."""
    if not _CVX_AVAILABLE:
        return None
    n = f.shape[0]
    g = cp.Variable(n, nonneg=True)

    constraints = []
    for k in range(2 * n - 1):
        low = max(0, k - (n - 1))
        high = min(n - 1, k)
        idx = np.arange(low, high + 1, dtype=int)
        partner = k - idx
        expr = cp.sum(cp.multiply(f[low : high + 1], g[partner]))
        constraints.append(expr <= M)

    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.GLPK, warm_start=True, verbose=False, glpkk=1e-08)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, verbose=False, max_iters=2000)
        except Exception:
            return None

    if prob.status not in ("optimal", "optimal_inaccurate"):
        return None
    g_val = np.asarray(g.value).flatten()
    if g_val is None:
        return None
    g_val = np.maximum(g_val, 0.0)
    return g_val


def _lp_refine(
    init: np.ndarray,
    target_sum: float,
    time_limit: float,
    mix: float = 0.2,
    max_iter: int = 6,
) -> tuple[np.ndarray, float]:
    """
    Apply a few LP‑updates starting from ``init``.
    Returns the best sequence seen and its score.
    """
    a = _project_onto_simplex(init.astype(float), target_sum)
    try:
        best_score = float(evaluate_sequence(a.tolist()))
    except Exception:
        best_score = math.inf
    best_seq = a.copy()

    t_start = time.time()
    for it in range(max_iter):
        if time.time() - t_start > time_limit:
            break
        conv = np.convolve(a, a)
        M = float(conv.max())

        g0 = _solve_lp_for_g(a, M)
        if g0 is None or np.sum(g0) == 0:
            break

        scale = target_sum / np.sum(g0)
        g = g0 * scale

        a = (1.0 - mix) * a + mix * g
        a = _project_onto_simplex(a, target_sum)

        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            score = math.inf

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(
                f"[{time.time():.2f}] LP‑refine iter {it+1} new best = {score:.6f} (n={len(a)})"
            )
    return best_seq, best_score

def _resize_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize seq to length new_n preserving non‑negativity."""
    old_n = seq.size
    if old_n == new_n:
        return seq.copy()
    if old_n > new_n:
        idx = np.linspace(0, old_n - 1, new_n).astype(int)
        return seq[idx]
    else:
        x_old = np.arange(old_n)
        x_new = np.linspace(0, old_n - 1, new_n)
        return np.interp(x_new, x_old, seq)


# ----------------------------------------------------------------------
# One optimisation run (choice of method)
# ----------------------------------------------------------------------
def _run_one(n: int, init_vec: np.ndarray, remaining: float) -> tuple[np.ndarray, float]:
    """
    Perform a single optimisation trial for length ``n``.
    ``init_vec`` is a candidate sequence (sum may be far from the target).
    Returns (best_sequence, best_score).
    """
    target = math.sqrt(2.0 * n)

    # ----- turn the initial vector into a probability vector -----
    total = np.sum(init_vec)
    if total <= 0:
        p_init = np.full(n, 1.0 / n)
    else:
        p_init = init_vec.astype(float) / total  # ∑p = 1

    # ----- smooth descent on the probability vector -----
    smooth_time = min(remaining * 0.45, 12.0)
    p_opt, max_conv = _smooth_descent(p_init, n, smooth_time)

    # Scale back to the required ℓ¹‑norm
    a_opt = p_opt * target

    try:
        best_score = float(evaluate_sequence(a_opt.tolist()))
    except Exception:
        best_score = math.inf
    best_seq = a_opt.copy()

    # ----- polish with multi‑peak projected gradient descent -----
    remaining = remaining - smooth_time
    if remaining > 4.0:
        cand_seq, cand_score = _multi_peak_descent(
            a_opt, target, remaining, topk=5, base_step=0.07, print_tag=f"n={n}"
        )
        if cand_score < best_score:
            best_seq, best_score = cand_seq, cand_score

    # ----- optional cheap LP refinement for moderate n -----
    if _CVX_AVAILABLE and n <= 600 and remaining > 6.0:
        cand_seq, cand_score = _lp_refine(best_seq, target, remaining, mix=0.2, max_iter=2)
        if cand_score < best_score:
            best_seq, best_score = cand_seq, cand_score

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a step‑function (list of non‑negative heights) with a low
    evaluation score.  Runs for at most ``budget_s`` seconds and returns
    the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # load historic best if available
    # ------------------------------------------------------------------
    best_seq = None
    best_score = math.inf
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(height_sequence_1, dtype=float)
            sc = float(evaluate_sequence(cand.tolist()))
            best_seq = cand.copy()
            best_score = sc
            print(f"[{time.time():.2f}] Started from historic seq, score={sc:.6f}")
        except Exception:
            best_seq = None
            best_score = math.inf

    # fallback: uniform 1000‑element sequence
    if best_seq is None:
        n0 = 1000
        target0 = math.sqrt(2.0 * n0)
        best_seq = np.full(n0, target0 / n0, dtype=float)
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"[{time.time():.2f}] No historic seq – using uniform, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Main search loop
    # ------------------------------------------------------------------
    while time.time() < deadline - 1.0:
        # pick a length – bias towards current best but allow exploration
        if best_seq is not None and np.random.rand() < 0.6:
            n = best_seq.size
            if np.random.rand() < 0.3:
                n = int(n * np.random.uniform(0.9, 1.1))
                n = max(200, min(2000, n))
        else:
            n = np.random.randint(200, 1800)

        target_sum = math.sqrt(2.0 * n)

        # initialise candidate
        if best_seq is not None:
            init = _resize_sequence(best_seq, n)
        else:
            init = np.random.rand(n)
            init = _project_onto_simplex(init, target_sum)

        # allocate a slice of the remaining time for this trial
        time_left = deadline - time.time()
        if time_left < 7.0:
            break

        cand_seq, cand_score = _run_one(n, init, time_left)

        if cand_seq is not None and cand_score < best_score:
            best_score = cand_score
            best_seq = cand_seq.copy()
            print(
                f"[{time.time():.2f}] *** GLOBAL NEW BEST = {best_score:.6f} (n={n})"
            )
            # early exit if we already beat the target bound
            if best_score < 1.503:
                break

    # ------------------------------------------------------------------
    # Final polishing on the global best if time permits
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 10.0 and best_seq is not None:
        try:
            # a short local refinement (the original sub‑gradient version) may still help
            cand_seq, cand_score = _multi_peak_descent(
                best_seq,
                math.sqrt(2.0 * best_seq.size),
                remaining,
                topk=5,
                base_step=0.07,
                print_tag="final",
            )
            if cand_score < best_score:
                best_seq, best_score = cand_seq, cand_score
                print(f"[{time.time():.2f}] Final polishing gave new best = {best_score:.6f}")
        except Exception:
            pass

    # safety fallback
    if best_seq is None:
        n = 1000
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=float)

    return best_seq.tolist()
