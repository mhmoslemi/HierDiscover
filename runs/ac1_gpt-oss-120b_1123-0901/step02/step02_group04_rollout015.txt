def _lp_refine(a_init, target, deadline, best, max_iters=5):
    try:
        from scipy.optimize import linprog
    except Exception:
        return a_init
    a = a_init.copy()
    n = a.shape[0]
    rows = 2 * n - 1
    # We'll allocate A each iteration because a updates.
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        conv = _conv_self(a)
        M = conv.max()
        # Build constraint matrix A
        # Each row k has entries a_{k - i} for i indices valid.
        A = np.zeros((rows, n), dtype=float)
        for k in range(rows):
            low = max(0, k - (n - 1))
            high = min(n - 1, k)
            cols = np.arange(low, high + 1, dtype=int)
            A[k, cols] = a[k - cols]
        # Solve LP: maximize sum(g) -> minimize -sum(g)
        res = linprog(
            c=-np.ones(n),
            A_ub=A,
            b_ub=np.full(rows, M),
            bounds=(0, None),
            method='highs',
            options={'presolve': True},
        )
        if not res.success:
            break
        g = res.x
        s = g.sum()
        if s == 0:
            break
        # Rescale to simplex
        g = g * (target / s)
        # Blend
        t = 0.1
        a = (1 - t) * a + t * g
        # Update best
        _update_best(a, best)
        # Check improvement
        if best['score'] < 1.503:
            break
    return a

"""
Black‑box optimiser for the step‑function evaluation problem.

The routine builds many candidate sequences from simple parametric families,
runs a fast sub‑gradient descent with momentum (averaging the gradients of the
few largest autocorrelation entries) from several random restarts, and finally
applies a short linear‑programming refinement (if SciPy is available).  The
search is time‑bounded – it stops automatically when the given budget expires
and returns the best sequence found so far.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the simplex {x ≥ 0, Σx = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project a vector onto the simplex {x >= 0, sum(x) = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # sorted decreasing
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper: Full convolution a * a (real FFT, fast for n up to a few thousand)
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Return the (full) convolution a * a using an efficient real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())      # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Helper: score update (internal objective = max autocorrelation)
# ----------------------------------------------------------------------
def _update_best(seq: np.ndarray, best: dict) -> float:
    """
    Update the global best dictionary if *seq* improves the score.
    Returns the internal score (max self‑convolution entry).
    """
    score = float(_conv_self(seq).max())
    if score < best["score"]:
        best["score"] = score
        best["seq"] = seq.copy()
        print(f"[{time.time():.2f}] New best = {score:.6f} (len={len(seq)})")
    return score


# ----------------------------------------------------------------------
# Shape factories (non‑negative, not yet normalised)
# ----------------------------------------------------------------------
def _shape_exp(n: int, alpha: float) -> np.ndarray:
    """decreasing exponential a_i = exp(-alpha * i)"""
    return np.exp(-alpha * np.arange(n, dtype=float))


def _shape_exp_rev(n: int, alpha: float) -> np.ndarray:
    """reverse exponential (largest at the end)"""
    return np.exp(-alpha * np.arange(n - 1, -1, -1, dtype=float))


def _shape_linear_desc(n: int, beta: float) -> np.ndarray:
    """linear decrease a_i = max(0, 1 - beta * i)"""
    i = np.arange(n, dtype=float)
    return np.maximum(0.0, 1.0 - beta * i)


def _shape_quad_desc(n: int, beta: float) -> np.ndarray:
    """quadratic decrease a_i = max(0, 1 - beta * i^2)"""
    i = np.arange(n, dtype=float)
    return np.maximum(0.0, 1.0 - beta * i * i)


def _shape_triangular(n: int) -> np.ndarray:
    """symmetric triangle a_i = min(i+1, n-i)"""
    i = np.arange(n, dtype=float)
    return np.minimum(i + 1.0, n - i)


def _shape_power_law(n: int, p: float) -> np.ndarray:
    """power‑law decay a_i = (i+1)^(-p) (always decreasing)"""
    i = np.arange(1, n + 1, dtype=float)
    return 1.0 / (i ** p)


def _shape_logistic(n: int, beta: float) -> np.ndarray:
    """logistic decay a_i = 1 / (1 + exp(beta*(i - n/2)))"""
    i = np.arange(n, dtype=float)
    mid = n / 2.0
    return 1.0 / (1.0 + np.exp(beta * (i - mid)))


def _shape_sqrt_desc(n: int) -> np.ndarray:
    """square‑root decay a_i = sqrt(n-i)"""
    i = np.arange(n, dtype=float)
    return np.sqrt(np.maximum(0.0, n - i))


# ----------------------------------------------------------------------
# Sub‑gradient descent with momentum (averaging the gradients of the
# top‑k autocorrelation entries).  Updates the *best* dict in‑place.
# ----------------------------------------------------------------------
def _subgradient_optimize(
    a0: np.ndarray,
    target: float,
    deadline: float,
    best: dict,
    max_iter: int = 5000,
    top_k: int = 3,
) -> None:
    """
    Perform projected sub‑gradient descent from start vector *a0*.
    The vector is kept on the simplex Σa = target.
    Periodically updates *best*.
    """
    a = a0.astype(float).copy()
    n = a.shape[0]
    m = np.zeros_like(a)            # momentum
    beta = 0.9                      # momentum decay
    base_step = 0.5                 # initial step magnitude

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        conv = _conv_self(a)
        max_val = conv.max()

        # indices of the top‑k largest autocorrelation entries
        if top_k == 1:
            ks = np.array([int(np.argmax(conv))])
        else:
            # use argpartition for speed
            ks = np.argpartition(conv, -top_k)[-top_k:]

        # average sub‑gradient over the selected shifts
        grad = np.zeros_like(a)
        for k in ks:
            low = max(0, k - (n - 1))
            high = min(n - 1, k)
            idx = np.arange(low, high + 1, dtype=int)
            partner = k - idx
            grad[idx] += 2.0 * a[partner]
        grad = grad / len(ks)

        # sub‑gradient step (direction = -grad)
        direction = -grad
        m = beta * m + (1.0 - beta) * direction

        # diminishing step size
        step = base_step / math.sqrt(it + 1)

        # normalise step w.r.t. ∞‑norm of momentum
        max_abs = np.max(np.abs(m))
        if max_abs == 0.0:
            max_abs = 1.0
        eta = step / max_abs

        a = a + eta * m
        a = _project_onto_simplex(a, target)

        # cheap evaluation each few iterations
        if it % 10 == 0:
            _update_best(a, best)
            # early exit if we have already passed the target bound
            if best["score"] < 1.503:
                return


# ----------------------------------------------------------------------
# Linear‑programming refinement (optional, uses SciPy if present)
# ----------------------------------------------------------------------
def _lp_refine(
    a0: np.ndarray,
    target: float,
    deadline: float,
    best: dict,
    max_iters: int = 3,
) -> None:
    """Iteratively improve *a0* by solving the LP described in the SOTA paper."""
    try:
        from scipy.optimize import linprog
    except Exception:            # SciPy not available
        return

    a = a0.copy()
    n = a.shape[0]
    rows = 2 * n - 1

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        conv = _conv_self(a)
        M = conv.max()

        # Build matrix A where (A g)_k = sum_i a_i g_{k-i}
        A = np.zeros((rows, n), dtype=float)
        for k in range(rows):
            low = max(0, k - (n - 1))
            high = min(n - 1, k)
            cols = np.arange(low, high + 1, dtype=int)
            A[k, cols] = a[k - cols]

        # maximise Σ g_j   ⇔ minimise  -Σ g_j
        res = linprog(
            c=-np.ones(n),
            A_ub=A,
            b_ub=np.full(rows, M),
            bounds=(0, None),
            method="highs",
            options={"presolve": True},
        )
        if not res.success:
            break

        g = res.x
        s = g.sum()
        if s <= 0.0:
            break

        # rescale g onto the simplex used by the rest of the algorithm
        g = g * (target / s)

        # convex combination step – a small blending factor keeps things stable
        t = 0.1
        a = (1.0 - t) * a + t * g

        _update_best(a, best)

        if best["score"] < 1.503:
            break

            a = _shape_quad_desc(n, beta)
            if a.sum() == 0:
                continue
            a = a / a.sum() * target
            _update_best(a, best)

        # triangular
        a = _shape_triangular(n)
        a = a / a.sum() * target
        _update_best(a, best)

        # power‑law
        for p in np.linspace(0.2, 2.0, num=12):
            a = _shape_power_law(n, p)
            if a.sum() == 0:
                continue
            a = a / a.sum() * target
            _update_best(a, best)

        # logistic
        for beta in np.linspace(0.001, 0.07, num=12):
            a = _shape_logistic(n, beta)
            if a.sum() == 0:
                continue
            a = a / a.sum() * target
            _update_best(a, best)

        # sqrt descending
        a = _shape_sqrt_desc(n)
        a = a / a.sum() * target
        _update_best(a, best)

    # ------------------------------------------------------------------
    # Phase 2 – heavy sub‑gradient descent (multiple restarts)
    # ------------------------------------------------------------------
    if best["seq"] is not None:
        _subgradient_optimize(best["seq"], math.sqrt(2.0 * len(best["seq"])), deadline, best)

    # Random restarts – generate Dirichlet‑like vectors and optimise them
    while time.time() < deadline:
        # Choose a length that balances expressivity and speed
        n = np.random.randint(600, 2501)            # 600 … 2500
        target = math.sqrt(2.0 * n)

        # random start: draw i.i.d. exponentials and normalise (Dirichlet)
        a0 = np.random.rand(n) + 1e-8
        a0 = a0 / a0.sum() * target

        _subgradient_optimize(a0, target, deadline, best)

        if best["score"] < 1.503:
            break

    # ------------------------------------------------------------------
    # Phase 3 – optional LP refinement (a few iterations)
    # ------------------------------------------------------------------
    if best["seq"] is not None and time.time() < deadline:
        _lp_refine(best["seq"], math.sqrt(2.0 * len(best["seq"])), deadline, best, max_iters=4)

    # ------------------------------------------------------------------
    # Fallback – uniform sequence if nothing succeeded
    # ------------------------------------------------------------------
    if best["seq"] is None:
        n = 1200
        target = math.sqrt(2.0 * n)
        best["seq"] = np.full(n, target / n, dtype=float)

    # Return plain Python list as required by the harness
    return best["seq"].tolist()
