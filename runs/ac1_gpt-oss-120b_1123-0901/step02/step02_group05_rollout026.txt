"""
Exponential‐decay step functions for the max‑convolution objective.

The evaluation metric simplifies to the maximal autocorrelation entry when the
sum of the heights is fixed to √(2 n).  For a geometric progression
a_i = C·r^i the convolution can be written in closed form:
    conv(k) = C²·(k+1)·r^k  (0 ≤ k < n) .
Choosing r = 1−λ/n (λ>0) and scaling C so that Σ a_i = √(2 n) yields
a one‑parameter family of admissible sequences.  The maximal entry of the
convolution depends only on λ:
    f(λ) = 2·λ²·e^{−λ} / (1−e^{−λ})² .
Its minimum is attained near λ≈2, giving a value ≈1.44 – already below the
target 1.503.  The code enumerates a handful of lengths n and λ values,
keeps the best construction, and finally runs a short projected‑gradient
refinement (sub‑gradient of the maximal entry) to squeeze a little more.

The routine respects all security checks of `evaluate_sequence`, works within
the 1000 s budget and returns a list of non‑negative floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simple Euclidean projection onto the simplex {x ≥ 0, Σ x = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex of non‑negative vectors with sum s."""
    n = v.shape[0]
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    # sort decreasing
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Gradient of the maximal convolution entry (sub‑gradient for the max)
# ----------------------------------------------------------------------
def _max_conv_and_grad(a: np.ndarray):
    """
    Compute max(conv(a, a)), its index and the sub‑gradient of that entry.
    Returns (max_val, grad, max_idx, full_conv).
    """
    n = a.shape[0]
    conv = np.convolve(a, a)
    max_idx = int(np.argmax(conv))
    max_val = conv[max_idx]

    # gradient of the k‑th entry: 2 * a shifted by k
    low = max(0, max_idx - (n - 1))
    high = min(n - 1, max_idx)
    idx = np.arange(low, high + 1, dtype=int)
    partner = max_idx - idx
    grad = np.zeros_like(a)
    grad[idx] = 2.0 * a[partner]
    return max_val, grad, max_idx, conv


# ----------------------------------------------------------------------
# Simple projected‑gradient descent (sub‑gradient of the max entry)
# ----------------------------------------------------------------------
def _local_descent(init: np.ndarray, target_sum: float,
                   time_limit: float, base_step: float = 0.04,
                   print_tag: str = "") -> tuple[np.ndarray, float]:
    """Run a short projected‑gradient descent starting from ``init``."""
    a = _project_onto_simplex(init.astype(float), target_sum)

    try:
        best_score = float(evaluate_sequence(a.tolist()))
    except Exception:
        best_score = math.inf
    best_seq = a.copy()

    step = base_step
    it = 0
    t0 = time.time()
    while time.time() < t0 + time_limit:
        max_val, grad, _, _ = _max_conv_and_grad(a)

        # move opposite to the sub‑gradient
        d = -grad
        # adapt step size to the magnitude of the gradient + current max
        max_abs = np.max(np.abs(d)) if np.max(np.abs(d)) > 0 else 1.0
        eta = step * max_val / max_abs

        a_new = a + eta * d
        a_new = _project_onto_simplex(a_new, target_sum)

        try:
            score = float(evaluate_sequence(a_new.tolist()))
        except Exception:
            score = math.inf

        if score < best_score:
            best_score = score
            best_seq = a_new.copy()
            print(f"[{time.time():.2f}] {print_tag} local best = {score:.6f}")

        a = a_new

        # decay step size occasionally
        if it % 300 == 0 and it > 0:
            step *= 0.9
        it += 1

    return best_seq, best_score


# ----------------------------------------------------------------------
# Exponential‑decay construction (parameter λ)
# ----------------------------------------------------------------------
def _exponential_sequence(n: int, lam: float, reverse: bool = False) -> np.ndarray:
    """
    Return a length‑n non‑negative sequence a_i = C·r^i (or reversed)
    with r = 1 − λ/n and Σ a_i = √(2 n).
    """
    if lam <= 0.0:
        lam = 1e-12
    alpha = lam / n                     # α = 1−r
    r = 1.0 - alpha                     # ratio 0 < r < 1
    # safe computation of rⁿ
    log_r = math.log(r) if r > 0 else -1e9
    rn = math.exp(n * log_r)           # r**n
    target = math.sqrt(2.0 * n)

    # C such that sum_i C·r^i = target
    # sum = C * (1 - rⁿ) / (1 - r) = C * (1 - rn) / alpha
    denom = 1.0 - rn
    if denom <= 0.0:
        denom = 1e-12
    C = target * alpha / denom

    # build the geometric progression
    idx = np.arange(n, dtype=float)
    seq = C * np.exp(idx * log_r)      # r**idx
    if reverse:
        seq = seq[::-1]
    # numerical safety – enforce Non‑negative and clamp to 1000 (evaluate does it)
    seq = np.maximum(seq, 0.0)
    return seq

# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: float = 1000.0, **kwargs):
    """
    Search for a step function (list of non‑negative heights) with a low
    evaluation score.  The routine explores exponential‑decay families,
    then applies a brief projected‑gradient refinement.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_score = math.inf

    # ------------------------------------------------------------------
    # 1) Scan a few lengths and λ‑values of the exponential family
    # ------------------------------------------------------------------
    lengths = [400, 600, 800, 1000, 1200, 1500, 1800]

            # decreasing exponential

            try:

            if score < best_score:

            # increasing exponential (mirror)
            seq_rev = _exponential_sequence(n, lam, reverse=True)
            try:
                score_rev = float(evaluate_sequence(seq_rev.tolist()))
            except Exception:
                score_rev = math.inf
            if score_rev < best_score:
                best_score = score_rev
                best_seq = seq_rev.copy()
                print(f"[{time.time():.2f}] new best (inc) n={n} λ={lam:.3f} → {score_rev:.6f}")

        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # 2) Local refinement from the best candidate (if any time left)
    # ------------------------------------------------------------------
    if best_seq is not None:

                best_score = refined_score
                print(f"[{time.time():.2f}] refined best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform sequence (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=float)
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"[{time.time():.2f}] fallback uniform, score={best_score:.6f}")

    # Return plain Python list
    return best_seq.tolist()
