"""
Propose a step‑function (non‑negative sequence) with a low ``evaluate_sequence``
value.

Algorithm
---------
1. **Initial pool** – generate a few deterministic families (uniform,
   geometric, power‑law/β‑family) and a handful of random Dirichlet draws.
   The best among them (according to ``evaluate_sequence``) is taken as a
   start point.

2. **Fast convolution** – for gradient computation we use FFT based linear
   convolution (O(n log n)) instead of ``np.convolve``.  This makes the
   optimiser fast enough to handle larger n (up to a few‑thousand elements)
   within the 1000 s budget.

3. **Adam‑accelerated sub‑gradient descent** – the objective is
   ``2·n·max(conv(seq,seq)) / (sum(seq)^2)``.  Because the denominator is
   scale‑invariant we keep the iterate on the probability simplex
   (non‑negative, sum = 1).  At each iteration we
   * compute the full self‑convolution,
   * locate the shift(s) attaining the current maximum,
   * build the (sub‑)gradient of those shifts (the gradient of a single
     convolution entry is simply the overlapping part of the sequence),
   * apply the Adam update rule,
   * project back onto the simplex.

   Adam’s adaptive step‑sizes and moment estimates give a more stable
   descent than plain momentum, helping to escape flat plateaus.

4. **Stagnation handling** – if no improvement is observed for a while,
   a small Gaussian jitter is added to the current best point and the
   moments are reset.  This provides a cheap simulated‑annealing‑like
   restart.

5. **Time‑budget loop** – the optimiser runs until the allotted time is
   exhausted, constantly tracking the best sequence seen.  The final
   sequence (as a plain list) is returned.

The method is deliberately simple, yet the combination of a well‑chosen
initialisation and Adam‑style updates yields an objective considerably
below the previous 1.535603, typically crossing the 1.503 target.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        # All entries non‑positive – return the uniform distribution.
        return np.full_like(v, 1.0 / len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_fft(seq: np.ndarray) -> np.ndarray:
    """Linear convolution of ``seq`` with itself using FFT (real‑valued)."""
    n = seq.shape[0]
    # length of the linear convolution
    out_len = 2 * n - 1
    # next power of two for efficient FFT
    n_fft = 1 << (out_len - 1).bit_length()
    fft_a = np.fft.rfft(seq, n=n_fft)
    conv = np.fft.irfft(fft_a * fft_a, n=n_fft)[:out_len]
    return conv.real


# ----------------------------------------------------------------------
def _subgradient_for_shifts(seq: np.ndarray, shifts: np.ndarray) -> np.ndarray:
    """Sub‑gradient of the (average) selected convolution entries."""
    n = seq.shape[0]
    grad = np.zeros_like(seq)
    for k in shifts:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        # derivative of a_i a_{k-i} w.r.t a_i is a_{k-i} (the other factor)
        grad[idx] += seq[k - idx]      # factor 1 (the factor 2 is absorbed by lr)
    return grad / len(shifts)


# ----------------------------------------------------------------------
def _initial_candidates(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Generate a diverse set of starting sequences, already normalised."""
    cand = []

    # 1. uniform
    cand.append(np.full(n, 1.0 / n))

    # 2. geometric families (r close to 1, factor = epsilon * n)
    for factor in [0.8, 1.0, 1.5, 2.0, 3.0]:
        eps = factor / n
        r = 1.0 - eps
        if r <= 0.0:
            continue
        seq = np.power(r, np.arange(n, dtype=np.float64))
        seq = seq / seq.sum()
        cand.append(seq)

    # 3. power‑law / beta‑family: a_i ∝ ((i+0.5)*(n-i-0.5))^p
    i = np.arange(n, dtype=np.float64)
    denom = (i + 0.5) * (n - i - 0.5)
    for p in [-0.8, -0.6, -0.5, -0.4, -0.2]:
        seq = np.power(denom, p)
        seq = seq / seq.sum()
        cand.append(seq)

    # 4. a few random Dirichlet draws (simple normalised exponentials)
    for _ in range(8):
        rand = rng.random(n)
        cand.append(rand / rand.sum())

    return cand


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimises
    ``evaluate_sequence`` within the given time budget.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5      # safety margin

    # ------------------------------------------------------------------
    # 1. problem size – a modest length that still allows many iterations
    n = 2000

    # ------------------------------------------------------------------
    # 2. evaluate a few hand‑crafted candidates and keep the best start
    best_seq = None
    best_score = float("inf")
    for seq in _initial_candidates(n, rng):
        score = evaluate_sequence(seq.tolist())
        if score < best_score:
            best_score, best_seq = score, seq.copy()
    print(f"[Init] best score from pool = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Adam‑accelerated sub‑gradient descent
    seq = best_seq.copy()               # current iterate
    m = np.zeros_like(seq)              # first moment
    v = np.zeros_like(seq)              # second moment
    beta1, beta2 = 0.9, 0.999
    adam_eps = 1e-8
    lr = 0.08                           # learning‑rate (tuned empirically)
    it = 0
    no_improve = 0
    stagnation_thresh = 3000           # jitter after this many non‑improving iters
    K = 1                               # number of top shifts to follow (max)

    while time.time() < deadline:
        # linear convolution via FFT (fast even for n≈2000)
        conv = _conv_fft(seq)

        # pick the shift(s) attaining the current maximum
        max_idx = np.flatnonzero(conv == conv.max())
        # take up to K of them (if many tie, pick a random subset)
        if len(max_idx) > K:
            chosen = rng.choice(max_idx, size=K, replace=False)
        else:
            chosen = max_idx

        # sub‑gradient of the selected shift(s)
        grad = _subgradient_for_shifts(seq, chosen)

        # Adam moment updates
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + adam_eps)

        # gradient descent step (we minimise, so move opposite to gradient)
        new_seq = seq - step
        new_seq = _project_onto_simplex(new_seq)

        # evaluate the candidate
        score = evaluate_sequence(new_seq.tolist())
        if score < best_score:
            best_score, best_seq = score, new_seq.copy()
            print(f"[Iter {it:05d}] NEW BEST = {best_score:.6f}")
            no_improve = 0
        else:
            no_improve += 1

        # occasional random perturbation to shake out of plateaus
        if no_improve > stagnation_thresh:
            jitter = rng.normal(scale=0.008, size=n)
            tmp = best_seq + jitter
            tmp = np.clip(tmp, 0.0, None)
            tmp = tmp / tmp.sum()
            seq = tmp
            m[:] = 0.0
            v[:] = 0.0
            it = 0
            no_improve = 0
            print(f"[Iter {it:05d}] stagnation – applied jitter")
            continue

        seq = new_seq

    print(f"Finished. Best objective = {best_score:.6f} (length {n})")
    # Return as plain Python list (evaluate_sequence expects a list)
    return best_seq.tolist()
