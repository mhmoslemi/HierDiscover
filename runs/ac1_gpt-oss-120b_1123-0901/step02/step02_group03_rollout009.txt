"""
Propose a non‑negative step‑function that minimizes

        2 * n * max(conv(a, a)) / (sum(a) ** 2)

using a projected Adam optimizer on a *soft‑max* approximation of the
max‑convolution.  At each iteration we compute the full convolution
with FFT, turn the max into a smooth logarithmic‑sum‑exp loss
(L_β = 1/β log Σ exp(β·conv)), obtain its gradient by one extra FFT,
and perform an Adam update followed by projection onto the simplex
{a_i ≥ 0, Σ a_i = √(2 n)}.  The temperature β is increased
gradually, turning the smooth loss into the true max‑loss.
Many short independent runs with random lengths (800–5000) are
executed; the best vector found is optionally refined in a final
longer run.  The algorithm stays fully within the 1000 s budget and
never uses closures or lambdas – all helpers are top‑level functions.
"""

import time
import math
import numpy as np

# --------------------------------------------------------------------------- #
# Helper: Euclidean projection onto the simplex { x_i ≥ 0 , Σ x_i = s }.
# --------------------------------------------------------------------------- #
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector ``v`` onto the simplex of radius ``s``."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")

    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------------------- #
# Fast full convolution (real FFT) – works for any two real vectors.
# --------------------------------------------------------------------------- #
def _conv_full_fft(x: np.ndarray, y: np.ndarray = None) -> np.ndarray:
    """
    Compute the full convolution of ``x`` with ``y`` (or with itself if ``y`` is None)
    using a real FFT.  Returns an array of length len(x)+len(y)-1.
    """
    if y is None:
        y = x
    n = x.shape[0]
    m = y.shape[0]
    n_fft = 1 << ((n + m - 1).bit_length())
    X = np.fft.rfft(x, n=n_fft)
    Y = np.fft.rfft(y, n=n_fft)
    conv = np.fft.irfft(X * Y, n=n_fft)[: n + m - 1]
    return conv


# --------------------------------------------------------------------------- #
# Gradient of the smooth (log‑sum‑exp) loss.
# For a weight vector w (non‑negative, same length as full convolution),
# the gradient w.r.t. a is
#   grad_i = 2 * Σ_j a_j * w_{i+j}
# which can be obtained by a single convolution of a with the reversed w.
# --------------------------------------------------------------------------- #
def _gradient_weighted(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute  ∂/∂a  Σ_k w_k * (a * a)[k] .
    Returns a vector of length len(a).
    """
    n = a.shape[0]
    # reverse the weight vector
    w_rev = w[::-1]
    # convolution a * w_rev (full)
    conv = _conv_full_fft(a, w_rev)        # length = n + (2n-1) - 1 = 3n-2
    # the needed slice starts at index 2*n-2 and goes backwards
    idx_start = 2 * n - 2
    slice_idx = idx_start - np.arange(n)
    grad = 2.0 * conv[slice_idx]
    return grad


# --------------------------------------------------------------------------- #
# Core optimisation routine for a single problem size.
# --------------------------------------------------------------------------- #
def _run_one_instance(
    n: int,
    init_vec: np.ndarray,
    best_seq_container: list,
    best_score_container: list,
    overall_deadline: float,
    max_run_time: float = 12.0,
) -> None:
    """
    Perform a projected Adam optimisation for length ``n``.
    Updates ``best_seq_container[0]`` and ``best_score_container[0]`` whenever
    a better candidate is found.
    """
    target_sum = math.sqrt(2.0 * n)                # Σ a_i = √(2n)
    a = _project_onto_simplex(init_vec.astype(float), target_sum)

    # Adam hyper‑parameters
    lr = 0.10
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    # soft‑max temperature (β in log‑sum‑exp)
    beta_temp = 1.0
    beta_max = 80.0
    beta_inc = 1.025

    run_deadline = min(overall_deadline, time.time() + max_run_time)

    while time.time() < run_deadline:
        # --------------------------------------------------- #
        # 1) Full self‑convolution and current objective value
        # --------------------------------------------------- #
        conv = _conv_full_fft(a)                     # length 2*n-1
        max_val = conv.max()
        cur_score = float(max_val)                  # with our normalisation this equals evaluate_sequence()
        if cur_score < best_score_container[0]:
            best_score_container[0] = cur_score
            best_seq_container[0] = a.copy()
            print(
                f"[{time.time():.2f}] New best = {cur_score:.6f} (n={n})"
            )

        # --------------------------------------------------- #
        # 2) Build a smooth approximation to the max (log‑sum‑exp)
        # --------------------------------------------------- #
        # weights w_k = exp(beta * (conv_k - max_val))   (0 ≤ w_k ≤ 1)
        w = np.exp(beta_temp * (conv - max_val))

        # --------------------------------------------------- #
        # 3) Gradient of the smooth loss via a single extra FFT
        # --------------------------------------------------- #
        grad = _gradient_weighted(a, w)

        # --------------------------------------------------- #
        # 4) Adam update + projection onto the simplex
        # --------------------------------------------------- #
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # Enforce feasibility
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_onto_simplex(a_new, target_sum)

        # --------------------------------------------------- #
        # 5) Occasional random shake (simulated annealing style)
        # --------------------------------------------------- #
        if np.random.rand() < 0.015:
            noise_scale = target_sum * 0.001 / n
            a_new = a_new + np.random.randn(n) * noise_scale
            a_new = _project_onto_simplex(np.maximum(a_new, 0.0), target_sum)
            # shrink learning rate a little after a shake
            lr *= 0.92

        a = a_new

        # --------------------------------------------------- #
        # 6) Scheduling of hyper‑parameters
        # --------------------------------------------------- #
        # slowly increase the temperature → sharper approximation of max
        beta_temp = min(beta_max, beta_temp * beta_inc)

        # exponential decay of the base learning rate
        lr *= 0.9996

        # tiny pruning – keep the vector sparse (helps numerical stability)
        prune_thr = 1e-9
        if np.any(a < prune_thr):
            a[a < prune_thr] = 0.0
            a = _project_onto_simplex(a, target_sum)


# --------------------------------------------------------------------------- #
# Main entry point expected by the grader.
# --------------------------------------------------------------------------- #
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs) -> list:
    """
    Search for a low‑value step‑function within ``budget_s`` seconds.
    Returns the best sequence found as a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # containers for the best solution discovered so far
    best_seq_container = [None]          # mutable holder for the numpy array
    best_score_container = [float("inf")]  # mutable holder for the score

    # ------------------------------------------------------------------- #
    # Main outer loop – many independent restarts with varying lengths.
    # ------------------------------------------------------------------- #
    while time.time() < deadline:
        # Choose a problem size.
        # Occasionally prefer a power of two (good for FFT) and occasionally
        # draw uniformly from a broader interval.
        if np.random.rand() < 0.30:
            # power‑of‑two between 2^10 (1024) and 2^13 (8192)
            n = 2 ** np.random.randint(10, 14)
        else:
            n = np.random.randint(800, 5001)

        # Initialise the vector.
        if np.random.rand() < 0.40 and "height_sequence_1" in globals():
            src = np.asarray(height_sequence_1, dtype=float)
            if src.size >= n:
                init = src[:n].copy()
            else:
                pad = np.random.rand(n - src.size) * 1e-3
                init = np.concatenate([src, pad])
        else:
            init = np.random.rand(n)

        # Run a short optimisation instance (≈10‑12 s)
        _run_one_instance(
            n,
            init,
            best_seq_container,
            best_score_container,
            deadline,
            max_run_time=12.0,
        )

        # Early exit if we have already passed the target bound
        if best_score_container[0] <= 1.503:
            break

    # ------------------------------------------------------------------- #
    # Final refinement on the current best vector, using a longer run.
    # ------------------------------------------------------------------- #
    remaining = deadline - time.time()
    if remaining > 15.0 and best_seq_container[0] is not None:
        n = best_seq_container[0].size
        _run_one_instance(
            n,
            best_seq_container[0],
            best_seq_container,
            best_score_container,
            deadline,
            max_run_time=remaining,
        )

    # ------------------------------------------------------------------- #
    # Fallback – produce a uniform vector if nothing was ever discovered.
    # ------------------------------------------------------------------- #
    if best_seq_container[0] is None:
        n = 1000
        target_sum = math.sqrt(2.0 * n)
        best_seq_container[0] = np.full(n, target_sum / n, dtype=float)

    # Return a plain Python list as required by the grader.
    return best_seq_container[0].tolist()
