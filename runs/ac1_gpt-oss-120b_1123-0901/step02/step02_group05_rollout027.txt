def max_conv_fft(x):
    n = len(x)
    size = int(2**np.ceil(np.log2(2*n-1)))
    fx = np.fft.rfft(x, n=size)
    conv = np.fft.irfft(fx * fx, n=size)
    conv = conv[:2*n-1]  # keep only needed part
    return np.max(conv)

def geometric_sequence(n, r):
    # generate decreasing geometric sequence with ratio r in (0,1)
    idx = np.arange(n)
    sequence = r**idx
    # Normalize to sum sqrt(2n) (or any sum, but scaling doesn't change evaluation)

    scale = S_target / sequence.sum()
    return sequence * scale

def fast_evaluation(seq):
    a = np.asarray(seq, dtype=np.float64)
    n = a.shape[0]
    size = int(2**np.ceil(np.log2(2*n-1)))
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[:2*n-1]
    max_b = float(conv.max())
    sum_a = float(a.sum())
    return 2 * n * max_b / (sum_a**2)

    best_seq = a.copy()

def gen_geometric(n):
    # Choose y around optimal 1.2
    y = np.random.uniform(0.5, 2.0)
    delta = y / n
    r = max(0.0, 1 - delta)
    seq = r ** np.arange(n, dtype=float)

def gen_triangular(n):
    # symmetric triangle with peak at centre
    i = np.arange(n)
    seq = 1.0 - np.abs(i - (n-1)/2) / ((n+1)/2)
    seq = np.maximum(seq, 0.0)
    return seq

    i = np.arange(n)

"""
Propose a step‑function (list of non‑negative heights) that minimises

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The programme works in three stages:

1.  **Fast random exploration** – a set of cheap candidate shapes
    (geometric, sine, triangular, power‑law, Dirichlet random) is
    generated for many different lengths (200 … 5000).  Their quality is
    estimated with a tiny FFT based evaluator (scale‑invariant, so the
    exact normalisation does not matter).

2.  **Gradient refinement** – the best few candidates are fed to a
    projected sub‑gradient optimiser that works on the true objective
    (via ``evaluate_sequence``).  The optimiser uses a momentum term,
    adaptive step size and keeps the sum of the coefficients fixed
    (simplex projection).  A small amount of random “shake‑up” is added
    periodically to escape shallow local minima.

3.  **Optional LP polishing** – if ``cvxpy`` is available an LP
    (the “LP‑update” from the paper sketch) is applied a few times.

The whole search respects the 1000 s budget; if the target value
(1.503) is reached the routine returns immediately.  All helper
functions are top‑level, there are no closures or file IO.
"""

import time
import math
import random
import numpy as np

# --------------------------------------------------------------
# Simple projection onto the simplex {x ≥ 0, Σ x = s}
# --------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex of non‑negative vectors with sum s."""
    n = v.shape[0]
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------
# Fast evaluation (scale‑invariant) using FFT – used during the
# cheap search phase.
# --------------------------------------------------------------
def _fast_evaluation(seq: np.ndarray) -> float:
    """Return 2*n*max(conv(seq,seq))/sum(seq)^2 using FFT (much faster than np.convolve)."""
    a = np.asarray(seq, dtype=np.float64)
    n = a.shape[0]
    if n == 0:
        return math.inf
    size = int(2 ** np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    max_b = float(conv.max())
    sum_a = float(a.sum())
    if sum_a < 1e-12:
        return math.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


# --------------------------------------------------------------
# Convolution + gradient of the max‑entry (sub‑gradient of the objective)
# --------------------------------------------------------------
def _max_conv_and_grad(a: np.ndarray):
    """
    Compute max(conv(a, a)), its index and the gradient of that entry.
    Returned ``grad`` satisfies sum(grad) == 0 (we later shift it to have zero mean).
    """
    n = a.shape[0]
    size = int(2 ** np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    max_idx = int(np.argmax(conv))
    max_val = conv[max_idx]

    low = max(0, max_idx - (n - 1))
    high = min(n - 1, max_idx)
    idx = np.arange(low, high + 1, dtype=int)
    partner = max_idx - idx
    grad = np.zeros_like(a)
    grad[idx] = 2.0 * a[partner]
    return max_val, grad, max_idx, conv


# --------------------------------------------------------------
# Local projected‑gradient descent (sub‑gradient of the maximal entry)
# --------------------------------------------------------------
def _local_descent(
    init: np.ndarray,
    target_sum: float,
    time_limit: float,
    base_step: float = 0.07,
    momentum: float = 0.9,
) -> tuple[np.ndarray, float]:
    """Run a short momentum‑gradient descent starting from ``init``."""
    a = _project_onto_simplex(init.astype(float), target_sum)

    best_seq = a.copy()
    best_score = _fast_evaluation(a)
    velocity = np.zeros_like(a)

    step = base_step
    it = 0
    t0 = time.time()

    while time.time() - t0 < time_limit:
        max_val, grad, _, _ = _max_conv_and_grad(a)

        # zero‑mean direction (keeps the sum constant after projection)
        d = -grad + grad.mean()

        # momentum update
        velocity = momentum * velocity + (1.0 - momentum) * d

        # step size scaling – larger max_val encourages larger steps
        max_abs = np.max(np.abs(velocity))
        if max_abs == 0:
            max_abs = 1.0
        eta = step * max_val / max_abs

        a_new = a + eta * velocity
        a_new = _project_onto_simplex(a_new, target_sum)

        score = _fast_evaluation(a_new)

        if score < best_score:
            best_score = score
            best_seq = a_new.copy()
            print(f"[{time.time():.2f}] descent new best = {score:.6f} (n={len(a)})")

        # occasional random shake‑up to escape plateaus
        if np.random.rand() < 0.03:
            sigma = 0.02 * target_sum / len(a)
            noise = np.random.randn(len(a)) * sigma
            a = _project_onto_simplex(a_new + noise, target_sum)
            step *= 0.9
            velocity[:] = 0.0
        else:
            a = a_new

        # step decay
        if it % 300 == 0 and it > 0:
            step *= 0.9
        it += 1

    # final accurate evaluation
    try:
        true_score = float(evaluate_sequence(best_seq.tolist()))
    except Exception:
        true_score = math.inf
    return best_seq, true_score

# Small mutation helper (Gaussian noise + simplex projection)
# --------------------------------------------------------------
def _mutate(a: np.ndarray, target_sum: float, sigma: float) -> np.ndarray:
    """Return a mutated copy of a (Gaussian noise, then simplex projection)."""
    b = a + np.random.randn(a.size) * sigma
    b = np.maximum(b, 0.0)
    b = _project_onto_simplex(b, target_sum)
    return b

# Optional LP refinement (requires cvxpy)
# --------------------------------------------------------------
try:
    import cvxpy as cp

    _CVX_AVAILABLE = True
except Exception:  # pragma: no cover
    _CVX_AVAILABLE = False


def _solve_lp_for_g(f: np.ndarray, M: float):
    """Solve the LP: maximise Σ g_j subject to g ≥ 0 and (f*g)_k ≤ M."""
    if not _CVX_AVAILABLE:
        return None
    n = f.shape[0]
    g = cp.Variable(n, nonneg=True)

    constraints = []
    for k in range(2 * n - 1):
        low = max(0, k - (n - 1))
        high = min(n - 1, k)
        idx = np.arange(low, high + 1, dtype=int)
        partner = k - idx
        expr = cp.sum(cp.multiply(f[low : high + 1], g[partner]))
        constraints.append(expr <= M)

    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.GLPK, warm_start=True, verbose=False, glpkk=1e-08)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, verbose=False, max_iters=2000)
        except Exception:
            return None

    if prob.status not in ("optimal", "optimal_inaccurate"):
        return None
    g_val = np.asarray(g.value).flatten()
    if g_val is None:
        return None
    return np.maximum(g_val, 0.0)


def _lp_refine(
    init: np.ndarray,
    target_sum: float,
    time_limit: float,
    mix: float = 0.2,
    max_iter: int = 6,
) -> tuple[np.ndarray, float]:
    """Perform a few LP‑updates starting from ``init``."""
    a = _project_onto_simplex(init.astype(float), target_sum)
    try:
        best_score = float(evaluate_sequence(a.tolist()))
    except Exception:
        best_score = math.inf
    best_seq = a.copy()

    start = time.time()
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        conv = np.convolve(a, a)
        M = float(conv.max())
        g0 = _solve_lp_for_g(a, M)
        if g0 is None or np.sum(g0) == 0:
            break
        g = g0 * (target_sum / np.sum(g0))
        a = (1.0 - mix) * a + mix * g
        a = _project_onto_simplex(a, target_sum)

        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            score = math.inf

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time():.2f}] LP‑refine iter {it+1} new best = {score:.6f} (n={len(a)})")
    return best_seq, best_score

# Shape generators for the cheap search phase
# --------------------------------------------------------------
def _gen_geometric(n: int) -> np.ndarray:
    """Exponential decay a_i = r**i, with r chosen via a random y≈optimal."""
    y = np.random.uniform(0.5, 2.0)          # y = (1‑r)·n, around optimal ≈1.256
    delta = y / n
    r = max(0.0, 1.0 - delta)
    return r ** np.arange(n, dtype=float)


def _gen_triangular(n: int) -> np.ndarray:
    """Symmetric triangle that peaks in the centre."""
    i = np.arange(n, dtype=float)
    seq = 1.0 - np.abs(i - (n - 1) / 2) / ((n + 1) / 2)
    return np.maximum(seq, 0.0)


def _gen_sine(n: int) -> np.ndarray:
    """Half‑sine wave – always non‑negative."""
    i = np.arange(n, dtype=float)
    return np.sin(np.pi * (i + 0.5) / n)


def _gen_power_decay(n: int) -> np.ndarray:
    """a_i = (i+1)^{-p} with random exponent p."""
    p = np.random.uniform(0.5, 3.0)
    return (np.arange(1, n + 1, dtype=float)) ** (-p)


def _gen_dirichlet(n: int) -> np.ndarray:
    """Random Dirichlet‑distributed vector (gamma sampling)."""
    alpha = np.random.uniform(0.5, 5.0)
    return np.random.gamma(alpha, 1.0, size=n)


_SHAPE_FUNCS = [_gen_geometric, _gen_triangular, _gen_sine,
                _gen_power_decay, _gen_dirichlet]


# --------------------------------------------------------------
# Simple resize (linear interpolation / down‑sampling)
# --------------------------------------------------------------
def _resize_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize seq to length new_n preserving non‑negativity (linear interpolation)."""
    old_n = seq.size
    if old_n == new_n:
        return seq.copy()
    if old_n > new_n:
        idx = np.linspace(0, old_n - 1, new_n).astype(int)
        return seq[idx]
    else:
        x_old = np.arange(old_n)
        x_new = np.linspace(0, old_n - 1, new_n)
        return np.interp(x_new, x_old, seq)

# Main entry point
# --------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a step‑function (list of non‑negative heights) with a low
    evaluation score.  The routine runs for at most ``budget_s`` seconds.
    """
    np.random.seed(seed)
    random.seed(seed)

    deadline = time.time() + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # Global best holder (initialised from historic solution if present)
    # ------------------------------------------------------------------
    best_seq = None
    best_score = math.inf

    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(height_sequence_1, dtype=float)
            sc = float(evaluate_sequence(cand.tolist()))
            best_seq = cand.copy()
            best_score = sc
            print(f"[{time.time():.2f}] historic start score = {sc:.6f}")
        except Exception:
            best_seq = None
            best_score = math.inf

    # ------------------------------------------------------------------
    # Phase 1 – cheap random exploration (FFT evaluator)
    # ------------------------------------------------------------------
    top_k = 8
    top = []                     # list of (score, seq, n)

    while time.time() < deadline - 30:          # keep ~30 s for refinement
        n = np.random.randint(300, 5000)       # varied lengths
        shape_func = random.choice(_SHAPE_FUNCS)
        seq = shape_func(n)
        if seq.sum() == 0:
            continue
        sc = _fast_evaluation(seq)
        if not np.isfinite(sc):
            continue

        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(f"[{time.time():.2f}] new fast global best = {sc:.6f} (n={n})")

        # keep a short list of the best candidates for later refinement
        if len(top) < top_k:
            top.append((sc, seq.copy(), n))
            top.sort(key=lambda x: x[0])
        elif sc < top[-1][0]:
            top[-1] = (sc, seq.copy(), n)
            top.sort(key=lambda x: x[0])

    # ------------------------------------------------------------------
    # Phase 2 – gradient descent refinements
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining <= 0:
        remaining = 0.0

    # allocate a chunk of time for each of the stored top candidates
    per_candidate = remaining / max(1, len(top))

    for sc0, seq0, n0 in top:
        if time.time() >= deadline - 2.0:
            break
        target = math.sqrt(2.0 * n0)
        t_alloc = min(per_candidate, deadline - time.time() - 2.0)
        cand, cand_score = _local_descent(seq0, target, t_alloc)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            print(f"[{time.time():.2f}] *** GLOBAL NEW BEST = {best_score:.6f} (n={n0})")
            if best_score < 1.503:
                break

    # ------------------------------------------------------------------
    # Phase 3 – optional LP polishing (if cvxpy is available)
    # ------------------------------------------------------------------
    if _CVX_AVAILABLE and best_seq is not None:
        try:
            n_best = best_seq.shape[0]
            target = math.sqrt(2.0 * n_best)
            cand, cand_score = _lp_refine(best_seq, target, deadline - time.time())
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                print(f"[{time.time():.2f}] LP‑final improvement to {best_score:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n0 = 1000
        best_seq = np.full(n0, math.sqrt(2.0 * n0) / n0, dtype=float)

    return best_seq.tolist()
