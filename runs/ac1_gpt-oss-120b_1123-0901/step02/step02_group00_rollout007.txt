def subgradient_one(a, k):
    n = len(a)
    i = np.arange(n)
    j = k - i
    mask = (j >= 0) & (j < n)
    rev = np.zeros_like(a)
    rev[mask] = a[j[mask]]
    return rev
```

def propose_candidate(seed=42, budget_s=1000):
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0

    # load base

        base = np.array(height_sequence_1, dtype=float)

        base = None

    lengths = [800, 1500, 3000, 5000, 8000]

    best_seq = None

    # if base exists, start from its length else use uniform of smallest length
    # we will gradually upscale

    # initialize current shape
    current_a = None
    current_len = None

    for n in lengths:
        if time.time() >= deadline:
            break
        target_sum = math.sqrt(2.0*n)

        # initialize a for length n
        if current_a is None:
            # first length: generate a set of seeds
            seeds = []
            # uniform
            seeds.append(np.full(n, target_sum/n))
            # random-like
            rand_vec = np.random.rand(n)
            rand_vec = rand_vec / rand_vec.sum() * target_sum
            seeds.append(rand_vec)
            # Gaussian with varied sigma
            center = (n-1)/2.0
            for sigma_factor in [0.1, 0.15, 0.2, 0.25]:
                sigma = sigma_factor * n
                x = np.arange(n)
                gauss = np.exp(-0.5*((x-center)/sigma)**2)
                gauss = gauss / gauss.sum() * target_sum
                seeds.append(gauss)
            # triangular
            tri = np.concatenate([np.arange(1, n//2+1), np.arange(n//2,0,-1)])
            tri = tri.astype(float)
            tri = tri / tri.sum() * target_sum
            seeds.append(tri)
            # exponential decay
            exp = np.exp(-np.arange(n)/ (0.2*n))
            exp = exp / exp.sum() * target_sum
            seeds.append(exp)
        else:
            # upscale previous best shape via interpolation
            old_a = current_a
            old_n = current_len
            # linear interpolation
            old_x = np.linspace(0, 1, old_n)
            new_x = np.linspace(0, 1, n)
            upsampled = np.interp(new_x, old_x, old_a)
            upsampled = np.maximum(upsampled, 0.0)
            upsampled = upsampled / upsampled.sum() * target_sum
            seeds = [upsampled]

        # best for this length
        best_local_seq = None
        best_local_score = float('inf')

        # run optimizer for each seed until time
        for init_a in seeds:
            if time.time() >= deadline:
                break
            # ensure non-negative and sum normalized
            init_a = np.maximum(init_a, 0.0)
            if init_a.sum() == 0:
                init_a = np.full_like(init_a, target_sum / n)

                init_a = init_a * (target_sum / init_a.sum())

    v = np.zeros_like(a)
    beta1=0.9
    beta2=0.999
    eps=1e-8
    lr0=0.1
    decay=5e-5
    iteration=0
    no_improve=0
    best_local_score=best_score

    while time.time() < deadline:
        # conv max
        conv = np.fft.rfft(a, n=fft_n)
        conv_full = np.fft.irfft(conv*conv, n=fft_n)[:2*n-1]
        M = conv_full.max()
        k_max = int(conv_full.argmax())
        # rev = subgradient for single active
        i = np.arange(n)
        j = k_max - i
        mask = (j>=0) & (j<n)
        rev = np.zeros_like(a)
        rev[mask] = a[j[mask]]
        coeff = 4.0 * n / (target_sum**2)
        const = 4.0 * n * M / (target_sum**3)
        grad = coeff * rev - const
        # Adam update
        iteration+=1
        m = beta1 * m + (1-beta1) * grad
        v = beta2 * v + (1-beta2) * (grad*grad)
        m_hat = m / (1 - beta1**iteration)
        v_hat = v / (1 - beta2**iteration)
        lr = lr0/(1+decay*iteration)

        if s > 0:

"""
Propose step‑function coefficients with a multi‑scale Adam optimiser.

The algorithm works as follows
--------------------------------
1.  Choose a list of increasing lengths (up to ~8000).  Starting from a
    short seed we gradually enlarge the vector by linear interpolation.
2.  For each length we generate several diverse initialisations
    (uniform, random, Gaussian with several widths, triangular, exponential).
3.  Every initialisation is refined by a projected Adam optimiser that
    minimises the maximal self‑convolution value M.
    * The gradient is the exact sub‑gradient of the max‑convolution
      (a single active shift is used – ties are rare).
    * After each update we project onto the non‑negative orthant and
      renormalise the sum to √(2 n) (the normalisation makes the objective
      equal to M).
    * An occasional tiny Gaussian jitter helps to escape plateaus.
    * Early stopping is performed when no improvement has been seen for
      many iterations.
4.  The best vector found over all runs (and all lengths) is returned.
   The whole search respects the 1000 s wall‑clock limit.

The implementation only uses numpy and the already‑imported
``evaluate_sequence`` function.
"""

import time
import math
import numpy as np


def _conv_max_fft(a: np.ndarray):
    """
    Compute the full self‑convolution a*a using an FFT of the next power
    of two, and return (max_value, index_of_max, full_convolution_array).
    """
    n = a.shape[0]
    # FFT length – next power of two ≥ 2*n‑1
    N = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=N)
    B = A * A
    conv = np.fft.irfft(B, n=N)[: 2 * n - 1]
    k_max = int(conv.argmax())
    return float(conv.max()), k_max, conv


def _subgradient_one(a: np.ndarray, k: int):
    """
    Sub‑gradient of the max‑convolution when the active shift is k.
    Returns a vector rev where rev[i] = a[k‑i] if the index is inside
    the array, otherwise 0.
    """
    n = a.shape[0]
    rev = np.zeros_like(a)
    i = np.arange(n)
    j = k - i
    mask = (j >= 0) & (j < n)
    rev[mask] = a[j[mask]]
    return rev


def _adam_optimize_length(
    n: int,
    init_a: np.ndarray,
    global_best_seq: np.ndarray,
    global_best_score: float,
    deadline: float,
):
    """
    Run an Adam optimiser on a vector of length ``n`` starting from
    ``init_a``.  The function returns the best (seq, score) found during
    this run.  ``global_best_seq``/``global_best_score`` are only used
    for occasional printing – the optimiser never modifies them.
    """
    target_sum = math.sqrt(2.0 * n)

    # normalise the starting vector
    a = init_a.copy()
    if a.sum() == 0:
        a = np.full_like(a, target_sum / n)
    else:
        a = a * (target_sum / a.sum())

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.12          # initial learning rate
    decay = 4e-5        # controls the decay of the learning rate
    jitter_interval = 4000
    iteration = 0
    no_improve = 0
    best_local_seq = a.copy()
    best_local_score = float("inf")

    while time.time() < deadline:
        # ----- 1) convolution and active shift -----
        M, k_max, conv_full = _conv_max_fft(a)

        # active set – we use the single maximal shift (ties are extremely rare)
        rev = _subgradient_one(a, k_max)

        # ----- 2) sub‑gradient of the objective (which equals M) -----
        coeff = 4.0 * n / (target_sum * target_sum)
        const = 4.0 * n * M / (target_sum ** 3)
        grad = coeff * rev - const

        # ----- 3) Adam update -----
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        lr = lr0 / (1.0 + decay * iteration)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # ----- 4) projection onto feasible set -----
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a = a * (target_sum / s)
        else:
            a = np.full_like(a, target_sum / n)

        # ----- 5) occasional jitter -----
        if iteration % jitter_interval == 0:
            jitter = np.random.randn(n) * (target_sum / (n * 800.0))
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        # ----- 6) keep the best we have seen -----
        cur_score = M   # because of the normalisation the objective equals M
        if cur_score < best_local_score:
            best_local_score = cur_score
            best_local_seq = a.copy()
            no_improve = 0
            if cur_score < global_best_score:
                # print only when we beat the *global* best
                print(
                    f"[len {n:5d} iter {iteration:7d}] "
                    f"new global best = {cur_score:.6f}"
                )
                global_best_score = cur_score
                global_best_seq = a.copy()
        else:
            no_improve += 1

        # early stop if stagnated for a long time
        if no_improve > 20000:
            break

    return best_local_seq, best_local_score


def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The function respects the given time budget and returns the best
    sequence (as a plain Python list) found so far.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin so the harness can still finish cleanly
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------
    # Load the previously known construction if it exists
    try:
        base = np.array(height_sequence_1, dtype=float)  # type: ignore
        base_len = base.shape[0]
    except Exception:
        base = None
        base_len = None

    # ------------------------------------------------------------
    # Choose a cascade of lengths – start small, finish with a large one
    length_chain = [800, 1500, 3000, 5000, 8000]
    # ensure we never exceed the allowed runtime
    length_chain = [l for l in length_chain if l >= 300]

    # ------------------------------------------------------------
    # Global best tracking
    best_seq = None
    best_score = float("inf")
    best_len = None

    # ------------------------------------------------------------
    # Helper: generate a few diverse seeds for a given length
    def _make_seeds(n):
        """Return a list of initial vectors (numpy arrays) for length n."""
        target_sum = math.sqrt(2.0 * n)
        seeds = []

        # 1. uniform vector
        seeds.append(np.full(n, target_sum / n))

        # 2. random Dirichlet‑like vector
        r = np.random.rand(n)
        r = r / r.sum() * target_sum
        seeds.append(r)

        # 3. a few Gaussians with different widths
        centre = (n - 1) / 2.0
        x = np.arange(n)
        for sigma_factor in (0.08, 0.12, 0.16, 0.20):
            sigma = sigma_factor * n
            g = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
            g = g / g.sum() * target_sum
            seeds.append(g)

        # 4. a simple triangular shape
        tri = np.concatenate([np.arange(1, n // 2 + 1),
                              np.arange(n // 2, 0, -1)])
        tri = tri.astype(float)
        tri = tri / tri.sum() * target_sum
        seeds.append(tri)

        # 5. exponential decay
        exp = np.exp(-np.arange(n) / (0.2 * n))
        exp = exp / exp.sum() * target_sum
        seeds.append(exp)

        return seeds

    # ------------------------------------------------------------
    # Main multi‑scale optimisation loop
    current_best = None   # best vector from the previous (smaller) scale
    current_len = None

    for n in length_chain:
        if time.time() >= deadline:
            break

        print(f"[stage] Optimising length {n}")

        # Produce seeds: either upscale the previous best or generate fresh ones
        if current_best is not None:
            # Upscale via linear interpolation
            old_x = np.linspace(0.0, 1.0, current_len)
            new_x = np.linspace(0.0, 1.0, n)
            upscaled = np.interp(new_x, old_x, current_best)
            upscaled = np.maximum(upscaled, 0.0)
            upscaled = upscaled / upscaled.sum() * math.sqrt(2.0 * n)
            seeds = [upscaled]
        else:
            # No previous scale – use the original construction if present,
            # otherwise fall back to generated seeds.
            if base is not None and base_len == n:
                seeds = [base.copy()]
            elif base is not None:
                # interpolate the supplied base to the required length
                old_x = np.linspace(0.0, 1.0, base_len)
                new_x = np.linspace(0.0, 1.0, n)
                inter = np.interp(new_x, old_x, base)
                inter = np.maximum(inter, 0.0)
                inter = inter / inter.sum() * math.sqrt(2.0 * n)
                seeds = [inter]
            else:
                seeds = _make_seeds(n)

        # Run a short optimisation for each seed
        # Allocate a fraction of the remaining time to this length

        # we give at most 40 % of the remaining budget to the current length,
        # the rest will be kept for larger scales.

    if best_seq is None:

        targ = math.sqrt(2.0 * n)
        best_seq = np.full(n, targ / n)

    print(f"[done] best score = {best_score:.6f}, length = {best_len}")
    return best_seq.tolist()
