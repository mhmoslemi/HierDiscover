"""
Propose a non‑negative step‑function (probability vector) that minimizes
the evaluation function `evaluate_sequence`.  The algorithm works in three
phases:

1. **Diverse seeding** – for several problem sizes (n = 500 … 8000) a
   handful of parametric families (geometric, power‑law, beta‑like,
   Gaussian, cosine‑bump, random Dirichlet) are generated and the best
   few are kept as seeds.

2. **Local optimisation** – each seed is refined by a hybrid
   projected‑gradient / simulated‑annealing loop:
   * the active convolution shift(s) are identified;
   * a (sub‑)gradient step is taken on the simplex;
   * if no improvement occurs a random mass‑transfer move is tried;
   * worse moves can be accepted with a temperature that decays over
     iterations, allowing the search to escape local plateaus.

3. **Selection** – the best sequence found over all lengths and seeds is
   returned as a plain Python list.  All work respects the 1000 s time
   budget.

The implementation uses an FFT‑based convolution (O(n log n)) and a fast
projection onto the probability simplex.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast utilities
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _self_convolve(seq: np.ndarray) -> np.ndarray:
    """Full convolution a*a (length 2*len(a)-1) using real FFT."""
    n = seq.shape[0]
    m = 2 * n - 1
    nfft = _next_pow2(m)
    A = np.fft.rfft(seq, n=nfft)
    conv = np.fft.irfft(A * A, n=nfft)[:m]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Local replica of evaluate_sequence (faster, no clipping)."""
    sum_a = seq.sum()
    if sum_a < 0.01:
        return float("inf")
    conv = _self_convolve(seq)
    max_b = conv.max()
    return float(2.0 * len(seq) * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Projection onto probability simplex {x ≥ 0, sum(x)=1}
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families (all return a *probability* vector, i.e. sum = 1)
def _geom_seq(n: int, alpha: float) -> np.ndarray | None:
    """Geometric progression with ratio r = 1 - alpha/n."""
    eps = alpha / n
    r = 1.0 - eps
    if r <= 0.0:
        return None
    powers = r ** np.arange(n, dtype=np.float64)
    seq = powers / powers.sum()
    return seq


def _powerlaw_seq(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    seq = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    return seq / seq.sum()


def _beta_seq(n: int, a: float, b: float) -> np.ndarray:
    """Beta‑like shape a_i ∝ (i+1)^{a-1} (n-i)^{b-1}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    seq = (i ** (a - 1.0)) * ((n + 1 - i) ** (b - 1.0))
    seq[seq < 0] = 0.0
    return seq / seq.sum()


def _gaussian_seq(n: int, sigma: float) -> np.ndarray:
    """Discrete Gaussian centred in the middle of the array."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    seq = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    return seq / seq.sum()


def _cosine_bump_seq(n: int, width: float) -> np.ndarray:
    """Half‑cosine bump; width ∈ (0,1] is the fraction of the support."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    half_width = width * (n - 1) / 2.0
    d = np.abs(i - centre) / half_width
    seq = np.clip(0.5 * (1 + np.cos(np.pi * d)), 0, None)
    return seq / seq.sum()


def _triangular_seq(n: int) -> np.ndarray:
    """Symmetric triangle rising then falling."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    seq = 1.0 - np.abs(i - centre) / centre
    seq[seq < 0] = 0.0
    return seq / seq.sum()


def _random_simplex(n: int) -> np.ndarray:
    """Draw a random point on the simplex (Dirichlet(1,…,1))."""
    x = np.random.exponential(1.0, size=n)
    return x / x.sum()


# ----------------------------------------------------------------------
# Local optimiser: projected sub‑gradient + simulated annealing
def _local_optimize(init_seq: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Refine `init_seq` (already normalised) until `deadline`.
    Returns (best_seq, best_score).
    """
    n = init_seq.shape[0]
    cur = init_seq.copy()
    cur_score = _objective(cur)
    best = cur.copy()
    best_score = cur_score

    it = 0
    # initial step for gradient moves
    step_size = 1.0
    # temperature schedule (initial temperature proportional to current score)
    T0 = 0.02 * cur_score if cur_score > 0 else 1e-3

    while time.time() < deadline:
        # --------------------------------------------------------------
        # 1) compute convolution and identify a set of almost‑active shifts
        conv = _self_convolve(cur)
        max_val = conv.max()
        # consider all shifts whose value is within 0.1 % of the maximum
        active = np.where(conv >= max_val * (1.0 - 1e-3))[0]

        # --------------------------------------------------------------
        # 2) build a sub‑gradient using all active shifts (averaged)
        grad = np.zeros_like(cur)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=int)
            # derivative contribution: 2 * a_{k-i} (except at exact centre)
            grad[idx] += 2.0 * cur[k - idx]
        # normalise gradient magnitude to avoid overly large steps
        norm = np.linalg.norm(grad)
        if norm > 0:
            grad = grad / norm

        # --------------------------------------------------------------
        # 3) try a projected gradient step
        cand = _project_onto_simplex(cur - step_size * grad)
        cand_score = _objective(cand)

        # temperature for this iteration (linear cooling)
        T = max(T0 * (1.0 - it / 20000.0), 1e-6)

        improved = False
        if cand_score < cur_score:
            cur, cur_score = cand, cand_score
            improved = True
        else:
            # accept worse move with SA probability
            if np.random.rand() < np.exp(-(cand_score - cur_score) / T):
                cur, cur_score = cand, cand_score

        # --------------------------------------------------------------
        # 4) occasional random mass‑transfer move (helps escape plateaus)
        if np.random.rand() < 0.03:
            i, j = np.random.choice(n, size=2, replace=False)
            delta = np.random.uniform(0.0, min(0.01, cur[i]))
            tmp = cur.copy()
            tmp[i] -= delta
            tmp[j] += delta
            tmp = _project_onto_simplex(tmp)
            tmp_score = _objective(tmp)
            if tmp_score < cur_score or np.random.rand() < 0.001:
                cur, cur_score = tmp, tmp_score

        # --------------------------------------------------------------
        # 5) bookkeeping
        if cur_score < best_score:
            best, best_score = cur.copy(), cur_score
            print(f"[Iter {it:05d}] new global best = {best_score:.6f}")

        # adapt step size (simple heuristic)
        if improved:
            step_size = min(step_size * 1.05, 5.0)
        else:
            step_size = max(step_size * 0.95, 1e-4)

        it += 1

    return best, best_score

# Main entry point required by the evaluation harness
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a step‑function with a low objective value.
    The function respects the provided time budget and returns
    a list of non‑negative floats (the heights).
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin for finalisation
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # 1) possible problem sizes (larger n gives a lower theoretical bound)
    candidate_lengths = [500, 1000, 2000, 4000, 8000]

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 2) use a global seed if present (the previous runner stores it in a global)
    if "height_sequence_1" in globals():
        try:
            glob = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            glob = np.clip(glob, 0.0, 1000.0)
            if glob.sum() > 0:
                glob = glob / glob.sum()
                s = _objective(glob)
                if s < best_score:
                    best_score, best_seq = s, glob.copy()
                    print(f"[Init] global seed score={s:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 3) iterate over lengths, generate seeds, refine them
    for n in candidate_lengths:
        if time.time() > deadline:
            break

        # ----------- generate a palette of seeds -----------------------
        seeds = []

        # uniform (always a valid baseline)
        seeds.append(np.full(n, 1.0 / n))

        # geometric family
        for alpha in np.linspace(0.5, 4.0, 9):
            seq = _geom_seq(n, alpha)
            if seq is not None:
                seeds.append(seq)

        # power‑law family
        for p in np.linspace(0.1, 2.5, 9):
            seeds.append(_powerlaw_seq(n, p))

        # beta‑like family (coarse sweep)
        for a in np.linspace(0.5, 3.0, 4):
            for b in np.linspace(0.5, 3.0, 4):
                seeds.append(_beta_seq(n, a, b))

        # Gaussian shapes
        for sigma in np.linspace(0.1, n / 2.0, 5):
            seeds.append(_gaussian_seq(n, sigma))

        # cosine bump
        for width in np.linspace(0.2, 1.0, 5):
            seeds.append(_cosine_bump_seq(n, width))

        # triangular
        seeds.append(_triangular_seq(n))

        # random Dirichlet samples
        for _ in range(15):
            seeds.append(_random_simplex(n))

        # evaluate all seeds quickly and keep the top few
        scored = [( _objective(s), s ) for s in seeds]
        scored.sort(key=lambda x: x[0])
        top_seeds = [s for _, s in scored[:4]]

        # ----------- run local optimiser on each top seed -------------
        for seed_seq in top_seeds:
            if time.time() > deadline:
                break
            # give each seed a slice of the remaining time (≈ 10 % of total)
            remaining = deadline - time.time()
            if remaining <= 0:
                break
            # allocate at most 30 s per seed (or the remaining time, whichever is smaller)
            seed_deadline = time.time() + min(30.0, remaining * 0.25)
            opt_seq, opt_score = _local_optimize(seed_seq, seed_deadline)
            if opt_score < best_score:
                best_score, best_seq = opt_score, opt_seq.copy()
                print(f"[n={n}] *** improved global best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) fallback – should never happen, but keep a safe uniform vector
    if best_seq is None:
        n = 2000
        best_seq = np.full(n, 1.0 / n)
        best_score = _objective(best_seq)

    # ------------------------------------------------------------------
    # 5) final post‑processing (respect clipping limits of evaluate_sequence)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Return as plain list of floats
    print(f"Final best objective = {best_score:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
