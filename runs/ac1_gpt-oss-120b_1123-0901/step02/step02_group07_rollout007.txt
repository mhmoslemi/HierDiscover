"""
propose_candidate
-----------------

A smoothed‑max (soft‑max) gradient descent on the probability simplex.
The objective `2·n·max(conv)/sum^2` is scale‑invariant, therefore we keep
the sequence normalized to sum = 1.  The algorithm proceeds in three
phases:

1. **Warm‑start** – generate a few hundred candidates from simple
   parametric families (geometric, power‑law, beta‑like) and from a
   random Dirichlet point.  The best of these seeds the local optimiser.
2. **Smooth‑max optimisation** – replace the non‑differentiable
   `max(conv)` by a soft‑max
   `g(a)= (1/β)·log Σ exp(β·conv_k)`.  Its gradient is
   `∇g = 2·a ⋆ w`, where `w` are the soft‑max weights and `⋆` denotes
   cross‑correlation.  The gradient is obtained with an FFT‑based
   convolution, then an Adam‑style update is taken and the result is
   projected back onto the simplex.
   The smoothing parameter `β` is increased linearly during the run,
   turning the smooth objective into the true max as the optimisation
   proceeds.
3. **Fallback moves** – if the Adam step does not improve the real
   (hard) objective, a back‑tracking line‑search on the raw gradient is
   tried, followed by a tiny random mass‑transfer move.

The whole routine respects the 1000 s budget, prints intermediate
best scores and finally returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
# ----------------------------------------------------------------------
def _project_onto_simplex(v):
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        # all entries are zero; return uniform vector
        return np.full_like(v, 1.0 / len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


# ----------------------------------------------------------------------
# Convolutions (FFT for speed)
# ----------------------------------------------------------------------
def _self_convolve(seq: np.ndarray) -> np.ndarray:
    """Full convolution seq * seq (length 2·n‑1) using a real FFT."""
    n = seq.shape[0]
    m = 2 * n - 1
    nfft = _next_pow2(m)
    A = np.fft.rfft(seq, n=nfft)
    conv = np.fft.irfft(A * A, n=nfft)[:m]
    return conv


def _cross_corr(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Cross‑correlation h_i = Σ_j w_{i+j}·a_j  (i = 0…n‑1).
    Equivalent to the slice `np.convolve(a[::-1], w)[n-1:n-1+n]` but
    computed with FFT for O(n log n) time.
    """
    n = a.shape[0]
    m = w.shape[0]                 # should be 2·n‑1
    nfft = _next_pow2(n + m - 1)
    A = np.fft.rfft(a[::-1], n=nfft)
    W = np.fft.rfft(w, n=nfft)
    conv = np.fft.irfft(A * W, n=nfft)   # length nfft
    start = n - 1
    return conv[start:start + n]


# ----------------------------------------------------------------------
# Objective (local fast version, identical to evaluate_sequence up to the
# scaling invariance)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """Return 2·n·max(conv)/sum(seq)^2 for a non‑negative seq."""
    sum_a = seq.sum()
    if sum_a < 0.01:
        return float("inf")
    conv = _self_convolve(seq)
    max_b = conv.max()
    return float(2.0 * len(seq) * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Parametric families for the initial sweep
# ----------------------------------------------------------------------
def _geom_seq(n: int, alpha: float) -> np.ndarray | None:
    """Geometric progression a_i ∝ r^i with r = 1−α/n."""
    eps = alpha / n
    r = 1.0 - eps
    if r <= 0.0:
        return None
    powers = r ** np.arange(n, dtype=np.float64)
    seq = powers / powers.sum()
    return seq


def _powerlaw_seq(n: int, p: float) -> np.ndarray:
    """Power‑law a_i ∝ (i+1)^{−p}."""
    seq = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    return seq / seq.sum()


def _beta_seq(n: int, a: float, b: float) -> np.ndarray:
    """Beta‑like shape a_i ∝ (i+1)^{a−1}·(n−i)^{b−1}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    seq = (i ** (a - 1.0)) * ((n + 1 - i) ** (b - 1.0))
    seq[seq < 0] = 0.0
    return seq / seq.sum()


def _random_simplex(n: int) -> np.ndarray:
    """Draw a random point on the simplex (Dirichlet(1,…,1))."""
    x = np.random.exponential(1.0, size=n)
    return x / x.sum()

# Main routine
# ----------------------------------------------------------------------

    within the allotted time.

    # Use the global seed if it exists.
