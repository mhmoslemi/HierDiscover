    conv = np.fft.irfft(fa * fa, n=size)

"""
Propose a step‑function (non‑negative height sequence) that minimises

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm consists of three stages:

1.  A few cheap parametric seeds (exponential, triangular, …) are
    normalised onto the probability simplex Σa_i = 1 and evaluated.
    The best seed becomes the initial “best”.

2.  A modern annealed gradient descent:
    - The objective `max(conv)` is smoothed by a log‑sum‑exp
      (temperature λ).  The gradient of the smoothed objective is
      computed analytically and efficiently with FFT‑based
      correlation.
    - Momentum (β=0.9) and a diminishing step size provide stability.
    - λ is increased geometrically, turning the smooth objective into
      the true max as optimisation proceeds.
    - After each update the vector is projected onto the simplex
      (non‑negative, Σa_i = 1).  Several random restarts with different
      lengths (500‑2500) are performed until the time budget expires or
      a target score (≈1.505) is reached.

3.  If SciPy is available, a single LP‑refinement (the “LP‑step” from the
    state‑of‑the‑art description) is executed on the current best
    sequence to squeeze a little more performance.

All helper routines are defined at module level (no closures).  The
function returns a plain ``list[float]`` ready for the evaluation harness.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper 1 – Euclidean projection onto the simplex {x ≥ 0, Σx = s}
# ----------------------------------------------------------------------
def project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex with sum `s` (non‑negative)."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper 2 – full self‑convolution via real FFT (fast for n up to a few k)
# ----------------------------------------------------------------------
def conv_full_fft(a: np.ndarray) -> np.ndarray:
    """Return the full convolution a * a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Helper 3 – smooth‑max gradient (log‑sum‑exp) using FFT for the correlation
# ----------------------------------------------------------------------
def gradient_and_conv(a: np.ndarray, lam: float):
    """
    For a vector a (already on the simplex) and a temperature λ,
    return (grad, conv) where:
        conv = a * a (full convolution)
        grad = ∇_a ( (1/λ) log Σ_k exp(λ * conv_k) )
    The gradient is computed with an FFT‑based correlation, O(n log n).
    """
    # 1) full convolution
    conv = conv_full_fft(a)                     # length 2n‑1

    # 2) soft‑max weights w_k ∝ exp(λ * conv_k)
    rc = lam * conv
    rc_max = rc.max()
    w = np.exp(rc - rc_max)                     # shift for numerical stability
    w /= w.sum()                                 # now Σ w = 1

    # 3) gradient = 2 * (a ⋆ w)   (valid cross‑correlation)
    n = a.shape[0]
    size = 1 << ((3 * n - 2).bit_length())      # power of two ≥ (2n‑1)+n‑1
    a_rev = a[::-1]                             # reversed a
    Fa = np.fft.rfft(a_rev, n=size)
    Fw = np.fft.rfft(w, n=size)
    corr = np.fft.irfft(Fa * Fw, n=size)        # length = size
    start = n - 1                               # first index of the “valid” part
    grad = 2.0 * corr[start:start + n]          # shape (n,)
    return grad, conv


# ----------------------------------------------------------------------
# Helper 4 – safe evaluation wrapper
# ----------------------------------------------------------------------
def safe_evaluate(seq) -> float:
    """Call the provided evaluate_sequence, returning +inf on error."""
    try:
        return evaluate_sequence(seq)                # the harness supplies this
    except Exception:                               # pragma: no cover
        return float("inf")

    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # 1️⃣  Quick parametric seeds (cheap, give a decent starting point)
    # --------------------------------------------------------------
    lengths = [200, 400, 600, 800, 1000, 1200, 1400, 1600]
    for n in lengths:
        if time.time() >= deadline:
            break
        # exponential decay
        for alpha in np.logspace(-4, -0.5, num=6):

            a = project_onto_simplex(a, 1.0)

        a = project_onto_simplex(a, 1.0)

            best_score = score
            best_seq = a.copy()
            print(f"[{time.time():.2f}] seed (tri) len={n} score={best_score:.6f}")

        except Exception:  # pragma: no cover

            print(f"[{time.time():.2f}] restart finished best {best_score:.6f} (len={n})")
        if best_score <= 1.505:
            break

    # --------------------------------------------------------------
    # 3️⃣  Optional linear‑programming refinement (if SciPy is present)
    # --------------------------------------------------------------
    try:
        from scipy.optimize import linprog
    except Exception:                     # pragma: no cover
        linprog = None

    if linprog is not None and best_seq is not None and time.time() < deadline:
        a = best_seq.copy()
        n = a.shape[0]

        # Build the (a * g) ≤ M constraints – matrix with 2n‑1 rows, n columns
        conv = conv_full_fft(a)
        M = conv.max()
        rows = 2 * n - 1
        A = np.zeros((rows, n), dtype=float)
        for k in range(rows):
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            cols = np.arange(lo, hi + 1, dtype=int)
            A[k, cols] = a[k - cols]

        # Linear programme: maximise Σ g_j   (i.e. minimise –Σ g_j)
        res = linprog(c=-np.ones(n),
                      A_ub=A,
                      b_ub=np.full(rows, M),
                      bounds=(0, None),
                      method="highs",
                      options={"presolve": True})

        if res.success:
            g = res.x
            g = project_onto_simplex(g, 1.0)          # keep the normalisation
            t = 0.2                                    # convex‑combination weight
            a = (1 - t) * a + t * g                     # blend
            score = safe_evaluate(a.tolist())
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"[{time.time():.2f}] LP refinement improved to {best_score:.6f}")

    # --------------------------------------------------------------
    # Fallback – uniform distribution if nothing else succeeded
    # --------------------------------------------------------------
    if best_seq is None:
        n = 1000
        best_seq = np.full(n, 1.0 / n, dtype=float)

    return best_seq.tolist()
