"""
Propose a step‑function (non‑negative heights) that minimizes

        2 * n * max(conv(a, a)) / (sum(a) ** 2)

The implementation combines three ideas:

1.  A fast simplex projector (keeps Σa = √(2n) and a_i ≥ 0).
2.  **Simulated annealing** – random mass transfers between two positions,
    accepted with a decreasing temperature schedule.  This gives a
    global‑search capability and helps the algorithm escape local minima.
3.  A **projected sub‑gradient descent** on the active (maximal)
    convolution entries – a fast local optimiser that refinements the
    result obtained by the annealer.

The outer loop repeatedly restarts the procedure with different lengths
(≈1200–3000) and different initialisations (including the best construction
found so far).  The best sequence discovered before the time budget expires
is returned as a plain Python list.
"""

import time
import numpy as np

# --------------------------------------------------------------------------- #
# Helper: Euclidean projection onto the simplex { x_i ≥ 0 , Σ x_i = s }.
# --------------------------------------------------------------------------- #
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector ``v`` onto the simplex of radius ``s``."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------------------- #
# Fast full convolution (real FFT) – returns a length 2*n‑1 array.
# --------------------------------------------------------------------------- #
def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    # next power of two for efficient FFT
    n_fft = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    conv = np.fft.irfft(A * A, n=n_fft)[: 2 * n - 1]
    return conv


# --------------------------------------------------------------------------- #
# Evaluate a vector – because we keep Σa = √(2n) the objective equals max(conv).
# --------------------------------------------------------------------------- #
def _max_conv(a: np.ndarray) -> float:
    return float(_conv_full_fft(a).max())


# --------------------------------------------------------------------------- #
# Simulated annealing – random mass transfers between two positions.
# --------------------------------------------------------------------------- #
def simulated_annealing(a_init: np.ndarray,
                        target_sum: float,
                        max_time: float,
                        start_time: float):
    """Return (best_vector, best_score) after SA of duration max_time."""
    a = a_init.copy()
    n = a.shape[0]

    cur_score = _max_conv(a)
    best_vec = a.copy()
    best_score = cur_score

    # temperature schedule
    T0 = max(0.01, cur_score * 0.05)   # start about 5 % of the current value
    T_min = 1e-6

    while time.time() < start_time + max_time:
        # pick two distinct indices
        i = np.random.randint(0, n)
        j = np.random.randint(0, n - 1)
        if j >= i:
            j += 1
        if a[i] <= 0.0:               # nothing to move from i
            continue

        # move a random fraction (up to 50 %) of a[i] from i to j
        frac = np.random.rand() * 0.5
        delta = a[i] * frac

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        # keep the simplex exactly (tiny numeric drift may occur)
        a_new = _project_onto_simplex(a_new, target_sum)

        new_score = _max_conv(a_new)
        delta_score = new_score - cur_score

        elapsed = time.time() - start_time
        frac_time = max(0.0, 1.0 - elapsed / max_time)
        T = T_min + T0 * frac_time

        accept = False
        if delta_score <= 0.0:
            accept = True
        else:
            # Metropolis acceptance
            prob = np.exp(-delta_score / (T + 1e-12))
            accept = np.random.rand() < prob

        if accept:
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_vec = a.copy()
                print(f"[{time.time():.2f}] SA new best = {best_score:.6f} (n={n})")

    return best_vec, best_score


# --------------------------------------------------------------------------- #
# Projected sub‑gradient descent on the active convolution entries.
# --------------------------------------------------------------------------- #
def subgradient_optimize(a_init: np.ndarray,
                         target_sum: float,
                         max_time: float,
                         start_time: float):
    """Refine a vector using a sub‑gradient method."""
    a = a_init.copy()
    n = a.shape[0]
    best_vec = a.copy()
    best_score = _max_conv(a)

    base_lr = 0.2          # base learning rate
    momentum = np.zeros_like(a)
    beta_mom = 0.9

    while time.time() < start_time + max_time:
        conv = _conv_full_fft(a)
        max_val = conv.max()
        cur_score = float(max_val)

        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()
            print(f"[{time.time():.2f}] GD new best = {best_score:.6f} (n={n})")

        # active (max‑tying) indices – tolerance is tiny because ties are exact
        eps = 1e-12
        active = np.where(conv >= max_val - eps)[0]

        # gradient of the sum of the active entries
        grad = np.zeros_like(a)
        for k in active:
            low = max(0, k - (n - 1))
            high = min(n - 1, k)
            idx = np.arange(low, high + 1, dtype=int)
            partner = k - idx
            grad[idx] += 2.0 * a[partner]

        max_grad = grad.max()
        if max_grad <= 0.0:
            break

        # step size – similar spirit to the original method
        eta = base_lr * max_val / max_grad

        # heavy‑ball momentum update
        momentum = beta_mom * momentum - eta * grad
        a_new = a + momentum

        # enforce feasibility
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_onto_simplex(a_new, target_sum)

        # occasional random shake (simulated‑annealing style)
        if np.random.rand() < 0.02:
            noise = np.random.randn(n) * (0.01 * target_sum / n)
            a_new = _project_onto_simplex(np.maximum(a_new + noise, 0.0), target_sum)
            base_lr *= 0.9

        a = a_new
        base_lr *= 0.99995   # mild decay for final fine‑tuning

    return best_vec, best_score


# --------------------------------------------------------------------------- #
# Main entry point expected by the grader.
# --------------------------------------------------------------------------- #
def propose_candidate(seed: int = 42,
                     budget_s: float = 1000.0,
                     **kwargs):
    """
    Search for a low‑value step‑function within ``budget_s`` seconds.
    Returns a plain Python list of non‑negative heights.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5          # small safety margin

    best_seq = None            # numpy array of the best candidate
    best_score = float("inf")  # objective value (max convolution)

    # ------------------------------------------------------------------- #
    # Helper to evaluate a candidate (fast, no extra clipping)
    # ------------------------------------------------------------------- #
    def _evaluate(a):
        return _max_conv(a)

    # ------------------------------------------------------------------- #
    # Outer loop – many restarts with varying lengths.
    # ------------------------------------------------------------------- #
    while time.time() < deadline:
        # Decide on a problem size.
        if best_seq is not None and np.random.rand() < 0.2:
            n = len(best_seq)
        else:
            n = np.random.randint(1200, 3001)

        target_sum = np.sqrt(2.0 * n)   # Σ a_i = √(2n)

        # --------------------------------------------------------------- #
        # Initialise a candidate.
        # --------------------------------------------------------------- #
        if "height_sequence_1" in globals() and np.random.rand() < 0.5:
            src = np.asarray(height_sequence_1, dtype=float)
            if src.size >= n:
                init = src[:n].copy()
            else:
                pad = np.random.rand(n - src.size) * 1e-3
                init = np.concatenate([src, pad])
        else:
            # U‑shaped seed: larger values near the ends, smaller in the middle.
            idx = np.arange(n, dtype=float)
            weight = 1.0 / (idx + 1) + 1.0 / (n - idx)
            init = np.random.rand(n) * weight

        a = _project_onto_simplex(init.astype(float), target_sum)

        # Quick check: update global best with the raw initial point.
        cur_score = _evaluate(a)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.2f}] Init best = {best_score:.6f} (n={n})")

        # --------------------------------------------------------------- #
        # Allocate time for this restart (≈12 s split 6 s SA / 6 s GD).
        # --------------------------------------------------------------- #
        time_left = deadline - time.time()
        if time_left <= 0:
            break
        phase_time = min(6.0, time_left * 0.5)

        # Simulated annealing stage
        a_sa, score_sa = simulated_annealing(a, target_sum, phase_time, time.time())
        if score_sa < best_score:
            best_score = score_sa
            best_seq = a_sa.copy()
            print(f"[{time.time():.2f}] After SA best = {best_score:.6f} (n={n})")

        # Gradient‑descent stage (use remaining time up to another 6 s)
        time_left = deadline - time.time()
        if time_left <= 0:
            break
        phase_time = min(6.0, time_left)

        a_gd, score_gd = subgradient_optimize(a_sa, target_sum,
                                              phase_time, time.time())
        if score_gd < best_score:
            best_score = score_gd
            best_seq = a_gd.copy()
            print(f"[{time.time():.2f}] After GD best = {best_score:.6f} (n={n})")

        # Early exit if we already beat the target bound.
        if best_score < 1.503:
            break

    # ------------------------------------------------------------------- #
    # Fallback – uniform vector if nothing was found (should never happen).
    # ------------------------------------------------------------------- #
    if best_seq is None:
        n = 1500
        target_sum = np.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)

    # Return a plain Python list as required by the grader.
    return best_seq.tolist()
