"""
Propose a non‑negative step‑function (vector) with a low value of

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm consists of two phases.

Phase 1 – Exploration:
    * Randomly sample a length n (800 … 5000) and a power‑law
      initialization a_i ∝ (i+δ)^{-p}.
    * Run a fast projected Adam optimiser that works on the
      “soft‑max” approximation of the max‑convolution.
    * Keep the best vector found among many short restarts.

Phase 2 – Refinement:
    * Take the current global best and continue optimisation for the
      remaining time, using a slower learning schedule and a higher
      soft‑max temperature so that the updates become more precise.

All updates are projected onto the simplex
    {x_i ≥ 0, Σ x_i = √(2 n)}.
The objective is scale‑invariant; with this normalisation the score
equals the maximal convolution value, so we can minimise ``max(conv)`` 
directly.

The code uses only NumPy (FFT based convolutions, O(n log n) per step)
and stays well within the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper – Euclidean projection onto the simplex Σ x_i = s, x_i ≥ 0
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex of radius s."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper – linear convolution via FFT (real inputs → real output)
# ----------------------------------------------------------------------
def _conv_fft_same(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using an FFT of power‑of‑2 size."""
    n = a.shape[0]
    m = 2 * n - 1
    size = 1 << ((m - 1).bit_length())  # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[:m]
    # Clip tiny negatives that may appear because of rounding
    np.maximum(conv, 0.0, out=conv)
    return conv


def _conv_fft_two(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two (real) vectors using FFT."""
    n, m = x.shape[0], y.shape[0]
    size = 1 << ((n + m - 2).bit_length())
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[: n + m - 1]
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Helper – compute gradient from soft‑max weights
# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Given a vector a (size n) and a weight vector w (size 2n‑1),
    compute the gradient of Σ_k w_k (a*a)_k with respect to a.
    The formula is   grad_i = 2 * Σ_k w_k a_{k-i}
    = 2 * correlation(a, w)[i].
    """
    # correlation(a, w) = convolution(a, w_rev) shifted by len(w)-1
    w_rev = w[::-1]
    conv = _conv_fft_two(a, w_rev)
    start = len(w_rev) - 1
    grad = 2.0 * conv[start:start + a.shape[0]]
    return grad


# ----------------------------------------------------------------------
# Core optimiser for a fixed length n
# ----------------------------------------------------------------------
def _optimise_one_length(
    n: int,
    init: np.ndarray,
    run_deadline: float,
    *,
    final_phase: bool = False,
) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimiser that minimises max(convolution(a, a)).
    Returns the best vector seen and its score.
    """
    # Target sum that makes the objective equal to the maximal convolution.
    target_sum = np.sqrt(2.0 * n)
    a = _project_onto_simplex(init.astype(np.float64), target_sum)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # Soft‑max temperature – start small (smooth) and increase.
    temp = 5.0          # inverse temperature β in exp(β·(conv-max))
    temp_max = 150.0   # final temperature (sharper)

    best_local = a.copy()
    conv = _conv_fft_same(a)
    best_score = 2.0 * n * conv.max() / (a.sum() ** 2)

    it = 0
    # Step‑size scaling factor – we keep updates bounded by the current
    # maximal convolution value.
    base_lr = 0.12 if not final_phase else 0.06

    while time.time() < run_deadline:
        it += 1

        # --- 1) convolution and soft‑max weights -----------------------
        conv = _conv_fft_same(a)
        max_val = conv.max()
        # soft‑max (the exponent is β = temp)
        w = np.exp(temp * (conv - max_val))
        # Numerical safety – very small weights add noise only.
        # No normalisation is required (the scale is absorbed by the step size).

        # --- 2) gradient of the soft‑max objective ---------------------
        grad = _gradient_from_weights(a, w)

        # --- 3) Adam update --------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # Adaptive step: keep the update magnitude comparable to max_val.
        raw_step = m_hat / (np.sqrt(v_hat) + eps)
        step_scale = base_lr * max_val / (np.max(np.abs(raw_step)) + 1e-12)
        step = step_scale * raw_step

        a_new = a - step
        a_new = _project_onto_simplex(a_new, target_sum)

        # --- 4) occasional random jitter to escape plateaus ------------
        if np.random.rand() < (0.03 if not final_phase else 0.01):
            noise = np.random.randn(n) * (target_sum * 0.002 / n)
            a_new = _project_onto_simplex(np.maximum(a_new + noise, 0.0), target_sum)
            # after a shake‑up we are slightly more cautious
            base_lr *= 0.94

        # --- 5) evaluate ------------------------------------------------
        conv_new = _conv_fft_same(a_new)
        max_val_new = conv_new.max()
        score_new = 2.0 * n * max_val_new / (a_new.sum() ** 2)  # equals max_val_new

        if score_new < best_score:
            best_score = score_new
            best_local = a_new.copy()
            print(f"[{time.time():.2f}] n={n}  new local best = {best_score:.6f}")

        a = a_new

        # --- 6) temperature schedule ------------------------------------
        if temp < temp_max:
            # increase temperature slowly (making the soft‑max sharper)
            temp *= 1.001

        # --- 7) decay learning rate slowly -------------------------------
        base_lr *= 0.9999

    return best_local, best_score

def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):

    function within the given time budget.

    Strategy
    --------
    1. Randomly explore many lengths (800 … 5000) with short optimisation
       runs (≈ 5–8 s each).  Initialisation follows a power‑law
       a_i ∝ (i+δ)^{-p}.
    2. Keep the globally best vector seen.
    3. With the remaining time, run a long refinement on that best
       vector using a slower Adam schedule and a higher soft‑max
       temperature.
    """

    deadline = time.time() + budget_s - 0.5   # safety margin for the caller

    best_seq = None
    best_score = float("inf")

    # quick exploratory restarts.
    final_refine_deadline = deadline - 200.0

    # Phase 1 – many short restarts
    # ------------------------------------------------------------------
    while time.time() < final_refine_deadline:
        remaining = deadline - time.time()
        # Choose a slice length: longer slices early, shorter later.
        slice_time = min(8.0, remaining * 0.07)

        # Random length – bias towards larger sizes at the start.
        if remaining > 400.0:
            n = np.random.randint(1200, 4500)
        else:

        idx = np.arange(1, n + 1, dtype=np.float64) + delta
        init = idx ** (-p)

        init += np.random.rand(n) * 1e-4

        if best_seq is not None and np.random.rand() < 0.35:

                init = best_seq[:n].copy()
            else:
                pad = np.random.rand(n - best_seq.size) * 1e-4

                f"[{time.time():.2f}] GLOBAL new best = {best_score:.6f}"

            break

    # ------------------------------------------------------------------
    # Phase 2 – long refinement on the best candidate so far
    # ------------------------------------------------------------------
    if best_seq is not None:
        n_best = best_seq.shape[0]
        # Give the remainder of the budget to a fine tuned optimisation.
        refine_deadline = deadline
        seq_refined, score_refined = _optimise_one_length(
            n=n_best,
            init=best_seq,
            run_deadline=refine_deadline,
            final_phase=True,
        )
        if score_refined < best_score:
            best_score = score_refined
            best_seq = seq_refined.copy()
            print(
                f"[{time.time():.2f}] FINAL refinement best = {best_score:.6f}"
                f" (length={n_best})"
            )
    else:
        # Extremely unlikely fallback: uniform distribution of modest size.
        n = 1500
        target_sum = np.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=np.float64)
        best_score = 2.0 * n * (target_sum / n) ** 2 / (target_sum ** 2)  # =2

    # Return a plain Python list as required by the evaluator.
    return best_seq.tolist()
