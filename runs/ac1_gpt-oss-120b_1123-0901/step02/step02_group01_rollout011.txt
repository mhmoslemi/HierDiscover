def _refine_lp_sequence(seq, deadline, t=0.2, max_iter=5):
    n = len(seq)
    best_seq = seq.copy()
    best_val = _objective(seq)
    # precompute initial M
    conv = _full_convolution(seq)
    M = conv.max()
    for it in range(max_iter):
        if time.time() > deadline:
            break
        # Build convolution matrix A (dense)
        from scipy.linalg import toeplitz
        col = np.concatenate([seq, np.zeros(n-1)])
        row = np.concatenate([seq[:1], np.zeros(n-1)])
        A = toeplitz(col, row)  # shape (2*n-1, n)
        # bounds for b
        bounds = [(0, None)] * n
        # objective for linprog: minimize -sum(b)
        c = -np.ones(n)
        # constraints: A @ b <= M
        b_ub = np.full(2*n - 1, M)
        # solve LP
        try:
            from scipy.optimize import linprog
            # Use method='highs' which is fast and robust
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': max(5.0, (deadline - time.time()))})
        except Exception as e:
            print(f"[LP] exception: {e}")
            break
        if not res.success:
            print(f"[LP] not solved: {res.message}")
            break
        b0 = res.x
        sum_b = b0.sum()
        if sum_b == 0:
            break
        b = b0 / sum_b
        # convex blend
        new_seq = (1 - t) * seq + t * b
        # ensure simplex (but blend of two simplex points stays simplex)
        # but due to numerical rounding ensure projection
        new_seq = _project_onto_simplex(new_seq)
        val = _objective(new_seq)
        print(f"[LP iter {it}] obj={val:.6f}")
        if val < best_val:
            best_val, best_seq = val, new_seq.copy()
            # update seq, conv, M for next iteration
            seq = new_seq
            conv = _full_convolution(seq)
            M = conv.max()
        else:
            # if no improvement, maybe stop early
            break
    return best_seq, best_val

"""
Hybrid optimizer for the step‑function problem.

The routine mixes three ideas:

1. **Fast Adam optimisation on a smooth‑max surrogate** – we minimise
   φ_λ(a) = (1/λ)·log∑exp(λ·(a*a)[k]) using projected Adam
   (the projection is onto the probability simplex so that
   ∑a_i = 1 and a_i ≥ 0).  λ is increased slowly (continuation) so
   the surrogate converges to the true max.

2. **Linear‑program refinement** – once a good candidate `f` is found,
   we solve a small LP:
        maximise   ∑ b_j
        s.t.       (f * b)_k ≤ M  for all k,
                    b_j ≥ 0,
   where M = max(f*f).  The LP returns an extreme point `b` that
   pushes the bound `M` as far as possible.  A convex combination
   (1‑t) f + t b (t≈0.2) is taken and the process may be repeated a
   few times.

3. **Tiny random local tweaks** – after the LP phase a cheap
   coordinate‑wise search moves a small amount of mass between two
   positions; any improvement is kept.

The whole search respects the 1000 s budget, prints progress and
returns the best sequence (as a plain list) found.
"""

import time
import numpy as np
from scipy.linalg import toeplitz
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# 1.  Projection onto the probability simplex {x ≥ 0, Σx = 1}
def _project_onto_simplex(v):
    """Euclidean projection of a vector onto the simplex."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:                       # all numbers non‑positive
        return np.full_like(v, 1.0 / len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# 2.  Fast linear (zero‑padded) convolution using FFT
def _full_convolution(a, size=None):
    """Return the linear self‑convolution a*a (length 2·len(a)‑1)."""
    n = len(a)
    if size is None:
        size = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# 3.  Exact objective (the one used by evaluate_sequence)
def _objective(a):
    """Compute 2·n·max(conv) / (sum(a)^2).  a is assumed non‑negative."""
    n = len(a)
    conv = _full_convolution(a)
    max_b = conv.max()
    sum_a = a.sum()
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# 4.  Gradient of the smooth‑max surrogate (soft‑max weights)
def _softmax_gradient(a, conv, lam, size_corr=None):
    """
    Gradient of φ_λ(a) = (1/λ)·log∑exp(λ·conv_k).

    The gradient is 2·(w ★ a) where w_k ∝ exp(λ·conv_k) and "★"
    denotes linear cross‑correlation.
    """
    n = len(a)
    # soft‑max weights (stable)
    m = conv.max()
    w = np.exp(lam * (conv - m))
    w /= w.sum()

    # cross‑correlation w ★ a (linear, not circular)
    rev_a = a[::-1]
    if size_corr is None:
        size_corr = 1 << int(np.ceil(np.log2(len(w) + n - 1)))
    fw = np.fft.rfft(w, size_corr)
    frev = np.fft.rfft(rev_a, size_corr)
    corr = np.fft.irfft(fw * frev, size_corr)

    # slice that corresponds to indices 0 … n‑1 of the original vector
    grad = 2.0 * corr[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# 5.  Adam optimiser on the surrogate
def _adam_optimize(init_seq, deadline, max_iters=None):
    """
    Projected Adam on the smooth‑max surrogate.
    Returns the best sequence seen and its exact objective value.
    """
    seq = init_seq.copy()
    n = len(seq)

    # hyper‑parameters
    lam = 1.0                # start of continuation
    lam_factor = 1.03        # increase λ each iteration
    alpha = 0.05             # base learning rate
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0

    # FFT plan size (kept constant for speed)
    size_fft = 1 << int(np.ceil(np.log2(3 * n)))

    best_seq = seq.copy()
    best_val = _objective(seq)
    last_improve = time.time()

    while time.time() < deadline:
        if max_iters is not None and t >= max_iters:
            break

        # 1) current convolution
        conv = _full_convolution(seq)

        # 2) gradient of the surrogate
        grad = _softmax_gradient(seq, conv, lam, size_corr=size_fft)

        # 3) Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        seq = seq - step

        # 4) projection back onto the simplex
        seq = _project_onto_simplex(seq)

        # 5) increase continuation parameter
        lam *= lam_factor

        # 6) occasional exact evaluation (every 20‑th step)
        if t % 20 == 0:
            val = _objective(seq)
            if val < best_val - 1e-9:
                best_val, best_seq = val, seq.copy()
                last_improve = time.time()
                print(f"[Adam] iter={t:06d}  obj={val:.6f}  lam={lam:.2f}")

        # 7) if stuck, add a tiny random shake and reset moments
        if time.time() - last_improve > 2.0:
            noise = (np.random.rand(n) - 0.5) * 1e-3
            seq = _project_onto_simplex(seq + noise)
            m.fill(0.0)
            v.fill(0.0)
            last_improve = time.time()

    return best_seq, best_val


# ----------------------------------------------------------------------
# 6.  Linear‑program refinement (extreme point improvement)
def _lp_refine(seq, deadline, t_blend=0.2, max_iter=4):
    """
    Given a feasible sequence `seq` (∑seq = 1), solve the LP
        maximise   Σ b_j
        s.t.       (seq * b)_k ≤ M   (M = max(seq*seq))
                    b_j ≥ 0
    and blend the result back with the current sequence.
    The routine is called at most `max_iter` times.
    """
    n = len(seq)
    best_seq = seq.copy()
    best_val = _objective(seq)
    conv = _full_convolution(seq)
    M = conv.max()

    for it in range(max_iter):
        if time.time() > deadline:
            break

        # Build the linear convolution matrix A (size (2n‑1) × n)
        col = np.concatenate([seq, np.zeros(n - 1)])
        row = np.concatenate([seq[:1], np.zeros(n - 1)])
        A = toeplitz(col, row)          # dense, but still modest (~64 MiB for n=2000)

        # LP: minimise -sum(b)  (i.e. maximise sum(b))
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None) for _ in range(n)]

        # time limit for the solver – leave a safety margin
        time_left = max(0.0, deadline - time.time() - 0.2)
        opts = {"presolve": True, "time_limit": max(5.0, time_left)}

        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                          method="highs", options=opts)
        except Exception as exc:
            print(f"[LP] exception: {exc}")
            break

        if not res.success:
            print(f"[LP] solver failed: {res.message}")
            break

        b0 = res.x
        sum_b = b0.sum()
        if sum_b <= 0.0:
            break

        # normalise to the simplex and blend
        b = b0 / sum_b
        new_seq = (1.0 - t_blend) * seq + t_blend * b
        new_seq = _project_onto_simplex(new_seq)

        val = _objective(new_seq)
        print(f"[LP] iter={it}  obj={val:.6f}")

        if val < best_val - 1e-9:
            best_val, best_seq = val, new_seq.copy()
            # prepare for the next LP round
            seq = new_seq
            conv = _full_convolution(seq)
            M = conv.max()
        else:
            # no improvement – stop early
            break

    return best_seq, best_val


# ----------------------------------------------------------------------
# 7.  Very cheap coordinate‑wise local search
def _local_search(seq, deadline, steps=1500):
    """
    Randomly shift a tiny amount of mass between two coordinates.
    Accept only improving moves.
    """
    n = len(seq)
    best_seq = seq.copy()
    best_val = _objective(seq)

    for _ in range(steps):
        if time.time() > deadline:
            break
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        eps = np.random.rand() * 5e-4          # tiny shift
        if best_seq[i] < eps:
            continue
        cand = best_seq.copy()
        cand[i] -= eps
        cand[j] += eps
        val = _objective(cand)
        if val < best_val:
            best_val, best_seq = val, cand.copy()
    return best_seq, best_val

# Main entry point required by the grader
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Hybrid search for a step‑function with minimal `evaluate_sequence` score.
    The function respects the time budget, prints progress and finally returns
    the best sequence (as a plain Python list).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # 1. Choose a moderate length – long enough for expressive power,
    #    short enough to keep FFTs cheap.
    n = 2000

    # ------------------------------------------------------------------
    # 2. Collect a pool of starting points
    starts = []

    # a) use the previously discovered sequence if it exists
    if "height_sequence_1" in globals():
        cand = np.array(height_sequence_1, dtype=np.float64)
        cand = np.maximum(cand, 0.0)
        if cand.sum() > 0.0:
            starts.append(cand / cand.sum())

    # b) geometric decays – a_i ∝ r^i
    for r in np.linspace(0.90, 0.998, 10):
        if time.time() > deadline:
            break
        vec = np.power(r, np.arange(n, dtype=np.float64))
        vec /= vec.sum()
        starts.append(vec)

    # c) a few random Dirichlet vectors
    for _ in range(12):
        if time.time() > deadline:
            break
        vec = np.random.dirichlet(np.ones(n))
        starts.append(vec)

    # ------------------------------------------------------------------
    # 3. Quick Adam passes on every start (limited iterations)
    best_seq = None
    best_val = float("inf")

    for idx, init in enumerate(starts):
        if time.time() > deadline:
            break
        # limited budget per starter (≈0.2 s each)
        sub_deadline = time.time() + min(0.25, deadline - time.time())
        seq_opt, val_opt = _adam_optimize(init, sub_deadline, max_iters=800)
        if val_opt < best_val:
            best_val, best_seq = val_opt, seq_opt.copy()
            print(f"[Init {idx:02d}] new global best = {val_opt:.6f}")

    # ------------------------------------------------------------------
    # 4. Long Adam refinement on the current best (uses remaining time)
    if best_seq is not None and time.time() < deadline:
        seq_ref, val_ref = _adam_optimize(best_seq, deadline)
        if val_ref < best_val:
            best_val, best_seq = val_ref, seq_ref.copy()
            print(f"[Fine‑Adam] improved global best = {val_ref:.6f}")

    # ------------------------------------------------------------------
    # 5. LP‑based extreme‑point refinement (few iterations)
    if best_seq is not None and time.time() < deadline:
        seq_lp, val_lp = _lp_refine(best_seq, deadline,
                                    t_blend=0.2, max_iter=3)
        if val_lp < best_val:
            best_val, best_seq = val_lp, seq_lp.copy()
            print(f"[LP‑refine] new global best = {val_lp:.6f}")

    # ------------------------------------------------------------------
    # 6. Final tiny local search
    if best_seq is not None and time.time() < deadline:
        seq_loc, val_loc = _local_search(best_seq, deadline, steps=1200)
        if val_loc < best_val:
            best_val, best_seq = val_loc, seq_loc.copy()
            print(f"[Local] final best = {val_loc:.6f}")

    # ------------------------------------------------------------------
    # 7. Return result
    if best_seq is None:
        # safety fallback – uniform distribution (guaranteed valid)
        best_seq = np.full(n, 1.0 / n)

    print(f"=== Final objective = {best_val:.6f} (length {n}) ===")
    return best_seq.tolist()
