    size = len(a)+len(b)-1
    nfft = 1 << (size-1).bit_length()
    fft_a = np.fft.rfft(a,nfft)
    fft_b = np.fft.rfft(b,nfft)
    fft_c = fft_a * fft_b
    c = np.fft.irfft(fft_c,nfft)[:size]
    return c

"""
Propose a step‑function (non‑negative sequence) with a low
`evaluate_sequence` value.

Algorithm
---------
1. **Initialization**
   * Generate a diverse pool of candidates for a moderate length
     (default n=2000): uniform, several power‑law “U‑shaped” families,
     a sine‑squared shape, arcsine, and a few random Dirichlet draws.
   * Pick the candidate with the smallest objective (computed directly
     from the convolution, avoiding the relatively expensive
     `evaluate_sequence`).

2. **Phase‑1 optimisation (n≈2000)**
   * Minimise a smooth approximation of the maximum convolution entry:
     L(β) = (1/β)·log Σₖ exp(β·cₖ) where c = a∗a.
   * The gradient of L is the correlation of the weight vector
     wₖ ∝ exp(β·cₖ) with the current sequence.
   * Apply the Adam optimiser (first‑ and second‑moment estimates) and
     after every step project back onto the probability simplex
     (∑aᵢ=1, aᵢ≥0).
   * β starts small (≈2) and is slowly increased (≈1 % per iteration)
     so the optimisation first smooths the landscape and later focuses
     on the true max‑convolution.
   * The best sequence seen so far is stored.

3. **Upscaling**
   * Interpolate the best Phase‑1 sequence to a larger size
     (default n=8000).  The interpolation preserves the overall shape.

4. **Phase‑2 optimisation (large n)**
   * Repeat the same Adam‑with‑log‑sum‑exp optimisation on the larger
     sequence.  Because the objective scales as 2 n·max(conv) the larger
     problem can achieve a lower bound (theoretical optimum ≈ 1).
   * Occasionally (after many iterations without improvement) add a
     small Gaussian jitter to escape plateaus.

5. **Return**
   * The best sequence found (projected onto the simplex) is returned.
   * The final objective is printed for reference.

The method uses only NumPy (no external solvers) and stays well
within the 1000 s budget on a single CPU core.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT (fast for n up to ~1e4)."""
    size = len(x) + len(y) - 1
    nfft = 1 << ((size - 1).bit_length())
    X = np.fft.rfft(x, nfft)
    Y = np.fft.rfft(y, nfft)
    Z = X * Y
    conv = np.fft.irfft(Z, nfft)[:size]
    return conv


# ----------------------------------------------------------------------
def _conv_self(seq: np.ndarray) -> np.ndarray:
    """Self‑convolution a∗a (full length)."""
    return _conv_fft(seq, seq)


# ----------------------------------------------------------------------
def _objective(seq: np.ndarray, conv: np.ndarray | None = None) -> float:
    """Compute the value returned by `evaluate_sequence` (sum is assumed ≈1)."""
    if conv is None:
        conv = _conv_self(seq)
    max_conv = conv.max()
    s = seq.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * len(seq) * max_conv / (s * s))


# ----------------------------------------------------------------------
def _initial_candidates(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Create a set of promising start vectors (already normalised to sum=1)."""
    cand = []

    # 1) uniform
    cand.append(np.full(n, 1.0 / n))

    # 2) power‑law “U‑shaped” families (heavier weight at the ends)
    for p in [0.5, 1.0, 1.5, 2.0, 2.5, 3.0]:
        i = np.arange(n, dtype=np.float64)
        a = ((i + 0.5) * (n - i - 0.5)) ** (-p)
        cand.append(a / a.sum())

    # 3) sine‑squared shape (smooth mound)
    s = np.sin(np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n) ** 2
    cand.append(s / s.sum())

    # 4) arcsine law
    i = np.arange(n, dtype=np.float64)
    arcs = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    cand.append(arcs / arcs.sum())

    # 5) a few random Dirichlet draws
    for _ in range(5):
        rnd = rng.random(n)
        cand.append(rnd / rnd.sum())

    return cand

    Return a list of non‑negative heights that minimises the evaluation
    function within the allotted time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # --------------------------------------------------------------
    # Phase‑1 configuration
    n1 = 2000                     # moderate size for many cheap iterations
    phase1_deadline = start_time + budget_s * 0.55

    # --------------------------------------------------------------
    # 1. initialise a diverse pool and pick the best start
    best_seq = None
    best_score = np.inf

    for cand in _initial_candidates(n1, rng):
        conv = _conv_self(cand)
        score = _objective(cand, conv)
        if score < best_score:
            best_score = score
            best_seq = cand.copy()
    print(f"[Init] length={n1}, best initial objective = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2. Phase‑1 optimisation (smooth max via log‑sum‑exp + Adam)
    seq = best_seq.copy()
    m = np.zeros_like(seq)          # first moment
    v = np.zeros_like(seq)          # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    lr = 0.25                       # initial learning rate
    beta = 2.0                      # temperature for soft‑max
    beta_max = 30.0
    it = 0
    no_improve = 0
    last_report = time.time()

    while time.time() < phase1_deadline:
        # full self‑convolution
        conv = _conv_self(seq)
        max_c = conv.max()

        # soft‑max weights for a smooth approximation of the maximum
        w = np.exp(beta * (conv - max_c))
        w /= w.sum()

        # gradient = correlation w * seq (valid mode gives length n)
        grad = np.correlate(w, seq, mode='valid')

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        it1 = it + 1
        m_hat = m / (1.0 - beta1 ** it1)
        v_hat = v / (1.0 - beta2 ** it1)
        seq_new = seq - lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto simplex (ensures non‑negativity and sum≈1)
        seq_new = _project_onto_simplex(seq_new)

        # evaluate current objective (cheap – we already have conv)
        score = _objective(seq_new, conv=np.convolve(seq_new, seq_new))

        # keep the best encountered sequence
        if score < best_score:
            best_score = score
            best_seq = seq_new.copy()
            print(f"[Phase1 iter {it:06d}] new best = {best_score:.6f}")
            no_improve = 0
        else:
            no_improve += 1

        # occasional random jitter to escape stagnation
        if no_improve > 50000:
            jitter = rng.normal(scale=0.005, size=n1)
            seq_new = seq_new + jitter
            seq_new = np.clip(seq_new, 0.0, None)
            seq_new /= seq_new.sum()
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0
            print(f"[Phase1 iter {it:06d}] stagnation – jitter applied")

        # schedule updates
        lr *= 0.9999
        beta = min(beta_max, beta * 1.001)

        seq = seq_new
        it += 1

        # sporadic progress report (avoid flooding stdout)
        if time.time() - last_report > 30:
            print(f"[Phase1] time elapsed {time.time() - start_time:.1f}s, "
                  f"current best {best_score:.6f}")
            last_report = time.time()

    # --------------------------------------------------------------
    # 3. Upscale to a larger problem size (more degrees of freedom)
    n2 = 8000
    # linear interpolation of the best Phase‑1 sequence
    x_old = np.linspace(0.0, 1.0, n1, endpoint=False)
    x_new = np.linspace(0.0, 1.0, n2, endpoint=False)
    seq_big = np.interp(x_new, x_old, best_seq)
    seq_big = _project_onto_simplex(seq_big)   # re‑normalise

    # --------------------------------------------------------------
    # 4. Phase‑2 optimisation (same scheme, larger n)
    seq = seq_big.copy()
    m = np.zeros_like(seq)
    v = np.zeros_like(seq)
    lr = 0.20
    beta = 2.0
    it = 0
    no_improve = 0
    last_report = time.time()

    while time.time() < deadline:
        conv = _conv_self(seq)
        max_c = conv.max()

        w = np.exp(beta * (conv - max_c))
        w /= w.sum()

        grad = np.correlate(w, seq, mode='valid')

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        it1 = it + 1
        m_hat = m / (1.0 - beta1 ** it1)
        v_hat = v / (1.0 - beta2 ** it1)
        seq_new = seq - lr * m_hat / (np.sqrt(v_hat) + eps)

        seq_new = _project_onto_simplex(seq_new)

        # cheap objective check
        score = _objective(seq_new, conv=np.convolve(seq_new, seq_new))
        if score < best_score:
            best_score = score
            best_seq = seq_new.copy()
            print(f"[Phase2 iter {it:06d}] new best = {best_score:.6f}")
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 80000:
            jitter = rng.normal(scale=0.003, size=n2)
            seq_new = seq_new + jitter
            seq_new = np.clip(seq_new, 0.0, None)
            seq_new /= seq_new.sum()
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0
            print(f"[Phase2 iter {it:06d}] stagnation – jitter applied")

        it += 1
