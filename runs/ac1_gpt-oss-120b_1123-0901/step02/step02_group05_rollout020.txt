A fast global optimiser for the step‑function bound.
===================================================

The objective is

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

with a_i ≥ 0.  Because the value is invariant under scaling we keep the
norm ‖a‖₁ fixed to √(2 n) and work on the simplex
{ a ≥ 0 , Σ a = √(2 n) }.

The search consists of three layers:

1.  **Diverse initialisations** – random Dirichlet draws,
    geometric (a_i ∝ r^i) and linear ramps.  The length n is
    sampled between 300 and 1800.

2.  **Adam‑based projected gradient descent** on the *sub‑gradient* of
    the maximal convolution entry (the same gradient used in the
    reference code).  Adam’s adaptive step‑size and bias‑correction
    give a much more robust descent than a plain fixed step.

3.  **Optional LP refinement** (when cvxpy is present) and a tiny
    evolutionary mutational loop.  The LP implements the update
    described in the paper sketch; the mutation is a Gaussian
    perturbation followed by a simplex projection.

The whole routine runs for at most the supplied budget (default
1000 s).  Progress and new best scores are printed.  The best
sequence found is returned as a plain Python list.
"""

# Simple Euclidean projection onto the simplex {x ≥ 0, Σ x = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex of non‑negative vectors with sum s."""
    n = v.shape[0]
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    # sort v in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper: convolution (full) and max entry together with the gradient
# ----------------------------------------------------------------------
def _max_conv_and_grad(a: np.ndarray):
    """
    Compute max(conv(a, a)), its index and the gradient of that entry.
    Returns (max_val, grad, max_idx, full_conv).
    """
    n = a.shape[0]
    conv = np.convolve(a, a)        # length 2n‑1
    max_idx = int(np.argmax(conv))
    max_val = conv[max_idx]

    # derivative of max_val w.r.t each a[i] (sub‑gradient)
    low = max(0, max_idx - (n - 1))
    high = min(n - 1, max_idx)
    idx = np.arange(low, high + 1, dtype=int)
    partner = max_idx - idx
    grad = np.zeros_like(a)
    grad[idx] = 2.0 * a[partner]
    return max_val, grad, max_idx, conv


# ----------------------------------------------------------------------
# Adam‑based projected gradient descent on the maximal convolution entry
# ----------------------------------------------------------------------
def _adam_descent(
    init: np.ndarray,
    target_sum: float,
    time_limit: float,
    base_lr: float = 0.12,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    print_tag: str = "",
) -> tuple[np.ndarray, float]:
    """
    Run Adam‑style descent on the sub‑gradient of the maximal convolution
    entry, projecting after each step onto the simplex.
    Returns the best sequence seen and its evaluation.
    """
    a = _project_onto_simplex(init.astype(float), target_sum)

    try:
        best_score = float(evaluate_sequence(a.tolist()))
    except Exception:
        best_score = math.inf
    best_seq = a.copy()

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    start = time.time()
    it = 0

    while time.time() < start + time_limit:
        max_val, grad, _, _ = _max_conv_and_grad(a)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        t += 1
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # raw Adam step
        delta = -base_lr * m_hat / (np.sqrt(v_hat) + eps)

        # scale step so that the largest per‑coordinate change is comparable to max_val
        max_abs = np.max(np.abs(delta)) if np.max(np.abs(delta)) > 0 else 1.0
        scale = base_lr * max_val / max_abs
        a_new = a + scale * delta

        # projection onto simplex (ensures non‑negativity and sum)
        a_new = _project_onto_simplex(a_new, target_sum)

        # evaluate
        try:
            score = float(evaluate_sequence(a_new.tolist()))
        except Exception:
            score = math.inf

        if score < best_score:
            best_score = score
            best_seq = a_new.copy()
            print(
                f"[{time.time():.2f}] {print_tag} new local best = {score:.6f} (n={len(a)})"
            )

        # simple occasional jitter to escape plateaus
        if np.random.rand() < 0.015:
            sigma = 0.02 * target_sum / len(a)
            a = _project_onto_simplex(a_new + np.random.randn(len(a)) * sigma, target_sum)
        else:
            a = a_new

        # decay learning‑rate a little every few hundred iterations
        if it % 300 == 0 and it > 0:
            base_lr *= 0.85
        it += 1

    return best_seq, best_score


# ----------------------------------------------------------------------
# Optional LP refinement (uses cvxpy if available)
# ----------------------------------------------------------------------
try:
    import cvxpy as cp

    _CVX_AVAILABLE = True
except Exception:  # pragma: no cover
    _CVX_AVAILABLE = False


def _solve_lp_for_g(f: np.ndarray, M: float):
    """Solve the LP: maximise Σ g_j subject to g ≥ 0 and f∗g ≤ M."""
    if not _CVX_AVAILABLE:
        return None
    n = f.shape[0]
    g = cp.Variable(n, nonneg=True)

    constraints = []
    for k in range(2 * n - 1):
        low = max(0, k - (n - 1))
        high = min(n - 1, k)
        idx = np.arange(low, high + 1, dtype=int)
        partner = k - idx
        expr = cp.sum(cp.multiply(f[low : high + 1], g[partner]))
        constraints.append(expr <= M)

    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.GLPK, warm_start=True, verbose=False, glpk={})
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, verbose=False, max_iters=2000)
        except Exception:
            return None

    if prob.status not in ("optimal", "optimal_inaccurate"):
        return None
    g_val = np.asarray(g.value).flatten()
    if g_val is None:
        return None
    g_val = np.maximum(g_val, 0.0)
    return g_val


def _lp_refine(
    init: np.ndarray,
    target_sum: float,
    time_limit: float,
    mix: float = 0.22,
    max_iter: int = 5,
) -> tuple[np.ndarray, float]:
    """A few LP‑updates starting from ``init``."""
    a = _project_onto_simplex(init.astype(float), target_sum)
    try:
        best_score = float(evaluate_sequence(a.tolist()))
    except Exception:
        best_score = math.inf
    best_seq = a.copy()
    t0 = time.time()

    for it in range(max_iter):
        if time.time() - t0 > time_limit:
            break
        M = np.convolve(a, a).max()
        g0 = _solve_lp_for_g(a, float(M))
        if g0 is None or g0.sum() == 0:
            break
        g = g0 * (target_sum / g0.sum())
        a = (1.0 - mix) * a + mix * g
        a = _project_onto_simplex(a, target_sum)

        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            score = math.inf

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time():.2f}] LP refine iter {it+1} new best = {score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Small mutation helper (Gaussian noise + projection)
# ----------------------------------------------------------------------
def _mutate(a: np.ndarray, target_sum: float, sigma: float) -> np.ndarray:
    """Return a mutated copy of a (Gaussian noise, then simplex projection)."""
    b = a + np.random.randn(a.size) * sigma
    b = np.maximum(b, 0.0)
    b = _project_onto_simplex(b, target_sum)
    return b


# ----------------------------------------------------------------------
# Resize helper – linear interpolation / down‑sampling
# ----------------------------------------------------------------------
def _resize_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize seq to length new_n preserving non‑negativity (linear interpolation)."""
    old_n = seq.size
    if old_n == new_n:
        return seq.copy()
    if old_n > new_n:
        idx = np.linspace(0, old_n - 1, new_n).astype(int)
        return seq[idx]
    else:
        x_old = np.arange(old_n)
        x_new = np.linspace(0, old_n - 1, new_n)
        return np.interp(x_new, x_old, seq)


# ----------------------------------------------------------------------
# Different initialisation strategies
# ----------------------------------------------------------------------
def _random_init(n: int) -> np.ndarray:
    """Return a random point on the simplex of size n."""
    v = np.random.rand(n)
    v = v / v.sum() * math.sqrt(2.0 * n)
    return v


def _geometric_init(n: int) -> np.ndarray:
    """Geometric decay a_i ∝ r^i with random ratio r∈(0.5,0.99)."""
    r = np.random.uniform(0.5, 0.99)
    v = r ** np.arange(n, dtype=float)
    v = v / v.sum() * math.sqrt(2.0 * n)
    return v


def _ramp_init(n: int) -> np.ndarray:
    """Linear ramp from 0 to a peak."""
    v = np.linspace(0.0, 1.0, n)
    v = v / v.sum() * math.sqrt(2.0 * n)
    return v


def _triangular_init(n: int) -> np.ndarray:
    """Symmetric triangle (increase then decrease)."""
    half = (n + 1) // 2
    inc = np.linspace(1.0, half, half)
    dec = inc[:-1][::-1] if n % 2 else inc[::-1]
    v = np.concatenate([inc, dec])
    v = v / v.sum() * math.sqrt(2.0 * n)
    return v


def _initial_candidate(n: int, reference: np.ndarray | None) -> np.ndarray:
    """Pick a starter: resized best, or a random parametrised shape."""
    if reference is not None and np.random.rand() < 0.45:
        return _resize_sequence(reference, n)

    # choose a random creation method
    creators = [_random_init, _geometric_init, _ramp_init, _triangular_init]
    return random.choice(creators)(n)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: float = 1000.0, **kwargs):
    """
    Search for a step‑function (list of non‑negative heights) with a low
    evaluation score.  The routine runs for at most ``budget_s`` seconds.
    """
    random.seed(seed)
    np.random.seed(seed)

    deadline = time.time() + budget_s - 0.5  # safety margin

    # --------------------------------------------------------------
    # Global best holder (try to use historic solution if present)
    # --------------------------------------------------------------
    best_seq = None
    best_score = math.inf

    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(height_sequence_1, dtype=float)
            sc = float(evaluate_sequence(cand.tolist()))
            best_seq = cand.copy()
            best_score = sc
            print(f"[{time.time():.2f}] Starting from historic seq, score={sc:.6f}")
        except Exception:
            best_seq = None
            best_score = math.inf

    # fallback: uniform sequence of length 1000
    if best_seq is None:
        n0 = 1000
        target0 = math.sqrt(2.0 * n0)
        best_seq = np.full(n0, target0 / n0, dtype=float)
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"[{time.time():.2f}] No historic seq – using uniform, score={best_score:.6f}")

    # --------------------------------------------------------------
    # Helper that runs a single optimisation round (Adam + optional LP)
    # --------------------------------------------------------------
    def _run_one_round(n: int, init_vec: np.ndarray, remaining: float):
        target = math.sqrt(2.0 * n)

        # allocate time proportionally – at most 12 s for the heavy part
        heavy_time = min(remaining * 0.75, 12.0)
        light_time = remaining - heavy_time

        # 1) Adam descent
        cand, sc = _adam_descent(init_vec, target, heavy_time,
                                 base_lr=0.12,
                                 print_tag=f"n={n}")

        # 2) optional LP refinement if cheap enough
        if _CVX_AVAILABLE and n <= 800 and random.rand() < 0.25:
            cand, sc = _lp_refine(cand, target,
                                  time_limit=min(light_time, 8.0),
                                  mix=0.22, max_iter=4)

        # 3) a tiny evolutionary perturbation (mutate + short Adam)
        if sc > best_score * 0.99 and random.rand() < 0.3:
            sigma = 0.018 * target / n
            mutated = _mutate(cand, target, sigma)
            cand2, sc2 = _adam_descent(mutated, target, min(light_time, 5.0),
                                       base_lr=0.07,
                                       print_tag=f"evo n={n}")
            if sc2 < sc:
                cand, sc = cand2, sc2

        return cand, sc

    # --------------------------------------------------------------
    # Main global loop
    # --------------------------------------------------------------
    while time.time() < deadline - 1.0:
        # pick a length – stay around current best but allow variation
        if best_seq is not None and random.random() < 0.65:
            n = best_seq.size
            if random.random() < 0.33:
                n = int(n * random.uniform(0.85, 1.15))
                n = max(250, min(2000, n))
        else:
            n = random.randint(250, 1900)

        target_sum = math.sqrt(2.0 * n)

        # initialise candidate (possibly resized best)
        init = _initial_candidate(n, best_seq)

        time_left = deadline - time.time()
        if time_left < 6.0:
            break

        cand_seq, cand_score = _run_one_round(n, init, time_left)

        if cand_seq is not None and cand_score < best_score:
            best_score = cand_score
            best_seq = cand_seq.copy()
            print(
                f"[{time.time():.2f}] *** GLOBAL NEW BEST = {best_score:.6f} (n={n})"
            )
            if best_score < 1.503:
                print("[INFO] Target reached – stopping early.")
                break

    # ------------------------------------------------------------------
    # Fallback safety (should never trigger)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=float)

    return best_seq.tolist()
