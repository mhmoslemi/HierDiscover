"""
Propose a non‑negative step function minimising

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

Algorithm
---------
* Search space: length n in [2000, 12000] (large n drives the theoretical bound
  towards 1).  The sum of the coefficients is forced to √(2 n); with this
  normalisation the evaluation reduces to the maximum entry of the full
  autoconvolution (objective = max(conv)).
* Initialise vectors by one of three cheap schemes:
    – power‑law a_i ∝ (i+1)^‑p (p≈0.5);
    – a random point on the simplex;
    – a “Fourier‑random” signal (random phases, real part taken, absolute value).
  In all cases the vector is projected onto the simplex {x_i≥0, Σx_i=√(2 n)}.
* Optimisation core is an ADAM‑style projected (sub‑)gradient descent.
  The gradient of the current maximum convolution is obtained from the
  “active‑set” – indices whose convolution value lies within a relative
  tolerance of the current maximum.  For a small active set a cheap O(|A|·n)
  routine is used; for larger active sets the gradient is computed by a
  second FFT‑based convolution.
* The step size is scaled by the current maximal autocorrelation, decays
  slowly and is capped by a back‑off factor when the objective does not
  improve.  Occasional random jitter (probability ≈2 %) helps escaping flat
  plateaus.
* Many independent restarts are performed within the 1000 s budget.
  The best sequence found overall is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# 1. Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# 2. FFT‑based full convolution (any two real vectors)
# ----------------------------------------------------------------------
def _conv_fft_full(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of real vectors x and y via FFT."""
    m = len(x) + len(y) - 1
    size = 1 << ((m - 1).bit_length())  # next power of two
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:m]
    # Clip tiny negatives caused by rounding
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# 3. Autoconvolution (a * a) – used for the objective
# ----------------------------------------------------------------------
def _auto_conv(a: np.ndarray) -> np.ndarray:
    return _conv_fft_full(a, a)


# ----------------------------------------------------------------------
# 4. Gradient from an arbitrary weight vector w (length 2n‑1)
#    grad_i = 2 * Σ_k w_k * a_{k‑i}
# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Compute the gradient of Σ_k w_k (a∗a)_k w.r.t. a."""
    # a_rev = a reversed – needed because conv uses (k‑i) indexing.
    a_rev = a[::-1]
    conv = _conv_fft_full(w, a_rev)          # length = (2n‑1)+n‑1 = 3n‑2
    start = len(a) - 1
    grad = 2.0 * conv[start:start + len(a)]
    return grad


# ----------------------------------------------------------------------
# 5. Gradient for the classic “active‑set” (binary weights)
# ----------------------------------------------------------------------
def _gradient_active_set(a: np.ndarray, active: np.ndarray) -> np.ndarray:
    """Fast O(|active|·n) gradient when the active set is small."""
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        partner = k - idx
        grad[idx] += 2.0 * a[partner]
    return grad


# ----------------------------------------------------------------------
# 6. One optimisation run for a fixed length n
# ----------------------------------------------------------------------
def _optimise_one_length_adam(
        n: int,
        init: np.ndarray,
        run_deadline: float,
        *,
        rel_tol_start: float = 0.02,
        rel_tol_decay: float = 0.995,
        min_rel_tol: float = 5e-4,
        base_lr: float = 0.5,
        beta1: float = 0.9,
        beta2: float = 0.999,
        jitter_prob: float = 0.02,
        jitter_scale: float = 1e-4,
) -> tuple[np.ndarray, float]:
    """
    Projected ADAM optimisation of max‑autocorrelation for a given length.
    Returns the best vector found in this run and its *evaluate_sequence*
    score.
    """
    target_sum = np.sqrt(2.0 * n)               # Σ a_i = √(2 n)
    a = _project_onto_simplex(init.astype(np.float64), target_sum)

    # bookkeeping for the best local solution
    conv = _auto_conv(a)
    best_score = evaluate_sequence(a.tolist())
    best_vec = a.copy()

    # ADAM accumulators
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    rel_tol = rel_tol_start

    while time.time() < run_deadline:
        conv = _auto_conv(a)
        max_val = conv.max()
        cur_score = evaluate_sequence(a.tolist())   # exact score (objective = max)

        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()
            print(f"[{time.time():.2f}] n={n}  local best = {cur_score:.6f}")

        # ---- active set -------------------------------------------------
        tol_val = max(rel_tol * max_val, 1e-12)
        active = np.where(conv >= max_val - tol_val)[0]

        # ---- gradient ---------------------------------------------------
        if active.size <= 200:                           # cheap exact gradient
            grad = _gradient_active_set(a, active)
        else:                                            # FFT‑based for larger sets
            w = np.zeros_like(conv)
            w[active] = 1.0
            grad = _gradient_from_weights(a, w)

        # ---- ADAM update -----------------------------------------------
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # Scale step so that it is roughly proportional to the current max.
        step_norm = np.linalg.norm(m_hat) + 1e-12
        eta = base_lr * max_val / step_norm
        a_new = a - eta * m_hat / (np.sqrt(v_hat) + 1e-8)

        # ---- jitter (helps escape plateaus) ---------------------------
        if np.random.rand() < jitter_prob:
            jitter = np.random.randn(n) * (target_sum * jitter_scale / n)
            a_new = a_new + jitter

        # ---- projection back onto the simplex -------------------------
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_onto_simplex(a_new, target_sum)

        a = a_new

        # ---- slowly tighten the active tolerance -----------------------
        rel_tol = max(rel_tol * rel_tol_decay, min_rel_tol)

    # Return the best vector seen in this run together with its true score.
    final_score = evaluate_sequence(best_vec.tolist())
    return best_vec, final_score


# ----------------------------------------------------------------------
# 7. Main entry point required by the evaluator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative height sequence with a low evaluation score.
    The routine runs for at most ``budget_s`` seconds and returns the best
    sequence discovered across many independent restarts.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    best_seq = None
    best_score = float("inf")

    # search over a fairly wide range of lengths – larger n gives a lower
    # theoretical bound (≈1) but is a little more expensive per iteration.
    min_n, max_n = 2000, 12000
    slice_seconds = 9.0   # time allocated to each restart

    while time.time() < deadline:
        # ---- pick a length -------------------------------------------------
        remaining = deadline - time.time()
        if remaining < slice_seconds:
            slice_seconds = max(2.0, remaining * 0.8)

        n = np.random.randint(min_n, max_n + 1)
        target_sum = np.sqrt(2.0 * n)

        # ---- initialise ----------------------------------------------------
        init_type = np.random.choice(["power", "simplex", "fourier"])
        if init_type == "power":
            p = np.random.uniform(0.4, 0.6)
            idx = np.arange(1, n + 1, dtype=np.float64)
            init = idx ** (-p)
            init += np.random.rand(n) * 1e-4
        elif init_type == "simplex":
            init = np.random.rand(n)
        else:   # fourier‑random
            size_fft = 1 << ((2 * n - 1 - 1).bit_length())
            phases = np.random.rand(size_fft // 2 + 1) * 2.0 * np.pi
            spec = np.exp(1j * phases)          # unit magnitude
            tmp = np.fft.irfft(spec, n=size_fft)[:n]
            init = np.abs(tmp) + 1e-6           # ensure >0

        # warm‑start from the global best with some probability
        if best_seq is not None and np.random.rand() < 0.4:
            if best_seq.size >= n:
                init = best_seq[:n].copy()
            else:
                pad = np.random.rand(n - best_seq.size) * 1e-4
                init = np.concatenate([best_seq, pad])

        # ---- run a short optimisation ---------------------------------------
        run_deadline = min(deadline, time.time() + slice_seconds)
        seq_local, score_local = _optimise_one_length_adam(
            n=n,
            init=init,
            run_deadline=run_deadline,
            rel_tol_start=0.02,
            rel_tol_decay=0.995,
            min_rel_tol=5e-4,
            base_lr=0.5,
            beta1=0.9,
            beta2=0.999,
            jitter_prob=0.02,
            jitter_scale=1e-4,
        )

        # ---- update global best --------------------------------------------
        if score_local < best_score:
            best_score = score_local
            best_seq = seq_local.copy()
            print(f"[{time.time():.2f}] GLOBAL new best = {best_score:.6f} (n={n})")

        # Early exit if we have already passed a comfortably low threshold.
        if best_score < 1.48:
            break

    # ----------------------------------------------------------------------
    # Fallback – should never be needed, but guarantees a valid return value.
    # ----------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        target_sum = np.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=np.float64)

    # Return a plain Python list, as required by the external evaluator.
    return best_seq.tolist()
