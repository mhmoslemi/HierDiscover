def _solve_lp(seq, M):
    n = len(seq)
    A = np.zeros((2*n-1, n), dtype=np.float64)
    for i in range(n):
        A[i:i+n, i] = seq
    b = np.full(2*n-1, M, dtype=np.float64)
    c = -np.ones(n)
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    if res.success:
        return res.x
    else:
        return None

def _self_convolve(seq):
    n = len(seq)
    m = 2*n - 1
    nfft = 1 << ((2*n-1)-1).bit_length()
    A = np.fft.rfft(seq, n=nfft)
    conv = np.fft.irfft(A*A, n=nfft)[:m]
    return conv

"""
Propose a non‑negative step‑function (list of heights) that minimises the
evaluation function:

    2·n·max(a*a) / (sum(a))²

The algorithm combines three ideas:

1. **Parametric seeding** – generate initial sequences from simple families
   (geometric, power‑law, beta‑like, random Dirichlet) for a few different
   lengths.

2. **Linear‑programming driven re‑balancing** – for a current candidate f,
   compute M = max(f*f).  Solve the linear program
       maximise   Σ g_i
       subject to g_i ≥ 0   and   (f * g)_k ≤ M   (∀k)
   This pushes mass to positions where f contributes little to the convolution,
   thereby reducing the peak.  The solution g is normalised to the simplex and
   the next iterate is f ← (1‑t)·f + t·g (t≈0.2).  Repeating this step shrinks
   the maximal autocorrelation.

3. **Projected sub‑gradient refinement** – after the LP phase a short
   projected‑gradient descent (using the full sub‑gradient from all active
   shifts) cleans up the remaining slack.

All steps respect the probability simplex {x≥0, Σx=1}; the objective is
scale‑invariant, so the sum is kept equal to 1 throughout.  The best sequence
found (over several lengths) before the time budget expires is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional linear‑programming support
try:
    import scipy.optimize
    _SCIPY_AVAILABLE = True
except Exception:     # pragma: no cover
    _SCIPY_AVAILABLE = False

# ----------------------------------------------------------------------
def _project_onto_simplex(v):
    """Projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power‑of‑two ≥ x."""
    return 1 << (x - 1).bit_length()


def _self_convolve(seq: np.ndarray) -> np.ndarray:
    """Full (non‑circular) self‑convolution using a real FFT."""
    n = seq.shape[0]
    m = 2 * n - 1
    nfft = _next_pow2(m)
    A = np.fft.rfft(seq, n=nfft)
    conv = np.fft.irfft(A * A, n=nfft)[:m]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Return 2·n·max(conv) / (sum(seq)²); assumes seq≥0."""
    s = seq.sum()
    if s < 0.01:
        return float("inf")
    conv = _self_convolve(seq)
    return float(2.0 * len(seq) * conv.max() / (s * s))


# ----------------------------------------------------------------------
# Simple parametric families (all return a normalised probability vector)
def _geom_seq(n: int, alpha: float):
    """Geometric progression a_i ∝ r**i with r = 1‑α/n (α>0)."""
    eps = alpha / n
    r = 1.0 - eps
    if r <= 0.0:
        return None
    powers = r ** np.arange(n, dtype=np.float64)
    seq = powers / powers.sum()
    return seq


def _powerlaw_seq(n: int, p: float):
    """Power‑law a_i ∝ (i+1)^(‑p)."""
    seq = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    return seq / seq.sum()


def _beta_seq(n: int, a: float, b: float):
    """Beta‑like shape a_i ∝ (i+1)^{a‑1}·(n‑i)^{b‑1}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    seq = (i ** (a - 1.0)) * ((n + 1 - i) ** (b - 1.0))
    seq[seq < 0] = 0.0
    return seq / seq.sum()


def _random_simplex(n: int):
    """Draw a random point on the simplex (Dirichlet(1,…,1))."""
    x = np.random.exponential(1.0, size=n)
    return x / x.sum()


# ----------------------------------------------------------------------
def _conv_matrix(seq: np.ndarray) -> np.ndarray:
    """
    Build matrix A (size (2n‑1)×n) such that A@g = seq * g (full convolution).
    Column i of A is ``seq`` shifted by i positions.
    """
    n = seq.shape[0]
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = seq
    return A


def _solve_lp(seq: np.ndarray, M: float):
    """
    Linear program: maximise Σ g_i   subject to   g≥0 and A@g ≤ M·1,
    where A is the convolution matrix built from ``seq``.
    Returns the optimal g (may be None if LP fails).
    """
    if not _SCIPY_AVAILABLE:          # pragma: no cover
        return None

    A = _conv_matrix(seq)
    m = A.shape[0]
    c = -np.ones(seq.shape[0], dtype=np.float64)   # maximise Σg  ⇔ minimise –Σg
    b = np.full(m, M, dtype=np.float64)
    bounds = [(0.0, None)] * seq.shape[0]

    res = scipy.optimize.linprog(
        c, A_ub=A, b_ub=b, bounds=bounds,
        method='highs',
        options={'presolve': True, 'time_limit': 5.0}
    )
    if res.success and res.x is not None:
        return res.x
    return None


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: float = 1000.0, **kwargs):
    """
    Search for a step‑function with low evaluation score.
    Returns the best list of heights found before the time budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Lengths to try (larger lengths give more freedom but cost more)
    lengths = [200, 400, 800, 1200, 2000]

    for n in lengths:
        if time.time() > deadline:
            break

        # ------------------------------------------------------------------
        # 1. Build a collection of seed vectors for this length.
        seeds = []

        # global seed if present (might have different length)
        if "height_sequence_1" in globals():
            try:
                g = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
                if g.size >= n:
                    seq = g[:n]
                else:
                    # pad with zeros – they are harmless but increase n,
                    # so we instead repeat the pattern to keep length n.
                    seq = np.concatenate([g, np.zeros(n - g.size)])
                seq = np.clip(seq, 0.0, 1000.0)
                if seq.sum() > 0:
                    seq = seq / seq.sum()
                    seeds.append(seq)
            except Exception:
                pass

        # systematic parametric families
        for alpha in np.linspace(0.5, 4.0, 7):
            seq = _geom_seq(n, alpha)
            if seq is not None:
                seeds.append(seq)

        for p in np.linspace(0.2, 2.2, 7):
            seeds.append(_powerlaw_seq(n, p))

        for a in [0.5, 1.0, 2.0]:
            for b in [0.5, 1.0, 2.0]:
                seeds.append(_beta_seq(n, a, b))

        # a handful of random points
        for _ in range(10):
            seeds.append(_random_simplex(n))

        # ------------------------------------------------------------------
        # 2. Run optimisation from each seed (stop early on time out)
        for init_seq in seeds:
            if time.time() > deadline:
                break

            seq = init_seq.copy()
            cur_score = _objective(seq)

            if cur_score < best_score:
                best_score = cur_score
                best_seq = seq.copy()
                print(f"[Init n={n}] score={cur_score:.6f}")

            # ------------------------------------------------------------------
            # 2a. LP‑driven re‑balancing (up to a modest number of iterations)
            t = 0.20                     # step size for convex combination
            max_lp_iters = 30
            for lp_iter in range(max_lp_iters):
                if time.time() > deadline:
                    break

                conv_ff = _self_convolve(seq)
                M = conv_ff.max()
                g0 = _solve_lp(seq, M)
                if g0 is None or g0.sum() < 1e-12:
                    break

                # normalise g to the simplex (sum = 1)
                g = g0 / g0.sum()
                # convex update
                seq = (1.0 - t) * seq + t * g

                cur_score = _objective(seq)
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = seq.copy()
                    print(f"[LP   n={n} it={lp_iter:02d}] score={cur_score:.6f}")

                # optional: if improvement stalls, shrink the mixing factor
                if lp_iter > 0 and cur_score >= prev_score:
                    t = max(t * 0.5, 0.01)
                prev_score = cur_score

            # ------------------------------------------------------------------
            # 2b. Projected sub‑gradient descent for fine tuning
            it = 0
            base_step = 5.0
            max_grad_iters = 2000
            while it < max_grad_iters and time.time() < deadline:
                conv_full = _self_convolve(seq)
                max_val = conv_full.max()
                # collect *all* active shifts (within a tiny tolerance)
                active = np.where(np.abs(conv_full - max_val) <= 1e-12)[0]

                # build sub‑gradient (sum over all active shifts)
                grad = np.zeros_like(seq)
                for k in active:
                    lo = max(0, k - (n - 1))
                    hi = min(k, n - 1)
                    idx = np.arange(lo, hi + 1, dtype=int)
                    grad[idx] += seq[k - idx]            # g_i = a_{k‑i}

                # back‑tracking line search
                step = base_step
                improved = False
                while step > 1e-7:
                    cand = _project_onto_simplex(seq - step * grad)
                    cand_score = _objective(cand)
                    if cand_score < cur_score:
                        improved = True
                        break
                    step *= 0.5

                if improved:
                    seq = cand
                    cur_score = cand_score
                    if cand_score < best_score:
                        best_score = cand_score
                        best_seq = seq.copy()
                        print(f"[Grad n={n} it={it:04d}] score={cand_score:.6f}")
                    # modestly increase step size for next iteration
                    base_step = max(step * 1.2, 1e-4)
                else:
                    # small random mass transfer if line‑search fails
                    i, j = np.random.choice(n, size=2, replace=False)
                    delta = 0.01 * min(seq[i], 0.1)
                    if delta > 0:
                        cand = seq.copy()
                        cand[i] -= delta
                        cand[j] += delta
                        cand = _project_onto_simplex(cand)
                        cand_score = _objective(cand)
                        if cand_score < cur_score:
                            seq = cand
                            cur_score = cand_score
                            if cand_score < best_score:
                                best_score = cand_score
                                best_seq = seq.copy()
                                print(f"[Rand n={n} it={it:04d}] score={cand_score:.6f}")
                it += 1

    # ------------------------------------------------------------------
    if best_seq is None:                     # fallback (should never happen)
        best_seq = np.full(200, 1.0 / 200)
        best_score = _objective(best_seq)

    print(f"Final best objective = {best_score:.6f} (length {best_seq.shape[0]})")
    return best_seq.tolist()
