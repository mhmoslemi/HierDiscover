"""
Lightweight optimizer for the step‑function bound.

The algorithm runs many short independent restarts.  For each chosen length
``n`` it starts from a monotone (sorted) random vector or from the previous
best construction (``height_sequence_1`` if present).  The vector is kept on
the simplex Σa_i = √(2n) – this makes the objective exactly the maximal
convolution value.

A smooth approximation of the max‑operator is used:
    f_τ(a) = τ·log ∑_k exp(conv_k / τ)   (τ → 0 gives the true max)
The gradient of f_τ is a weighted sum of the gradients of the individual
convolution entries, which can be evaluated efficiently by a single FFT‑based
convolution:
    grad_i = 2·∑_j a_j·w_{i+j},
where w are the soft‑max weights.

A projected (heavy‑ball) gradient step is taken, followed by occasional
random “shake’’ steps to escape shallow local minima.  The smoothing parameter
τ is slowly decreased, and the learning‑rate factor is annealed.  The best
vector seen over the whole run is returned.

The implementation uses only NumPy and the standard library and respects the
required 1000‑second budget.
"""
import time
import numpy as np

# --------------------------------------------------------------------------- #
# Helper 1 – Euclidean projection onto the simplex {x_i ≥ 0, Σ x_i = s}.
# --------------------------------------------------------------------------- #
def project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector ``v`` onto the simplex of radius ``s``."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]                 # descending sort
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------------------- #
# Helper 2 – Full convolution via real FFT (fast, works for any two vectors).
# --------------------------------------------------------------------------- #
def fft_convolve_full(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Return the full (linear) convolution of ``a`` and ``b`` using FFT."""
    n = a.shape[0] + b.shape[0] - 1
    n_fft = 1 << ((n - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=n_fft)
    B = np.fft.rfft(b, n=n_fft)
    conv = np.fft.irfft(A * B, n=n_fft)[:n]
    return conv


# --------------------------------------------------------------------------- #
# Helper 3 – Objective (exactly the value returned by evaluate_sequence for
#             a vector that satisfies Σa = √(2n)).
# --------------------------------------------------------------------------- #
def max_convolution_value(a: np.ndarray) -> float:
    """Compute max_{k} (a * a)_k for a given vector a."""
    conv = fft_convolve_full(a, a)
    return conv.max()


# --------------------------------------------------------------------------- #
# Helper 4 – Gradient of the τ‑smoothed max‑convolution.
# --------------------------------------------------------------------------- #
def smoothed_gradient(a: np.ndarray, conv: np.ndarray, tau: float) -> np.ndarray:
    """
    Return the gradient of the smoothed max function
    f_τ(a) = τ·log ∑_k exp(conv_k/τ).
    """
    if tau <= 0:
        raise ValueError("tau must be positive")
    # soft‑max weights (stable: subtract max)
    maxc = conv.max()
    w = np.exp((conv - maxc) / tau)
    w_sum = w.sum()
    if w_sum == 0:
        # fallback to uniform weights (should not happen)
        w = np.full_like(conv, 1.0 / conv.size)
    else:
        w /= w_sum                     # now Σ w_k = 1

    # gradient = 2 * correlation of a with w (no reversal)
    # correlation can be obtained via convolution with the reversed weight vector
    w_rev = w[::-1]
    grad_full = fft_convolve_full(a, w_rev)          # length = 3n‑2
    n = a.shape[0]
    idx = (2 * n - 2) - np.arange(n)                 # indices for valid part
    grad = 2.0 * grad_full[idx]
    return grad


# --------------------------------------------------------------------------- #
# Main entry point – required by the grader.
# --------------------------------------------------------------------------- #
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a step‑function (non‑negative heights) that minimizes
    the evaluation function within the allotted time budget.
    Returns a plain Python list of the best sequence found.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5          # a small safety margin

    best_seq = None
    best_val = float("inf")

    # reuse the previously discovered construction if it exists
    prev_global = globals().get("height_sequence_1")
    if isinstance(prev_global, (list, tuple, np.ndarray)):
        prev_global = np.asarray(prev_global, dtype=float)

    # ------------------------------------------------------------------- #
    # One optimisation run for a given length n.
    # ------------------------------------------------------------------- #
    def optimise_length(n: int, init_vec: np.ndarray):
        nonlocal best_seq, best_val

        target_sum = np.sqrt(2.0 * n)          # Σ a_i = √(2n)

        # start from the supplied vector, project onto the simplex
        a = project_onto_simplex(init_vec.astype(float), target_sum)

        # hyper‑parameters (they will be annealed during the run)
        base_lr = 0.30                         # learning‑rate factor
        beta_mom = 0.90                        # heavy‑ball momentum coefficient
        momentum = np.zeros_like(a)

        tau = 1.0                              # smoothing parameter
        tau_min = 0.001
        tau_decay = 0.9995

        max_iters = 4000
        it = 0
        last_improve_it = 0

        while time.time() < deadline and it < max_iters:
            # ---- convolution and current objective ---------------------------------
            conv = fft_convolve_full(a, a)                 # length 2n‑1
            maxb = conv.max()
            # because Σa = √(2n) the objective equals maxb
            cur_val = float(maxb)

            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                print(f"[{time.time():.2f}] New best = {best_val:.6f} (n={n})")
                last_improve_it = it
                # early exit if we beat the target bound
                if best_val < 1.503:
                    return

            # ---- update smoothing parameter ---------------------------------------
            if tau > tau_min:
                tau *= tau_decay

            # ---- gradient of the smoothed max ------------------------------------
            grad = smoothed_gradient(a, conv, tau)

            # ---- step size (scale‑invariant) ------------------------------------
            # use a step proportional to the current max value divided by
            # the gradient magnitude – similar spirit to the original code
            grad_norm = np.linalg.norm(grad) + 1e-12
            eta = base_lr * maxb / grad_norm

            # ---- heavy‑ball momentum update ---------------------------------------
            momentum = beta_mom * momentum - eta * grad
            a_new = a + momentum

            # ---- projection onto the feasible simplex -------------------------------
            a_new = np.maximum(a_new, 0.0)                   # non‑negativity
            a_new = project_onto_simplex(a_new, target_sum)

            # ---- occasional random shake (simulated‑annealing style) -------------
            if np.random.rand() < 0.02:
                sigma = 0.02 * target_sum / n
                noise = np.random.randn(n) * sigma
                a_new = np.maximum(a_new + noise, 0.0)
                a_new = project_onto_simplex(a_new, target_sum)
                base_lr *= 0.92                               # slowly shrink step

            a = a_new

            # ---- minor learning‑rate decay ----------------------------------------
            base_lr *= 0.9999

            # ---- stagnation check – restart if no progress for a while ----------
            if it - last_improve_it > 1500:
                break               # give up this length, start a new one

            it += 1

    # ------------------------------------------------------------------- #
    # Outer loop – many restarts with varying lengths.
    # ------------------------------------------------------------------- #
    while time.time() < deadline:
        # Choose a problem size.  The previously successful size was ≈1480,
        # so we favour that neighbourhood but also explore a broader range.
        if np.random.rand() < 0.6:
            n = np.random.randint(1300, 1701)   # focused band
        else:
            n = np.random.randint(1200, 3001)   # full allowed band

        # Initialise
        if prev_global is not None and prev_global.size >= n and np.random.rand() < 0.45:
            init = prev_global[:n].copy()
        else:
            # random monotone start – sort descending to encourage asymmetry
            init = np.random.rand(n)
            init = np.sort(init)[::-1]

        optimise_length(n, init)

        # Stop early if we have already reached the ambitious target.
        if best_val < 1.503:
            break

    # ------------------------------------------------------------------- #
    # Fallback – if nothing was found (should never happen).
    # ------------------------------------------------------------------- #
    if best_seq is None:
        n = 1000
        target_sum = np.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)

    return best_seq.tolist()
