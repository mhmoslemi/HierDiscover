"""
Search for a step‑function (non‑negative heights) that minimises

        2 * n * max(conv(a, a)) / (sum(a) ** 2)

The objective is equal to the maximal entry of the *full* convolution
because we keep the vector on the simplex Σa_i = √(2n).  The algorithm
combines three ideas:

1.  A fast projected (Adam) gradient descent on the sub‑gradient of the
    maximal convolution entry.
2.  Periodic random “shakes’’ (small Gaussian perturbations) to escape
    local minima.
3.  A short simulated‑annealing refinement that moves a tiny amount of
    mass between two coordinates.

Many independent restarts with different problem sizes (up to a few
thousand points) are performed until the allotted time budget expires.
The best vector found is returned as a plain Python list.
"""

import time
import numpy as np

# --------------------------------------------------------------------------- #
# Helper: projection onto the simplex {x_i ≥ 0 , Σ x_i = s}.
# --------------------------------------------------------------------------- #
def project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex of radius s."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------------------- #
# Fast full convolution (real FFT) – returns an array of length 2*n‑1.
# --------------------------------------------------------------------------- #
def full_conv_fft(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    n_fft = 1 << ((2 * n - 1).bit_length())      # next power of two
    A = np.fft.rfft(a, n=n_fft)
    conv = np.fft.irfft(A * A, n=n_fft)[: 2 * n - 1]
    return conv


# --------------------------------------------------------------------------- #
# Gradient of the *maximal* convolution entry (or a set of ties).
# grad_i = Σ_{k∈active} 2 * a_{k-i}   (only when the index is inside the overlap)
# --------------------------------------------------------------------------- #
def active_gradient(a: np.ndarray, active_idxs: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in active_idxs:
        low = max(0, k - (n - 1))
        high = min(n - 1, k)
        idx = np.arange(low, high + 1, dtype=int)
        partner = k - idx
        grad[idx] += 2.0 * a[partner]
    return grad


# --------------------------------------------------------------------------- #
# Compute the maximal entry of the convolution (the objective value).
# --------------------------------------------------------------------------- #
def conv_max(a: np.ndarray) -> float:
    return float(full_conv_fft(a).max())


# --------------------------------------------------------------------------- #
# Adam‑based projected gradient descent for a fixed amount of wall‑clock time.
# --------------------------------------------------------------------------- #
def optimise_sequence(
    a0: np.ndarray,
    target_sum: float,
    time_limit: float,
    *,
    base_lr: float = 0.5,
    shake_prob: float = 0.02,
) -> tuple[np.ndarray, float]:
    """
    Returns (best_vector, best_value) found within `time_limit` seconds.
    The vector is kept on the simplex Σa_i = target_sum.
    """
    n = a0.shape[0]
    a = project_onto_simplex(a0.astype(float), target_sum)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    best_a = a.copy()
    best_val = conv_max(a)

    start = time.time()
    while time.time() - start < time_limit:
        # current objective
        cur_val = conv_max(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.2f}]   New local best = {best_val:.6f} (n={n})")

        # sub‑gradient of the active (max‑tight) convolution entries
        conv = full_conv_fft(a)
        max_val = conv.max()
        active = np.where(conv >= max_val - 1e-12)[0]
        grad = active_gradient(a, active)

        # Adam update (descent direction = -grad)
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step                     # descend
        a = np.maximum(a, 0.0)           # non‑negativity
        a = project_onto_simplex(a, target_sum)

        # occasional random shake (helps escape shallow minima)
        if np.random.rand() < shake_prob:
            noise = np.random.randn(n) * (target_sum * 0.001 / n)
            a = a + noise
            a = np.maximum(a, 0.0)
            a = project_onto_simplex(a, target_sum)

        # a small learning‑rate decay to fine‑tune at the end
        base_lr *= 0.99997

    return best_a, best_val


# --------------------------------------------------------------------------- #
# Simple simulated‑annealing refinement that moves a tiny amount of mass.
# --------------------------------------------------------------------------- #
def anneal_refine(
    a0: np.ndarray,
    target_sum: float,
    time_limit: float,
    init_temp: float = 0.3,
    cooling: float = 0.9995,
) -> tuple[np.ndarray, float]:
    """
    Runs a cheap annealing loop for `time_limit` seconds.
    Returns the best vector seen.
    """
    a = a0.copy()
    cur_val = conv_max(a)
    best_a = a.copy()
    best_val = cur_val
    n = a.shape[0]
    T = init_temp * cur_val  # temperature scales with current objective
    start = time.time()

    while time.time() - start < time_limit:
        # propose a move: transfer a small amount from i to j
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # magnitude of the transfer (a few percent of the average entry)
        delta = np.random.rand() * target_sum * 0.005 / n
        if a[i] < delta:
            delta = a[i]          # cannot go negative
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        # compute new objective
        new_val = conv_max(a_new)

        # Metropolis acceptance
        if new_val < cur_val or np.random.rand() < np.exp(-(new_val - cur_val) / max(T, 1e-12)):
            a = a_new
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
                print(f"[{time.time():.2f}]   New annealed best = {best_val:.6f} (n={n})")

        # temperature schedule
        T *= cooling

    return best_a, best_val


# --------------------------------------------------------------------------- #
# Main entry point – the grader will call this function.
# --------------------------------------------------------------------------- #
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a low‑value step‑function using many restarts.
    The best sequence found before the time budget expires is returned.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    best_seq = None
    best_score = float("inf")                       # this is the maximal convolution value

    # ------------------------------------------------------------------- #
    # Helper to run a full optimisation (gradient + annealing) for a given n.
    # ------------------------------------------------------------------- #
    def run_one_size(n: int):
        nonlocal best_seq, best_score

        target_sum = np.sqrt(2.0 * n)                # Σ a_i = √(2n)

        # ---- initialise -------------------------------------------------
        # Mix of a uniform part and a gentle sine‑shaped part.
        uniform = np.full(n, target_sum / n)
        sine = np.sin(np.pi * np.arange(n) / (n - 1))   # 0 … 1 … 0
        sine = sine / sine.sum() * target_sum           # scale to same L1
        init = 0.6 * uniform + 0.4 * sine
        # add a bit of noise
        init += np.random.randn(n) * (target_sum * 1e-3 / n)
        init = np.clip(init, 0.0, None)

        # Occasionally reuse a good old construction if it is long enough.
        if np.random.rand() < 0.3 and "height_sequence_1" in globals():
            src = np.asarray(height_sequence_1, dtype=float)
            if src.size >= n:
                init = src[:n].copy()
            else:
                pad = np.random.rand(n - src.size) * (target_sum * 1e-3 / n)
                init = np.concatenate([src, pad])

        # ---- allocate time slices for this size -------------------------
        remaining = deadline - time.time()
        if remaining <= 0:
            return
        # about 25 s for gradient descent, the rest for annealing
        grad_time = min(25.0, remaining * 0.7)
        anneal_time = min(10.0, remaining - grad_time)

        # ---- gradient descent -------------------------------------------
        seq, val = optimise_sequence(init, target_sum, grad_time)
        if val < best_score:
            best_score = val
            best_seq = seq.copy()
            print(f"[{time.time():.2f}] New global best (gradient) = {best_score:.6f} (n={n})")

        # ---- annealing refinement ---------------------------------------
        if anneal_time > 0.5:
            seq2, val2 = anneal_refine(seq, target_sum, anneal_time)
            if val2 < best_score:
                best_score = val2
                best_seq = seq2.copy()
                print(f"[{time.time():.2f}] New global best (anneal) = {best_score:.6f} (n={n})")

    # ------------------------------------------------------------------- #
    # Main outer loop – random problem sizes, many restarts.
    # ------------------------------------------------------------------- #
    while time.time() < deadline:
        # pick a length – larger lengths give more freedom but cost more
        n = np.random.randint(1800, 5001)   # 1.8k … 5k
        run_one_size(n)

        # early stop if we already surpass the target bound
        if best_score < 1.503:
            break

    # ------------------------------------------------------------------- #
    # Fallback in the unlikely event that nothing was found.
    # ------------------------------------------------------------------- #
    if best_seq is None:
        n = 2000
        target_sum = np.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)

    # Return a plain Python list as required by the grader
    return best_seq.tolist()
