def cross_corr_grad(a, w):
    # a length n, w length 2n-1
    rev_a = a[::-1]
    size = 1 << int(np.ceil(np.log2(len(w) + len(a) - 1)))
    fw = np.fft.rfft(w, size)
    frev = np.fft.rfft(rev_a, size)
    corr = np.fft.irfft(fw * frev, size)
    n = len(a)
    grad = 2.0 * corr[n-1:n-1+n]
    return grad

"""
Propose a step‑function sequence with a low evaluation score.

The algorithm is a projected Adam optimizer on a smooth‑max surrogate of the
self‑convolution maximum.  The surrogate uses a temperature λ that is
gradually increased (continuation) so the optimisation first smooths the
objective and later behaves like a sub‑gradient descent on the true max.
After each Adam step the vector is projected onto the probability simplex
(non‑negative, sum = 1).  Random “shakes” are added when progress stalls.
Multiple cheap initialisations (geometric, Dirichlet, possible global
candidate) are tried and the best ever sequence is returned.

The method avoids the aggressive step‑size decay of the previous solution,
allowing continued improvement within the 1000‑second budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Projection onto the probability simplex {x ≥ 0, sum(x)=1}
def _project_onto_simplex(v):
    """Euclidean projection of a vector onto the simplex."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        # all non‑positive -> uniform distribution
        return np.full_like(v, 1.0 / v.shape[0])
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear (full) convolution a*a using FFT
def _full_convolution(a, size=None):
    """Linear self‑convolution a*a, length 2·len(a)‑1."""
    n = a.shape[0]
    if size is None:
        size = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Gradient of the smooth‑max surrogate.
# Input:
#   a   – current sequence (length n)
#   w   – weight vector over convolution indices (length 2n‑1), sums to 1
# Output:
#   grad – ∇  (1/λ)·log∑ e^{λ·c_k} = 2·(w ★ a)  (length n)
def _cross_corr_gradient(a, w, size_corr):
    """Compute 2·(w ★ a) using FFT‑based cross‑correlation."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, size_corr)
    frev = np.fft.rfft(rev_a, size_corr)
    corr = np.fft.irfft(fw * frev, size_corr)
    n = a.shape[0]
    grad = 2.0 * corr[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Exact objective used by `evaluate_sequence`.
def _objective(a):
    """2·n·max(conv) / (sum(a)²); a is assumed non‑negative."""
    n = a.shape[0]
    conv = _full_convolution(a)
    max_b = conv.max()
    sum_a = a.sum()
    # sum_a is forced to be 1 by construction, but keep formula general.
    return 2.0 * n * max_b / (sum_a * sum_a)


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    `evaluate_sequence`.  Returns the best sequence found before the time
    budget expires.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # Choose a problem size.  2000 provides a good trade‑off between
    # expressiveness and speed.
    n = 2000

    # ------------------------------------------------------------------
    # Helpers for the optimisation loop.
    size_fft = 1 << int(np.ceil(np.log2(2 * n - 1)))      # for self‑conv
    size_corr = 1 << int(np.ceil(np.log2(3 * n)))        # for correlation

    # ------------------------------------------------------------------
    # Initialise with a few cheap candidates and keep the best.
    best_seq = None
    best_val = float("inf")

    # 1) possible global variable from a previous run
    try:
        global height_sequence_1
        cand = np.asarray(height_sequence_1, dtype=float)
        if cand.shape[0] == n:
            cand = _project_onto_simplex(cand)
            val = _objective(cand)
            if val < best_val:
                best_val, best_seq = val, cand.copy()
                print("[Init global] obj={:.6f}".format(val))
    except Exception:
        pass

    # 2) a few geometric progressions (a_i ∝ r^i)
    factors = np.linspace(0.8, 3.0, 20)
    for f in factors:
        if time.time() > deadline:
            break
        r = 1.0 - f / n
        if r <= 0:
            continue
        powers = np.power(r, np.arange(n, dtype=np.float64))
        seq = powers / powers.sum()
        val = _objective(seq)
        if val < best_val:
            best_val, best_seq = val, seq.copy()
            print("[Init geom] f={:.3f}  obj={:.6f}".format(f, val))

    # 3) several random Dirichlet starts
    for _ in range(12):
        if time.time() > deadline:
            break
        seq = np.random.dirichlet(np.ones(n))
        val = _objective(seq)
        if val < best_val:
            best_val, best_seq = val, seq.copy()
            print("[Init rand] obj={:.6f}".format(val))

    # Fallback – uniform distribution (should never be used)
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)
        best_val = _objective(best_seq)

    # ------------------------------------------------------------------
    # Optimisation parameters (Adam)
    seq = best_seq.copy()                     # current iterate
    m = np.zeros_like(seq)                    # 1st moment
    v = np.zeros_like(seq)                    # 2nd moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.12                                 # base learning rate
    lam = 5.0                                 # temperature for smooth‑max
    lam_factor = 1.02                         # continuation scaling
    iteration = 0
    last_improve = time.time()

    # ------------------------------------------------------------------
    while time.time() < deadline:
        iteration += 1

        # 1) self‑convolution
        conv = _full_convolution(seq, size=size_fft)

        # 2) soft‑max weights (smooth‑max surrogate)
        c_max = conv.max()
        # shift by max to keep numbers in a stable range
        w_raw = np.exp(lam * (conv - c_max))
        w_sum = w_raw.sum()
        # guard against under‑flow (should not happen because max term gives 1)
        if w_sum == 0.0:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w = w_raw / w_sum

        # 3) gradient (2·(w ★ a))
        grad = _cross_corr_gradient(seq, w, size_corr)

        # 4) Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        # bias‑corrected estimates
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        # optional clipping to keep steps moderate
        max_step = 0.05
        if np.max(np.abs(step)) > max_step:
            step = step * (max_step / np.max(np.abs(step)))

        new_seq = seq - step
        new_seq = _project_onto_simplex(new_seq)

        # 5) evaluate exact objective
        val = _objective(new_seq)

        # 6) keep the best ever found
        if val < best_val:
            best_val, best_seq = val, new_seq.copy()
            last_improve = time.time()
            print("[Iter {:06d}] new best obj = {:.6f}".format(iteration, val))

        # 7) shake if we are stuck for a few seconds
        if time.time() - last_improve > 3.0:
            noise = (np.random.rand(n) - 0.5) * 1e-3
            new_seq = _project_onto_simplex(new_seq + noise)
            last_improve = time.time()
            # reset Adam moments a little to avoid being stuck in stale direction
            m *= 0.5
            v *= 0.5

        # 8) update for next iteration
        seq = new_seq
        lam = lam * lam_factor

        # occasional progress report (every ~30 s) without flooding output
        if iteration % 2000 == 0:
            elapsed = time.time() - (deadline - budget_s + 0.5)
            print("[Iter {:06d}] elapsed {:.1f}s, cur obj {:.6f}, best {:.6f}".format(
                iteration, elapsed, val, best_val))

    # ------------------------------------------------------------------
    print("Final best objective = {:.6f} (length {})".format(best_val, n))
    # `evaluate_sequence` expects a plain Python list
    return best_seq.tolist()
