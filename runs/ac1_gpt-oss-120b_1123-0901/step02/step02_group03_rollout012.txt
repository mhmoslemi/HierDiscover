def run_optim_instance(n, init_vec, time_limit):
    import time, numpy as np
    target_sum = np.sqrt(2.0 * n)
    a = _project_onto_simplex(init_vec.astype(float), target_sum)
    best_seq_local = a.copy()
    conv = _conv_full_fft(a)
    best_score_local = conv.max()
    # Adam hyperparameters
    lr = 0.1
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t_it = 0
    # temperature (softmax)
    tau = None
    run_deadline = time.time() + time_limit
    while time.time() < run_deadline:
        conv = _conv_full_fft(a)
        max_conv = conv.max()
        if max_conv < best_score_local:
            best_score_local = max_conv
            best_seq_local = a.copy()
        # update tau
        if tau is None:
            tau = max_conv * 0.2  # initial smoothing
        else:
            tau = max(tau * 0.98, max_conv * 1e-6)  # decay, keep >0
        # softmax weights
        # subtract max for stability
        w_raw = np.exp((conv - max_conv) / tau)
        w_sum = w_raw.sum()
        if w_sum == 0 or np.isinf(w_sum) or np.isnan(w_sum):
            # fallback to active set gradient
            active = np.where(conv >= max_conv - 1e-12)[0]
            grad = _gradient_of_active(a, active)
        else:
            w = w_raw / w_sum
            # compute correlation gradient
            nfft = 1 << ((n + (2*n-1) - 1).bit_length())
            A = np.fft.rfft(a, n=nfft)
            W = np.fft.rfft(w, n=nfft)
            corr_full = np.fft.irfft(A * np.conj(W), n=nfft)
            start = (2*n-2)
            grad = 2.0 * corr_full[start:start + n]
        # Adam update
        t_it += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t_it)
        v_hat = v / (1 - beta2 ** t_it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        # enforce constraints
        a = np.maximum(a, 0.0)
        a = _project_onto_simplex(a, target_sum)
        # occasional random perturbation
        if np.random.rand() < 0.01:
            noise = np.random.randn(n) * (target_sum * 0.01 / np.sqrt(n))
            a = _project_onto_simplex(np.maximum(a + noise, 0.0), target_sum)
            # resetting Adam momentum may help
            m = np.zeros_like(a)
            v = np.zeros_like(a)
            t_it = 0
    return best_seq_local, best_score_local

"""
Propose a step‑function (non‑negative heights) that minimizes

        2 * n * max(conv(a, a)) / (sum(a) ** 2)

using a projected Adam optimizer on a smooth (soft‑max) surrogate of the
maximum convolution value.  The algorithm works on the simplex
{a_i ≥ 0, Σ a_i = √(2n)} so the objective equals the maximal convolution
term.  For each restart we:
    • initialise a (either from the known good construction or at random)
    • repeatedly compute the full autocorrelation via a real FFT
    • form a soft‑max over the convolution entries (temperature τ → 0)
    • obtain a (sub‑)gradient of the soft‑max using a correlation via FFT
    • perform an Adam update and project back onto the simplex
    • occasionally add a small random shake to escape plateaus.
The best sequence seen over all restarts (within the time budget) is
returned as a plain python list.
"""

import time
import numpy as np

# --------------------------------------------------------------------------- #
# Helper: projection onto the simplex {x_i ≥ 0 , Σ x_i = s}.
# --------------------------------------------------------------------------- #
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex of radius s."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------------------- #
# Fast full convolution (real FFT) – returns a length 2*n‑1 array.
# --------------------------------------------------------------------------- #
def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    """Return a * a (linear convolution) using a real FFT."""
    n = a.shape[0]
    n_fft = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    conv = np.fft.irfft(A * A, n=n_fft)[: 2 * n - 1]
    return conv


# --------------------------------------------------------------------------- #
# Gradient of the active (max‑tying) convolution entries (fallback).
# --------------------------------------------------------------------------- #
def _gradient_of_active(a: np.ndarray, active_idxs: np.ndarray) -> np.ndarray:
    """Sub‑gradient obtained from the set of active (max‑tying) indices."""
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in active_idxs:
        low = max(0, k - (n - 1))
        high = min(n - 1, k)
        idx = np.arange(low, high + 1, dtype=int)
        partner = k - idx
        grad[idx] += 2.0 * a[partner]
    return grad


# --------------------------------------------------------------------------- #
# Gradient of the soft‑max surrogate of max(conv) via FFT correlation.
# --------------------------------------------------------------------------- #
def _gradient_via_softmax(a: np.ndarray, conv: np.ndarray, tau: float) -> np.ndarray:
    """
    Compute ∇ [ τ·log Σ exp(conv_k/τ) ].
    The result equals 2 · correlate(a, w) where w are soft‑max weights.
    """
    n = a.shape[0]
    max_val = conv.max()
    # stable soft‑max
    shifted = (conv - max_val) / max_val if max_val != 0 else conv
    exps = np.exp((conv - max_val) / tau)
    w_sum = exps.sum()
    if w_sum == 0 or not np.isfinite(w_sum):
        # fall back to the active set if numerics fail
        active = np.where(conv >= max_val - 1e-12)[0]
        return _gradient_of_active(a, active)

    w = exps / w_sum                     # length 2*n‑1
    len_w = w.shape[0]                   # = 2*n‑1
    nfft = 1 << ((n + len_w - 1).bit_length())   # >= 3*n‑2
    A = np.fft.rfft(a, n=nfft)
    W = np.fft.rfft(w, n=nfft)
    corr_full = np.fft.irfft(A * np.conj(W), n=nfft)
    start = len_w - 1                    # = 2*n‑2
    grad = 2.0 * corr_full[start:start + n]
    return grad


# --------------------------------------------------------------------------- #
# Run a single optimisation instance for a given length ``n``.
# --------------------------------------------------------------------------- #
def _run_one_instance(
    n: int,
    init_vec: np.ndarray,
    time_limit: float,
) -> tuple[np.ndarray, float]:
    """
    Optimise a sequence of length ``n`` for at most ``time_limit`` seconds.
    Returns the best sequence found and its (max‑convolution) score.
    """
    target_sum = np.sqrt(2.0 * n)          # Σ a_i = √(2n)
    a = _project_onto_simplex(init_vec.astype(float), target_sum)

    # best local record
    conv = _conv_full_fft(a)
    best_seq = a.copy()
    best_score = conv.max()

    # Adam hyper‑parameters
    lr = 0.12
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t_it = 0

    # temperature (soft‑max smoothing)
    tau = None

    deadline = time.time() + time_limit
    while time.time() < deadline:
        # ----- convolution & current value -----
        conv = _conv_full_fft(a)
        cur_max = conv.max()
        if cur_max < best_score:
            best_score = cur_max
            best_seq = a.copy()
        # ----- temperature schedule -----
        if tau is None:
            tau = cur_max * 0.2               # start relatively smooth
        else:
            tau = max(tau * 0.98, cur_max * 1e-6)

        # ----- gradient (soft‑max surrogate) -----
        grad = _gradient_via_softmax(a, conv, tau)

        # ----- Adam update -----
        t_it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t_it)
        v_hat = v / (1.0 - beta2 ** t_it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step                     # descent
        a = np.maximum(a, 0.0)           # non‑negativity
        a = _project_onto_simplex(a, target_sum)

        # ----- occasional random shake (helps escape plateaus) -----
        if np.random.rand() < 0.015:
            noise = np.random.randn(n) * (target_sum * 0.01 / np.sqrt(n))
            a = _project_onto_simplex(np.maximum(a + noise, 0.0), target_sum)
            # reset Adam state after a shake
            m.fill(0.0)
            v.fill(0.0)
            t_it = 0

    return best_seq, best_score
