def propose_candidate(seed=42, budget_s=1000.0, **kwargs):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5

    best_seq = None
    best_score = float('inf')
    
    # Helper to evaluate and update best
    def try_candidate(arr):
        nonlocal best_seq, best_score
        score = _evaluate(arr.tolist())
        if score < best_score:
            best_score = score
            best_seq = arr.copy()
            print(f"[{time.time():.2f}] New best = {best_score:.6f} (len={len(arr)})")
        return score

    # Initialize from global if available
    init = globals().get('height_sequence_1')
    if init:
        arr = np.array([float(x) for x in init])
        n = len(arr)
        target = math.sqrt(2.0 * n)
        arr = _project_onto_simplex(arr, target)
        try_candidate(arr)

    # Phase 1: parametric families
    lengths = list(range(300, 2001, 300)) + [2500]
    for n in lengths:
        target = math.sqrt(2.0 * n)
        i = np.arange(n, dtype=float)

        # exponential from left
        for alpha in np.logspace(-4, -0.2, num=10):
            a = np.exp(-alpha * i)
            a = _project_onto_simplex(a, target)
            try_candidate(a)
        # reversed exponential
        for alpha in np.logspace(-4, -0.2, num=10):
            a = np.exp(-alpha * (n-1 - i))
            a = _project_onto_simplex(a, target)
            try_candidate(a)
        # U-shaped exponential (min distance to edge)
        for alpha in np.logspace(-4, -0.2, num=10):
            d = np.minimum(i, n-1 - i)
            a = np.exp(-alpha * d)
            a = _project_onto_simplex(a, target)
            try_candidate(a)
        # Sum of two exponentials
        for alpha in np.logspace(-4, -0.2, num=8):
            a = np.exp(-alpha * i) + np.exp(-alpha * (n-1 - i))
            a = _project_onto_simplex(a, target)
            try_candidate(a)
        # Polynomial edge: (i+1)^p + (n-i)^p
        for p in np.linspace(0.5, 3.0, num=6):
            a = (i+1)**p + (n-i)**p
            a = _project_onto_simplex(a, target)
            try_candidate(a)

        if time.time() >= deadline:
            break

    # Phase 2: smooth gradient descent
    def run_opt(start_vec, n):
        nonlocal best_seq, best_score
        a = start_vec.copy()
        target = math.sqrt(2.0 * n)
        m = np.zeros_like(a)
        beta = 0.9
        base_step = 0.5
        lam_start = 5.0
        lam_end = 30.0
        max_iter = 5000
        eval_every = 10

        for it in range(max_iter):
            if time.time() >= deadline:
                break

            conv = _conv_self(a)
            # schedule lam
            lam = lam_start + (lam_end - lam_start) * min(1.0, it / max_iter)
            # compute weighted gradient
            max_c = conv.max()
            exp_vals = np.exp(lam * (conv - max_c))
            w = exp_vals / exp_vals.sum()
            w_rev = w[::-1]
            # convolution a with w_rev
            full = _conv_fft(a, w_rev)
            n_loc = n
            grad = 2 * full[n_loc-1:2*n_loc-1][::-1]

            direction = -grad
            m = beta * m + (1-beta) * direction

            step = base_step / math.sqrt(it + 1)
            max_abs = np.max(np.abs(m))
            if max_abs == 0:
                max_abs = 1.0
            eta = step / max_abs

            a_new = a + eta * m
            a_new = _project_onto_simplex(a_new, target)
            a = a_new

            if it % eval_every == 0:
                try_candidate(a)

            # early break if best score already low
            if best_score < 1.503:
                break

    # Run optimization on current best if exists
    if best_seq is not None:
        n = len(best_seq)
        run_opt(best_seq, n)

    # Random restarts
    while time.time() < deadline:
        n = np.random.randint(500, 2501)
        target = math.sqrt(2.0 * n)
        # random shape: blend two exponential sides
        alpha = np.random.rand()*0.02 + 1e-4
        i = np.arange(n)
        a0 = np.exp(-alpha * i) + np.exp(-alpha * (n-1 - i))
        a0 = _project_onto_simplex(a0, target)
        run_opt(a0, n)

    # Phase 3: LP refinement (optional, same as before)
    try:
        from scipy.optimize import linprog
    except Exception:
        linprog = None

    if linprog is not None and best_seq is not None and time.time() < deadline:
        a = best_seq.copy()
        while time.time() < deadline:
            conv = _conv_self(a)
            M = conv.max()
            n = len(a)
            rows = 2*n-1
            A = np.zeros((rows, n))
            for k in range(rows):
                low = max(0, k-(n-1))
                high = min(n-1, k)
                cols = np.arange(low, high+1)
                A[k, cols] = a[k - cols]
            # solve LP, maximize sum of g
            res = linprog(c=-np.ones(n), A_ub=A, b_ub=np.full(rows, M), bounds=(0,None), method="highs")
            if not res.success:
                break
            g = res.x
            s = g.sum()
            if s <= 0:

            g = g * (math.sqrt(2.0 * n) / s)
            a = 0.8 * a + 0.2 * g   # moderate blend

        n = 1000
        target = math.sqrt(2.0 * n)
        a = np.full(n, target / n)
        best_seq = a

    return best_seq.tolist()

"""
Propose a step‑function that minimises
    2 * n * max(conv(a, a)) / (sum(a)**2)

The evaluation simplifies to the maximum entry of the self‑convolution
when the sequence is normalised to Σ a_i = √(2 n).  The optimiser works in
three stages:

1.  Fast sweep of several parametric families (exponential, U‑shaped,
    sum‑of‑two‑exponentials, polynomial edge).  Every candidate is projected
    onto the simplex Σ a_i = √(2 n) and evaluated.

2.  Projected “smooth‑max” gradient descent (Adam‑style momentum) on a
    single start vector.  The objective is a soft‑max approximation of the
    L∞‑norm of the autocorrelation; the gradient is obtained analytically via
    FFT‑based convolutions.  The step‑size decays as 1/√t and the smoothing
    parameter λ is increased from 5 to 30.

3.  Optional LP refinement (exactly as in the reference code) if SciPy is
    available.

The whole routine respects the supplied time budget (default 1000 s) and
keeps the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Global evaluation helper (provided by the harness)
# ----------------------------------------------------------------------
def _evaluate(seq: list) -> float:
    """Safely call the external evaluate_sequence."""
    try:
        return evaluate_sequence(seq)  # type: ignore[name-defined]
    except Exception:
        return float("inf")


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution helpers – FFT based (real input)
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full convolution a * a using real FFT (returns length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors via FFT."""
    size = 1 << ((len(x) + len(y) - 1).bit_length())
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)
    return conv[: len(x) + len(y) - 1]


# ----------------------------------------------------------------------
# Gradient of the smooth‑max (log‑sum‑exp) of the convolution
# ----------------------------------------------------------------------
def _smooth_gradient(a: np.ndarray, conv: np.ndarray, lam: float) -> np.ndarray:
    """
    Compute ∇ ϕ_λ(a) where ϕ_λ = (1/λ) log Σ exp(λ·(a*a)).
    Returns a gradient of shape a.shape.
    """
    # soft‑max weights (stable)
    max_c = conv.max()
    exp_vals = np.exp(lam * (conv - max_c))
    w = exp_vals / exp_vals.sum()                # length 2·n‑1
    w_rev = w[::-1]                              # reverse for the gradient
    full = _conv_fft(a, w_rev)                   # length 3·n‑2
    n = a.shape[0]
    # grab the slice corresponding to indices (n‑1 … 2·n‑2) and reverse
    grad = 2.0 * full[n - 1 : 2 * n - 1][::-1]
    return grad


# ----------------------------------------------------------------------
# Projected Adam‑style optimiser on the smooth‑max objective
# ----------------------------------------------------------------------
def _run_opt(
    init_vec: np.ndarray,
    n: int,
    deadline: float,
    lam_start: float = 5.0,
    lam_end: float = 30.0,
    max_iter: int = 5000,
    eval_every: int = 10,
) -> tuple[np.ndarray, float]:
    """
    Run a smooth‑max gradient descent from ``init_vec`` (already on the simplex).
    Returns the best sequence seen and its score.
    """
    target = math.sqrt(2.0 * n)
    a = init_vec.copy()
    # Adam‑style moments
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    best_local = a.copy()
    best_score = _evaluate(a.tolist())

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # current autocorrelation
        conv = _conv_self(a)

        # annealing of λ
        lam = lam_start + (lam_end - lam_start) * min(1.0, it / max_iter)

        # gradient of the smooth max
        grad = _smooth_gradient(a, conv, lam)

        # Adam update (we minimise, so step opposite to grad)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = 0.3 / (np.sqrt(v_hat) + eps)       # 0.3 is a tunable base step
        a_new = a - step * m_hat
        a_new = _project_onto_simplex(a_new, target)

        a = a_new

        if it % eval_every == 0:
            score = _evaluate(a.tolist())
            if score < best_score:
                best_score = score
                best_local = a.copy()
                # early stopping if we beat the target bound
                if best_score < 1.503:
                    break

    return best_local, best_score

# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs) -> list:
    """
    Search for a non‑negative height sequence that minimises the evaluation
    function.  The routine respects the given time budget and returns the
    best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # safety margin

    best_seq = None
    best_score = float("inf")

    # Helper to evaluate a candidate and possibly update the global best
    # ------------------------------------------------------------------

        nonlocal best_seq, best_score
        score = _evaluate(arr.tolist())
        if score < best_score:
            best_score = score
            best_seq = arr.copy()
            print(

            )

    # ------------------------------------------------------------------

    init_global = globals().get("height_sequence_1")
    if init_global:

        n0 = a0.shape[0]
        target0 = math.sqrt(2.0 * n0)
        a0 = _project_onto_simplex(a0, target0)
        _try_candidate(a0)

    # ------------------------------------------------------------------

    lengths = list(range(300, 2001, 300)) + [2500]

    while time.time() < deadline:

        a = best_seq.copy()
        while time.time() < deadline:
            conv = _conv_self(a)
            M = conv.max()
            n = a.shape[0]
            rows = 2 * n - 1

            # Build constraint matrix for (a * g)_k ≤ M
            A = np.zeros((rows, n), dtype=float)
            for k in range(rows):
                low = max(0, k - (n - 1))
                high = min(n - 1, k)
                cols = np.arange(low, high + 1, dtype=int)
                A[k, cols] = a[k - cols]

        best_seq = np.full(n, target / n, dtype=float)

    return best_seq.tolist()
