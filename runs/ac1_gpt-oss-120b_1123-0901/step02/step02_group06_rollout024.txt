def _project_onto_simplex(v, target_sum=1.0):
    # Basic simplex projection onto sum = target_sum
    if np.all(v >= 0) and np.isclose(v.sum(), target_sum):
        return v
    # follow algorithm for sum=1 then scale:
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(1, len(v)+1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        # all non-positive
        return np.full_like(v, target_sum / len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a step‑function (non‑negative sequence) with a low
evaluate_sequence value.

Algorithm
---------
1. Choose a moderate length `n` (default 2000).  
   Normalise every candidate so that  Σ a_i = √(2 n).  With this
   normalisation the evaluation function equals   max(a*a) .
2. Build a small pool of diverse starting points:
   uniform, geometric progressions, arcsine‑shaped law, a few
   random Dirichlet draws, and – if present – the historic
   construction `height_sequence_1`.  Keep the best one.
3. Refine by *projected sub‑gradient descent* on the max‑convolution
   functional.
   *   Compute the self‑convolution `c = a*a`.
   *   Let `K` be all indices where `c` attains its maximum.
   *   The sub‑gradient of `max(c)` w.r.t. `a` is the sum over those
       shifts of the corresponding overlapping slices of `a`.
   *   Perform a heavy‑ball update and project back onto the simplex
       `{x≥0, Σx = √(2 n)}`.
   *   The step size decays slowly; after long stagnation a tiny
       Gaussian jitter is added.
4. When only a short time budget (≈ 20 s) remains, switch to the
   *linear‑programming refinement* described in the statement:
   for the current vector `f` (with max‑convolution `M`) solve

        maximise  Σ g_j
        subject to  (f * g)_k ≤ M   for all k
                    g_j ≥ 0

   This is a linear program because each convolution entry is a linear
   form in `g`.  The optimal `g` is rescaled to the required sum and
   blended with `f` (`f ← (1‑t)f + t g`, `t≈0.3`).  The procedure is
   repeated a few times – each iteration can only lower the maximum
   convolution value.
5. Throughout the search keep the best sequence found (according to
   `evaluate_sequence`).  Return it as a plain Python list.

The whole routine respects the 1000 s budget and needs only
numpy + scipy (both pre‑installed in the execution environment).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, target_sum: float = 1.0) -> np.ndarray:
    """
    Project the vector v onto the probability simplex of given total mass.
    Result satisfies:  w_i ≥ 0 and Σ w_i = target_sum.
    """
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), target_sum):
        return v

    # Scale to the unit simplex, project, then rescale.
    # This is the algorithm of Wang & Carreira‑Perpiñán (2013).
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(1, len(v) + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:               # all entries are non‑positive
        return np.full_like(v, target_sum / len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _initial_candidates(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """
    Produce a handful of promising initial sequences,
    already normalised to sum = sqrt(2 n).
    """
    target_sum = np.sqrt(2.0 * n)
    cand = []

    # 1. uniform
    cand.append(np.full(n, target_sum / n))

    # 2. geometric families (different decay rates)
    for factor in [0.7, 0.85, 0.95, 1.0, 1.2, 1.5, 2.0]:
        eps = factor / n
        r = max(0.0, 1.0 - eps)
        seq = np.power(r, np.arange(n, dtype=np.float64))
        seq = seq / seq.sum() * target_sum
        cand.append(seq)

    # 3. arcsine shape  a_i ∝ 1/√((i+½)(n‑i‑½))
    i = np.arange(n, dtype=np.float64)
    arcs = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    cand.append(arcs / arcs.sum() * target_sum)

    # 4. a few random Dirichlet draws
    for _ in range(8):
        rand = rng.random(n)
        cand.append(rand / rand.sum() * target_sum)

    # 5. historic construction if available
    old = globals().get("height_sequence_1")
    if isinstance(old, (list, tuple, np.ndarray)):
        old_arr = np.asarray(old, dtype=np.float64)
        if old_arr.size > 0:
            # truncate or pad with zeros to length n
            if old_arr.size >= n:
                seq = old_arr[:n]
            else:
                seq = np.r_[old_arr, np.zeros(n - old_arr.size)]
            seq = np.maximum(seq, 0.0)
            if seq.sum() > 0:
                seq = seq / seq.sum() * target_sum
                cand.append(seq)

    return cand


# ----------------------------------------------------------------------
def _subgradient_descent(
    start: np.ndarray,
    target_sum: float,
    time_limit: float,
    *,
    step0: float = 0.30,
    beta: float = 0.85,
    max_no_improve: int = 50000,
) -> tuple[np.ndarray, float]:
    """
    Perform projected (heavy‑ball) sub‑gradient descent on
    `max(convolution)`.  Returns the best sequence seen and its
    evaluation value.
    """
    rng = np.random.default_rng(0)
    n = len(start)
    seq = start.copy()
    step = step0
    momentum = np.zeros_like(seq)
    best_seq = seq.copy()
    best_score = float("inf")
    no_improve = 0
    it = 0

    while time.time() < time_limit:
        conv = np.convolve(seq, seq)        # length 2n‑1
        max_val = conv.max()

        # all positions attaining the current maximum
        top_idx = np.where(conv == max_val)[0]

        # sub‑gradient of the max‑convolution functional
        grad = np.zeros_like(seq)
        for k in top_idx:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            grad[idx] += seq[k - idx]
        grad /= max(1, len(top_idx))        # average if several maximisers

        # heavy‑ball update
        momentum = beta * momentum + (1.0 - beta) * grad
        new_seq = seq - step * momentum
        new_seq = _project_onto_simplex(new_seq, target_sum)

        # evaluate
        score = evaluate_sequence(new_seq.tolist())
        if score < best_score:
            best_score = score
            best_seq = new_seq.copy()
            no_improve = 0
            # optional logging
            print(f"[SGD {it:06d}] new best = {best_score:.6f}")
        else:
            no_improve += 1

        # step‑size schedule (slow decay)
        step *= 0.9997

        # occasional jitter to escape plateaus
        if no_improve > max_no_improve:
            jitter = rng.normal(scale=0.01, size=n)
            new_seq = new_seq + jitter
            new_seq = np.clip(new_seq, 0.0, None)
            new_seq = new_seq / new_seq.sum() * target_sum
            momentum[:] = 0.0
            no_improve = 0
            print(f"[SGD {it:06d}] stagnation – jitter applied")

        seq = new_seq
        it += 1

    return best_seq, best_score


# ----------------------------------------------------------------------
def _lp_refine(
    f: np.ndarray,
    target_sum: float,
    time_limit: float,
    *,
    mix_factor: float = 0.30,
    max_iters: int = 30,
) -> tuple[np.ndarray, float]:
    """
    Repeatedly improve `f` by solving the linear programme described
    in the problem statement.  Each iteration potentially lowers the
    maximal convolution entry.  Returns the best sequence encountered.
    """
    from scipy.optimize import linprog
    from scipy.linalg import toeplitz

    n = len(f)
    best_seq = f.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    it = 0

    while time.time() < time_limit and it < max_iters:
        conv = np.convolve(f, f)
        M = conv.max()
        # Build convolution matrix A such that (A @ g) == f * g
        col0 = np.r_[f, np.zeros(n - 1)]
        row0 = np.r_[f[0], np.zeros(n - 1)]
        A = toeplitz(col0, row0)          # shape (2n‑1, n), dense but modest

        # Linear programme: maximise Σ g_j   ⇔ minimise −Σ g_j
        c = -np.ones(n)
        b = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n

        res = linprog(
            c,
            A_ub=A,
            b_ub=b,
            bounds=bounds,
            method="highs",
            options={"presolve": True},
        )
        if not res.success:
            # If the LP fails (rare), break out.
            print(f"[LP {it}] linprog failed: {res.message}")
            break

        g0 = res.x
        sum_g0 = g0.sum()
        if sum_g0 == 0:
            break

        # Rescale to the required total mass
        g = g0 * (target_sum / sum_g0)
        g = np.clip(g, 0.0, None)                     # numerical safety
        g = _project_onto_simplex(g, target_sum)     # enforce exact sum

        # Mix the new vector with the current one
        f_new = (1.0 - mix_factor) * f + mix_factor * g

        # Evaluate the mixed candidate
        score = evaluate_sequence(f_new.tolist())
        if score < best_score:
            best_score = score
            best_seq = f_new.copy()
            print(f"[LP {it}] improvement: {best_score:.6f}")

        # Prepare for next iteration
        f = f_new
        it += 1

    return best_seq, best_score
