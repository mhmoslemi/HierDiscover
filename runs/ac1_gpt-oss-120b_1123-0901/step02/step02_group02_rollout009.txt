Propose a non‑negative step‑function (list of heights) with a low
value of

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm mixes three ideas that complement the previous
gradient‑descent approach:

1. **Diverse analytically‑inspired seeds** – power‑law, integral of
   1/√x and symmetric “U‑shaped’’ seeds with random offsets and
   exponents.  These give a much richer set of starting points than a
   single fixed law.

2. **Adam‑style projected optimisation** – for a fixed length we run a
   few seconds of Adam on the sub‑gradient of the maximum convolution
   entry (active set of indices whose value is within a tiny tolerance
   of the current maximum).  After each step the vector is projected
   onto the simplex `{x ≥ 0, Σ x = √(2 n)}` which keeps the objective
   scale‑invariant.

3. **Heavy‑weight restart loop** – many independent restarts with
   varying lengths (up to ten‑thousand entries) are performed until the
   time budget expires.  The best sequence found overall is returned.

The implementation is completely self‑contained, uses only NumPy and
standard library modules, and respects the 1000 s time budget.

# ----------------------------------------------------------------------
# Helper – Euclidean projection onto the simplex {x_i ≥ 0, Σ x_i = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex of radius s (Σ x = s, x ≥ 0)."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper – linear convolution via FFT (real → real)
# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using an FFT of power‑of‑2 size."""
    n = a.shape[0]
    m = 2 * n - 1
    size = 1 << ((m - 1).bit_length())  # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[:m]
    np.maximum(conv, 0.0, out=conv)  # clip tiny negatives caused by round‑off
    return conv


# ----------------------------------------------------------------------
# Helper – active indices (those close to the current maximum)
# ----------------------------------------------------------------------
def _active_indices(conv: np.ndarray, rel_tol: float = 5e-5) -> tuple[float, np.ndarray]:
    """Return (max_value, indices) where conv ≥ max – rel_tol·max."""
    max_val = conv.max()
    tol = max(1e-12, max_val * rel_tol)
    active = np.where(conv >= max_val - tol)[0]
    return max_val, active


# ----------------------------------------------------------------------
# Helper – sub‑gradient of the sum of the active convolution entries
# ----------------------------------------------------------------------
def _gradient(a: np.ndarray, active: np.ndarray) -> np.ndarray:
    """Gradient of Σ_{k∈active} (a * a)[k] with respect to a."""
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in active:
        low = max(0, k - (n - 1))
        high = min(n - 1, k)
        idx = np.arange(low, high + 1, dtype=int)
        partner = k - idx
        grad[idx] += 2.0 * a[partner]          # ∂/∂a_i a_i a_{k-i} = 2 a_{k-i}
    return grad


# ----------------------------------------------------------------------
# Adam optimiser for a fixed length
# ----------------------------------------------------------------------
def _adam_optimize_one_length(
    n: int,
    init: np.ndarray,
    run_deadline: float,
    target_sum: float,
    *,
    rel_tol: float = 5e-5,
    lr: float = 0.04,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimiser that minimises the maximal entry of a*a.
    Returns the best vector discovered in the allocated time and its
    objective value (the same quantity that `evaluate_sequence` returns).
    """
    a = _project_onto_simplex(init.astype(np.float64), target_sum)

    # keep the best vector seen in this run
    conv = _conv_fft(a)
    max_val, _ = _active_indices(conv, rel_tol)
    best_local = a.copy()
    best_score = 2.0 * n * max_val / (a.sum() ** 2)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    cur_lr = lr

    while time.time() < run_deadline:
        conv = _conv_fft(a)
        max_val, active = _active_indices(conv, rel_tol)

        grad = _gradient(a, active)

        # Adam moment updates
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = cur_lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = _project_onto_simplex(a_new, target_sum)

        # occasional stochastic shake‑up (helps escape flat plateaus)
        if np.random.rand() < 0.015:
            noise = np.random.randn(n) * (target_sum * 0.001 / n)
            a_new = _project_onto_simplex(np.maximum(a_new + noise, 0.0), target_sum)
            cur_lr *= 0.93  # become a bit more cautious after a shake‑up

        # evaluate new candidate
        conv_new = _conv_fft(a_new)
        max_val_new = conv_new.max()
        score_new = 2.0 * n * max_val_new / (a_new.sum() ** 2)

        if score_new < best_score:
            best_score = score_new
            best_local = a_new.copy()
            print(f"[{time.time():.2f}] n={n}  Adam local best = {score_new:.6f}")

        a = a_new
        cur_lr *= 0.9997          # exponential decay of the learning rate

    return best_local, best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative height sequence that minimises the
    evaluation function.  The routine runs for at most `budget_s`
    seconds, trying many random initialisations (power‑law,
    integral‑of‑1/√x, symmetric “U’’ shapes) and refines each with a
    short Adam optimisation.  The best sequence found overall is
    returned as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.3  # small safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Warm‑start from a previously discovered construction if available
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        seq0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if seq0.ndim == 1 and seq0.size > 0:
            best_seq = seq0.copy()
            best_score = 2.0 * seq0.size * np.max(np.convolve(seq0, seq0)) / (np.sum(seq0) ** 2)
            print(f"[{time.time():.2f}] Warm‑start from height_sequence_1, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Independent restarts – each gets a few seconds of optimisation
    # ------------------------------------------------------------------
    while time.time() < deadline:
        remaining = deadline - time.time()

        # Choose a problem size.
        if remaining < 25.0:
            n = np.random.randint(800, 1500)
        else:
            # Occasionally try very large sizes (up to 10 000)
            if np.random.rand() < 0.22:
                n = np.random.randint(5000, 10001)
            else:
                n = np.random.randint(800, 3501)

        # --------------------------------------------------------------
        # Pick a seed shape
        # --------------------------------------------------------------
        shape_type = np.random.choice(["power", "integral", "symmetric"])

        if shape_type == "power":
            # a_i ∝ (i+offset)^{-p}
            p = np.random.uniform(0.45, 0.55)
            offset = np.random.uniform(0.0, 5.0)
            idx = np.arange(n, dtype=np.float64) + offset
            init = (idx + 1.0) ** (-p)

        elif shape_type == "integral":
            # a_i = 2 [√(i+β+1) – √(i+β)]
            beta = np.random.uniform(0.0, 6.0)
            i = np.arange(n, dtype=np.float64)
            init = 2.0 * (np.sqrt(i + beta + 1.0) - np.sqrt(i + beta))

        else:  # symmetric “U’’ shape
            p = np.random.uniform(0.45, 0.55)
            offset = np.random.uniform(0.0, 5.0)
            i = np.arange(n, dtype=np.float64) + offset
            left = (i + 1.0) ** (-p)
            # mirrored index
            j = (n - 1 - i) + offset
            right = (j + 1.0) ** (-p)
            init = left + right

        # Tiny random jitter to break perfect symmetry
        init += np.random.rand(n) * 1e-6

        # --------------------------------------------------------------
        # Normalise onto the simplex Σ a_i = √(2 n)
        # --------------------------------------------------------------
        target_sum = np.sqrt(2.0 * n)
        init = _project_onto_simplex(init.astype(np.float64), target_sum)

        # --------------------------------------------------------------
        # Quick evaluation of the raw seed
        # --------------------------------------------------------------
        conv = _conv_fft(init)
        max_val = conv.max()
        score = 2.0 * n * max_val / (init.sum() ** 2)

        if score < best_score:
            best_score = score
            best_seq = init.copy()
            print(f"[{time.time():.2f}] New global seed = {score:.6f} (n={n})")

        # --------------------------------------------------------------
        # Allocate a time slice for this run (≈ 4‑6 s)
        # --------------------------------------------------------------
        slice_seconds = min(6.0, deadline - time.time())
        if slice_seconds < 0.8:
            continue
        run_deadline = time.time() + slice_seconds

        # --------------------------------------------------------------
        # Optimise the seed with Adam
        # --------------------------------------------------------------
        seq_local, score_local = _adam_optimize_one_length(
            n=n,
            init=init,
            run_deadline=run_deadline,
            target_sum=target_sum,
            rel_tol=5e-5,
            lr=0.04,
        )

        # --------------------------------------------------------------
        # Update the global best if we improved
        # --------------------------------------------------------------
        if score_local < best_score:
            best_score = score_local
            best_seq = seq_local.copy()
            print(f"[{time.time():.2f}] GLOBAL new best = {best_score:.6f} (length={n})")

        # Early exit if we have already crossed the target threshold
        if best_score < 1.503:
            break

    # ------------------------------------------------------------------
    # Fallback – should never be needed but guards against an empty result
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        target_sum = np.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=np.float64)

    # Return a plain Python list as required by the evaluation harness
    return best_seq.tolist()
