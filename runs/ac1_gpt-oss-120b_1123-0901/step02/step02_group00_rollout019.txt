"""
Propose a step‑function (non‑negative vector) that minimizes the
``evaluate_sequence`` objective.

Algorithm
---------
* The objective is     f(a) = 2·n·max(a*a) / (∑a)² .
  After each update we re‑scale to keep ∑a = √(2n), therefore we only
  have to minimise the maximal self‑convolution value M = max(a*a).

* We use a *continuation* approach: instead of the non‑smooth max we
  minimise a smooth approximation
        M_T(a) = Σ_k w_k·(a*a)_k ,   w_k ∝ exp((a*a)_k / T) .
  For large temperature T the weights are almost uniform (smooth landscape);
  as T → 0 the weights concentrate on the true maximiser.

* The gradient of M_T is
        ∂M_T/∂a_i = 2· Σ_k w_k·a_{k‑i}
      which can be written as a *correlation* of the weight vector w with a.
  Using FFT we obtain both the convolution a*a and the correlation w⋆a
  in O(N log N) time.

* Adam (with bias‑correction) performs the updates.
  After each step we project onto the non‑negative orthant and renormalise
  the sum to √(2n).  Occasionally a tiny random jitter is added to escape
  plateaus.

* Several sequence lengths are tried (the original length and larger
  interpolated versions).  For each length we run a few random restarts
  (including the interpolated base shape).  The best sequence over all
  runs is returned.

* The whole routine stops before the 1000 s wall‑clock limit; intermediate
  progress is printed.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: next power of two (for FFT padding)
def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << ((x - 1).bit_length())

# ----------------------------------------------------------------------
# Core optimisation for a fixed length -------------------------------------------------
def _smooth_adam_optimize(
    n: int,
    init_a: np.ndarray,
    best_seq: np.ndarray,
    best_score: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation that minimises the smooth approximation of
    max(a*a).  The temperature T is exponentially decreased.
    The function updates ``best_seq``/``best_score`` whenever a better
    vector is found.
    """
    # target sum = √(2n) – the feasible set after projection
    target_sum = math.sqrt(2.0 * n)

    a = init_a.copy()
    # enforce the normalisation once at the start
    a = a * (target_sum / a.sum())

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.07                # base learning rate (empirically good)
    iteration = 0

    # temperature schedule for the soft‑max smoothing
    T = 0.8                   # start fairly smooth
    T_min = 5e-5
    T_decay = 0.9996          # multiplied every iteration

    jitter_interval = 4000

    # pre‑compute constants that occur in the gradient formula
    coeff = 4.0 * n / (target_sum * target_sum)   # = 2 for the normalised case
    const_factor = 4.0 * n / (target_sum ** 3)    # multiplies the current M

    while time.time() < deadline:
        iteration += 1

        # ------------------------ 1) a*a (full) ---------------------------------
        N_conv = _next_pow2(2 * n - 1)                # enough for a*a
        Af = np.fft.rfft(a, n=N_conv)
        conv_full = np.fft.irfft(Af * Af, n=N_conv)[: 2 * n - 1]

        # ------------------------ 2) smooth weights ----------------------------
        # temperature update (geometric decay, never below T_min)
        if T > T_min:
            T = max(T_min, T * T_decay)

        if T <= 1e-12:                               # effectively the hard max
            # binary weight on the (possibly multiple) maximal indices
            M = conv_full.max()
            tol = 1e-12
            act = np.where(conv_full >= M - tol)[0]
            w = np.zeros_like(conv_full)
            w[act] = 1.0 / max(1, act.size)
        else:
            max_c = conv_full.max()
            # soft‑max (shifted by max for numerical stability)
            exp_vals = np.exp((conv_full - max_c) / T)
            w = exp_vals / exp_vals.sum()
            M = np.dot(w, conv_full)                 # smooth approximation of the max

        # ------------------------ 3) rev = Σ_k w_k·a_{k‑i} --------------------
        # This is a linear correlation (valid mode).  We compute it with FFT.
        N_corr = _next_pow2(3 * n - 2)               # length needed for correlation
        # Pad both vectors to the same length
        Af_full = np.fft.rfft(a, n=N_corr)
        Wf = np.fft.rfft(w, n=N_corr)
        # correlation = irfft( W · conj(A) )
        corr_full = np.fft.irfft(Wf * np.conj(Af_full), n=N_corr)
        # valid correlation indices correspond to lag i = 0 … n‑1 (shift of w)
        rev = corr_full[n - 1 : n - 1 + n]           # length n

        # ------------------------ 4) gradient ----------------------------------
        const = const_factor * M
        grad = coeff * rev - const

        # ------------------------ 5) Adam update --------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        lr = lr0 / (1.0 + 1e-4 * iteration)          # simple decay
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # ------------------------ 6) projection ---------------------------------
        a = np.maximum(a, 0.0)                       # non‑negative
        s = a.sum()
        if s > 0.0:
            a = a * (target_sum / s)                # renormalise to √(2n)
        else:
            # extremely unlikely – fall back to uniform vector
            a = np.full_like(a, target_sum / n)

        # ------------------------ 7) occasional jitter -------------------------
        if iteration % jitter_interval == 0:
            jitter = np.random.randn(n) * 0.001 * target_sum / n
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        # ------------------------ 8) current score (exact) --------------------
        cur_max = conv_full.max()                    # the true max a*a after the step
        cur_score = 2.0 * n * cur_max / (target_sum ** 2)   # identical to evaluate_sequence

        if cur_score < best_score:
            best_score = float(cur_score)
            best_seq = a.copy()
            print(
                f"[len {n:5d} iter {iteration:7d}] new best = {best_score:.6f}"
            )

        # --------------------------------------------------------------------
        # Optional early exit if no improvement for many iterations
        if iteration % 20000 == 0 and time.time() >= deadline:
            break

    return best_seq, best_score

        interp = np.maximum(interp, 0.0)
        inits.append(interp)

        # 2) interpolation of the original base (if length differs)
        if n != n0:
            old_x2 = np.linspace(0.0, 1.0, n0)
            new_x2 = np.linspace(0.0, 1.0, n)
            a = np.interp(new_x2, old_x2, base)
            a = np.maximum(a, 0.0)
            inits.append(a)

        # 3) a few random initialisations
        for _ in range(3):
            rand_vec = np.random.rand(n)
            rand_vec = rand_vec * (target_sum / rand_vec.sum())
            inits.append(rand_vec)

        # ------------------------------------------------------------------
        # Run the optimiser on each initialisation (subject to remaining time)
        for init_a in inits:
            if time.time() >= deadline:
                break
            print(f"[run] length={n}, start optimisation (init {len(inits)})")
            best_seq, best_score = _smooth_adam_optimize(
                n=n,
                init_a=init_a,
                best_seq=best_seq,
                best_score=best_score,
                deadline=deadline,
            )

    # ------------------------------------------------------------------
    # Safety fallback (should never be needed)
    if best_seq is None:
        n = 2100
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n)
        best_score = evaluate_sequence(best_seq.tolist())
        print("[fallback] using uniform sequence")

    print(f"[done] best score = {best_score:.6f}, length = {best_seq.shape[0]}")
    return best_seq.tolist()
