def adam_update(a, grad, m, v, t, lr=0.1, beta1=0.9, beta2=0.999, eps=1e-8):
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * (grad * grad)
    m_hat = m / (1 - beta1 ** t)
    v_hat = v / (1 - beta2 ** t)
    a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
    return a_new, m, v

"""
Propose a low‑objective step‑function sequence.

Algorithm (single‑run, time‑bounded)
------------------------------------
* Work on a few different lengths (2000, 3000, 4000 …).  Larger
  lengths give more “degrees of freedom’’ and can reduce the
  self‑convolution peak, at the cost of slightly larger FFTs.
* For each length initialise the vector a on the probability
  simplex (sum = 1, a_i ≥ 0) with several simple starters:
  – a geometric decay,
  – a few Dirichlet draws,
  – the best known global sequence (if the variable
    ``height_sequence_1`` exists) interpolated to the new length.
* Refine the best starter with a *continuation* scheme:
  1. Replace the hard max in the objective
        F(a)=2·n·max(a*a)/‖a‖₁²
     by the smooth log‑sum‑exp surrogate
        φ_λ(a)= (1/λ)·log Σ_k exp(λ·c_k) ,  c=a*a .
     When λ is small the surrogate is very smooth; as λ grows it
     approaches the true max.
  2. Compute the gradient of φ_λ efficiently using a single FFT‑based
     cross‑correlation (see `_softmax_gradient`).
  3. Apply Adam (adaptive‑momentum) updates on the gradient and
     project the iterate back onto the simplex.
  4. Increase λ slowly (λ ← λ·1.02) so that early steps move
     far from the start and later steps focus on the true max.
  5. If the objective does not improve for a couple of seconds,
     add a tiny random perturbation (helps escape shallow minima).
* Keep the globally best sequence (across all lengths and starts) and
  return it as a plain Python list.  The routine respects the
  1000 s budget by checking the wall‑clock time before each heavy
  operation.

The method is deliberately lightweight (pure NumPy, FFT‑based) and
fits comfortably into the time limit while typically driving the
objective well below the target 1.503.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Projection onto the probability simplex {x ≥ 0, sum(x) = 1}
def _project_onto_simplex(v):
    """Euclidean projection of a vector onto the simplex."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:               # all numbers non‑positive
        return np.full_like(v, 1.0 / len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear (non‑circular) self‑convolution using FFT.
def _full_convolution(a, size=None):
    """Return a*a (linear convolution) of length 2·len(a)‑1."""
    n = len(a)
    if size is None:
        size = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Exact objective as used by evaluate_sequence (sum is not forced to 1 here).
def _objective(a):
    """Compute 2·n·max(conv) / (sum(a)²). a must be non‑negative."""
    n = len(a)
    conv = _full_convolution(a)
    max_b = conv.max()
    sum_a = a.sum()
    # sum_a is always > 0 in our usage (simplex projection)
    return 2.0 * n * max_b / (sum_a * sum_a)


# ----------------------------------------------------------------------
# Gradient of the smooth‑max (log‑sum‑exp) surrogate.
def _softmax_gradient(a, conv, lam, size_corr=None):
    """
    Return ∇_a ( (1/lam)·log Σ_k exp(lam·conv_k) ).
    The result is 2·(w ★ a) where w is the soft‑max over λ·conv.
    """
    n = len(a)
    # soft‑max weights (numerically stable)
    m = conv.max()
    w = np.exp(lam * (conv - m))
    w /= w.sum()

    # cross‑correlation w ★ a (linear, not circular)
    rev_a = a[::-1]
    if size_corr is None:
        size_corr = 1 << int(np.ceil(np.log2(len(w) + n - 1)))
    fw = np.fft.rfft(w, size_corr)
    frev = np.fft.rfft(rev_a, size_corr)
    corr = np.fft.irfft(fw * frev, size_corr)

    # slice giving the gradient of length n
    grad = 2.0 * corr[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step‑function sequence with a small
    evaluation score.  The routine respects a wall‑clock budget,
    prints progress and finally returns the best list of heights found.
    """
    np.random.seed(seed)
    # safety margin: leave a little time for the interpreter after the loop
    deadline = time.time() + budget_s - 0.5

    # ------------------------------------------------------------------
    # Length schedule – we start small and move to larger vectors.
    # Larger n offers more flexibility but costs a bit more per iteration.
    length_schedule = [2000, 3000, 4000]

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    for n in length_schedule:
        if time.time() >= deadline:
            break
        print(f"\n=== Optimising length {n} ===")
        # FFT size for the *self* convolution (2·n‑1 → next power of two)
        conv_fft_size = 1 << int(np.ceil(np.log2(2 * n - 1)))
        # Size for the cross‑correlation (len(w)+n‑1 ≤ 3·n)
        corr_fft_size = 1 << int(np.ceil(np.log2(3 * n)))

        # --------------------------------------------------------------
        # Build a modest set of initial points (must lie on the simplex)
        init_points = []

        # geometric decays (slow decay → fairly flat shape)
        for r in [0.9995, 0.999, 0.9985, 0.998]:
            powers = np.power(r, np.arange(n, dtype=np.float64))
            seq = powers / powers.sum()
            init_points.append(seq)

        # a handful of random Dirichlet draws
        for _ in range(4):
            seq = np.random.dirichlet(np.ones(n))
            init_points.append(seq)

        # optionally reuse the globally available sequence if present
        old_seq = globals().get('height_sequence_1')
        if isinstance(old_seq, (list, tuple, np.ndarray)):
            old_arr = np.asarray(old_seq, dtype=np.float64)
            if old_arr.ndim == 1 and old_arr.size > 0:
                # linear interpolation to new length
                x_old = np.linspace(0.0, 1.0, old_arr.size)
                x_new = np.linspace(0.0, 1.0, n)
                interp = np.interp(x_new, x_old, old_arr)
                interp = np.maximum(interp, 0.0)
                if interp.sum() == 0.0:
                    interp = np.ones(n) / n
                else:
                    interp = interp / interp.sum()
                init_points.append(interp)

        # --------------------------------------------------------------
        # Run the Adam‑based continuation optimisation for each starter
        for start_id, a0 in enumerate(init_points):
            if time.time() >= deadline:
                break
            seq = a0.copy()
            cur_val = _objective(seq)
            if cur_val < best_val:
                best_val, best_seq = cur_val, seq.copy()
                print(f"[Init {start_id}] new global best = {cur_val:.6f}")

            # Adam state
            m = np.zeros_like(seq)
            v = np.zeros_like(seq)
            t = 0

            # continuation parameters
            lam = 5.0                 # starting λ (smoothness)
            lam_factor = 1.02         # gradual increase
            lam_max = 200.0

            # Adam hyper‑parameters
            lr = 0.10                 # base learning rate (after scaling)
            beta1, beta2 = 0.9, 0.999
            eps = 1e-8

            last_improve = time.time()
            # Main optimisation loop (stops when overall deadline hits)
            while time.time() < deadline:
                # 1) self‑convolution
                conv = _full_convolution(seq, size=conv_fft_size)

                # 2) smooth‑max gradient
                grad = _softmax_gradient(seq, conv, lam, size_corr=corr_fft_size)

                # 3) Adam update
                t += 1
                m = beta1 * m + (1.0 - beta1) * grad
                v = beta2 * v + (1.0 - beta2) * (grad * grad)
                m_hat = m / (1.0 - beta1 ** t)
                v_hat = v / (1.0 - beta2 ** t)
                step = lr * m_hat / (np.sqrt(v_hat) + eps)

                seq_new = seq - step
                seq_new = _project_onto_simplex(seq_new)

                # 4) exact evaluation (no surrogate)
                val = _objective(seq_new)
                if val < best_val:
                    best_val, best_seq = val, seq_new.copy()
                    last_improve = time.time()
                    print(f"[Iter n={n} s={start_id} it={t}] new best = {val:.6f}")

                # 5) stagnation handling: small random shake
                if time.time() - last_improve > 2.0:
                    noise = (np.random.rand(n) - 0.5) * 1e-3
                    seq_new = _project_onto_simplex(seq_new + noise)
                    last_improve = time.time()

                # 6) advance state
                seq = seq_new
                lam = min(lam * lam_factor, lam_max)

    # ------------------------------------------------------------------
    # Fallback in the extremely unlikely case that nothing was found
    if best_seq is None:
        n = 2000
        best_seq = np.full(n, 1.0 / n)
        best_val = _objective(best_seq)

    print(f"\nFinal best objective = {best_val:.6f} (length {len(best_seq)})")
    # The evaluator expects a plain Python list
    return best_seq.tolist()
