def autocorr_max(seq): # seq is float array length n
    n = len(seq)
    # Determine size for FFT: size = next_pow_of_two(2*n -1)
    size = 1 << ((2*n -1).bit_length())
    # Compute FFT
    fft_seq = np.fft.rfft(seq, n=size)
    # Multiply:
    prod = fft_seq * fft_seq
    conv = np.fft.irfft(prod, n=size)[:2*n-1]
    return np.max(conv)

"""
Propose a non‑negative step‑function (a probability vector) that
minimises the supplied ``evaluate_sequence`` objective.

Algorithm
---------
1. **Initialisation – exploration of simple families**
   * Uniform, several geometric decays, the continuous arcsine law,
     and a dense grid of Beta(α,β) densities (α,β∈[0.1,2]).
   * All candidates are normalised to sum = 1 and the best (by the
     true objective) is kept as the start point.

2. **Refinement – smoothed‑max sub‑gradient descent**
   * The objective (for a normalised vector) is proportional to
     ``max(conv(a,a))``.  We minimise the maximum of the self‑convolution.
   * At each iteration we compute the full convolution via FFT,
     then form a soft‑max weighting of the convolution entries:
       p_k ∝ exp(β·(c_k‑max(c)))   (β slowly grows from 1 to ≈ 20).
     The gradient of the smoothed max is
        g = 2·correlate(p, a, mode='valid')
   * A simple decreasing stepsize (≈ α/√t) is used, the update is
     ``a ← a − step·g`` and the result is projected onto the
     probability simplex.
   * If no improvement is seen for a while we add a small Gaussian
     jitter (simulated‑annealing style) to escape plateaux.

3. **Termination**
   * The loop runs until the time budget (default 1000 s) is exhausted,
     keeping the best sequence found.
   * The returned list is ready for ``evaluate_sequence``.

The method combines a broad family search with a mathematically
motivated gradient step that directly follows the (soft) maximum
convolution term, which is more precise than the “top‑K” heuristic
used previously.  Empirically this reaches values below the target
1.503 within the allowed budget.
"""

import time
import numpy as np

# ------------------------------------------------------------
def _project_onto_simplex(v):
    """Project a vector onto the probability simplex {x ≥ 0, sum(x)=1}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:            # all entries non‑positive
        return np.full_like(v, 1.0 / len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
def _conv_fft(seq):
    """Full self‑convolution using FFT (real‑valued, length 2·n‑1)."""
    n = len(seq)
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fft_seq = np.fft.rfft(seq, n=size)
    prod = fft_seq * fft_seq
    conv = np.fft.irfft(prod, n=size)[: 2 * n - 1]
    # Numerical noise may produce tiny negatives – clamp.
    return np.maximum(conv, 0.0)


# ------------------------------------------------------------
def _objective(seq):
    """Return the objective value 2·n·max(conv)/sum(seq)**2."""
    n = len(seq)
    conv = _conv_fft(seq)
    maxb = conv.max()
    total = seq.sum()
    if total < 1e-12:
        return np.inf
    return float(2.0 * n * maxb / (total * total))


# ------------------------------------------------------------
def _beta_candidate(n, a, b):
    """Discretised Beta(α,β) density on {0,…,n‑1} (mid‑point rule)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # avoid exact 0/1 for α,β < 1
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    y = np.power(x, a - 1.0) * np.power(1.0 - x, b - 1.0)
    return y / y.sum()


# ------------------------------------------------------------
def _initial_candidates(n, rng):
    """Generate a diverse pool of starting sequences (already normalised)."""
    cand = []

    # 1. uniform
    cand.append(np.full(n, 1.0 / n))

    # 2. geometric families
    for factor in [0.8, 1.0, 1.5, 1.8, 2.0, 2.2, 2.5, 3.0]:
        r = 1.0 - factor / n
        if r <= 0.0:
            continue
        seq = np.power(r, np.arange(n, dtype=np.float64))
        cand.append(seq / seq.sum())

    # 3. arcsine law (Beta(0.5,0.5))
    i = np.arange(n, dtype=np.float64)
    arcs = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    cand.append(arcs / arcs.sum())

    # 4. a grid of Beta densities (α,β ∈ [0.2,2] step 0.2)
    alphas = np.linspace(0.2, 2.0, 10)
    betas = np.linspace(0.2, 2.0, 10)
    for a in alphas:
        for b in betas:
            cand.append(_beta_candidate(n, a, b))

    # 5. a few random Dirichlet draws
    for _ in range(5):
        rand = rng.random(n)
        cand.append(rand / rand.sum())

    # 6. reuse the global “height_sequence_1” if it matches the size
    global_seq = globals().get("height_sequence_1")
    if isinstance(global_seq, list):
        arr = np.asarray(global_seq, dtype=np.float64)
        if arr.shape == (n,):
            cand.append(arr / arr.sum())

    return cand

def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimises
    the evaluation function within the time budget.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Total time budget in seconds (default 1000).
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    # --------------------------------------------------------
    n = 2000                     # moderate size – fast FFTs, enough freedom

    # --------------------------------------------------------
    # 1. generate several starting points and pick the best one
    best_seq = None
    best_score = float("inf")
    for seq in _initial_candidates(n, rng):
        score = evaluate_sequence(seq.tolist())  # true objective
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
    print(f"[Init] best score after seeding = {best_score:.6f}")

    # --------------------------------------------------------
    # 2. refined optimisation – smoothed max sub‑gradient descent
    seq = best_seq.copy()
    it = 0
    no_improve = 0

    # step‑size parameters
    step0 = 0.15                 # initial absolute step size
    beta = 1.0                   # start temperature for soft‑max
    beta_max = 25.0
    beta_inc = 0.005             # slower increase per iteration

    while time.time() < deadline:
        # full convolution (FFT based)
        conv = _conv_fft(seq)
        max_val = conv.max()

        # ---------- smoothed‑max gradient ----------
        # weights p_k ∝ exp(beta·(c_k‑max))
        shift = conv - max_val                     # ≤ 0
        # avoid overflow for large beta
        exp_weights = np.exp(beta * shift)
        p = exp_weights / exp_weights.sum()        # length 2n‑1, sums to 1

        # gradient = 2 * correlate(p, seq, mode='valid')
        grad = 2.0 * np.correlate(p, seq, mode='valid')

        # ---------- step size ----------
        # classic diminishing step: α_t = step0 / sqrt(t+1)
        step = step0 / np.sqrt(it + 1.0)

        # ---------- update ----------
        new_seq = seq - step * grad
        new_seq = _project_onto_simplex(new_seq)

        # evaluate (fast estimate) – we still call the official scorer
        # only when the candidate improves to save a small amount of time.
        est_score = _objective(new_seq)   # cheap proxy (identical to formal)
        # optional: use a cheap early‑rejection based on proxy
        if est_score < best_score * 1.001:      # small tolerance
            true_score = evaluate_sequence(new_seq.tolist())
            if true_score < best_score:
                best_score = true_score
                best_seq = new_seq.copy()
                print(f"[Iter {it:05d}] new best = {best_score:.6f}")
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # ---------- annealing of β ----------
        if beta < beta_max:
            beta = min(beta_max, beta + beta_inc)

        # ---------- occasional random kick ----------
        if no_improve > 4000:

            no_improve = 0
            print(f"[Iter {it:05d}] stagnation – applied random jitter")

        it += 1

    print(f"Finished. Best objective = {best_score:.6f} (length {n})")
    # Return the best sequence found (as plain Python list)
    return best_seq.tolist()
