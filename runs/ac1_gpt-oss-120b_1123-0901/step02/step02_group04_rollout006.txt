def run_subgradient_multi(init_seq, best_seq, best_score, deadline):
    a = init_seq.copy()
    n = a.shape[0]
    target = math.sqrt(2.0 * n)
    m = np.zeros_like(a)
    beta = 0.9
    base_step = 0.2
    it = 0
    while time.time() < deadline:
        conv = _conv_self(a)
        M = conv.max()
        # active set
        epsilon = 1e-4
        active = np.where(conv >= M - epsilon)[0]
        grad = np.zeros_like(a)
        for k in active:
            low = max(0, k - (n - 1))
            high = min(n - 1, k)
            idx = np.arange(low, high + 1, dtype=int)
            partner = k - idx
            grad[idx] += 2.0 * a[partner]
        direction = -grad
        m = beta * m + (1.0 - beta) * direction
        step = base_step / math.sqrt(it + 1)
        max_abs = np.max(np.abs(m))
        if max_abs == 0:
            max_abs = 1.0
        eta = step / max_abs
        a_new = a + eta * m
        # projection onto simplex
        a_new = _project_onto_simplex(a_new, target)
        # compute new score
        new_score = np.max(_conv_self(a_new))
        # update best
        if new_score < best_score:
            best_score = new_score
            best_seq = a_new.copy()
            print(f"[{time.time():.2f}] New best = {best_score:.6f} (len={n})")
        # Accept if improvement else keep a unchanged? Should we accept? In subgradient we replace a always.
        a = a_new
        it += 1
        # optional early stop if best_score below threshold
        if best_score < 1.503:
            break
    return best_seq, best_score
```

def random_hill_climb(seq, best_seq, best_score, deadline):
    cur = seq.copy()
    cur_score = best_score
    n = cur.shape[0]
    start_time = time.time()
    total_time = deadline - start_time
    init_T = 0.02
    final_T = 1e-5
    iterations = 0
    while time.time() < deadline:

        T = init_T * (final_T / init_T) ** frac
        # propose move
        i = np.random.randint(n)
        if cur[i] <= 0:

        j = np.random.randint(n-1)

        if iterations % 1000 == 0:

"""
Propose a (non‑negative) height sequence that minimizes

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

Because the evaluator normalises the sum to √(2 n), the score is simply
the maximum entry of the autocorrelation (the full self‑convolution).
The algorithm consists of four stages:

1. **Parametric sweep** – cheap families (exponential, linear, quadratic,
   triangular, half‑sine, raised‑cosine, beta‑shaped) are generated for a
   variety of lengths and parameters.  Every candidate is projected onto
   the simplex Σa_i = √(2 n) and its score is evaluated.  The best
   candidate so far is kept.

2. **Multi‑constraint sub‑gradient descent** – starting from the current
   best vector we minimise the infinity‑norm of the autocorrelation.
   Unlike the previous implementation we use *all* near‑maximal entries
   (|conv‑max| ≤ ε) to form the sub‑gradient, apply momentum and a
   diminishing step size, and re‑project onto the simplex after each step.

3. **Random hill‑climbing with simulated annealing** – a cheap move
   transfers a random amount of mass from one index to another while
   preserving the total sum.  Moves that increase the objective are
   occasionally accepted according to a temperature schedule, allowing
   the search to escape shallow local minima.

4. **Optional LP refinement** – if SciPy is present we run a few high‑
   level linear‑programming refinements exactly as described in the
   state‑of‑the‑art algorithm.

The routine respects the supplied time budget (default 1000 s) and
returns the best sequence found as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the simplex {x ≥ 0, Σx = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper: full convolution a * a (real FFT, fast for n up to a few thousand)
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Return the full self‑convolution of a (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Compute the score (max entry of the autocorrelation)
# ----------------------------------------------------------------------
def _score_of_seq(a: np.ndarray) -> float:
    """Return max(conv(a, a)) – the value returned by evaluate_sequence."""
    return float(np.max(_conv_self(a)))


# ----------------------------------------------------------------------
# Multi‑constraint sub‑gradient descent (uses all nearly‑maximal entries)
# ----------------------------------------------------------------------
def _subgradient_descent_multi(init_seq: np.ndarray,
                               best_seq: np.ndarray,
                               best_score: float,
                               deadline: float) -> tuple:
    """
    Perform sub‑gradient descent on `init_seq`.  The global best is
    updated whenever a better point is found.  Returns (best_seq, best_score).
    """
    a = init_seq.astype(float).copy()
    n = a.shape[0]
    target = math.sqrt(2.0 * n)

    m = np.zeros_like(a)                # momentum
    beta = 0.9                           # momentum decay
    base_step = 0.2

    it = 0
    while time.time() < deadline:
        conv = _conv_self(a)
        M = conv.max()
        # active set: all entries within a tiny absolute tolerance of the maximum
        eps = max(1e-8, 1e-4 * M)
        active = np.where(conv >= M - eps)[0]

        # compute sub‑gradient from all active constraints
        grad = np.zeros_like(a)
        for k in active:
            low = max(0, k - (n - 1))
            high = min(n - 1, k)
            idx = np.arange(low, high + 1, dtype=int)
            partner = k - idx
            grad[idx] += 2.0 * a[partner]

        direction = -grad
        m = beta * m + (1.0 - beta) * direction

        step = base_step / math.sqrt(it + 1)
        max_abs = np.max(np.abs(m))
        if max_abs == 0.0:
            max_abs = 1.0
        eta = step / max_abs

        a_new = a + eta * m
        a_new = _project_onto_simplex(a_new, target)

        # score of the new point
        new_score = _score_of_seq(a_new)

        # update the global best if we improved
        if new_score < best_score:
            best_score = new_score
            best_seq = a_new.copy()
            print(f"[{time.time():.2f}] New best = {best_score:.6f} (len={n})")

        a = a_new
        it += 1

        if best_score < 1.503:          # target already reached
            break
    return best_seq, best_score


# ----------------------------------------------------------------------
# Simple random hill‑climbing with simulated annealing
# ----------------------------------------------------------------------
def _random_hill_climb(start_seq: np.ndarray,
                       best_seq: np.ndarray,
                       best_score: float,
                       deadline: float) -> tuple:
    """
    Perform many small mass‑transfer moves, accepting worsening moves
    according to a slowly cooling temperature schedule.
    Returns the updated (best_seq, best_score).
    """
    cur = start_seq.astype(float).copy()
    cur_score = _score_of_seq(cur)
    n = cur.shape[0]
    start_time = time.time()
    total_time = deadline - start_time
    if total_time <= 0:
        return best_seq, best_score

    # temperature schedule
    T0 = 0.02
    Tf = 1e-5

    it = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / total_time
        T = T0 * (Tf / T0) ** frac

        # propose a move: transfer a random fraction of mass from i to j
        i = np.random.randint(n)
        if cur[i] <= 0.0:
            continue
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = cur[i] * np.random.rand() * 0.5   # up to 50 % of source mass
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta

        # ensure non‑negativity (numerical safety)
        cand[i] = max(cand[i], 0.0)
        cand[j] = max(cand[j], 0.0)

        cand_score = _score_of_seq(cand)

        # Metropolis acceptance
        if cand_score < cur_score or np.random.rand() < math.exp(-(cand_score - cur_score) / (T + 1e-12)):
            cur = cand
            cur_score = cand_score

        # global best update
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            print(f"[{time.time():.2f}] New best = {best_score:.6f} (len={n})")
            if best_score < 1.503:
                break

        it += 1
        # occasional re‑projection to correct tiny drift
        if it % 1000 == 0:
            target = math.sqrt(2.0 * n)
            cur = _project_onto_simplex(cur, target)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs) -> list:
    """
    Search for a non‑negative step‑function height sequence that minimises
    the provided evaluation function.  The routine respects the time budget
    and returns the best sequence found (as a plain Python list).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5      # a small safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Helper to evaluate a candidate and possibly update the global best
    # ------------------------------------------------------------------
    def _maybe_update(candidate: np.ndarray) -> float:
        nonlocal best_seq, best_score
        sc = _score_of_seq(candidate)
        if sc < best_score:
            best_score = sc
            best_seq = candidate.copy()
            print(f"[{time.time():.2f}] New best = {best_score:.6f} (len={candidate.shape[0]})")
        return sc

    # ------------------------------------------------------------------
    # 0.  Seed from a previously discovered construction if available
    # ------------------------------------------------------------------
    seed_seq = globals().get('height_sequence_1')
    if seed_seq is not None:
        a0 = np.array(seed_seq, dtype=float)
        n0 = a0.shape[0]
        target0 = math.sqrt(2.0 * n0)
        a0 = _project_onto_simplex(a0, target0)
        _maybe_update(a0)

    # ------------------------------------------------------------------
    # 1.  Parametric families (cheap sweep)
    # ------------------------------------------------------------------
    lengths = list(range(200, 1601, 200)) + [1326, 1500, 1800]
    for n in lengths:
        target = math.sqrt(2.0 * n)

        # exponential decay (forward)
        for alpha in np.logspace(-4, -0.3, num=12):
            a = np.exp(-alpha * np.arange(n, dtype=float))
            a = a / a.sum() * target
            _maybe_update(a)

        # exponential decay (reversed)
        for alpha in np.logspace(-4, -0.3, num=12):
            a = np.exp(-alpha * np.arange(n - 1, -1, -1, dtype=float))
            a = a / a.sum() * target
            _maybe_update(a)

        # linear descending
        for beta in np.linspace(0.0, 2.0 / max(1, n - 1), num=10):
            i = np.arange(n, dtype=float)
            a = np.maximum(0.0, 1.0 - beta * i)
            if a.sum() == 0.0:
                continue
            a = a / a.sum() * target
            _maybe_update(a)

        # quadratic descending
        for beta in np.linspace(0.0, 1.0 / max(1, n ** 2), num=10):
            i = np.arange(n, dtype=float)
            a = np.maximum(0.0, 1.0 - beta * i * i)
            if a.sum() == 0.0:
                continue
            a = a / a.sum() * target
            _maybe_update(a)

        # triangular (symmetric)
        i = np.arange(n, dtype=float)
        a = np.minimum(i + 1.0, n - i)
        a = a / a.sum() * target
        _maybe_update(a)

        # half‑sine shape
        for p in [0.5, 1.0, 1.5, 2.0, 3.0, 4.0]:
            a = np.sin(np.pi * i / (n - 1)) ** p
            a = a / a.sum() * target
            _maybe_update(a)

        # raised‑cosine shape (1‑cos)
        for p in [0.5, 1.0, 1.5, 2.0]:
            a = (1.0 - np.cos(np.pi * i / (n - 1))) ** p
            a = a / a.sum() * target
            _maybe_update(a)

        # beta‑type shape
        for alpha in [0.5, 1.0, 2.0, 3.0]:
            for beta in [0.5, 1.0, 2.0, 3.0]:
                t = i / (n - 1)
                a = (t ** alpha) * ((1 - t) ** beta)
                if a.sum() == 0.0:
                    continue
                a = a / a.sum() * target
                _maybe_update(a)

        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # 2.  Multi‑constraint sub‑gradient descent (starting from best)
    # ------------------------------------------------------------------
    if best_seq is not None:
        best_seq, best_score = _subgradient_descent_multi(best_seq, best_seq, best_score, deadline)

    # ------------------------------------------------------------------
    # 3.  Random restarts + sub‑gradient refinement
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # pick a random length (moderate to keep each run fast)
        n = np.random.randint(300, 2001)

        # initialise with a random positive vector, then project
        a0 = np.random.rand(n)
        a0 = _project_onto_simplex(a0, math.sqrt(2.0 * n))

        best_seq, best_score = _subgradient_descent_multi(a0, best_seq, best_score, deadline)

        if best_score < 1.503:
            break

    # ------------------------------------------------------------------
    # 4.  Fine‑grained random hill‑climbing (simulated annealing)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        best_seq, best_score = _random_hill_climb(best_seq, best_seq, best_score, deadline)

    # ------------------------------------------------------------------
    # 5.  Optional linear‑programming refinement (if SciPy is available)
    # ------------------------------------------------------------------
    try:
        from scipy.optimize import linprog
    except Exception:                         # pragma: no cover
        linprog = None

    if linprog is not None and best_seq is not None and time.time() < deadline:
        a = best_seq.copy()
        n = a.shape[0]
        target = math.sqrt(2.0 * n)

        while time.time() < deadline:
            conv = _conv_self(a)
            M = conv.max()
            rows = 2 * n - 1

            # Build constraint matrix A for (a * g)_k ≤ M
            A = np.zeros((rows, n), dtype=float)
            for k in range(rows):
                low = max(0, k - (n - 1))
                high = min(n - 1, k)
                cols = np.arange(low, high + 1, dtype=int)
                A[k, cols] = a[k - cols]

            res = linprog(
                c=-np.ones(n),                # maximise Σ g_j
                A_ub=A,
                b_ub=np.full(rows, M),
                bounds=(0, None),
                method="highs",
                options={"presolve": True},
            )
            if not res.success:
                break
            g = res.x
            s = g.sum()
            if s <= 0:
                break
            g = g * (target / s)

            # convex combination (moderate mixing)
            t = 0.2
            a = (1 - t) * a + t * g

            # update best if improved
            cur_score = _score_of_seq(a)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time():.2f}] New best = {best_score:.6f} (len={n})")
                if best_score < 1.503:
                    break

            if time.time() + 0.5 > deadline:
                break

    # ------------------------------------------------------------------
    # 6.  Fallback – uniform sequence if nothing else succeeded
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=float)
        best_score = _score_of_seq(best_seq)

    # Return as plain Python list (evaluate_sequence expects a list)
    return best_seq.tolist()
