"""
Multi‑restart Adam optimiser for the step‑function attack.

We explore several sequence lengths (original length and a few larger
multiples) and, for each length, many random initialisations (uniform,
linear, exponential, Dirichlet‑sampled, sparse).  Each candidate is
optimised with a short Adam run that uses the exact sub‑gradient of the
max‑convolution.  The best local solution found is then refined with a
longer Adam run.  All updates are projected onto the feasible set
(non‑negative entries, fixed ℓ₁‑norm √(2 n)).  The routine stops well
before the 1000 s wall‑clock limit and returns the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _conv_max_fft(a: np.ndarray):
    """Return (max_conv, index_of_max, full_convolution) using an FFT."""
    n = a.shape[0]
    # size for linear convolution (next power of two)
    N = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=N)
    B = A * A
    conv = np.fft.irfft(B, n=N)
    conv = conv[: 2 * n - 1]
    k_max = int(conv.argmax())
    return float(conv.max()), k_max, conv


# ----------------------------------------------------------------------
def _subgradient(a: np.ndarray, active_idxs: np.ndarray):
    """
    Sub‑gradient of the max‑convolution.
    It is the average reversed vector a_{k‑i} over all active shift indices.
    """
    n = a.shape[0]
    idx = np.arange(n)
    rev = np.zeros_like(a)
    for k in active_idxs:
        j = k - idx
        mask = (j >= 0) & (j < n)
        rev[mask] += a[j[mask]]
    if active_idxs.size > 0:
        rev /= active_idxs.size
    return rev


# ----------------------------------------------------------------------
def _adam_optimize(init_a: np.ndarray,
                   target_sum: float,
                   max_iters: int,
                   deadline: float):
    """
    Adam optimiser for a fixed length.
    Returns (best_sequence, best_score) where best_score = max convolution
    (identical to evaluate_sequence after normalisation).
    """
    n = init_a.shape[0]
    # enforce exact ℓ₁‑norm
    a = init_a.astype(float)
    s = a.sum()
    if s > 0:
        a = a * (target_sum / s)
    else:
        a = np.full_like(a, target_sum / n)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.05
    iteration = 0
    jitter_interval = 2000

    best_seq = a.copy()
    # after scaling, the objective equals the maximal convolution value
    best_score = float('inf')

    while iteration < max_iters and time.time() < deadline:
        # 1) max‑convolution and active set
        M, _, conv_full = _conv_max_fft(a)
        tol = 1e-12
        active = np.where(conv_full >= M - tol)[0]
        rev = _subgradient(a, active)

        # 2) sub‑gradient of the objective
        coeff = 4.0 * n / (target_sum * target_sum)
        const = 4.0 * n * M / (target_sum ** 3)
        grad = coeff * rev - const

        # 3) Adam update
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        lr = lr0 / (1.0 + 1e-4 * iteration)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # 4) projection onto non‑negative orthant and re‑normalise
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a = a * (target_sum / s)
        else:
            a = np.full_like(a, target_sum / n)

        # 5) occasional jitter to escape plateaus
        if iteration % jitter_interval == 0:
            jitter = np.random.randn(n) * 0.001 * target_sum / n
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        # 6) evaluate current objective (equals M after normalisation)
        cur_score = M
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(
                f"[len {n:5d} iter {iteration:6d}] new local best = {best_score:.6f}"
            )

    return best_seq, best_score


# ----------------------------------------------------------------------
def _generate_initial(n: int, typ: str, rng: np.random.Generator) -> np.ndarray:
    """Create a random non‑negative start vector of length n."""
    if typ == "uniform":
        return np.ones(n)
    if typ == "linear":
        return np.linspace(1.0, 0.0, n)
    if typ == "exp":
        return np.exp(-np.linspace(0.0, 5.0, n))
    if typ == "dirichlet":
        # sample from a Dirichlet distribution (uniform on simplex)
        return rng.dirichlet(np.ones(n))
    if typ == "sparse":
        # a few random non‑zeros
        m = max(2, int(n * 0.02))
        a = np.zeros(n)
        idx = rng.choice(n, size=m, replace=False)
        a[idx] = rng.random(m) + 0.01
        return a
    # fallback: pure random
    return rng.random(n)
