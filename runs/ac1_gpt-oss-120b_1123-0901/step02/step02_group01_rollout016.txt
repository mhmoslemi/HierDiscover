def _next_pow2(x):
    return 1 << (x - 1).bit_length()

def _softmax(t):
    t_shift = t - np.max(t)
    e = np.exp(t_shift)
    return e / np.sum(e)

def _full_conv(a):
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return conv

def _smooth_obj_grad(t, lam, n):
    a = _softmax(t)
    # convolution
    size_conv = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, size_conv)
    conv = np.fft.irfft(fa * fa, size_conv)[: 2 * n - 1]
    m = conv.max()

    diff = conv - m

"""
A continuation‑L‑BFGS search on the probability simplex.

The step‑function coefficients a_i are constrained to a≥0 and
∑a_i=1.  We parametrize a by logits t (soft‑max) so that the
constraints are automatically satisfied.  For a fixed smoothing
parameter λ a smooth surrogate of the max‑convolution is

    φ_λ(a) = (1/λ)·log∑_k e^{λ·(a*a)[k]},

whose gradient w.r.t. a is 2·(w ★ a) where w is the soft‑max
vector of the self‑convolution.  The exact objective is

    F(a) = 2·n·max_k (a*a)[k]   (because ∑a_i=1).

We minimise the smooth surrogate 2·n·φ_λ(a) with L‑BFGS, gradually
increasing λ (continuation).  Several random starts (Dirichlet,
geometric sequences and the previous best solution, if available)
are explored within the 1000 s budget.  Small random “shakes”
are added after each continuation step to help escape shallow minima.
The best sequence found is returned as a plain Python list.
"""

import time
import numpy as np
from scipy.optimize import minimize


# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Small utility: smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _softmax(t: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (maps ℝⁿ → simplex)."""
    t_shift = t - np.max(t)
    e = np.exp(t_shift)
    return e / np.sum(e)


def _full_conv(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of size power‑of‑2."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return conv


def _smooth_obj_grad(t: np.ndarray, lam: float, n: int):
    """
    Smooth surrogate of the evaluation function together with its gradient.

    Parameters
    ----------
    t : ndarray
        Logits (real numbers) – after soft‑max they give the coefficients.
    lam : float
        Smoothing parameter λ>0.
    n : int
        Length of the coefficient vector.

    Returns
    -------
    obj : float
        2·n·φ_λ(a) where a = softmax(t).
    grad_t : ndarray
        Gradient of the above objective w.r.t. the logits t.
    """
    # a lies on the simplex
    a = _softmax(t)

    # ----- self‑convolution a*a (linear) -----
    size_conv = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, size_conv)
    conv = np.fft.irfft(fa * fa, size_conv)[: 2 * n - 1]

    # ----- soft‑max weights over the convolution entries -----
    m = conv.max()
    diff = conv - m                     # ≤0, improves numeric stability
    ex = np.exp(lam * diff)
    sum_ex = ex.sum()
    w = ex / sum_ex                     # shape (2·n‑1,)
    phi = m + np.log(sum_ex) / lam      # smooth approximation of max(conv)

    # ----- gradient wrt a : 2·(w ★ a) (linear cross‑correlation) -----
    size_corr = _next_pow2(len(w) + n - 1)
    fw = np.fft.rfft(w, size_corr)
    frev = np.fft.rfft(a[::-1], size_corr)
    corr_full = np.fft.irfft(fw * frev, size_corr)
    # slice to obtain the n components that correspond to a’s indices
    corr = corr_full[n - 1 : n - 1 + n]
    grad_phi_a = 2.0 * corr

    # ----- chain rule through the soft‑max -----
    g_a = 2.0 * n * grad_phi_a               # ∂(2·n·φ)/∂a
    dot = np.dot(a, g_a)
    grad_t = a * (g_a - dot)                # ∂(2·n·φ)/∂t
    obj = 2.0 * n * phi

    return obj, grad_t


def _objective(a: np.ndarray) -> float:
    """
    Exact objective used by ``evaluate_sequence`` (scale‑invariant).

    Computes   2·n·max(a*a) / (∑a)² .
    """
    n = a.shape[0]
    if n == 0:
        return np.inf
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 1e-12:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)
