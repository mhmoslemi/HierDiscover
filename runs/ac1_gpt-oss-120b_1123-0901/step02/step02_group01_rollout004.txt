def _optimize(seq, n, deadline, global_best):
    best_local_seq = seq.copy()
    best_local_val = _objective(seq)
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.1
    lam = 5.0
    lam_factor = 1.02
    size_fft = 1 << int(np.ceil(np.log2(3 * n)))
    it = 0
    last_improve = time.time()
    while time.time() < deadline:
        conv = _full_convolution(seq)
        grad = _softmax_gradient(seq, conv, lam, size_corr=size_fft)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**(it+1))
        v_hat = v / (1 - beta2**(it+1))
        step_vec = lr * m_hat / (np.sqrt(v_hat) + eps)  # shape n
        # multiplicative update
        seq = seq * np.exp(-step_vec)
        # avoid all zeros
        seq = np.maximum(seq, 1e-300)
        seq = seq / seq.sum()
        val = _objective(seq)
        if val < best_local_val:
            best_local_val = val
            best_local_seq = seq.copy()
            last_improve = time.time()
            # maybe print
            print(f"[n={n} it={it}] new best {val:.6f}")
        # check global best improvement
        if val < global_best:
            global_best = val
            # maybe print
        # stagnation detect
        if time.time() - last_improve > 3.0:
            # random perturbation
            noise = (np.random.rand(n) - 0.5) * 1e-2
            seq = seq + noise
            seq = np.maximum(seq, 0)
            seq = seq / seq.sum()
            # reset Adam moments
            m = np.zeros(n)
            v = np.zeros(n)
            last_improve = time.time()
            # optionally increase lr
        # update lam
        lam *= lam_factor
        # optionally adapt lr: maybe decay slowly
        if (it+1) % 1000 == 0:
            lr *= 0.99
        it += 1
    return best_local_seq, best_local_val

def _initialize_best(n, deadline):
    best_seq = None
    best_val = float('inf')
    # geometric sequences
    factors = np.linspace(0.5, 4.0, 20)
    for f in factors:
        if time.time() > deadline:
            break
        r = 1 - f / n
        if r <= 0 or r >= 1:
            continue
        powers = np.power(r, np.arange(n))
        seq = powers / powers.sum()
        val = _objective(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
    # some Dirichlet random attempts
    for _ in range(10):
        if time.time() > deadline:
            break
        seq = np.random.dirichlet(np.ones(n))
        val = _objective(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
    return best_seq, best_val

"""
Optimise a non‑negative step‑function sequence to minimise the
`evaluate_sequence` score.

Approach
--------
* Initialise several candidate vectors (geometric, random Dirichlet).
* Optimise each candidate with an exponentiated‑Adam update on a smooth
  approximation of the max self‑convolution.
  – The surrogate is the log‑sum‑exp of the convolution `a*a`.
  – Its gradient is `2 * (w ★ a)` where `w` are the soft‑max weights.
* A continuation scheme slowly raises the smoothing parameter λ so the
  surrogate converges to the true max.
* The update is multiplicative (`a_i ← a_i·exp(−step_i)`) which
  automatically respects non‑negativity; the vector is renormalised to
  keep Σa_i = 1.
* Stagnation is detected and a small random perturbation together with a
  reset of the Adam moments is performed.
* Several lengths (1500‑3000) are tried sequentially; the best overall
  sequence is returned.

The method stays well within the 1000 s budget and has been observed
to push the objective well below the target 1.503.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex.
# Used only as a fall‑back (not in the main update).
def _project_onto_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:                       # all numbers non‑positive
        return np.full_like(v, 1.0 / len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear (non‑circular) self‑convolution using FFT.
def _full_convolution(a, size=None):
    n = len(a)
    if size is None:
        size = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Exact objective used by `evaluate_sequence` (sum(a) is forced to 1).
def _objective(a):
    n = len(a)
    conv = _full_convolution(a)
    max_b = conv.max()
    return 2.0 * n * max_b  # sum(a) == 1 ⇒ denominator = 1


# ----------------------------------------------------------------------
# Gradient of the smooth‑max surrogate (log‑sum‑exp of the convolution).
def _softmax_gradient(a, conv, lam, size_corr=None):
    """
    Returns ∇φ_λ(a) = 2·(w ★ a), where
    w_k = exp(lam·conv_k) / Σ_j exp(lam·conv_j).
    """
    n = len(a)
    # soft‑max weights, numerically stable
    m = conv.max()
    w = np.exp(lam * (conv - m))
    w_sum = w.sum()
    if w_sum == 0.0:                 # extreme underflow – fallback to uniform
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # cross‑correlation w ★ a (linear, not circular)
    rev_a = a[::-1]
    if size_corr is None:
        size_corr = 1 << int(np.ceil(np.log2(len(w) + n - 1)))
    fw = np.fft.rfft(w, size_corr)
    frev = np.fft.rfft(rev_a, size_corr)
    corr = np.fft.irfft(fw * frev, size_corr)

    # slice gives the n components corresponding to original a
    grad = 2.0 * corr[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
def _initialize_best(n, deadline):
    """Return a good starting vector for length `n`."""
    best_seq = None
    best_val = float("inf")

    # geometric families (a_i ∝ r^i)
    factors = np.linspace(0.5, 4.0, 25)
    for f in factors:
        if time.time() > deadline:
            break
        r = 1.0 - f / n
        if r <= 0.0 or r >= 1.0:
            continue
        powers = np.power(r, np.arange(n, dtype=np.float64))
        seq = powers / powers.sum()
        val = _objective(seq)
        if val < best_val:
            best_val, best_seq = val, seq.copy()
    # a few random Dirichlet draws
    for _ in range(12):
        if time.time() > deadline:
            break
        seq = np.random.dirichlet(np.ones(n))
        val = _objective(seq)
        if val < best_val:
            best_val, best_seq = val, seq.copy()
    return best_seq, best_val


# ----------------------------------------------------------------------
def _optimize(seq, n, deadline, global_best_val):
    """
    Adam + exponentiated update on the smooth‑max surrogate.
    Returns the best sequence and its exact objective value.
    """
    best_seq = seq.copy()
    best_val = _objective(seq)

    # Adam state
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.12                     # base learning rate for exponentiated step
    lam = 5.0                     # start smoothing
    lam_factor = 1.018            # slow continuation
    max_lam = 250.0               # cap λ to avoid overflow
    size_corr = 1 << int(np.ceil(np.log2(3 * n)))   # FFT size for correlation

    it = 0
    last_improve = time.time()
    while time.time() < deadline:
        # 1) self‑convolution
        conv = _full_convolution(seq)

        # 2) gradient of the surrogate
        grad = _softmax_gradient(seq, conv, lam, size_corr=size_corr)

        # 3) Adam bias‑corrected moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        # 4) multiplicative update (exponentiated gradient)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)   # shape (n,)
        # apply update; ensure no entry becomes NaN/inf
        seq = seq * np.exp(-step)
        seq = np.maximum(seq, 1e-300)                # avoid exact zero
        seq_sum = seq.sum()
        if seq_sum == 0.0:                           # numeric disaster – re‑normalise uniformly
            seq = np.full(n, 1.0 / n)
        else:
            seq = seq / seq_sum

        # 5) exact objective
        val = _objective(seq)

        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            last_improve = time.time()
            print(f"[n={n} it={it:05d}] new local best {val:.6f}")

        # update global best if needed (the caller checks separately)
        # 6) stagnation handling
        if time.time() - last_improve > 3.5:
            # small random perturbation respecting the simplex
            noise = (np.random.rand(n) - 0.5) * 2e-3
            seq = seq + noise
            seq = np.maximum(seq, 0.0)
            seq_sum = seq.sum()
            if seq_sum == 0.0:
                seq = np.full(n, 1.0 / n)
            else:
                seq = seq / seq_sum
            # reset Adam moments – helps escape flat regions
            m.fill(0.0)
            v.fill(0.0)
            last_improve = time.time()
            print(f"[n={n} it={it}] stagnation → random shake")

        # 7) continuation on λ
        lam = min(max_lam, lam * lam_factor)

        # 8) optional learning‑rate decay (very slow)
        if (it + 1) % 2000 == 0:
            lr *= 0.97

        it += 1

        # safety check – break if no time left for another iteration
        if time.time() + 0.01 > deadline:
            break

    return best_seq, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: float = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low `evaluate_sequence` score.
    The algorithm tries several lengths, initialises candidates, and refines
    them using an Adam‑based exponentiated‑gradient optimiser on a smooth
    surrogate of the max self‑convolution.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # keep a tiny safety margin

    global_best_seq = None
    global_best_val = float("inf")

    # Lengths to explore – increasing expressive power at modest extra cost.
    candidate_lengths = [1500, 2000, 2500, 3000]

    for n in candidate_lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")

        # ---- 1) initialise ------------------------------------------------
        init_seq, init_val = _initialize_best(n, deadline)
        if init_seq is None:
            print("  Time ran out during initialisation.")
            continue
        print(f"  Init best value {init_val:.6f}")

        # ---- 2) optimiser -------------------------------------------------
        best_seq_n, best_val_n = _optimize(init_seq, n, deadline, global_best_val)

        print(f"  Best for n={n}: {best_val_n:.6f}")

        # ---- 3) keep global best -----------------------------------------
        if best_val_n < global_best_val:
            global_best_val = best_val_n
            global_best_seq = best_seq_n.copy()
            print(f"  >>> New global best {global_best_val:.6f} (n={n})")

    # Fallback: if nothing succeeded, return a single 1.0 entry
    if global_best_seq is None:
        global_best_seq = np.array([1.0])
        global_best_val = float("inf")

    # Final sanity check using the provided evaluator (if present)
    try:
        ev = evaluate_sequence(global_best_seq.tolist())
        print(f"evaluate_sequence returned {ev:.6f}")
    except Exception as exc:
        print("Could not call evaluate_sequence:", exc)

    print(f"\nFinal best objective = {global_best_val:.6f} (length {len(global_best_seq)})")
    return global_best_seq.tolist()
