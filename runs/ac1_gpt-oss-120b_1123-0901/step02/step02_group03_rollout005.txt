"""
Propose a non‑negative step‑function that minimizes

        2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm is a *temperature‑controlled Adam* optimiser on the
simplex {a_i ≥ 0, Σ a_i = √(2n)}.  At each iteration we minimise a
smooth approximation of the max‑convolution

        f_τ(a) = τ·log Σ_k exp(conv_k(a)/τ)

whose gradient can be written as a convolution and therefore evaluated
with a pair of FFTs.  The temperature τ is slowly lowered, turning the
smooth loss into the true ℓ_∞ loss.  Random restarts, occasional
Gaussian “shakes”, and a final fallback to the best sequence seen so
far make the search robust.  The code respects the 1000 s time budget
and returns the best list of heights found.
"""

import time
import numpy as np

# --------------------------------------------------------------------------- #
# Helper: Euclidean projection onto the simplex { x_i ≥ 0 , Σ x_i = s }.
# --------------------------------------------------------------------------- #
def project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project the vector ``v`` onto the simplex of radius ``s``."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]

    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------------------- #
# FFT‑based full convolution (real input → real output).
# --------------------------------------------------------------------------- #
def conv_full_fft(a: np.ndarray) -> np.ndarray:
    """Return a full linear convolution a * a (length 2·len(a)-1)."""
    n = a.shape[0]
    n_fft = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    conv = np.fft.irfft(A * A, n=n_fft)[: 2 * n - 1]
    return conv


# --------------------------------------------------------------------------- #
# Compute soft‑max (temperature‑controlled) weights from a convolution.
# --------------------------------------------------------------------------- #
def softmax_weights(conv: np.ndarray, tau: float) -> np.ndarray:
    """Return the soft‑max weights w_i ∝ exp(conv_i / τ)."""
    # Numerical stability: shift by the maximum value.
    m = conv.max()
    if tau <= 0.0:
        # Exact max: distribute weight uniformly over all maximal entries.
        mask = conv == m
        w = np.zeros_like(conv, dtype=np.float64)
        w[mask] = 1.0 / mask.sum()
        return w
    shifted = (conv - m) / tau          # ≤ 0
    exp_vals = np.exp(shifted)
    w = exp_vals / exp_vals.sum()
    return w


# --------------------------------------------------------------------------- #
# Gradient of the smoothed loss:  grad = 2·(w_rev ⋆ a) evaluated at proper indices.
# --------------------------------------------------------------------------- #
def gradient_from_weights(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Given soft‑max weights w (length 2·n‑1) and current a (length n),
    return the gradient of τ·log Σ exp(conv/τ) w.r.t. a.
    """
    n = a.shape[0]
    w_rev = w[::-1]                                   # length 2·n‑1
    size = 3 * n - 2                                  # length of w_rev ⋆ a
    n_fft = 1 << ((size - 1).bit_length())            # next power‑of‑two

    # real FFTs (fast)
    W = np.fft.rfft(w_rev, n=n_fft)
    A = np.fft.rfft(a, n=n_fft)
    conv = np.fft.irfft(W * A, n=n_fft)[:size]        # w_rev ⋆ a

    # indices 2·n‑2, 2·n‑3, …, n‑1 give the needed inner products
    idx = 2 * n - 2 - np.arange(n)
    grad = 2.0 * conv[idx]
    return grad


# --------------------------------------------------------------------------- #
# Run a single optimisation instance for a fixed length n.
# --------------------------------------------------------------------------- #
def _run_optimisation_instance(
    n: int,
    init_vec: np.ndarray,
    time_budget: float,
) -> tuple[np.ndarray, float]:
    """
    Optimise a vector of length ``n`` starting from ``init_vec``.
    Returns the best vector encountered and its maximal convolution value.
    """
    target_sum = np.sqrt(2.0 * n)                     # Σ a_i = √(2n)
    a = project_onto_simplex(init_vec.astype(np.float64), target_sum)

    # Adam parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    # learning‑rate & temperature schedule
    lr = 0.5
    tau = 0.5
    tau_min = 0.02
    decay_tau = 0.9985                                 # per iteration

    best_a = a.copy()
    best_score = float("inf")
    it = 0
    deadline = time.time() + time_budget

    while time.time() < deadline:
        # ----- evaluate current objective (max convolution) -----
        conv = conv_full_fft(a)
        cur_max = conv.max()
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        # ----- smoothed max (soft‑max) -----
        w = softmax_weights(conv, tau)

        # ----- gradient (smooth) -----
        grad = gradient_from_weights(w, a)

        # ----- Adam update -----
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = np.maximum(a_new, 0.0)                # keep non‑negative
        a_new = project_onto_simplex(a_new, target_sum)

        # ----- occasional random shake (simulated‑annealing style) -----
        if np.random.rand() < 0.01:
            noise = np.random.randn(n) * (target_sum * 0.004 / n)
            a_new = project_onto_simplex(np.maximum(a_new + noise, 0.0), target_sum)
            lr *= 0.92                                 # a tiny step‑size reduction

        a = a_new

        # ----- schedule decay -----
        lr *= 0.9999
        tau = max(tau * decay_tau, tau_min)

    # Return the best vector discovered (already on the simplex)
    return best_a, best_score

# Main entry point expected by the grader.
# --------------------------------------------------------------------------- #
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a low‑value step‑function within ``budget_s`` seconds.
    The best sequence discovered is returned as a plain Python list.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5            # safety margin

    best_seq = None
    best_score = float("inf")
    best_n = None

    # optional warm start from a previously stored sequence
    warm_start = None
    if "height_sequence_1" in globals():
        try:
            warm_start = np.asarray(height_sequence_1, dtype=np.float64)
        except Exception:
            warm_start = None

    # ------------------------------------------------------------------- #
    # Outer loop: many independent restarts with varying lengths.
    # ------------------------------------------------------------------- #
    while time.time() < deadline:
        # Choose a problem size.  1400‑2600 is a sweet spot (enough freedom,
        # still cheap enough for many iterations).
        n = np.random.randint(1400, 2601)

        # Initialise the vector.
        if warm_start is not None and np.random.rand() < 0.45 and warm_start.size >= n:
            init = warm_start[:n].copy()
            # tiny jitter to escape exact copies
            init += np.random.rand(n) * 1e-4
        else:
            # Dirichlet‑like random start (positive entries).
            init = np.random.rand(n)

        # Allocate time for this run (≈ 12 s, leaving room for many restarts).
        remaining = deadline - time.time()
        run_time = min(12.0, remaining)
        if run_time <= 0.0:
            break

        seq, score = _run_optimisation_instance(n, init, run_time)

        if score < best_score:
            best_seq = seq.copy()
            best_score = score
            best_n = n
            print(f"[{time.time():.2f}] New global best = {best_score:.6f} (n={best_n})")

        # Early exit if we already beat the target bound.
        if best_score < 1.503:
            break

    # ------------------------------------------------------------------- #
    # Fallback in the unlikely case no candidate was produced.
    # ------------------------------------------------------------------- #
    if best_seq is None:
        n = 2000
        target = np.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=np.float64)
        best_score = 2.0                     # uniform sequence yields 2.0

    # Store for possible later runs.
    globals()["height_sequence_1"] = best_seq.tolist()

    # Return plain Python list as required by the grader.
    return best_seq.tolist()
