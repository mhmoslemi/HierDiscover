"""
Hybrid LP‑guided refinement + projected gradient descent.
-------------------------------------------------------

* Initialise a non‑negative sequence (length n≈2000) from a known seed
  (the global variable ``height_sequence_1``) or from a random simplex.
* Repeatedly:
    1. Compute the self‑convolution f*f and the current maximum M.
    2. Take the (few) shift positions where the convolution is close to M
       (active set).  For each active shift k build the linear constraint
       (f ∗ g)[k] ≤ M.  This gives a small dense LP:
           maximise   Σ g_i
           subject to A·g ≤ M,   g_i ≥ 0 .
       The LP tells us how to move mass towards indices that do not enlarge
       the critical convolution entries.
    3. Scale the LP solution to the same ℓ₁‑norm as f and line‑search a
       convex combination  f←(1−t)f+t·g  (t∈{1,½,¼,…}) keeping the sum fixed.
       The best t that strictly lowers the objective is kept.
    4. If the LP step yields no improvement, fall back to a few steps of
       aggressive projected gradient descent (the method used previously).

The loop stops when the time budget (≈1000 s) is exhausted.  The best
sequence found is returned as a plain Python list.
"""

import time
import numpy as np

# optional SciPy – if unavailable we fall back to pure gradient descent
try:
    from scipy.optimize import linprog
except Exception:  # pragma: no cover
    linprog = None


# ----------------------------------------------------------------------
def _project_onto_simplex(v):
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:                 # all entries were zero
        return np.full_like(v, 1.0 / len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _self_convolve(seq: np.ndarray) -> np.ndarray:
    """Full convolution of a real vector with itself (length 2n‑1)."""
    n = seq.shape[0]
    m = 2 * n - 1
    nfft = _next_pow2(m)
    A = np.fft.rfft(seq, n=nfft)
    conv = np.fft.irfft(A * A, n=nfft)[:m]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective used by evaluate_sequence (scale‑invariant)."""
    sum_a = seq.sum()
    if sum_a < 0.01:
        return float("inf")
    conv = _self_convolve(seq)
    max_b = conv.max()
    return float(2.0 * len(seq) * max_b / (sum_a * sum_a))


def _build_active_A(f: np.ndarray, shifts: np.ndarray) -> np.ndarray:
    """Construct dense matrix A (|shifts|×n) with
       (A·g)[k] = (f∗g)[k] for each active shift k."""
    n = f.shape[0]
    m = shifts.shape[0]
    A = np.zeros((m, n), dtype=np.float64)
    for row, k in enumerate(shifts):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i = np.arange(i_start, i_end + 1, dtype=np.int32)
        j = k - i
        A[row, j] = f[i]
    return A


def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """Return a list of non‑negative heights that (hopefully) beats the
    previous best bound (≈1.5449).  The routine mixes a small LP step with
    gradient descent and respects the 1000 s time limit."""
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5          # small safety margin

    # ------------------------------------------------------------------
    # 1. Length of the step function.  2000 is a good compromise.
    n = 2000

    # ------------------------------------------------------------------
    # 2. Initialise the sequence.
    best_seq = None
    best_score = float("inf")

    # try to use the previously discovered construction if present
    if "height_sequence_1" in globals():
        try:
            init = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            init = np.clip(init, 0.0, 1000.0)
            if init.sum() > 0.0:
                init = init / init.sum()          # normalise to sum = 1
                sc = _objective(init)
                best_seq = init.copy()
                best_score = sc
                print(f"[Init] global seed      score={sc:.6f}")
        except Exception:
            pass

    # fall back to a random point on the simplex if needed
    if best_seq is None:
        init = np.random.exponential(1.0, size=n)
        init = init / init.sum()
        best_seq = init.copy()
        best_score = _objective(best_seq)
        print(f"[Init] random simplex    score={best_score:.6f}")

    f = best_seq.copy()
    cur_score = best_score

    # ------------------------------------------------------------------
    # 3. Main optimisation loop – LP steps followed by a tiny gradient
    #    fallback if the LP cannot improve further.
    it = 0
    max_iter = 50                 # safety cap; real stop is time‑budget
    active_thresh = 0.99          # keep constraints where conv ≥ thresh·M
    max_active = 12               # never more than this many rows in the LP

    while time.time() < deadline and it < max_iter:
        # ----- (a) compute current max convolution -----
        conv_f = _self_convolve(f)
        M = conv_f.max()
        obj_f = 2.0 * n * M / (f.sum() ** 2)      # same as _objective(f)

        # early exit if we already beat the target
        if obj_f < 1.503:
            print(f"[Done] reached target  obj={obj_f:.6f} at iter {it}")
            best_seq = f.copy()
            break

        # ----- (b) choose active shifts (those close to the maximum) -----
        active_shifts = np.where(conv_f >= active_thresh * M)[0]
        if active_shifts.size == 0:               # should not happen

        # keep only a few of the most critical shifts

        g_candidate = None

            c = -np.ones(n)                         # maximise Σ g_i

            b = np.full(A.shape[0], M, dtype=np.float64)

                pass
