def propose_candidate(seed=0, budget_s=1000):
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5
    n = 5000
    # initialize candidate arcsine shape
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    a = a / a.sum()  # sum=1
    best_seq = a.copy()
    best_score = 2.0 * n * np.max(np.convolve(a, a))
    print(f"[Init] score {best_score:.6f}")

    # precompute constants
    L = 1 << ((3*n - 2).bit_length())
    # Adam parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.1  # initial learning rate
    alpha = 1.0  # softmax temperature
    alpha_max = 30.0
    alpha_growth = 1.001
    it = 0

    while time.time() < deadline:

        A = np.fft.fft(a, L)
        # convolution
        conv = np.fft.ifft(A * A).real[:2*n-1]

        max_c = conv.max()

        grad = 2.0 * np.fft.ifft(W * np.conj(A)).real[:n]

def _softmax_weights(conv, alpha):
    max_c = np.max(conv)
    exp_vals = np.exp(alpha * (conv - max_c))
    sum_exp = np.sum(exp_vals)
    return exp_vals / sum_exp

def _descent(initial_seq, n, deadline, rng):
    # returns best_seq, best_score
    # We may pass L = _next_power_of_two(3*n-2) once.
    # Use Adam hyperparameters
    a = initial_seq.copy()
    # ensure sum=1
    a = a / a.sum()
    best_seq = a.copy()
    # compute initial conv and score
    L = _next_power_of_two(3*n - 2)
    A = np.fft.fft(a, L)
    conv = np.fft.ifft(A * A).real[:2*n-1]
    best_score = 2.0 * n * np.max(conv)
    # init Adam
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.2   # maybe large?
    alpha = 1.0
    alpha_max = 30.0
    alpha_growth = 1.001
    it = 1
    no_improve = 0
    while time.time() < deadline:
        # compute softmax weights
        w = _softmax_weights(conv, alpha)
        # gradient
        W = np.fft.fft(w, L)
        grad = 2.0 * np.fft.ifft(A * np.conj(W)).real[:n]
        # adam moments
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        # projection onto simplex
        a_new = _project_onto_simplex(a_new)
        # compute new convolution for next iteration
        A = np.fft.fft(a_new, L)
        conv = np.fft.ifft(A * A).real[:2*n-1]
        # evaluate objective
        cur_score = 2.0 * n * np.max(conv)
        if cur_score < best_score - 1e-12:   # improvement
            best_score = cur_score
            best_seq = a_new.copy()
            no_improve = 0
            print(f"[Iter {it:07d}] new best {best_score:.6f}")
        else:
            no_improve += 1
        # stalling: add jitter
        if no_improve >= 2000:
            jitter = rng.normal(scale=0.001, size=n)
            a_new = a_new + jitter
            a_new = np.clip(a_new, 0, None)

            # reset Adam moments
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0
            # recompute A, conv for this jitter
            A = np.fft.fft(a_new, L)
            conv = np.fft.ifft(A * A).real[:2*n-1]
            print(f"[Iter {it:07d}] jitter applied")
        # update variables
        a = a_new
        # update alpha

        # time check
        it += 1
        if time.time() >= deadline:

    # final evaluation with evaluate_sequence to get proper result

"""
Propose a step‑function (non‑negative sequence) with a very low evaluation value.

Algorithm
---------
1. Choose a relatively large length `n` (5000) – the theoretical lower bound
   for the objective is `2*n/(2*n‑1) ≈ 1`, so a well‑flattened convolution can
   easily beat the target 1.503.
2. Initialise several diverse candidates:
   * the exact arcsine density (the continuous solution of f*f = const),
   * the uniform distribution,
   * a few random Dirichlet samples.
3. For each candidate run a projected Adam optimiser that
   * computes the self‑convolution with an FFT (O(n log n)),
   * builds a *soft‑max* approximation of the max‑operator
     (`w_k ∝ exp(alpha * conv_k)`) and uses the gradient
     `∇ = 2·corr(a , w)`,
   * updates the coefficients with Adam,
   * projects the result onto the probability simplex (non‑negative,
     sum = 1).
   The temperature `alpha` is slowly increased so that the optimisation
   starts smooth and ends focusing on the true maximum.
4. Keep the best sequence (according to the exact `evaluate_sequence`
   function) discovered over all restarts and return it.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x ≥ 0, Σx = 1}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.size + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / v.size)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_power_of_two(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max over the convolution values (used as a smooth max)."""
    max_c = np.max(conv)
    # stable exponentiation
    exp_vals = np.exp(alpha * (conv - max_c))
    return exp_vals / exp_vals.sum()


def _descent(initial_seq: np.ndarray,
             n: int,
             deadline: float,
             rng: np.random.Generator):
    """
    Run a projected Adam optimisation that flattens the self‑convolution.

    Returns the best sequence found (as a NumPy array) and its exact
    `evaluate_sequence` score.
    """
    # --- parameters ----------------------------------------------------
    beta1 = 0.9          # Adam momentum
    beta2 = 0.999        # Adam second‑moment decay
    eps = 1e-8
    lr = 0.20            # base learning rate
    alpha = 1.0          # start temperature for the soft‑max
    alpha_max = 30.0
    alpha_growth = 1.001
    # ------------------------------------------------------------------

    a = _project_onto_simplex(initial_seq.copy())
    L = _next_power_of_two(3 * n - 2)           # length for all FFTs

    # initialise Adam moments
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # initial convolution & score
    A = np.fft.fft(a, L)
    conv = np.fft.ifft(A * A).real[: 2 * n - 1]
    conv = np.maximum(conv, 0.0)                # eliminate tiny negatives
    best_seq = a.copy()
    best_score = 2.0 * n * np.max(conv)        # sum(a)=1 ⇒ objective = 2·n·max(conv)
    no_improve = 0
    it = 1

    while time.time() < deadline:
        # soft‑max weights – smoother surrogate of max(conv)
        w = _softmax_weights(conv, alpha)

        # gradient = 2·corr(a , w)
        W = np.fft.fft(w, L)
        grad = 2.0 * np.fft.ifft(A * np.conj(W)).real[:n]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1**it)
        v_hat = v / (1.0 - beta2**it)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto the simplex (keeps sum = 1 and non‑negative)
        a_new = _project_onto_simplex(a_new)

        # new convolution (will be used in the next iteration)
        A = np.fft.fft(a_new, L)
        conv = np.fft.ifft(A * A).real[: 2 * n - 1]
        conv = np.maximum(conv, 0.0)

        cur_score = 2.0 * n * np.max(conv)
        if cur_score + 1e-12 < best_score:      # improvement
            best_score = cur_score
            best_seq = a_new.copy()
            no_improve = 0
            print(f"[Iter {it:07d}] new best {best_score:.6f}")
        else:
            no_improve += 1

        # occasional jitter to escape plateaus
        if no_improve >= 2000:
            jitter = rng.normal(scale=0.001, size=n)
            a_new = a_new + jitter
            a_new = np.clip(a_new, 0.0, None)
            a_new = a_new / a_new.sum()
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0
            print(f"[Iter {it:07d}] jitter applied")

        # prepare for next loop
        a = a_new
        alpha = min(alpha_max, alpha * alpha_growth)
        it += 1
        if time.time() >= deadline:
            break

    # final exact evaluation (uses the given security‑checked function)
    final_obj = evaluate_sequence(best_seq.tolist())
    print(f"Descent finished. Exact objective = {final_obj:.6f}")
    return best_seq, final_obj


def propose_candidate(seed: int = 42,
                     budget_s: float = 1000,
                     **kwargs):
    """
    Return a list of non‑negative heights that minimises ``evaluate_sequence``
    within the allotted time.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # a small safety margin

    # --------------------------------------------------------------
    # problem size – larger *n* gives a tighter theoretical lower bound.
    n = 5000

    # --------------------------------------------------------------
    # build a small pool of promising starts
    init_pool = []

    # 1. arcsine law (continuous solution of f*f = const)
    i = np.arange(n, dtype=np.float64)
    arcs = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    arcs = arcs / arcs.sum()
    init_pool.append(arcs)

    # 2. uniform distribution (baseline)
    init_pool.append(np.full(n, 1.0 / n))

    # 3. a few random Dirichlet draws
    for _ in range(3):
        rnd = rng.random(n)
        init_pool.append(rnd / rnd.sum())

    best_overall_seq = None
    best_overall_score = float("inf")

    # -----------------------------------------------------------------
    # run the optimiser on each start while time permits
    for idx, start in enumerate(init_pool):
        if time.time() >= deadline:
            break
        print(f"[Start {idx}] running descent from initial candidate")
        seq, score = _descent(start, n, deadline, rng)
        if score < best_overall_score:
            best_overall_score = score
            best_overall_seq = seq.copy()
            print(f"[Pool] new overall best = {best_overall_score:.6f}")

    # -----------------------------------------------------------------
    # fallback (should never happen) – return a simple uniform sequence
    if best_overall_seq is None:
        best_overall_seq = np.full(n, 1.0 / n)

    print(f"Finished. Best objective = {best_overall_score:.6f} (length {n})")
    return best_overall_seq.tolist()
