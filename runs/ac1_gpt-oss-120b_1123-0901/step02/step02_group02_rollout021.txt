def solve_lp_active(f, M, active_idxs):
    n = f.shape[0]
    m = len(active_idxs)
    if m == 0:
        return None
    A = np.zeros((m, n), dtype=np.float64)
    for row_idx, i in enumerate(active_idxs):
        low = max(0, i - (n - 1))
        high = min(n - 1, i)
        cols = np.arange(low, high+1, dtype=int)
        A[row_idx, cols] = f[i - cols]
    # b_ub vector:
    ub = np.full(m, M, dtype=np.float64)
    # equality constraints? none
    res = linprog(c=-np.ones(n), A_ub=A, b_ub=ub, bounds=(0, None), method='highs')
    if res.success:
        return res.x
    else:
        return None
```

"""
propose_candidate – hybrid search for a low‑value step function.

Strategy
--------
* Randomly pick a length n (1200 … 3000) and initialise a non‑negative
  vector a_i ≈ (i+1)^{‑p} with a random exponent p∈[0.2,0.8] and a tiny
  noisy perturbation.
* Normalise the vector to the simplex Σ a_i = √(2n).  The evaluation
  function then returns exactly max(conv(a,a)), so we can use this
  cheaper proxy.
* Run a projected (momentum) gradient descent on the “active’’ set
  – indices whose convolution value is within a small relative tolerance
  of the current maximum.  The gradient of Σ_{k∈active}(a∗a)_k is
  2 a_{k‑i} summed over the active indices, which we compute in
  _O(n·|active|)_.  A simple back‑tracking line‑search (steps 1, ½, ¼)
  avoids overshooting.
* Every few iterations we solve a small linear programme:
      maximise   Σ b_j
      subject to f*b ≤ M   (full convolution constraints)
                b_j ≥ 0
  where f is the current vector and M is its maximal convolution entry.
  The LP produces a feasible direction g (scaled to the simplex) that
  typically reduces the peak when mixed into f:   f ← (1‑t) f + t g.
* After the time slice for a given length expires we keep the best local
  vector.  The outer loop performs many independent restarts; the best
  vector seen so far is kept globally.
* Finally, if a few seconds remain, a lightweight simulated‑annealing
  refinement perturbs individual coordinates of the global best and
  accepts improvements (or, with a decaying temperature, occasional
  uphill moves).

All heavy work (FFT convolution, gradient, LP) is vectorised; the
algorithm stays well within the 1000 s budget on a single CPU.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper 1 – Euclidean projection onto the simplex {x_i ≥ 0, Σ x_i = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex of radius s (s≥0)."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper 2 – convolution via FFT (real inputs → real output)
# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using a power‑of‑2 FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    size = 1 << ((m - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[:m]
    np.maximum(conv, 0.0, out=conv)             # eliminate tiny negatives
    return conv


# ----------------------------------------------------------------------
# Helper 3 – active set of indices close to the current maximum
# ----------------------------------------------------------------------
def _active_indices(conv: np.ndarray, rel_tol: float = 5e-4) -> tuple[float, np.ndarray]:
    """Return (max value, indices) where conv ≥ max − rel_tol·max."""
    max_val = conv.max()
    tol = max(1e-12, max_val * rel_tol)
    active = np.where(conv >= max_val - tol)[0]
    return max_val, active


# ----------------------------------------------------------------------
# Helper 4 – gradient of the sum of active convolution entries
# ----------------------------------------------------------------------
def _gradient(a: np.ndarray, active: np.ndarray) -> np.ndarray:
    """Gradient of Σ_{k∈active}(a∗a)_k w.r.t. a."""
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in active:
        low = max(0, k - (n - 1))
        high = min(n - 1, k)
        idx = np.arange(low, high + 1, dtype=int)
        partner = k - idx
        grad[idx] += 2.0 * a[partner]      # ∂/∂a_i Σ a_i a_{k-i} = 2 a_{k-i}
    return grad


# ----------------------------------------------------------------------
# Helper 5 – solve the linear programme  max Σ b_j  s.t. f∗b ≤ M , b≥0
# ----------------------------------------------------------------------
def _solve_lp_full(f: np.ndarray, M: float) -> np.ndarray | None:
    """Return an optimal b (as dense ndarray) or None on failure."""
    n = f.shape[0]
    m = 2 * n - 1
    # Build the (m×n) constraint matrix A where row i is f shifted.
    A = np.zeros((m, n), dtype=np.float64)
    for i in range(m):
        low = max(0, i - (n - 1))
        high = min(n - 1, i)
        cols = np.arange(low, high + 1, dtype=int)
        A[i, cols] = f[i - cols]
    # Linear programme (maximise Σ b) → minimise −Σ b
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=np.full(m, M, dtype=np.float64),
                  bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


# ----------------------------------------------------------------------
# Core optimisation for a single length n
# ----------------------------------------------------------------------
def _optimise_one_length(
    n: int,
    init: np.ndarray,
    run_deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Hybrid projected momentum gradient descent with occasional LP updates.
    Returns the best vector found and its max‑convolution value (which equals
    the evaluation score after normalisation).
    """
    target_sum = np.sqrt(2.0 * n)          # Σ a_i = √(2n)
    a = _project_onto_simplex(init.astype(np.float64), target_sum)

    # current best (local) and its score
    conv = _conv_fft(a)
    best_score = conv.max()
    best_vec = a.copy()

    # momentum bookkeeping
    velocity = np.zeros_like(a)
    base_step = 0.22          # relatively aggressive initial step
    momentum_beta = 0.9
    rel_tol = 1e-3           # a little looser than in the original code
    iteration = 0

    while time.time() < run_deadline:
        conv = _conv_fft(a)
        max_val, active = _active_indices(conv, rel_tol)

        # ----- gradient descent step ------------------------------------
        grad = _gradient(a, active)
        velocity = momentum_beta * velocity + (1.0 - momentum_beta) * grad
        max_grad = np.max(np.abs(velocity)) + 1e-12
        step = base_step * max_val / max_grad

        # simple back‑tracking line search (1, ½, ¼)
        best_local_a = None
        best_local_score = np.inf
        for factor in (1.0, 0.5, 0.25):
            cand = a - factor * step * velocity
            cand = _project_onto_simplex(cand, target_sum)
            cand_score = _conv_fft(cand).max()
            if cand_score < best_local_score:
                best_local_score = cand_score
                best_local_a = cand

        # accept if we made progress
        if best_local_score < max_val - 1e-12:
            a = best_local_a
            max_val = best_local_score
            if max_val < best_score:
                best_score = max_val
                best_vec = a.copy()
                print(f"[{time.time():.2f}] n={n} GD best={best_score:.6f}")
        else:
            # occasional jitter to leave possible plateaux
            if np.random.rand() < 0.015:
                noise = np.random.randn(n) * (target_sum * 1e-3 / n)
                a = _project_onto_simplex(np.maximum(a + noise, 0.0), target_sum)

        # ----- occasional LP‑based update --------------------------------
        if iteration % 35 == 0 and iteration > 0:
            # solve the full LP (still cheap for n≈2‑3k)
            b = _solve_lp_full(a, max_val)
            if b is not None and b.sum() > 0:
                # scale to the simplex
                g = b * (target_sum / b.sum())
                t = 0.12                     # mixing factor (kept modest)
                a_lp = (1.0 - t) * a + t * g
                a_lp = _project_onto_simplex(a_lp, target_sum)
                lp_score = _conv_fft(a_lp).max()
                if lp_score < max_val:
                    a = a_lp
                    max_val = lp_score
                    if max_val < best_score:
                        best_score = max_val
                        best_vec = a.copy()
                        print(f"[{time.time():.2f}] n={n} LP improvement={best_score:.6f}")

        # decay step size gently
        base_step *= 0.9994
        iteration += 1

    return best_vec, best_score


# ----------------------------------------------------------------------
# Light simulated‑annealing fine‑tuner (run on the global best at the end)
# ----------------------------------------------------------------------
def _anneal_refine(seq: np.ndarray, deadline: float) -> np.ndarray:
    """Small random walk with a decaying temperature to squeeze out a few more bits."""
    n = seq.shape[0]
    target_sum = np.sqrt(2.0 * n)
    a = seq.copy()
    best_score = _conv_fft(a).max()
    best = a.copy()
    temp = 0.08

    while time.time() < deadline:
        i = np.random.randint(n)
        delta = (np.random.randn() * 0.02) * a[i]   # ∼2 % relative change
        a_new = a.copy()
        a_new[i] = max(a_new[i] + delta, 0.0)
        a_new = _project_onto_simplex(a_new, target_sum)
        sc = _conv_fft(a_new).max()
        if sc < best_score:
            best_score = sc
            best = a_new.copy()
            a = a_new
            temp = max(temp * 0.995, 1e-6)
            print(f"[{time.time():.2f}] anneal new best={best_score:.6f}")
        else:
            # Metropolis acceptance
            if np.random.rand() < np.exp(-(sc - best_score) / temp):
                a = a_new
            temp *= 0.9997
    return best

# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative height sequence that minimises the evaluation
    function within the allotted wall‑clock budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    best_seq = None
    best_score = np.inf

    # typical useful range for the length – larger n gives a smoother kernel
    min_n, max_n = 1200, 3000

    # ------------------------------------------------------------------
    # Restart loop – each restart gets a few seconds of work.
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # – choose a length -------------------------------------------------
        remaining = deadline - time.time()
        # as time runs out we bias towards smaller n (cheaper runs)
        if remaining < 30.0:
            n = np.random.randint(1200, 2000)
        else:
            n = np.random.randint(min_n, max_n + 1)

        # – initial shape ---------------------------------------------------
        p = np.random.uniform(0.20, 0.80)          # wider exponent range
        idx = np.arange(1, n + 1, dtype=np.float64)
        init = idx ** (-p)

        # small random jitter to break symmetry
        init += np.random.rand(n) * 1e-4

        # warm‑start from the current global best with some probability
        if best_seq is not None and np.random.rand() < 0.35:
            if best_seq.size >= n:
                init = best_seq[:n].copy()
            else:
                pad = np.random.rand(n - best_seq.size) * 1e-4
                init = np.concatenate([best_seq, pad])

        # – allocate a time slice for this run -----------------------------
        run_deadline = min(deadline, time.time() + 11.0)   # ≈10 s per restart

        # – run the hybrid optimiser ---------------------------------------
        seq_local, score_local = _optimise_one_length(n, init, run_deadline)

        # – update the global best -----------------------------------------
        if score_local < best_score:
            best_score = score_local
            best_seq = seq_local.copy()
            print(
                f"[{time.time():.2f}] GLOBAL new best = {best_score:.6f} (n={n})"
            )

        # early exit if we already beat the target comfortably
        if best_score < 1.45:      # well below the 1.503 target
            break

    # ------------------------------------------------------------------
    # Final simulated‑annealing refinement (if time permits)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        refined = _anneal_refine(best_seq, deadline)
        refined_score = _conv_fft(refined).max()
        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined
            print(f"[{time.time():.2f}] FINAL refined best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1500
        target_sum = np.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=np.float64)

    # Return as a plain Python list (exactly what the evaluator expects)
    return best_seq.tolist()
