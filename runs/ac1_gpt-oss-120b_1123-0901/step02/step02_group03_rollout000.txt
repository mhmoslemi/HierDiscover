def propose_candidate(seed=42, budget_s=1000.0, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5
    best_seq = None
    best_score = float('inf')
    best_n = None

    # params
    REFINE_TIME = 200.0   # time for local refinement

    # Stage 1: many gradient (Adam) runs
    while time.time() < deadline - REFINE_TIME:
        # choose n
        if best_n is not None and rng.random() < 0.3:
            n = int(best_n + rng.integers(-200, 201))
            n = max(500, min(4000, n))
        else:
            n = rng.integers(800, 4001)  # choose between 800 and 4000

        # initialise a
        if best_seq is not None and best_n == n:
            a0 = np.array(best_seq, dtype=float)
        elif best_seq is not None and best_n is not None and best_n > n:
            # take first n entries of best_seq (maybe cut)
            a0 = np.array(best_seq[:n], dtype=float)
        else:
            a0 = rng.random(n)

        # run optimization
        run_best_a, run_best_score = _run_adam_instance(a0, n, deadline - time.time())
        if run_best_score < best_score:
            best_score = run_best_score
            best_seq = run_best_a.copy()
            best_n = n
            print(f"[{time.time():.2f}] New global best = {best_score:.6f} (n={n})")

    # Stage 2: local refinement via random pair moves
    if best_seq is None:
        # fallback to uniform
        n = 1000
        target_sum = np.sqrt(2.0 * n)

    target_sum = np.sqrt(2.0 * n)
    a = _project_onto_simplex(init_vec.astype(float), target_sum)
    best_a = a.copy()
    best_score = float('inf')
    # Adam parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    lr = 0.05  # learning rate

            best_score = max_val
            best_a = a.copy()

"""
Step‑function optimizer for the objective

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm works in two phases:

1.  Many short Adam‑based subgradient runs on random lengths (800 s).
    Each run keeps the vector on the simplex {x_i≥0, Σx_i = √(2n)}.
    The subgradient is the sum of the gradients of all convolution entries
    that attain the current maximum.  After each update we project back onto
    the simplex.  Random Gaussian shakes are added occasionally to escape
    shallow local minima.

2.  A long “pairwise transfer” refinement (≈200 s) on the best vector found.
    In each step a small amount of mass is moved from one index to another;
    the move is kept if it reduces the maximal convolution value.
    This simple hill‑climbing usually squeezes a few percent more out of a
    good candidate.

The whole search respects the 1000 s budget, returns a plain Python list,
and uses only NumPy (no external solvers).  The internal score that is
tracked equals the maximal autocorrelation entry because we keep the sum
normalised to √(2n); this is exactly the value returned by ``evaluate_sequence``.
"""

import time
import numpy as np

# --------------------------------------------------------------------------- #
# Helper: Euclidean projection onto the simplex { x_i ≥ 0, Σ x_i = s }.
# --------------------------------------------------------------------------- #
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project ``v`` onto the simplex of radius ``s``."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------------------- #
# Fast full convolution (real FFT) – returns a length 2*n‑1 array.
# --------------------------------------------------------------------------- #
def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    n_fft = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    conv = np.fft.irfft(A * A, n=n_fft)[: 2 * n - 1]
    return conv


# --------------------------------------------------------------------------- #
# Gradient of the *maximal* convolution entry (or a set of ties).
# grad_i = Σ_{k∈active} 2 * a_{k-i}   (only when the index is inside the overlap)
# --------------------------------------------------------------------------- #
def _gradient_of_active(a: np.ndarray, active_idxs: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in active_idxs:
        low = max(0, k - (n - 1))
        high = min(n - 1, k)
        idx = np.arange(low, high + 1, dtype=int)
        partner = k - idx
        grad[idx] += 2.0 * a[partner]
    return grad


# --------------------------------------------------------------------------- #
# Adam‑based subgradient optimisation for a single length.
# --------------------------------------------------------------------------- #
def _run_adam_instance(init_vec: np.ndarray, n: int, time_budget: float):
    """
    Optimise a vector of length ``n`` for at most ``time_budget`` seconds.
    Returns the best vector seen in this run and its maximal convolution value.
    """
    target_sum = np.sqrt(2.0 * n)
    a = _project_onto_simplex(init_vec.astype(float), target_sum)

    best_a = a.copy()
    best_score = float("inf")

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.07                     # base learning rate
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    end_time = time.time() + time_budget
    while time.time() < end_time:
        # ----- current score -------------------------------------------------
        conv = _conv_full_fft(a)
        max_val = float(conv.max())
        if max_val < best_score:
            best_score = max_val
            best_a = a.copy()
            # tiny progress report (can be silenced if too noisy)
            # print(f"    [inner] best = {best_score:.6f}")

        # ----- active set ----------------------------------------------------
        eps_rel = 1e-12
        active = np.where(conv >= max_val - eps_rel)[0]

        # ----- sub‑gradient --------------------------------------------------
        grad = _gradient_of_active(a, active)
        if np.allclose(grad, 0.0, atol=1e-13):
            break

        # ----- Adam update ---------------------------------------------------
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)   # descent step

        a_new = a - step                # minimise the max‑entry
        a_new = np.maximum(a_new, 0.0)   # enforce non‑negativity
        a_new = _project_onto_simplex(a_new, target_sum)

        # ----- occasional Gaussian shake ------------------------------------
        if np.random.rand() < 0.02:
            noise = np.random.randn(n) * (0.005 * target_sum / n)
            a_new = _project_onto_simplex(np.maximum(a_new + noise, 0.0), target_sum)
            lr *= 0.94                    # slowly shrink LR after a shake

        a = a_new
        lr *= 0.99995                    # gentle decay

    return best_a, best_score


# --------------------------------------------------------------------------- #
# Main entry point expected by the grader.
# --------------------------------------------------------------------------- #
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a low‑value step‑function within ``budget_s`` seconds.
    Returns a plain Python list of non‑negative heights.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # safety margin

    best_seq = None          # numpy array of the best candidate
    best_score = float("inf")
    best_n = None

    # ------------------------------------------------------------------- #
    # Phase 1 – many short Adam runs on random lengths.
    # ------------------------------------------------------------------- #
    REFINE_TIME = 200.0                     # seconds reserved for Phase 2
    while time.time() < deadline - REFINE_TIME:
        remaining = deadline - time.time()
        # allocate up to 12 s for the next run, but leave REFINE_TIME untouched
        run_time = min(12.0, max(1.0, remaining - REFINE_TIME))

        # choose a length; bias a little towards the best length seen so far
        if best_n is not None and np.random.rand() < 0.3:
            n = int(best_n + np.random.randint(-200, 201))
            n = max(600, min(4000, n))
        else:
            n = np.random.randint(800, 4001)    # 800 … 4000

        # initialise candidate
        if best_seq is not None and best_n == n:
            init = np.array(best_seq, dtype=float)
        elif best_seq is not None and best_n is not None and best_n > n:
            init = np.concatenate([np.array(best_seq[:n], dtype=float),
                                   np.random.rand(n - best_seq[:n].size) * 1e-3])
        else:
            init = np.random.rand(n)

        # run a short Adam optimisation
        cand, cand_score = _run_adam_instance(init, n, run_time)

        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            best_n = n
            print(f"[{time.time():.2f}] New global best = {best_score:.6f} (n={n})")

        # early break if we already beat the target bound
        if best_score < 1.503:
            break

    # ------------------------------------------------------------------- #
    # Phase 2 – simple pairwise‑transfer hill‑climbing on the best vector.
    # ------------------------------------------------------------------- #
    if best_seq is None:
        # fallback: uniform distribution of length 1000
        n = 1000
        target_sum = np.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)
        best_score = float('inf')

    # keep refining until the overall deadline
    while time.time() < deadline:
        a = best_seq.copy()
        n = a.shape[0]
        target_sum = np.sqrt(2.0 * n)

        # pick two distinct indices
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue

        # move a small fraction of mass from i to j
        if a[i] <= 0:
            continue
        frac = np.random.uniform(0.001, 0.01)   # 0.1 %–1 % of a[i]
        delta = frac * a[i]
        a[i] -= delta
        a[j] += delta

        # re‑project onto the simplex to correct any rounding drift
        a = _project_onto_simplex(a, target_sum)

        # compute new maximal convolution entry
        conv = _conv_full_fft(a)
        new_score = float(conv.max())

        if new_score < best_score:
            best_score = new_score
            best_seq = a.copy()
            print(f"[{time.time():.2f}] Refine improved = {best_score:.6f} (n={n})")

        # (optional) simulated‑annealing acceptance of worse moves could be added,
        # but keeping only improvements already yields good progress within the budget.

    # return as plain Python list
    return best_seq.tolist()
