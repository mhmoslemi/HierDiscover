def _optimize_with_smoothing(n, init_a, best_seq, best_score, deadline):
    target_sum = math.sqrt(2.0 * n)
    a = init_a.copy()
    a = a * (target_sum / a.sum())
    m = np.zeros_like(a); v = np.zeros_like(a)
    beta = 0.5
    beta_max = 64.0
    beta_growth_interval = 3000
    iteration = 0
    beta_growth_factor = 1.5
    lr0 = 0.1
    eps = 1e-8
    beta_update_counter = 0
    jitter_interval = 5000
    # main loop
    while time.time() < deadline:
        # convolution
        conv = _full_convolution(a)
        M = conv.max()
        # weights smoothing
        exp_vals = np.exp(beta * (conv - M))
        w = exp_vals / exp_vals.sum()
        # gradient
        w_rev = w[::-1]
        grad_full = _convolve_fft(a, w_rev)
        grad_slice = grad_full[(len(w)-1) : (len(w)-1)+n]
        grad = 2.0 * grad_slice
        # Adam update
        iteration += 1
        m = 0.9 * m + 0.1 * grad
        v = 0.999 * v + 0.001 * (grad * grad)
        m_hat = m / (1 - 0.9 ** iteration)
        v_hat = v / (1 - 0.999 ** iteration)
        lr = lr0 / (1 + 1e-4 * iteration)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        # projection
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 0:
            a = a * (target_sum / s)
        else:
            a = np.full_like(a, target_sum / n)
        # jitter
        if iteration % jitter_interval == 0:
            jitter = np.random.randn(n) * 0.001 * target_sum / n
            a = np.maximum(a + jitter, 0)
            a = a * (target_sum / a.sum())
        # beta growth
        if iteration % beta_growth_interval == 0 and beta < beta_max:

        # break if time near deadline
        if time.time() + 0.01 >= deadline:

"""
Propose a step‑function (non‑negative vector) that minimises the
``evaluate_sequence`` objective.

Algorithm
---------
* Several vector lengths are tried (original, 1.5×, 2×, 3×, 4×,
  capped at 8000).  For every length a few different initial shapes are
  generated (interpolated previous best, uniform random, exponential).
* For each initial vector a gradient descent is performed on a *smoothed*
  version of the max‑convolution objective:
        f_β(a) = Σ_k w_k·(a∗a)[k]   with  w_k ∝ exp(β·((a∗a)[k]−M)),
  where M = max (a∗a) and β starts small (≈0.5) and is multiplied by
  1.5 every few thousand iterations up to β≈64.  When β is large the
  weights concentrate on the active shifts, so the method gradually
  turns into a sub‑gradient descent on the true max.
* Gradients of the smoothed objective are exact and computed by a
  single FFT‑based convolution:
        ∇f_β(a)_i = 2·Σ_j a_j·w_{i+j},
  i.e. a correlation of a with the weight vector w.
* Adam optimiser (β₁=0.9, β₂=0.999) is used, followed by projection onto
  the feasible set: non‑negative entries and Σ a = √(2·n).
* Occasionally a tiny Gaussian jitter is added to escape plateaux.
* All candidates are evaluated (via the known relation
  ``evaluate_sequence = max_convolution`` when the sum is normalised)
  and the best sequence found before the time budget expires is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper FFT‑based convolutions -----------------------------------------

def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full convolution a∗a (real vectors), returned as a 1‑D NumPy array."""
    n = a.shape[0]
    N = 1 << ((2 * n - 1).bit_length())        # next power of two ≥ 2n‑1
    A = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(A * A, n=N)[: 2 * n - 1]
    return conv


def _convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution a∗b via FFT (both real vectors)."""
    n, m = a.shape[0], b.shape[0]
    N = 1 << ((n + m - 1).bit_length())
    A = np.fft.rfft(a, n=N)
    B = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(A * B, n=N)[: n + m - 1]
    return conv


# ----------------------------------------------------------------------
# Gradient of the smoothed max‑convolution ------------------------------

def _gradient_smoothed(a: np.ndarray, beta: float) -> tuple[float, np.ndarray]:
    """
    Returns (M, grad) where
        M   = max(a∗a)
        grad = ∇_a Σ_k w_k·(a∗a)[k]   with w_k ∝ exp(β·( (a∗a)[k]−M ))
    The gradient is exact and computed with two FFTs.
    """
    n = a.shape[0]

    # 1) compute full auto‑convolution and its maximum
    conv = _full_convolution(a)          # length 2n‑1
    M = float(conv.max())

    # 2) softmax weights on the convolution values (numerically stable)
    #    w_k ≥ 0, Σ w_k = 1
    #    Using exp(beta*(conv-M)) guarantees the max gets weight 1.
    exp_vals = np.exp(beta * (conv - M))
    w_sum = exp_vals.sum()
    if w_sum == 0.0:                     # extreme under‑flow – fall back to uniform
        w = np.full_like(conv, 1.0 / conv.shape[0])
    else:
        w = exp_vals / w_sum

    # 3) gradient of the weighted sum:
    #    ∂/∂a_i Σ_k w_k·(a∗a)[k] = 2·Σ_j a_j·w_{i+j}
    #    which is a correlation of a with w.  Implemented as a convolution
    #    with the reversed weight vector.
    w_rev = w[::-1]                      # reverse for correlation
    grad_full = _convolve_fft(a, w_rev)  # length 3n‑2
    start = len(w) - 1                   # = 2n‑2
    grad = 2.0 * grad_full[start:start + n]
    return M, grad


# ----------------------------------------------------------------------
# Adam optimiser on a single length using the smoothed objective --------

def _optimize_length(
    n: int,
    init_a: np.ndarray,
    best_seq: np.ndarray,
    best_score: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """Run Adam optimisation (with a temperature schedule) for a given length."""
    target_sum = math.sqrt(2.0 * n)

    # initialise and enforce the ℓ¹‑normalisation
    a = init_a.copy()
    a = a * (target_sum / a.sum())

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr0 = 0.10               # base learning rate

    # temperature (smoothness) schedule
    beta_temp = 0.5          # start very smooth
    beta_max = 64.0
    beta_growth = 1.5
    beta_growth_interval = 3000

    iteration = 0
    jitter_interval = 5000

    while time.time() < deadline:
        # --------------------------------------------------------------
        # 1) smoothed gradient
        M, grad = _gradient_smoothed(a, beta_temp)

        # --------------------------------------------------------------
        # 2) Adam update
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        lr = lr0 / (1.0 + 1e-4 * iteration)
        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # --------------------------------------------------------------
        # 3) projection onto feasibility set (non‑negative + fixed sum)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a = a * (target_sum / s)
        else:                                   # should never happen
            a = np.full_like(a, target_sum / n)

        # --------------------------------------------------------------
        # 4) occasional jitter to leave shallow plateaus
        if iteration % jitter_interval == 0:
            jitter = np.random.randn(n) * 0.001 * target_sum / n
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        # --------------------------------------------------------------
        # 5) temperature (β) increase
        if iteration % beta_growth_interval == 0 and beta_temp < beta_max:
            beta_temp = min(beta_temp * beta_growth, beta_max)

        # --------------------------------------------------------------
        # 6) evaluate the *real* objective (max‑convolution)
        cur_score = M                     # because Σa = √(2n) ⇒ eval = M
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[len {n:5d} iter {iteration:7d}] new best M = {cur_score:.6f}")

        # --------------------------------------------------------------
        # 7) stop if we are about to exceed the wall‑clock budget
        if time.time() + 0.01 >= deadline:
            break

    return best_seq, best_score


# ----------------------------------------------------------------------
# Initialisation helpers -------------------------------------------------

def _interpolate(base: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of ``base`` onto a grid of size ``new_len``."""
    old_len = base.shape[0]
    old_x = np.linspace(0.0, 1.0, old_len)
    new_x = np.linspace(0.0, 1.0, new_len)
    return np.interp(new_x, old_x, base)


def _random_init(n: int, target_sum: float) -> np.ndarray:
    """Uniform‑random positive vector, ℓ¹‑normalised to ``target_sum``."""
    a = np.random.rand(n)
    return a * (target_sum / a.sum())


def _exp_init(n: int, target_sum: float) -> np.ndarray:
    """Exponential decay shape a_i = exp(-i/τ), τ≈n/5."""
    tau = max(1.0, n / 5.0)
    i = np.arange(n)
    a = np.exp(-i / tau)
    return a * (target_sum / a.sum())


# ----------------------------------------------------------------------
# Main entry point -------------------------------------------------------

def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective within the given time budget.
    Returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the harness
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Load the previously discovered construction if it exists.
    try:
        base = np.array(height_sequence_1, dtype=float)  # type: ignore
    except Exception:
        # fallback: a modest uniform vector
        base = np.full(2100, 1.0, dtype=float)

    n0 = base.shape[0]

    # ------------------------------------------------------------------
    # Global best tracking
    best_seq = None
    best_score = float("inf")

    # Evaluate the starting sequence (it is already normalised later)
    target_sum0 = math.sqrt(2.0 * n0)
    base = base * (target_sum0 / base.sum())
    init_score = evaluate_sequence(base.tolist())
    best_seq = base.copy()
    best_score = init_score
    print(f"[init] length={n0}, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Lengths to explore (original and several enlargements)
    length_factors = [1.0, 1.5, 2.0, 3.0, 4.0]
    lengths = sorted({int(n0 * f) for f in length_factors})
    lengths = [min(l, 8000) for l in lengths if l >= 300]

    # ------------------------------------------------------------------
    # Main optimisation loop over selected lengths
    for n in lengths:
        if time.time() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)

        # ----------------------------------------------------------------
        # Prepare a few different initialisations for this length
        init_candidates = []

        # 1) interpolated version of the current best (or of base if no best)
        if best_seq is not None:
            init_candidates.append(_interpolate(best_seq, n))
        else:
            init_candidates.append(_interpolate(base, n))

        # 2) pure random uniform
        init_candidates.append(_random_init(n, target_sum))

        # 3) exponential decay shape (some extra bias)
        init_candidates.append(_exp_init(n, target_sum))

        print(f"[run] starting optimisation for length = {n} "
              f"({len(init_candidates)} initial candidates)")

        # ----------------------------------------------------------------
        # Run the optimiser on each candidate (time‑limited)
        for init_a in init_candidates:
            if time.time() >= deadline:
                break
            best_seq, best_score = _optimize_length(
                n=n,
                init_a=init_a,
                best_seq=best_seq,
                best_score=best_score,
                deadline=deadline,
            )

    # ------------------------------------------------------------------
    # Final sanity check – fall back to a simple uniform vector if needed.
    if best_seq is None or best_score == float("inf"):
        n = 2100
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n)
        best_score = evaluate_sequence(best_seq.tolist())
        print("[fallback] using uniform sequence")

    print(f"[done] best score = {best_score:.6f}, length = {best_seq.shape[0]}")
    return best_seq.tolist()
