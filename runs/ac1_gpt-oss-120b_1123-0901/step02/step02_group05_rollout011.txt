def weighted_gradient(a, w):
    n = a.shape[0]
    grad = np.zeros_like(a)
    conv_len = 2*n - 1
    for k in range(conv_len):
        wk = w[k]
        if wk == 0:
            continue
        low = max(0, k - (n-1))
        high = min(n-1, k)
        # indices i in [low, high]
        idx = np.arange(low, high + 1)
        partner = k - idx
        # partner also in [0,n-1], as ensured
        # derivative: 2*a[partner] (for each i)

    return grad

    max_beta_conv = np.max(beta * conv)  # to avoid overflow: not needed if we handle stable.

"""
Propose a low‑autocorrelation step‑function.

Algorithm
---------
*  The evaluation objective is      2 n·max(conv(a,a)) / (sum(a)²) .
   It is invariant to uniform scaling, so we fix the L¹‑norm to
   √(2 n) (the “simplex” constraint Σa_i = √(2 n), a_i ≥ 0).

*  Global search:
      – Choose a length n (200…3000, favouring the length of the current best).
      – Initialise from the best known sequence, resized, or from a random Dirichlet draw.
      – Run a time‑bounded Adam optimisation that minimises a *smooth*
         approximation of the max‑convolution:
               f_β(a) = (1/β)·log Σ_k exp(β·(a⋆a)_k)
         The gradient of f_β can be written in closed form using a single
         convolution (see `_softmax_grad`).

      – β is ramped from a small value (β≈0.5) to a large one (β≈20) so that the
         optimisation first follows a smoothed landscape and then focuses on the
         true maximum.

      – After each Adam step we project back onto the simplex (Euclidean projection)
         to keep the sum fixed.

      – The true objective (via `evaluate_sequence`) is sampled every few iterations;
         the best sequence seen overall is kept.

*  Occasionally an LP‐refinement step (the “LP‑update” from the paper) is performed
  when `cvxpy` is available and the problem size is modest (n≤800).

*  The whole process continues until the time budget (≈1000 s) is exhausted,
  always returning the best sequence discovered.

The combination of a smoothed max‑convolution, adaptive Adam steps,
simplex projection and occasional LP‑updates yields a more robust
exploration of the non‑convex landscape and typically beats the
previous 1.523‑bound, aiming for the target 1.503.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simple Euclidean projection onto the simplex {x ≥ 0, Σ x = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex of non‑negative vectors with sum s."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Small mutation helper (Gaussian noise + simplex projection)
# ----------------------------------------------------------------------
def _mutate(a: np.ndarray, target_sum: float, sigma: float) -> np.ndarray:
    """Return a mutated copy of a (Gaussian noise, then simplex projection)."""
    b = a + np.random.randn(a.size) * sigma
    b = np.maximum(b, 0.0)
    b = _project_onto_simplex(b, target_sum)
    return b


# ----------------------------------------------------------------------
# Resize helper – linear interpolation / down‑sampling
# ----------------------------------------------------------------------
def _resize_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize seq to length new_n preserving non‑negativity (linear interpolation)."""
    old_n = seq.size
    if old_n == new_n:
        return seq.copy()
    if old_n > new_n:
        idx = np.linspace(0, old_n - 1, new_n).astype(int)
        return seq[idx]
    else:
        x_old = np.arange(old_n)
        x_new = np.linspace(0, old_n - 1, new_n)
        return np.interp(x_new, x_old, seq)


# ----------------------------------------------------------------------
# Soft‑max weighted gradient of the smoothed max‑convolution
# ----------------------------------------------------------------------
def _softmax_grad(a: np.ndarray, beta: float) -> np.ndarray:
    """
    Compute ∇_a  (1/β)·log Σ_k exp(β·(a⋆a)_k) .
    Returns a gradient vector of shape a.shape.
    """
    n = a.shape[0]
    conv = np.convolve(a, a)                     # length 2n‑1
    # soft‑max weights (stable)
    scaled = beta * conv
    c = np.max(scaled)
    exp_scaled = np.exp(scaled - c)
    w = exp_scaled / exp_scaled.sum()           # weights sum to 1

    # weighted gradient:
    #   grad_i = 2 * Σ_j a_j * w_{i+j}
    # This can be obtained by a single convolution of reversed a with w.
    a_rev = a[::-1]
    conv_rev = np.convolve(a_rev, w)            # length 3n‑2
    grad = 2.0 * conv_rev[n - 1 : 2 * n - 1]     # slice of length n
    return grad


# ----------------------------------------------------------------------
# Adam optimiser with β‑ramp and simplex projection
# ----------------------------------------------------------------------
def _adam_optimize(
    init: np.ndarray,
    target_sum: float,
    time_limit: float,
    beta_start: float = 0.5,
    beta_end: float = 20.0,
    lr: float = 0.05,
    eval_every: int = 20,
    print_tag: str = "",
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smoothed max‑convolution.
    Returns the best sequence encountered and its true objective value.
    """
    a = _project_onto_simplex(init.astype(float), target_sum)

    # initial evaluation
    try:
        best_score = float(evaluate_sequence(a.tolist()))
    except Exception:
        best_score = math.inf
    best_seq = a.copy()

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    it = 0
    start = time.time()
    while time.time() < start + time_limit:
        elapsed = time.time() - start
        frac = elapsed / time_limit
        beta = beta_start + (beta_end - beta_start) * frac

        # gradient of smoothed max‑convolution
        grad = _softmax_grad(a, beta)

        # Adam moments
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        it += 1
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = _project_onto_simplex(a, target_sum)

        # occasional evaluation of the true objective
        if it % eval_every == 0:
            try:
                score = float(evaluate_sequence(a.tolist()))
            except Exception:
                score = math.inf
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(
                    f"[{time.time():.2f}] {print_tag} new local best = {score:.6f} (n={len(a)})"
                )

        # occasional random shake (helps escape shallow plateaus)
        if np.random.rand() < 0.005:
            sigma = 0.02 * target_sum / len(a)
            a = _mutate(a, target_sum, sigma)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Optional LP refinement (the “LP‑update” from the paper)
# ----------------------------------------------------------------------
try:
    import cvxpy as cp

    _CVX_AVAILABLE = True
except Exception:  # pragma: no cover
    _CVX_AVAILABLE = False


def _solve_lp_for_g(f: np.ndarray, M: float):
    """Solve   max Σ g_j   s.t.   g≥0,   f⋆g ≤ M .   Returns the optimal g or None."""
    if not _CVX_AVAILABLE:
        return None
    n = f.shape[0]
    g = cp.Variable(n, nonneg=True)
    constraints = []
    # Build each convolution inequality (linear in g)
    for k in range(2 * n - 1):
        low = max(0, k - (n - 1))
        high = min(n - 1, k)
        idx = np.arange(low, high + 1, dtype=int)
        partner = k - idx
        expr = cp.sum(cp.multiply(f[low : high + 1], g[partner]))
        constraints.append(expr <= M)
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.GLPK, warm_start=True, verbose=False, glpkk=1e-08)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, verbose=False, max_iters=2000)
        except Exception:
            return None
    if prob.status not in ("optimal", "optimal_inaccurate"):
        return None
    g_val = np.asarray(g.value).flatten()
    if g_val is None:
        return None
    return np.maximum(g_val, 0.0)


def _lp_refine(
    init: np.ndarray,
    target_sum: float,
    time_limit: float,
    mix: float = 0.2,
    max_iter: int = 6,
) -> tuple[np.ndarray, float]:
    """Perform a few LP‑updates starting from ``init``."""
    a = _project_onto_simplex(init.astype(float), target_sum)
    try:
        best_score = float(evaluate_sequence(a.tolist()))
    except Exception:
        best_score = math.inf
    best_seq = a.copy()

    start = time.time()
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        conv = np.convolve(a, a)
        M = float(conv.max())
        g0 = _solve_lp_for_g(a, M)
        if g0 is None or np.sum(g0) == 0:
            break
        g0 = _project_onto_simplex(g0, target_sum)  # ensure feasibility
        a = (1.0 - mix) * a + mix * g0
        a = _project_onto_simplex(a, target_sum)
        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            score = math.inf
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(
                f"[{time.time():.2f}] LP‑refine iter {it+1} new best = {score:.6f} (n={len(a)})"
            )
    return best_seq, best_score
