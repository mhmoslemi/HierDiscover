def adam_optimize(a_init, target_sum, deadline, n):
    a = a_init.copy()
    # ensure sum normalization
    a = a * (target_sum / a.sum())
    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    iteration = 0

      – non‑negative,
      – forced to be (approximately) symmetric (a_i = a_{n‑1‑i}),
      – renormalised so that ∑a = √(2 n) (this makes the objective equal to the
        maximal convolution value).
* Occasionally a tiny symmetric Gaussian jitter is added to escape plateaus.
* The best sequence (lowest maximal convolution) found over all runs and
  lengths is returned.

The routine respects a wall‑clock budget (default 1000 s) and prints progress
information.  No external files or network access are used.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _conv_max_fft(a: np.ndarray):
    """
    Compute the full self‑convolution of a (real, zero‑padded) using FFT.
    Returns (M, conv_full) where M is the maximal entry.
    """
    n = a.shape[0]
    # Next power of two ≥ 2·n‑1  (FFT zero‑padding)
    N = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=N)
    B = A * A
    conv = np.fft.irfft(B, n=N)
    conv = conv[: 2 * n - 1]          # keep only the valid part
    M = float(conv.max())
    return M, conv


# ----------------------------------------------------------------------
def _subgradient(a: np.ndarray, active_idxs: np.ndarray):
    """
    Average reversed vector a_{k‑i} over all active shift indices.
    This yields a (sub‑)gradient of the maximal convolution term.
    """
    n = a.shape[0]
    idx = np.arange(n)
    rev = np.zeros_like(a)
    for k in active_idxs:
        j = k - idx
        mask = (j >= 0) & (j < n)
        rev[mask] += a[j[mask]]
    if active_idxs.size > 0:
        rev /= float(active_idxs.size)
    return rev


# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    target_sum: float,
    n: int,
    deadline: float,
    verbose: bool = True,
):
    """
    Run a projected Adam optimiser on a single length `n` starting from `a0`.
    The optimisation respects the non‑negativity and sum constraints and
    enforces (approximate) symmetry after each update.

    Returns the best vector found and its maximal convolution value.
    """
    a = a0.copy()
    # initial normalisation (sum must be target_sum)
    a = a * (target_sum / a.sum())

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.10                     # relatively aggressive start
    iteration = 0
    jitter_interval = 3000

    best_M = float("inf")
    best_a = a.copy()

    # constant part of gradient (derivative of the denominator)
    const_factor = 4.0 * n / (target_sum ** 3)

    while time.time() < deadline:
        # --------------------------------------------------------------
        # 1) full convolution and active set
        M, conv_full = _conv_max_fft(a)
        tol = 1e-12
        active = np.where(conv_full >= M - tol)[0]

        # --------------------------------------------------------------
        # 2) sub‑gradient of the objective (see derivation in the docstring)
        rev = _subgradient(a, active)
        coeff = 4.0 * n / (target_sum ** 2)
        grad = coeff * rev - const_factor * M   # same as in the reference code

        # --------------------------------------------------------------
        # 3) Adam update
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        # simple learning‑rate decay
        lr = lr0 / (1.0 + 1e-5 * iteration)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # --------------------------------------------------------------
        # 4) projection onto feasible set
        a = np.maximum(a, 0.0)                # non‑negativity
        a = (a + a[::-1]) * 0.5               # enforce (approximate) symmetry
        s = a.sum()
        if s > 0.0:
            a = a * (target_sum / s)          # renormalise sum
        else:                                 # fallback – should never happen
            a = np.full_like(a, target_sum / n)

        # --------------------------------------------------------------
        # 5) occasional symmetric jitter
        if iteration % jitter_interval == 0:
            jitter = np.random.randn(n) * 0.001 * target_sum / n
            a = np.maximum(a + jitter, 0.0)
            a = (a + a[::-1]) * 0.5
            a = a * (target_sum / a.sum())

        # --------------------------------------------------------------
        # 6) keep track of the best (lowest maximal convolution) seen
        if M < best_M:
            best_M = M
            best_a = a.copy()
            if verbose:
                print(
                    f"[len {n:5d} iter {iteration:7d}] new local best M = {best_M:.6f}"
                )

        # --------------------------------------------------------------
        # optional early stop when improvement stalls – omitted for simplicity
    # Ensure the final best vector respects symmetry and normalisation
    best_a = np.maximum(best_a, 0.0)
    best_a = (best_a + best_a[::-1]) * 0.5
    best_a = best_a * (target_sum / best_a.sum())
    best_M, _ = _conv_max_fft(best_a)
    return best_a, best_M

def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative height sequence that minimizes the
    ``evaluate_sequence`` objective.  The function returns the best sequence
    (as a plain Python list) before the time budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0      # a small safety margin

    # otherwise fall back to a modest uniform vector.

    base_len = base.shape[0]

    # --------------------------------------------------------------

    best_score = float("inf")

    # Evaluate the supplied initial construction (if any)
    S0 = math.sqrt(2.0 * base_len)
    base = base * (S0 / base.sum())
    init_score = evaluate_sequence(base.tolist())
    best_seq = base.copy()
    best_score = init_score
    print(f"[init] length={base_len}, initial score={best_score:.6f}")

    # --------------------------------------------------------------
    # Choose a set of lengths (original plus several larger ones)
    length_factors = [1.0, 1.33, 1.66, 2.0]
    lengths = sorted({int(base_len * f) for f in length_factors})
    lengths = [l for l in lengths if 300 <= l <= 9600]   # sanity caps
    print("[info] lengths to explore:", lengths)

    # --------------------------------------------------------------
    # Main optimisation loop over the selected lengths
    for n in lengths:
        if time.time() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)

        # Rough estimate of time needed for a single run on this size
        est_run = max(8.0, 15.0 * (n / 3000.0))

        # Allocate a small number of restarts depending on remaining time
        remaining = deadline - time.time()
        possible_runs = max(1, int(remaining // (est_run * 2)))
        runs = min(possible_runs, 5)          # never explode the number of runs
        print(f"[len {n}] planning {runs} runs (≈{est_run:.1f}s each)")

        for run_idx in range(runs):
            if time.time() >= deadline:
                break

            # ----------------------------------------------------------
            # Initial vector for this run
            if n == base_len:
                a0 = base.copy()
                # add a tiny symmetric jitter to diversify the start
                jitter = np.random.randn(n) * 0.001 * target_sum / n
                a0 = np.maximum(a0 + jitter, 0.0)
                a0 = a0 * (target_sum / a0.sum())
            else:
                # --- option A: interpolated base shape ---
                old_x = np.linspace(0.0, 1.0, base_len)
                new_x = np.linspace(0.0, 1.0, n)
                a0 = np.interp(new_x, old_x, base)
                a0 = np.maximum(a0, 0.0)

                # add a moderate symmetric noise term
                noise = np.random.randn(n) * 0.005 * target_sum / n
                a0 = np.maximum(a0 + noise, 0.0)
                a0 = a0 * (target_sum / a0.sum())

                # --- with probability 0.3 use a fresh Dirichlet start ---
                if np.random.rand() < 0.30:
                    rand_vals = np.random.exponential(scale=1.0, size=n)
                    a0 = rand_vals
                    a0 = a0 * (target_sum / a0.sum())

            # ----------------------------------------------------------
            # Run the optimiser for this restart
            run_deadline = min(deadline, time.time() + est_run)
            a_opt, M_opt = _adam_optimize(a0, target_sum, n, run_deadline, verbose=False)

            # ----------------------------------------------------------
            # Record if this run improves the global best
            if M_opt < best_score:
                best_score = M_opt
                best_seq = a_opt.copy()
                print(
                    f"[global best] length {n}, run {run_idx}, "
                    f"new best max‑conv = {best_score:.6f}"
                )

    # --------------------------------------------------------------
    # Fallback in the unlikely case that nothing was found
    if best_seq is None:
        n = base_len
        S = math.sqrt(2.0 * n)
        best_seq = np.full(n, S / n)
        best_score = evaluate_sequence(best_seq.tolist())
        print("[fallback] using uniform sequence")

    print(f"[done] best score = {best_score:.6f}, length = {best_seq.shape[0]}")
    return best_seq.tolist()
