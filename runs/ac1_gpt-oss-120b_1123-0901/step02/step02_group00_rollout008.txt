Pseudo:

def smooth_max(a, beta):
    conv = conv_full_fft(a)
    # compute exp term; careful with overflow: use max subtraction.
    max_val = conv.max()
    # stable: exp(beta*(conv - max_val)), sumExp = sumexp, M_s = max_val + log(sumExp)/beta.
    exp_terms = np.exp(beta * (conv - max_val))
    sum_exp = exp_terms.sum()
    smooth_val = max_val + np.log(sum_exp) / beta
    # weights:
    w = exp_terms / sum_exp
    # gradient = 2 * conv_of_a_and_w_rev slice
    rev_w = w[::-1]  # length 2n-1
    # conv of a and rev_w:
    # using FFT: compute FFT of a with size N >= n + (2n-1) - 1 = 3n-2 next power of two.
    N = 1 << ((n + (2*n-1) - 1).bit_length())
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(rev_w, N)
    grad_full = np.fft.irfft(A * W, N)  # convolution a * rev_w
    # Extract gradient entries for j from 0 to n-1: indices (2n-2 - j)
    # compute p = 2*n-2 - np.arange(n)
    idx = (2*n - 2) - np.arange(n)
    g = 2.0 * grad_full[idx]
    return smooth_val, g

"""
Step‑function optimizer (smooth max + variance regularisation).

The evaluator is scale‑invariant, so we fix Σ a_i = √(2n) and minimise
the (scaled) maximal auto‑convolution value.  The objective
max conv(a,a) is non‑smooth; we replace it by a temperature‑controlled
soft‑max (β) and optionally add a variance penalty (λ·Var(conv)).  The
gradient of both terms can be written as a convolution, which we compute
with FFTs (O(n log n)).  An Adam optimiser with projection onto the
non‑negative simplex drives the search.  β is increased gradually (annealing)
while λ is decreased, so the algorithm first flattens the convolution
profile and then concentrates on the true maximum.

Multiple lengths (original and a few enlarged versions) are tried, each
with a few random restarts.  The best sequence found before the time
budget expires is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Small helper: next power‑of‑two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a * a using FFT.
    Returns an array of length 2·n‑1 (real values).
    """
    n = a.shape[0]
    N = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(A * A, n=N)
    return conv[: 2 * n - 1]


def _conv_fft(a: np.ndarray, v: np.ndarray, N: int) -> np.ndarray:
    """
    Full convolution of a (length n) with v (length m) using FFT.
    N must be a power‑of‑two ≥ n+m‑1.
    Returns the full convolution (length n+m‑1) as a real array.
    """
    A = np.fft.rfft(a, n=N)
    B = np.fft.rfft(v, n=N)
    full = np.fft.irfft(A * B, n=N)
    return full[: a.shape[0] + v.shape[0] - 1]


def _smooth_max_and_grad(a: np.ndarray,
                         beta: float,
                         lam: float,
                         target_sum: float) -> tuple[float, np.ndarray, float]:
    """
    Compute (smooth_max, gradient, actual_max) for a given vector a.

    * smooth_max  –  soft‑max approximation of max(conv) with temperature beta.
    * gradient    –  sub‑gradient of (smooth_max + lam·Var(conv)).
    * actual_max  –  true max(conv) (used for bookkeeping).

    The gradient uses FFT‑based convolutions and works for any length.
    """
    n = a.shape[0]
    L = 2 * n - 1                     # length of the full convolution
    # ----- FFT size for convolutions (needs space for a * v where v length = L) -----
    N = _next_pow2(n + L - 1)         # ≥ 3·n‑2

    # ----- convolution a * a (full) -----
    A_fft = np.fft.rfft(a, n=N)
    conv_full = np.fft.irfft(A_fft * A_fft, n=N)[:L]   # length L
    actual_max = float(conv_full.max())

    # ----- smooth max (soft‑max) -----
    # For numerical stability subtract the true max
    shift = actual_max
    exp_terms = np.exp(beta * (conv_full - shift))
    sum_exp = exp_terms.sum()
    smooth_val = shift + np.log(sum_exp) / beta
    w = exp_terms / sum_exp                     # soft‑max weights, sum to 1

    # ----- gradient of the smooth max -----
    rev_w = w[::-1]                               # length L
    grad_full_w = _conv_fft(a, rev_w, N)          # length n+L-1
    idx = (L - 1) - np.arange(n)                  # indices that map to grad components
    grad_smooth = 2.0 * grad_full_w[idx]          # factor 2 from d(conv_i)/da_j = 2 a_{i‑j}

    # ----- variance regularisation (optional) -----
    if lam > 0.0:
        mu = conv_full.mean()
        delta = conv_full - mu                     # length L, zero‑mean
        rev_delta = delta[::-1]
        grad_full_delta = _conv_fft(a, rev_delta, N)
        grad_var = (4.0 / L) * grad_full_delta[idx]   # derived in the analysis
    else:
        grad_var = np.zeros_like(a)

    # ----- total gradient -----
    grad = grad_smooth + lam * grad_var
    return smooth_val, grad, actual_max


def _project_onto_simplex(a: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Project a onto the feasible set {x ≥ 0, Σx = target_sum}.
    This is a simple scaling after truncating negatives.
    """
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 0.0:
        a = a * (target_sum / s)
    else:
        a = np.full_like(a, target_sum / a.shape[0])
    return a


def _symmetrize(a: np.ndarray) -> np.ndarray:
    """Force mirror symmetry a_i = a_{n‑1‑i}."""
    return (a + a[::-1]) * 0.5


def _optimize_length_smooth(n: int,
                            init_a: np.ndarray,
                            best_seq: np.ndarray | None,
                            best_score: float,
                            deadline: float,
                            *,
                            seed: int = 0) -> tuple[np.ndarray, float]:
    """
    Adam optimiser for a fixed length `n` using smooth‑max + variance term.
    Returns the (possibly updated) global best sequence and its score.
    """
    np.random.seed(seed)
    target_sum = math.sqrt(2.0 * n)

    # start from a well‑scaled version of init_a
    a = init_a.astype(np.float64)
    a = _project_onto_simplex(a, target_sum)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.07                     # a slightly larger base learning rate
    iteration = 0

    # annealing schedules
    beta = 0.5                     # start temperature (soft max)
    beta_step = 2500               # increase β every this many iterations
    beta_multiplier = 1.6
    max_beta = 32.0

    lam = 8.0                      # start variance weight
    lam_step = 2500
    lam_decay = 0.85               # multiply λ every lam_step iterations

    jitter_interval = 4000
    jitter_scale = 0.001           # relative to target_sum / n

    while time.time() < deadline:
        iteration += 1

        # ----- evaluate objective and gradient -----
        smooth_val, grad, cur_max = _smooth_max_and_grad(
            a, beta=beta, lam=lam, target_sum=target_sum)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        # simple decay of the learning rate
        lr = lr0 / (1.0 + 1e-4 * iteration)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # ----- enforce symmetry (optional but helps) -----
        a = _symmetrize(a)

        # ----- feasibility projection -----
        a = _project_onto_simplex(a, target_sum)

        # ----- occasional jitter to escape plateaus -----
        if iteration % jitter_interval == 0:
            jitter = np.random.randn(n) * jitter_scale * target_sum / n
            a = _project_onto_simplex(a + jitter, target_sum)

        # ----- keep global best (using true max, not smoothed) -----
        if cur_max < best_score:
            best_score = cur_max
            best_seq = a.copy()
            print(f"[len {n:5d} iter {iteration:7d}] new best = {best_score:.6f}")

        # ----- annealing schedules -----
        if iteration % beta_step == 0:
            beta = min(max_beta, beta * beta_multiplier)
        if iteration % lam_step == 0:
            lam = lam * lam_decay   # bring variance weight down gradually

        # stop condition is only the global time limit
    return best_seq, best_score
