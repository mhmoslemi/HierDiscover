def _optimize_sequence(n, init, deadline):
    target_sum = np.sqrt(2.0 * n)
    a = init.astype(np.float64)
    a = _project_onto_simplex(a, target_sum)

    best_local = a.copy()
    conv = _conv_fft_self(a)
    max_val = conv.max()
    best_score = 2.0 * n * max_val / (a.sum() * a.sum())

    v = np.zeros_like(a)
    momentum = 0.9
    base_step = 0.1
    beta = 0.5
    beta_max = 150.0
    beta_factor = 1.02 # increase each iteration

    iteration = 0

    while time.time() < deadline:
        # Compute conv and weights
        conv = _conv_fft_self(a)
        max_val = conv.max()

        # Determine beta (increase gradually)
        if beta < beta_max:
            beta = min(beta_max, beta * beta_factor)

        # compute softmax weights
        # subtract max to avoid overflow
        exp_vals = np.exp(beta * (conv - max_val))
        w = exp_vals / exp_vals.sum()

        # compute gradient using w
        # grad = 2 * correlation(w, a)
        # Use convolution: conv_full = np.convolve(w[::-1], a); then grad_i = 2 * conv_full[(len(w)-1) - i]
        conv_full = _conv_full_fft(w[::-1], a)
        # conv_full length = len(w)+len(a)-1 = 3n-2
        idx = (len(w)-1) - np.arange(n)
        grad = 2.0 * conv_full[idx]

        # momentum
        v = momentum * v + (1 - momentum) * grad

        # step size: could use scaling similar to previous: step = base_step * max_val / np.max(np.abs(v)) (if >0)
        max_abs_v = np.max(np.abs(v))
        if max_abs_v == 0:
            eta = 0.0
        else:
            eta = base_step * max_val / max_abs_v

        a_new = a - eta * v

        # project onto simplex
        a_new = _project_onto_simplex(a_new, target_sum)

        # random jitter occasional
        if np.random.rand() < 0.01:
            noise = np.random.randn(n) * (target_sum * 0.001 / n)
            a_new = _project_onto_simplex(np.maximum(a_new + noise, 0.0), target_sum)
            base_step *= 0.95

        # evaluate new score
        conv_new = _conv_fft_self(a_new)
        max_val_new = conv_new.max()
        score_new = 2.0 * n * max_val_new / (a_new.sum() ** 2)

        if score_new < best_score:
            best_score = score_new
            best_local = a_new.copy()
            print(f"[{time.time():.2f}] n={n}, new local best={best_score:.6f}, beta={beta:.2f}")

        a = a_new
        iteration += 1
        # optionally decay base_step slowly
        base_step *= 0.9999

        # maybe break if improvement < tolerance for many iterations
        # We'll just continue until deadline.

    return best_local, best_score

def _resample_sequence(seq, new_n):
    old_n = len(seq)
    if new_n == old_n:
        return seq.copy()
    # Interpolation on index normalized to [0,1]
    x_old = np.linspace(0, 1, old_n)

    new_seq = np.interp(x_new, x_old, seq)

"""
Propose a step‑function (non‑negative vector) with a low value of

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

Algorithm
---------

1. **Search space** – choose a length `n` (typically 2000‑8000).  Larger
   vectors give a better approximation to the continuous optimum
   (≈ π/2), but we keep FFT sizes modest.

2. **Initialisation** – start either from a power‑law shape
   `a_i ∝ (i+1)^{-p}` with a random exponent `p≈0.5`, or by
   resampling the best vector found so far.

3. **Smoothing‑gradient descent** – we minimise the *maximum*
   autocorrelation by minimising a smooth approximation:

       L_β(a) = (1/β)·log ∑_k exp(β·conv_k(a))

   For a given temperature `β` the gradient of `L_β` is
   `∇L = 2 · corr(w, a)`, where `w_k ∝ exp(β·conv_k)`.  The gradient
   is obtained with a single FFT‑based convolution.

   An Adam‑style update (with momentum and per‑coordinate adaptivity)
   is performed, after which the vector is projected onto the simplex
   `{x_i≥0, Σx_i = √(2 n)}` – this keeps the objective invariant.

4. **Annealing** – `β` is increased geometrically (starting from 0.5
   up to ≈ 150).  Early phases move the vector quickly; later phases
   focus on the true max‑convolution.

5. **Refinement** – when time permits we run a few seconds of the
   earlier active‑set gradient descent (the “previous program”) on the
   current best vector for a final fine‑tuning.

All heavy work (convolutions, gradients) uses real FFTs, so even
thousands of iterations on vectors of a few thousand entries fit easily
into the 1000 s budget.

The routine keeps the overall best sequence seen and returns it as a
plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper 1 – Euclidean projection onto the simplex {x_i ≥ 0, Σ x_i = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Return the Euclidean projection of v onto the simplex of radius s."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper 2 – Full linear convolution via FFT (real inputs → real output)
# ----------------------------------------------------------------------
def _conv_full_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using an FFT of power‑of‑2 size."""
    n = x.shape[0]
    m = y.shape[0]
    size = 1 << ((n + m - 2).bit_length())  # next power of two ≥ n+m-1
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[: n + m - 1]
    # numerical noise may produce tiny negatives – clip
    np.maximum(conv, 0.0, out=conv)
    return conv


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full convolution a * a using an FFT of power‑of‑2 size."""
    return _conv_full_fft(a, a)


# ----------------------------------------------------------------------
# Helper 3 – Resample (linear interpolation) a sequence to a new length
# ----------------------------------------------------------------------
def _resample_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_len)
    x_new = np.linspace(0.0, 1.0, new_len)
    return np.interp(x_new, x_old, seq)


# ----------------------------------------------------------------------
# Helper 4 – Gradient of the soft‑max smoothed max‑convolution
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Compute ∇ L_β(a) where
        L_β = (1/β)·log Σ_k exp(β·conv_k)
    The gradient is 2·corr(w, a) with w_k ∝ exp(β·conv_k).
    """
    n = a.shape[0]
    max_val = conv.max()
    # soft‑max weights (stable)
    exp_vals = np.exp(beta * (conv - max_val))
    w = exp_vals / exp_vals.sum()

    # correlation w ⋆ a  (w shifted against a)
    # implement via convolution: corr_i = Σ_k w_k a_{k-i}
    # This equals 2·((w[::-1] * a))[len(w)-1 - i] after linear conv.
    conv_full = _conv_full_fft(w[::-1], a)          # length = 3n‑2
    idx = (len(w) - 1) - np.arange(n)               # indices of interest
    grad = 2.0 * conv_full[idx]
    return grad


# ----------------------------------------------------------------------
# Core optimiser – Adam with soft‑max smoothing of the max‑convolution
# ----------------------------------------------------------------------
def _adam_smooth_optimize(
    n: int,
    init: np.ndarray,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smoothed max‑convolution objective.
    Returns the best vector found and its (unsmoothed) score.
    """
    # target sum = √(2 n); scaling does not affect the objective.
    target_sum = np.sqrt(2.0 * n)
    a = _project_onto_simplex(init.astype(np.float64), target_sum)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.05                     # base learning rate
    m = np.zeros_like(a)         # first moment
    v = np.zeros_like(a)         # second moment

    # annealing schedule for β
    beta = 0.5
    beta_max = 150.0
    beta_mult = 1.015

    best_a = a.copy()
    conv = _conv_self_fft(a)
    best_score = 2.0 * n * conv.max() / (a.sum() ** 2)

    iteration = 0
    while time.time() < deadline:
        iteration += 1

        # 1) current convolution and gradient (soft‑max)
        conv = _conv_self_fft(a)
        grad = _softmax_gradient(a, conv, beta)

        # 2) Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = _project_onto_simplex(a_new, target_sum)

        # 3) occasional jitter to escape plateaus
        if np.random.rand() < 0.01:
            noise = np.random.randn(n) * (target_sum * 0.0005 / n)
            a_new = _project_onto_simplex(np.maximum(a_new + noise, 0.0), target_sum)
            lr *= 0.95  # be a little more cautious after a shake‑up

        # 4) evaluate true (unsmoothed) objective
        conv_new = _conv_self_fft(a_new)
        score = 2.0 * n * conv_new.max() / (a_new.sum() ** 2)

        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            print(
                f"[{time.time():.2f}] n={n}  smooth‑Adam local best = {score:.6f}   β={beta:.2f}"
            )

        # 5) prepare for next iteration
        a = a_new
        if beta < beta_max:
            beta = min(beta_max, beta * beta_mult)

        # tiny learning‑rate decay
        lr *= 0.99995

    return best_a, best_score


# ----------------------------------------------------------------------
# Helper – small active‑set optimisation (the previous method)
# ----------------------------------------------------------------------
def _active_set_optimize(
    n: int,
    init: np.ndarray,
    deadline: float,
    *,
    rel_tol: float = 5e-4,
    base_step: float = 0.12,
    momentum_beta: float = 0.9,
) -> tuple[np.ndarray, float]:
    """
    Gradient descent on the *exact* max‑convolution objective, using the
    active‑set of near‑maximal entries (identical to the code provided
    in the prompt).  This serves as a fine‑tuning step.
    """
    target_sum = np.sqrt(2.0 * n)          # Σ a_i = √(2 n)
    a = _project_onto_simplex(init.astype(np.float64), target_sum)

    conv = _conv_self_fft(a)
    max_val = conv.max()
    best_local = a.copy()
    best_score = 2.0 * n * max_val / (a.sum() ** 2)

    velocity = np.zeros_like(a)
    step_factor = base_step

    while time.time() < deadline:
        conv = _conv_self_fft(a)
        max_val, active = _active_indices(conv, rel_tol)

        # gradient of sum of active entries
        grad = _gradient_active_set(a, active)

        # momentum
        velocity = momentum_beta * velocity + (1.0 - momentum_beta) * grad
        max_grad = np.max(velocity) if np.max(velocity) > 0 else 1.0
        eta = step_factor * max_val / max_grad

        a_new = a - eta * velocity
        a_new = _project_onto_simplex(a_new, target_sum)

        # occasional shake‑up
        if np.random.rand() < 0.02:
            noise = np.random.randn(n) * (target_sum * 0.001 / n)
            a_new = _project_onto_simplex(np.maximum(a_new + noise, 0.0), target_sum)
            step_factor *= 0.92

        conv_new = _conv_self_fft(a_new)
        max_val_new = conv_new.max()
        score = 2.0 * n * max_val_new / (a_new.sum() ** 2)

        if score < best_score:
            best_score = score
            best_local = a_new.copy()
            print(f"[{time.time():.2f}] n={n}  active‑set local best = {score:.6f}")

        a = a_new
        step_factor *= 0.9997

    return best_local, best_score


def _active_indices(conv: np.ndarray, rel_tol: float = 5e-4) -> tuple[float, np.ndarray]:
    """Return (max_value, indices) where conv ≥ max – rel_tol·max."""
    max_val = conv.max()
    tol = max(1e-12, max_val * rel_tol)
    active = np.where(conv >= max_val - tol)[0]
    return max_val, active


def _gradient_active_set(a: np.ndarray, active: np.ndarray) -> np.ndarray:
    """Gradient of Σ_{k∈active} conv_k w.r.t. a (same as in the prompt)."""
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in active:
        low = max(0, k - (n - 1))
        high = min(n - 1, k)
        idx = np.arange(low, high + 1, dtype=int)
        partner = k - idx
        grad[idx] += 2.0 * a[partner]
    return grad

# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative height sequence that minimises the evaluation
    function.  The routine runs for at most `budget_s` seconds and returns
    the best sequence discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.3   # small safety margin

    best_seq = None
    best_score = float("inf")

    # admissible lengths – we explore a fairly wide range
    min_n, max_n = 1800, 8000

    # ------------------------------------------------------------------
    # Main restart loop – each restart gets a time slice.
    # ------------------------------------------------------------------
    while time.time() < deadline:
        remaining = deadline - time.time()
        # adapt length choice to remaining time (longer runs for larger n)
        if remaining > 200:
            n = np.random.randint(min_n, max_n + 1)
            run_time = min(remaining * 0.55, 20.0)
        elif remaining > 60:
            n = np.random.randint(1200, 5000)
            run_time = min(remaining * 0.6, 12.0)
        else:
            n = np.random.randint(800, 2500)
            run_time = remaining - 0.2   # leave a tiny buffer

        # --------------------------------------------------------------
        # Initial vector
        # --------------------------------------------------------------
        if best_seq is not None and np.random.rand() < 0.35:
            # warm‑start from the current best (resample or truncate)
            init = _resample_sequence(best_seq, n)
        else:
            # power‑law start, p ≈ 0.5 with a little randomisation
            p = np.random.uniform(0.45, 0.55)
            idx = np.arange(1, n + 1, dtype=np.float64)
            init = idx ** (-p)
            init += np.random.rand(n) * 1e-5  # tiny jitter

        # --------------------------------------------------------------
        # Allocate a time slice for this run
        # --------------------------------------------------------------
        run_deadline = time.time() + run_time

        # --------------------------------------------------------------
        # Phase 1 – Adam with smooth max (most of the allocated time)
        # --------------------------------------------------------------
        seq_local, score_local = _adam_smooth_optimize(n, init, run_deadline)

        # --------------------------------------------------------------
        # Phase 2 – (optional) short active‑set fine‑tuning if time left
        # --------------------------------------------------------------
        leftover = run_deadline - time.time()
        if leftover > 2.0:

                seq_local,
                time.time() + leftover,

            )

        # --------------------------------------------------------------
        # Update global best
        # --------------------------------------------------------------
        if score_local < best_score:
            best_score = score_local
            best_seq = seq_local.copy()
            print(
                f"[{time.time():.2f}] GLOBAL new best = {best_score:.6f} (length={n})"
            )

        # early exit if we have comfortably passed the target
        if best_score < 1.45:   # well below the 1.503 target
            break

    # ------------------------------------------------------------------
    # Final fine‑tuning on the overall best (time permitting)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 0.5:
        n = best_seq.size
        final_deadline = deadline
        best_seq, best_score = _active_set_optimize(
            n,
            best_seq,
            final_deadline,
            rel_tol=5e-5,   # tighter tolerance for final polish
            base_step=0.04,
            momentum_beta=0.9,
        )
        print(f"[{time.time():.2f}] FINAL polished best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback – if nothing was found (extremely unlikely)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        target_sum = np.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=np.float64)

    # Return a plain Python list as required by the evaluator.
    return best_seq.tolist()
