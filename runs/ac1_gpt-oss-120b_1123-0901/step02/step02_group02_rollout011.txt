"""
Smoothing–gradient descent optimisation for the max‑convolution objective.
The algorithm fixes the ℓ₁‑norm Σa_i = √(2n) so that the evaluation
function reduces to max(conv(a,a)).  It minimises a smooth approximation
of the max (log‑sum‑exp) by projected gradient descent with momentum and a
temperature schedule.  Many short restarts over a range of lengths are
performed; the best sequence found is returned.
"""

import time
import numpy as np

def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _next_power_of_two(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (real inputs) using FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    size = _next_power_of_two(m)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[:m]
    np.maximum(conv, 0.0, out=conv)  # clip tiny negatives
    return conv

def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Return soft‑max weights ∝ exp(beta * conv). Numerically stable."""
    # subtract max to avoid overflow
    max_val = conv.max()
    exps = np.exp(beta * (conv - max_val))
    total = exps.sum()
    if total == 0:
        # fallback to uniform
        return np.full_like(conv, 1.0 / conv.size)
    return exps / total

def _gradient_from_weights(a: np.ndarray, w: np.ndarray, size_grad: int) -> np.ndarray:
    """Compute ∇_a  Σ_i w_i (a*a)[i] = 2 * correlation(a, w)."""
    # correlation(a,w) = (rev(a) * w)[lag + n-1]
    rev_a = a[::-1]
    fa = np.fft.rfft(rev_a, n=size_grad)
    fw = np.fft.rfft(w, n=size_grad)
    conv_rev = np.fft.irfft(fa * fw, n=size_grad)
    # slice to obtain indices 0 … n‑1
    n = a.shape[0]
    start = n - 1
    grad = 2.0 * conv_rev[start:start + n]
    return grad

def _optimise_one_length_smooth(
    n: int,
    init: np.ndarray,
    run_deadline: float,
    *,
    base_step: float = 0.25,
    momentum_beta: float = 0.9,
    beta_init: float = 0.2,
    beta_max: float = 8.0,
    beta_growth: float = 1.001,
    step_decay: float = 0.9995,
) -> tuple[np.ndarray, float]:
    """
    Gradient descent on a smooth approximation of max(conv) for a fixed length.
    Returns (best_vector, best_max_conv).
    """
    target_sum = np.sqrt(2.0 * n)
    a = _project_onto_simplex(init.astype(np.float64), target_sum)

    # Pre‑compute FFT sizes
    size_conv = _next_power_of_two(2 * n - 1)
    size_grad = _next_power_of_two(3 * n - 2)

    best_vec = a.copy()
    conv = _conv_fft(a)
    best_max = conv.max()

    velocity = np.zeros_like(a)
    step = base_step
    beta = beta_init

    while time.time() < run_deadline:
        # convolution and soft‑max weights
        conv = _conv_fft(a)
        max_conv = conv.max()
        w = _softmax_weights(conv, beta)

        # gradient of smoothed objective
        grad = _gradient_from_weights(a, w, size_grad)

        # momentum update
        velocity = momentum_beta * velocity + (1.0 - momentum_beta) * grad

        # simple step (not normalising by norm; step already small)
        a_new = a - step * velocity
        a_new = _project_onto_simplex(a_new, target_sum)

        # occasional random shake‑up to escape flat plateaus
        if np.random.rand() < 0.02:
            noise = np.random.randn(n) * (target_sum * 1e-5)
            a_new = _project_onto_simplex(np.maximum(a_new + noise, 0.0), target_sum)
            step *= 0.92  # become a touch more conservative after a shake‑up

        # evaluate
        conv_new = _conv_fft(a_new)
        max_new = conv_new.max()
        if max_new < best_max:
            best_max = max_new
            best_vec = a_new.copy()
            # Log improvement
            print(f"[{time.time():.2f}] n={n}  new best max_conv = {best_max:.6f}")

        # update for next iteration
        a = a_new
        step *= step_decay
        beta = min(beta_max, beta * beta_growth)

    return best_vec, best_max

    The routine performs many short runs of a smoothed‑gradient descent optimizer
    on different lengths and initial shapes, keeps the best vector seen and
    finally returns it as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_score = float("inf")

    if isinstance(maybe_old, (list, tuple)) and maybe_old:
        arr = np.array(maybe_old, dtype=np.float64)
        if arr.size > 0:
            # normalise to the required ℓ₁‑norm for its length
            n_old = arr.shape[0]
            target_sum_old = np.sqrt(2.0 * n_old)
            arr = _project_onto_simplex(arr, target_sum_old)
            best_seq = arr
            best_score = 2.0 * n_old * _conv_fft(arr).max() / (arr.sum() ** 2)
            print(f"Loaded previous construction with score {best_score:.6f}")

        # keep runs short when we are near the deadline

"""
Smoothing‑gradient descent optimisation for the max‑convolution bound.

The evaluation function is homogeneous, therefore we fix the ℓ₁‑norm
Σ a_i = √(2 n).  With this normalisation the score returned by
``evaluate_sequence`` equals ``max(conv(a,a))``.  The algorithm
minimises a smooth approximation of the max (log‑sum‑exp) by
projected gradient descent with momentum, a temperature schedule and
occasional random shake‑ups.  Many short restarts over a range of
lengths are performed; the best sequence discovered is returned.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Helper 1 – Euclidean projection onto the simplex {x_i ≥ 0, Σ x_i = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex of radius s (non‑negative, sum = s)."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper 2 – Utility functions for FFT sizes and convolution
# ----------------------------------------------------------------------
def _next_power_of_two(x: int) -> int:
    """Smallest power of two greater than or equal to x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a power‑of‑2 FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    size = _next_power_of_two(m)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[:m]
    # clip tiny negative noise that may appear because of round‑off
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Helper 3 – Soft‑max weights for a smooth max approximation
# ----------------------------------------------------------------------
def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Return a probability vector ∝ exp(beta * conv), numerically stable."""
    max_val = conv.max()
    # shift by max to avoid overflow
    exps = np.exp(beta * (conv - max_val))
    total = exps.sum()
    if total == 0.0:                     # fallback – should not happen
        return np.full_like(conv, 1.0 / conv.size)
    return exps / total


# ----------------------------------------------------------------------
# Helper 4 – Gradient of Σ_i w_i (a*a)[i]  (w are the soft‑max weights)
# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray, size_grad: int) -> np.ndarray:
    """
    Compute ∇_a Σ_i w_i (a*a)[i] = 2 * correlation(a, w).

    The correlation is obtained as a convolution of the reversed a with w,
    then the appropriate slice (indices n‑1 … 2n‑2) yields the gradient.
    """
    rev_a = a[::-1]                       # a reversed
    fa = np.fft.rfft(rev_a, n=size_grad)
    fw = np.fft.rfft(w, n=size_grad)
    conv_rev = np.fft.irfft(fa * fw, n=size_grad)
    n = a.shape[0]
    start = n - 1
    grad = 2.0 * conv_rev[start:start + n]
    return grad


# ----------------------------------------------------------------------
# Core optimiser for a fixed length
# ----------------------------------------------------------------------
def _optimise_one_length_smooth(
    n: int,
    init: np.ndarray,
    run_deadline: float,
    *,
    base_step: float = 0.25,
    momentum_beta: float = 0.9,
    beta_init: float = 0.2,
    beta_max: float = 8.0,
    beta_growth: float = 1.001,
    step_decay: float = 0.9995,
) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on a smooth approximation of max(conv).

    Returns the best vector encountered and its maximal convolution value.
    """
    target_sum = np.sqrt(2.0 * n)               # Σ a_i = √(2 n) → score = max(conv)
    a = _project_onto_simplex(init.astype(np.float64), target_sum)

    size_conv = _next_power_of_two(2 * n - 1)           # for a*a
    size_grad = _next_power_of_two(3 * n - 2)           # for correlation

    best_vec = a.copy()
    best_max = _conv_fft(a).max()

    velocity = np.zeros_like(a)
    step = base_step
    beta = beta_init

    while time.time() < run_deadline:
        # 1) current convolution and soft‑max weights
        conv = _conv_fft(a)
        max_conv = conv.max()
        w = _softmax_weights(conv, beta)

        # 2) gradient of the smoothed objective
        grad = _gradient_from_weights(a, w, size_grad)

        # 3) momentum update
        velocity = momentum_beta * velocity + (1.0 - momentum_beta) * grad

        # 4) take a step and re‑project onto the simplex
        a_new = a - step * velocity
        a_new = _project_onto_simplex(a_new, target_sum)

        # 5) occasional random shake‑up (helps escaping flat plateaux)
        if np.random.rand() < 0.02:
            noise = np.random.randn(n) * (target_sum * 1e-5)
            a_new = _project_onto_simplex(np.maximum(a_new + noise, 0.0), target_sum)
            step *= 0.92

        # 6) evaluate new max‑convolution
        conv_new = _conv_fft(a_new)
        max_new = conv_new.max()
        if max_new < best_max:
            best_max = max_new
            best_vec = a_new.copy()
            print(f"[{time.time():.2f}] n={n}  new best max_conv = {best_max:.6f}")

        # 7) prepare next iteration
        a = a_new
        step *= step_decay
        beta = min(beta_max, beta * beta_growth)

    return best_vec, best_max
