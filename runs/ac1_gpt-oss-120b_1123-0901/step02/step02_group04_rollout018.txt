```
def _softmax_weights(conv, lam):
    # compute w = exp(lam*conv) / sum(exp(lam*conv))
    # Use stable version
    x = lam * conv
    mx = np.max(x)
    e = np.exp(x - mx)
    sum_e = np.sum(e)
    w = e / sum_e
    return w
```

```python
# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the simplex {x ≥ 0, Σx = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Projection of a vector onto the simplex with sum `s`."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper: FFT‑based convolution (real inputs)
# ----------------------------------------------------------------------
def _conv_fft(u: np.ndarray, v: np.ndarray) -> np.ndarray:
    """Full convolution u * v using real FFT (fast for a few thousand entries)."""
    n = u.shape[0] + v.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power of two
    fu = np.fft.rfft(u, n=size)
    fv = np.fft.rfft(v, n=size)
    conv = np.fft.irfft(fu * fv, n=size)
    return conv[:n]


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a*a (full length) via FFT."""
    return _conv_fft(a, a)


# ----------------------------------------------------------------------
# Helper: soft‑max weights for a smooth max‑approximation
# ----------------------------------------------------------------------
def _softmax_weights(conv: np.ndarray, lam: float) -> np.ndarray:
    """Return soft‑max weights w_k ∝ exp(lam·conv_k)."""
    x = lam * conv
    mx = np.max(x)
    e = np.exp(x - mx)           # subtract max for stability
    return e / np.sum(e)

# Shape factories (raw, not normalised)

def _shape_exp(n: int, alpha: float) -> np.ndarray:
    return np.exp(-alpha * np.arange(n, dtype=float))


def _shape_exp_rev(n: int, alpha: float) -> np.ndarray:
    return np.exp(-alpha * np.arange(n - 1, -1, -1, dtype=float))


def _shape_power(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    return i ** (-p)


def _shape_logistic(n: int, s: float) -> np.ndarray:
    """Symmetric logistic decay with scale s."""
    mid = (n - 1) / 2.0
    i = np.arange(n, dtype=float)
    return 1.0 / (1.0 + np.exp((i - mid) / s))


def _shape_piecewise_exp(n: int, k_frac: float, alpha: float) -> np.ndarray:
    """Plateau of ones for the first k_frac·n entries, then exponential tail."""
    k = max(1, int(k_frac * n))
    plateau = np.ones(k, dtype=float)
    tail_len = n - k
    tail = np.exp(-alpha * np.arange(tail_len, dtype=float))
    return np.concatenate([plateau, tail])

        for k_frac in [0.2, 0.35, 0.5, 0.65, 0.8]:
            for alpha in np.logspace(-4, -0.2, num=8):
                a = _shape_piecewise_exp(n, k_frac, alpha)
                a = a / a.sum() * target
                score = _fast_score(a)
                if score < best_score:
                    best_seq = a.copy()
                    best_score = score
                    print(f"[{time.time():.2f}] Sweep (piece) n={n} k={k_frac:.2f} α={alpha:.5g} → {best_score:.6f}")

        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # 2.  Projected gradient descent (soft‑max surrogate)
    # ------------------------------------------------------------------
    def _pgd_optimize(start: np.ndarray) -> np.ndarray:
        """Run a long PGD chain on `start` (already on the simplex)."""
        nonlocal best_seq, best_score
        a = start.astype(float)
        n = a.shape[0]
        target = math.sqrt(2.0 * n)

        # momentum
        m = np.zeros_like(a)
        beta = 0.9
        base_step = 0.6

        it = 0
        while time.time() < deadline:
            # current convolution and (fast) score
            conv = _conv_self(a)
            cur_score = float(conv.max())
            if cur_score < best_score:
                # evaluate the *exact* score via the provided routine
                _maybe_update(a)

            # soft‑max temperature (grow slowly)
            lam = min(50.0, 0.2 + it * 0.02)          # λ∈[~0.2,50]

            # weights for the smooth max
            w = _softmax_weights(conv, lam)

            # gradient = 2·correlation(w, a)  (see derivation)
            a_rev = a[::-1]
            corr = _conv_fft(w, a_rev)               # length 3n‑2
            grad = 2.0 * corr[n - 1:n - 1 + n]       # slice of length n

            # descent direction with momentum
            direction = -grad
            m = beta * m + (1.0 - beta) * direction

            # step‑size (norm‑normalised)
            step = base_step / math.sqrt(it + 1.0)
            max_abs = np.max(np.abs(m))
            if max_abs == 0.0:
                max_abs = 1.0
            eta = step / max_abs

            a = a + eta * m
            a = _project_onto_simplex(a, target)

            # occasional random jitter to escape plateaus
            if it % 150 == 0 and np.random.rand() < 0.15:
                a += np.random.normal(scale=target * 0.001, size=n)
                a = _project_onto_simplex(a, target)

            it += 1

            # early exit if already excellent
            if best_score < 1.503:
                break

        return a

    # run PGD on the current best (if we have one)
    if best_seq is not None:
        best_seq = _pgd_optimize(best_seq)

    # a few random restarts, each followed by PGD
    while time.time() < deadline:
        # choose a random length (bigger lengths give a lower theoretical bound)
        n = np.random.randint(800, 5001)
        target = math.sqrt(2.0 * n)

        # random mixture of two exponentials (forward/backward)
        alpha1 = np.random.rand() * 0.3 + 5e-5
        alpha2 = np.random.rand() * 0.3 + 5e-5
        w = np.random.rand()
        raw = w * _shape_exp(n, alpha1) + (1.0 - w) * _shape_exp_rev(n, alpha2)
        raw = np.maximum(raw, 0.0)
        a0 = raw / raw.sum() * target

        a_opt = _pgd_optimize(a0)

        # stop early once target achieved
        if best_score < 1.503:
            break

    # ------------------------------------------------------------------
    # 3.  Optional linear‑programming refinement (if SciPy is present)
    # ------------------------------------------------------------------
    try:
        from scipy.optimize import linprog
    except Exception:          # pragma: no cover
        linprog = None

    if linprog is not None and best_seq is not None and time.time() < deadline:
        a = best_seq.copy()
        n = a.shape[0]
        target = math.sqrt(2.0 * n)

        while time.time() < deadline:
            conv = _conv_self(a)
            M = conv.max()
            rows = 2 * n - 1

            # Build constraint matrix for (a * g)_k ≤ M
            A = np.zeros((rows, n), dtype=float)
            for k in range(rows):
                lo = max(0, k - (n - 1))
                hi = min(n - 1, k)
                cols = np.arange(lo, hi + 1, dtype=int)
                A[k, cols] = a[k - cols]

            res = linprog(
                c=-np.ones(n),           # maximise Σ g_j
                A_ub=A,
                b_ub=np.full(rows, M),
                bounds=(0, None),
                method="highs",
                options={"presolve": True, "time_limit": max(0.0, deadline - time.time())},
            )
            if not res.success:
                break

            g = res.x
            s = g.sum()
            if s == 0:
                break
            g = g * (target / s)

            # convex combination step
            t = 0.2
            a = (1.0 - t) * a + t * g

            # evaluate – this also updates the global best if improved
            _maybe_update(a)

            if best_score < 1.503:
                break

    # ------------------------------------------------------------------
    # Fallback – uniform distribution if nothing else succeeded
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=float)

    return best_seq.tolist()
```
