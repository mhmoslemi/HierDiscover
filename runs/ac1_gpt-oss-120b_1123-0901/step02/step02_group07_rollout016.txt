def propose_candidate(seed=42, budget_s=1000, **kwargs):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5
    n = 2000
    best_seq = None
    best_score = float('inf')
    # start with global seed or generate some random
    init_seq = None
    if 'height_sequence_1' in globals():
        init_seq = np.asarray(globals()['height_sequence_1'], dtype=np.float64)
        init_seq = np.clip(init_seq, 0.0, 1000.0)
        if init_seq.sum() > 0:
            init_seq = init_seq / init_seq.sum()
    else:
        init_seq = np.full(n, 1.0 / n)
    # create list of start candidates to try: uniform, random Dirichlet, current best etc.
    start_candidates = [init_seq, np.full(n, 1.0/n), _random_simplex(n)]
    # Add extra random Dirichlet seeds
    for _ in range(5):
        start_candidates.append(_random_simplex(n))
    # Evaluate each and keep best
    for cand in start_candidates:
        if time.time() > deadline: break
        score = _objective(cand)
        if score < best_score:
            best_score = score
            best_seq = cand.copy()
    # Now do optimization from best_seq
    seq = best_seq.copy()
    cur_score = best_score
    it_total = 0
    # Loop improvements: alternate LP and gradient descent.
    while time.time() < deadline:
        # if we have no progress for many iterations, break maybe.
        # Alternate: try LP refinement every few iterations
        # We'll attempt LP refinement each outer loop
        # Compute conv and M
        conv = _self_convolve(seq)
        M = conv.max()
        # Build convolution matrix A
        # Build dense matrix A (2n-1, n)
        rows = 2 * n - 1
        # allocate array
        A = np.zeros((rows, n), dtype=np.float64)
        # fill columns
        for i in range(n):
            A[i:i+n, i] = seq
        # Solve LP
        c = -np.ones(n)  # maximize sum(b)
        b_ub = M * np.ones(rows)
        bounds = [(0, None) for _ in range(n)]
        # call linprog
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs', options={'presolve': True})
        if res.success:
            b = res.x
            sum_b = b.sum()
            if sum_b > 0:
                g = b / sum_b
                # line search mixing
                step = 0.5
                improved = False
                while step > 1e-4:
                    cand_seq = (1 - step) * seq + step * g
                    cand_score = _objective(cand_seq)
                    if cand_score < cur_score - 1e-9:
                        improved = True
                        break
                    step *= 0.5
                if improved:
                    seq = cand_seq
                    cur_score = cand_score
                    if cand_score < best_score:
                        best_score = cand_score
                        best_seq = seq.copy()
                        print(f"[LP] iter {it_total:05d} improved to {best_score:.6f}")
        # gradient descent step
        # compute subgradient using active shift
        k_star = int(conv.argmax())
        grad = np.zeros_like(seq)
        lo = max(0, k_star - (n - 1))
        hi = min(k_star, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] = seq[k_star - idx]
        # backtracking line search
        step = 5.0
        improved = False
        while step > 1e-6:
            cand = _project_onto_simplex(seq - step * grad)
            cand_score = _objective(cand)
            if cand_score < cur_score:
                improved = True
                break
            step *= 0.5
        if improved:
            seq = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = seq.copy()
                print(f"[GD] iter {it_total:05d} improved to {best_score:.6f}")
        else:
            # small random move
            i,j = np.random.choice(n, 2, replace=False)
            delta = 0.01 * min(seq[i], 0.1)
            if delta > 0:
                cand = seq.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_onto_simplex(cand)
                cand_score = _objective(cand)
                if cand_score < cur_score:
                    seq = cand
                    cur_score = cand_score
                    if cand_score < best_score:
                        best_score = cand_score
                        best_seq = seq.copy()
                        print(f"[Rnd] iter {it_total:05d} improved to {best_score:.6f}")
        it_total += 1
    print(f"Final best = {best_score:.6f}")
    return best_seq.tolist()

import time
import numpy as np

def _project_onto_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        w = np.full_like(v, 1.0 / n)
        return w
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def _next_pow2(x):
    return 1 << (x - 1).bit_length()

def _self_convolve(seq):
    n = seq.shape[0]
    m = 2 * n - 1
    nfft = _next_pow2(m)
    A = np.fft.rfft(seq, n=nfft)
    conv = np.fft.irfft(A * A, n=nfft)[:m]
    return conv

def _objective(seq):
    sum_a = seq.sum()
    if sum_a < 0.01:
        return float('inf')
    conv = _self_convolve(seq)
    max_b = conv.max()
    return float(2.0 * len(seq) * max_b / (sum_a * sum_a))

def _geom_seq(n, alpha):
    eps = alpha / n
    r = 1.0 - eps
    if r <= 0.0:
        return None
    powers = r ** np.arange(n, dtype=np.float64)
    seq = powers / powers.sum()
    return seq

def _powerlaw_seq(n, p):
    seq = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    return seq / seq.sum()

def _beta_seq(n, a, b):
    i = np.arange(1, n + 1, dtype=np.float64)
    seq = (i ** (a - 1.0)) * ((n + 1 - i) ** (b - 1.0))
    seq = np.where(seq > 0, seq, 0.0)
    return seq / seq.sum()

    center = (n - 1) / 2.0
    seq = np.exp(-0.5 * ((i - center) / sigma) ** 2)
    return seq / seq.sum()

    vals = np.sin(np.pi * (i + 0.5) / n) ** p
    seq = np.where(vals > 0, vals, 0.0)

def _project_onto_simplex(v):
    """Project vector v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)

            pass

    # Uniform baseline (always feasible)
    _consider(np.full(n, 1.0 / n))

    # --------------------------------------------------------------
    # 3. Systematic sweep over a few parametric families.
    #    The loops break early if the time budget is exhausted.

    # exponential family
    for alpha in np.linspace(0.5, 8.0, 20):
        if time.time() > deadline:
            break
        seq = _geom_seq(n, alpha)
        if seq is not None:
            _consider(seq)

    # power‑law family
    for p in np.linspace(0.2, 5.0, 20):
        if time.time() > deadline:
            break
        seq = _powerlaw_seq(n, p)
        _consider(seq)

    # beta‑like family (two parameters)
    a_vals = np.linspace(0.2, 8.0, 12)
    b_vals = np.linspace(0.2, 8.0, 12)
    for a in a_vals:
        for b in b_vals:
            if time.time() > deadline:
                break
            seq = _beta_seq(n, a, b)
            _consider(seq)

    # Gaussian shapes
    for sigma in np.linspace(0.05 * n, 0.5 * n, 12):
        if time.time() > deadline:
            break
        seq = _gaussian_seq(n, sigma)
        _consider(seq)

    # Sine‑power shapes
    for p in np.linspace(0.5, 10.0, 12):
        if time.time() > deadline:
            break
        seq = _sin_seq(n, p)
        _consider(seq)

    # Random points on the simplex (Dirichlet(1,…,1))
    for _ in range(40):
        if time.time() > deadline:
            break
        seq = np.random.exponential(1.0, size=n)
        _consider(seq)

    # --------------------------------------------------------------
    # 4. Fallback if nothing was generated (should never happen)
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)
        best_score = _objective(best_seq)

    # --------------------------------------------------------------
    # 5. Local refinement – accelerated projected gradient descent.
    seq = best_seq.copy()
    cur_score = best_score
    v = np.zeros_like(seq)          # momentum vector
    beta_mom = 0.9
    base_step = 1.0
    it = 0
    stagnation = 0

    while time.time() < deadline:
        # ----------------------------------------------------------
        # compute convolution and the active shift k*
        conv = _self_convolve(seq)
        k_star = int(conv.argmax())

        # sub‑gradient of the active convolution entry (factor 2 omitted)
        grad = np.zeros_like(seq)
        lo = max(0, k_star - (n - 1))
        hi = min(k_star, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] = seq[k_star - idx]

        # Nesterov momentum update
        v = beta_mom * v + (1.0 - beta_mom) * grad

        # back‑tracking line search on the momentum direction
        step = base_step
        improved = False
        while step > 1e-7:
            cand = _project_onto_simplex(seq - step * v)
            cand_score = _objective(cand)
            if cand_score < cur_score:
                improved = True
                break
            step *= 0.5

        if improved:
            seq = cand
            cur_score = cand_score
            v = v * (step / base_step)          # adapt magnitude
            base_step = max(step * 1.2, 1e-4)
            stagnation = 0
            if cand_score < best_score:
                best_score = cand_score
                best_seq = seq.copy()
                print(f"[GD {it:05d}] improved to {best_score:.6f}")
        else:
            stagnation += 1
            # every 20 unsuccessful attempts try a larger random transfer
            if stagnation % 20 == 0:
                i, j = np.random.choice(n, 2, replace=False)
                delta = 0.05 * min(seq[i], 0.2)   # up to 5 % of total mass
                if delta > 0:
                    cand = seq.copy()
                    cand[i] -= delta
                    cand[j] += delta
                    cand = _project_onto_simplex(cand)
                    cand_score = _objective(cand)
                    if cand_score < cur_score:
                        seq = cand
                        cur_score = cand_score
                        if cand_score < best_score:
                            best_score = cand_score
                            best_seq = seq.copy()
                            print(f"[Rnd {it:05d}] improved to {best_score:.6f}")

        # ----------------------------------------------------------
        # Occasionally perform a quick LP‑based refinement (if SciPy is present)
        if it % 250 == 0 and time.time() < deadline:
            try:
                from scipy.optimize import linprog
                # Build dense matrix A where (A @ b)[k] = (seq * b)[k]
                rows = 2 * n - 1
                A = np.zeros((rows, n), dtype=np.float64)
                for i in range(n):
                    A[i:i + n, i] = seq
                M = conv.max()
                # maximise sum(b)  <=> minimise -sum(b)
                c = -np.ones(n)
                b_ub = M * np.ones(rows)
                bounds = [(0, None)] * n
                # a moderate time limit for the LP (if the solver respects it)
                res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                              method='highs', options={'presolve': True})
                if res.success:
                    b = res.x
                    sum_b = b.sum()
                    if sum_b > 0:
                        g = b / sum_b
                        # try blending a bit of the LP solution
                        step_lp = 0.4
                        cand = (1 - step_lp) * seq + step_lp * g
                        cand_score = _objective(cand)
                        if cand_score < cur_score:
                            seq = cand
                            cur_score = cand_score
                            if cand_score < best_score:
                                best_score = cand_score
                                best_seq = seq.copy()
                                print(f"[LP {it:05d}] improved to {best_score:.6f}")
            except Exception:
                # SciPy not available or LP failed – ignore and continue
                pass

        it += 1

    # --------------------------------------------------------------
    print(f"Final best objective = {best_score:.6f} (length {n})")
    return best_seq.tolist()
