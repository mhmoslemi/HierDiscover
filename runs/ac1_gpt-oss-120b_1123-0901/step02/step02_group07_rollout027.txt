def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))

"""
Propose a step‑function that (hopefully) beats the previous best ≈ 1.5449.
The algorithm is based on the classic “alternating linear programme”:

1. Keep a non‑negative vector `f` (normalised to sum = 1).
2. Compute `M = max(f*f)` (full self‑convolution).
3. Solve the linear programme
        maximise    Σ g_j
        subject to  g_j ≥ 0
                     (f∗g)[k] ≤ M   for all shifts k
   The convolution `(f∗g)[k]` is linear in `g`, therefore the constraints are
   a standard set of linear inequalities `A @ g ≤ M`.
4. Rescale the LP solution `g` so that it also has unit sum and perform a
   short back‑tracking line search on the convex combination
        f_new = (1‑t)·f + t·g .
   The step size `t` is decreased until the objective strictly improves.
5. Repeat until the time budget (≈ 1000 s) expires, always keeping the best
   vector seen.  The implementation uses a fast FFT‑based convolution for
   objective evaluation and `scipy.optimize.linprog` (HiGHS) to solve the LP.
   It starts from the global seed `height_sequence_1` if present, otherwise
   from the uniform distribution.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex (used for fallback moves)
def _project_onto_simplex(v):
    """Project a vector v onto the probability simplex {x≥0, Σx=1}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast self‑convolution using real FFT
def _next_pow2(x: int) -> int:
    """Small utility: smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _self_convolve(seq: np.ndarray) -> np.ndarray:
    """Full convolution of seq with itself (length 2·n‑1)."""
    n = seq.shape[0]
    m = 2 * n - 1
    nfft = _next_pow2(m)
    A = np.fft.rfft(seq, n=nfft)
    conv = np.fft.irfft(A * A, n=nfft)[:m]
    return conv


# ----------------------------------------------------------------------
# Local objective (identical to evaluate_sequence but faster, no clipping)
def _objective(seq: np.ndarray) -> float:
    """Return 2·n·max(conv)/sum(seq)^2 for a non‑negative sequence."""
    sum_a = seq.sum()
    if sum_a < 0.01:
        return float("inf")
    conv = _self_convolve(seq)
    max_b = conv.max()
    return float(2.0 * len(seq) * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Build the convolution matrix A such that (f∗g) = A @ g
def _conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Return a dense matrix A of shape (2·n‑1, n) with
    A[k, j] = f[k‑j]  if 0 ≤ k‑j < n else 0.
    Then A @ g = np.convolve(f, g)   (full convolution).
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    # fill column‑wise: column j is f shifted by +j
    for j in range(n):
        A[j:j + n, j] = f
    return A


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000,
                     n: int = 2000,
                     **kwargs):
    """
    Search for a sequence of non‑negative heights that minimises the
    provided `evaluate_sequence`.  The method starts from a uniform or a
    saved global seed, then repeatedly solves a linear programme to obtain
    a better direction and performs a back‑tracking line‑search on the
    convex combination.  The best sequence found before the time limit is
    returned as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # 0. initialise `f` (unit‑sum vector)
    if "height_sequence_1" in globals():
        try:
            f0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            f0 = np.clip(f0, 0.0, 1000.0)
            if f0.sum() > 0.0:
                f0 = f0 / f0.sum()
                print("[Init] global seed used")
        except Exception:
            f0 = None
    else:
        f0 = None

    if f0 is None or f0.shape[0] != n:
        # uniform seed
        f0 = np.full(n, 1.0 / n, dtype=np.float64)

    f = f0.copy()
    cur_score = _objective(f)
    best_seq = f.copy()
    best_score = cur_score
    print(f"[Init] start objective = {cur_score:.6f}")

    it = 0
    base_step = 0.2   # first guess for the convex‑combination weight

    while time.time() < deadline:
        # --------------------------------------------------------------
        # 1. current bound M = max(f*f)
        conv_ff = _self_convolve(f)
        M = conv_ff.max()

        # 2. Build linear programme: maximise Σ g_j  subject to A@g ≤ M, g≥0
        A = _conv_matrix(f)                     # shape (2n‑1, n)
        b = np.full(A.shape[0], M, dtype=np.float64)

        # Solve LP (HiGHS, fast for these sizes)
        #  maximise Σ g_j  → minimise −Σ g_j
        c = -np.ones(n, dtype=np.float64)
        lp_res = linprog(c, A_ub=A, b_ub=b,
                         bounds=(0, None),
                         method='highs',
                         options={"presolve": True, "time_limit": max(0.0, deadline - time.time())})

        if not lp_res.success:
            # fallback: random perturbation if LP fails
            i, j = np.random.choice(n, 2, replace=False)
            delta = 0.01 * min(f[i], 0.1)
            if delta > 0.0:
                cand = f.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_onto_simplex(cand)
                cand_score = _objective(cand)
                if cand_score < cur_score:
                    f, cur_score = cand, cand_score
                    if cand_score < best_score:
                        best_score, best_seq = cand_score, cand.copy()
                        print(f"[Iter {it:05d}] random tweak improved to {best_score:.6f}")
            it += 1
            continue

        g_raw = lp_res.x
        sum_g_raw = g_raw.sum()
        if sum_g_raw <= 0:
            # degenerate LP, skip
            it += 1
            continue

        # 3. Rescale g to unit sum (same normalisation as f)
        g = g_raw / sum_g_raw

        # 4. Line‑search on convex combination f_new = (1‑t)*f + t*g
        t = base_step
        improved = False
        while t > 1e-5:
            cand = (1.0 - t) * f + t * g
            cand_score = _objective(cand)
            if cand_score < cur_score:
                improved = True
                break
            t *= 0.5

        if improved:
            f = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score, best_seq = cand_score, cand.copy()
                print(f"[Iter {it:05d}] new best = {best_score:.6f} (t={t:.4f})")
            # slightly enlarge step for next iteration if the move succeeded
            base_step = max(t * 1.3, 0.02)
        else:
            # no improvement – try a tiny random mass transfer
            i, j = np.random.choice(n, 2, replace=False)
            delta = 0.01 * min(f[i], 0.1)
            if delta > 0.0:
                cand = f.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_onto_simplex(cand)
                cand_score = _objective(cand)
                if cand_score < cur_score:
                    f, cur_score = cand, cand_score
                    if cand_score < best_score:
                        best_score, best_seq = cand_score, cand.copy()
                        print(f"[Iter {it:05d}] tiny random move improved to {best_score:.6f}")

            # keep the step size modest
            base_step = max(base_step * 0.7, 0.02)

        it += 1

    print(f"Finished – best objective = {best_score:.6f} (iterations {it})")
    # Return a plain Python list (evaluate_sequence expects a list)
    return best_seq.tolist()
