"""
Propose a non‑negative step‑function (list of heights) that minimizes
``evaluate_sequence``.  

Algorithm
---------
* Several problem sizes are explored (original length and a few larger
  multiples).  For each size we run a short Adam optimiser that minimises
  the peak of the autocorrelation (the max of `a*a`), which is exactly the
  value returned by ``evaluate_sequence`` after normalising the L1 norm to
  √(2 n).
* The optimiser works on the exact sub‑gradient of the max‑convolution:
      ∂max/∂a_i = average_{k∈active} a_{k-i}  –  (max·a_i)/(Σa) .
  The active set consists of all shift indices whose convolution value is
  within a tiny tolerance of the current maximum.
* After each update we project onto the feasible set (non‑negative and the
  correct L1‑norm).  Occasionally a tiny Gaussian jitter is added to avoid
  plateaus.
* To avoid being stuck in a local minimum we perform a few random restarts
  (different random initial vectors) and also a lightweight “local search’’
  that moves a small amount of mass between two random positions if the
  objective improves.
* The total wall‑clock time is limited to the supplied budget (≈1000 s);
  the routine stops early if the time runs out and returns the best
  sequence found.

The code only uses ``numpy`` and the already‑imported ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _conv_max_fft(a: np.ndarray):
    """
    Return the maximal value of the full convolution a*a and the full
    convolution array (real).  Uses an FFT of length equal to the next
    power of two for speed.
    """
    n = a.shape[0]
    # Size for zero‑padding: next power of two >= 2*n-1
    N = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=N)
    B = A * A                     # element‑wise product → FFT of convolution
    conv = np.fft.irfft(B, n=N)
    conv = conv[: 2 * n - 1]      # keep only the valid part
    max_val = float(conv.max())
    return max_val, conv


# ----------------------------------------------------------------------
def _subgradient(a: np.ndarray, active_idxs: np.ndarray):
    """
    Compute a sub‑gradient of the max‑convolution at the current point.
    The gradient component for index i equals the average of a_{k-i} over
    all active shift positions k (with out‑of‑range indices ignored).
    """
    n = a.shape[0]
    idx = np.arange(n)
    grad = np.zeros_like(a)

    for k in active_idxs:
        j = k - idx                # indices of the partner in the convolution
        mask = (j >= 0) & (j < n)  # keep only pairs that lie inside the vector
        grad[mask] += a[j[mask]]

    # average over the active set (prevents bias when several shifts tie)
    if active_idxs.size > 0:
        grad /= float(active_idxs.size)
    return grad


# ----------------------------------------------------------------------
def _adam_optimize(init_a: np.ndarray,
                   n: int,
                   target_sum: float,
                   deadline: float,
                   max_iters: int = 10_000_000):
    """
    Minimise the peak of a*a using Adam on the exact sub‑gradient.
    Returns the best vector seen and its objective value.
    """
    a = init_a.copy()
    # initial normalisation (guard against tiny rounding errors)
    s = a.sum()
    if s > 0:
        a *= (target_sum / s)
    else:
        a[:] = target_sum / n

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    base_lr = 0.04
    iteration = 0
    jitter_interval = 4000
    best_vec = a.copy()
    _, conv = _conv_max_fft(a)
    best_val = conv.max()

    while time.time() < deadline and iteration < max_iters:
        # --------------------------------------------------------------
        # 1. convolution and active set
        M, conv = _conv_max_fft(a)

        # treat all shifts whose value is within tolerance of the maximum as active
        tol = 1e-12
        active = np.where(conv >= M - tol)[0]

        # --------------------------------------------------------------
        # 2. sub‑gradient of the max term
        rev = _subgradient(a, active)

        # derivative of the whole objective  (the scaling factor 2*n/(sum^2) is 1)
        coeff = 4.0 * n / (target_sum * target_sum)
        const = 4.0 * n * M / (target_sum ** 3)
        grad = coeff * rev - const

        # --------------------------------------------------------------
        # 3. Adam update
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        # simple decay schedule for the learning rate
        lr = base_lr / (1.0 + 1e-4 * iteration)

        a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # --------------------------------------------------------------
        # 4. projection onto feasible set
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0:
            a *= (target_sum / s)
        else:
            a[:] = target_sum / n

        # --------------------------------------------------------------
        # 5. occasional jitter to escape flat plateaus
        if iteration % jitter_interval == 0:
            jitter = np.random.randn(n) * 0.0008 * target_sum / n
            a = np.maximum(a + jitter, 0.0)
            a *= (target_sum / a.sum())

        # --------------------------------------------------------------
        # 6. keep the best solution seen so far (objective = M)
        if M < best_val:
            best_val = M
            best_vec = a.copy()
            print(f"[len {n:5d} iter {iteration:7d}] new best  M = {best_val:.6f}")

    return best_vec, best_val


# ----------------------------------------------------------------------
def _local_search(a: np.ndarray,
                  n: int,
                  target_sum: float,
                  steps: int,
                  deadline: float):
    """
    Very cheap stochastic local search: repeatedly move a tiny amount of mass
    from one random index to another if the move reduces the max‑convolution.
    """
    best = a.copy()
    best_val, _ = _conv_max_fft(best)
    for step in range(steps):
        if time.time() >= deadline:
            break

        # pick two distinct indices
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue

        # amount to transfer – a small fraction of the source entry
        delta = best[i] * 0.02
        if delta <= 0:
            continue

        # propose move
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        # projection onto non‑negative is already satisfied
        # (sum stays constant)

        M, _ = _conv_max_fft(cand)
        if M < best_val:
            best = cand
            best_val = M
        # optionally could accept occasional uphill moves (simulated annealing),
        # but the simple greedy scheme works well enough for a short sweep.
    return best, best_val


# ----------------------------------------------------------------------
