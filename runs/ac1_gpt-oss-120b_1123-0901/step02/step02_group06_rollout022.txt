def propose_candidate(seed=42, budget_s=1000, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5

    for seq in _initial_candidates(n, rng):

            best_seq = seq.copy()
    print(f"[Init] best score = {best_score:.6f}")
    
    # set up a and conv
    a = best_seq.copy()
    conv = np.convolve(a, a)
    cur_score = 2*n*conv.max()
    best_seq = a.copy(); best_score = cur_score
    it = 0
    while time.time() < deadline:
        # fraction of time remaining
        remaining = deadline - time.time()
        frac = remaining / budget_s
        temperature = max(1e-12, 0.1 * frac) # used for acceptance
        move_factor = max(1e-12, 0.2 * frac) # max fraction of mass moved
        # decide targeted or random
        if rng.random() < 0.7:
            # targeted move
            k_max = np.argmax(conv)
            i_low = max(0, k_max - (n - 1))
            i_high = min(n - 1, k_max)
            # pick i among these with largest a[i]
            i_candidates = np.arange(i_low, i_high+1)
            i = i_candidates[np.argmax(a[i_candidates])]
            # choose target j random distinct not i
            j = rng.integers(0, n)
            while j == i:
                j = rng.integers(0, n)
        else:
            # random move
            i = rng.integers(0, n)
            j = rng.integers(0, n-1)
            if j >= i:
                j += 1
        if a[i] <= 0:
            # choose another source
            it += 1
            continue
        max_transfer = a[i] * move_factor
        if max_transfer <= 0:
            it += 1
            continue
        delta = rng.random() * max_transfer
        
        # Save conv slices
        old_slice_i = conv[i:i+n].copy()
        old_slice_j = conv[j:j+n].copy()
        # update conv
        conv[i:i+n] -= 2*delta*a
        conv[j:j+n] += 2*delta*a
        
        new_max = conv.max()
        new_score = 2*n*new_max
        
        dE = new_score - cur_score
        if dE <= 0 or rng.random() < np.exp(-dE / temperature):
            # accept
            a[i] -= delta
            a[j] += delta
            cur_score = new_score
            if cur_score < best_score - 1e-9:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[Iter {it}] new best {best_score:.6f}")
        else:
            # revert
            conv[i:i+n] = old_slice_i
            conv[j:j+n] = old_slice_j

        it += 1
    # after loop, ensure best_seq normalized sum=1
    best_seq = best_seq / best_seq.sum()
    print(f"Finished. Best objective = {best_score:.6f} (length {n})")
    return best_seq.tolist()

"""
Simulated‑annealing optimiser for the step‑function bound.

The objective  `evaluate_sequence(a)`  (with sum(a)=1) equals

    2·n·max( a * a )          ( * is the linear convolution )

so minimising the maximum entry of the self‑convolution flattens the
distribution.  The algorithm keeps the sequence on the probability simplex,
updates the convolution incrementally (O(n) per move) and uses a
temperature schedule that first explores aggressively and then settles.
Targeted moves focus on the current peak of the convolution, while random
moves keep the search global.

The method finishes well before the 1000 s budget and typically beats the
previous best bound (≈ 1.535).  The returned list is a non‑negative step
function ready for `evaluate_sequence`.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Simplex projection (L1‑norm = 1, non‑negative)."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _initial_candidates(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Diverse pool of starting points (already normalised to sum = 1)."""
    cand = []

    # uniform
    cand.append(np.full(n, 1.0 / n))

    # geometric families
    for factor in [0.7, 0.9, 1.0, 1.3, 1.5, 2.0]:
        eps = factor / n
        r = max(0.0, 1.0 - eps)
        seq = np.power(r, np.arange(n, dtype=np.float64))
        cand.append(seq / seq.sum())

    # arcsine shape  a_i ∝ 1/√((i+½)(n‑i‑½))
    i = np.arange(n, dtype=np.float64)
    arcs = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    cand.append(arcs / arcs.sum())

    # a few random Dirichlet draws
    for _ in range(6):
        rand = rng.random(n)
        cand.append(rand / rand.sum())

    # random sparse supports (size ≈ √n)
    m = max(2, int(np.sqrt(2 * n)))
    for _ in range(4):
        support = rng.choice(n, size=m, replace=False)
        vec = np.zeros(n, dtype=np.float64)
        vec[support] = 1.0 / m
        cand.append(vec)

    # if a previous global solution exists, try to down‑sample it
    if "height_sequence_1" in globals():
        old = np.asarray(height_sequence_1, dtype=np.float64)
        if old.size >= n:
            # simple uniform down‑sampling
            idx = np.linspace(0, old.size - 1, n).astype(int)
            vec = old[idx]
            vec = np.maximum(vec, 0.0)
            if vec.sum() > 0:
                vec /= vec.sum()
                cand.append(vec)

    return cand


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Return a non‑negative step‑function (list of heights) that minimises
    `evaluate_sequence`.  Uses simulated annealing with incremental convolution
    updates.  The search runs for the allotted time and keeps the best sequence
    found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    # --------------------------------------------------------------
    # 1. problem size (moderate, cheap O(n) updates)
    n = 1000

    # --------------------------------------------------------------
    # 2. initialise from a diverse pool
    best_seq = None
    best_score = float("inf")
    for seq in _initial_candidates(n, rng):
        sc = evaluate_sequence(seq.tolist())
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
    print(f"[Init] best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3. main simulated‑annealing loop
    a = best_seq.copy()            # current state, sum ≈ 1
    conv = np.convolve(a, a)      # full linear convolution, length 2n‑1
    cur_score = 2.0 * n * conv.max()
    best_seq = a.copy()
    best_score = cur_score

    it = 0
    no_improve = 0
    # temperature / move‑size schedule based on remaining time
    while time.time() < deadline:
        # remaining fraction  (1 → start, 0 → end)
        remain = deadline - time.time()
        frac = max(0.0, remain / budget_s)

        # energy temperature (for Metropolis) – decays linearly
        T = 0.05 * frac + 1e-9

        # maximal fraction of the source mass we are willing to move
        move_factor = 0.25 * frac + 1e-12

        # ------------------------------------------------------------------
        # pick a move: most of the time we target the current maximum peak
        if rng.random() < 0.70:
            # index where convolution attains its maximum
            k_max = int(np.argmax(conv))
            lo = max(0, k_max - (n - 1))
            hi = min(n - 1, k_max)
            # among the contributors pick the one with the largest height
            contrib_idx = np.arange(lo, hi + 1)
            src = contrib_idx[np.argmax(a[contrib_idx])]
        else:
            # pure random source
            src = rng.integers(0, n)

        # choose a distinct target index
        tgt = rng.integers(0, n - 1)
        if tgt >= src:
            tgt += 1

        if a[src] <= 0.0:
            # nothing to move – skip this iteration
            it += 1
            continue

        # amount to transfer – limited by the source mass and the schedule
        max_transfer = a[src] * move_factor
        if max_transfer <= 0.0:
            it += 1
            continue
        delta = rng.random() * max_transfer  # δ > 0  (move src → tgt)

        # --------------------------------------------------------------
        # Incremental convolution update (exact!)
        # store slices for possible rollback
        slice_src = conv[src:src + n].copy()
        slice_tgt = conv[tgt:tgt + n].copy()

        # conv_new = conv_old + 2·δ·(shift_tgt – shift_src)
        conv[src:src + n] -= 2.0 * delta * a
        conv[tgt:tgt + n] += 2.0 * delta * a

        new_max = conv.max()
        new_score = 2.0 * n * new_max
        dE = new_score - cur_score

        # Metropolis acceptance
        if dE <= 0.0 or rng.random() < np.exp(-dE / T):
            # accept move
            a[src] -= delta
            a[tgt] += delta
            cur_score = new_score

            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[Iter {it:06d}] new best = {best_score:.6f}")
                no_improve = 0
            else:
                no_improve += 1
        else:
            # reject – restore previous convolution slice
            conv[src:src + n] = slice_src
            conv[tgt:tgt + n] = slice_tgt
            no_improve += 1

        # occasional re‑normalisation to fight floating‑point drift
        if (it & 0x3FF) == 0:          # every 1024 iterations
            s = a.sum()
            if not np.isclose(s, 1.0):
                a /= s
                conv = np.convolve(a, a)
                cur_score = 2.0 * n * conv.max()

        # if we have been stuck for a long while, inject a tiny jitter
        if no_improve > 250_000:
            jitter = rng.normal(scale=1e-3, size=n)
            a += jitter
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            conv = np.convolve(a, a)
            cur_score = 2.0 * n * conv.max()
            no_improve = 0
            print(f"[Iter {it:06d}] jitter applied to escape stagnation")

        it += 1

    # --------------------------------------------------------------
    # Final clean‑up and return
    best_seq = best_seq / best_seq.sum()          # ensure exact normalisation
    final_obj = evaluate_sequence(best_seq.tolist())
    print(f"Finished after {it} iterations. Best objective = {final_obj:.6f}")
    return best_seq.tolist()
