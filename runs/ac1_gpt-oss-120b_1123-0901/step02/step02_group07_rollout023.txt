def _cross_corr_weighted(a, w):
    # a: length n
    # w: length 2n-1
    # return h length n: h[j] = Σ_k w[k] * a[k - j] where a out-of-range zero.
    a_rev = a[::-1]
    m = len(w) + len(a_rev) - 1
    nfft = 1 << ((m - 1).bit_length())
    A = np.fft.rfft(a_rev, n=nfft)
    W = np.fft.rfft(w, n=nfft)
    conv = np.fft.irfft(A * W, n=nfft)[:m]
    start = len(a_rev) - 1  # n-1
    h = conv[start:start+len(a)]
    return h

"""
Propose a non‑negative step‑function (a list of heights) that minimises
evaluate_sequence(...).  The algorithm works in three stages:

1.  Broad parametric sweep (geometric, power‑law, beta‑like and Dirichlet)
    – a few hundred cheap candidates are generated and the best few are kept.

2.  Refine each promising seed with a smooth “soft‑max” approximation of the
    max‑convolution and Adam optimisation on the log‑probabilities.
    The parameters are the logits of a probability vector; after each update
    a soft‑max gives a valid non‑negative sequence with unit sum.
    The smoothed objective J_ρ = (1/ρ) log Σ exp(ρ·conv) approximates the true
    max‑convolution; its gradient can be computed by a single FFT‑based
    convolution, which makes the optimisation fast (O(n log n) per step).

3.  The best sequence seen during the whole run (actual objective, not the
    surrogate) is returned as a plain Python list.

The implementation respects the 1000 s budget, prints progress information
and contains only top‑level helper functions (no closures or lambdas).
"""

import time
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex (used only for fall‑back)
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex {x ≥ 0, sum(x)=1}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast convolution utilities (real FFT)
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _self_convolve(seq: np.ndarray) -> np.ndarray:
    """Full convolution of seq with itself (length 2·n‑1)."""
    n = seq.shape[0]
    m = 2 * n - 1
    nfft = _next_pow2(m)
    A = np.fft.rfft(seq, n=nfft)
    conv = np.fft.irfft(A * A, n=nfft)[:m]
    return conv


def _cross_corr_weighted(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute h_i = Σ_k w_k * a_{k-i}   (indices out of range give 0).

    a : length n,   w : length 2·n‑1
    Returns an array of length n.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    m = len(w) + len(a_rev) - 1
    nfft = _next_pow2(m)
    A = np.fft.rfft(a_rev, n=nfft)
    W = np.fft.rfft(w, n=nfft)
    conv = np.fft.irfft(A * W, n=nfft)[:m]
    start = n - 1                # index where a_rev starts in the full conv
    return conv[start:start + n]


# ----------------------------------------------------------------------
# Objective (local fast version).  With sum(a)=1 it equals max(conv).
def _objective(seq: np.ndarray) -> float:
    """2·n·max(conv)/sum(seq)^2 for a non‑negative sequence."""
    sum_a = seq.sum()
    if sum_a < 0.01:
        return float("inf")
    conv = _self_convolve(seq)
    max_b = conv.max()
    return float(2.0 * len(seq) * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Parametric families
def _geom_seq(n: int, alpha: float) -> np.ndarray | None:
    """Geometric progression a_i ∝ r^i with r = 1-α/n."""
    eps = alpha / n
    r = 1.0 - eps
    if r <= 0.0:
        return None
    powers = r ** np.arange(n, dtype=np.float64)
    seq = powers / powers.sum()
    return seq


def _powerlaw_seq(n: int, p: float) -> np.ndarray:
    """Power‑law a_i ∝ (i+1)^{-p}."""
    seq = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    return seq / seq.sum()


def _beta_seq(n: int, a: float, b: float) -> np.ndarray:
    """Beta‑like shape a_i ∝ (i+1)^{a‑1}·(n‑i)^{b‑1}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    seq = (i ** (a - 1.0)) * ((n + 1 - i) ** (b - 1.0))
    seq[seq < 0] = 0.0
    return seq / seq.sum()


def _random_dirichlet(n: int, conc: float = 1.0) -> np.ndarray:
    """Draw a point on the simplex from Dirichlet(conc,…,conc)."""
    x = np.random.gamma(conc, 1.0, size=n)
    if x.sum() == 0.0:
        return np.full(n, 1.0 / n)
    return x / x.sum()


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs) -> List[float]:
    """
    Search for a step‑function with a low evaluate_sequence value.
    The method combines a parametric sweep and an Adam optimiser on a
    smooth surrogate of the max‑convolution.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5            # leave a safety margin

    # ------------------------------------------------------------------
    # Parameters
    n = 2000                                            # length of the sequence
    sum_target = np.sqrt(2.0 * n)                       # normalisation used in the paper

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 1. Optional initial seed from the global variable
    if "height_sequence_1" in globals():
        try:
            tmp = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            tmp = np.clip(tmp, 0.0, 1000.0)
            if tmp.sum() > 0.0:
                tmp = tmp / tmp.sum()
                sc = _objective(tmp)
                if sc < best_score:
                    best_score, best_seq = sc, tmp.copy()
                    print(f"[Init] global seed score={sc:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 2. Broad parametric sweep – keep the 10 best seeds
    candidate_list = []

    # geometric family (α from 0.2 to 5.0)
    for alpha in np.linspace(0.2, 5.0, 20):
        if time.time() > deadline:
            break
        seq = _geom_seq(n, alpha)
        if seq is None:
            continue
        candidate_list.append(seq)

    # power‑law family (p from -2.0 to 2.5)
    for p in np.linspace(-2.0, 2.5, 20):
        if time.time() > deadline:
            break
        candidate_list.append(_powerlaw_seq(n, p))

    # beta‑like family (a,b in [0.1,4.0])
    for a in np.linspace(0.1, 4.0, 5):
        for b in np.linspace(0.1, 4.0, 5):
            if time.time() > deadline:
                break
            candidate_list.append(_beta_seq(n, a, b))

    # random Dirichlet with different concentration parameters
    for conc in [0.2, 0.5, 1.0, 2.0, 5.0]:
        for _ in range(6):
            if time.time() > deadline:
                break
            candidate_list.append(_random_dirichlet(n, conc))

    # ------------------------------------------------------------------
    # Evaluate all candidates, retain the top 10
    scores = [_objective(seq) for seq in candidate_list]
    order = np.argsort(scores)[:10]                      # take the best 10
    seeds = [candidate_list[i] for i in order]
    for i, seq in enumerate(seeds):
        sc = scores[order[i]]
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
        print(f"[Sweep] seed {i:02d} score={sc:.6f}")

    # ------------------------------------------------------------------
    # 3. Adam optimisation on a smooth surrogate of max‑convolution
    # --------------------------------------------------------------
    # Adam hyper‑parameters
    lr = 0.15
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # start smoothing parameter ρ (rho) – will be increased during optimisation
    rho = 5.0
    rho_max = 200.0
    rho_step = 5.0

    max_outer_iters = 12        # how many different seed runs
    max_inner_iters = 400       # Adam steps per seed (will stop early if time runs out)

    for outer in range(max_outer_iters):
        if time.time() > deadline:
            break

        # pick a seed – either the current best or a random Dirichlet
        if outer == 0 and best_seq is not None:
            seed_vec = best_seq.copy()
        else:
            seed_vec = _random_dirichlet(n, conc=1.0)

        # initialise logits (b) from the seed (add tiny epsilon to avoid log(0))
        epsilon = 1e-12
        b = np.log(seed_vec + epsilon)

        # Adam moment vectors
        m = np.zeros_like(b)
        v = np.zeros_like(b)

        for t in range(1, max_inner_iters + 1):
            if time.time() > deadline:
                break

            # ------- obtain probabilities (a) from the logits -----------------
            b_max = np.max(b)
            a = np.exp(b - b_max)
            a /= a.sum()                               # now Σ a_i = 1

            # ------- true objective (for bookkeeping) -------------------------
            conv = _self_convolve(a)
            cur_score = np.max(conv)                   # because Σ a = 1
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[Iter {outer:02d}/{t:04d}] new best score={best_score:.6f}")

            # ------- smooth surrogate weights ---------------------------------
            # w_k = softmax_ρ(conv_k)
            max_conv = conv.max()
            exp_term = np.exp(rho * (conv - max_conv))
            w = exp_term / exp_term.sum()              # Σ w_k = 1

            # ------- gradient of the surrogate w.r.t. a -----------------------
            # grad_a = 2 * Σ_k w_k * a_{k-i}  (cross‑correlation)
            h = _cross_corr_weighted(a, w)             # length n
            grad_a = 2.0 * h

            # ------- chain rule to logits b (soft‑max) -----------------------
            dot = np.dot(a, grad_a)                    # scalar ⟨a,∇_a J⟩
            grad_b = a * (grad_a - dot)                # ∇_b J

            # ------- Adam update ---------------------------------------------
            m = beta1 * m + (1.0 - beta1) * grad_b
            v = beta2 * v + (1.0 - beta2) * (grad_b ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            b -= lr * m_hat / (np.sqrt(v_hat) + eps)

            # ------- slowly increase smoothing parameter ----------------------
            if rho < rho_max and (t % 50 == 0):
                rho = min(rho + rho_step, rho_max)

    # ------------------------------------------------------------------
    # 4. Final safeguard – if nothing was generated, fall back to uniform.
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)

    # Return a simple Python list with sum = 1 (evaluate_sequence rescales anyway)
    final_seq = (best_seq / best_seq.sum()).tolist()
    final_obj = _objective(np.asarray(final_seq))
    print(f"Final best objective = {final_obj:.6f} (length {n})")
    return final_seq
