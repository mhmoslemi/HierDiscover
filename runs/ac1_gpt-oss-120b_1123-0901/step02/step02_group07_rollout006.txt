def _gradient_smoothed(seq, beta):
    conv = _self_convolve(seq)
    maxc = conv.max()
    w = np.exp(beta * (conv - maxc))

def _smooth_gradient(seq, beta):

    conv = _self_convolve(seq)
    maxc = conv.max()

"""
Search for a step‑function (non‑negative sequence) that minimises

    2·n·max(conv(a,a)) / (sum a)^2

using a smoothed‑max gradient descent on the probability simplex.
The algorithm consists of three stages:

1.  A broad parametric sweep (geometric, power‑law, beta‑like,
    Gaussian and several Dirichlet draws) to obtain a good seed.

2.  An aggressive projected gradient descent where the gradient of a
    *soft‑max* of the self‑convolution is used.  The smoothing parameter
    β is increased slowly so that early iterations move the mass
    globally while later iterations focus on the true maximal entry.
    A back‑tracking line‑search guarantees monotone improvement.
    Random tiny mass‑transfer moves and occasional restarts help escape
    plateaus.

3.  (Optionally) enlarge the support by interpolating the current best
    sequence to a longer length and continue the descent – this
    exploits the fact that the objective contains a factor `n`, allowing
    larger sequences to achieve a smaller bound.

The routine respects the 1000 s time budget, logs progress and returns
the best sequence found as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection ----------------------------------------------------
def _project_onto_simplex(v):
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        # all entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _self_convolve(seq: np.ndarray) -> np.ndarray:
    """Full convolution of seq with itself (length 2·n‑1) using real FFT."""
    n = seq.size
    m = 2 * n - 1
    nfft = _next_pow2(m)
    A = np.fft.rfft(seq, n=nfft)
    conv = np.fft.irfft(A * A, n=nfft)[:m]
    return conv


# ----------------------------------------------------------------------
# Fast local objective --------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv)/sum(seq)^2 for a non‑negative seq."""
    s = seq.sum()
    if s < 0.01:           # protects against division by near‑zero
        return float("inf")
    conv = _self_convolve(seq)
    return float(2.0 * seq.size * conv.max() / (s * s))


# ----------------------------------------------------------------------
# Smoothed gradient of the max‑convolution -------------------------------
def _smooth_gradient(seq: np.ndarray, beta: float) -> np.ndarray:
    """
    Gradient of `max(conv(seq,seq))` approximated by the gradient of a
    soft‑max with temperature 1/β.
    """
    n = seq.size
    conv = _self_convolve(seq)                # length 2n‑1
    maxc = conv.max()
    # soft‑max weights (max weight = 1)
    w = np.exp(beta * (conv - maxc))

    # gradient = 2 * correlation(w, seq)
    rev_seq = seq[::-1]
    L = conv.size                              # = 2n‑1
    nfft = _next_pow2(L + n - 1)               # length for convolution
    W = np.fft.rfft(w, n=nfft)
    R = np.fft.rfft(rev_seq, n=nfft)
    corr = np.fft.irfft(W * R, n=nfft)[:L + n - 1]   # length 3n‑2

    # grad_j = 2 * corr[L‑1‑j]   (see derivation in analysis)
    grad = 2.0 * corr[L - 1 - np.arange(n)]
    return grad


# ----------------------------------------------------------------------
# Parametric families ----------------------------------------------------
def _geom_seq(n: int, alpha: float) -> np.ndarray | None:
    """Geometric progression a_i ∝ r^i,  r = 1‑α/n."""
    r = 1.0 - alpha / n
    if r <= 0.0:
        return None
    powers = r ** np.arange(n, dtype=np.float64)
    return powers / powers.sum()


def _powerlaw_seq(n: int, p: float) -> np.ndarray:
    """Power‑law a_i ∝ (i+1)^{‑p}."""
    seq = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    return seq / seq.sum()


def _beta_seq(n: int, a: float, b: float) -> np.ndarray:
    """Beta‑like shape a_i ∝ (i+1)^{a‑1}·(n‑i)^{b‑1}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    seq = (i ** (a - 1.0)) * ((n + 1 - i) ** (b - 1.0))
    seq[seq < 0] = 0.0
    return seq / seq.sum()


def _gaussian_seq(n: int, sigma: float) -> np.ndarray:
    """Discrete centred Gaussian."""
    x = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
    seq = np.exp(-0.5 * (x / sigma) ** 2)
    return seq / seq.sum()


def _random_dirichlet(n: int, concentration: float) -> np.ndarray:
    """Dirichlet(α,…,α) with α = concentration."""
    if concentration <= 0:
        concentration = 1e-3
    g = np.random.gamma(concentration, 1.0, size=n)
    return g / g.sum()


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a low‑objective step‑function.
    Returns a list of floats (the best sequence found before the budget runs out).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # 1. Choose an initial length (a compromise between expressivity and speed)
    n = 2000

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 2. If a global seed exists, use it
    if "height_sequence_1" in globals():
        try:
            init = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            init = np.clip(init, 0.0, 1000.0)
            if init.sum() > 0.0:
                init = init / init.sum()
                sc = _objective(init)
                if sc < best_score:
                    best_score = sc
                    best_seq = init.copy()
                    print(f"[Init] global seed score={sc:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 3. Broad parametric sweep (stops early if time runs out)
    # -------------------------------------------------- geometric
    for alpha in np.linspace(0.5, 5.0, 20):
        if time.time() > deadline:
            break
        seq = _geom_seq(n, alpha)
        if seq is None:
            continue
        sc = _objective(seq)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(f"[Init] geometric α={alpha:.3f}  score={sc:.6f}")

    # -------------------------------------------------- power‑law
    for p in np.linspace(0.1, 2.5, 20):
        if time.time() > deadline:
            break
        seq = _powerlaw_seq(n, p)
        sc = _objective(seq)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(f"[Init] power‑law p={p:.3f}  score={sc:.6f}")

    # -------------------------------------------------- beta‑like
    for a in np.linspace(0.4, 3.0, 7):
        for b in np.linspace(0.4, 3.0, 7):
            if time.time() > deadline:
                break
            seq = _beta_seq(n, a, b)
            sc = _objective(seq)
            if sc < best_score:
                best_score = sc
                best_seq = seq.copy()
                print(f"[Init] beta a={a:.2f} b={b:.2f}  score={sc:.6f}")

    # -------------------------------------------------- Gaussian
    for sigma_factor in np.linspace(0.2, 0.5, 7):
        if time.time() > deadline:
            break
        sigma = sigma_factor * n
        seq = _gaussian_seq(n, sigma)
        sc = _objective(seq)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(f"[Init] gaussian σ={sigma:.1f}  score={sc:.6f}")

    # -------------------------------------------------- Dirichlet draws (various concentrations)
    for conc in [0.2, 0.5, 1.0, 2.0, 5.0]:
        for _ in range(10):
            if time.time() > deadline:
                break
            seq = _random_dirichlet(n, conc)
            sc = _objective(seq)
            if sc < best_score:
                best_score = sc
                best_seq = seq.copy()
                print(f"[Init] Dirichlet α={conc:.2f}  score={sc:.6f}")

    # -------------------------------------------------- pure random points
    for _ in range(30):
        if time.time() > deadline:
            break
        seq = np.random.exponential(1.0, size=n)
        seq = seq / seq.sum()
        sc = _objective(seq)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(f"[Init] random draw score={sc:.6f}")

    # ------------------------------------------------------------------
    # 4. Fallback – uniform (should never be needed)
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)
        best_score = _objective(best_seq)

    # ------------------------------------------------------------------
    # 5. Local optimisation – smoothed‑max projected gradient descent
    seq = best_seq.copy()
    cur_score = best_score
    it = 0
    base_step = 1.0                     # initial step size for line‑search
    beta = 0.5                          # smoothing parameter (starts low)
    beta_max = 30.0
    no_improve = 0

    while time.time() < deadline:
        # ----- compute gradient of the soft‑max of the convolution -----
        grad = _smooth_gradient(seq, beta)

        # ----- back‑tracking line‑search on the simplex -----
        step = base_step
        improved = False
        while step > 1e-8:
            cand = _project_onto_simplex(seq - step * grad)
            cand_score = _objective(cand)
            if cand_score < cur_score - 1e-12:   # strict improvement
                improved = True
                break
            step *= 0.5

        if improved:
            seq = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = seq.copy()
                print(f"[Iter {it:06d}] new best = {best_score:.6f}")
            # adapt step size and smoothing
            base_step = max(step * 1.2, 1e-4)
            beta = min(beta * 1.02, beta_max)
            no_improve = 0
        else:
            # ----- no improvement: small random mass‑transfer -----
            i, j = np.random.choice(n, size=2, replace=False)
            delta = 0.01 * min(seq[i], 0.1)   # move at most 1 % of total mass
            if delta > 0:
                cand = seq.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = _project_onto_simplex(cand)
                cand_score = _objective(cand)
                if cand_score < cur_score - 1e-12:
                    seq = cand
                    cur_score = cand_score
                    if cand_score < best_score:
                        best_score = cand_score
                        best_seq = seq.copy()
                        print(f"[Iter {it:06d}] random move improved to {best_score:.6f}")
                    base_step = max(base_step * 0.9, 1e-4)
                    beta = max(beta * 0.98, 0.5)
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1

        # ----- occasional restart if stuck for a while -----
        if no_improve >= 300:
            # perturb the current best a little and restart from it
            noise = np.random.exponential(0.001, size=n)
            seq = best_seq + noise
            seq = _project_onto_simplex(seq)
            cur_score = _objective(seq)
            base_step = 1.0
            beta = 0.5
            no_improve = 0
            print(f"[Iter {it:06d}] restart after plateau (score={cur_score:.6f})")

        # ----- optional expansion to a larger support (one‑time) -----
        # If we still have ample time and have not yet expanded, double the length.
        if n < 4000 and (time.time() < deadline - 200) and it % 5000 == 0 and it > 0:
            new_n = min(4000, int(n * 1.5))
            print(f"[Iter {it:06d}] expanding from {n} → {new_n}")
            # Linear interpolation onto the larger grid
            new_seq = np.interp(np.linspace(0, 1, new_n),
                                np.linspace(0, 1, n),
                                seq)
            new_seq = _project_onto_simplex(new_seq)
            seq = new_seq
            n = new_n
            cur_score = _objective(seq)
            # reset optimisation parameters for the new dimension
            base_step = 1.0
            beta = 0.5
            no_improve = 0

        it += 1

    # ------------------------------------------------------------------
    print(f"Final best objective = {best_score:.6f} (length {len(best_seq)})")
    # The evaluator normalises internally, so returning a simplex vector is fine.
    return best_seq.tolist()
