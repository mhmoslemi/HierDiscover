def subgradient_descent(a0, target_sum, max_seconds, start_time):
    a = a0.copy()
    n = a.shape[0]

    # Adam hyperparameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.2  # base learning rate
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    best_a = a.copy()
    conv = conv_full_fft(a)
    best_val = float(conv.max())
    # output? we may print.
    # But we should also keep variable for best_val_local.

    # loop
    while True:
        now = time.time()
        if now - start_time >= max_seconds:
            break
        # Compute convolution
        conv = conv_full_fft(a)
        max_val = float(conv.max())
        # Update best if improvement
        if max_val < best_val:
            best_val = max_val
            best_a = a.copy()
        # Determine active indices
        eps_act = 1e-12
        active = np.where(conv >= max_val - eps_act)[0]
        # Compute gradient
        grad = gradient_of_active(a, active)
        # If zero gradient, break
        if grad.max() <= 0:
            break
        # Adam update
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        # Step direction: negative gradient
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        # Update a with step
        a_new = a - step
        # Clip to non-negative
        a_new = np.maximum(a_new, 0.0)
        # Project onto simplex
        a_new = project_onto_simplex(a_new, target_sum)

        # Occasionally add random noise
        if np.random.rand() < 0.02:
            noise = np.random.randn(n) * 0.001 * target_sum / n
            a_new = project_onto_simplex(np.maximum(a_new + noise, 0.0), target_sum)
            # optionally decay learning rate
            lr *= 0.95

        a = a_new

        # decay learning rate slowly
        lr *= 0.9999

    return best_a, best_val

def gradient_of_active(a, active_idxs):
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in active_idxs:
        low = max(0, k - (n - 1))
        high = min(n - 1, k)
        idx = np.arange(low, high + 1, dtype=int)
        partner = k - idx
        grad[idx] += 2.0 * a[partner]
    return grad

"""
search_step_function.py

A time‑bounded multi‑restart optimizer for the non‑negative step‑function
problem

    minimize 2 * n * max(conv(a, a)) / (sum(a)**2)

The objective is scale‑invariant; we therefore fix the ℓ₁‑norm of the
vector to √(2 n).  The algorithm consists of many independent runs.
Each run performs a projected Adam‑style sub‑gradient descent on the
maximum entry of the self‑convolution, with occasional random “shakes”
to escape plateaus.  The best sequence found over the whole budget is
returned as a plain Python list.

All helper functions are defined at module level (no closures) and the
search respects the 1000 s time limit.
"""

import time
import numpy as np

# --------------------------------------------------------------------------- #
# Simplex projection (Euclidean) – {x_i ≥ 0, Σ x_i = s}
# --------------------------------------------------------------------------- #
def project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector ``v`` onto the simplex of radius ``s``."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    # Sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------------------- #
# Full linear convolution using real FFT (returns length 2*n‑1 array)
# --------------------------------------------------------------------------- #
def conv_full_fft(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    # next power of two for efficient FFT
    n_fft = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    conv = np.fft.irfft(A * A, n=n_fft)[: 2 * n - 1]
    return conv


# --------------------------------------------------------------------------- #
# Sub‑gradient of the maximum convolution entry.
# For each active index k, ∂max/∂a_i receives 2*a_{k‑i} whenever i participates.
# --------------------------------------------------------------------------- #
def gradient_of_active(a: np.ndarray, active_idxs: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in active_idxs:
        low = max(0, k - (n - 1))
        high = min(n - 1, k)
        idx = np.arange(low, high + 1, dtype=int)
        partner = k - idx
        grad[idx] += 2.0 * a[partner]
    return grad


# --------------------------------------------------------------------------- #
# Compute the objective (max convolution) for a given vector.
# --------------------------------------------------------------------------- #
def max_convolution(a: np.ndarray) -> float:
    """Return max(conv(a, a)) – the objective after fixing Σa = √(2n)."""
    return float(conv_full_fft(a).max())


# --------------------------------------------------------------------------- #
# Adam‑style projected sub‑gradient descent with periodic random shakes.
# --------------------------------------------------------------------------- #
def subgradient_descent_adam(
    a0: np.ndarray,
    target_sum: float,
    time_budget: float,
    start_time: float,
) -> tuple[np.ndarray, float]:
    """
    Perform a projected Adam sub‑gradient descent on ``a0``.
    Returns the best vector seen and its objective value.
    """
    a = a0.copy()
    n = a.shape[0]

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.02                # base learning rate (tuned empirically)
    m = np.zeros_like(a)     # first moment
    v = np.zeros_like(a)     # second moment
    t = 0

    best_a = a.copy()
    best_val = max_convolution(a)

    # Tolerance for deciding which entries are “active”
    act_eps = 1e-12

    # Main loop – stops when the allotted time expires
    while True:
        now = time.time()
        if now - start_time >= time_budget:
            break

        # === evaluate current point ===
        conv = conv_full_fft(a)
        cur_max = float(conv.max())

        if cur_max < best_val:
            best_val = cur_max
            best_a = a.copy()
            print(
                f"[{now:.2f}]  Local best = {best_val:.6f} (n={n})"
            )

        # === sub‑gradient of the active entries ===
        active = np.where(conv >= cur_max - act_eps)[0]
        grad = gradient_of_active(a, active)

        if grad.max() <= 0.0:
            # No direction to descend
            break

        # === Adam update (projected) ===
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step                      # move opposite to the sub‑gradient
        a_new = np.maximum(a_new, 0.0)         # enforce non‑negativity
        a_new = project_onto_simplex(a_new, target_sum)

        # === occasional random shake (helps escape plateaus) ===
        if np.random.rand() < 0.02:
            noise = np.random.randn(n) * (0.01 * target_sum / n)
            a_new = project_onto_simplex(
                np.maximum(a_new + noise, 0.0), target_sum
            )
            lr *= 0.9   # slowly shrink learning rate after a shake

        a = a_new

        # Decay the base learning rate very gently
        lr *= 0.99995

    return best_a, best_val


# --------------------------------------------------------------------------- #
# Generate a random initial vector of a given length.
# --------------------------------------------------------------------------- #
def generate_initial_vector(rng: np.random.Generator, n: int, mode: str) -> np.ndarray:
    """
    Produce an initial vector of length ``n``.
    ``mode`` chooses the distribution:
        - 'uniform'  : Uniform(0,1)
        - 'exp'      : Exponential decay e^{-λ i}
        - 'pow'      : Power law (i+1)^(-p)
        - 'copy'     : Truncate / pad from the current global best if available
    """
    if mode == "uniform":
        return rng.random(n)
    if mode == "exp":
        lam = rng.uniform(0.001, 0.03)          # decay rate
        return np.exp(-lam * np.arange(n, dtype=float))
    if mode == "pow":
        p = rng.uniform(0.5, 2.5)               # power‑law exponent
        return (np.arange(1, n + 1, dtype=float) ** (-p))
    if mode == "copy":
        # If a global best exists, reuse it (truncate or pad)
        if "height_sequence_1" in globals():
            src = np.asarray(height_sequence_1, dtype=float)
            if src.size >= n:
                return src[:n].copy()
            else:
                pad = rng.random(n - src.size) * 1e-3
                return np.concatenate([src, pad])
        else:
            # fall back to uniform if no global best is present
            return rng.random(n)
    # default
    return rng.random(n)


# --------------------------------------------------------------------------- #
# Main entry point expected by the grader.
# --------------------------------------------------------------------------- #
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs) -> list:
    """
    Search for a low‑value step‑function within ``budget_s`` seconds.
    Returns the best sequence (list of floats) discovered.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5   # tiny safety margin

    best_seq = None
    best_score = float("inf")

    # Main outer loop – many independent restarts with varying lengths
    while time.time() < deadline:
        # ------------------------------------------------------------------- #
        # Choose a problem size.  We mix medium and larger sizes.
        # ------------------------------------------------------------------- #
        if rng.random() < 0.8:
            n = rng.integers(1500, 3001)          # typical range used before
        else:
            n = rng.integers(3000, 6001)          # occasional larger instance

        target_sum = np.sqrt(2.0 * n)               # Σ a_i = √(2n)

        # ------------------------------------------------------------------- #
        # Initialise the vector – several random “modes” are tried.
        # ------------------------------------------------------------------- #
        mode = rng.choice(["uniform", "exp", "pow", "copy"], p=[0.4, 0.2, 0.2, 0.2])
        init_vec = generate_initial_vector(rng, n, mode)
        a0 = project_onto_simplex(init_vec.astype(float), target_sum)

        # ------------------------------------------------------------------- #
        # Allocate a time slice for this run.
        # ------------------------------------------------------------------- #
        remaining = deadline - time.time()
        # Give each run at most 30 s, but never exceed the remaining budget.
        run_budget = min(30.0, remaining)
        if run_budget < 5.0:   # not enough time for a useful run
            break

        # ------------------------------------------------------------------- #
        # Run the Adam‑style sub‑gradient descent.
        # ------------------------------------------------------------------- #
        a_opt, val_opt = subgradient_descent_adam(
            a0, target_sum, run_budget, time.time()
        )

        # ------------------------------------------------------------------- #
        # Update the global best if we improved.
        # ------------------------------------------------------------------- #
        if val_opt < best_score:
            best_score = val_opt
            best_seq = a_opt.copy()
            print(
                f"[{time.time():.2f}]  New global best = {best_score:.6f} (n={n})"
            )
            # Early exit if we reach the target bound.
            if best_score < 1.503:
                break

    # ------------------------------------------------------------------- #
    # Fallback – should never happen, but keep a safe default.
    # ------------------------------------------------------------------- #
    if best_seq is None:
        n = 2000
        target_sum = np.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)

    # Convert to a plain Python list as required by the grader.
    return best_seq.tolist()
