"""
Propose a step‑function (non‑negative vector) with a low value of

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm consists of many short independent runs.  For each run

* a length ``n`` (800 … 4500) is chosen,
* a simple power‑law initialization is used (occasionally warm‑started
  from the global best that has been found so far),
* a **smooth‑max** surrogate of the peak of the auto‑convolution is
  minimised by projected (momentum) gradient descent.
  The smooth‑max is ``log‑sum‑exp``; its gradient can be written as a
  convolution and is evaluated by FFT, i.e. each iteration costs
  ``O(n log n)`` only.
* the temperature of the smooth‑max (parameter ``theta``) is increased
  gradually – a continuation method that first moves globally and then
  focuses on the true maximum.
* occasional random jitter helps escaping plateaus.

All iterates are projected onto the simplex
    {x_i ≥ 0, Σ x_i = √(2 n)},
which makes the evaluation function exactly equal to the current
maximum of the convolution.  The best sequence seen across all runs is
returned.

The routine respects the 1000 s budget, prints occasional progress
messages and never uses closures or lambdas.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# 1. Euclidean projection onto the simplex {x≥0, Σx = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector *v* onto the simplex of radius *s*."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# 2. Generic convolution via FFT (real inputs)
# ----------------------------------------------------------------------
def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution x * y using FFT (real valued)."""
    n = x.shape[0]
    m = y.shape[0]
    size = 1 << ((n + m - 2).bit_length())  # next power of two
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[: n + m - 1]
    # eliminate tiny negative round‑off
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# 3. Smooth‑max of the auto‑convolution and its gradient
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, theta: float):
    """
    Returns (max_conv, gradient) where the objective is the smooth
    max (log‑sum‑exp) of the auto‑convolution of *a*.
    Gradient is the exact derivative of the smooth‑max.
    """
    n = a.shape[0]

    # 1) auto‑convolution a*a
    conv = _conv_fft(a, a)                     # length 2n‑1
    max_val = conv.max()

    # 2) soft‑max weights (numerically stable)
    #    w_k = exp(theta * (conv_k - max_val)) / Z
    #    Z = sum_k exp(...)
    #    when theta → ∞ the weights concentrate on the true maximisers
    exp_shift = np.exp(theta * (conv - max_val))
    Z = exp_shift.sum()
    w = exp_shift / Z

    # 3) gradient = 2 * conv(a, w_rev) evaluated at indices 2n‑2‑i
    w_rev = w[::-1]                            # length 2n‑1
    grad_full = _conv_fft(a, w_rev)            # length 3n‑2
    idx = 2 * n - 2 - np.arange(n)             # descending slice
    grad = 2.0 * grad_full[idx]

    return max_val, grad


# ----------------------------------------------------------------------
# 4. Optimisation of a single length (projected gradient descent)
# ----------------------------------------------------------------------
def _optimise_one_length(
    n: int,
    init: np.ndarray,
    run_deadline: float,
    *,
    theta_start: float = 0.1,
    theta_end: float = 30.0,
    base_lr: float = 0.7,
    momentum_beta: float = 0.9,
    total_iters: int = 20000,
) -> tuple[np.ndarray, float]:
    """
    Minimise the smooth‑max surrogate for a fixed length ``n``.
    Returns the best vector found and its (exact) evaluation‑score
    ``2*n*max(conv)/(sum**2)`` (which equals the maximal auto‑convolution
    because the sum is fixed to sqrt(2*n)).
    """
    target_sum = np.sqrt(2.0 * n)               # Σ a_i = √(2n)
    a = _project_onto_simplex(init.astype(np.float64), target_sum)

    # keep the best sequence seen in this run
    conv = _conv_fft(a, a)
    best_local = a.copy()
    best_score = conv.max()                     # equal to the true score

    # momentum buffer
    vel = np.zeros_like(a)

    it = 0
    start_time = time.time()
    while time.time() < run_deadline and it < total_iters:
        # ----- continuation on theta (smoothness) --------------------
        frac = min(it / total_iters, 1.0)
        theta = theta_start * (theta_end / theta_start) ** frac

        # ----- compute smooth‑max and its gradient -------------------
        max_val, grad = _smooth_max_and_grad(a, theta)

        # ----- momentum step -----------------------------------------
        vel = momentum_beta * vel + (1.0 - momentum_beta) * grad
        # normalise step size: we want a change of order ``target_sum / n``
        step_scale = base_lr * max_val / (np.max(np.abs(vel)) + 1e-12)
        a_new = a - step_scale * vel

        # ----- projection onto the simplex (enforces non‑negativity)-
        a_new = _project_onto_simplex(a_new, target_sum)

        # ----- evaluate true (non‑smoothed) objective ----------------
        conv_new = _conv_fft(a_new, a_new)
        score_new = conv_new.max()

        # ----- simple downhill acceptance (always keep improvement)-
        if score_new < best_score:
            best_score = score_new
            best_local = a_new.copy()
            print(f"[{time.time():.2f}] n={n}  new best = {best_score:.6f}")

        # ----- keep the new point if it improves; otherwise accept
        #       with a tiny probability that decays with time (helps escape
        #       shallow plateaus).
        if score_new <= max_val or np.random.rand() < 0.001:
            a = a_new

        # ----- occasional random shake‑up -----------------------------
        if np.random.rand() < 0.015:
            noise = np.random.randn(n) * (target_sum * 0.001 / n)
            a = _project_onto_simplex(np.maximum(a + noise, 0.0), target_sum)
            # after a disturbance we become slightly more cautious
            base_lr *= 0.95

        # ----- anneal learning rate ----------------------------------
        base_lr *= 0.9996

        it += 1

    return best_local, best_score


# ----------------------------------------------------------------------
# 5. Upsampling helper (interpolates an existing vector to a new length)
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of *seq* to length *new_n*."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    x_old = np.arange(old_n, dtype=np.float64)
    x_new = np.linspace(0, old_n - 1, new_n, dtype=np.float64)
    return np.interp(x_new, x_old, seq)


# ----------------------------------------------------------------------
# 6. Main entry point required by the evaluator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative height sequence that minimises the
    evaluation function.  The routine runs for at most ``budget_s`` seconds
    and returns the best sequence discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    best_seq = None
    best_score = float("inf")

    # reasonable range for the length – longer vectors give a lower
    # theoretical bound but are also more expensive.
    min_n, max_n = 800, 4500

    # ------------------------------------------------------------------
    # Main optimisation loop – many short independent restarts.
    # ------------------------------------------------------------------
    while time.time() < deadline:
        remaining = deadline - time.time()

        # Dynamic choice of length: as the deadline shrinks use smaller
        # vectors to keep each run cheap.
        if remaining < 30.0:
            n = np.random.randint(800, 1500)
        else:
            n = np.random.randint(min_n, max_n + 1)

        # ----- initialise -------------------------------------------------
        # power‑law shape a_i ∝ (i+1)^{-p}, p ∈ [0.42, 0.58]
        p = np.random.uniform(0.42, 0.58)
        idx = np.arange(1, n + 1, dtype=np.float64)
        init = idx ** (-p)

        # small jitter to break perfect symmetry
        init += np.random.rand(n) * 1e-5

        # Occasionally warm‑start from the best vector found so far.
        # If the global best is shorter we up‑sample; if it is longer we
        # truncate and add a tiny amount of noise.
        if best_seq is not None and np.random.rand() < 0.45:
            if best_seq.size == n:
                init = best_seq.copy()
            elif best_seq.size < n:
                init = _upsample(best_seq, n)
            else:
                init = best_seq[:n].copy()
            init += np.random.randn(n) * 1e-5   # tiny perturbation

        # ----- allocate a time slice for this run -----------------------
        # 7‑10 s per restart is a good compromise between exploration
        # and depth of optimisation.
        run_deadline = min(deadline, time.time() + np.random.uniform(7.0, 10.0))

        # ----- run the optimiser ----------------------------------------
        seq_local, score_local = _optimise_one_length(
            n,
            init,
            run_deadline,
            theta_start=0.1,
            theta_end=30.0,
            base_lr=0.7,
            momentum_beta=0.9,
            total_iters=20000,
        )

        # ----- update global best ----------------------------------------
        if score_local < best_score:
            best_score = score_local
            best_seq = seq_local.copy()
            print(
                f"[{time.time():.2f}] GLOBAL new best = {best_score:.6f} (length={n})"
            )

        # Early exit if we already beat the target comfortably.
        if best_score < 1.45:    # target is 1.503, give some margin
            break

    # ------------------------------------------------------------------
    # Fallback – in the unlikely case that nothing was found.
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        target_sum = np.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=np.float64)

    # Return a plain Python list as required by the evaluator.
    return best_seq.tolist()
