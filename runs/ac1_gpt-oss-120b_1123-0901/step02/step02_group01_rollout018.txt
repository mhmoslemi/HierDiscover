"""
Search for a low‑value step‑function using exponentiated (multiplicative) mirror
descent on a smooth‑max surrogate of the max‑self‑convolution.
The algorithm works on the probability simplex (∑a_i = 1, a_i ≥ 0) so the
objective coincides with that used by `evaluate_sequence`.

Key ideas
---------
* Initialise from several simple families (geometric, power‑law, Dirichlet)
  and keep the best start point.
* Smooth the hard max with a temperature λ:
        φ_λ(a) = (1/λ)·log∑_k exp(λ·(a*a)[k]).
  Its Euclidean gradient is   ∇φ_λ(a) = 2·(w ★ a)  with
        w_k = exp(λ·c_k) / ∑_j exp(λ·c_j)   (c = a*a).
* Perform mirror descent on the simplex:
        a_i ← a_i·exp(‑η·∇φ_λ(a)_i)   and renormalise.
  This automatically respects non‑negativity and the sum‑to‑one constraint.
* Increase λ gradually (continuation) and decay the step size η.
* If progress stalls, inject a tiny random perturbation (“shake”) and,
  after a longer pause, restart from a fresh random Dirichlet point.
* All convolutions are carried out with FFTs (O(n log n)).

The routine runs for at most `budget_s` seconds (default 1000 s) and
returns the best sequence found as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Small utility – next power of two ≥ x."""
    return 1 << (x - 1).bit_length()


# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray, size: int | None = None) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) via FFT."""
    n = a.shape[0]
    if size is None:
        size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective used by evaluate_sequence:
       2·n·max(a*a) / (∑a)^2  (a assumed non‑negative)."""
    n = a.shape[0]
    conv = _full_convolution(a)
    max_b = conv.max()
    sum_a = a.sum()
    # sum_a is forced to be > 0 by the caller.
    return 2.0 * n * max_b / (sum_a * sum_a)


# ----------------------------------------------------------------------
def _smooth_max_weights(conv: np.ndarray, lam: float) -> np.ndarray:
    """Soft‑max weights w_k = exp(λ·conv_k) / Σ exp(λ·conv)."""
    # subtract the max for numerical stability
    shift = conv.max()
    w = np.exp(lam * (conv - shift))
    w_sum = w.sum()
    if w_sum == 0.0:
        return np.full_like(conv, 1.0 / conv.size)
    return w / w_sum


# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray,
                      conv: np.ndarray,
                      lam: float,
                      size_corr: int) -> np.ndarray:
    """
    Gradient of the smooth‑max surrogate φ_λ.
    Returns 2·(w ★ a) (linear correlation, slice of length len(a)).
    """
    n = a.shape[0]
    w = _smooth_max_weights(conv, lam)

    # linear cross‑correlation of w with a (reversed)
    rev_a = a[::-1]
    fw = np.fft.rfft(w, size_corr)
    frev = np.fft.rfft(rev_a, size_corr)
    corr = np.fft.irfft(fw * frev, size_corr)

    # slice that corresponds to indices 0 … n‑1 of the original vector
    grad = 2.0 * corr[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    Returns the best sequence (plain Python list) found within the time budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # 1. Choose a problem size.
    #    2000–3000 gives a good trade‑off between expressiveness and speed.
    n = 2500

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # 2. Warm‑start from a few simple families.
    #    (geometric, power‑law, random Dirichlet)
    candidates = []

    # geometric families a_i ∝ r^i  (r close to 1)
    for r in np.linspace(0.995, 0.9999, 6):
        powers = np.power(r, np.arange(n, dtype=np.float64))
        seq = powers / powers.sum()
        candidates.append(seq)

    # power‑law families a_i ∝ (i+1)^{-p}
    for p in [0.5, 0.8, 1.0, 1.2, 1.5, 2.0]:
        idx = np.arange(1, n + 1, dtype=np.float64)
        seq = idx ** (-p)
        seq = seq / seq.sum()
        candidates.append(seq)

    # a few random Dirichlet points
    for _ in range(12):
        seq = np.random.dirichlet(np.ones(n))
        candidates.append(seq)

    # keep the best of the cheap initialisations
    for seq in candidates:
        if time.time() > deadline:
            break
        val = _objective(seq)
        if val < best_val:
            best_val, best_seq = val, seq.copy()
            print(f"[Init] obj={val:.6f}")

    # ------------------------------------------------------------------
    # 3. Mirror‑descent optimisation on the smooth‑max surrogate.
    #    We work on the probability simplex (∑a = 1) – scaling is irrelevant.
    a = best_seq.copy()
    lam = 2.0               # start with a very smooth approximation
    lam_factor = 1.05       # accelerate the continuation
    eta = 0.6               # initial multiplicative step size
    eta_decay = 0.9996      # slow decay
    size_fft = _next_pow2(2 * n - 1)          # for self‑convolution
    size_corr = _next_pow2(3 * n)             # for correlation (len(w)+n‑1)
    it = 0
    last_improve = time.time()

    while time.time() < deadline:
        # 3a) self‑convolution
        conv = _full_convolution(a, size=size_fft)

        # 3b) gradient of the smooth‑max surrogate
        grad = _softmax_gradient(a, conv, lam, size_corr=size_corr)

        # 3c) exponentiated (mirror) update on the simplex
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0.0:                     # safety fallback
            a = np.random.dirichlet(np.ones(n))
        else:
            a /= s

        # 3d) evaluate the true objective (exact)
        val = _objective(a)

        if val < best_val:
            best_val, best_seq = val, a.copy()
            last_improve = time.time()
            print(f"[Iter {it:06d}] new best obj={val:.6f}  λ={lam:.3f}  η={eta:.5f}")

        # 3e) occasional random “shake” to escape shallow plateaus
        if time.time() - last_improve > 2.0:
            noise = (np.random.rand(n) - 0.5) * 1e-3
            a = a + noise
            a = np.maximum(a, 0.0)
            a_sum = a.sum()
            if a_sum == 0.0:
                a = np.random.dirichlet(np.ones(n))
            else:
                a /= a_sum
            last_improve = time.time()

        # 3f) longer stagnation → restart from a fresh random point
        if time.time() - last_improve > 20.0:
            print(f"[Iter {it:06d}] stagnation – restarting from scratch")
            a = np.random.dirichlet(np.ones(n))
            lam = 2.0
            eta = 0.6
            last_improve = time.time()

        # 3g) schedule updates
        lam *= lam_factor
        eta *= eta_decay
        it += 1

    # ------------------------------------------------------------------
    # 4. Final report
    print(f"Final best objective = {best_val:.6f} (length {n})")
    # Convert to a plain Python list as required by the evaluator
    return best_seq.tolist()
