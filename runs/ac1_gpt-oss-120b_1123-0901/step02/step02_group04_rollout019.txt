"""
propose_candidate
-----------------

A fresh optimisation pipeline for the step‑function bound.

The routine consists of three parts:

1.  Initialise with the best known construction (if available) or a
    simple exponential shape, projected onto the simplex
    Σa_i = √(2 n).

2.  Smooth‑max gradient descent.  The non‑smooth objective
    max(conv(a,a)) is replaced by the smooth surrogate
        M_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k)
    for a sequence of increasing β.  The exact gradient of M_β can be
    written as a cross‑correlation; both the autocorrelation and the
    gradient are evaluated by real FFTs (O(n log n)).  After each step
    the vector is projected back onto the simplex.  The current best
    candidate is recorded periodically.

3.  Optional linear‑programming refinement (the “LP‑step’’) using
    SciPy, applied a few times to the best sequence found so far.

All heavy work respects the global time budget (default 1000 s) and
updates the global best via a top‑level helper – no closures are used.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Global state (updated by the helper functions)
# ----------------------------------------------------------------------
_GLOBAL_BEST_SEQ = None          # type: np.ndarray | None
_GLOBAL_BEST_SCORE = float("inf")
_GLOBAL_DEADLINE = 0.0           # absolute timestamp when we must stop

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the simplex {x ≥ 0, Σx = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper: full autocorrelation a * a (real FFT, length 2n‑1)
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    """Return the full convolution a * a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 2).bit_length())  # next power of two ≥ 2n‑1
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Helper: gradient of the smooth‑max (cross‑correlation w ⋆ a)
# ----------------------------------------------------------------------
def _conv_weight_rev(w: np.ndarray, a_rev: np.ndarray) -> np.ndarray:
    """
    Compute the cross‑correlation (w ⋆ a)[j] = Σ_k w[k]·a[k‑j]
    via convolution w * a_rev.  Returns an array of length len(a_rev)
    containing the values for j = 0 … n‑1.
    """
    L = w.shape[0] + a_rev.shape[0] - 1          # = 3n‑2
    size = 1 << ((L - 1).bit_length())            # next power of two
    fw = np.fft.rfft(w, n=size)
    fr = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(fw * fr, n=size)
    conv = conv[:L]                               # trim excess tail
    start = a_rev.shape[0] - 1                    # = n‑1
    return conv[start:start + a_rev.shape[0]]


# ----------------------------------------------------------------------
# Helper: evaluate a candidate and update the global best if improved
# ----------------------------------------------------------------------
def _maybe_update_candidate(seq: np.ndarray) -> None:
    """Evaluate `seq` (list of floats) and keep it if it improves the best."""
    global _GLOBAL_BEST_SEQ, _GLOBAL_BEST_SCORE, _GLOBAL_DEADLINE
    if time.time() >= _GLOBAL_DEADLINE:
        return
    try:
        score = evaluate_sequence(seq.tolist())
    except Exception:
        score = float("inf")
    if score < _GLOBAL_BEST_SCORE:
        _GLOBAL_BEST_SCORE = score
        _GLOBAL_BEST_SEQ = seq.copy()
        print(f"[{time.time():.2f}] New best = {score:.6f} (len={len(seq)})")


# ----------------------------------------------------------------------
# Gradient descent on the smooth‑max objective
# ----------------------------------------------------------------------
def _run_smoothgd(init_arr: np.ndarray) -> None:
    """
    Perform many gradient‑descent steps on `init_arr`
    (already projected onto the simplex).  The routine updates the
    global best whenever a better sequence is encountered.
    """
    global _GLOBAL_DEADLINE, _GLOBAL_BEST_SCORE
    a = init_arr.astype(float, copy=True)
    n = a.shape[0]
    target = math.sqrt(2.0 * n)

    # keep the initial point on the simplex (just in case)
    a = _project_onto_simplex(a, target)
    _maybe_update_candidate(a)

    base_lr = 0.5                # base step size
    it = 0                       # iteration counter

    # schedule of smoothing parameters β (increasing → sharper max)
    beta_schedule = [0.2, 0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 40.0]

    for beta in beta_schedule:
        # cap the number of inner iterations per β (enough for convergence)
        max_iter_beta = 800
        for _ in range(max_iter_beta):
            if time.time() >= _GLOBAL_DEADLINE:
                return

            # autocorrelation and smooth‑max weights
            conv = _conv_self(a)
            max_c = conv.max()
            exp_vals = np.exp(beta * (conv - max_c))          # stabilised softmax
            sum_exp = exp_vals.sum()
            if sum_exp == 0 or np.isnan(sum_exp):
                w = np.full_like(conv, 1.0 / conv.size)
            else:
                w = exp_vals / sum_exp

            # gradient = 2·(w ⋆ a)
            a_rev = a[::-1]
            grad = 2.0 * _conv_weight_rev(w, a_rev)

            # diminishing step size (1/√it) and normalisation
            step = base_lr / math.sqrt(it + 1)
            max_abs = np.max(np.abs(grad))
            if max_abs > 0.0:
                eta = step / max_abs
            else:
                eta = 0.0

            # gradient step and projection
            a = a - eta * grad
            a = _project_onto_simplex(a, target)

            it += 1
            # occasional evaluation (cheap compared to per‑iteration cost)
            if it % 10 == 0:
                _maybe_update_candidate(a)

            # early exit if target bound already reached
            if _GLOBAL_BEST_SCORE < 1.503:
                return


# ----------------------------------------------------------------------
# Optional linear‑programming refinement (standard “LP‑step’’)
# ----------------------------------------------------------------------
def _lp_refine(seq: np.ndarray) -> None:
    """
    Perform a tiny LP‑based refinement on `seq`.
    Maximises Σ g_j under a * g ≤ M where M = max(a*a).
    The resulting g is blended back into a.
    """
    global _GLOBAL_DEADLINE, _GLOBAL_BEST_SCORE
    try:
        from scipy.optimize import linprog
    except Exception:
        return

    a = seq.astype(float, copy=True)
    n = a.shape[0]
    target = math.sqrt(2.0 * n)

    while time.time() < _GLOBAL_DEADLINE - 0.1:
        conv = _conv_self(a)
        M = conv.max()
        rows = 2 * n - 1

        # Build Toeplitz‑like constraint matrix A (a * g ≤ M)
        A = np.zeros((rows, n), dtype=float)
        for k in range(rows):
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            cols = np.arange(lo, hi + 1, dtype=int)
            A[k, cols] = a[k - cols]

        res = linprog(
            c=-np.ones(n),                # maximise Σ g_j
            A_ub=A,
            b_ub=np.full(rows, M),
            bounds=(0, None),
            method="highs",
            options={"presolve": True},
        )
        if not res.success:
            break

        g = res.x
        s = g.sum()
        if s <= 0.0:
            break
        g = g * (target / s)                # re‑scale to the simplex

        # convex combination (small mixing factor)
        t = 0.2
        a = (1.0 - t) * a + t * g

        _maybe_update_candidate(a)

        # stop if we have already crossed the target
        if _GLOBAL_BEST_SCORE < 1.503:
            break


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs) -> list:
    """
    Search for a non‑negative height sequence that minimises
    ``evaluate_sequence`` under the given time budget.
    Returns the best sequence discovered as a plain Python list.
    """
    global _GLOBAL_BEST_SEQ, _GLOBAL_BEST_SCORE, _GLOBAL_DEADLINE

    # ------------------------------------------------------------------
    # Initialisation
    # ------------------------------------------------------------------
    np.random.seed(seed)
    start_time = time.time()
    _GLOBAL_DEADLINE = start_time + budget_s - 0.5   # tiny safety margin
    _GLOBAL_BEST_SEQ = None
    _GLOBAL_BEST_SCORE = float("inf")

    # ------------------------------------------------------------------
    # 1.  Start from the known construction if it exists
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        seq0 = np.array(height_sequence_1, dtype=float)
        n0 = len(seq0)
        target0 = math.sqrt(2.0 * n0)
        seq0 = _project_onto_simplex(seq0, target0)
        _maybe_update_candidate(seq0)
        _run_smoothgd(seq0)

    # ------------------------------------------------------------------
    # 2.  Randomised restarts with smooth‑max gradient descent
    # ------------------------------------------------------------------
    while time.time() < _GLOBAL_DEADLINE:
        # pick a moderate length (large enough to give a good bound,
        # small enough to stay fast)
        n = np.random.randint(800, 2501)               # 800 … 2500
        target = math.sqrt(2.0 * n)

        # initialise with a random exponential tail plus tiny noise
        alpha = np.random.rand() * 4.0 + 0.001          # 0.001 … 4.001
        shape = np.exp(-alpha * np.arange(n, dtype=float))
        shape += 0.01 * np.random.rand(n)               # tiny perturbation
        shape = shape / shape.sum() * target
        shape = np.maximum(shape, 0.0)

        _maybe_update_candidate(shape)
        _run_smoothgd(shape)

        # stop early if we already beat the target bound
        if _GLOBAL_BEST_SCORE < 1.503:
            break

    # ------------------------------------------------------------------
    # 3.  Optional LP refinement on the best sequence found so far
    # ------------------------------------------------------------------
    if _GLOBAL_BEST_SEQ is not None and time.time() < _GLOBAL_DEADLINE:
        _lp_refine(_GLOBAL_BEST_SEQ)

    # ------------------------------------------------------------------
    # 4.  Fallback – uniform sequence (should never happen)
    # ------------------------------------------------------------------
    if _GLOBAL_BEST_SEQ is None:
        n = 1000
        target = math.sqrt(2.0 * n)
        _GLOBAL_BEST_SEQ = np.full(n, target / n, dtype=float)

    return _GLOBAL_BEST_SEQ.tolist()
