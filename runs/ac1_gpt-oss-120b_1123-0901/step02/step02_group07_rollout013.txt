"""
Propose a step‑function that minimizes ``evaluate_sequence``.

The search combines three complementary ingredients:

1. **Multi‑restart stochastic initialization** – several families
   (geometric, power‑law, beta‑like, Dirichlet) are sampled and the
   best one seeds the optimisation.

2. **Linear‑programming mixing** – given the current sequence ``a`` we
   compute ``M = max(a*a)`` and solve the LP
   ``max 1ᵀg   s.t.   (a*g)[k] ≤ M  (∀k),   g ≥ 0``.
   The LP finds a new direction ``g`` that can add a large amount of
   mass without increasing the maximal convolution.  A convex combination
   ``a←(1‑t)a + t g`` (followed by a projection onto the probability
   simplex) strictly reduces the objective in practice.

3. **Local refinement** – after each LP step we perform a short
   simulated‑annealing / random‑mass‑transfer phase and a simple
   projected sub‑gradient step that targets the currently active shift.
   These moves help to escape shallow local minima.

The whole procedure runs until the 1000 s time budget expires (a small
safety margin is kept).  The best sequence discovered is returned as a
plain Python list.  All numbers are non‑negative and the sequence is
normalised to unit sum; the objective is homogeneous, so any scaling is
equivalent for the external ``evaluate_sequence`` call.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import – used for the LP step.
try:
    from scipy.optimize import linprog
    from scipy.sparse import csr_matrix
    _SCIPY_AVAILABLE = True
except Exception:               # pragma: no cover
    _SCIPY_AVAILABLE = False

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    v = np.maximum(v, 0.0)
    if v.sum() == 0.0:
        return np.full_like(v, 1.0 / len(v))
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _next_pow2(x: int) -> int:
    """Small helper – smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _self_convolve(a: np.ndarray) -> np.ndarray:
    """Full (real) self‑convolution using FFT (length 2·n‑1)."""
    n = a.shape[0]
    m = 2 * n - 1
    nfft = _next_pow2(m)
    A = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(A * A, n=nfft)[:m]
    return conv


def _objective(a: np.ndarray) -> float:
    """Local copy of the objective (identical to evaluate_sequence up to clipping)."""
    s = a.sum()
    if s < 0.01:
        return float("inf")
    conv = _self_convolve(a)
    return float(2.0 * len(a) * conv.max() / (s * s))


# ----------------------------------------------------------------------
def _solve_lp(a: np.ndarray, M: float):
    """
    Solve  max 1ᵀg  s.t. (a*g)[k] ≤ M  for all k, g ≥ 0.
    Returns ``g`` (numpy array) or ``None`` if the LP fails.
    """
    if not _SCIPY_AVAILABLE:               # pragma: no cover
        return None

    n = len(a)
    # build the Toeplitz constraint matrix A_ub (size (2n‑1)×n)
    nnz = n * n
    rows = np.empty(nnz, dtype=np.int32)
    cols = np.empty(nnz, dtype=np.int32)
    data = np.empty(nnz, dtype=np.float64)

    idx = 0
    for j in range(n):
        rows[idx:idx + n] = np.arange(j, j + n, dtype=np.int32)
        cols[idx:idx + n] = j
        data[idx:idx + n] = a
        idx += n

    A_ub = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)          # maximise sum(g) → minimise -sum(g)

    res = linprog(c,
                  A_ub=A_ub,
                  b_ub=b_ub,
                  bounds=[(0, None)] * n,
                  method='highs',
                  options={'presolve': True, 'time_limit': 10})
    if not res.success:
        return None
    g = res.x
    # If the LP returns the trivial all‑zero solution, we are already optimal.
    if g.sum() < 1e-12:
        return None
    return g


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step‑function with a low value of
    ``evaluate_sequence``.  The function respects the 1000 s time budget
    and returns the best sequence found (as a Python list).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5    # safety margin

    # --------------------------------------------------------------
    # 1. Choose a moderate length – large enough for expressive power,
    #    small enough for fast LP solves.
    n = 800

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # Helper: evaluate and potentially update the global best.
    def _maybe_update(seq, score, tag=""):
        nonlocal best_seq, best_score
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"[{tag}] new global best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2. Generate a few seed candidates from simple parametric families.
    def _geom(alpha):
        r = max(0.0, 1.0 - alpha / n)
        if r <= 0.0:
            return None
        p = r ** np.arange(n, dtype=np.float64)
        if p.sum() == 0.0:
            return None
        return p / p.sum()

    def _power(p):
        v = np.arange(1, n + 1, dtype=np.float64) ** (-p)
        return v / v.sum()

    def _beta(a, b):
        i = np.arange(1, n + 1, dtype=np.float64)
        v = (i ** (a - 1.0)) * ((n + 1 - i) ** (b - 1.0))
        v = np.clip(v, 0, None)
        if v.sum() == 0.0:
            return None
        return v / v.sum()

    def _rand_simplex():
        x = np.random.exponential(1.0, size=n)
        return x / x.sum()

    seed_fns = [
        lambda: _geom(np.random.uniform(0.5, 3.0)),
        lambda: _power(np.random.uniform(0.2, 2.5)),
        lambda: _beta(np.random.uniform(0.5, 3.0), np.random.uniform(0.5, 3.0)),
        _rand_simplex,
    ]

    for _ in range(30):
        if time.time() > deadline:
            break
        seq = None
        while seq is None:
            seq = np.random.choice(seed_fns)()
        score = _objective(seq)
        _maybe_update(seq, score, tag="seed")

    # If a global variable ``height_sequence_1`` exists we also try it.
    if "height_sequence_1" in globals():
        try:
            init = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            init = np.clip(init, 0.0, 1000.0)
            if init.sum() > 0.0:
                init = init / init.sum()
                score = _objective(init)
                _maybe_update(init, score, tag="global")
        except Exception:               # pragma: no cover
            pass

    # --------------------------------------------------------------
    # 3. Core optimisation loop – a few outer LP phases, each followed
    #    by a burst of cheap stochastic moves.
    outer_iter = 0
    while time.time() < deadline and outer_iter < 30:
        if best_seq is None:                      # safety fallback
            a = _rand_simplex()
            cur_score = _objective(a)
            _maybe_update(a, cur_score, tag="fallback")
        else:
            a = best_seq.copy()
            cur_score = best_score

        # ---------- LP phase ----------
        conv = _self_convolve(a)
        M = conv.max()
        g = _solve_lp(a, M)
        if g is not None:
            # Mix a little bit of g into a – line‑search on the mixing weight.
            t = 0.5
            mixed = False
            while t > 1e-5:
                cand = (1.0 - t) * a + t * g
                cand = _project_onto_simplex(cand)
                cand_score = _objective(cand)
                if cand_score < cur_score:
                    a = cand
                    cur_score = cand_score
                    _maybe_update(a, cur_score, tag="LPmix")
                    mixed = True
                    break
                t *= 0.5
            if not mixed:
                # If mixing did not help we still keep the LP direction
                # as a candidate for random moves later.
                pass

        # ---------- Local stochastic refinement ----------
        inner = 0
        while time.time() < deadline and inner < 400:
            # Random mass‑transfer move
            i, j = np.random.choice(n, 2, replace=False)
            delta = 0.02 * min(a[i], 0.05)   # move at most a few percent of total mass
            if delta <= 0:
                inner += 1
                continue
            cand = a.copy()
            cand[i] -= delta
            cand[j] += delta
            cand = _project_onto_simplex(cand)
            cand_score = _objective(cand)
            if cand_score < cur_score:
                a = cand
                cur_score = cand_score
                _maybe_update(a, cur_score, tag="rand")
                inner = 0               # reset inner counter after an improvement
                continue

            # Occasionally perform a simple sub‑gradient step
            if inner % 20 == 0:
                conv = _self_convolve(a)
                k_star = int(conv.argmax())
                lo = max(0, k_star - (n - 1))
                hi = min(k_star, n - 1)
                idx = np.arange(lo, hi + 1, dtype=int)
                grad = np.zeros_like(a)
                grad[idx] = a[k_star - idx]            # reversed fragment
                # step opposite to the gradient (projected)
                step = 0.5
                cand = _project_onto_simplex(a - step * grad)
                cand_score = _objective(cand)
                if cand_score < cur_score:
                    a = cand
                    cur_score = cand_score
                    _maybe_update(a, cur_score, tag="grad")
                    inner = 0
                    continue
            inner += 1

        outer_iter += 1

    # --------------------------------------------------------------
    if best_seq is None:          # extremely unlikely, but keep program safe
        best_seq = np.full(n, 1.0 / n)

    print(f"Final best objective = {best_score:.6f} (length {n})")
    return best_seq.tolist()
