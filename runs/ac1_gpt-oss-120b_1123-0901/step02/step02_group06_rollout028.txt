def subgradient_maxconv(a):
    n = len(a)
    n_fft = next_pow_two(2*n-1)
    # conv using FFT
    f = np.fft.rfft(a, n_fft)
    conv = np.fft.irfft(f * f, n_fft)[:2*n-1]
    max_b = conv.max()
    # find indices within tolerance
    active = np.where(conv >= max_b - 1e-15)[0]  # tolerance small
    # compute gradient
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k-(n-1))
        hi = min(n-1, k)
        # vectors
        idx = np.arange(lo, hi+1)
        grad[idx] += 2 * a[k - idx]
    return max_b, grad

def _conv_self(seq):
    n = len(seq)
    size = 2*n - 1
    n_fft = 1 << (size - 1).bit_length()
    fft_seq = np.fft.rfft(seq, n=n_fft)
    conv = np.fft.irfft(fft_seq * fft_seq, n=n_fft)[:size]
    return conv

"""
Propose a non‑negative step‑function (a probability vector) with a low
`evaluate_sequence` score.

Strategy
--------
1.  Initialise a diverse pool of candidates (uniform, linear ramps,
    arcsine law, random Dirichlet draws and a few Beta‑shaped densities).
    Keep the best of them.

2.  Optimise the shape with a mirror‑descent / exponentiated‑gradient
    scheme that works directly on the probability simplex.
    For the current sequence we compute the self‑convolution via FFT,
    locate the shift(s) where the convolution attains its maximum and
    construct the exact sub‑gradient of the objective
        f(a) = max_k (a * a)_k .
    The update is `a_i ← a_i * exp( - η·∂f/∂a_i )` followed by a
    renormalisation to sum‑to‑one.

3.  The step‑size η decays slowly (η ← η·ρ).  When the search stalls we
    inject a small Gaussian jitter and reset the step‑size.

4.  If time permits we double the problem size (interpolating the best
    found sequence) and continue the optimisation – larger n gives a
    finer discretisation of the optimal continuous shape.

The whole routine stops before the 1000 s budget and returns the best
sequence discovered.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Small utility: next power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_self(seq: np.ndarray) -> np.ndarray:
    """
    Self‑convolution of a non‑negative vector using a real FFT.
    Returns a vector of length 2·len(seq)‑1.
    """
    n = seq.shape[0]
    size = 2 * n - 1
    n_fft = _next_pow2(size)
    f = np.fft.rfft(seq, n=n_fft)
    conv = np.fft.irfft(f * f, n=n_fft)[:size]
    # tiny negative artefacts caused by round‑off are clipped
    np.maximum(conv, 0.0, out=conv)
    return conv


def _maxconv_and_active(seq: np.ndarray):
    """
    Compute the maximal convolution value and the set of shift indices
    where this maximum is attained (within a tiny tolerance).
    """
    conv = _conv_self(seq)
    max_val = conv.max()
    # tolerance: a few ulps below the maximum
    tol = max_val * 1e-12 + 1e-15
    active = np.where(conv >= max_val - tol)[0]
    return max_val, active


def _subgradient(seq: np.ndarray):
    """
    Exact sub‑gradient of `max_k (seq*seq)_k`.
    Returns (max_value, gradient_vector).
    """
    n = seq.shape[0]
    max_val, active = _maxconv_and_active(seq)

    grad = np.zeros_like(seq)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        # contribution a_{k-i} for each i in the overlap
        grad[idx] += 2.0 * seq[k - idx]

    # average over the active shifts (keeps the scale stable)
    grad /= max(1, len(active))
    return max_val, grad


def _objective(seq: np.ndarray) -> float:
    """2·n·max(conv) – the value that `evaluate_sequence` returns
    (the sum of the vector is 1, clipping never happens)."""
    n = seq.shape[0]
    max_val, _ = _maxconv_and_active(seq)
    return float(2.0 * n * max_val)


def _initial_candidates(n: int, rng: np.random.Generator):
    """Create a rich set of starting vectors (already normalised)."""
    cand = []

    # 1. uniform
    cand.append(np.full(n, 1.0 / n))

    # 2. linear ramps (decreasing / increasing)
    ramps = np.arange(n, 0, -1, dtype=np.float64)      # decreasing
    cand.append(ramps / ramps.sum())
    cand.append(ramps[::-1] / ramps.sum())            # increasing

    # 3. arcsine law   a_i ∝ 1/√((i+½)(n‑i‑½))
    i = np.arange(n, dtype=np.float64)
    arcs = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    cand.append(arcs / arcs.sum())

    # 4. a few random Dirichlet draws
    for _ in range(6):
        r = rng.random(n)
        cand.append(r / r.sum())

    # 5. Beta‑shaped densities – many random (α,β) pairs
    for _ in range(12):
        # log‑uniform sampling of the parameters
        loga = rng.uniform(-2.0, 2.0)
        logb = rng.uniform(-2.0, 2.0)
        a = np.exp(loga)
        b = np.exp(logb)
        x = (np.arange(n) + 0.5) / n
        beta = np.power(x, a - 1.0) * np.power(1.0 - x, b - 1.0)
        # avoid a completely flat vector (might happen for extreme params)
        if beta.sum() == 0.0:
            continue
        cand.append(beta / beta.sum())

    return cand

    deadline = time.time() + budget_s - 0.5  # a small safety margin
