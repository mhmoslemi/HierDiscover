"""
Improved optimizer for the step‑function evaluation problem.

We use a smoothed max‑convolution gradient (softmax weighting of all
convolution entries) combined with an Adam‑style adaptive step size and a
simple momentum term.  The algorithm is run repeatedly for several
randomly chosen lengths, interpolating the current best solution to the
new length before refinement.  Optional LP refinement is kept for
fallback.  The routine respects the 1000 s budget and returns the best
sequence found.
"""
import time, math, numpy as np

def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σ x = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _conv_full(a: np.ndarray) -> np.ndarray:
    """Full convolution a * a (length 2*len(a)-1)."""
    return np.convolve(a, a)

def _weighted_grad(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)[k] with respect to a.
    w must have length 2*len(a)-1.
    """
    n = a.shape[0]
    # reverse the weight vector
    w_rev = w[::-1]
    # Convolution a with reversed weights
    conv = np.convolve(a, w_rev)
    # Slice the part that corresponds to indices 0..n‑1 (see derivation)
    # conv indices n‑1 … 2n‑2 contain the needed values in reversed order.
    slice_part = conv[n - 1 : 2 * n - 1][::-1]
    return 2.0 * slice_part

def _smooth_max_and_grad(a: np.ndarray, tau_factor: float = 0.1):
    """
    Return (max_val, grad) where max_val = max(conv(a,a))
    and grad is the gradient of a smooth approximation of the max,
    obtained with a soft‑max weighting of all convolution entries.
    """
    conv = _conv_full(a)
    max_val = float(conv.max())
    # Temperature proportional to the current maximum – ensures numerical stability.
    tau = max(1e-12, max_val * tau_factor)
    # Soft‑max weights (larger values receive larger weight, all sum to 1)
    exp_vals = np.exp((conv - max_val) / tau)
    w = exp_vals / np.sum(exp_vals)
    grad = _weighted_grad(a, w)
    return max_val, grad, conv

def _adam_descent(init: np.ndarray, target_sum: float,
                  timelimit: float,
                  base_lr: float = 0.2,
                  print_tag: str = "",
                  tau_factor: float = 0.1) -> tuple[np.ndarray, float]:
    """
    Adam‑style projected optimisation of the smoothed max‑convolution.
    Returns the best sequence (projected onto the simplex) encountered
    and its evaluation score.
    """
    a = _project_onto_simplex(init.astype(float), target_sum)
    best_seq = a.copy()
    try:
        best_score = float(evaluate_sequence(best_seq.tolist()))
    except Exception:
        best_score = math.inf

    # Adam hyper‑parameters
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    t0 = time.time()
    it = 0
    while time.time() < t0 + timelimit:
        max_val, grad, conv = _smooth_max_and_grad(a, tau_factor)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        # Bias‑correction
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        # Adaptive learning rate – we scale by the current max value
        step = base_lr * max_val / (np.sqrt(v_hat) + eps)

        a_new = a - step * m_hat
        a_new = _project_onto_simplex(a_new, target_sum)

        # occasional random kick to escape flat regions
        if np.random.rand() < 0.01:
            sigma = 0.02 * target_sum / len(a)
            a_new = _mutate(a_new, target_sum, sigma)

        # evaluate
        try:
            score = float(evaluate_sequence(a_new.tolist()))
        except Exception:
            score = math.inf

        if score < best_score:
            best_score = score
            best_seq = a_new.copy()
            print(f"[{time.time():.2f}] {print_tag} new best = {score:.6f} (n={len(a)})")

        a = a_new
        it += 1

    return best_seq, best_score

def _mutate(a: np.ndarray, target_sum: float, sigma: float) -> np.ndarray:
    """Gaussian mutation followed by simplex projection."""
    b = a + np.random.randn(a.size) * sigma
    b = np.maximum(b, 0.0)
    b = _project_onto_simplex(b, target_sum)
    return b

def _lp_refine(init: np.ndarray, target_sum: float,
               timelimit: float, mix: float = 0.2,
               max_iter: int = 4) -> tuple[np.ndarray, float]:
    """Same LP refinement as the reference implementation (kept for safety)."""
    try:
        import cvxpy as cp
        _cvx = True
    except Exception:
        _cvx = False

    if not _cvx:
        return init, float(evaluate_sequence(init.tolist()))
    n = init.shape[0]
    a = _project_onto_simplex(init, target_sum)
    best_seq = a.copy()
    try:
        best_score = float(evaluate_sequence(a.tolist()))
    except Exception:
        best_score = math.inf

    t0 = time.time()
    for it in range(max_iter):
        if time.time() - t0 > timelimit:
            break
        conv = np.convolve(a, a)
        M = float(conv.max())
        # build LP
        g = cp.Variable(n, nonneg=True)
        constraints = []
        for k in range(2 * n - 1):
            low = max(0, k - (n - 1))
            high = min(n - 1, k)
            idx = np.arange(low, high + 1, dtype=int)
            partner = k - idx
            expr = cp.sum(cp.multiply(a[low : high + 1], g[partner]))
            constraints.append(expr <= M)
        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
        try:
            prob.solve(solver=cp.GLPK, warm_start=True, verbose=False, glpkk=1e-08)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS, warm_start=True, verbose=False, max_iters=2000)
            except Exception:
                break
        if prob.status not in ("optimal", "optimal_inaccurate"):
            break
        g_val = np.asarray(g.value).flatten()
        if g_val is None or g_val.sum() == 0:
            break
        g_val = np.maximum(g_val, 0.0)
        scale = target_sum / g_val.sum()
        g_val *= scale
        a = (1 - mix) * a + mix * g_val
        a = _project_onto_simplex(a, target_sum)
        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            score = math.inf
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time():.2f}] LP‑refine iter {it+1} new best = {score:.6f} (n={len(a)})")
    return best_seq, best_score

def _resize_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation / sub‑sampling to change the length of a sequence."""
    old_n = seq.size
    if old_n == new_n:
        return seq.copy()
    if old_n > new_n:
        idx = np.linspace(0, old_n - 1, new_n).astype(int)
        return seq[idx]
    else:
        x_old = np.arange(old_n)
        x_new = np.linspace(0, old_n - 1, new_n)
        return np.interp(x_new, x_old, seq)

"""
Improved optimizer for the step‑function evaluation problem.

The algorithm uses a *smoothed* sub‑gradient of the maximum convolution
value (soft‑max weighting of all convolution entries) together with an
Adam‑style adaptive step size and occasional random mutations.  For each
candidate length we run a short optimisation, sometimes followed by an
LP‑refinement (if cvxpy is available).  The best sequence seen across all
runs is returned.  The routine respects a 1000 s budget and never
returns an empty or invalid list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simple Euclidean projection onto the simplex {x ≥ 0, Σ x = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex of non‑negative vectors with sum s."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution helpers
# ----------------------------------------------------------------------
def _conv_full(a: np.ndarray) -> np.ndarray:
    """Full convolution a * a (length 2*len(a)-1)."""
    return np.convolve(a, a)


def _weighted_grad(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)[k] w.r.t. a.
    w must have length 2*len(a)-1.
    """
    n = a.shape[0]
    w_rev = w[::-1]
    conv = np.convolve(a, w_rev)               # length 3n‑2
    # slice indices n‑1 … 2n‑2 (inclusive) and reverse them
    grad = 2.0 * conv[n - 1 : 2 * n - 1][::-1]
    return grad


def _smooth_max_and_grad(a: np.ndarray, tau_factor: float = 0.1):
    """
    Compute a smooth approximation of max(conv(a,a)) together with its
    gradient.  The smoothness is controlled by ``tau_factor`` (relative
    to the current maximum).
    Returns (max_val, grad, full_conv).
    """
    conv = _conv_full(a)
    max_val = float(conv.max())
    tau = max(1e-12, max_val * tau_factor)
    # soft‑max weights – values far below the maximum get exponentially
    # small weight, but we clip the exponent to avoid overflow.
    exp_vals = np.exp(np.clip((conv - max_val) / tau, -50, 0))
    w = exp_vals / np.sum(exp_vals)
    grad = _weighted_grad(a, w)
    return max_val, grad, conv


# ----------------------------------------------------------------------
# Small mutation helper (Gaussian noise + simplex projection)
# ----------------------------------------------------------------------
def _mutate(a: np.ndarray, target_sum: float, sigma: float) -> np.ndarray:
    """Return a mutated copy of a (Gaussian noise, then projection)."""
    b = a + np.random.randn(a.size) * sigma
    b = np.maximum(b, 0.0)
    b = _project_onto_simplex(b, target_sum)
    return b


# ----------------------------------------------------------------------
# Adam‑style projected descent on the smoothed max‑convolution
# ----------------------------------------------------------------------
def _adam_descent(init: np.ndarray,
                  target_sum: float,
                  timelimit: float,
                  base_lr: float = 0.2,
                  print_tag: str = "",
                  tau_factor: float = 0.1) -> tuple[np.ndarray, float]:
    """
    Adam optimiser with projection onto the simplex.
    Returns the best sequence encountered and its evaluation score.
    """
    a = _project_onto_simplex(init.astype(float), target_sum)

    best_seq = a.copy()
    try:
        best_score = float(evaluate_sequence(best_seq.tolist()))
    except Exception:
        best_score = math.inf

    # Adam hyper‑parameters
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    t0 = time.time()
    it = 0
    while time.time() < t0 + timelimit:
        max_val, grad, _ = _smooth_max_and_grad(a, tau_factor)

        # Adam moment updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        # bias‑corrected estimates
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        # step size – we scale with the current maximal convolution entry
        step = base_lr * max_val / (np.sqrt(v_hat) + eps)

        a_new = a - step * m_hat
        a_new = _project_onto_simplex(a_new, target_sum)

        # occasional random shake‑up to escape flat regions
        if np.random.rand() < 0.01:
            sigma = 0.02 * target_sum / len(a)
            a_new = _mutate(a_new, target_sum, sigma)

        try:
            score = float(evaluate_sequence(a_new.tolist()))
        except Exception:
            score = math.inf

        if score < best_score:
            best_score = score
            best_seq = a_new.copy()
            print(f"[{time.time():.2f}] {print_tag} new best = {score:.6f} (n={len(a)})")

        a = a_new
        it += 1

    return best_seq, best_score


# ----------------------------------------------------------------------
# Optional LP refinement (kept for safety)
# ----------------------------------------------------------------------
try:
    import cvxpy as cp
    _CVX_AVAILABLE = True
except Exception:  # pragma: no cover
    _CVX_AVAILABLE = False


def _lp_refine(init: np.ndarray,
               target_sum: float,
               timelimit: float,
               mix: float = 0.2,
               max_iter: int = 4) -> tuple[np.ndarray, float]:
    """
    Small LP‑update loop (same idea as the paper sketch).  Returns the best
    sequence obtained and its score.
    """
    if not _CVX_AVAILABLE:
        return init, float(evaluate_sequence(init.tolist()))

    n = init.shape[0]
    a = _project_onto_simplex(init, target_sum)

    try:
        best_score = float(evaluate_sequence(a.tolist()))
    except Exception:
        best_score = math.inf
    best_seq = a.copy()

    t0 = time.time()
    for it in range(max_iter):
        if time.time() - t0 > timelimit:
            break
        conv = np.convolve(a, a)
        M = float(conv.max())

        g = cp.Variable(n, nonneg=True)
        constraints = []
        for k in range(2 * n - 1):
            low = max(0, k - (n - 1))
            high = min(n - 1, k)
            idx = np.arange(low, high + 1, dtype=int)
            partner = k - idx
            expr = cp.sum(cp.multiply(a[low : high + 1], g[partner]))
            constraints.append(expr <= M)

        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
        try:
            prob.solve(solver=cp.GLPK, warm_start=True, verbose=False,
                       glpkk=1e-08)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS, warm_start=True,
                           verbose=False, max_iters=2000)
            except Exception:
                break

        if prob.status not in ("optimal", "optimal_inaccurate"):
            break
        g_val = np.asarray(g.value).flatten()
        if g_val is None or g_val.sum() == 0:
            break
        g_val = np.maximum(g_val, 0.0)
        scale = target_sum / g_val.sum()
        g_val *= scale

        a = (1.0 - mix) * a + mix * g_val
        a = _project_onto_simplex(a, target_sum)

        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            score = math.inf
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time():.2f}] LP‑refine iter {it+1} new best = {score:.6f} (n={n})")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Simple resize (linear interpolation / down‑sampling) to reuse a known good
# sequence
# ----------------------------------------------------------------------
def _resize_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize seq to length new_n preserving non‑negativity (linear interpolation)."""
    old_n = seq.size
    if old_n == new_n:
        return seq.copy()
    if old_n > new_n:
        idx = np.linspace(0, old_n - 1, new_n).astype(int)
        return seq[idx]
    else:
        x_old = np.arange(old_n)
        x_new = np.linspace(0, old_n - 1, new_n)
        return np.interp(x_new, x_old, seq)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000.0, **_):
    """
    Search for a step‑function (list of non‑negative heights) that minimises

        2 * n * max(conv(a, a)) / (sum(a) ** 2)

    The routine runs for at most ``budget_s`` seconds and returns the best
    sequence seen.  It mixes a smoothed Adam optimiser, occasional random
    mutations and a tiny LP‑refinement pass.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.3   # safety margin

    # ------------------------------------------------------------------
    # Global holder (use historic sequence if present)
    # ------------------------------------------------------------------
    global height_sequence_1
    best_seq = None
    best_score = math.inf

    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(height_sequence_1, dtype=float)
            sc = float(evaluate_sequence(cand.tolist()))
            best_seq = cand.copy()
            best_score = sc
            print(f"[{time.time():.2f}] start from historic seq, score={sc:.6f}")
        except Exception:
            pass

    if best_seq is None:
        # fallback: modest uniform sequence (fast and valid)
        n0 = 800
        target0 = math.sqrt(2.0 * n0)
        best_seq = np.full(n0, target0 / n0, dtype=float)
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"[{time.time():.2f}] start from uniform, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Helper that runs a single optimisation run (choice of method)
    # ------------------------------------------------------------------
    def _run_one(n: int, init_vec: np.ndarray, time_left: float):
        target = math.sqrt(2.0 * n)

        pick = np.random.rand()
        # 60 % Adam, 25 % LP (if available), rest: tiny evolutionary step
        if pick < 0.60:
            run_time = min(time_left, 12.0)
            return _adam_descent(init_vec, target, run_time,
                                 base_lr=0.3,
                                 print_tag=f"adam n={n}",
                                 tau_factor=0.08)
        elif pick < 0.85 and _CVX_AVAILABLE:
            run_time = min(time_left, 8.0)
            return _lp_refine(init_vec, target, run_time)
        else:
            # Evolutionary: mutate a few times and run a short Adam on each
            best_local = None
            best_local_score = math.inf
            trials = 4
            for _ in range(trials):
                sigma = 0.015 * target / n
                mutated = _mutate(init_vec, target, sigma)
                cand, sc = _adam_descent(mutated, target,
                                         min(time_left / (trials + 1), 6.0),
                                         base_lr=0.2,
                                         print_tag=f"evo n={n}",
                                         tau_factor=0.10)
                if sc < best_local_score:
                    best_local_score = sc
                    best_local = cand
            return best_local, best_local_score

    # ------------------------------------------------------------------
    # Main search loop
    # ------------------------------------------------------------------
    while time.time() < deadline - 1.0:
        # Choose a length – stay near the current best but allow some wander.
        if best_seq is not None and np.random.rand() < 0.70:
            n = best_seq.size
            if np.random.rand() < 0.25:
                n = int(n * np.random.uniform(0.85, 1.15))
                n = max(300, min(2500, n))
        else:
            n = np.random.randint(400, 2200)

        target_sum = math.sqrt(2.0 * n)

        # Initialise candidate (resize the current best or start random)
        if best_seq is not None:
            init = _resize_sequence(best_seq, n)
        else:
            init = np.random.rand(n)

        time_left = deadline - time.time()
        if time_left < 5.0:
            break

        cand_seq, cand_score = _run_one(n, init, time_left)

        if cand_seq is not None and cand_score < best_score:
            best_score = cand_score
            best_seq = cand_seq.copy()
            print(f"[{time.time():.2f}] *** GLOBAL NEW BEST = {best_score:.6f} (n={n})")
            if best_score < 1.503:          # target reached – stop early
                break

    # ------------------------------------------------------------------
    # Final safety – ensure a valid sequence is returned
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        target = math.sqrt(2.0 * n)
        best_seq = np.full(n, target / n, dtype=float)

    return best_seq.tolist()
