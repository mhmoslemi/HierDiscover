def _project_to_simplex(v):
    """Project onto probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    if v.ndim != 1:
        raise ValueError("v must be 1-d")
    n = v.shape[0]

import numpy as np
import time
import math

def _project_onto_simplex(v):
    """Projection of a vector onto the probability simplex {x>=0, sum=1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        # all nonpositive, return uniform distribution
        return np.full_like(v, 1.0/n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def _next_pow2(x):
    return 1 << ((x - 1).bit_length())

def _score_from_conv(conv, n):
    """Compute the objective value given convolution and length."""
    return 2.0 * n * np.max(conv)

def propose_candidate(budget_s=1000, seed=0, **kwargs):
    """
    Find a non‑negative sequence (step‑function heights) that minimizes

"""
Propose a sequence of non‑negative heights that minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm departs from the simple sub‑gradient method used previously.
It employs a *smooth* approximation of the max (soft‑max), Adam‑style
gradient descent, and projection onto the probability simplex.
A continuation on the soft‑max temperature τ gradually turns the smooth
objective into the true peak, helping the optimiser escape plateaus.
Symmetry of the sequence is enforced (a_i = a_{n‑1‑i}) because the
optimal step‑function is believed to be symmetric.

All operations use FFTs, so each iteration costs O(n log n) and easily
fits into the 1000 s time budget even for several thousand coefficients.
"""

import numpy as np
import time
import math

# ----------------------------------------------------------------------
# Helper functions
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    { x ≥ 0, sum(x) = 1 }.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]

    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)

    # find the largest rho satisfying the condition
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are non‑positive – fall back to the uniform vector
        return np.full_like(v, 1.0 / n)

    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _score_from_conv(conv: np.ndarray, n: int) -> float:
    """Return the evaluation score 2·n·max(conv) (sum of a is 1)."""
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000, seed: int = 0, **kwargs) -> list[float]:
    """
    Search for a good step‑function height vector.

    The routine:
      * starts from the previously discovered sequence if available,
        otherwise from the central‑binomial (Catalan) coefficients,
      * normalises it to a probability vector,
      * runs Adam optimisation on a soft‑max approximation of the peak
        of the self‑convolution,
      * projects onto the simplex after every update and re‑enforces
        symmetry,
      * gradually increases the soft‑max temperature so the objective
        becomes the true max,
      * performs occasional random kicks to avoid stagnation,
      * returns the best sequence seen before the time limit expires.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0      # safety margin

    # ------------------------------------------------------------------
    # 1. Initialise a
    # ------------------------------------------------------------------
    try:
        # Use the sequence created by earlier runs if it exists
        init = np.array(height_sequence_1, dtype=np.float64)
    except NameError:
        # Fallback: central‑binomial coefficients (C(2i,i)/4**i)
        n = 3000
        init = np.empty(n, dtype=np.float64)
        init[0] = 1.0
        for i in range(1, n):
            init[i] = init[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    else:
        n = init.shape[0]

    # Normalise to the simplex (sum = 1)
    a = _project_onto_simplex(init)

    # Enforce symmetry (the optimum is believed to be symmetric)
    a = (a + a[::-1]) / 2.0
    a = _project_onto_simplex(a)

    # ------------------------------------------------------------------
    # 2. Optimisation parameters
    # ------------------------------------------------------------------
    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    # Soft‑max temperature (τ).  Start smooth → end close to true max.
    tau = 1.0
    tau_target = 40.0                # final temperature
    tau_step = (tau_target - tau) / 1000.0  # linear schedule

    # FFT size that safely covers both convolution (2n‑1) and correlation
    # (n + (2n‑1) - 1 = 3n‑2).
    fft_size = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # 3. Initial score
    # ------------------------------------------------------------------
    fa = np.fft.rfft(a, fft_size)
    conv_full = np.fft.irfft(fa * fa, fft_size)[: 2 * n - 1]
    best_score = _score_from_conv(conv_full, n)
    best_a = a.copy()
    print(f"Initial score (n={n}): {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Main optimisation loop
    # ------------------------------------------------------------------
    iteration = 0
    while time.time() < deadline:
        iteration += 1

        # ----- a) self‑convolution (already needed for the loss) -----
        fa = np.fft.rfft(a, fft_size)
        conv_full = np.fft.irfft(fa * fa, fft_size)[: 2 * n - 1]

        # ----- b) soft‑max weights w_k ≈ exp(τ·conv_k) / Σ -----
        scaled = tau * conv_full
        max_scaled = np.max(scaled)
        exp_shifted = np.exp(scaled - max_scaled)
        w = exp_shifted / exp_shifted.sum()

        # ----- c) gradient of the smooth objective:
        #     ∇J = 2·(a ⋆ w)  (cross‑correlation)
        # ------------------------------------------------------------
        fw = np.fft.rfft(w, fft_size)
        corr_full = np.fft.irfft(fa * np.conj(fw), fft_size)[: 3 * n - 2]
        grad = 2.0 * corr_full[:n]                 # first n entries

        # ----- d) Adam update ------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        step = 0.02 * m_hat / (np.sqrt(v_hat) + eps)   # learning rate ≈ 0.02

        a_candidate = a - step
        a_candidate = _project_onto_simplex(a_candidate)

        # Re‑enforce symmetry (helps search and respects the problem structure)
        a_candidate = (a_candidate + a_candidate[::-1]) / 2.0
        a_candidate = _project_onto_simplex(a_candidate)

        # ----- e) evaluate candidate -----------------------------------------
        fa_cand = np.fft.rfft(a_candidate, fft_size)
        conv_cand = np.fft.irfft(fa_cand * fa_cand, fft_size)[: 2 * n - 1]
        cand_score = _score_from_conv(conv_cand, n)

        # ----- f) Accept if we do not worsen the objective --------------------
        cur_score = _score_from_conv(conv_full, n)
        if cand_score <= cur_score:
            a = a_candidate
            if cand_score < best_score:
                best_score = cand_score
                best_a = a.copy()
                print(f"Iter {iteration:6d}: new best {best_score:.6f}")
        else:
            # occasional random shake‑up to avoid being stuck
            if iteration % 500 == 0:
                noise = np.random.randn(n) * 1e-4
                a = _project_onto_simplex(a + noise)
                a = (a + a[::-1]) / 2.0
                a = _project_onto_simplex(a)

        # ----- g) increase temperature (make the smooth max sharper) -----
        tau = min(tau_target, tau + tau_step)

        # ----- h) small random perturbation every few thousand steps -----
        if iteration % 2000 == 0:
            perturb = np.random.randn(n) * 0.003
            a = _project_onto_simplex(a + perturb)
            a = (a + a[::-1]) / 2.0
            a = _project_onto_simplex(a)

    print(f"Finished after {iteration} iterations. Best score = {best_score:.6f}")
    # Return plain Python list as required by the evaluator
    return best_a.tolist()
