def project_to_simplex(v, s):
    if np.sum(v) == s and np.all(v >= 0):
        return v
    # shift v to be sorted descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def project_to_simplex(v, s):
    # Duchi algorithm.
    # v: 1d ndarray of any shape.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v)+1)
    # find rho
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero?
        return np.zeros_like(v)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a candidate step‑function using a smoothed‑max Adam optimiser.

The evaluation metric is   E(a)=2·n·max(a*a)/(∑a)² .
Because the sum is fixed to √(2n) during the search, minimising E is
equivalent to minimising the maximal convolution value M=max(a*a).
M is a non‑convex, non‑smooth functional.  We approximate the max by a
soft‑max (log‑sum‑exp) with a temperature β that is slowly increased.
For a given β the objective becomes smooth and its gradient can be
written in closed form:

    p_k = exp(β·c_k) / Σ_j exp(β·c_j)         (c = a*a)
    ∇β M ≈ 2·Σ_k p_k·a_{k-i}  for each i

The convolution in the gradient is obtained efficiently with an FFT‑based
convolution.  An Adam optimiser provides adaptive step‑sizes and a
projection onto the probability simplex (non‑negative entries with a
fixed sum) enforces the constraints required by ``evaluate_sequence``.
Occasional random jitter and a simple stagnation restart keep the
search from being trapped in poor local minima.

The routine runs until the time budget (default 1000 s) expires and
returns the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power‑of‑2 ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution of real vectors a and b using FFT.
    Returns the full convolution of length len(a)+len(b)-1.
    """
    n = a.size
    m = b.size
    N = _next_pow2(n + m - 1)
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    Fc = Fa * Fb
    conv = np.fft.irfft(Fc, N)
    return conv[: n + m - 1]


def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        {x | x_i ≥ 0 , Σ_i x_i = s }.
    Implements the algorithm of Duchi et al. (2008).
    """
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

        else:
            probs = exp_terms / sum_exp

        # ---- gradient of the smoothed max ---------------------------------------
        # grad_i = 2·Σ_k probs_k·a_{k-i}
        probs_rev = probs[::-1]
        conv_grad = _conv_fft(a, probs_rev)        # length n + (2n‑1) –1 = 3n‑2
        L = conv.size                              # = 2n‑1
        grad = 2.0 * conv_grad[L - 1 - np.arange(n)]

        # ---- Adam update ---------------------------------------------------------
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)
        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step

        # ---- projection onto the feasible simplex ---------------------------------
        a_new = _project_simplex(a_new, target_sum)
        np.clip(a_new, 0.0, 1000.0, out=a_new)    # respect the hard bounds of evaluate_sequence

        a = a_new

        # ---- periodic exact evaluation -------------------------------------------
        if iteration % 200 == 0:
            cur_score = evaluate_sequence(a.tolist())
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                last_improve = iteration
                print(f"[iter {iteration:07d}] new best = {best_score:.6f}")

            # small random jitter every few thousand iterations to aid exploration
            if iteration % 2000 == 0:
                jitter = np.random.randn(n) * target_sum * 0.001 / n
                a = _project_simplex(a + jitter, target_sum)

        # ---- simple stagnation restart -------------------------------------------
        if iteration - last_improve > 50000:
            print(f"[iter {iteration}] stagnation detected – re‑initialising from best")
            a = best_seq.copy()
            a += np.random.randn(n) * target_sum * 0.01 / n
            a = _project_simplex(a, target_sum)
            last_improve = iteration

    print(f"[done] iterations={iteration}, best score={best_score:.6f}")
    return best_seq.tolist()
