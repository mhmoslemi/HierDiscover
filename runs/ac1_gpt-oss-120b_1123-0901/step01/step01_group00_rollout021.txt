def compute_grad(a, active_indices):
    n = len(a); grad = np.zeros_like(a)
    for k in active_indices:
        low = max(0, k-(n-1))
        high = min(n-1, k)
        idx = np.arange(low, high+1)
        partner = k - idx
        grad[idx] += 2.0 * a[partner]
    return grad

def propose_candidate(seed=123, budget_s=1000.0):
    import time, numpy as np, math
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5

    best_seq = None
    best_score = float('inf')
    # optionally seed with existing
    if 'height_sequence_1' in globals():
        seq_arr = np.array(height_sequence_1, dtype=float)
        # ensure sum exact? We'll just evaluate.
        try:
            sc = evaluate_sequence(seq_arr.tolist())
            best_seq = seq_arr.copy()
            best_score = sc
            print(f"Initial seed from height_sequence_1: score {best_score}")
        except Exception:
            pass

    # helper functions defined earlier: _project_onto_simplex, compute_grad, refine_with_gradient

    while time.time() < deadline:
        # random choose method type
        method = np.random.choice(['binary','dirichlet','uniform'])
        n = int(np.random.randint(500, 2000))
        target_sum = math.sqrt(2.0 * n)

        if method == 'binary':
            # pick k between 5% and 50% of n
            k = int(np.random.randint(max(2, int(0.05*n)), max(3, int(0.5*n))))
            indices = np.random.choice(n, size=k, replace=False)
            a = np.zeros(n, dtype=float)
            a[indices] = target_sum / k
        elif method == 'dirichlet':
            alpha = np.random.uniform(0.2, 5.0)
            g = np.random.gamma(alpha, 1.0, size=n)
            if g.sum() == 0:
                continue
            a = g / g.sum() * target_sum
        else:  # uniform random (without Dirichlet)
            a = np.random.rand(n)
            a_sum = a.sum()
            a = a / a_sum * target_sum

        # evaluate
        try:
            score = evaluate_sequence(a.tolist())
        except Exception:
            continue

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time():.2f}] New global best {best_score:.6f} (n={n}, method={method})")

        # If score is promising, apply small gradient refinement (e.g., if score < 1.8)
        if score < 1.8 and time.time() < deadline - 5.0:
            a_refined, sc_refined = _refine_sequence(a, target_sum, deadline)
            if sc_refined < best_score:
                best_score = sc_refined
                best_seq = a_refined.copy()
                print(f"[{time.time():.2f}] Refined best {best_score:.6f} (n={n})")
            # if refined becomes very good break
            if best_score < 1.503:
                break

    # fallback uniform
    if best_seq is None:
        n = 1000
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum/n)

    return best_seq.tolist()

def _refine_sequence(a, target_sum, deadline):
    import numpy as np, time
    # ensure on simplex exactly:
    a = a.copy()
    # maybe project: a = _project_onto_simplex(a, target_sum)
    best_local = a.copy()
    best_score = evaluate_sequence(best_local.tolist())
    base_step = 0.05
    n = len(a)
    while time.time() < deadline and (time.time() < deadline - 1.0):
        conv = np.convolve(a, a)
        max_val = conv.max()
        # active indices within tolerance
        tol = max_val * 1e-12 + 1e-12
        active = np.where(conv >= max_val - tol)[0]
        # compute gradient
        grad = np.zeros_like(a)
        for k in active:
            low = max(0, k-(n-1))
            high = min(n-1, k)
            idx = np.arange(low, high+1)
            partner = k - idx
            grad[idx] += 2.0 * a[partner]
        max_grad = grad.max()
        if max_grad <= 0:
            break
        eta = base_step * max_val / max_grad
        a_new = a - eta * grad
        # project onto simplex
        a_new = _project_onto_simplex(a_new, target_sum)
        # evaluate
        sc = evaluate_sequence(a_new.tolist())
        if sc < best_score:
            best_score = sc
            best_local = a_new.copy()
        # acceptance: always move
        a = a_new
        # adapt step - maybe reduce if improvement slow
        base_step *= 0.999  # small decay
        # occasional random perturbation

"""
Randomized construction of step function heights with optional gradient refinement.

The algorithm repeatedly draws candidate sequences on the simplex:
*  binary support with uniform heights (chosen support size proportional to n);
*  Dirichlet‐distributed random vectors;
*  plain random vectors.

Each candidate is evaluated with `evaluate_sequence`.  The global best
candidate is kept.  Whenever a candidate reaches a reasonably good score
(e.g. < 1.8) a cheap projected gradient descent (sub‑gradient of the
maximum convolution entry) is run for a few iterations to smooth the
convolution further.  The total running time is limited by the supplied
budget (default 1000 s).  Finally the best sequence found is returned.
"""

import time
import math
import numpy as np

def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σ x = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _gradient_of_active(a: np.ndarray, active_idx: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max_{k} (a*a)[k] for the set of active indices."""
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in active_idx:
        low = max(0, k - (n - 1))
        high = min(n - 1, k)
        idx = np.arange(low, high + 1, dtype=int)
        partner = k - idx
        grad[idx] += 2.0 * a[partner]
    return grad

def _refine_sequence(a: np.ndarray, target_sum: float, deadline: float,
                     max_iters: int = 300, base_step: float = 0.07) -> tuple[np.ndarray, float]:
    """Run a few projected‑gradient steps starting from `a`.
    Returns the best (vector, score) seen during the run."""
    n = a.shape[0]
    # ensure feasibility
    a = _project_onto_simplex(a, target_sum)
    # initial evaluation
    best_vec = a.copy()
    try:
        best_score = float(evaluate_sequence(a.tolist()))
    except Exception:
        best_score = float('inf')
    step = base_step
    while time.time() < deadline and max_iters > 0:
        max_iters -= 1
        conv = np.convolve(a, a)
        max_val = conv.max()
        # active constraints – all indices attaining the maximum (numerical tolerance)
        tol = max_val * 1e-12 + 1e-14
        active = np.where(conv >= max_val - tol)[0]
        grad = _gradient_of_active(a, active)
        max_grad = grad.max()
        if max_grad <= 0:
            # nothing to change – exit early
            break
        # step length normalised as in the reference code
        eta = step * max_val / max_grad
        a_new = a - eta * grad
        a_new = _project_onto_simplex(a_new, target_sum)

        # occasional random shake‑up to avoid stagnation
        if np.random.rand() < 0.01:
            noise = np.random.randn(n) * (0.01 * target_sum / n)
            a_new = _project_onto_simplex(np.maximum(a_new + noise, 0.0), target_sum)
            step *= 0.95

        a = a_new
        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            score = float('inf')
        if score < best_score:
            best_score = score
            best_vec = a.copy()
            # shrink step slightly when improvement is found
            step *= 0.97

    return best_vec, best_score

"""
Randomized construction of a step‑function sequence with optional gradient polishing.

The search works as follows (all time is bounded by the supplied budget, 1000 s
by default).

1. Repeatedly draw a fresh candidate on the simplex
   Σ a_i = √(2 n), a_i ≥ 0.
   Three families are used:
   * *binary*: pick a random support size k ≈ (30‑50)% of n and assign the
     same height S/k to the k chosen indices (the remaining entries stay 0).
   * *dirichlet*: draw a Dirichlet vector with a random concentration α ∈[0.2,5].
   * *uniform*: a plain random vector normalised to the simplex.

2. Evaluate the candidate with the provided ``evaluate_sequence`` (which,
   after the normalisation, equals the maximal convolution entry).

3. Keep the best candidate seen so far.  When a candidate reaches a moderate
   quality (score < 1.8) a short projected‑gradient descent (sub‑gradient of
   the maximal convolution entry) is launched to smooth the convolution a
   little further.

4. The process stops when the time budget expires or when a sequence with
   score < 1.503 (the target) is found.  If nothing is found a uniform

def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of `v` onto the simplex of radius `s`."""
    if s < 0:

# ----------------------------------------------------------------------
def _gradient_of_active(a: np.ndarray, active_idx: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of   max_k (a*a)[k]   for the set of active indices.
    The gradient for a single index k is
        (∂/∂a_i) (a*a)[k] = 2·a_{k−i}   when i participates in the sum.

    grad = np.zeros_like(a)
    for k in active_idx:
        low = max(0, k - (n - 1))
        high = min(n - 1, k)
        idx = np.arange(low, high + 1, dtype=int)
        partner = k - idx
        grad[idx] += 2.0 * a[partner]
    return grad

                                               max_iters=200, base_step=0.07)
            if sc_ref < best_score:
                best_score = sc_ref
                best_seq = refined.copy()
                print(f"[{time.time():.2f}] Refined best {best_score:.6f} (n={n})")
            # early exit when the target is reached
            if best_score < 1.503:
                break

    # ------------------------------------------------------------------
    # Fallback – a safe uniform sequence if nothing else succeeded
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"[fallback] uniform sequence score = {best_score:.6f}")

    return best_seq.tolist()
