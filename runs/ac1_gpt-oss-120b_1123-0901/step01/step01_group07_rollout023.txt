"""
Propose a non‑negative step‑function that beats the 1.503 target.

The theoretical analysis shows that a power‑law
        a_i = (i + γ)^(‑p)
with an exponent p > 0.5 yields a constant asymptotic score
    score(p) = 2·(1‑p)²·B(1‑p,1‑p),
which drops below 1.5 for p≈0.57‑0.62 if the shift γ is large enough that
the endpoint term a₀² does not dominate the convolution.
The previous code never explored this region (it stayed close to p≈0.5).

The new search therefore

1. scans a fine grid for p∈[0.55,0.70] and γ∈[0.8,3.0],
   optionally adding a small prefix of zeros (offset) to shrink a₀.
2. performs a stochastic refinement around the best grid point.
3. (optional) tests a DPSS window – a smooth, non‑negative shape that often
   has low aperiodic autocorrelation.

The whole routine respects the 1000 s wall‑time and always returns the best
sequence found as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: build the candidate sequence
# ----------------------------------------------------------------------
def make_powerlaw_sequence(n: int, p: float, gamma: float, offset: int = 0) -> np.ndarray:
    """
    Return a sequence of length n:
        a_i = 0                      for i < offset
        a_i = (i‑offset + γ)^(-p)    for i ≥ offset
    All numbers are non‑negative.
    """
    seq = np.zeros(n, dtype=float)
    if offset < n:
        length = n - offset
        indices = np.arange(length, dtype=float) + gamma          # (i‑offset+γ)
        seq[offset:] = indices ** (-p)
    return seq


# ----------------------------------------------------------------------
# Optional DPSS candidate (requires scipy)
# ----------------------------------------------------------------------
def dpss_candidate(n: int) -> np.ndarray:
    """Return the first DPSS window (if scipy is available); otherwise empty."""
    try:
        from scipy.signal.windows import dpss
        # NW chosen heuristically; the first DPSS is non‑negative.
        win = dpss(n, NW=2.5, Kmax=1, sym=False)[0]
        # Clip any tiny negative round‑off errors.
        win = np.maximum(win, 0.0)
        return win
    except Exception:
        return np.array([])   # scipy not present


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: float = 1000) -> list:
    """
    Search for a step‑function with a low value of evaluate_sequence().
    Returns the best sequence (as a list of floats) found before the time
    budget expires.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.1   # safety margin

    # Length used in the previous best construction – a good trade‑off
    n = 3428

    best_seq = None
    best_score = float("inf")
    best_params = None   # (p, gamma, offset)

    # ------------------------------------------------------------------
    # 1) Coarse grid over (p, gamma, offset)
    # ------------------------------------------------------------------
    p_grid = np.arange(0.55, 0.71, 0.01)          # 0.55 … 0.70
    gamma_grid = np.arange(0.8, 3.01, 0.10)      # 0.8 … 3.0
    offset_grid = [0, 5, 10, 20, 50]             # few prefixes of zeros

    for p in p_grid:
        for gamma in gamma_grid:
            for offset in offset_grid:
                if time.time() > deadline:
                    break
                # quick feasibility check: a₀ must not be huge
                if offset == 0:
                    a0 = (gamma) ** (-p)
                else:
                    a0 = 0.0
                # Build and evaluate
                seq = make_powerlaw_sequence(n, p, gamma, offset)
                score = evaluate_sequence(seq.tolist())
                if score < best_score:
                    best_score = score
                    best_seq = seq.copy()
                    best_params = (p, gamma, offset)
                    print(f"[grid] p={p:.3f}, γ={gamma:.2f}, off={offset}, "
                          f"score={score:.6f}")

            if time.time() > deadline:
                break
        if time.time() > deadline:
            break

    # ------------------------------------------------------------------
    # 2) DPSS baseline (if available)
    # ------------------------------------------------------------------
    dpss_seq = dpss_candidate(n)
    if dpss_seq.size != 0:
        dpss_score = evaluate_sequence(dpss_seq.tolist())
        if dpss_score < best_score:
            best_score = dpss_score
            best_seq = dpss_seq.copy()
            best_params = ("DPSS", None, None)
            print(f"[DPSS] score={dpss_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Random local refinement (Gaussian steps)
    # ------------------------------------------------------------------
    if best_params is None:         # safety fallback – should never happen
        best_params = (0.60, 1.5, 0)

    p_cur, gamma_cur, off_cur = best_params if isinstance(best_params, tuple) else (0.60, 1.5, 0)

    # refinement schedule – more iterations when we have time
    while time.time() < deadline:
        # propose new parameters
        p_new = p_cur + np.random.randn() * 0.008     # fine‑grained step
        gamma_new = gamma_cur + np.random.randn() * 0.12
        off_new = off_cur + int(np.random.randn() * 2)

        # keep within sensible bounds
        if not (0.51 <= p_new < 0.80):
            continue
        if not (0.2 <= gamma_new <= 8.0):
            continue
        if not (0 <= off_new <= 200):
            continue
        if off_new >= n - 10:          # leave at least a few non‑zero entries
            continue

        seq = make_powerlaw_sequence(n, p_new, gamma_new, off_new)
        score = evaluate_sequence(seq.tolist())
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            p_cur, gamma_cur, off_cur = p_new, gamma_new, off_new
            print(f"[refine] p={p_new:.4f}, γ={gamma_new:.3f}, off={off_new}, "
                  f"score={score:.6f}")

    # ------------------------------------------------------------------
    # 4) Final conversion and report
    # ------------------------------------------------------------------
    result = best_seq.tolist()
    print(f"Finished search – best score {best_score:.6f} "
          f"with params p={p_cur:.4f}, γ={gamma_cur:.3f}, offset={off_cur}")
    return result
