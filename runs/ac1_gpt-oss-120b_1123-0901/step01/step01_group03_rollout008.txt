```
import numpy as np
import time
import math

def _project_onto_simplex(v):
    v = np.maximum(v, 0.0)
    sum_v = v.sum()
    if sum_v == 0:
        return np.full_like(v, 1.0 / len(v))
    return v / sum_v

def _full_convolution_fft(a):
    n = len(a)
    size = 1 << ((2*n - 1).bit_length())
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa*fa, size)[:2*n-1]
    return conv

def _smooth_gradient(a, conv, eps):
    # conv is a*a
    max_conv = conv.max()
    # stable softmax
    exp_part = np.exp(eps * (conv - max_conv))
    w = exp_part / exp_part.sum()
    # convolution of reversed a with w
    a_rev = a[::-1]
    # compute convolution via FFT
    size = 1 << ((len(w) + len(a_rev) - 1).bit_length())
    fw = np.fft.rfft(w, size)
    frev = np.fft.rfft(a_rev, size)
    conv_rev = np.fft.irfft(fw * frev, size)[:len(w)+len(a_rev)-1]
    # extract slice n-1 to n-1+n-1? Actually length n from index n-1 inclusive
    start = len(a)-1
    grad = 2.0 * conv_rev[start:start+len(a)]
    return grad
```

A pure sub‑gradient step on the *hard* maximum gets stuck very easily.

"""
Propose a step‑function (non‑negative vector) with a low self‑convolution peak.
The method combines a smoothed‑max gradient (soft‑max weighting of the
convolution) with the exact sub‑gradient at the current maximal index,
projects onto the probability simplex after each step and injects occasional
non‑linear shape changes (power‑transform) and tiny random perturbations.
The loop runs until the 1000 s budget is exhausted and returns the best
vector encountered.
"""

import time
import math
import numpy as np


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Return the Euclidean projection of v onto {x ≥ 0, Σx = 1}."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        # uniform distribution as a safe fallback
        return np.full_like(v, 1.0 / len(v))
    return v / s


def _full_convolution_fft(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a*a (full length 2·len(a)‑1) using FFT.
    """
    n = len(a)
    size = 1 << ((2 * n - 1).bit_length())  # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return conv


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray, k_max: int) -> np.ndarray:
    """
    Sub‑gradient of max(conv) w.r.t. a for a single maximising index k_max.
    (∂/∂a_i max(conv) = 2·a_{k_max-i} when the index is valid, otherwise 0.)
    """
    n = len(a)
    grad = np.zeros_like(a)
    i_start = max(0, k_max - (n - 1))
    i_end = min(n, k_max + 1)   # exclusive
    # j = k_max - i
    js = k_max - np.arange(i_start, i_end)
    grad[i_start:i_end] = 2.0 * a[js]
    return grad


def _smooth_gradient(a: np.ndarray, conv: np.ndarray, eps: float) -> np.ndarray:
    """
    Gradient of the soft‑max approximation of max(conv).
    The weights are w_i ∝ exp(eps·conv[i]), i.e. a smooth version of the max.
    The gradient ends up being a convolution of the reversed a with w.
    """
    # stable soft‑max
    max_c = conv.max()
    exp_part = np.exp(eps * (conv - max_c))
    w = exp_part / exp_part.sum()            # shape (2n‑1,)

    # convolution a_rev * w  (full length = n + (2n‑1) – 1 = 3n‑2)
    a_rev = a[::-1]
    size = 1 << ((len(w) + len(a_rev) - 1).bit_length())
    fw = np.fft.rfft(w, size)
    frev = np.fft.rfft(a_rev, size)
    conv_rev = np.fft.irfft(fw * frev, size)[: len(w) + len(a_rev) - 1]

    start = len(a) - 1          # position where the slice of length n starts
    grad = 2.0 * conv_rev[start:start + len(a)]
    return grad

    Search for a coefficient vector that minimises
        2·n·max(conv(a,a)) / (sum(a)^2)
    under the constraints a_i ≥ 0.
    The function respects the 1000 s time budget and returns the best
    vector found as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0   # a small safety margin

    # ------------------------------------------------------------

    # ------------------------------------------------------------
    try:

    except Exception:

    # 2. Initialise a (central‑binomial coefficients or previous best)

    if 'height_sequence_1' in globals():
        a = np.array(height_sequence_1[:n], dtype=np.float64)  # type: ignore[name-defined]
        if len(a) < n:                         # pad with zeros if needed
            a = np.concatenate([a, np.zeros(n - len(a))])
    else:
        # central‑binomial series a_i = C(2i,i)/4^i
        a = np.empty(n, dtype=np.float64)
        a[0] = 1.0
        for i in range(1, n):
            a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)

    a = _project_onto_simplex(a)

    # ------------------------------------------------------------
    # 3. Main optimisation loop (projected smoothed‑gradient descent)
    # ------------------------------------------------------------
    best_a = a.copy()
    conv = _full_convolution_fft(a)

    print(f"Initial: n={n}, M={best_M:.6e}, score={best_score:.6f}")

    iteration = 0
    eta0 = 0.6               # initial step‑size factor
    eps_start = 5.0          # start of soft‑max smoothing
    eps_end = 45.0           # final smoothing (close to true max)
    gamma = 0.35             # mixing weight: 0 = pure sub‑gradient,

    power_candidates = [0.7, 0.85, 0.9, 1.1, 1.2, 1.4]

    while time.time() < deadline:
        # --- current convolution and its peak ---
        conv = _full_convolution_fft(a)
        M = conv.max()
        k_max = int(conv.argmax())

        # --- schedule parameters ---
        # epsilon grows slowly with the iteration count
        eps = eps_start + (eps_end - eps_start) * min(1.0, iteration / 20000.0)

        g_max = _subgradient_maxconv(a, conv, k_max)

        grad = (1.0 - gamma) * g_max + gamma * g_smooth

        improved = False
        for factor in (1.0, 0.5, 0.25, 0.125):

            a_new = _project_onto_simplex(a_new)

                M = M_new
                improved = True
                break

        if not improved:

            a = _project_onto_simplex(a + np.random.randn(n) * 1e-4)

        # --- occasional non‑linear “shape” moves (power transform) ---
        if iteration % 2000 == 0 and iteration > 0:

                conv_pow = _full_convolution_fft(a_pow)
                M_pow = conv_pow.max()
                if M_pow < M:

                    M = M_pow
                    break

        # --- bookkeeping ---
        score = 2.0 * n * M
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"Iter {iteration:6d}: new best score {best_score:.6f}")

        # slowly reduce the influence of the smooth part (focus on true max)
        if iteration % 5000 == 0 and iteration > 0:
            gamma = max(0.0, gamma - 0.05)

        iteration += 1

    # ------------------------------------------------------------
    # 4. Return the best vector (plain Python floats)
    # ------------------------------------------------------------
    final_conv = _full_convolution_fft(best_a)
    final_score = 2.0 * n * final_conv.max()
    print(f"Finished after {iteration} iterations. Best score = {final_score:.6f}")
    return best_a.tolist()
