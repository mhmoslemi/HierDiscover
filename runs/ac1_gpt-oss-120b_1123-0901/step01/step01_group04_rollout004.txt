"""
propose_candidate
=================

Generate a non‑negative step‑function (list of floats) that improves on the
classic “arcsine” construction.  The algorithm proceeds in three stages.

1️⃣  **Parameter sweep** – try a family of symmetric Beta‑like shapes  
    a_i ∝ (x_i(1‑x_i))^e  with exponent *e* in a neighbourhood of the
    continuous optimum *e = –0.5*.  The best shape (by calling
    ``evaluate_sequence``) is kept.

2️⃣  **Linear‑programming refinement on a small grid** –  
    work on a reduced length (≈250) so that a dense LP is cheap.
    Given a current candidate *f* we compute  
    `M = max(conv(f,f))` and solve  

        max   Σ g_j
        s.t.  (f * g)_k ≤ M   for all k
              g_j ≥ 0

    (the constraints are linear in *g*).  The solution *g* is rescaled
    to the same ℓ₁‑norm as *f* and mixed with *f* (`f ← (1‑t)f + t·g`).
    Repeating a few times strictly reduces the objective.

3️⃣  **Fine‑grained hill‑climbing** on the final length (default 6027).  
    Randomly move a small fraction of mass from one index to another,
    accept the move only if the evaluation improves.  The loop stops
    when the time budget is exhausted.

All steps respect the non‑negativity and ℓ₁‑normalisation required by
``evaluate_sequence``.  The function returns a plain ``list[float]``.
"""

import time
import math
import random
import numpy as np

# SciPy is optional – the LP stage is simply skipped if it cannot be imported.
try:
    from scipy.optimize import linprog
except Exception:  # pragma: no cover
    linprog = None


def _score(seq: np.ndarray) -> float:
    """Convenient wrapper around the provided evaluator."""
    return evaluate_sequence(seq.tolist())


def _beta_shape(n: int, e: float) -> np.ndarray:
    """
    Symmetric Beta‑like shape a_i ∝ (x_i·(1‑x_i))^e,
    where x_i = (i+0.5)/n ∈ (0,1).  The result is ℓ₁‑normalised.
    """
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    # Guard against overflow for very negative exponents.
    with np.errstate(divide='ignore'):
        w = (x * (1.0 - x)) ** e
    # Very small values become inf when e<0; replace infinities.
    w[~np.isfinite(w)] = 0.0
    # Clip extreme values to keep the numbers well‑behaved.
    w = np.clip(w, 0.0, 1e12)
    total = w.sum()
    if total == 0.0:                     # fall back to uniform
        w = np.ones_like(w)
        total = w.sum()
    return w / total


def _lp_refine(f: np.ndarray, max_iters: int, mix: float, deadline: float):
    """
    Perform the LP‑based refinement on a short vector ``f``.
    ``mix`` is the convex‑combination factor (t in the description).
    Returns the best vector found and its score.
    """
    if linprog is None:
        # No linear programming support – just return the original vector.
        return f, _score(f)

    n = len(f)
    best_vec = f.copy()
    best_score = _score(best_vec)

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        conv = np.convolve(f, f)
        M = conv.max()

        # Build the (2n‑1)×n matrix that maps g → f*g.
        rows = 2 * n - 1
        A = np.zeros((rows, n), dtype=np.float64)
        for k in range(rows):
            start = max(0, k - (n - 1))
            end = min(n - 1, k)
            # indices of g that appear in (f*g)[k]
            js = np.arange(start, end + 1, dtype=np.int64)
            is_ = k - js
            A[k, js] = f[is_]

        # Linear program: maximise Σ g_j  ⇔ minimise −Σ g_j
        c = -np.ones(n, dtype=np.float64)
        b = np.full(rows, M, dtype=np.float64)

        res = linprog(
            c,
            A_ub=A,
            b_ub=b,
            bounds=[(0, None)] * n,
            method="highs",
            options={"presolve": True, "time_limit": max(1, int(deadline - time.time()))},
        )
        if not res.success:
            # LP failed – abort the refinement.
            break

        g = res.x
        sum_g = g.sum()
        if sum_g == 0.0:
            break
        # Rescale so that Σ g = Σ f (we keep the ℓ₁‑norm = 1).
        g = g * (1.0 / sum_g)

        # Convex combination.
        f_new = (1.0 - mix) * f + mix * g
        # Re‑normalise to avoid tiny drift.
        f_new /= f_new.sum()

        # Evaluate the new candidate.
        sc = _score(f_new)
        if sc < best_score:
            best_vec = f_new.copy()
            best_score = sc

        # Prepare for the next iteration.
        f = f_new

    return best_vec, best_score


def _hill_climb(a: np.ndarray, max_iters: int, deadline: float):
    """
    Very light stochastic hill‑climbing on a full‑size sequence.
    Moves a small fraction of mass from a random source index to a random
    destination index and keeps the move if it lowers the evaluation.
    """
    n = len(a)
    best = a.copy()
    best_score = _score(best)

    # Pre‑compute a list of indices with non‑zero mass (updated lazily).
    for it in range(max_iters):
        if time.time() >= deadline:
            break

        # Pick a source index with non‑trivial mass.
        src_candidates = np.where(best > 1e-12)[0]
        if src_candidates.size == 0:
            break
        src = int(np.random.choice(src_candidates))
        dst = int(np.random.randint(0, n))
        if src == dst:
            continue

        # Transfer a random fraction (1 % … 10 %) of the source mass.
        frac = np.random.uniform(0.01, 0.10)
        delta = frac * best[src]
        if delta <= 0.0:
            continue

        cand = best.copy()
        cand[src] -= delta
        cand[dst] += delta

        # Re‑normalise to keep the ℓ₁‑norm exactly 1 (avoids tiny numerical drift).
        cand /= cand.sum()

        sc = _score(cand)
        if sc < best_score:
            best = cand
            best_score = sc
            # occasional progress report
            if it % 100 == 0:
                print(f"[hill‑climb] it={it:05d} score={best_score:.6f}")

    return best, best_score
