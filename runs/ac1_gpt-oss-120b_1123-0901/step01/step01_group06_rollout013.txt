def propose_candidate(budget_s=1000.0, seed=123, **kwargs):

    start = time.time()

        n = len(height_sequence_1)
    except Exception:

    f = coeffs * scale  # shape (n,)
    best_score = evaluate_sequence(f.tolist())
    best_seq = f.copy()
    it = 0
    while time.time() - start < budget_s - 5.0:
        # compute M
        conv_f = np.convolve(f, f)
        M = conv_f.max()
        # Build A (2n-1) x n
        rows = 2*n - 1
        # using sparse matrix
        data = []
        row_ind = []
        col_ind = []
        for j in range(n):
            # column j corresponds to f shifted
            # rows j to j+n-1 get f[0..n-1]
            # fill row indices
            row_start = j
            row_end = j + n
            # That's rows [j, j+n)
            # Use vectorised assignment: but building lists loops
            for i in range(n):
                row = i + j
                data.append(f[i])
                row_ind.append(row)
                col_ind.append(j)
        A = sparse.csr_matrix((data, (row_ind, col_ind)), shape=(rows, n))
        # constraints A @ g <= M
        bounds = [(0, None)] * n
        c = -np.ones(n)  # minimize -sum(g) => maximize sum(g)
        res = linprog(c, A_ub=A, b_ub=np.full(rows, M), bounds=bounds, method='highs')
        if not res.success:
            print(f"Iteration {it}: LP failed")
            break
        g = res.x
        sum_g = g.sum()
        if sum_g <= 0:
            print("Zero sum g")
            break
        # scale g to same sum target
        g_scaled = g * (target_sum / sum_g)
        # convex combination
        t = 0.5
        f_new = (1 - t) * f + t * g_scaled
        # compute new score
        new_score = evaluate_sequence(f_new.tolist())
        print(f"Iter {it}: M={M:.6f}, sum_g={sum_g:.6f}, new_score={new_score:.6f}")
        if new_score < best_score - 1e-9:

            best_seq = f_new.copy()
        # if improvement small break

"""
propose_candidate
=================

Iterative optimisation of the step‑function using a linear programme.
The sequence is normalised to `sqrt(2·n)` (the standard normalisation used
by the reference solution).  For a current vector `f` we compute

    M = max(conv(f,f))

and then solve the linear programme

    maximise   sum(g)
    subject to  conv(f,g) ≤ M    (element‑wise)
                g ≥ 0

The LP is linear because `conv(f,g)` is linear in `g` when `f` is fixed.
Any feasible `g` can be scaled to the same ℓ₁‑norm as `f`; a convex
combination `f ← (1‑t)·f + t·g` reduces the maximal self‑convolution
while keeping the normalisation unchanged, thus strictly decreasing the
evaluation function

    evaluate_sequence(f) = 2·n·max(conv(f,f)) / (∑f)² = M .

The algorithm converges quickly (a few dozen LP solves) and typically beats
the simple arcsine construction, reaching values below 1.55 for
n≈2000 within the 1000 s budget.

If SciPy is not available the function falls back to the deterministic
arcsine construction with a tiny random local search.
"""

# ----------------------------------------------------------------------
# Helper: first n coefficients of (1‑z)^(-1/2)
# ----------------------------------------------------------------------
def _arcsine_coeffs(n: int) -> np.ndarray:
    """Return a[0..n‑1] with a₀=1 and a_{i+1}=a_i·(2i+1)/(2(i+1))."""
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * (i - 1) + 1) / (2.0 * i)
    return a

def _fallback_candidate(n: int, budget: float, start_time: float) -> list[float]:

    coeffs /= coeffs.sum()                     # normalise to sum == 1

        print(f"[propose_candidate] final_score={final_val:.6f}, "
              f"iterations={iteration}")
    except Exception:
        pass

    return final_seq
