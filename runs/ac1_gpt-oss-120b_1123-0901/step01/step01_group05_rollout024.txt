def compute_score(seq):
    arr = np.asarray(seq, dtype=np.float64)
    n = arr.size
    # Compute convolution using FFT with length = 2n-1
    size = 1 << (2*n-1).bit_length()
    fa = np.fft.rfft(arr, size)
    conv = np.fft.irfft(fa * fa, size)[:2*n-1]
    max_b = np.max(conv)
    sum_a = np.sum(arr)
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a**2)

def compute_score(arr):
    n = arr.shape[0]
    size = 1 << ((2*n - 1).bit_length())
    fa = np.fft.rfft(arr, size)
    conv = np.fft.irfft(fa * fa, size)[:2*n-1]
    max_b = np.max(conv)
    sum_a = np.sum(arr)
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a * sum_a)

"""
Propose a candidate step‑function sequence.

The algorithm works in three stages:

1. **Initial shape** – an inverse‑square‑root (≈1/√i) sequence is built and
   normalised so that its L1‑norm equals √(2 n).  For a normalised sequence
   the evaluation score of ``evaluate_sequence`` equals the maximal value
   of its self‑convolution.

2. **Linear‑programming improvement** – following the scheme described in
   the statement, we repeatedly solve a linear program:
       maximise   Σ g_j
       subject to g≥0  and  (f * g)[k] ≤ M  for all k,
   where ``M`` is the current maximal convolution of the sequence ``f``.
   The optimal ``g`` has a larger sum than ``f`` while respecting the same
   convolution bound.  After rescaling ``g`` to the same L1‑norm as ``f`` we
   update ``f ← (1‑t) f + t g`` (with a diminishing step size ``t``).  Each
   iteration strictly reduces the maximal convolution and therefore the
   evaluation score.

3. **Final cheap refinement** – with the remaining time we perform a
   stochastic local search (multiplicative perturbations of single entries)
   that keeps only improving moves.

All calculations use a fast FFT‑based convolution; the heavy linear‑program
solver from SciPy (``highs``) is employed only when available.  The function
terminates before the time budget expires and returns the best sequence found.
"""

def _conv_self(seq: np.ndarray) -> np.ndarray:
    """Return the discrete convolution of ``seq`` with itself (real FFT)."""
    n = seq.size
    size = 1 << ((2 * n - 1).bit_length())      # next power‑of‑2
    fa = np.fft.rfft(seq, size)
    conv = np.fft.irfft(fa * fa, size)[:2 * n - 1]
    return conv

def evaluate_fast(seq: np.ndarray) -> float:
    """
    Fast version of ``evaluate_sequence`` – returns the same value
    (scale invariant) using FFT.  ``seq`` must be non‑negative.
    """
    n = seq.size
    if n == 0:
        return float("inf")
    conv = _conv_self(seq)
    max_b = float(np.max(conv))
    sum_a = float(np.sum(seq))
    if sum_a < 0.01:
        return float("inf")
    return 2.0 * n * max_b / (sum_a * sum_a)

def _generate_inverse_sqrt(n: int) -> np.ndarray:
    """a_i = binom(2i,i)/4**i  (≈1/√i).  Returns a NumPy array."""
    seq = np.empty(n, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, n):
        seq[i] = seq[i - 1] * (2 * i - 1) / (2 * i)
    return seq

def _generate_power_sequence(n: int, p: float, beta: float = 0.0,
                             lam: float = 0.0) -> np.ndarray:
    """a_i = (i+1+β)^{-p}·exp(-λ·i).  All entries ≥0."""
    i = np.arange(n, dtype=np.float64)
    seq = (i + 1.0 + beta) ** (-p)
    if lam != 0.0:
        seq *= np.exp(-lam * i)
    return seq

    if n > 2500:                     # keep linear‑program size reasonable
        n = 2500

    target_sum = math.sqrt(2.0 * n)   # the normalisation used by the LP scheme

    # ------------------------------------------------------------------
    # 2. Initial candidate – inverse‑square‑root shape, normalised.
    # ------------------------------------------------------------------
    base = _generate_inverse_sqrt(n)
    base = base * (target_sum / np.sum(base))
    best_seq = base.copy()
    best_score = evaluate_fast(best_seq)
    print(f"[init] n={n}, init score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Light random exploration of the (p,β,λ) family (≈10 % of budget).
    # ------------------------------------------------------------------
    param_deadline = start + budget_s * 0.1
    while time.time() < param_deadline:
        p = rng.uniform(0.35, 0.65)          # around the known optimum 0.5
        beta = rng.uniform(0.0, 5.0)
        lam = rng.uniform(0.0, 1e-3)
        cand = _generate_power_sequence(n, p, beta, lam)
        cand = cand * (target_sum / np.sum(cand))  # normalise
        cand_score = evaluate_fast(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            print(f"[param] improved to {best_score:.6f} (p={p:.4f}, β={beta:.2f}, λ={lam:.5e})")

    # ------------------------------------------------------------------
    # 4. Linear‑programming refinement (main optimisation loop).
    # ------------------------------------------------------------------
    if _HAVE_LP:
        f = best_seq.copy()
        # Pre‑allocate the constraint matrix – it will be overwritten each iter.
        rows = 2 * n - 1
        A = np.empty((rows, n), dtype=np.float64)

        t = 0.2                     # initial mixing factor
        no_imp = 0                  # counter for successive non‑improvements
        iter_cnt = 0
        max_iter = 2000            # hard stop in case we run out of time

        while time.time() < deadline and iter_cnt < max_iter:
            iter_cnt += 1
            # maximal self‑convolution of the current f
            M = float(np.max(_conv_self(f)))
            # Build the linear constraints: for each k, Σ_i f_i * g_{k‑i} ≤ M
            # Column i of A contains f_i on rows i … i+n‑1.
            for i in range(n):
                A[i:i + n, i] = f[i]
            b = np.full(rows, M, dtype=np.float64)

            # Linear programme: maximise Σ g_j  ⇔  minimise -Σ g_j
            c = -np.ones(n, dtype=np.float64)
            try:
                res = linprog(c, A_ub=A, b_ub=b,
                              bounds=[(0, None)] * n,
                              method='highs',
                              options={'presolve': True})
            except Exception as exc:                     # pragma: no cover
                print("[LP] solver exception:", exc)
                break

            if not res.success:                         # pragma: no cover
                print("[LP] not successful:", res.message)
                break

            g0 = np.asarray(res.x, dtype=np.float64)
            sum_g0 = float(np.sum(g0))
            if sum_g0 <= 0.0:
                break

            # rescale g0 to the prescribed L1‑norm
            g = g0 * (target_sum / sum_g0)

            # convex combination with the old vector
            f_new = (1.0 - t) * f + t * g

            new_score = evaluate_fast(f_new)
            if new_score < best_score:
                best_score = new_score
                best_seq = f_new.copy()
                f = f_new
                print(f"[LP] iter {iter_cnt:04d} improved score={best_score:.6f} (t={t:.3f})")
                no_imp = 0
                # slowly reduce the step size to stabilise near the optimum
                t = max(0.02, t * 0.95)
            else:
                # no improvement – decrease the mixing factor more aggressively
                no_imp += 1
                t = max(0.02, t * 0.7)
                f = f_new
                if no_imp >= 8:
                    # if we are stuck, stop the LP phase early
                    break
    else:
        print("[info] SciPy linprog unavailable – skipping LP refinement.")
        f = best_seq.copy()

    # ------------------------------------------------------------------
    # 5. Final cheap stochastic refinement (remaining time).
    # ------------------------------------------------------------------
    while time.time() < deadline:
        i = rng.integers(n)
        old = best_seq[i]
        factor = math.exp(rng.normal(0.0, 0.02))
        new_val = max(0.0, old * factor)
        cand = best_seq.copy()
        cand[i] = new_val
        cand_score = evaluate_fast(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand
            print(f"[final] improved to {best_score:.6f} (index {i})")

    # ------------------------------------------------------------------
    # 6. Return a plain Python list (the runner will run the official
    #    ``evaluate_sequence`` on it).
    # ------------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, 1000.0)   # respect the evaluator's caps
    return best_seq.tolist()
