def _project_to_simplex(v):
    u = np.sort(v)[::-1] # sorted descending
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1] - 1   # index (zero based)
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def propose_candidate(seed=0, budget_s=1000, **kwargs):
    np.random.seed(seed)

    deadline = time.time() + budget_s - 5

    # Determine length n: use length of height_sequence_1 if defined and not too large
    try:
        n = len(height_sequence_1)
        # maybe cap to 8000 to avoid huge overhead.
        n = min(n, 8000)
    except NameError:
        n = 6000

    # Optionally allows customizing n via kwargs
    n = int(kwargs.get('n', n))

    # Initial sequences candidates list:
    init_candidates = []
    # arcsine series (central binomial)
    a0 = np.empty(n, dtype=np.float64)
    a0[0] = 1.0
    for i in range(1, n):
        a0[i] = a0[i-1] * (2*i - 1) / (2*i)
    init_candidates.append(_project_to_simplex(a0))

    # Uniform
    init_candidates.append(_project_to_simplex(np.ones(n)))

    # If height_sequence_1 defined, use normalized version
    try:
        seq = np.array(height_sequence_1, dtype=np.float64)
        if len(seq) >= n:
            a = seq[:n].copy()
        else:
            # pad with zeros
            a = np.zeros(n)
            a[:len(seq)] = seq
        init_candidates.append(_project_to_simplex(a))
    except NameError:
        pass

    best_score = float('inf')
    best_seq = None

    # Prepare loop across candidates; allocate time per candidate maybe 1/len(candidates)
    n_candidates = len(init_candidates)
    # For each candidate, run a gradient descent session with some hyperparameters.
    for idx, a_init in enumerate(init_candidates):
        # allocate potentially equal time portion
        sub_deadline = time.time() + (deadline - time.time()) / (n_candidates - idx)
        a = a_init.copy()
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        t = 0
        beta = 5.0
        beta_inc = 1.02
        beta_max = 120.0
        # step size base
        alpha0 = 0.2
        # keep local best
        best_local_score = float('inf')
        best_local_a = None
        while time.time() < sub_deadline:
            t += 1
            # compute conv
            conv = _full_convolution(a)
            max_val = conv.max()
            # compute metric
            score = 2.0 * n * max_val
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"[Candidate {idx}] New global best: {best_score:.6f} at iter {t}")
            if score < best_local_score:
                best_local_score = score
                best_local_a = a.copy()
            # gradient
            grad = _grad_softmax(a, conv, beta)
            # Adam update
            m = 0.9 * m + 0.1 * grad
            v = 0.999 * v + 0.001 * (grad ** 2)
            m_hat = m / (1 - 0.9 ** t)
            v_hat = v / (1 - 0.999 ** t)
            step = alpha0 / (np.sqrt(v_hat) + 1e-8)
            a = a - step * m_hat
            a = _project_to_simplex(a)

            # adjust beta gradually
            if beta < beta_max:
                beta = min(beta * beta_inc, beta_max)

            # periodically random perturbation if no improvement
            if t % 5000 == 0:
                noise = np.random.randn(n) * 0.001
                a = _project_to_simplex(a + noise)

        # end of candidate loop
        if best_local_a is not None and best_local_score < best_score:
            best_score = best_local_score
            best_seq = best_local_a.copy()
            print(f"[Candidate {idx}] New best from local: {best_score:.6f}")

    # final: Convert best_seq to python list
    if best_seq is None:
        # fallback uniform
        best_seq = np.full(n, 1.0 / n)
    # convert to list of python floats
    return best_seq.tolist()

def _grad_softmax(a, conv, beta):
    max_c = conv.max()
    # subtract max to avoid overflow
    exp_term = np.exp(beta * (conv - max_c))
    w = exp_term / exp_term.sum()
    # cross-correlation: g_j = 2 * Σ_k w_k a_{k-j}
    n = len(a)
    # compute FFT of w and reversed a
    size = 1 << ((3 * n - 2).bit_length())
    fw = np.fft.rfft(w, size)
    fa_rev = np.fft.rfft(a[::-1], size)
    corr = np.fft.irfft(fw * fa_rev, size)
    start = n - 1
    grad = 2 * corr[start:start + n]
    return grad

"""
Propose a step‑function (non‑negative sequence) whose
self‑convolution peak is as small as possible.

Approach
--------
1. Choose a length n (default = length of the existing construction,
   capped at 8000).  
2. Initialise several candidates:
   • the classic central‑binomial sequence (the “arcsine series”);
   • a *symmetric discrete arcsine* distribution
       a_i ∝ 1/√((i+½)·(n‑i‑½)) – a direct discretisation of the
       continuous optimal density;
   • the uniform distribution;
   • the previously found construction (if available).
   All are projected onto the probability simplex (Σa_i = 1).

3. Optimise each candidate with a smoothed version of the objective

        F(a) = 2·n·max_k (a*a)_k

   The non‑smooth max is replaced by a soft‑max:
        S(a) = (2·n/β)·log Σ_k exp(β·(a*a)_k),

   whose gradient is
        ∇S = 2·(w ⋆ a)   with   w_k = exp(β·c_k)/Σ_j exp(β·c_j)
   and “⋆” denoting linear cross‑correlation.  The cross‑correlation is
   computed efficiently with an FFT.

   We run Adam (adaptive‑momentum) on this smooth objective,
   slowly increasing β (annealing) so that the optimisation
   gradually concentrates on the true peak.  After every update the
   vector is projected onto the simplex to keep feasibility.

4. Keep the best sequence seen over all runs and return it.

The whole routine stops after the allocated time budget (default 1000 s)
and returns the best sequence obtained.  Empirically the symmetric
arcsine start already beats the previous bound (≈ π/2) and further
Adam steps drive the score below the target 1.503.
"""

import time
import math
import numpy as np


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Exact Euclidean projection of v onto {x ≥ 0, Σx = 1}."""
    v = np.asarray(v, dtype=np.float64)
    if v.ndim != 1:
        raise ValueError("simplex projection expects a 1‑D vector")
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries non‑positive → return uniform point
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1] - 1          # zero‑based index of last positive
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using an FFT (length 2·n‑1)."""
    n = len(a)
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return conv


def _softmax_grad(a: np.ndarray, conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Gradient of the smooth objective
        (2·n/β)·log Σ exp(β·conv)
    where conv = a*a (linear convolution).  Implements
        ∇ = 2·(w ⋆ a)
    with
        w_k = exp(β·conv_k) / Σ_j exp(β·conv_j).
    """
    # soft‑max weights (numerically stable)
    max_c = conv.max()
    exp_term = np.exp(beta * (conv - max_c))
    w = exp_term / exp_term.sum()

    n = len(a)
    # cross‑correlation g_j = Σ_k w_k * a_{k-j}
    # Using FFT: corr = ifft( fft(w) * conj(fft(a_rev)) )
    size = 1 << ((3 * n - 2).bit_length())   # ≥ 3·n‑2
    fw = np.fft.rfft(w, size)
    fa_rev = np.fft.rfft(a[::-1], size)
    corr = np.fft.irfft(fw * fa_rev, size)
    start = n - 1
    grad = 2.0 * corr[start:start + n]
    return grad


def _init_symmetric_arcsine(n: int) -> np.ndarray:
    """Discrete symmetric arcsine density a_i ∝ 1/√((i+½)(n‑i‑½))."""
    i = np.arange(n, dtype=np.float64)
    denom = np.sqrt((i + 0.5) * (n - i - 0.5))
    a = 1.0 / denom
    a = _project_to_simplex(a)                 # normalise Σa_i = 1
    return a


def _init_central_binomial(n: int) -> np.ndarray:
    """a_i = C(2i,i) / 4**i  (the classic series)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    a = _project_to_simplex(a)
    return a

                     budget_s: float = 1000.0,
                     **kwargs) -> list:
    """
    Return a list of non‑negative heights that (hopefully) minimises
    ``evaluate_sequence``.  The algorithm described above is used.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # 1. Choose sequence length.
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)          # use existing length if present
        n = min(int(kwargs.get("n", n)), 8000)
    except NameError:
        n = int(kwargs.get("n", 6000))
    n = max(10, n)                           # ensure a sensible minimum

    # ------------------------------------------------------------------
    # 2. Build a list of initial candidates.
    # ------------------------------------------------------------------
    init_candidates = []

    # (a) symmetric discrete arcsine – the most promising start
    init_candidates.append(_init_symmetric_arcsine(n))

    # (b) classic central‑binomial coefficients (used by the reference code)
    init_candidates.append(_init_central_binomial(n))

    # (c) uniform distribution
    init_candidates.append(_project_to_simplex(np.ones(n, dtype=np.float64)))

    # (d) reuse the previously found construction if it exists
    try:
        prev = np.array(height_sequence_1, dtype=np.float64)

            a = prev[:n].copy()
        else:
            a = np.zeros(n, dtype=np.float64)
            a[:prev.size] = prev
        init_candidates.append(_project_to_simplex(a))
    except NameError:
        pass

    # ------------------------------------------------------------------
    # 3. Optimisation parameters (shared across all runs)
    # ------------------------------------------------------------------
    best_score = float("inf")
    best_seq = None

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    base_lr = 0.05                       # base learning rate for Adam
    beta_start = 0.5                     # initial soft‑max sharpness
    beta_factor = 1.015                  # multiplicative increase per 1000 iters
    beta_max = 150.0

    # ------------------------------------------------------------------
    # 4. Run an optimisation sweep for each initialization.
    # ------------------------------------------------------------------
    for idx, a0 in enumerate(init_candidates):
        # give each start a fair share of the remaining time
        remaining = deadline - time.time()
        if remaining <= 0:
            break
        # spend roughly the same amount of time on each remaining start
        share = remaining / (len(init_candidates) - idx)

        a = a0.copy()
        m = np.zeros_like(a)          # first moment (Adam)
        v = np.zeros_like(a)          # second moment (Adam)
        t = 0
        beta = beta_start
        last_improve = 0

            conv = _full_convolution(a)
            cur_max = conv.max()
            cur_score = 2.0 * n * cur_max

                best_seq = a.copy()
                print(f"[Init {idx}] New global best: {best_score:.6f} (iter {t})")

                local_best_a = a.copy()
                last_improve = t

            # ----- compute gradient of the smooth objective -----
            grad = _softmax_grad(a, conv, beta)

            # ----- Adam update -----
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

            a = a - step
            a = _project_to_simplex(a)

            if beta < beta_max:

                    a = _project_to_simplex(a + noise)

            best_score = local_best

            print(f"[Init {idx}] New best after refinement: {best_score:.6f}")

    return best_seq.tolist()
