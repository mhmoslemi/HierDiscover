def eval_obj(a):
    n = len(a)
    # Convolution via FFT
    m = 1 << (2 * n - 1).bit_length()
    fa = np.fft.rfft(a, n=m) # rfft for real
    conv = np.fft.irfft(fa * fa, n=m)[:2*n-1]
    max_b = conv.max()
    return 2 * n * max_b / (a.sum() ** 2)

    start = time.time()
    deadline = start + budget_s - 5

    best_score = float('inf')

    # set of lengths to try
    lengths = [800, 1500, 2100, 3000, 5000]
    # allocate time proportionally to length maybe
    for n in lengths:
        if time.time() >= deadline:

        # allocate max 0.18 of remaining time per n
        remaining = deadline - time.time()
        allocate = remaining * 0.18  # some fraction
        # run SA for this n

        if score < best_score:
            best_score = score
            best_seq = seq

"""
Propose a step‑function height sequence with a low evaluation score.

The algorithm is a hybrid of several ideas:

1. **Adam‑style projected gradient descent**  
   The objective (when the total sum is fixed to √(2·n)) equals the
   maximum entry of the self‑convolution `a*a`.  The gradient of this
   max‑convolution is known analytically (see the comment in the code).
   An Adam optimizer with projection onto the non‑negative orthant and a
   renormalisation step is used.  A tiny random jitter is added periodically
   to escape shallow plateaus.

2. **Multiple random restarts & different lengths**  
   The search is performed for several sequence lengths (including the
   length of the provided `height_sequence_1`).  For each length several
   restarts are launched with different Dirichlet initialisations (α=0.5,
   1.0, 2.0).  The best solution across all runs is kept.

3. **Fast convolution via FFT**  
   The convolution and its maximum are computed with an FFT of length
   `next_pow2(2·n‑1)`.  This makes each iteration O(n log n) and allows
   thousands of iterations within the 1000 s budget.

The program respects all constraints of `evaluate_sequence` (non‑negative,
≤1000, fixed total sum).  The final sequence is returned as a list of
floats.  Progress and intermediate best scores are printed for monitoring.
"""

import time
import math
import numpy as np

# --------------------------------------------------------------
# Helper: next power of two (for FFT length)
def _next_pow2(x: int) -> int:
    return 1 << (x - 1).bit_length()

# --------------------------------------------------------------
# Fast convolution max using a real FFT
def _conv_max_fft(a: np.ndarray) -> tuple[float, int]:
    """
    Return (max_value, k_max) of the full linear convolution a*a.
    Uses a real FFT of length a power of two.
    """
    n = a.shape[0]
    m = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[: 2 * n - 1]
    k_max = int(conv.argmax())
    return float(conv[k_max]), k_max

# --------------------------------------------------------------
# Direct evaluation (max of convolution) – the objective when sum=√(2n)
def _objective(a: np.ndarray) -> float:
    """Return the value of evaluate_sequence for a (fast version)."""
    max_b, _ = _conv_max_fft(a)
    # The sum of a is (by construction) sqrt(2n); the objective equals max_b.
    return max_b

    rng: np.random.Generator,
    *,
    lr0: float = 0.12,
    decay: float = 4e-5,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    jitter_interval: int = 800,
) -> tuple[np.ndarray, float]:
    """
    Perform Adam optimisation on the max‑convolution objective.

    The sum of the vector is kept fixed to sqrt(2·n) after each update.
    Returns the best saw sequence and its objective value.
    """
    a = a0.copy()
    n = a.shape[0]
    target_sum = math.sqrt(2.0 * n)

    # normalise once (guards against tiny numerical drift)
    a *= target_sum / a.sum()

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = _objective(best_a)

    start = time.time()
    it = 0
    while time.time() - start < time_budget:
        # ------------------------------------------------------
        # 1. compute max‑convolution and the corresponding index
        M, k_max = _conv_max_fft(a)

        # ------------------------------------------------------
        # 2. gradient of the objective (sub‑gradient at the active diagonal)
        idx = np.arange(n)
        j = k_max - idx
        mask = (j >= 0) & (j < n)
        rev = np.zeros_like(a)
        rev[mask] = a[j[mask]]

        coeff = (4.0 * n) / (target_sum * target_sum)      # 4n / S²
        const = (4.0 * n * M) / (target_sum ** 3)          # 4n·M / S³
        grad = coeff * rev - const

        # ------------------------------------------------------
        # 3. Adam update (learning rate decays slowly)
        lr = lr0 / (1.0 + decay * it)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # ------------------------------------------------------
        # 4. projection onto the feasible set (non‑negative, correct sum)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a *= (target_sum / s)
        else:
            # should never happen, but guard against collapse
            a = best_a.copy()

        # ------------------------------------------------------
        # 5. occasional jitter to promote exploration
        if it % jitter_interval == 0 and it > 0:
            jitter = rng.normal(scale=0.0008 * target_sum / n, size=n)
            a = np.maximum(a + jitter, 0.0)
            a *= (target_sum / a.sum())

        # ------------------------------------------------------
        # 6. evaluate and keep the best
        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            # log progress
            print(
                f"[adam n={n:4d} it={it:7d}] new best = {best_score:.6f}"
            )

        it += 1

    return best_a, best_score

# Optimisation driver for a particular length
def _optimise_length(
    n: int,
    time_budget: float,
    rng: np.random.Generator,
    base_seq: np.ndarray | None = None,
) -> tuple[np.ndarray, float]:
    """
    Run several Adam restarts for a fixed length `n`.
    Returns the best sequence found and its objective.
    """
    target_sum = math.sqrt(2.0 * n)

    # a small safety margin so that the outer loop can still perform a few
    # extra checks before the global deadline
    per_restart = max(5.0, time_budget / 4.0)

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------
    # 1. use the supplied base sequence (if any) as the first restart
    if base_seq is not None and base_seq.shape[0] == n:
        a0 = base_seq.astype(float).copy()
        a0 *= target_sum / a0.sum()
        seq, sc = _adam_optimize(a0, per_restart, rng)
        if sc < best_score:
            best_seq, best_score = seq, sc

    # ------------------------------------------------------
    # 2. random‑Dirichlet restarts with different concentration parameters
    alphas = [0.5, 1.0, 2.0]               # from spiky to almost uniform
    for alpha in alphas:
        # stop early if we run out of allocated time
        if time.time() >= _global_deadline - 5.0:
            break
        a0 = rng.dirichlet(np.full(n, alpha)) * target_sum
        seq, sc = _adam_optimize(a0, per_restart, rng)
        if sc < best_score:
            best_seq, best_score = seq, sc

    return best_seq, best_score

# Global deadline – will be set inside `propose_candidate`
_global_deadline = 0.0


def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative height sequence that minimises `evaluate_sequence`.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Wall‑clock time budget (seconds).  The function aims to return a
        result before the budget expires.
    kwargs : dict
        Unused – kept for compatibility with the harness.

    Returns
    -------
    list[float]
        The best sequence discovered within the allotted time.
    """
    global _global_deadline
    rng = np.random.default_rng(seed)
    start_time = time.time()
    _global_deadline = start_time + budget_s - 4.0  # a tiny safety margin

    # ------------------------------------------------------
    # Determine the reference sequence (if available)
    base_seq = None
    try:
        base_seq = np.array(height_sequence_1, dtype=float)
    except Exception:
        base_seq = None

    # ------------------------------------------------------
    # Lengths to explore – start with the reference length,
    # then a few larger sizes to exploit extra degrees of freedom.
    lengths = []
    if base_seq is not None:
        lengths.append(base_seq.shape[0])
    # add a few handcrafted sizes (the values were chosen empirically)
    for n in (1200, 2100, 3000, 4200, 5600, 7200):
        if n not in lengths:
            lengths.append(n)

    best_overall_seq = None
    best_overall_score = float("inf")

    # ------------------------------------------------------
    # Allocate time roughly equally among the different lengths,
    # but keep a little reserve for the final check.
    remaining_lengths = len(lengths)
    for n in lengths:
        now = time.time()
        if now >= _global_deadline:
            break

        # proportion of remaining budget
        time_left = _global_deadline - now
        alloc = min(180.0, time_left * 0.9 / remaining_lengths)  # cap per length

        print(f"[master] optimise length {n} – budget {alloc:.1f}s")
        seq, sc = _optimise_length(n, alloc, rng, base_seq if n == (base_seq.shape[0] if base_seq is not None else -1) else None)

        if sc < best_overall_score:
            best_overall_score = sc
            best_overall_seq = seq.copy()
            print(f"[master] NEW GLOBAL BEST (len={n}) score={best_overall_score:.6f}")

        # Early exit if we already beat the target bound
        if best_overall_score < 1.503:
            print("[master] target bound reached – stopping early")
            break

        remaining_lengths -= 1

    # ------------------------------------------------------
    # Fallback: uniform sequence (should never be hit)
    if best_overall_seq is None:
        n = 2100
        target_sum = math.sqrt(2.0 * n)
        best_overall_seq = np.full(n, target_sum / n)

    # Final sanity check with the official evaluator (adds clipping etc.)
    final_score = evaluate_sequence(best_overall_seq.tolist())
    print(f"[final] best sequence length={best_overall_seq.shape[0]}, score={final_score:.6f}")

    return best_overall_seq.tolist()
