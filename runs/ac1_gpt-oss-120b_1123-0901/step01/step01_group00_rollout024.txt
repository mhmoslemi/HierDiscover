"""Propose a step‑function height vector that minimises
2·n·max(conv(a,a))/sum(a)^2 ≈ max(conv(a,a)) after normalising
∑a = √(2n).

The algorithm combines two ideas:
1.  Projected gradient descent on the *squared* convolution error
    L(a)=∑_k ( (a∗a)_k – μ )² where μ is the mean of the convolution.
    This drives all convolution entries towards the common value μ,
    which also reduces the max‑entry that the final score depends on.
2.  Random restarts, occasional Gaussian “shakes’’ and a simple
    adaptive step‑size.  Each iterate is Euclidean‑projected onto the
    simplex {a_i≥0, Σa_i = √(2n)} so the objective stays scale‑invariant.

The method is quick (≈ O(n²) per iteration) and works well for n up to a few
thousand within the 1000‑second budget.
"""

import time
import numpy as np

def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    return np.maximum(v - theta, 0.0)


def _convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (real, no FFT – fast enough for n≈2000)."""
    return np.convolve(a, a)


def _gradient_L2(a: np.ndarray, diff: np.ndarray) -> np.ndarray:
    """
    Gradient of   L(a) = Σ_k ( (a∗a)_k – μ )² .
    Using the identity  ∂L/∂a = 4 * ( diff ⋆ a_rev ),
    where ⋆ denotes correlation (implemented via convolution with the reversed a).
    """
    n = a.shape[0]
    # correlation = convolution of diff with reversed a
    rev = a[::-1]
    full = np.convolve(diff, rev)          # length = 3n‑2
    start = n - 1
    return 4.0 * full[start:start + n]


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) yields a low
    evaluation score.  The routine runs for at most ``budget_s`` seconds,
    keeps the best sequence discovered and finally returns it.
    """
    np.random.seed(seed)
    deadline = time.time() + max(0.0, budget_s - 0.5)   # safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Helper that performs a few gradient‑descent iterations for a fixed n.
    # ------------------------------------------------------------------
    def _optimise_fixed_n(n: int, init: np.ndarray):
        nonlocal best_seq, best_score

        target_sum = np.sqrt(2.0 * n)               # simplex radius
        a = _project_onto_simplex(init.astype(np.float64), target_sum)

        # hyper‑parameters (they will be adapted during the run)
        base_lr = 0.6
        decay = 0.9995        # slowly decay the learning‑rate
        momentum = np.zeros_like(a)

        # allocate a little time slice for this optimisation run
        run_deadline = min(deadline, time.time() + 8.0)

        while time.time() < run_deadline:
            # ----- current convolution, mean and deviation -----
            conv = _convolution(a)
            mu = np.mean(conv)                # target value (≈ sum(a)^2/(2n‑1))
            diff = conv - mu

            # ----- gradient of the L2 loss -----
            grad = _gradient_L2(a, diff)

            # adaptive step size: scale according to the current worst deviation
            max_dev = np.max(conv) - mu          # how far the max entry is above μ
            max_grad = np.max(np.abs(grad)) + 1e-12
            eta = base_lr * max_dev / max_grad

            # ----- momentum update -----
            momentum = 0.9 * momentum - eta * grad
            a_new = a + momentum

            # project back onto the simplex (guarantees feasibility)
            a_new = _project_onto_simplex(a_new, target_sum)

            # ----- evaluate -----
            try:
                score = evaluate_sequence(a_new.tolist())
            except Exception:                  # safety net
                score = float("inf")

            if score < best_score:
                best_score = score
                best_seq = a_new.copy()
                print(f"[{time.time():.2f}] New global best = {best_score:.6f} (n={n})")

            # accept the new point regardless – it is usually an improvement,
            # but we keep the old point if the update was too aggressive.
            if score <= best_score * 1.01:
                a = a_new
            else:
                # undo momentum a little when the step hurt the objective
                momentum *= 0.5

            # decay learning‑rate
            base_lr *= decay

            # occasional random shake‑up to escape flat plateaus
            if np.random.rand() < 0.015:
                noise = np.random.randn(n) * (0.02 * target_sum / n)
                a = _project_onto_simplex(np.maximum(a + noise, 0.0), target_sum)
                momentum[:] = 0.0          # reset momentum after a shake

    # ------------------------------------------------------------------
    # Main optimisation loop – many independent restarts with varying n.
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # choose problem size (larger n gives a lower theoretical bound)
        n = np.random.randint(800, 2500)

        # initialise the vector
        if 'height_sequence_1' in globals() and np.random.rand() < 0.6:
            src = np.array(height_sequence_1, dtype=float)
            if src.size >= n:
                init = src[:n].copy()
            else:
                pad = np.random.rand(n - src.size) * 1e-3
                init = np.concatenate([src, pad])
        else:
            # start from a smooth exponential shape (helps convergence)
            r = 0.998  # decay produces a gentle slope
            init = np.power(r, np.arange(n, dtype=float))
            init *= np.random.rand()  # random scale, will be projected anyway

        _optimise_fixed_n(n, init)

        # early exit if we already hit the target region
        if best_score < 1.505:
            break

    # ------------------------------------------------------------------
    # Fallback – if nothing was found, return a uniform sequence.
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n, dtype=float)

    # final sanity projection (protects against any accidental overflow)
    best_seq = _project_onto_simplex(best_seq, np.sqrt(2.0 * best_seq.size))

    return best_seq.tolist()
