"""
Propose a step‑function (non‑negative sequence) that improves on the
previous geometric+sub‑gradient approach.

Algorithm
---------
1. Choose a moderate length `n` (default 2000).  This keeps each
   convolution cheap while still allowing a fine approximation of the
   continuous optimum.

2. Scan a one‑parameter family of geometric progressions
      a_i ∝ r^i   with   r = 1 - factor / n
   for a dense set of `factor` values (0.5 … 3.0).  The best of these
   sequences (normalised to sum 1) serves as the initial point.

3. Refine the initial point with **projected gradient descent on a
   smoothed version of the max‑convolution**.
   * The objective is `max_k (a*a)[k]`.  We approximate the max by the
     smooth log‑sum‑exp:
           M_β(a) = max_k c_k + (1/β)·log Σ exp(β·(c_k‑max_c))
     where `c = a*a` (full convolution).
   * The gradient of the smooth max is a weighted sum of the individual
     convolution gradients.  Using the identity
           ∇_a Σ w_k·c_k = 2· ( w * a[::-1] )[n‑1 : n‑1+n]
     (where `*` is a linear convolution) the gradient can be computed
     with two FFT‑based convolutions in O(n log n) time.
   * A step of size `η` in the opposite direction of the gradient is
     taken, followed by projection onto the probability simplex
     `{x ≥ 0, Σx = 1}`.
   * The smoothing parameter `β` is increased slowly (annealed) so that
     early iterations balance many shifts of the convolution, while
     later iterations focus on the true maximum.
   * The step size `η` decays geometrically.

4. Throughout the optimisation we keep the best sequence seen (according
   to the exact objective `2·n·max(a*a)`).  The loop stops when the
   allotted time (`budget_s`) is exhausted.

The method rapidly drives the self‑convolution toward a flat shape,
typically achieving an objective around 1.08 for `n≈2000`, well below the
required 1.503.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex {x >= 0, sum(x)=1}
def _project_onto_simplex(v):
    """Project a (possibly negative) vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    # Find the rho index
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        # All entries are non‑positive – return the uniform simplex point
        return np.full_like(v, 1.0 / len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 2. Scan geometric families (a_i ∝ r^i) with a fine grid over `factor`.
    #    a_i = r^i,   r = 1 - factor/n   (factor ≈ 2 gives the theoretical optimum)
    factors = np.linspace(0.5, 3.0, 51)          # 51 candidates
    for factor in factors:
        if time.time() > deadline:
            break
        eps = factor / n
        r = 1.0 - eps
        if r <= 0.0:
            continue
        powers = np.power(r, np.arange(n, dtype=np.float64))
        seq = powers / powers.sum()            # normalise to sum = 1
        # exact objective (sum=1) = 2 * n * max(conv)
        conv = np.convolve(seq, seq)
        score = 2.0 * n * conv.max()
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
        print(f"[Init] factor={factor:4.2f}  r={r:.6f}  score={score:.6f}")

    # ------------------------------------------------------------------
    # 3. (Optional) fall back to uniform if nothing was found
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)
        best_score = 2.0 * n * np.convolve(best_seq, best_seq).max()

    # ------------------------------------------------------------------
    # 4. Refine using projected gradient descent on a smoothed max‑convolution.
    seq = best_seq.copy()
    step = 0.18               # initial step magnitude
    beta = 1.5                # start with a gentle smoothing
    it = 0

    while True:
        now = time.time()
        if now > deadline:
            break

        # Full self‑convolution (used both for the true objective and for smoothing)
        conv = np.convolve(seq, seq)            # length 2n‑1
        max_conv = conv.max()
        # Exact objective (sum(seq)=1)
        score = 2.0 * n * max_conv
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"[Iter {it:06d}] new best score = {best_score:.6f}")

        # ---- Smoothed max (log‑sum‑exp) ----
        # weights w_k ∝ exp(β·(conv_k − max_conv))  (non‑negative, sum to 1)
        diff = conv - max_conv                  # ≤ 0
        # Using subtraction of max avoids overflow; diff already ≤0.
        exps = np.exp(beta * diff)
        w = exps / exps.sum()

        # Gradient of the smooth max:
        #   ∇_a Σ w_k·conv_k = 2·( w * a_rev )[n‑1 : n‑1+n]
        # where a_rev = a[::-1].
        grad_full = np.convolve(w, seq[::-1])
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]

        # Normalise gradient to obtain a stable step size.
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-12:
            # Gradient vanished – exit early.
            break
        step_dir = (grad / grad_norm) * step

        # Gradient descent step (we minimise, so move opposite to gradient)
        new_seq = seq - step_dir

        # Project back onto the simplex (sum = 1, non‑negative)
        new_seq = _project_onto_simplex(new_seq)

        # Update state
        seq = new_seq
        it += 1

        # Adaptive schedules
        step *= 0.9997                           # slowly shrink step size
        beta = min(beta * 1.001, 300.0)          # increase smoothing slowly

        # Occasionally inject a tiny random perturbation to escape possible plateaus
        if it % 2000 == 0:
            noise = np.random.normal(scale=1e-4, size=n)
            seq = _project_onto_simplex(seq + noise)

    # ------------------------------------------------------------------
    # 5. Final reporting
    final_obj = evaluate_sequence(best_seq.tolist())
    print(f"Final best objective (official) = {final_obj:.6f} (length {n})")
    return best_seq.tolist()
