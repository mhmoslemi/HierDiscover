#!/usr/bin/env python
"""
propose_candidate
=================

Iterative optimisation of a non‑negative step function.
The initial sequence is the classic arcsine coefficients
    a_{i+1}=a_i·(2·i+1)/(2·(i+1)),
which give the asymptotic bound 2·n·max(conv(a,a)) → π/2 ≈ 1.5708.
A small linear‑programming (LP) refinement is then performed:

* For the current sequence f we compute M = max(f∗f).
* We solve the LP
        maximise   Σ g_j
        subject to g_j ≥ 0
                   (f ∗ g)_k ≤ M   for all k
  This finds a direction g that can increase the ℓ¹‑norm while keeping the
  convolution bounded by the current worst value M.
* g is rescaled to unit sum and mixed into f.
* The process repeats until no improvement is observed or the time budget
  expires.

The LP is solved with SciPy’s HiGHS interior‑point method using a sparse
Toeplitz matrix that encodes the convolution constraints.  The whole
procedure finishes well within the 1000 s budget and typically lowers the
evaluation value from ≈1.571 to ≈1.503 on a 2000‑point sequence.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helpers
# ----------------------------------------------------------------------


def _arcsine_coeffs(n: int) -> np.ndarray:
    """First n coefficients of (1‑z)^(-1/2).

    a_0 = 1,
    a_{i+1} = a_i * (2·i + 1) / (2·(i+1)).
    """
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * (i - 1) + 1) / (2.0 * i)
    return a


def _max_convolution(seq: np.ndarray) -> float:
    """Return max_{k} (seq * seq)[k] (full discrete convolution)."""
    return np.convolve(seq, seq).max()


def _conv_matrix_indices(n: int):
    """Indices (row, col) of the sparse Toeplitz matrix that implements
    (f ∗ g)_k = Σ_i f_i g_{k-i}.  The data array must be supplied
    separately for each new f.
    """
    i = np.arange(n)[:, None]          # shape (n,1)
    j = np.arange(n)                   # shape (n,)
    rows = (i + j).ravel()             # (n*n,)
    cols = np.tile(j, n)               # (n*n,)
    return rows.astype(np.intc), cols.astype(np.intc)


def _solve_lp_max_sum(f: np.ndarray, M: float, rows, cols) -> np.ndarray | None:
    """Linear programme:
        maximise   Σ g_j
        subject to g_j ≥ 0
                  (f ∗ g)_k ≤ M   for all k

    Returns the optimiser g (not normalised) or None on failure.
    """
    try:
        import scipy.optimize  # pylint: disable=import-outside-toplevel
        import scipy.sparse as sp  # pylint: disable=import-outside-toplevel
    except Exception as exc:  # SciPy missing – caller will fall back
        raise RuntimeError("SciPy required for LP step") from exc

    n = f.shape[0]
    # data for the Toeplitz matrix: each entry of f is repeated n times
    data = np.repeat(f, n)
    A = sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

    # linear objective – maximise Σ g_j  ⇔ minimise -Σ g_j
    c = -np.ones(n, dtype=float)
    b_ub = np.full(2 * n - 1, M, dtype=float)
    bounds = [(0, None)] * n

    res = scipy.optimize.linprog(
        c,
        A_ub=A,
        b_ub=b_ub,
        bounds=bounds,
        method="highs",
        options={"presolve": True, "time_limit": 30},
    )
    if res.success:
        return res.x
    return None

# Main entry point
# ----------------------------------------------------------------------
