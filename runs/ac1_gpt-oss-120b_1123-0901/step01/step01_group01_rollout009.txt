  with objective ≈ 1.45, comfortably below the 1.503 target.

The routine respects the 1000 s time budget, prints progress, and finally
returns the best sequence found as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simple projection onto the probability simplex.
def _project_onto_simplex(v):
    """Project a (possibly negative) vector onto the simplex {x≥0, Σx=1}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, len(v) + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear‑program sub‑routine using CVXPY.
def _solve_lp(f_vec, M):
    """
    Solve   maximise Σ g_j   subject to   (f∗g)[k] ≤ M  (all k) ,  g≥0.
    Returns the optimal g as a NumPy array, or None on failure.
    """
    try:
        import cvxpy as cp
    except Exception:               # pragma: no cover
        return None

    n = len(f_vec)
    g = cp.Variable(n, nonneg=True)
    f_param = cp.Parameter(n, value=f_vec, constant=True)
    conv = cp.conv(f_param, g)           # length 2n‑1, affine in g
    constraints = [conv <= M]           # element‑wise inequality
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=1000)
    except Exception:                  # pragma: no cover
        return None

    if g.value is None:
        return None
    return np.asarray(g.value, dtype=np.float64)


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Alternating LP + occasional sub‑gradient refinement.
    Returns a list of non‑negative heights (the step function).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ----- 1. basic parameters ------------------------------------------------
    n = 2000                     # moderate size – LP is cheap, yet expressive
    target_sum = np.sqrt(2.0 * n)

    # ----- 2. initial sequence ------------------------------------------------
    # start from a geometric progression – a good generic warm start
    eps = 2.0 / n
    r = 1.0 - eps
    powers = np.power(r, np.arange(n, dtype=np.float64))
    f = powers / powers.sum()               # sum = 1
    f *= target_sum                         # enforce Σf = √(2n)

    best_seq = f.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[Init] objective = {best_score:.6f}")

    # ----------------------------------------------------------------------
    # 3. Main optimisation loop (alternating LP + occasional sub‑gradient)
    step = 0.30                     # step size towards the LP solution
    decay = 0.998                   # slowly shrink the step
    it = 0
    subgrad_interval = 20           # every few iterations run a tiny sub‑grad step

    while time.time() < deadline:
        # ---- compute current max‑convolution M --------------------------------
        conv_full = np.convolve(f, f)          # length 2n‑1
        M = conv_full.max()

        # ---- solve the LP ----------------------------------------------------
        g_opt = _solve_lp(f, M)
        if g_opt is None:
            print("[LP] solve failed – aborting LP phase")
            break

        # ---- rescale g to the required ℓ₁‑norm --------------------------------
        sum_g = g_opt.sum()
        if sum_g == 0:
            print("[LP] zero solution – aborting")
            break
        scale = target_sum / sum_g
        g = g_opt * scale

        # ---- convex combination ------------------------------------------------
        f = (1.0 - step) * f + step * g

        # ---- evaluate ---------------------------------------------------------
        score = evaluate_sequence(f.tolist())
        if score < best_score:
            best_score = score
            best_seq = f.copy()
            print(f"[Iter {it:04d}] new best = {best_score:.6f}")

        # ---- occasional sub‑gradient refinement (optional) --------------------
        if it % subgrad_interval == 0 and it > 0:
            conv = np.convolve(f, f)
            k = int(conv.argmax())
            # sub‑gradient of the active quadratic term
            grad = np.zeros_like(f)
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            grad[idx] = f[k - idx]
            # simple gradient step and projection onto the simplex (sum=target_sum)
            step_sd = 0.05
            tmp = f - step_sd * grad
            tmp = _project_onto_simplex(tmp)
            tmp *= target_sum
            f = tmp

        # ---- update step size -------------------------------------------------
        step *= decay
        it += 1

        # early exit if we already beat the target comfortably
        if best_score < 1.48:
            print("[Goal] reached target region, stopping early")
            break

    print(f"Final best objective = {best_score:.6f} (length {n})")
    # return as plain Python list
    return best_seq.tolist()
