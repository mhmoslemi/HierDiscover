"""
Propose a new step‑function sequence.

The algorithm works directly on the probability simplex Σa_i = 1.
Starting from the best known construction (the global variable
`height_sequence_1` if present, otherwise the central‑binomial series)
it alternates:

1. Compute the self‑convolution a*a (FFT, length 2·n‑1) and locate the current
   maximal value M = max(conv).  The evaluation metric for Σa_i = 1 is
   2·n·M, so minimising M is enough.

2. Build an *active set* of convolution indices whose value is within a small
   tolerance of M.  The indicator of that set (0/1) is cross‑correlated with
   the current a; the resulting vector g = 2·(a ⋆ ind) tells how much each
   coefficient contributes to the peak region.

3. Move a small amount of mass ε from the index with largest g (src) to the
   index with smallest g (dst).  The move is accepted only if the new peak
   does not increase; otherwise it is rejected.

4. The step size ε is proportional to a[src] and slowly decays.
   Every few hundred iterations the full convolution is recomputed (to avoid
   drift) and a tiny random jitter is added to escape plateaus.

The loop runs until the time budget (≈ 1000 s) expires.  The best sequence
found (lowest 2·n·max(conv)) is returned as a plain list of floats.
"""

import time
import math
import numpy as np

def _fft_size_for_length(length: int) -> int:
    """Return the next power of two >= length (for FFT)."""
    return 1 << (length - 1).bit_length()


def propose_candidate(seed: int = 0, budget_s: float = 1000.0) -> list[float]:
    """Return a non‑negative sequence that (hopefully) improves the bound.

    Args:
        seed: random seed for reproducibility.
        budget_s: total time budget in seconds.

    Returns:
        List of floats – the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0          # small safety margin

    # ----------------------------------------------------------------------
    # 1. Choose a length and an initial vector a (normalised to sum = 1)
    # ----------------------------------------------------------------------
    try:
        a = np.array(height_sequence_1, dtype=np.float64)   # type: ignore
        n = a.size
    except NameError:
        # fallback: 6000 entries of the central‑binomial series
        n = 6000
        a = np.empty(n, dtype=np.float64)
        a[0] = 1.0
        for i in range(1, n):
            a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)

    # enforce non‑negativity and normalise to a probability simplex
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s

    # FFT size sufficient for both the self‑convolution (2n‑1) and the
    # cross‑correlation with an indicator of length 2n‑1 (needs 3n‑2).
    fft_size = _fft_size_for_length(3 * n - 2)

    # ----------------------------------------------------------------------
    # Helper: compute the full linear convolution a*a (length 2·n‑1)
    # ----------------------------------------------------------------------
    def _self_convolution(x: np.ndarray) -> np.ndarray:
        fx = np.fft.rfft(x, fft_size)
        conv_full = np.fft.irfft(fx * fx, fft_size)
        return conv_full[: 2 * n - 1]

    # ----------------------------------------------------------------------
    # Helper: cross‑correlation a ⋆ ind (return first n entries)
    #           ind is a 0/1 array of length 2·n‑1
    # ----------------------------------------------------------------------
    def _cross_correlation(x: np.ndarray, ind: np.ndarray) -> np.ndarray:
        fx = np.fft.rfft(x, fft_size)
        fi = np.fft.rfft(ind, fft_size)
        corr_full = np.fft.irfft(fx * np.conj(fi), fft_size)
        # the full correlation has length n + (2n‑1) – 1 = 3n‑2
        return corr_full[:n]        # first n entries correspond to a ⋆ ind

    # ----------------------------------------------------------------------
    # Initialise bookkeeping
    # ----------------------------------------------------------------------
    conv = _self_convolution(a)
    cur_max = conv.max()
    best_a = a.copy()
    best_max = cur_max
    best_score = 2.0 * n * best_max
    iteration = 0

    # hyper‑parameters
    step_factor = 0.2            # relative size of the transferred mass
    delta_frac = 1e-3            # width of the active set (relative to max)
    decay_rate = 0.9995          # slowly shrink step size

    print(f"Start: n={n}, initial score={best_score:.6f}")

    # ----------------------------------------------------------------------
    # Main optimisation loop
    # ----------------------------------------------------------------------
    while time.time() < deadline:
        # recompute convolution (occasionally it drifts)
        if iteration % 250 == 0:
            conv = _self_convolution(a)
            cur_max = conv.max()
        else:
            cur_max = conv.max()

        # active set where the convolution is close to its peak
        delta = cur_max * delta_frac
        active = (conv >= cur_max - delta).astype(np.float64)

        # aggregated gradient: g = 2 * (a ⋆ active)
        g = 2.0 * _cross_correlation(a, active)

        # choose source (largest contribution) and destination (smallest)
        src = int(np.argmax(g))
        dst = int(np.argmin(g))

        if src == dst:
            # very unlikely, pick a random pair instead
            src, dst = np.random.choice(n, 2, replace=False)

        eps = a[src] * step_factor * np.random.rand()
        # keep the move safe
        eps = min(eps, a[src])
        if eps <= 0.0:
            # nothing to move – break out
            break

        # propose a new vector
        a_new = a.copy()
        a_new[src] -= eps
        a_new[dst] += eps

        # compute the new convolution and its peak
        conv_new = _self_convolution(a_new)
        new_max = conv_new.max()

        # accept only non‑worsening moves (greedy)
        if new_max <= cur_max * (1 + 1e-14):
            a = a_new
            conv = conv_new
            cur_max = new_max

            if new_max < best_max:
                best_max = new_max
                best_a = a.copy()
                best_score = 2.0 * n * best_max
                print(f"Iter {iteration:6d}: new best score = {best_score:.6f}")
        else:
            # reject – no change to a or conv
            pass

        # slowly shrink the step size
        step_factor = max(step_factor * decay_rate, 1e-4)

        # occasional random jitter to escape plateaus
        if iteration % 1000 == 0 and iteration > 0:
            jitter = np.random.randn(n) * 5e-5
            a = np.maximum(a + jitter, 0.0)
            a /= a.sum()
            conv = _self_convolution(a)

        iteration += 1

    final_score = 2.0 * n * best_max
    print(f"Finished after {iteration} iterations. Best score = {final_score:.6f}")

    # The evaluation metric is homogeneous, so any scaling works.
    # We keep the sum = 1 (the safest choice for the evaluator).
    return best_a.tolist()
