def _conv_fft(x, y):
    n = len(x)
    m = len(y)
    size = 1 << ((n + m - 1).bit_length())
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[: n + m -1]
    return conv

"""
Propose a candidate step‑function sequence that (hopefully) improves the
self‑convolution peak metric.

Algorithm
---------
1. Choose several problem sizes (lengths).  If a previous solution
   `height_sequence_1` exists its length is also tried.
2. For each length run a few *restarts* with different initialisations:
   - central‑binomial coefficients (the classic start)
   - a simple power‑law a_i ∝ 1/√(i+1)
   - a random Dirichlet vector.
3. Optimise a *smooth surrogate* of the peak:
   f(a)= ε·log( Σ_k exp( (a∗a)_k / ε) )
   whose gradient is
        ∇f_i = 2· Σ_k w_k·a_{k‑i} ,   w_k ∝ exp( (a∗a)_k / ε )
   This is computed efficiently with FFTs.
4. Perform projected (Euclidean) gradient descent on the simplex
   {a≥0, Σa=1}.  The step size decays like 1/√t and a small back‑tracking
   line‑search guarantees non‑increase of the true objective
   (2·n·max(a∗a)).
5. Occasionally inject tiny random perturbations and slowly decay ε so
   that the optimisation first smooths the whole convolution and finally
   concentrates on the true maximum.
6. Keep the globally best vector (across all lengths/restarts) and return
   it as a plain Python list.

The method works directly with the homogeneous formulation (Σa=1), so the
returned sequence is already feasible for `evaluate_sequence`.
"""

import numpy as np
import time
import math

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    { x ∈ ℝⁿ | x_i ≥ 0, Σ_i x_i = 1 }.
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # Find the rho index
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero; return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Linear convolution of two real vectors using FFT.
    Returns a vector of length len(x)+len(y)-1.
    """
    n = x.shape[0]
    m = y.shape[0]
    size = 1 << ((n + m - 1).bit_length())
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[: n + m - 1]
    return conv


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a ∗ a (length 2·len(a)‑1) via FFT."""
    return _conv_fft(a, a)


def _initial_central_binomial(n: int) -> np.ndarray:
    """Central‑binomial coefficients a_i = C(2i,i)/4**i, truncated to length n."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return _project_onto_simplex(a)


def _initial_power_law(n: int, p: float = 0.5) -> np.ndarray:
    """Simple power‑law a_i ∝ (i+1)^{-p}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    return _project_onto_simplex(a)


def _initial_random_dirichlet(n: int) -> np.ndarray:
    """Random Dirichlet(1,…,1) – i.e. normalised Gamma(1,1)."""
    a = np.random.gamma(1.0, 1.0, size=n)
    return _project_onto_simplex(a)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------

    Search for a non‑negative sequence that minimises the evaluation metric.
    The function runs for at most `budget_s` seconds and returns the best
    sequence discovered as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # safety margin

    best_a = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 1. Choose lengths to explore
    # ------------------------------------------------------------------
    length_options = [6000, 8000, 10000]
    try:

        if height_sequence_1:   # type: ignore[name-defined]

    except NameError:
        pass
    length_options = sorted(set(length_options))

    # ------------------------------------------------------------------
    # 2. Optimisation loop over lengths
    # ------------------------------------------------------------------
    for n in length_options:
        if time.time() >= deadline:

        print(f"\n=== Optimising length n = {n} ===")
        # Allocate a roughly equal slice of the remaining budget to this n
        length_deadline = time.time() + (deadline - time.time()) / 2.0

        # ------------------------------------------------------------------

        # ------------------------------------------------------------------
        for restart_id in range(3):

            elif restart_id == 1:

    return best_a.tolist()
