import time, numpy as np

def propose_candidate(seed=123, budget_s=1000, **kwargs):
    rng = np.random.default_rng(seed)

    n = len(height_sequence_1) if defined else 2000
    base_seq = generate_inverse_sqrt_sequence(n)
    seq = np.array(base_seq, dtype=float)

    best_seq = seq.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] n={n} score={best_score}")

    deadline = time.time() + budget_s - 0.5
    # parameters
    epsilon = 0.05  # fraction of donor amount to move
    early_fraction = 0.25
    late_fraction = 0.75

    while time.time() < deadline:
        # compute convolution and max
        b = np.convolve(seq, seq)
        M = b.max()
        # choose donor from early indices
        donor_range = int(n * early_fraction)
        receiver_range_start = int(n * late_fraction)

        donor = rng.integers(donor_range)  # 0..donor_range-1
        receiver = rng.integers(receiver_range_start, n)  # later half

        # amount to transfer
        amount = seq[donor] * rng.uniform(0.001, epsilon)  # small fraction
        if amount <= 0:
            continue

        new_seq = seq.copy()
        new_seq[donor] -= amount
        new_seq[receiver] += amount

        # ensure nonnegative
        if new_seq[donor] < 0:
            continue

        cand_score = evaluate_sequence(new_seq.tolist())
        if cand_score < best_score:
            best_score = cand_score
            best_seq = new_seq.copy()
            seq = new_seq  # continue from improved
            print(f"[improve] score={best_score:.6f}")
            continue

        # optionally accept with probability if slight worsening (simulated annealing)
        # but we might skip

        # also occasional random multiplicative move to escape
        if rng.random() < 0.001:

"""
Propose a candidate step‑function sequence that improves on the classic
inverse‑square‑root (central‑binomial) construction.

Algorithm
---------

1. **Base shape** – Use the well‑known sequence  
   a_i = C(2i,i)/4**i  (generated by the recurrence a_{i+1}=a_i*(2i+1)/(2i+2)).
   This shape makes the convolution exactly flat (=1) for the first half of the
   indices, giving a score of π/2 ≈ 1.5708.

2. **Mass‑transfer hill‑climbing** – The current maximum of the convolution
   (≈1) is attained for many “early’’ indices (k ≤ n‑1).  For those indices the
   contribution of the first quarter of the array is large, while the tail
   (k > n‑1) still has a large slack (its convolution value is well below 1).
   We therefore repeatedly *move a tiny amount of mass* from an early element
   to a late element.  The move keeps the vector non‑negative, is inexpensive,
   and usually reduces the evaluation because the sum grows while the maximal
   convolution stays at the same bound.

3. **Acceptance rule** – A move is kept if it strictly improves the score.
   In addition a simple simulated‑annealing rule allows occasional uphill
   moves, helping the search escape shallow plateaus.

4. **Fine‑grained local tweaks** – When the time budget is almost exhausted a
   cheap single‑entry log‑normal perturbation (the trick used in the reference
   solution) is applied to polish the result.

The whole routine respects the 1000 s budget, prints progress information and
returns the best list of non‑negative floats found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: produce the “inverse‑square‑root’’ (central‑binomial) sequence.
# ----------------------------------------------------------------------
def generate_inverse_sqrt_sequence(n: int) -> np.ndarray:
    """Return a NumPy array a[0..n‑1] with a_i = C(2i,i)/4**i.
    The recurrence a_{i+1}=a_i*(2i+1)/(2i+2) avoids large intermediate numbers."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


# ----------------------------------------------------------------------
# Main entry point – the runner will call this function.
# ----------------------------------------------------------------------

    """
    Search for a sequence of non‑negative heights that minimizes
    evaluate_sequence.  The routine starts from the optimal inverse‑square‑root
    shape and performs a specialised hill‑climbing that transfers mass from
    early to late indices, optionally accepting occasional uphill moves.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1. Length of the vector.
    # --------------------------------------------------------------
    try:
        n = len(height_sequence_1)      # type: ignore[name-defined]
    except Exception:

    # --------------------------------------------------------------
    # 2. Base sequence (central‑binomial coefficients).
    # --------------------------------------------------------------
    base_seq = generate_inverse_sqrt_sequence(n)
    seq = base_seq.copy()               # current working sequence (NumPy array)

    # --------------------------------------------------------------
    # 3. Initial score.
    # --------------------------------------------------------------
    best_seq = seq.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] length={n}, initial score={best_score:.6f}")

    # --------------------------------------------------------------
    # 4. Hill‑climbing parameters.
    # --------------------------------------------------------------
    # fraction of the array considered “early’’ (source of mass)
    early_frac = 0.25
    # fraction where we try to deposit mass
    late_frac = 0.75
    early_end = int(n * early_frac)
    late_start = int(n * late_frac)

    # maximum fraction of the donor entry that may be moved in one step
    max_transfer_frac = 0.03    # 3 % of the donor’s current value
    # annealing schedule (temperature decays linearly to 0)
    start_temp = 5e-3           # small temperature – moves are mostly greedy

    deadline = time.time() + budget_s - 0.2    # tiny margin for safety

    # --------------------------------------------------------------
    # 5. Main optimisation loop.
    # --------------------------------------------------------------
    while time.time() < deadline:
        # -----------------------------------------------------------------
        # 5.1  Choose a donor from the “early’’ part of the array.  Heavier
        #      entries are more likely to be donors because they influence
        #      the maximal convolution the most.
        # -----------------------------------------------------------------
        donor = rng.integers(early_end)
        donor_val = seq[donor]
        if donor_val <= 0.0:
            continue

        # -----------------------------------------------------------------
        # 5.2  Choose a receiver from the “late’’ part of the array.
        # -----------------------------------------------------------------
        receiver = rng.integers(late_start, n)
        # -----------------------------------------------------------------
        # 5.3  Amount of mass to transfer – a random fraction of the donor.
        # -----------------------------------------------------------------
        transfer = donor_val * rng.uniform(1e-4, max_transfer_frac)

        # Apply the transfer tentatively.
        new_seq = seq.copy()
        new_seq[donor] -= transfer
        new_seq[receiver] += transfer

        # --------------------------------------------------------------
        # 5.4  Evaluate the candidate.
        # --------------------------------------------------------------
        cand_score = evaluate_sequence(new_seq.tolist())
        if not np.isfinite(cand_score):
            # Safety: should never happen because we stay non‑negative.
            continue

        # --------------------------------------------------------------
        # 5.5  Acceptance test.
        # --------------------------------------------------------------
        if cand_score < best_score:
            # Strict improvement – accept unconditionally.
            best_score = cand_score
            best_seq = new_seq.copy()
            seq = new_seq
            print(f"[improve] score={best_score:.6f} "
                  f"(donor {donor}, receiver {receiver})")
            continue

        # Simulated‑annealing style occasional uphill acceptance.
        # Temperature decreases linearly with the remaining time.
        remaining = deadline - time.time()
        total = budget_s - 0.2
        temp = start_temp * (remaining / total)
        prob = np.exp((best_score - cand_score) / max(temp, 1e-12))
        if rng.random() < prob:
            # Accept the worse move (helps escape flat plateaus).
            seq = new_seq
            # No update of best_seq / best_score because it is not better.
            continue

        # --------------------------------------------------------------

        #      refinement used in the reference solution) – this helps to
        #      explore directions that the mass‑transfer operator does not
        #      reach.
        # --------------------------------------------------------------
        if rng.random() < 0.001:   # very rare
            i = rng.integers(n)
            factor = np.exp(rng.normal(0.0, 0.06))
            alt_seq = seq.copy()
            alt_seq[i] = max(0.0, alt_seq[i] * factor)
            alt_score = evaluate_sequence(alt_seq.tolist())
            if alt_score < best_score:
                best_score = alt_score
                best_seq = alt_seq.copy()
                seq = alt_seq
                print(f"[variant] score={best_score:.6f} (index {i})")

    # --------------------------------------------------------------
    # 6. Final polishing: a short burst of single‑entry log‑normal moves.
    # --------------------------------------------------------------
    polishing_deadline = time.time() + 0.15
    while time.time() < polishing_deadline:

        old = best_seq[i]
        factor = np.exp(rng.normal(0.0, 0.02))
        cand = best_seq.copy()
        cand[i] = max(0.0, old * factor)
        cand_score = evaluate_sequence(cand.tolist())
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand
            # keep the improvement for a while
            # (no need to recompute anything else)

    print(f"[final] best score={best_score:.6f}")
    # Return a plain Python list (the runner expects a list, not a NumPy array)
    return best_seq.tolist()
