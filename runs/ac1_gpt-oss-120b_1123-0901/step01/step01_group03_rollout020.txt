    if kind == 'binomial':
        a = np.empty(n)
        a[0] = 1.0
        for i in range(1,n):
            a[i]=a[i-1]*(2*i-1)/(2*i)

        a = np.ones(n)
    elif kind == 'random':
        a = np.random.rand(n)  # random positive
    else:
        a = np.ones(n)
    a = _project_onto_simplex(a)

def _project_onto_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), 1):
        return v
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

def _full_convolution(x, y=None):
    if y is None:
        y = x
    n = len(x)
    m = len(y)
    size = 1 << ((n + m - 1).bit_length())
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[: n + m - 1]
    return conv

def _optimize(a, n, deadline, max_iter=5000):
    best_a = a.copy()
    conv = _full_convolution(a)
    best_score = 2.0 * n * conv.max()
    iteration = 0
    momentum = np.zeros_like(a)
    beta = 0.9
    eta0 = 1.0  # base learning rate
    p = 4.0
    max_p = 64.0
    for iteration in range(max_iter):
        if time.time() >= deadline:
            break
        conv = _full_convolution(a)
        max_val = conv.max()
        # compute gradient for smooth max
        Num = np.sum(conv ** p)
        f = Num ** (1.0 / p)
        # factor = 2 * f^{1-p}
        factor = 2.0 * (f ** (1.0 - p))
        w = conv ** (p - 1)
        a_rev = a[::-1]
        corr = _full_convolution(w, a_rev)[n-1:n-1+n]   # length n
        grad = factor * corr
        # momentum update
        momentum = beta * momentum + (1 - beta) * grad
        eta = eta0 / np.sqrt(iteration + 1)
        a_new = a - eta * momentum
        a_new = _project_onto_simplex(a_new)
        # compute new max to accept or revert
        conv_new = _full_convolution(a_new)
        new_max = conv_new.max()
        if new_max <= max_val:
            a = a_new
            max_val = new_max
        else:
            # backtrack attempt smaller step
            for scale in [0.5, 0.2, 0.1]:
                a_candidate = a - eta * scale * momentum
                a_candidate = _project_onto_simplex(a_candidate)
                conv_c = _full_convolution(a_candidate)
                if conv_c.max() <= max_val:
                    a = a_candidate
                    max_val = conv_c.max()
                    break
            else:
                # random perturb
                a = _project_onto_simplex(a + np.random.randn(n) * 0.005)
        # update best
        score = 2.0 * n * max_val
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"N={n}, iter={iteration}, p={p:.1f}, score={best_score:.6f}")
        # increase p occasionally
        if iteration % 500 == 0 and p < max_p:
            p = min(p * 2, max_p)
    return best_a, best_score

  sum = 1) and their metric `2·n·max(conv(a,a))` is evaluated.
* **Local smoothing optimisation** – the best few candidates are refined by
  gradient descent on a smooth approximation of the peak:
        f_p(a) = ( Σ (a∗a)^p )^{1/p}   (p→∞ gives the true max).
  The gradient of `f_p` is
        ∇f = 2·f^{1‑p} · corr( (a∗a)^{p‑1} , a )
  where `corr` is the (linear) correlation.  Momentum and a simple
  back‑tracking line search are employed; after each step the vector is
  projected back onto the simplex.
* **Mixing** – occasional convex combinations of the current best vector with
  the “arcsine” (central‑binomial) distribution are tried.
* The search runs until the supplied time budget (≈ 1000 s) expires.  The
  best sequence discovered is returned as a plain list of floats.
"""

import numpy as np
import time
import math

# ----------------------------------------------------------------------
# Helper functions
# ----------------------------------------------------------------------


def _project_onto_simplex(v):
    """Euclidean projection of a vector onto the simplex {x ≥ 0, Σx = 1}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # avoid division by zero in pathological case
    s = w.sum()
    if s <= 0:
        w.fill(1.0 / n)
    else:
        w /= s
    return w


def _full_convolution(x, y=None):
    """Linear convolution using real FFT (full length)."""
    if y is None:
        y = x
    n = len(x)
    m = len(y)
    size = 1 << ((n + m - 1).bit_length())  # next power of two
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[: n + m - 1]
    return conv


def _init_power_law(n, alpha):
    """a_i ∝ (i+1)^{-α}."""
    i = np.arange(n, dtype=np.float64) + 1.0
    a = 1.0 / (i ** alpha)
    return _project_onto_simplex(a)


def _init_exponential(n, lam):
    """a_i ∝ exp(-λ i)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    return _project_onto_simplex(a)


def _init_beta_distribution(n, p, q):
    """Beta‑type distribution a_i ∝ (i+½)^{p‑1}·(n‑½‑i)^{q‑1}."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 0.5) ** (p - 1.0) * (n - 0.5 - i) ** (q - 1.0)
    # Guard against infinities / NaNs for extreme parameters
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    return _project_onto_simplex(a)


def _evaluate_metric(a):
    """Metric 2·n·max(conv(a,a)) for a that already sums to 1."""
    n = len(a)
    conv = _full_convolution(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Smoothing + momentum optimiser
# ----------------------------------------------------------------------


def _smooth_max_optimize(a0, deadline, max_iter=5000,
                         p_start=4.0, p_max=64.0):
    """
    Refine a starting vector a0 by minimising a smoothed sup‑norm
    of the self‑convolution using momentum and projection onto the simplex.
    """
    a = a0.copy()
    n = len(a)
    best_a = a.copy()
    best_score = _evaluate_metric(a)
    # optimisation state
    momentum = np.zeros_like(a)
    beta = 0.9
    eta0 = 1.0
    p = p_start
    it = 0
    # main loop
    while time.time() < deadline and it < max_iter:
        conv = _full_convolution(a)                     # a*a
        max_val = conv.max()
        # smooth‑max (p‑norm) and its derivative factor
        conv_pow = conv ** p
        f = conv_pow.sum() ** (1.0 / p)                # (Σ conv^p)^{1/p}
        factor = 2.0 * (f ** (1.0 - p))                # 2·f^{1‑p}
        # gradient = factor * corr( conv^{p‑1} , a )
        w = conv ** (p - 1.0)                          # (a*a)^{p‑1}
        a_rev = a[::-1]
        corr_full = _full_convolution(w, a_rev)        # length 3n‑2
        grad = factor * corr_full[n - 1:n - 1 + n]      # keep n entries
        # momentum update
        momentum = beta * momentum + (1.0 - beta) * grad
        eta = eta0 / math.sqrt(it + 1.0)
        # trial step
        a_candidate = _project_onto_simplex(a - eta * momentum)
        conv_cand = _full_convolution(a_candidate)
        if conv_cand.max() <= max_val:
            a = a_candidate
            max_val = conv_cand.max()
        else:
            # back‑track: try smaller steps
            improved = False
            for factor_shrink in (0.5, 0.2, 0.1):
                a_tmp = _project_onto_simplex(a - eta * factor_shrink * momentum)
                if _full_convolution(a_tmp).max() <= max_val:
                    a = a_tmp
                    improved = True
                    break
            if not improved:
                # small random kick to escape
                a = _project_onto_simplex(
                    a + np.random.randn(n) * 3e-4
                )
        # record best
        score = 2.0 * n * max_val
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"  *** refinement n={n} iter={it} p={p:.1f} → {best_score:.6f}")
        # schedule for p (increase every few hundred iterations)
        if it % 400 == 0 and p < p_max:
            p = min(p * 2.0, p_max)
        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the evaluation metric.
    Returns the best sequence found before the time budget expires.
    """
    np.random.seed(0)               # deterministic enough for reproducibility
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # keep a small safety margin

    best_seq = None
    best_score = float('inf')

    # ------------------------------------------------------------------
    # 1.  Existing construction (if any) – use as initial baseline.
    # ------------------------------------------------------------------
    try:
        baseline = height_sequence_1          # type: ignore
        baseline_score = evaluate_sequence(baseline)
        if baseline_score < best_score:
            best_score = baseline_score
            best_seq = np.array(baseline, dtype=np.float64)
        print(f"Baseline (height_sequence_1) score = {baseline_score:.6f}")
    except NameError:
        pass

    # ------------------------------------------------------------------
    # 2.  Parameter‑space exploration (random families).
    # ------------------------------------------------------------------
    # a few useful lengths (they fit easily into memory & FFT fast)
    lengths = [1200, 2000, 3000, 4000, 6000, 8000]

    # main search loop
    while time.time() < deadline:
        # pick a length (biased towards larger for better asymptotics)
        n = int(np.random.choice(lengths))
        # pick a parametric family
        family = np.random.choice(['power', 'exp', 'beta'])
        if family == 'power':
            alpha = np.random.uniform(0.05, 2.5)
            a = _init_power_law(n, alpha)
        elif family == 'exp':
            lam = np.random.uniform(0.0, 6.0)
            a = _init_exponential(n, lam)
        else:  # beta
            p = np.random.uniform(0.1, 3.0)
            q = np.random.uniform(0.1, 3.0)
            a = _init_beta_distribution(n, p, q)

        # quick evaluation
        score = _evaluate_metric(a)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"Random {family} n={n} → new best {best_score:.6f}")

        # Occasionally launch a short refinement on the current candidate
        if np.random.rand() < 0.07:   # ~7 % of the trials get refined
            # give each refinement a tiny time slice (~0.4 s)
            sub_deadline = min(deadline, time.time() + 0.4)
            a_ref, score_ref = _smooth_max_optimize(
                a, sub_deadline, max_iter=2000, p_start=4.0, p_max=48.0
            )
            if score_ref < best_score:
                best_score = score_ref
                best_seq = a_ref.copy()
                print(f"  Refined {family} n={n} → new best {best_score:.6f}")

        # Now try a convex mix with the classical arcsine (central‑binomial) shape
        if best_seq is not None and np.random.rand() < 0.03:
            # build arcsine‑type shape of same length
            arc = np.empty(n, dtype=np.float64)
            arc[0] = 1.0
            for i in range(1, n):
                arc[i] = arc[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
            arc = _project_onto_simplex(arc)
            # convex combination
            t = np.random.uniform(0.01, 0.15)
            mixed = _project_onto_simplex((1 - t) * best_seq[:n] + t * arc)
            mixed_score = _evaluate_metric(mixed)
            if mixed_score < best_score:
                best_score = mixed_score
                best_seq = mixed.copy()
                print(f"  Mixed with arcsine (t={t:.3f}) → new best {best_score:.6f}")

        # Periodically run a longer refinement on the current global best
        if best_seq is not None and np.random.rand() < 0.02:
            sub_deadline = min(deadline, time.time() + 0.8)
            a_ref, score_ref = _smooth_max_optimize(
                best_seq, sub_deadline, max_iter=5000, p_start=8.0, p_max=64.0
            )
            if score_ref < best_score:
                best_score = score_ref
                best_seq = a_ref.copy()
                print(f"Long refinement → new best {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3.  Return the best sequence found (as plain Python list).
    # ------------------------------------------------------------------
    if best_seq is None:
        # fail‑safe: return a tiny uniform distribution
        n = 1000
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    print(f"Search finished. Best score = {best_score:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
