    n = len(seq)

"""Step‑function optimizer.

This implementation mixes a few deterministic seeds (uniform, geometric,
cosine‑window, sine‑window, Gaussian) and a smooth “log‑sum‑exp’’ gradient
descent.  The gradient is computed from a smoothed version of the
max‑convolution, allowing the optimizer to move past the flat plateaus that
plain sub‑gradient descent gets stuck in.  The iterates are always projected
onto the probability simplex so the sum of the heights stays equal to one.
A tiny amount of random jitter is injected every few thousand iterations
to avoid local minima.  The routine respects the given time budget and
returns the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Projection onto the probability simplex {x >= 0, sum(x)=1}
def _project_onto_simplex(v):
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:               # should not happen, but be safe
        return np.full_like(v, 1.0 / len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full linear convolution using FFT (fast for many repeated calls)
def _full_convolution(a, b):
    """Linear convolution of real vectors a and b using FFT."""
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    n = len(a) + len(b) - 1
    m = 1 << (n - 1).bit_length()          # next power of two
    fa = np.fft.rfft(a, m)
    fb = np.fft.rfft(b, m)
    conv = np.fft.irfft(fa * fb, m)[:n]
    return conv


# ----------------------------------------------------------------------
# Scoring function – reproduces evaluate_sequence without the safety checks
def _score(seq):
    """Return the evaluation metric for a non‑negative probability vector."""
    s = seq.sum()
    if s < 0.01:
        return float("inf")
    n = len(seq)
    max_b = _full_convolution(seq, seq).max()
    return float(2.0 * n * max_b / (s * s))

# Families of candidate initialisations
def _uniform_seq(n):
    return np.full(n, 1.0 / n)


def _geometric_seq(n, factor):
    eps = factor / n
    r = max(0.0, 1.0 - eps)
    powers = np.power(r, np.arange(n, dtype=np.float64))
    seq = powers / powers.sum()
    return seq


def _cosine_seq(n):
    i = np.arange(n, dtype=np.float64)
    seq = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    return seq / seq.sum()


def _sine_seq(n):
    i = np.arange(n, dtype=np.float64)
    seq = np.sin(np.pi * (i + 1) / (n + 1))
    return seq / seq.sum()


def _gaussian_seq(n, sigma_factor=0.2):
    i = np.arange(n, dtype=np.float64)
    mu = (n - 1) / 2.0
    sigma = sigma_factor * n
    seq = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    return seq / seq.sum()


def _random_dirichlet_seq(n, alpha=0.3):
    seq = np.random.gamma(alpha, 1.0, size=n)
    return seq / seq.sum()


# ----------------------------------------------------------------------
# Smooth (log‑sum‑exp) gradient step
def _smooth_step(seq, beta, step):
    """
    One optimisation step using the smoothed max (log‑sum‑exp) of the
    self‑convolution.  Returns the new sequence and its score.
    """
    n = len(seq)

    # Full convolution a*a
    conv = _full_convolution(seq, seq)          # length 2n‑1
    m = conv.max()

    # Soft‑max weights (stable by subtracting the max)
    exp_vals = np.exp(beta * (conv - m))
    Z = exp_vals.sum()
    if Z == 0.0:                               # numerical safety
        w = np.full_like(conv, 1.0 / conv.size)
    else:
        w = exp_vals / Z

    # Gradient of the smoothed objective:
    #   grad_i = 2 * sum_k w_k * a_{k-i}   (indices out of range are ignored)
    rev = seq[::-1]                              # reversed a
    grad_full = _full_convolution(w, rev)        # length (2n‑1)+n‑1 = 3n‑2
    start = conv.size - 1                        # = 2n‑2
    grad = 2.0 * grad_full[start:start + n]

    # Gradient descent step + projection onto simplex
    new_seq = seq - step * grad
    new_seq = _project_onto_simplex(new_seq)
    return new_seq, _score(new_seq)
