def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Generate a non‑negative sequence that (approximately) minimizes
    2·n·max(conv(a,a))/(sum a)^2.
    The algorithm starts from the discrete arcsine distribution, then
    repeatedly solves a linear program that finds a feasible direction
    improving the L∞ bound, and finally performs a brief random hill‑climb.
    """
    import time
    import numpy as np
    # attempt to import scipy
    try:
        from scipy.linalg import toeplitz
        from scipy.optimize import linprog
    except Exception as e:
        toeplitz = None
        linprog = None
        print("[fallback] SciPy not available, using simple hill‑climbing only.")
    #-----------------
    np.random.seed(seed)
    # Determine length
    n = kwargs.get("length")
    if n is None:
        try:
            n = len(height_sequence_1)  # type: ignore[name-defined]
        except Exception:
            # reasonable default that balances speed & quality
            n = 2000
    #-----------------
    # 1. initial arcsine distribution
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k-1] * ((2*k-1)*(2*k)) / (4.0 * k * k)
    a /= a.sum()   # normalise to 1
    #-----------------
    def score(vec):
        # compute objective using same formula as evaluate_sequence (with sum=1)
        conv = np.convolve(vec, vec)
        return float(2 * n * conv.max())
    best_score = score(a)
    print(f"[init] n={n}, score={best_score:.6f}")
    start = time.time()
    deadline = start + budget_s - 0.5  # safety margin
    #----------------------------------
    # 2. LP‑based improvement phase
    # use at most 25 iterations or until time runs out
    if linprog is not None:
        t = 0.4        # initial mixing coefficient
        max_lp_iters = 30
        for it in range(max_lp_iters):
            now = time.time()
            if now > deadline:
                break
            # compute current max convolution
            conv_a = np.convolve(a, a)
            M = conv_a.max()
            # Build Toeplitz matrix for constraints (f * b)_i <= M
            # A shape (2n-1, n)
            c = np.concatenate([a, np.zeros(n-1, dtype=np.float64)])
            r = np.zeros(n, dtype=np.float64)
            r[0] = a[0]
            A = toeplitz(c, r)   # dense
            # Linear program: maximise sum(b)  <=> minimise -sum(b)
            c_obj = -np.ones(n, dtype=np.float64)
            b_ub = np.full(2*n-1, M, dtype=np.float64)
            bounds = [(0, None)] * n
            # solve LP
            res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')

                print(f"[lp] iteration {it}: failed ({res.message})")
                break

                print(f"[lp] iteration {it}: zero sum")
                break

            # Re‑normalise (numerical safety)
            a_candidate /= a_candidate.sum()
            cand_score = score(a_candidate)
            print(f"[lp] it={it:02d}, t={t:.3f}, M={M:.8e}, sum_b0={sum_b0:.6f}, cand_score={cand_score:.6f}")
            if cand_score < best_score - 1e-8:
                a = a_candidate
                best_score = cand_score
                # keep mixing coefficient, maybe increase a little to speed up convergence
                t = min(0.9, t * 1.1)
                print(f"[improve] new best={best_score:.6f}")
            else:
