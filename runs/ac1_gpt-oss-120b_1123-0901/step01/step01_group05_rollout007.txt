    from scipy.optimize import linprog
    from scipy.sparse import csr_matrix

    start = time.time()
    deadline = start + budget_s - 0.5  # margin

    # Determine n and initial sequence
    try:
        init_seq = np.array(height_sequence_1, dtype=float)
        n = len(init_seq)
    except Exception:
        n = 3000
        init_seq = np.array(generate_inverse_sqrt_sequence(n), dtype=float)

    # Normalize sum to sqrt(2*n)
    target_sum = math.sqrt(2 * n)
    current_seq = init_seq * (target_sum / init_seq.sum())
    # Ensure non-negative and within bounds 0-1000 (should be)
    current_seq = np.clip(current_seq, 0, 1000)

    best_seq = current_seq.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] n={n}, initial score={best_score:.6f}")

    iter_count = 0
    no_improve = 0

    # Precompute possible t values
    t_values = [0.1, 0.2, 0.3, 0.5, 0.7]

    while time.time() < deadline and no_improve < 30:
        iter_count += 1
        # Compute M for current sequence
        conv = np.convolve(current_seq, current_seq)
        M = conv.max()
        # Build constraint matrix A_ub (sparse)
        data = []
        row_ind = []
        col_ind = []
        # Pre-allocate list length? We'll just use loops; speed okay for moderate n.
        for j in range(n):
            rows_j = np.arange(j, j + n)  # shape (n,)
            row_ind.extend(rows_j.tolist())
            col_ind.extend([j] * n)
            data.extend(current_seq.tolist())
        A_ub = csr_matrix((data, (row_ind, col_ind)), shape=(2 * n - 1, n))

        # Setup LP
        c = -np.ones(n)  # maximize sum -> minimize -sum
        b_ub = np.full(2 * n - 1, M, dtype=float)

        # bounds
        bounds = [(0, None)] * n

        # Solve LP
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs', options={'presolve': True, 'time_limit': max(0, deadline - time.time())})
        if not res.success:
            print(f"[iter {iter_count}] LP failed: {res.message}")
            break

        b_opt = res.x
        sum_b = b_opt.sum()
        # scaling factor
        scale = target_sum / sum_b
        b_scaled = b_opt * scale

        # Try several convex combos
        improved = False
        for t in t_values:
            new_seq = (1 - t) * current_seq + t * b_scaled
            # Ensure within bounds (0-1000)
            # Actually evaluate_sequence will clip but inside we also clip maybe
            # Evaluate score
            try:
                score = evaluate_sequence(new_seq.tolist())
            except Exception:
                continue
            if score < best_score - 1e-8:
                best_score = score
                best_seq = new_seq.copy()
                current_seq = new_seq.copy()
                improved = True
                no_improve = 0
                print(f"[iter {iter_count}] improve t={t:.2f} score={best_score:.6f}")
                break  # break t loop, go to next iteration
        if not improved:
            # if none of the t values improved, maybe try t=0 (no change)
            no_improve += 1
            # Optionally keep current_seq unchanged.
        # also could break if no improvement over many iterations
        if time.time() > deadline:
            break

    # Return best sequence as list (converted to Python floats)
    print(f"[final] iterations={iter_count}, best_score={best_score:.6f}")
    # Clip to non-negative and <=1000 if needed
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    return best_seq.tolist()

"""
Propose a candidate step‑function sequence using an
iterative convex‑combination / linear‑programming refinement.
The algorithm:
1. Initialise a non‑negative sequence (the known inverse‑sqrt shape
   or the previously found `height_sequence_1`) and normalise it so that
   the sum equals √(2 n).  With this normalisation the evaluation function
   reduces to the maximum value of the self‑convolution.
2. Repeatedly
   a) compute M = max(f * f)
   b) solve the LP  max Σ b_i  s.t.  b_i ≥ 0  and  f * b ≤ M .
      The constraints are linear in b because f is fixed.
   c) scale the LP solution to the required normalisation,
      and replace f by a convex combination (1‑t) f + t b
      (several values of t are tried, the best improvement is kept).
   The convex combination keeps the total sum unchanged while
   strictly decreasing the maximum convolution.
3. When the LP no longer yields improvement a cheap random local search
   (multiplicative perturbation of a single entry) is performed for the
   remaining time.
The function returns the best sequence found within the allotted budget.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

def generate_inverse_sqrt_sequence(n: int) -> np.ndarray:
    """Return a NumPy array a[0..n‑1] with a_i = binom(2i,i)/4^i."""
    seq = np.empty(n, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, n):
        seq[i] = seq[i - 1] * (2 * i - 1) / (2 * i)
    return seq


def _build_toeplitz_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the matrix A such that (A @ b) == f * b (full convolution).
    A has shape (2*n-1, n) and is extremely sparse,
    but we store it as a dense float array because the LP solver
    (highs) accepts dense matrices and the size (≈ 200 MiB for n≈3400)
    fits comfortably into memory.
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    # each column j contains f shifted by j places
    for j in range(n):
        A[j:j + n, j] = f
    return A


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Entry point used by the runner. Returns a list of non‑negative floats.
    The routine spends most of the allotted time running the LP‑based
    refinement described in the module docstring and finishes with a
    lightweight stochastic fine‑tuning.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # ----------------------------------------------------------------------
    # 1. Choose the length n and an initial shape.
    # ----------------------------------------------------------------------
    try:
        init_seq = np.array(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        n = init_seq.size
    except Exception:
        n = 3400  # a length that gave a good starting bound in earlier runs
        init_seq = generate_inverse_sqrt_sequence(n)

    # Normalise the sum to sqrt(2*n) – with this normalisation the score equals max(conv)
    target_sum = math.sqrt(2.0 * n)
    init_seq = init_seq * (target_sum / init_seq.sum())
    init_seq = np.clip(init_seq, 0.0, 1000.0)

    # ----------------------------------------------------------------------
    # 2. Initial evaluation
    # ----------------------------------------------------------------------
    best_seq = init_seq.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] n={n}, initial score={best_score:.6f}")

    current_seq = best_seq.copy()
    no_improve_lp = 0
    max_no_improve_lp = 30                # stop LP phase after this many non‑improving steps
    iteration = 0

    # ----------------------------------------------------------------------
    # 3. Main LP‑based refinement loop
    # ----------------------------------------------------------------------
    while time.time() < deadline and no_improve_lp < max_no_improve_lp:
        iteration += 1
        # Compute M = max convolution of the current sequence
        conv = np.convolve(current_seq, current_seq)
        M = conv.max()

        # Build the Toeplitz constraint matrix (f * b <= M)
        A_ub = _build_toeplitz_matrix(current_seq)     # shape (2*n-1, n)

        # Linear programme: maximise sum(b)  <=> minimise -sum(b)
        c = -np.ones(n, dtype=np.float64)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)
        bounds = [(0.0, None)] * n

        # Solve the LP
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
        if not res.success:
            print(f"[iter {iteration}] LP failed ({res.message}), exiting LP phase.")
            break

        b_opt = res.x
        sum_b_opt = b_opt.sum()
        if sum_b_opt <= 0:
            print(f"[iter {iteration}] LP returned zero sum, exiting LP phase.")
            break

        # Rescale to keep the total sum equal to target_sum
        scale = target_sum / sum_b_opt
        b_scaled = b_opt * scale

        # Try several convex‑combination coefficients and keep the best move
        improved = False
        # candidate mixing parameters (including full replacement)
        t_candidates = [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]
        for t in t_candidates:
            cand = (1.0 - t) * current_seq + t * b_scaled
            # small numerical noise can produce tiny negatives – clip
            cand = np.clip(cand, 0.0, 1000.0)

            # Since the sum is preserved we can use the max‑convolution as a proxy
            conv_cand = np.convolve(cand, cand)
            score_cand = conv_cand.max()

            if score_cand < best_score - 1e-9:
                best_score = score_cand
                best_seq = cand.copy()
                current_seq = cand.copy()
                improved = True
                no_improve_lp = 0
                print(f"[iter {iteration}] improve t={t:.2f} score={best_score:.6f}")
                break   # accept the first improvement (largest t tried later may be even better)

        if not improved:
            no_improve_lp += 1
            # No improvement – keep the current sequence unchanged
            # and continue the loop; eventually the LP will stop improving.

    # ----------------------------------------------------------------------
    # 4. Light‑weight random local search for the remaining time
    # ----------------------------------------------------------------------
    if time.time() < deadline:
        # Re‑evaluate the best sequence to obtain a clean score value
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"[lp‑phase] finished with score={best_score:.6f}")

        # Random multiplicative perturbations of a single entry
        # (the same idea that gave the inverse‑sqrt baseline)
        while time.time() < deadline:
            i = rng.integers(n)
            old = best_seq[i]
            # Log‑normal perturbation keeps positivity
            factor = math.exp(rng.normal(0.0, 0.02))
            new_val = max(0.0, old * factor)
            cand = best_seq.copy()
            cand[i] = new_val
            cand = np.clip(cand, 0.0, 1000.0)
            try:
                cand_score = evaluate_sequence(cand.tolist())
            except Exception:
                continue
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand
                print(f"[local] improve idx={i} score={best_score:.6f}")

    # ----------------------------------------------------------------------
    # 5. Return the final sequence
    # ----------------------------------------------------------------------
    final_seq = np.clip(best_seq, 0.0, 1000.0).tolist()
    print(f"[final] iterations={iteration}, final score={best_score:.6f}")
    return final_seq

"""
Propose a candidate step‑function sequence using an LP‑based refinement.

The algorithm works in two phases:

1. **LP refinement**
   - Start from a known non‑negative shape (the previous best sequence if
     available, otherwise the classical inverse‑square‑root construction)
     and normalise it so that the sum equals √(2 n).  With this normalisation
     the evaluation function reduces to the maximum entry of the self‑convolution.
   - Repeatedly:
       * Compute M = max(f * f).
       * Solve the linear programme  
            maximise   Σ b_i  
            subject to b_i ≥ 0 and (f * b)_k ≤ M for every k.  
         Because f is fixed, the constraints are linear in b.
       * Scale the LP solution to keep the total sum equal to √(2 n).
       * Replace f by a convex combination (1‑t) f + t b for several
         values of t (including t = 1).  The combination keeps the sum
         unchanged while strictly decreasing the maximal convolution.
   - Stop when no improvement is observed for a number of successive
     iterations or when the time budget is exhausted.

2. **Local random search**
   - With the remaining time perform cheap multiplicative perturbations
     of single entries, accepting only moves that improve the evaluation.
