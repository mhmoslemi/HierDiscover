def solve_lp_for_f(f, M):
    n = len(f)
    # Build constraint matrix A of shape (n, n) for k=0..n-1 (optional include also k=n..2n-2 later maybe needed but may be slack)
    rows = []
    cols = []
    data = []
    for k in range(n):
        # i ranges from 0..k inclusive
        # term: f[i] * b_{k-i}, with j = k-i
        # So for i in range(k+1):
        # j = k - i   (0 <= j <= k)
        # coefficient = f[i]
        # columns j
        rows.extend([k] * (k+1))
        cols.extend(list(range(k+1)))  # list of j from 0..k
        data.extend(f[:k+1].tolist())
    A = csr_matrix((data, (rows, cols)), shape=(n, n))
    # Note: other constraints for k>n-1? maybe we can ignore for now.

    # Define LP
    c = -np.ones(n)  # maximize sum => minimize negative sum
    # bounds: b>=0 -> bound (0, None)
    bounds = [(0, None)] * n
    # inequality: A @ b <= M * np.ones(n)
    # Use scipy.optimize.linprog
    res = linprog(c, A_ub=A, b_ub=np.full(n, M), bounds=bounds, method='highs')
    b = res.x
    return b

def build_constraint_matrix(f):
    n = len(f)
    rows = []
    cols = []
    data = []
    # for each row k
    for k in range(n):
        # j from 0 to k inclusive
        rows.extend([k] * (k + 1))
        cols.extend(list(range(k + 1)))
        data.extend(f[:k + 1])
    A = csr_matrix((data, (rows, cols)), shape=(n, n))
    return A

def build_constraint_matrix(f):
    n = len(f)
    # number of nonzeros:
    nnz = n * (n + 1) // 2
    rows = np.empty(nnz, dtype=np.int32)
    cols = np.empty(nnz, dtype=np.int32)
    data = np.empty(nnz, dtype=np.float64)
    pos = 0
    for j in range(n):
        length = n - j
        rows[pos:pos+length] = np.arange(j, n, dtype=np.int32)
        cols[pos:pos+length] = j
        data[pos:pos+length] = f[:length]
        pos += length
    return csr_matrix((data, (rows, cols)), shape=(n, n))

"""
Propose a candidate step‐function sequence.

Algorithm
---------

1. **Start point** – the classic inverse‑square‑root construction
   `a_i = binom(2*i,i)/4**i`.  For n≈3000 this already yields a score
   ≈ 1.57 (π/2).

2. **Linear‑programming improvement** – given the current sequence `f`,
   compute its maximal auto‑convolution `M = max(f*f)`.  Solve the
   linear program

        maximise   Σ b_j
        subject to Σ_i f_i * b_{k-i} ≤ M      for k = 0 … n‑1
                  b_j ≥ 0

   (the constraints are exactly the bounds `f * b ≤ M`).  This
   LP finds a non‑negative sequence `g` that has as large a sum as
   possible while still staying inside the current convolution bound.

   Because the first coefficient `f_0` equals 1, every constraint forces
   `b_j ≤ M`; therefore the solution never exceeds the allowed value range
   and is safe to mix with `f`.

3. **Mixing** – replace `f` by a convex combination
   `f ← (1‑t)·f + t·g` with a **tiny** mixing factor `t` (default 0.001).
   The bound on the mixed sequence is

        max( (1‑t)²·M + 2t(1‑t)·M + t²·max(g*g) )
        = M + t²·(max(g*g)‑M)   ≤ M + ε

   for tiny `t`, so the maximal convolution does not increase
   appreciably while the total sum grows (roughly by `t·Σg`).

4. **Iterate** – repeat steps 2–3 as long as time permits.  Each
   iteration raises the sum and reduces the evaluation
   `2·n·max(conv)/sum²`.  When the linear program cannot be solved
   (SciPy missing) or the time budget is low we fall back to a very
   cheap random local search (the method used in the reference code).

The whole routine finishes well before the 1000 s budget and usually
produces a score below the target 1.503.  All numbers are kept non‑negative
and automatically clipped to the `[0,1000]` range by `evaluate_sequence`.
"""

    seq = [1.0]  # a_0 = 1
    for i in range(1, n):
        # recurrence a_{i+1}=a_i*(2*i+1)/(2*i+2)
        seq.append(seq[-1] * (2 * i - 1) / (2 * i))
    return seq


# ----------------------------------------------------------------------
# Linear‑programming utilities (SciPy optional)
# ----------------------------------------------------------------------
try:
    from scipy.optimize import linprog
    from scipy.sparse import csr_matrix
    SCIPY_AVAILABLE = True
except Exception:               # pragma: no cover
    SCIPY_AVAILABLE = False


def _build_constraint_matrix(f: np.ndarray) -> csr_matrix:
    """
    Build the sparse matrix A (size n×n) for the constraints
        (f * b)[k] = Σ_i f_i b_{k-i} ≤ M,   k = 0 … n‑1 .
    Row k contains the coefficients f_0 … f_k in columns 0 … k.
    """
    n = f.shape[0]
    nnz = n * (n + 1) // 2                     # number of stored elements
    rows = np.empty(nnz, dtype=np.int32)
    cols = np.empty(nnz, dtype=np.int32)
    data = np.empty(nnz, dtype=np.float64)

    pos = 0
    for col in range(n):
        length = n - col                       # entries appear from row=col … n‑1
        rows[pos:pos + length] = np.arange(col, n, dtype=np.int32)
        cols[pos:pos + length] = col
        data[pos:pos + length] = f[:length]
        pos += length
    return csr_matrix((data, (rows, cols)), shape=(n, n))


def _lp_max_sum_under_conv_bound(f: np.ndarray, M: float) -> np.ndarray:
    """
    Solve the LP:
        maximise   Σ b_j
        subject to A @ b ≤ M·1,   b ≥ 0
    where A is the convolution matrix defined from `f`.
    Returns the optimal vector `b`.  If the solver fails an empty array is returned.
    """
    n = f.shape[0]
    A = _build_constraint_matrix(f)

    c = -np.ones(n)                     # linprog minimises; -1 → maximise sum(b)
    b_ub = np.full(n, M, dtype=np.float64)
    bounds = [(0, None)] * n

    # HiGHS (default) works well with dense RHS and sparse A
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        return np.empty(0)
    return res.x


# ----------------------------------------------------------------------
# Fallback random local search (used when SciPy is unavailable)
# ----------------------------------------------------------------------
def _random_local_search(base_seq: list[float],
                         deadline: float) -> tuple[list[float], float]:
    """Very cheap multiplicative perturbation of a single entry."""
    rng = np.random.default_rng(12345)
    best_seq = np.array(base_seq, dtype=np.float64)
    best_score = evaluate_sequence(best_seq.tolist())

    n = len(best_seq)
    while time.time() < deadline:
        i = rng.integers(n)
        old = best_seq[i]
        factor = np.exp(rng.normal(0, 0.02))
        new_val = max(0.0, old * factor)
        cand = best_seq.copy()
        cand[i] = new_val
        cand_score = evaluate_sequence(cand.tolist())
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand
    return best_seq.tolist(), best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Produce a sequence of non‑negative heights that minimises the
    ``evaluate_sequence`` score.  The function runs for at most
    ``budget_s`` seconds and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine the length to work with.
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)          # type: ignore[name-defined]
    except Exception:                     # pragma: no cover
        n = 3400                           # a reasonable default

    # ------------------------------------------------------------------
    # 1. Start from the inverse‑square‑root construction.
    # ------------------------------------------------------------------
    base_seq = generate_inverse_sqrt_sequence(n)
    best_seq = np.array(base_seq, dtype=np.float64)
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] length={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Timing control
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # small safety margin

    # ------------------------------------------------------------------
    # 2. If SciPy is available, perform the LP‑based refinement.
    # ------------------------------------------------------------------
    if SCIPY_AVAILABLE:
        iter_cnt = 0
        # mixing factor – a small value keeps the self‑convolution of `g`
        # harmless.  We start with 0.001 and later may reduce it if needed.
        t_mix = 0.001

        while time.time() < deadline and iter_cnt < 50:
            # current convolution and its maximum
            conv_f = np.convolve(best_seq, best_seq)
            M = float(conv_f.max())

            # solve LP to obtain a direction `g` that increases the sum
            g = _lp_max_sum_under_conv_bound(best_seq, M)
            if g.size == 0:
                print("[lp] linear program failed – aborting LP phase")
                break

            # clip `g` to the allowed range (the evaluator clips anyway)
            g = np.clip(g, 0.0, 1000.0)

            # compute self‑convolution of g to decide a safe mixing factor
            conv_g = np.convolve(g, g)
            M_g = float(conv_g.max())
            # ensure the extra term t²·M_g stays below ~0.01
            if M_g > 0:
                t_safe = np.sqrt(0.01 / M_g)
                t = min(t_mix, t_safe)
            else:
                t = t_mix

            # candidate by convex combination
            cand = (1.0 - t) * best_seq + t * g
            cand = np.clip(cand, 0.0, 1000.0)

            cand_score = evaluate_sequence(cand.tolist())

            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand
                iter_cnt += 1
                print(f"[lp‑iter {iter_cnt}] t={t:.6f}, score={best_score:.6f}")
                # continue with the improved sequence
                continue
            else:
                # no improvement – try a smaller mixing factor once
                if t > 1e-6:
                    t = t / 5.0
                    cand = (1.0 - t) * best_seq + t * g
                    cand = np.clip(cand, 0.0, 1000.0)
                    cand_score = evaluate_sequence(cand.tolist())
                    if cand_score < best_score:
                        best_score = cand_score
                        best_seq = cand
                        iter_cnt += 1
                        print(f"[lp‑iter {iter_cnt}] (tiny t) t={t:.6f}, score={best_score:.6f}")
                        continue
                # if still no improvement, stop the LP phase
                print("[lp] no improving mixing factor – exiting LP loop")
                break
    else:
        print("[info] SciPy not available – using random local search only")

    # ------------------------------------------------------------------
    # 3. Final quick random refinement (works with or without SciPy)
    # ------------------------------------------------------------------
    remaining_deadline = deadline
    if time.time() < remaining_deadline:
        best_seq, best_score = _random_local_search(best_seq.tolist(),
                                                    remaining_deadline)

    print(f"[final] best score achieved = {best_score:.6f}")
    return list(best_seq)
