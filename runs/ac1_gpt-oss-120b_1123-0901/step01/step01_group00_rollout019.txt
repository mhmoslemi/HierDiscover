def run_optim(n, init, time_limit):
    """
    Optimize a sequence of length n starting from init until time_limit timestamp.
    Returns best sequence (numpy array) and its evaluation (max conv).
    """
    import time, numpy as np

    target_sum = np.sqrt(2.0 * n)
    a = np.array(init, dtype=float)
    a = _project_onto_simplex(a, target_sum)
    best_a = a.copy()
    conv = np.convolve(a, a)
    best_val = float(conv.max())
    # parameters
    base_step = 0.05  # step scale
    beta = 0.5
    beta_max = 12.0
    beta_factor = 1.001
    momentum = 0.9
    grad_prev = None
    no_improve_counter = 0

    while time.time() < time_limit:
        conv = np.convolve(a, a)
        max_val = conv.max()
        if max_val < best_val - 1e-9:
            best_val = max_val
            best_a = a.copy()
            no_improve_counter = 0
        else:
            no_improve_counter += 1

        # compute softmax weights
        # subtract max to avoid overflow (already done by using conv - max_val)
        exponents = beta * (conv - max_val)
        # Clip to avoid overflow in exp
        exponents = np.clip(exponents, -700, 0)
        w = np.exp(exponents)
        w_sum = w.sum()
        if w_sum == 0:
            # fallback to one-hot at max index
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
            w_sum = 1.0
        w = w / w_sum

        # gradient via convolution
        grad = 2.0 * np.convolve(w, a[::-1])[n-1:2*n-1]

        # momentum
        if grad_prev is not None:
            grad = momentum * grad_prev + (1 - momentum) * grad
        grad_prev = grad

        # step size
        max_grad = np.max(np.abs(grad))
        if max_grad == 0:
            max_grad = 1.0
        eta = base_step * max_val / max_grad

        a_new = a - eta * grad
        # ensure non-negative before projection? not needed because projection handles negative.
        a_new = _project_onto_simplex(a_new, target_sum)

        # occasional random perturbation
        if np.random.rand() < 0.01:
            noise = np.random.randn(n) * (0.01 * target_sum / n)
            a_new = _project_onto_simplex(np.maximum(a_new + noise, 0.0), target_sum)
            # reset momentum to avoid stale.
            grad_prev = None
            base_step *= 0.9

        a = a_new

        # increase beta slowly
        if beta < beta_max:
            beta = min(beta_max, beta * beta_factor)

        # adapt step size if stuck
        if no_improve_counter > 200:
            base_step *= 0.8
            no_improve_counter = 0

    return best_a, best_val
```

def lp_step(f, M, max_iter=200):
    n = len(f)
    rows = 2*n-1
    # coefficient matrix A_ub shape (rows, n)
    A_ub = np.zeros((rows, n), dtype=float)
    for k in range(rows):
        low = max(0, k - (n - 1))
        high = min(k, n - 1)
        # i from low..high
        for i in range(low, high+1):
            j = k - i
            A_ub[k, j] = f[i]
    b_ub = np.full(rows, M, dtype=float)

    bounds = [(0, None)] * n

    c = np.full(n, -1.0, dtype=float)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs', options={'presolve': True, 'time_limit': 10})
    if not res.success:

        return f

    sum_g0 = np.sum(g0)
    if sum_g0 > 0:

    t = 0.1  # small step
    new_f = (1-t) * f + t * g
    new_f = _project_onto_simplex(new_f, target_sum)

"""
Propose a non‑negative step‑function sequence with a low evaluation score.

Algorithm
---------
1. **Soft‑max gradient descent**  
   - The objective is 2·n·max(conv(a,a))/sum(a)².  By projecting onto the
     simplex {x≥0, Σx = √(2n)} the denominator becomes constant, so we only
     need to minimise the maximal convolution entry.
   - At each iteration we compute the full convolution, build a soft‑max
     weight vector `w` on the convolution entries and take a gradient step of
     the weighted sum.  This spreads the descent direction over many
     anti‑diagonals rather than only the current maximum, helping to flatten
     the autocorrelation.
   - The step is projected back onto the simplex; a small random shake‑up
     and a simple momentum scheme help to escape plateaux.

2. **Random restarts**  
   - Many independent runs are performed with different lengths `n` (≈500–2500)
     and initial points (random or a trimmed version of the previously known
     best sequence).  Each run is limited to a short time slice.

3. **Refinement**  
   - After the exploration phase the best sequence found so far is refined
     for the remaining time budget using the same gradient scheme with a
     longer deadline, a slower step‑size decay and a higher soft‑max temperature.

4. **Optional LP polishing** (uses `scipy.optimize.linprog`)  
   - A few Frank‑Wolfe style linear‑programming steps are applied to the
     refined candidate.  For a fixed `f`, we maximise Σgᵢ subject to
     `f*g ≤ M` (where `M = max(conv(f,f))`).  The resulting direction `g` is
     combined with `f` (convex combination) and projected onto the simplex.
   - This often reduces the peak convolution further.

All numbers are kept non‑negative; the final sequence is returned as a plain
Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the simplex {x≥0, Σx = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex of radius s."""
    n = v.shape[0]
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Core optimisation routine for a fixed length n
# ----------------------------------------------------------------------
def _run_optim(
    n: int,
    init: np.ndarray,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Perform soft‑max gradient descent on a sequence of length ``n`` until ``deadline``.
    Returns the best sequence discovered during the run and its maximal convolution value.
    """
    target_sum = np.sqrt(2.0 * n)

    # initialise and enforce feasibility
    a = _project_onto_simplex(init.astype(np.float64), target_sum)

    # record best encountered during this run
    conv = np.convolve(a, a)
    best_val = float(conv.max())
    best_seq = a.copy()

    # algorithmic parameters
    base_step = 0.05          # raw step‑size scale
    beta = 0.5                # soft‑max temperature
    beta_max = 12.0
    beta_factor = 1.001       # slow temperature increase
    momentum = 0.9
    grad_prev = None
    no_improve = 0

    while time.time() < deadline:
        conv = np.convolve(a, a)
        max_val = conv.max()

        # keep the best ever seen in this run
        if max_val < best_val - 1e-12:
            best_val = max_val
            best_seq = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # ---------- soft‑max weights ----------
        # w_i ∝ exp(beta * (conv_i - max_val)), max weight = 1
        exponents = beta * (conv - max_val)
        exponents = np.clip(exponents, -700, 0)  # avoid overflow/underflow
        w = np.exp(exponents)
        w_sum = w.sum()
        if w_sum == 0.0:                     # numerical safety
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
            w_sum = 1.0
        w /= w_sum

        # ---------- gradient of weighted sum ----------
        # grad = 2 * (w * a) convolution (see derivation in analysis)
        grad = 2.0 * np.convolve(w, a[::-1])[n - 1 : 2 * n - 1]

        # momentum blending
        if grad_prev is not None:
            grad = momentum * grad_prev + (1.0 - momentum) * grad
        grad_prev = grad

        # step size scaled to the magnitude of the gradient
        max_grad = np.max(np.abs(grad))
        if max_grad == 0.0:
            max_grad = 1.0
        eta = base_step * max_val / max_grad

        a_new = a - eta * grad
        a_new = _project_onto_simplex(a_new, target_sum)

        # occasional random shake‑up
        if np.random.rand() < 0.01:
            noise = np.random.randn(n) * (0.01 * target_sum / n)
            a_new = _project_onto_simplex(np.maximum(a_new + noise, 0.0), target_sum)
            base_step *= 0.9   # slightly damp the step after a perturbation
            grad_prev = None   # reset momentum

        a = a_new

        # temperature schedule
        if beta < beta_max:
            beta = min(beta_max, beta * beta_factor)

        # simple adaptive step‑size reduction if stuck
        if no_improve > 200:
            base_step *= 0.8
            no_improve = 0

    return best_seq, best_val


# ----------------------------------------------------------------------
# optional LP polishing step (Frank‑Wolfe style)
# ----------------------------------------------------------------------
def _lp_polish(seq: np.ndarray, time_limit: float, max_iters: int = 10) -> np.ndarray:
    """
    Perform a few linear‑programming polishing iterations.
    Returns a (hopefully) improved sequence.
    """
    try:
        from scipy.optimize import linprog
    except Exception:   # SciPy not available – silently skip polishing
        return seq

    n = seq.size
    target_sum = np.sqrt(2.0 * n)

    # build the inequality matrix A_ub (shape (2n‑1, n))
    rows = 2 * n - 1
    A_ub = np.zeros((rows, n), dtype=np.float64)
    for k in range(rows):
        low = max(0, k - (n - 1))
        high = min(k, n - 1)
        # for each i in [low, high] we have contribution f_i * g_{k-i}
        # column index j = k - i
        for i in range(low, high + 1):
            j = k - i
            A_ub[k, j] = seq[i]

    bounds = [(0, None)] * n
    c = -np.ones(n, dtype=np.float64)          # maximise Σg  ⇔ minimise -Σg

    # run a few polishing iterations while time permits
    it = 0
    while it < max_iters and time.time() < time_limit:
        M = np.max(np.convolve(seq, seq))

        # solve the LP
        res = linprog(
            c,
            A_ub=A_ub,
            b_ub=np.full(rows, M, dtype=np.float64),
            bounds=bounds,
            method="highs",
            options={"presolve": True},
        )
        if not res.success:
            break

        g0 = res.x
        # scale g0 to the required simplex radius
        sum_g0 = np.sum(g0)
        if sum_g0 > 0:
            g = g0 * (target_sum / sum_g0)
        else:
            g = g0

        # convex combination (small step towards the LP direction)
        t = 0.2
        seq = _project_onto_simplex((1.0 - t) * seq + t * g, target_sum)

        it += 1

    return seq

        seq_local, val_local = _run_optim(n, init, run_deadline)

        if val_local < best_val:
            best_val = val_local
            best_seq = seq_local.copy()
            best_n = n
            print(f"[{time.time():.2f}] New global best = {best_val:.6f} (n={best_n})")

        # early stop if we already beat the target comfortably
        if best_val < 1.51:
            break

    # ------------------------------------------------------------------
    # Phase 2 – longer refinement on the currently best length
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        n = best_n
        init = best_seq.copy()
        run_deadline = deadline
        seq_local, val_local = _run_optim(n, init, run_deadline)

        if val_local < best_val:
            best_val = val_local
            best_seq = seq_local.copy()
            best_n = n
            print(f"[{time.time():.2f}] Refined best = {best_val:.6f} (n={best_n})")

    # ------------------------------------------------------------------
    # Phase 3 – optional LP polishing (uses remaining time)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        polish_deadline = deadline
        seq_polished = _lp_polish(best_seq, polish_deadline, max_iters=12)
        # evaluate the polished candidate (real objective)
        try:
            score = evaluate_sequence(seq_polished.tolist())
        except Exception:
            score = float("inf")
        if score < best_val:
            best_val = score
            best_seq = seq_polished
            print(f"[{time.time():.2f}] LP‑polish improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – very small uniform vector if nothing was found
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        best_seq = np.full(n, np.sqrt(2.0 * n) / n, dtype=float)

    return best_seq.tolist()
