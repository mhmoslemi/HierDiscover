def _solve_lp(f, M, n):
    # f: np.ndarray shape (n,)
    # M: scalar
    # returns b (np.ndarray shape (n,))
    # Build A_ub
    rows = np.empty(n*n, dtype=int)
    cols = np.empty(n*n, dtype=int)
    data = np.empty(n*n, dtype=float)
    idx = 0
    for i in range(n):
        # row indices for j=0..n-1: i + j
        rows[idx:idx+n] = i + np.arange(n)
        cols[idx:idx+n] = np.arange(n)
        data[idx:idx+n] = f[i]   # coefficient f[i]
        idx += n
    from scipy.sparse import csr_matrix
    A_ub = csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    # bounds
    bounds = [(0, None) for _ in range(n)]
    # objective: maximize sum(b) -> minimize -sum(b)
    c = -np.ones(n)
    # Solve
    # Using method high performance: 'highs' default
    from scipy.optimize import linprog
    res = linprog(c, A_ub=A_ub, b_ub=np.full(2*n-1, M), bounds=bounds, method='highs')
    if res.success:
        return res.x
    else:
        return None

"""
Search for a non‑negative step function that minimizes

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm is a hybrid of the linear‑programming “LP‑boost” described in
the prompt and a small projected‑gradient refinement.

1. **LP‑boost**  
   For a given length `n` we keep a vector `f` with the standard normalisation
   Σ f_i = √(2 n).  Let `M = max(conv(f,f))`.  We solve the linear program

        maximise   Σ b_i
        subject to conv(f, b) ≤ M   (for every convolution index)
                  b_i ≥ 0

   The solution `b` is as large as possible while keeping the convolution with
   the current `f` below the current worst value `M`.  After scaling `b` to the
   required sum we obtain a new candidate `g`.  Mixing `f` and `g` with a
   suitable step size strictly reduces the maximal autocorrelation; the
   process is repeated until progress stalls.

2. **Projected gradient fine‑tuning**  
   Once the LP stage stops improving, a few iterations of the simple
   projected‑gradient descent (the method that produced 1.524…) are applied to
   squeeze out the remaining slack.

The whole search runs for at most the supplied time budget (default 1000 s)
and keeps the best sequence seen.  The function `propose_candidate` returns
that sequence as a plain Python list.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the simplex {x_i ≥ 0, Σ x_i = s}.
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project `v` onto the simplex of radius `s`."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper: build the (2n‑1) × n convolution matrix A such that
#         conv(f, b) = A @ b
# ----------------------------------------------------------------------
def _build_conv_matrix(f: np.ndarray) -> csr_matrix:
    """
    Return a sparse matrix A where (A @ b)[k] = Σ_i f[i] * b[k-i]
    (zero when the index is out of range).  Shape is (2n‑1, n).
    """
    n = f.shape[0]
    size = n * n
    rows = np.empty(size, dtype=np.int32)
    cols = np.empty(size, dtype=np.int32)
    data = np.empty(size, dtype=np.float64)

    idx = 0
    for i, fi in enumerate(f):
        rows[idx:idx + n] = i + np.arange(n, dtype=np.int32)
        cols[idx:idx + n] = np.arange(n, dtype=np.int32)
        data[idx:idx + n] = fi
        idx += n

    return csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


# ----------------------------------------------------------------------
# Helper: solve the LP that maximises Σ b_i under conv(f,b) ≤ M.
# ----------------------------------------------------------------------
def _solve_lp_max_sum(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve the LP:
        maximise   Σ b_i
        subject to conv(f, b) ≤ M  (all 2n‑1 entries)
                  b_i ≥ 0
    Returns the optimal `b` as a dense array, or None if the solver fails.
    """
    n = f.shape[0]
    A = _build_conv_matrix(f)                     # shape (2n‑1, n)
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)

    # Objective: minimise -Σ b_i  (i.e. maximise Σ b_i)
    c = -np.ones(n, dtype=np.float64)

    # Bounds: b_i ≥ 0
    bounds = [(0, None) for _ in range(n)]

    # hiGHS (the default method) handles sparse matrices efficiently.
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True})

    if res.success:
        return res.x
    return None


# ----------------------------------------------------------------------
# Helper: compute the maximal autocorrelation value of a sequence.
# ----------------------------------------------------------------------
def _max_autocorr(seq: np.ndarray) -> float:
    """Return max(conv(seq, seq))."""
    return np.max(np.convolve(seq, seq))


# ----------------------------------------------------------------------
# LP‑boost optimisation for a single length `n`.
# ----------------------------------------------------------------------
def _lp_optimize_length(n: int,
                        init_seq: np.ndarray,
                        deadline: float,
                        best_seq: np.ndarray | None,
                        best_score: float) -> tuple[np.ndarray, float]:
    """
    Run the LP‑boost scheme on a vector of length `n` starting from `init_seq`.
    The function updates and returns the globally best sequence and its score.
    """
    target_sum = np.sqrt(2.0 * n)

    # Normalise the start point onto the simplex Σ a_i = √(2n)
    f = init_seq.astype(np.float64)
    if f.sum() == 0:
        f = np.full(n, target_sum / n, dtype=np.float64)
    else:
        f = f * (target_sum / f.sum())

    cur_M = _max_autocorr(f)
    cur_score = 2.0 * n * cur_M / (target_sum ** 2)
    if cur_score < best_score:
        best_score, best_seq = cur_score, f.copy()
        print(f"[{time.time():.2f}] New best = {best_score:.6f} (n={n})")

    # Parameters for the inner loop
    max_iter = 30                     # safety guard – cheap enough
    no_improve = 0
    t_candidates = [0.1, 0.3, 0.5, 0.7, 0.9, 1.0]

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        # Solve the linear program
        b = _solve_lp_max_sum(f, cur_M)
        if b is None:
            # LP failed – abort this length
            break

        # Scale the LP solution back to the required sum
        sum_b = b.sum()
        if sum_b <= 0:
            break
        scale = target_sum / sum_b
        g = b * scale

        # Try several mixing factors and keep the best reduction in M
        best_local_M = cur_M
        best_f_candidate = None
        best_t = 0.0

        for t in t_candidates:
            f_candidate = (1.0 - t) * f + t * g
            M_candidate = _max_autocorr(f_candidate)
            if M_candidate < best_local_M - 1e-12:   # strict improvement
                best_local_M = M_candidate
                best_f_candidate = f_candidate
                best_t = t

        if best_t > 0.0:
            # Accept the improvement
            f = best_f_candidate
            cur_M = best_local_M
            cur_score = 2.0 * n * cur_M / (target_sum ** 2)
            no_improve = 0

            if cur_score < best_score:
                best_score, best_seq = cur_score, f.copy()
                print(f"[{time.time():.2f}] New best = {best_score:.6f} (n={n})")
        else:
            # No mixing factor helped – try a small random perturbation
            no_improve += 1
            if no_improve >= 3:
                # Restart from a fresh random point to escape a plateau
                f = np.random.rand(n)
                f = f * (target_sum / f.sum())
                cur_M = _max_autocorr(f)
                no_improve = 0
            else:
                # Gentle shake‑up
                noise = np.random.randn(n) * (target_sum / (20 * n))
                f = np.maximum(f + noise, 0.0)
                f = _project_onto_simplex(f, target_sum)
                cur_M = _max_autocorr(f)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Projected‑gradient refinement (the original simple method).
# ----------------------------------------------------------------------
def _gradient_refinement(seq: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float) -> tuple[np.ndarray, float]:
    """
    Perform a few iterations of projected gradient descent on the current
    best sequence to polish the result.  The optimisation target is
    `max(conv(seq, seq))` (equivalently the evaluation score because the
    sum is fixed).
    """
    n = seq.shape[0]
    target_sum = np.sqrt(2.0 * n)
    a = seq.copy()
    base_step = 0.05

    while time.time() < deadline:
        conv = np.convolve(a, a)
        k_max = int(np.argmax(conv))
        max_val = conv[k_max]

        # compute gradient of the *selected* maximal convolution entry
        low = max(0, k_max - (n - 1))
        high = min(n - 1, k_max)
        idx = np.arange(low, high + 1, dtype=int)
        partner = k_max - idx
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[partner]

        # step size adaptation
        max_grad = grad.max() if grad.max() > 0 else 1.0
        eta = base_step * max_val / max_grad

        a_new = a - eta * grad
        a_new = _project_onto_simplex(a_new, target_sum)

        # evaluate via cheap surrogate (max autocorrelation)
        score = 2.0 * n * _max_autocorr(a_new) / (target_sum ** 2)

        if score < best_score:
            best_score, best_seq = score, a_new.copy()
            print(f"[{time.time():.2f}] New best (gradient) = {best_score:.6f} (n={n})")

        # occasional random shake‑up
        if np.random.rand() < 0.02:
            noise = np.random.randn(n) * (0.01 * target_sum / n)
            a = _project_onto_simplex(np.maximum(a_new + noise, 0.0), target_sum)
            base_step *= 0.9
        else:
            a = a_new

    return best_seq, best_score
