def solve_lp_max_sum(f, M, bounds=(0,1000)):
    n = len(f)
    num_constraints = 2*n - 1
    # Build A_ub matrix (num_constraints, n) where each row corresponds to constraint conv(f,g)[k] <= M.
    A = np.zeros((num_constraints, n), dtype=np.float64)
    for k in range(num_constraints):
        # contribution from g_j for j=0..n-1 where i=k-j must be between 0..n-1
        # So j from max(0, k-(n-1)) to min(k, n-1)
        j_start = max(0, k - (n-1))
        j_end = min(k, n-1)
        # for each j in that range, i = k - j
        # set A[k, j] = f[i]
        # Could vectorize:
        idxs = np.arange(j_start, j_end+1)
        i_vals = k - idxs
        A[k, idxs] = np.array(f)[i_vals]
    # objective: maximize sum(g) -> minimize -sum
    c = -np.ones(n)
    # RHS: M repeated for each constraint
    b = np.full(num_constraints, M)
    # bounds: (0, 1000) for each var
    bounds = [(0, 1000) for _ in range(n)]
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs', options={'presolve': True})
    if res.success:
        return res.x
    else:
        raise RuntimeError('LP failed')

from scipy.optimize import linprog
from scipy import sparse

def solve_lp_max_sum(f, M):
    n = len(f)
    num_cons = 2 * n - 1
    # Build A_ub as list of rows
    data = []
    rows = []
    cols = []
    for k in range(num_cons):
        # indices j range
        j_start = max(0, k - (n - 1))
        j_end = min(k, n - 1)
        # Append each coefficient f_i where i = k - j
        for j in range(j_start, j_end + 1):
            i = k - j
            rows.append(k)
            cols.append(j)
            data.append(f[i])
    A_ub = sparse.csr_matrix((data, (rows, cols)), shape=(num_cons, n), dtype=np.float64)
    c = -np.ones(n)
    b_ub = np.full(num_cons, M, dtype=np.float64)
    bounds = [(0, M) for _ in range(n)]   # M is bound
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs', options={'presolve': True})
    if not res.success:
        raise RuntimeError("LP failed")
    return res.x

"""
Propose a candidate sequence that beats the π/2 ≈ 1.5708 bound.

Algorithm
---------
1. Initialise with the classic inverse‑square‑root sequence
   a_i = C(2i,i)/4^i (its convolution is almost flat, score ≈ π/2).

2. Systematically explore a one‑parameter family that shifts weight from the
   beginning of the sequence toward the end:

        a_i(s) = a_i · (1 + s·((2*i)/(n‑1) – 1))

   For s>0 early entries are multiplied by (1‑s) and the last entries
   by (1+s).  The parameter s is scanned on a coarse grid; the best
   candidate is kept.

3. Refine the best candidate with cheap stochastic moves that are known
   to be effective for this problem:
   * a single multiplicative perturbation of one entry,
   * transfer a tiny amount from a “early’’ index to a “late’’ index,
   * simultaneous multiplicative perturbations of two random indices.

   Every move is accepted only if the evaluation score improves.

All steps respect the security checks of `evaluate_sequence`.  The routine
runs for the allotted time (default 1000 s) and returns the best sequence
found.
"""

import time
import numpy as np

def generate_inverse_sqrt_sequence(n: int) -> list[float]:
    """Return a_i = binom(2*i,i)/4**i via the recurrence
       a_{i+1}=a_i·(2*i+1)/(2*i+2).  The series solves (1‑z)^‑½."""
    seq = [1.0]                     # a_0 = 1
    for i in range(1, n):
        # recurrence a_i = a_{i-1} * (2*i-1)/(2*i)
        seq.append(seq[-1] * (2 * i - 1) / (2 * i))
    return seq


def linear_transform(seq: list[float], s: float) -> list[float]:
    """
    Apply the linear weight shift a_i ← a_i·(1 + s·((2*i)/(n‑1) – 1)).
    Values forced to be non‑negative.
    """
    n = len(seq)
    i_arr = np.arange(n, dtype=np.float64)
    factor = 1.0 + s * ((2.0 * i_arr) / (n - 1) - 1.0)
    factor = np.maximum(factor, 0.0)            # protect against negatives
    return (np.array(seq, dtype=np.float64) * factor).tolist()
