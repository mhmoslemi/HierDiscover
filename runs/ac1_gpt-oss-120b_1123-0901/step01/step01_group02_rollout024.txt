def solve_lp(f, M):
    n = len(f)
    # Build A_ub matrix: for each k, coefficients f_i for b_{k-i}
    rows = []
    cols = []
    data = []
    for i in range(n):
        # For each i, column i has coefficient f_i for rows k where 0 <= k-i < n -> k in [i, i + n - 1]
        start = i
        end = i + n
        # For each k from start to end-1:
        # coefficient at row k, col i = f_i
        # But we need to ensure k length 2n-1.
        for k in range(start, min(end, 2*n-1)):
            rows.append(k)
            cols.append(i)
            data.append(f[i])
    A_ub = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    b_ub = M * np.ones(2*n-1)
    res = scipy.optimize.linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
    return res.x

def _lp_max_sum_under_M(f, M, target_sum):
    n = len(f)
    # Build sparse matrix A where A[k,i] = f[i] if 0 <= k - i < n else 0
    rows = []
    cols = []
    data = []
    for i in range(n):

        end = i + n

        return f.copy()
    b = b_raw * (target_sum / total)
    # Clip to allowed range
    np.clip(b, 0.0, 1000.0, out=b)
    return b

"""
Hybrid optimizer for the step‑function bound.

The algorithm keeps the ℓ₁–norm fixed to `sqrt(2·n)` (the normalisation used by
the reference implementation).  It combines three ideas:

1. **Projected sub‑gradient descent** – the gradient of the objective
   `E = 2·n·max_k (a⋆a)[k] / (∑a)²` is derived analytically.
   When several convolution entries attain the maximum, the sub‑gradient is
   taken as the average over all active shifts, which gives a smoother descent.

2. **Random jitter / simulated‑annealing moves** – every few thousand steps a
   tiny Gaussian perturbation is added and a very small probability of
   accepting a worsening move is kept, to escape shallow plateaus.

3. **Periodic linear‑programming boost** – for the current iterate `f`,
   the LP `max Σ b_j   s.t.   (f⋆b)[k] ≤ M_f` (with `M_f` the current maximal
   convolution) yields an extreme direction `g`.  The new point
   `f ← (1‑t)·f + t·g` (with `t≈0.5`) often reduces the peak dramatically.
   The LP is built only occasionally because it is relatively expensive.

Multiple simple initialisations (uniform, geometric, random Dirichlet, a
Sidon‑like pattern and the provided `height_sequence_1`) are tried and the
best solution seen so far is kept.  The whole search stops a few seconds
before the wall‑clock budget runs out.

All operations are performed with NumPy (FFT‑based convolution) and, when
available, SciPy's `linprog`.  The returned list is guaranteed to satisfy the
security checks of `evaluate_sequence`.
"""

import time
import math
import numpy as np

# optional SciPy – used only for the LP boost
try:
    import scipy.optimize
    import scipy.sparse
    _HAVE_SCIPY = True
except Exception:               # pragma: no cover
    _HAVE_SCIPY = False


def _conv_stats(a: np.ndarray):
    """Return (M, active_indices, full_convolution) for vector a."""
    n = a.shape[0]
    L = 2 * n - 1
    # FFT convolution (real → real)
    fa = np.fft.rfft(a, n=L)
    b = np.fft.irfft(fa * fa, n=L)
    b = np.real(b)                       # eliminate tiny imag part
    M = b.max()
    atol = 1e-12 * max(1.0, M)
    active = np.where(b >= M - atol)[0]
    return M, active, b


def _gradient_step_multi(a: np.ndarray, target_sum: float, lr: float):
    """
    One projected (sub‑)gradient step.
    The gradient is averaged over all active convolution indices.
    """
    n = a.shape[0]
    S = a.sum()
    if S == 0.0:
        return a.copy()

    M, active, _ = _conv_stats(a)

    # sum of the reversed contributions for all active shifts
    rev_total = np.zeros_like(a)
    for k in active:
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        rev_total[mask] += a[j[mask]]
    # average over the active set
    rev_avg = rev_total / max(1, len(active))

    # sub‑gradient of the objective (see derivation in the docstring)
    grad = (4.0 * n / (S * S)) * rev_avg - (4.0 * n * M / (S ** 3))

    # gradient step, projection onto the non‑negative orthant and renorm
    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0.0)
    sum_new = a_new.sum()
    if sum_new > 0.0:
        a_new = a_new * (target_sum / sum_new)
    else:                                 # should not happen, but stay safe
        a_new = a.copy()
    np.clip(a_new, 0.0, 1000.0, out=a_new)
    return a_new


def _lp_max_sum_under_M(f: np.ndarray, M: float, target_sum: float):
    """
    Solve   max Σ b_j   s.t.   (f⋆b)[k] ≤ M  for all k,
    b ≥ 0.  Returns the optimal b scaled to `target_sum`.
    Uses SciPy's HiGHS interior‑point solver (if available).
    """
    if not _HAVE_SCIPY:                     # pragma: no cover
        raise RuntimeError("SciPy not available – cannot run LP step")

    n = f.shape[0]
    rows = []
    cols = []
    data = []

    # Build the (2n‑1) × n constraint matrix  A  where
    #   (A·b)_k = Σ_i f_i * b_{k‑i}   (zero padding outside the range)
    for i in range(n):
        start = i
        end = i + n                         # exclusive
        # rows k = start … end‑1 (clipped to length 2n‑1)
        row_start = start
        row_end = min(end, 2 * n - 1)
        cnt = row_end - row_start
        rows.extend(range(row_start, row_end))
        cols.extend([i] * cnt)
        data.extend([f[i]] * cnt)

    A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)                         # minimise –Σb ⇔ maximise Σb
    b_ub = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n

    res = scipy.optimize.linprog(
        c, A_ub=A, b_ub=b_ub, bounds=bounds,
        method='highs', options={'presolve': True}
    )
    if not res.success:
        raise RuntimeError(f"LP failed: {res.message}")

    b_raw = res.x
    total = b_raw.sum()
    if total == 0.0:
        return f.copy()
    b = b_raw * (target_sum / total)
    np.clip(b, 0.0, 1000.0, out=b)
    return b


def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step‑function with a small value of
    `evaluate_sequence`.  Returns the best sequence found within the time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0      # safety margin for cleanup

    # ------------------------------------------------------------------
    # determine length and normalisation
    try:
        base = np.array(height_sequence_1, dtype=float)   # type: ignore
    except Exception:
        base = np.ones(2100, dtype=float)                 # fallback length

    n = base.shape[0]
    target_sum = math.sqrt(2.0 * n)                       # normalisation used by the paper

    # ------------------------------------------------------------------
    # generate a few diverse starting points
    starts = []

    # 1) the supplied SOTA construction (if any)
    a0 = base.copy()
    a0 = a0 * (target_sum / a0.sum())
    starts.append(a0)

    # 2) uniform vector
    uniform = np.full(n, target_sum / n)
    starts.append(uniform)

    # 3) geometric decay (r close to 1)
    r = 0.9995
    geom = r ** np.arange(n)
    geom = geom * (target_sum / geom.sum())
    starts.append(geom)

    # 4) random Dirichlet (Gamma(1) → exponential)
    rand = np.random.rand(n)
    rand = rand * (target_sum / rand.sum())
    starts.append(rand)

    # 5) pseudo‑Sidon pattern (roughly √n equally spaced spikes)
    K = int(math.sqrt(2.0 * n)) + 10
    idx = np.linspace(0, n - 1, K, dtype=int)
    sid = np.zeros(n)
    sid[idx] = target_sum / K
    starts.append(sid)

    # ------------------------------------------------------------------
    # helper to compute the objective directly (identical to evaluate_sequence)
    def _score(a):
        M = np.max(np.convolve(a, a))
        return 2.0 * n * M / (target_sum ** 2)

    # evaluate the seeds, keep the current best
    best_seq = None
    best_score = float('inf')
    for a in starts:
        sc = _score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
        print(f"[init] seed score = {sc:.6f}, current best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # main optimisation loop – start from the best seed
    a = best_seq.copy()
    iteration = 0
    lr0 = 0.9                               # initial learning rate
    decay = 4e-5                            # controls decay of the step size
    patience = 0
    max_patience = 8000                     # re‑initialise after long stagnation

    while time.time() < deadline:
        # learning rate schedule
        lr = lr0 / (1.0 + decay * iteration)

        # ---- projected sub‑gradient descent step (multi‑active) ----
        a_new = _gradient_step_multi(a, target_sum, lr)

        # occasional tiny random jitter (helps escape plateaus)
        if iteration % 2000 == 0 and iteration > 0:
            jitter = np.random.randn(n) * 0.001 * target_sum / n
            a_new = a_new + jitter
            a_new = np.maximum(a_new, 0.0)
            a_new = a_new * (target_sum / a_new.sum())

        # evaluate current objective (fast, no call to evaluate_sequence)
        cur_score = _score(a_new)

        # keep the best ever seen
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a_new.copy()
            print(f"[iter {iteration:07d}] new best = {best_score:.6f}")
            patience = 0
        else:
            patience += 1

        # accept or reject the step (simple hill‑climb with occasional uphill jump)
        if cur_score <= best_score:
            a = a_new
        else:
            if np.random.rand() < 0.01:      # rare uphill move
                a = a_new

        # ---- periodic LP boost ------------------------------------------------
        if iteration % 5000 == 0 and iteration > 0 and _HAVE_SCIPY:
            try:
                M_cur, _, _ = _conv_stats(a)
                g = _lp_max_sum_under_M(a, M_cur, target_sum)
                # blend the current point toward the LP direction
                t = 0.5
                a = (1 - t) * a + t * g
                a = a * (target_sum / a.sum())      # renormalise
                # re‑evaluate after the blend
                cur_score = _score(a)
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = a.copy()
                    print(f"[lp {iteration}] post‑mix new best = {best_score:.6f}")
            except Exception as exc:                # pragma: no cover
                print(f"[lp {iteration}] LP failed: {exc}")

        # ------------------------------------------------------------------------
        # if we have been stuck for a while, restart from a fresh random point
        if patience >= max_patience:
            a = np.random.rand(n)
            a = a * (target_sum / a.sum())
            patience = 0
            print(f"[restart {iteration}] random re‑initialisation")

        iteration += 1

    print(f"[done] iterations = {iteration}, best score = {best_score:.6f}")
    # final sanity check – the harness will apply the same clipping rules,
    # but we keep the values within bounds already.
    return best_seq.tolist()
