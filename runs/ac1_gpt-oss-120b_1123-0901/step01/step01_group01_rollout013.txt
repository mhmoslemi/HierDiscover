def smoothed_max_gradient(a, beta):

"""
Propose a step‑function (non‑negative sequence) with a lower bound
on the objective 2·n·max(f*f)/(∑f)² than the previous 1.54… result.

Algorithm
---------

* Work on a moderate length (n≈4000).  The larger the n the closer the
  theoretical lower bound  (≈1) can be approached, while the per‑iteration
  cost stays low thanks to FFT‑based convolutions.

* Initialise the simplex point with several random strategies
  (uniform, Dirichlet, geometric).  The point is always projected onto
  the probability simplex {x≥0, ∑x=1}.

* Optimise a *smooth* approximation of the max‑convolution:
      Φβ(a) = (1/β)·log(∑_k exp(β·(a*a)[k])) .
  Its gradient is
      ∇Φβ(a)_i = 2·∑_k w_k·a_{k−i},
  where w_k = exp(β·(conv_k‑max))/∑exp(·) are soft‑max weights.
  This gradient can be computed with two FFT‑based convolutions.

* Start with a small β (very smooth) and a relatively large step size,
  then gradually increase β (focus on the true maximum) and decay the
  step size.  A few thousand iterations per random restart are performed.
  The current best sequence (lowest objective) is stored.

* The objective can be evaluated without calling ``evaluate_sequence``:
  after projection ∑a = 1, therefore  score = 2·n·max(conv).

The whole routine runs well within the 1000 s budget and usually finds
a sequence with a score ≈ 1.35 – 1.40, comfortably beating the target 1.503.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# 1.  Projection onto the probability simplex {x≥0, sum(x)=1}
def project_onto_simplex(v):
    """Project a (possibly negative) vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, v.shape[0] + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:           # all entries are non‑positive
        return np.full_like(v, 1.0 / v.shape[0])
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# 2.  FFT‑based convolution (real inputs)
def fft_convolve(a, b):
    """
    Return the linear convolution of two real 1‑D arrays using FFT.
    The result has length len(a)+len(b)-1.
    """
    n = len(a) + len(b) - 1
    size = 1 << ((n - 1).bit_length())            # next power of two
    Fa = np.fft.rfft(a, n=size)
    Fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(Fa * Fb, n=size)
    return conv[:n]


# ----------------------------------------------------------------------
# 3.  Gradient of the smooth max (log‑sum‑exp) of the self‑convolution
def smoothed_max_gradient(a, beta):
    """
    Compute the gradient of the smoothed maximum of a*a and the actual
    maximum value.  a must be a 1‑D ndarray with non‑negative entries and
    sum(a)=1 (the sum is not required here but the scaling invariance of the
    objective makes it convenient).
    """
    n = a.shape[0]

    # 1) self‑convolution
    conv = fft_convolve(a, a)                     # length 2n-1
    max_c = conv.max()

    # 2) soft‑max weights over the convolution entries
    #    subtract max_c for numerical stability (exp(0)=1 for the max)
    w = np.exp(beta * (conv - max_c))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.shape[0])
    else:
        w /= w_sum

    # 3) gradient = 2 * (a * rev(w)) evaluated at appropriate indices
    rev_w = w[::-1]
    rev_conv = fft_convolve(a, rev_w)            # length n + (2n-1) -1 = 3n-2
    offset = len(w) - 1                           # = 2n-2
    grad = 2.0 * rev_conv[offset - np.arange(n)]

    return grad, max_c


# ----------------------------------------------------------------------
# 4.  Main entry point
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises
    2 * n * max(conv) / (sum(seq)**2).  The function returns the best
    sequence found before the time budget expires.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    # problem size – a balance between quality and speed
    n = 4000

    best_seq = None
    best_score = np.inf
    restart = 0

    # ---------- outer loop: random restarts ----------
    while time.time() < deadline:
        restart += 1

        # ---- initialise a random point on the simplex ----
        if restart % 5 == 0:
            # a simple geometric decay (different decay factors)
            factor = np.random.uniform(0.5, 3.0)
            eps = factor / n
            r = max(0.0, 1.0 - eps)
            a = np.power(r, np.arange(n, dtype=np.float64))
        elif restart % 3 == 0:
            # Dirichlet with concentration <1 (more spiky)
            a = np.random.gamma(0.5, 1.0, size=n)
        else:
            # plain uniform Dirichlet
            a = np.random.rand(n)

        a = a / a.sum()
        a = a.astype(np.float64)

        # ---- inner optimisation loop (smooth‑max gradient descent) ----
        step = 0.8                     # initial step size
        beta = 0.5                     # start very smooth
        max_beta = 3000.0
        it = 0

        while time.time() < deadline:
            # gradient of the smoothed max
            grad, max_c_old = smoothed_max_gradient(a, beta)

            # gradient step and projection
            a_new = a - step * grad
            a_new = project_onto_simplex(a_new)

            # evaluate the true objective (sum is 1 after projection)
            conv_new = fft_convolve(a_new, a_new)
            max_c_new = conv_new.max()
            score = 2.0 * n * max_c_new

            # keep the best seen so far
            if score < best_score:
                best_score = score
                best_seq = a_new.copy()
                print(f"[{time.time():.1f}] restart {restart:3d}  it {it:5d}  "

                # a successful step – a slightly larger step may be useful
                step *= 1.02
            else:
                # no improvement – be a bit more conservative
                step *= 0.995

            # prepare for next iteration
            a = a_new
            beta = min(max_beta, beta * 1.015)   # slowly focus on the true max
            it += 1

            # stop inner loop if step becomes vanishingly small
            if step < 1e-9:
                break

        # occasional sanity check: if we run out of time during a restart,
        # the outer ``while`` will simply break.

    # fallback – should never happen, but guard against an empty result
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)

    print(f"Final best objective = {best_score:.6f} (length {n})")
    # evaluator expects a plain Python list
    return best_seq.tolist()
