"""
Improved search for a step‑function that minimises
    2 * n * max(conv(a,a)) / (sum(a))**2

The algorithm works on a fairly large support (n = 5000) and combines:

* FFT‑based convolution (O(n log n)) for speed.
* A smoothed max using a small soft‑max over the K largest convolution values.
* Exponential‑gradient (mirror descent) on the probability simplex.
* Adaptive learning‑rate decay and occasional random restarts to escape plateaus.
* Several simple initialisations (uniform, geometric, triangular, random and the
  previous global solution if available).

The optimiser runs until the time budget (≈1000 s) expires and returns the

import time
import numpy as np

# --------------------------------------------------------------
# Simple projection onto the simplex via normalisation (exponential update keeps positivity)
def _normalize(v: np.ndarray) -> np.ndarray:
    """Scale a non‑negative vector to sum to 1 (handles the degenerate zero case)."""
    s = v.sum()
    if s <= 0.0 or np.isnan(s):
        return np.full_like(v, 1.0 / v.size, dtype=np.float64)
    return v / s


# --------------------------------------------------------------
# FFT‑based linear convolution a*a (real vectors)
def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Return the linear convolution a*a using FFT of length `fft_len`."""
    A = np.fft.rfft(a, n=fft_len)
    c = np.fft.irfft(A * A, n=fft_len)
    # keep only the first 2*n-1 entries (the linear convolution length)
    return c[: 2 * a.size - 1]

def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative sequence that beats the uniform bound.
    The routine stays well within the given time budget and returns the
    best sequence discovered (as a Python list of floats).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    # -------------------- 1. Choose problem size --------------------
    n = 5000                               # larger support gives a lower asymptotic bound
    fft_len = 1 << ((2 * n - 1).bit_length())   # next power‑of‑two for efficient FFT

    # -------------------- 2. Build a few starting candidates ----------
    candidates = []

    # a) previous global solution, if it matches the chosen length
    if 'height_sequence_1' in globals():
        prev = np.asarray(globals()['height_sequence_1'], dtype=np.float64)
        if prev.size != n:
            # truncate or pad with a tiny uniform mass
            if prev.size > n:
                prev = prev[:n]
            else:
                pad = np.full(n - prev.size, 1e-12, dtype=np.float64)
                prev = np.concatenate([prev, pad])
        candidates.append(_normalize(prev))

    # b) simple geometric progression (the “theoretical” starter)
    factor = 2.0
    eps = factor / n
    r = 1.0 - eps
    if r > 0.0:
        geo = np.power(r, np.arange(n, dtype=np.float64))
        candidates.append(_normalize(geo))

    # c) uniform distribution (the baseline)
    candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

    # d) a symmetric triangular shape
    t = np.arange(1, n // 2 + 1, dtype=np.float64)
    if n % 2 == 0:
        tri = np.concatenate([t, t[::-1]])
    else:
        tri = np.concatenate([t, t[-2::-1]])   # odd length – centre repeated once
    candidates.append(_normalize(tri))

    # e) a random Dirichlet draw (adds stochastic diversity)
    candidates.append(np.random.dirichlet(np.ones(n)))

    # -------------------- 3. Initialise best solution --------------------
    best_seq = None
    best_score = float("inf")
    for seq in candidates:
        score = evaluate_sequence(seq.tolist())
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
    print(f"[Init] best objective = {best_score:.6f}")

    # -------------------- 4. Optimisation loop --------------------
    # hyper‑parameters (tuned empirically)
    lr = 0.8                # base learning rate for exponential update
    lr_min = 1e-4
    decay = 0.9995          # exponential decay per iteration
    topK = 5                # number of largest convolution entries considered
    temp = 200.0            # temperature for the soft‑max over the topK values
    restart_after = 3000    # iterations without improvement → restart
    it = 0
    no_improve = 0

    # start from the best of the initial candidates
    seq = best_seq.copy()

    while time.time() < deadline:
        # -------- convolution and current objective ----------
        conv = _conv_fft(seq, fft_len)          # length 2n‑1
        max_val = conv.max()
        score = 2.0 * n * max_val               # sum(seq) == 1 → denominator = 1

        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"[Iter {it:06d}] new best = {best_score:.6f}")
            no_improve = 0
        else:
            no_improve += 1

        # -------- occasional random restart to escape plateaus ----------
        if no_improve >= restart_after:
            # perturb the current best slightly and re‑normalise
            perturb = best_seq * (0.9 + 0.2 * np.random.rand(n))
            seq = _normalize(perturb)
            no_improve = 0
            print(f"[Iter {it:06d}] restart triggered")
            continue

        # -------- form a smoothed sub‑gradient ----------
        # indices of the topK convolution entries
        if topK == 1:
            active = [int(conv.argmax())]
            weights = [1.0]
        else:
            # argpartition gives the K largest (unsorted)
            top_idx = np.argpartition(-conv, topK - 1)[:topK]
            # soft‑max weights on those values (sharper for larger `temp`)
            vals = conv[top_idx]
            vmax = vals.max()
            soft = np.exp(temp * (vals - vmax))
            weights = (soft / soft.sum()).tolist()
            active = top_idx.tolist()

        grad = np.zeros_like(seq)
        for k, wgt in zip(active, weights):
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            # sub‑gradient of (a*a)[k] w.r.t. a_i is a_{k-i} on the overlap
            grad[idx] += wgt * seq[k - idx]

        # -------- exponential (mirror) update on the simplex ----------
        seq = seq * np.exp(-lr * grad)   # element‑wise
        seq = _normalize(seq)

        # decay learning rate
        lr = max(lr_min, lr * decay)
        it += 1

    print(f"Finished after {it} iterations – best objective = {best_score:.6f}")
    # The evaluator expects a plain Python list
    return best_seq.tolist()
