"""
Propose a step‑function (non‑negative heights) that minimises

        2 * n * max(conv(a, a)) / (sum(a) ** 2)

The classic (i+γ)^‑p family attains a value ≈π/2≈1.57 because the
zero‑lag term a₀² dominates the convolution.  By forcing a₀=0 (a leading
zero) we remove that large peak; the maximum then comes from the
central part of the self‑convolution and the bound

        2·(1‑p)²·B(1‑p,1‑p)

becomes attainable.  The search therefore explores

    a_i = (i+γ)⁻ᵖ  for i ≥ offset
    a_i = 0        for i < offset

with a small integer offset (0 or 1), exponent p∈[0.55,0.90] and a few
γ‑values.  A coarse grid is examined first, then a short random
refinement around the best triple (p,γ,offset).  The algorithm respects
the 1000 s budget and returns the best sequence (as a plain Python list)
found.
"""

import time
import numpy as np
from math import log, exp

# Optional: use scipy for the beta function if you want a more precise
# theoretical guide.  It is not required for the search itself.
try:
    from scipy.special import beta as beta_func
except Exception:  # pragma: no cover
    beta_func = None


def _make_sequence(n: int, p: float, gamma: float = 0.0, offset: int = 0) -> np.ndarray:
    """
    Build a step‑function of length n.
    - a[0:offset] = 0
    - a[offset:] = (i + gamma) ** (-p)   where i runs from offset … n‑1
    """
    seq = np.zeros(n, dtype=float)
    # indices that actually host the power‑law part
    idx = np.arange(offset, n, dtype=float)
    seq[offset:] = (idx + gamma) ** (-p)
    return seq


def propose_candidate(seed: int = 42, budget_s: float = 1000) -> list:
    """
    Search for a low‑score sequence using a grid over (p, γ, offset)
    followed by a small stochastic refinement.
    Returns a plain Python list that can be passed directly to
    ``evaluate_sequence``.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.1   # tiny safety margin

    n = 3428                     # length used in the reference solution
    best_seq = None
    best_score = float("inf")
    best_params = None           # (p, gamma, offset)

    # ------------------------------------------------------------------
    # 1) Coarse grid search
    # ------------------------------------------------------------------
    # exponent grid – we focus on the region where the theoretical bound
    # is low (p ≳ 0.6).  31 points give ~0.01 resolution.
    p_grid = np.linspace(0.55, 0.90, 31)

    # a few gamma candidates (including 0 – allowed when offset ≥ 1)
    gamma_grid = [0.0, 0.05, 0.2, 0.5, 1.0, 2.0, 5.0]

    # offset values – 0 (no leading zero) reproduces the earlier
    # implementation, 1 removes the dominant a0² term.
    offset_grid = [0, 1]

    for offset in offset_grid:
        for p in p_grid:
            for gamma in gamma_grid:
                if time.time() > deadline:
                    break

                # avoid division by zero when offset==0 and gamma==0
                if offset == 0 and gamma == 0.0:
                    continue

                seq = _make_sequence(n, p, gamma, offset)
                # evaluate; convert to plain list to match the expected type
                score = evaluate_sequence(seq.tolist())
                if score < best_score:
                    best_score = score
                    best_seq = seq.copy()
                    best_params = (p, gamma, offset)
                    print(f"[grid] new best p={p:.4f}, γ={gamma:.4f}, offset={offset}, "
                          f"score={score:.6f}")

    # ------------------------------------------------------------------
    # 2) Random local refinement around the best triple
    # ------------------------------------------------------------------
    if best_seq is None:  # safety fallback – should never happen
        best_seq = _make_sequence(n, 0.75, 0.0, 1)
        best_score = evaluate_sequence(best_seq.tolist())
        best_params = (0.75, 0.0, 1)

    # refinement settings – a few thousand cheap trials are enough
    refine_iters = 8000
    p_best, gamma_best, offset_best = best_params

    for _ in range(refine_iters):
        if time.time() > deadline:
            break

        # propose a perturbation; use smaller steps as we get close
        p_new = p_best + np.random.randn() * 0.006   # ~0.6 % steps
        gamma_new = gamma_best + np.random.randn() * 0.12
        # offset is integer; try -1, 0, +1 relative move
        offset_new = offset_best + np.random.choice([-1, 0, 1])

        # keep parameters in sensible ranges
        if not (0.5 < p_new < 0.95):
            continue
        if gamma_new < 0.0:
            continue
        if offset_new < 0:
            offset_new = 0
        if offset_new > 5:          # no need for many leading zeros
            offset_new = 5

        # guard against the zero‑division case
        if offset_new == 0 and gamma_new == 0.0:
            continue

        seq = _make_sequence(n, p_new, gamma_new, offset_new)
        score = evaluate_sequence(seq.tolist())
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            p_best, gamma_best, offset_best = p_new, gamma_new, offset_new
            print(f"[refine] improved p={p_new:.4f}, γ={gamma_new:.4f}, offset={offset_new}, "

    # Ensure the output is a plain Python list (no NumPy array)
    result = best_seq.tolist()
    print(f"Finished search – best score {best_score:.6f} "
          f"with p={p_best:.4f}, γ={gamma_best:.4f}, offset={offset_best}")
    return result
