def propose_candidate(budget_s=1000.0, seed=42, **kwargs):
    import numpy as np, time
    np.random.seed(seed)
    try:
        n = len(height_sequence_1)
    except NameError:
        n = 2000

    # generate arcsine coeffs
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i-1] * (2*(i-1) + 1) / (2.0 * i)
    seq = a / a.sum()   # sum = 1

    # ensure non-zero length
    conv = np.convolve(seq, seq)  # length 2n-1
    cur_score = 2.0 * n * conv.max()
    best_score = cur_score
    best_seq = seq.copy()
    current_seq = seq.copy()
    current_conv = conv.copy()
    current_score = cur_score

    start = time.time()
    # parameters
    initial_temp = 0.02
    final_temp = 1e-4
    max_factor_start = 0.2
    max_factor_end = 0.01
    target_move_prob = 0.3

    it = 0
    while True:
        elapsed = time.time() - start
        if elapsed > budget_s - 0.5:
            break
        # compute temperature
        prog = elapsed / budget_s
        temp = initial_temp * (1 - prog) + final_temp * prog
        max_factor = max_factor_start * (1 - prog) + max_factor_end * prog

        # decide move type
        if np.random.rand() < target_move_prob:
            # targeted move
            # find index of max conv
            k_max = int(np.argmax(current_conv))
            # range for i indices
            lo = max(0, k_max - (n - 1))
            hi = min(n - 1, k_max)
            i_candidates = np.arange(lo, hi + 1, dtype=int)
            # contributions to k_max
            contrib = current_seq[i_candidates] * current_seq[k_max - i_candidates]
            i = i_candidates[np.argmax(contrib)]

            # choose j outside the contributing region
            if k_max < n - 1:
                # j > k_max
                j_options = np.arange(k_max + 1, n, dtype=int)
            else:
                # j < k_max - (n - 1) + 1?
                upper = k_max - (n - 1)
                if upper <= 0:
                    j_options = np.arange(0, n, dtype=int)
                else:
                    j_options = np.arange(0, upper, dtype=int)
            # ensure j_options not empty and exclude i
            if j_options.size == 0:
                # fallback to random
                j = np.random.choice([idx for idx in range(n) if idx != i])
            else:
                # exclude i if present
                if i in j_options:
                    j_options = j_options[j_options != i]
                if j_options.size == 0:
                    j = np.random.choice([idx for idx in range(n) if idx != i])
                else:
                    j = np.random.choice(j_options)
        else:
            # random pair weighted by seq magnitude for i
            i = np.random.choice(n, p=current_seq / current_seq.sum())
            # choose j != i uniformly
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

        # move mass
        ai = current_seq[i]
        if ai <= 0:
            continue
        eps = ai * max_factor * np.random.rand()
        if eps <= 0:
            continue

        # Save old state
        a_old = current_seq.copy()
        c_old = current_conv.copy()

        delta_i = -eps
        delta_j = eps

        # linear updates
        current_conv[i:i+n] += 2.0 * delta_i * a_old
        current_conv[j:j+n] += 2.0 * delta_j * a_old

        # quadratic updates
        idx_2i = 2 * i
        idx_2j = 2 * j
        idx_ij = i + j
        current_conv[idx_2i] += eps * eps
        current_conv[idx_2j] += eps * eps
        current_conv[idx_ij] -= 2.0 * eps * eps

        # update seq
        current_seq[i] = a_old[i] + delta_i
        current_seq[j] = a_old[j] + delta_j

        new_score = 2.0 * n * current_conv.max()
        accept = False
        if new_score < current_score - 1e-12:
            accept = True
        else:
            # occasional uphill acceptance
            if np.random.rand() < np.exp(-(new_score - current_score) / (temp + 1e-12)):
                accept = True

        if accept:
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = current_seq.copy()
                # optionally log improvement
                if it % 1000 == 0:
                    print(f"[propose_candidate] iter={it}, time={elapsed:.1f}s, best_score={best_score:.6f}")
        else:
            # revert
            current_seq[:] = a_old
            current_conv[:] = c_old
            # current_score unchanged

        # occasional renormalisation
        if (it & 0xFFF) == 0:  # every 4096 iterations
            s = current_seq.sum()
            if abs(s - 1.0) > 1e-8:
                current_seq /= s
                # recompute conv from scratch to maintain consistency
                current_conv = np.convolve(current_seq, current_seq)
                current_score = 2.0 * n * current_conv.max()

        it += 1

    # after loop, final best_seq
    seq_out = best_seq.tolist()
    try:
        final_score = evaluate_sequence(seq_out)
        print(f"[propose_candidate] final_length={n}, final_score={final_score:.6f}")
    except Exception:
        pass

    return seq_out

"""
Propose a (near‑optimal) non‑negative step function.

The algorithm
---------------
1.  Initialise the sequence with the first *n* coefficients of the arcsine
    series (the known asymptotically optimal shape).  These coefficients are
    normalised so that the sum equals one.

2.  Keep the current sequence `a` and its full self‑convolution `c = a*a`
    (length 2 n – 1).  The evaluation function for a normalised sequence is

        score = 2·n·max(c).

3.  Optimise `a` by a stochastic local‑search / simulated‑annealing loop.
    In each iteration a small amount `ε` is moved from an index `i` to another
    index `j`.  The convolution `c` is updated **exactly** in O(n) time using
    the analytical formula for the effect of a rank‑2 update on a self‑convolution.

    *Two kinds of moves are tried*:
    – **targeted**: pick the index `i` that contributes most to the current
      maximal convolution entry and move mass to a `j` that contributes the
      least to that entry (often zero).  This directly flattens the peak.
    – **random**: pick `i` with probability proportional to its weight
      (large entries are explored more) and a random `j ≠ i`.  This provides
      global exploration.

    Moves are accepted if they improve the score, otherwise they are accepted
    with a probability `exp(-(Δ)/T)` where the temperature `T` decays linearly
    over the time budget (simulated annealing).

4.  Occasionally the sequence is re‑normalised to counteract floating‑point
    drift; the convolution is recomputed from scratch at these points.

5.  The best sequence seen during the run is returned.  The whole optimisation
    respects the 1000 s budget and uses only `numpy` operations, avoiding the
    O(n²) cost of repeatedly calling `np.convolve`.

The approach is able to push the bound below the previous value ≈ 1.571,
approaching the target ≈ 1.503.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: arcsine coefficients (1‑z)^(-1/2) series
# ----------------------------------------------------------------------
def _arcsine_coeffs(n: int) -> np.ndarray:
    """Return the first n terms a_i = C(2i,i) / 4^i using a stable recurrence."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * (i - 1) + 1) / (2.0 * i)
    return a


# ----------------------------------------------------------------------
# Core optimisation routine
# ----------------------------------------------------------------------
def propose_candidate(budget_s: float = 1000.0, seed: int = 42, **kwargs):
    """
    Return a non‑negative sequence that (approximately) minimises the
    evaluation function defined in the problem statement.

    Parameters
    ----------
    budget_s : float
        Time budget in seconds.
    seed : int
        Random seed (deterministic run).

    Returns
    -------
    list[float]
        The best sequence found (normalised to sum = 1).
    """
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # Choose a length.  If a previous construction exists we reuse its
    # length; otherwise we default to 2000.
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)      # type: ignore[name-defined]
    except Exception:
        n = 2000

    # ------------------------------------------------------------------
    # Initialise with the arcsine coefficients and normalise.
    # ------------------------------------------------------------------
    a = _arcsine_coeffs(n)
    seq = a / a.sum()                     # sum == 1
    conv = np.convolve(seq, seq)          # length 2n‑1
    cur_score = 2.0 * n * conv.max()
    best_score = cur_score
    best_seq = seq.copy()

    # ------------------------------------------------------------------
    # Simulated‑annealing parameters
    # ------------------------------------------------------------------
    start = time.time()
    init_temp = 0.02
    final_temp = 1e-4
    init_max_factor = 0.20   # at the start we may move up to 20 % of a[i]
    final_max_factor = 0.01
    target_move_prob = 0.35   # probability of using the “targeted” move

    iteration = 0

    while True:
        elapsed = time.time() - start
        if elapsed > budget_s - 0.5:
            break

        # temperature and maximal move size (linear schedules)
        progress = elapsed / budget_s
        temp = init_temp * (1.0 - progress) + final_temp * progress
        max_factor = init_max_factor * (1.0 - progress) + final_max_factor * progress

        # ------------------------------------------------------------------
        # Choose a move.
        # ------------------------------------------------------------------
        if np.random.rand() < target_move_prob:
            # ----- targeted move -------------------------------------------------
            # index where the convolution attains its current maximum
            k_max = int(np.argmax(conv))

            # indices that actually participate in conv[k_max]
            lo = max(0, k_max - (n - 1))
            hi = min(n - 1, k_max)
            i_candidates = np.arange(lo, hi + 1, dtype=np.int64)

            # contribution of each candidate to the maximal entry
            contrib = seq[i_candidates] * seq[k_max - i_candidates]
            i = i_candidates[np.argmax(contrib)]

            # find a partner j that contributes the least to the same entry
            # (zero if the complementary index lies outside the support)
            idx = np.arange(n, dtype=np.int64)
            complement = k_max - idx
            mask = (complement >= 0) & (complement < n)
            contrib_all = np.full(n, np.inf, dtype=np.float64)
            contrib_all[mask] = seq[mask] * seq[complement[mask]]
            # exclude i
            contrib_all[i] = np.inf
            j = int(np.argmin(contrib_all))
        else:

            # pick i biased towards larger coefficients
            i = np.random.choice(n, p=seq / seq.sum())
            # pick j uniformly, j != i
            j = np.random.randint(n - 1)
            if j >= i:

        # Decide how much mass to move; never exceed the source amount.
        src_mass = seq[i]
        if src_mass <= 0.0:

        eps = src_mass * max_factor * np.random.rand()
        if eps <= 0.0:

        # ------------------------------------------------------------------

        # ------------------------------------------------------------------

        conv_old = conv.copy()

        seq[i] = seq_old[i] + delta_i
        seq[j] = seq_old[j] + delta_j

        # ------------------------------------------------------------------

        # ------------------------------------------------------------------
        new_score = 2.0 * n * conv.max()

        # ------------------------------------------------------------------
        # Accept / reject using simulated annealing.
        # ------------------------------------------------------------------
        accept = False
        if new_score < cur_score_old - 1e-12:
            accept = True
        else:
            # occasional uphill moves
            if np.random.rand() < np.exp(-(new_score - cur_score_old) / (temp + 1e-12)):
                accept = True

        if accept:

            if new_score < best_score:
                best_score = new_score
                best_seq = seq.copy()
                # occasional progress report

                          f"time={elapsed:.1f}s best_score={best_score:.6f}")
        else:
            # revert to the previous state
            seq[:] = seq_old
            conv[:] = conv_old
            cur_score = cur_score_old

        # ------------------------------------------------------------------
        # Periodic re‑normalisation to avoid drift (and a full conv recompute).
        # ------------------------------------------------------------------

            s = seq.sum()

                seq /= s
                conv = np.convolve(seq, seq)
                cur_score = 2.0 * n * conv.max()
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = seq.copy()

        iteration += 1

    # Final sanity checks and output.
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)

        print(f"[propose_candidate] final_length={n}, final_score={final_score:.6f}")
    except Exception:
        pass
