def propose_candidate(seed=0, budget_s=1000):
    import time, numpy as np
    from scipy.optimize import differential_evolution

    np.random.seed(seed)
    deadline = time.time() + budget_s - 1.0

    n = 3428 # or could be 4000 for more points
    K = 20

    # define interpolation: index positions = np.linspace(0, n-1, K)
    positions = np.linspace(0, n-1, K)

    # objective function: given vector x (size K) produce sequence by linear interpolation and evaluate.
    # We will enforce x >= epsilon to avoid sum near zero.
    # We'll scale to have sum = 1 (or to have max <= 1000). We'll keep scaling factor: seq = x_interp.
    # Actually we should rescale after interpolation: seq = seq / np.mean(seq) or something; but scaling irrelevant.
    # But shape matters. Use scaling to ensure sum of seq = 1 for stability.

    def objective(x):
        # early exit if time out
        if time.time() > deadline:
            return 1e9
        # ensure non-negative
        seq_vals = np.maximum(0, x)
        # Interpolate
        seq = np.interp(np.arange(n), positions, seq_vals)
        # Avoid all zeros
        total = np.sum(seq)
        if total < 1e-6:
            return 1e9
        # scale to sum = 1 (optional)
        seq = seq / total
        # evaluate
        return evaluate_sequence(seq.tolist())

    Returns the best sequence found as a plain Python list.

    start = time.time()
    deadline = start + budget_s - 0.5  # safety margin

"""
Propose a step‑function (non‑negative heights) that minimises

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm consists of two phases:

1. **Global search** – a differential‑evolution (DE) optimisation on a
   piecewise‑linear parametrisation.  K break‑points (default 20) define a
   monotone‑decreasing shape; the break‑point values are interpolated to the
   full length n, normalised to unit L1‑norm (the objective is scale‑invariant)
   and fed to the supplied `evaluate_sequence`.

2. **Local refinement** – a simple random‑walk hill‑climbing that perturbs
   a few break‑points while always keeping the shape monotone.  It runs until
   the time budget expires.

Both phases respect the hard time limit (`budget_s`).  No closures are used –
all helpers are defined at module level.  The function returns a plain Python
list ready for the scorer.
"""

import time
import numpy as np
from functools import partial
from scipy.optimize import differential_evolution

# ----------------------------------------------------------------------
# Helper functions (top‑level – no closures)
# ----------------------------------------------------------------------


def _build_sequence(params: np.ndarray, n: int, positions: np.ndarray) -> np.ndarray:
    """
    Convert a raw parameter vector into a full step‑function.

    * params    – raw break‑point heights (may be unsorted)
    * n         – desired length of the sequence
    * positions – equally spaced positions of the break‑points (size K)

    The heights are forced non‑negative, sorted in descending order to obtain a
    monotone decreasing shape, linearly interpolated to length `n`,
    and finally normalised so that the sum equals 1 (scale‑invariance of the
    objective).  The returned array contains Python‑compatible floats.
    """
    # enforce non‑negativity
    heights = np.maximum(0.0, params)
    # monotone decreasing order
    heights = np.sort(heights)[::-1]
    # linear interpolation to full length
    seq = np.interp(np.arange(n, dtype=float), positions, heights)
    # normalise L1‑norm to 1 (avoids the <0.01 sum penalty)
    total = seq.sum()
    if total < 1e-12:
        # this will be penalised by evaluate_sequence, but we keep the array
        return seq
    seq = seq / total
    return seq


def _evaluate_params(params: np.ndarray, n: int, positions: np.ndarray) -> float:
    """
    Objective for the optimiser: build the sequence from `params` and evaluate
    it with the provided `evaluate_sequence`.  The return value is the scalar
    score; lower is better.
    """
    seq = _build_sequence(params, n, positions)
    # convert to a plain Python list – evaluate_sequence expects it
    return evaluate_sequence(seq.tolist())


def _de_callback(xk: np.ndarray, convergence: float, deadline: float) -> bool:
    """
    Callback for scipy's DE – stops the optimisation when the global time
    budget is exhausted.
    """
    return time.time() > deadline


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000) -> list:
    """
    Search for a low‑score step function under the given time budget.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Total wall‑clock time allowed (seconds).

    Returns
    -------
    list[float]
        Heights of the best sequence found (plain Python list).
    """
    # ------------------------------------------------------------------
    # Basic configuration
    # ------------------------------------------------------------------
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # small safety margin

    # Length of the candidate – the original construction used 3428.
    n = 3428

    # Number of break‑points for the piecewise‑linear shape.
    K = 20

    # Positions of the break‑points (equally spaced across the index range).
    positions = np.linspace(0, n - 1, K, dtype=float)

    # ------------------------------------------------------------------
    # 1) Global optimisation with Differential Evolution
    # ------------------------------------------------------------------
    # Parameter bounds – heights are unrestricted up to 10; final normalisation
    # makes the actual magnitude irrelevant.
    bounds = [(0.0, 10.0) for _ in range(K)]

    # Partially bind the evaluation function to avoid closures.
    obj = partial(_evaluate_params, n=n, positions=positions)

    # Callback that checks the remaining time.
    cb = partial(_de_callback, deadline=deadline)

    # Run DE.  The number of generations is limited so that we still have time
    # for the local refinement afterwards.
    result = differential_evolution(
        func=obj,
        bounds=bounds,
        strategy="best1bin",
        maxiter=80,
        popsize=15,
        tol=1e-7,
        mutation=(0.5, 1.0),
        recombination=0.7,
        seed=seed,
        polish=False,
        callback=cb,
        init="latinhypercube",
        updating="deferred",
        disp=False,
    )

    best_params = result.x
    best_score = result.fun

    # Occasionally DE can terminate with an infinite score (e.g. if all
    # candidates collapsed to sum≈0).  Guard against that.
    if not np.isfinite(best_score):
        best_params = np.full(K, 1.0)
        best_score = obj(best_params)

    print(f"[DE] best score after global search: {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Local random refinement (hill climbing)
    # ------------------------------------------------------------------
    # Simple stochastic walk: perturb a couple of break‑points, keep the change
    # if it improves the objective.  The loop runs until the time budget is
    # exhausted.
    while time.time() < deadline:
        # Create a copy and apply a small perturbation to 1–2 random indices.
        new_params = best_params.copy()
        for _ in range(np.random.randint(1, 3)):
            idx = np.random.randint(K)

            step = np.random.randn() * 0.2 * (abs(new_params[idx]) + 0.1)
            new_params[idx] = max(0.0, new_params[idx] + step)

        cand_score = obj(new_params)

    # ------------------------------------------------------------------

    print(f"Finished search – final best score {best_score:.6f}")
    return result_list
