"""
A simple yet effective search for a low‑score step‑function.

The candidate shapes are
    a_i = (i + γ) ** (‑p) ,  i = 0 … n‑1
where the exponent *p* (> 0.5) flattens the aperiodic autocorrelation and the
offset *γ* reduces the large contributions from the first few terms.
The theoretical bound  S(p)=2·(1‑p)²·B(1‑p,1‑p)  is below 1.5 for
p≈0.75, but only if the early terms are small enough (γ≈7 for the default
length n=3428).  The previous implementation explored γ only up to 5,
so it missed this region.

The algorithm works in two phases:

1. **Coarse grid** over (p,γ) with a generous range
   (p∈[0.55,0.90], γ∈[0.5,12]) on the default length n=3428.
2. **Random local refinement** around the best pair found,
   allowing p up to 0.95 and γ up to 30.

All evaluations use the provided ``evaluate_sequence`` function; the best
list of heights is returned.  The total running time never exceeds the
budget (default 1000 s).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper – creates a power‑law step function
# ----------------------------------------------------------------------
def make_power_law(n: int, p: float, gamma: float) -> np.ndarray:
    """Return a_i = (i + gamma) ** (-p) for i=0..n‑1 (as a NumPy array)."""
    return (np.arange(n, dtype=float) + gamma) ** (-p)


# ----------------------------------------------------------------------
# Main entry point required by the challenge
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                      budget_s: float = 1000,
                      length: int = 3428,
                      **kwargs) -> list[float]:
    """
    Search for a step‑function (non‑negative heights) that minimises

        2 * n * max(conv(a, a)) / (sum(a) ** 2)

    The search explores the (p, γ)‑plane for power‑law shapes.
    It first runs a coarse grid, then performs a random refinement.
    The function respects the time budget ``budget_s`` (seconds) and
    returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.1   # safety margin

    n = length                               # number of coefficients

    # ------------------------------------------------------------------
    # Start from a known construction if it exists (global variable).
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    best_p = best_gamma = None

    # ``height_sequence_1`` may be injected by the surrounding environment.
    try:
        hs = globals()["height_sequence_1"]
        if isinstance(hs, (list, tuple)):
            score = evaluate_sequence(list(hs))
            best_seq = np.array(hs, dtype=float)
            best_score = score
            print(f"[init] Using provided height_sequence_1, score = {score:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1) Coarse grid search on (p, gamma)
    # ------------------------------------------------------------------
    # p in [0.55, 0.90] with step ≈0.01, gamma in [0.5, 12] with step 0.5
    p_vals = np.linspace(0.55, 0.90, 36)          # 0.55 … 0.90
    gamma_vals = np.linspace(0.5, 12.0, 24)       # 0.5 … 12.0

    for p in p_vals:
        for gamma in gamma_vals:
            if time.time() > deadline:
                break

            seq = make_power_law(n, p, gamma)
            # ``evaluate_sequence`` expects a plain Python list
            score = evaluate_sequence(seq.tolist())

            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                best_p, best_gamma = p, gamma
                print(f"[grid] new best p={p:.4f}, γ={gamma:.4f}, score={score:.6f}")

        if time.time() > deadline:
            break

    # ------------------------------------------------------------------
    # 2) Random local refinement around the current best point
    # ------------------------------------------------------------------
    if best_seq is None:   # safety fallback – should never happen
        best_p, best_gamma = 0.75, 7.0
        best_seq = make_power_law(n, best_p, best_gamma)
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"[fallback] start from p={best_p}, γ={best_gamma}")

    # number of refinement attempts – capped by remaining time
    refinement_iters = 15000
    for _ in range(refinement_iters):
        if time.time() > deadline:
            break

        # Gaussian perturbation (tuned empirically)
        p_new = best_p + np.random.randn() * 0.015
        gamma_new = best_gamma + np.random.randn() * 0.7

        # keep parameters inside a sensible domain
        if not (0.50 < p_new < 0.95 and 0.1 < gamma_new < 30.0):
            continue

        seq = make_power_law(n, p_new, gamma_new)
        score = evaluate_sequence(seq.tolist())

        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            best_p, best_gamma = p_new, gamma_new
            print(f"[refine] improved p={p_new:.4f}, γ={gamma_new:.4f}, score={score:.6f}")

    # ------------------------------------------------------------------
    # 3) Very short deterministic fine‑tuning: tweak p only (γ fixed)
    # ------------------------------------------------------------------
    # Sometimes a tiny shift of p can still improve the score.
    for _ in range(2000):

        p_new = best_p + np.random.uniform(-0.005, 0.005)

        seq = make_power_law(n, p_new, best_gamma)

            best_seq = seq.copy()
            best_p = p_new
            print(f"[fine] p={p_new:.4f}, γ={best_gamma:.4f}, score={score:.6f}")

    # ------------------------------------------------------------------
    # Final result (plain Python list)
    # ------------------------------------------------------------------
    result = best_seq.tolist()
    print(f"Finished search – best score {best_score:.6f} "
          f"with p={best_p:.5f}, γ={best_gamma:.5f}, length={n}")
    return result
