    import time, numpy as np
    rng = np.random.default_rng(seed)

    # Determine length n
    try:
        n = len(height_sequence_1)
    except Exception:
        n = 2000

    # base sequence
    base_seq = generate_inverse_sqrt_sequence(n)
    
    # ensure within bounds 0-1000 -> values are <= 1.
    f = np.array(base_seq, dtype=np.float64)
    best_seq = f.copy()
    best_score = evaluate_sequence(best_seq.tolist())

    start = time.time()
    deadline = start + budget_s - 0.5  # margin

    # Try LP improvement if scipy available
    try:
        import scipy.optimize as opt
        import scipy.sparse as sp
        have_scipy = True
    except Exception as e:
        have_scipy = False
        print("scipy not available, falling back to simple local search")

    if have_scipy:
        t = 0.5   # mixing factor
        max_iter = 8
        iter_num = 0
        while time.time() < deadline and iter_num < max_iter:
            iter_num += 1
            # compute current max convolution
            conv_f = np.convolve(f, f)
            M = conv_f.max()
            # Build convolution matrix A (size (2n-1) x n)
            rows = []
            cols = []
            data_vals = []
            for k in range(2*n - 1):
                start_j = max(0, k - (n - 1))
                end_j = min(k, n - 1)
                cols_row = np.arange(start_j, end_j + 1, dtype=np.int32)
                # f indices reversed slice
                f_vals = f[start_j: end_j + 1][::-1]   # length = len(cols_row)
                rows.extend([k] * len(cols_row))
                cols.extend(cols_row.tolist())
                data_vals.extend(f_vals.tolist())
            # Convert to csr
            A = sp.csr_matrix((np.array(data_vals, dtype=np.float64),
                               (np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32))),
                               shape=(2*n - 1, n))
            # LP: maximize sum(g) => minimize -sum(g)
            c = -np.ones(n, dtype=np.float64)
            b = np.full(2*n - 1, M, dtype=np.float64)
            res = opt.linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n,
                              method='highs', options={'presolve': True, 'time_limit': min(60.0, deadline - time.time())})
            if not res.success:
                print(f"[LP] iteration {iter_num}: LP failed, status {res.message}")
                break
            g = res.x
            sum_f = f.sum()
            sum_g = g.sum()
            if sum_g == 0:
                print("[LP] sum_g zero, break")
                break
            # Scale g to same L1 as f

            # If improvement:

                best_score = new_score
                best_seq = f_new.copy()

            f = f_new  # continue

    # After LP steps, do cheap random local search for remaining time
    deadline = start + budget_s - 0.2
    while time.time() < deadline:
        i = rng.integers(n)

"""
Propose a candidate step‑function sequence using an iterative LP‑based
refinement (the method described in the statement) followed by a cheap
random local search.

1.  Start with the “inverse‑square‑root’’ sequence a_i = C(2i,i)/4^i,
    which gives a score ≈ π/2 ≈ 1.571.
2.  Repeatedly:
        • compute M = max((a * a))                     (discrete convolution)
        • solve the linear program
              maximise   Σ g_j
              s.t.       (a ⊛ g)_k ≤ M   for all k
                         g_j ≥ 0
          (the convolution is linear in g, so this is a genuine LP)
        • rescale g so that Σ g = Σ a and replace a by the convex
          combination a ← (1‑t)·a + t·g   (t = 0.5).  This never
          increases the maximum convolution value, thus strictly
          reduces the evaluation score.
        • keep the best sequence seen.
   Only a few LP iterations are needed; each LP uses a sparse
   Toeplitz matrix of size (2n‑1)×n.
3.  When the time budget is almost exhausted perform a cheap
   multiplicative perturbation of single entries (the same idea as the
   baseline) to extract any remaining improvements.

The algorithm is scale‑invariant, works only with non‑negative numbers,
and respects the security checks of ``evaluate_sequence``.
"""

import time
import numpy as np

def generate_inverse_sqrt_sequence(n: int) -> list[float]:
    """Return a_i = binom(2*i,i) / 4**i using the recurrence
    a_{i+1}=a_i*(2*i+1)/(2*i+2)."""
    seq = [1.0]
    for i in range(1, n):
        seq.append(seq[-1] * (2 * i - 1) / (2 * i))
    return seq


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):

    # ----------------------------------------------------------------------
    # Determine a reasonable length

    try:
        n = len(height_sequence_1)          # type: ignore[name-defined]
    except Exception:

    # ----------------------------------------------------------------------
    # Base sequence: inverse‑square‑root shape
    # ----------------------------------------------------------------------
    base = np.array(generate_inverse_sqrt_sequence(n), dtype=np.float64)
    # ``evaluate_sequence`` already clips to the allowed range [0,1000]
    best_seq = base.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] n={n}, score={best_score:.6f}")

    start_time = time.time()
    deadline = start_time + budget_s - 0.5               # tiny safety margin

    # ----------------------------------------------------------------------
    # Try the LP‑based refinement if SciPy is available
    # ----------------------------------------------------------------------
    try:
        import scipy.optimize as opt
        import scipy.sparse as sp
        have_scipy = True
    except Exception:

        print("[info] SciPy not available – falling back to simple local search.")

    if have_scipy:
        # mixing weight (0 < t ≤ 1); 0.5 gives rapid reduction while keeping
        # the convex‑combination guarantee.
        t = 0.5
        max_lp_iters = 8               # keep the total run time well below the limit
        lp_iter = 0

        while time.time() < deadline and lp_iter < max_lp_iters:
            lp_iter += 1

            # ----- current maximal convolution value ----------
            conv_a = np.convolve(best_seq, best_seq)
            M = conv_a.max()

            # ----- build the (2n‑1)×n convolution matrix A such that A @ g = a ⊛ g ----------
            # The matrix is Toeplitz: A[k, j] = a_{k‑j} if 0≤k‑j<n else 0.
            # Build it row‑wise as a CSR matrix (≈ n² non‑zeros).
            rows = []
            cols = []
            data = []

            c = -np.ones(n, dtype=np.float64)               # minimise –sum(g)
