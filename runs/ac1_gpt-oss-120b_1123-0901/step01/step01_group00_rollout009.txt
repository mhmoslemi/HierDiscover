def full_convolve(a, b):
    size = len(a) + len(b) - 1
    # next power of two for efficiency
    nfft = 1 << (size-1).bit_length()
    Fa = np.fft.rfft(a, n=nfft)
    Fb = np.fft.rfft(b, n=nfft)
    conv = np.fft.irfft(Fa * Fb, n=nfft)[:size]
    return conv

"""
Gradient‑descent on a smooth approximation of the max‑convolution.
The objective for a length‑n non‑negative vector a with Σa = √(2n) is

    score = 2·n·max(conv(a,a)) / (Σa)² = 2·n·max(conv(p,p)),

where p = a / √(2n) is a probability vector (Σp = 1).  
We minimize  max(conv(p,p))  by projected gradient descent on the
simplex.  The non‑smooth max is replaced by a soft‑max with a temperature
β that is increased during the run (continuation).  Each gradient step is
projected back onto the simplex {x≥0, Σx=1}.  Several random restarts of
different lengths are performed inside the allowed time budget; the
best sequence found is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Wang & Carreira‑Perpiñán 2013)
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto {x_i ≥ 0, Σx_i = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast full convolution using FFT (real inputs)
# ----------------------------------------------------------------------
def _full_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Return a * b (full linear convolution) using FFT."""
    size = a.shape[0] + b.shape[0] - 1
    # next power of two for efficient FFT
    nfft = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, n=nfft)
    B = np.fft.rfft(b, n=nfft)
    conv = np.fft.irfft(A * B, n=nfft)[:size]
    return conv


# ----------------------------------------------------------------------
# Gradient of the β‑soft‑max of the convolution
# ----------------------------------------------------------------------
def _gradient_softmax(p: np.ndarray, beta: float):
    """
    Compute the gradient of  f(p)= (1/β)·log Σ_k exp(β·conv_k(p))
    with respect to p.
    Returns (grad, conv) where conv = p*p (full) and grad has shape (len(p),).
    """
    conv = _full_convolve(p, p)                     # length 2n‑1
    # soft‑max weights (stable)
    scaled = beta * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    w = exp_scaled / np.sum(exp_scaled)            # length 2n‑1, Σw = 1

    # grad_i = Σ_k w_k (p_{k‑i} + p_{i‑k})
    conv1 = _full_convolve(w, p)                   # length 3n‑2
    conv2 = _full_convolve(w[::-1], p)             # length 3n‑2
    L = conv.shape[0]                               # = 2n‑1
    n = p.shape[0]
    idx = np.arange(n)
    grad = conv1[idx] + conv2[L - 1 - idx]
    return grad, conv


# ----------------------------------------------------------------------
# Main entry point required by the competition framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) yields a low
    evaluation score.  The routine runs for at most ``budget_s`` seconds
    and keeps the best sequence discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Helper that runs a single optimisation for a given length `n`
    # ------------------------------------------------------------------
    def _run_one(n: int, p_init: np.ndarray):
        nonlocal best_seq, best_score

        # ensure we start on the simplex Σp = 1
        p = _project_onto_simplex(p_init.astype(np.float64), 1.0)

        # hyper‑parameters (tuned empirically)
        beta_start = 1.0
        beta_end = 150.0                    # final temperature (more peaked)
        max_iters = 1200
        base_eta = 0.15                     # base step size

        for it in range(max_iters):
            now = time.time()
            if now >= deadline:
                break

            # continuation on β
            beta = beta_start + (beta_end - beta_start) * (it / max_iters)

            grad, conv = _gradient_softmax(p, beta)

            # adaptive step – scale by current max‑convolution and max‑gradient
            max_conv = np.max(conv)
            max_grad = np.max(np.abs(grad)) + 1e-12
            eta = base_eta * max_conv / max_grad

            # gradient step and projection back onto the simplex
            p_new = p - eta * grad
            p_new = _project_onto_simplex(p_new, 1.0)

            # cheap objective (no call to evaluate_sequence)
            conv_new = _full_convolve(p_new, p_new)
            score_new = 2.0 * n * np.max(conv_new)   # identical to evaluate_sequence’s result

            if score_new < best_score:
                best_score = score_new
                best_seq = (p_new * np.sqrt(2.0 * n)).copy()
                print(f"[{now:.2f}] New global best = {best_score:.6f} (n={n})")

            # simple acceptance – always move forward (helps exploration)
            p = p_new

            # occasional random perturbation to escape plateaus
            if np.random.rand() < 0.01:
                noise = np.random.randn(n) * (0.005 / n)
                p = _project_onto_simplex(p + noise, 1.0)
                base_eta *= 0.9   # slowly shrink step after a shake‑up

    # ------------------------------------------------------------------
    # Main optimisation loop: many random restarts of different sizes
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # pick a length – shorter runs for very large n keep each iteration cheap
        n = np.random.randint(350, 1500)

        # initialise either from the existing construction or randomly
        if np.random.rand() < 0.25 and "height_sequence_1" in globals():
            src = np.asarray(height_sequence_1, dtype=float)
            if src.size >= n:
                init = src[:n].copy()
            else:
                pad = np.random.rand(n - src.size) * 1e-4
                init = np.concatenate([src, pad])
            # convert to probability vector
            p0 = init / np.sqrt(2.0 * n)
            p0 = np.maximum(p0, 0.0)
            if p0.sum() > 0:
                p0 = p0 / p0.sum()
            else:
                p0 = np.full(n, 1.0 / n)
        else:
            # Dirichlet‑like random start
            p0 = np.random.rand(n)
            p0 = p0 / p0.sum()

        _run_one(n, p0)

        # early stop if we already beat the target
        if best_score < 1.503:
            break

    # ------------------------------------------------------------------
    # Final refinement of the best candidate (if any time remains)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        n_best = len(best_seq)
        p_best = np.asarray(best_seq, dtype=float) / np.sqrt(2.0 * n_best)

        # a longer, more aggressive run on the winning length
        _run_one(n_best, p_best)

    # ------------------------------------------------------------------
    # Fallback – return a safe uniform sequence if everything failed
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 800
        p = np.full(n, 1.0 / n)
        best_seq = (p * np.sqrt(2.0 * n)).tolist()
        best_score = 2.0 * n * np.max(_full_convolve(p, p))

    return best_seq.tolist()
