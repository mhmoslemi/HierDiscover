        # --------------------------------------------------------------
        # 3) compute current peak of the self‑convolution
        conv_ff = np.convolve(f, f)
        M = conv_ff.max()

        # --------------------------------------------------------------
        # 4) solve the LP: maximise Σ g_j  s.t.   f * g ≤ M
        A = _build_convolution_matrix(f)
        b = np.full(2 * n - 1, M, dtype=np.float64)
        c = -np.ones(n, dtype=np.float64)               # linprog minimises – we want max Σg

        # solve with HiGHS (fast, supports sparse matrices)
        res = linprog(c,
                      A_ub=A,
                      b_ub=b,
                      bounds=(0, None),
                      method='highs',
                      options={'presolve': True})
        if not res.success:
            print(f"[Iter {it}] LP failed: {res.message}")
            break

        g = res.x
        if g.sum() == 0.0:
            print(f"[Iter {it}] LP returned zero vector – terminating.")
            break
        g = g / g.sum()                     # re‑normalise to sum = 1

        # --------------------------------------------------------------
        # 5) mixture step – find a t that actually lowers the peak
        improved = False
        # try a few decreasing step sizes; the first one that reduces M is kept
        for t_try in (t, t * 0.5, t * 0.2, 0.05, 0.02):
            f_new = (1.0 - t_try) * f + t_try * g
            conv_new = np.convolve(f_new, f_new)
            if conv_new.max() < M - 1e-12:
                f = f_new
                M = conv_new.max()
                improved = True
                break
        if not improved:
            # very small step – should never increase the objective
            f = (1.0 - 0.02) * f + 0.02 * g
            M = np.convolve(f, f).max()

        # --------------------------------------------------------------
        # 6) evaluate the true objective; keep the best sequence
        score = evaluate_sequence(f.tolist())
        if score < best_score:
            best_score = score
            best_seq = f.copy()
            print(f"[Iter {it:04d}] new best score = {best_score:.6f}")

        # --------------------------------------------------------------
        # 7) update mixture coefficient and iteration counter
        t = max(0.02, t * decay)          # never let t go below 0.02
        it += 1

        # early exit if we have already crossed the required threshold
        if best_score < 1.503:
            print(f"Target reached (score={best_score:.6f}) after {it} iterations.")
            break

    print(f"Finished – best objective {best_score:.6f} (iterations={it})")
    # convert to plain list as required by the evaluator
    return best_seq.tolist()
