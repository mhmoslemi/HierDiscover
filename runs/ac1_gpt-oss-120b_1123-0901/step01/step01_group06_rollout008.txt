```
def project_simplex(v):
    # projection onto probability simplex (sum=1, v_i >= 0)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

#!/usr/bin/env python
"""
propose_candidate
=================

A two‑phase optimiser for the step‑function bound.

Phase 1 – LP improvement
------------------------
Start from the classic arcsine sequence (coefficients of (1‑z)^‑½) normalised
to have ℓ₁‑norm √(2 n).  For such a sequence the evaluation function
``evaluate_sequence`` equals the maximal value of the self‑convolution.
We then solve a linear programme

    maximise   Σ b_j
    subject to   (f * b)[k] ≤ M   for all k,
                 b_j ≥ 0,

where ``f`` is the current sequence and ``M`` its current maximal
convolution value.  The LP finds a vector ``b`` that can carry more total
mass while respecting the same convolution bound.  Rescaling ``b`` to the
same ℓ₁‑norm and moving a small fraction towards it (convex combination)
strictly reduces the bound.  Repeating a few times quickly drives the
objective below the π/2 ≈ 1.571 limit.

Phase 2 – focused hill‑climbing
-------------------------------
After the LP phase we perform a cheap stochastic refinement.  In each
iteration we locate a convolution entry attaining the current maximum,
identify the index contributing most to it and the index contributing
least, and shift a tiny amount of mass from the former to the latter.
If the move improves the score we keep it; otherwise we occasionally
accept worsening moves (simulated‑annealing style) to escape plateaus.
The whole search runs well within the 1000 s budget and typically reaches
values ≲ 1.50 for sequences of a few thousand coefficients.

The final returned sequence is normalised to sum = 1, which is accepted
by ``evaluate_sequence``.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: arcsine (C(2i,i)/4^i) coefficients
# ----------------------------------------------------------------------
def _arcsine_coeffs(n: int) -> np.ndarray:
    """First n terms of (1‑z)^(-1/2) = Σ C(2i,i)/4^i·z^i."""
    a = np.empty(n, dtype=float)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * (i - 1) + 1) / (2.0 * i)
    return a


# ----------------------------------------------------------------------
# Helper: next power of two (for FFT padding)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


# ----------------------------------------------------------------------
# Helper: max self‑convolution (via FFT, returns only the maximum)
# ----------------------------------------------------------------------
def _max_convolution(seq: np.ndarray) -> float:
    """Return max_k (seq * seq)[k] using an FFT‑based convolution."""
    n = len(seq)
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return float(conv.max())


# ----------------------------------------------------------------------
# Helper: build the linear constraints matrix A for conv(f, b) ≤ M
# ----------------------------------------------------------------------
def _build_conv_constraints(f: np.ndarray) -> np.ndarray:
    """
    Construct matrix A (shape (2n‑1, n)) such that
        (A @ b)[k] = (f * b)[k]   for every k.
    """
    n = len(f)
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=float)
    # Fill row‑wise – each row corresponds to a specific output index k.
    for k in range(rows):
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        # coefficient for variable b_j (j = idx) is f_{k-j}
        A[k, idx] = f[k - idx]
    return A


# ----------------------------------------------------------------------
# Helper: project a vector onto the probability simplex (sum = 1, ≥0)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = 1}."""
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # numerical safety – fall back to uniform distribution
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def propose_candidate(budget_s: float = 1000.0,
                      seed: int = 42,
                      **kwargs):
    """
    Produce a candidate sequence of non‑negative heights.

    Parameters
    ----------
    budget_s : float, optional
        Time budget in seconds.  The algorithm uses the full budget.
    seed : int, optional
        Random seed for reproducibility.
    **kwargs
        Ignored; present for API compatibility.

    Returns
    -------
    list[float]
        Normalised sequence (sum == 1.0) that (hopefully) minimises the
        evaluation function.
    """
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # Choose a length.  Prefer the length of a previously found solution
    # if available; otherwise pick a sensible default.
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)   # pylint: disable=undefined-variable
    except Exception:
        n = 2000

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    coeffs = _arcsine_coeffs(n)
    # Normalise to ℓ₁‑norm sqrt(2 n) as described in the reference method.
    S = np.sqrt(2.0 * n)
    f = coeffs / coeffs.sum() * S   # current sequence (sum = S)

    # Current maximum of self‑convolution (this is also the evaluation
    # value for a sequence of ℓ₁‑norm S).
    M = _max_convolution(f)

    best_M = M

    try:

    except Exception:               # pragma: no cover

    start_time = time.time()

            # Build the linear constraints A·b ≤ M (b ≥ 0)
            A = _build_conv_constraints(f)

            # Linear programme: maximise Σb   ⇔   minimise −Σb
            c = -np.ones(n, dtype=float)
            b_ub = np.full(2 * n - 1, M, dtype=float)
            bounds = [(0, None)] * n

            res = linprog(c, A_ub=A, b_ub=b_ub,
                          bounds=bounds, method='highs')
            if not res.success:

                break

            sum_b = b.sum()

            M = M_new
            t *= 0.9                   # slowly shrink the step size

    # Phase 2 – focused stochastic hill‑climbing refinement.
    # ------------------------------------------------------------------
    # Convert the best ℓ₁‑norm‑S vector to a probability vector (sum = 1)
    a = best_f / best_f.sum()
    best_score = evaluate_sequence(a.tolist())   # pylint: disable=undefined-variable
    best_seq = a.copy()

    max_local_iters = 20000
    local_start = time.time()
    for it in range(max_local_iters):
        if time.time() - local_start > budget_s - 0.5:
            break

        max_idxs = np.where(conv == cur_max)[0]

        # Find the index that contributes most to this peak and the index
        # that contributes least (the latter is simply the smallest component).
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
