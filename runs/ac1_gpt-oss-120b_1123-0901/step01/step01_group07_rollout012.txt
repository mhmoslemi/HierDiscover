    import time, numpy as np
    from scipy.optimize import linprog
    from scipy.sparse import diags

    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.1

    n = 3428
    # initial shape from power law
    p0, gamma0 = 0.5, 0.3
    f = (np.arange(n)+gamma0) ** (-p0)
    f = f.astype(float)
    # maybe also apply clipping to [0,1000] not needed
    best_f = f
    best_score = evaluate_sequence(f.tolist())
    print(f"Initial score: {best_score}")

    iteration = 0
    while time.time() < deadline:
        iteration += 1
        # compute M
        conv_ff = np.convolve(f, f)
        M = conv_ff.max()
        # solve linear program
        A = diags([np.full(n, fi) for fi in f], offsets=np.arange(n), shape=(2*n-1, n), format='csr')
        b_ub = np.full(2*n-1, M)
        c = -np.ones(n)   # maximize sum -> minimize -sum
        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n, method='highs', options={'presolve': True})
        except Exception as e:
            print(f"LP failed: {e}")
            break
        if not res.success:
            print(f"LP not successful: {res.message}")
            break
        g = res.x
        sum_f = f.sum()
        sum_g = g.sum()
        # if sum not improved, break
        if sum_g <= sum_f * (1 + 1e-8):
            print("No sum improvement, terminating")
            break
        # compute M_g
        M_g = np.convolve(g, g).max()
        # try several alphas
        alphas = [0.1, 0.25, 0.5, 0.75]
        improved = False
        for alpha in alphas:
            if time.time() > deadline: break
            cand = (1 - alpha) * f + alpha * g
            cand_score = evaluate_sequence(cand.tolist())
            if cand_score < best_score:
                best_score = cand_score
                best_f = cand.copy()
                print(f"[iter {iteration}] alpha {alpha:.2f} improved score {cand_score:.6f}")
                improved = True
        if improved:
            f = best_f.copy()
        else:
            # no improvement from blending; maybe set f = g ?
            # compute candidate using g alone
            cand_score = evaluate_sequence(g.tolist())
            if cand_score < best_score:
                best_score = cand_score
                best_f = g.copy()
                f = best_f.copy()
                print(f"[iter {iteration}] direct g improved score {cand_score:.6f}")
            else:
                # reduce step size maybe stop
                print(f"[iter {iteration}] No improvement, exit")
                break

    result = best_f.tolist()
    print(f"Finished iterative search – best score {best_score}")
    return result

"""
Propose a step‑function using a Frank‑Wolfe style iteration.

The evaluation metric   score = 2 * n * max(conv(a,a)) / (sum(a)^2)
is scale‑invariant: multiplying a by a constant does not change the score.
Hence we can try to increase the total mass Σa while keeping the peak of the
convolution (the “peak‑correlation” M) as low as possible.

Algorithm
---------
1. Start from a good analytic shape (here a power‑law with p≈0.5).  
   The shape already yields a score close to the continuous bound
   (≈ π/2 ≈ 1.57).

2. At each iteration compute M = max(conv(f,f)).  Solve the linear programme

        maximise   Σ b_j
        subject to conv(f,b) ≤ M      (element‑wise)
                   b_j ≥ 0

   The constraints are linear in the unknown vector `b`.  The LP therefore
   finds the sequence `g` that adds as much mass as possible without ever
   increasing the peak‑correlation with the current sequence `f`.

3. Blend the new direction `g` with the current candidate:
        f←(1−α) f + α g
   for several small values of α (0.05 … 0.4).  The blend keeps the peak
   correlation roughly bounded (the increase is O(α²)) while the denominator
   Σ f grows linearly with α, thus the score tends to drop.

4. Keep the best sequence seen so far and repeat until the time budget
   expires.

The linear programme is solved with SciPy’s high‑performance “highs”
interior‑point method.  The convolution matrix A (size (2n‑1) × n) is very
structured – each diagonal offset i contains the constant value f[i] –
and is built as a sparse `csc_matrix` with `scipy.sparse.diags`.

All numbers are forced into the allowed interval [0, 1000] before they are
handed to `evaluate_sequence`.  The function respects the 1000 s budget
and prints progress information.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import diags


def _make_powerlaw(n: int, p: float, gamma: float) -> np.ndarray:
    """Return the power‑law a_i = (i+γ)^{‑p} for i=0..n‑1."""
    return (np.arange(n, dtype=float) + gamma) ** (-p)


def _conv_matrix(f: np.ndarray) -> "scipy.sparse.csc_matrix":
    """
    Build the matrix A such that (A @ b)[k] = (conv(f, b))[k]  (full convolution).
    A has shape (2*len(f)-1, len(f)) and consists of n constant diagonals.
    """
    n = f.size
    # each diagonal i (offset i) is filled with the constant f[i]
    data = [np.full(n, fi, dtype=float) for fi in f]
    offsets = np.arange(n, dtype=int)                # 0 … n‑1
    return diags(data, offsets, shape=(2 * n - 1, n), format="csc")


def _solve_direction(f: np.ndarray, M: float, time_limit: float) -> np.ndarray | None:
    """
    Solve  maximise Σ b_j   subject to   conv(f, b) ≤ M  and b ≥ 0.
    Returns the optimal vector `b` (or None on failure).
    """
    n = f.size
    A = _conv_matrix(f)                     # (2n‑1) × n
    b_ub = np.full(2 * n - 1, M, dtype=float)
    bounds = [(0.0, None)] * n               # b_j ≥ 0
    c = -np.ones(n, dtype=float)            # maximise Σ b  ⇔ minimise –Σ b

    # linprog “highs” automatically uses a sparse matrix when A is sparse.
    # The solver respects a time limit via the `options['time_limit']`
    # (available in recent SciPy versions).  If the current SciPy does not
    # support it we simply let the solver run – the outer loop checks the
    # global deadline anyway.
    try:
        res = linprog(
            c,
            A_ub=A,
            b_ub=b_ub,
            bounds=bounds,
            method="highs",
            options={"presolve": True, "time_limit": max(0.0, time_limit)},
        )
    except Exception as exc:  # pragma: no cover – safeguard for unexpected errors
        print(f"LP exception: {exc}")
        return None

    if not res.success:
        print(f"LP failed: {res.message}")
        return None
    return res.x


def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative height sequence that minimises the evaluation score.
    Returns the best sequence found within the time budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.1   # tiny safety margin

    # ------------------------------------------------------------------
    # 0) Choose a length and an initial sequence.
    # ------------------------------------------------------------------
    n = 3428                                # length used by the reference solution

    # If the runner provides a previously found sequence, start from it.
    if "height_sequence_1" in globals():
        init = np.asarray(height_sequence_1, dtype=float)
        if init.size >= n:
            f = init[:n].copy()
        else:                               # pad with zeros if too short
            f = np.concatenate([init, np.zeros(n - init.size, dtype=float)])
    else:
        # power‑law close to the analytic bound (p≈0.5, small γ)
        f = _make_powerlaw(n, p=0.5002, gamma=0.3210)

    # safety clipping (the scorer also clips, but we keep the numbers modest)
    f = np.clip(f, 0.0, 1000.0)

    best_seq = f.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 1) Frank‑Wolfe style iterations.
    # ------------------------------------------------------------------
    iteration = 0
    # a few blending factors to try each round
    alphas = [0.05, 0.10, 0.20, 0.40]

    while time.time() < deadline:
        iteration += 1
        # current peak of the autocorrelation
        conv_ff = np.convolve(f, f)
        M = conv_ff.max()
        sum_f = f.sum()

        # --------------------------------------------------------------
        # 1a) Solve the LP: maximise total mass while keeping
        #     the cross‑convolution with the current f bounded by M.
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        g = _solve_direction(f, M, time_limit=remaining)
        if g is None:

        sum_g = g.sum()
        print(

            f"M={M:.3e}"
        )

        # If the new direction does not increase the L1‑mass, we are at a
        # (local) optimum for the current peak constraint.
        if sum_g <= sum_f * (1.0 + 1e-12):
            print("[stop] No increase in total mass – terminating.")
            break

        # --------------------------------------------------------------
        # 1b) Test a few convex combinations f←(1‑α)f+αg.
        # --------------------------------------------------------------
        improved = False
        for alpha in alphas:
            if time.time() > deadline:
                break
            cand = (1.0 - alpha) * f + alpha * g
            cand = np.clip(cand, 0.0, 1000.0)   # respect hard caps
            cand_score = evaluate_sequence(cand.tolist())
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                f = cand.copy()               # continue from the improved point
                improved = True
                print(
                    f"[iter {iteration}] α={alpha:.2f} → new best score {cand_score:.6f}"
                )
                break   # accept the first improvement (larger α later would only increase M)

        if improved:
            continue

        # --------------------------------------------------------------
        # 1c) If blending did not improve, try the direction itself.
        # --------------------------------------------------------------
        cand_score = evaluate_sequence(g.tolist())
        if cand_score < best_score:
            best_score = cand_score
            best_seq = g.copy()
            f = g.copy()
            print(f"[iter {iteration}] direct g improved score {cand_score:.6f}")
            continue

        # --------------------------------------------------------------
        # 1d) No improvement found – stop the loop.
        # --------------------------------------------------------------
        print("[stop] No improvement in this iteration – terminating.")
        break

    # ------------------------------------------------------------------
    # 2) Return the best sequence discovered.
    # ------------------------------------------------------------------
    print(f"Finished search after {iteration} iteration(s); best score {best_score:.6f}")
    # Convert to plain Python list as required by the external scorer.
    return best_seq.tolist()
