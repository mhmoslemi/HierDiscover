"""
Propose a non‑negative step sequence that minimizes
    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm combines three ideas:

1. **Simplex projection** – every candidate is projected onto
   the simplex {x_i ≥ 0, Σ x_i = √(2 n)} so that the scale‑invariant
   objective can be evaluated as `max(conv(a, a))`.

2. **LP “improvement” step** – for a fixed sequence f the
   linear program

       maximise   Σ g_i
       subject to g_i ≥ 0
                  (f * g)_k ≤ M   for all k

   where M = max(conv(f, f)),
   yields a direction g that cannot increase the maximal convolution.
   The solution g is rescaled to satisfy the simplex constraint and the
   current point is moved a small convex combination
   `f ← (1‑t) f + t g`.  Repeating this step drives the maximal
   convolution down.

3. **Fallback stochastic gradient descent** – if cvxpy is not available or
   the LP fails, a cheap projected gradient method on the few largest
   convolution entries is applied.

Multiple random restarts with different lengths (≈ 800‑2000) are performed
within the time budget (≈ 1000 s).  The best sequence found overall is
returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project a vector onto the simplex {x_i ≥ 0, Σ x_i = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Fast objective (no clipping because we stay inside the safe region)
def _objective(a: np.ndarray) -> float:
    """Return 2 * n * max(conv(a,a)) / (sum(a)**2)."""
    n = a.shape[0]
    sum_a = a.sum()
    if sum_a <= 0:
        return float("inf")
    maxb = np.max(np.convolve(a, a))
    return 2.0 * n * maxb / (sum_a * sum_a)

# ----------------------------------------------------------------------
# Gradient based helper (used when the LP route is unavailable)
def _gradient_multi(a: np.ndarray, top_k: int = 5) -> np.ndarray:
    """Gradient of the sum of the `top_k` largest convolution entries."""
    n = a.shape[0]
    conv = np.convolve(a, a)
    # indices of the largest entries
    if top_k >= conv.size:
        idxs = np.arange(conv.size)
    else:
        idxs = np.argpartition(conv, -top_k)[-top_k:]
    grad = np.zeros_like(a)
    for k in idxs:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=int)
        partner = k - idx
        grad[idx] += 2.0 * a[partner]
    return grad

# ----------------------------------------------------------------------
# LP improvement using cvxpy (if available)
def _lp_improve(f: np.ndarray, t: float, max_iter: int = 12):
    """
    Perform up to `max_iter` LP improvement steps on vector `f`.
    Returns the improved vector (or the original one if the LP fails).
    """
    try:
        import cvxpy as cp
    except Exception:
        return f  # cvxpy not available – caller will fall back

    n = f.shape[0]
    target_sum = math.sqrt(2.0 * n)
    a = f.copy()

    for _ in range(max_iter):
        M = np.max(np.convolve(a, a))
        # build and solve the LP: maximise sum(g) subject to g>=0 and conv(f,g) ≤ M
        g = cp.Variable(n, nonneg=True)
        # cvxpy's convolve works with constant first argument and variable second
        constraints = [cp.convolve(a, g) <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, verbose=False, max_iters=1000)
        except Exception:
            break
        if g.value is None:
            break
        g_val = np.asarray(g.value).flatten()
        sum_g = g_val.sum()
        if sum_g <= 0:
            break
        # rescale to meet the simplex radius
        g_val = g_val * (target_sum / sum_g)
        # convex combination
        a = (1.0 - t) * a + t * g_val
    return a

        r = np.random.rand()
        if r < 0.15 and best_seq is not None and best_seq.shape[0] == n:

        elif r < 0.30 and 'height_sequence_1' in globals():

            else:

        elif r < 0.55:

            lam = np.random.rand() * 5.0
            i = np.arange(n, dtype=float)
            a = np.exp(-lam * i)
            a = _project_onto_simplex(a, target_sum)
        else:

            a = _project_onto_simplex(a, target_sum)

        cur_score = _objective(a)

        # --------------------------------------------------------------------
        use_lp = False

            use_lp = True
        except Exception:
            use_lp = False

            src = np.asarray(height_sequence_1, dtype=float)
            if src.size >= n:
                a = src[:n].copy()
            else:
                pad = np.random.rand(n - src.size) * 1e-4
                a = np.concatenate([src, pad])
            a = _project_onto_simplex(a, target_sum)
        elif r < 0.55:
            # exponential decay shape – λ drawn uniformly
            lam = np.random.rand() * 5.0
            i = np.arange(n, dtype=float)
            a = np.exp(-lam * i)
            a = _project_onto_simplex(a, target_sum)
        else:
            a = np.random.rand(n)
            a = _project_onto_simplex(a, target_sum)

        cur_score = _objective(a)
        _maybe_update(a, cur_score)

        # ----------------------------------------------------------
        # Try the LP improvement (if cvxpy is present)
        # ----------------------------------------------------------
        use_lp = False
        try:
            import cvxpy  # noqa: F401
            use_lp = True
        except Exception:
            use_lp = False

        if use_lp:
            t = 0.25
            for _ in range(8):
                if time.time() >= deadline:
                    break
                a_new = _lp_improve(a, t, max_iter=1)  # one LP step per outer loop
                if a_new is None:
                    break
                a = a_new
                cur_score = _objective(a)
                _maybe_update(a, cur_score)
                t *= 0.85                     # shrink step size gradually
        else:
            # ----------------------------------------------------------
            # Fallback stochastic projected gradient descent
            # ----------------------------------------------------------
            step_factor = target_sum * 0.08 / n         # typical move magnitude
            top_k = 5
            for it in range(250):
                if time.time() >= deadline:
                    break

                grad = _gradient_multi(a, top_k=top_k)
                gnorm = np.linalg.norm(grad)
                if gnorm > 0:
                    maxb = np.max(np.convolve(a, a))
                    eta = step_factor * maxb / (gnorm + 1e-12)
                    a_candidate = a - eta * grad
                else:
                    a_candidate = a + np.random.randn(n) * step_factor

                a_candidate = _project_onto_simplex(np.maximum(a_candidate, 0.0), target_sum)

                # occasional random shake to escape flat regions
                if np.random.rand() < 0.03:
                    noise = np.random.randn(n) * step_factor * 0.3
                    a_candidate = _project_onto_simplex(np.maximum(a_candidate + noise, 0.0), target_sum)

                cand_score = _objective(a_candidate)

                # simple Metropolis acceptance; temperature cools linearly
                elapsed = time.time() - start_time
                frac = elapsed / budget_s
                T = 0.02 * max(0.0, 1.0 - frac) + 1e-8
                if cand_score < cur_score or np.random.rand() < math.exp(-(cand_score - cur_score) / T):
                    a = a_candidate
                    cur_score = cand_score
                    _maybe_update(a, cur_score)

                step_factor *= 0.998
                if it % 50 == 0 and top_k < 12:
                    top_k += 1

        # early exit if we already beat the target
        if best_score < 1.503:
            break

    # ----------------------------------------------------------------------
    # Fallback – return a safe uniform sequence if everything else failed
    # ----------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        best_seq = np.full(n, math.sqrt(2.0 * n) / n, dtype=float)

    return best_seq.tolist()
